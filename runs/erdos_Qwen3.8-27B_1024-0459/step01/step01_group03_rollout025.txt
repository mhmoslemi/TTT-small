        else:
            ka = -k
            g[:n - ka] += p * (1.0 - h[ka:])
            g[ka:] -= p * h[:n - ka]
    return val, dx * g

def slsqp_optimize(h0, n, alpha, maxiter, deadline):
    """Run SLSQP with smooth objective"""
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        v, _ = smooth_obj_and_grad(x, alpha)
        return v
    
    def grad(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g
    
    constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)},)
    
    res = minimize(obj, h0, method='SLSQP', jac=grad,
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10, 'disp': False})
    
    if time.perf_counter() > deadline:
        return h0, evaluate_erdos(h0)
    
    if res.success or hasattr(res, 'x'):
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            x = project_sum(x, target)
            c = evaluate_erdos(x)
            return x, c
    return h0, evaluate_erdos(h0)

def binary_anneal(h0, deadline, target, max_iter=20000, start_temp=0.05, end_temp=0.001):
    """Simulated annealing on binary/step function"""
    n = h.size if hasattr(h0, 'size') else len(h0)
    best = np.clip(h0.copy(), 0.0, 1.0)
    best_c = evaluate_erdos(best)
    cur = best.copy()
    cur_c = best_c
    
    ones = np.where(cur > 0.5)[0]
    zeros = np.where(cur <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return best, best_c
    
    n_iter = max_iter
    t0 = time.perf_counter()
    
    for it in range(n_iter):
        if time.perf_counter() > deadline:
            break
        progress = it / float(n_iter)
        temp = start_temp * ((end_temp / start_temp) ** progress)
        
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        cur[i], cur[j] = cur[j], cur[i]
        new_c = evaluate_erdos(cur)
        
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c:
                best = cur.copy()
                best_c = new_c
        else:
            delta = new_c - cur_c
            if np.random.random() < np.exp(-delta / max(temp, 1e-10)):
                cur_c = new_c
            else:
                cur[i], cur[j] = cur[j], cur[i]
    
    return best, best_c

def gradient_descent(h0, deadline, target, max_iter=5000):
    """Projected gradient descent on smooth objective"""
    n = h0.size
    best = project_sum(h0, target)
    best_c = evaluate_erdos(best)
    alpha = 20.0
    
    for it in range(max_iter):
        if time.perf_counter() > deadline:
            break
        
        _, g = smooth_obj_and_grad(best, alpha)
        # Project gradient to feasible direction (sum=0)
        g = g - g.mean()
        norm = np.linalg.norm(g)
        if norm < 1e-14:
            break
        
        # Backtracking line search
        improved = False
        for step in (0.05, 0.02, 0.01, 0.005, 0.002):
            cand = project_sum(best - step * g / norm, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        
        if not improved:
            break
    
    return best, best_c

def targeted_swap_search(h0, deadline, target, max_iter=10000):
    """Swap search guided by which lags are at maximum"""
    n = h0.size
    best = project_sum(h0.copy(), target)
    best_c = evaluate_erdos(best)
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() > deadline:
            break
        if stall > 300:
            break
        
        corr = get_corr(best)
        max_val = np.max(corr)
        # Find lags near the maximum
        thresh = max_val - 0.02 * max_val
        active_lags = np.where(corr >= thresh)[0]
        center = n - 1
        active_ks = active_lags - center
        
        # Compute gradient at active lags
        g = weighted_lag_gradient(best, active_lags, np.ones(len(active_lags)))
        g = g - g.mean()
        
        # Find indices to swap: increase where g is negative, decrease where g is positive
        neg_idx = np.argsort(g)[:5]
        pos_idx = np.argsort(-g)[:5]
        
        found = False
        for ii in neg_idx:
            for jj in pos_idx:
                ii, jj = int(ii), int(jj)
                if ii == jj:
                    continue
                if best[ii] <= 0.01 and best[jj] >= 0.99:
                    continue
                if best[ii] >= 0.99 and best[jj] <= 0.01:
                    continue
                
                # Try swapping or partial adjustment
                delta = 0.5 * (best[jj] - best[ii])
                if abs(delta) < 1e-8:
                    continue
                
                cand = best.copy()
                cand[ii] += delta
                cand[jj] -= delta
                cand = project_sum(cand, target)
                c = evaluate_erdos(cand)
                
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
            if found:
                break
        
        if not found:
            # Try random swap
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and abs(best[i] - best[j]) > 0.05:
                cand = best.copy()
                delta = 0.3 * (best[j] - best[i])
                cand[i] += delta
                cand[j] -= delta
                cand = project_sum(cand, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
        
        if not found:
            stall += 1
            # Perturb to escape
            if stall % 50 == 0:
                idx = np.random.randint(n)
                best[idx] = np.clip(best[idx] + np.random.uniform(-0.1, 0.1), 0, 1)
                best = project_sum(best, target)
                best_c = evaluate_erdos(best)
    
    return best, best_c

def make_initial(n, rng, mode):
    """Generate initial candidate"""
    half = n // 2
    if mode == 0:
        # Random binary
        h = np.zeros(n)
        ones = rng.choice(n, half, replace=False)
        h[ones] = 1.0
    elif mode == 1:
        # Block pattern
        h = np.zeros(n)
        h[:half] = 1.0
    elif mode == 2:
        # Sine-weighted
        x = np.arange(n) / n
        vals = 0.5 + 0.35 * np.sin(2 * np.pi * rng.integers(1, 5) * x + rng.uniform(0, 6.28))
        vals += 0.2 * np.sin(2 * np.pi * rng.integers(1, 7) * x + rng.uniform(0, 6.28))
        order = np.argsort(vals)[::-1]
        h = np.zeros(n)
        h[order[:half]] = 1.0
    elif mode == 3:
        # Two-scale pattern
        h = np.zeros(n)
        block = max(1, n // 8)
        for b in range(0, n, block * 2):
            end = min(b + block, n)
            h[b:end] = 1.0
        # Fix sum
        s = int(h.sum())
        if s > half:
            ones = np.where(h > 0.5)[0]
            rng.shuffle(ones)
            for i in range(s - half):
                h[ones[i]] = 0
        elif s < half:
            zeros = np.where(h <= 0.5)[0]
            rng.shuffle(zeros)
            for i in range(half - s):
                h[zeros[i]] = 1
    else:
        # Smooth random
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 7 * np.arange(n) / n + rng.uniform(0, 6.28))
        h = np.clip(h, 0, 1)
    
    return project_sum(h, half)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start_time + budget - 15.0
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float("inf")
    
    # Try to use parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 10:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        pn = parent.size
        p_target = 0.5 * pn
        p_proj = project_sum(parent, p_target)
        p_c = evaluate_erdos(p_proj)
        if p_c < best_c:
            best_h = p_proj.copy()
            best_c = p_c
    
    if best_h is None:
        best_h = make_initial(90, rng, 0)
        best_c = evaluate_erdos(best_h)
    
    n_best = best_h.size
    
    # Phase 1: Optimize at current resolution
    # Use SLSQP with multiple alphas
    if best_h is not None and time.perf_counter() < deadline - 60:
        n = best_h.size
        target = 0.5 * n
        h_work = best_h.copy()
        
        for alpha in (15.0, 40.0, 100.0):
            if time.perf_counter() > deadline - 40:
                break
            d = deadline - 35
            res_h, res_c = slsqp_optimize(h_work, n, alpha, maxiter=200, deadline=d)
            if res_c < best_c:
                best_h = res_h.copy()
                best_c = res_c
            h_work = res_h.copy()
    
    # Phase 2: Gradient descent refinement
    if best_h is not None and time.perf_counter() < deadline - 30:
        n = best_h.size
        target = 0.5 * n
        g_h, g_c = gradient_descent(best_h, deadline - 25, target, max_iter=3000)
        if g_c < best_c:
            best_h = g_h.copy()
            best_c = g_c
    
    # Phase 3: Targeted swap search
    if best_h is not None and time.perf_counter() < deadline - 20:
        n = best_h.size
        target = 0.5 * n
        s_h, s_c = targeted_swap_search(best_h, deadline - 15, target, max_iter=8000)
        if s_c < best_c:
            best_h = s_h.copy()
            best_c = s_c
    
    # Phase 4: Try binary annealing at same resolution
    if best_h is not None and time.perf_counter() < deadline - 50:
        n = best_h.size
        target = 0.5 * n
        # Binarize
        h_bin = (best_h > 0.5).astype(float)
        # Fix to exactly n/2 ones
        s = int(h_bin.sum())
        if s > n // 2:
            ones = np.where(h_bin > 0.5)[0]
            rng.shuffle(ones)
            for i in range(s - n // 2):
                h_bin[ones[i]] = 0
        elif s < n // 2:
            zeros = np.where(h_bin <= 0.5)[0]
            rng.shuffle(zeros)
            for i in range(n // 2 - s):
                h_bin[zeros[i]] = 1
        
        a_h, a_c = binary_anneal(h_bin, deadline - 45, target, max_iter=15000,
                                  start_temp=0.03, end_temp=0.001)
        if a_c < best_c:
            best_h = a_h.copy()
            best_c = a_c
    
    # Phase 5: Try different initializations
    if time.perf_counter() < deadline - 100:
        n = best_h.size
        for mode in range(5):
            if time.perf_counter() > deadline - 60:
                break
            h_init = make_initial(n, rng, mode)
            # Quick SLSQP
            h_opt, c_opt = slsqp_optimize(h_init, n, 30.0, maxiter=100,
                                          deadline=deadline - 55)
            if c_opt < best_c:
                best_h = h_opt.copy()
                best_c = c_opt
                n_best = n
    
    # Phase 6: Try higher resolution
    if best_h is not None and time.perf_counter() < deadline - 80:
        n_curr = best_h.size
        n_high = n_curr * 2
        if n_high <= 360:
            h_up = upsample(best_h, n_high)
            c_up = evaluate_erdos(h_up)
            if time.perf_counter() < deadline - 40:
                h_opt, c_opt = slsqp_optimize(h_up, n_high, 25.0, maxiter=80,
                                              deadline=deadline - 35)
                if c_opt < best_c:
                    best_h = h_opt.copy()
                    best_c = c_opt
                h_opt2, c_opt2 = targeted_swap_search(h_opt, deadline - 20, 0.5 * n_high,
                                                      max_iter=3000)
                if c_opt2 < best_c:
                    best_h = h_opt2.copy()
                    best_c = c_opt2
    
    # Phase 7: Final intensive polish
    if best_h is not None and time.perf_counter() < deadline - 5:
        n_final = best_h.size
        target_final = 0.5 * n_final
        
        # More SLSQP with high alpha
        for alpha in (50.0, 150.0):
            if time.perf_counter() > deadline - 3:
                break
            h_p, c_p = slsqp_optimize(best_h, n_final, alpha, maxiter=100,
                                      deadline=deadline - 2)
            if c_p < best_c:
                best_h = h_p.copy()
                best_c = c_p
        
        # Final gradient descent
        if time.perf_counter() < deadline - 2:
            h_g, c_g = gradient_descent(best_h, deadline - 1, target_final, max_iter=2000)
            if c_g < best_c:
                best_h = h_g.copy()
                best_c = c_g
    
    # Final projection
    if best_h is not None:
        final_n = best_h.size
        best_h = project_sum(best_h, 0.5 * final_n)
        best_c = evaluate_erdos(best_h)
    
    if best_h is None:
        best_h = project_sum(np.full(90, 0.5), 45.0)
        best_c = evaluate_erdos(best_h)
    
    return best_h, float(best_c), int(best_h.size)
