```python
# Pre-compute objective and gradient functions
def compute_c5(h, n_points):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * (2.0 / n_points)

def compute_grad(h, n_points):
    # Gradient via forward finite-difference for cheap quick estimate
    n = n_points
    grad = np.zeros(n)
    for idx in range(n):
        # f(idx+ε) - f(idx) / ε
        h_plus = h.copy()
        h_plus[idx] += 1e-5
        f_plus = compute_c5(h_plus, n_points) - compute_c5(h, n_points)  # Direction derivative
        grad[idx] = f_plus / 1e-5
    return grad
```
