This gives a linear approximation, and we can solve a linear program (with the max constraint) to get a direction. Then do a line search.

This is essentially a cutting plane / linearization approach for the non-convex problem. It might work well in practice.

def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    target = float(target)
    s = float(np.sum(x))
    if abs(s - target) <= 1e-12 * max(1.0, n):
        return x.copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_test = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s_test < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-13:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if free.size > 0:
            if res > 0:
                idx = int(free[np.argmax(1.0 - h[free])])
            else:
                idx = int(free[np.argmax(h[free])])
            h[idx] = float(np.clip(h[idx] + res, 0.0, 1.0))
    return h

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if corr.size == 0:
        return float('inf')
    return float(np.max(corr) * dx)

    _g_alpha = float(alpha)
    _g_target = float(target)

def _flu_obj(x):
    n = _g_n
    alpha = _g_alpha
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode='full')

        return 1e100
    m = float(np.max(corr))

    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100

    n = _g_n
    alpha = _g_alpha
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)

        return np.zeros(n, dtype=float)

def grad_descent(h0, n, deadline, alpha=50.0, max_iter=2000):
    """Gradient descent with backtracking line search on the projected feasible set."""
    n = int(n)
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    _set_ctx(n, alpha, target)
    
    lr = 0.05
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        g = _flu_jac(h)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        g = g / gn
        # Project gradient to feasible direction (maintain sum)
        g = g - np.mean(g)
        gn2 = float(np.linalg.norm(g))
        if gn2 < 1e-14:
            break
        g = g / gn2
        
        # Backtracking line search
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            step = lr * factor
            cand = project_density(h + step * g, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                h = cand
                best_h = cand.copy()
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
            lr = min(lr * 1.1, 0.5)
        else:
            stall += 1
            lr *= 0.7
            if stall > 50:
                break
            if lr < 1e-6:
                break
    
    return best_h, best_c
