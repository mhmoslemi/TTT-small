            # d/dh[i] = -h[i+m] for i < n-m
            if m < n:
                for i in range(m, n):
                    g[i] += (1.0 - h[i - m])
                for i in range(n - m):
                    g[i] -= h[i + m]
        grad += g
    grad /= len(active)
    return grad, max_val * dx, corr * dx

import numpy as np
from scipy.optimize import minimize
import functools
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def get_grad_active(h, n_active=12):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    active_mask = corr >= max_corr * 0.92
    active_idx = np.where(active_mask)[0]
    if len(active_idx) > n_active:
        active_idx = active_idx[np.argsort(corr[active_idx])[-n_active:]]
    elif len(active_idx) == 0:
        active_idx = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for k_idx in active_idx:
        shift = k_idx - (n - 1)
        if shift == 0:
            grad += 1.0 - 2.0 * h
        elif shift > 0:
            m = shift
            if m < n:
                grad[:n - m] += (1.0 - h[m:])
                grad[m:] -= h[:n - m]
        else:
            m = -shift
            if m < n:
                grad[m:] += (1.0 - h[:n - m])
                grad[:n - m] -= h[m:]
    grad /= len(active_idx)
    c5_val = max_corr * (2.0 / n)
    return grad, c5_val


def smooth_max_obj(x, tau):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -60, 60)
    logsum = np.log(np.sum(np.exp(shifted)))
    return float((m + tau * logsum) * dx)


def eq_con(x):
    return np.array([x.sum() - len(x) / 2.0])


def eq_jac(x):
    return np.ones((1, len(x)))


def periodic_init(n, period_frac, duty_frac):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    period = 2.0 * period_frac
    h = np.zeros(n)
    for i in range(n):
        phase = (x[i] % period) / period
        h[i] = 1.0 if phase < duty_frac else 0.0
    h = fix_sum(h, n)
    return h


def randomized_block_init(n, rng, n_blocks=6):
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        val = 1.0 if rng.random() < 0.5 else 0.0
        h[start:end] = val
        h[start:end] += rng.uniform(-0.1, 0.1, end - start)
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(180, 0.5)

    n = len(parent_h)
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient with momentum (fast global search)
    h_curr = best_h.copy()
    velocity = np.zeros(n)
    beta = 0.85
    base_step = 4.0 / (n + 1)
    best_pg1 = best_c5

    for it in range(4000):
        if time.time() - t0 > budget_s * 0.22:
            break
        grad, c5_val = get_grad_active(h_curr, n_active=15)
        step_t = base_step / np.sqrt(1.0 + it * 0.01)
        velocity = beta * velocity - step_t * grad
        h_new = h_curr + velocity
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        c5_new = compute_c5(h_new)
        h_curr = h_new
        if c5_new < best_pg1 - 1e-9:
            best_pg1 = c5_new
            best_h = h_new.copy()

    best_c5 = min(best_c5, compute_c5(best_h))

    # Phase 2: SLSQP refinement (moderate tau → tight)
    bounds_list = [(0.0, 1.0)] * n
    x0 = best_h.copy()
    for tau in [0.06, 0.015, 0.004, 0.001]:
        if time.time() - t0 > budget_s * 0.42:
            break
        obj_p = functools.partial(smooth_max_obj, tau=tau)
        try:
            res = minimize(
                fun=obj_p, x0=x0, method='SLSQP',
                bounds=bounds_list,
                constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
                options={'maxiter': 1000, 'ftol': 1e-13, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
            x0 = h_new
        except Exception:
            pass

    # Phase 3: Diverse kicks + SLSQP
    for r in range(14):
        if time.time() - t0 > budget_s * 0.58:
            break
        noise_scale = 0.07 * (1.0 - r / 20.0)
        pert = rng.normal(0, noise_scale, n)
        x_pert = np.clip(best_h + pert, 0.0, 1.0)
        x_pert = fix_sum(x_pert, n)
        tau = 0.006
        obj_p = functools.partial(smooth_max_obj, tau=tau)
        try:
            res = minimize(
                fun=obj_p, x0=x_pert, method='SLSQP',
                bounds=bounds_list,
                constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
                options={'maxiter': 700, 'ftol': 1e-13, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                x0 = h_new
        except Exception:
            pass

    # Phase 4: Periodic and structured initializations
    for pf, df in [(1.0/3, 0.5), (1.0/4, 0.5), (1.0/5, 0.5), (1.0/2, 0.5),
                   (1.0/3, 0.4), (1.0/3, 0.35), (2.0/7, 0.5)]:
        if time.time() - t0 > budget_s * 0.65:
            break
        h_per = periodic_init(n, pf, df)
        tau = 0.02
        obj_p = functools.partial(smooth_max_obj, tau=tau)
        try:
            res = minimize(
                fun=obj_p, x0=h_per, method='SLSQP',
                bounds=bounds_list,
                constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
                options={'maxiter': 600, 'ftol': 1e-12, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 5: Randomized block initializations
    for nb in [4, 6, 8, 10]:
        if time.time() - t0 > budget_s * 0.70:
            break
        h_blk = randomized_block_init(n, rng, n_blocks=nb)
        tau = 0.02
        obj_p = functools.partial(smooth_max_obj, tau=tau)
        try:
            res = minimize(
                fun=obj_p, x0=h_blk, method='SLSQP',
                bounds=bounds_list,
                constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
                options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 6: Higher resolution
    for n_hi in [2 * n, 3 * n]:
        if time.time() - t0 > budget_s * 0.85:
            break
        x_hi = upsample_to(best_h, n_hi)
        x_hi = fix_sum(x_hi, n_hi)
        bounds_hi = [(0.0, 1.0)] * n_hi
        for tau in [0.04, 0.008]:
            if time.time() - t0 > budget_s * 0.92:
                break
            obj_p = functools.partial(smooth_max_obj, tau=tau)
            try:
                res = minimize(
                    fun=obj_p, x0=x_hi, method='SLSQP',
                    bounds=bounds_hi,
                    constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
                    options={'maxiter': 500, 'ftol': 1e-12, 'disp': False}
                )
                h_new = np.clip(res.x, 0.0, 1.0)
                h_new = fix_sum(h_new, n_hi)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                x_hi = h_new
            except Exception:
                pass

    # Phase 7: Final micro-polish at best resolution
    n_f = len(best_h)
    bounds_f = [(0.0, 1.0)] * n_f
    for r in range(25):
        if time.time() - t0 > budget_s * 0.995:
            break
        small = rng.normal(0, 0.006, n_f)
        x_pert = np.clip(best_h + small, 0.0, 1.0)
        x_pert = fix_sum(x_pert, n_f)
        tau = 0.001
        obj_p = functools.partial(smooth_max_obj, tau=tau)
        try:
            res = minimize(
                fun=obj_p, x0=x_pert, method='SLSQP',
                bounds=bounds_f,
                constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
                options={'maxiter': 800, 'ftol': 1e-14, 'disp': False}
            )
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, n_f)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
