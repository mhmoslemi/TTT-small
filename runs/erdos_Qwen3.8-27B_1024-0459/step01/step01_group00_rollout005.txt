def project_sum(h, target):
    h = np.asarray(h,dtype=float)
    n = len(h)
    target = float(np.clip(target,0.0,float(n)))
    h = np.clip(h,0.0,1.0)
    s = float(h.sum())
    if abs(s-target) <= 1e-13:
        return h.copy()
    lo=-1.0; hi=1.0
    # verify f(lo)<=target<=f(hi); if not extend.
    # h already clip so f(-1)=0, f(1)=n.
    for _ in range(50):
       mid=0.5*(lo+hi)
       val=float(np.sum(np.clip(h+mid,0.0,1.0)))
       if val < target:
          lo=mid
       else:

def smooth_grad(x,n,beta):
   C=np.correlate(x,1.0-x)
   cmax=float(C.max())
   shifted=beta*(C-cmax)
   # avoid under/overflow
   e=np.exp(shifted)
   ssum=float(e.sum())
   if ssum <=0: return np.zeros(n)
   w=e/ssum
   grad=np.zeros(n,float)
   z=n-1
   for idx in range(2*n-1):
      ww=float(w[idx])
      if ww == 0.0: continue
      d=idx-z
      if d == 0:
          grad += ww*(1.0-2.0*x)
      elif d > 0:
          grad[d:n] += ww*(1.0 - x[0:n-d])
          grad[0:n-d] -= ww*x[d:n]
      else:
          dd=-d
          grad[0:n-dd] += ww*(1.0 - x[dd:n])
          grad[dd:n] -= ww*x[0:n-dd]
   return grad

def project_sum(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = float(np.clip(target, 0.0, float(n)))
    if n == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) <= 1e-14:
        out = h.copy()
    else:
        lo = -1.0
        hi = 1.0
        for _ in range(55):
            mid = 0.5 * (lo + hi)
            val = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
            if val < target:
                lo = mid
            else:
                hi = mid
        alpha = 0.5 * (lo + hi)
        out = np.clip(h + alpha, 0.0, 1.0)

    for _ in range(6):
        diff = target - float(out.sum())
        if abs(diff) < 1e-15:
            break
        mask = (out > 1e-14) & (out < 1.0 - 1e-14)
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def interpolate_to_n(src, n):
    src = np.asarray(src, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    if src.size == 0:
        return np.full(n, 0.5, dtype=float)

    x_new = np.linspace(0.0, 2.0, n, endpoint=False)

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return float("inf")
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / float(n)))


def sum_eq(x, target):
    return float(np.sum(x) - target)


def sum_jac(x, target):
    return np.ones_like(x)


def smooth_obj(x, n, beta):
    x = np.asarray(x, dtype=float)
    corr = np.correlate(x, 1.0 - x, mode="full")
    cmax = float(np.max(corr))
    shifted = beta * (corr - cmax)
    s = float(np.sum(np.exp(shifted)))
    if s <= 0.0:
        return cmax
    return cmax + float(np.log(s)) / float(beta)


def smooth_grad(x, n, beta):
    x = np.asarray(x, dtype=float)
    corr = np.correlate(x, 1.0 - x, mode="full")
    cmax = float(np.max(corr))
    shifted = beta * (corr - cmax)
    e = np.exp(shifted)
    s = float(e.sum())
    if s <= 0.0:
        return np.zeros(n, dtype=float)
    w = e / s
    grad = np.zeros(n, dtype=float)
    z = n - 1
    m = 2 * n - 1
    for idx in range(m):
        ww = float(w[idx])
        if abs(ww) < 1e-16:
            continue
        d = idx - z
        if d == 0:
            grad += ww * (1.0 - 2.0 * x)
        elif d > 0:
            a = n - d
            grad[d:n] += ww * (1.0 - x[0:a])
            grad[0:a] -= ww * x[d:n]
        else:
            dd = -d
            a = n - dd
            grad[0:a] += ww * (1.0 - x[dd:n])
            grad[dd:n] -= ww * x[0:a]
    return grad


def run_slsqp_smooth(h, n, beta, maxiter=300):
    n = int(n)
    if n <= 0:
        return None, None
    x0 = make_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [
        {
            "type": "eq",
            "fun": sum_eq,
            "jac": sum_jac,
            "args": (0.5 * float(n),),
        }
    ]
    try:
        res = minimize(
            smooth_obj,
            x0,
            jac=smooth_grad,
            args=(n, float(beta)),
            bounds=bounds,
            constraints=cons,
            method="SLSQP",
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False},
        )
    except Exception:
        return None, None
    if not hasattr(res, "x"):
        return None, None
    hf = make_feasible(res.x, n)
    c = compute_c5(hf)
    if not np.isfinite(c):
        return None, None
    return hf, c

def unique_inits(inits, tol=0.04, max_n=6):
    out = []
    for h in inits:
        if h is None:
            continue
        h = np.asarray(h, dtype=float).reshape(-1)
        if h.size == 0:
            continue
        dup = False
        for u in out:
            if u.shape[0] == h.shape[0] and float(np.max(np.abs(h - u))) <= tol:
                dup = True
                break
        if not dup:
            out.append(h)
            if len(out) >= int(max_n):
                break
    return out


def get_coord_cache(n):

        rc.append(z + mm - j)

    return cache


def coord_1d_min(c_eval, a_arr, b0, s0, lam):

            c = hi - invphi * (hi - lo)

        else:
            lo = c
            c = d
            hc = hd
            d = lo + invphi * (hi - lo)
            hd = float(np.max(c_eval + a_arr * (d - s0) + lam * d))

    hmin_s = 0.5 * (lo + hi)
    cands = [0.0, 1.0, hmin_s]

    grid = GRID64

    linmax = lin.max(axis=0)

    fvals = qtilde - linmax

    roots = 0
    for i in range(len(grid) - 1):
        fa = float(fvals[i])
        fb = float(fvals[i + 1])
        if (fa < -1e-13 and fb > 1e-13) or (fa > 1e-13 and fb < -1e-13):
            if roots >= 2:
                break
            a = float(grid[i])
            b = float(grid[i + 1])
            fa0 = fa

                    b0 + mid * (1.0 - mid) + lam * mid
                    - np.max(c_eval + a_arr * (mid - s0) + lam * mid)
                )
                if fa0 * fmid > 0.0:
                    a = mid
                    fa0 = fmid
                else:
                    b = mid
            cands.append(0.5 * (a + b))
            roots += 1

    best_s = 0.0
    best_val = float("inf")
    for s in cands:
        if s < 0.0:
            s = 0.0
        elif s > 1.0:
            s = 1.0

        val = hv if hv > qv else qv
        if val < best_val:
            best_val = val
            best_s = s
    return float(best_s)


def coordinate_descent(h, n, cycles=5, lam=0.0, stop_time=0.0):
    h = make_feasible(h, n)
    n = int(n)
    if n <= 1:
        return h, compute_c5(h)

    C = np.correlate(h, 1.0 - h, mode="full")
    target = 0.5 * float(n)
    S = float(h.sum())
    best_c = compute_c5(h)
    best_h = h.copy()

    mc, lc, rc = get_coord_cache(n)
    M = 2 * n - 1
    z = n - 1

    for _ in range(int(cycles)):
        if stop_time > 0.0 and time.time() > stop_time:
            break
        for j in range(n):
            if stop_time > 0.0 and time.time() > stop_time:
                break
            s0 = float(h[j])
            A = np.zeros(M, dtype=float)
            mm = mc[j]
            A[lc[j]] += 1.0 - h[mm]
            A[rc[j]] -= h[mm]
            A[z] = 0.0

            c_eval = C.copy()
            c_eval[z] = -np.inf
            b0 = C[z] - s0 * (1.0 - s0)

            s_new = coord_1d_min(c_eval, A, b0, s0, float(lam))
            delta = s_new - s0
            if abs(delta) > 1e-15:
                C += A * delta
                C[z] += s_new * (1.0 - s_new) - s0 * (1.0 - s0)
                h[j] = s_new
                S += delta

        S_pre = S
        h = project_sum(h, target)
        C = np.correlate(h, 1.0 - h, mode="full")
        S = float(h.sum())

        c = compute_c5(h)
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = h.copy()

        err = S_pre - target
        lam = 0.5 * float(lam) + 0.04 * err
        if lam > 1.0:
            lam = 1.0
        elif lam < -1.0:
            lam = -1.0

    return best_h, best_c


def random_perturb(h, n, n_trials=200, step=0.02, stop_time=0.0):
    h = make_feasible(h, n)
    n = int(n)
    best_c = compute_c5(h)
    best_h = h.copy()
    kmax = min(5, n)

    for _ in range(int(n_trials)):
        if stop_time > 0.0 and time.time() > stop_time:
            break
        if kmax <= 1:
            k = 1
        else:
            k = int(np.random.randint(1, kmax + 1))
        idx = np.random.choice(n, k, replace=False)
        h_try = h.copy()
        h_try[idx] += np.random.uniform(-step, step, size=k)
        h_try = make_feasible(h_try, n)
        c = compute_c5(h_try)

        hs = make_feasible(symmetrize(h_try, n), n)
        cs = compute_c5(hs)
        if cs < c:
            c = cs
            h_try = hs

        if c < best_c - 1e-13:
            best_c = c
            best_h = h_try.copy()

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    budget_s = max(60.0, float(budget_s))
    deadline = start_time + (budget_s - 30.0)

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float).reshape(-1)
    except Exception:
        parent = None

    best_h = None
    best_n = 160
    best_c = float("inf")

    if parent is not None and parent.size > 0:
        best_n = int(parent.size)
        best_h = make_feasible(parent, best_n)
        best_c = compute_c5(best_h)
    else:
        n0 = 160
        best_h = make_feasible(np.random.rand(n0) * 1.5, n0)
        best_n = n0
        best_c = compute_c5(best_h)
        for h0 in generate_structured(n0, int(seed) + 7):
            if time.time() > deadline - 20:
                break
            c0 = compute_c5(h0)
            if c0 < best_c:
                best_c = c0
                best_h = h0.copy()

    preferred = [
        best_n,
        160,
        128,
        200,
        240,
        320,
        192,
        256,
        384,
        448,
    ]
    if budget_s > 900:
        append_unique(preferred, 512)

    job_ns = []
    for n in preferred:
        n = int(n)
        if 32 <= n <= 512:
            append_unique(job_ns, n)

    for n in job_ns:
        n = int(n)
        if time.time() > deadline - 60:
            break
        if n > 256 and time.time() > deadline - 170:
            break

        inits = []
        if parent is not None and parent.size > 0:
            if parent.size == n:
                hp = make_feasible(parent, n)
            else:
                hp = make_feasible(interpolate_to_n(parent, n), n)
            inits.append(hp)
            inits.append(make_feasible(symmetrize(hp, n), n))

        if best_h is not None:
            if best_h.size == n:
                hb = best_h.copy()
            else:
                hb = make_feasible(interpolate_to_n(best_h, n), n)
            inits.append(hb)
            inits.append(make_feasible(symmetrize(hb, n), n))

        struct = generate_structured(n, int(seed) + n)
        add_struct = 4 if n <= 160 else 3 if n <= 256 else 2
        for hs in struct[:add_struct]:
            inits.append(hs)

        inits.append(make_feasible(np.random.rand(n) * 1.5, n))

        max_inits = 6 if n <= 160 else 4 if n <= 256 else 3
        inits = unique_inits(inits, tol=0.04, max_n=max_inits)

        if n <= 160:
            beta_steps = [(20.0, 350), (80.0, 250), (250.0, 180)]
        elif n <= 256:
            beta_steps = [(20.0, 250), (80.0, 180)]
        else:
            beta_steps = [(20.0, 180), (80.0, 130)]

        for h0 in inits:
            if time.time() > deadline - 30:
                break

            current_h = h0
            current_c = compute_c5(current_h)
            if current_c < best_c:
                best_c = current_c
                best_h = current_h.copy()
                best_n = n

            hs0 = make_feasible(symmetrize(h0, n), n)
            cs0 = compute_c5(hs0)
            if cs0 < best_c:
                best_c = cs0
                best_h = hs0.copy()
                best_n = n
            if cs0 <= current_c:
                current_h = hs0
                current_c = cs0

            for beta, maxit in beta_steps:
                rem = deadline - time.time()
                if rem < 25:
                    break
                eff_maxit = int(maxit)
                if rem < 70:
                    eff_maxit = min(eff_maxit, 120)
                if rem < 40:
                    eff_maxit = min(eff_maxit, 70)
                h_opt, c_opt = run_slsqp_smooth(current_h, n, beta, eff_maxit)
                if h_opt is None:
                    continue
                c_opt = float(c_opt)
                hs_opt = make_feasible(symmetrize(h_opt, n), n)
                cs_opt = compute_c5(hs_opt)
                if cs_opt < c_opt:
                    c_opt = cs_opt
                    h_opt = hs_opt

                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
                    best_n = n
                current_h = h_opt
                current_c = c_opt

            if time.time() > deadline - 90:
                continue
            if current_c < best_c + 0.004:
                cycles = 5 if n <= 160 else 3 if n <= 256 else 2
                h_pol, c_pol = coordinate_descent(
                    current_h, n, cycles=cycles, lam=0.0, stop_time=deadline - 25
                )
                c_pol = compute_c5(h_pol)
                if c_pol < best_c:
                    best_c = c_pol
                    best_h = h_pol.copy()
                    best_n = n
                current_h = h_pol
                current_c = c_pol

                if time.time() < deadline - 70:
                    hs = make_feasible(symmetrize(current_h, n), n)
                    cs = compute_c5(hs)
                    if cs < best_c:
                        best_c = cs
                        best_h = hs.copy()
                        best_n = n
                    h_pol2, c_pol2 = coordinate_descent(
                        hs, n, cycles=max(2, cycles - 2), lam=0.0, stop_time=deadline - 25
                    )
                    c_pol2 = compute_c5(h_pol2)
                    if c_pol2 < best_c:
                        best_c = c_pol2
                        best_h = h_pol2.copy()
                        best_n = n

    if best_h is not None and time.time() < deadline - 50:
        h_r, c_r = random_perturb(
            best_h,
            best_n,
            n_trials=500,
            step=0.02,
            stop_time=deadline - 15,
        )
        c_r = compute_c5(h_r)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r.copy()
            best_n = int(h_r.size)

    if best_h is not None and time.time() < deadline - 70:
        rem = deadline - time.time()
        if rem > 120:
            eff_max = int(min(600, max(220, rem / 3.0)))
            h_fin, c_fin = run_slsqp_smooth(best_h, best_n, 120.0, eff_max)
            if h_fin is not None:
                c_fin = float(c_fin)
                hs_fin = make_feasible(symmetrize(h_fin, best_n), best_n)
                cs_fin = compute_c5(hs_fin)
                if cs_fin < c_fin:
                    c_fin = cs_fin
                    h_fin = hs_fin
                if c_fin < best_c:
                    best_c = c_fin
                    best_h = h_fin.copy()
                    best_n = int(h_fin.size)

    if best_h is not None and time.time() < deadline - 15:
        cycles = 7 if best_n <= 160 else 4 if best_n <= 256 else 2
        h_pol, c_pol = coordinate_descent(
            best_h, best_n, cycles=cycles, lam=0.0, stop_time=max(start_time, deadline - 5.0)
        )
        c_pol = compute_c5(h_pol)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol.copy()
            best_n = int(h_pol.size)

    best_h = make_feasible(best_h, best_n)
    best_c = compute_c5(best_h)

    return best_h, float(best_c), int(best_n)
