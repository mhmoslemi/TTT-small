    h = z.copy()

    corr = compute_corr(h)

    return float((m + log_sum) * dx)

    h = h.copy()
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt

    return np.interp(x_new, x_old, h_old)

            i = i_idx

                    ones = np.where(h == 1.0)[0]

                    h[j] = 0.0

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_obj(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    return float((m + tau * np.log(np.sum(np.exp(shifted)))) * dx)


def eq_con(x, n):
    return x.sum() - n / 2.0


def eq_jac(x, n):
    return np.ones(n)


def fix_sum(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def to_binary(h, n):
    """Convert h to binary with exactly n/2 ones, using the values of h as priority."""
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1.0)[0]
        # Remove the ones with lowest h values
        order = np.argsort(h[one_idx])[:n_ones - target]
        for idx in order:
            h_bin[one_idx[idx]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0.0)[0]
        # Add the zeros with highest h values
        order = np.argsort(h[zero_idx])[::-1][:target - n_ones]
        for idx in order:
            h_bin[zero_idx[idx]] = 1.0
    return h_bin


def stochastic_binary_search(h_bin, n, n_steps=200000, time_limit_s=60, seed=0):

    t_start = time.time()

    h = h_bin.copy()
    curr_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    rng = np.random.default_rng(seed)
    
    for step in range(n_steps):
        if time.time() - t_start > time_limit_s:
            break
        
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        
        if new_max < curr_max - 1e-15:
            curr_max = new_max
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return h, curr_max * dx

    """Run SLSQP with a schedule of tau values."""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    x = x0.copy()
    
    for i, tau in enumerate(tau_list):
        elapsed = time.time() - t0

            break
        maxiter = maxiter_list[i] if i < len(maxiter_list) else 2000
        try:
            res = minimize(
                fun=smooth_max_obj, args=(n, dx, tau),
                x0=x, method='SLSQP',

                constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac, 'args': (n,)}],
                options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
            )
            h_new = fix_sum(res.x, n)
            x = h_new
        except Exception:
            pass
    
    return x

def eq_con(x, n):
    return x.sum() - n / 2.0


def eq_jac(x, n):
    return np.ones(n)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def to_binary(h, n):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[one_idx])[:n_ones - target]
        for idx in order:
            h_bin[one_idx[idx]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(h[zero_idx])[::-1][:target - n_ones]
        for idx in order:
            h_bin[zero_idx[idx]] = 1.0
    return h_bin


def stochastic_binary_search(h_bin, n, n_steps, time_limit_s, seed):
    t_start = time.time()
    dx = 2.0 / n
    h = h_bin.copy()
    curr_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    rng = np.random.default_rng(seed)
    stagnant = 0
    max_stagnant = 50000
    for step in range(n_steps):
        if time.time() - t_start > time_limit_s:
            break
        if stagnant > max_stagnant:
            break
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        new_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        if new_max < curr_max - 1e-15:
            curr_max = new_max
            stagnant = 0
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
            stagnant += 1
    return h, curr_max * dx


def slsqp_anneal(x0, n, tau_list, maxiter_list, t0, budget_s, frac_end):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    x = x0.copy()
    for i, tau in enumerate(tau_list):
        if time.time() - t0 > budget_s * frac_end:
            break
        maxiter = maxiter_list[i] if i < len(maxiter_list) else 2000
        try:
            res = minimize(
                fun=smooth_max_obj, args=(n, dx, tau),
                x0=x, method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac, 'args': (n,)}],
                options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
            )
            h_new = fix_sum(res.x, n)
            if compute_c5(h_new) < compute_c5(x):
                x = h_new
            else:
                x = h_new
        except Exception:
            pass
    return x


def lbfgsb_anneal(x0, n, tau_list, penalty_list, maxiter_list, t0, budget_s, frac_end):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    x = x0.copy()
    for i in range(len(tau_list)):
        if time.time() - t0 > budget_s * frac_end:
            break
        tau = tau_list[i]
        pen = penalty_list[i] if i < len(penalty_list) else 100.0
        maxiter = maxiter_list[i] if i < len(maxiter_list) else 2000
        try:
            res = minimize(
                fun=penalized_obj, args=(n, dx, tau, pen),
                x0=x, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
            )
            h_new = fix_sum(res.x, n)
            x = h_new
        except Exception:
            pass
    return x


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(180, 0.5)

    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    n_start = len(best_h)

    # Phase 1: SLSQP at parent resolution with aggressive tau annealing
    tau1 = [0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005]
    mi1 = [3000, 3000, 3000, 3000, 3000, 3000, 3000, 3000, 3000]
    x1 = slsqp_anneal(fix_sum(parent_h, n_start), n_start, tau1, mi1, t0, budget_s, 0.28)
    c5_1 = compute_c5(x1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = x1.copy()

    # Phase 2: Binary stochastic search at parent resolution
    if time.time() - t0 < budget_s * 0.42:
        h_bin = to_binary(x1, n_start)
        c5_chk = compute_c5(h_bin)
        tl = max(10, min(50, (budget_s * 0.42 - (time.time() - t0)) * 0.8))
        if c5_chk < best_c5 + 0.05:
            h_b, c5_b = stochastic_binary_search(h_bin, n_start, 500000, tl, seed)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Also try random binary starts
    if time.time() - t0 < budget_s * 0.48:
        for trial in range(5):
            if time.time() - t0 > budget_s * 0.48:
                break
            h_rand = np.zeros(n_start)
            idx = rng.permutation(n_start)[:n_start // 2]
            h_rand[idx] = 1.0
            tl = max(5, min(15, (budget_s * 0.48 - (time.time() - t0)) * 0.5))
            h_r, c5_r = stochastic_binary_search(h_rand, n_start, 100000, tl, seed + 100 + trial)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            # Try SLSQP from this binary start
            if time.time() - t0 < budget_s * 0.50:
                tau_t = [0.02, 0.005, 0.001]
                mi_t = [2000, 2000, 2000]
                x_t = slsqp_anneal(h_r, n_start, tau_t, mi_t, t0, budget_s, 0.50)
                c5_t = compute_c5(x_t)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = x_t.copy()

    # Phase 3: Upsample to 2x and optimize
    n2 = 2 * n_start
    if time.time() - t0 < budget_s * 0.50:
        x2 = upsample(best_h, n2)
        x2 = fix_sum(x2, n2)
        tau2 = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        mi2 = [2000, 2000, 2000, 2000, 2000, 2000, 2000]
        x2 = slsqp_anneal(x2, n2, tau2, mi2, t0, budget_s, 0.72)
        c5_2 = compute_c5(x2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = x2.copy()

        # Use L-BFGS-B for additional refinement at this resolution
        if time.time() - t0 < budget_s * 0.78:
            tau2b = [0.005, 0.002, 0.001, 0.0005]
            pen2b = [50.0, 200.0, 500.0, 1000.0]
            mi2b = [3000, 3000, 3000, 3000]
            x2b = lbfgsb_anneal(x2, n2, tau2b, pen2b, mi2b, t0, budget_s, 0.78)
            c5_2b = compute_c5(x2b)
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = x2b.copy()

        # Binary search at 2x
        if time.time() - t0 < budget_s * 0.82:
            h_bin2 = to_binary(x2, n2)
            c5_chk2 = compute_c5(h_bin2)
            tl2 = max(5, min(30, (budget_s * 0.82 - (time.time() - t0)) * 0.7))
            if c5_chk2 < best_c5 + 0.05:
                h_b2, c5_b2 = stochastic_binary_search(h_bin2, n2, 300000, tl2, seed + 50)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_b2.copy()

    # Phase 4: Upsample to 4x and optimize (if time permits)
    n4 = 4 * n_start
    if time.time() - t0 < budget_s * 0.78:
        x4 = upsample(best_h, n4)
        x4 = fix_sum(x4, n4)
        tau4 = [0.03, 0.01, 0.005, 0.002, 0.001]
        mi4 = [1500, 1500, 1500, 1500, 1500]
        x4 = slsqp_anneal(x4, n4, tau4, mi4, t0, budget_s, 0.90)
        c5_4 = compute_c5(x4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = x4.copy()

        if time.time() - t0 < budget_s * 0.93:
            tau4b = [0.005, 0.002, 0.001]
            pen4b = [100.0, 500.0, 1000.0]
            mi4b = [2000, 2000, 2000]
            x4b = lbfgsb_anneal(x4, n4, tau4b, pen4b, mi4b, t0, budget_s, 0.93)
            c5_4b = compute_c5(x4b)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = x4b.copy()

    # Phase 5: Final polish at best resolution
    if time.time() - t0 < budget_s * 0.94:
        n_best = len(best_h)
        x0f = best_h.copy()
        for tau_f in [0.0005, 0.0002, 0.0001]:
            if time.time() - t0 > budget_s * 0.985:
                break
            dx_f = 2.0 / n_best
            bounds_f = [(0.0, 1.0)] * n_best
            try:
                res = minimize(
                    fun=smooth_max_obj, args=(n_best, dx_f, tau_f),
                    x0=x0f, method='SLSQP',
                    bounds=bounds_f,
                    constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac, 'args': (n_best,)}],
                    options={'maxiter': 5000, 'ftol': 1e-14, 'disp': False}
                )
                h_f = fix_sum(res.x, n_best)
                c5_f = compute_c5(h_f)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f
                    x0f = h_f
            except Exception:
                pass

        # Tiny binary refinement at the end
        if time.time() - t0 < budget_s * 0.99:
            h_final_bin = to_binary(x0f, n_best)
            c5_fb = compute_c5(h_final_bin)
            tl_f = max(3, budget_s * 0.005)
            h_fb, c5_fb = stochastic_binary_search(h_final_bin, n_best, 50000, tl_f, seed + 999)
            if c5_fb < best_c5:
                best_c5 = c5_fb
                best_h = h_fb.copy()

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
