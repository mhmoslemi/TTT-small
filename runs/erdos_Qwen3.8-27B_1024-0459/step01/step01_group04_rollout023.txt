                nv = n - d
                grad[:nv] += wk * (1.0 - hx[d:])
                grad[d:] += -wk * hx[:nv]
            else:
                d = -lag
                nv = n - d
                grad[d:] += wk * (1.0 - hx[:nv])
                grad[:nv] += -wk * hx[d:]
        return grad

    def cons_eq(x):
        return (float(np.sum(x)) - target_sum) / n

    def cons_jac(x):
        return np.full((1, n), 1.0/n)

    try:
        res = minimize(obj, h, jac=obj_grad, method="SLSQP",
                       bounds=bounds,
                       constraints=[{"type": "eq", "fun": cons_eq, "jac": cons_jac}],
                       options={"maxiter": 200, "ftol": 1e-12, "disp": 0})
        if res.success or res.fun < obj(h):
            h_out = project_feasible(res.x, target_sum)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    _RUN_DEADLINE = t0 + budget_s - 10.0

    rng = np.random.default_rng(seed)

    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim != 1 or parent.size < 2:
            parent = None
    except Exception:
        parent = None

    if parent is not None:
        n_parent = int(parent.size)
        h_parent = parent.copy()
    else:
        n_parent = 70
        x = (np.arange(n_parent) + 0.5) * (2.0 / n_parent)
        h_parent = 0.5 + 0.3 * np.cos(math.pi * x) + 0.1 * np.cos(2 * math.pi * x + 0.5)
        h_parent = project_feasible(h_parent, 0.5 * n_parent)

    best_h = h_parent.copy()
    best_c = compute_c5(best_h)
    best_n = n_parent

    def update_best(h_new, c_new, n_new):
        nonlocal best_h, best_c, best_n
        if c_new < best_c - 1e-10:
            best_h = h_new.copy()
            best_c = float(c_new)
            best_n = int(n_new)

    # Resolution schedule: (n, time_fraction)
    # Prioritize moderate increases over the parent
    trial_resolutions = []
    for mult in [1.0, 1.5, 2.0, 3.0, 4.0]:
        nn = int(round(mult * n_parent / 2.0)) * 2
        if nn >= 4 and nn <= 500:
            trial_resolutions.append(nn)
    # Deduplicate, sort
    trial_resolutions = sorted(set(trial_resolutions))

    total_budget = budget_s * 0.88  # leave 12% for polish
    # Allocate time: weight by 1/sqrt(n) to favor smaller n
    weights = [1.0 / math.sqrt(n) for n in trial_resolutions]
    wsum = sum(weights)
    time_fractions = [w / wsum for w in weights]

    # Also reserve 15% for binary search
    binary_budget = total_budget * 0.15
    cont_budget = total_budget - binary_budget

    # Continuous optimization at each resolution
    prev_best = h_parent.copy()
    for ri, n in enumerate(trial_resolutions):
        if time.time() > _RUN_DEADLINE - 30:
            break
        cont_time = cont_budget * time_fractions[ri]
        if cont_time < 15:
            continue
        # Initialize from previous best (upsampled)
        h_init = upsample_linear(prev_best, n)
        h_init = project_feasible(h_init, 0.5 * n)

        # Also try a slightly perturbed version
        h_init2 = h_init + rng.normal(0, 0.05, n)
        h_init2 = project_feasible(h_init2, 0.5 * n)

        n_inits = 2 if cont_time > 60 else 1
        per_init_time = cont_time / n_inits - 5

        inits = [h_init]
        if n_inits > 1:
            inits.append(h_init2)

        for h_i in inits:
            if time.time() > _RUN_DEADLINE - 25:
                break
            if per_init_time < 10:
                break
            try:
                h_opt, c_opt = gradient_descent(h_i, n, per_init_time, seed=seed + ri * 100 + int(np.mean(h_i)*1000))
                update_best(h_opt, c_opt, n)
                if c_opt < compute_c5(prev_best):
                    prev_best = h_opt.copy()
            except Exception:
                pass

    # Binary local search at a couple of resolutions
    binary_budgets = [binary_budget * 0.5, binary_budget * 0.5]
    binary_ns = [n_parent, 2 * n_parent]
    for bi, bn in enumerate(binary_ns[:2]):
        if time.time() > _RUN_DEADLINE - 25:
            break
        if binary_budgets[bi] < 10:
            continue
        # Initialize from continuous solution
        h_bin_init = upsample_linear(prev_best, bn)
        try:
            h_b, c_b = binary_local_search(h_bin_init, bn, binary_budgets[bi], rng)
            update_best(h_b, c_b, bn)
        except Exception:
            pass
        # Random restart
        try:
            h_b2, c_b2 = binary_local_search(None, bn, binary_budgets[bi] * 0.5, rng)
            update_best(h_b2, c_b2, bn)
        except Exception:
            pass

    # Final SLSQP polish on the best solution
    remaining = max(5.0, (_RUN_DEADLINE - time.time()) * 0.9)
    if remaining > 10:
        try:
            h_pol, c_pol = slsqp_polish(best_h, best_n, remaining)
            update_best(h_pol, c_pol, best_n)
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, 0.5 * best_n)
    best_c = compute_c5(best_h)

    return best_h, float(best_c), int(best_n)
