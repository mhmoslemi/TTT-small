    return np.correlate(h, 1.0 - h, mode="full")

    return float(np.max(corr_full(h)) * (2.0 / n))

def soft_max_objective(h, n, beta):
    c = corr_full(h)
    dx = 2.0 / n
    scaled = beta * c
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

    c = corr_full(h)
    dx = 2.0 / n
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    w /= w_sum

        else:

            grad[d:] -= wk * h[:n-d]

Actually, for a fixed k (the active one), c_k as a function of h[m] is:

- If m is the "second" index: h[partner] * (1-h[m]) → linear in h[m]
- If m appears in both: h[m]*(1-h[partner1]) + h[partner2]*(1-h[m]) → still linear
- If m appears in its own pair (lag 0): h[m]*(1-h[m]) → quadratic (concave)

import numpy as np
from scipy.optimize import minimize
import time


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_value(h, n):
    return float(np.max(corr_full(h)) * (2.0 / n))


def sum_eq_fun(h, n):
    return np.sum(h) - n / 2.0


def sum_eq_jac(h, n):
    return np.ones(n)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-14
        else:
            mask = h < 1.0 - 1e-14
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_obj(h, n, beta):
    c = corr_full(h)
    dx = 2.0 / n
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx)


def soft_grad(h, n, beta):
    c = corr_full(h)
    dx = 2.0 / n
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    grad = np.zeros(n)
    center = n - 1
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-13:
            continue
        lag = k_idx - center
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            d = lag
            grad[d:] += wk * (1.0 - h[:n - d])
            grad[:n - d] -= wk * h[d:]
        else:
            d = -lag
            grad[:n - d] += wk * (1.0 - h[d:])
            grad[d:] -= wk * h[:n - d]
    return grad * dx


def hard_obj(h, n):
    return float(np.max(corr_full(h)) * (2.0 / n))


def hard_grad(h, n):
    c = corr_full(h)
    k_max = int(np.argmax(c))
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    lag = k_max - center
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        d = lag
        grad[d:] += (1.0 - h[:n - d])
        grad[:n - d] -= h[d:]
    else:
        d = -lag
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] -= h[:n - d]
    return grad * dx


def hard_grad_from_k(h, n, k_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    center = n - 1
    lag = k_idx - center
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        d = lag
        grad[d:] += (1.0 - h[:n - d])
        grad[:n - d] -= h[d:]
    else:
        d = -lag
        grad[:n - d] += (1.0 - h[d:])
        grad[d:] -= h[:n - d]
    return grad * dx


def subgradient_refine(h, n, n_iter=20000, lr_init=0.004):
    h = h.copy()
    lr = lr_init
    best_local = c5_value(h, n)
    best_local_h = h.copy()
    for i in range(n_iter):
        c = corr_full(h)
        k_max = int(np.argmax(c))
        grad = hard_grad_from_k(h, n, k_max)
        h_new = project_feasible(h - lr * grad, n)
        c5 = c5_value(h_new, n)
        if c5 < best_local:
            best_local = c5
            best_local_h = h_new.copy()
        h = h_new
        lr = max(lr * 0.9995, lr_init * 0.05)
    return best_local_h, best_local


def slsqp_optimize(h0, n, use_hard, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    if use_hard:
        res = minimize(hard_obj, h0, jac=hard_grad, args=(n,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(soft_obj, h0, jac=soft_grad, args=(n, 50),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(np.clip(res.x, 0.0, 1.0), n)
