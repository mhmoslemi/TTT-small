            m = min(n, h.size)
            if m > 0:
                tmp[:m] = h[:m]
            h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) <= 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                per = diff / float(np.count_nonzero(mask))
                h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
        else:
            mask = h > 0.0
            if np.any(mask):
                per = -diff / float(np.count_nonzero(mask))
                h[mask] = np.clip(h[mask] - per, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def add_lag_gradient(grad, idx, coef, h, n):
    if coef == 0.0:
        return
    if idx == 0:
        grad += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
    else:
        d = idx // 2
    if d >= n:
        return
    m = n - d
    if m <= 0:
        return
    if idx % 2 == 1:
        grad[:m] += coef * (1.0 - h[d:])
        grad[d:] += coef * (-h[:m])
    else:
        grad[:m] += coef * (-h[d:])
        grad[d:] += coef * (1.0 - h[:m])


def softmax_objective(x, alpha, n):
    check_deadline()
    h = np.clip(x, 0.0, 1.0)
    scores = compute_scores(h, n)
    m = float(np.max(scores))
    if alpha > 1e-14:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0:
            val = m
        else:
            val = m + alpha * math.log(A)
    else:
        val = m
    return float(val) * (2.0 / n) if math.isfinite(val) else 1e6


def softmax_gradient(x, alpha, n):
    check_deadline()
    h = np.clip(x, 0.0, 1.0)
    scores = compute_scores(h, n)
    m = float(np.max(scores))
    grad = np.zeros(n, dtype=float)
    if alpha <= 1e-14:
        tol = max(1e-8, 1e-6 * max(1.0, m))
        active = np.where(scores >= m - tol)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(scores))])
        coef = 1.0 / float(active.size)
        for i in active:
            add_lag_gradient(grad, int(i), coef, h, n)
    else:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if math.isfinite(A) and A > 0:
            thresh = 1e-12
            for i in range(len(w)):
                c = w[i] / A
                if c >= thresh:
                    add_lag_gradient(grad, i, c, h, n)
    grad *= 2.0 / n
    if not np.all(np.isfinite(grad)):
        grad[:] = 0.0
    return grad


def constraint_eq(x, *args):
    n = max(1, len(x))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)


def constraint_jac(x, *args):
    n = max(1, len(x))
    return np.full((1, n), 1.0 / float(n), dtype=float)


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None


def upsample_linear(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent.astype(float)).astype(float)


class OptState:
    def __init__(self, h, c, n):
        self.h = np.array(h, dtype=float)
        self.c = float(c)
        self.n = int(n)

    def update(self, h, n):
        if h is None or n is None:
            return
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size == 0:
            return
        n = int(n)
        if h.size != n:
            h = h[:n] if h.size > n else np.pad(h, (0, n - h.size))
        h2 = as_feasible(h, n)
        c2 = compute_c5(h2)
        if math.isfinite(c2) and c2 < self.c - 1e-15:
            self.h = h2
            self.c = float(c2)
            self.n = n


def slsqp_refine(h, n, state, time_limit, alpha_start=0.1, alpha_end=0.0):
    if time_limit < 10.0:
        return
    h0 = as_feasible(h, n)
    state.update(h0, n)
    start_t = time.time()
    stage_end = start_t + time_limit
    if _RUN_DEADLINE is not None:
        stage_end = min(stage_end, _RUN_DEADLINE)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]

    # Adaptive alpha schedule
    n_alphas = 4
    alphas = []
    for i in range(n_alphas):
        t = i / max(1, n_alphas - 1)
        a = alpha_start * (alpha_end + (1 - alpha_end) * (1 - t) ** 2)
        if i == n_alphas - 1:
            a = 0.0
        alphas.append(a)

    maxiter_per = max(50, int(200.0 * (80.0 / float(n))))
    maxiter_per = min(maxiter_per, 500)

    remaining_time = stage_end - start_t
    time_per_alpha = remaining_time / float(n_alphas)

    for ai, alpha in enumerate(alphas):
        if time.time() >= stage_end - 5.0:
            break
        check_deadline()
        an = as_feasible(h, n)
        t_alloc = min(time_per_alpha, stage_end - time.time() - 5.0)
        if t_alloc < 8.0:
            break
        old_dl = _RUN_DEADLINE
        _RUN_DEADLINE = min(stage_end - 5.0, time.time() + t_alloc)
        try:
            res = minimize(
                softmax_objective, an, args=(alpha, n),
                jac=softmax_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": maxiter_per, "ftol": 1e-11, "disp": 0},
            )
            if res is not None and getattr(res, "x", None) is not None:
                h = as_feasible(res.x, n)
                state.update(h, n)
        except BudgetError:
            pass
        except Exception:
            pass
        finally:
            _RUN_DEADLINE = old_dl

    # Final exact-max polish
    if time.time() < stage_end - 10.0:
        check_deadline()
        an = as_feasible(h, n)
        old_dl = _RUN_DEADLINE
        _RUN_DEADLINE = stage_end - 5.0
        try:
            res = minimize(
                softmax_objective, an, args=(0.0, n),
                jac=softmax_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(maxiter_per * 1.5), "ftol": 1e-12, "disp": 0},
            )
            if res is not None and getattr(res, "x", None) is not None:
                state.update(res.x, n)
        except (BudgetError, Exception):
            pass
        finally:
            _RUN_DEADLINE = old_dl


def subgradient_polish(h, n, state, time_limit, n_iter=5000):
    """Subgradient method with projection for the exact max."""
    h = as_feasible(h, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / n
    start_t = time.time()
    end_t = start_t + time_limit
    if _RUN_DEADLINE is not None:
        end_t = min(end_t, _RUN_DEADLINE)

    # Estimate Lipschitz constant
    L = 4.0 * n
    D = math.sqrt(float(n))

    # Polyak step with tuning
    c_opt = 0.35  # optimistic lower bound on optimal value
    for it in range(n_iter):
        if time.time() >= end_t:
            break
        check_deadline()
        scores = compute_scores(h, n)
        f_val = float(np.max(scores)) * dx
        active = np.where(scores >= np.max(scores) - 1e-9)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(scores))])
        # Subgradient
        sg = np.zeros(n, dtype=float)
        coef = 1.0 / float(active.size)
        for idx in active:
            add_lag_gradient(sg, int(idx), coef, h, n)
        sg *= dx
        if not np.all(np.isfinite(sg)):
            break
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        # Polyak step
        gap = f_val - c_opt
        if gap > 1e-12:
            step = gap / (sg_norm * sg_norm + 1e-12)
        else:
            step = 1.0 / (L * math.sqrt(it + 1))
        step = min(step, 0.05)  # cap step size
        h_new = h - step * sg
        h_new = as_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            state.update(best_h, n)
        h = h_new

    state.update(best_h, n)
    return best_h, best_c


def smooth_perturb(h, n, rng, amp=0.05):
    """Add small smooth perturbation."""
    h2 = h.copy()
    x = np.arange(n, dtype=float)
    for _ in range(rng.integers(1, 4)):
        freq = 2.0 * math.pi * rng.integers(1, 5) / n * 2.0
        phase = rng.uniform(0, 2 * math.pi)
        h2 += amp * amp * np.sin(freq * x + phase)
    return as_feasible(h2, n)


def structured_start(n, rng):
    """Generate structured starting points."""
    candidates = []
    # Cosine family
    for f in (1, 2, 3):
        for amp in (0.2, 0.3, 0.4):
            x = np.arange(n, dtype=float) * 2.0 / n
            h = 0.5 + amp * np.cos(2.0 * math.pi * f * x / 2.0)
            h = as_feasible(h, n)
            candidates.append(h)
    # Two-level
    target = n // 2
    for _ in range(3):
        idx = rng.choice(n, size=target, replace=False)
        h = np.zeros(n)
        h[idx] = 1.0
        h = as_feasible(h, n)
        # Smooth it
        h = 0.6 * h + 0.4 * np.convolve(h, np.ones(5)/5.0, mode='same')
        h = as_feasible(h, n)
        candidates.append(h)
    return candidates


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 15.0 if budget_s > 60.0 else max(2.0, 0.1 * budget_s)
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = global_deadline

    rng = np.random.default_rng(seed)

    parent = get_parent()
    if parent is not None and parent.size > 0:
        n0 = int(parent.size)
        h0 = as_feasible(parent, n0)
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = as_feasible(np.full(n0, 0.5), n0)
        # Add structure
        x = np.arange(n0, dtype=float) * 2.0 / n0
        h0 = 0.5 + 0.3 * np.cos(2.0 * math.pi * x)
        h0 = as_feasible(h0, n0)
        c0 = compute_c5(h0)

    state = OptState(h0.copy(), c0, n0)

    # Phase 1: Optimize at parent resolution
    if time.time() < global_deadline - 200.0:
        t1 = min(300.0, global_deadline - time.time() - 200.0)
        slsqp_refine(h0, n0, state, t1, alpha_start=0.08, alpha_end=0.0)

    # Phase 2: Try higher resolution
    ns_list = []
    if n0 % 2 == 0:
        ns_list.append(2 * n0)
    # Try n=100, 120, 140 if reasonable
    for nn in (80, 100, 120, 140):
        if nn != n0 and nn != 2 * n0 and nn <= 4 * n0:
            ns_list.append(nn)
    # Remove duplicates and sort
    ns_list = sorted(set(ns_list))

    for nn in ns_list:
        if time.time() >= global_deadline - 100.0:
            break
        remaining = global_deadline - time.time()
        if remaining < 120.0:
            break

        # Create starting point for this resolution
        if nn == n0:
            h_start = state.h
        else:
            h_start = upsample_linear(state.h, nn)
            h_start = as_feasible(h_start, nn)
            # Also try upsampled parent
            if parent is not None:
                h_parent_up = upsample_linear(parent, nn)
                h_parent_up = as_feasible(h_parent_up, nn)
                c_parent_up = compute_c5(h_parent_up)
                if c_parent_up < compute_c5(h_start):
                    h_start = h_parent_up

        t_alloc = min(200.0, remaining - 100.0)
        if t_alloc < 60.0:
            break
        slsqp_refine(h_start, nn, state, t_alloc, alpha_start=0.06, alpha_end=0.0)

    # Phase 3: Subgradient polishing at best resolution
    if time.time() < global_deadline - 30.0:
        remaining = global_deadline - time.time() - 30.0
        if remaining > 30.0:
            subgradient_polish(state.h, state.n, state, remaining, n_iter=20000)

    # Phase 4: Final SLSQP polish at best resolution
    if time.time() < global_deadline - 15.0:
        remaining = global_deadline - time.time() - 15.0
        if remaining > 20.0:
            n_best = state.n
            bounds = [(0.0, 1.0)] * n_best
            cons = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]
            h_current = as_feasible(state.h, n_best)
            old_dl = _RUN_DEADLINE
            _RUN_DEADLINE = global_deadline - 10.0
            try:
                res = minimize(
                    softmax_objective, h_current, args=(0.0, n_best),
                    jac=softmax_gradient,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": int(300.0 * (80.0 / float(n_best))),
                             "ftol": 1e-12, "disp": 0},
                )
                if res is not None and getattr(res, "x", None) is not None:
                    state.update(res.x, n_best)
            except (BudgetError, Exception):
                pass
            finally:
                _RUN_DEADLINE = old_dl

    # Phase 5: Try a few random perturbations + re-optimize
    if time.time() < global_deadline - 20.0:
        remaining = global_deadline - time.time() - 20.0
        n_best = state.n
        for _ in range(3):
            if time.time() >= global_deadline - 20.0:
                break
            h_pert = smooth_perturb(state.h, n_best, rng, amp=0.03)
            c_pert = compute_c5(h_pert)
            if c_pert < state.c + 0.002:
                t_p = min(60.0, remaining / 2.0)
                if t_p > 20.0:
                    slsqp_refine(h_pert, n_best, state, t_p, alpha_start=0.02, alpha_end=0.0)

    # Final: ensure validity
    h_final = as_feasible(state.h, state.n)
    c_final = compute_c5(h_final)
    n_final = int(state.n)

    if not is_valid_candidate(h_final):
        h_final = as_feasible(h_final, n_final)
        c_final = compute_c5(h_final)

    # Also check parent
    if parent is not None:
        c_parent = compute_c5(parent)
        if c_parent < c_final:
            h_final = parent.copy()
            c_final = c_parent
            n_final = int(parent.size)

    return h_final, float(c_final), int(n_final)
