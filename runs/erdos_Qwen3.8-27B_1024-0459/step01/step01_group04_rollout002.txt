```
def add_lag_grad(grad, idx, coef, h, n):
   if coef == 0: return
   if idx == 0:
      grad += coef * (1.0 - 2.0*h)
      return
   if idx % 2 == 1:
      d=(idx+1)//2
   else:
      d=idx//2
   m=n-d
   if m <=0: return
   if idx % 2 ==1:
      grad[:m] += coef*(1.0 - h[d:])
      grad[d:] += -coef*h[:m]
   else:
      grad[:m] += -coef*h[d:]
      grad[d:] += coef*(1.0 - h[:m])
```

```python
import numpy as np
import time
import math
from scipy.optimize import minimize

_RUN_DEADLINE = None


class BudgetError(Exception):
    pass


def check_deadline():
    if _RUN_DEADLINE is not None and time.time() >= _RUN_DEADLINE:
        raise BudgetError


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv) * (2.0 / float(n)))
    if not math.isfinite(c):
        return 1e6
    return c


def as_feasible(h, n):
    n = int(n)
    h = np.array(h, dtype=float, copy=True)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n, dtype=float)
            m = min(n, h.size)
            if m > 0:
                tmp[:m] = h[:m]
            h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)

    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) <= 1e-13:
        return h

    lo = -1.0
    hi = 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            order = np.argsort(1.0 - h)
            for idx in order:
                if diff <= 1e-14:
                    break
                add = min(diff, 1.0 - float(h[idx]))
                if add <= 0.0:
                    continue
                h[idx] += add
                diff -= add
        else:
            order = np.argsort(h)
            need = -diff
            for idx in order:
                if need <= 1e-14:
                    break
                dec = min(need, float(h[idx]))
                if dec <= 0.0:
                    continue
                h[idx] -= dec
                need -= dec

    h = np.clip(h, 0.0, 1.0)
    return h


def resample_linear(parent, n):
    n = int(n)
    parent = np.asarray(parent, dtype=float)
    n0 = parent.size
    if n0 == n:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5, dtype=float)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    return np.interp(x, x0, parent)


def threshold_binary(h, n):
    n = int(n)
    h = resample_linear(np.asarray(h, dtype=float), n)
    order = np.argsort(h, kind="mergesort")
    out = np.zeros(n, dtype=float)
    if n > 0:
        out[order[n // 2:]] = 1.0
    return out


def block_pattern(n, rng, min_len=2, max_len=10):
    n = int(n)
    h = np.empty(n, dtype=float)
    pos = 0
    val = float(rng.integers(2))
    min_len = max(1, int(min_len))
    max_len = max(min_len, int(max_len))
    while pos < n:
        length = int(rng.integers(min_len, max_len + 1))
        length = min(length, n - pos)
        if length <= 0:
            length = 1
        h[pos:pos + length] = val
        val = 1.0 - val
        pos += length

    target = n // 2
    ones = int(np.sum(h > 0.5))
    if ones < target:
        zeros = np.flatnonzero(h <= 0.5)
        k = target - ones
        if k > 0 and zeros.size > 0:
            if zeros.size >= k:
                idx = zeros[rng.choice(zeros.size, k, replace=False)]
            else:
                idx = zeros
                k = zeros.size
            if k > 0:
                h[idx] = 1.0
    elif ones > target:
        ones_idx = np.flatnonzero(h > 0.5)
        k = ones - target
        if k > 0 and ones_idx.size > 0:
            if ones_idx.size >= k:
                idx = ones_idx[rng.choice(ones_idx.size, k, replace=False)]
            else:
                idx = ones_idx
                k = ones_idx.size
            if k > 0:
                h[idx] = 0.0
    return h

def init_scores_float(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    scores = np.zeros(2 * n - 1, dtype=float)
    if n == 0:
        return scores

    total = float(np.sum(h))
    ac = np.correlate(h, h, mode="full")
    ac = ac[n - 1:n - 1 + n]

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])

    scores[0] = total - float(np.dot(h, h))
    if n > 1:
        d = np.arange(1, n, dtype=int)
        N = n - d
        scores[1::2] = prefix[N] - ac[1:]
        scores[2::2] = total - prefix[d] - ac[1:]
    return scores


def apply_change(p, new_val, h, scores):
    p = int(p)
    n = h.size
    old = float(h[p])
    new = float(new_val)
    if new < 0.0:
        new = 0.0
    elif new > 1.0:
        new = 1.0
    delta = new - old
    if abs(delta) < 1e-15:
        return

    scores[0] += delta - (new * new - old * old)

    if p > 0:
        left = h[p - 1::-1]
        scores[1:2 * p:2] -= delta * left
        scores[2:2 * p + 1:2] += delta * (1.0 - left)

    m = n - p - 1
    if m > 0:
        right = h[p + 1:]
        scores[1:2 * m:2] += delta * (1.0 - right)
        scores[2:2 * m + 1:2] -= delta * right

    h[p] = new


def energy_from_max(scores, alpha, m):
    m = float(m)
    if alpha <= 1e-12:
        return m
    s = (scores - m) / float(alpha)
    A = float(np.sum(np.exp(s)))
    if not math.isfinite(A) or A <= 0.0:
        return m
    return m + float(alpha) * math.log(A)


def choose_delta(h, p, q, rng):
    maxd = min(float(h[q]), 1.0 - float(h[p]))
    if maxd <= 1e-12:
        return 0.0
    u = rng.random()
    if u < 0.60:
        return float(maxd)
    return float(maxd * rng.uniform(0.25, 1.0))


def random_high_low(h, n, rng):
    n = int(n)
    for _ in range(25):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if h[i] > h[j] + 1e-12:
            q = i
            p = j
        elif h[j] > h[i] + 1e-12:
            q = j
            p = i
        else:
            if rng.random() < 0.5:
                p = i
                q = j
            else:
                p = j
                q = i
        delta = choose_delta(h, p, q, rng)
        if delta > 1e-12:
            return int(p), int(q), float(delta)
    return -1, -1, 0.0


def zero_move(h, n, rng):
    frac = np.flatnonzero((h > 0.02) & (h < 0.98))
    if frac.size < 2:
        return random_high_low(h, n, rng)
    if frac.size > 30:
        samp = frac[rng.choice(frac.size, 30, replace=False)]
    else:
        samp = frac

    imax = int(samp[np.argmax(h[samp])])
    imin = int(samp[np.argmin(h[samp])])
    if imax == imin:
        pair = rng.choice(frac.size, 2, replace=False)
        imax = int(frac[pair[0]])
        imin = int(frac[pair[1]])

    p_inc = imax
    q_dec = imin
    if float(h[p_inc]) < float(h[q_dec]):
        p_inc, q_dec = q_dec, p_inc

    delta = choose_delta(h, p_inc, q_dec, rng)
    if delta > 1e-12:
        return int(p_inc), int(q_dec), float(delta)
    return random_high_low(h, n, rng)


def propose_move(h, scores, n, cur_max, rng):
    n = int(n)
    if n < 2:
        return -1, -1, 0.0

    if rng.random() < 0.62:
        if rng.random() < 0.35:
            idx = int(np.argmax(scores))
        else:
            tol = max(0.25, 0.05 * max(1.0, float(cur_max)))
            act = np.flatnonzero(scores >= float(cur_max) - tol)
            if act.size == 0:
                idx = int(np.argmax(scores))
            else:
                idx = int(act[int(rng.integers(act.size))])

        if idx == 0:
            return zero_move(h, n, rng)
        if idx >= 2 * n - 1:
            idx = 1

        if idx % 2 == 1:
            d = (idx + 1) // 2
        else:
            d = idx // 2

        if d < 1 or d >= n:
            return random_high_low(h, n, rng)

        for _ in range(12):
            i = int(rng.integers(0, n - d))
            a = float(h[i])
            b = float(h[i + d])
            if idx % 2 == 1:
                if a > b:
                    p_inc = i + d
                    q_dec = i
                    delta = choose_delta(h, p_inc, q_dec, rng)
                    if delta > 1e-12:
                        return int(p_inc), int(q_dec), float(delta)
            else:
                if b > a:
                    p_inc = i
                    q_dec = i + d
                    delta = choose_delta(h, p_inc, q_dec, rng)
                    if delta > 1e-12:
                        return int(p_inc), int(q_dec), float(delta)

        return random_high_low(h, n, rng)

    return random_high_low(h, n, rng)


def perturb_best(ref, n, rng, strength=0.05):
    n = int(n)
    h = np.array(ref, dtype=float, copy=True)
    if h.size != n:
        h = as_feasible(h, n)

    if n > 10 and rng.random() < min(0.8, 0.3 + float(strength)):
        l = int(rng.integers(0, n - 5))
        r = int(rng.integers(l + 5, n))
        h[l:r] = h[l:r][::-1]

    swaps = int(max(5, float(strength) * float(n)))
    for _ in range(swaps):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if h[i] > h[j]:
            q = i
            p = j
        elif h[j] > h[i]:
            q = j
            p = i
        else:
            if rng.random() < 0.5:
                p = i
                q = j
            else:
                p = j
                q = i
        delta = choose_delta(h, p, q, rng)
        if delta > 1e-12:
            h[p] += delta
            h[q] -= delta

    return as_feasible(h, n)


def maybe_update(holder, h, n):
    if h is None:
        return 1e30
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or h.size == 0:
        return 1e30
    n = int(n)
    if h.size != n:
        return 1e30
    try:
        h2 = as_feasible(h, n)
        c2 = compute_c5(h2)
    except Exception:
        return 1e30
    if not math.isfinite(c2):
        return 1e30
    if holder is not None and c2 < float(holder[1]) - 1e-12:
        holder[0] = h2.copy()
        holder[1] = float(c2)
        holder[2] = n
    return float(c2)


def continuous_local_search(start_h, n, time_limit, rng, holder):
    n = int(n)
    if n < 4 or time_limit <= 5.0:
        return None, 1e30

    h = as_feasible(start_h, n)
    scores = init_scores_float(h)
    cur_max = float(np.max(scores))

    local_best_h = h.copy()
    local_best_c = maybe_update(holder, local_best_h, n)

    if holder is not None and holder[0] is not None and int(holder[2]) == n:
        if float(holder[1]) < float(local_best_c) - 1e-12:
            local_best_h = as_feasible(holder[0], n)
            local_best_c = float(holder[1])

    alpha = max(2.5, 0.04 * float(n))
    temp0 = max(1.5, 0.25 * alpha)
    temp_end = 0.2

    cur_energy = energy_from_max(scores, alpha, cur_max)

    if n <= 100:
        max_ops = 700000
        kick_threshold = 3000
    elif n <= 300:
        max_ops = 250000
        kick_threshold = 2500
    else:
        max_ops = 90000
        kick_threshold = 1500

    start_t = time.time()
    ops = 0
    stall = 0
    kicks = 0

    while ops < max_ops:
        if time.time() - start_t >= time_limit:
            break
        if _RUN_DEADLINE is not None and time.time() >= _RUN_DEADLINE:
            break

        frac = (time.time() - start_t) / max(1.0, float(time_limit))
        if frac > 1.0:
            frac = 1.0
        cur_temp = temp0 * (1.0 - frac) * (1.0 - frac) + temp_end

        p_idx, q_idx, delta = propose_move(h, scores, n, cur_max, rng)
        if p_idx >= 0 and q_idx >= 0 and delta > 1e-12:
            old_p = float(h[p_idx])
            old_q = float(h[q_idx])
            maxd = min(old_q, 1.0 - old_p)
            delta = min(float(delta), maxd)
            if delta > 1e-12:
                new_p = old_p + delta
                new_q = old_q - delta

                apply_change(p_idx, new_p, h, scores)
                apply_change(q_idx, new_q, h, scores)

                new_max = float(np.max(scores))
                new_energy = energy_from_max(scores, alpha, new_max)

                accept = False
                if new_max < cur_max - 1e-10:
                    accept = True
                elif abs(new_max - cur_max) <= 1e-10:
                    if new_energy < cur_energy - 1e-10:
                        accept = True

                max_jump = max(2.0, 0.02 * float(n))
                if (not accept) and cur_temp > 1e-8 and new_max <= cur_max + max_jump:
                    diff = new_energy - cur_energy
                    if diff < 0.0:
                        accept = True
                    elif diff < 50.0:
                        accept = bool(rng.random() < math.exp(-diff / cur_temp))

                new_c = (new_max * 2.0) / float(n)
                if new_c < float(local_best_c) - 1e-12:
                    local_best_h = h.copy()
                    local_best_c = maybe_update(holder, local_best_h, n)
                    stall = 0

                if accept:
                    cur_max = new_max
                    cur_energy = new_energy
                    if ((cur_max * 2.0) / float(n)) >= float(local_best_c) - 1e-12:
                        stall += 1
                else:
                    apply_change(q_idx, old_q, h, scores)
                    apply_change(p_idx, old_p, h, scores)
                    stall += 1
            else:
                stall += 1
        else:
            stall += 1

        ops += 1

        if stall > kick_threshold:
            strength = 0.03 + 0.02 * float(kicks)
            if strength > 0.25:
                strength = 0.25
            kicks += 1
            try:
                h = perturb_best(local_best_h, n, rng, strength)
                h = as_feasible(h, n)
                scores = init_scores_float(h)
                cur_max = float(np.max(scores))
                cur_energy = energy_from_max(scores, alpha, cur_max)
            except Exception:
                h = as_feasible(local_best_h, n)
                scores = init_scores_float(h)
                cur_max = float(np.max(scores))
                cur_energy = energy_from_max(scores, alpha, cur_max)
            stall = 0
            if kicks > 14:
                break

    local_best_h = as_feasible(local_best_h, n)
    local_best_c = maybe_update(holder, local_best_h, n)
    return local_best_h, local_best_c


def slsqp_scores(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    scores = np.empty(2 * n - 1, dtype=float)
    total = float(np.sum(h))
    ac = np.correlate(h, h, mode="full")
    ac = ac[n - 1:n - 1 + n]

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])

    scores[0] = total - float(np.dot(h, h))
    if n > 1:
        d = np.arange(1, n, dtype=int)
        N = n - d
        scores[1::2] = prefix[N] - ac[1:]
        scores[2::2] = total - prefix[d] - ac[1:]
    m = float(np.max(scores))
    return scores, m


def slsqp_objective(x, alpha):
    check_deadline()
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0 or not np.all(np.isfinite(x)):
        return 1e6
    h = np.clip(x, 0.0, 1.0)
    scores, m = slsqp_scores(h)

    if alpha > 1e-10:
        s = (scores - m) / float(alpha)
        A = float(np.sum(np.exp(s)))
        if not math.isfinite(A) or A <= 0.0:
            val = m
        else:
            val = m + float(alpha) * math.log(A)
    else:
        val = m

    if not math.isfinite(val):
        return 1e6
    return float(val) * (2.0 / float(n))


def add_lag_grad(grad, idx, coef, h, n):
    if coef == 0.0:
        return
    if idx == 0:
        grad += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
    else:
        d = idx // 2
    m = n - d
    if m <= 0:
        return
    if idx % 2 == 1:
        grad[:m] += coef * (1.0 - h[d:])
        grad[d:] += -coef * h[:m]
    else:
        grad[:m] += -coef * h[d:]
        grad[d:] += coef * (1.0 - h[:m])


def slsqp_gradient(x, alpha):
    check_deadline()
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return np.zeros(0, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)

    h = np.clip(x, 0.0, 1.0)
    scores, m = slsqp_scores(h)
    grad = np.zeros(n, dtype=float)

    if alpha <= 1e-10:
        tol = max(1e-8, 1e-7 * max(1.0, float(m)))
        active = np.flatnonzero(scores >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(scores))], dtype=int)
        coef = 1.0 / float(active.size)
        for idx in active:
            add_lag_grad(grad, int(idx), coef, h, n)
    else:
        s = (scores - m) / float(alpha)
        w = np.exp(s)
        A = float(np.sum(w))
        if math.isfinite(A) and A > 0.0:
            w = w / A
            for idx in range(w.size):
                wi = float(w[idx])
                if wi > 1e-10:
                    add_lag_grad(grad, int(idx), wi, h, n)

    grad *= 2.0 / float(n)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=float)
    return grad


def constraint_eq(x, *args):
    n = max(1, int(x.size))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)


def constraint_jac(x, *args):
    n = max(1, int(x.size))
    return np.full((1, n), 1.0 / float(n), dtype=float)


def polish_slsqp(start_h, n, holder, time_limit):
    n = int(n)
    if n < 4 or time_limit < 30.0:
        h = as_feasible(start_h, n)
        c = compute_c5(h)
        maybe_update(holder, h, n)
        return h, c

    h = as_feasible(start_h, n)
    c = compute_c5(h)
    maybe_update(holder, h, n)

    if holder is not None and holder[0] is not None and int(holder[2]) == n:
        if float(holder[1]) < float(c) - 1e-12:
            h = as_feasible(holder[0], n)
            c = float(holder[1])

    old_deadline = _RUN_DEADLINE
    rem = time_limit
    if old_deadline is not None:
        rem = min(rem, max(1.0, float(old_deadline) - time.time() - 8.0))
    stage = time.time() + max(5.0, rem)
    if old_deadline is not None:
        stage = min(stage, float(old_deadline))

    _RUN_DEADLINE = stage
    try:
        bounds = [(0.0, 1.0)] * n
        constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]

        alpha1 = max(1.0, 0.02 * float(n))
        maxiter1 = 90 if n <= 100 else 55
        try:
            res1 = minimize(
                slsqp_objective,
                h,
                args=(alpha1,),
                jac=slsqp_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter1, "ftol": 1e-10, "disp": 0},
            )
        except BudgetError:
            res1 = None
        except Exception:
            res1 = None

        if res1 is not None and getattr(res1, "x", None) is not None:
            h1 = as_feasible(res1.x, n)
            c1 = compute_c5(h1)
            maybe_update(holder, h1, n)
            if math.isfinite(c1) and c1 < c - 1e-12:
                h = h1
                c = c1
            if holder is not None and holder[0] is not None and int(holder[2]) == n:
                if float(holder[1]) < c - 1e-12:
                    h = as_feasible(holder[0], n)
                    c = float(holder[1])

        if old_deadline is None or time.time() + 40.0 < float(stage):
            alpha2 = max(0.05, 0.001 * float(n))
            maxiter2 = 70 if n <= 100 else 45
            try:
                res2 = minimize(
                    slsqp_objective,
                    h,
                    args=(alpha2,),
                    jac=slsqp_gradient,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": maxiter2, "ftol": 1e-10, "disp": 0},
                )
            except BudgetError:
                res2 = None
            except Exception:
                res2 = None

            if res2 is not None and getattr(res2, "x", None) is not None:
                h2 = as_feasible(res2.x, n)
                c2 = compute_c5(h2)
                maybe_update(holder, h2, n)
                if math.isfinite(c2) and c2 < c - 1e-12:
                    h = h2
                    c = c2
                if holder is not None and holder[0] is not None and int(holder[2]) == n:
                    if float(holder[1]) < c - 1e-12:
                        h = as_feasible(holder[0], n)
                        c = float(holder[1])
    finally:
        _RUN_DEADLINE = old_deadline

    h = as_feasible(h, n)
    c = compute_c5(h)
    maybe_update(holder, h, n)
    return h, c


def cand_key(item):
    return item[1]


def add_candidate(cands, h, n, limit=3):
    if len(cands) >= limit:
        return
    if h is None:
        return
    n = int(n)
    try:
        h2 = as_feasible(h, n)
        c2 = compute_c5(h2)
    except Exception:
        return
    if not math.isfinite(c2):
        return
    for hh, _ in cands:
        if hh.size == h2.size and np.allclose(hh, h2, rtol=1e-6, atol=1e-6):
            return
    cands.append((h2, float(c2)))


def make_starts(n, parent, holder, rng, limit=2):
    n = int(n)
    cands = []
    limit = max(1, int(limit))

    if parent is not None:
        hp = resample_linear(parent, n)
        add_candidate(cands, hp, n, limit)
        hb = threshold_binary(hp, n)
        add_candidate(cands, hb, n, limit)

    if holder is not None and holder[0] is not None:
        hb = holder[0]
        hbr = resample_linear(hb, n)
        add_candidate(cands, hbr, n, limit)
        hbt = threshold_binary(hbr, n)
        add_candidate(cands, hbt, n, limit)

    if n <= 500:
        ml = max(2, n // 40)
        xl = max(ml + 1, n // 8)
        hb = block_pattern(n, rng, ml, xl)
        add_candidate(cands, hb, n, limit)

    if len(cands) == 0:
        h0 = np.full(n, 0.5, dtype=float)
        add_candidate(cands, h0, n, limit)

    cands.sort(key=cand_key)
    if len(cands) > limit:
        cands = cands[:limit]
    return cands


def build_base_tasks(n0):
    n0 = int(n0)
    if n0 < 4:
        n0 = 4
    if n0 % 2 != 0:
        n0 += 1

    tasks = [(n0, 0.42)]

    specs = [
        (1.5, 0.10),
        (2.0, 0.20),
        (2.5, 0.08),
        (3.0, 0.10),
        (4.0, 0.10),
    ]

    for mult, wt in specs:
        n = int(round(float(mult) * float(n0)))
        if n % 2 != 0:
            n += 1
        if n < 6 or n > 800:
            continue
        duplicate = False
        for en, _ in tasks:
            if en == n:
                duplicate = True
                break
        if not duplicate:
            tasks.append((n, wt))

    total_w = sum(w for _, w in tasks)
    if total_w <= 0.0:
        return tasks
    return [(n, w / total_w) for n, w in tasks]


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE

    t0 = time.time()
    try:
        b = float(budget_s)
    except Exception:
        b = 1300.0
    if not math.isfinite(b) or b <= 0.0:
        b = 30.0

    reserve = 20.0 if b > 60.0 else max(2.0, 0.1 * b)
    _RUN_DEADLINE = t0 + max(1.0, b - reserve)

    rng = np.random.default_rng(seed)
    parent = get_parent()
    holder = [None, 1e30, 0]

    if parent is not None and parent.size >= 4:
        n0 = int(parent.size)
        if n0 % 2 != 0:
            n0 += 1
            h0 = resample_linear(parent, n0)
        else:
            h0 = parent.astype(float).copy()
        h0 = as_feasible(h0, n0)
    else:
        n0 = 70
        h0 = synthetic_fallback(n0, rng)
        h0 = as_feasible(h0, n0)

    maybe_update(holder, h0, n0)
    initial_h = h0.copy()
    initial_n = int(holder[2]) if int(holder[2]) > 0 else int(n0)
    initial_h = as_feasible(initial_h, initial_n)
    initial_c = compute_c5(initial_h)

    if _RUN_DEADLINE is not None and time.time() + 45.0 >= float(_RUN_DEADLINE):
        return initial_h, float(initial_c), int(initial_n)

    tasks = build_base_tasks(n0)
    avail = max(10.0, min(b - 1.5 * reserve, float(_RUN_DEADLINE - t0)))
    polished = {}

    for n_task, wt in tasks:
        if _RUN_DEADLINE is None or time.time() >= float(_RUN_DEADLINE) - 30.0:
            break
        rem_global = max(0.0, float(_RUN_DEADLINE - time.time()))
        if rem_global < 40.0:
            break

        lim = min(float(wt) * float(avail), rem_global - 25.0)
        if lim < 25.0:
            continue
        if n_task < 6 or n_task > 1200:
            continue

        cands = make_starts(n_task, parent, holder, rng, 2)
        if not cands:
            continue

        task_time = lim
        k = len(cands)

        for idx_cand in range(k):
            if task_time < 25.0:
                break
            if _RUN_DEADLINE is None or time.time() >= float(_RUN_DEADLINE) - 30.0:
                break

            allocated = min(
                task_time / float(k - idx_cand),
                max(20.0, float(_RUN_DEADLINE - time.time()) - 40.0),
            )
            if allocated <= 20.0:
                break

            t_cand = time.time()
            h_res, c_res = continuous_local_search(cands[idx_cand][0], n_task, allocated, rng, holder)
            task_time -= max(0.0, time.time() - t_cand)

            if n_task not in polished and n_task <= 200:
                rem_pol = max(0.0, float(_RUN_DEADLINE - time.time()) - 25.0)
                if task_time > 40.0 and rem_pol > 80.0:
                    if holder is not None and holder[0] is not None and int(holder[2]) == n_task:
                        pol_h = holder[0]
                    else:
                        pol_h = h_res if h_res is not None else cands[idx_cand][0]
                    # Also try the local result if it is clearly better than the global holder at this n.
                    if h_res is not None:
                        hr = compute_c5(h_res)
                        if hr < (float(holder[1]) if holder is not None else 1e30) + 1e-9:
                            pol_h = h_res

                    if pol_h is not None:
                        t_pol = time.time()
                        pol_h, pol_c = polish_slsqp(pol_h, n_task, holder, min(90.0, task_time - 20.0, rem_pol - 20.0))
                        task_time -= max(0.0, time.time() - t_pol)
                        polished[n_task] = 1

    if holder[0] is not None and _RUN_DEADLINE is not None:
        nf = int(holder[2])
        rem_final = max(0.0, float(_RUN_DEADLINE - time.time()) - 25.0)
        if rem_final > 90.0:
            if nf <= 200:
                if nf not in polished:
                    t_pol = time.time()
                    polish_slsqp(holder[0], nf, holder, min(80.0, rem_final - 20.0))
                    rem_final = max(0.0, float(_RUN_DEADLINE - time.time()) - 25.0)
                if rem_final > 70.0:
                    h_res, _ = continuous_local_search(holder[0], nf, min(80.0, rem_final - 25.0), rng, holder)
                    if rem_final > 40.0:
                        polish_slsqp(h_res if h_res is not None else holder[0], nf, holder, min(50.0, rem_final - 20.0))
            else:
                continuous_local_search(holder[0], nf, min(120.0, rem_final - 30.0), rng, holder)

    if holder[0] is None:
        h_final = initial_h
        c_final = initial_c
        n_final = initial_n
    else:
        h_final = holder[0]
        c_final = float(holder[1])
        n_final = int(holder[2])

    try:
        h_final = as_feasible(h_final, n_final)
        c_final = compute_c5(h_final)
    except Exception:
        h_final = initial_h
        c_final = initial_c
        n_final = initial_n

    if not math.isfinite(c_final) or c_final > 1e5:
        h_final = initial_h
        c_final = initial_c
        n_final = initial_n

    if initial_c < c_final - 1e-12:
        h_final = initial_h
        c_final = initial_c
        n_final = initial_n

    return np.asarray(h_final, dtype=float), float(c_final), int(n_final)
```
