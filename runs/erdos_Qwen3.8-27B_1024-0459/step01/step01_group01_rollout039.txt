    # Use bisection on threshold t
    # x_i = clip(h_i - t, 0, 1)
    # Find t such that sum(x) = c

    for _ in range(60):
        t = (lo + hi) / 2
        x = np.clip(h - t, 0, 1)
        s = x.sum()
        if s > c:
            lo = t  # need to subtract more
        else:
            hi = t  # need to subtract less
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Final correction
    diff = c - x.sum()
    if diff > 0:
        idx = np.where(x < 1)[0]
    else:
        idx = np.where(x > 0)[0]
    if len(idx) > 0:
        x[idx] += diff / len(idx)
        x = np.clip(x, 0, 1)
    return x

def compute_grad(h, w_full, n, dx):
    """
    h: length n
    w_full: length 2n-1, where w_full[k] is weight for shift s = k - (n-1)
    """
    # Build the two Toeplitz-vector products
    # T1 @ (1-h): (T1 @ x)[i] = sum_j T1[i,j] * x[j] where T1[i,j] = w_full[j-i+n-1]
    # This is the "upper triangular part" of the Toeplitz with the full series w_full
    
    # Using FFT: a Toeplitz matrix-vector product can be done via embedding in a circulant.
    # The first column of T1 is: w_full[n-1], w_full[n-2], ..., w_full[0] (length n)
    # The first row of T1 is: w_full[n-1], w_full[n], ..., w_full[2n-2] (length n)
    
    # For FFT-based Toeplitz matvec:
    # Pad the first column to length 2n-1 (or next power of 2), then:
    # result = ifft(fft(padded_col) * fft(x_padded))[:n]
    # where padded_col = [first_col, 0, ..., 0, first_row[-1], ..., first_row[1]]
    # Hmm, this is the standard trick.
    
    # Actually, let me just use scipy.linalg.toeplitz for n <= 500 or so,
    # and FFT for larger n.
    
    # For simplicity and correctness, let me use the direct approach with np.einsum
    # or a loop over shifts (which is O(n^2) but with good cache behavior).
    
    grad = np.zeros(n)
    om_h = 1.0 - h
    for s in range(-(n-1), n):
        w_s = w_full[s + n - 1]
        if w_s < 1e-15:
            continue
        # Contribution to grad from shift s:
        # w_s * (om_h[i+s] - h[i-s]) for valid i
        i_lo = max(0, -s)      # so that i+s >= 0
        i_hi = min(n, n - s) if s > 0 else n  # so that i+s < n
        j_lo = max(0, s)       # so that i-s >= 0
        j_hi = min(n, n + s) if s < 0 else n  # so that i-s < n
        
        # Actually, let me be more careful:
        # For term1: need 0 <= i+s < n, so i >= -s and i < n-s
        #   i ranges from max(0, -s) to min(n-1, n-1-s)
        # For term2: need 0 <= i-s < n, so i >= s and i < n+s
        #   i ranges from max(0, s) to min(n-1, n+s-1)

        i1_lo = max(0, -s)
        i1_hi = min(n-1, n-1-s)
        if i1_lo <= i1_hi:

Wait, the coordinate descent as written is O(n² × 4) per sweep (calling compute_c5 4n times per sweep, each O(n²)). For n=720, that's 4*720*720² ≈ 1.5 billion ops per sweep. Way too slow!

Let me fix the coordinate descent. Instead of checking 4 values per coordinate, I can derive the optimal value analytically.

When we change h[i] from old_val to new_val, the change in corr[k] is:
Δcorr[k] = new_val * (1-h[i+s_k]) - old_val * (1-h[i+s_k]) - new_val * h[i-s_k] + old_val * h[i-s_k]
= (new_val - old_val) * [(1-h[i+s_k]) - h[i-s_k]]

for valid indices.

So corr_new[k] = corr_old[k] + (new_val - old_val) * d_k

where d_k = [(1-h[i+s_k]) - h[i-s_k]] (0 if both invalid)

The objective is max_k (corr_new[k] * dx) = dx * max_k (corr_old[k] + (new_val - old_val) * d_k)

This is a convex piecewise-linear function of new_val. The minimum is at a "kink" point, which occurs when two terms are equal:
corr_old[k1] + (new_val - old_val) * d_{k1} = corr_old[k2] + (new_val - old_val) * d_{k2}
(new_val - old_val) * (d_{k1} - d_{k2}) = corr_old[k2] - corr_old[k1]
new_val = old_val + (corr_old[k2] - corr_old[k1]) / (d_{k1} - d_{k2})

So the optimal new_val is either 0, 1, or at one of these kink points.

But finding all kink points is O(n) per coordinate (there are at most 2n-1 terms, so O(n) pairs, giving O(n²) kink points). That's still O(n³) per sweep.

For n=720, that's 720³ ≈ 373M per sweep. Too slow.

Let me just use a few candidate values: 0, 1, and the current value ± small amounts. Or better, use the subgradient to determine the direction:

subgradient of max_k (corr_old[k] + t * d_k) at t=0:
If the max is unique at k*, the subgradient is d_{k*}.
If the max is shared by a set S, the subgradient is any convex combination of {d_k : k ∈ S}.

The direction of steepest decrease is -d_{k*} (if unique max). So:
- If d_{k*} > 0, decrease new_val (move h[i] down).
- If d_{k*} < 0, increase new_val (move h[i] up).

And the step size can be determined by the next kink point.

This is getting complicated. Let me just use a simpler approach for polishing: **a few iterations of the gradient method with very small step size** (essentially a local refinement).

import numpy as np
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def project_onto_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        t = (lo + hi) / 2.0
        x = np.clip(h - t, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = t
        else:
            hi = t
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - x.sum()
    if diff > 0:
        idx = np.where(x < 1.0 - 1e-10)[0]
    else:
        idx = np.where(x > 1e-10)[0]
    if len(idx) > 0:
        x[idx] += diff / len(idx)
        x = np.clip(x, 0.0, 1.0)
    return x


def smooth_max_and_weights(corr, tau):
    m = np.max(corr)
    shifted = (corr - m) / tau
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    logsum = max_shifted + np.log(np.sum(exps))
    smooth_val = m + tau * logsum
    weights = exps / np.sum(exps)
    return smooth_val, weights


def toeplitz_matvec(w_full, x, n, N):
    c = w_full[n - 1:0:-1]
    r = w_full[n - 1:2 * n - 1]
    v = np.zeros(N)
    v[:n] = c
    v[N - (n - 1):N] = r[n - 1:0:-1]
    X = np.zeros(N)
    X[:n] = x
    result = np.fft.ifft(np.fft.fft(v) * np.fft.fft(X)).real[:n]
    return result


def toeplitz_matvec_T(w_full, x, n, N):
    c_T = w_full[n - 1:2 * n - 1]
    r_T = w_full[n - 1:0:-1]
    v = np.zeros(N)
    v[:n] = c_T
    v[N - (n - 1):N] = r_T[n - 1:0:-1]
    X = np.zeros(N)
    X[:n] = x
    result = np.fft.ifft(np.fft.fft(v) * np.fft.fft(X)).real[:n]
    return result


def compute_gradient(h, w_full, n, dx, N):
    om_h = 1.0 - h
    term1 = toeplitz_matvec(w_full, om_h, n, N)
    term2 = toeplitz_matvec_T(w_full, h, n, N)
    return dx * (term1 - term2)


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def next_power_of_2(x):
    n = 1
    while n < x:
        n *= 2
    return n


def adam_descent(h, n, N, tau_schedule, lr_init, lr_decay, beta1, beta2, eps,
                 time_limit, check_fn):
    dx = 2.0 / n
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    adam_t = 0
    lr = lr_init
    
    best_local = compute_c5(h)
    best_local_h = h.copy()
    
    for tau_idx, tau in enumerate(tau_schedule):
        if check_fn():
            break
        max_iter = 2000
        for it in range(max_iter):
            if (it % 200 == 0) and check_fn():
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            smooth_val, weights = smooth_max_and_weights(corr, tau)
            grad = compute_gradient(h, weights, n, dx, N)
            
            adam_t += 1
            m_adam = beta1 * m_adam + (1.0 - beta1) * grad
            v_adam = beta2 * v_adam + (1.0 - beta2) * grad * grad
            m_hat = m_adam / (1.0 - beta1 ** adam_t)
            v_hat = v_adam / (1.0 - beta2 ** adam_t)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_onto_feasible(h_new, n)
            
            if smooth_val * dx < best_local - 1e-8:
                c5_check = compute_c5(h_new)
                if c5_check < best_local:
                    best_local = c5_check
                    best_local_h = h_new.copy()
            
            h = h_new
            
            if it % 500 == 0 and it > 0:
                lr *= lr_decay
    
    c5_final = compute_c5(h)
    if c5_final < best_local:
        best_local = c5_final
        best_local_h = h.copy()
    
    return best_local_h, best_local
