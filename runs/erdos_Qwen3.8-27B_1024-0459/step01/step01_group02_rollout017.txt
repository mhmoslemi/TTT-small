import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def corr_max_raw(h):
    """Return the max correlation value (before multiplying by dx)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    # Try uniform adjustment
    h_adj = h + diff / n
    if np.all(h_adj >= 0) and np.all(h_adj <= 1):
        return h_adj
    # Water-filling / bisection
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_mid = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final fix
    remaining = target - np.sum(h)
    if abs(remaining) > 1e-12:
        mask = (h > 0.01) & (h < 0.99)
        if np.sum(mask) > 0:
            h[mask] += remaining / np.sum(mask)
        else:
            h[np.argmax(h < 1.0)] += remaining if remaining > 0 else 0
    h = np.clip(h, 0.0, 1.0)
    return h

def cd_1d_min(slopes, intercepts):
    """Find t in [0,1] minimizing max_k (slopes[k]*t + intercepts[k])."""
    # Convex piecewise linear function
    # Check endpoints
    f0 = np.max(intercepts)
    f1 = np.max(slopes + intercepts)
    
    # If f is monotone
    if np.all(slopes >= 0):
        return 0.0
    if np.all(slopes <= 0):
        return 1.0
    
    # Bisection on subgradient
    # At t, active set = {k : slopes[k]*t + intercepts[k] == max}
    # Subgradient = convex hull of {slopes[k] : k active}
    lo, hi = 0.0, 1.0
    for _ in range(40):

        vals = slopes * t + intercepts
        mx = np.max(vals)
        active = vals >= mx - 1e-12
        grad_lo = np.min(slopes[active])
        grad_hi = np.max(slopes[active])
        if grad_lo >= 0:

            hi = t
        elif grad_hi <= 0:
            # f is decreasing at t, minimum is to the right
            lo = t
        else:
            # 0 is in subgradient, t is optimal
            break
    t_star = (lo + hi) / 2.0
    # Round to nearest 1/256 for stability
    return np.clip(t_star, 0.0, 1.0)

def coordinate_descent(h, n, max_sweeps=200, time_limit=300, start_time=None):
    """Run coordinate descent on h."""
    if start_time is None:
        start_time = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    no_improve = 0

        order = np.random.permutation(n)
        for j in order:

            # Compute corr with h[j]=0 and h[j]=1

            h[j] = h_save

            intercepts = corr_0
            
            t_star = cd_1d_min(slopes, intercepts)
            h[j] = t_star

37. **One more thought: the "sum constraint" in coordinate descent**

When I update h[j] from its current value to t*, the sum changes. I handle this by projecting at the end of each sweep. But this means that during the sweep, the sum constraint is violated, and the "optimal" t* is found without respecting the sum. This is an approximation.

A better approach: when updating h[j], also adjust h[j'] (some other coordinate) to maintain the sum. But this couples the coordinates.

Simplest: just project at the end of each sweep. In practice, this works well because the projection is a small adjustment (the sum drift is at most O(1/n) per coordinate update).

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = np.argmax(h < 1.0) if diff > 0 else np.argmin(h > 0.0)
            h[idx] += diff
    h = np.clip(h, 0.0, 1.0)
    return h


def cd_1d_min(slopes, intercepts):
    mx_s = np.max(slopes)
    mn_s = np.min(slopes)
    if mx_s <= 1e-14:
        return 1.0
    if mn_s >= -1e-14:
        return 0.0
    lo = 0.0
    hi = 1.0
    for _ in range(40):
        t = 0.5 * (lo + hi)
        vals = slopes * t + intercepts
        mx = np.max(vals)
        active = vals >= mx - 1e-10
        if not np.any(active):
            active = vals >= mx - 1.0
        g = slopes[active]
        if np.min(g) > 0.0:
            hi = t
        elif np.max(g) < 0.0:
            lo = t
        else:
            break
    return np.clip(0.5 * (lo + hi), 0.0, 1.0)


def coordinate_descent(h0, n, max_sweeps, time_limit, rng):
    h = h0.copy()
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stalls = 0
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        order = rng.permutation(n)
        for j in order:
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = cd_1d_min(c1 - c0, c0)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stalls = 0
        else:
            stalls += 1
            if stalls >= 15:
                break
    return best_h, best_c5


def smooth_obj_fn(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx


def slsqp_polish(h, n, beta, maxiter, time_limit):
    t0 = time.time()
    if time.time() - t0 > time_limit:
        return h, compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for b in [beta, beta * 3, beta * 9]:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj_fn(x, n, b), best_h,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5


def local_search(h, n, time_limit, rng):
    t0 = time.time()
    h = h.copy()
    c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, c5
    for iteration in range(50000):
        if time.time() - t0 > time_limit:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        save_i, save_j = h[i], h[j]
        h[i], h[j] = save_j, save_i
        new_c5 = compute_c5(h)
        if new_c5 <= c5 + 1e-14:
            c5 = new_c5
        else:
            h[i], h[j] = save_i, save_j
            if rng.rand() < 0.01:
                h[i], h[j] = save_j, save_i
                h = project_feasible(h, n)
    return h, decode_c5(h)


def decode_c5(h):
    return compute_c5(h)


def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def make_periodic(n, freq, rng):
    x = np.arange(n)
    phase = rng.rand() * 2.0 * np.pi
    h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / n + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def interpolate_to_resolution(src, n_new):
    n_old = len(src)
    if n_old == n_new:
        return src.copy()
    xp = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    val = np.interp(xn, xp, src)
    return project_feasible(val, int(n_new))
