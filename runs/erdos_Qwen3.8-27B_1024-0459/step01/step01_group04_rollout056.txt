def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv)) * (2.0 / n)
    return c if math.isfinite(c) else 1e6

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()

        h2 = np.zeros(n)
        m = min(n, h.size)
        if m > 0:

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid

                          constraints=[{"type": "eq", "fun": eq_fun, "jac": eq_jac}],
                          options={"maxiter": 80, "ftol": 1e-10})
            if res.x is not None:
                h_res = make_feasible(res.x, nb)
                tracker.update(h_res, nb)
        except Exception:
            pass
    
    # Final
    if tracker.h is not None:
        h_final = make_feasible(tracker.h, tracker.n)
        c_final = compute_c5(h_final)
        return h_final, float(c_final), int(tracker.n)
    
    h_fb = make_feasible(h_cont0, n0)
    return h_fb, compute_c5(h_fb), n0
