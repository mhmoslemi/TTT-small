import numpy as np
import time
import math

def project_sum(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1}"""
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    # Binary search for shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s_mid = np.clip(h + mid, 0, 1).sum()
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo+hi)/2, 0, 1)
    # Fix residual
    diff = target - h.sum()
    if diff > 0:
        mask = h < 1.0
        if mask.sum() > 0:
            h[mask] = np.minimum(1.0, h[mask] + diff / mask.sum())
    elif diff < 0:
        mask = h > 0.0
        if mask.sum() > 0:
            h[mask] = np.maximum(0.0, h[mask] + diff / mask.sum())
    return h

def compute_c5(h):
    n = h.size
    if n == 0:
        return 1e6
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / n)

def compute_gradient(h, alpha):
    """Compute gradient of soft-max objective. Returns (objective_value, gradient)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    
    if alpha < 1e-14:
        active = corr >= m - 1e-10
        w = np.zeros(2*n-1)
        w[active] = 1.0 / active.sum()
        obj = m
    else:
        exp_vals = np.exp((corr - m) / alpha)
        A = exp_vals.sum()
        w = exp_vals / A
        obj = m + alpha * np.log(A)
    
    grad = np.zeros(n)
    # d=0 term
    w0 = w[n-1]
    if w0 > 1e-15:
        grad += w0 * (1.0 - 2.0 * h)
    
    # d>0 and d<0 terms (combined)
    for m_d in range(1, n):
        w_comb = w[n-1+m_d] + w[n-1-m_d]
        if w_comb < 1e-15:
            continue
        grad[:n-m_d] += w_comb * (1.0 - h[m_d:])
        grad[m_d:] += w_comb * (-h[:n-m_d])
    
    return obj * (2.0/n), grad * (2.0/n)

def pgd_optimize(h0, n, alpha_seq, max_iter, lr, deadline):

    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for alpha in alpha_seq:
        # Adaptive learning rate
        cur_lr = lr
        for it in range(max_iter):
            if time.time() > deadline:
                break
            obj, grad = compute_grad(h, alpha)
            # Backtracking line search
            for _ in range(30):
                step = cur_lr
                h_new = h - step * grad
                h_new = project_sum(h_new, 0.5 * n)
                c_new = compute_c5(h_new)
                # Check sufficient decrease
                if c_new < obj - 1e-10 * step * np.dot(grad, h - h_new):
                    h = h_new
                    break
                cur_lr *= 0.5
            else:
                h = h_new  # Accept even if not sufficient decrease
            c_cur = compute_c5(h)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h.copy()
            # Reduce alpha within this phase if stuck
            if it % 1000 == 999:
                if c_cur > best_c * 1.001:
                    break
    
    return best_h, best_c

    n = h.size
