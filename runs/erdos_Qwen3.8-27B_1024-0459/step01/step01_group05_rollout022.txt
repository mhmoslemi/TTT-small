def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    m = np.max(x)

    sum_exp = np.sum(exp_vals)
    val = m + np.log(sum_exp) / beta
    grad = exp_vals / sum_exp  # softmax weights
    return val, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max_and_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -60.0, 60.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    val = m + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    return val, weights

def corr_smooth_max_and_grad(h, beta):
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    val, weights = smooth_max_and_weights(corr, beta)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-16:
            continue
        d = k - (n - 1)
        # Term 1: (1 - h[j+d]) for j where 0<=j<n and 0<=j+d<n

        je = min(n, n - d)

            grad[js:je] += w * (1.0 - h[js+d:je+d])

        js2 = max(0, d)
        je2 = min(n, n + d)
        if je2 > js2:
            grad[js2:je2] -= w * h[js2-d:je2-d]
    
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

_current_beta = 10.0
_current_target = 50.0
_current_n = 100

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max_and_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -60.0, 60.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    val = m + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    return val, weights

def corr_smooth_max_and_grad(h, beta):
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    val, weights = smooth_max_and_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if abs(w) < 1e-16:
            continue
        d = k - (n - 1)
        js = max(0, -d)
        je = min(n, n - d)
        if je > js:
            grad[js:je] += w * (1.0 - h[js + d:je + d])
        js2 = max(0, d)
        je2 = min(n, n + d)
        if je2 > js2:
            grad[js2:je2] -= w * h[js2 - d:je2 - d]
    return val, grad

def _obj(x):
    x_cl = np.clip(x, 0.0, 1.0)
    val, _ = corr_smooth_max_and_grad(x_cl, _current_beta)
    return val

def _obj_grad(x):
    x_cl = np.clip(x, 0.0, 1.0)
    _, g = corr_smooth_max_and_grad(x_cl, _current_beta)
    return g

def _con_fun(x):
    return np.sum(x) - _current_target

def _con_jac(x):
    return np.ones(_current_n)

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-15
            if np.any(mask):
                h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run_slsqp(h_start, n, target, beta, max_iter):
    global _current_beta, _current_target, _current_n
    _current_beta = beta
    _current_target = target
    _current_n = n
    h0 = project_sum(h_start, target)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        h_out = project_sum(res.x, target)
    except Exception:
        h_out = h0
    return h_out

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def make_binary_start(h, n, rng):
    target = n / 2.0
    if len(h) == n:
        h_bin = (h > 0.5).astype(float)
    else:
        h_interp = interpolate_h(h, len(h), n)
        h_bin = (h_interp > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = int(target)
    if n_ones > target_ones:
        excess = min(n_ones - target_ones, n_ones)
        ones_idx = np.where(h_bin == 1.0)[0]
        if len(ones_idx) > 0:
            to_remove = rng.choice(ones_idx, excess, replace=False)
            h_bin[to_remove] = 0.0
    elif n_ones < target_ones:
        deficit = min(target_ones - n_ones, n - n_ones)
        zeros_idx = np.where(h_bin == 0.0)[0]
        if len(zeros_idx) > 0:
            to_add = rng.choice(zeros_idx, deficit, replace=False)
            h_bin[to_add] = 1.0
    return h_bin

def thue_morse_pattern(n, invert=False):
    tm = np.array([(i & (i >> 1)) ^ (i >> 1) for i in range(n * 4)])[:n]
    # Actually generate properly
    tm_list = []
    x = 0
    for i in range(n):
        c = bin(i).count('1') % 2
        tm_list.append(c)
    tm = np.array(tm_list, dtype=float)
    target = n / 2.0
    n_ones = int(np.sum(tm))
    if n_ones < target:
        zeros = np.where(tm == 0)[0]
        need = int(target) - n_ones
        if len(zeros) >= need:
            idx = np.random.choice(zeros, need, replace=False)
            tm[idx] = 1.0
    elif n_ones > target:
        ones = np.where(tm == 1)[0]
        excess = n_ones - int(target)
        if len(ones) >= excess:
            idx = np.random.choice(ones, excess, replace=False)
            tm[idx] = 0.0
    return tm

def local_search_block(h, n, target, rng, n_trials):
    best = h.copy()
    best_val = compute_c5(h)
    for _ in range(n_trials):
        h_try = best.copy()
        block_size = rng.randint(2, min(8, n // 4))
        start = rng.randint(0, max(1, n - block_size))
        end = min(start + block_size, n)
        r = rng.random()
        if r < 0.4:
            h_try[start:end] = 1.0 - h_try[start:end]
        elif r < 0.7:
            h_try[start:end] = 1.0
        else:
            h_try[start:end] = 0.0
        h_try = project_sum(h_try, target)
        val = compute_c5(h_try)
        if val < best_val - 1e-10:
            best = h_try.copy()
            best_val = val
    return best, best_val
