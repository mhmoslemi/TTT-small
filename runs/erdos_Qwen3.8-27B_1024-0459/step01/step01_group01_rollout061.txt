def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, N):
    h = np.clip(h, 0.0, 1.0).copy()
    target = N / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h

def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_corr_entry(h, j, N):
    """Gradient of corr[j] w.r.t. h, where corr = correlate(h, 1-h, 'full')"""
    m = j - (N - 1)  # the lag
    grad = np.zeros(N)
    for p in range(N):
        # First term: (1-h[p-m]) if 0 <= p-m < N
        if 0 <= p - m and p - m < N:
            grad[p] += 1.0 - h[p - m]
        # Second term: -h[p+m] if 0 <= p+m < N
        if 0 <= p + m and p + m < N:
            grad[p] -= h[p + m]
    return grad

def subgradient_topk(h, N, k=5):
    """Subgradient of max_k corr[k] (times dx)"""
    dx = 2.0 / N
    corr = get_corr(h)
    top_idx = np.argsort(corr)[-k:]
    g = np.zeros(N)
    for j in top_idx:
        g += grad_corr_entry(h, j, N)
    g *= dx / k
    return g, corr

    m = np.max(corr)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)

def projected_step(h, g, lr, N):
    """Take a step and project to feasible set"""
    h_new = h - lr * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, N)
    return h_new

def subgradient_method(h, N, lr_init, n_iter, top_k=5, t0=None, deadline=None):
    """Run subgradient method with backtracking"""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    
    for it in range(n_iter):
        if t0 is not None and deadline is not None:
            if time.time() > deadline:
                break
        
        g, corr = subgradient_topk(h, N, top_k)
        h_new = projected_step(h, g, lr, N)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Slightly increase lr
            lr = min(lr * 1.01, lr_init * 2)
        else:
            # Reduce lr
            lr *= 0.9
            h = h_new  # still move, but with smaller step next time
    
    return best_h, best_c5

    return h

This is feasible! For each coordinate, the "kink" values of v[i] are where corr[k1](v[i]) = corr[k2](v[i]) for some pair k1, k2. Since corr[k] is linear in v[i] for each k, the kink points are at v[i] = (corr_k2_const - corr_k1_const) / (corr_k1_slope - corr_k2_slope).

But there are O(N²) pairs of kink points, which is too many. In practice, we only need the top few k (active constraints), so we only check O(k²) kink points plus the boundaries.

This is a very practical approach! Let me code it up:

```python
def coord_optimize_one(h, i, N, top_k=10):
    """Optimize h[i] given all other h[j] fixed, using sum constraint."""
    # With sum constraint: sum(h) = N/2
    # If we change h[i] by delta, we need to distribute -delta among other coordinates.
    # Simplest: distribute uniformly. h[j] -= delta/(N-1) for j≠i.
    # But this might violate 0≤h[j]≤1.
    # Alternative: just optimize h[i] in [0,1] and then fix_sum.
    # But that's what we're trying to avoid.
    
    # Actually, let's just do: optimize h[i] in [0,1] (ignoring sum), then fix_sum.
    # This is approximate but should work in practice.
    
    # For a given h[i] = t, compute the top_k correlations and their values.
    # corr[k] = a_k * t + b_k (linear in t)
    # Find t that minimizes max_k (a_k * t + b_k) * dx
    
    # First, identify which k's are active (top_k)
    # Compute a_k = d(corr[k])/d(h[i]) and b_k = corr[k] - a_k * h[i]
    # Then minimize max_k (a_k * t + b_k) * dx over t in [0,1]
    
    current_h = h[i]
    corr = get_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    
    # For each active k, compute slope and intercept w.r.t. h[i]
    # corr[k] = sum_j h[j]*(1-h[j+m_k])
    # d/dh[i]: (1-h[i+m_k])*I(0≤i+m_k<N) - h[i-m_k]*I(0≤i-m_k<N)
    slopes = np.zeros(top_k)
    for ii, k_idx in enumerate(top_idx):
        m = k_idx - (N-1)
        s = 0.0
        if 0 <= i - m and i - m < N:
            s += 1.0 - h[i - m]
        if 0 <= i + m and i + m < N:
            s -= h[i + m]
        slopes[ii] = s
    
    intercepts = corr[top_idx] - slopes * current_h
    
    # Minimize max_k (slopes[k]*t + intercepts[k]) * dx over t in [0,1]
    # = dx * min_t max_k (slopes[k]*t + intercepts[k])
    # The optimal t is where the two largest lines cross (or at a boundary)
    
    # Find the minimum of f(t) = max_k (slopes[k]*t + intercepts[k])
    # Evaluate at boundaries and kink points
    candidates = [0.0, 1.0]
    for a in range(top_k):
        for b in range(a+1, top_k):
            if abs(slopes[a] - slopes[b]) > 1e-12:
                t_cross = (intercepts[b] - intercepts[a]) / (slopes[a] - slopes[b])
                if 0 <= t_cross <= 1:
                    candidates.append(t_cross)
    
    best_t = current_h
    best_val = np.inf
    for t in candidates:
        vals = slopes * t + intercepts
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_t = t
    
    return best_t
```

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, N):
    h = np.clip(h, 0.0, 1.0).copy()
    target = N / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def coord_optimize_one(h, i, N, top_k=12):
    corr = get_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    current_t = h[i]
    slopes = np.zeros(top_k)
    for ii, k_idx in enumerate(top_idx):
        m = k_idx - (N - 1)
        s = 1.0
        if 0 <= i + m and i + m < N:
            s -= h[i + m]
        if 0 <= i - m and i - m < N:
            s -= h[i - m]
        slopes[ii] = s
    intercepts = corr[top_idx] - slopes * current_t
    candidates = [0.0, 1.0, current_t]
    for a in range(top_k):
        for b in range(a + 1, top_k):
            ds = slopes[a] - slopes[b]
            if abs(ds) > 1e-14:
                tc = (intercepts[b] - intercepts[a]) / ds
                if -0.01 <= tc <= 1.01:
                    candidates.append(np.clip(tc, 0.0, 1.0))
    best_t = current_t
    best_val = np.inf
    for t in candidates:
        t = np.clip(t, 0.0, 1.0)
        vals = slopes * t + intercepts
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_t = t
    return best_t


def coordinate_descent(h, N, n_sweeps, top_k=12, order='sequential', rng=None, t0=None, deadline=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for sweep in range(n_sweeps):
        if t0 is not None and deadline is not None and time.time() > deadline:
            break
        if order == 'random' and rng is not None:
            idx_order = rng.permutation(N)
        else:
            idx_order = range(N)
        for i in idx_order:
            h[i] = coord_optimize_one(h, i, N, top_k)
        h = fix_sum(h, N)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5


def smooth_max_obj(x, N, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)


def eq_con(x, N):
    return [x.sum() - N / 2.0]


def eq_jac(x, N):
    return np.ones((1, N))


def slsqp_polish(h, N, tau, maxiter, t0, deadline):
    if time.time() > deadline:
        return h, compute_c5(h)
    dx = 2.0 / N
    bounds = [(0.0, 1.0)] * N
    obj_arg = (N, dx, tau)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, N, dx, tau),
            x0=h.copy(), method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': lambda x: eq_con(x, N), 'jac': lambda x: eq_jac(x, N)}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        h_new = fix_sum(np.clip(res.x, 0, 1), N)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        if c5_new < c5_old:
            return h_new, c5_new
        return h, c5_old
    except Exception:
        return h, compute_c5(h)


def subgradient_step(h, N, lr, top_k=5):
    dx = 2.0 / N
    corr = get_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    g = np.zeros(N)
    for k_idx in top_idx:
        m = k_idx - (N - 1)
        for p in range(N):
            if 0 <= p - m and p - m < N:
                g[p] += 1.0 - h[p - m]
            if 0 <= p + m and p + m < N:
                g[p] -= h[p + m]
    g *= dx / top_k
    h_new = h - lr * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, N)
    return h_new, g

    # Phase 2: Subgradient at parent resolution for global improvement

        if c5_new < best_c5:

            best_h = h_new
            h_sub = h_new
            lr = min(lr * 1.02, 200.0)
