def local_search(h, n, n_trials=3000, step_size=0.03, seed=42):
    """Random local search with feasibility repair."""
    rng = np.random.RandomState(seed)
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = step_size
    for t in range(n_trials):
        h_try = best_h.copy()
        n_mod = rng.randint(1, 5)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        # Repair sum constraint
        diff = n/2.0 - np.sum(h_try)
        mask = (h_try > 0.01) & (h_try < 0.99)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        # Final repair if needed
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
        # Annealing: slowly reduce step size
        step *= 0.9995
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def binary_projection(h, n):
    """Push h toward binary, then repair sum constraint."""
    h_bin = (h > 0.5).astype(float)
    # h_bin has some number of 1s, need exactly n/2
    n_ones = int(np.sum(h_bin))
    if n_ones == n/2:
        return h_bin
    elif n_ones > n/2:
        # Need to turn off some 1s: choose the ones with lowest h values
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for idx in order[:n_ones - n//2]:
            h_bin[ones_idx[idx]] = 0
    else:
        # Need to turn on some 0s: choose the zeros with highest h values
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for idx in order[:n//2 - n_ones]:
            h_bin[zeros_idx[idx]] = 1
    return h_bin
