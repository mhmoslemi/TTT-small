def project_feasible(h, n):

    if time.time() < deadline - 15:
        # Try a few more SLSQP rounds with different betas
        for beta_val in [40.0, 120.0, 200.0]:
            if time.time() > deadline - 5:
                break
            pf_h, pf_c5 = slsqp_polish(best_h, best_n, beta_val, 350)
            if pf_c5 < best_c5:
                best_c5 = pf_c5
                best_h = pf_h.copy()
                print("  Final polish beta={}: C5={:.6f}".format(int(beta_val), best_c5))

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Verify with evaluator if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
