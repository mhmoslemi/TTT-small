                h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

    m = np.max(x)
    sh = np.clip(beta * (x - m), -100.0, 100.0)
    return m + np.log(np.sum(np.exp(sh))) / beta

def soft_max_weights(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -100.0, 100.0)
    e = np.exp(sh)
    return e / np.sum(e)

def obj_fn(h, beta):
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    return soft_max_val(corr, beta)

def obj_grad(h, beta):
    n = len(h)
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    w = soft_max_weights(corr, beta)
    
    grad = np.zeros(n)
    grad += w[n-1] * (1.0 - 2.0 * hc)
    
    for s in range(1, n):
        ws = w[n-1+s]
        if ws > 1e-14:
            grad[s:] += ws * (1.0 - hc[:n-s])
            grad[:n-s] -= ws * hc[s:]
    
    for m in range(1, n):
        wm = w[n-1-m]
        if wm > 1e-14:
            grad[:n-m] += wm * (1.0 - hc[m:])
            grad[m:] -= wm * hc[:n-m]
    
    return grad

def run_slsqp(h0, n, target, beta, max_iter=500, ftol=1e-12):
    def f(h):
        return obj_fn(h, beta)
    
    def g(h):
        return obj_grad(h, beta)
    
    def con_f(h):
        return np.sum(h) - target
    
    def con_g(h):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': con_f, 'jac': con_g}]
    bounds = [(0.0, 1.0)] * n
    
    h0 = project_sum(h0, target)
    
    try:
        res = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': ftol})
        if res.success or res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_sum(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
        else:
            free = np.where(h > 1e-12)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h


def soft_max_val(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -100.0, 100.0)
    return m + np.log(np.sum(np.exp(sh))) / beta


def soft_max_weights(x, beta):
    m = np.max(x)
    sh = np.clip(beta * (x - m), -100.0, 100.0)
    e = np.exp(sh)
    s = np.sum(e)
    if s < 1e-300:
        return np.full_like(x, 1.0 / len(x))
    return e / s


def obj_fn(h, beta):
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    return soft_max_val(corr, beta)


def obj_grad(h, beta):
    n = len(h)
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    w = soft_max_weights(corr, beta)

    grad = np.zeros(n)
    # s = 0
    grad += w[n - 1] * (1.0 - 2.0 * hc)

    # Positive shifts
    for s in range(1, n):
        ws = w[n - 1 + s]
        if ws > 1e-14:
            grad[s:] += ws * (1.0 - hc[:n - s])
            grad[:n - s] -= ws * hc[s:]

    # Negative shifts
    for m in range(1, n):
        wm = w[n - 1 - m]
        if wm > 1e-14:
            grad[:n - m] += wm * (1.0 - hc[m:])
            grad[m:] -= wm * hc[:n - m]

    return grad


def run_slsqp(h0, n, target, beta, max_iter=400, ftol=1e-12):
    def f(h):
        return obj_fn(h, beta)

    def g(h):
        return obj_grad(h, beta)

    def con_f(h):
        return np.sum(h) - target

    def con_g(h):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_f, 'jac': con_g}]
    bounds = [(0.0, 1.0)] * n
    h0 = project_sum(h0, target)

    try:
        res = minimize(f, h0, jac=g, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol, 'eps': 1e-7})
        if res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0


def projected_grad_step(h, grad, step_size, target):
    n = len(h)
    y = h - step_size * grad
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        if np.sum(x) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    return project_sum(x, target)


def run_projected_gradient(h0, n, target, beta, max_iter=2000):
    h = h0.copy()
    step = 0.01
    best_val = obj_fn(h, beta)
    for it in range(max_iter):
        g = obj_grad(h, beta)
        cnorm = np.linalg.norm(g)
        if cnorm < 1e-12:
            break
        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(30):
            h_new = projected_grad_step(h, g, alpha, target)
            f_new = obj_fn(h_new, beta)
            if f_new < best_val - 1e-15:
                h = h_new
                best_val = f_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return h


def make_periodic_init(n, period, seed=0):
    rng = np.random.RandomState(seed)
    p = period
    base = rng.rand(p) * 0.4 + 0.3
    full = np.tile(base, n // p + 1)[:n]
    return project_sum(full, n / 2.0)
