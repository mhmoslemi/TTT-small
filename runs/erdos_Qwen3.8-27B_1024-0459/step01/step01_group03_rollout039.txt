def subgradient_step(h, top_k=5, lr=0.1):
    """Take a projected subgradient step to reduce max correlation."""
    n = h.size
    target = 0.5 * n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top k correlation indices
    top_idx = np.argsort(corr)[-top_k:]
    # Compute subgradient (average gradient of top-k correlations)
    g = np.zeros(n)
    center = n - 1
    for t in top_idx:
        k = int(t) - center
        if k == 0:
            g += 1.0 - 2.0 * h
        else:
            lag = abs(k)
            # For shift k: c_k = sum_i h[i] * (1-h[i+k]) (approximately)
            # dc_k/dh[i] = (1-h[i+k]) if i in valid range for first factor
            #             - h[i-k] if i in valid range for second factor
            # This depends on the exact convention, so let me compute it properly
            # corr[t] = sum_i h[i] * (1-h)[i + t - (n-1)]
            # Let j = i + t - (n-1), so i = j - t + (n-1)
            # corr[t] = sum_j h[j - t + n - 1] * (1-h[j]) (for valid indices)
            
            # Actually, let me just use the formula:
            # np.correlate(h, 1-h, 'full')[t] = sum_i h[i] * (1-h)[i + t - (n-1)]
            # (where we sum over i such that both indices are valid)
            
            # For the gradient with respect to h[m]:
            # h[m] appears as:
            # - In h[i] when i = m: contributes (1-h[m + t - n + 1]) = (1-h[m+t-center])
            # - In (1-h)[j] when j + t - center = m, i.e., j = m - t + center: contributes -h[m - t + center]
            
            # So dc/dh[m] = (1-h[m+t-center]) * [0<=m<n and 0<=m+t-center<n] 
            #               - h[m-t+center] * [0<=m<n and 0<=m-t+center<n]
            
            # Let a = t - center = k (the shift)
            # dc/dh[m] = (1-h[m+k]) * [0<=m<n, 0<=m+k<n] - h[m-k] * [0<=m<n, 0<=m-k<n]
            
            if k > 0:
                # First term: m from 0 to n-1-k
                g[0:n-k] += (1.0 - h[k:])
                # Second term: m from k to n-1
                g[k:n] -= h[0:n-k]
            else:  # k < 0, lag = -k
                # First term: m from -k to n-1, i.e., m from lag to n-1
                g[lag:n] += (1.0 - h[0:n-lag])
                # Second term: m from 0 to n-1-lag
                g[0:n-lag] -= h[lag:n]
    
    g /= float(len(top_idx))
    
    # Projected gradient step
    h_new = h - lr * g
    h_new = project_density(h_new, target)
    return h_new

    # Phase 3: Try structured initializations at a few resolutions
    if deadline - time.perf_counter() > 30.0:
        for mode in [0, 1, 5, 2]:
            if deadline - time.perf_counter() < 15.0:
                break
            for n_try in [best_n, min(best_n * 2, 500)]:
                if n_try % 2 != 0:
                    n_try += 1
                if deadline - time.perf_counter() < 10.0:
                    break
                h_struct = make_structured(n_try, rng, mode)
                c_struct = evaluate_erdos(h_struct)
                
                if c_struct < best_c + 0.05:
                    sub_dl = time.perf_counter() + min(30.0, deadline - time.perf_counter() - 10.0)
                    h_pol, c_pol = polish_top_k(h_struct, sub_dl, top_k_list=(5, 8, 12), max_iter=200)
                    if c_pol < best_c:
                        best_h = h_pol.copy()
                        best_c = c_pol
                        best_n = int(n_try)
                    
                    if n_try % 2 == 0 and sub_dl - time.perf_counter() > 5.0:
                        h_bl, c_bl = binary_local_search(h_pol, sub_dl, max_iter=5000, rng=rng)
                        if c_bl < best_c:
                            best_h = h_bl.copy()
                            best_c = c_bl
                            best_n = int(n_try)
    
    # Phase 4: Final intensive polish on the best solution
    if best_h is not None and deadline - time.perf_counter() > 15.0:
        n_best = int(best_h.size)
        
        # Polish with many top_k values
        h_fp, c_fp = polish_top_k(best_h, deadline, top_k_list=(2, 3, 4, 5, 7, 10, 15, 20), max_iter=200)
        if c_fp < best_c:
            best_h = h_fp.copy()
            best_c = c_fp
        
        # Binary polish if even
        if n_best % 2 == 0 and deadline - time.perf_counter() > 10.0:
            h_bp, c_bp = binary_local_search(best_h, deadline, max_iter=10000, rng=rng)
            if c_bp < best_c:
                best_h = h_bp.copy()
                best_c = c_bp
    
    # Final verification
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
