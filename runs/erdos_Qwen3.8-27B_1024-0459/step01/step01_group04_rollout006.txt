def score_max(h, n=None):
    h = np.asarray(h, dtype=float)
    n = int(h.size) if n is None else int(n)
    if n == 0:
        return np.zeros(0), 1e6
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    if n == 1:
        m = 0.0
        return np.array([m]), m
    corr = np.correlate(h, h, mode="full")
    ac = corr[n-1:n-1+n]
    prefix = np.empty(n+1, dtype=float)
    prefix[0]=0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2*n-1, dtype=float)
    scores[0] = total - ac[0]
    for d in range(1,n):
        scores[2*d-1] = prefix[n-d] - ac[d]
        scores[2*d] = total - prefix[d] - ac[d]
    m = float(np.max(scores))
    if not math.isfinite(m):
       scores.fill(1e6); m=1e6
    return scores,m

def score_max(h, n=None):
    h = np.asarray(h, dtype=float)
    n = int(h.size) if n is None else int(n)
    if n <= 0:
        return np.zeros(0, dtype=float), 1e6
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    if n == 1:
        return np.array([0.0], dtype=float), 0.0

    corr = np.correlate(h, h, mode="full")
    ac = corr[n - 1:n - 1 + n]
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]

    scores = np.empty(2 * n - 1, dtype=float)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2 * d - 1] = prefix[n - d] - ac[d]
        scores[2 * d] = total - prefix[d] - ac[d]

    m = float(np.max(scores))
    if not math.isfinite(m):
        scores.fill(1e6)
        m = 1e6
    return scores, m


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n <= 0 or not np.all(np.isfinite(h)):
        return 1e6
    if float(np.min(h)) < -1e-8 or float(np.max(h)) > 1.0 + 1e-8:
        h = as_feasible(h, n)
    _, m = score_max(h, n)
    c = float(m) * (2.0 / float(n))
    if not math.isfinite(c):
        return 1e6
    return c


def _adjust_residual(h, target):
    h = np.asarray(h, dtype=float).copy()
    n = int(h.size)
    if n <= 0:
        return h
    diff = target - float(h.sum())
    guard = 0
    while abs(diff) > 1e-12 and guard <= n + 8:
        guard += 1
        if diff > 0.0:
            mask = h < 1.0
            if not np.any(mask):
                break
            cnt = int(np.count_nonzero(mask))
            per = diff / float(cnt)
            inc = min(per, float(1.0 - h[mask].min()))
            if inc <= 1e-15:
                break
            h[mask] += inc
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            cnt = int(np.count_nonzero(mask))
            per = -diff / float(cnt)
            dec = min(per, float(h[mask].min()))
            if dec <= 1e-15:
                break
            h[mask] -= dec
        diff = target - float(h.sum())

    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

def soft_value(scores, m, alpha):
    if scores.size == 0:
        return m
    if alpha <= 1e-12:
        return m
    w = np.exp((scores - m) / alpha)
    a = float(np.sum(w))
    if math.isfinite(a) and a > 0.0:
        return m + alpha * math.log(a)
    return m


def soft_weights(scores, m, alpha):
    n = int(scores.size)
    w = np.zeros(n, dtype=float)
    if n == 0:
        return w
    if alpha > 1e-12:
        e = np.exp((scores - m) / alpha)
        a = float(np.sum(e))
        if math.isfinite(a) and a > 1e-300:
            w = e / a
            return w
    tol = max(1e-8, 1e-6 * max(1.0, abs(float(m))))
    active = np.flatnonzero(scores >= m - tol)
    if active.size == 0:
        active = np.asarray([int(np.argmax(scores))])
    w[active] = 1.0 / float(active.size)
    return w


def add_lag_grad(grad, idx, coef, h, n):
    if coef == 0.0:
        return
    idx = int(idx)
    n = int(n)
    if idx < 0 or idx >= 2 * n - 1:
        return
    if idx == 0:
        grad += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
    else:
        d = idx // 2
    m = n - d
    if m <= 0:
        return
    if idx % 2 == 1:
        grad[:m] += coef * (1.0 - h[d:])
        grad[d:] += coef * (-h[:m])
    else:
        grad[:m] += coef * (-h[d:])
        grad[d:] += coef * (1.0 - h[:m])


def regular_fg(x, alpha, pen):
    x = np.asarray(x, dtype=float)
    n = int(x.size)
    if n <= 0:
        return 0.0, np.zeros(0, dtype=float)
    h = np.clip(x, 0.0, 1.0)
    scores, m = score_max(h, n)
    val = soft_value(scores, m, alpha)
    f = val * (2.0 / float(n))

    weights = soft_weights(scores, m, alpha)
    g = np.zeros(n, dtype=float)
    for i in range(scores.size):
        wi = float(weights[i])
        if wi > 1e-12:
            add_lag_grad(g, int(i), wi, h, n)
    g *= 2.0 / float(n)

    if pen > 0.0:
        e = (float(np.sum(h)) - 0.5 * float(n)) / float(n)
        f += float(pen) * e * e
        g += (2.0 * float(pen) * e) / float(n)

    if not np.all(np.isfinite(g)):
        g.fill(0.0)
    if not math.isfinite(f):
        f = 1e6
    return float(f), g

        best_c = compute_c5(best_h)

            state.update(best_h, n, best_c)

        base_alpha = max(0.35, min(1.6, 0.008 * float(n)))
        base_iter = int(max(15, min(130, 6000.0 / float(n))))
        specs = [
            (1.0, 1.0),
            (0.5, 0.9),
            (0.18, 0.8),
            (0.06, 0.7),
            (0.00, 0.55),
        ]
        bounds = [(0.0, 1.0)] * n
        constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]

        for af, ifac in specs:
            if time.time() >= phase - 8.0:
                break
            alpha = base_alpha * af
            iters = int(base_iter * ifac) + 5
            res = None
            try:
                res = minimize(
                    slsqp_obj,
                    h,
                    args=(alpha,),
                    method="SLSQP",
                    jac=slsqp_grad,
                    bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": iters, "ftol": 1e-9, "disp": 0},
                )
            except Exception:
                res = None

            if res is not None and getattr(res, "x", None) is not None:
                h_try = as_feasible(res.x, n)
                c_try = compute_c5(h_try)
                h = h_try
                if math.isfinite(c_try) and c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
                    if state is not None:
                        state.update(best_h, n, best_c)

        if state is not None:
            state.update(best_h, n, best_c)
        return best_h, float(best_c)
    finally:
        _RUN_DEADLINE = old_dl


def deadline_callback(ck):
    return _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE


def upsample_linear(parent, n):
    parent = np.asarray(parent, dtype=float)
    n0 = int(parent.size)
    n = int(n)
    if n0 <= 0:
        return np.full(n, 0.5, dtype=float)
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    return np.interp(x, x0, parent).astype(float)


def downsample_large(h, m):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    m = int(m)
    if n <= 0 or m < 2:
        return None
    if n % m == 0:
        r = n // m
        out = np.mean(h.reshape(m, r), axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / float(n))
        x = (np.arange(m) + 0.5) * (2.0 / float(m))
        out = np.interp(x, x0, h)
    return as_feasible(out, m)


def smooth1(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return h.copy()
    if n == 1:
        return h.copy()
    if n == 2:
        return np.array([0.75 * h[0] + 0.25 * h[1],
                         0.25 * h[0] + 0.75 * h[1]], dtype=float)
    out = h.copy()
    out[0] = 0.75 * h[0] + 0.25 * h[1]
    out[-1] = 0.25 * h[-2] + 0.75 * h[-1]
    out[1:-1] = 0.25 * h[:-2] + 0.5 * h[1:-1] + 0.25 * h[2:]
    return np.clip(out, 0.0, 1.0)


def make_binary_random(n, rng):
    n = int(n)
    target = n // 2
    h = np.zeros(n, dtype=np.int16)
    if target > 0:
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1
    return h


def make_binary_from_float(vals, n):
    n = int(n)
    target = n // 2
    h = np.zeros(n, dtype=np.int16)
    vals = np.asarray(vals, dtype=float)
    if vals.size != n:
        if vals.size > n:
            vals = vals[:n]
        else:
            tmp = np.zeros(n, dtype=float)
            tmp[:vals.size] = vals
            vals = tmp
    vals = np.nan_to_num(vals, nan=0.5, posinf=1.0, neginf=0.0)
    order = np.argsort(vals)
    if target > 0:
        h[order[target:]] = 1
    return h


def binary_scores(h, n):
    s, _ = score_max(np.asarray(h, dtype=float), n)
    return np.rint(s).astype(float)


def binary_flip_update(scores, h, idx, new_val, n):
    i = int(idx)
    new_val = int(new_val)
    for d in range(1, n):
        pidx = 2 * d - 1
        midx = 2 * d
        if i - d >= 0:
            hj = int(h[i - d])
            if new_val == 1:
                scores[pidx] -= hj
                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += hj
                scores[midx] -= 1.0 - hj
        if i + d < n:
            hj = int(h[i + d])
            if new_val == 1:
                scores[pidx] += 1.0 - hj
                scores[midx] -= hj
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += hj
    h[i] = new_val


def binary_swap_apply(scores, h, i, j, n):
    i = int(i)
    j = int(j)
    if i == j or i < 0 or j < 0 or i >= n or j >= n:
        return False
    hi = int(h[i])
    hj = int(h[j])
    if hi == hj:
        return False
    if hi == 0 and hj == 1:
        binary_flip_update(scores, h, i, 1, n)
        binary_flip_update(scores, h, j, 0, n)
    else:
        binary_flip_update(scores, h, i, 0, n)
        binary_flip_update(scores, h, j, 1, n)
    return True


def binary_propose(h, scores, cur, rng, n):
    if rng.random() < 0.65:
        active = np.flatnonzero(scores >= (cur - 1e-9))
        if active.size == 0:
            active = np.asarray([int(np.argmax(scores))])
        idx = int(active[int(rng.integers(active.size))])
        if idx != 0:
            if idx % 2 == 1:
                d = (idx + 1) // 2
                L = n - d
                if L > 0:
                    mask = (h[:L] == 1) & (h[d:] == 0)
                    cand = np.flatnonzero(mask)
                    if cand.size > 0:
                        pos = int(cand[int(rng.integers(cand.size))])
                        return pos, pos + d
            else:
                d = idx // 2
                L = n - d
                if L > 0:
                    mask = (h[:L] == 0) & (h[d:] == 1)
                    cand = np.flatnonzero(mask)
                    if cand.size > 0:
                        pos = int(cand[int(rng.integers(cand.size))])
                        return pos, pos + d

    zeros = np.flatnonzero(h == 0)
    ones = np.flatnonzero(h == 1)
    if zeros.size == 0 or ones.size == 0:
        return -1, -1
    i = int(zeros[int(rng.integers(zeros.size))])
    j = int(ones[int(rng.integers(ones.size))])
    return i, j


def binary_local_once(start_h, n, deadline, max_ops, stall_limit, t0, tmin, rng):
    n = int(n)
    if n % 2 != 0 or n <= 0:
        return None, 1e18
    if start_h is not None:
        h = make_binary_from_float(start_h, n)
    else:
        h = make_binary_random(n, rng)
    scores = binary_scores(h, n)
    cur = float(np.max(scores))
    best = cur
    best_h = h.copy()

    ops = 0
    stall = 0
    while ops < max_ops and time.time() < deadline:
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        ops += 1
        i, j = binary_propose(h, scores, cur, rng, n)
        if i < 0:
            stall += 50
            continue
        if not binary_swap_apply(scores, h, i, j, n):
            continue
        new_cur = float(np.max(scores))
        if new_cur < cur:
            cur = new_cur
            stall = 0
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            delta = new_cur - cur
            temp = max(tmin, t0 * (tmin / t0) ** (float(ops) / float(max(1, max_ops))))
            if delta <= 1e-12:
                accept = (rng.random() < 0.18)
            else:
                accept = (rng.random() < math.exp(-float(delta) / max(1e-12, temp)))
            if accept:
                cur = new_cur
                stall += 1
                if cur < best:
                    best = cur
                    best_h = h.copy()
            else:
                binary_swap_apply(scores, h, i, j, n)
                stall += 1
        if stall > stall_limit:
            break

    return best_h.astype(float), float(best) * (2.0 / float(n))


def binary_hill_search(start_h, n, state, deadline, rng, time_limit):
    n = int(n)
    if n % 2 != 0 or time_limit <= 0.0:
        return None, 1e18
    start = time.time()
    stage = min(float(deadline), start + float(time_limit))
    best_h = None
    best_c = 1e18
    max_restarts = 3 if n <= 110 else 2
    rate = 900000.0 / max(50.0, float(n))

    for r in range(max_restarts):
        if time.time() >= stage - 1.0:
            break
        rt = (stage - time.time()) / float(max_restarts - r)
        if rt < 1.0:
            rt = 1.0
        sh = start_h if r == 0 else None
        max_ops = int(rt * rate)
        max_ops = max(1000, min(350000, max_ops))
        stall_limit = max(1000, int(max_ops // 22))
        try:
            h, c = binary_local_once(sh, n, stage, max_ops, stall_limit, 3.0, 0.3, rng)
        except Exception:
            h, c = None, 1e18
        if h is not None and math.isfinite(c) and c < best_c:
            best_c = c
            best_h = h
            if state is not None:
                state.update(h, n, best_c)

    return best_h, float(best_c)


def swap_refine(h0, n, state, deadline, time_limit, rng):
    n = int(n)
    if time_limit <= 0.0 or n <= 1:
        return None, 1e18
    phase = min(float(deadline), time.time() + float(time_limit))
    h = as_feasible(h0, n)
    best_c = compute_c5(h)
    best_h = h.copy()
    if state is not None:
        state.update(best_h, n, best_c)

    stall = 0
    ratios = (0.25, 0.5, 0.75, 1.0)
    while time.time() < phase and stall < 900:
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        scores, m = score_max(h, n)
        active = np.flatnonzero(scores >= m - 1e-9)
        if active.size == 0:
            break
        idx = int(active[int(rng.integers(active.size))])
        improved = False

        if idx == 0:
            high = np.flatnonzero(h > 0.5 + 1e-6)
            low = np.flatnonzero(h < 0.5 - 1e-6)
            if high.size > 0 and low.size > 0:
                hi = int(high[int(rng.integers(high.size))])
                lo = int(low[int(rng.integers(low.size))])
                for rr in ratios:
                    amt = float(rr) * min(h[hi] - 0.5, 0.5 - h[lo])
                    if amt <= 1e-12:
                        continue
                    cand = h.copy()
                    cand[hi] += amt
                    cand[lo] -= amt
                    cand = np.clip(cand, 0.0, 1.0)
                    c2 = compute_c5(as_feasible(cand, n))
                    if math.isfinite(c2) and c2 < best_c - 1e-12:
                        improved = True
                        best_c = c2
                        best_h = cand.copy()
                        break
        else:
            if idx % 2 == 1:
                d = (idx + 1) // 2
                L = n - d
                mask = (h[:L] - h[d:d + L]) > 1e-7
                if mask.size > 0:
                    pos = int(mask[int(rng.integers(mask.size))])
                    a = float(h[pos])
                    b = float(h[pos + d])
                    for rr in ratios:
                        amt = float(rr) * min(a, 1.0 - b)
                        if amt <= 1e-12:
                            continue
                        cand = h.copy()
                        cand[pos] -= amt
                        cand[pos + d] += amt
                        c2 = compute_c5(cand)
                        if math.isfinite(c2) and c2 < best_c - 1e-12:
                            improved = True
                            best_c = c2
                            best_h = cand.copy()
                            break
            else:
                d = idx // 2
                L = n - d
                mask = (h[d:d + L] - h[:L]) > 1e-7
                if mask.size > 0:
                    pos = int(mask[int(rng.integers(mask.size))])
                    a = float(h[pos])
                    b = float(h[pos + d])
                    for rr in ratios:
                        amt = float(rr) * min(b, 1.0 - a)
                        if amt <= 1e-12:
                            continue
                        cand = h.copy()
                        cand[pos + d] -= amt
                        cand[pos] += amt
                        c2 = compute_c5(cand)
                        if math.isfinite(c2) and c2 < best_c - 1e-12:
                            improved = True
                            best_c = c2
                            best_h = cand.copy()
                            break

        if improved:
            h = as_feasible(best_h, n)
            c_now = compute_c5(h)
            if state is not None:
                state.update(h, n, c_now)
            stall = 0
        else:
            stall += 1
            if stall % 80 == 0:
                hi = int(np.argmax(h))
                lo = int(np.argmin(h))
                amt = 0.02 * min(float(h[hi]), 1.0 - float(h[lo]))
                if amt > 1e-9:
                    cand = h.copy()
                    cand[hi] -= amt
                    cand[lo] += amt
                    cand = as_feasible(cand, n)
                    c2 = compute_c5(cand)
                    if math.isfinite(c2) and c2 < best_c:
                        best_c = c2
                        best_h = cand.copy()
                        h = cand
                        stall = 0
                        if state is not None:
                            state.update(h, n, best_c)

    return best_h, float(best_c)


def synthetic_regular(n, rng):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    best_h = as_feasible(np.full(n, 0.5, dtype=float), n)
    best_c = compute_c5(best_h)

    for _ in range(12):
        a1 = rng.uniform(-0.45, 0.45)
        a2 = rng.uniform(-0.45, 0.45)
        a3 = rng.uniform(-0.35, 0.35)
        a4 = rng.uniform(-0.25, 0.25)
        h = (0.5 + a1 * np.cos(np.pi * x) + a2 * np.sin(np.pi * x)
             + a3 * np.cos(2.0 * np.pi * x) + a4 * np.sin(4.0 * np.pi * x))
        h = as_feasible(np.clip(h, 0.0, 1.0), n)
        c = compute_c5(h)
        if math.isfinite(c) and c < best_c:
            best_c = c
            best_h = h.copy()

    for _ in range(4):
        hb = make_binary_random(n, rng).astype(float)
        hs = as_feasible(smooth1(hb), n)
        c = compute_c5(hs)
        if math.isfinite(c) and c < best_c:
            best_c = c
            best_h = hs.copy()

    return best_h


def synthetic_anti(n, rng):
    n = int(n)
    if n % 2 != 0 or n <= 2:
        return synthetic_regular(n, rng)
    p = n // 2
    y = (np.arange(p) + 0.5) / float(p)
    best_h = as_feasible(make_anti_h(np.full(p, 0.5, dtype=float), n), n)
    best_c = compute_c5(best_h)

    for _ in range(14):
        a1 = rng.uniform(-0.45, 0.45)
        a2 = rng.uniform(-0.45, 0.45)
        a3 = rng.uniform(-0.30, 0.30)
        a4 = rng.uniform(-0.35, 0.35)
        f = (0.5 + a1 * np.sin(2.0 * np.pi * y) + a2 * np.cos(2.0 * np.pi * y)
             + a3 * np.sin(4.0 * np.pi * y) + a4 * (y - 0.5))
        f = np.clip(f, 0.0, 1.0)
        h = as_feasible(make_anti_h(f, n), n)
        c = compute_c5(h)
        if math.isfinite(c) and c < best_c:
            best_c = c
            best_h = h.copy()

    for _ in range(4):
        cnt = int(p * (0.35 + 0.30 * rng.random()))
        cnt = max(0, min(p, cnt))
        f = np.zeros(p, dtype=float)
        if cnt > 0:
            f[rng.choice(p, size=cnt, replace=False)] = 1.0
        h = as_feasible(make_anti_h(f, n), n)
        c = compute_c5(h)
        if math.isfinite(c) and c < best_c:
            best_c = c
            best_h = h.copy()

    return best_h


class BestState:
    def __init__(self, h, c, n):
        self.h = np.asarray(h, dtype=float).copy()
        self.c = float(c)
        self.n = int(n)

    def update(self, h, n, c=None):
        if h is None:
            return
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size == 0:
            return
        if n is None:
            n = h.size
        n = int(n)
        if h.size != n:
            if h.size > n:
                h = h[:n]
            else:
                return
        if not np.all(np.isfinite(h)):
            return
        if float(h.min()) < -1e-9 or float(h.max()) > 1.0 + 1e-9:
            h = as_feasible(h, n)
        else:
            h = np.clip(h, 0.0, 1.0)
            if abs(float(h.sum()) - 0.5 * float(n)) > 1e-7:
                h = as_feasible(h, n)
        if c is None or not math.isfinite(c):
            c = compute_c5(h)
        c = float(c)
        if math.isfinite(c) and c < self.c - 1e-14:
            self.h = h.copy()
            self.c = c
            self.n = n


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None


def record_best(history, n, h, c):
    if h is None or not math.isfinite(c):
        return
    n = int(n)
    if n <= 0:
        return
    try:
        h2 = as_feasible(np.asarray(h, dtype=float), n)
        c2 = compute_c5(h2)
    except Exception:
        return
    if not math.isfinite(c2):
        return
    old = history.get(n)
    if old is None or c2 < old[1] - 1e-13:
        history[n] = (h2.copy(), float(c2))


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE

    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 1300.0
    reserve = 25.0 if budget_s > 120.0 else max(2.0, 0.1 * budget_s)
    deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = deadline

    rng = np.random.default_rng(seed)

    parent = get_parent()
    if parent is not None and parent.size >= 4:
        n0 = int(parent.size)
        h0 = parent.copy()
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = synthetic_regular(n0, rng)
        c0 = compute_c5(h0)

    state = BestState(h0, c0, n0)
    if not is_valid_candidate(state.h):
        hf = as_feasible(state.h, state.n)
        state = BestState(hf, compute_c5(hf), state.n)
        h0 = hf.copy()
        c0 = state.c

    n0e = n0 if n0 % 2 == 0 else n0 + 1
    if n0e != state.n:
        he = upsample_linear(state.h, n0e)
        ce = compute_c5(he)
        state.update(he, n0e, ce)

    history = {}
    record_best(history, state.n, state.h, state.c)
    record_best(history, n0e, upsample_linear(state.h, n0e), None)

    if time.time() < deadline - 90.0:
        h, c = optimize_regular_lbfgs(state.h, state.n, state, deadline,
                                      min(70.0, deadline - time.time()))
        if h is not None:
            record_best(history, state.n if state.n >= 0 else state.n, h, c)

    if time.time() < deadline - 70.0 and state.n % 2 == 0:
        hb = best_anti_from_h(state.h, state.n)
        f = hb[:state.n // 2]
        h, c = optimize_anti_lbfgs(f, state.n, state, deadline,
                                   min(50.0, deadline - time.time()))
        if h is not None:
            record_best(history, state.n, h, c)

    if n0e <= 220 and time.time() < deadline - 80.0:
        bh = make_binary_from_float(upsample_linear(state.h, n0e), n0e)
        bh, bc = binary_hill_search(bh, n0e, state, deadline, rng,
                                    min(25.0, deadline - time.time()))
        if bh is not None:
            record_best(history, n0e, bh, bc)

    fracs = (1.0, 2.0, 1.5, 3.0, 2.5, 3.5)
    n_order = []
    for fr in fracs:
        nn = int(round(n0e * float(fr) / 2.0) * 2)
        if nn < 8:
            nn = 8
        if nn > 330:
            continue
        if nn not in n_order:
            n_order.append(nn)
    if 140 <= 330 and 140 not in n_order:
        pos = len(n_order)
        for i, nn in enumerate(n_order):
            if nn > 140:
                pos = i
                break
        n_order.insert(pos, 140)

    for n in n_order:
        if time.time() >= deadline - 100.0:
            break
        rem = float(deadline - time.time())
        if rem < 80.0:
            break

        if n <= n0e + 2:
            tn = min(35.0, rem - 80.0)
        elif n <= 2 * n0e + 10:
            tn = min(230.0, rem - 80.0)
        elif n <= 3 * n0e + 10:
            tn = min(170.0, rem - 80.0)
        else:
            tn = min(130.0, rem - 80.0)
        if tn < 40.0:
            continue

        if n in history:
            base = history[n][0]
        elif state.n == n:
            base = state.h.copy()
        else:
            base = upsample_linear(state.h, n)
        base = as_feasible(base, n)
        state.update(base, n, compute_c5(base))
        record_best(history, n, base, None)

        if n <= 220 and rem > 110.0:
            bh = make_binary_from_float(base, n)
            bh, bc = binary_hill_search(bh, n, state, deadline, rng,
                                        min(12.0, tn * 0.15))
            if bh is not None:
                record_best(history, n, bh, bc)
                if state.n == n:
                    base = state.h.copy()

        if n % 2 == 0 and rem > 95.0:
            hb = best_anti_from_h(base, n)
            record_best(history, n, hb, compute_c5(hb))
            f = hb[:n // 2]
            t_anti = min(80.0, tn * 0.40, rem - 60.0)
            if t_anti > 18.0:
                h, c = optimize_anti_lbfgs(f, n, state, deadline, t_anti)
                if h is not None:
                    record_best(history, n, h, c)

            if rem > 130.0:
                hs = synthetic_anti(n, rng)
                hb2 = best_anti_from_h(hs, n)
                record_best(history, n, hb2, compute_c5(hb2))
                f2 = hb2[:n // 2]
                t2 = min(35.0, tn * 0.22, rem - 55.0)
                if t2 > 10.0:
                    h, c = optimize_anti_lbfgs(f2, n, state, deadline, t2)
                    if h is not None:
                        record_best(history, n, h, c)

        if rem > 75.0:
            if n in history:
                hstart = history[n][0]
            elif state.n == n:
                hstart = state.h.copy()
            else:
                hstart = upsample_linear(state.h, n)
            hstart = as_feasible(hstart, n)

            h, c = swap_refine(hstart, n, state, deadline, min(5.0, tn * 0.10), rng)
            if h is not None:
                record_best(history, n, h, c)

            if rem > 80.0:
                if n <= 150:
                    t_reg = min(95.0, tn * 0.45, rem - 45.0)
                    if t_reg > 20.0:
                        h, c = optimize_slsqp_regular(hstart, n, state, deadline, t_reg)
                else:
                    t_reg = min(150.0, tn * 0.55, rem - 45.0)
                    if t_reg > 20.0:
                        h, c = optimize_regular_lbfgs(hstart, n, state, deadline, t_reg)
                if h is not None:
                    record_best(history, n, h, c)

        if rem > 65.0:
            if n in history:
                hpol = history[n][0]
                npol = n
            elif state.n == n:
                hpol = state.h.copy()
                npol = n
            else:
                hpol = upsample_linear(state.h, n)
                npol = n
            h, c = swap_refine(hpol, npol, state, deadline, min(10.0, tn * 0.15), rng)
            if h is not None:
                record_best(history, npol, h, c)

    if time.time() < deadline - 55.0:
        if state.n in history:
            hp = history[state.n][0]
        else:
            hp = state.h.copy()
        h, c = swap_refine(hp, state.n, state, deadline, min(25.0, deadline - time.time() - 10.0), rng)
        if h is not None:
            record_best(history, state.n, h, c)

    if time.time() < deadline - 45.0 and state.n % 2 == 0:
        if state.n in history:
            hb = best_anti_from_h(history[state.n][0], state.n)
        else:
            hb = best_anti_from_h(state.h, state.n)
        record_best(history, state.n, hb, compute_c5(hb))
        f = hb[:state.n // 2]
        h, c = optimize_anti_lbfgs(f, state.n, state, deadline, min(35.0, deadline - time.time() - 10.0))
        if h is not None:
            record_best(history, state.n, h, c)

    if time.time() < deadline - 35.0:
        if state.n in history:
            hp = history[state.n][0]
        else:
            hp = state.h.copy()
        h, c = optimize_regular_lbfgs(hp, state.n, state, deadline, min(50.0, deadline - time.time() - 8.0))
        if h is not None:
            record_best(history, state.n, h, c)

    ms = []
    for m in (state.n // 2, state.n // 4, 70, 100, 140, 180, 210, 280):
        try:
            m = int(m)
        except Exception:
            m = -1
        if m >= 4 and m not in ms:
            ms.append(m)

    for m in ms:
        if time.time() > deadline - 10.0:
            break
        hd = downsample_large(state.h, m)
        if hd is not None:
            record_best(history, m, hd, compute_c5(hd))
            if state.n == m and m % 2 == 0:
                ha = best_anti_from_h(hd, m)
                record_best(history, m, ha, compute_c5(ha))

    final_h = state.h
    final_n = int(state.n)
    final_c = float(state.c)

    if history:
        global_best = None
        for nd, item in history.items():
            h, c = item
            if global_best is None or c < global_best[1] - 1e-14:
                global_best = (nd, h, c)
        if global_best is not None:
            nd, h, c = global_best
            if math.isfinite(c) and c <= final_c + 1e-12:
                final_h = h
                final_n = nd
                final_c = c

    if final_h is None or final_n <= 0:
        final_h = as_feasible(h0, h0.size)
        final_n = int(final_h.size)
        final_c = compute_c5(final_h)

    if not is_valid_candidate(final_h):
        final_h = as_feasible(final_h, final_h.size)
        final_c = compute_c5(final_h)
        final_n = int(final_h.size)

    return final_h.copy(), float(final_c), int(final_n)
