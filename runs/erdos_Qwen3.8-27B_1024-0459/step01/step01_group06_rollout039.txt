            h_opt = run_slsqp_hard(h_opt, n_try, 500, 1e-13)
            c5_struct = compute_c5(h_opt, n_try)
            if c5_struct < best_c5:
                best_h = h_opt
                best_c5 = c5_struct
                best_n = n_try

    # Phase 4: Random restarts at best resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 80
    n_restarts = max(5, int(remaining / 20))
    for i in range(n_restarts):
        if time.time() - t0 > budget_s - 60:
            break
        scale = 0.03 + 0.05 * (i / max(1, n_restarts))
        h_pert = best_h + rng.randn(best_n) * scale
        h_pert = project_feasible(h_pert, best_n)
        h_opt = run_slsqp_hard(h_pert, best_n, 300, 1e-12)
        c5_opt = compute_c5(h_opt, best_n)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 5: Local search polish
    if time.time() - t0 < budget_s - 40:
        t_limit_ls = t0 + budget_s - 25
        best_h, best_c5 = local_search(best_h, best_n, rng, t_limit_ls, max_rounds=15)

    # Phase 6: Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        h_new = run_slsqp_hard(best_h, best_n, 2000, 1e-15)
        c5_new = compute_c5(h_new, best_n)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
