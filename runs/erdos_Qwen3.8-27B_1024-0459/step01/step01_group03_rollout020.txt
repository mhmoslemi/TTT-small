def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) <= 1e-14 * max(1, n):
        return x
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]

            idx = int(free[np.argmax(np.abs(h[free] - 0.5))])

        h[start:max(start+1, end)] = float(rng.choice([0.0, 1.0, 0.5]))

    alpha = _ALPHA

        return 0.0

def smooth_jacobian(x):
    n = _NC
    alpha = _ALPHA
    if n <= 1:
        return np.zeros(max(1, n))
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")

        return np.zeros(n)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    idx = np.where(w > 1e-10)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))])

                g[lag:] += p
            else:
                g[:n - lag] += p
            g[:n - lag] -= p * h[lag:]
            g[lag:] -= p * h[:n - lag]
    return (2.0 / float(n)) * g

def corr_grad_for_lags(h, lags):

    g = np.zeros(n)

    if n < 4:
        return best, best_c
    rng_local = np.random.default_rng(12345)
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        group_size = int(rng_local.integers(2, max(3, n // 20)))
        i1 = int(rng_local.integers(0, n - group_size))
        i2 = int(rng_local.integers(0, n - group_size))
        if abs(i1 - i2) < group_size:
            continue
        cand = best.copy()
        tmp = cand[i1:i1+group_size].copy()
        cand[i1:i1+group_size] = cand[i2:i2+group_size]
        cand[i2:i2+group_size] = tmp
        cand = project_density(cand, target)
        c = evaluate_erdos(cand)
        if c < best_c - 1e-12:
            best = cand
            best_c = c
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
    return best, best_c


def optimize_at_resolution(h0, n, deadline, use_slqp=True):
    n = int(n)
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    now = time.perf_counter()
    span = max(1.0, deadline - now)
    polish_time = min(60.0, max(10.0, 0.25 * span))
    slqp_deadline = deadline - polish_time

    if use_slqp and n <= 1200:
        if n <= 120:
            alphas = (10.0, 40.0, 120.0, 300.0)
            maxiter = 200
        elif n <= 300:
            alphas = (15.0, 50.0, 150.0, 400.0)
            maxiter = 120
        elif n <= 600:
            alphas = (15.0, 50.0, 150.0)
            maxiter = 80
        else:
            alphas = (15.0, 50.0)
            maxiter = 50

        bounds = [(0.0, 1.0)] * n
        constraints = ({"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac},)

        for alpha in alphas:
            if time.perf_counter() >= slqp_deadline:
                break
            set_context(n, alpha, target)
            try:
                res = minimize(
                    smooth_objective, h,
                    method="SLSQP",
                    jac=smooth_jacobian,
                    bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": maxiter, "ftol": 1e-10, "disp": False}
                )
                xg = getattr(res, "x", None)
                if xg is not None:
                    xg = np.asarray(xg, dtype=float)
                    if np.all(np.isfinite(xg)):
                        cand = project_density(xg, target)
                        c = evaluate_erdos(cand)
                        if c < best_c:
                            best_h = cand
                            best_c = c
                        if c <= best_c + 0.03:
                            h = cand
                        else:
                            h = best_h
            except Exception:
                h = best_h

    if time.perf_counter() < deadline - 2.0:
        rem = deadline - time.perf_counter()
        if rem > 5.0:
            h2, c2 = active_set_polish(best_h, time.perf_counter() + rem * 0.4, max_iter=400, top_k=5)
            if c2 < best_c:
                best_h, best_c = h2, c2
        if time.perf_counter() < deadline - 3.0:
            h3, c3 = pair_coordinate_search(best_h, time.perf_counter() + (deadline - time.perf_counter()) * 0.5, max_iter=600, top_k=5)
            if c3 < best_c:
                best_h, best_c = h3, c3

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)

    rng = np.random.default_rng(seed)
    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        nparent = int(parent.size)
        pproj = project_density(parent, 0.5 * nparent)
        cp = evaluate_erdos(pproj)
        if cp < best_c:
            best_h = pproj
            best_c = cp
            best_n = nparent

    if best_h is None:
        n0 = 120
        best_h = make_candidate(n0, rng, 0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Phase 1: Optimize at base resolution
    base_n = max(best_n, 90)
    if deadline - time.perf_counter() > 60:
        h1, c1 = optimize_at_resolution(best_h, base_n, time.perf_counter() + min(80.0, (deadline - start) * 0.12))
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, base_n

    # Phase 2: Multi-resolution optimization
    target_ns = []
    for mult in (1, 2, 3, 4, 6, 8):
        n = int(base_n * mult)
        if 60 <= n <= 1440:
            target_ns.append(n)
    for n in (180, 270, 360, 540, 720, 900, 1080, 1440):
        if n not in target_ns and 60 <= n <= 1440:
            target_ns.append(n)
    target_ns = sorted(set(target_ns))

    # Sort by resolution, optimize progressively
    for ni in target_ns:
        if deadline - time.perf_counter() < 40.0:
            break
        rem = deadline - time.perf_counter()
        # Allocate time based on resolution
        if ni <= 200:
            alloc = min(100.0, rem * 0.3)
        elif ni <= 500:
            alloc = min(200.0, rem * 0.3)
        elif ni <= 900:
            alloc = min(300.0, rem * 0.35)
        else:
            alloc = min(400.0, rem * 0.4)
        alloc = max(30.0, alloc)

        # Build starting points
        starts = []
        starts.append(upsample_h(best_h, ni))
        if parent is not None:
            starts.append(upsample_h(parent, ni))
        # Add a random start for diversity
        if ni <= 600:
            starts.append(make_candidate(ni, rng, int(rng.integers(0, 4))))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if vector_close(s, u):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        # Optimize from each starting point (best first)
        uniq.sort(key=lambda x: evaluate_erdos(x))
        for idx_s, s in enumerate(uniq):
            sub_rem = deadline - time.perf_counter()
            if sub_rem < 20.0:
                break
            # Give more time to the first (best) starting point
            if idx_s == 0:
                sub_alloc = min(alloc * 0.55, sub_rem * 0.6)
            else:
                sub_alloc = min(alloc * 0.3, sub_rem * 0.4)
            sub_alloc = max(15.0, sub_alloc)
            sub_deadline = time.perf_counter() + min(sub_alloc, sub_rem - 5.0)

            # Only use SLSQP for smaller n or first start
            use_slqp = (ni <= 600) or (idx_s == 0)
            h_out, c_out = optimize_at_resolution(s, ni, sub_deadline, use_slqp=use_slqp)
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * ni)
                c_out = evaluate_erdos(h_out)
                if c_out < best_c:
                    best_h = h_out
                    best_c = c_out
                    best_n = ni
                    # After improvement, upsample and try next resolution earlier
                    if best_c < 0.381:
                        break

    # Phase 3: Final intensive polish at best resolution
    if best_h is not None and deadline - time.perf_counter() > 20.0:
        nbest = int(best_h.size)
        # Try going one step higher in resolution
        if nbest < 1440 and deadline - time.perf_counter() > 120.0:
            n_try = min(nbest * 2, 1440)
            if n_try > nbest:
                h_try = upsample_h(best_h, n_try)
                h_try, c_try = optimize_at_resolution(h_try, n_try, time.perf_counter() + min(300.0, (deadline - start) * 0.3), use_slqp=(n_try <= 720))
                if c_try < best_c:
                    best_h, best_c, best_n = h_try, c_try, n_try

        # Aggressive final polish
        rem = deadline - time.perf_counter()
        if rem > 10.0:
            h_p, c_p = active_set_polish(best_h, time.perf_counter() + rem * 0.35, max_iter=1000, top_k=8)
            if c_p < best_c:
                best_h, best_c = h_p, c_p
        if deadline - time.perf_counter() > 8.0:
            h_p2, c_p2 = pair_coordinate_search(best_h, time.perf_counter() + (deadline - time.perf_counter()) * 0.5, max_iter=2000, top_k=6)
            if c_p2 < best_c:
                best_h, best_c = h_p2, c_p2
        if deadline - time.perf_counter() > 5.0:
            h_p3, c_p3 = active_set_polish(best_h, time.perf_counter() + (deadline - time.perf_counter()) * 0.6, max_iter=800, top_k=10)
            if c_p3 < best_c:
                best_h, best_c = h_p3, c_p3

    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
