def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def active_gradient(h, n_points):
    """Compute gradient of the active (max) constraint."""
    n = n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    k_act = int(np.argmax(conv))
    j = k_act - (n - 1)  # shift
    
    n_pts = n  # avoid shadowing
    grad = np.zeros(n_pts)
    
    if j >= 0:
        # out = sum_{n=0}^{n-1-j} h[n+j]*(1-h[n])
        # d/dh[p] = (1-h[p-j])*I[j<=p<=n-1] - h[p+j]*I[0<=p<=n-1-j]
        if j == 0:
            grad[:] = 1.0 - 2.0 * h
        else:

            grad[j:] += (1.0 - h[:n_pts-j])
            # p in [0, n-1-j]: contributes -h[p+j]
            grad[:n_pts-j] += -h[j:]
    else:

            grad[:n_pts-t] += (1.0 - h[t:])
            # p in [t, n-1]: contributes -h[p-t]
            grad[t:] += -h[:n_pts-t]
