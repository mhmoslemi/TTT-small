def linear_conv(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    la = a.size
    lb = b.size
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    if la * lb <= 1000000:
        return np.convolve(a, b)
    nout = la + lb - 1
    nfft = next_fast_len(nout)
    fa = rfft(a, nfft)
    fb = rfft(b, nfft)
    out = irfft(fa * fb, nfft)
    return out[:nout].astype(float, copy=False)


def compute_corr_fft(h):
    h = np.asarray(h, dtype=float)

        return np.array([], dtype=float)
    g = 1.0 - h
    return linear_conv(h, g[::-1])

def smooth_value_from_c(c, hn, beta):

def call_lbfgs(fun, jac, x0, args, nvars, maxfun, ftol=1e-12, gtol=1e-8):
    if maxfun is None or maxfun < 10:
        return None
    x0 = np.asarray(x0, dtype=float).ravel()
    if x0.size != nvars:
        return None
    x0 = np.clip(x0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * nvars
    opts = {
        "maxiter": int(maxfun),
        "maxfun": int(maxfun),
        "ftol": ftol,
        "gtol": gtol,
        "maxcor": 20,
        "maxls": 20,
        "disp": False,
    }
    try:
        res = minimize(
            fun,
            x0,
            method="L-BFGS-B",
            jac=jac,
            args=args,
            bounds=bounds,
            options=opts,
        )
        x = getattr(res, "x", None)
        if x is None:
            return None
        x = np.asarray(x, dtype=float).ravel()
        if x.size != nvars:
            return None
        if not np.all(np.isfinite(x)):
            return None
        return x
    except OptTimeout:
        return None
    except Exception:
        return None

        hfull, _ = improve_full(best_n, best_h, BETAS_MAIN, cap_for_n(best_n))
        hnew, cnew = consider_candidate(hfull, best_n, best_c5)
        if hnew is not None:
            best_h = hnew
            best_c5 = cnew
            best_n = int(best_n)

    raw_ns = [128, 192, 256, 384, 512, 768, 1024]
    if budget_s > 1100.0:
        raw_ns.extend([1536, 2048])
    if best_n * 2 <= 2048:
        raw_ns.append(int(best_n * 2))
    raw_ns = sorted(set([int(x) for x in raw_ns]))

    for n in raw_ns:
        if n <= best_n:
            continue
        reserve = 120.0 + 0.06 * float(n)
        if time.time() > _DEADLINE - reserve:
            break

        h0 = resample_h(best_h, n)
        h0 = project_box_sum(h0, 0.5 * float(n))
        cap = cap_for_n(n)

        v0 = project_anti(h0)
        h_ant, _ = improve_anti(n, v0, BETAS_MAIN, cap)
        hnew, cnew = consider_candidate(h_ant, n, best_c5)
        anti_improved = hnew is not None
        if anti_improved:
            best_h = hnew
            best_c5 = cnew
            best_n = n

        if time.time() < _DEADLINE - 80.0:
            if n <= 384:
                fb = BETAS_MAIN
                fc = cap
            elif n <= 1024:
                fb = BETAS_SHORT
                fc = max(100, cap // 2)
            else:
                fb = BETAS_SHORT[:2]
                fc = max(80, cap // 2)
            hfull, _ = improve_full(n, h0, fb, fc)
            hnew, cnew = consider_candidate(hfull, n, best_c5)
            if hnew is not None:
                best_h = hnew
                best_c5 = cnew
                best_n = n

        if n <= 1024 and h_ant is not None and time.time() < _DEADLINE - 70.0:
            hstart = project_box_sum(h_ant, 0.5 * float(n))
            hfull2, _ = improve_full(n, hstart, BETAS_SHORT[:2], max(80, cap // 3))
            hnew, cnew = consider_candidate(hfull2, n, best_c5)
            if hnew is not None:
                best_h = hnew
                best_c5 = cnew
                best_n = n

    if time.time() < _DEADLINE - 120.0:
        if best_n <= 256:
            n_rand = 512
        elif best_n <= 1024:
            n_rand = max(512, int(best_n))
        else:
            n_rand = 768

        for s in range(2):
            if time.time() > _DEADLINE - 60.0:
                break
            hrand = make_fourier_h(n_rand, seed + s)
            h_ant = improve_anti(n_rand, project_anti(hrand), BETAS_SHORT, 200)[0]
            hnew, cnew = consider_candidate(h_ant, n_rand, best_c5)
            if hnew is not None:
                best_h = hnew
                best_c5 = cnew
                best_n = n_rand

        if time.time() < _DEADLINE - 80.0:
            hrand = make_block_h(n_rand, seed + 11)
            h_ant = improve_anti(n_rand, project_anti(hrand), BETAS_SHORT, 220)[0]
            hnew, cnew = consider_candidate(h_ant, n_rand, best_c5)
            if hnew is not None:
                best_h = hnew
                best_c5 = cnew
                best_n = n_rand

    best_h = project_box_sum(best_h, 0.5 * float(best_n))
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = exact_c5(best_h)

    if best_n <= 300 and time.time() < _DEADLINE - 60.0:
        for beta in (150.0, 600.0, 1500.0):
            if time.time() > _DEADLINE - 25.0:
                break
            hs = call_slq(best_h, beta, best_n, maxiter=80)
            hnew, cnew = consider_candidate(hs, best_n, best_c5)
            if hnew is not None:
                best_h = hnew
                best_c5 = cnew

    if time.time() < _DEADLINE - 60.0:
        v = project_anti(best_h)
        h_ant = improve_anti(best_n, v, BETAS_FINE, cap_for_n(best_n))[0]
        hnew, cnew = consider_candidate(h_ant, best_n, best_c5)
        if hnew is not None:
            best_h = hnew
            best_c5 = cnew

        hfull = improve_full(best_h, BETAS_FINE[:2], cap_for_n(best_n))[0]
        hnew, cnew = consider_candidate(hfull, best_n, best_c5)
        if hnew is not None:
            best_h = hnew
            best_c5 = cnew

    if best_n <= 1536 and time.time() < _DEADLINE - 45.0:
        hwr = weighted_refine(project_box_sum(best_h, 0.5 * float(best_n)), reps=2, cap=150)[0]
        hnew, cnew = consider_candidate(hwr, best_n, best_c5)
        if hnew is not None:
            best_h = hnew
            best_c5 = cnew

    if time.time() < _DEADLINE - 30.0:
        mir = max(40, min(120, 22000 // max(1, best_n)))
        hr = exact_refine(project_box_sum(best_h, 0.5 * float(best_n)), max_iter=mir)[0]
        hnew, cnew = consider_candidate(hr, best_n, best_c5)
        if hnew is not None:
            best_h = hnew
            best_c5 = cnew

    if time.time() < _DEADLINE - 25.0:
        v = project_anti(best_h)
        h_ant = improve_anti(best_n, v, (500.0,), 60)[0]
        hnew, cnew = consider_candidate(h_ant, best_n, best_c5)
        if hnew is not None:
            best_h = hnew
            best_c5 = cnew

    best_h = project_box_sum(best_h, 0.5 * float(best_n))
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = exact_c5(best_h)

    return best_h, float(best_c5), int(best_n)
