def optimize_with_analytic_grad(h_start, n, target, beta, max_iter=300):
    dx = 2.0 / n
    h0 = project_sum(h_start, target)
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr, beta) * dx
    
    def obj_grad(h):
        return compute_gradient(h, beta, dx)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': max_iter, 'ftol': 1e-12})
        if result.x is not None:
            return project_sum(result.x, target)
    except Exception:
        pass
    return h0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_shifted = np.exp(shifted)
    return exp_shifted / np.sum(exp_shifted)

def compute_gradient_fast(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, beta)
    
    lags = np.arange(2*n-1) - (n-1)
    js = np.arange(n)[:, None]
    ls = lags[None, :]
    
    idx1 = js - ls
    idx2 = js + ls
    
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    idx1_c = np.clip(idx1, 0, n-1)
    idx2_c = np.clip(idx2, 0, n-1)
    
    s1 = np.sum(weights[None, :] * (1.0 - h[idx1_c]) * valid1, axis=1)
    s2 = np.sum(weights[None, :] * h[idx2_c] * valid2, axis=1)
    
    return dx * (s1 - s2)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def _objective_grad(h, beta, dx):
    return compute_gradient_fast(h, beta, dx)

def _sum_constraint(h, target):
    return np.sum(h) - target

def _sum_constraint_jac(h, target):
    return np.ones(len(h))

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            free = (h < 1.0)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
        else:
            free = (h > 0.0)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def optimize_analytic(h_start, n, target, beta, max_iter=300, ftol=1e-12):
    dx = 2.0 / n
    h0 = project_sum(h_start, target)
    
    cons = [{'type': 'eq',
             'fun': partial(_sum_constraint, target=target),
             'jac': partial(_sum_constraint_jac, target=target)}]
    bounds = [(0.0, 1.0)] * n
    
    obj_fn = partial(_objective, beta=beta, dx=dx)
    obj_grad_fn = partial(_objective_grad, beta=beta, dx=dx)
    
    try:
        result = minimize(obj_fn, h0, jac=obj_grad_fn, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': ftol})
        if result.x is not None:
            return project_sum(result.x, target)
    except Exception:
        pass
    return h0

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    if s == 0:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
        return w
    return exp_shifted / s

def compute_gradient_fast(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, beta)
    lags = np.arange(2 * n - 1) - (n - 1)
    js = np.arange(n)[:, None]
    ls = lags[None, :]
    idx1 = js - ls
    idx2 = js + ls
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1_c = np.clip(idx1, 0, n - 1)
    idx2_c = np.clip(idx2, 0, n - 1)
    s1 = np.sum(weights[None, :] * (1.0 - h[idx1_c]) * valid1, axis=1)
    s2 = np.sum(weights[None, :] * h[idx2_c] * valid2, axis=1)
    return dx * (s1 - s2)

def _objective(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def _objective_grad(h, beta, dx):
    return compute_gradient_fast(h, beta, dx)

def _sum_constraint(h, target):
    return np.sum(h) - target

def _sum_constraint_jac(h, target):
    return np.ones(len(h))

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            free = (h < 1.0)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
        else:
            free = (h > 0.0)
            if np.any(free):
                per = diff / np.sum(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def optimize_analytic(h_start, n, target, beta, max_iter=300, ftol=1e-12):
    dx = 2.0 / n
    h0 = project_sum(h_start, target)
    cons = [{'type': 'eq',
             'fun': partial(_sum_constraint, target=target),
             'jac': partial(_sum_constraint_jac, target=target)}]
    bounds = [(0.0, 1.0)] * n
    obj_fn = partial(_objective, beta=beta, dx=dx)
    obj_grad_fn = partial(_objective_grad, beta=beta, dx=dx)
    try:
        result = minimize(obj_fn, h0, jac=obj_grad_fn, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': ftol})
        if result.x is not None:
            return project_sum(result.x, target)
    except Exception:
        pass
    return h0

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    h[h < 0.5] = 0.0
    h[h >= 0.5] = 1.0
    target = n / 2.0
    h = project_sum(h, target)
    return h

def smooth_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    a = rng.uniform(0.3, 0.7)
    b = rng.uniform(0.2, 0.6)
    c = rng.uniform(0.5, 3.0)
    d = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.3 * np.sin(c * x + d) + b * np.sin(2 * c * x + a * x)
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, n / 2.0)
    return h

def interpolate_to_resolution(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_sum(h_new, n_new / 2.0)
    return h_new
