    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection for the offset
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo+hi)/2, 0, 1)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)

import time
import math
from scipy.optimize import minimize

def _compute_scores(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full")

def _compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    scri = _compute_scores(h)
    c = float(np.max(scri)) * (2.0 / n)
    return c if math.isfinite(c) else 1e6

def _project(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n)
            m = min(n, h.size)
            if m > 0:
                tmp[:m] = h[:m]
            h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
            cnt = float(np.count_nonzero(mask))
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            mask = h > 0.0
            cnt = float(np.count_nonzero(mask))
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def _lse_val(scores, alpha):
    m = float(np.max(scores))
    if alpha < 1e-14:
        return m
    return m + alpha * math.log(float(np.sum(np.exp((scores - m) / alpha))))

def _sor_cost(h, alpha, n):
    scores = _compute_scores(h)
    return _lse_val(scores, alpha) * (2.0 / n)

def _sor_grad(h, alpha, n):
    scores = _compute_scores(h)
    m = float(np.max(scores))
    if alpha < 1e-14:
        tol = max(1e-9, 1e-8 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(scores))])
        w = np.zeros(len(scores))
        w[active] = 1.0 / float(active.size)
    else:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if A <= 0 or not math.isfinite(A):
            return np.zeros(n)
        w /= A
    grad = np.zeros(n, dtype=float)
    for k in range(len(scores)):
        ck = w[k]
        if ck < 1e-13:
            continue
        if k == n - 1:
            grad += ck * (1.0 - 2.0 * h)
        elif k < n - 1:
            d = (n - 1) - k
            if d <= 0 or d >= n:
                continue
            mm = n - d
            grad[:mm] += ck * (1.0 - h[d:])
            grad[d:] += ck * (-h[:mm])
        else:
            d = k - (n - 1)
            if d <= 0 or d >= n:
                continue
            mm = n - d
            grad[:mm] += ck * (-h[d:])
            grad[d:] += ck * (1.0 - h[:mm])
    grad *= (2.0 / n)
    return grad

def _penalty_cost(h, alpha, lam, n):
    scores = _compute_scores(h)
    val = _lse_val(scores, alpha) * (2.0 / n)
    cv = float(h.sum()) - 0.5 * float(n)
    val += lam * cv * cv
    return val

def _penalty_grad(h, alpha, lam, n):
    g = _sor_grad(h, alpha, n)
    cv = float(h.sum()) - 0.5 * float(n)
    g += 2.0 * lam * cv
    return g

def _con_eq(x, *a):
    n = max(1, x.size)
    return (float(x.sum()) - 0.5 * float(n)) / float(n)

def _con_jac(x, *a):
    n = max(1, x.size)
    return np.full((1, n), 1.0 / float(n))

def _slsqp_opt(h0, n, alpha, maxiter, deadline, ftol=1e-10):
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _con_eq, "jac": _con_jac}]
    try:
        res = minimize(
            _sor_cost, h0.copy(), args=(alpha, n),
            jac=_sor_grad, method="SLSQP",
            bounds=bounds, constraints=cons,
            options={"maxiter": maxiter, "ftol": ftol, "disp": 0}
        )
        if res is not None and res.x is not None:
            h_out = _project(res.x, n)
            return h_out
    except Exception:
        pass
    if time.time() >= deadline:
        return None
    return h0

def _lbfgs_opt(h0, n, alpha, lam, maxiter, deadline):
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            _penalty_cost, h0.copy(), args=(alpha, lam, n),
            jac=_penalty_grad, method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-8, "disp": 0, "maxfun": maxiter * 3}
        )
        if res is not None and res.x is not None:
            h_out = _project(res.x, n)
            return h_out
    except Exception:
        pass
    return h0

def _binary_local(h_start, n, deadline, rng, max_iters=50000, stall_lim=5000):
    if n % 2 != 0:
        n2 = (n // 2) * 2
        h_start = _project(h_start[:n2] if h_start.size >= n2 else np.zeros(n2), n2)
        n = n2
    target = n // 2
    h = np.zeros(n, dtype=np.int8)
    vals = np.asarray(h_start, dtype=float)
    if vals.size == n:
        order = np.argsort(vals)
        h[order[target:]] = 1
    else:
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1
    hf = h.astype(float)
    scores = _compute_scores(hf)
    cur = float(np.max(scores))
    best = cur
    best_h = h.copy()
    ops = 0
    stall = 0
    while ops < max_iters and time.time() < deadline:
        ops += 1
        if ops % 200 == 0 and time.time() >= deadline:
            break
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if zeros.size == 0 or ones.size == 0:
            break
        i = int(zeros[rng.integers(zeros.size)])
        j = int(ones[rng.integers(ones.size)])
        if i == j:
            continue
        old_i = h[i]
        old_j = h[j]
        h[i] = 1
        h[j] = 0
        scores_new = _compute_scores(h.astype(float))
        new_max = float(np.max(scores_new))
        if new_max < cur - 1e-12:
            cur = new_max
            scores = scores_new
            stall = 0
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            if new_max > cur + 1e-9 and rng.random() > 0.95:
                h[i] = old_i
                h[j] = old_j
                stall += 1
            elif new_max <= cur + 1e-9 and rng.random() < max(0.01, 0.3 * (1.0 - ops / float(max_iters))):
                scores = scores_new
                stall += 1
            else:
                h[i] = old_i
                h[j] = old_j
                stall += 1
        if stall > stall_lim:
            break
    final_c = best * (2.0 / float(n))
    return best_h.astype(float), final_c

def _upsample(h_old, n_new):
    n_old = h_old.size
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old.astype(float))

def _synthetic_start(n, rng):
    best_h = np.full(n, 0.5)
    best_c = _compute_c5(best_h)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    for f in range(1, 6):
        for amp in [0.3, 0.4, 0.5, 0.6, 0.7]:
            for ph in [0.0, 0.5 * math.pi, 0.25 * math.pi]:
                h = 0.5 + amp * np.cos(math.pi * f * x + ph)
                h = _project(h, n)
                c = _compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
    for trial in range(5):
        h = 0.5 + 0.3 * np.sin(3.0 * math.pi * x + rng.random() * 2 * math.pi)
        h += 0.15 * np.random.default_rng(trial).normal(0, 0.1, n)
        h = _project(h, n)
        c = _compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except:
        budget_s = 1300.0
    deadline = t0 + budget_s - 5.0
    rng = np.random.default_rng(seed)

    # Get initial
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim != 1 or parent.size < 2:
            raise ValueError
    except:
        parent = None

    if parent is not None:
        n = parent.size
        h = parent.copy()
    else:
        n = 70
        h = _synthetic_start(n, rng)

    h = _project(h, n)
    best_h = h.copy()
    best_c = _compute_c5(best_h)

    def try_update(h_new, n_new):
        nonlocal best_h, best_c
        if h_new is None:
            return
        h_new = _project(h_new, n_new)
        c_new = _compute_c5(h_new)
        if math.isfinite(c_new) and c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 1: SLSQP alpha annealing at n
    # Alpha schedule: 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0, 0
    alpha_schedule = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.0, 0.0]
    h_work = best_h.copy()
    phase_deadline = t0 + 400.0
    iters_per_stage = max(30, int(120 * (70.0 / float(n))))
    for alpha in alpha_schedule:
        if time.time() > min(phase_deadline, deadline) - 10:
            break
        maxit = iters_per_stage
        if alpha == 0:
            maxit = iters_per_stage * 2
        st = time.time()
        min_time = 5.0
        rem = min(phase_deadline, deadline) - st - 5.0
        if rem < 10:
            break
        res = _slsqp_opt(h_work, n, alpha, maxit, min(phase_deadline, deadline))
        if res is not None:
            c_res = _compute_c5(res)
            if c_res < _compute_c5(h_work):
                h_work = res
            try_update(h_work, n)
        else:
            break

    # Phase 2: L-BFGS-B with penalty
    if time.time() < deadline - 60:
        for lam in [50.0, 200.0, 1000.0]:
            if time.time() > deadline - 30:
                break
            alpha_pb = 0.01
            h_work2 = best_h.copy()
            res = _lbfgs_opt(h_work2, n, alpha_pb, lam, 500, deadline)
            if res is not None:
                try_update(res, n)

    # Phase 3: Continue SLSQP with very small alpha from best
    if time.time() < deadline - 60:
        remaining3 = deadline - time.time() - 30
        maxit3 = max(50, int(300 * (70.0 / float(n))))
        h_work3 = best_h.copy()
        res = _slsqp_opt(h_work3, n, 0.0, maxit3, deadline - 25)
        if res is not None:
            try_update(res, n)

    # Phase 4: Binary local search
    if time.time() < deadline - 80 and n % 2 == 0:
        bin_deadline = deadline - 60
        bh, bc = _binary_local(best_h, n, bin_deadline, rng,
                                max_iters=100000, stall_lim=8000)
        if bh is not None and bc < best_c:
            best_c = bc
            best_h = bh.copy()

    # Phase 5: Multi-resolution - try n*2
    if time.time() < deadline - 120:
        n2 = 2 * n
        if n2 <= 280:
            h_up = _upsample(best_h, n2)
            h_up = _project(h_up, n2)
            try_update(h_up, n2)
            # SLSQP at n2
            alpha2 = [0.2, 0.1, 0.05, 0.03, 0.01, 0.0]
            it2 = max(20, int(80 * (70.0 / float(n2))))
            phase5_deadline = time.time() + min(250.0, deadline - time.time() - 30)
            for a2 in alpha2:
                if time.time() > phase5_deadline - 10:
                    break
                res2 = _slsqp_opt(h_up, n2, a2, it2, phase5_deadline)
                if res2 is not None:
                    c2 = _compute_c5(res2)
                    if c2 < _compute_c5(h_up):
                        h_up = res2
                    try_update(h_up, n2)

    # Phase 6: Try a different starting point (synthetic) and optimize
    if time.time() < deadline - 50:
        h_alt = _synthetic_start(n, rng)
        h_alt = _project(h_alt, n)
        c_alt = _compute_c5(h_alt)
        if c_alt < best_c + 0.02:
            for a in [0.2, 0.1, 0.01, 0.0]:
                if time.time() > deadline - 20:
                    break
                res_alt = _slsqp_opt(h_alt, n, a, 80, deadline - 15)
                if res_alt is not None:
                    h_alt = res_alt
                    try_update(h_alt, n)

    # Phase 7: Final careful polish at best resolution
    if time.time() < deadline - 15:
        h_final = best_h.copy()
        n_final = int(h_final.size)
        res_final = _slsqp_opt(h_final, n_final, 0.0,
                                max(30, int(100 * (70.0 / float(n_final)))),
                                deadline - 8)
        if res_final is not None:
            try_update(res_final, n_final)

    # Ensure best solution is valid
    best_h = _project(best_h, int(best_h.size))
    best_c = _compute_c5(best_h)
    n_final = int(best_h.size)

    return best_h, float(best_c), n_final
