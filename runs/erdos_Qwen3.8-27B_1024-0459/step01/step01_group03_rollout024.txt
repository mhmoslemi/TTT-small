        if parent is not None:
            starts.append(upsample_h(parent, ni))

        # Random starts (limited)
        nrand = 2 if ni <= 180 else 1
        for _ in range(nrand):
            starts.append(make_random_start(ni, rng))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        if not uniq:
            continue

        # Optimize from each start
        n_starts = len(uniq)
        per_time = (sub_deadline - time.perf_counter()) / float(n_starts)
        per_time = max(10.0, per_time)

        for s in uniq:
            if deadline - time.perf_counter() < 15.0:
                break
            sd = min(time.perf_counter() + per_time, sub_deadline)
            h_out, c_out = optimize_at_n(s, ni, sd)
            if c_out < best_c:
                best_h = h_out
                best_c = c_out
                best_n = ni

    # Final polishing on best solution
    if best_h is not None and deadline - time.perf_counter() > 20.0:
        nf = best_h.size
        tf = 0.5 * nf
        # Extra active-set polish
        h_pol, c_pol = cycle_active_polish(best_h, nf, deadline, max_iter=500)
        if c_pol < best_c:
            best_h = h_pol
            best_c = c_pol
        # Extra pair search
        if deadline - time.perf_counter() > 10.0:
            h_pr, c_pr = pair_local_search(best_h, nf, deadline, max_iter=1000, top_k=8)
            if c_pr < best_c:
                best_h = h_pr
                best_c = c_pr
        # Final block search
        if deadline - time.perf_counter() > 5.0:
            h_bl, c_bl = block_local_search(best_h, nf, deadline, max_iter=200)
            if c_bl < best_c:
                best_h = h_bl
                best_c = c_bl

    # Ensure feasibility
    if best_h is None:
        best_h = project_density(np.full(90, 0.5), 45.0)
        best_c = evaluate_erdos(best_h)
        best_n = 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
