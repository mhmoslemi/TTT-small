import numpy as np
from scipy.optimize import minimize
import time
import functools

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        per = diff / mask.sum()
        h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def smooth_max_obj(x, n, dx, tau):
    x = np.clip(x, 0.0, 1.0)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    val = m + tau * np.log(np.sum(np.exp((c - m) / tau)))
    return float(val * dx)

def eq_con(x, n):
    return np.array([x.sum() - n / 2.0])

def eq_jac(x, n):
    return np.ones((1, n))

    best_h = parent_h.copy()

Key insight: the correlation at lag k is:

r(-k) = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]) (for k ≥ 0)

Note that r(k) + r(-k) = sum_{i=0}^{n-1-k} h[i+k] + sum_{i=0}^{n-1-k} (1-h[i+k])(1-h[i])
... this doesn't simplify nicely.

Actually: r(k) = sum_{i=0}^{n-1-k} h[i+k] - sum_{i=0}^{n-1-k} h[i+k]*h[i]
And r(-k) = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i+k]*h[i]

So r(k) - r(-k) = sum_{i=0}^{n-1-k} (h[i+k] - h[i])

And r(k) + r(-k) = sum_{i=0}^{n-1-k} (h[i+k] + h[i]) - 2*sum_{i=0}^{n-1-k} h[i+k]*h[i]
= (n-k)*average of (h[i+k]+h[i]) - 2*autocorr(k)

This is interesting. The sum r(k) + r(-k) is related to the autocorrelation.

If h is "symmetric" in some sense, r(k) ≈ r(-k), so we just need to minimize r(k) for all k.
