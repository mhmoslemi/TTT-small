import numpy as np
import math
import time
from scipy.optimize import minimize


def time_remaining(t0, budget_s):
    return float(budget_s) - (time.time() - t0)


def project_box_sum(x, S):
    x = np.asarray(x, dtype=float).ravel()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(x)
    if n == 0:
        return x
    S = float(S)
    if S <= 0.0:
        return np.zeros(n, dtype=float)
    if S >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = float(np.sum(y))
        if s > S:
            lo = mid
        else:
            hi = mid

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(30):
        diff = S - float(np.sum(y))
        if abs(diff) < 1e-14:
            break
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if not np.any(mask):
            break
        y[mask] += diff / float(np.sum(mask))
        y = np.clip(y, 0.0, 1.0)

    return y


def evaluate_c5(h, n):
    h = np.asarray(h, dtype=float).ravel()
    if not np.all(np.isfinite(h)):
        return 1e8
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def sum_eq_fun(x, n):
    x = np.asarray(x, dtype=float).ravel()
    return float(np.sum(x) - 0.5 * float(n))


def sum_eq_jac(x, n):
    return np.ones(len(x), dtype=float)


def add_lag_gradient(g, h, d, wk):
    n = len(h)
    if wk == 0.0:
        return
    if d == 0:
        g += wk * (1.0 - 2.0 * h)
        return
    if d > 0:
        t = int(d)
        if t >= n:
            return
        g[:n - t] += wk * (1.0 - h[t:])
        g[t:] -= wk * h[:n - t]
    else:
        t = int(-d)
        if t >= n:
            return
        g[t:] += wk * (1.0 - h[:n - t])
        g[:n - t] -= wk * h[t:]


def soft_objective(x, n, beta):
    h = np.asarray(x, dtype=float).ravel()
    if not np.all(np.isfinite(h)):
        return 1e8
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = float(beta) * c
    m = float(np.max(scaled))
    if not np.isfinite(m):
        return 1e8
    z = scaled - m
    s = float(np.sum(np.exp(z)))
    if s <= 0.0 or not np.isfinite(s):
        return m / float(beta) * (2.0 / float(n))
    return float((m + math.log(s)) / float(beta) * (2.0 / float(n)))


def soft_gradient(x, n, beta):
    h = np.asarray(x, dtype=float).ravel()
    if not np.all(np.isfinite(h)):
        return np.zeros(len(h), dtype=float)
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = float(beta) * c
    m = float(np.max(scaled))
    z = scaled - m
    w = np.exp(z)
    sw = float(np.sum(w))
    if sw <= 0.0 or not np.isfinite(sw):
        w = np.zeros_like(w)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / sw

    g = np.zeros(len(h), dtype=float)
    base = n - 1
    for idx, wk in enumerate(w):
        if wk < 1e-14:
            continue
        d = base - int(idx)
        add_lag_gradient(g, h, d, float(wk))
    return g * (2.0 / float(n))


def slsqp_refine(h, n, beta, maxiter):
    h = np.asarray(h, dtype=float).ravel()
    n = int(n)
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = evaluate_c5(best_h, n)

    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [
        {"type": "eq", "fun": sum_eq_fun, "jac": sum_eq_jac, "args": (n,)}
    ]

    try:
        res = minimize(
            soft_objective,
            h,
            jac=soft_gradient,
            args=(n, float(beta)),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False},
        )
        x = np.asarray(res.x, dtype=float).ravel()
        if np.all(np.isfinite(x)):
            x = np.clip(x, 0.0, 1.0)
            x = project_box_sum(x, 0.5 * n)
            c = evaluate_c5(x, n)
            if c < best_c:
                return x, c
    except Exception:
        pass

    return best_h, best_c


def interpolate_grid(h, n_new):
    h = np.asarray(h, dtype=float).ravel()
    n = len(h)
    n_new = int(n_new)
    if n == n_new:
        return project_box_sum(h, 0.5 * n_new)
    x_old = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, h)
    return project_box_sum(y, 0.5 * n_new)


def fast_local_search(h, n, deadline, seed, max_trials=70):
    rng = np.random.RandomState(seed)
    n = int(n)
    S = 0.5 * float(n)
    h = project_box_sum(np.asarray(h, dtype=float).ravel(), S).astype(float)
    dx = 2.0 / float(n)
    base_corr = np.correlate(h, 1.0 - h, mode="full")
    base_C = float(np.max(base_corr)) * dx

    best_h = h.copy()
    best_C = base_C
    base_idx = n - 1
    no_improve = 0
    T = 2.0e-5

    if n > 150:
        max_trials = max(25, int(max_trials) // 2)

    while time.time() < deadline:
        cs = base_corr
        cur_C = base_C

        if cur_C < best_C:
            best_C = cur_C
            best_h = h.copy()

        if no_improve == 0:
            T = max(1.0e-6, T * 0.97)
        if no_improve > 500:
            T = max(T, 5.0e-5)
        if no_improve > 2000:
            T = min(2.0e-4, T * 1.05)

        M = len(cs)
        top_n = min(M, max(3, n // 30))
        top_band = max(5, min(M, M // 4))
        order = np.argsort(-cs)

        focus_idxs = []
        for idx in order[:top_n]:
            focus_idxs.append(int(idx))
        for _ in range(top_n):
            focus_idxs.append(int(order[rng.randint(0, top_band)]))

        focus = []
        seen = set()
        for idx in focus_idxs:
            if idx not in seen:
                seen.add(idx)
                focus.append(idx)

        trial_C = np.inf
        trial_h = None
        trial_corr = None
        cand_count = 0

        for idx in focus:
            if cand_count >= max_trials:
                break
            d = base_idx - int(idx)
            if d == 0:
                continue
            if d > 0:
                t = int(d)
                if t >= n:
                    continue
                src = np.arange(n - t)
                tgt = src + t
                contrib = h[src] * (1.0 - h[tgt])
            else:
                t = int(-d)
                if t >= n:
                    continue
                src = np.arange(t, n)
                tgt = src - t
                contrib = h[src] * (1.0 - h[tgt])

            if len(contrib) == 0:
                continue
            kpair = min(4, len(contrib))
            pair_order = np.argsort(-contrib)[:kpair]

            for p in pair_order:
                if cand_count >= max_trials:
                    break
                s = int(src[p])
                u = int(tgt[p])
                maxd = min(float(h[s]), 1.0 - float(h[u]))
                if maxd < 1e-12:
                    continue
                for frac in (0.25, 0.5, 1.0):
                    if cand_count >= max_trials:
                        break
                    delta = maxd * float(frac)
                    if delta <= 1e-12:
                        continue
                    ht = h.copy()
                    va = ht[s] - delta
                    vb = ht[u] + delta
                    if va < 0.0:
                        va = 0.0
                    if vb > 1.0:
                        vb = 1.0
                    ht[s] = va
                    ht[u] = vb
                    ct = np.correlate(ht, 1.0 - ht, mode="full")
                    Cv = float(np.max(ct)) * dx
                    if Cv < trial_C:
                        trial_C = Cv
                        trial_h = ht
                        trial_corr = ct
                    cand_count += 1

        if cand_count < max_trials:
            attempts = 0
            while cand_count < max_trials and attempts < 30:
                attempts += 1
                a = int(rng.randint(n))
                b = int(rng.randint(n))
                if a == b:
                    continue
                if rng.rand() < 0.5:
                    s, u = a, b
                else:
                    s, u = b, a
                maxd = min(float(h[s]), 1.0 - float(h[u]))
                if maxd < 1e-12:
                    continue
                delta = maxd * (0.1 + 0.9 * float(rng.rand()))
                if delta <= 1e-12:
                    continue
                ht = h.copy()
                va = ht[s] - delta
                vb = ht[u] + delta
                if va < 0.0:
                    va = 0.0
                    delta = float(h[s])
                if vb > 1.0:
                    vb = 1.0
                    delta = 1.0 - float(h[u])
                ht[s] = float(h[s]) - delta
                ht[u] = float(h[u]) + delta
                ht = np.clip(ht, 0.0, 1.0)
                ct = np.correlate(ht, 1.0 - ht, mode="full")
                Cv = float(np.max(ct)) * dx
                if Cv < trial_C:
                    trial_C = Cv
                    trial_h = ht
                    trial_corr = ct
                cand_count += 1

        if cand_count < max_trials:
            for sig in (0.02, 0.06):
                if cand_count >= max_trials:
                    break
                hp = project_box_sum(h + rng.randn(n) * float(sig), S)
                ct = np.correlate(hp, 1.0 - hp, mode="full")
                Cv = float(np.max(ct)) * dx
                if Cv < trial_C:
                    trial_C = Cv
                    trial_h = hp.copy()
                    trial_corr = ct
                cand_count += 1

        if trial_h is not None:
            if trial_C < cur_C - 1e-15:
                h = trial_h
                base_corr = trial_corr
                base_C = trial_C
                no_improve = 0
                if trial_C < best_C:
                    best_C = trial_C
                    best_h = h.copy()
            else:
                dval = trial_C - cur_C
                if dval <= 0.0:
                    prob = 1.0
                else:
                    x = dval / T
                    prob = math.exp(-x) if x < 50.0 else 0.0
                if rng.random_sample() < prob:
                    h = trial_h
                    base_corr = trial_corr
                    base_C = trial_C
                    no_improve += 1
                    if trial_C < best_C:
                        best_C = trial_C
                        best_h = h.copy()
                else:
                    no_improve += 1
        else:
            no_improve += 1
            if no_improve > 200:
                h = project_box_sum(h + rng.randn(n) * 0.05, S)
                base_corr = np.correlate(h, 1.0 - h, mode="full")
                base_C = float(np.max(base_corr)) * dx
                no_improve = 0

        if best_C < 0.5 and base_C > best_C + 4.0e-4:
            h = project_box_sum(best_h + rng.randn(n) * 0.02, S)
            base_corr = np.correlate(h, 1.0 - h, mode="full")
            base_C = float(np.max(base_corr)) * dx
            no_improve = 0

        if no_improve > 4000:
            scale = 0.05 + 0.10 * float(rng.random_sample())
            h = project_box_sum(best_h + rng.randn(n) * scale, S)
            base_corr = np.correlate(h, 1.0 - h, mode="full")
            base_C = float(np.max(base_corr)) * dx
            T = 1.0e-4
            no_improve = 0

    best_h = project_box_sum(best_h, S)
    best_C = evaluate_c5(best_h, n)
    return best_h, best_C


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)

    try:
        h_raw = np.asarray(initial_h_values, dtype=float).ravel()
        n = int(len(h_raw))
    except Exception:
        n = 190
        h_raw = np.full(n, 0.5, dtype=float)

    if n < 2:
        n = 2
        h_raw = np.array([0.5, 0.5], dtype=float)

    h0 = project_box_sum(h_raw, 0.5 * float(n))
    best_h = h0.copy()
    best_n = n
    best_c = evaluate_c5(best_h, best_n)

    target = 0.38085

    left = time_remaining(t0, budget_s)
    if left > 120:
        for beta in (5.0, 20.0, 80.0):
            if time_remaining(t0, budget_s) <= 260:
                break
            maxit = 260 if best_n <= 120 else 160
            h_new, c_new = slsqp_refine(best_h, best_n, beta, maxit)
            if c_new < best_c:
                best_h = h_new
                best_c = c_new

    left = time_remaining(t0, budget_s)
    if left > 350:
        dl = time.time() + min(left - 260.0, 750.0)
        if dl > time.time() + 20.0:
            h_loc, c_loc = fast_local_search(
                best_h,
                best_n,
                dl,
                int(seed),
                max_trials=(80 if best_n <= 120 else 50),
            )
            if c_loc < best_c:
                best_h = h_loc

    left = time_remaining(t0, budget_s)
    if left > 120:
        for beta in (200.0, 600.0):
            if time_remaining(t0, budget_s) <= 80:
                break
            maxit = 200 if best_n <= 120 else 100
            h_new, c_new = slsqp_refine(best_h, best_n, beta, maxit)

    left = time_remaining(t0, budget_s)

            if time_remaining(t0, budget_s) > 180:

                        dl,
                        int(seed) + 17,

                    )
                    if c_alt2 < c_alt:

            if time_remaining(t0, budget_s) > 80:
                h_alt, c_alt = slsqp_refine(h_alt, n_alt, 400.0, 120)
            if c_alt < best_c:
                best_h = h_alt
                best_c = c_alt
                best_n = n_alt

    left = time_remaining(t0, budget_s)
    if left > 120:

    except Exception:
        pass

    return best_h, best_c, int(best_n)
