    def objective(x):
        return compute_c5_smooth(x, n, temp) + penalty * (np.sum(x) - target) ** 2

    def grad_objective(x):
        return compute_grad_smooth(x, n, temp) + 2.0 * penalty * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    res = minimize(objective, x0, jac=grad_objective, bounds=bounds,
                   method='L-BFGS-B', options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9})
    return project_feasible(res.x, n), compute_c5(project_feasible(res.x, n), n)


def optimize_slsqp(h0, n, maxiter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return compute_c5(x, n)

    def grad(x):
        return compute_grad_hard(x, n)

    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    x0 = project_feasible(h0, n)
    res = minimize(obj, x0, jac=grad, bounds=bounds, constraints=cons,
                   method='SLSQP', options={'maxiter': maxiter, 'ftol': 1e-13})
    hx = project_feasible(res.x, n)
    return hx, compute_c5(hx, n)


def projected_momentum(h0, n, n_iter=300, lr=0.015, momentum=0.85, temp_init=0.05):
    h = project_feasible(h0, n)
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    temp = temp_init
    for it in range(n_iter):
        g = compute_grad_smooth(h, n, temp)
        v = momentum * v + (1.0 - momentum) * g
        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        temp *= 0.997
        lr *= 0.9995
    return best_h, best_val


def binary_local_search(h0, n, n_iter=5000, rng=np.random.RandomState(0)):
    h = np.array(h0, dtype=float).copy()
    h = (h >= 0.5).astype(float)
    for _ in range(100):
        s = np.sum(h)
        if s > n / 2:
            idx = rng.choice(np.where(h == 1)[0])
            h[idx] = 0
        elif s < n / 2:
            idx = rng.choice(np.where(h == 0)[0])
            h[idx] = 1
        else:
            break
    best_h = h.copy()
    best_val = compute_c5(h, n)
    t = 0.02
    for it in range(n_iter):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        h[i1] = 0
        h[i0] = 1
        val = compute_c5(h, n)
        delta = val - best_val
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-8)):
            best_h = h.copy()
            best_val = val
        else:
            h[i1] = 1
            h[i0] = 0
        t *= 0.9995
    return best_h, best_val


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is None or len(h0) == 0:
        n_base = 70
        h0 = np.full(n_base, 0.5)
    else:
        n_base = len(h0)

    best_h = project_feasible(h0, n_base)
    best_val = compute_c5(best_h, n_base)
    best_n = n_base

    # Phase 1: Smooth max L-BFGS at base resolution with temperature annealing
    if time_left() > 120:
        for temp in [0.15, 0.08, 0.04, 0.02, 0.01, 0.005]:
            if time_left() < 60:
                break
            try:
                h_try, v_try = optimize_lbfgs_smooth(best_h, best_n, temp, maxiter=400)
                if v_try < best_val:
                    best_val, best_h = v_try, h_try
            except Exception:
                pass

    # Phase 2: SLSQP polish at base resolution
    if time_left() > 100:
        for _ in range(3):
            if time_left() < 50:
                break
            try:
                h_try, v_try = optimize_slsqp(best_h, best_n, maxiter=800)
                if v_try < best_val:
                    best_val, best_h = v_try, h_try
            except Exception:
                pass

    # Phase 3: Projected momentum
    if time_left() > 80:
        try:
            h_try, v_try = projected_momentum(best_h, best_n, n_iter=500, lr=0.01, temp_init=0.05)
            if v_try < best_val:
                best_val, best_h = v_try, h_try
        except Exception:
            pass

    # Phase 4: Random restarts at base resolution
    attempt = 0
    while time_left() > 250 and attempt < 80:
        strategy = attempt % 6
        if strategy == 0:
            h_rand = rng.uniform(0, 1, best_n)
        elif strategy == 1:
            h_rand = (rng.rand(best_n) < 0.5).astype(float)
        elif strategy == 2:
            h_rand = best_h + rng.normal(0, 0.25, best_n)
        elif strategy == 3:
            block = max(2, best_n // 14)
            pattern = np.tile(np.concatenate([np.ones(block), np.zeros(block)]), best_n // (2 * block) + 1)
            h_rand = pattern[:best_n].astype(float)
            h_rand = h_rand + rng.normal(0, 0.05, best_n)
        elif strategy == 4:
            h_rand = best_h.copy()
            n_flip = best_n // 5
            idx = rng.choice(best_n, size=n_flip, replace=False)
            h_rand[idx] = 1.0 - h_rand[idx]
        else:
            # Symmetric random
            half = best_n // 2
            h_half = rng.uniform(0, 1, half)
            h_rand = np.concatenate([h_half, h_half[::-1]])
            if best_n % 2 == 1:
                h_rand = h_rand[:best_n]

        h_rand = project_feasible(h_rand, best_n)
        attempt += 1

        try:
            h_try, v_try = optimize_lbfgs_smooth(h_rand, best_n, 0.03, maxiter=200)
            if v_try < best_val:
                best_val, best_h = v_try, h_try
                # Quick SLSQP follow-up
                if time_left() > 50:
                    try:
                        h_try2, v_try2 = optimize_slsqp(h_try, best_n, maxiter=200)
                        if v_try2 < best_val:
                            best_val, best_h = v_try2, h_try2
                    except Exception:
                        pass
        except Exception:
            pass

    # Phase 5: Binary local search
    if time_left() > 100:
        try:
            h_bin, v_bin = binary_local_search(best_h, best_n, n_iter=8000, rng=rng)
            if v_bin < best_val:
                best_val, best_h = v_bin, h_bin
            # Optimize the binary solution with continuous methods
            h_try, v_try = optimize_lbfgs_smooth(h_bin, best_n, 0.02, maxiter=300)
            if v_try < best_val:
                best_val, best_h = v_try, h_try
        except Exception:
            pass

    # Phase 6: Multi-resolution refinement
    for n_fine in [100, 140, 200, 280]:
        if time_left() < 120:
            break
        x_coarse = np.linspace(0, 2, best_n, endpoint=False)
        x_fine = np.linspace(0, 2, n_fine, endpoint=False)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)

        for temp in [0.05, 0.02, 0.01]:
            if time_left() < 60:
                break
            try:
                h_try, v_try = optimize_lbfgs_smooth(h_fine, n_fine, temp, maxiter=300)
                h_fine = h_try
                if v_try < best_val:
                    best_val, best_h, best_n = v_try, h_try, n_fine
            except Exception:
                pass

    # Phase 7: Perturbation + re-optimization around best
    for i in range(20):
        if time_left() < 50:
            break
        for scale in [0.02, 0.04, 0.08, 0.15]:
            if time_left() < 30:
                break
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = project_feasible(h_pert, best_n)
            try:
                h_try, v_try = optimize_lbfgs_smooth(h_pert, best_n, 0.015, maxiter=250)
                if v_try < best_val:
                    best_val, best_h = v_try, h_try
            except Exception:
                pass

    # Phase 8: Final SLSQP polish (multiple passes)
    for _ in range(8):
        if time_left() < 15:
            break
        try:
            h_try, v_try = optimize_slsqp(best_h, best_n, maxiter=1000)
            if v_try < best_val:
                best_val, best_h = v_try, h_try
            else:
                break
        except Exception:
            break

    # Phase 9: Final momentum pass
    if time_left() > 25:
        try:
            h_try, v_try = projected_momentum(best_h, best_n, n_iter=300, lr=0.005, momentum=0.9, temp_init=0.01)
            if v_try < best_val:
                best_val, best_h = v_try, h_try
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    s = np.sum(best_h)
    if abs(s - best_n / 2.0) > 1e-8:
        best_h += (best_n / 2.0 - s) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
