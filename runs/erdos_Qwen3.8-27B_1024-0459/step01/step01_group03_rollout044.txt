```python
        g[k:] += (1.0 - h[:n-k])  # (1-h[j-k]) for j=k..n-1
        g[:n-k] -= h[k:]  # -h[j+k] for j=0..n-1-k
    else:
        m = -k
        # corr = sum_{i=m}^{n-1} h[i-m]*(1-h[i])
        # d/dh[j]:
        #   if 0 <= j <= n-1-m: (1-h[j+m]) [from h[i-m] term]
        #   if m <= j <= n-1: -h[j-m] [from (1-h[i]) term]
        g[:n-m] += (1.0 - h[m:])  # (1-h[j+m]) for j=0..n-1-m
        g[m:] -= h[:n-m]  # -h[j-m] for j=m..n-1
    return g
```

```python
import time
import numpy as np
from scipy.optimize import minimize

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x.copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-9) & (h < 1.0 - 1e-9))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        m = -k
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def top_k_gradient(h, corr, K):
    n = h.size
    center = n - 1
    idxs = np.argsort(corr)[-K:]
    g = np.zeros(n, dtype=float)
    for t in idxs:
        g += lag_gradient(h, int(t) - center)
    return g / float(K), idxs

def subgradient_polish(h, target, deadline, K=8, max_iter=2000):
    n = h.size
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    vel = np.zeros(n)
    mu = 0.7
    stall = 0
    base_step = 0.05
    shrunk = 0

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        g, _ = top_k_gradient(best, corr, K)
        g -= float(np.mean(g))
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        d = -g / gn
        vel = mu * vel + 0.3 * d

        improved = False
        for step in (0.02, 0.05, 0.1, 0.2, 0.4, 0.8):
            cand = project_density(best + step * vel, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                best = cand
                best_c = c
                improved = True
                break
        if not improved:
            for step in (0.01, 0.03, 0.07, 0.15):
                cand = project_density(best + step * d, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-12:
                    best = cand
                    best_c = c
                    improved = True
                    break
        if not improved:
            shrunk += 1
            if shrunk > 8:
                mu *= 0.9
            if shrunk > 30:
                break
        else:
            shrunk = 0
            stall = 0
        if it % 50 == 49:
            cand = project_density(best + 0.01 * np.random.randn(n) * 0.1, target)
            c = evaluate_erdos(cand)
            if c < best_c:
                best = cand
                best_c = c
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c

def pair_optimize(h, target, deadline, K=8, max_iter=300):
    n = h.size
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    stall = 0

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        g, _ = top_k_gradient(best, corr, K)
        if (float(np.max(g)) - float(np.min(g))) < 1e-13:
            break
        neg = np.argsort(g)[:5]
        pos = np.argsort(-g)[:5]
        found = False
        for ii in neg:
            ii = int(ii)
            for jj in pos:
                jj = int(jj)
                if ii == jj:
                    continue
                cap_up = 1.0 - best[ii]
                cap_dn = best[jj]
                cap = min(cap_up, cap_dn)
                if cap <= 1e-12:
                    continue
                for frac in (0.25, 0.5, 0.75, 1.0):
                    d = frac * cap
                    if d <= 1e-12:
                        continue
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            d = -g
            d -= float(np.mean(d))
            nn = float(np.linalg.norm(d))
            if nn > 1e-12:
                d /= nn
                for step in (0.03, 0.1, 0.3):
                    cand = project_density(best + step * d, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
        if not found:
            stall += 1
            if stall >= 20:
                break
        else:
            stall = 0
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c

def binary_try(h, target, deadline):
    n = h.size
    if n % 2 != 0:
        return project_density(h, target), evaluate_erdos(project_density(h, target))
    nones = target
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:int(nones)]] = 1.0
    h_bin = project_density(h_bin, target)
    c_bin = evaluate_erdos(h_bin)
    best = h_bin.copy()
    best_c = c_bin
    for _ in range(50):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        g, _ = top_k_gradient(best, corr, 6)
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        g_ones = g[ones]
        g_zeros = g[zeros]
        i_one = ones[np.argmax(-g_ones)]
        j_zero = zeros[np.argmax(g_zeros)]
        if g[i_one] >= g[j_zero]:
            break
        cand = best.copy()
        cand[i_one] = 0.0
        cand[j_zero] = 1.0
        c = evaluate_erdos(cand)
        if c < best_c - 1e-13:
            best = cand
            best_c = c
    return best, best_c

def smooth_objective(x, n, alpha, target):
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    dx = 2.0 / float(n)
    return dx * (m + np.log(z) / alpha)

def smooth_jacobian(x, n, alpha, target):
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    g = np.zeros(n)
    center = n - 1
    threshold = 1e-10
    for t in range(corr.size):
        p = w[t]
        if p < threshold:
            continue
        k = t - center
        lg = lag_gradient(h, k)
        g += p * lg
    dx = 2.0 / float(n)
    return dx * g

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def sum_con_fun(x, target):
    return float(np.sum(x)) - target

def sum_con_jac(x, target):
    return np.ones_like(x)

def slsqp_optimize(h0, n, target, deadline):
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    now = time.perf_counter()
    remaining = deadline - now
    if remaining < 5.0:
        return best_h, best_c

    alphas = []
    maxiters = []
    if n <= 200:
        alphas = [40.0, 100.0, 300.0]
        maxiters = [80, 60, 40]
    elif n <= 500:
        alphas = [50.0, 150.0]
        maxiters = [50, 35]
    else:
        alphas = [60.0]
        maxiters = [30]

    bounds = [(0.0, 1.0)] * n

    for alpha, mi in zip(alphas, maxiters):
        if time.perf_counter() >= deadline - 3.0:
            break
        cons = ({'type': 'eq',
                 'fun': lambda x: sum_con_fun(x, target),
                 'jac': lambda x: sum_con_jac(x, target)},)
        try:
            res = minimize(
                lambda x: smooth_objective(x, n, alpha, target),
                h,
                method='SLSQP',
                jac=lambda x: smooth_jacobian(x, n, alpha, target),
                bounds=bounds,
                constraints=cons,
                options={'maxiter': mi, 'ftol': 1e-10, 'disp': False}
            )
            xg = getattr(res, 'x', None)
            if xg is not None:
                xg = np.asarray(xg, dtype=float)
                if np.all(np.isfinite(xg)):
                    cand = project_density(xg, target)
                    c = evaluate_erdos(cand)
                    if c < best_c:
                        best_h = cand
                        best_c = c
                    h = cand
        except Exception:
            pass
    return best_h, best_c

def generate_random(n, rng, mode):
    if mode == 0:
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
        h = np.full(n, 0.5)
        for m in range(1, 8):
            amp = rng.uniform(0.05, 0.3) * (0.6 ** (m - 1))
            freq = int(max(1, np.rint(m * rng.uniform(0.5, 2.0))))
            phase = rng.uniform(0.0, 2.0 * np.pi)
            h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    elif mode == 1:
        h = np.random.choice([0.0, 1.0], size=n, p=[0.5, 0.5])
    else:
        h = rng.random(n)
        h = (h > 0.5).astype(float)
    return project_density(h, 0.5 * n)
```
