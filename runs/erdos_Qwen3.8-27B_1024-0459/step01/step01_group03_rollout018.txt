            resolutions.append(nn)
    for nn in [60, 90, 120, 180, 240, 360]:
        if nn not in resolutions:
            resolutions.append(nn)
    resolutions = sorted(set(resolutions))

    for ni in resolutions:
        if deadline - time.perf_counter() < 20.0:
            break

        starts = []
        if best_h is not None:
            starts.append(upsample_h(best_h, ni))
        if parent is not None:
            starts.append(upsample_h(parent, ni))

        # Add a random smooth start
        x = (np.arange(ni) + 0.5) * (2.0 / ni)
        h_rand = 0.5
        for m in range(1, 5):
            amp = rng.uniform(0.05, 0.25) * (0.7 ** (m - 1))
            freq = int(max(1, m * rng.uniform(0.5, 2.0)))
            phase = rng.uniform(0, 2 * np.pi)
            h_rand = h_rand + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
        starts.append(project_density(h_rand, 0.5 * ni))

        seen = []
        uniq_starts = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in seen:
                if np.allclose(s, u, atol=1e-8):
                    dup = True
                    break
            if not dup:
                seen.append(s.copy())
                uniq_starts.append(s)

        rem_total = deadline - time.perf_counter()
        per_start = max(8.0, rem_total / (len(uniq_starts) + 1))

        for s in uniq_starts:
            rem = deadline - time.perf_counter()
            if rem < 10.0:
                break

            if ni <= 100:
                time_solve = min(per_start * 0.4, rem * 0.35)
                time_cd = min(per_start * 0.5, rem * 0.5)
                time_fine = rem * 0.15
            elif ni <= 200:
                time_solve = min(per_start * 0.3, rem * 0.25)
                time_cd = min(per_start * 0.55, rem * 0.55)
                time_fine = rem * 0.2
            else:
                time_solve = min(per_start * 0.3, rem * 0.2)
                time_cd = min(per_start * 0.6, rem * 0.6)
                time_fine = rem * 0.15

            hcur = s.copy()
            ccur = evaluate_erdos(hcur)

            # Phase 1: Smooth SLSQP
            if time_solve > 3.0:
                sol_deadline = time.perf_counter() + time_solve
                h_s, c_s = smooth_optimize(hcur, ni, sol_deadline)
                if c_s < ccur:
                    hcur = h_s
                    ccur = c_s
                if ccur < best_c:
                    best_h = hcur.copy()
                    best_c = ccur
                    best_n = ni

            # Phase 2: Coordinate descent (true objective)
            rem2 = deadline - time.perf_counter()
            if rem2 > 8.0:
                cd_deadline = time.perf_counter() + min(time_cd, rem2 * 0.7)
                h_cd, c_cd = fast_coord_descend(hcur, ni, cd_deadline, rng,
                                                max_iter=8000, top_k=4)
                if c_cd < ccur:
                    hcur = h_cd
                    ccur = c_cd
                if ccur < best_c:
                    best_h = hcur.copy()
                    best_c = ccur
                    best_n = ni

            # Phase 3: Fine polish
            rem3 = deadline - time.perf_counter()
            if rem3 > 5.0:
                fp_deadline = time.perf_counter() + min(time_fine, rem3 * 0.8)
                h_fp, c_fp = fine_polish(hcur, ni, fp_deadline, rng, top_k=5)
                if c_fp < ccur:
                    hcur = h_fp
                    ccur = c_fp
                if ccur < best_c:
                    best_h = hcur.copy()
                    best_c = ccur
                    best_n = ni

    # Final concentrated effort on best resolution
    rem = deadline - time.perf_counter()
    if rem > 10.0 and best_h is not None:
        nb = int(best_h.size)
        h_final = project_density(best_h, 0.5 * nb)
        c_final = evaluate_erdos(h_final)

        # More coordinate descent
        if rem > 15.0:
            cd_deadline2 = time.perf_counter() + rem * 0.4
            h2, c2 = fast_coord_descend(h_final, nb, cd_deadline2, rng,
                                        max_iter=5000, top_k=5)
            if c2 < c_final:
                h_final = h2
                c_final = c2

        # Fine polish
        rem2 = deadline - time.perf_counter()
        if rem2 > 5.0:
            fp_deadline2 = time.perf_counter() + rem2 * 0.8
            h3, c3 = fine_polish(h_final, nb, fp_deadline2, rng, top_k=6)
            if c3 < c_final:
                h_final = h3
                c_final = c3

        if c_final < best_c:
            best_h = h_final
            best_c = c_final
            best_n = nb

    if best_h is None:
        fallback = project_density(np.full(90, 0.5, dtype=float), 45.0)
        return fallback, evaluate_erdos(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
