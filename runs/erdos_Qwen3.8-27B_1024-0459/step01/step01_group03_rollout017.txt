        improved = False
        neg = np.argsort(g)[:5]
        pos = np.argsort(-g)[:5]
        for ii in neg:
            for jj in pos:
                ii, jj = int(ii), int(jj)
                if ii == jj:
                    continue
                delta = min(h[ii], 1.0 - h[jj])
                if delta < 1e-12:
                    continue
                for frac in (0.3, 0.6, 1.0):
                    d = frac * delta
                    cand = h.copy()
                    cand[ii] -= d
                    cand[jj] += d
                    cand = _project_density(cand, target)
                    c = _erdos_c(cand)
                    if c < best_c - 1e-12:
                        h = cand
                        best_c = c
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 25:
                break
        else:
            stall = 0
    return _project_density(h, target), _erdos_c(h)


def _binary_local_search(h0, deadline, max_iter=2000):
    h = _project_density(np.asarray(h0, dtype=float), 0.5 * h0.size)
    n = h.size
    target = 0.5 * n
    best_c = _erdos_c(h)
    stall = 0
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

    for it in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[np.random.randint(len(ones))])
        j = int(zeros[np.random.randint(len(zeros))])
        if i == j:
            continue

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = 5
        idxs = np.argsort(corr)[-top_k:]
        g = _corr_subgrad(h, idxs)
        gain = g[i] - g[j]
        if gain >= 0:
            continue

        delta = min(h[i], 1.0 - h[j])
        if delta < 1e-12:
            continue

        cand = h.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_density(cand, target)
        c = _erdos_c(cand)
        if c < best_c - 1e-12:
            h = cand
            best_c = c
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            stall = 0
        else:
            stall += 1
            if it % 50 == 0:
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            if stall > 200:
                break

    return _project_density(h, target), _erdos_c(h)


def _make_random_binary(n, rng):
    h = np.zeros(n)
    idxs = rng.choice(n, size=n // 2, replace=False)
    h[idxs] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = t0 + budget - 8.0

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = parent.size
        proj = _project_density(parent, 0.5 * np_)
        c = _erdos_c(proj)
        if c < best_c:
            best_h, best_c, best_n = proj.copy(), c, np_

    if best_h is None:
        n_fallback = 90
        best_h = _make_random_binary(n_fallback, rng)
        best_h = _project_density(best_h, 0.5 * n_fallback)
        best_c = _erdos_c(best_h)
        best_n = n_fallback

    base_n = best_n

    # Determine resolutions to try
    resolutions = []
    for mult in (2, 3, 4):
        n = base_n * mult
        if 60 <= n <= 500:
            resolutions.append(n)
    if 180 not in resolutions:
        resolutions.append(180)
    if 240 not in resolutions and 240 not in [r for r in resolutions]:
        resolutions.append(240)
    resolutions = sorted(set(resolutions))
    # Cap at reasonable
    resolutions = [n for n in resolutions if 60 <= n <= 450]
    resolutions = sorted(set(resolutions))

    for n in resolutions:
        if time.perf_counter() >= deadline - 30:
            break

        # Build initial guess for this resolution
        if best_h is not None and best_h.size == n:
            h_start = best_h.copy()
        elif best_h is not None:
            h_start = _upsample(best_h, n)
        elif parent is not None:
            h_start = _upsample(parent, n)
        else:
            h_start = _make_random_binary(n, rng)
        h_start = _project_density(h_start, 0.5 * n)

        # Also try a random binary start
        h_rand = _make_random_binary(n, rng)

        # SLSQP optimization on both starts
        for h0 in [h_start, h_rand]:
            if time.perf_counter() >= deadline - 25:
                break

            if n <= 180:
                alphas = [(25.0, 150), (60.0, 100)]
            elif n <= 300:
                alphas = [(25.0, 100), (60.0, 80)]
            else:
                alphas = [(20.0, 80)]

            for alpha, miter in alphas:
                if time.perf_counter() >= deadline - 20:
                    break
                h_opt = _slsqp_optimize(h0, n, alpha, miter, deadline)
                c_opt = _erdos_c(h_opt)
                if c_opt < best_c:
                    best_h, best_c, best_n = h_opt.copy(), c_opt, n
                # Use the better of h0 and h_opt as new h0
                if c_opt < _erdos_c(h0):
                    h0 = h_opt
                else:
                    h0 = _project_density(h0, 0.5 * n)

        # Coordinate descent polish on best for this resolution
        if time.perf_counter() < deadline - 15:
            if best_n == n:
                h_pol, c_pol = _coordinate_polish(best_h, deadline, max_iter=500, k_active=5)
                if c_pol < best_c:
                    best_h, best_c = h_pol, c_pol
            else:
                h_up = _upsample(best_h, n)
                h_pol, c_pol = _coordinate_polish(h_up, deadline, max_iter=300, k_active=5)
                if c_pol < best_c:
                    best_h, best_c, best_n = h_pol, c_pol, n

    # Final polish phase
    if best_h is not None and time.perf_counter() < deadline - 5:
        h_final = _project_density(best_h, 0.5 * best_h.size)
        c_final = _erdos_c(h_final)

        # Coordinate polish
        h_p, c_p = _coordinate_polish(h_final, deadline, max_iter=800, k_active=6)
        if c_p < c_final:
            h_final, c_final = h_p, c_p

        # Pair polish
        if time.perf_counter() < deadline - 3:
            h_p2, c_p2 = _pair_polish(h_final, deadline, max_iter=400, k_active=5)
            if c_p2 < c_final:
                h_final, c_final = h_p2, c_p2

        # Binary local search (only if values are near-binary)
        if time.perf_counter() < deadline - 2:
            frac_binary = np.sum((h_final < 0.05) | (h_final > 0.95)) / h_final.size
            if frac_binary > 0.85:
                h_p3, c_p3 = _binary_local_search(h_final, deadline, max_iter=1500)
                if c_p3 < c_final:
                    h_final, c_final = h_p3, c_p3

        best_h, best_c, best_n = h_final, c_final, int(h_final.size)

    # Third polish pass if time remains
    if best_h is not None and time.perf_counter() < deadline - 3:
        h3, c3 = _coordinate_polish(best_h, time.perf_counter() + 5, max_iter=200, k_active=8)
        if c3 < best_c:
            best_h, best_c, best_n = h3, c3, int(h3.size)

    if best_h is None:
        fallback = _project_density(np.full(90, 0.5), 45.0)
        return fallback, _erdos_c(fallback), 90

    final_h = _project_density(best_h, 0.5 * int(best_h.size))
    final_c = _erdos_c(final_h)
    return final_h, float(final_c), int(final_h.size)
