    c = np.correlate(h, 1.0 - h, mode="full")

    return np.correlate(h, 1.0 - h, mode="full")

    c = get_correlations(h, n)
    dx = 2.0 / n

    m = np.max(scaled)
    return float(m + np.log(np.sum(np.exp(scaled - m)))) / beta

def smoothed_grad(h, n, beta):
    c = get_correlations(h, n)
    dx = 2.0 / n
    scaled = beta * c * dx
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    w = w / w_sum  # softmax weights
    
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        if w[k] < 1e-18:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # c[k] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            overlap = n - lag
            # ∂c/∂h[j] for j in [lag, n-1]: (1-h[j-lag])
            # ∂c/∂h[j] for j in [0, overlap-1]: -h[j+lag]
            # Overlap region j in [lag, overlap-1]: (1-h[j-lag]) - h[j+lag]
            grad[lag:n] += w[k] * (1.0 - h[:overlap])  # first part
            if overlap > 0:
                grad[:overlap] -= w[k] * h[lag:lag+overlap]  # second part
        else:
            m_lag = -lag
            overlap = n - m_lag
            # c[k] = sum_{i=0}^{overlap-1} h[i]*(1-h[i+m_lag])
            grad[:overlap] += w[k] * (1.0 - h[m_lag:m_lag+overlap])

            break

Now let me think about the most efficient implementation. For the smoothed gradient:

```python
def smoothed_grad(h, n, beta):
    c = np.correlate(h, 1.0-h, mode='full')
    dx = 2.0 / n
    scaled = beta * c * dx
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    w = w / w_sum
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if w[k] < 1e-15:
            continue
        wk = w[k]
        lag = k - (n-1)
        if lag >= 0:
            ov = n - lag
            # h[j] for j in [lag, n-1]: contributes wk*(1-h[j-lag])
            grad[lag:] += wk * (1.0 - h[:ov])
            # h[j] for j in [0, ov-1]: contributes -wk*h[j+lag]
            grad[:ov] -= wk * h[lag:lag+ov]
        else:
            m_lag = -lag
            ov = n - m_lag
            grad[:ov] += wk * (1.0 - h[m_lag:m_lag+ov])
            grad[m_lag:] -= wk * h[:ov]
    
    return grad * dx
```

The main cost is np.correlate which is also O(n²) (or O(n log n) with FFT). For n=200, this is fine.

The problem is: min max_k c_k, subject to 0≤h≤1, sum h = n/2.

Each c_k is a quadratic in h. Specifically, c_k = b_k^T h - h^T A_k h where A_k is a symmetric matrix with 1s on the k-th sub/super diagonal.
