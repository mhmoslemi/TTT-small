import numpy as np
from scipy.optimize import minimize, NonlinearConstraint
import time

_N = 100

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    mask = (h > 1e-9) & (h < 1.0 - 1e-9)
    if np.sum(mask) > 0:
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def constraint_sum(x):
    return np.sum(x) - _N / 2.0

def constraint_jac(x):
    return np.ones(_N)

def make_constraint():
    return NonlinearConstraint(constraint_sum, _N / 2.0, _N / 2.0, jac=constraint_jac)

def obj_smooth(x, beta):
    corr = get_corr(x)
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0 / beta, 0.0)
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def coordinate_descent(h, n, max_rounds=30):
    h = h.copy()
    current_c5 = compute_c5(h)
    step_sizes = [0.01, 0.03, 0.05, 0.1]
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            saved_h = h.copy()
            orig = h[i]
            best_c5_i = current_c5
            best_h_i = saved_h.copy()
            for delta in step_sizes:
                for sign in (1, -1):
                    new_val = orig + sign * delta
                    if new_val < -1e-6 or new_val > 1.0 + 1e-6:
                        continue
                    h[:] = saved_h
                    h[i] = new_val
                    h = project_feasible(h, n)
                    c5_t = compute_c5(h)
                    if c5_t < best_c5_i - 1e-14:
                        best_c5_i = c5_t
                        best_h_i = h.copy()
            if best_c5_i < current_c5 - 1e-14:
                h = best_h_i
                current_c5 = best_c5_i
                improved = True
        if not improved:
            break
    return h, current_c5

def slsqp_optimize(h0, n, maxiter=300):
    global _N
    _N = n
    bounds = [(0.0, 1.0)] * n
    cons = [make_constraint()]
    h_cur = h0.copy()
    for beta in (8.0, 25.0, 80.0, 200.0):
        try:
            res = minimize(obj_smooth, h_cur, args=(beta,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-13, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def perturb_reoptimize(h, n, rng, n_perturb=3):
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    for _ in range(n_perturb):
        h_pert = h_best.copy()
        magnitude = 0.05 + 0.1 * rng.rand()
        h_pert += rng.randn(n) * magnitude
        h_pert = project_feasible(h_pert, n)
        h_opt = slsqp_optimize(h_pert, n, maxiter=200)
        h_cd, c5_cd = coordinate_descent(h_opt, n, max_rounds=15)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd
    return h_best, c5_best
