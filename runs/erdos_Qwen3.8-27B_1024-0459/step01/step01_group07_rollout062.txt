```python
def project_box_hyperplane(v, target_sum):
    """Project v onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(v)
    # Water-filling: find tau such that sum(min(max(v[i]-tau, 0), 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(v - tau, 0, 1)
```

```python
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def project_h(v, n):
    """Project v onto {x ∈ [0,1]^n : sum(x) = n/2}"""
    target = n / 2.0
    # Water-filling
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, 0.0, 1.0)

def smooth_obj(h, n, T):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    if T < 1e-10:
        return np.max(conv)
    cmax = np.max(conv)
    exp_c = np.exp((conv - cmax) / T)
    return T * (np.log(np.sum(exp_c)) + cmax / T)

def smooth_grad(h, n, T):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(conv)
    exp_c = np.exp((conv - cmax) / T)
    p = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        k = m + (n - 1)
        pk = p[k]
        if pk < 1e-14:
            continue
        if m >= 0:
            n1 = n - m
            if n1 > 0:
                grad[:n1] += pk * (1.0 - h[m:m+n1]) * dx
                grad[m:m+n1] += pk * (-h[:n1]) * dx
        else:
            pp = -m
            n1 = n - pp
            if n1 > 0:
                grad[pp:pp+n1] += pk * (1.0 - h[:n1]) * dx
                grad[:n1] += pk * (-h[pp:pp+n1]) * dx
    return grad

def pgd_optimize(h0, n, T_schedule, iters_per_T=500, time_limit=60):
    h = h0.copy()
    best_h = h.copy()
    best_f = compute_c5(h, n)
    t_start = time.time()
    prev_h = None
    prev_g = None
    
    for T in T_schedule:
        for it in range(iters_per_T):
            if time.time() - t_start > time_limit:
                break
            g = smooth_grad(h, n, T)
            
            # BB step size
            if prev_h is not None and prev_g is not None:
                s = h - prev_h
                ys = g - prev_g
                sys = np.dot(s, ys)
                nys = np.dot(ys, ys)
                if sys > 1e-12 and nys > 1e-12:
                    alpha = sys / nys
                else:
                    alpha = 1.0 / max(np.linalg.norm(g), 1e-12)
            else:
                alpha = 1.0 / max(np.linalg.norm(g), 1e-12)
            
            # Clamp alpha
            alpha = np.clip(alpha, 1e-8, 1.0)
            
            # Backtracking
            f_cur = smooth_obj(h, n, T)
            for bt in range(20):
                h_new = project_h(h - alpha * g, n)
                f_new = smooth_obj(h_new, n, T)
                if f_new < f_cur + 1e-4 * alpha * np.dot(g, h_new - h):
                    break
                alpha *= 0.5
            
            prev_h, prev_g = h.copy(), g.copy()
            h = h_new
            
            # Track best (using true C5, not smoothed)
            true_f = compute_c5(h, n)
            if true_f < best_f:
                best_f = true_f
                best_h = h.copy()
        
        if time.time() - t_start > time_limit:
            break
    
    return best_h, best_f
```

Actually, c_k = sum_j h[j](1-h[j+k]) * dx. The terms involving h[i] are:

- For m ≥ 0: h[i] appears as h[i]*(1-h[i+m]) in c_m when i ≤ n-1-m
  → contribution to c_m: h[i]*(1-h[i+m])*dx
  → also h[i] appears in h[i-m]*(1-h[i]) when i-m ≥ 0, i.e., i ≥ m
  → contribution to c_m: -h[i-m]*h[i]*dx
