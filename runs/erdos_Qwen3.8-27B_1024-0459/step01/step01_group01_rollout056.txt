    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)

def eq_fun(x):
    return np.array([x.sum() - len(x) / 2.0])

def eq_jac(x):
    return np.ones((1, len(x)))

def slsqp_optimize(h0, n, tau, maxiter, ftol):
    dx = 2.0 / n
    bounds = [(0, 1)] * n
    res = minimize(
        lambda x: smooth_obj(x, n, dx, tau),
        h0, method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}

    one_idx = set(np.where(b == 1.0)[0])
    zero_idx = set(np.where(b == 0.0)[0])
    
    for p in range(max_passes):
        improved = False
