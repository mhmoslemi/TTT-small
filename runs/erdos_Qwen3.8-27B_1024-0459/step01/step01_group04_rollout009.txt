        # Also upsample current best
        if state.h is not None:
            h_up = project_box_sum(upsample(state.h, n), n)
            c_up = compute_c5(h_up)
            cands.append((h_up, c_up))
        # Sort by quality
        cands.sort(key=lambda x: x[1])
        # Take top 2-3
        top_cands = cands[:min(3, len(cands))]

        # PGD on best candidate
        if t_pgd > 10.0 and len(top_cands) > 0:
            for ci, (hc, cc) in enumerate(top_cands[:2]):
                if time.time() >= _DEADLINE - 30.0:
                    break
                t_alloc = t_pgd / 2.0
                try:
                    pgd_optimize(hc, n, state, t_alloc, rng)
                except BudgetError:
                    pass
                except Exception:
                    pass

        # Coordinate descent from best
        if t_cd > 8.0:
            try:
                coordinate_descent(state.h, n, state, t_cd, rng)
            except BudgetError:
                pass
            except Exception:
                pass

        # Binary local search
        if t_bin > 8.0 and n % 2 == 0:
            try:
                binary_local_search(state.h, n, state, t_bin, rng)
            except BudgetError:
                pass
            except Exception:
                pass

    # Final: if we have a good solution, do one more round of refinement
    if time.time() < _DEADLINE - 30.0 and state.h is not None:
        n_best = state.n
        t_rem = min(60.0, _DEADLINE - time.time() - 20.0)
        if t_rem > 15.0:
            try:
                pgd_optimize(state.h, n_best, state, t_rem * 0.6, rng)
            except BudgetError:
                pass
            except Exception:
                pass
            if time.time() < _DEADLINE - 20.0:
                try:
                    coordinate_descent(state.h, n_best, state, t_rem * 0.4, rng)
                except BudgetError:
                    pass
                except Exception:
                    pass

    # Final answer
    if state.h is not None and state.c < 1e5:
        h_final = project_box_sum(state.h, state.n)
        c_final = compute_c5(h_final)
        n_final = state.n
    else:
        h_final = project_box_sum(initial_h, initial_n)
        c_final = compute_c5(h_final)
        n_final = initial_n

    # Safety: keep best of initial vs final
    if initial_c < c_final:
        h_final = initial_h
        c_final = initial_c
        n_final = initial_n

    # Final validation
    h_final = np.clip(h_final, 0.0, 1.0)
    c_final = compute_c5(h_final)
    n_final = int(h_final.size)

    return h_final, float(c_final), n_final
