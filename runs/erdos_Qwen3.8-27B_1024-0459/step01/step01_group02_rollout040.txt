```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_max_corr(h):
    """Returns the max correlation value (before multiplying by dx)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def binary_local_search(h, max_iters=1000, time_limit=30):
    """Local search on binary pattern using pair flips"""
    n = len(h)
    start = time.time()
    current_max = int(compute_max_corr(h))
    best_h = h.copy()
    best_max = current_max
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Try all pairs (a from ones, b from zeros)
        best_delta = 0
        best_pair = None
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # For efficiency, only try a subset if n is large
        # For n=128, n_ones=n_zeros=64, pairs = 4096, each eval is O(n) = 128
        # Total: ~500K ops per iteration, should be fast in numpy
        
        for ia in range(n_ones):
            a = ones[ia]
            for ib in range(n_zeros):
                b = zeros[ib]
                # Flip a: 1->0, b: 0->1
                h[a] = 0
                h[b] = 1
                new_max = int(compute_max_corr(h))
                h[a] = 1
                h[b] = 0
                delta = new_max - current_max
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (a, b)
        
        if best_pair is None:
            break  # Local minimum
        
        a, b = best_pair
        h[a] = 0
        h[b] = 1
        current_max += best_delta
        
        # Update ones and zeros
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if current_max < best_max:
            best_max = current_max
            best_h = h.copy()
    
    return best_h, best_max
```

  Valid when 0 ≤ b+k-(N-1) ≤ N-1, i.e., N-1-b ≤ k ≤ N-1+b.
  
- In the factor (1-h[n]): changes when n = b
  Before: (1-h[b]) = 1. After: (1-h[b]) = 0.
  Change: h[b+k-(N-1)] * (0-1) = -h[b+k-(N-1)]
  Valid when 0 ≤ b+k-(N-1) ≤ N-1, i.e., k ∈ [N-1-b, 2N-1-b].

So the total change in corr[k] for a flip (a: 1→0, b: 0→1):

Δcorr[k] = [from h[a] change in h factor] + [from h[a] change in (1-h) factor] + [from h[b] change in h factor] + [from h[b] change in (1-h) factor]

= -h[a+k-(N-1)] * I[N-1-a ≤ k ≤ N-1+a]
  + h[a+k-(N-1)] * I[N-1-a ≤ k ≤ 2N-1-a]
  + (1-h[b+k-(N-1)]) * I[N-1-b ≤ k ≤ N-1+b]
  - h[b+k-(N-1)] * I[N-1-b ≤ k ≤ 2N-1-b]

where I[condition] is 1 if condition holds, 0 otherwise. And h[.] = 0 for out-of-range indices.

This is O(1) to compute for each k, and O(N) total for all k. So we can compute the effect of a flip in O(N) time.

With N² pairs, one full pass is O(N³). For N=128, that's 2M operations. In pure Python, this would be slow, but we can vectorize using numpy.

```python
def compute_flip_effects(h, a, b, N):
    """Compute Δcorr for flipping h[a]: 1→0 and h[b]: 0→1"""
    k = np.arange(2*N - 1)
    
    # From h[a] change (1→0):
    # h factor: -h[a+k-(N-1)] for k in [N-1-a, N-1+a]
    idx_a = a + k - (N-1)
    term1 = -h * 0  # will fill in
    
    mask1 = (k >= N-1-a) & (k <= N-1+a)
    valid1 = (idx_a >= 0) & (idx_a < N)
    term1 = np.zeros(2*N-1)
    term1[mask1 & valid1] = -h[idx_a[mask1 & valid1]]
    
    # (1-h) factor: +h[a+k-(N-1)] for k in [N-1-a, 2N-1-a]
    mask2 = (k >= N-1-a) & (k <= 2*N-1-a)
    valid2 = (idx_a >= 0) & (idx_a < N)
    term2 = np.zeros(2*N-1)
    term2[mask2 & valid2] = h[idx_a[mask2 & valid2]]
    
    # From h[b] change (0→1):
    idx_b = b + k - (N-1)
    mask3 = (k >= N-1-b) & (k <= N-1+b)
    valid3 = (idx_b >= 0) & (idx_b < N)
    term3 = np.zeros(2*N-1)
    term3[mask3 & valid3] = 1 - h[idx_b[mask3 & valid3]]
    
    mask4 = (k >= N-1-b) & (k <= 2*N-1-b)
    valid4 = (idx_b >= 0) & (idx_b < N)
    term4 = np.zeros(2*N-1)
    term4[mask4 & valid4] = -h[idx_b[mask4 & valid4]]
    
    return term1 + term2 + term3 + term4
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def project_onto_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    n_mask = np.sum(mask)
    if n_mask > 0:
        h[mask] += diff / n_mask
    h = np.clip(h, 0.0, 1.0)
    return h

def binary_steepest_descent(h, max_iters=500, time_limit=60):
    """Steepest descent local search using pair flips on binary h"""
    n = len(h)
    start = time.time()
    current_max = compute_max_corr(h)
    best_h = h.copy()
    best_max = current_max
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_delta = 0
        best_pair = None
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Try all pairs
        for ia in range(n_ones):
            a = ones[ia]
            for ib in range(n_zeros):
                b = zeros[ib]
                h[a] = 0
                h[b] = 1
                new_max = compute_max_corr(h)
                h[a] = 1
                h[b] = 0
                delta = new_max - current_max
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (a, b)
                    if best_delta == -1:  # Can't do better than -1 for max reduction
                        # But there might be multiple -1 moves, keep looking for -2 etc
                        pass
        
        if best_pair is None:
            no_improve_count += 1
            if no_improve_count > 3:
                break
            # Perturb: random swap
            ia = np.random.randint(n_ones)
            ib = np.random.randint(n_zeros)
            h[ones[ia]], h[zeros[ib]] = h[zeros[ib]], h[ones[ia]]
            current_max = compute_max_corr(h)
        else:
            a, b = best_pair
            h[a] = 0
            h[b] = 1
            current_max += best_delta
            no_improve_count = 0
        
        if current_max < best_max:
            best_max = current_max
            best_h = h.copy()
    
    return best_h, best_max

def simulated_annealing(h, t_start=10.0, t_end=0.1, n_iter=5000, time_limit=120):
    """Simulated annealing with pair flips"""
    n = len(h)
    start = time.time()
    current = h.copy()
    current_max = compute_max_corr(current)
    best_h = current.copy()
    best_max = current_max
    
    n_iter_actual = n_iter
    for it in range(n_iter_actual):
        if time.time() - start > time_limit:
            break
        
        progress = it / n_iter_actual
        T = t_start * (t_end / t_start) ** progress
        
        ones = np.where(current == 1)[0]
        zeros = np.where(current == 0)[0]
        
        a = ones[np.random.randint(len(ones))]
        b = zeros[np.random.randint(len(zeros))]
        
        current[a] = 0
        current[b] = 1
        new_max = compute_max_corr(current)
        delta = new_max - current_max
        
        if delta <= 0 or np.random.random() < np.exp(-delta / T):
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = current.copy()
        else:
            current[a] = 1
            current[b] = 0
    
    return best_h, best_max

def obj_smooth(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    return (m + np.log(np.sum(exps)) / beta) * dx

def obj_raw(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx

def slsqp_refine(h, n, time_limit=30):
    """SLSQP refinement of a solution"""
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        return obj_smooth(x, n, beta=50)
    
    def constraint(x):
        return np.sum(x) - n / 2.0
    
    cons = [{'type': 'eq', 'fun': constraint}]
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in [20, 50, 100, 200]:
        if time.time() - start > time_limit:
            break
        
        def obj_local(x):
            return obj_smooth(x, n, beta)
        
        try:
            result = minimize(obj_local, best_h, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 200, 'ftol': 1e-12, 'iprint': 0})
            h_new = project_onto_feasible(result.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    
    # Final raw objective polish
    if time.time() - start < time_limit:
        try:
            result = minimize(lambda x: obj_raw(x, n), best_h, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_onto_feasible(result.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass
    
    return best_h, best_c5
```
