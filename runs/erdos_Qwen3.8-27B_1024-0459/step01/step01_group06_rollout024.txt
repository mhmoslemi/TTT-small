        scaled = beta * c
        m = np.max(scaled)
        all_w = np.exp(scaled - m)
        all_w /= np.sum(all_w)
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = all_w[k_idx]
        if wk < 1e-15:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            m_val = lag
            # r[n-1+m] = sum_{j=0}^{n-1-m} (1-h[j])*h[j+m]
            # grad[0:n-m] += w * (-h[m:])
            # grad[m:n]   += w * (1-h[:n-m])
            grad[:n - m_val] += wk * (-h[m_val:])
            grad[m_val:] += wk * (1.0 - h[:n - m_val])
        else:
            m_val = -lag
            # r[n-1-m] = sum_{i=0}^{n-1-m} (1-h[i+m])*h[i]
            # grad[0:n-m] += w * (1-h[m:])
            # grad[m:n]   += w * (-h[:n-m])
            grad[:n - m_val] += wk * (1.0 - h[m_val:])
            grad[m_val:] += wk * (-h[:n - m_val])
    
    grad *= dx
    if penalty > 0:
        grad += 2.0 * penalty * (h.sum() - n / 2.0)
    return grad
