def project(h, n):

    corr = get_corr(h)
    dx = 2.0 / n
    scaled = beta * corr * dx
    m = np.max(scaled)
    e = np.exp(scaled - m)
    sm = m + np.log(np.mean(e))
    s = h.sum() - n / 2.0
    return sm + lam * s * s


def soft_grad(h, n, beta, lam):

            grad[lag:] -= f * h[:n - lag]

            grad[:n - m2] -= f * h[m2:]
