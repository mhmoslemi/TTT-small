def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0: return x
    if target <= 0.0: return np.zeros(n)
    if target >= n: return np.ones(n)
    lo = float(np.min(x) - 1.0)
    hi = float(np.max(x) + 1.0)
    for _ in range(80):
        tau = 0.5*(lo+hi)
        y = np.clip(x - tau, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            lo = tau
        else:
            hi = tau
    tau=0.5*(lo+hi)
    y=np.clip(x - tau, 0.0, 1.0)
    diff = target - float(np.sum(y))
    if abs(diff) > 1e-14:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)

def compute_q_list(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    if h.size != n:
        h = h[:n]
    dx = 2.0 / float(n)
    oh = 1.0 - h
    q = np.empty(2 * n - 1, dtype=float)
    q[0] = dx * float(np.dot(h, oh))
    for m in range(1, n):
        L = n - m
        q[m] = dx * float(np.dot(h[:L], oh[m:]))
        q[n - 1 + m] = dx * float(np.dot(h[m:], oh[:L]))
    return q


def compute_grad_weighted(h, n, w):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    w = np.asarray(w, dtype=float).reshape(-1)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)

    if w[0] != 0.0:
        g += (w[0] * dx) * (1.0 - 2.0 * h)

    for m in range(1, n):
        L = n - m
        wp = w[m]
        if wp != 0.0:
            c = wp * dx
            g[:L] += c * (1.0 - h[m:])
            g[m:] -= c * h[:L]

        wn = w[n - 1 + m]
        if wn != 0.0:
            c = wn * dx
            g[m:] += c * (1.0 - h[:L])
            g[:L] -= c * h[m:]

    return g


def smooth_max_and_weights(h, n, temp):

    s = float(np.sum(e))
    F = m + temp * float(np.log(s))
    return F, e / s

def binary_round(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    if h.size < n:
        h = np.pad(h, (0, n - h.size), mode="wrap")
    else:
        h = h[:n]
    target = n // 2
    order = np.argsort(-h)
    hb = np.zeros(n, dtype=float)
    hb[order[:target]] = 1.0
    return hb


def balance_binary(h, n, rng):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    if h.size < n:
        h = np.pad(h, (0, n - h.size), mode="wrap")
    else:
        h = h[:n]
    target = n // 2
    hb = np.where(h >= 0.5, 1.0, 0.0)
    ones = int(np.sum(hb))
    if ones < target:
        zeros = np.flatnonzero(hb == 0.0)
        need = target - ones
        if zeros.size < need:
            return hb
        idx = rng.choice(zeros, size=need, replace=False)
        hb[idx] = 1.0
    elif ones > target:
        pos = np.flatnonzero(hb == 1.0)
        need = ones - target
        if pos.size < need:
            return hb
        idx = rng.choice(pos, size=need, replace=False)
        hb[idx] = 0.0
    return hb


def binary_eval(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def make_binary_runs(n, rng, q):
    n = int(n)
    h = np.empty(n, dtype=float)
    state = 1.0 if rng.random() < 0.5 else 0.0
    q = float(q)

def binary_state_from_z(z, n):
    z = np.asarray(z, dtype=float).reshape(-1)
    n = int(n)
    C = np.empty(n, dtype=float)
    for m in range(n):
        L = n - m
        C[m] = float(np.dot(z[:L], z[m:]))
    P = np.empty(n + 1, dtype=float)
    P[0] = 0.0
    P[1:] = np.cumsum(z)
    return C, P


def binary_objective(C, P, midx, Lvals, n):
    B = P[n - midx] + P[midx]
    M = (Lvals - C + np.abs(B)) * 0.25
    M = np.maximum(M, 0.0)
    return float(np.max(M)) * (2.0 / float(n))


def binary_trial_objective(z, C, P, a, b, midx, Lvals, n):
    z = np.asarray(z, dtype=float)
    C = np.asarray(C, dtype=float)
    P = np.asarray(P, dtype=float)
    n = int(n)
    a = int(a)
    b = int(b)

    Cn = C.copy()

    rem = n - 1 - a
    if rem > 0:
        Cn[1:rem + 1] += -2.0 * z[a + 1:a + 1 + rem]
    if a > 0:
        Cn[1:a + 1] += -2.0 * z[a - 1::-1]

    rem = n - 1 - b
    if rem > 0:
        Cn[1:rem + 1] += 2.0 * z[b + 1:b + 1 + rem]
    if b > 0:
        Cn[1:b + 1] += 2.0 * z[b - 1::-1]

    if a != b:
        Cn[abs(a - b)] += -4.0

    Pt = P.copy()
    if a < b:
        Pt[a + 1:b + 1] -= 2.0
    elif a > b:
        Pt[b + 1:a + 1] += 2.0

    B = Pt[n - midx] + Pt[midx]
    M = (Lvals - Cn + np.abs(B)) * 0.25
    M = np.maximum(M, 0.0)
    return float(np.max(M)) * (2.0 / float(n))


def binary_apply_swap(z, C, P, a, b, n):
    n = int(n)
    a = int(a)
    b = int(b)

    rem = n - 1 - a
    if rem > 0:
        C[1:rem + 1] += -2.0 * z[a + 1:a + 1 + rem]
    if a > 0:
        C[1:a + 1] += -2.0 * z[a - 1::-1]

    rem = n - 1 - b
    if rem > 0:
        C[1:rem + 1] += 2.0 * z[b + 1:b + 1 + rem]
    if b > 0:
        C[1:b + 1] += 2.0 * z[b - 1::-1]

    if a != b:
        C[abs(a - b)] += -4.0

    if a < b:
        P[a + 1:b + 1] -= 2.0
    elif a > b:
        P[b + 1:a + 1] += 2.0

    z[a] = -z[a]
    z[b] = -z[b]


def binary_best_swap_for_a(z, C, P, a, midx, Lvals, n, currC):
    z = np.asarray(z, dtype=float)
    C = np.asarray(C, dtype=float)
    P = np.asarray(P, dtype=float)
    n = int(n)
    a = int(a)
    currC = float(currC)
    zeros = np.flatnonzero(z < 0.0)
    best_val = currC
    best_b = -1
    for b in zeros:
        val = binary_trial_objective(z, C, P, a, int(b), midx, Lvals, n)
        if val < best_val - 1e-12:
            best_val = val
            best_b = int(b)
    return best_b, best_val

def make_sinusoid(n, freq, phase, edge_power):
    n = int(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    edge = np.cos(np.pi * x / 2.0)
    edge = np.clip(edge, 0.0, 1.0) ** float(edge_power)
    h = 0.5 + 0.5 * edge * np.sin(np.pi * float(freq) * x + float(phase))
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, 0.5 * float(n))


def make_random_smooth(n, rng, max_freq=6, amp_scale=0.45):
    n = int(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    edge = np.cos(np.pi * x / 2.0)
    edge = np.clip(edge, 0.0, 1.0) ** 2.0
    y = np.zeros(n, dtype=float)
    for f in range(1, int(max_freq) + 1):
        if rng.random() < 0.8:
            amp = float(amp_scale) * rng.uniform(0.2, 1.0) / max(1.0, float(f) ** 0.5)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            y += amp * np.sin(np.pi * float(f) * x + ph)
    h = 0.5 + edge * y
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, 0.5 * float(n))

        hp = project_box_sum(parent, 0.5 * float(n0))
        hp = force_exact_sum(hp, 0.5 * float(n0))
        starts.append((hp, n0, 3))
        starts.append((binary_round(hp, n0), n0, 2))

        for sc in (0.01, 0.03, 0.06, 0.10):
            hn = project_box_sum(parent + rng.normal(0.0, sc, n0), 0.5 * float(n0))
            hn = force_exact_sum(hn, 0.5 * float(n0))
            starts.append((hn, n0, 2 if sc <= 0.03 else 1))
            starts.append((binary_round(hn, n0), n0, 2 if sc <= 0.03 else 1))

    for f, ph, ep in (
        (2, 0.0, 1.0),
        (3, 1.5707963267948966, 1.0),
        (4, 0.7853981633974483, 2.0),
        (5, 1.25, 1.0),
        (3, 0.5235987755982988, 2.0),
    ):
        starts.append((make_sinusoid(n0, f, ph, ep), n0, 2))

    for i in range(3):
        starts.append((make_random_smooth(n0, rng), n0, 2 if i == 0 else 1))

    for q in (0.03, 0.08, 0.15, 0.25, 0.40):
        starts.append((make_binary_runs(n0, rng, q), n0, 1))

    processed = 0
    for hstart, nstart, lev in starts:
        rem = deadline - time.time()
        if rem < 90:
            break
        if processed >= 8:
            break

        if lev >= 3 and rem < 450:
            lev = 2
        elif lev >= 2 and rem < 180:
            lev = 1

        if rem < 45:
            break

        try:
            h2, v2 = improve_candidate(hstart, nstart, rng, deadline, lev)
        except Exception:
            continue

        if v2 < best_val:
            best_h, best_val, best_n = h2, float(v2), int(nstart)
        processed += 1

    bn = int(best_n)
    if bn % 2 == 0 and best_h is not None:
        bin_starts = [binary_round(best_h, bn)]
        for q in (0.05, 0.15, 0.35):
            bin_starts.append(make_binary_runs(bn, rng, q))

        for hb in bin_starts:
            rem = deadline - time.time()
            if rem < 160:
                break
            base_steps = int(30000 * 70.0 / float(bn))
            base_steps = max(3000, min(50000, base_steps))
            if rem > 500:
                base_steps = int(1.5 * base_steps)
            base_steps = int(base_steps)

            try:
                hb2, vb2 = binary_local_search(hb, bn, rng, base_steps, 0.002, deadline - 45)
            except Exception:
                hb2, vb2 = hb, np.inf

            if vb2 < best_val:
                best_h, best_val, best_n = hb2, float(vb2), bn

            rem = deadline - time.time()
            if rem > 220 and vb2 < best_val + 0.01:
                try:
                    hc, vc = improve_candidate(hb2, bn, rng, deadline, 1)
                except Exception:
                    hc, vc = hb2, vb2
                if vc < best_val:
                    best_h, best_val, best_n = hc, float(vc), bn

    bf = int(best_n)
    if best_h is not None:
        nf_candidates = []
        rem = deadline - time.time()
        if rem > 500:
            if bf < 100:
                nf_candidates.extend([100, 140])
            elif bf < 140:
                nf_candidates.extend([140])
            elif bf < 200:
                nf2 = 2 * bf
                if nf2 > bf and nf2 <= 200:
                    nf_candidates.append(nf2)

        for nf in nf_candidates:
            rem = deadline - time.time()
            if rem < 300:
                break
            if nf <= int(best_n):
                continue
            nf = int(nf)
            if nf % 2 != 0:
                nf -= 1
            if nf < 4:
                continue

            x_old = np.linspace(0.0, 2.0, int(best_n), endpoint=False)
            x_new = np.linspace(0.0, 2.0, nf, endpoint=False)
            hf = np.interp(x_new, x_old, best_h)
            hf = project_box_sum(hf, 0.5 * float(nf))
            hf = force_exact_sum(hf, 0.5 * float(nf))

            try:
                h2, v2 = improve_candidate(hf, nf, rng, deadline, 1)
            except Exception:
                h2, v2 = hf, np.inf

            if v2 < best_val:
                best_h, best_val, best_n = h2, float(v2), nf

            rem = deadline - time.time()
            if rem > 200 and int(best_n) % 2 == 0:
                hn = int(best_n)
                hb = binary_round(h2, hn)
                steps = int(20000 * 70.0 / float(hn))
                steps = max(2500, min(35000, steps))
                if rem > 400:
                    steps = int(1.5 * steps)
                try:
                    hb2, vb2 = binary_local_search(hb, hn, rng, steps, 0.002, deadline - 45)
                except Exception:
                    hb2, vb2 = hb, np.inf
                if vb2 < best_val:
                    best_h, best_val, best_n = hb2, float(vb2), hn

                rem = deadline - time.time()
                if rem > 180:
                    try:
                        hc, vc = improve_candidate(hb2, hn, rng, deadline, 1)
                    except Exception:
                        hc, vc = hb2, vb2
                    if vc < best_val:
                        best_h, best_val, best_n = hc, float(vc), hn

    rem = deadline - time.time()
    if rem > 60 and best_h is not None:
        iters = 600 if rem > 200 else 300
        if int(best_n) > 120:
            iters = max(60, iters // 2)
        try:
            hp, vp = active_polish(best_h, int(best_n), iters, 1e-7, 0.01, rng, deadline - 3)
        except Exception:
            hp, vp = best_h, best_val
        if vp < best_val:
            best_h, best_val, best_n = hp, float(vp), int(best_n)

    best_h = project_box_sum(best_h, 0.5 * float(int(best_n)))
    best_h = force_exact_sum(best_h, 0.5 * float(int(best_n)))
    best_val = exact_c5(best_h, int(best_n))
    return (best_h, float(best_val), int(best_n))
