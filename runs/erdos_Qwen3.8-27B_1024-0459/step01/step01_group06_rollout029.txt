def obj_and_grad_smooothmax(h, n, beta):
    """Smooth max approximation with gradient"""
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    # Log-sum-exp for stability
    shifted = scaled - m
    log_sum = m + np.log(np.sum(np.exp(shifted)))
    val = (log_sum / beta) * (2.0 / n)
    
    # Gradient
    w = np.exp(shifted)
    w /= np.sum(w)
    dx = 2.0 / n
    n_full = len(c)  # 2n-1
    k_max_offset = n - 1
    
    grad = np.zeros(n)
    for k_idx in range(n_full):
        if w[k_idx] < 1e-18:
            continue
        wk = w[k_idx]
        lag = k_idx - k_max_offset
        if lag >= 0:
            # c[k_idx] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
            # d/dc_k_idx/dh[j] = (1-h[j-lag]) for j >= lag, and -h[j+lag] for j < n-lag
            # Actually: c = sum_i h[i+lag]*(1-h[i]) for i=0..n-1-lag
            # d/dh[j] = (1-h[j-lag]) if j >= lag (from the h[i+lag] term, i = j-lag)
            # d/dh[j] + (-h[j+lag]) if j+lag < n (from the (1-h[i]) term, i = j, i+lag = j+lag)
            overlap = n - lag
            # h[j] appears in h[i+lag] when i+lag=j, i=j-lag, valid if 0<=j-lag<=overlap-1
            # h[j] appears in (1-h[i]) when i=j, valid if 0<=j<overlap

            # j must be < overlap for second term

            grad[lag2:] -= wk * h[:overlap]
    
    return val, grad * dx

def obj_hard(h, n):
    return compute_c5(h, n)

def grad_hard(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    k_idx = np.argmax(c)
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        overlap = n - lag
        grad[lag:] += (1.0 - h[:overlap])
        grad[:overlap] -= h[lag:]
    else:
        lag2 = -lag
        overlap = n - lag2
        grad[:overlap] += (1.0 - h[lag2:])
        grad[lag2:] -= h[:overlap]
    return grad * dx

def adam_projected_grad(h, n, lr=0.01, n_iter=5000, beta1=0.9, beta2=0.999):
    """Adam optimizer with projected gradient for the hard max C5"""
    h = h.copy().astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for t in range(1, n_iter + 1):
        val = compute_c5(h, n)
        if val < best_c5:

            best_h = h.copy()
        
        g = grad_hard(h, n)
        
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h, n)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-14
        else:
            mask = h < 1.0 - 1e-14
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
    h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = h.sum() - target
    if abs(diff) > 1e-15:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def corr_and_gradient(h, n):
    """Compute correlation values and gradient w.r.t. a single lag."""
    c = np.correlate(h, 1.0 - h, mode="full")
    return c


def grad_for_lag(h, n, lag):
    """Gradient of c_k (for given lag) w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        overlap = n - lag
        if overlap > 0:
            grad[lag:] += (1.0 - h[:overlap])
            grad[:overlap] -= h[lag:overlap + lag] if overlap + lag <= n else h[lag:]
            # Correct: d/dh[j] of sum_{i=0}^{overlap-1} h[i+lag]*(1-h[i])
            # = (1-h[j-lag]) for j in [lag, n-1]
            #   - h[j+lag] for j in [0, overlap-1]
            grad = np.zeros(n)
            grad[lag:] += (1.0 - h[:overlap])
            grad[:overlap] -= h[lag:lag + overlap]
    else:
        lag2 = -lag
        overlap = n - lag2
        if overlap > 0:
            grad = np.zeros(n)
            # d/dh[j] of sum_{i=0}^{overlap-1} h[i]*(1-h[i+lag2])
            # = (1-h[j+lag2]) for j in [0, overlap-1]
            #   - h[j-lag2] for j in [lag2, n-1]
            grad[:overlap] += (1.0 - h[lag2:lag2 + overlap])
            grad[lag2:] -= h[:overlap]
    return grad * dx


def softmax_obj_val(h, n, beta):
    """Evaluate softmax approximation of max correlation."""
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n))


def softmax_obj_grad(h, n, beta):
    """Evaluate softmax value and gradient."""
    h = np.clip(h, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    shifted = scaled - m
    w = np.exp(shifted)
    w /= np.sum(w)
    val = float((m + np.log(np.sum(w))) / beta * (2.0 / n))
    
    dx = 2.0 / n
    grad = np.zeros(n)
    n_full = len(c)
    
    for k_idx in range(n_full):
        if w[k_idx] < 1e-18:
            continue
        wk = w[k_idx]
        lag = k_idx - (n - 1)
        if lag >= 0:
            overlap = n - lag
            if overlap > 0:
                grad[lag:] += wk * (1.0 - h[:overlap])
                grad[:overlap] -= wk * h[lag:lag + overlap]
        else:
            lag2 = -lag
            overlap = n - lag2
            if overlap > 0:
                grad[:overlap] += wk * (1.0 - h[lag2:lag2 + overlap])
                grad[lag2:] -= wk * h[:overlap]
    
    return val, grad * dx


def lbgfs_objective(x, n, beta):
    return softmax_obj_grad(np.clip(x, 0.0, 1.0), n, beta)[0]


def lbgfs_gradient(x, n, beta):
    return softmax_obj_grad(np.clip(x, 0.0, 1.0), n, beta)[1]


def adam_optimize(h, n, lr, n_iter, use_smooth):
    """Adam optimizer for C5 minimization."""
    h = h.copy().astype(float)
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    for t in range(1, n_iter + 1):
        if use_smooth:
            _, g = softmax_obj_grad(h, n, 100.0)
        else:
            c = np.correlate(h, 1.0 - h, mode="full")
            k_max = int(np.argmax(c))
            g = grad_for_lag(h, n, k_max - (n - 1))
        
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g * g
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5


def subgradient_optimize(h, n, n_iter, step_scale):
    """Subgradient method with diminishing step size for hard max."""
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    momentum = 0.9
    vel = np.zeros(n)
    
    for t in range(1, n_iter + 1):
        lr = step_scale / np.sqrt(t)
        
        c = np.correlate(h, 1.0 - h, mode="full")
        k_max = int(np.argmax(c))
        g = grad_for_lag(h, n, k_max - (n - 1))
        
        vel = momentum * vel + (1 - momentum) * g
        h_new = h - lr * vel
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5


def try_resolution(h_from, n_from, n_to, beta_seq, t0, budget_s, rng):
    """Optimize at a new resolution, starting from interpolation of h_from."""
    x_old = np.linspace(0, 2, n_from, endpoint=False) + 1.0 / n_from
    x_new = np.linspace(0, 2, n_to, endpoint=False) + 1.0 / n_to
    h_curr = np.interp(x_new, x_old, h_from)
    h_curr = project_feasible(h_curr, n_to)
    c5_curr = compute_c5(h_curr, n_to)
    
    for beta in beta_seq:
        if time.time() - t0 > budget_s - 40:
            break
        try:
            res = fmin_l_bfgs_b(
                lbgfs_objective, h_curr.copy(),
                fprime=lbgfs_gradient,
                args=(n_to, beta),
                pgtol=1e-12, maxiter=1500, maxfun=5000
            )
            h_new = project_feasible(np.clip(res, 0.0, 1.0), n_to)
            c5_new = compute_c5(h_new, n_to)
            if c5_new < c5_curr:
                h_curr = h_new
                c5_curr = c5_new
        except Exception:
            pass
    
    return h_curr, c5_curr
