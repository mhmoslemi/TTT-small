Let me formalize:
- h is a step function with breakpoints 0 = b_0 < b_1 < ... < b_m < b_{m+1} = 2
- h = c_i on [b_i, b_{i+1}) where c_i ∈ {0, 1}
- Constraint: sum of (b_{i+1} - b_i) over i where c_i = 1 equals 1

For a shift k > 0:
∫_0^{2-k} h(x)(1-h(x+k)) dx

This can be computed by iterating over the segments and finding the overlap.

The critical values of k are: all differences |b_i - b_j| for i, j in {0, ..., m+1}.

def eval_step_function(gaps, n_trans):

    gaps[i] is the length of segment i (0-indexed), h starts at 1.
    """

    m = n_trans  # number of transitions
    # t_0 = 0, t_1 = gaps[0], t_2 = gaps[0]+gaps[1], ..., t_{m+1} = 2
    t = np.zeros(m + 2)
    t[1:] = np.cumsum(gaps)
    # t should end at 2
    # Segments: segment i (0-indexed) is [t[i], t[i+1]) with value 1 if i even, 0 if i odd
    n_seg = m + 1  # number of segments
    
    # For binary, h(x)(1-h(x+k)) = 1 iff h(x)=1 and h(x+k)=0
    # i.e., x is in an even-indexed segment and x+k is in an odd-indexed segment

    shifts = set()
    for i in range(m + 2):
        for j in range(m + 2):
            d = abs(t[i] - t[j])
            if d > 0 and d < 2:
                shifts.add(round(d, 15))
            elif d == 0:
                shifts.add(0.0)
    shifts = sorted(shifts)
    
    # Compute overlap at each critical shift
    max_overlap = 0.0
    for k in shifts:
        if k >= 2.0:
            continue

        # where x is in [t[i], t[i+1]] and x is in [0, 2-k]
        o = 0.0
        for i in range(n_seg):

            s1, e1 = t[i], t[i+1]

            hi = min(e1, 2.0 - k)
            if lo >= hi:

            for j in range(n_seg):
                if j % 2 != 1:  # h=1, skip (we need h(x+k)=0)

                s2, e2 = t[j], t[j+1]
                olo = max(xlo, s2)
                ohi = min(xhi, e2)
                if olo < ohi:
                    o += ohi - olo
        max_overlap = max(max_overlap, o)
    
    return max_overlap

def upsample(h, n_new):

    n0 = h.size
