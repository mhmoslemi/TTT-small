```python
def project_onto_feasible(h, n):
    """Project h onto {0 ≤ h_i ≤ 1, Σh_i = n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 2.0  # Range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fine-tune to get exact sum
    diff = target - projected.sum()
    if abs(diff) > 1e-15:
        # Distribute the difference
        if diff > 0:
            mask = projected < 1.0 - 1e-15
        else:
            mask = projected > 1e-15
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0.0, 1.0)
    return projected
```

```python
        if time.time() > deadline - 30:
            break

            if time.time() > deadline - 30:
                break

            b = end - start

            block_sum = h[start:end].sum()

            target = block_sum  # We're not changing the sum, just redistributing
```
