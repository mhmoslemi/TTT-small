def compute_c5(h):
    n = len(h)

def project_continuous(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        free = (h > 1e-10) & (h < 1.0 - 1e-10)

            diff2 = target - np.sum(h)
            h[free] += diff2 / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
