import time
import numpy as np
from scipy.optimize import minimize


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_sum(h, target):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    if h.size == 0:
        return h
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / h.size
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = np.nonzero(h < 1.0 - 1e-15)[0]
        else:
            idx = np.nonzero(h > 1e-15)[0]
        if idx.size:
            per = diff / idx.size
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def smooth_max(x, tau):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return 0.0
    m = float(np.max(x))
    if tau <= 1e-12:
        return m
    y = (x - m) / tau
    y = np.clip(y, -60.0, 0.0)
    return m + tau * float(np.log(np.sum(np.exp(y))))


def softmax_weights(x, tau):
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n == 0:
        return np.array([], dtype=float)
    m = float(np.max(x))
    w = np.zeros(n, dtype=float)
    if tau <= 1e-12:
        w[int(np.argmax(x))] = 1.0
        return w
    y = (x - m) / tau
    y = np.clip(y, -60.0, 0.0)
    w = np.exp(y)
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        w[int(np.argmax(x))] = 1.0
        return w
    return w / s


def sigmoid(z):
    z = np.asarray(z, dtype=float)
    out = np.empty_like(z)
    pos = z >= 0.0
    out[pos] = 1.0 / (1.0 + np.exp(-z[pos]))
    ez = np.exp(z[~pos])
    out[~pos] = ez / (1.0 + ez)
    return out


def compute_shift_a(s, target_sum):
    s = np.asarray(s, dtype=float)
    if s.size == 0:
        return 0.0
    lo = -60.0
    hi = 60.0
    target = float(target_sum)
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        h = sigmoid(s + mid)
        diff = float(np.sum(h)) - target
        if diff > 0.0:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)


def make_logit(h, level=1e-3):
    h = np.clip(np.asarray(h, dtype=float), level, 1.0 - level)
    s = np.log(h / (1.0 - h))
    s -= float(np.mean(s))
    return np.clip(s, -25.0, 25.0)


def resample_h(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = len(h_old)
    if n_old == 0:
        return None
    if n_old == n_new:
        return project_sum(h_old, 0.5 * n_new)
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h = np.interp(x_new, x_old, h_old)
    return project_sum(h, 0.5 * n_new)


def make_sym_h(t):
    t = np.asarray(t, dtype=float)
    m = t.size
    h = np.empty(2 * m, dtype=float)
    if m:
        h[:m] = t
        h[m:] = 1.0 - t[::-1]
    return h


def make_symmetric_h(h, n):
    if n % 2 != 0:
        n = n - 1
    h = np.asarray(h, dtype=float)
    if len(h) != n:
        h = resample_h(h, n)
        if h is None:
            return None
    h = np.clip(h, 0.0, 1.0)
    hs = 0.5 * (h + 1.0 - h[::-1])
    return project_sum(hs, 0.5 * n)


def make_symmetric_t(h, n):
    if n % 2 != 0:
        n = n - 1
    hs = make_symmetric_h(h, n)
    if hs is None:
        return None
    return hs[: n // 2]


def make_random_t(n, style=0):
    m = n // 2
    if m <= 0:
        return np.array([], dtype=float)
    if style == 0:
        return (np.random.rand(m) < 0.5).astype(float)
    if style == 1:
        return np.clip(0.5 + 0.45 * np.random.randn(m), 0.0, 1.0)
    if style == 3:
        return np.clip(0.5 + (np.random.rand(m) - 0.5), 0.0, 1.0)
    x = np.linspace(0.0, 1.0, m, endpoint=False)
    freq = 1 + int(np.random.randint(0, 4))
    phase = 2.0 * np.pi * np.random.rand()
    return np.clip(0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x + phase), 0.0, 1.0)


def random_s_h(n, scale=4.0):
    s = np.random.uniform(-scale, scale, n)
    s -= float(np.mean(s))
    a = compute_shift_a(s, 0.5 * n)
    h = sigmoid(s + a)
    return project_sum(h, 0.5 * n)


def binary_sym_local_search(t, n, epochs, deadline):
    if n % 2 != 0 or n < 4:
        return np.asarray(t, dtype=float)
    t = np.round(np.clip(np.asarray(t, dtype=float), 0.0, 1.0)).copy()
    m = t.size
    if m == 0 or 2 * m != n:
        t = t[: n // 2]
        m = t.size
        if 2 * m != n:
            return t
    h = make_sym_h(t)
    conv = np.convolve(h, h, "full")
    cur = float(np.max(conv)) if conv.size else 0.0
    best = t.copy()
    best_cur = cur
    temp = 2.0

    for ep in range(max(0, int(epochs))):
        if time.time() > deadline:
            break
        conv = np.convolve(h, h, "full")
        cur = float(np.max(conv)) if conv.size else 0.0
        idx = np.random.permutation(m)
        improved = False
        L = 2 * n - 1
        for i in idx:
            if time.time() > deadline:
                break
            ti = int(round(float(t[i])))
            if ti < 0 or ti > 1:
                continue
            p = n - 1 - i
            delta = 1.0 - 2.0 * float(ti)
            inc = np.zeros(L, dtype=float)
            inc[i:i + n] += 2.0 * delta * h
            inc[p:p + n] -= 2.0 * delta * h
            inc[2 * i] += 1.0
            inc[2 * p] += 1.0
            inc[i + p] -= 2.0
            new_conv = conv + inc
            sc = float(np.max(new_conv)) if new_conv.size else 0.0
            accept = sc < cur - 1e-12
            if not accept:
                diff = sc - cur
                if diff <= 1e-12:
                    accept = True
                elif diff < 2.0 and temp > 0.01 and np.random.rand() < np.exp(-diff / temp):
                    accept = True
            if accept:
                t[i] = float(1 - ti)
                h[i] = float(np.clip(h[i] + delta, 0.0, 1.0))
                h[p] = float(np.clip(h[p] - delta, 0.0, 1.0))
                conv = new_conv
                cur = sc
                improved = True
                if cur < best_cur - 1e-12:
                    best_cur = cur
                    best = t.copy()
        temp *= 0.6
        if not improved and ep > 2:
            break
    return best


def sym_objective(t, n, tau):
    t = np.asarray(t, dtype=float)
    h = make_sym_h(t)
    if h.size == 0:
        return 0.0
    conv = np.convolve(h, h, "full")
    x = conv * (2.0 / float(n))
    return smooth_max(x, tau)


def sym_jacobian(t, n, tau):
    t = np.asarray(t, dtype=float)
    m = t.size
    if m == 0:
        return np.array([], dtype=float)
    n = 2 * m
    h = make_sym_h(t)
    conv = np.convolve(h, h, "full")
    x = conv * (2.0 / float(n))
    p = softmax_weights(x, tau)
    base = np.convolve(p, h[::-1], "full")[n - 1:2 * n - 1]
    grad_counts = base + h * p[0::2]
    grad_h = (2.0 / float(n)) * grad_counts
    return grad_h[:m] - grad_h[m:][::-1]


def optimize_symmetric(t0, n, tau_schedule, maxiter, maxfun, ftol, deadline):
    if n % 2 != 0 or n < 4:

    h0 = project_sum(h0, 0.5 * n)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

            break
        try:
            res = minimize(
                sym_objective,
                t_work,
                method="L-BFGS-B",

                    "maxfun": int(maxfun),

            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-14:
                best_c = c_new

def s_objective(s, n, tau):
    s = np.asarray(s, dtype=float)
    if s.size == 0:
        return 0.0
    a = compute_shift_a(s, 0.5 * n)
    h = sigmoid(s + a)
    h = np.clip(h, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, "full")
    x = c * (2.0 / float(n))
    return smooth_max(x, tau)


def s_jacobian(s, n, tau):
    s = np.asarray(s, dtype=float)
    n = int(n)
    if s.size == 0:
        return np.array([], dtype=float)
    a = compute_shift_a(s, 0.5 * n)
    h = sigmoid(s + a)
    h = np.clip(h, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, "full")
    x = c * (2.0 / float(n))
    p = softmax_weights(x, tau)

    L = 2 * n - 1
    pref = np.empty(L + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(p)
    A = pref[n:2 * n] - pref[:n]
    T1 = np.convolve(p, h, "full")[n - 1:2 * n - 1]
    T2 = np.convolve(p[::-1], h, "full")[n - 1:2 * n - 1]
    grad_counts = A - T1 - T2
    grad_h = (2.0 / float(n)) * grad_counts

    sp = h * (1.0 - h)
    S = float(np.sum(sp))
    if S < 1e-14:
        return np.zeros_like(s)
    gdot = float(np.dot(grad_h, sp))
    return sp * (grad_h - gdot / S)

                tau_s = (0.05, 0.025, 0.012, 0.006, 0.003)
            elif rem > 350.0:
                tau_s = (0.03, 0.015, 0.008, 0.004)
            elif rem > 180.0:
                tau_s = (0.02, 0.01, 0.005)
            else:
                tau_s = (0.01, 0.005)

            if n <= 160:
                maxit_s = 130
            elif n <= 320:
                maxit_s = 100
            else:
                maxit_s = 80
            maxfun_s = maxit_s * 2 + 20

            hopt, copt = optimize_s_full(h_for, n, tau_s, maxit_s, maxfun_s, 1e-9, deadline)
            full_runs += 1
            if copt < best_c:
                best_h = hopt
                best_c = copt
                best_n = n

    if time.time() < deadline - 320.0 and best_n < 512 and best_h is not None:
        n_up = 512
        h_up = resample_h(best_h, n_up)
        if h_up is not None:
            h_eval = project_sum(h_up, 0.5 * n_up)
            ceval = compute_c5(h_eval)
            if ceval < best_c:
                best_h = h_eval
                best_c = ceval
                best_n = n_up
            h_for = np.clip(h_eval, 0.05, 0.95)
            hopt, copt = optimize_s_full(h_for, n_up, (0.03, 0.015, 0.008, 0.004), 90, 240, 1e-9, deadline)
            if copt < best_c:
                best_h = hopt
                best_c = copt
                best_n = n_up

    if best_h is not None and time.time() < deadline - 150.0:
        for attempt in range(3):
            if time.time() > deadline - 60.0:
                break
            noise = 0.07 * (1.0 - 0.25 * attempt)
            hpert = best_h + noise * np.random.randn(best_n)
            hpert = project_sum(hpert, 0.5 * best_n)
            tau_f = (0.02, 0.01, 0.006) if best_n <= 256 else (0.016, 0.008)
            hopt, copt = optimize_s_full(hpert, best_n, tau_f, 70, 200, 1e-9, deadline)
            if copt < best_c:
                best_h = hopt
                best_c = copt

    if best_h is None:
        n_final = 96
        best_n = n_final
        best_h = np.full(n_final, 0.5, dtype=float)
        best_c = 0.5

    best_n = int(best_n)
    best_h = np.asarray(best_h, dtype=float)
    if best_h.size != best_n:
        best_h = resample_h(best_h, best_n)
        if best_h is None:
            best_h = np.full(best_n, 0.5, dtype=float)
    best_h = project_sum(best_h, 0.5 * best_n)
    best_h = np.where(np.isfinite(best_h), best_h, 0.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5(best_h)

    return (best_h, float(best_c), int(best_n))
