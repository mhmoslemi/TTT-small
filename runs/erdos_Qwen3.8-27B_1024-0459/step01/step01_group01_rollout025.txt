import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        per = diff / cnt
        h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def gradient_of_smooth_max(h, tau, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
    else:
        # Use all but weight by softmax
        top_idx = np.arange(len(corr))
    
    # Softmax weights
    shifted = corr[top_idx] - np.max(corr[top_idx])
    weights = np.exp(shifted / tau)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)  # shift in sample units
        w = weights[np.where(top_idx == idx)[0][0]]
        if k >= 0:
            if k == 0:
                # h[i]*(1-h[i]) = h[i] - h[i]^2
                grad += w * (1.0 - 2.0 * h)
            else:
                # Σ_{i=0}^{n-1-k} h[i]*(1-h[i+k])
                # ∂/∂h[m] for m in [0, n-1-k]: (1-h[m+k])
                # ∂/∂h[m] for m in [k, n-1]: -h[m-k]
                # Combined for m in [k, n-1-k]: (1-h[m+k]) - h[m-k]
                mask1 = np.zeros(n, dtype=bool)
                mask1[:n-k] = True
                mask2 = np.zeros(n, dtype=bool)
                mask2[k:] = True
                both = mask1 & mask2
                only1 = mask1 & ~mask2
                only2 = mask2 & ~mask1
                
                if both.any():
                    grad[both] += w * ((1.0 - h[np.where(both)[0] + k]) - h[np.where(both)[0] - k])
                if only1.any():
                    idx1 = np.where(only1)[0]
                    grad[idx1] += w * (1.0 - h[idx1 + k])
                if only2.any():
                    idx2 = np.where(only2)[0]
                    grad[idx2] -= w * h[idx2 - k]
        else:
            s = -k
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                # Σ_{i=s}^{n-1} h[i]*(1-h[i-s])
                # ∂/∂h[m] for m in [s, n-1]: (1-h[m-s])
                # ∂/∂h[m] for m in [0, n-1-s]: -h[m+s]

    if top_k is not None and top_k < len(corr):

        c_top = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        c_top = corr
    
    c_max = np.max(c_top)
    exp_c = np.exp((c_top - c_max) / tau)
    weights = exp_c / exp_c.sum()

def gradient_descent_step(h, n, dx, tau, lr, top_k=None):
    grad = grad_smooth_max(h, tau, n, dx, top_k=top_k)
    h_new = h - lr * grad
    h_new = project_h(h_new, n)
    return h_new

def backtracking_line_search(h, n, dx, grad, lr, max_iter=50, c1=1e-4):
    f0 = compute_c5(h)
    for i in range(max_iter):
        h_try = h - lr * grad
        h_try = project_h(h_try, n)
        f_try = compute_c5(h_try)
        if f_try <= f0 - c1 * lr * np.sum(grad * (h - h_try)):
            return h_try, f_try
        lr *= 0.5
    return h_try, f_try

            best_c5 = c5_new
            best_h = h_new
    
    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
