_n_cur = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n, rng):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = int(rng.randint(0, n))
            if diff > 0:
                h[idx] = min(1.0, h[idx] + 0.005)
            else:
                h[idx] = max(0.0, h[idx] - 0.005)
            continue
        h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-13:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h
