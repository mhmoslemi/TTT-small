        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_h = project_feasible(parent_h, parent_n)
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
    except Exception:
        pass

    if best_h is None:
        best_n = 96
        best_h = np.zeros(best_n)
        best_h[:48] = 1.0
        best_c5 = compute_c5(best_h)

    print("Initial: n={}, C5={:.8f}".format(best_n, best_c5))

    # Determine resolutions to try
    n_list = set()
    if parent_n > 0:
        n_list.add(parent_n)
    for nn in [96, 128, 160, 192, 256]:
        if 48 <= nn <= 300:
            n_list.add(nn)
    n_list = sorted(n_list)

    # Phase A: Multi-resolution multi-stage SLSQP
    for n in n_list:
        if remaining() < 35:
            break

        # Build initializations
        inits = []
        if parent_n > 0:
            inits.append(interp_to(parent_h, parent_n, n))
        if best_n > 0:
            inits.append(interp_to(best_h, best_n, n))
        # Random
        h_rand = rng.rand(n)
        inits.append(project_feasible(h_rand, n))
        # Structured (block + noise)
        h_struct = make_structured_init(n, rng)
        inits.append(h_struct)
        # Low-frequency
        t = np.arange(n) / n
        h_lf = 0.5 + 0.3 * np.sin(2 * np.pi * t * (1 + rng.rand()) + rng.rand() * 2 * np.pi)
        inits.append(project_feasible(h_lf, n))

        for h_init in inits:
            if remaining() < 25:
                break
            h_cur = project_feasible(h_init, n)
            c5_cur = compute_c5(h_cur)

            # Progressive smoothing stages
            stages = [
                (8, 200, 1e-10),
                (25, 300, 1e-11),
                (80, 300, 1e-12),
                (250, 400, 1e-12),
            ]
            for beta, maxit, ftol in stages:
                if remaining() < 12:
                    break
                h_cur, c5_cur = slsqp_opt(h_cur, n, beta, maxit, ftol)

            # Top-k sum stage
            if remaining() > 15:
                h_k, c5_k = topk_sum_opt(h_cur, n, k=5, maxiter=200, ftol=1e-12)
                if c5_k < c5_cur:
                    h_cur, c5_cur = h_k, c5_k
            if remaining() > 15:
                h_k, c5_k = topk_sum_opt(h_cur, n, k=10, maxiter=200, ftol=1e-12)
                if c5_k < c5_cur:
                    h_cur, c5_cur = h_k, c5_k

            # Very high beta final
            if remaining() > 12:
                h_cur, c5_cur = slsqp_opt(h_cur, n, 500, 300, 1e-13)

            if c5_cur < best_c5 - 1e-11:
                best_c5 = c5_cur
                best_h = h_cur.copy()
                best_n = n
                print("  SLSQP: n={}, C5={:.8f}".format(n, best_c5))

    # Phase B: Nelder-Mead coordinate polish
    if remaining() > 40:
        h_nm, c5_nm = nemesis_polish(best_h, best_n, rng, max_iter=100)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()
            print("  NM: C5={:.8f}".format(best_c5))

    # Phase C: Coordinate descent
    if remaining() > 20:
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=3, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CoordDesc: C5={:.8f}".format(best_c5))

    # Phase D: Swap annealing on binary-ish solutions
    if remaining() > 80:
        n = best_n
        # Binary from best
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_want = n // 2
        if n_ones > n_want:
            ones = np.where(h_bin == 1.0)[0]
            order = np.argsort(best_h[ones])[:n_ones - n_want]
            h_bin[ones[order]] = 0.0
        elif n_ones < n_want:
            zeros = np.where(h_bin == 0.0)[0]
            order = np.argsort(-best_h[zeros])[:n_want - n_ones]
            h_bin[zeros[order]] = 1.0
        h_bin = project_feasible(h_bin, n)
        c5_bin = compute_c5(h_bin)

        # Anneal
        if remaining() > 40:
            h_ann, c5_ann = swap_anneal(h_bin, n, 5000, 0.008, 0.0003, rng)
            if c5_ann < best_c5:
                best_c5 = c5_ann
                best_h = h_ann.copy()
                print("  Anneal: C5={:.8f}".format(best_c5))
            # SLSQP polish on annealed
            if remaining() > 20:
                h_pol, c5_pol = slsqp_opt(h_ann, n, 100, 200, 1e-12)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    print("  Anneal+SLSQP: C5={:.8f}".format(best_c5))

        # Second anneal from random binary
        h_bin2 = np.zeros(n)
        idx2 = rng.choice(n, size=n // 2, replace=False)
        h_bin2[idx2] = 1.0
        h_bin2 = project_feasible(h_bin2, n)
        if remaining() > 40:
            h_ann2, c5_ann2 = swap_anneal(h_bin2, n, 5000, 0.01, 0.0005, rng)
            if c5_ann2 < best_c5:
                best_c5 = c5_ann2
                best_h = h_ann2.copy()
                print("  Anneal2: C5={:.8f}".format(best_c5))

    # Phase E: Final aggressive polishing
    if remaining() > 25:
        n = best_n
        # Run one more SLSQP at very high beta
        h_final, c5_final = slsqp_opt(best_h, n, 1000, 300, 1e-14)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            print("  Final SLSQP: C5={:.8f}".format(best_c5))

        # One more coord descent
        if remaining() > 15:
            h_cd2, c5_cd2 = coord_descent(best_h, best_n, n_passes=2, rng=rng)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
