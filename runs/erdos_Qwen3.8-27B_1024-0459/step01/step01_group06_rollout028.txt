    # Coordinate descent polish at parent resolution
    if time.time() - t0 < time_limit - 100:
        h_cd, c5_cd = coordinate_descent_polish(best_h, n_parent, max_iters=200)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 2: Try higher resolutions
    resolutions = [190, 285, 380]
    for n_new in resolutions:
        if time.time() - t0 > time_limit - 120:
            break
        if n_new % 2 != 0:
            continue  # need sum = n/2 to be integer... actually n/2 can be non-integer

        h_interp = interpolate_to_resolution(best_h, best_n, n_new)
        bounds_n = [(0.0, 1.0)] * n_new
        cons_n = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_new,)}]

        # Soft-max optimisation at new resolution
        for beta in [15.0, 60.0, 250.0]:
            if time.time() - t0 > time_limit - 80:
                break
            try:
                res = minimize(
                    fun=obj_soft, x0=h_interp, jac=obj_soft_grad,
                    args=(n_new, beta), method='SLSQP',
                    bounds=bounds_n, constraints=cons_n,
                    options={'maxiter': 500, 'ftol': 1e-13}
                )
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_new)
                c5_new = compute_c5(h_new, n_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    best_n = n_new
                h_interp = project_feasible(np.clip(res.x, 0.0, 1.0), n_new)
            except Exception:
                pass

        # Hard-max refinement
        if time.time() - t0 < time_limit - 60:
            for _ in range(2):
                try:
                    res = minimize(
                        fun=obj_hard, x0=h_interp, jac=obj_hard_grad,
                        args=(n_new,), method='SLSQP',
                        bounds=bounds_n, constraints=cons_n,
                        options={'maxiter': 800, 'ftol': 1e-14}
                    )
                    h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_new)
                    c5_new = compute_c5(h_new, n_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        best_n = n_new
                    h_interp = project_feasible(np.clip(res.x, 0.0, 1.0), n_new)
                except Exception:
                    break

    # Phase 3: L-BFGS-B style with projection at best resolution
    if time.time() - t0 < time_limit - 60:
        n_cur = best_n
        for beta in [50.0, 200.0]:
            if time.time() - t0 > time_limit - 40:
                break
            h_l = best_h.copy()
            for step in range(3000):
                h_l = project_feasible(h_l, n_cur)
                val = obj_soft(h_l, n_cur, beta)
                g = obj_soft_grad(h_l, n_cur, beta)
                # Project gradient to satisfy sum constraint
                g_centered = g - np.mean(g)
                # Simple gradient step with backtracking
                lr = 0.5
                for _ in range(20):
                    h_trial = h_l - lr * g_centered
                    h_trial = project_feasible(h_trial, n_cur)
                    val_trial = obj_soft(h_trial, n_cur, beta)
                    if val_trial < val:
                        break
                    lr *= 0.5
                h_l = h_trial
                c5_trial = compute_c5(h_l, n_cur)
                if c5_trial < best_c5:
                    best_h = h_l.copy()
                    best_c5 = c5_trial

    # Phase 4: Multiple random restarts at best resolution
    if time.time() - t0 < time_limit - 40:
        rng = np.random.RandomState(seed)
        n_cur = best_n
        bounds_c = [(0.0, 1.0)] * n_cur
        cons_c = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_cur,)}]
        for i in range(60):
            if time.time() - t0 > time_limit - 20:
                break
            scale = 0.02 + 0.04 * (i / 30.0)
            h_pert = best_h + rng.randn(n_cur) * scale
            h_pert = project_feasible(h_pert, n_cur)
            try:
                res = minimize(
                    fun=obj_hard, x0=h_pert, jac=obj_hard_grad,
                    args=(n_cur,), method='SLSQP',
                    bounds=bounds_c, constraints=cons_c,
                    options={'maxiter': 300, 'ftol': 1e-13}
                )
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_cur)
                c5_new = compute_c5(h_new, n_cur)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 5: Final coordinate descent polish
    if time.time() - t0 < time_limit - 15:
        h_final, c5_final = coordinate_descent_polish(best_h, best_n, max_iters=300)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    # Phase 6: One more SLSQP pass
    if time.time() - t0 < time_limit - 8:
        bounds_f = [(0.0, 1.0)] * best_n
        cons_f = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (best_n,)}]
        try:
            res = minimize(
                fun=obj_hard, x0=best_h, jac=obj_hard_grad,
                args=(best_n,), method='SLSQP',
                bounds=bounds_f, constraints=cons_f,
                options={'maxiter': 500, 'ftol': 1e-14}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), best_n)
            c5_new = compute_c5(h_new, best_n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
