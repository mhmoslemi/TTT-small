                        k = int(np.argmax(1.0 - h))
                        if k != j:
                            h[k] = min(1.0, h[k] + diff)
                        else:
                            h[j] = old_v
                            continue
                    else:
                        k = int(np.argmax(h))
                        if k != j:
                            h[k] = max(0.0, h[k] + diff)
                        else:
                            h[j] = old_v
                            continue
                c = _c5(h)
                if c < best_local_c - 1e-12:
                    best_local_c = c
                    best_local = cv
            h[j] = best_local
            # Re-fix sum
            diff = 0.5 * n - h.sum()
            if abs(diff) > 1e-9:
                if diff > 0:
                    k = int(np.argmax(1.0 - h))
                    h[k] = min(1.0, h[k] + diff)
                else:
                    k = int(np.argmax(h))
                    h[k] = max(0.0, h[k] + diff)
            c = _c5(h)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def _binary_local(h_start, n, deadline, max_ops=50000, rng=None):
    h = _feasible(h_start, n)
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Fix count
    cnt = int(h_bin.sum())
    target = n // 2
    if cnt > target:
        vals = np.where(h_bin == 1)[0]
        # Remove the ones that the continuous h prefers to be 0
        order = np.argsort(h[vals])[:cnt - target]
        h_bin[vals[order]] = 0
    elif cnt < target:
        vals = np.where(h_bin == 0)[0]
        order = np.argsort(-h[vals])[:target - cnt]
        h_bin[vals[order]] = 1
    best_h = h_bin.copy()
    best_c = _c5(h_bin)
    scores = np.correlate(h_bin, 1.0 - h_bin, mode="full").copy()
    cur_max = float(np.max(scores))
    ops = 0
    stall = 0
    while ops < max_ops and time.time() < deadline:
        ops += 1
        _chk()
        # Find best swap: try to reduce the max score
        # Identify the shift with max score
        max_idx = int(np.argmax(scores))
        d = max_idx - (n - 1)
        if d > 0:
            # score = sum h[i]*(1-h[i+d])
            # To decrease: find i where h[i]=1, h[i+d]=0, and swap h[i+d] to 1 (if h[i] can be 0)
            # Or find i where h[i]=0, h[i+d]=1, swap to reduce
            mlen = n - d
            ones_before = np.where(h_bin[:mlen] == 1)[0]
            zeros_after = np.where(h_bin[d:] == 0)[0]
            if len(ones_before) > 0 and len(zeros_after) > 0:
                # Try swapping: h[i]=1->0 and h[i+d]=0->1 for some pair
                # This reduces score by 1 for this lag but may increase others
                i = int(ones_before[0])
                j = i + d  # h[j] is 0, set to 1; h[i] set to 0
                if h_bin[j] == 0 and h_bin[i] == 1:
                    h_bin[i] = 0
                    h_bin[j] = 1
                    new_scores = np.correlate(h_bin, 1.0 - h_bin, mode="full")
                    new_max = float(np.max(new_scores))
                    if new_max <= cur_max + 1e-9:
                        cur_max = new_max
                        scores = new_scores
                        if cur_max < float(best_c * n / 2.0) - 1e-9:
                            best_c = cur_max * (2.0 / n)
                            best_h = h_bin.copy()
                        stall = 0
                    else:
                        h_bin[i] = 1
                        h_bin[j] = 0
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        elif d < 0:
            dd = -d
            mlen = n - dd
            ones_after = np.where(h_bin[dd:] == 1)[0]
            zeros_before = np.where(h_bin[:mlen] == 0)[0]
            if len(ones_after) > 0 and len(zeros_before) > 0:
                j = int(ones_after[0]) + dd
                i = j - dd
                if h_bin[j] == 1 and h_bin[i] == 0:
                    h_bin[i] = 1
                    h_bin[j] = 0
                    new_scores = np.correlate(h_bin, 1.0 - h_bin, mode="full")
                    new_max = float(np.max(new_scores))
                    if new_max <= cur_max + 1e-9:
                        cur_max = new_max
                        scores = new_scores
                        if cur_max < float(best_c * n / 2.0) - 1e-9:
                            best_c = cur_max * (2.0 / n)
                            best_h = h_bin.copy()
                        stall = 0
                    else:
                        h_bin[i] = 0
                        h_bin[j] = 1
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        else:
            # Self-correlation, skip (should be 0 for binary)
            stall += 1
        if stall > 5000:
            break
    return best_h, best_c

def _get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None

def run(seed=42, budget_s=1300, **kwargs):
    global _run_deadline
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 15.0
    deadline = t0 + max(5.0, budget_s - reserve)
    _run_deadline = deadline

    rng = np.random.default_rng(seed)

    parent = _get_parent()
    if parent is not None and parent.size >= 4:
        n0 = int(parent.size)
        h0 = _feasible(parent, n0)
    else:
        n0 = 70
        h0 = _cosine_init(n0, 1.0, 0.0, rng)
        h0 = _feasible(h0, n0)

    best_h = h0.copy()
    best_c = _c5(h0)
    best_n = n0

    print(f"Initial: n={n0}, C5={best_c:.6f}", flush=True)

    # Resolutions to try (must be even for binary compatibility, but not required)
    resolutions = [n0]
    if n0 % 2 == 0:
        for mult in [2, 3, 4]:
            nn = n0 * mult
            if nn <= n0 * 6:
                resolutions.append(nn)
    # Also try n0+1, n0+2 if odd adjustments help
    if n0 not in [50, 60, 70, 80, 90, 100]:
        for alt in [70, 100, 140]:
            if alt > 4 and alt not in resolutions:
                resolutions.append(alt)
    resolutions = sorted(set(resolutions))

    # Precompute initializations per resolution
    inits = {}
    for nn in resolutions:
        cands = []
        if parent is not None:
            h_up = _upsample(parent, nn)
            h_up = _feasible(h_up, nn)
            cands.append(h_up)
        cands.append(_cosine_init(nn, 1.0, 0.0, rng))
        cands.append(_cosine_init(nn, 2.0, math.pi/3, rng))
        # Balanced pattern: evenly spaced
        n2 = nn // 2
        h_bal = np.zeros(nn)
        positions = np.linspace(0, nn, n2 + 1, dtype=int)[:n2]
        h_bal[positions] = 1.0
        cands.append(_feasible(h_bal, nn))
        # Random binary
        cands.append(_random_binary(nn, rng))
        # Deduplicate and keep best 4
        seen = []
        for c in cands:
            c = _feasible(c, nn)
            dup = False
            for s in seen:
                if np.allclose(s, c, atol=1e-6):
                    dup = True
                    break
            if not dup:
                seen.append(c)
        inits[nn] = seen[:4]

    # Time allocation
    time_per_res = max(20.0, (budget_s - 60.0) / max(1, len(resolutions)))
    # Give more time to primary resolution
    time_weights = {}
    for i, nn in enumerate(resolutions):
        if nn == n0:
            time_weights[nn] = time_per_res * 2.2
        elif nn <= 2 * n0:
            time_weights[nn] = time_per_res * 1.2
        else:
            time_weights[nn] = time_per_res * 0.7

    for nn in resolutions:
        if time.time() > deadline - 30:
            break
        t_res_start = time.time()
        t_res_end = min(deadline - 10, t_res_start + time_weights[nn])

        print(f"\nOptimizing n={nn} (budget {time_weights[nn]:.0f}s)", flush=True)

        for init_idx, h_init in enumerate(inits[nn]):
            if time.time() > t_res_end - 15:
                break
            remaining = t_res_end - time.time()
            if remaining < 15:
                break

            h_work = _feasible(h_init, nn)
            c_work = _c5(h_work)
            print(f"  Init {init_idx}: C5={c_work:.6f}", flush=True)

            # Stage 1: L-BFGS with smooth max (medium alpha)
            alpha1 = max(1.0, 0.03 * nn)
            maxit1 = max(30, int(min(200, 30.0 * (70.0 / nn))))
            t_stage = t_res_start + (t_res_end - t_res_start) * 0.3
            try:
                h1, c1 = _lbfgs_optimize(h_work, nn, alpha1, maxit1, t_stage)
                if c1 < c_work:
                    h_work, c_work = h1, c1
                    print(f"    After L-BFGS: C5={c_work:.6f}", flush=True)
            except Exception:
                pass

            # Stage 2: SLSQP with geometric alpha annealing
            alphas = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
            # Scale alphas by n/70
            scale = max(1.0, nn / 70.0)
            alphas = [a * scale for a in alphas]
            iters_each = max(20, int(min(150, 80.0 * (70.0 / nn))))
            t_stage2 = t_res_start + (t_res_end - t_res_start) * 0.85
            try:
                h2, c2 = _slsqp_optimize(h_work, nn, alphas, iters_each, t_stage2)
                if c2 < c_work:
                    h_work, c_work = h2, c2
                    print(f"    After SLSQP: C5={c_work:.6f}", flush=True)
            except Exception:
                pass

            # Stage 3: Coordinate descent polish
            n_cd = max(5, int(30.0 * (70.0 / nn)))
            try:
                h3, c3 = _coord_descent(h_work, nn, n_cd, t_res_end - 5)
                if c3 < c_work:
                    h_work, c_work = h3, c3
                    print(f"    After CoordDesc: C5={c_work:.6f}", flush=True)
            except Exception:
                pass

            # Update global best
            if c_work < best_c - 1e-12:
                best_c = c_work
                best_h = h_work.copy()
                best_n = nn
                print(f"  *** NEW BEST: n={nn}, C5={best_c:.6f} ***", flush=True)

    # Final: try binary local search on best solution
    if time.time() < deadline - 20:
        n_bin = best_n if best_n % 2 == 0 else best_n
        h_bin_start = _feasible(best_h, n_bin)
        try:
            hb, cb = _binary_local(h_bin_start, n_bin, deadline - 5, max_ops=30000, rng=rng)
            if cb < best_c - 1e-12:
                best_c = cb
                best_h = hb.copy()
                best_n = n_bin
                print(f"  Binary local: C5={best_c:.6f}", flush=True)
        except Exception:
            pass

    # Final feasibility check
    best_h = _feasible(best_h, best_n)
    best_c = _c5(best_h)

    # Validate
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10), "h out of bounds"
    assert abs(best_h.sum() - 0.5 * best_n) < 1e-6, f"sum constraint violated: {best_h.sum()} vs {0.5*best_n}"

    print(f"\nFinal: n={best_n}, C5={best_c:.6f}", flush=True)
    return best_h, float(best_c), int(best_n)
