    """Project h onto [0,1]^n ∩ {sum(h) = n/2}"""
    h = np.array(h, dtype=float)
    target = n / 2.0
    for _ in range(1000):
        # First clip to [0,1]
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - target) < 1e-15:
            break
        diff = target - s

    lo, hi = -1.0, 2.0
    for _ in range(100):

        x = np.clip(h - t, 0.0, 1.0)

        if s > target:
            lo = t
        else:
            hi = t

    # Final adjustment
    diff = target - np.sum(x)
    if abs(diff) > 1e-14:

        if np.sum(mask) > 0:

def sum_constraint(h, n):

def run_slsq(h_init, n, beta, max_iter=2000, ftol=1e-14):
    def obj(x):
        return smooth_max_obj(x, beta)
    
    def grad(x):
        return smooth_max_grad(x, beta)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:

def smooth_anneal(h_init, n, betas, max_iter_each=1500):

        rng = np.random.RandomState(42)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for it in range(n_iter):
        corr_vals = get_corr_values(h)
        cmax = np.max(corr_vals)
        # Find top shifts (as indices in the full correlation)
        top_k = np.argsort(corr_vals)[-5:]
        
        # Build a "pressure" vector: for each h[j], what direction reduces the active correlations?
        # For shift s >= 0: reducing h[j] for j in [0, n-1-s] helps if h[j] large and h[j+s] small
        #                   increasing h[j] for j in [s, n-1] helps if h[j+s] was small (makes 1-h[j+s] smaller)
        # Simplified: use subgradient of the max
        weight = np.zeros(2 * n - 1)
        for k_idx in top_k:
            weight[k_idx] += 1.0
        weight /= np.sum(weight)
        
        pressure = np.zeros(n)
        ns = 2 * n - 1
        for s in range(0, n):
            m = n - 1 + s
            w = weight[m]
            if w < 1e-12:
                continue
            pressure[s:] += w * (1.0 - h[:n - s])
            pressure[:n - s] -= w * h[s:]
        for mu in range(1, n):
            m = n - 1 - mu
            w = weight[m]
            if w < 1e-12:
                continue
            pressure[:n - mu] += w * (1.0 - h[mu:])
            pressure[mu:] -= w * h[:n - mu]
        
        # Negative gradient step (to reduce the objective)
        step_size = 0.01 / (1.0 + 0.01 * it)
        h_try = h - step_size * pressure
        h_try = project_feasible(h_try, n)
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            h = h_try
            best_c5 = c5_try
        
        # Also try random perturbations biased by the pressure
        if it % 5 == 0:
            h_rand = h + rng.randn(n) * 0.02 * (1.0 + np.abs(pressure))
            h_rand = project_feasible(h_rand, n)
            c5_rand = compute_c5(h_rand)
            if c5_rand < best_c5 - 1e-12:
                h = h_rand
                best_c5 = c5_rand
    
    return h, best_c5

def binary_swap_search(h, n, n_iter=5000, rng=None):
    """Try pushing towards binary and doing 1↔0 swaps"""
    if rng is None:
        rng = np.random.RandomState(123)
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    
    # First try to make it more binary
    for strength in [0.1, 0.3, 0.5, 0.7, 1.0]:
        h_bin = np.where(h > 0.5, 1.0, 0.0)
        h_blend = (1 - strength) * h + strength * h_bin
        h_blend = project_feasible(h_blend, n)
        if compute_c5(h_blend) < best_c5:
            h = h_blend
            best_c5 = compute_c5(h)
    
    # Now do swap-based local search
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for it in range(n_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_try = h.copy()
        h_try[i] = 1.0
        h_try[j] = 0.0
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    
    return h, best_c5

def generate_inits(n, rng, parent_h):
    inits = []
    
    # Interpolated parent
    if parent_h is not None and len(parent_h) != n:
        inits.append(project_feasible(interpolate_to_n(parent_h, n), n))
    elif parent_h is not None:
        inits.append(parent_h.copy())
    
    # Constant 0.5
    inits.append(np.ones(n) * 0.5)
    
    # Alternating binary
    if n % 2 == 0:
        alt = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n)])
        inits.append(alt.copy())
    
    # Block structure: [1...1, 0...0]
    block = np.zeros(n)
    block[:n//2] = 1.0
    inits.append(block.copy())
    
    # Two-block: [1...1, 0...0, 1...1, 0...0]
    if n % 4 == 0:
        two_block = np.zeros(n)
        two_block[:n//4] = 1.0
        two_block[n//2:3*n//4] = 1.0
        inits.append(two_block.copy())
    
    # Thue-Morse
    tm = np.array([1.0 if bin(i).count('1') % 2 == 0 else 0.0 for i in range(n)])
    inits.append(project_feasible(tm, n))
    
    # Random
    h_r = rng.rand(n) * 1.2
    inits.append(project_feasible(h_r, n))
    
    # Smooth random (low-pass filtered)
    h_smooth = rng.rand(n)
    kernel = np.array([0.1, 0.2, 0.4, 0.2, 0.1])
    h_smooth = np.convolve(h_smooth, kernel, mode='same')
    inits.append(project_feasible(h_smooth, n))
    
    # M-sequence inspired (for n close to 2^m - 1)
    for m in [5, 6, 7, 8]:
        L = 2**m - 1
        if L <= n:
            # Generate m-sequence
            seq = [1] * m
            state = 1
            seq_bits = []
            for _ in range(L):
                seq_bits.append(state & 1)
                new_bit = ((state << 1) ^ (((state >> (m-1)) * 0x95) & state + (state >> (m-1))) & ((1<<(m-1))-1)) & 1
                # Simplified: just use a linear pattern
                state = ((state << 1) | (1 if (bin(state).count('1') % 2 == 0) else 0)) & ((1<<m)-1)
            # Just use a pseudo-random pattern
            h_ms = np.zeros(n)
            h_ms[:L] = np.array(seq_bits, dtype=float)
            h_ms[L:] = h_ms[:n-L]  # repeat
            inits.append(project_feasible(h_ms, n))
    
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        if parent_h is None or len(parent_h) == 0:
            parent_h = None
    except Exception:
        parent_h = None
    
    if parent_h is not None:
        best_h = parent_h.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Polish parent at its resolution
    if time_left() > 100:
        betas = [10, 30, 80, 200, 500]
        h_pol, c5_pol = smooth_anneal(best_h, best_n, betas, max_iter_each=1500)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol
    
    # Phase 2: Multi-resolution optimization
    resolutions = sorted(set([best_n, 200, 240, 256, 320, 400, 512]))
    # Remove resolutions too close to best_n
    resolutions = [r for r in resolutions if r == best_n or abs(r - best_n) > 20]
    
    for n in resolutions:
        if time_left() < 60:
            break
        
        inits = generate_inits(n, rng, best_h if best_n != n else parent_h)
        if best_h is not None and len(best_h) != n:
            h_interp = interpolate_to_n(best_h, n)
            inits.insert(0, h_interp)
        
        # Limit inits to manage time
        n_inits = min(len(inits), 4 if time_left() > 400 else 3)
        
        for h0 in inits[:n_inits]:
            if time_left() < 30:
                break
            
            # Smooth annealing SLSQP
            if n <= 320:
                betas = [10, 40, 100, 300, 800]
            else:
                betas = [10, 40, 100, 300]
            
            h_opt, c5_opt = smooth_anneal(h0, n, betas, max_iter_each=2000)
            if c5_opt < best_c5 - 1e-10:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
    
    # Phase 3: Peak-targeted local search on best solution
    if time_left() > 60:
        n_iter = min(3000, int((time_left() - 30) / 0.05))
        best_h, best_c5 = peak_targeted_search(best_h, best_n, n_iter=max(n_iter, 500), rng=rng)
    
    # Phase 4: Try different resolutions with the current best
    if time_left() > 100:
        for n_try in [best_n - 40, best_n + 40, best_n + 80, best_n + 128]:
            if n_try <= 0 or time_left() < 50:
                continue
            if n_try in [best_n]:
                continue
            h_interp = interpolate_to_n(best_h, n_try)
            betas = [20, 80, 200, 500]
            h_opt, c5_opt = smooth_anneal(h_interp, n_try, betas, max_iter_each=2000)
            if c5_opt < best_c5 - 1e-10:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try
    
    # Phase 5: Final SLSQP polish with very high beta
    if time_left() > 40:
        h_fin, c5_fin = run_slsq(best_h, best_n, beta=2000, max_iter=3000, ftol=1e-16)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
    
    # Phase 6: Final peak-targeted refinement
    if time_left() > 30:
        n_iter = min(2000, int((time_left() - 15) / 0.03))
        best_h, best_c5 = peak_targeted_search(best_h, best_n, n_iter=max(n_iter, 200), rng=rng)
    
    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
