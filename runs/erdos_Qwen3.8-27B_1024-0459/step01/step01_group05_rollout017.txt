def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_box_sum(h, target):
    h = h.astype(float).copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:

def obj_and_grad(h, beta):

    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

            grad[j] += np.dot(w[w_idx], 1.0 - h[t_vals])
        
        # Part B: -sum_s w[n-1+s] * h[j-s] for j-s in [0, n-1]
        s_lo2 = max(-(n-1), j-(n-1))
        s_hi2 = min(n-1, j)
        if s_hi2 >= s_lo2:

            w_idx2 = s_vals2 + (n-1)
            t_vals2 = j - s_vals2  # = j-s, ranges over [0, n-1]
            grad[j] -= np.dot(w[w_idx2], h[t_vals2])
    
    return f, grad

def projected_gradient_descent(h0, target, beta, max_iter=5000, tol=1e-12):
    """PGD with backtracking line search."""
    h = h0.copy()
    n = len(h)
    step = 1.0
    prev_f = obj_and_grad(h, beta)[0]
    
    for it in range(max_iter):
        f, g = obj_and_grad(h, beta)
        
        # Project gradient onto feasible set (subtract mean for sum constraint)
        g_proj = g - np.mean(g)
        
        # Backtracking line search
        alpha = step
        for _ in range(30):

            h_new = project_box_sum(h_new, target)
            f_new, _ = obj_and_grad(h_new, beta)
            if f_new <= f - 1e-4 * alpha * np.dot(g_proj, h_new - h):
                break
            alpha *= 0.5
        
        h = h_new
        if abs(prev_f - f) < tol:
            break
        prev_f = f
        step = alpha * 1.5  # increase step if we used full step
        
    return h

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_obj_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    f = m + np.log(Z) / beta
    w = exps / Z
    grad = np.zeros(n)
    for j in range(n):
        s_lo = -j
        s_hi = n - 1 - j
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            w_idx = s_vals + (n - 1)
            t_vals = np.arange(0, n - j)
            grad[j] += float(np.dot(w[w_idx], 1.0 - h[t_vals]))
        s_lo2 = j - (n - 1)
        s_hi2 = j
        if s_hi2 >= s_lo2:
            s_vals = np.arange(s_lo2, s_hi2 + 1)
            w_idx = s_vals + (n - 1)
            t_vals = np.arange(0, j + 1)
            grad[j] -= float(np.dot(w[w_idx], h[t_vals]))
    return f, grad

def lp_obj_grad(h, p):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_pos = np.maximum(corr, 1e-30)
    f = np.sum(corr_pos ** p) ** (1.0 / p)
    w = corr_pos ** (p - 1) / np.sum(corr_pos ** p)
    grad = np.zeros(n)
    for j in range(n):
        s_lo = -j
        s_hi = n - 1 - j
        if s_hi >= s_lo:
            s_vals = np.arange(s_lo, s_hi + 1)
            w_idx = s_vals + (n - 1)
            t_vals = np.arange(0, n - j)
            grad[j] += f * float(np.dot(w[w_idx], 1.0 - h[t_vals]))
        s_lo2 = j - (n - 1)
        s_hi2 = j
        if s_hi2 >= s_lo2:
            s_vals = np.arange(s_lo2, s_hi2 + 1)
            w_idx = s_vals + (n - 1)
            t_vals = np.arange(0, j + 1)
            grad[j] -= f * float(np.dot(w[w_idx], h[t_vals]))
    return f, grad

def run_slsqp_analytical(h0, target, beta, maxiter, ftol, use_lp=False, lp_p=50.0):
    n = len(h0)
    h0 = project_box_sum(h0, target)
    bounds = [(0.0, 1.0)] * n

    if use_lp:
        def obj(h):
            return lp_obj_grad(h, lp_p)[0]
        def obj_grad(h):
            return lp_obj_grad(h, lp_p)[1]
    else:
        def obj(h):
            return smooth_max_obj(h, beta)
        def obj_grad(h):
            return smooth_max_obj_grad(h, beta)[1]

    def cons_fun(h):
        return np.sum(h) - target
    def cons_jac(h):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': maxiter, 'ftol': ftol})
        if result.x is not None:
            return project_box_sum(result.x, target)
    except Exception:
        pass
    return h0

def binary_local_search(h_bin, n, target, max_passes=20):
    """Local search on binary patterns: try swapping a 1 and 0."""
    h = h_bin.astype(float).copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    improved = True
    passes = 0
    while improved and passes < max_passes:
        improved = False
        passes += 1
        for i in ones.copy():
            for j in zeros.copy():
                if i == j:
                    continue
                h_trial = h.copy()
                h_trial[i] = 0.0
                h_trial[j] = 1.0
                c5_trial = compute_c5(h_trial)
                c5_curr = compute_c5(h)
                if c5_trial < c5_curr - 1e-14:
                    h = h_trial
                    improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    break
            if improved:
                break
    return h
