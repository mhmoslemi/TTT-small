    else:
        m = -lag
        # Case 1: j from 0 to n-m-1 (0 <= j < n + lag)
        if n + lag > 0:
            grad[:n + lag] += (1.0 - h[m:])
        # Case 2: j from m to n-1
        grad[m:] -= h[:n - m]
    return grad

def smooth_max_obj_and_grad(h, n, beta):
    """Compute smooth max objective and its gradient"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    # Softmax weights (numerically stable)
    exps = np.exp(beta * (corr - cmax))
    w = exps / exps.sum()
    # Objective
    obj = cmax + (1.0 / beta) * np.log(np.sum(exps))
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k(h, k)
    return obj, grad, cmax

def projected_gradient_anneal(h0, n, beta_start=5.0, beta_end=200.0, n_beta_steps=8,
                                max_iter_per_beta=500, ls_min=1e-8, ls_max=0.1):
    """Projected gradient method with smooth-max annealing"""
    h = np.array(h0, dtype=float).copy()
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for bstep in range(n_beta_steps):
        frac = bstep / max(n_beta_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        step = ls_max
        
        for it in range(max_iter_per_beta):
            obj, grad, cmax = smooth_max_obj_and_grad(h, n, beta)
            
            # Backtracking line search
            improved = False
            for ls in range(30):
                alpha = step * (0.5 ** ls)
                h_try = project_feasible(h - alpha * grad, n)
                c5_try = np.max(np.correlate(h_try, 1.0 - h_try, mode='full'))
                if c5_try <= cmax + 1e-14:
                    h = h_try
                    improved = True
                    break
            if improved:
                step = min(step * 1.5, ls_max)
            else:
                step *= 0.5
                if step < ls_min:
                    break
            
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
    
    return best_h, best_c5

def greedy_transfer_search(h0, n, max_steps=500000, time_limit=60.0):
    """Greedy transfer local search using gradient information"""
    h = np.array(h0, dtype=float).copy()
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stall = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        max_idx = int(np.argmax(corr))
        
        if max_val * (2.0 / n) < best_c5 - 1e-15:
            best_c5 = max_val * (2.0 / n)
            best_h = h.copy()
            stall = 0
            continue
        
        # Compute gradient at the max shift
        g = grad_corr_k(h, max_idx)
        
        # Find best transfer: decrease h at argmax(g), increase h at argmin(g)
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        
        if i == j:
            break
        
        # Try several delta values
        max_delta = min(h[i], 1.0 - h[j])
        found = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            delta = max_delta * frac
            if delta < 1e-12:
                continue
            h_try = h.copy()
            h_try[i] -= delta
            h_try[j] += delta
            h_try = np.clip(h_try, 0.0, 1.0)
            # Sum is preserved by construction, just clip
            h_try = make_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                found = True
                break
        
        if not found:
            stall += 1
            if stall > 10000:
                # Random kick
                h_try = h + np.random.randn(n) * 0.02
                h_try = make_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 * 1.01:
                    h = h_try
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
                    stall = 0
                else:
                    h = best_h.copy()
                    stall = 0
            elif stall % 10000 == 0:
                # Try random pair
                i2 = np.random.randint(n)
                j2 = np.random.randint(n)
                if i2 != j2:
                    delta = 0.05
                    direction = 1 if h[i2] > h[j2] else -1
                    h_try = h.copy()
                    h_try[i2] -= direction * delta
                    h_try[j2] += direction * delta
                    h_try = make_feasible(np.clip(h_try, 0, 1), n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        h = h_try
                        best_c5 = c5_try
                        best_h = h.copy()
                        stall = 0
        else:
            stall = 0
    
    return best_h, best_c5

def multi_shift_greedy(h0, n, max_steps=200000, time_limit=60.0):
    """Greedy search considering top-k shifts simultaneously"""
    h = np.array(h0, dtype=float).copy()
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stall = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        
        if max_val * (2.0 / n) < best_c5 - 1e-15:
            best_c5 = max_val * (2.0 / n)
            best_h = h.copy()
            stall = 0
            continue
        
        # Get top-3 shifts
        top_idx = np.argsort(corr)[-3:]
        
        # Average gradient over top shifts
        g = np.zeros(n)
        for k in top_idx:
            g += grad_corr_k(h, int(k))
        g /= len(top_idx)
        
        i = int(np.argmax(g))
        j = int(np.argmin(g))
        
        if i == j:
            break
        
        max_delta = min(h[i], 1.0 - h[j])
        found = False
        for frac in [1.0, 0.5, 0.25, 0.1]:
            delta = max_delta * frac
            if delta < 1e-12:
                continue
            h_try = h.copy()
            h_try[i] -= delta
            h_try[j] += delta
            h_try = make_feasible(np.clip(h_try, 0.0, 1.0), n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                found = True
                break
        
        if not found:
            stall += 1
            if stall > 50000:
                h = best_h + np.random.randn(n) * 0.03
                h = make_feasible(h, n)
                c5_try = compute_c5(h)
                if c5_try < best_c5 * 1.005:
                    best_c5 = c5_try
                    best_h = h.copy()
                stall = 0
        else:
            stall = 0
    
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def slsqp_optimize(h0, n, max_iter=3000):
    """SLSQP optimization with smooth-max objective"""
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj_fun(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = corr.max()
        # Smooth max with moderate beta
        beta = 50.0
        return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = corr.max()
        beta = 50.0
        exps = np.exp(beta * (corr - cmax))
        w = exps / exps.sum()
        obj = cmax + (1.0 / beta) * np.log(np.sum(exps))
        grad = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-12:
                grad += w[k] * grad_corr_k(x, k)
        return obj, grad
    
    def constraint_fun(x):
        return np.sum(x) - target
    
    try:
        obj0, grad0 = obj_grad(h0)
        res = minimize(obj_fun, h0, jac=lambda x: obj_grad(x)[1],
                      method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': constraint_fun}],
                      options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.success or res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def generate_structured_init(n, rng):
    """Generate diverse structured initial guesses"""
    inits = []
    
    # Block patterns
    if n % 4 == 0:
        inits.append(np.tile([1.0, 1.0, 0.0, 0.0], n // 4).copy())
        inits.append(np.tile([1.0, 0.0, 1.0, 0.0], n // 4).copy())
        inits.append(np.tile([1.0, 0.0, 0.0, 1.0], n // 4).copy())
    
    # Two-block
    h = np.zeros(n)
    h[:n//4] = 1.0
    h[n//2:n//2+n//4] = 1.0
    inits.append(h)
    
    # Thue-Morse
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(tm.copy())
    
    # Random
    for _ in range(2):
        inits.append(rng.rand(n).copy())
    
    # All at 0.5 (flat)
    inits.append(np.ones(n) * 0.5)
    
    return [make_feasible(h, n) for h in inits]

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Load parent
    global initial_h_values
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    
    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP on parent
    if remaining() > 100:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, max_iter=2000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: Projected gradient annealing on parent
    if remaining() > 200:
        t_limit = min(remaining() * 0.2, 200)
        h_pg, c5_pg = projected_gradient_anneal(best_h, best_n,
                                                beta_start=3.0, beta_end=300.0,
                                                n_beta_steps=10,
                                                max_iter_per_beta=300)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    
    # Phase 3: Greedy transfer search
    if remaining() > 100:
        t_limit = min(remaining() * 0.2, 150)
        h_g, c5_g = greedy_transfer_search(best_h, best_n,
                                           max_steps=1000000, time_limit=t_limit)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
    
    # Phase 4: Multi-shift greedy
    if remaining() > 100:
        t_limit = min(remaining() * 0.15, 120)
        h_ms, c5_ms = multi_shift_greedy(best_h, best_n,
                                          max_steps=500000, time_limit=t_limit)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()
    
    # Phase 5: Try other resolutions
    other_ns = [best_n]
    for n_try in [best_n - 20, best_n + 20, best_n + 40, 200, 240, 280, 320]:
        if n_try >= 40 and n_try not in other_ns:
            other_ns.append(n_try)
    other_ns = sorted(set(other_ns))
    
    for n_try in other_ns:
        if remaining() < 120:
            break
        if n_try == best_n:
            continue
        
        # Interpolate current best to new resolution
        h_interp = interpolate_to_n(best_h, n_try)
        h_interp = make_feasible(h_interp, n_try)
        
        # Also try parent interpolated
        inits = [h_interp]
        if h_parent is not None and len(h_parent) != n_try:
            h_p = make_feasible(interpolate_to_n(h_parent, n_try), n_try)
            inits.append(h_p)
        
        # Structured inits
        for h_s in generate_structured_init(n_try, rng)[:3]:
            inits.append(h_s)
        
        for h0 in inits:
            if remaining() < 60:
                break
            
            # SLSQP
            if remaining() > 60:
                h_opt, c5_opt = slsqp_optimize(h0, n_try, max_iter=2000)
                if h_opt is not None and c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_try
            
            # Projected gradient
            if remaining() > 60:
                h_pg, c5_pg = projected_gradient_anneal(h0, n_try,
                                                        beta_start=5.0, beta_end=200.0,
                                                        n_beta_steps=6,
                                                        max_iter_per_beta=200)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = n_try
    
    # Phase 6: Fine-tuning with SLSQP (high precision)
    if remaining() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, max_iter=5000)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 7: Final greedy polish
    if remaining() > 30:
        t_limit = min(remaining() - 15, 120)
        if t_limit > 10:
            h_g, c5_g = greedy_transfer_search(best_h, best_n,
                                               max_steps=2000000, time_limit=t_limit)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
    
    # Phase 8: Very fine projected gradient
    if remaining() > 30:
        h_pg, c5_pg = projected_gradient_anneal(best_h, best_n,
                                                beta_start=100.0, beta_end=500.0,
                                                n_beta_steps=4,
                                                max_iter_per_beta=500,
                                                ls_min=1e-10, ls_max=0.01)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    
    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
