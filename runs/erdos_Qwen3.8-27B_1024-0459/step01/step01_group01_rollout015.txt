    """Compute all C_j values."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    C = corr[n-1 : 2*n-1]  # lags 0 to n-1
    return C

def compute_grad(h, C, tau):
    """Compute gradient of soft-max objective."""
    n = len(h)
    maxC = np.max(C)
    exp_C = np.exp((C - maxC) / tau)
    W = exp_C / np.sum(exp_C)
    
    # g[k] = sum_j W[j] * dC_j/dh[k]
    # dC_j/dh[k] = dx * [(1-h[k+j])*1{k+j<n} - h[k-j]*1{k>=j}]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        # First term: sum_{j=0}^{n-1-k} W[j] * (1-h[k+j])
        s1 = 0.0
        for j in range(n - k):
            s1 += W[j] * (1.0 - h[k+j])
        # Second term: sum_{j=0}^{k} W[j] * h[k-j]
        s2 = 0.0
        for j in range(k + 1):
            s2 += W[j] * h[k-j]
        g[k] = dx * (s1 - s2)
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr[n-1:]) * dx)

def compute_C_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[n-1:] * dx

def softmax_weights(C, tau):
    maxC = np.max(C)
    e = np.exp((C - maxC) / tau)
    s = np.sum(e)
    if s < 1e-300:
        return np.full(len(C), 1.0/len(C))
    return e / s

def compute_grad(h, W):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    omh = 1.0 - h
    for k in range(n):
        g[k] = dx * (np.dot(W[:n-k], omh[k:]) - np.dot(W[:k+1], h[k::-1]))
    return g

def project_box_sum(x, target):
    lo = -1.0 - x.max()
    hi = 1.0 - x.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    return np.interp(x_new, x_old, h)

def pgd_refine(h0, target, taus, iters_per_tau, verbose=False):
    n = len(h0)
    dx = 2.0 / n
    h = project_box_sum(h0.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for tau in taus:
        alpha = tau / (2.0 * dx)
        C = compute_C_all(h)
        for it in range(iters_per_tau):
            W = softmax_weights(C, tau)
            g = compute_grad(h, W)
            h_new = project_box_sum(h - alpha * g, target)
            C = compute_C_all(h_new)
            h = h_new
            c5_val = float(np.max(C))
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
            if it % 10000 == 0 and verbose:
                print(f"  tau={tau:.4f}, it={it}, c5={c5_val:.8f}")
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get parent
    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(90, 0.5)
    
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at parent resolution
    n0 = len(parent_h)
    target0 = n0 / 2.0
    taus1 = np.geomspace(0.5, 0.01, 15)
    h1, c5_1 = pgd_refine(parent_h, target0, taus1, 300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
    
    # Phase 2: Optimize at 2x resolution
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.4:
        n1 = 2 * n0
        target1 = n1 / 2.0
        x0 = upsample(best_h, n1)
        x0 = project_box_sum(x0, target1)
        taus2 = np.geomspace(0.3, 0.005, 20)
        h2, c5_2 = pgd_refine(x0, target1, taus2, 300)
        if c5_2 < best_c5:

            best_h = h2

    elapsed = time.time() - t0
    if elapsed < budget_s * 0.6:

        target2 = n2 / 2.0
        x0 = upsample(best_h, n2)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr[n-1:]) * dx)


def compute_C_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr[n-1:] * dx


def softmax_weights(C, tau):
    maxC = np.max(C)
    e = np.exp((C - maxC) / tau)
    s = np.sum(e)
    if s < 1e-300:
        return np.full(len(C), 1.0 / len(C))
    return e / s


def build_masks(n):
    J = np.arange(n)
    mask1 = (J[None, :] <= (n - 1 - J)[:, None]).astype(np.float64)
    mask2 = (J[None, :] <= J[:, None]).astype(np.float64)
    return mask1, mask2


def compute_grad_mat(h, W, dx, mask1, mask2):
    F1 = mask1 * W[None, :]
    F2 = mask2 * W[None, :]
    return dx * (F1 @ (1.0 - h) - F2 @ h)


def project_box_sum(x, target):
    lo = -1.0 - float(x.max())
    hi = 1.0 - float(x.min()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def upsample_fn(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    return np.interp(x_new, x_old, h)


def pgd_refine(h0, target, taus, iters_per_tau):

    dx = 2.0 / n
    mask1, mask2 = build_masks(n)

    best_c5 = float(np.max(compute_C_all(h)))

    for tau in taus:
        alpha = tau / (2.0 * dx)
        C = compute_C_all(h)
        for it in range(iters_per_tau):
            W = softmax_weights(C, tau)
            g = compute_grad_mat(h, W, dx, mask1, mask2)
            h_new = project_box_sum(h - alpha * g, target)
            C = compute_C_all(h_new)
            h = h_new
            c5_val = float(np.max(C))
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
    return best_h, best_c5


def slsqp_obj(x, n, dx, tau):
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode="full")
    C = corr[n - 1:] * dx
    m = float(np.max(C))
    return m + tau * float(np.log(np.sum(np.exp((C - m) / tau))))


def slsqp_eq(x, n):
    return np.array([x.sum() - n / 2.0])


def slsqp_jac(x, n):
    return np.ones((1, n))

    h = np.zeros(n)
