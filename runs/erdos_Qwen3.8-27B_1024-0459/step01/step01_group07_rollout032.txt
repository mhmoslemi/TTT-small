    return np.max(conv) * dx

def smooth_obj_grad(h, n, temp):
    dx = 2.0 / n
    v = 1.0 - h
    c = np.correlate(h, v, mode='full') * dx
    max_c = np.max(c)
    shifted = (c - max_c) / temp
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = temp * (np.log(sum_exp) + max_c / temp)
    
    # Add penalty for sum constraint
    target = n / 2.0
    s = np.sum(h)
    pen = 1000.0 * (s - target) ** 2
    obj += pen
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s_k = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-15:
            continue
        # Term 1: (1-h[i-s_k]) for valid i

        i_max1 = min(n - 1, s_k + n - 1)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def smooth_obj_grad(h, n, temp):
    dx = 2.0 / n
    v = 1.0 - h
    c = np.correlate(h, v, mode='full') * dx
    max_c = np.max(c)
    shifted = (c - max_c) / temp
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = temp * (np.log(sum_exp) + max_c / temp)
    
    # Penalty for sum constraint
    target = n / 2.0
    s = np.sum(h)
    pen_weight = 100.0
    obj += pen_weight * (s - target) ** 2
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s_k = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-16:
            continue
        # Term 1: (1-h[i-s_k]) for valid i: s_k <= i <= s_k+n-1
        i_min1 = max(0, s_k)
        i_max1 = min(n - 1, s_k + n - 1)
        if i_min1 <= i_max1:
            j_idx = np.arange(i_min1, i_max1 + 1) - s_k
            grad[i_min1:i_max1+1] += w * (1.0 - h[j_idx])
        # Term 2: -h[i+s_k] for valid i: -s_k <= i <= n-1-s_k
        i_min2 = max(0, -s_k)
        i_max2 = min(n - 1, n - 1 - s_k)
        if i_min2 <= i_max2:
            j_idx2 = np.arange(i_min2, i_max2 + 1) + s_k
            grad[i_min2:i_max2+1] -= w * h[j_idx2]
    
    grad += 2.0 * pen_weight * (s - target)
    
    return obj, grad

        h = np.clip(h, 0.0, 1.0)

def compute_c5(h, n_points):

def smooth_obj_grad(h, n, temp):
    dx = 2.0 / n
    v = 1.0 - h
    c = np.correlate(h, v, mode='full') * dx
    max_c = np.max(c)
    shifted = (c - max_c) / temp
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    obj = temp * (np.log(sum_exp) + max_c / temp)
    
    target = n / 2.0
    s = np.sum(h)
    pw = 100.0
    obj += pw * (s - target) ** 2
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s_k = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-16:
            continue
        i_min1 = max(0, s_k)
        i_max1 = min(n - 1, s_k + n - 1)
        if i_min1 <= i_max1:

        i_min2 = max(0, -s_k)

            j_idx2 = np.arange(i_min2, i_max2 + 1) + s_k
            grad[i_min2:i_max2+1] -= w * h[j_idx2]
    
    grad += 2.0 * pw * (s - target)
    return obj, grad

            grad += w * (1.0 - 2.0 * h)
        elif s_k > 0:
            grad[s_k:] += w * (1.0 - h[:n - s_k])
            grad[:n - s_k] -= w * h[s_k:]
        else:
            p = -s_k
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
    grad += 2.0 * pw * (s - target)
    return obj, grad


def obj_func(x, n, temp):
    return smooth_obj_grad(x, n, temp)[0]


def grad_func(x, n, temp):
    return smooth_obj_grad(x, n, temp)[1]


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = h < 1.0 - 1e-14
            cnt = float(np.sum(free))
            if cnt < 1e-12:
                break
            step = diff / cnt
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = h > 1e-14
            cnt = float(np.sum(free))
            if cnt < 1e-12:
                break
            step = diff / cnt
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample_fn(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def lbbf(h0, n, temp, maxit):
    res = minimize(
        obj_func, h0,
        jac=grad_func,
        args=(n, temp),
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-7,
                 'maxfun': maxit * max(2, n) * 3}
    )
    return res.x


def gen_periodic_init(n, rng):
    period = rng.randint(max(3, n // 10), max(4, n // 3))
    h = np.zeros(n)
    for p in range(period):
        if rng.random() < 0.5:
            h[p::period] = 1.0
    return h


def gen_spectral_init(n, rng):
    nf = np.fft.rfftfreq(n)
    ph = rng.uniform(0, 2.0 * np.pi, len(nf))
    am = np.zeros(len(nf))
    for j in range(1, len(nf)):
        am[j] = rng.exponential(0.15) * np.exp(-3.0 * j / n)
    h = np.fft.irfft(am * np.exp(1j * ph) * n, n=n) + 0.5
    return np.clip(h, 0.0, 1.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is None or len(h0) == 0:
        n0 = 120
        h0 = np.full(n0, 0.5)
    else:
        n0 = len(h0)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_val = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Polish initial resolution with smooth max (annealing)
    temps1 = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01]
    for temp in temps1:
        if budget_s - (time.time() - t0) < 50:
            break
        try:
            h_try = lbbf(best_h.copy(), best_n, temp, 300)
            h_try = project_feasible(h_try, best_n)
            v = compute_c5(h_try, best_n)
            if v < best_val:
                best_h, best_val = h_try, v
        except Exception:
            pass

    # Phase 2: Upsample to higher resolutions and optimize
    target_resolutions = [140, 200, 280, 400]
    for n_new in target_resolutions:
        if budget_s - (time.time() - t0) < 150:
            break
        h_f = upsample_fn(best_h, best_n, n_new)
        h_f = project_feasible(h_f, n_new)
        temps2 = [0.25, 0.1, 0.04, 0.015, 0.008]
        for temp in temps2:
            if budget_s - (time.time() - t0) < 30:
                break
            try:
                h_try = lbbf(h_f, n_new, temp, 250)
                h_try = project_feasible(h_try, n_new)
                v = compute_c5(h_try, n_new)
                if v < best_val:
                    best_h, best_val, best_n = h_try, v, n_new
                h_f = best_h.copy() if best_n == n_new else h_try
            except Exception:
                pass

    # Phase 3: Diverse random restarts at best resolution
    n_opt = best_n
    attempt = 0
    max_attempts = 150
    while attempt < max_attempts:
        if budget_s - (time.time() - t0) < 100:
            break
        st = attempt % 7
        if st == 0:
            h_r = rng.uniform(0.0, 1.0, n_opt)
        elif st == 1:
            h_r = rng.binomial(1, 0.5, n_opt).astype(float)
        elif st == 2:
            h_r = gen_periodic_init(n_opt, rng)
        elif st == 3:
            h_r = gen_spectral_init(n_opt, rng)
        elif st == 4:
            h_r = np.clip(best_h + rng.normal(0.0, 0.1, n_opt), 0.0, 1.0)
        elif st == 5:
            # Block pattern with random lengths
            h_r = np.zeros(n_opt)
            pos = 0
            while pos < n_opt:
                blen = rng.randint(2, max(3, n_opt // 12))
                gap = rng.randint(2, max(3, n_opt // 12))
                h_r[pos:pos + blen] = 1.0
                pos += blen + gap
        else:
            # Two-scale
            nr = max(3, n_opt // 4)
            h_coarse = rng.uniform(0.0, 1.0, nr)
            h_r = np.interp(np.arange(n_opt) * (nr - 1.0) / max(1, n_opt - 1),
                            np.arange(nr), h_coarse)
            h_r += rng.normal(0.0, 0.08, n_opt)
            h_r = np.clip(h_r, 0.0, 1.0)

        h_r = project_feasible(h_r, n_opt)
        for temp in [0.15, 0.04, 0.01]:
            try:
                h_try = lbbf(h_r, n_opt, temp, 120)
                h_try = project_feasible(h_try, n_opt)
                v = compute_c5(h_try, n_opt)
                if v < best_val:
                    best_h, best_val = h_try, v
                h_r = best_h.copy() if v < best_val else h_try
            except Exception:
                break
        attempt += 1

    # Phase 4: Local perturbation / coordinate search
    for _ in range(50):
        if budget_s - (time.time() - t0) < 30:
            break
        npert = rng.randint(1, max(2, best_n // 12) + 1)
        idx = rng.choice(best_n, size=npert, replace=False)
        h_p = best_h.copy()
        h_p[idx] = np.clip(h_p[idx] + rng.normal(0.0, 0.18, npert), 0.0, 1.0)
        h_p = project_feasible(h_p, best_n)
        for temp in [0.04, 0.01]:
            try:
                h_try = lbbf(h_p, best_n, temp, 100)
                h_try = project_feasible(h_try, best_n)
                v = compute_c5(h_try, best_n)
                if v < best_val:
                    best_h, best_val = h_try, v
                h_p = best_h.copy() if v < best_val else h_try
            except Exception:
                break

    # Phase 5: Try near-binary solutions at higher resolution
    if budget_s - (time.time() - t0) > 100:
        for n_high in [350, 500]:
            if budget_s - (time.time() - t0) < 60:
                break
            h_bin = upsample_fn(best_h, best_n, n_high)
            h_bin = project_feasible(h_bin, n_high)
            # Soft-threshold toward binary
            h_bin = 0.5 * np.tanh((h_bin - 0.5) * 4.0) + 0.5
            h_bin = project_feasible(h_bin, n_high)
            for temp in [0.1, 0.03, 0.01]:
                if budget_s - (time.time() - t0) < 20:
                    break
                try:
                    h_try = lbbf(h_bin, n_high, temp, 200)
                    h_try = project_feasible(h_try, n_high)
                    v = compute_c5(h_try, n_high)
                    if v < best_val:
                        best_h, best_val, best_n = h_try, v, n_high
                    h_bin = best_h.copy() if best_n == n_high else h_try
                except Exception:
                    pass

    # Phase 6: Final aggressive polish
    if budget_s - (time.time() - t0) > 25:
        for temp in [0.01, 0.005, 0.002]:
            if budget_s - (time.time() - t0) < 12:
                break
            try:
                h_try = lbbf(best_h.copy(), best_n, temp, 800)
                h_try = project_feasible(h_try, best_n)
                v = compute_c5(h_try, best_n)
                if v < best_val:
                    best_h, best_val = h_try, v
            except Exception:
                break

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    return (best_h, best_val, best_n)
