import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50.0, 50.0)
    return m + float(np.log(np.sum(np.exp(shifted)))) / beta

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                h[mask] += diff / int(np.sum(mask))
        else:
            mask = h > 0.0
            if np.any(mask):
                h[mask] += diff / int(np.sum(mask))
    np.clip(h, 0.0, 1.0, out=h)
    return h

def thue_morse_binary(n):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        t[i] = float(bin(i).count('1') % 2)
    num_ones = int(np.sum(t))
    target_ones = n // 2
    if num_ones > target_ones:
        ones = np.where(t == 1)[0]
        t[ones[:num_ones - target_ones]] = 0
    elif num_ones < target_ones:
        zeros = np.where(t == 0)[0]
        t[zeros[:target_ones - num_ones]] = 1
    return t

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def sa_binary(n, n_iter, t_high, t_low, seed, h_init=None):
    rng = np.random.RandomState(seed)
    if h_init is not None:
        h = h_init.copy()
    else:
        h = random_binary(n, seed)
    
    n_half = n // 2
    ones_count = int(np.sum(h > 0.5))
    if ones_count != n_half:
        h = random_binary(n, seed)
    
    curr_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    
    for it in range(n_iter):
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        
        diff = new_c5 - curr_c5
        if diff > 0:
            frac = it / max(n_iter - 1, 1)
            T = t_high * (t_low / t_high) ** frac
            if T > 1e-15 and rng.random() < np.exp(-diff / T):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        else:
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5

def slsqp_refine(h0, n, beta, max_iter):
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        xc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(xc, 1.0 - xc, mode='full') * dx
        return smooth_max(corr, beta)
    
    def cons_f(x):
        return np.sum(x) - target
    
    def cons_g(x):
        return np.ones(n)
    
    try:
        res = minimize(obj, h0.copy(), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': cons_f, 'jac': cons_g}],
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return project_sum(h0, target)

def lbfgsb_refine(h0, n, beta, max_iter):

    dx = 2.0 / n
    lam = 10.0

    except Exception:
        pass
    return project_sum(h0, target)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50.0, 50.0)
    return m + float(np.log(np.sum(np.exp(shifted)))) / beta

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                h[mask] += diff / int(np.sum(mask))
        else:
            mask = h > 0.0
            if np.any(mask):
                h[mask] += diff / int(np.sum(mask))
    np.clip(h, 0.0, 1.0, out=h)
    return h

def thue_morse_binary(n):
    t = np.zeros(n, dtype=float)
    for i in range(n):
        t[i] = float(bin(i).count('1') % 2)
    num_ones = int(np.sum(t))
    target_ones = n // 2
    if num_ones > target_ones:
        ones = np.where(t == 1)[0]
        t[ones[:num_ones - target_ones]] = 0
    elif num_ones < target_ones:
        zeros = np.where(t == 0)[0]
        t[zeros[:target_ones - num_ones]] = 1
    return t

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h

def sa_binary(n, n_iter, t_high, t_low, seed, h_init=None):
    rng = np.random.RandomState(seed)
    if h_init is not None:
        h = h_init.copy()
    else:
        h = random_binary(n, seed)
    curr_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    for it in range(n_iter):
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        diff = new_c5 - curr_c5
        if diff > 0:
            frac = it / max(n_iter - 1, 1)
            T = t_high * (t_low / t_high) ** frac
            if T > 1e-15 and rng.random() < np.exp(-diff / T):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        else:
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    return best_h, best_c5

def _lbfgsb_pen_obj(x, n, beta, lam):
    dx = 2.0 / n
    xc = np.clip(x, 0.0, 1.0)
    corr = np.correlate(xc, 1.0 - xc, mode='full') * dx
    sm = smooth_max(corr, beta)
    target = n / 2.0
    penalty = lam * (np.sum(x) - target) ** 2
    return sm + penalty

def continuous_optimize(h0, n, beta, max_iter, lam=50.0):
    target = n / 2.0
    h_start = project_sum(h0, target)
    res = minimize(_lbfgsb_pen_obj, h_start, args=(n, beta, lam),
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    if res.x is not None:
        return project_sum(res.x, target)
    return h_start
