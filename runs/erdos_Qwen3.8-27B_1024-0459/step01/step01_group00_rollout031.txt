            delta = t_new - t_old
            h[p] = t_new
            
            # Fix sum: redistribute delta to other coordinates
            target_sum = n / 2.0
            current_sum = np.sum(h)
            correction = target_sum - current_sum  # = -delta
            if abs(correction) > 1e-14:
                # Distribute to coordinates with room
                mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                mask[p] = False
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h[mask] += correction / cnt
                    h = np.clip(h, 0.0, 1.0)
                    # Re-fix sum if clipping changed it
                    for _ in range(100):
                        s = np.sum(h)
                        d = target_sum - s
                        if abs(d) < 1e-12:
                            break
                        mask2 = (h > 1e-10) & (h < 1.0 - 1e-10)
                        cnt2 = np.count_nonzero(mask2)
                        if cnt2 == 0:
                            break
                        h[mask2] += d / cnt2
                        h = np.clip(h, 0.0, 1.0)
            
            # Recompute correlation
            corr = compute_corr(h)
            c5_new = np.max(corr) * (2.0 / n)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
        
        if verbose and sweep % 10 == 0:
            print(f"Sweep {sweep}: C5 = {best_c5:.6f}, improved={improved}")
        if not improved and sweep > 5:
            break
    
    return h, best_c5

∇f(h) = dx * ∇corr[k*](h) where k* = argmax_k corr[k]

∇corr[k*](h)[j] for the full correlation element at position m:
The correlation element at position m (lag = m - (n-1)):
corr[m] = sum_i h[i]*(1-h[i+lag])

∂corr[m]/∂h[j] = (1-h[j+lag])*I(0≤j+lag≤n-1) - h[j-lag]*I(0≤j-lag≤n-1)

This is the same b vector I computed above! So the subgradient is just b * dx (for the active m).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, corr, alpha):
    """One projected subgradient step for the max of correlations."""
    n = len(h)
    m_active = np.argmax(corr)
    dx = 2.0 / n
    p = None
    # Compute gradient of the active correlation w.r.t. h
    # corr[m] = sum_i h[i]*(1-h[i+lag]) where lag = m - (n-1)
    # d/dh[j] = (1-h[j+lag])*I(0<=j+lag<n) - h[j-lag]*I(0<=j-lag<n)
    lag = m_active - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j + lag < n:
            val += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            val -= h[j - lag]
        grad[j] = val
    h_new = h - alpha * dx * grad
    return make_feasible(h_new, n)

def subgradient_descent(h, n, n_iters=2000, alpha_init=0.02, alpha_decay=0.999):
    """Projected subgradient descent on the max correlation."""
    h = make_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = alpha_init
    corr = compute_corr_vec(h)
    for it in range(n_iters):
        h_try = subgradient_step(h, corr, alpha)
        c5_try = np.max(compute_corr_vec(h_try)) * (2.0 / n)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
            h = h_try
        else:
            alpha *= alpha_decay
        corr = compute_corr_vec(h)
        if alpha < 1e-8:
            break
    return best_h, best_c5

def coordinate_descent(h, n, n_sweeps=100):
    """Coordinate descent exploiting 1D convex structure of max of linears."""
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    corr = compute_corr_vec(h)
    
    for sweep in range(n_sweeps):
        improved = False
        for p in range(n):
            t_old = h[p]
            # Compute b[m] and a[m] such that corr[m] = a[m] + b[m]*h[p]
            # b[m] = (1-h[p+lag])*I(0<=p+lag<n) - h[p-lag]*I(0<=p-lag<n)
            # where lag = m - (n-1)
            b = np.zeros(2 * n - 1)
            # p+lag in [0,n-1] means n-1-p <= m <= 2n-2-p
            i1s = max(0, n - 1 - p)
            i1e = min(2 * n - 2, 2 * n - 2 - p)
            if i1s <= i1e:
                idxs1 = np.arange(i1s, i1e + 1)
                h_off1 = p + idxs1 - (n - 1)
                b[idxs1] += (1.0 - h[h_off1])
            # p-lag in [0,n-1] means p <= m <= p+n-1
            i2s = max(0, p)
            i2e = min(2 * n - 2, p + n - 1)
            if i2s <= i2e:
                idxs2 = np.arange(i2s, i2e + 1)
                h_off2 = p - idxs2 + (n - 1)
                b[idxs2] -= h[h_off2]
            
            a = corr - b * t_old
            
            # Minimize f(t) = max_m(a[m] + b[m]*t) over t in [0,1]
            t_grid = np.linspace(0.0, 1.0, 50)
            f_vals = np.max(a[None, :] + b[None, :] * t_grid[:, None], axis=1)
            t_best = t_grid[np.argmin(f_vals)]
            
            # Refine
            if 0.02 < t_best < 0.98:
                t_lo = max(0.0, t_best - 0.02)
                t_hi = min(1.0, t_best + 0.02)
                t_fine = np.linspace(t_lo, t_hi, 50)
                f_fine = np.max(a[None, :] + b[None, :] * t_fine[:, None], axis=1)
                t_best = t_fine[np.argmin(f_fine)]
            
            t_new = t_best
            if abs(t_new - t_old) < 1e-13:
                continue
            
            h[p] = t_new
            # Fix sum constraint
            target_sum = n / 2.0
            for _ in range(100):
                s = np.sum(h)
                d = target_sum - s
                if abs(d) < 1e-13:
                    break
                mask = (h > 1e-12) & (h < 1.0 - 1e-12)
                mask[p] = False
                cnt = np.count_nonzero(mask)
                if cnt == 0:
                    break
                h[mask] += d / cnt
                h = np.clip(h, 0.0, 1.0)
            
            corr = compute_corr_vec(h)
            c5_new = np.max(corr) * (2.0 / n)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
        
        if not improved and sweep > 3:
            break
    
    return h, best_c5

def swap_local_search(h, n, n_trials=5000):
    """Local search using pairwise swaps (preserves sum exactly)."""
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    corr = compute_corr_vec(h)
    rng = np.random.RandomState(12345)
    
    for trial in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if abs(h[i] - h[j]) < 1e-12:
            continue
        # Swap
        h[i], h[j] = h[j], h[i]
        c5_try = compute_c5(h)
        if c5_try > best_c5 + 1e-14:
            # Revert
            h[i], h[j] = h[j], h[i]
        else:
            best_c5 = c5_try
    return h, best_c5

def perturb_local_search(h, n, n_trials=3000, step=0.05):
    """Local search with small perturbations and sum redistribution."""
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(67890)
    
    for trial in range(n_trials):
        h_try = h.copy()
        n_mod = rng.randint(1, 4)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        # Fix sum
        target = n / 2.0
        for _ in range(200):
            d = target - np.sum(h_try)
            if abs(d) < 1e-13:
                break
            mask = (h_try > 1e-12) & (h_try < 1.0 - 1e-12)
            cnt = np.count_nonzero(mask)
            if cnt == 0:

            h_try[mask] += d / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
    return h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def generate_structured_inits(n, seed_val):
    rng = np.random.RandomState(seed_val)
    inits = []
    # Block pattern
    if n % 4 == 0:
        inits.append(np.tile([1.0, 1.0, 0.0, 0.0], n // 4))

    if n % 2 == 0:

        h[:n//4] = 1.0
        h[n//2:n//2+n//4] = 1.0
        inits.append(h)

    # Golden ratio based

    for j in range(n // 2):
        h_gold[positions[j]] = 1.0
    inits.append(h_gold)
    # Random
    inits.append(rng.rand(n))
    return inits

def smooth_max_objective(x, n, beta):
    corr = compute_corr_vec(x)
    cmax = np.max(corr)
    # Numerically stable soft max
    shifted = beta * (corr - cmax)
    e = np.exp(np.clip(shifted, -50, 0))
    return (cmax + (1.0/beta) * np.log(np.sum(e))) * (2.0/n)

def slsqp_optimize(n, h_init, max_iter=3000):
    def obj(x):
        return smooth_max_objective(x, n, 200.0)
    def obj_grad(x):
        corr = compute_corr_vec(x)
        beta = 200.0
        cmax = np.max(corr)
        shifted = beta * (corr - cmax)
        e = np.exp(np.clip(shifted, -50, 0))
        w = e / np.sum(e)
        # Gradient of soft max
        # d/dcmax = 1 - sum(w) = 0 (since w sums to 1)
        # dcorr[m]/dx = grad of corr[m]
        # Full gradient: sum_m w[m] * dcorr[m]/dx
        # dcorr[m]/dx[j] = (1-h[j+lag])*I - h[j-lag]*I
        grad = np.zeros(n)
        for m in range(2*n-1):
            if w[m] < 1e-10:
                continue
            lag = m - (n-1)
            g = np.zeros(n)
            for j in range(n):
                if 0 <= j + lag < n:
                    g[j] += (1.0 - x[j+lag])
                if 0 <= j - lag < n:
                    g[j] -= x[j-lag]
            grad += w[m] * g
        return (2.0/n) * grad
    
    def constraint_eq(x):
        return np.sum(x) - n / 2.0
    
    def constraint_eq_grad(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_grad}]
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None
