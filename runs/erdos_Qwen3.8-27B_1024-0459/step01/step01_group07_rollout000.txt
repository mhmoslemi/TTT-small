import numpy as np

def exact_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)


def feasible_projection(x, n, target, use_sym=False):
    y = np.asarray(x, dtype=float).reshape(n)
    if use_sym:
        y = 0.5 * (y + y[::-1])

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        p = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(p))
        if s > target:
            lo = mid
        else:
            hi = mid

    p = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    if use_sym:
        p = 0.5 * (p + p[::-1])

    s = float(np.sum(p))
    diff = target - s
    if abs(diff) > 1e-11:
        for _ in range(300):
            s = float(np.sum(p))
            diff = target - s
            if abs(diff) < 1e-13:
                break
            if diff > 0.0:
                free = (p < 1.0 - 1e-12)
            else:
                free = (p > 1e-12)
            cnt = float(np.sum(free))
            if cnt <= 0.0:
                break
            p[free] = np.clip(p[free] + diff / cnt, 0.0, 1.0)
            if use_sym:
                p = 0.5 * (p + p[::-1])

    p = np.clip(p, 0.0, 1.0)
    if use_sym:
        p = 0.5 * (p + p[::-1])
    return p


def compute_f_values(h, n):
    oh = 1.0 - h
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = float(np.dot(h, oh))

    p = np.empty(n - 1, dtype=float)
    q = np.empty(n - 1, dtype=float)
    for m in range(1, n):
        L = n - m
        p[m - 1] = float(np.dot(h[:L], oh[m:]))
        q[m - 1] = float(np.dot(h[m:], oh[:L]))

    vals[1:n] = p
    vals[n:] = q
    return vals, p, q


def compute_smooth_value(h, n, temp):
    vals, _, _ = compute_f_values(h, n)
    fmax = float(np.max(vals))
    dx = 2.0 / n
    if temp <= 1e-12:
        return fmax * dx, fmax * dx
    ex = np.exp((vals - fmax) / temp)
    s = float(np.sum(ex))
    if s <= 0.0:
        s = 1.0
    smooth = temp * (fmax + math.log(s)) * dx
    return smooth, fmax * dx


def compute_smooth_grad(h, n, temp):
    vals, p, q = compute_f_values(h, n)
    fmax = float(np.max(vals))
    dx = 2.0 / n

    if temp <= 1e-12:
        w = np.zeros_like(vals)
        idx = np.nonzero(vals >= fmax - 1e-12)[0]
        if len(idx) == 0:
            idx = [int(np.argmax(vals))]
        w[idx] = 1.0 / len(idx)
        s = float(np.sum(np.exp((vals - fmax) / temp)))
    else:
        ex = np.exp((vals - fmax) / temp)
        s = float(np.sum(ex))
        if s <= 0.0:
            s = 1.0
        w = ex / s

    smooth = temp * (fmax + math.log(s)) * dx
    exact = fmax * dx

    grad = w[0] * (1.0 - 2.0 * h)
    wp = w[1:n]
    wq = w[n:]

    for m in range(1, n):
        a = float(wp[m - 1])
        b = float(wq[m - 1])
        if a == 0.0 and b == 0.0:
            continue
        L = n - m
        grad[:L] += a * (1.0 - h[m:]) - b * h[m:]
        grad[m:] += b * (1.0 - h[:L]) - a * h[:L]

    grad *= dx
    return smooth, exact, grad

def upsample_h(h, n):
    h = np.asarray(h, dtype=float)
    m = len(h)
    if m == 0:
        return 0.5 * np.ones(n, dtype=float)
    if m == n:
        return h.copy()
    x_old = (np.arange(m) + 0.5) * (2.0 / m)
    x_new = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x_new, x_old, h)


def binary_round(h, n):
    h = np.asarray(h, dtype=float)
    if n % 2 == 0:
        target = n // 2
    else:
        target = int(round(n / 2.0))
    out = np.zeros(n, dtype=float)
    if target > 0:
        idx = np.argsort(-h)
        out[idx[:target]] = 1.0
    return out


def binary_local_search(h_prior, n, deadline, rng):
    if n < 12 or n % 2 != 0 or time.time() > deadline - 30.0:
        return h_prior, exact_c5(h_prior, n)

    h = binary_round(h_prior, n)
    best_h = h.copy()
    best_max = int(np.max(np.correlate(h, 1.0 - h, mode='full')))
    cur_max = best_max
    T = 2.0
    no_improve = 0

    while time.time() < deadline - 2.0:
        ones = np.nonzero(h > 0.5)[0]
        zeros = np.nonzero(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        p = int(ones[int(rng.randint(len(ones)))])
        q = int(zeros[int(rng.randint(len(zeros)))])

        h[p] = 0.0
        h[q] = 1.0
        mx = int(np.max(np.correlate(h, 1.0 - h, mode='full')))

        accept = False
        if mx <= cur_max:
            accept = True
        elif T > 1e-8 and rng.random() < math.exp(-max(0.0, mx - cur_max) * 2.0 / T):
            accept = True

        if accept:
            cur_max = mx
            if mx < best_max:
                best_max = mx
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[p] = 1.0
            h[q] = 0.0

        if no_improve > 5000:
            h = best_h.copy()
            cur_max = best_max
            T = 1.8
            no_improve = 0

        if (no_improve % 1000) == 0:
            T = max(0.08, T * 0.93)

    c = float(best_max) * (2.0 / n)
    return best_h, c


def analyze_initial(h_parent, n, rng):
    starts = []
    x = (np.arange(n) + 0.5) * (2.0 / n)
    y = x - 1.0

    starts.append(1.0 - np.abs(x - 1.0))
    starts.append(np.abs(x - 1.0))

    for a in (0.25, 0.5, 0.75, 1.0):
        q = (a + 1.0) / 2.0
        starts.append(1.0 - q * np.power(np.abs(x - 1.0), a))

    for a in (0.5, 1.0, 2.0):
        starts.append(np.power(np.abs(x - 1.0), a))

    for B in (-0.5, -0.25, 0.0, 0.25, 0.5):
        A = 0.5 - B / 3.0
        starts.append(A + B * y * y)

    for C in (-0.35, -0.15, 0.15, 0.35):
        A = 0.5 - 0.2 * C
        starts.append(A + C * y ** 4)

    for f in (1, 2, 3):
        for phase in (0.0, math.pi):
            starts.append(0.5 + 0.5 * np.cos(math.pi * f * y + phase))
        starts.append(0.5 + 0.35 * np.sin(math.pi * f * y))

    for f in (1, 2, 3, 5):
        phase = 2.0 * math.pi * rng.uniform()
        amp = rng.uniform(0.15, 0.40)
        starts.append(0.5 + amp * np.sin(math.pi * f * x + phase)
                      + rng.normal(0.0, 0.05, n))

    for _ in range(3):
        starts.append(0.5 + 0.35 * rng.normal(0.0, 1.0, n))

    if h_parent is not None and len(h_parent) > 0:
        if len(h_parent) == n:
            hp = np.array(h_parent, dtype=float).copy()
        else:
            hp = upsample_h(h_parent, n)
        hp = np.clip(hp, 0.0, 1.0)
        starts.append(hp)
        starts.append(1.0 - hp)
        starts.append(hp[::-1].copy())
        starts.append(0.5 * (hp + hp[::-1]))
        for sc in (0.03, 0.06, 0.10, 0.18):
            starts.append(hp + sc * rng.normal(0.0, 1.0, n))

    proj = []
    target = n / 2.0
    for h in starts:
        h = feasible_projection(h, n, target, False)
        proj.append(h)
    return proj


def cand_key(item):
    return item[0]


def build_candidate_list(starts, n, kmax=10):
    target = n / 2.0
    raw = []
    for h in starts:
        c = exact_c5(h, n)
        if math.isfinite(c):
            raw.append((c, h))
        hs = feasible_projection(h, n, target, True)
        cs = exact_c5(hs, n)
        if math.isfinite(cs):
            raw.append((cs, hs))

    raw.sort(key=cand_key)

    seen = set()
    unique = []
    for c, h in raw:
        key = tuple(np.round(h, 4).tolist())
        if key in seen:
            continue
        seen.add(key)
        unique.append((c, h))
        if len(unique) >= kmax:
            break
    return unique


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    global_deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = 1e100
    best_n = 0

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None

    n_main = 70
    if parent is not None and len(parent) > 0:
        n_main = int(len(parent))
        if n_main < 16:
            n_main = 70
        if n_main > 160:
            n_main = 140
        if n_main % 2 != 0:
            n_main += 1

    x0 = (np.arange(n_main) + 0.5) * (2.0 / n_main)
    if parent is not None and len(parent) > 0:
        h_par = upsample_h(parent, n_main)
    else:
        h_par = 1.0 - np.abs(x0 - 1.0)
    h_par = feasible_projection(h_par, n_main, n_main / 2.0, False)
    best_h = h_par.copy()
    best_c = exact_c5(best_h, n_main)
    best_n = n_main

    reserve = max(140.0, 0.25 * float(budget_s))
    main_deadline = max(t0 + 60.0, global_deadline - reserve)

    starts = analyze_initial(parent, n_main, rng)
    cands = build_candidate_list(starts, n_main, 10)

    stage_times = [180, 150, 120, 100, 90, 70, 60, 50, 40, 40]
    for i, item in enumerate(cands):
        if time.time() >= main_deadline - 20.0:
            break
        c0, h0 = item
        stage = float(stage_times[i]) if i < len(stage_times) else 35.0
        end = time.time() + stage
        if end > main_deadline:
            end = main_deadline
        if time.time() > end - 12.0:
            break
        use_sym = (i % 2 == 0)
        h2, c2 = smooth_descent(h0, n_main, end, use_sym=use_sym, rng=rng)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = n_main

    n2 = 0
    if n_main <= 100:
        n2 = 2 * n_main
    elif n_main <= 140:
        n2 = 2 * n_main

    if n2 <= 200 and time.time() < global_deadline - 180.0:
        x2 = (np.arange(n2) + 0.5) * (2.0 / n2)
        y2 = x2 - 1.0
        h_hi = upsample_h(best_h, n2)
        h_hi = feasible_projection(h_hi, n2, n2 / 2.0, False)

        hi_starts = [
            h_hi,
            feasible_projection(1.0 - np.abs(x2 - 1.0), n2, n2 / 2.0, True),
            feasible_projection(0.5 + 0.5 * np.cos(math.pi * y2), n2, n2 / 2.0, True),
            feasible_projection(0.5 - 0.5 * np.cos(math.pi * y2), n2, n2 / 2.0, True),
        ]

        hh = build_candidate_list(hi_starts, n2, 3)
        j = 0
        for item in hh:
            if time.time() >= global_deadline - 140.0:
                break
            stage = 80.0 - 10.0 * j
            if stage < 25.0:
                stage = 25.0
            end = min(global_deadline - 130.0, time.time() + stage)
            if time.time() > end - 12.0:
                break
            use_sym = (j % 2 == 0)
            h2, c2 = smooth_descent(item[1], n2, end, use_sym=use_sym, rng=rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = n2
            j += 1

    if n_main % 2 == 0 and n_main <= 96 and time.time() < global_deadline - 180.0:
        end = min(global_deadline - 110.0, time.time() + 90.0)
        h_bin, c_bin = binary_local_search(best_h, n_main, end, rng)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()
            best_n = n_main

        if time.time() < global_deadline - 90.0:
            h_smooth = feasible_projection(h_bin + 0.04 * rng.normal(0.0, 1.0, n_main),
                                           n_main, n_main / 2.0, False)
            end = min(global_deadline - 80.0, time.time() + 55.0)
            h2, c2 = smooth_descent(h_smooth, n_main, end, use_sym=False, rng=rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = n_main

    if time.time() < global_deadline - 50.0:
        end = min(global_deadline - 20.0, time.time() + 110.0)
        if time.time() < end - 15.0:
            h2, c2 = smooth_descent(best_h, best_n, end, use_sym=False, rng=rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = best_n

    if time.time() < global_deadline - 30.0:
        end = min(global_deadline - 8.0, time.time() + 60.0)
        if time.time() < end - 10.0:
            h2, c2 = smooth_descent(best_h, best_n, end, use_sym=True, rng=rng)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = best_n

    if best_h is None:
        best_n = n_main
        best_h = feasible_projection(1.0 - np.abs((np.arange(best_n) + 0.5) * 2.0 / best_n - 1.0),
                                     best_n, best_n / 2.0, False)
    else:
        best_h = feasible_projection(best_h, best_n, best_n / 2.0, False)

    best_c = exact_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
