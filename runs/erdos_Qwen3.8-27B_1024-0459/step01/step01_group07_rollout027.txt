def monograd(h, n, shift):
    """Subgradient of a single shift value (times dx) w.r.t. h"""
    dx = 2.0 / n
    g = np.zeros(n)
    if shift >= 0:
        m = shift
        # h[j] as first factor: j-m in [0, n-1-m] → j in [m, n-1]
        # contribution: (1-h[j-m])
        j_start, j_end = m, n
        g[j_start:j_end] += (1.0 - h[j_start - m:j_end - m]) * dx
        # h[j] in (1-h[i]): j in [0, n-1-m], contribution: -h[j+m]
        j_start, j_end = 0, n - m
        if j_end > 0:
            g[j_start:j_end] -= h[j_start + m:j_end + m] * dx
    else:
        p = -shift
        # h[j] as first factor: j in [0, n-1-p], contribution: (1-h[j+p])
        j_start, j_end = 0, n - p
        if j_end > 0:
            g[j_start:j_end] += (1.0 - h[j_start + p:j_end + p]) * dx
        # h[j] in (1-h[i+p]): j-p in [0, n-1-p] → j in [p, n-1], contribution: -h[j-p]
        j_start, j_end = p, n
        g[j_start:j_end] -= h[j_start - p:j_end - p] * dx
    return g

def smooth_max_obj(h, n, beta=30.0):
    conv = all_shifts(h, n)
    mx = np.max(conv)
    # exp(β*(conv - mx)) to avoid overflow
    shifted = np.clip(conv - mx, -100.0/beta, 0.0) * beta
    logsum = np.log(np.sum(np.exp(shifted))) / beta
    return mx + logsum

def smooth_max_grad(h, n, beta=30.0):
    conv = all_shifts(h, n)
    mx = np.max(conv)
    shifted = np.clip(conv - mx, -100.0/beta, 0.0) * beta

            continue
        shift = idx - (n - 1)
        grad += weights[idx] * monograd(h, n, shift)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def all_shifts(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
            cnt = int(np.sum(free))
            if cnt == 0:
                break
            h[free] = np.clip(h[free] + diff / cnt, 0.0, 1.0)
        else:
            free = h > 0.0
            cnt = int(np.sum(free))
            if cnt == 0:
                break
            h[free] = np.clip(h[free] + diff / cnt, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        interior = (h > 0.05) & (h < 0.95)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += (target - s) / cnt
        else:
            h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def monograd(h, n, shift):
    dx = 2.0 / n
    g = np.zeros(n)
    if shift >= 0:
        m = shift
        if m < n:
            js, je = m, n
            g[js:je] += (1.0 - h[js - m:je - m]) * dx
            js, je = 0, n - m
            if je > 0:
                g[js:je] -= h[js + m:je + m] * dx
    else:
        p = -shift
        if p < n:
            js, je = 0, n - p
            if je > 0:
                g[js:je] += (1.0 - h[js + p:je + p]) * dx
            js, je = p, n
            g[js:je] -= h[js - p:je - p] * dx
    return g


def slsqp_opt(h0, n, maxiter=400, ftol=1e-13):
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        return compute_c5(x, n)

    def grad(x):
        conv = np.correlate(x, 1.0 - x, mode='full')
        k = int(np.argmax(conv))
        m = k - (n - 1)
        return monograd(x, n, m)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        hx = project_feasible(res.x, n)
        return hx, compute_c5(hx, n)
    except Exception:
        return h0.copy(), compute_c5(h0, n)


def smooth_max_obj_val(h, n, beta):
    conv = all_shifts(h, n)
    mx = np.max(conv)
    sh = np.clip(conv - mx, -80.0 / beta, 0.0) * beta
    return mx + np.log(np.sum(np.exp(sh))) / beta


def smooth_max_slsq(h0, n, maxiter=300, beta=60.0, ftol=1e-13):
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        conv = all_shifts(x, n)
        mx = np.max(conv)
        sh = np.clip(conv - mx, -80.0 / beta, 0.0) * beta
        return mx + np.log(np.sum(np.exp(sh))) / beta

    def grad(x):
        conv = all_shifts(x, n)
        mx = np.max(conv)
        sh = np.clip(conv - mx, -80.0 / beta, 0.0) * beta
        w = np.exp(sh)
        wsum = np.sum(w)
        w = w / wsum
        g = np.zeros(n)
        for idx in range(len(conv)):
            if w[idx] < 1e-10:
                continue
            m = idx - (n - 1)
            g += w[idx] * monograd(x, n, m)
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        hx = project_feasible(res.x, n)
        return hx, compute_c5(hx, n)
    except Exception:
        return h0.copy(), compute_c5(h0, n)


def subgrad_descent(h0, n, n_iter=200, lr_init=0.005):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    lr = lr_init
    for it in range(n_iter):
        conv = all_shifts(h, n)
        k_active = np.argmax(conv)
        m = k_active - (n - 1)
        g = monograd(h, n, m)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        if it % 30 == 29:
            lr *= 0.85
    return best_h, best_val


def coord_descent(h0, n, n_passes=50, n_cands=5):
    h = h0.copy()
    best_val = compute_c5(h, n)
    for p in range(n_passes):
        improved = False
        order = np.random.permutation(n)
        for j in order:
            old_v = h[j]
            cands = np.linspace(max(0, old_v - 0.3), min(1, old_v + 0.3), n_cands)
            cands = np.unique(np.concatenate([cands, [old_v]]))
            best_lv = best_val
            best_lv_h = old_v
            for cv in cands:
                h[j] = cv
                hx = project_feasible(h, n)
                hv = compute_c5(hx, n)
                if hv < best_lv:
                    best_lv = hv
                    best_lv_h = cv
            h[j] = best_lv_h
            h = project_feasible(h, n)
            if best_lv < best_val - 1e-12:
                best_val = best_lv
                improved = True
            else:
                h[j] = old_v
        if not improved:
            break
    return h, best_val


def make_random_start(n, rng, strategy=0):
    if strategy == 0:
        h = rng.uniform(0, 1, n)
    elif strategy == 1:
        h = rng.binomial(1, 0.5, n).astype(float)
    elif strategy == 2:
        # Block pattern
        block = rng.randint(3, max(4, n // 10))
        pattern = np.zeros(block)
        pattern[:block // 2] = 1.0
        h = np.tile(pattern, n // block + 1)[:n]
        h += rng.normal(0, 0.1, n)
    elif strategy == 3:
        # Sinusoidal
        x = np.linspace(0, 2, n, endpoint=False)
        freq = rng.choice([1, 2, 3, 4, 5])
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.sin(np.pi * freq * x + phase)
        h += rng.normal(0, 0.05, n)
    elif strategy == 4:
        # Two-scale pattern
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.2 * np.sin(6 * np.pi * x + 0.5)
        h += rng.normal(0, 0.05, n)
    else:
        h = rng.uniform(0.1, 0.9, n)
    return project_feasible(h, n)
