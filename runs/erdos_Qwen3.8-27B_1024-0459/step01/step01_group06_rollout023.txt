            x = np.arange(n) / n
            h_pert = best_h + scale * np.sin(2 * np.pi * freq * x + phase)
        h_pert = project_feasible(h_pert, n)
        try:
            res = minimize(
                fun=hard_obj,
                x0=h_pert,
                jac=hard_grad_analytic,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-13}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Stage 6: Higher resolution attempt
    if elapsed() < budget_s - 150:
        for n_alt in [n * 2, n + 24, n + 16]:
            if elapsed() > budget_s - 50:
                break
            if n_alt > 400:
                continue
            # Interpolate to new resolution
            x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0.0, 2.0, n_alt, endpoint=False) + 1.0 / n_alt
            h_alt_init = np.interp(x_new, x_old, best_h, period=2.0)
            h_alt_init = project_feasible(h_alt_init, n_alt)

            bounds_alt = [(0.0, 1.0)] * n_alt
            cons_alt = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_alt,)}]

            # Softmax at higher res
            c5_alt = compute_c5(h_alt_init, n_alt)
            for beta in [10.0, 50.0, 200.0, 500.0]:
                if elapsed() > budget_s - 40:
                    break
                try:
                    res = minimize(
                        fun=soft_obj,
                        x0=h_alt_init,
                        jac=soft_grad,
                        args=(n_alt, beta),
                        method='SLSQP',
                        bounds=bounds_alt,
                        constraints=cons_alt,
                        options={'maxiter': 600, 'ftol': 1e-13}
                    )
                    h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_alt)
                    c5_new = compute_c5(h_new, n_alt)
                    if c5_new < c5_alt:
                        h_alt_init = h_new
                        c5_alt = c5_new
                except Exception:
                    pass

            # Hard max refinement
            if elapsed() < budget_s - 30:
                try:
                    res = minimize(
                        fun=hard_obj,
                        x0=h_alt_init,
                        jac=hard_grad_analytic,
                        args=(n_alt,),
                        method='SLSQP',
                        bounds=bounds_alt,
                        constraints=cons_alt,
                        options={'maxiter': 1000, 'ftol': 1e-14}
                    )
                    h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_alt)
                    c5_new = compute_c5(h_new, n_alt)
                    if c5_new < c5_alt:
                        h_alt_init = h_new
                        c5_alt = c5_new
                except Exception:
                    pass

            if c5_alt < best_c5:
                best_h = h_alt_init
                best_c5 = c5_alt
                n = n_alt
                bounds = bounds_alt
                cons = cons_alt

    # Stage 7: Additional coordinate descent on best
    if elapsed() < budget_s - 20:
        best_h = coord_descent_step(best_h, n, max_sweeps=5)
        best_c5 = compute_c5(best_h, n)

    # Stage 8: Final SLSQP polish
    if elapsed() < budget_s - 10:
        try:
            res = minimize(
                fun=hard_obj,
                x0=best_h,
                jac=hard_grad_analytic,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 800, 'ftol': 1e-15}
            )
            h_final = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c5_final = compute_c5(h_final, n)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except Exception:
            pass

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    # Validate
    try:
        val = evaluate_erdos_solution(best_h, n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, n)
