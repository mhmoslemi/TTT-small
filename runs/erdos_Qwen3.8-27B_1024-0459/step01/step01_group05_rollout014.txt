import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_constraint(h, target):
    h = np.array(h, dtype=float)
    # Bisection on lambda
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h + lam, 0.0, 1.0)
    # Fix residual
    res = target - np.sum(h)
    if abs(res) > 1e-12:
        if res > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += res / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(corr)
    s = beta * (corr - m)
    s = np.clip(s, -50.0, 50.0)
    e = np.exp(s)
    tot = np.sum(e)
    w = e / tot
    val = m + np.log(tot) / beta
    return val, w

    beta_list = [3.0, 8.0, 20.0, 50.0, 150.0, 400.0]
    phase1_end = 0.35
    n_betas = len(beta_list)
    for bi, b in enumerate(beta_list):
        deadline_frac = 0.05 + phase1_end * (bi + 1) / n_betas
        deadline = t0 + budget_s * deadline_frac
        n_iter = 30000 if b < 50 else 15000
        h1 = project_grad_optimize(h1, n1, tgt1, b, n_iter, deadline)
        try_update(h1, n1)

    # Phase 2: Try multiple resolutions from interpolated parent
    res_list = [100, 128, 160, 200, 256]
    phase2_start = 0.36
    phase2_end = 0.72
    for ri, n_try in enumerate(res_list):
        if remaining_frac() < 0.29:
            break
        h_interp = interp_to_resolution(best_h, n_try)
        h_interp = project_to_constraint(h_interp, n_try / 2.0)
        tgt_try = n_try / 2.0

        d1 = t0 + budget_s * (phase2_start + phase2_end * (2*ri + 1) / (2 * len(res_list)))
        h_a = project_grad_optimize(h_interp, n_try, tgt_try, 30.0, 20000, d1)
        try_update(h_a, n_try)

        if remaining_frac() < 0.28:
            break
        d2 = t0 + budget_s * (phase2_start + phase2_end * (2*ri + 2) / (2 * len(res_list)))
        h_b = project_grad_optimize(h_a, n_try, tgt_try, 200.0, 15000, d2)
        try_update(h_b, n_try)

    # Phase 3: Threshold refinement on best
    if remaining_frac() > 0.25:
        n_b = best_n
        h_thr = threshold_and_fix(best_h, n_b / 2.0, thresh=0.03)
        deadline = t0 + budget_s * 0.80
        h_thr_opt = project_grad_optimize(h_thr, n_b, n_b / 2.0, 100.0, 15000, deadline)
        try_update(h_thr_opt, n_b)

    # Phase 4: Multi-start perturbations
    n_p = best_n
    tgt_p = n_p / 2.0
    n_perts = 12
    phase4_end = 0.93
    for pi in range(n_perts):
        if remaining_frac() < 0.08:
            break
        frac_slice = (phase4_end - 0.82) / n_perts
        deadline = t0 + budget_s * (0.82 + frac_slice * (pi + 1))
        scale = 0.04 / (1.0 + pi * 0.3)
        h_pert = best_h + np.random.randn(n_p) * scale
        h_pert = project_to_constraint(h_pert, tgt_p)
        h_opt = project_grad_optimize(h_pert, n_p, tgt_p, 80.0, 12000, deadline)
        try_update(h_opt, n_p)
        # Threshold variant
        if pi % 3 == 0:
            h_thr2 = threshold_and_fix(h_opt, tgt_p, thresh=0.02)
            if remaining_frac() > 0.07:
                h_opt2 = project_grad_optimize(h_thr2, n_p, tgt_p, 150.0, 8000, deadline)
                try_update(h_opt2, n_p)

    # Phase 5: Try a different resolution with random init
    if remaining_frac() > 0.12:
        n_rand = 128
        h_rand = np.random.rand(n_rand) * 0.3 + 0.35
        h_rand = project_to_constraint(h_rand, n_rand / 2.0)
        deadline = t0 + budget_s * 0.95
        h_rand_opt = project_grad_optimize(h_rand, n_rand, n_rand / 2.0, 20.0, 20000, deadline)
        try_update(h_rand_opt, n_rand)
        if remaining_frac() > 0.10:
            h_rand_opt2 = project_grad_optimize(h_rand_opt, n_rand, n_rand / 2.0, 150.0, 15000,
                                                t0 + budget_s * 0.96)
            try_update(h_rand_opt2, n_rand)

    # Phase 6: Final sharp refinement
    if remaining_frac() > 0.04:
        h_fin = project_grad_optimize(best_h, best_n, best_n / 2.0, 500.0,
                                       10000, t0 + budget_s * 0.98)
        try_update(h_fin, best_n)

    # Final: ensure exact constraint satisfaction
    best_h = project_to_constraint(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, best_n)
