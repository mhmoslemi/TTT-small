import numpy as np
from scipy.optimize import minimize
import time

_GP = {}

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h

def upsample_lin(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def smooth_obj(h, tau, n, mu_sum, mu_bin):
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode="full")
    m = np.max(corr)
    sm = m + tau * np.log(np.sum(np.exp((corr - m) / tau)))
    dx = 2.0 / n
    result = float(sm * dx)
    s = np.sum(hc) - n / 2.0
    result += mu_sum * s * s
    if mu_bin > 0:
        result += mu_bin * float(np.sum(hc * (1.0 - hc)))
    return result

def _obj(x):
    n = len(x)
    return smooth_obj(x, _GP['tau'], n, _GP['mu_sum'], _GP['mu_bin'])

def _con_fun(x):
    return np.array([x.sum() - len(x) / 2.0])

def _con_jac(x):

def cd_polish(h, n, time_limit):

        for sweep in range(4):
            if time.time() - t0 > time_limit:
                break
            for i in range(n):
                if time.time() - t0 > time_limit:
                    break
                orig = h[i]
                for d in [-step, step]:
                    new_val = np.clip(orig + d, 0.0, 1.0)
                    if new_val == orig:
                        continue
                    h[i] = new_val
                    c = compute_c5(h)
                    if c < best_c5 - 1e-15:
                        best_c5 = c
                    else:
                        h[i] = orig
    return h, best_c5

def binary_cd(h, n, time_limit):

            else:
                h = h.copy()
                h[i] = orig
                h = fix_sum(h, n)
    return h, best_c5

def run_slsqp(x0, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new

def run_lbfgsb(x0, n, maxiter, ftol):
    bounds = list(zip(np.zeros(n), np.ones(n)))
    res = minimize(_obj, x0, method='L-BFGS-B', bounds=bounds,

    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = fix_sum(h_new, n)
    return h_new
