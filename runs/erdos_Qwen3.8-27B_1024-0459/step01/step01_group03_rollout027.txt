def project_density(x, target):

    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    n = x.size

        return x
    # Bisection for the water-filling parameter
    lo, hi = -max(s-target, 0) - 2, min(target-s, 0) + 2
    # Actually, simpler: add a constant to get the right sum
    # h = clip(x + t, 0, 1), find t such that sum = target

        else:

        active_corr = corr[top_idx].astype(float)
        grads = np.array([grad_for_lag(best, int(t) - center) for t in top_idx])
        mean_g = grads.mean(axis=0)
        pos_idx = np.argsort(-mean_g)[:top_l]
        neg_idx = np.argsort(mean_g)[:top_l]
        best_delta_overall = 0.0
        best_ij = None
        best_val = float(np.max(active_corr))
        for i in neg_idx:
            i = int(i)
            if best[i] >= 1.0 - 1e-12:
                continue
            for j in pos_idx:
                j = int(j)
                if i == j:
                    continue
                if best[j] <= 1e-12:
                    continue
                cap = min(1.0 - best[i], best[j])
                if cap < 1e-12:
                    continue
                d_vec = grads[:, i] - grads[:, j]
                d_opt, v_opt = optimal_pair_delta(active_corr, d_vec, 1e-12, cap - 1e-12)
                if v_opt < best_val - 1e-12:
                    best_val = v_opt
                    best_delta_overall = d_opt
                    best_ij = (i, j)
        if best_ij is not None:
            i, j = best_ij
            best[i] += best_delta_overall
            best[j] -= best_delta_overall
            best = np.clip(best, 0.0, 1.0)
            best = project_density(best, target)
            c_new = eval_c5(best)
            if c_new < best_c - 1e-12:
                best_c = c_new
            else:
                break
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    best = project_density(best, target)
    best_c = eval_c5(best)
    return best, best_c

def random_restart_optimize(h0, n, deadline, rng, max_iter_sg=2000, max_iter_ps=3000):
    target = 0.5 * n
    h = project_density(h0, target)
    best = h.copy()
    best_c = eval_c5(best)
    t1 = time.perf_counter() + (deadline - time.perf_counter()) * 0.5
    h2, c2 = subgradient_phase(h, n, t1, max_iter_sg)
    if c2 < best_c:
        best = h2
        best_c = c2
    h3, c3 = pair_swap_phase(best, n, deadline, max_iter_ps)
    if c3 < best_c:
        best = h3
        best_c = c3
    return best, best_c

def make_init_binary(n, rng):
    h = np.zeros(n, dtype=float)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return project_density(h, 0.5 * n)

def make_init_smooth(n, rng, k_freq):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)
    for m in range(1, k_freq + 1):
        amp = rng.uniform(0.1, 0.4) * (0.6 ** (m - 1))
        freq = int(max(1, np.rint(m * rng.uniform(0.8, 1.5))))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return project_density(h, 0.5 * n)

def make_init_block(n, rng, nblocks=4):
    h = np.zeros(n, dtype=float)
    block_size = n // nblocks
    for b in range(nblocks):
        if b % 2 == 0:
            h[b * block_size:(b + 1) * block_size] = 1.0
    return project_density(h, 0.5 * n)

def make_init_graded(n, rng, exponent=1.0):
    x = np.linspace(0, 1, n)
    h = x ** float(exponent)
    return project_density(h, 0.5 * n)

def make_init_pattern(n, rng, period_style=3):
    h = np.zeros(n, dtype=float)
    if period_style == 1:
        period = max(3, n // 8)
        for i in range(n):
            h[i] = 1.0 if (i % period) < period // 2 else 0.0
    elif period_style == 2:
        period = max(4, n // 6)
        for i in range(n):
            frac = (i % period) / float(period)
            h[i] = 1.0 if 0.2 < frac < 0.7 else 0.0
    else:
        periods = [max(2, n // 10), max(3, n // 7), max(5, n // 4)]
        for i in range(n):
            v = 0.0
            for p in periods:
                if (i % p) < p // 2:
                    v += 1.0
            h[i] = v / float(len(periods))
    return project_density(h, 0.5 * n)

def full_optimize(h0, n, deadline, rng):
    target = 0.5 * n
    h = project_density(h0, target)
    best = h.copy()
    best_c = eval_c5(best)
    elapsed = time.perf_counter()
    span = max(1.0, deadline - elapsed)
    # Phase 1: Subgradient
    t1 = elapsed + span * 0.35
    h1, c1 = subgradient_phase(h, n, t1, max_iter=5000)
    if c1 < best_c:
        best = h1
        best_c = c1
    # Phase 2: Pair swap
    t2 = elapsed + span * 0.75
    h2, c2 = pair_swap_phase(best, n, t2, max_iter=8000, top_k=8, top_l=20)
    if c2 < best_c:
        best = h2
        best_c = c2
    # Phase 3: Another round of subgradient with more active lags
    if deadline - time.perf_counter() > 5.0:
        h3, c3 = subgradient_phase(best, n, deadline - 2.0, max_iter=2000)
        if c3 < best_c:
            best = h3
            best_c = c3
    # Phase 4: Final pair swap
    if deadline - time.perf_counter() > 3.0:
        h4, c4 = pair_swap_phase(best, n, deadline, max_iter=5000, top_k=10, top_l=25)
        if c4 < best_c:
            best = h4
            best_c = c4
    best = project_density(best, target)
    best_c = eval_c5(best)
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)
    best_h = None
    best_c = float('inf')
    best_n = 0
    # Get parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 4:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None
    if parent is not None:
        np_ = parent.size
        tp = 0.5 * np_
        pp = project_density(parent, tp)
        cp = eval_c5(pp)
        if cp < best_c:
            best_h = pp
            best_c = cp
            best_n = np_
    # Define resolutions to try
    res_list = [90, 120, 180, 240, 300, 360, 450, 600]
    # Filter to reasonable range
    res_list = [r for r in res_list if 40 <= r <= 720]
    # Allocate time to each resolution
    n_res = len(res_list)
    time_per_res = max(30.0, (deadline - start) / float(n_res))
    for ri, n_r in enumerate(res_list):
        if time.perf_counter() >= deadline - 20.0:
            break
        # Build initializations for this resolution
        inits = []
        if best_h is not None:
            inits.append(upsample_h(best_h, n_r))
        if parent is not None:
            inits.append(upsample_h(parent, n_r))
        # Add diverse random inits
        if n_r <= 180:
            for _ in range(2):
                inits.append(make_init_binary(n_r, rng))
            inits.append(make_init_smooth(n_r, rng, 4))
            inits.append(make_init_pattern(n_r, rng, 3))
        elif n_r <= 360:
            inits.append(make_init_binary(n_r, rng))
            inits.append(make_init_smooth(n_r, rng, 3))
        else:
            inits.append(make_init_smooth(n_r, rng, 3))
        # Deduplicate
        uniq_inits = []
        for s in inits:
            s = project_density(s, 0.5 * n_r)
            dup = False
            for u in uniq_inits:
                if np.allclose(s, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                uniq_inits.append(s)
        # Try each init
        rem = deadline - time.perf_counter()
        n_try = max(1, len(uniq_inits))
        per_init = min(rem / float(n_try), time_per_res)
        per_init = max(15.0, per_init)
        best_local = None
        best_local_c = float('inf')
        for s in uniq_inits:
            sub_dead = time.perf_counter() + per_init
            if time.perf_counter() >= deadline - 10.0:
                break
            hs, cs = full_optimize(s, n_r, sub_dead, rng)
            if cs < best_local_c:
                best_local = hs
                best_local_c = cs
        if best_local is not None:
            if best_local_c < best_c:
                best_h = best_local
                best_c = best_local_c
                best_n = n_r
    # Final fine-tuning at the best resolution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 10.0:
        n_b = best_h.size
        # Try a few more pair swap rounds
        for _ in range(3):
            if time.perf_counter() >= deadline - 3.0:
                break
            h_fin, c_fin = pair_swap_phase(best_h, n_b, deadline - 2.0, max_iter=3000, top_k=10, top_l=30)
            if c_fin < best_c:
                best_h = h_fin
                best_c = c_fin
    # Fallback
    if best_h is None:
        fallback = project_density(np.full(90, 0.5, dtype=float), 45.0)
        return fallback, eval_c5(fallback), 90
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = eval_c5(final_h)
    return final_h, float(final_c), int(final_h.size)
