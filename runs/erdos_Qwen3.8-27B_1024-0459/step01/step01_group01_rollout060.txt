from scipy.optimize import minimize

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fast_correlate(a, b):
    """FFT-based cross-correlation, equivalent to np.correlate(a, b, 'full')."""
    n = len(a)
    m = len(b)
    N = n + m - 1
    # Cross-correlation: c[k] = sum_n a[n+k] * b[n]
    # = convolution of a with reversed b
    Nfft = 1
    while Nfft < 2 * N:
        Nfft *= 2
    A = np.fft.rfft(a, Nfft)
    B = np.fft.rfft(b[::-1], Nfft)
    result = np.fft.irfft(A * B, Nfft)
    return result[:N]

def compute_c5_fast(h):
    dx = 2.0 / len(h)
    corr = fast_correlate(h, 1.0 - h)
    return float(np.max(corr) * dx)

        n = len(h)
    h = h.copy()

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n=None):
    if n is None:
        n = len(h)
    h = h.copy()
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        count = mask.sum()
        per = diff / count
        h[mask] = np.clip(h[mask] + per, 0.0, 1.0)
    return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def obj_func(x, tau, n, pen_weight):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / tau, -80, 0)
    smooth_max = m + tau * np.log(np.sum(np.exp(shifted)) + 1e-300)
    sum_pen = pen_weight * (h.sum() - n / 2.0) ** 2
    return float(smooth_max * dx + sum_pen)


def eq_constraint(x, n):
    return np.array([x.sum() - n / 2.0])

    h = h.copy()
    target = n // 2
    order = np.argsort(-h)
    h[:] = 0.0
