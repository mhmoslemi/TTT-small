def subgradient_polish(h, deadline, max_iter=500, active_k=6, lr_start=0.05):

    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    ncorr = 2 * n - 1
    center = n - 1
    lr = lr_start
    stall = 0

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        k = min(active_k, ncorr)
        top_idx = np.argsort(corr)[-k:]
        w = corr[top_idx]
        w = w - w.min() + 1e-8
        w = w / w.sum()
        g = np.zeros(n, dtype=float)
        for t, wt in zip(top_idx, w):
            t = int(t)
            p = float(wt)
            lag = t - center
            if lag == 0:
                g += p * (1.0 - 2.0 * best)
            elif lag > 0:
                g[:n - lag] += p * (1.0 - best[lag:])
                g[lag:] -= p * best[:n - lag]
            else:
                lag2 = -lag
                g[lag2:] += p * (1.0 - best[:n - lag2])
                g[:n - lag2] -= p * best[lag2:]
        g -= float(np.mean(g))
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        d = -g / gn
        moved = False
        for lr_try in (lr, 0.5 * lr, 0.25 * lr, 0.1 * lr, 0.05):
            cand = project_density(best + lr_try * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                best = cand
                best_c = c
                moved = True
                lr = min(lr * 1.2, 0.15)
                break
        if not moved:
            stall += 1
            lr *= 0.7
            if stall > 20:
                break
        else:
            stall = max(0, stall - 1)

    return project_density(best, target), evaluate_erdos(best)


def two_coord_descent(h, deadline, max_rounds=200, active_k=8):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    ncorr = 2 * n - 1
    center = n - 1

    for rnd in range(max_rounds):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        k = min(active_k, ncorr)
        top_idx = np.argsort(corr)[-k:]
        top_vals = sorted(corr[top_idx], reverse=True)

        g = np.zeros(n, dtype=float)
        for t in top_idx:
            t = int(t)
            lag = t - center
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[:n - lag] += (1.0 - best[lag:])
                g[lag:] -= best[:n - lag]
            else:
                lag2 = -lag
                g[lag2:] += (1.0 - best[:n - lag2])
                g[:n - lag2] -= best[lag2:]
        g /= float(k)

        neg_i = np.argsort(g)[:5]
        pos_j = np.argsort(-g)[:5]

        improved = False
        for ii in neg_i:
            for jj in pos_j:
                ii, jj = int(ii), int(jj)
                if ii == jj:
                    continue
                if g[ii] >= g[jj]:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                # Golden section search for optimal t in [0, cap]
                a, b = 0.0, cap
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = b - gr * (b - a)
                c2 = a + gr * (b - a)
                for _ in range(12):
                    h1 = best.copy()
                    h1[ii] += c1
                    h1[jj] -= c1
                    h1 = np.clip(h1, 0, 1)
                    f1 = evaluate_erdos(h1)
                    h2 = best.copy()
                    h2[ii] += c2
                    h2[jj] -= c2
                    h2 = np.clip(h2, 0, 1)
                    f2 = evaluate_erdos(h2)
                    if f1 < f2:
                        b = c2
                        c2 = c1
                        c1 = b - gr * (b - a)
                    else:
                        a = c1
                        c1 = c2
                        c2 = a + gr * (b - a)
                t_opt = 0.5 * (a + b)
                if t_opt < 1e-10:
                    continue
                cand = best.copy()
                cand[ii] += t_opt
                cand[jj] -= t_opt
                cand = np.clip(cand, 0, 1)
                c_val = evaluate_erdos(cand)
                if c_val < best_c - 1e-12:
                    best = cand
                    best_c = c_val
                    improved = True

        if not improved:
            # Try gradient step
            d = -g
            d -= np.mean(d)
            dn = float(np.linalg.norm(d))
            if dn > 1e-13:
                d /= dn
                for step in (0.03, 0.08, 0.15, 0.3):
                    cand = project_density(best + step * d, target)
                    c_val = evaluate_erdos(cand)
                    if c_val < best_c - 1e-12:
                        best = cand
                        best_c = c_val
                        improved = True
                        break

        if not improved:
            break

    return project_density(best, target), evaluate_erdos(best)


def optimize_resolution(h0, n, deadline):
    n = int(n)
    target = 0.5 * n
    h = project_density(np.asarray(h0, dtype=float), target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    remaining = deadline - time.perf_counter()
    if remaining < 5.0:
        return best_h, best_c

    lbfgs_time = min(60.0, 0.5 * remaining)
    if n > 200:
        lbfgs_time = min(40.0, 0.4 * remaining)
    solve_deadline = time.perf_counter() + lbfgs_time

    alphas = (15.0, 50.0, 150.0) if n <= 120 else (15.0, 50.0)
    if n > 200:
        alphas = (15.0, 40.0)

    for alpha in alphas:
        if time.perf_counter() >= solve_deadline - 2.0:
            break
        set_ctx(n, alpha, target)
        bounds = list(zip(np.zeros(n), np.ones(n)))
        try:
            res = fmin_l_bfgs_b(
                smooth_obj, h, fprime=smooth_grad,
                bounds=bounds, maxiter=80,
                pgtol=1e-8, iprint=0
            )
            xg = res[0]
            cand = project_density(xg, target)
            c = evaluate_erdos(cand)
            if c < best_c:
                best_h = cand
                best_c = c
            if c <= best_c + 0.03:
                h = cand
            else:
                h = best_h.copy()
        except Exception:
            h = best_h.copy()

    # Subgradient polish
    if time.perf_counter() < deadline - 10.0:
        h2, c2 = subgradient_polish(best_h, min(deadline - 5.0, time.perf_counter() + 40.0),
                                     max_iter=300, active_k=6)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Two-coordinate descent
    if time.perf_counter() < deadline - 5.0:
        h3, c3 = two_coord_descent(best_h, min(deadline - 2.0, time.perf_counter() + 50.0),
                                    max_rounds=150, active_k=8)
        if c3 < best_c:
            best_h = h3
            best_c = c3

    # Final subgradient
    if time.perf_counter() < deadline - 3.0:
        h4, c4 = subgradient_polish(best_h, min(deadline - 1.0, time.perf_counter() + 20.0),
                                     max_iter=200, active_k=8, lr_start=0.02)
        if c4 < best_c:
            best_h = h4
            best_c = c4

    best_h = project_density(best_h, target)
    best_c = evaluate_erdos(best_h)
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0).copy()
    except Exception:
        parent = None

    if parent is not None:
        np_parent = int(parent.size)
        parent = project_density(parent, 0.5 * np_parent)
        cp = evaluate_erdos(parent)
        if cp < best_c:
            best_h = parent.copy()
            best_c = cp
            best_n = np_parent

    if best_h is None:
        n0 = 90
        best_h = make_fourier_start(n0, rng)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    base_n = best_n

    # Determine resolutions to try
    resolutions = []
    for mult in (1, 2, 3, 4):
        nn = int(base_n * mult)
        if 30 <= nn <= 500 and nn % 2 == 0:
            resolutions.append(nn)
    for nn in (60, 90, 120, 180, 240, 360, 450):
        if 30 <= nn <= 500 and nn % 2 == 0:
            resolutions.append(nn)
    resolutions = sorted(set(resolutions))

    # Time allocation per resolution
    n_res = len(resolutions)
    base_time = max(30.0, (deadline - time.perf_counter()) / max(1, n_res))

    for res_n in resolutions:
        if deadline - time.perf_counter() < 25.0:
            break

        # Time for this resolution
        if res_n <= 100:
            alloc = min(base_time * 1.5, 120.0)
        elif res_n <= 200:
            alloc = min(base_time * 1.2, 90.0)
        else:
            alloc = min(base_time, 60.0)
        alloc = max(20.0, min(alloc, deadline - time.perf_counter() - 5.0))
        res_deadline = time.perf_counter() + alloc

        # Build diverse starts
        starts = []
        if best_h is not None:
            if int(best_h.size) == res_n:
                starts.append(best_h.copy())
            else:
                starts.append(upsample(best_h, res_n))

        if parent is not None:
            starts.append(upsample(parent, res_n))

        # Add random starts
        n_rand = 2 if res_n <= 100 else 1
        for rr in range(n_rand):
            mode = rr % 3
            if mode == 0:
                starts.append(make_random_binary(res_n, rng))
            elif mode == 1:
                starts.append(make_fourier_start(res_n, rng))
            else:
                starts.append(make_block_start(res_n, rng))

        # Deduplicate
        uniq_starts = []
        for s in starts:
            s = project_density(s, 0.5 * res_n)
            dup = False
            for u in uniq_starts:
                if np.allclose(s, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                uniq_starts.append(s)

        if not uniq_starts:
            continue

        per_start = max(10.0, (res_deadline - time.perf_counter()) / len(uniq_starts))

        for si, s in enumerate(uniq_starts):
            if time.perf_counter() >= res_deadline - 3.0:
                break
            per_dl = min(time.perf_counter() + per_start, res_deadline - 2.0)
            try:
                h_out, c_out = optimize_resolution(s, res_n, per_dl)
                if h_out is not None:
                    h_out = project_density(h_out, 0.5 * res_n)
                    c_out = evaluate_erdos(h_out)
                    if c_out < best_c:
                        best_h = h_out
                        best_c = c_out
                        best_n = res_n
            except Exception:
                pass

    # Final polish on best solution
    if best_h is not None and deadline - time.perf_counter() > 15.0:
        n_best = int(best_h.size)
        if n_best <= 200:
            h_f, c_f = two_coord_descent(best_h, deadline, max_rounds=300, active_k=10)
            if c_f < best_c:
                best_h = h_f
                best_c = c_f
                best_n = n_best

        if deadline - time.perf_counter() > 8.0:
            h_g, c_g = subgradient_polish(best_h, deadline, max_iter=400, active_k=10, lr_start=0.01)
            if c_g < best_c:
                best_h = h_g
                best_c = c_g
                best_n = int(best_h.size)

    # One more final 2-coord pass
    if best_h is not None and deadline - time.perf_counter() > 5.0:
        h_z, c_z = two_coord_descent(best_h, deadline, max_rounds=50, active_k=12)
        if c_z < best_c:
            best_h = h_z
            best_c = c_z

    if best_h is None:
        h_fb = project_density(np.full(45, 0.5), 22.5)
        return h_fb, evaluate_erdos(h_fb), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
