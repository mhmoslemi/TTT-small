Let me implement the gradient of the smoothed objective:

The smoothed objective is: f(h) = (m + (1/beta) * log(sum(exp(beta*(corr-k)))) ) * dx

where m = max(corr) and corr = cross_correlate(h, 1-h, 'full').

The gradient is:
f'(h) = (1/beta) * sum_k exp(beta*(corr[k]-m)) * corr_k'(h) * dx / (sum_k exp(beta*(corr[k]-m))) * dx

Wait, let me be more careful:
f = (m + (1/beta) * log(S)) * dx where S = sum_k exp(beta*(corr[k]-m))

df/d(h[i]) = (1/beta) * (S'/S) * dx = (1/beta) * sum_k exp(beta*(corr[k]-m)) * d(corr[k])/d(h[i]) / S * dx

= sum_k w_k * d(corr[k])/d(h[i]) * dx

where w_k = exp(beta*(corr[k]-m)) / S (softmax weights)

So I need d(corr[k])/d(h[i]) for all k, weighted by w_k.

d(corr[k])/d(h[i]) where s = k-(n-1):
= (1-h[i-s]) * [valid(i-s)] - h[i+s] * [valid(i+s)]

where [valid(x)] = 1 if 0<=x<n and the range condition holds.

Actually, the range condition is automatically satisfied by the validity of i-s and i+s being in [0,n-1] combined with the definition of the correlation.

Let me just verify: corr[k] = sum_j h[j+s] * (1-h[j]) where s=k-(n-1) and j ranges over valid values.
d(corr[k])/d(h[m]):
- m appears as j+s when j = m-s: contributes (1-h[m-s]) if m-s is a valid j
- m appears as j when j = m: contributes -h[m+s] if m is a valid j

Valid j range: 0 <= j < n AND 0 <= j+s < n, i.e., max(0,-s) <= j <= min(n-1, n-1-s).

So:
d(corr[k])/d(h[m]) = (1-h[m-s]) * I[max(0,-s) <= m-s <= min(n-1,n-1-s)] - h[m+s] * I[max(0,-s) <= m <= min(n-1,n-1-s)]

With the additional condition that m-s and m+s are in [0,n-1] (which is implied by the above for the respective terms).

def _smooth_gradient(h):

    beta = _g_beta
    dx = 2.0 / n

    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    S = np.sum(exps)
    w = exps / S  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    # For each k (0 to 2n-2), s = k - (n-1)
    # grad[m] += w[k] * [(1-h[m-s])*I - h[m+s]*I] * dx

        hi = min(n-1, n-1-s)

    m = np.max(corr)

def _correlate_gradient(h, k):
    """Exact gradient of np.correlate(h, 1-h, 'full')[k] w.r.t. h."""
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    
    m1_lo = max(0, lo + s)
    m1_hi = min(n - 1, hi + s)
    if m1_lo <= m1_hi:
        idx = np.arange(m1_lo, m1_hi + 1)
        grad[idx] = (1.0 - h[idx - s])
    
    m2_lo = max(lo, -s)
    m2_hi = min(hi, n - 1 - s)
    if m2_lo <= m2_hi:
        idx = np.arange(m2_lo, m2_hi + 1)
        grad[idx] -= h[idx + s]

    best_local = _compute_c5(h)

        k_star = np.argmax(corr)
        g = _correlate_gradient(h, k_star)

        c_new = _compute_c5(h_new)

            h = h_new

        lr *= lr_decay

def _golden_section_1d(f, lo, hi, iters=80):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(iters):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _coord_descent(h, n, sweeps=3):
    """Optimize each coordinate using golden section, projecting after each."""
    h = h.copy()
    for sweep in range(sweeps):
        for i in range(n):
            save = h[i]
            def f1d(t):
                h[i] = t
                h_p = _project(h, n)
                return _compute_c5(h_p)
            h[i] = (save + 0.0) / 2.0  # center
            # Golden section on [0, 1]
            optimum = _golden_section_1d(f1d, 0.0, 1.0, iters=60)
            h[i] = optimum
            h = _project(h, n)
    return h
