    center = n - 1
    for t in idx:
        t = int(t)
        p = float(w[t])
        if p <= 0.0:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
            else:
                g[:n - lag] += p
            g[:n - lag] -= p * h[lag:]
            g[lag:] -= p * h[:n - lag]
    return (2.0 / float(n)) * g


def sum_constraint(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _TARGET)


def sum_constraint_jac(x):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)


def corr_gradient_for_lag(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g += 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g


def active_subgradient(h, deadline, max_iter=400, active_k=6, step_scale=0.05):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = 0.5 * n
    best = h.copy()
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    active = min(int(active_k), 2 * n - 1)
    stall = 0
    for it in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        idxs = np.argsort(corr)[-active:]
        g = np.zeros(n, dtype=float)
        for t in idxs:
            k = int(t) - (n - 1)
            g += corr_gradient_for_lag(best, k)
        g /= float(active)
        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm < 1e-13:
            break
        d /= norm
        improved = False
        for step in (step_scale * 0.2, step_scale * 0.5, step_scale, step_scale * 2, step_scale * 4):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-12:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def local_pair_search(h, deadline, max_iter=600, active_k=5):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 4:
        return best, best_c
    k_act = min(int(active_k), 2 * n - 1)
    stall = 0
    for _ in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        if not np.all(np.isfinite(corr)):
            break
        idxs = np.argsort(corr)[-k_act:]
        lags = np.abs(idxs - (n - 1)).astype(int)
        g = np.zeros(n, dtype=float)
        for t in idxs:
            k = int(t) - (n - 1)
            g += corr_gradient_for_lag(best, k)
        g /= float(k_act)
        if g.size == 0 or (float(np.max(g)) - float(np.min(g))) < 1e-12:
            break
        n_neg = min(4, n // 10 + 1)
        n_pos = min(4, n // 10 + 1)
        neg = np.argsort(g)[:n_neg]
        pos = np.argsort(-g)[:n_pos]
        found = False
        for ii in neg:
            ii = int(ii)
            for jj in pos:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap <= 1e-13:
                    continue
                a = float(g[ii] - g[jj])
                if a >= -1e-12:
                    continue
                b = 0.0
                for lag in lags:
                    lag = int(lag)
                    if lag == 0:
                        b += -2.0
                    elif abs(ii - jj) == lag:
                        b += 1.0
                b /= float(k_act)
                deltas = []
                if b > 1e-13:
                    dv = -a / (2.0 * b)
                    if dv > 0.0:
                        if dv > cap:
                            dv = cap
                        deltas.append(float(dv))
                deltas.append(0.5 * cap)
                deltas.append(0.1 * cap)
                deltas.append(cap)
                deltas.append(0.85 * cap)
                seen = []
                for d in deltas:
                    if d <= 1e-13 or d > cap:
                        continue
                    d = min(d, cap - 1e-15)
                    if d <= 1e-13:
                        continue
                    if any(abs(d - sd) < 1e-10 for sd in seen):
                        continue
                    seen.append(d)
                    cand = best.copy()
                    cand[ii] = best[ii] + d
                    cand[jj] = best[jj] - d
                    cand[ii] = np.clip(cand[ii], 0.0, 1.0)
                    cand[jj] = np.clip(cand[jj], 0.0, 1.0)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
        if not found:
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-13:
                d /= norm
                for step in (0.015, 0.05, 0.15, 0.3):
                    cand = project_density(best + step * d, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
        if found:
            stall = 0
        else:
            stall += 1
            if stall >= 18:
                break
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c

def run_slsqp(h, n, alpha, maxiter, sub_deadline):
    n = int(n)
    target = 0.5 * n
    h = project_density(h, target)
    set_context(n, alpha, target)
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac},)
    try:
        res = minimize(
            smooth_objective, h,
            method="SLSQP", jac=smooth_jacobian,
            bounds=bounds, constraints=constraints,
            options={"maxiter": maxiter, "ftol": 1e-10, "disp": False},
        )
        xg = getattr(res, "x", None)
        if xg is not None:
            xg = np.asarray(xg, dtype=float)
            if np.all(np.isfinite(xg)):
                return project_density(xg, target)
    except Exception:
        pass
    return h


def optimize_at_resolution(h0, n, deadline, rng):
    n = int(n)
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    elapsed_when_start = time.perf_counter()
    span = max(5.0, deadline - time.perf_counter())

    polish_time = min(40.0, max(8.0, 0.15 * span))
    solve_deadline = deadline - polish_time

    if n <= 100:
        alphas = (5.0, 15.0, 40.0, 100.0, 250.0)
        maxiters = (200, 150, 120, 100, 80)
    elif n <= 200:
        alphas = (8.0, 25.0, 70.0, 170.0)
        maxiters = (150, 120, 90, 70)
    elif n <= 400:
        alphas = (8.0, 25.0, 70.0)
        maxiters = (120, 80, 60)
    else:
        alphas = (8.0, 30.0)
        maxiters = (80, 50)

    for i, alpha in enumerate(alphas):
        if time.perf_counter() >= solve_deadline:
            break
        sub_dl = solve_deadline
        if i < len(alphas) - 1:
        sub_dl = time.perf_counter() + (solve_deadline - time.perf_counter()) / (len(alphas) - i)
        out = run_slsqp(h, n, alpha, maxiters[i], sub_dl)
        c = evaluate_erdos(out)
        if c < best_c:
            best_h = out
            best_c = c
        if c <= best_c + 0.03:
            h = out
        else:
            h = best_h

    if time.perf_counter() < deadline - 5.0:
        h2, c2 = active_subgradient(best_h, deadline - 2.0, max_iter=200, active_k=6, step_scale=0.04)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.perf_counter() < deadline - 3.0:
        h3, c3 = local_pair_search(best_h, deadline - 1.0, max_iter=400, active_k=5)
        if c3 < best_c:
            best_h = h3
            best_c = c3

    best_h = project_density(best_h, target)
    best_c = evaluate_erdos(best_h)
    return best_h, best_c


def perturb_restart(h, n, rng, deadline, n_tries=3):
    h = np.asarray(h, dtype=float)
    n = int(n)
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    for _ in range(int(n_tries)):
        if time.perf_counter() >= deadline:
            break
        noise_scale = 0.08
        noisy = best + noise_scale * rng.standard_normal(n)
        cand = project_density(noisy, target)
        sub_dl = min(deadline, time.perf_counter() + 40.0)
        c = evaluate_erdos(cand)
        if c < best_c:
            best = cand
            best_c = c
        h2, c2 = local_pair_search(cand, sub_dl, max_iter=150, active_k=5)
        if c2 < best_c:
            best = h2
            best_c = c2
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(5.0, budget - 15.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        nparent = int(parent.size)
        target_p = 0.5 * nparent
        pproj = project_density(parent, target_p)
        cp = evaluate_erdos(pproj)
        if cp < best_c:
            best_h = pproj
            best_c = cp
            best_n = nparent

    if best_h is None:
        n0 = 90
        best_h = make_random_candidate(n0, rng, 0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    base_n = best_n if best_n >= 2 else 90

    resolutions = []
    n_primary = base_n
    resolutions.append(n_primary)
    n2 = int(base_n * 2)
    if n2 <= 720 and n2 != n_primary:
        resolutions.append(n2)
    n3 = int(base_n * 4)
    if n3 <= 720 and n3 not in resolutions:
        resolutions.append(n3)
    for nn in (180, 270, 360):
        if nn not in resolutions and nn <= 720:
            resolutions.append(nn)
    resolutions = sorted(set(resolutions))
    resolutions = [r for r in resolutions if r >= 2 and r <= 720]

    for ni in resolutions:
        if deadline - time.perf_counter() < 40.0:
            break

        starts = []
        starts.append(upsample_h(best_h, ni))
        if parent is not None:
            starts.append(upsample_h(parent, ni))
        if ni <= 180:
            starts.append(make_random_candidate(ni, rng, rng.integers(0, 3)))
        else:
            starts.append(make_random_candidate(ni, rng, 0))

        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if vector_close(s, u):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        if not uniq:
            continue

        sub_budget = deadline - time.perf_counter()
        n_starts = len(uniq)
        if ni <= 100:
            per_start = min(120.0, sub_budget / float(n_starts))
        elif ni <= 200:
            per_start = min(100.0, sub_budget / float(n_starts))
        else:
            per_start = min(70.0, sub_budget / float(n_starts))
        per_start = max(15.0, per_start)

        for s in uniq:
            rem = deadline - time.perf_counter()
            if rem < 15.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem)
            h_out, c_out = optimize_at_resolution(s, ni, sub_dl, rng)
            h_out = project_density(h_out, 0.5 * ni)
            c_out = evaluate_erdos(h_out)
            if c_out < best_c:
                best_h = h_out
                best_c = c_out
                best_n = ni

    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 30.0:
        nbest = int(best_h.size)

        if nbest >= 90:
            h_down = downsample_h(best_h, 90)
            c_down = evaluate_erdos(h_down)
            sub_dl = time.perf_counter() + min(60.0, rem * 0.25)
            h_p, c_p = optimize_at_resolution(h_down, 90, sub_dl, rng)
            c_p = evaluate_erdos(h_p)
            if c_p < best_c:
                best_h = h_p
                best_c = c_p
                best_n = 90

        if deadline - time.perf_counter() > 20.0:
            n_now = int(best_h.size)
            h2, c2 = local_pair_search(best_h, deadline - 5.0, max_iter=800, active_k=5)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = n_now

        if deadline - time.perf_counter() > 10.0:
            h3, c3 = fine_polish(best_h, deadline - 2.0, max_iter=500, active_k=4)
            if c3 < best_c:
                best_h = h3
                best_c = c3
                best_n = int(best_h.size)

        if deadline - time.perf_counter() > 20.0:
            n_now = int(best_h.size)
            h4, c4 = perturb_restart(best_h, n_now, rng, deadline - 2.0, n_tries=3)
            if c4 < best_c:
                best_h = h4
                best_c = c4
                best_n = n_now

        if deadline - time.perf_counter() > 10.0:
            h5, c5 = fine_polish(best_h, deadline - 1.0, max_iter=300, active_k=4)
            if c5 < best_c:
                best_h = h5
                best_c = c5
                best_n = int(best_h.size)

    if best_h is None:
        fallback = project_density(np.full(90, 0.5, dtype=float), 45.0)
        return fallback, evaluate_erdos(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
