def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    c = get_corr(h, n)
    return float(np.max(c) * (2.0 / n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-12
        else:
            mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) > 0:
        if diff > 0:
            mask = h > 1e-12
        else:
            mask = h < 1.0 - 1e-12
        if mask.sum() > 0:
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
