def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def hard_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = get_corr(h)
    return float(np.max(c) * dx)


def hard_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = get_corr(h)
    k_max = int(np.argmax(c))
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] -= h[:n - l]
    return grad * dx


def soft_obj(h, n, beta):
    dx = 2.0 / n
    c = get_corr(h) * dx
    m = np.max(c)
    return float(m + np.log(np.sum(np.exp(c - m))) / beta)


def soft_grad(h, n, beta):
    dx = 2.0 / n
    c = get_corr(h) * dx
    m = np.max(c)
    w = np.exp(beta * (c - m))
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros(len(w))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            l = -lag
            grad[:n - l] += wk * (1.0 - h[l:])
            grad[l:] -= wk * h[:n - l]
    return grad * dx


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-14
        else:
            mask = h < 1.0 - 1e-14
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] -= diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def constraint_fun(x):
    return x.sum() - x.size / 2.0


def constraint_jac(x):
    return np.ones(x.size)


def make_cosine_init(n, freq, amp, phase):
    x = np.arange(n, dtype=float) / n
    h = 0.5 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def make_block_init(n, num_blocks):
    h = np.zeros(n)
    block_size = n // (2 * num_blocks)
    remainder = n - block_size * 2 * num_blocks
    pos = 0
    for i in range(num_blocks):
        sz1 = block_size + (1 if i < remainder // num_blocks else 0)
        h[pos:pos + sz1] = 1.0
        pos += sz1
        if pos >= n:
            break
        h[pos:pos + block_size + (1 if i < remainder % num_blocks else 0)] = 0.0
        pos += block_size + (1 if i < (remainder % num_blocks) else 0)
    return project_feasible(h, n)


def make_alternating_init(n):
    h = np.zeros(n)
    h[0::2] = 1.0
    h[1::2] = 0.0
    return project_feasible(h, n)


def interp_to_n(h, n_old, n_new):
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def slsqp_smooth(h, n, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(
            fun=soft_obj, x0=h, jac=soft_grad,
            args=(n, beta),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-13}
        )
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        return h_new
    except Exception:
        return h


def slsqp_hard(h, n, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(
            fun=hard_obj, x0=h, jac=hard_grad,
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        return h_new
    except Exception:
        return h


def perturb_and_optimize(h_best, n, rng, scale, maxiter):
    h_pert = h_best + rng.randn(n) * scale
    h_pert = project_feasible(h_pert, n)
    h_new = slsqp_hard(h_pert, n, maxiter)
    return h_new


def local_coordinate_descent(h, n, steps):
    h = h.copy()
    c5_cur = compute_c5(h, n)
    dx = 2.0 / n
    for _ in range(steps):
        improved = False
        perm = np.random.permutation(n)
        for idx in perm:
            for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
                h_try = h.copy()
                new_val = np.clip(h[idx] + delta, 0.0, 1.0)
                if new_val == h[idx]:
                    continue
                adj = h[idx] - new_val
                h_try[idx] = new_val
                # Adjust one other variable to maintain sum
                if abs(adj) > 1e-15:
                    # Find the best variable to adjust
                    best_j = -1
                    best_c5 = compute_c5(h_try, n) + 1e-6
                    for j in range(n):
                        if j == idx:
                            continue
                        h_try[j] = np.clip(h[j] + adj, 0.0, 1.0)
                        if 0.0 <= h_try[j] <= 1.0:
                            c5_try = compute_c5(h_try, n)
                            if c5_try < best_c5:
                                best_c5 = c5_try
                                best_j = j
                        h_try[j] = h[j]
                    if best_j >= 0:
                        h_try[idx] = new_val
                        h_try[best_j] = np.clip(h[best_j] + adj, 0.0, 1.0)
                        h_try = project_feasible(h_try, n)
                        if compute_c5(h_try, n) < c5_cur - 1e-10:
                            h = h_try
                            c5_cur = compute_c5(h, n)
                            improved = True
                            break
                    h_try.clear()
                else:
                    h_try[idx] = new_val
                    c5_try = compute_c5(h_try, n)
                    if c5_try < c5_cur - 1e-10:
                        h = h_try
                        c5_cur = c5_try
                        improved = True
                        break
        if not improved:
            break
    return h, c5_cur


def binary_push(h, n):
    """Try pushing variables toward 0 or 1 while maintaining feasibility."""
    h = h.copy()
    c5_cur = compute_c5(h, n)
    n_push = int(n * 0.3)
    idxs = np.argsort(np.abs(h - 0.5))[-n_push:]
    for idx in idxs:
        target = 0.0 if h[idx] < 0.5 else 1.0
        h_try = h.copy()
        adj = h[idx] - target
        h_try[idx] = target
        # Distribute adjustment to maintain sum
        others = [j for j in range(n) if j != idx]
        if adj != 0:
            per = adj / len(others)
            feasible = True
            for j in others:
                new_v = h[j] + per
                if new_v < -1e-12 or new_v > 1.0 + 1e-12:
                    feasible = False
                    break
                h_try[j] = np.clip(new_v, 0.0, 1.0)
            if feasible:
                h_try = project_feasible(h_try, n)
                if compute_c5(h_try, n) < c5_cur - 1e-12:
                    h = h_try
                    c5_cur = compute_c5(h, n)
    return h, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get parent
    h_parent = None
    n_parent = 95
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        pass

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Evaluate parent if available
    if h_parent is not None:
        h_p = project_feasible(h_parent, n_parent)
        c5_p = compute_c5(h_p, n_parent)
        best_h = h_p.copy()
        best_c5 = c5_p
        best_n = n_parent

    # Resolutions to try (even n for clean binary-ness)
    n_list = [96, 100, 104, 120, 128]

    # Build initializations for each resolution
    for n in n_list:
        if time.time() - t0 > budget_s - 150:
            break

        # Interpolated parent
        if h_parent is not None:
            h_init = interp_to_n(h_parent, n_parent, n)
        else:
            h_init = make_cosine_init(n, 2, 0.4, 0.5)

        # Phase 1: Smooth max SLSQP
        for beta in [8.0, 30.0, 120.0, 500.0]:
            if time.time() - t0 > budget_s - 120:
                break
            h_init = slsqp_smooth(h_init, n, beta, 600)
            c5_check = compute_c5(h_init, n)
            if c5_check < best_c5:
                best_h = h_init.copy()
                best_c5 = c5_check
                best_n = n

        # Phase 2: Hard max SLSQP (multiple passes)
        for _ in range(3):
            if time.time() - t0 > budget_s - 80:
                break
            h_init = slsqp_hard(h_init, n, 1200)
            c5_check = compute_c5(h_init, n)
            if c5_check < best_c5:
                best_h = h_init.copy()
                best_c5 = c5_check
                best_n = n

    # Try diverse initializations at the best resolution
    n_work = best_n
    if n_work % 2 != 0:
        n_work = 96

    if time.time() - t0 < budget_s - 200:
        # Cosine initializations
        for freq in [1, 2, 3, 4, 5]:
            for amp in [0.35, 0.42, 0.48, 0.5]:
                for phase in [0.0, 0.7, 1.5, 2.2]:
                    if time.time() - t0 > budget_s - 100:
                        break
                    h_cos = make_cosine_init(n_work, freq, amp, phase)
                    h_cos = slsqp_smooth(h_cos, n_work, 30.0, 400)
                    h_cos = slsqp_hard(h_cos, n_work, 800)
                    c5_cos = compute_c5(h_cos, n_work)
                    if c5_cos < best_c5:
                        best_h = h_cos.copy()
                        best_c5 = c5_cos
                        best_n = n_work

    # Block initializations
    if time.time() - t0 < budget_s - 150:
        for nb in [2, 3, 4, 5, 6]:
            if time.time() - t0 > budget_s - 80:
                break
            h_blk = make_block_init(n_work, nb)
            h_blk = slsqp_smooth(h_blk, n_work, 20.0, 300)
            h_blk = slsqp_hard(h_blk, n_work, 600)
            c5_blk = compute_c5(h_blk, n_work)
            if c5_blk < best_c5:
                best_h = h_blk.copy()
                best_c5 = c5_blk
                best_n = n_work

    # Perturbation restarts from best
    if time.time() - t0 < budget_s - 100:
        for i in range(150):
            if time.time() - t0 > budget_s - 40:
                break
            scale = 0.02 + 0.04 * (i / 50.0)
            h_pert = best_h + rng.randn(best_n) * scale
            h_pert = project_feasible(h_pert, best_n)
            h_pert = slsqp_hard(h_pert, best_n, 500)
            c5_pert = compute_c5(h_pert, best_n)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert

    # Try a few more resolutions with the best solution interpolated
    if time.time() - t0 < budget_s - 120:
        for n_alt in [128, 192, 64]:
            if n_alt == best_n:
                continue
            if time.time() - t0 > budget_s - 60:
                break
            h_alt = interp_to_n(best_h, best_n, n_alt)
            h_alt = slsqp_smooth(h_alt, n_alt, 50.0, 400)
            h_alt = slsqp_hard(h_alt, n_alt, 800)
            c5_alt = compute_c5(h_alt, n_alt)
            if c5_alt < best_c5:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt

    # Local coordinate descent polish
    if time.time() - t0 < budget_s - 30:
        h_cd, c5_cd = local_coordinate_descent(best_h, best_n, 3)
        if c5_cd < best_c5:
            best_h = h_cd
            best_c5 = c5_cd

    # Binary push
    if time.time() - t0 < budget_s - 20:
        h_bp, c5_bp = binary_push(best_h, best_n)
        if c5_bp < best_c5:
            best_h = h_bp
            best_c5 = c5_bp

    # Final SLSQP refinement
    if time.time() - t0 < budget_s - 10:
        best_h = slsqp_hard(best_h, best_n, 800)
        best_c5 = compute_c5(best_h, best_n)

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    # Validate
    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
