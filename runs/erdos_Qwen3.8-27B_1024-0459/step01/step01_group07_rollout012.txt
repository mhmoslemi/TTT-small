    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    shifted = conv - m

        if w < 1e-12:

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            if not np.any(free):
                break
            h[free] = np.minimum(h[free] + diff / np.sum(free), 1.0)
        else:
            free = h > 1e-15
            if not np.any(free):
                break
            h[free] = np.maximum(h[free] + diff / np.sum(free), 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        diff = target - s
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)

                  beta1=0.9, beta2=0.999, eps=1e-8, temp_decay=0.98):
    n = n_points
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    temp = temp_init
    
    for it in range(n_iter):
        temp = max(temp_end, temp * temp_decay)
        grad = compute_grad_smooth(h, n, temp)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**(it + 1))
        v_hat = v / (1 - beta2**(it + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        val = compute_c5(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_c5_smooth(h, n_points, temp):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    shifted = conv - m
    return m + temp * np.log(np.sum(np.exp(shifted / temp)))


def compute_grad_smooth(h, n_points, temp):
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    shifted = conv - m
    exp_vals = np.exp(shifted / temp)
    weights = exp_vals / np.sum(exp_vals)

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-13:
            continue
        s = k - (n - 1)
        n_lo = max(0, -s)
        n_hi = min(n - 1, n - 1 - s)

        # h[i] as h[n+s]: i-s in [n_lo, n_hi], so i in [n_lo+s, n_hi+s]
        i_lo1 = max(0, n_lo + s)
        i_hi1 = min(n - 1, n_hi + s)
        if i_lo1 <= i_hi1:
            n_idx = np.arange(i_lo1 - s, i_hi1 - s + 1)
            grad[i_lo1:i_hi1 + 1] += w * (1.0 - h[n_idx])

        # h[i] in (1-h[n]): i in [n_lo, n_hi]
        i_lo2 = max(0, n_lo)
        i_hi2 = min(n - 1, n_hi)
        if i_lo2 <= i_hi2:
            i_idx = np.arange(i_lo2, i_hi2 + 1)
            grad[i_lo2:i_hi2 + 1] -= w * h[i_idx + s]

    grad *= dx
    return grad


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            if not np.any(free):
                break
            h[free] = np.minimum(h[free] + diff / np.sum(free), 1.0)
        else:
            free = h > 1e-15
            if not np.any(free):
                break
            h[free] = np.maximum(h[free] + diff / np.sum(free), 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        diff = target - s
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h


def adam_optimize(h0, n_points, n_iter, lr=0.03, temp_init=0.08, temp_end=0.005,
                  beta1=0.9, beta2=0.999, eps=1e-8, temp_decay=0.995,
                  rng_seed=42, anneal_start=0):
    n = n_points
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    temp = temp_init
    rng = np.random.RandomState(rng_seed)

    for it in range(n_iter):
        temp_cur = max(temp_end, temp * (temp_decay ** it))
        if it >= anneal_start:
            temp_cur = max(temp_end, temp_init * (temp_decay ** (it - anneal_start)))

        grad = compute_grad_smooth(h, n, temp_cur)
        norm = np.linalg.norm(grad)
        if norm < 1e-15:
            break

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)

        val = compute_c5(h_new, n)

            best_val = val
            best_h = h_new.copy()
        h = h_new

    return best_h, best_val

def try_slsqp_polish(h0, n_points, maxiter=300):

    def obj(x):
        return compute_c5_smooth(x, n, 0.01)

        return compute_grad_smooth(x, n, 0.01)

                                         temp_init=0.12, temp_end=0.005, rng_seed=seed + 600)
            if v_try < best_val:
                best_h, best_val, best_n = h_try, v_try, n_fine

    # Phase 6: Final aggressive polish
    if time_left() > 60:
        h_try, v_try = annealed_adam(best_h, best_n, n_iter=3000, lr=0.015,
                                     temp_init=0.04, temp_end=0.001, rng_seed=seed + 900)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    if time_left() > 40:
        h_try, v_try = local_search(best_h, best_n, n_iter=10000, rng_seed=seed + 999)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    if time_left() > 30:
        h_try, v_try = try_slsqp_polish(best_h, best_n, maxiter=500)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
