2. Adjust the sum to n/2 by adding a constant to all non-boundary elements

```python
def project_constraint(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}."""
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-10
        else:
            free = h > 1e-10
        if not np.any(free):
            break
        h[free] += diff / np.count_nonzero(free)
    return h
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_direct(h):
    """Compute C5 using direct correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_fft(h):
    """Compute C5 using FFT-based correlation."""
    n = len(h)
    dx = 2.0 / n
    N = 1 << (2 * n - 1).bit_length()
    # correlate(h, g, 'full') = convolve(g, h, 'full')
    # convolve(g, h, 'full') = ifft(fft(g, N) * fft(h, N))[:2n-1]
    g = 1.0 - h
    G = np.fft.rfft(g, N)
    H = np.fft.rfft(h, N)
    corr = np.fft.irfft(G * H, N)[:2*n-1]
    # Remove small imaginary artifacts (shouldn't be any for real inputs)
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = n/2}."""
    h = h.copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
        else:
            free = h > 1e-12
        cnt = np.count_nonzero(free)
        if cnt == 0:
            break
        h[free] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, beta):
    """Compute smooth max of corr using log-sum-exp."""
    cmax = np.max(corr)
    return cmax + np.log(np.sum(np.exp(beta * (corr - cmax)))) / beta

def smooth_max_grad(corr, beta, h, n):
    """
    Compute gradient of smooth max objective w.r.t. h.
    
    corr[k] = sum_j (1-h[j]) * h[j+k-(n-1)] for valid j
    d_corr[k]/d_h[i] = (1-h[i-d]) * I(0≤i-d<n) - h[i+d] * I(0≤i+d<n)
    where d = k - (n-1)
    
    Returns gradient of (1/beta) * log(sum(exp(beta*corr))) w.r.t. h
    """
    # Compute weights
    cmax = np.max(corr)
    exp_beta = np.exp(beta * (corr - cmax))
    Z = np.sum(exp_beta)
    w = exp_beta / Z  # w[k] for k = 0..2n-2, corresponds to d = k-(n-1)
    
    # Gradient computation
    # dL/dh[i] = sum_k w[k] * [(1-h[i-d_k])*I(0≤i-d_k<n) - h[i+d_k]*I(0≤i+d_k<n)]
    # where d_k = k - (n-1)
    
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for k in range(len(w)):
            d = k - (n - 1)
            if w[k] < 1e-15:
                continue
            # First term: (1-h[i-d]) if 0 <= i-d < n
            if 0 <= i - d < n:
                s += w[k] * (1.0 - h[i - d])
            # Second term: -h[i+d] if 0 <= i+d < n
            if 0 <= i + d < n:
                s -= w[k] * h[i + d]
        grad[i] = s
    return grad
