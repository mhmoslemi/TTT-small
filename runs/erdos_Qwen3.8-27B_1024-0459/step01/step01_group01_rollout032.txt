    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return fix_sum(h_new, n_new)

def smooth_penalty_obj(x, n, tau, pen_mult):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(corr.max())
    shifted = np.clip((corr - m) / tau, -600.0, 0.0)
    smooth_max = m + tau * np.log(np.sum(np.exp(shifted)))
    base = smooth_max * (2.0 / n)
    penalty = pen_mult * (float(h.sum()) - n / 2.0) ** 2
    return base + penalty

def do_slsqp(x0, n, tau, pen_mult, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_penalty_obj, x0, method='SLSQP',
            bounds=bounds, args=(n, tau, pen_mult),
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
        )
        h_out = np.clip(res.x, 0.0, 1.0)
        return fix_sum(h_out, n)
    except Exception:
        return fix_sum(x0, n)

def periodic_init(n, period_frac, duty_frac):
    h = np.zeros(n)
    period_len = max(1, int(period_frac * n))
    duty_len = int(duty_frac * period_len)
    for i in range(n):
        if (i % period_len) < duty_len:
            h[i] = 1.0
    return fix_sum(h, n)

def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def smooth_init(n, f1, f2, a1, a2, ph1, ph2):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + a1 * np.sin(2.0 * np.pi * f1 * x + ph1) + a2 * np.sin(2.0 * np.pi * f2 * x + ph2)
    return fix_sum(np.clip(h, 0.0, 1.0), n)
