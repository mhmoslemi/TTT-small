        if m >= 0:
            if m < n:
                grad[m:] += (1.0 - x[:n - m]) * dx
            if n - m > 0:
                grad[:n - m] += (-x[m:]) * dx
        else:
            p = -m
            if n - p > 0:
                grad[:n - p] += (1.0 - x[p:]) * dx
            if p < n:
                grad[p:] += (-x[:n - p]) * dx
        return grad

    # Try multiple SLSQP runs with perturbation
    for attempt in range(5):
        if time.time() - t_start > time_limit:
            break
        if attempt == 0:
            h_try = h.copy()
        else:
            rng2 = np.random.RandomState(seed + attempt * 100)
            h_try = h + rng2.normal(0, 0.05, n)
            h_try = project_feasible(h_try, n)
        try:
            res = minimize(obj_slsqp, h_try, jac=jac_slsqp, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq',
                                        'fun': lambda x: np.sum(x) - target,
                                        'jac': lambda x: np.ones(n)}],
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_try = project_feasible(res.x, n)
            v_try = compute_c5_val(h_try, n)
            if v_try < best_v:
                best_v = v_try
                best_h = h_try.copy()
        except Exception:
            pass

    # Projected gradient descent as backup
    t_pg = time.time()
    h_pg = best_h.copy()
    lr = 0.02
    for it in range(500):
        if time.time() - t_pg > max(5, time_limit * 0.2):
            break
        conv = np.correlate(h_pg, 1.0 - h_pg, mode='full')
        k_max = np.argmax(conv)
        m = k_max - (n - 1)
        dx = 2.0 / n
        grad = np.zeros(n)
        if m >= 0:
            if m < n:
                grad[m:] += (1.0 - h_pg[:n - m]) * dx
            if n - m > 0:
                grad[:n - m] += (-h_pg[m:]) * dx
        else:
            p = -m
            if n - p > 0:
                grad[:n - p] += (1.0 - h_pg[p:]) * dx
            if p < n:
                grad[p:] += (-h_pg[:n - p]) * dx
        h_new = h_pg - lr * grad
        h_new = project_feasible(h_new, n)
        v_new = compute_c5_val(h_new, n)
        if v_new < best_v:
            best_v = v_new
            best_h = h_new.copy()
        h_pg = h_new
        if it % 50 == 49:
            lr *= 0.9

    return best_h, best_v


def structured_binary(n, seed):
    """Generate structured binary sequences for initialization."""
    rng = np.random.RandomState(seed)
    target = n // 2
    candidates = []

    # Thue-Morse
    tm = thue_morse_seq(n)
    if tm.sum() != target:
        # Adjust to have exactly target ones
        if tm.sum() > target:
            idx1 = np.where(tm == 1)[0]
            flip = rng.choice(idx1, size=tm.sum() - target, replace=False)
            tm[flip] = 0
        else:
            idx0 = np.where(tm == 0)[0]
            flip = rng.choice(idx0, size=target - tm.sum(), replace=False)
            tm[flip] = 1
    candidates.append(tm.astype(float))

    # Alternating with noise
    alt = np.tile([1.0, 0.0], n // 2)
    if n % 2 == 1:
        alt = np.concatenate([alt, [0.0]])
    if alt.sum() != target:
        if alt.sum() > target:
            idx1 = np.where(alt == 1)[0]
            flip = rng.choice(idx1, size=alt.sum() - target, replace=False)
            alt[flip] = 0
        else:
            idx0 = np.where(alt == 0)[0]
            flip = rng.choice(idx0, size=target - alt.sum(), replace=False)
            alt[flip] = 1
    candidates.append(alt)

    # Blocked pattern: blocks of size 2 or 3
    block_sizes = rng.choice([2, 3, 4], size=4)
    h_block = np.zeros(n)
    pos = 0
    val = 1.0
    for bs in block_sizes:
        for _ in range(2):  # one block of 1s, one of 0s
            end = min(pos + bs, n)
            if pos < n:
                h_block[pos:end] = val
                pos = end
            if pos >= n:
                break
            val = 1.0 - val
        if pos >= n:
            break
    # Fix count
    if h_block.sum() > target:
        idx1 = np.where(h_block == 1)[0]
        flip = rng.choice(idx1, size=int(h_block.sum()) - target, replace=False)
        h_block[flip] = 0
    elif h_block.sum() < target:
        idx0 = np.where(h_block == 0)[0]
        flip = rng.choice(idx0, size=target - int(h_block.sum()), replace=False)
        h_block[flip] = 1
    candidates.append(h_block)

    # Golden ratio based
    phi = (1 + np.sqrt(5)) / 2
    h_gold = np.zeros(n)
    for i in range(n):
        h_gold[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    if h_gold.sum() > target:
        idx1 = np.where(h_gold == 1)[0]
        flip = rng.choice(idx1, size=int(h_gold.sum()) - target, replace=False)
        h_gold[flip] = 0
    elif h_gold.sum() < target:
        idx0 = np.where(h_gold == 0)[0]
        flip = rng.choice(idx0, size=target - int(h_gold.sum()), replace=False)
        h_gold[flip] = 1
    candidates.append(h_gold)

    # Random
    h_rand = rng.binomial(1, 0.5, n).astype(float)
    if h_rand.sum() != target:
        if h_rand.sum() > target:
            idx1 = np.where(h_rand == 1)[0]
            flip = rng.choice(idx1, size=int(h_rand.sum()) - target, replace=False)
            h_rand[flip] = 0
        else:
            idx0 = np.where(h_rand == 0)[0]
            flip = rng.choice(idx0, size=target - int(h_rand.sum()), replace=False)
            h_rand[flip] = 1
    candidates.append(h_rand)

    return candidates


def interpolate_h(h_coarse, n_coarse, n_fine):
    x_c = np.linspace(0, 2, n_coarse, endpoint=False)
    x_f = np.linspace(0, 2, n_fine, endpoint=False)
    h_fine = np.interp(x_f, x_c, h_coarse)
    return h_fine


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial from parent
    n_parent = 70
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        pass

    if h_parent is None:
        h_parent = np.full(n_parent, 0.5)

    best_h = project_feasible(h_parent.copy(), n_parent)
    best_v = compute_c5_val(best_h, n_parent)
    best_n = n_parent

    # Phase 1: Binary search at parent resolution
    h_bin_parent = binary_round(h_parent, n_parent)
    h_bin_parent, v_bin_parent = sa_hc_combined(h_bin_parent, n_parent, min(40, time_left() * 0.15), seed)
    if v_bin_parent < best_v:
        best_v = v_bin_parent
        best_h = h_bin_parent.copy()
        best_n = n_parent

    # Continuous polish at parent resolution
    if time_left() > 100:
        h_pol, v_pol = continuous_polish(h_bin_parent, n_parent, min(80, time_left() * 0.25), seed + 10)
        if v_pol < best_v:
            best_h = h_pol
            best_v = v_pol
            best_n = n_parent

    # Phase 2: Try multiple resolutions with binary + continuous
    resolutions = [100, 120, 140, 160, 200]
    for n_res in resolutions:
        if time_left() < 120:
            break
        if n_res == best_n:
            continue
        alloc = min(90, time_left() * 0.5 / len([r for r in resolutions if r > best_n]))

        # Interpolate best to this resolution
        h_interp = interpolate_h(best_h, best_n, n_res)
        h_bin_res = binary_round(h_interp, n_res)

        # Multiple initializations
        init_candidates = [h_bin_res]
        init_candidates.extend(structured_binary(n_res, seed + n_res))

        best_at_res = None
        best_v_res = float('inf')

        for ci, h_init in enumerate(init_candidates):
            if time_left() < 60:
                break
            t_alloc = min(alloc / len(init_candidates) * 1.5, time_left() * 0.3)
            h_b, v_b = sa_hc_combined(h_init, n_res, t_alloc, seed + n_res + ci)
            if v_b < best_v_res:
                best_v_res = v_b
                best_at_res = h_b

        if best_at_res is None:
            continue

        # Continuous polish at this resolution
        if time_left() > 80:
            h_cp, v_cp = continuous_polish(best_at_res, n_res, min(40, time_left() * 0.3), seed + n_res + 50)
            if v_cp < best_v_res:
                best_v_res = v_cp
                best_at_res = h_cp

        if best_v_res < best_v:
            best_v = best_v_res
            best_h = best_at_res
            best_n = n_res

    # Phase 3: Aggressive local search on best solution
    if time_left() > 60:
        h_cur = best_h.copy()
        if best_n <= 200:
            # If nearly binary, do more local search
            n_ones = int(np.sum(h_cur > 0.5))
            if abs(n_ones - best_n // 2) <= 3:
                h_b = np.zeros(best_n)
                h_b[h_cur > 0.5] = 1.0
                if sum(h_b) != best_n // 2:
                    h_b = binary_round(h_cur, best_n)
                h_sa, v_sa = sa_binary(h_b, best_n, min(30, time_left() * 0.4), seed + 200)
                h_hc, v_hc = hc_binary(h_sa, best_n, min(20, time_left() * 0.3), seed + 210)
                if v_hc < best_v:
                    best_v = v_hc
                    best_h = h_hc

    # Phase 4: Final polish
    if time_left() > 30:
        h_final, v_final = continuous_polish(best_h, best_n, min(30, time_left() * 0.5), seed + 500)
        if v_final < best_v:
            best_v = v_final
            best_h = h_final

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_v = compute_c5_val(best_h, best_n)

    return (best_h, best_v, best_n)
