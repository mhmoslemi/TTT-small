def thue_morse_binary(n):
    """Generate Thue-Morse-like binary sequence with n/2 ones"""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        # Count bits
        c = 0
        x = i
        while x > 0:
            c += x & 1
            x >>= 1
        tm[i] = c % 2
    # Ensure exactly n/2 ones
    k = n // 2
    num_ones = np.sum(tm)
    if num_ones > k:
        # Flip some 1s to 0s
        ones_idx = np.where(tm == 1)[0]
        rng_t = np.random.default_rng(0)
        to_flip = rng_t.choice(ones_idx, size=num_ones - k, replace=False)
        tm[to_flip] = 0
    elif num_ones < k:
        zeros_idx = np.where(tm == 0)[0]
        rng_t = np.random.default_rng(0)
        to_flip = rng_t.choice(zeros_idx, size=k - num_ones, replace=False)
        tm[to_flip] = 1
    # Shuffle to remove position bias
    rng_s = np.random.default_rng(1)
    permutation = rng_s.permutation(n)
    tm = tm[permutation]
    return tm.astype(float)

def slsqp_optimize(x0, n, tau, maxiter, time_fn=None):
    """Run SLSQP optimization"""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, n, dx, tau)
    
    def con(x):
        return eq_con(x, n)
    
    def jac(x):
        return eq_jac(x, n)
    
    try:
        res = minimize(
            fun=obj, x0=x0, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con, 'jac': jac}],
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )
        if time_fn is not None and time_fn():
            return None
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = fix_sum(h_new, n)
        return h_new
    except Exception:
        return None

def multi_level_lsqp(h0, n, time_fn=None):
    """Multi-level SLSQP with decreasing tau"""
    current = h0.copy()
    current = fix_sum(current, n)
    best_h = current.copy()
    best_c5 = compute_c5(best_h)
    
    tau_list = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for tau in tau_list:
        if time_fn is not None and time_fn():
            break
        h_new = slsqp_optimize(current, n, tau, maxiter=500, time_fn=time_fn)
        if h_new is not None:
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                current = h_new.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    def time_up():
        return (time.time() - t0) > budget_s * 0.95
    
    # Get initial values
    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(180, 0.5)
    
    n = len(best_h)
    best_h = fix_sum(best_h, n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Binarize and local search (fast)
    b_init = to_binary(best_h, n)
    b_ls, mc_ls = binary_full_local_search(b_init, n, max_rounds=50, time_fn=time_up)
    c5_ls = mc_ls * (2.0 / n)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = b_ls.copy()
    
    # Phase 2: Simulated annealing from best binary
    elapsed = time.time() - t0
    sa_budget = budget_s * 0.25
    n_sa_iter = max(5000, int(n * 200))
    
    sa_starts = [b_ls.copy()]
    # Add some random binary starts
    for _ in range(2):
        if elapsed > sa_budget:
            break
        rb = np.zeros(n)
        ones_positions = rng.permutation(n)[:n//2]
        rb[ones_positions] = 1.0
        sa_starts.append(rb)
    # Add Thue-Morse
    sa_starts.append(thue_morse_binary(n))
    
    for sb in sa_starts:
        elapsed = time.time() - t0
        remain_frac = max(0.0, 1.0 - elapsed / sa_budget)
        if remain_frac <= 0:
            break
        n_iter_this = max(1000, int(n_sa_iter * remain_frac / max(1, len(sa_starts))))
        b_sa, mc_sa = binary_sa(sb, n, n_iter_this, rng, time_fn=time_up)
        if time_up():
            break
        # Local search after SA
        b_sa2, mc_sa2 = binary_full_local_search(b_sa, n, max_rounds=30, time_fn=time_up)
        c5_sa = mc_sa2 * (2.0 / n)
        if c5_sa < best_c5 - 1e-13:
            best_c5 = c5_sa
            best_h = b_sa2.copy()
    
    # Phase 3: Try different structured initializations
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.5:
        structured_init = []
        # Various periodic patterns
        for period in [3, 4, 5, 6, 7, 8, 9, 10, 12, 15]:
            for duty in [0.3, 0.4, 0.5, 0.6, 0.7]:
                if period * duty == 1.0:
                    continue
                structured_init.append(periodic_binary(n, period, duty))
        # Random
        for _ in range(3):
            rb = np.zeros(n)
            ones_positions = rng.permutation(n)[:n//2]
            rb[ones_positions] = 1.0
            structured_init.append(rb)
        
        for si in structured_init:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.55:
                break
            b_si, mc_si = binary_full_local_search(si, n, max_rounds=20, time_fn=time_up)
            if time_up():
                break
            c5_si = mc_si * (2.0 / n)
            if c5_si < best_c5 - 1e-13:
                best_c5 = c5_si
                best_h = b_si.copy()
    
    # Phase 4: Try different resolution
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.6:
        for n_alt in [120, 240]:
            if n_alt == n:
                continue
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.7:
                break
            
            # Upsample/downsample
            h_alt = upsample_to(best_h, n_alt)
            h_alt = fix_sum(h_alt, n_alt)
            
            # Binarize and local search
            b_alt = to_binary(h_alt, n_alt)
            b_alt_ls, mc_alt = binary_full_local_search(b_alt, n_alt, max_rounds=30, time_fn=time_up)
            if time_up():
                break
            c5_alt = mc_alt * (2.0 / n_alt)
            if c5_alt < best_c5 - 1e-13:
                best_c5 = c5_alt
                best_h = b_alt_ls.copy()
                n = n_alt
    
    # Phase 5: Continuous optimization from best solution
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.75:
        n_best = len(best_h)
        # Multi-level SLSQP
        h_cont = best_h.astype(float).copy()
        h_cont = fix_sum(h_cont, n_best)
        
        h_opt, c5_opt = multi_level_lsqp(h_cont, n_best, time_fn=time_up)
        if c5_opt < best_c5 - 1e-13:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 6: More continuous optimization with perturbations
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.88:
        n_best = len(best_h)
        cont_starts = [best_h.astype(float).copy()]
        
        # Add perturbed versions
        for _ in range(3):
            if time_up():
                break
            noise = rng.normal(0, 0.05, n_best)
            hp = np.clip(best_h + noise, 0.0, 1.0)
            hp = fix_sum(hp, n_best)
            cont_starts.append(hp)
        
        for cs in cont_starts:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.93:
                break
            h_opt, c5_opt = multi_level_lsqp(cs, n_best, time_fn=time_up)
            if time_up():
                break
            if c5_opt < best_c5 - 1e-13:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 7: Final fine polish with very small tau
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.96:
        n_best = len(best_h)
        h_final = best_h.astype(float).copy()
        h_final = fix_sum(h_final, n_best)
        h_polish = slsqp_optimize(h_final, n_best, tau=0.001, maxiter=1000, time_fn=time_up)
        if h_polish is not None:
            c5_polish = compute_c5(h_polish)
            if c5_polish < best_c5 - 1e-13:
                best_c5 = c5_polish
                best_h = h_polish.copy()
    
    # Final verification and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
