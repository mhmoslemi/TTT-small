def compute_scores(h, n):
    """Compute all lag scores s_k = sum_i h[i](1-h[i+k]) * dx"""
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, h, mode='full')
    ac = corr[n-1:]  # autocorrelation lags 0..n-1
    prefix = np.concatenate([[0], np.cumsum(h)])
    total = prefix[n]
    scores = np.zeros(2*n-1)

        scores[2*d] = total - prefix[d] - ac[d]  # negative lag (by symmetry)
    return scores

def gradient_scores(h, n, active_indices):

    grad = np.zeros(n)

            grad += (1 - 2*h)
        else:

            else:

                m = n - d
                grad[:m] += (-h[d:])
                grad[d:] += (1 - h[:m])

One important detail: the evaluation uses `np.correlate(h, 1-h, mode='full') * dx` and takes the max. My score computation should match this.

np.correlate(h, 1-h, mode='full') gives:
For the full array of length 2n-1:
result[k] for k=0,...,2n-2

The formula: result[k] = sum_i h[i] * (1-h[i - (k - (n-1))])
Actually, numpy's correlate is:
c_k = sum_n a_n * conj(v_{n+k})

For np.correlate(h, 1-h):
c_k = sum_n h_n * (1-h)_{n+k} = sum_n h_n * (1 - h_{n+k})

where the sum is over valid indices.

So c_k for k = -(n-1), ..., 0, ..., n-1:
c_0 = sum_n h_n (1-h_n) = sum h - sum h²
c_k for k>0: sum_{n=0}^{n-1-k} h_n (1-h_{n+k})
c_k for k<0: sum_{n=-k}^{n-1} h_n (1-h_{n+k}) = sum_{n=0}^{n-1+|k|} h_{n-|k|} (1-h_n)

In the array (0-indexed):
result[n-1+k] = c_k

So:
result[n-1] = c_0 = sum h - sum h²
result[n-1+k] for k>0: overlap with positive shift
result[n-1-k] for k>0: overlap with negative shift

The max of result * dx is C₅.

My score computation needs to give the same values. Let me verify:
scores[0] = total - ac[0] where ac[0] = dot(h, h) = sum h²
So scores[0] = sum(h) - sum(h²) = c_0. ✓

scores[2d-1] for d>=1: prefix[n-d] - ac[d]
= sum_{i=0}^{n-d-1} h[i] - sum_{i=0}^{n-d-1} h[i]*h[i+d]

import numpy as np
import time
import math
from scipy.optimize import minimize

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv) * (2.0 / n))
    if not math.isfinite(c):
        return 1e6
    return c

def enforce_constraints(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n)
            m = min(n, h.size)
            if m > 0:
                tmp[:m] = h[:m]
            h = tmp
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    s = float(h.sum())
    if abs(s - target) > 1e-12:
        lo = -2.0
        hi = 2.0
        for _ in range(100):
            mid = 0.5 * (lo + hi)
            proj = np.clip(h + mid, 0.0, 1.0)
            s_mid = float(proj.sum())
            if s_mid < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
            if np.any(mask):
                h[mask] = np.minimum(1.0, h[mask] + diff / max(1, mask.sum()))
        else:
            mask = h > 0.0
            if np.any(mask):
                h[mask] = np.maximum(0.0, h[mask] + diff / max(1, mask.sum()))
    return np.clip(h, 0.0, 1.0)

def compute_scores(h, n):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, h, mode="full")
    ac = corr[n - 1:]
    prefix = np.empty(n + 1)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2 * n - 1)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2 * d - 1] = prefix[n - d] - ac[d]
        scores[2 * d] = total - prefix[d] - ac[d]
    return scores

def grad_single_score(idx, h, n):
    grad = np.zeros(n)
    if idx == 0:
        grad += (1.0 - 2.0 * h)
    else:
        if idx % 2 == 1:
            d = (idx + 1) // 2
            m = n - d
            if m <= 0:
                return grad
            grad[:m] += (1.0 - h[d:])
            grad[d:] += (-h[:m])
        else:
            d = idx // 2
            m = n - d
            if m <= 0:
                return grad
            grad[:m] += (-h[d:])
            grad[d:] += (1.0 - h[:m])
    return grad

def project_feasible(v, n):
    target = 0.5 * float(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(v - mid, 0.0, 1.0)
        s = float(proj.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = x < 1.0
            cnt = int(np.count_nonzero(mask))
            if cnt > 0:
                per = diff / cnt
                inc = min(per, float(1.0 - x[mask].min()))
                x[mask] += inc
        else:
            mask = x > 0.0
            cnt = int(np.count_nonzero(mask))
            if cnt > 0:
                per = -diff / cnt
                dec = min(per, float(x[mask].min()))
                x[mask] -= dec
    return np.clip(x, 0.0, 1.0)

def upsample(parent, n):
    n0 = parent.size
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent).astype(float)

def make_synthetic(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    best_h = enforce_constraints(np.full(n, 0.5), n)
    best_c = compute_c5(best_h)
    for f in range(1, 5):
        for ph in (0.0, 0.5 * math.pi, math.pi):
            h = 0.5 + 0.45 * np.cos(f * math.pi * x + ph)
            h = np.clip(h, 0.0, 1.0)
            h = enforce_constraints(h, n)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h
    h = make_smoothed_binary(n, rng)
    c = compute_c5(h)
    if c < best_c:
        best_h = h
    return best_h

def make_smoothed_binary(n, rng):
    target = n // 2
    idx = rng.choice(n, size=target, replace=False)
    hb = np.zeros(n)
    hb[idx] = 1.0
    if n > 2:
        out = hb.copy()
        out[0] = 0.75 * hb[0] + 0.25 * hb[1]
        out[-1] = 0.25 * hb[-2] + 0.75 * hb[-1]
        out[1:-1] = 0.25 * hb[:-2] + 0.5 * hb[1:-1] + 0.25 * hb[2:]
        hb = out
    return enforce_constraints(hb, n)

def projected_gradient(h0, n, time_limit, rng):
    h = h0.copy()
    scores = compute_scores(h, n)
    m = float(scores.max())
    best_h = h.copy()
    best_m = m
    alpha = 0.3
    alpha_min = 0.003
    lr = 4.0 / float(n)
    lr_decay = 0.9995
    it = 0
    t_start = time.time()
    stall = 0
    best_stall_time = t_start
    
    while time.time() < time_limit:
        it += 1
        if it % 500 == 0:
            if time.time() - best_stall_time > 30.0:
                break
        
        alpha = max(alpha_min, alpha * 0.998)
        lr = max(0.5 / float(n), lr * lr_decay)
        
        if alpha > alpha_min:
            shifted = (scores - m) / alpha
            shifted = np.clip(shifted, -50, 0)
            w = np.exp(shifted)
            w_sum = float(w.sum())
            if w_sum < 1e-30:
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(scores))] = 1.0
                w_sum = 1.0
            w /= w_sum
            thresh = 1e-8
            active = np.flatnonzero(w > thresh)
        else:
            tol = max(1e-7, 1e-6 * max(1.0, m))
            active = np.flatnonzero(scores >= m - tol)
            if active.size == 0:
                active = np.array([int(np.argmax(scores))])
        
        grad = np.zeros(n)
        for idx in active:
            w_i = w[idx] if alpha > alpha_min else 1.0 / len(active)
            if w_i < 1e-10:
                continue
            if idx == 0:
                grad += w_i * (1.0 - 2.0 * h)
            else:
                if idx % 2 == 1:
                    d = (idx + 1) // 2
                    mm = n - d
                    if mm <= 0:
                        continue
                    grad[:mm] += w_i * (1.0 - h[d:])
                    grad[d:] += w_i * (-h[:mm])
                else:
                    d = idx // 2
                    mm = n - d
                    if mm <= 0:
                        continue
                    grad[:mm] += w_i * (-h[d:])
                    grad[d:] += w_i * (1.0 - h[:mm])
        
        h_new = project_feasible(h - lr * grad, n)
        
        h = h_new
        scores = compute_scores(h, n)
        m = float(scores.max())

            best_m = m
            best_h = h.copy()
            stall = 0
            best_stall_time = time.time()

            if lr > 0.2 / float(n):
                lr *= 0.5
                stall = 0
            else:
                h = enforce_constraints(0.5 * best_h + 0.5 * h, n)
                scores = compute_scores(h, n)
                m = float(scores.max())
                if m < best_m - 1e-10:
                    best_m = m
                    best_h = h.copy()
                break

    c_cur = m * (2.0 / n)
    if c_cur < c_best:
        best_h = h.copy()
        c_best = c_cur
    return best_h, c_best

def slsqp_obj(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    scores = compute_scores(h, n)
    m = float(scores.max())
    if alpha > 1e-14:
        shifted = np.clip((scores - m) / alpha, -50, 0)
        w = np.exp(shifted)
        A = float(w.sum())
        if A > 1e-30:
            val = m + alpha * math.log(A)
        else:
            val = m
    else:
        val = m
    return float(val) * (2.0 / n)

def slsqp_grad(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    scores = compute_scores(h, n)
    m = float(scores.max())
    grad = np.zeros(n)
    if alpha > 1e-14:
        shifted = np.clip((scores - m) / alpha, -50, 0)
        w = np.exp(shifted)
        A = float(w.sum())
        if A > 1e-30:
            w /= A
            for i, wi in enumerate(w):
                if wi < 1e-8:
                    continue
                if i == 0:
                    grad += wi * (1.0 - 2.0 * h)
                else:
                    if i % 2 == 1:
                        d = (i + 1) // 2
                        mm = n - d
                        if mm <= 0:
                            continue
                        grad[:mm] += wi * (1.0 - h[d:])
                        grad[d:] += wi * (-h[:mm])
                    else:
                        d = i // 2
                        mm = n - d
                        if mm <= 0:
                            continue
                        grad[:mm] += wi * (-h[d:])
                        grad[d:] += wi * (1.0 - h[:mm])
    else:
        tol = max(1e-7, 1e-6 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(scores))])
        for i in active:
            co = 1.0 / len(active)
            if i == 0:
                grad += co * (1.0 - 2.0 * h)
            else:
                if i % 2 == 1:
                    d = (i + 1) // 2
                    mm = n - d
                    if mm <= 0:
                        continue
                    grad[:mm] += co * (1.0 - h[d:])
                    grad[d:] += co * (-h[:mm])
                else:
                    d = i // 2
                    mm = n - d
                    if mm <= 0:
                        continue
                    grad[:mm] += co * (-h[d:])
                    grad[d:] += co * (1.0 - h[:mm])
    grad *= 2.0 / n
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return grad

def constraint_eq(x, n):
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)

def constraint_jac(x, n):
    return np.full((1, n), 1.0 / float(n))

        inits.append(enforce_constraints(0.5 + 0.45 * np.cos(2 * math.pi * x), n))
        inits.append(enforce_constraints(0.5 + 0.4 * np.sin(math.pi * x + 0.3), n))
        
        # Some random perturbation of parent
        h_pert = upsample(parent, n)
        h_pert += 0.05 * rng.standard_normal(n)
        inits.append(enforce_constraints(h_pert, n))
        
        n_inits = len(inits)
        time_per_init = max(5.0, remaining_pg / (n_inits * 1.2))
        
        for init in inits:
            if time.time() > pg_deadline - 10:
                break
            h_init = enforce_constraints(init, n)
            if h_init is None or not np.all(np.isfinite(h_init)):
                continue
            tl = time.time() + min(time_per_init, pg_deadline - time.time() - 5)
            if tl <= time.time() + 3:
                break
            h_opt, c_opt = projected_gradient(h_init, n, tl, rng)
            if c_opt < best_c:
                best_c = float(c_opt)
                best_h = h_opt.copy()
                best_n = n
    
    # Phase 2: SLSQP polish
    if time.time() < deadline - 20:
        slsqp_dl = deadline - 5
        # Polish current best
        h_poly = slsqp_polish(best_h, best_n, slsqp_dl)
        c_poly = compute_c5(h_poly)
        if c_poly < best_c:
            best_c = float(c_poly)
            best_h = h_poly
        
        # Try polishing at parent resolution if different
        if best_n != n0 and time.time() < slsqp_dl - 30:
            h_at_n0 = upsample(best_h, n0)
            h_at_n0 = enforce_constraints(h_at_n0, n0)
            h_poly2 = slsqp_polish(h_at_n0, n0, slsqp_dl - 10)
            c_poly2 = compute_c5(h_poly2)
            if c_poly2 < best_c:
                best_c = float(c_poly2)
                best_h = h_poly2
                best_n = n0
    
    # Phase 3: Final verification and enforcement
    best_h = enforce_constraints(best_h, best_n)
    best_c = compute_c5(best_h)
    
    return best_h, float(best_c), int(best_n)
