def project_density(x,target):
  x=np.asarray(x,dtype=float)
  x=np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
  x=np.clip(x,0,1)
  n=x.size
  target=float(target)
  if n==0: return x
  s=x.sum()
  if abs(s-target)<1e-14*max(1,n): return x
  lo,hi=-1,1
  for _ in range(80):
    mid=(lo+hi)/2
    s=np.clip(x+mid,0,1).sum()
    if s<target: lo=mid
    else: hi=mid
  h=np.clip(x+(lo+hi)/2,0,1)
  res=target-h.sum()
  # distribute residual among free elements with largest slack/eaters
  if abs(res)>1e-12:
     free=np.where((h>1e-15)&(h<1-1e-15))[0]
     if free.size:
        h[free] += res/free.size
        h=np.clip(h,0,1)
  # final exact? floating may still res; adjust one free
  res=target-h.sum()
  if abs(res)>1e-15:

        h[idx]+=res
        if h[idx]<0: h[idx]=0
        elif h[idx]>1: h[idx]=1
  return np.clip(h,0,1)

def gradient_for_lags(h,lag_indices):
   n=h.size; g=np.zeros(n)
   if n==0 or len(lag_indices)==0: return g
   center=n-1
   for t in lag_indices:
      t=int(t)
      if t<0 or t>=2*n-1: continue
      k=t-center
      if k==0:

      else:
         lag=abs(k)
         if k>0:
            g[lag:] += 1.0
         else:

         g[lag:] -= h[:n-lag]

        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    val = float(np.max(corr) * dx)
    if not np.isfinite(val):
        return float("inf")
    return val


def corr_array(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    s = float(np.sum(x))
    if abs(s - target) <= 1e-14 * max(1, n):
        return x

    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        ss = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if ss < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    res = target - float(np.sum(h))
    if abs(res) > 1e-14:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)

    for _ in range(30):
        res = target - float(np.sum(h))
        if abs(res) <= 1e-15:
            break
        if res > 0.0:
            free = np.where(h < 1.0 - 1e-15)[0]
            if free.size == 0:
                break
            idx = int(free[np.argmax(1.0 - h[free])])
            add = min(res, 1.0 - h[idx])
            h[idx] += add
        else:
            free = np.where(h > 1e-15)[0]
            if free.size == 0:
                break
            idx = int(free[np.argmax(h[free])])
            sub = min(-res, h[idx])
            h[idx] -= sub

    return h


def is_feasible(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -1e-7) or np.any(h > 1.0 + 1e-7):
        return False
    if abs(float(np.sum(h)) - 0.5 * float(n)) > 1e-5:
        return False
    return True


def vector_close(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    if a.size != b.size:
        return False
    return bool(np.allclose(a, b, rtol=0.0, atol=1e-7))


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n <= 0:
        return project_density(np.full(1, 0.5), 0.5)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * float(n))

    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * float(n))


def make_candidate(n, rng, mode):
    n = int(n)
    if n < 2:
        n = 2
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    mode = int(mode)

    if mode == 0:
        h = 0.5 * x
    elif mode == 1:
        h = 1.0 - np.abs(x - 1.0)
    elif mode == 2:
        h = np.abs(x - 1.0)
    elif mode == 3:
        h = 0.5 + 0.5 * np.cos(np.pi * x)
    elif mode == 4:
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        h = 0.5 + 0.5 * np.sin(np.pi * x + phase)
    elif mode == 5:
        h = np.full(n, 0.5, dtype=float)
        for m in range(1, 7):
            amp = float(rng.uniform(0.02, 0.35)) * (0.65 ** (m - 1))
            freq = int(max(1, np.rint(float(m) * rng.uniform(0.7, 1.6))))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            h = h + amp * np.sin(np.pi * freq * x + phase)
    else:
        h = np.zeros(n, dtype=float)
        pos = 0
        val = float(rng.choice([0.0, 1.0]))
        while pos < n:
            hi = max(2, n - pos + 1)
            length = int(rng.integers(1, hi))
            h[pos:pos + length] = val
            pos += length
            val = 1.0 - val

    return project_density(h, 0.5 * float(n))


def gradient_for_lags(h, lag_indices):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0 or len(lag_indices) == 0:
        return g

    center = n - 1
    for t in lag_indices:
        t = int(t)
        if t < 0 or t >= 2 * n - 1:
            continue
        k = t - center
        if k == 0:
            g = g + (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] = g[lag:] + 1.0
            else:
                g[:n - lag] = g[:n - lag] + 1.0
            g[:n - lag] = g[:n - lag] - h[lag:]
            g[lag:] = g[lag:] - h[:n - lag]
    return g


def add_direction(dirs, d):
    d = np.asarray(d, dtype=float)
    if d.size == 0:
        return
    d = d - float(np.mean(d))
    nrm = float(np.linalg.norm(d))
    if not np.isfinite(nrm) or nrm < 1e-12:
        return
    d = d / nrm
    for e in dirs:
        dot = float(np.dot(d, e))
        if abs(dot) > 0.999:
            return
    dirs.append(d)


def _line_value(tau, A, B, Q, dx):
    c = A + tau * B + tau * tau * Q
    if not np.all(np.isfinite(c)):
        c = np.nan_to_num(c, nan=1e100, posinf=1e100, neginf=1e100)
    return float(np.max(c) * dx)


def line_search(h, d, tau_max, A, C0, dx, target):
    h = np.asarray(h, dtype=float)
    d = np.asarray(d, dtype=float)
    if d.size != h.size:
        return None

    d = d - float(np.mean(d))
    pos = d > 1e-14
    neg = d < -1e-14
    if not (pos.any() or neg.any()):
        return None

    tmax = float(tau_max)
    if not np.isfinite(tmax) or tmax <= 0.0:
        return None

    if pos.any():
        tmax = min(tmax, float(np.min((1.0 - h[pos]) / d[pos])))
    if neg.any():
        tmax = min(tmax, float(np.min(h[neg] / (-d[neg]))))

    if not np.isfinite(tmax) or tmax <= 1e-12:
        return None
    tmax *= (1.0 - 1e-12)
    if tmax <= 1e-12:
        return None

    try:
        B = np.correlate(d, 1.0 - h, mode="full") - np.correlate(h, d, mode="full")
        Q = -np.correlate(d, d, mode="full")
    except Exception:
        return None

    if B.size != A.size:
        A = np.correlate(h, 1.0 - h, mode="full")

    if not (np.all(np.isfinite(A)) and np.all(np.isfinite(B)) and np.all(np.isfinite(Q))):
        return None

    K = 49 if A.size <= 400 else 33
    try:
        tau = np.linspace(0.0, tmax, K)
        tt = tau[:, None]
        cc = A[None, :] + tt * B[None, :] + (tt * tt) * Q[None, :]
        if not np.all(np.isfinite(cc)):
            cc = np.nan_to_num(cc, nan=1e100, posinf=1e100, neginf=1e100)
        vals = np.max(cc, axis=1) * dx
    except Exception:
        return None

    if vals.size == 0:
        return None

    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return None
    if vals[best_idx] >= C0 - 1e-12:
        return None

    left = float(tau[best_idx - 1]) if best_idx > 0 else 0.0
    right = float(tau[best_idx + 1]) if best_idx < K - 1 else float(tmax)
    if right <= left + 1e-14:
        left = 0.0
        right = float(tmax)

    best_t = float(tau[best_idx])
    best_v = float(vals[best_idx])

    try:
        opts = {"xatol": max(1e-8, float(tmax) * 1e-9), "maxiter": 80}
        res = minimize_scalar(_line_value, args=(A, B, Q, dx), bounds=(left, right),
                              method="bounded", options=opts)
        if getattr(res, "success", False):
            vt = float(res.x)
            cv = float(res.fun)
            if np.isfinite(cv) and cv < best_v - 1e-13:
                best_t = vt
                best_v = cv
    except Exception:
        pass

    if best_v >= C0 - 1e-11:
        return None

    cand = h + best_t * d
    cand = np.clip(cand, 0.0, 1.0)
    if abs(float(np.sum(cand)) - float(target)) > 1e-9:
        cand = project_density(cand, target)

    val = evaluate_erdos(cand)
    if not np.isfinite(val) or val >= C0 - 1e-12:
        return None

    return val, cand


def block_direction(h, g, rng, n):
    if rng is None:
        return None
    n = int(n)
    for _ in range(12):
        lo = max(1, n // 12)
        hi = max(lo + 1, n // 3)
        len1 = int(rng.integers(lo, hi))
        len2 = int(rng.integers(lo, hi))
        if n - len1 < 0 or n - len2 < 0:
            continue
        s1 = int(rng.integers(0, n - len1 + 1))
        s2 = int(rng.integers(0, n - len2 + 1))

        d = np.zeros(n, dtype=float)
        d[s1:s1 + len1] = 1.0
        d[s2:s2 + len2] = -float(len1) / float(len2)
        d = d - float(np.mean(d))
        gd = float(np.dot(g, d))
        if gd > 0.0:
            d = -d
        nrm = float(np.linalg.norm(d))
        if np.isfinite(nrm) and nrm > 1e-10:
            return d / nrm
    return None


def local_search(h0, n, deadline, rng=None, max_iter=1000):
    n = int(n)
    if n < 2:
        h = project_density(h0, 0.5 * float(max(1, n)))
        return h, evaluate_erdos(h)

    target = 0.5 * float(n)
    dx = 2.0 / float(n)

    h = project_density(h0, target)
    A = corr_array(h)
    C = evaluate_erdos(h)

    best_h = h.copy()
    best_c = C

    if n <= 120:
        dir_limit = 24
        k_list = (1, 2, 3, 5, 8)
        pair_limit = 14
    elif n <= 240:
        dir_limit = 18
        k_list = (1, 2, 3, 5)
        pair_limit = 10
    else:
        dir_limit = 13
        k_list = (1, 2, 3)
        pair_limit = 7

    stall = 0
    jumps = 0
    max_jumps = 6
    jump_sigs = (0.07, 0.045, 0.028, 0.016, 0.009, 0.005)

    for _ in range(int(max_iter)):
        now = time.perf_counter()
        if now >= deadline:
            break

        order = np.argsort(A)
        max_lags = min(len(order), 12)
        dirs = []

        for k in k_list:
            kk = min(k, max_lags)
            if kk <= 0:
                continue
            g = gradient_for_lags(h, order[-kk:])
            add_direction(dirs, -g)
            if len(dirs) >= dir_limit:
                break

        for t in order[-min(4, max_lags):]:
            if len(dirs) >= dir_limit:
                break
            g = gradient_for_lags(h, np.array([int(t)], dtype=int))
            add_direction(dirs, -g)

        if len(dirs) < dir_limit:
            d = np.where(h > 0.5, 1.0, -1.0)
            add_direction(dirs, d)

        if len(dirs) < dir_limit:
            kk = min(4, max_lags)
            g = gradient_for_lags(h, order[-kk:])
            hi = np.argsort(g)[-8:]
            lo = np.argsort(g)[:8]
            pairs = []
            for ii in lo:
                ii = int(ii)
                for jj in hi:
                    jj = int(jj)
                    if ii != jj:
                        diff = float(g[jj] - g[ii])
                        if diff > 1e-12:
                            pairs.append((diff, ii, jj))
            pairs.sort(reverse=True)
            for _diff, ii, jj in pairs[:pair_limit]:
                if len(dirs) >= dir_limit:
                    break
                d = np.zeros(n, dtype=float)
                d[ii] = 1.0
                d[jj] = -1.0
                add_direction(dirs, d)

        if rng is not None and stall >= 1 and len(dirs) < dir_limit:
            d = np.asarray(rng.normal(size=n), dtype=float)
            add_direction(dirs, d)

        if rng is not None and stall >= 3 and len(dirs) < dir_limit:
            kk = min(4, max_lags)
            g = gradient_for_lags(h, order[-kk:])
            d = block_direction(h, g, rng, n)
            if d is not None:
                add_direction(dirs, d)

        if len(dirs) == 0:
            stall += 1
            if stall > 20:
                break
            continue

        best_trial = None
        for d in dirs:
            if time.perf_counter() >= deadline:
                break
            trial = line_search(h, d, 1e12, A, C, dx, target)
            if trial is not None:
                cv, ch = trial
                if best_trial is None or cv < best_trial[0] - 1e-12:
                    best_trial = (cv, ch)

        if best_trial is not None:
            h_proj = project_density(best_trial[1], target)
            c_proj = evaluate_erdos(h_proj)
            if np.isfinite(c_proj) and c_proj < C - 1e-12:
                h = h_proj
                C = c_proj
                A = corr_array(h)
                if C < best_c - 1e-13:
                    best_h = h.copy()
                    best_c = C
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 25 and rng is not None and jumps < max_jumps:
            sig = float(jump_sigs[min(jumps, len(jump_sigs) - 1)])
            h = project_density(h + sig * rng.normal(size=n), target)
            A = corr_array(h)
            C = evaluate_erdos(h)
            jumps += 1
            stall = 0
        elif stall > 35:
            break

    final_h = project_density(best_h, target)
    final_c = evaluate_erdos(final_h)
    return final_h, final_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 20.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0
    parent = None

    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            p = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0)
            p = np.clip(p, 0.0, 1.0)
            nparent = int(p.size)
            targetp = 0.5 * float(nparent)
            if abs(float(np.sum(p)) - targetp) <= 1e-6:
                candp = p
            else:
                candp = project_density(p, targetp)
            cp = evaluate_erdos(candp)
            parent = candp
            if np.isfinite(cp) and cp < best_c:
                best_h = candp
                best_c = cp
                best_n = nparent
    except Exception:
        parent = None

    if best_h is None:
        n0 = 90
        best_h = make_candidate(n0, rng, 3)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    if best_h is None or not np.isfinite(best_c):
        n0 = 90
        best_h = make_candidate(n0, rng, 0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    if best_n < 30 or best_n > 450:
        base = 90
    else:
        base = int(best_n)
    if base % 2 == 1:
        base += 1

    nset = set()
    nset.add(base)
    for mult in (1, 2, 4):
        nn = base * mult
        if 30 <= nn <= 450:
            if nn % 2 == 1:
                nn += 1
            nset.add(nn)
    for nn in (90, 180, 270, 360):
        if base <= nn <= 450:
            nset.add(nn)
    if parent is not None and parent.size > 450:
        nset.add(450)

    nlist = sorted(nn for nn in nset if 30 <= nn <= 450)
    if not nlist:
        nlist = [90]

    work_deadline = deadline - min(180.0, 0.08 * max(1.0, deadline - start))
    if work_deadline < start + 60.0:
        work_deadline = deadline - 10.0

    weights = [0.50, 0.30, 0.15, 0.05, 0.05, 0.05, 0.05]
    while len(weights) < len(nlist):
        weights.append(0.05)
    weights = weights[:len(nlist)]
    wsum = float(sum(weights))
    if wsum <= 0.0:
        weights = [1.0 / float(len(nlist))] * len(nlist)
        wsum = 1.0
    weights = [float(w) / wsum for w in weights]

    now = start
    for idx, n in enumerate(nlist):
        now = time.perf_counter()
        if now >= work_deadline:
            break
        rem_stages = len(nlist) - idx
        if rem_stages <= 1:
            stage_deadline = work_deadline
        else:
            ws = float(sum(weights[idx:]))
            frac = weights[idx] / ws if ws > 0.0 else 1.0
            stage_deadline = now + frac * (work_deadline - now)

        if stage_deadline - now < 8.0:
            continue

        starts = []

        if best_h is not None:
            s = upsample_h(best_h, n)
            s = project_density(s, 0.5 * float(n))
            if s.size == n and np.all(np.isfinite(s)):
                starts.append(s)

        if parent is not None:
            s = upsample_h(parent, n)
            s = project_density(s, 0.5 * float(n))
            dup = False
            for u in starts:
                if vector_close(s, u):
                    dup = True
                    break
            if not dup and s.size == n and np.all(np.isfinite(s)):
                starts.append(s)

        modes = (3, 0, 5, 1, 2, 4, 6)
        max_starts = 6 if n <= 120 else 4
        for mm in modes:
            if len(starts) >= max_starts:
                break
            s = make_candidate(n, rng, mm)
            dup = False
            for u in starts:
                if vector_close(s, u):
                    dup = True
                    break
            if not dup:
                starts.append(s)

        if not starts:
            starts = [make_candidate(n, rng, 3)]

        evals = []
        for s in starts:
            s = project_density(s, 0.5 * float(n))
            c = evaluate_erdos(s)
            if np.isfinite(c):
                evals.append((c, s))

        if not evals:
            continue

        evals.sort(key=_eval_key)

        selected = []
        for i in range(min(2, len(evals))):
            selected.append(evals[i][1])

        if len(evals) > 2 and (stage_deadline - now) > 70.0:
            ri = int(rng.integers(2, len(evals)))
            selected.append(evals[ri][1])

        for si in range(len(selected)):
            now = time.perf_counter()
            if now >= stage_deadline:
                break
            rem_sel = len(selected) - si
            per = (stage_deadline - now) / float(max(1, rem_sel))
            sub_deadline = now + per
            if sub_deadline - now < 4.0:
                break

            h_opt, c_opt = local_search(selected[si], n, sub_deadline, rng, max_iter=1000)
            if np.isfinite(c_opt) and c_opt < best_c - 1e-12:
                best_h = h_opt
                best_c = c_opt
                best_n = n

            if time.perf_counter() >= stage_deadline - 1.0:
                break

    now = time.perf_counter()
    if best_h is not None and deadline - now > 8.0:
        n_polish = int(best_n)
        if n_polish < 2:
            n_polish = 90
            h_start = make_candidate(n_polish, rng, 3)
        elif n_polish > 450:
            n_polish = 450
            h_start = upsample_h(best_h, n_polish)
        else:
            h_start = best_h

        h_opt, c_opt = local_search(h_start, n_polish, deadline, rng, max_iter=1000)
        if np.isfinite(c_opt) and c_opt < best_c - 1e-12:
            best_h = h_opt
            best_c = c_opt
            best_n = n_polish

    now = time.perf_counter()
    if best_h is not None and deadline - now > 25.0 and int(best_n) < 360:
        n2 = min(2 * int(best_n), 450)
        h_start = upsample_h(best_h, n2)
        h_opt, c_opt = local_search(h_start, n2, deadline, rng, max_iter=700)
        if np.isfinite(c_opt) and c_opt < best_c - 1e-12:
            best_h = h_opt
            best_c = c_opt
            best_n = n2

    if best_h is None:
        n0 = 90
        best_h = make_candidate(n0, rng, 3)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    final_h = project_density(best_h, 0.5 * float(int(best_h.size)))
    final_c = evaluate_erdos(final_h)

    if not is_feasible(final_h) or not np.isfinite(final_c):
        n0 = 90
        final_h = make_candidate(n0, rng, 3)
        final_c = evaluate_erdos(final_h)

    return final_h, float(final_c), int(final_h.size)
