def project_box_sum(z,S):
    z=np.nan_to_num(np.asarray(z,dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    n=z.size
    if n==0: return z.copy()
    if S < 0: S=0.0
    if S > n: S=float(n)
    lo=float(np.min(z)) - 1.0
    hi=float(np.max(z)) + 1.0
    for _ in range(70):
       mid=0.5*(lo+hi)
       h=np.clip(z-mid,0.0,1.0)
       if h.sum() > S:
          lo=mid
       else:
          hi=mid
    h=np.clip(z-0.5*(lo+hi),0.0,1.0)
    for _ in range(100):
       diff=S - float(h.sum())
       if abs(diff) < 1e-13: break
       if diff > 0:
          mask = h < 1.0 - 1e-12
       else:
          mask = h > 1e-12
       cnt=int(np.count_nonzero(mask))
       if cnt == 0: break
       h[mask] += diff/cnt
       h = np.clip(h,0.0,1.0)
    diff=S - float(h.sum())
    if abs(diff) > 1e-12:
       if diff > 0:
          order=np.argsort(1.0-h)
          for idx in order:
             cap=1.0-h[idx]
             add=cap if cap < diff else diff
             if add <= 0: continue
             h[idx] += add; diff -= add
             if abs(diff) < 1e-13: break
       else:
          order=np.argsort(h)
          for idx in order:
             cap=h[idx]
             sub=cap if cap < -diff else -diff
             if sub <=0: continue
             h[idx] -= sub; diff += sub
             if abs(diff) < 1e-13: break
    return h

def compute_c5(h, n=None):
    h = np.asarray(h, dtype=float)
    if n is None:
        n = int(len(h))
    n = int(n)
    if n <= 0:
        return float("inf")
    if h.size != n:
        h = np.resize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fast_c5(h, n):
    n = int(n)

    h = np.asarray(h, dtype=float)

def full_corr_h(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = np.resize(h, n)
    b = 1.0 - h
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    Fh = np.fft.rfft(h, n=L)
    Fb = np.fft.rfft(b, n=L)
    r = np.fft.irfft(Fh * np.conj(Fb), n=L).real
    c = np.empty(need, dtype=float)
    c[n - 1:] = r[:n]
    if n > 1:
        c[:n - 1] = r[L - (n - 1):L][::-1]
    return c


def project_box_sum(z, S):
    z = np.nan_to_num(np.asarray(z, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    n = int(z.size)

        h = np.clip(z - mid, 0.0, 1.0)

    x = np.nan_to_num(np.asarray(x, dtype=float), nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(x, 0.0, 1.0)

    cmax = float(np.max(c))
    sh = beta * (c - cmax)
    ex = np.exp(sh)
    Z = float(ex.sum())
    if Z <= 1e-300:
        W = np.full(2 * n - 1, 1.0 / max(1, 2 * n - 1), dtype=float)
    else:
        W = ex / Z
    obj = scale * (cmax + np.log(Z) / beta)
    return h, W, obj

def compute_logsum_grad(h, n, beta):
    h = np.clip(np.nan_to_num(np.asarray(h, dtype=float), nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    c = full_corr_h(h, n)
    cmax = float(np.max(c))
    ex = np.exp(beta * (c - cmax))

    if Z <= 1e-300:
        W = np.full(2 * n - 1, 1.0 / max(1, 2 * n - 1), dtype=float)
    else:
        W = ex / Z
    b = 1.0 - h
    off = n - 1
    A = np.convolve(W[::-1], b, mode="full")[off:off + n]
    B = np.convolve(W, h, mode="full")[off:off + n]
    return A - B

def unique_inits(inits, tol, max_out):

def improve_by_shifts(h, n, rng, deadline, full_search):
    n = int(n)
    if n <= 0:
        return h, float("inf")
    h = make_feasible(h, n)
    best_h = h.copy()
    best_c = fast_c5(h, n)
    deadline = float(deadline)

    variants = [h, h[::-1], 1.0 - h, (1.0 - h)[::-1], 0.5 * (h + h[::-1])]
    for v in variants:
        v = make_feasible(v, n)
        if full_search and n <= 200:
            step = 1
        else:
            stop = min(n, 60)
            step = max(1, int(np.ceil(n / float(max(1, stop)))))

                break
            hh = np.roll(v, s)
            c = fast_c5(hh, n)
            if c < best_c - 1e-14:
                best_h = hh.copy()
                best_c = c
    return best_h, best_c

        rng = np.random.RandomState(int(seed))
    except Exception:
        rng = np.random.RandomState(42)

    parent = None
    try:
        parent = initial_h_values
    except Exception:
        parent = None
    if parent is not None:
        try:
            parent = np.asarray(parent, dtype=float)
        except Exception:
            parent = None
        if parent is not None and parent.size == 0:
            parent = None

    if parent is not None:
        best_n = int(parent.size)
        best_h = np.array(parent, dtype=float, copy=True)
    else:
        best_n = 160
        inits = make_structured_inits(best_n, rng)
        best_h = None
        best_tmp = 1e100
        for h0 in inits[:8]:
            h0 = make_feasible(h0, best_n)
            c0 = fast_c5(h0, best_n)
            if best_h is None or c0 < best_tmp:
                best_h = h0
                best_tmp = c0
        if best_h is None:
            best_h = np.full(best_n, 0.5, dtype=float)

    best_n = int(best_n)
    best_h = make_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    rem = _GLOBAL_DEADLINE - time.time()
    if rem > 60 and best_n <= 600:
        hh, cc = improve_by_shifts(best_h, best_n, rng, _GLOBAL_DEADLINE - 5.0, best_n <= 200)
        if cc < best_c - 1e-12:
            best_h = make_feasible(hh, best_n)
            exact = compute_c5(best_h, best_n)
            best_c = exact

    ordered = order_candidate_ns(best_n)
    for n in ordered:
        n = int(n)
        if n <= 0 or n > 600:
            continue
        rem = _GLOBAL_DEADLINE - time.time()
        if rem < 120:
            break

        inits = []
        inits.append(interpolate_to_n(best_h, n))
        if parent is not None and parent.size != n and parent.size <= 600:
            inits.append(interpolate_to_n(parent, n))
        struct = make_structured_inits(n, rng)
        for h0 in struct[:4]:
            inits.append(h0)
        inits = unique_inits(inits, 0.04, 8)

        ntry = 3
        if n == best_n:
            ntry = 4
        if rem < 400:
            ntry = 2

        maxit = 200
        call_limit = 60.0
        if n <= 180:
            maxit = 350
            call_limit = 80.0
        elif n <= 240:
            maxit = 250
            call_limit = 100.0
        elif n <= 320:
            maxit = 150
            call_limit = 120.0
        else:
            maxit = 80
            call_limit = 80.0

        for hi in range(min(ntry, len(inits))):
            h0 = inits[hi]
            rem = _GLOBAL_DEADLINE - time.time()
            if rem < 90:
                break

            cur = make_feasible(h0, n)
            c_cur = fast_c5(cur, n)
            if c_cur < best_c - 1e-12:
                best_h = make_feasible(cur, n)
                exact = compute_c5(best_h, n)
                if exact < best_c:
                    best_c = exact
                    best_n = n

            beta_list = [8, 20, 50, 100]
            if n > 240:
                beta_list = [10, 30, 80]
            if rem < 300:
                beta_list = beta_list[:2]

            for beta in beta_list:
                rem = _GLOBAL_DEADLINE - time.time()
                if rem < 45:
                    break
                h1, c1, _ = run_smooth_slsqp(n, cur, beta, maxit, call_limit)
                if h1 is not None:
                    if c1 < c_cur + 0.01:
                        cur = h1
                        c_cur = c1
                    if c1 < best_c - 1e-12:
                        best_h = make_feasible(h1, n)
                        exact = compute_c5(best_h, n)
                        if exact < best_c:
                            best_c = exact
                            best_n = n

            rem = _GLOBAL_DEADLINE - time.time()
            if rem > 45:
                est = int(max(0, rem - 25) * 2000)
                if n > 240:
                    est = int(est * 0.25)
                maxit_local = min(300000, max(2000, est))
                hh, cc = local_anneal(cur, n, maxit_local, rng, _GLOBAL_DEADLINE - 5.0, max(1e-7, 3e-5))
                if cc < c_cur:
                    cur = hh
                    c_cur = cc

            rem = _GLOBAL_DEADLINE - time.time()
            if n <= 220 and rem > 30:
                hh, cc = improve_by_shifts(cur, n, rng, _GLOBAL_DEADLINE - 5.0, n <= 180)
                if cc < c_cur:
                    cur = hh
                    c_cur = cc

            if c_cur < best_c - 1e-12:
                best_h = make_feasible(cur, n)
                exact = compute_c5(best_h, n)
                if exact < best_c:
                    best_c = exact
                    best_n = n

    rem = _GLOBAL_DEADLINE - time.time()
    if rem > 40 and 0 < best_n <= 600:
        h1, c1, _ = run_smooth_slsqp(best_n, best_h, 150, 300, 80)
        if h1 is not None and c1 < best_c + 0.01:
            best_h = make_feasible(h1, best_n)
            exact = compute_c5(best_h, best_n)
            if exact < best_c:
                best_c = exact

        rem = _GLOBAL_DEADLINE - time.time()
        if rem > 30:
            est = int(max(0, rem - 15) * 2500)
            if best_n > 240:
                est = int(est * 0.25)
            maxit_local = min(400000, max(5000, est))
            hh, cc = local_anneal(best_h, best_n, maxit_local, rng, _GLOBAL_DEADLINE - 2.0, 1e-7)
            if cc < best_c - 1e-12:
                best_h = make_feasible(hh, best_n)
                exact = compute_c5(best_h, best_n)
                if exact < best_c:
                    best_c = exact

        rem = _GLOBAL_DEADLINE - time.time()
        if best_n <= 220 and rem > 20:
            hh, cc = improve_by_shifts(best_h, best_n, rng, _GLOBAL_DEADLINE - 1.0, best_n <= 180)
            if cc < best_c - 1e-12:
                best_h = make_feasible(hh, best_n)
                exact = compute_c5(best_h, best_n)
                if exact < best_c:
                    best_c = exact

        rem = _GLOBAL_DEADLINE - time.time()
        if rem > 60:
            h1, c1, _ = run_smooth_slsqp(best_n, best_h, 220, 250, 45)
            if h1 is not None and c1 < best_c - 1e-12:
                best_h = make_feasible(h1, best_n)
                exact = compute_c5(best_h, best_n)
                if exact < best_c:
                    best_c = exact

            rem = _GLOBAL_DEADLINE - time.time()
            if rem > 20:
                est = int(max(0, rem - 5) * 2500)
                if best_n > 240:
                    est = int(est * 0.25)
                maxit_local = min(120000, max(1000, est))
                hh, cc = local_anneal(best_h, best_n, maxit_local, rng, _GLOBAL_DEADLINE - 1.0, 1e-8)
                if cc < best_c - 1e-12:
                    best_h = make_feasible(hh, best_n)
                    exact = compute_c5(best_h, best_n)
                    if exact < best_c:
                        best_c = exact

    best_h = make_feasible(best_h, best_n)
    final_c = compute_c5(best_h, best_n)
    if np.isfinite(final_c) and final_c < best_c + 1e-14:
        best_c = final_c

    return best_h, float(best_c), int(best_n)
