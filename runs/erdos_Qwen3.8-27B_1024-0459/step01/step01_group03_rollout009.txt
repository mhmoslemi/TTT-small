def fourier_h(params, n):
    """Evaluate Fourier series h(x) = 0.5 + sum(params) on grid of n points."""
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)
    m = len(params) // 2
    for k in range(m):
        a = params[2 * k]
        b = params[2 * k + 1]
        h += a * np.cos(k * np.pi * x) + b * np.sin(k * np.pi * x)
    return h

        pproj = project_density(parent, 0.5 * nparent)
        cp = evaluate_erdos(pproj)
        if cp < best_c:
            best_h = pproj
            best_c = cp
            best_n = nparent

    # Phase 1: Generate multiple starting candidates at various resolutions
    resolutions = [180, 360, 480]
    candidates = []  # list of (h, n, c)

    if best_h is not None:
        for n_res in resolutions:
            if time.perf_counter() >= deadline:
                break
            h_up = upsample_h(best_h, n_res)
            c = evaluate_erdos(h_up)
            candidates.append((h_up, n_res, c))
            if c < best_c:
                best_h = h_up
                best_c = c
                best_n = n_res

    # Add random structured candidates
    for n_res in (180, 360):
        if time.perf_counter() >= deadline:
            break
        for mode in range(4):
            h_rand = random_structured_h(n_res, rng, mode)
            c = evaluate_erdos(h_rand)
            candidates.append((h_rand, n_res, c))

    # Fourier-optimized candidates
    if parent is not None:
        for M in (8, 12):
            if time.perf_counter() >= deadline:
                break
            n_f = 360
            fourier_dl = min(deadline, time.perf_counter() + 40.0)
            h_f = fourier_optimize(n_f, M, fourier_dl, rng)
            if h_f is not None:
                c = evaluate_erdos(h_f)
                candidates.append((h_f, n_f, c))

    # Sort candidates by quality
    candidates.sort(key=lambda x: x[2])

    # Phase 2: Subgradient optimization on best candidates
    for idx, (h_cand, n_cand, c_cand) in enumerate(candidates[:6]):
        if deadline - time.perf_counter() < 30.0:
            break

        # Time allocation: more time for better candidates
        if idx == 0:
            sub_dl = min(deadline, time.perf_counter() + 300.0)
        elif idx == 1:
            sub_dl = min(deadline, time.perf_counter() + 150.0)
        else:
            sub_dl = min(deadline, time.perf_counter() + 80.0)

        if sub_dl - time.perf_counter() < 20.0:
            break

        h_out, c_out = subgradient_optimize(h_cand, n_cand, sub_dl, top_k=4, max_iter=60000)
        if c_out < best_c:
            best_h = h_out
            best_c = c_out
            best_n = n_cand

    # Phase 3: Lag attack on best solution
    if best_h is not None and deadline - time.perf_counter() > 60.0:
        n_best = int(best_h.size)
        la_dl = min(deadline, time.perf_counter() + 200.0)
        h_la, c_la = lag_attack(best_h, n_best, la_dl, max_iter=3000, top_k=3)
        if c_la < best_c:
            best_h = h_la
            best_c = c_la
            best_n = n_best

    # Phase 4: Subgradient again with different top_k
    if best_h is not None and deadline - time.perf_counter() > 40.0:
        n_best = int(best_h.size)
        sg2_dl = min(deadline, time.perf_counter() + 150.0)
        h_sg2, c_sg2 = subgradient_optimize(best_h, n_best, sg2_dl, top_k=2, max_iter=40000)
        if c_sg2 < best_c:
            best_h = h_sg2
            best_c = c_sg2
            best_n = n_best

    # Phase 5: Fine pair search
    if best_h is not None and deadline - time.perf_counter() > 30.0:
        n_best = int(best_h.size)
        ps_dl = min(deadline, time.perf_counter() + 120.0)
        # Need to pass rng - use a helper
        global _RNG
        _RNG = rng
        h_ps, c_ps = pair_search_fine_wrapper(best_h, n_best, ps_dl)
        if c_ps < best_c:
            best_h = h_ps
            best_c = c_ps
            best_n = n_best

    # Phase 6: Try upsample + re-optimize at higher resolution
    if best_h is not None and deadline - time.perf_counter() > 80.0:
        n_cur = int(best_h.size)
        if n_cur < 480:
            n_hi = 480
            h_hi = upsample_h(best_h, n_hi)
            hi_dl = min(deadline, time.perf_counter() + 60.0)
            h_hi2, c_hi = subgradient_optimize(h_hi, n_hi, hi_dl, top_k=4, max_iter=30000)
            if c_hi < best_c:
                best_h = h_hi2
                best_c = c_hi
                best_n = n_hi

    # Final: one more round of refinement
    if best_h is not None and deadline - time.perf_counter() > 20.0:
        n_best = int(best_h.size)
        fin_dl = min(deadline, time.perf_counter() + 15.0)
        h_fin, c_fin = subgradient_optimize(best_h, n_best, fin_dl, top_k=3, max_iter=10000)
        if c_fin < best_c:
            best_h = h_fin
            best_c = c_fin
            best_n = n_best

    # Final projection and evaluation
    if best_h is None:
        # Fallback
        n_fallback = 90
        best_h = project_density(np.full(n_fallback, 0.5), 0.5 * n_fallback)
        best_c = evaluate_erdos(best_h)
        best_n = n_fallback

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)

def pair_search_fine_wrapper(h0, n, deadline):
    """Wrapper that uses global RNG."""
    target = 0.5 * n
    h = project_density(h0, target)
    best_c = evaluate_erdos(h)
    center = n - 1
    stall = 0
    top_k = 4
    max_iter = 5000

    local_rng = np.random.default_rng(12345)

    for itr in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break
        if stall > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        if not np.all(np.isfinite(corr)):
            break
        top_idx = np.argsort(corr)[-top_k:]
        improved = False

        g = corr_gradient_at_lags(h, top_idx)
        if float(np.var(g)) < 1e-14:
            break

        sorted_g = np.argsort(g)
        neg_candidates = sorted_g[:5]
        pos_candidates = sorted_g[-5:]

        for ii in neg_candidates:
            if improved:
                break
            ii = int(ii)
            for jj in pos_candidates:
                if improved:
                    break
                jj = int(jj)
                if ii == jj:
                    continue
                cap_pos = 1.0 - h[ii]
                cap_neg = h[jj]
                cap = min(cap_pos, cap_neg)
                if cap < 1e-12:
                    continue
                for frac in (1.0, 0.5, 0.2):
                    delta = frac * cap
                    cand = h.copy()
                    cand[ii] += delta
                    cand[jj] -= delta
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        h = cand
                        best_c = c
                        improved = True
                        break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall % 200 == 50:
                mask = local_rng.random(n) < 0.05
                h = project_density(h + local_rng.normal(0, 0.01, n) * mask.astype(float), target)

    return project_density(h, target), evaluate_erdos(project_density(h, target))
