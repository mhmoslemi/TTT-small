    else:

        g[nL:] += (1.0 - h[:n-nL])
        g[:n-nL] -= h[nL:]
    return g

def project_h(h, target):
    """Project to {h : 0<=h<=1, sum(h)=target}"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Bisection for the right shift
    lo = min(-1.0, -diff) - 0.1
    hi = max(1.0, diff) + 0.1
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    # Fix any remaining error
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = np.where((h > 0) & (h < 1))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
            # Final fix
            diff = target - h.sum()
            if abs(diff) > 1e-15:
                if len(free) > 0:
                    h[free[0]] += diff
                    h = np.clip(h, 0.0, 1.0)
    return h

    """Upsample h to n_new points using linear interpolation"""
    n_old = h.size
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new / 2.0)

        g[mL:] += (1.0 - h[:n-mL])
        g[:n-mL] -= h[mL:]

import time
import numpy as np

def eval_c5(h):
    """Evaluate the C5 objective."""
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}."""
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        free = np.where((h > 1e-15) & (h < 1.0 - 1e-15))[0]
        if len(free) > 0:
            per = diff / len(free)
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
    # Final correction
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        free = np.where((h > 0) & (h < 1))[0]
        if len(free) > 0:
            h[free[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, n_new):
    """Upsample h to n_new points."""
    n_old = h.size
    if n_old == n_new:
        return h.copy()
    if n_old == 0:
        return project_feasible(np.full(n_new, 0.5), n_new/2.0)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new * 0.5)

def grad_at_lag(h, idx):
    """Gradient of full-correlation value at index idx w.r.t. h."""
    n = h.size
    center = n - 1
    L = idx - center
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0*h
    elif L > 0:
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    else:
        mL = -L
        g[mL:] += (1.0 - h[:n-mL])
        g[:n-mL] -= h[mL:]
    return g

def grad_descent_step(h, target, k_active, step_sizes):
    """One step of gradient descent on active lags. Returns (new_h, improved)."""
    n = h.size
    corr = np.correlate(h, 1.0-h, mode='full')
    top_k = np.argpartition(corr, -k_active)[-k_active:]
    g = np.zeros(n)
    for k in top_k:
        g += grad_at_lag(h, int(k))
    g /= len(top_k)
    g -= g.mean()
    
    best_h = h
    best_c = float(np.max(corr))
    for alpha in step_sizes:
        cand = h - alpha * g
        cand = project_feasible(cand, target)
        c = float(np.max(np.correlate(cand, 1.0-cand, mode='full')))
        if c < best_c - 1e-14:
            best_h = cand
            best_c = c
    return best_h, best_c < float(np.max(corr))

def pair_descent(h, target, deadline, max_iter=10000, k_active=8, p_pairs=8, shake_every=200):
    """Coordinate descent via pair swaps."""
    n = h.size
    best_h = h.copy()
    best_raw = float(np.max(np.correlate(best_h, 1.0-best_h, mode='full')))
    stall = 0
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        cur_raw = float(np.max(corr))
        
        top_k = np.argpartition(corr, -k_active)[-k_active:]
        g = np.zeros(n)
        for k in top_k:
            g += grad_at_lag(best_h, int(k))
        g /= len(top_k)
        g -= g.mean()
        
        # Get candidate indices
        order = np.argsort(g)
        negs = order[:p_pairs]
        poss = order[-p_pairs:][::-1]
        
        best_delta = 0.0
        best_ii = -1
        best_jj = -1
        best_cand = best_raw
        
        for ii in negs:
            ii = int(ii)
            for jj in poss:
                jj = int(jj)
                if ii == jj:
                    continue
                max_d = min(best_h[jj], 1.0 - best_h[ii])
                if max_d < 1e-12:
                    continue
                for frac in (0.5, 0.8, 1.0):
                    d = frac * max_d
                    cand = best_h.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = float(np.max(np.correlate(cand, 1.0-cand, mode='full')))
                    if c < best_cand:
                        best_cand = c
                        best_delta = d
                        best_ii = ii
                        best_jj = jj
        
        if best_ii >= 0:
            best_h[best_ii] += best_delta
            best_h[best_jj] -= best_delta
            best_h = np.clip(best_h, 0.0, 1.0)
            best_h = project_feasible(best_h, target)
            best_raw = best_cand
            stall = 0
        else:
            stall += 1
            if stall > shake_every:
                # Shake: add small random perturbation
                noise = rng.normal(0, 0.02, n)
                cand = project_feasible(best_h + noise, target)
                c = float(np.max(np.correlate(cand, 1.0-cand, mode='full')))
                if c < best_raw + 0.01:  # Allow slight worsening
                    best_h = cand
                    best_raw = c
                stall = 0
            
            if stall > shake_every * 3:
                break
    
    return best_h, best_raw * (2.0/n)
