```python
def smooth_max_grad(h, n, tau):
    """Compute gradient of smooth max correlation objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -40, 40)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)  # softmax weights, length 2n-1
    
    # grad[h[j]] = dx * sum_k w[k] * d(corr_k)/dh[j]
    # corr_{n-1+s} depends on h[j] through:
    #   (1-h[j+s]) if 0<=j+s<n, coefficient +1
    #   -h[j-s] if 0<=j-s<n, coefficient -1
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        # s = k - (n-1), k is the index in corr (0 to 2n-2)
        k = n - 1 + s
        wk = w[k] * dx
        if abs(wk) < 1e-15:
            continue
        # For each j: contrib = wk * [(1-h[j+s]) if valid - h[j-s] if valid]
        # (1-h[j+s]) is valid when 0 <= j+s < n, i.e., max(0,-s) <= j <= min(n-1, n-1-s)
        lo1 = max(0, -s)
        hi1 = min(n-1, n-1-s)
        if lo1 <= hi1:
            # h[j+s] for j in [lo1, hi1], j+s in [s+lo1, s+hi1] = valid range
            # (1-h[j+s])
            idx1 = np.arange(lo1, hi1+1)
            grad[idx1] += wk * (1.0 - h[idx1 + s])
        
        # -h[j-s] is valid when 0 <= j-s < n, i.e., max(0,s) <= j <= min(n-1, n-1+s)
        lo2 = max(0, s)
        hi2 = min(n-1, n-1+s)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2+1)
            grad[idx2] += wk * (-h[idx2 - s])
    
    return grad
```

import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        for i in idx:
            h[i] = np.clip(h[i] + per, 0.0, 1.0)
    return h

def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def obj_smooth(x, n, tau, lam):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -40.0, 40.0)
    exp_s = np.exp(shifted)
    sm = (m + tau * np.log(np.sum(exp_s))) * (2.0 / n)
    pen = lam * (h.sum() - n / 2.0) ** 2
    return sm + pen

def grad_smooth(x, n, tau, lam):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -40.0, 40.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    dx = 2.0 / n
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        k = n - 1 + s
        wk = w[k] * dx
        if abs(wk) < 1e-15:
            continue
        # Term 1: (1-h[j+s]) for 0<=j+s<n
        lo1 = max(0, -s)
        hi1 = min(n-1, n-1-s)
        if lo1 <= hi1:
            idx1 = np.arange(lo1, hi1 + 1)
            grad[idx1] += wk * (1.0 - h[idx1 + s])
        # Term 2: -h[j-s] for 0<=j-s<n  (i.e., j in [max(0,s), min(n-1, n-1+s)])
        lo2 = max(0, s)
        hi2 = min(n-1, n-1 + s)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            grad[idx2] -= wk * h[idx2 - s]
    
    # Penalty gradient
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    
    return grad

    elif style == 3:

    else:

        h = 0.5 + 0.35 * np.cos(np.pi * 4.0 * x / 2.0) + 0.1 * np.sin(np.pi * 7.0 * x / 2.0)
        return np.clip(h, 0.0, 1.0)
    else:
        h = rng.random(n) * 0.5 + 0.25
        return np.clip(h, 0.0, 1.0)


def run_slsqp_one(h0, n, tau, lam, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        fun=obj_smooth, x0=h0, method='SLSQP',
        jac=grad_smooth, args=(n, tau, lam),
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
    )
    return np.clip(res.x, 0.0, 1.0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    parent_h = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            arr = np.asarray(initial_h_values, dtype=float)
            if np.size(arr) > 0:
                parent_h = arr.copy()
    except Exception:
        pass
    if parent_h is None or np.size(parent_h) == 0:
        parent_h = np.full(180, 0.5)

    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    n_parent = len(best_h)

    lam = 80.0

    # Phase 1: Optimize at parent resolution with decreasing temperature
    n = n_parent
    x0 = best_h.copy()
    tau_list = [0.4, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008]

    for tau in tau_list:
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.45:
            break
        mi = 4000 if n <= 200 else 2000
        try:
            h_new = run_slsqp_one(x0, n, tau, lam, mi, 1e-15)
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            x0 = h_new
        except Exception:
            pass

    # Phase 2: Diverse starts at parent resolution
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.55:
        for style in [1, 2, 3, 4, 5, 6]:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.65:
                break
            x_init = make_initial(n, rng, style)
            x_init = fix_sum(x_init, n)
            for tau in [0.15, 0.05, 0.02, 0.008]:
                elapsed = time.time() - t0
                if elapsed > budget_s * 0.62:
                    break
                mi = 2000 if n <= 200 else 1000
                try:
                    h_new = run_slsqp_one(x_init, n, tau, lam, mi, 1e-13)
                    h_new = fix_sum(h_new, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    x_init = h_new
                except Exception:
                    pass

    # Phase 3: Try different resolutions
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.70:
        n_options = sorted(set([n, n * 2, n + 40]))
        n_options = [no for no in n_options if 40 <= no <= 500]
        for n_alt in n_options:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.75:
                break
            if n_alt == n:
                continue
            x_alt = upsample_to(best_h, n_alt)
            x_alt = fix_sum(x_alt, n_alt)
            for tau in [0.1, 0.04, 0.015, 0.005]:
                elapsed = time.time() - t0
                if elapsed > budget_s * 0.72:
                    break
                mi = 2000 if n_alt <= 200 else 1000
                try:
                    h_new = run_slsqp_one(x_alt, n_alt, tau, lam, mi, 1e-13)
                    h_new = fix_sum(h_new, n_alt)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    x_alt = h_new
                except Exception:
                    pass
            # Go back to original n if we found something better at n_alt
            if best_h.size != n:
                best_h_local = upsample_to(best_h, n)
                best_h_local = fix_sum(best_h_local, n)
                best_c5_local = compute_c5(best_h_local)
                if best_c5_local < best_c5 - 1e-12:
                    best_c5 = best_c5_local
                    best_h = best_h_local

    # Phase 4: Fine polish - very low temperature, small perturbations
    n_best = len(best_h)
    best_h = fix_sum(best_h, n_best)
    best_c5 = compute_c5(best_h)

    for r in range(40):
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.97:
            break
        scale = 0.003 + 0.012 * (r / 40.0)
        x_pert = best_h + rng.normal(0, scale, n_best)
        x_pert = fix_sum(x_pert, n_best)
        tau = 0.001 if r < 20 else 0.0005
        mi = 2500 if n_best <= 200 else 1200
        try:
            h_new = run_slsqp_one(x_pert, n_best, tau, lam, mi, 1e-15)
            h_new = fix_sum(h_new, n_best)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 5: Ultra-fine final push on the best solution
    elapsed = time.time() - t0
    if elapsed < budget_s * 0.98:
        for tau in [0.0005, 0.0003, 0.0002]:
            elapsed = time.time() - t0
            if elapsed > budget_s * 0.99:
                break
            mi = 5000 if n_best <= 200 else 2500
            try:
                h_new = run_slsqp_one(best_h, n_best, tau, lam, mi, 1e-16)
                h_new = fix_sum(h_new, n_best)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
