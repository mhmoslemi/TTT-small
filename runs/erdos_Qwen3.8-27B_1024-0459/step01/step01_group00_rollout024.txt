```python
    max_val = np.max(corr)
```

```python
    n = len(h)
    h = h.copy()
    for _ in range(50):
        # First clip
        h = np.clip(h, 0, 1)
```

```python
    """Return the correlation vector (before multiplying by dx)."""

def binary_local_search(h_binary, n, max_iters=100000):
    """Local search: flip pairs of bits to maintain sum constraint."""
    h = h_binary.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    best_max = np.max(corr)
    
    for _ in range(max_iters):
        # Try flipping a 1 to 0 and a 0 to 1
        i_1 = ones[np.random.randint(len(ones))]
        i_0 = zeros[np.random.randint(len(zeros))]
        if i_1 == i_0:
            continue
        h[i_1] = 0
        h[i_0] = 1
        corr = np.correlate(h, 1-h, mode='full')
        new_max = np.max(corr)
        if new_max > best_max:
            h[i_1] = 1
            h[i_0] = 0
        elif new_max < best_max:
            best_max = new_max
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    return h, best_max * dx
```

```python
def make_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:

            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_constraints(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_gradient(corr, beta):
    """Compute weights for smooth max gradient."""
    shifted = corr - np.max(corr)
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / np.sum(exp_vals)
    return weights

def compute_gradient(h, n, weights):
    """Compute gradient of weighted sum of correlations.
    weights[i] corresponds to correlation at lag (i - (n-1))."""
    grad = np.zeros(n)
    n1 = n - 1
    for idx in range(2 * n - 1):
```

```python
def smooth_max_optimize(h_init, n, beta=50.0, n_iterations=5000, lr_start=0.001, lr_end=0.0001):
    h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    corr = np.correlate(h, 1-h, mode='full')
    for it in range(n_iterations):
        # Linear schedule for beta and learning rate
        t = it / max(n_iterations - 1, 1)
        beta_cur = beta * (1 + 2*t)  # increase beta over time
        lr = lr_start * (lr_end / lr_start) ** t
        
        # Compute smooth max weights
        weights = smooth_max_gradient(corr, beta_cur)
        
        # Compute gradient
        grad = compute_gradient(h, n, weights)
        
        # Quadratic regularization for stability
        # Add small L2 to prevent oscillation
        grad -= 0.01 * (h - 0.5)
        
        # Take step
        h_new = h - lr * grad
        h_new = project_onto_constraints(h_new, n)
        
        # Evaluate
        corr_new = np.correlate(h_new, 1-h_new, mode='full')
        c5_new = np.max(corr_new) * (2.0 / n)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            corr = corr_new
        else:
            # Reject, maybe reduce lr
            h = h  # keep current
            corr = corr  # keep current
            # Don't change h
    
    return best_h, best_c5
```
