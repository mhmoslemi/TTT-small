def project_density(x, target):

    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)

    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    # Bisection for water-filling
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(h))
    if abs(res) > 0:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:

            h = np.clip(h, 0.0, 1.0)

    res = target - float(np.sum(h))
