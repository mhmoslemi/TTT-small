def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            # All saturated, distribute to all
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(corr)
    # Use a "soft max" that's close to max for large beta
    shifted = corr - m
    # Clip to avoid overflow
    shifted = np.clip(shifted, -100.0/beta, 0.0)
    exps = np.exp(beta * shifted)
    return m + np.log(np.sum(exps)) / beta

    """Local search on binary solutions using pair swaps."""
    start = time.time()
    h = h_bin.copy()
    current_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -100.0 / beta, 0.0)
    exps = np.exp(beta * shifted)
    s = np.sum(exps)
    obj = (m + np.log(s) / beta) * dx
    weights = exps / s
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        s_val = k_idx - (n - 1)
        for i in range(n):
            if 0 <= i + s_val < n:
                grad[i] += w * (1.0 - h[i + s_val])
            if 0 <= i - s_val < n:
                grad[i] -= w * h[i - s_val]
    grad *= dx
    return obj, grad

def slsqp_optimize(h0, n, beta, maxiter=400):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        return smooth_max_obj_and_grad(x, n, beta)[0]

    def grad(x):
        return smooth_max_obj_and_grad(x, n, beta)[1]

    try:
        result = minimize(obj, h0, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_opt = project_feasible(result.x, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0.copy(), compute_c5(h0)

            h_bin_best[idx[np.argsort(best_h[idx])[:n_ones - n_want]]] = 0.0
        elif n_ones < n_want:
            idx = np.where(h_bin_best == 0.0)[0]
            h_bin_best[idx[np.argsort(-best_h[idx])[:n_want - n_ones]]] = 1.0

        c5_bin = compute_c5(h_bin_best)
        t_sa = min(30, remaining() - 10)
        if t_sa > 5:
            h_sa_final, c5_sa_final = sla_binary(h_bin_best, best_n, t_sa, rng)
            if c5_sa_final < best_c5:
                best_c5 = c5_sa_final
                best_h = h_sa_final.copy()
                print("Final SA improved: C5={:.6f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
