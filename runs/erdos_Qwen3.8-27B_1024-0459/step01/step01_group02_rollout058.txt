def soft_max_obj(h, beta):

                    best_n = n
                    print("  SLSQP raw n={}: C5={:.8f}".format(n, best_c5))
            except Exception:
                pass

    # Phase 2: Binary local search with multiple strategies
    if remaining() > 100:
        even_ns = [n for n in [100, 128, 160, best_n] if n % 2 == 0 and 50 <= n <= 256]
        even_ns = sorted(set(even_ns))
        for n_bs in even_ns:
            if remaining() < 60:
                break
            if n_bs == best_n and len(best_h) == n_bs:
                h_bs = best_h.copy()
            elif parent_h is not None:
                h_bs = interpolate_to_n(parent_h, parent_n, n_bs)
            else:
                h_bs = project_feasible(rng.rand(n_bs), n_bs)

            # Direct binary search
            h_bin, c5_bin = binary_local_search(h_bs, n_bs, 30000, seed)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n_bs
                print("  Binary SA n={}: C5={:.8f}".format(n_bs, best_c5))

            # Kick + search
            if remaining() > 50:
                h_kick, c5_kick = kick_and_search(h_bin, n_bs, 10, 5000, seed + 123)
                if c5_kick < best_c5:
                    best_c5 = c5_kick
                    best_h = h_kick.copy()
                    best_n = n_bs
                    print("  Kick+SA n={}: C5={:.8f}".format(n_bs, best_c5))

            # Random restart binary
            if remaining() > 40:
                h_rand = (rng.rand(n_bs) < 0.5).astype(float)
                h_rand, c5_rand = binary_local_search(h_rand, n_bs, 20000, seed + 777)
                if c5_rand < best_c5:
                    best_c5 = c5_rand
                    best_h = h_rand.copy()
                    best_n = n_bs
                    print("  Random binary n={}: C5={:.8f}".format(n_bs, best_c5))

    # Phase 3: Further SLSQP polishing on best solution
    if remaining() > 60:
        _N_OPT = best_n
        _BETA_OPT = 500.0
        bounds = [(0.0, 1.0)] * best_n
        h_polish = project_feasible(best_h, best_n)
        for beta_val in [300, 800, 2000]:
            if remaining() < 30:
                break
            _BETA_OPT = float(beta_val)
            try:
                res = minimize(_obj_smooth, h_polish, method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-14, 'iprint': 0})
                h_new = project_feasible(res.x, best_n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Polish beta={}: C5={:.8f}".format(beta_val, best_c5))
                h_polish = h_new
            except Exception:
                pass

    # Phase 4: Additional binary search rounds if time allows
    if remaining() > 80 and best_n % 2 == 0:
        n_extra = best_n
        h_seed = best_h
        for round_i in range(3):
            if remaining() < 40:
                break
            h_res, c5_res = kick_and_search(h_seed, n_extra, 8, 10000, seed + 5000 + round_i * 31)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
                best_n = n_extra
                h_seed = h_res
                print("  Extra round {}: C5={:.8f}".format(round_i, best_c5))
            else:
                h_seed = h_seed

    # Final: verify feasibility and compute
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-12:
        final_c5 = best_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
