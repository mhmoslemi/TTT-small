import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(5000):
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-15
        else:
            mask = h < 1.0 - 1e-15
        n_m = mask.sum()
        if n_m == 0:
            break
        h[mask] -= diff / n_m
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(h, n, beta, mu):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    scaled = beta * c
    m = np.max(scaled)
    lse = (m + np.log(np.sum(np.exp(scaled - m)))) / beta
    sv = h.sum() - n / 2.0
    obj = lse + mu * sv * sv
    return float(obj)


def lse_grad(h, n, beta, mu):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w_sum = np.sum(w)
    w = w / w_sum

    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += wk * (1.0 - h[j1 - lag])
            j2 = np.arange(0, n - lag)
            grad[j2] -= wk * h[j2 + lag]
        else:
            m_ = -lag
            j1 = np.arange(m_, n)
            grad[j1] += wk * (1.0 - h[j1 - m_])
            j2 = np.arange(0, n - m_)
            grad[j2] -= wk * h[j2 + m_]
    grad *= dx
    sv = h.sum() - n / 2.0
    grad += 2.0 * mu * sv
    return grad


def hard_max_obj(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def hard_max_grad(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(c)
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        j1 = np.arange(lag, n)
        grad[j1] += (1.0 - h[j1 - lag])
        j2 = np.arange(0, n - lag)
        grad[j2] -= h[j2 + lag]
    else:
        m_ = -lag
        j1 = np.arange(m_, n)
        grad[j1] += (1.0 - h[j1 - m_])
        j2 = np.arange(0, n - m_)
        grad[j2] -= h[j2 + m_]
    return grad * dx


def sum_con_fun(h, n):
    return h.sum() - n / 2.0


def sum_con_jac(h, n):
    return np.ones(n)


def coordinate_polish(h, n, max_rounds=200, t0=None, budget=None):
    """Coordinate-wise adjustment to reduce C5 while maintaining feasibility."""
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h, n)
    for r in range(max_rounds):
        if t0 is not None and budget is not None:
            if time.time() - t0 > budget - 15:
                break
        improved = False
        for i in range(n):
            # Try moving h[i] by delta, compensate with h[j] for some j
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_trial = best.copy()
                h_trial[i] += delta
                if h_trial[i] < 0 or h_trial[i] > 1:
                    continue
                # Compensate: find a j to adjust
                # Simple: adjust the element with most "room"
                if delta > 0:
                    comp_mask = h_trial < 1.0 - 1e-10
                else:
                    comp_mask = h_trial > 1e-10
                    comp_mask[i] = False
                if not comp_mask.any():
                    continue
                comp_mask[i] = False
                if not comp_mask.any():
                    continue
                j = np.argmax(h_trial[comp_mask]) if delta > 0 else np.argmin(h_trial[comp_mask])
                h_trial[j] -= delta
                if h_trial[j] < 0 or h_trial[j] > 1:
                    continue
                c5_trial = compute_c5(h_trial, n)
                if c5_trial < best_c5 - 1e-12:
                    best = h_trial.copy()
                    best_c5 = c5_trial
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5


def interpolate_to(n_target, h_src, n_src):
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    x_tgt = np.linspace(0, 2, n_target, endpoint=False) + 1.0 / n_target
    return np.interp(x_tgt, x_src, h_src)


def make_binary_start(n, rng):
    """Create a binary-like starting vector with exactly n/2 units of mass."""
    h = np.zeros(n)
    ones = np.arange(0, n, 2)  # alternating
    idx = rng.permutation(n)
    # Use a structured pattern: groups of consecutive 1s
    n_ones = n // 2
    # Random binary with correct count
    h[idx[:n_ones]] = 1.0
    # Add small noise to make it non-boundary
    h += rng.randn(n) * 0.05
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    return h


def make_smooth_start(n, freq, rng):
    """Create a smooth sinusoidal starting vector."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    h += rng.randn(n) * 0.03
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    return h
