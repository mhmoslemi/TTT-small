def smooth_max_and_grad(h, n, temp):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax
    max_c = conv.max()
    exp_vals = np.exp((conv - max_c) / temp)
    p = exp_vals / exp_vals.sum()
    obj = temp * (np.log(exp_vals.sum()) + max_c / temp)
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        m = k - (n - 1)
        # Part 1: (1-h[j-m]) for valid j
        # Need: 0 <= j-m <= n-1, i.e., m <= j <= m+n-1
        # Also: 0 <= j <= n-1
        j_lo = max(0, m)
        j_hi = min(n - 1, m + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] += p[k] * dx * (1.0 - h[idx - m])
        # Part 2: -h[j+m] for valid j
        # Need: 0 <= j+m <= n-1, i.e., -m <= j <= n-1-m
        # Also: 0 <= j <= n-1
        j_lo2 = max(0, -m)
        j_hi2 = min(n - 1, n - 1 - m)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad[j_lo2:j_hi2+1] -= p[k] * dx * h[idx2 + m]
    return obj, grad, p

    # When lambda is very positive, clip(h-lambda, 0, 1) = 0 for all i, sum = 0

    j_hi = min(n - 1, m + n - 1)
    if j_lo <= j_hi:
        idx = np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi+1] += dx * (1.0 - h[idx - m])
    j_lo2 = max(0, -m)
    j_hi2 = min(n - 1, n - 1 - m)
    if j_lo2 <= j_hi2:
        idx2 = np.arange(j_lo2, j_hi2 + 1)
        grad[j_lo2:j_hi2+1] -= dx * h[idx2 + m]
    return grad

def adam_run(h0, n, time_limit_s, temp_start=0.08, temp_end=0.005, lr=0.02):
    h = h0.copy()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_t = np.zeros(n)
    v_t = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_val = compute_c5(h, n)
    t0 = time.time()
    
    while time.time() - t0 < time_limit_s:
        t += 1
        # Temperature schedule: exponential decay over the run
        frac = min(1.0, (time.time() - t0) / time_limit_s)
        if frac < 0.8:
            temp = temp_start * (temp_end / temp_start) ** (frac / 0.8)
        else:
            temp = temp_end
        
        obj, grad = smooth_max_grad(h, n, temp)
        
        # Adam update
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * (grad * grad)
        m_hat = m_t / (1.0 - beta1 ** t)
        v_hat = v_t / (1.0 - beta2 ** t)
        
        # Learning rate decay
        lr_eff = lr / (1.0 + 0.0005 * t)
        
        h_new = h - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_val

def exact_gradient_descent(h0, n, n_iter=500, lr=0.01):
    """Final polish with exact max gradient."""
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)

def random_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n)
    indices = rng.choice(n, size=n//2, replace=False)

def golden_ratio_binary(n, rng):
    """Generate a binary vector based on golden ratio fraction."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    offset = rng.uniform(0, 1)
    h = np.zeros(n)
    for i in range(n):
        if ((i * phi + offset) % 1.0) < 0.5:
            h[i] = 1.0
    # Adjust to have exactly n/2 ones
    n_ones = int(h.sum())
    if n_ones > n // 2:
        one_idx = np.where(h == 1.0)[0]
        flip = rng.choice(one_idx, size=n_ones - n//2, replace=False)
        h[flip] = 0.0
    elif n_ones < n // 2:
        zero_idx = np.where(h == 0.0)[0]
        flip = rng.choice(zero_idx, size=n//2 - n_ones, replace=False)
        h[flip] = 1.0
    return h

def periodic_binary(n, period, ones_per_period, rng):
    """Generate a periodic binary pattern."""
    h = np.zeros(n)
    offset = rng.randint(0, period)
    for i in range(n):
        pos_in_period = (i + offset) % period
        if pos_in_period < ones_per_period:
            h[i] = 1.0
    # Adjust

        one_idx = np.where(h == 1.0)[0]

        zero_idx = np.where(h == 0.0)[0]
        flip = rng.choice(zero_idx, size=n//2 - n_ones, replace=False)
        h[flip] = 1.0

def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()

        y = np.clip(h - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        nf = np.count_nonzero(free)

            y = np.clip(y, 0.0, 1.0)

def compute_c5(h, n):

    conv = np.correlate(h, 1.0 - h, mode='full')

    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = conv.max()

    p = exp_vals / exp_vals.sum()
    obj = temp * (np.log(exp_vals.sum()) + max_c / temp)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue

        j_hi = min(n - 1, m + n - 1)

    lr = lr_init
    for i in range(n_iter):
        grad = exact_grad(h, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        if i % 100 == 99:
            lr *= 0.7
    return best_h, best_val

def random_binary(n, rng):
    h = np.zeros(n)
    indices = rng.choice(n, size=n // 2, replace=False)
    h[indices] = 1.0
    return h

def golden_ratio_binary(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    offset = rng.uniform(0, 1)
    h = np.zeros(n)
    fracs = (np.arange(n) * phi + offset) % 1.0
    h = (fracs < 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > n // 2:
        one_idx = np.where(h == 1.0)[0]
        flip = rng.choice(one_idx, size=n_ones - n // 2, replace=False)
        h[flip] = 0.0
    elif n_ones < n // 2:
        zero_idx = np.where(h == 0.0)[0]
        flip = rng.choice(zero_idx, size=n // 2 - n_ones, replace=False)
        h[flip] = 1.0
    return h

def periodic_binary(n, period, ones_per_period, rng):
    h = np.zeros(n)
    offset = rng.randint(0, period)
    positions = (np.arange(n) + offset) % period
    h[positions < ones_per_period] = 1.0
    n_ones = int(h.sum())
    if n_ones > n // 2:
        one_idx = np.where(h == 1.0)[0]
        flip = rng.choice(one_idx, size=n_ones - n // 2, replace=False)
        h[flip] = 0.0
    elif n_ones < n // 2:
        zero_idx = np.where(h == 0.0)[0]
        flip = rng.choice(zero_idx, size=n // 2 - n_ones, replace=False)
        h[flip] = 1.0
    return h

def local_search_bit_flip(h, n, rng, n_iter=200):
    h = h.copy()
    best_val = compute_c5(h, n)
    one_idx = np.where(h > 0.5)[0]
    zero_idx = np.where(h < 0.5)[0]
    for _ in range(n_iter):
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        a = one_idx[rng.randint(len(one_idx))]
        b = zero_idx[rng.randint(len(zero_idx))]
        h[a], h[b] = 1.0, 0.0
        val = compute_c5(h, n)
        if val < best_val - 1e-12:
            best_val = val
            one_idx = np.where(h > 0.5)[0]
            zero_idx = np.where(h < 0.5)[0]
        else:
            h[a], h[b] = 0.0, 1.0
    return h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    best_h = None
    best_val = float('inf')
    best_n = 0
    main_n = 140

    # Phase 1: Optimize from parent (90s)
    if h_parent is not None:
        n_p = len(h_parent)
        x_p = np.linspace(0, 2, n_p, endpoint=False)
        x_m = np.linspace(0, 2, main_n, endpoint=False)
        h_interp = np.interp(x_m, x_p, h_parent)
        h_interp = project_feasible(h_interp, main_n)
        if time_left() > 80:
            h_opt, val = adam_run(h_interp, main_n, time_limit_s=70, temp_start=0.05, temp_end=0.004, lr=0.03)
            h_opt, val = exact_gradient_descent(h_opt, main_n, n_iter=300, lr_init=0.005)
            if val < best_val:
                best_val = val
                best_h = h_opt.copy()
                best_n = main_n

    # Phase 2: Diverse multi-start (500s)
    initializations = []
    # 1: Perturbed best
    if best_h is not None:
        initializations.append(best_h + rng.normal(0, 0.12, main_n))
    # 2,3: Random binary
    initializations.append(random_binary(main_n, rng))
    initializations.append(random_binary(main_n, rng))
    # 4,5: Golden ratio
    initializations.append(golden_ratio_binary(main_n, rng))
    initializations.append(golden_ratio_binary(main_n, rng))
    # 6: Periodic 2/4
    initializations.append(periodic_binary(main_n, 4, 2, rng))
    # 7: Periodic 2/5
    initializations.append(periodic_binary(main_n, 5, 2, rng))
    # 8: Periodic 3/7
    initializations.append(periodic_binary(main_n, 7, 3, rng))
    # 9: Random continuous
    initializations.append(rng.uniform(0, 1, main_n))
    # 10: Sine pattern
    f1 = rng.uniform(2, 4)
    initializations.append((np.sin(2 * np.pi * f1 * np.arange(main_n) / main_n + rng.uniform(0, 6.28)) > 0).astype(float))
    # 11: Random walk threshold
    walk = np.cumsum(rng.normal(0, 1, main_n))
    initializations.append((walk > np.median(walk)).astype(float))
    # 12: Another periodic
    initializations.append(periodic_binary(main_n, 6, 3, rng))
    # 13: Bimodal continuous
    initializations.append(rng.choice([0.1, 0.9], size=main_n))
    # 14: Perturbed best larger
    if best_h is not None:
        initializations.append(best_h + rng.normal(0, 0.25, main_n))
    # 15: Low freq
    f2 = rng.uniform(1, 2)
    initializations.append((np.cos(2 * np.pi * f2 * np.arange(main_n) / main_n + rng.uniform(0, 6.28)) > 0).astype(float))

    n_starts = len(initializations)
    time_per_start = 22
    for i in range(n_starts):
        if time_left() < 180:
            break
        h0 = project_feasible(initializations[i], main_n)
        tl = min(time_per_start, max(10, time_left() - 180))
        # Vary hyperparameters slightly per start
        ts = 0.04 + 0.03 * (i % 3) / 2.0
        lr_v = 0.02 + 0.01 * (i % 4) / 3.0
        try:
            h_opt, val = adam_run(h0, main_n, time_limit_s=tl, temp_start=ts, temp_end=0.003, lr=lr_v)
            h_opt2, val2 = exact_gradient_descent(h_opt, main_n, n_iter=400, lr_init=0.006)
            if val2 < best_val:
                best_val = val2
                best_h = h_opt2.copy()
                best_n = main_n
        except Exception:
            pass

    # Phase 3: Intensive refinement (300s)
    if time_left() > 200 and best_h is not None:
        for attempt in range(6):
            if time_left() < 100:
                break
            scale = [0.03, 0.05, 0.08, 0.12, 0.06, 0.1][attempt % 6]
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = project_feasible(h_pert, best_n)
            ts = 0.03 + 0.02 * (attempt % 3)
            try:
                h_opt, val = adam_run(h_pert, best_n, time_limit_s=40, temp_start=ts, temp_end=0.002, lr=0.015)
                h_opt, val = exact_gradient_descent(h_opt, best_n, n_iter=300, lr_init=0.004)
                if val < best_val:
                    best_val = val
                    best_h = h_opt.copy()
            except Exception:
                pass

        # Local search on best
        if time_left() > 60:
            h_ls, val_ls = local_search_bit_flip(best_h, best_n, rng, n_iter=500)
            if val_ls < best_val:
                best_val = val_ls
                best_h = h_ls.copy()
            # Re-optimize after local search
            try:
                h_opt, val = adam_run(best_h, best_n, time_limit_s=20, temp_start=0.03, temp_end=0.002, lr=0.01)
                h_opt, val = exact_gradient_descent(h_opt, best_n, n_iter=200, lr_init=0.003)
                if val < best_val:
                    best_val = val
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 4: Higher resolution (200s)
    if time_left() > 150 and best_h is not None:
        for n_high in [200, 280]:
            if time_left() < 80:
                break
            x_low = np.linspace(0, 2, best_n, endpoint=False)
            x_high = np.linspace(0, 2, n_high, endpoint=False)
            h_high = np.interp(x_high, x_low, best_h)
            h_high = project_feasible(h_high, n_high)
            try:
                h_opt, val = adam_run(h_high, n_high, time_limit_s=50, temp_start=0.03, temp_end=0.003, lr=0.012)
                h_opt, val = exact_gradient_descent(h_opt, n_high, n_iter=300, lr_init=0.003)
                if val < best_val:
                    best_val = val
                    best_h = h_opt.copy()
                    best_n = n_high
            except Exception:
                pass
            # Also try fresh at higher res
            if time_left() > 60:
                try:
                    h_rand_high = project_feasible(random_binary(n_high, rng), n_high)
                    h_opt, val = adam_run(h_rand_high, n_high, time_limit_s=30, temp_start=0.04, temp_end=0.003, lr=0.015)
                    if val < best_val:
                        best_val = val
                        best_h = h_opt.copy()
                        best_n = n_high
                except Exception:
                    pass

    # Phase 5: Final polish
    if time_left() > 30 and best_h is not None:
        for _ in range(3):
            if time_left() < 15:
                break
            try:
                h_opt, val = exact_gradient_descent(best_h, best_n, n_iter=500, lr_init=0.003)
                if val < best_val:
                    best_val = val
                    best_h = h_opt.copy()
                else:
                    break
            except Exception:
                break

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    if best_h is None:
        best_n = 100
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
