        inits.append(np.tile([1.0, 0.0, 1.0, 0.0], n // 4).copy())
    if n % 2 == 0:
        h3 = np.zeros(n)
        h3[:n // 4] = 1.0
        h3[n // 2:n // 2 + n // 4] = 1.0
        inits.append(h3)
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm, n))
    inits.append(rng.rand(n) * 1.4)
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    def remaining():
        return budget_s - (time.time() - start_time)
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
    best_c5 = compute_c5(best_h)
    resolutions = sorted(set([best_n, 160, 200, 240, 320, 400]))
    resolutions = [r for r in resolutions if r % 2 == 0]
    for n in resolutions:
        if remaining() < 30:
            break
        inits = [make_feasible(interpolate_to_n(best_h, n), n)]
        if h_parent is not None and len(h_parent) != n:
            inits.append(make_feasible(interpolate_to_n(h_parent, n), n))
        for hs in structured_init(n, seed=n + seed):
            inits.append(make_feasible(hs, n))
        inits = inits[:5]
        for h0 in inits:
            if remaining() < 15:
                break
            best_n_h = h0.copy()
            best_n_c5 = compute_c5(h0)
            t_rem = remaining()
            t_grad = min(0.35 * t_rem, 200.0)
            t_slsqp = min(0.25 * t_rem, 120.0)
            t_local = min(0.2 * t_rem, 80.0)
            if t_grad > 5:
                h_g, c5_g = nesterov_smooth_descent(h0, n, max_time=t_grad,
                    beta_start=5.0, beta_end=300.0, n_betas=5, lr_init=0.02, momentum=0.85)
                if c5_g < best_n_c5:
                    best_n_c5 = c5_g
                    best_n_h = h_g
            if remaining() > 10 and t_slsqp > 10:
                h_s, c5_s = slsqp_polish(best_n_h, n, max_iter=2500)
                if h_s is not None and c5_s < best_n_c5:
                    best_n_c5 = c5_s
                    best_n_h = h_s
                h_g2, c5_g2 = nesterov_smooth_descent(best_n_h, n, max_time=min(30, remaining()*0.15),
                    beta_start=30.0, beta_end=500.0, n_betas=3, lr_init=0.005, momentum=0.8)
                if c5_g2 < best_n_c5:
                    best_n_c5 = c5_g2
                    best_n_h = h_g2
            if remaining() > 10 and t_local > 5:
                h_l, c5_l = local_search(best_n_h, n, n_trials=8000,
                    step_size=0.02, budget_time=min(40, remaining()*0.3), seed=seed+n)
                if c5_l < best_n_c5:
                    best_n_c5 = c5_l
                    best_n_h = h_l
            if best_n_c5 < best_c5 - 1e-13:
                best_c5 = best_n_c5
                best_h = best_n_h.copy()
                best_n = n
    if remaining() > 20:
        h_f, c5_f = nesterov_smooth_descent(best_h, best_n, max_time=min(80, remaining()*0.4),
            beta_start=20.0, beta_end=600.0, n_betas=4, lr_init=0.008, momentum=0.82)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if remaining() > 20:
        h_f2, c5_f2 = local_search(best_h, best_n, n_trials=20000,
            step_size=0.008, budget_time=min(50, remaining()*0.5), seed=seed+999)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2
    if remaining() > 5:
        h_f3, c5_f3 = slsqp_polish(best_h, best_n, max_iter=5000)
        if h_f3 is not None and c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_f3
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
