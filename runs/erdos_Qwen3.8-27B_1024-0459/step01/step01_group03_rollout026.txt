import time
import numpy as np
from scipy.optimize import minimize

_GN = 90
_GALPHA = 10.0
_GTARGET = 45.0
_GPEN = 1e6


def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) <= 1e-14 * max(1, n):
        return x.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if free.size > 0:
            per = res / float(free.size)
            h[free] = np.clip(h[free] + per, 0.0, 1.0)
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-12:
                if res2 > 0:
                    idx = free[np.argmax(1.0 - h[free])]
                else:
                    idx = free[np.argmax(h[free])]
                h[idx] = np.clip(h[idx] + res2, 0.0, 1.0)
    return h


def is_feasible(h, target, tol=1e-6):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - float(target)) > max(1e-8, tol * max(1, h.size)):
        return False
    return True


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def make_candidate(n, rng, mode):
    n = int(n)
    if n <= 1:
        return project_density(np.array([0.5]), 0.5)
    if mode == 0:
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
        h = np.full(n, 0.5, dtype=float)
        for m in range(1, 7):
            amp = float(rng.uniform(0.05, 0.3)) * (0.65 ** (m - 1))
            freq = int(max(1, np.rint(m * float(rng.uniform(0.5, 1.8)))))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    elif mode == 1:
        u = rng.random(n)
        thresh = float(np.median(u))
        h = (u > thresh).astype(float)
        for _ in range(3):
            h = project_density(h + 0.05 * rng.standard_normal(n), 0.5 * n)
    elif mode == 2:
        h = np.empty(n, dtype=float)
        levels = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]
        pos = 0
        while pos < n:
            rem = n - pos
            length = int(rng.integers(1, max(2, rem // 4 + 1)))
            val = float(rng.choice(levels))
            h[pos:pos + length] = val
            pos += length
    else:
        h = np.full(n, 0.5, dtype=float)
        h += 0.15 * rng.standard_normal(n)
    return project_density(h, 0.5 * n)


def set_ctx(n, alpha, target, pen):
    global _GN, _GALPHA, _GTARGET, _GPEN
    _GN = int(n)
    _GALPHA = float(alpha)
    _GTARGET = float(target)
    _GPEN = float(pen)


def smoothed_obj(x):
    n = _GN
    alpha = _GALPHA
    target = _GTARGET
    pen = _GPEN
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(corr)):
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    base = float((2.0 / n) * (m + np.log(z) / alpha))
    s = float(np.sum(h)) - target
    return base + pen * s * s


def smoothed_grad(x):
    n = _GN
    alpha = _GALPHA
    target = _GTARGET
    pen = _GPEN
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(corr)):
        return np.zeros(n, dtype=float)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n, dtype=float)
    w = e / z
    idx = np.where(w > 1e-10)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))], dtype=int)
    g = np.zeros(n, dtype=float)
    center = n - 1
    for t in idx:
        t = int(t)
        p = float(w[t])
        if p <= 0.0:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
            else:
                g[:n - lag] += p
            g[:n - lag] -= p * h[lag:]
            g[lag:] -= p * h[:n - lag]
    g *= (2.0 / n)
    s = float(np.sum(h)) - target
    g += 2.0 * pen * s * np.ones(n, dtype=float)
    return g


def run_lbfgsb(h0, n, deadline):
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    bounds = [(0.0, 1.0)] * n
    alphas = [8.0, 15.0, 30.0, 60.0]
    if n <= 120:
        maxit = 300
    elif n <= 300:
        maxit = 200
    else:
        maxit = 120
    for alpha in alphas:
        if time.perf_counter() >= deadline:
            break
        set_ctx(n, alpha, target, 1e5)
        try:
            res = minimize(
                smoothed_obj, h,
                method='L-BFGS-B',
                jac=smoothed_grad,
                bounds=bounds,
                options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxfun': maxit * 3, 'disp': False}
            )
            if res.x is not None:
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                if c <= best_c + 0.01:
                    h = cand
                else:
                    h = best_h
        except Exception:
            h = best_h
    set_ctx(n, 15.0, target, 1e7)
    if time.perf_counter() < deadline - 2.0:
        try:
            res = minimize(
                smoothed_obj, h,
                method='L-BFGS-B',
                jac=smoothed_grad,
                bounds=bounds,
                options={'maxiter': maxit, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxfun': maxit * 2, 'disp': False}
            )
            if res.x is not None:
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
        except Exception:
            pass
    return best_h, best_c


def corr_grad_active(h, indices):
    n = h.size
    g = np.zeros(n, dtype=float)
    center = n - 1
    for t in indices:
        t = int(t)
        k = t - center
        if k == 0:
            g += 1.0 - 2.0 * h
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += 1.0
            else:
                g[:n - lag] += 1.0
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
    return g


def local_search(h, deadline, max_iter=2000, active_k=5):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    k = min(int(active_k), 2 * n - 1)
    stall = 0
    for _ in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        if not np.all(np.isfinite(corr)):
            break
        idxs = np.argsort(corr)[-k:]
        g = corr_grad_active(best, idxs) / float(k)
        if g.size == 0 or (float(np.max(g)) - float(np.min(g))) < 1e-12:
            break
        found = False
        neg = np.argsort(g)[:5]
        pos = np.argsort(-g)[:5]
        for ii in neg:
            ii = int(ii)
            for jj in pos:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap <= 1e-13:
                    continue
                deltas = [0.25 * cap, 0.5 * cap, 0.75 * cap, cap - 1e-14]
                for d in deltas:
                    if d <= 1e-14 or d >= cap:
                        continue
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-13:
                d /= norm
                for step in (0.02, 0.05, 0.1, 0.2):
                    cand = project_density(best + step * d, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
        if found:
            stall = 0
        else:
            stall += 1
            if stall >= 20:
                break
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def block_search(h, deadline, max_iter=500):

        block_sizes = [2, 3, max(2, n // 15)]

        nbest = int(best_h.size)
        # First pass: local pair search
        if deadline - time.perf_counter() > 15.0:
            h2, c2 = local_search(best_h, deadline - 10.0, max_iter=3000, active_k=5)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = nbest
        # Block search
        if deadline - time.perf_counter() > 10.0:
            h3, c3 = block_search(best_h, deadline - 8.0, max_iter=500)
            if c3 < best_c:
                best_h = h3
                best_c = c3
                best_n = int(best_h.size)
        # Second local pass
        if deadline - time.perf_counter() > 10.0:
            h4, c4 = local_search(best_h, deadline - 5.0, max_iter=2000, active_k=6)
            if c4 < best_c:
                best_h = h4
                best_c = c4
                best_n = int(best_h.size)

    # Phase 4: Final gradient polish
    if best_h is not None and deadline - time.perf_counter() > 5.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        best = project_density(best_h, target)
        best_cg = evaluate_erdos(best)
        stall = 0
        for _ in range(500):
            if time.perf_counter() >= deadline:
                break
            corr = np.correlate(best, 1.0 - best, mode="full")
            idxs = np.argsort(corr)[-5:]
            g = corr_grad_active(best, idxs) / 5.0
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm < 1e-14:
                break
            d /= norm
            improved = False
            for step in (0.01, 0.03, 0.05, 0.1, 0.2, 0.5):
                cand = project_density(best + step * d, target)
                c = evaluate_erdos(cand)
                if c < best_cg - 1e-12:
                    best = cand
                    best_cg = c
                    improved = True
                    break
            if improved:
                stall = 0
            else:
                stall += 1
                if stall > 20:
                    break
        if best_cg < best_c:
            best_h = best
            best_c = best_cg
            best_n = nbest

    if best_h is None:
        fallback = project_density(np.full(90, 0.5, dtype=float), 45.0)
        return fallback, evaluate_erdos(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
