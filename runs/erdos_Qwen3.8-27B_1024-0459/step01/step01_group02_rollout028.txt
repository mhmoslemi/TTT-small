from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    # Project onto {h: 0<=h<=1, sum(h)=n/2}
    h = h.copy()
    for it in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = n/2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute uniformly among free variables
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(free)
        if cnt == 0:
            free = np.ones(n, dtype=bool)
            cnt = n
        h[free] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Tiny residual
    diff = n/2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def smooth_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(corr - m, -5.0, 0.0)
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def bmd_smooth(h, n, beta):
    """Bound minimization deviation - smooth objective with constraints handled by SLSQP"""
    return smooth_obj(h, n, beta)

    """Coordinate descent for minimizing max correlation"""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old_val = h[j]
            # Compute b[m] = change in corr[m] per unit change in h[j]
            # corr[m] = sum_i h[i]*(1-h[i+m-(n-1)])
            # d(corr[m])/d(h[j]) for i=j: (1-h[j+m-(n-1)]) if valid
            # d(corr[m])/d(h[j]) for i+m-(n-1)=j: -h[i] = -h[j-m+n-1] if valid
            b = np.zeros(2*n-1)
            
            # i=j term: m+j-(n-1) = idx in [0,n)
            m_lo1 = max(0, n-1-j)
            m_hi1 = min(2*n-2, 2*(n-1)-j)
            for m in range(m_lo1, m_hi1+1):
                idx = j + m - (n-1)
                b[m] += (1.0 - h[idx])
            
            # i+m-(n-1)=j term: i = j-m+n-1 in [0,n)
            m_lo2 = max(0, j)
            m_hi2 = min(2*n-2, j+n-1)
            for m in range(m_lo2, m_hi2+1):
                idx = j - m + (n-1)
                b[m] += (-h[idx])
            
            # corr[m] + delta*b[m] where delta = new_h[j] - old_h[j]
            # = (h[j] + delta) related
            # Actually corr[m] already has h[j] in it
            # New corr[m] = corr[m] + (new_val - old_val) * b[m]
            # We want to find new_val in [0,1] minimizing max_m (corr[m] + (new_val-old_val)*b[m])
            # Let x = new_val - old_val, x in [-old_val, 1-old_val]

            a = corr.copy()  # base

            x_lo = -old_val

            lo, hi = x_lo, x_hi

                else:
                    lo = mid

                h[j] = new_val

                    corr[m] += (new_val - old_val) * b[m]
        
        if not improved:
            break

        h = project_feasible(h, n)
