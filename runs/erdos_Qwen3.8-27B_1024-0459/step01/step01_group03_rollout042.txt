                h = best.copy()
                stall = 0
            elif stall % 10 == 0:
                step_size *= decay
        
        # Update step
        step_size = max(0.01, min(1.0, step_size * decay))
    
    return best, best_c

def pair_polish(h0, deadline, max_iter=500, topk=4):
    """Local pair search: find best pair move to reduce top-k correlations."""
    h = project_density(h0, 0.5 * h0.size)
    n = h.size
    best = h.copy()
    best_c = evaluate_erdos(best)
    target = 0.5 * n
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr, g, idxs = corr_and_grad_topk(best, topk)
        
        # Find best pair: decrease g at one position, increase at another
        # (reducing the top-k correlation)
        # Sort by gradient: want to decrease h where g is large, increase where g is small
        sorted_idx = np.argsort(g)
        neg_candidates = sorted_idx[:8]
        pos_candidates = sorted_idx[-8:][::-1]
        
        found = False
        for ii in neg_candidates:
            ii = int(ii)
            for jj in pos_candidates:
                jj = int(jj)
                if ii == jj:
                    continue
                # Move: h[ii] += d, h[jj] -= d
                # h[ii] currently has low g, so increasing it helps
                # h[jj] currently has high g, so decreasing it helps
                cap_inc = 1.0 - best[ii]
                cap_dec = best[jj]
                cap = min(cap_inc, cap_dec)
                if cap < 1e-10:
                    continue
                
                a = g[ii] - g[jj]  # should be negative for improvement
                if a >= -1e-10:
                    continue
                
                # Try full cap and fractions
                for d in (cap, 0.7*cap, 0.4*cap, 0.2*cap):
                    d = min(d, cap - 1e-12)
                    if d < 1e-10:
                        continue
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    cand = np.clip(cand, 0.0, 1.0)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        
        # Also try gradient direction
        if not found:
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-12:
                d /= norm
                for step in (0.02, 0.05, 0.1, 0.2, 0.4):
                    cand = project_density(best + step * d, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    
    return best, best_c

def stochastic_polish(h0, deadline, max_iter=200, topk=4, rng=None):
    """Stochastic perturbation to escape local minima."""
    if rng is None:
        rng = np.random.default_rng(42)
    h = project_density(h0, 0.5 * h0.size)
    n = h.size
    best = h.copy()
    best_c = evaluate_erdos(best)
    target = 0.5 * n
    
    temp = 0.1
    decay = 0.99
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        # Random perturbation
        n_pert = max(2, n // 10)
        perturb_idx = rng.choice(n, size=n_pert, replace=False)
        noise = rng.uniform(-temp, temp, size=n_pert)
        cand = best.copy()
        cand[perturb_idx] += noise
        cand = project_density(cand, target)
        c = evaluate_erdos(cand)
        
        if c < best_c:
            best = cand
            best_c = c
            temp = max(0.01, temp * 0.9)
        else:
            temp = max(0.01, temp * decay)
            # Also try small gradient step
            corr, g, idxs = corr_and_grad_topk(best, topk)
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-12:
                d /= norm
                for step in (0.05, 0.1, 0.2):
                    cand2 = project_density(best + step * d, target)
                    c2 = evaluate_erdos(cand2)
                    if c2 < best_c:
                        best = cand2
                        best_c = c2
                        break
    
    return best, best_c

def make_initial(n, rng, mode):
    """Create diverse initial candidates."""
    if mode == 0:
        # Random smooth
        x = (np.arange(n) + 0.5) * (2.0/n)
        h = np.full(n, 0.5)
        for m in range(1, 8):
            amp = rng.uniform(0.05, 0.3) * (0.7 ** m)
            freq = int(max(1, np.rint(m * rng.uniform(0.5, 2.0))))
            phase = rng.uniform(0, 2*np.pi)
            h += amp * np.sin(2*np.pi*freq*x/2.0 + phase)
    elif mode == 1:
        # Step function
        h = np.empty(n)
        nblocks = rng.integers(n//6, n//2)
        boundaries = np.sort(rng.choice(np.arange(1, n), size=nblocks-1, replace=False))
        boundaries = np.concatenate([[0], boundaries, [n]])
        for i in range(nblocks):
            h[boundaries[i]:boundaries[i+1]] = rng.choice([0, 0.25, 0.5, 0.75, 1.0])
    elif mode == 2:
        # Quarter pattern
        nq = n // 4
        h = np.concatenate([np.full(nq, 0.0), np.full(nq, 1.0), np.full(nq, 0.5), np.full(min(nq, n-3*nq), np.random.uniform(0,1))])
        if h.size < n:
            h = np.concatenate([h, np.full(n - h.size, 0.5)])
        h = h[:n]
    else:
        # Binary with noise
        h = (rng.random(n) > 0.5).astype(float)
        h += rng.uniform(-0.1, 0.1, size=n)
    
    return project_density(h, 0.5 * n)

def optimize_at_resolution(h0, n, deadline):
    """Optimize at a specific resolution."""
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    
    now = time.perf_counter()
    remaining = deadline - now
    if remaining < 10:
        return best_h, best_c
    
    # Phase 1: SLSQP at increasing alpha
    alphas_and_iters = []
    if n <= 100:
        alphas_and_iters = [(5.0, 200), (15.0, 150), (40.0, 100), (80.0, 80), (150.0, 60)]
    elif n <= 200:
        alphas_and_iters = [(5.0, 150), (15.0, 120), (40.0, 80), (80.0, 60)]
    elif n <= 360:
        alphas_and_iters = [(5.0, 100), (15.0, 80), (40.0, 50)]
    else:
        alphas_and_iters = [(5.0, 80), (15.0, 60)]
    
    time_per_alpha = remaining * 0.4 / len(alphas_and_iters)
    
    for alpha, maxit in alphas_and_iters:
        if time.perf_counter() >= deadline - remaining * 0.4:
            break
        sub_dl = time.perf_counter() + time_per_alpha
        h = slsqp_optimize(best_h, n, alpha, maxit, sub_dl)
        c = evaluate_erdos(h)
        if c < best_c:
            best_h = h.copy()
            best_c = c
    
    # Phase 2: Subgradient polish
    if time.perf_counter() < deadline - remaining * 0.15:
        sub_dl = time.perf_counter() + remaining * 0.15
        h2, c2 = subgrad_polish(best_h, sub_dl, max_iter=1000, topk=4)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    # Phase 3: Pair polish
    if time.perf_counter() < deadline - remaining * 0.1:
        sub_dl = time.perf_counter() + remaining * 0.1
        h2, c2 = pair_polish(best_h, sub_dl, max_iter=1000, topk=4)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    # Phase 4: Stochastic perturbation
    if time.perf_counter() < deadline - remaining * 0.05:
        sub_dl = time.perf_counter() + remaining * 0.05
        rng = np.random.default_rng(int(hash(tuple(np.round(best_h, 6))) % 2**31))
        h2, c2 = stochastic_polish(best_h, sub_dl, max_iter=100, topk=4, rng=rng)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    # Phase 5: Final fine polish
    if time.perf_counter() < deadline - 2:
        h2, c2 = pair_polish(best_h, deadline - 2, max_iter=500, topk=6)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 10.0)
    
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float("inf")
    best_n = 0
    
    # Get parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None
    
    # Evaluate parent
    if parent is not None:
        nparent = parent.size
        pproj = project_density(parent, 0.5 * nparent)
        cp = evaluate_erdos(pproj)
        best_h = pproj.copy()
        best_c = cp
        best_n = nparent
    
    if best_h is None:
        best_h = make_initial(90, rng, 0)
        best_c = evaluate_erdos(best_h)
        best_n = 90
    
    # Strategy: optimize at multiple resolutions, keeping the best
    # Focus on n=90 (fast), n=180 (medium), n=180 or 270 (higher)
    
    resolutions = [90]
    if best_n != 90:
        resolutions.append(best_n)
    resolutions.extend([180, 270])
    
    # Deduplicate and sort
    seen = set()
    res_list = []
    for r in resolutions:
        r = int(r)
        if 10 <= r <= 500 and r not in seen:
            seen.add(r)
            res_list.append(r)
    res_list.sort()
    
    # Allocate time: more time to promising resolutions
    # SLSQP at n=90 is fast (~5s), n=180 is ~20s, n=270 is ~40s
    time_allocs = {}
    for r in res_list:
        if r <= 100:
            time_allocs[r] = 80.0
        elif r <= 200:
            time_allocs[r] = 120.0
        else:
            time_allocs[r] = 100.0
    
    # Do first pass: quick optimization at all resolutions
    for r in res_list:
        if time.perf_counter() >= deadline - 30:
            break
        alloc = min(time_allocs.get(r, 60.0), deadline - time.perf_counter() - 20.0)
        if alloc < 15:
            break
        sub_dl = time.perf_counter() + alloc
        
        # Start from upsampled best
        h_start = upsmap_h(best_h, r)
        h_out, c_out = optimize_at_resolution(h_start, r, sub_dl)
        
        if c_out < best_c:
            best_h = h_out
            best_c = c_out
            best_n = r
    
    # Second pass: more time on the best resolution
    if time.perf_counter() < deadline - 60:
        alloc = min(200.0, deadline - time.perf_counter() - 30.0)
        sub_dl = time.perf_counter() + alloc
        h_out, c_out = optimize_at_resolution(best_h, best_n, sub_dl)
        if c_out < best_c:
            best_h = h_out
            best_c = c_out
            best_n = best_h.size
    
    # Try a different starting point at the best resolution
    if time.perf_counter() < deadline - 40:
        h_alt = make_initial(best_n, rng, 1)
        alloc = min(80.0, deadline - time.perf_counter() - 20.0)
        sub_dl = time.perf_counter() + alloc
        h_out, c_out = optimize_at_resolution(h_alt, best_n, sub_dl)
        if c_out < best_c:
            best_h = h_out
            best_c = c_out
            best_n = best_h.size
    
    # Final intensive polish on the best solution
    if time.perf_counter() < deadline - 10:
        # Multiple rounds of pair polish with different topk
        for topk in (3, 4, 5, 6, 8):
            if time.perf_counter() >= deadline - 5:
                break
            h2, c2 = pair_polish(best_h, deadline - 5, max_iter=2000, topk=topk)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        
        # Subgradient with different topk
        for topk in (3, 5):
            if time.perf_counter() >= deadline - 5:
                break
            h2, c2 = subgrad_polish(best_h, deadline - 5, max_iter=1000, topk=topk)
            if c2 < best_c:
                best_h = h2
                best_c = c2
    
    # Try upsampled higher resolution if time permits
    if time.perf_counter() < deadline - 50:
        for n_try in [best_n * 2]:
            if n_try > 600 or n_try <= best_n:
                continue
            alloc = min(80.0, deadline - time.perf_counter() - 15.0)
            if alloc < 20:
                break
            sub_dl = time.perf_counter() + alloc
            h_start = upsmap_h(best_h, n_try)
            h_out, c_out = optimize_at_resolution(h_start, n_try, sub_dl)
            if c_out < best_c:
                best_h = h_out
                best_c = c_out
                best_n = n_try
    
    # Final cleanup
    final_h = project_density(best_h, 0.5 * best_h.size)
    final_c = evaluate_erdos(final_h)
    
    if final_h is None or final_c >= float("inf"):
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90
    
    return final_h, float(final_c), int(final_h.size)
