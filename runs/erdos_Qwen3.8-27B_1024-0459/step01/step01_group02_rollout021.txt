Hmm, this gets complicated with the indexing. Let me just use the direct formula.

For the "natural" correlation at lag d (d can be negative):
corr[d] = ∑_i h[i] * (1-h[i+d]) (summing over valid i)

When h[i] changes by δ:
- Terms where h[i] appears in the first factor: Δ = δ * (1-h[i+d])
- Terms where h[i] appears in the second factor (as h[i+d] where the index is i): Δ = -δ * h[i-d]

So Δcorr[d] = δ * (1-h[i+d]) - δ * h[i-d] (with appropriate boundary handling)

For a 2-opt move (flip i from 1→0, j from 0→1):
Δcorr[d] = -1 * (1-h[i+d]) + h[i-d] + 1 * (1-h[j+d]) - h[j-d]
= -(1-h[i+d]) + h[i-d] + (1-h[j+d]) - h[j-d]
= -1 + h[i+d] + h[i-d] + 1 - h[j+d] - h[j-d]
= h[i+d] + h[i-d] - h[j+d] - h[j-d]

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            h[mask] += diff / nm
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    return (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx

def slsqp_optimize(h0, n, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return smooth_obj(x, n, beta)
    def constraint(x):
        return np.sum(x) - n / 2.0
    cons = [{'type': 'eq', 'fun': constraint}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except:
        return h0.copy()
