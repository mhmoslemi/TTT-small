        h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad_soft(h, n, beta):
    c = c5_vals(h)
    w = soft_max_weights(c, beta)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            m = -lag
            grad[m:] += wk * (1.0 - h[:n - m])
            grad[:n - m] -= wk * h[m:]
    return grad * dx


def compute_grad_hard(h, n):
    c = c5_vals(h)
    k_max = int(np.argmax(c))
    lag = k_max - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] -= h[m:]
    return grad * dx


def adam_step(h, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    return h_new, m, v


def upsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new


def fourier_perturb(h, n, rng, amp):
    h_p = h.copy()
    for mode in range(1, 8):
        amp_m = amp / (mode * 0.7)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        h_p += amp_m * np.sin(2.0 * np.pi * mode * x / n + phase)
    return h_p


def obj_func_hard(h_v, nn):
    return compute_c5(h_v, nn)


def jac_func_hard(h_v, nn):
    return compute_grad_hard(h_v, nn)


def obj_func_soft(h_v, nn, beta):
    c = c5_vals(h_v)
    scaled = beta * c
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m))) / beta) * (2.0 / nn))


def jac_func_soft(h_v, nn, beta):
    return compute_grad_soft(h_v, nn, beta)


def sum_eq_fun(h_v, nn):
    return np.sum(h_v) - nn / 2.0


def sum_eq_jac(h_v, nn):
    return np.ones(nn)


def binary_snip(h, n):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    return h_bin


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    except Exception:
        n_init = 95
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_n = n_init
    best_c5 = compute_c5(best_h, best_n)

    # Phase 1: Adam at n=190 with annealed beta
    n1 = 190
    h1 = project_feasible(upsample(h_init, n_init, n1), n1)
    c5_1 = compute_c5(h1, n1)
    if c5_1 < best_c5:
        best_h = h1.copy()
        best_n = n1
        best_c5 = c5_1

    m_adam = np.zeros(n1)
    v_adam = np.zeros(n1)
    t_adam = 0

    beta_schedule1 = [5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]
    for beta in beta_schedule1:
        if time.time() - t0 > budget_s - 350:
            break
        lr_base = 0.004 / (1.0 + 0.05 * beta)
        n_steps = int(4000 * (1.0 / (1.0 + 0.3 * beta)))
        n_steps = max(n_steps, 800)
        for step in range(n_steps):
            if time.time() - t0 > budget_s - 350:
                break
            grad = compute_grad_soft(h1, n1, beta)
            t_adam += 1
            h1_raw, m_adam, v_adam = adam_step(h1, grad, m_adam, v_adam, t_adam, lr_base)
            h1 = project_feasible(h1_raw, n1)

        c5_1 = compute_c5(h1, n1)
        if c5_1 < best_c5:
            best_h = h1.copy()
            best_n = n1
            best_c5 = c5_1

    # Phase 2: Adam at n=380
    if time.time() - t0 < budget_s - 250:
        n2 = 380
        h2 = project_feasible(upsample(best_h, best_n, n2), n2)
        m_adam2 = np.zeros(n2)
        v_adam2 = np.zeros(n2)
        t_adam2 = 0

        beta_schedule2 = [8.0, 20.0, 50.0, 100.0, 200.0]
        for beta in beta_schedule2:
            if time.time() - t0 > budget_s - 150:
                break
            lr_base = 0.002 / (1.0 + 0.04 * beta)
            n_steps = int(3000 * (1.0 / (1.0 + 0.2 * beta)))
            n_steps = max(n_steps, 600)
            for step in range(n_steps):
                if time.time() - t0 > budget_s - 150:
                    break
                grad = compute_grad_soft(h2, n2, beta)
                t_adam2 += 1
                h2_raw, m_adam2, v_adam2 = adam_step(h2, grad, m_adam2, v_adam2, t_adam2, lr_base)
                h2 = project_feasible(h2_raw, n2)

            c5_2 = compute_c5(h2, n2)
            if c5_2 < best_c5:
                best_h = h2.copy()
                best_n = n2
                best_c5 = c5_2

    # Phase 3: SLSQP refinement on best
    if time.time() - t0 < budget_s - 80:
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (best_n,)}]
        try:
            res = minimize(
                fun=obj_func_hard, x0=best_h, jac=jac_func_hard,
                args=(best_n,), method='SLSQP', bounds=bounds,
                constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), best_n)
            c5_new = compute_c5(h_new, best_n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 4: Fourier perturbation restarts
    if time.time() - t0 < budget_s - 60:
        n_try = best_n
        bounds_p = [(0.0, 1.0)] * n_try
        cons_p = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_try,)}]
        n_restarts = 30
        for i in range(n_restarts):
            if time.time() - t0 > budget_s - 30:
                break
            amp = 0.04 + 0.06 * (i / max(n_restarts - 1, 1))
            h_p = fourier_perturb(best_h, n_try, rng, amp)
            h_p = project_feasible(h_p, n_try)
            try:
                res = minimize(
                    fun=obj_func_hard, x0=h_p, jac=jac_func_hard,
                    args=(n_try,), method='SLSQP', bounds=bounds_p,
                    constraints=cons_p, options={'maxiter': 500, 'ftol': 1e-12}
                )
                h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_try)
                c5_new = compute_c5(h_new, n_try)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 5: Binary projection heuristic
    if time.time() - t0 < budget_s - 40:
        h_bin = binary_snip(best_h, best_n)
        c5_bin = compute_c5(h_bin, best_n)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
        # Refine binary with SLSQP
        bounds_b = [(0.0, 1.0)] * best_n
        cons_b = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (best_n,)}]
        try:
            res = minimize(
                fun=obj_func_hard, x0=best_h, jac=jac_func_hard,
                args=(best_n,), method='SLSQP', bounds=bounds_b,
                constraints=cons_b, options={'maxiter': 1500, 'ftol': 1e-14}
            )
            h_new = project_feasible(np.clip(res.x, 0.0, 1.0), best_n)
            c5_new = compute_c5(h_new, best_n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 6: Final Adam polish at higher beta
    if time.time() - t0 < budget_s - 20:
        h_final = best_h.copy()
        m_f = np.zeros(best_n)
        v_f = np.zeros(best_n)
        t_f = 0
        for beta in [500.0, 1000.0]:
            if time.time() - t0 > budget_s - 5:
                break
            lr_f = 0.001 / (1.0 + 0.02 * beta)
            for step in range(1000):
                if time.time() - t0 > budget_s - 5:
                    break
                grad = compute_grad_soft(h_final, best_n, beta)
                t_f += 1
                h_final_raw, m_f, v_f = adam_step(h_final, grad, m_f, v_f, t_f, lr_f)
                h_final = project_feasible(h_final_raw, best_n)
            c5_f = compute_c5(h_final, best_n)
            if c5_f < best_c5:
                best_h = h_final
                best_c5 = c5_f

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
