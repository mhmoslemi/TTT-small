                        cand[ii] = 0
                        cand[jj] += (dv - cand[ii] + best[ii])
                    elif cand[ii] > 1:
                        cand[ii] = 1
                        cand[jj] -= (cand[ii] - 1 + best[jj] - best[jj])
                    cand[jj] = np.clip(cand[jj], 0, 1)
                    c = _eval(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                if found:
                    break
            if found:
                break
        if not found:
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-13:
                d /= norm
                for step in (0.05, 0.15, 0.4):
                    cand = _proj(best + step * d, target)
                    c = _eval(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
        if not found:
            stall += 1
            if stall >= 20:
                break
        else:
            stall = 0
    best = _proj(best, target)
    best_c = _eval(best)
    return best, best_c


def _triple_search(h, deadline, max_iter, topk):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    target = 0.5 * n
    h = _proj(h, target)
    best = h.copy()
    best_c = _eval(best)
    k = min(topk, 2 * n - 1)
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        idxs = np.argsort(corr)[-k:]
        g = np.zeros(n)
        ctr = n - 1
        for t in idxs:
            p = 1.0 / k
            kk = int(t) - ctr
            if kk == 0:
                g += p * (1.0 - 2.0 * best)
            else:
                lag = abs(kk)
                if lag >= n:
                    continue
                if kk > 0:
                    g[lag:] += p
                else:
                    g[:n - lag] += p
                g[:n - lag] -= p * best[lag:]
                g[lag:] -= p * best[:n - lag]
        neg = np.argsort(g)[:5]
        pos = np.argsort(-g)[:5]
        found = False
        for ii in neg:
            ii = int(ii)
            for jj in pos:
                jj = int(jj)
                if ii == jj:
                    continue
                for kk2 in neg:
                    kk2 = int(kk2)
                    if kk2 == ii or kk2 == jj:
                        continue
                    cap1 = min(1.0 - best[ii], best[jj])
                    cap2 = min(1.0 - best[kk2], best[jj]) / 2.0
                    cap = min(cap1, cap2)
                    if cap <= 1e-13:
                        continue
                    d1 = float(g[ii] - g[jj])
                    d2 = float(g[kk2] - g[jj])
                    if d1 >= -1e-12 and d2 >= -1e-12:
                        continue
                    for dv in (cap * 0.5, cap * 0.9):
                        cand = best.copy()
                        cand[ii] += dv
                        cand[kk2] += dv
                        cand[jj] -= 2.0 * dv
                        if cand[ii] < 0:
                            cand[ii] = 0
                        if cand[kk2] < 0:
                            cand[kk2] = 0
                        if cand[jj] < 0:
                            cand[jj] = 0
                        if cand[jj] > 1:
                            cand[jj] = 1
                        cand = np.clip(cand, 0, 1)
                        c = _eval(cand)
                        if c < best_c - 1e-12:
                            best = cand
                            best_c = c
                            found = True
                    if found:
                        break
                if found:
                    break
            if found:
                break
        if not found:
            stall += 1
            if stall >= 25:
                break
        else:
            stall = 0
    best = _proj(best, target)
    best_c = _eval(best)
    return best, best_c


def _upsample(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return _proj(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return _proj(h, 0.5 * n)


def _make_start(n, rng, mode):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    if mode == 0:
        h = 0.5 * np.ones(n)
        for m in range(1, 8):
            amp = 0.3 * (0.75 ** (m - 1))
            freq = m + int(rng.integers(0, 2))
            phase = rng.uniform(0, 2 * np.pi)
            h += amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
        h = np.clip(h, 0, 1)
    elif mode == 1:
        h = 0.5 * np.ones(n)
        h[:n // 3] = 0.7
        h[n // 3:2 * n // 3] = 0.5
        h[2 * n // 3:] = 0.3
    elif mode == 2:
        h = np.zeros(n)
        block = max(2, n // 10)
        for i in range(0, n, block):
            h[i:min(i + block // 2, n)] = 1.0
            h[min(i + block // 2, n):min(i + block, n)] = 0.0
    else:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * 3 * x / 2.0 + rng.uniform(0, 6))
        h = np.clip(h, 0, 1)
    return _proj(h, 0.5 * n)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + budget - 15.0

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        pn = parent.size
        pt = 0.5 * pn
        pp = _proj(parent, pt)
        cp = _eval(pp)
        if cp < best_c:
            best_h = pp
            best_c = cp
            best_n = pn

    if best_h is None:
        best_h = _make_start(100, rng, 0)
        best_c = _eval(best_h)
        best_n = best_h.size

    def try_optimize(h0, n, time_alloc):
        nonlocal best_h, best_c, best_n
        h0 = _proj(np.asarray(h0, dtype=float), 0.5 * n)
        c0 = _eval(h0)
        if c0 < best_c:
            best_h = h0.copy()
            best_c = c0
            best_n = n
        sub_dl = time.perf_counter() + time_alloc
        if n <= 120:
            alphas = (15.0, 50.0, 150.0, 400.0)
            mx = 200
        elif n <= 300:
            alphas = (20.0, 60.0, 200.0)
            mx = 120
        elif n <= 600:
            alphas = (20.0, 80.0)
            mx = 70
        else:
            alphas = (20.0, 60.0)
            mx = 40
        h1, c1 = _slsqp(h0, n, sub_dl, alphas, mx)
        if c1 < best_c:
            best_h = h1
            best_c = c1
            best_n = n
        if time.perf_counter() < sub_dl:
            h2, c2 = _grad_polish(h1, sub_dl, 300, topk=6)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = n
        if time.perf_counter() < sub_dl:
            h3, c3 = _pair_search(h1, sub_dl, 400, topk=6)
            if c3 < best_c:
                best_h = h3
                best_c = c3
                best_n = n
        return best_h, best_c

    # Phase 1: Optimize at parent resolution
    pn = best_n
    rem = deadline - time.perf_counter()
    t1 = min(180.0, rem * 0.20)
    if rem > 30:
        try_optimize(best_h, pn, t1)

    # Phase 2: Upsample to 2x
    n2 = min(200, pn * 2 + 10)
    rem = deadline - time.perf_counter()
    t2 = min(250.0, rem * 0.25)
    if rem > 40:
        h2 = _upsample(best_h, n2)
        try_optimize(h2, n2, t2)

    # Phase 3: Try n=100 from scratch
    rem = deadline - time.perf_counter()
    t3 = min(100.0, rem * 0.12)
    if rem > 40:
        h3 = _upsample(best_h, 100)
        try_optimize(h3, 100, t3)

    # Phase 4: Go to higher resolution
    n4 = min(400, pn * 4 + 20)
    rem = deadline - time.perf_counter()
    t4 = min(250.0, rem * 0.25)
    if rem > 50:
        h4 = _upsample(best_h, n4)
        try_optimize(h4, n4, t4)

    # Phase 5: Try another structured start at moderate n
    n5 = 200
    rem = deadline - time.perf_counter()
    t5 = min(120.0, rem * 0.15)
    if rem > 40:
        h5 = _make_start(n5, rng, 3)
        try_optimize(h5, n5, t5)

    # Phase 6: Try n=300
    rem = deadline - time.perf_counter()
    t6 = min(150.0, rem * 0.18)
    if rem > 40:
        h6 = _upsample(best_h, 300)
        try_optimize(h6, 300, t6)

    # Final polish on best found
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 20:
        bn = best_h.size
        hp, cp = _pair_search(best_h, deadline, 2000, topk=8)
        if cp < best_c:
            best_h = hp
            best_c = cp
            best_n = bn
    if best_h is not None and deadline - time.perf_counter() > 10:
        hp, cp = _grad_polish(best_h, deadline, 1000, topk=8)
        if cp < best_c:
            best_h = hp
            best_c = cp
            best_n = best_h.size
    if best_h is not None and deadline - time.perf_counter() > 10:
        hp, cp = _triple_search(best_h, deadline, 500, topk=6)
        if cp < best_c:
            best_h = hp
            best_c = cp
            best_n = best_h.size

    # Final: re-evaluate and ensure feasibility
    if best_h is None:
        fb = _proj(np.full(100, 0.5), 50.0)
        return fb, _eval(fb), 100

    final = _proj(best_h, 0.5 * best_h.size)
    final_c = _eval(final)
    return final, float(final_c), int(final.size)
