                local_best_h = h0.copy()

            for beta in [20, 60, 150, 300]:
                if t_left() < 15:
                    break
                h_new, c5_new = slsqp_run(local_best_h, n, beta, 350, t_left)
                if c5_new < local_best_c5:
                    local_best_c5 = c5_new
                    local_best_h = h_new.copy()
                    print("  SLSQP n={} beta={}: C5={:.6f}".format(n, beta, c5_new))

            if t_left() > 20:
                h_np, c5_np = nelder_polish(local_best_h, n, t_left)
                if c5_np < local_best_c5:
                    local_best_c5 = c5_np
                    local_best_h = h_np.copy()
                    print("  NM n={}: C5={:.6f}".format(n, c5_np))

            if t_left() > 25:
                h_sa, c5_sa = binary_sa(local_best_h, n,
                                        min(8000, int(t_left() * 80)),
                                        0.008, 0.0001, rng, t_left)
                if c5_sa < local_best_c5:
                    local_best_c5 = c5_sa
                    local_best_h = h_sa.copy()
                    print("  SA n={}: C5={:.6f}".format(n, c5_sa))

            if t_left() > 15:
                h_sg = local_best_h.copy()
                c5_sg = local_best_c5
                for sg_iter in range(200):
                    if t_left() < 5:
                        break
                    ss = max(0.01, 0.1 * (0.95 ** sg_iter))
                    h_sg = subgrad_step(h_sg, n, ss, t_left)
                    c5_test = compute_c5(h_sg)
                    if c5_test < c5_sg:
                        c5_sg = c5_test
                    else:
                        h_sg = h_sg + 0.3 * ss * (h_sg - local_best_h)
                        h_sg = project_feasible(h_sg, n)
                        c5_test = compute_c5(h_sg)
                        if c5_test < c5_sg:
                            c5_sg = c5_test
                if c5_sg < local_best_c5:
                    local_best_c5 = c5_sg
                    local_best_h = h_sg.copy()
                    print("  Subgrad n={}: C5={:.6f}".format(n, c5_sg))

            if local_best_h is not None and t_left() > 10:
                for beta in [500]:
                    h_new, c5_new = slsqp_run(local_best_h, n, beta, 300, t_left)
                    if c5_new < local_best_c5:
                        local_best_c5 = c5_new
                        local_best_h = h_new.copy()

        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = local_best_h.copy()
            best_n = n
            print("  Best updated: n={}, C5={:.6f}".format(n, best_c5))

    if t_left() > 10 and best_n > 0:
        h_round = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_round))
        n_want = best_n // 2
        if n_ones > n_want:
            oi = np.where(h_round == 1.0)[0]
            ord_i = np.argsort(best_h[oi])[:n_ones - n_want]
            h_round[oi[ord_i]] = 0.0
        elif n_ones < n_want:
            zi = np.where(h_round == 0.0)[0]
            ord_i = np.argsort(-best_h[zi])[:n_want - n_ones]
            h_round[zi[ord_i]] = 1.0
        c5_round = compute_c5(h_round)
        if c5_round < best_c5:
            best_c5 = c5_round
            best_h = h_round.copy()
            print("  Binary rounding: C5={:.6f}".format(best_c5))

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
