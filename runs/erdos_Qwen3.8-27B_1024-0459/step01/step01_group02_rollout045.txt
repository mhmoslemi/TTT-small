def project_feasible(h, n):

        if np.sum(mask) == 0:

        h[mask] += step
        h = np.clip(h, 0.0, 1.0)

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):

        if abs(diff) < 1e-14:

                best_c5 = c_rand2
                best_h = h_rand2.copy()
                best_n = n
                print("  Random 2-opt: n={}, C5={:.6f}".format(n, best_c5))

    # Final continuous refinement on best solution
    if time_left() > 15 and best_n > 0:
        n = best_n
        h0 = project_feasible(best_h, n)
        try:
            h_fin = optimize_slsqp(h0, n, beta=500.0, maxiter=500)
            h_fin = project_feasible(h_fin, n)
            c_fin = compute_c5(h_fin)
            if c_fin < best_c5:
                best_c5 = c_fin
                best_h = h_fin
                print("  Final polish: C5={:.6f}".format(best_c5))
        except Exception:
            pass

        # Also try L-BFGS-B one more time
        try:
            h_fin2 = optimize_lbb(h_fin, n, beta=200.0, maxiter=1000)
            h_fin2 = project_feasible(h_fin2, n)
            c_fin2 = compute_c5(h_fin2)
            if c_fin2 < best_c5:
                best_c5 = c_fin2
                best_h = h_fin2
                print("  Final L-BFGS: C5={:.6f}".format(best_c5))
        except Exception:
            pass

    # Final binary rounding attempt
    if time_left() > 10 and best_n > 0:
        h_br = round_to_binary(best_h, best_n)
        c_br = compute_c5(h_br)
        if c_br < best_c5:
            best_c5 = c_br
            best_h = h_br
            print("  Final binary: C5={:.6f}".format(best_c5))

        h_br, c_br2 = binary_2opt(h_br, best_n, max_iters=2000)
        if c_br2 < best_c5:
            best_c5 = c_br2
            best_h = h_br
            best_n = best_n
            print("  Final 2-opt: C5={:.6f}".format(best_c5))

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
