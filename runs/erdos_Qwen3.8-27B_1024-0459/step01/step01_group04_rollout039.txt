def score_max(h, n):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, h, mode="full")

        scores[2*d] = total - prefix[d] - ac[d]
    return scores, float(np.max(scores))

def constraint_eq(x, *args):
    n = max(1, int(x.size))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)

    n = max(1, int(x.size))
    return np.full((1, n), 1.0 / float(n), dtype=float)

# Thue-Morse sequence
def thue_morse(n):
    """Generate Thue-Morse sequence of length n (0-indexed)."""
    h = np.zeros(n, dtype=np.int8)
    for i in range(n):
        # Count number of 1-bits in i
        h[i] = bin(i).count('1') % 2
    return h

def generate_structured(n, rng):
    """Generate structured binary sequences."""
    cands = []
    # Thue-Morse
    tm = thue_morse(n)
    cands.append(tm.astype(float))
    # Reverse Thue-Morse
    cands.append(tm[::-1].astype(float))
    # Complementary
    cands.append((1 - tm).astype(float))
    # Random
    idx = rng.choice(n, size=n//2, replace=False)
    r = np.zeros(n, dtype=np.int8)
    r[idx] = 1
    cands.append(r.astype(float))
    return cands

def slsqp_optimize(h0, n, time_limit, alpha_start=None, alpha_end=None):
    """Run SLSQP with adaptive alpha."""
    check_deadline()
    n = int(n)
    h0 = as_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]
    
    if alpha_start is None:
        alpha_start = 0.5
    if alpha_end is None:
        alpha_end = 0.0
    
    # Multi-stage alpha schedule
    stages = []
    if alpha_start > alpha_end + 1e-10:
        # Geometric schedule from alpha_start to alpha_end
        num_stages = 4
        if num_stages > 1:
            ratio = (alpha_end + 1e-6) / (alpha_start + 1e-6)
            for s in range(num_stages):
                a = alpha_start * (ratio ** (s / (num_stages - 1))) if num_stages > 1 else alpha_end
                a = max(a, alpha_end)
                stages.append(min(a, alpha_start))
        else:
            stages.append(alpha_end)
    else:
        stages.append(alpha_end)
    
    # Add final exact stage
    if stages[-1] > 1e-8:
        stages.append(0.0)
    
    h_cur = h0.copy()
    best_c = compute_c5(h_cur)
    
    total_iter_budget = int(200.0 * (70.0 / float(n)))
    total_iter_budget = max(50, min(500, total_iter_budget))
    iters_per_stage = total_iter_budget // len(stages)
    iters_per_stage = max(20, iters_per_stage)
    
    start_t = time.time()
    for alpha in stages:
        if time.time() - start_t > time_limit * 0.85:
            break
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 30:
            break
        res = None
        try:

        except BudgetError:

        except Exception:
            break

            c_new = compute_c5(h_new)

                h_cur = h_new

def _random_swap(h, rng, n):

    ones = np.flatnonzero(h == 1)

        return -1, -1
    i = int(zeros[int(rng.integers(zeros.size))])
    j = int(ones[int(rng.integers(ones.size))])

def _swap_delta(h, i, j, old_hi, new_hi, n):

    _apply_single_flip_delta(delta, h, i, old_hi, new_hi, n)

    _apply_single_flip_delta(delta, h, j, old_hj, new_hj, n)

        if elapsed > time_limit * 0.8:
            break
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 30:
            break
        remaining = time_limit - elapsed
        if remaining < 15:
            break
        maxiter = int(iters_base * 0.3)
        maxiter = max(20, min(150, maxiter))
        h_new = slsqp_stage(h0, n, alpha, maxiter, 1e-10)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h0 = h_new
        else:
            h0 = best_h.copy()
    
    return best_h, best_c

def perturb(h, n, rng, amount):
    h = h.astype(float).copy()
    noise = rng.normal(0, amount, size=n)
    h_new = h + noise
    return as_feasible(h_new, n)

def thue_morse_seq(n):
    h = np.zeros(n, dtype=np.int8)
    for i in range(n):
        c = 0
        v = i
        while v > 0:
            c += v & 1
            v >>= 1
        h[i] = c % 2
    return h

def binary_score_full(hb, n):
    hf = hb.astype(float)
    max_s = 0.0
    for d in range(1, n):
        s1 = float(np.dot(hf[:n-d], 1.0 - hf[d:]))
        s2 = float(np.dot(hf[d:], 1.0 - hf[:n-d]))
        if s1 > max_s:
            max_s = s1
        if s2 > max_s:
            max_s = s2
    return max_s

def binary_hill_climb(h_start, n, time_limit, rng):
    n = int(n)
    h = np.array(h_start, dtype=np.int8)
    target = n // 2
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    if ones.size > target:
        to_remove = rng.choice(ones, size=ones.size - target, replace=False)
        h[to_remove] = 0
    elif ones.size < target:
        to_add = rng.choice(zeros, size=target - ones.size, replace=False)
        h[to_add] = 1
    
    cur = binary_score_full(h, n)
    best = cur
    best_h = h.copy()
    start_t = time.time()
    stall = 0
    T = 1.0
    
    while time.time() - start_t < time_limit:
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        if stall > 10000:
            break
        
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if zeros.size == 0 or ones.size == 0:
            break
        
        if rng.random() < 0.5 and cur > 0:
            # Targeted: find a swap that reduces an active lag
            active_d = _find_active_dags(h, n, cur)
            if len(active_d) > 0:
                d = active_d[int(rng.integers(len(active_d)))]
                if rng.random() < 0.5:
                    mask = (h[:n-d] == 1) & (h[d:] == 0)
                else:
                    mask = (h[:n-d] == 0) & (h[d:] == 1)
                cand = np.nonzero(mask)[0]
                if cand.size > 0:
                    i = int(cand[int(rng.integers(cand.size))])
                    j = i + d
                else:
                    i = int(zeros[int(rng.integers(zeros.size))])
                    j = int(ones[int(rng.integers(ones.size))])
            else:
                i = int(zeros[int(rng.integers(zeros.size))])
                j = int(ones[int(rng.integers(ones.size))])
        else:
            i = int(zeros[int(rng.integers(zeros.size))])
            j = int(ones[int(rng.integers(ones.size))])
        
        if i == j:
            continue
        if h[i] == h[j]:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_cur = binary_score_full(h, n)
        
        if new_cur <= cur + 1e-9:
            cur = new_cur
            if cur < best - 1e-9:
                best = cur
                best_h = h.copy()
                stall = 0
                T = max(T * 0.9, 0.05)
            else:
                stall += 1
        else:
            if T > 0.05 and rng.random() < math.exp(-(new_cur - cur) / T):
                cur = new_cur
                stall += 1
            else:
                h[i], h[j] = h[j], h[i]
                stall += 1
            T = max(T * 0.998, 0.05)
    
    best_c = best * (2.0 / n)
    return best_h.astype(float), best_c

def _find_active_dags(h, n, cur):
    hf = h.astype(float)
    active = []
    for d in range(1, n):
        s1 = float(np.dot(hf[:n-d], 1.0 - hf[d:]))
        s2 = float(np.dot(hf[d:], 1.0 - hf[:n-d]))
        if max(s1, s2) >= cur - 1e-9:
            active.append(d)
    return active

def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 20.0 if budget_s > 60 else max(2.0, 0.1 * budget_s)
    deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = deadline
    
    rng = np.random.default_rng(seed)
    parent = get_parent()
    
    if parent is not None:
        n0 = int(parent.size)
        h0 = parent.copy()
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = as_feasible(np.full(n0, 0.5), n0)
        c0 = compute_c5(h0)
    
    best_h = h0.copy()
    best_c = c0
    best_n = n0
    
    def update_best(h, c, n):
        nonlocal best_h, best_c, best_n
        if c < best_c - 1e-12:
            best_h = h.copy()
            best_c = c
            best_n = n
    
    # Phase 1: SLSQP on parent resolution
    remaining = deadline - time.time()
    if remaining > 60:
        t_lim = min(300.0, remaining * 0.35)
        h_opt, c_opt = adaptive_slsqp(h0, n0, t_lim)
        if c_opt < best_c:
            update_best(h_opt, c_opt, n0)
        h0 = h_opt  # use improved solution for subsequent phases
    
    # Phase 2: SLSQP with perturbation restarts on n0
    remaining = deadline - time.time()
    if remaining > 40:
        n_restarts = 3
        t_per = (remaining * 0.25) / n_restarts
        for r in range(n_restarts):
            if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE - 30:
                break
            if time.time() - t0 > budget_s - 40:
                break
            h_pert = perturb(best_h, best_n, rng, 0.15)
            h_opt, c_opt = adaptive_slsqp(h_pert, best_n, t_per)
            if c_opt < best_c:
                update_best(h_opt, c_opt, best_n)
    
    # Phase 3: Upsample to 2*n0 and optimize
    n2 = 2 * n0
    remaining = deadline - time.time()
    if n2 <= 280 and remaining > 60:
        h_up = upsample(best_h, n2)
        h_up = as_feasible(h_up, n2)
        c_up = compute_c5(h_up)
        if c_up < best_c:
            update_best(h_up, c_up, n2)
        t_lim = min(250.0, remaining * 0.35)
        h_opt, c_opt = adaptive_slsqp(h_up, n2, t_lim)
        if c_opt < best_c:
            update_best(h_opt, c_opt, n2)
    
    # Phase 4: Binary search at n0
    remaining = deadline - time.time()
    if n0 % 2 == 0 and remaining > 30:
        # Threshold the best continuous solution
        h_bin = (best_h > 0.5).astype(float)
        # Fix to have exactly n0/2 ones
        ones = np.flatnonzero(h_bin == 1)
        target = n0 // 2
        if ones.size > target:
            # Remove ones with smallest values
            order = np.argsort(-best_h[ones])
            for i in range(ones.size - target):
                h_bin[ones[order[i]]] = 0
        elif ones.size < target:
            zeros = np.flatnonzero(h_bin == 0)
            order = np.argsort(best_h[zeros])
            for i in range(target - ones.size):
                h_bin[zeros[order[i]]] = 1
        
        t_lim = min(80.0, remaining * 0.3)
        h_b, c_b = binary_hill_climb(h_bin, n0, t_lim, rng)
        if c_b < best_c:
            update_best(h_b, c_b, n0)
        
        # Also try Thue-Morse
        if n0 <= 256:
            tm = thue_morse_seq(n0).astype(float)
            # Check if it has exactly n0/2 ones (it does for powers of 2)
            if int(tm.sum()) == target:
                h_b2, c_b2 = binary_hill_climb(tm, n0, min(30.0, remaining * 0.15), rng)
                if c_b2 < best_c:
                    update_best(h_b2, c_b2, n0)
    
    # Phase 5: Binary search at n2 (if n2 is even and reasonable)
    remaining = deadline - time.time()
    if n2 % 2 == 0 and n2 <= 280 and remaining > 30:
        h_bin2 = (best_h > 0.5).astype(float)
        if h_bin2.size != n2:
            h_bin2 = (upsample(best_h, n2) > 0.5).astype(float)
        ones2 = np.flatnonzero(h_bin2 == 1)
        target2 = n2 // 2
        if ones2.size > target2:
            order = np.argsort(-h_bin2[ones2] * 0 + rng.random(ones2.size))
            for i in range(ones2.size - target2):
                h_bin2[ones2[order[i]]] = 0
        elif ones2.size < target2:
            zeros2 = np.flatnonzero(h_bin2 == 0)
            order = np.argsort(rng.random(zeros2.size))
            for i in range(target2 - ones2.size):
                h_bin2[zeros2[order[i]]] = 1
        
        t_lim = min(60.0, remaining * 0.4)
        h_b3, c_b3 = binary_hill_climb(h_bin2, n2, t_lim, rng)
        if c_b3 < best_c:
            update_best(h_b3, c_b3, n2)
    
    # Phase 6: Final refinement of best solution
    remaining = deadline - time.time()
    if remaining > 30:
        h_final = as_feasible(best_h, best_n)
        t_lim = min(120.0, remaining * 0.7)
        h_opt, c_opt = adaptive_slsqp(h_final, best_n, t_lim)
        if c_opt < best_c:
            update_best(h_opt, c_opt, best_n)
    
    # Ensure validity
    h_final = as_feasible(best_h, best_n)
    c_final = compute_c5(h_final)
    n_final = int(best_n)
    
    if not math.isfinite(c_final):
        h_final = as_feasible(h0, n0)
        c_final = compute_c5(h_final)
        n_final = n0
    
    _RUN_DEADLINE = None
    return h_final, float(c_final), int(n_final)
