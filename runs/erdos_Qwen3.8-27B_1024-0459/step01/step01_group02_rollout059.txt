    if n % 2 == 0:
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0
    else:
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0
        h[rng.randint(n)] = 0.5  # This might overlap... let me handle differently
    return h

def binary_local_search(h, n, max_iters, rng, n_tries=5):
    h = h.copy()
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    best_raw = np.max(corr_raw)
    
    ones = np.where(h > 0.99)[0]
    zeros = np.where(h < 0.01)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h, best_raw * dx
    
    no_improve = 0
    stall_limit = min(8000, max(2000, max_iters // 4))
    
    for it in range(max_iters):
        if no_improve >= stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        for _ in range(n_tries):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            
            old_hi = h[i]
            old_hj = h[j]
            h[i] = old_hj
            h[j] = old_hi
            
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_raw = np.max(new_corr)
            
            if new_raw < best_raw - 1e-15:
                best_raw = new_raw
                no_improve = 0
                improved = True
                ones = np.where(h > 0.99)[0]
                zeros = np.where(h < 0.01)[0]
                break
            else:
                h[i] = old_hi
                h[j] = old_hj
                no_improve += 1
    
    return h, best_raw * dx

def smooth_obj(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _g_beta
    vals = beta * (corr - m)
    vals = np.clip(vals, -100, 0)
    log_sum = np.log(np.sum(np.exp(vals)))
    return (m + log_sum / beta) * dx

def raw_obj(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx

def eq_con(x):
    return np.sum(x) - _g_target
