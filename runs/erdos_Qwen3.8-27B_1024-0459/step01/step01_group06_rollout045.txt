    """Subgradient of c[k_idx] w.r.t. h, times dx."""

    grad = np.zeros(n)
    lag = k_idx - (n - 1)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        if n - lag > 0:
            grad[:n - lag] -= h[lag:]
    else:
        p = -lag
        grad[:n - p] += (1.0 - h[p:])
        if p < n:
            grad[p:] -= h[:n - p]
    return grad * dx

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(np.max(compute_corr(h, n)) * (2.0 / n))


def compute_subgradient_at_lags(h, n, lags, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag, w in zip(lags, weights):
        if w < 1e-15:
            continue
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= w * h[lag:]
        else:
            p = -lag
            grad[:n - p] += w * (1.0 - h[p:])
            if p < n:
                grad[p:] -= w * h[:n - p]
    return grad * dx


def project_feasible(h, n):

def slsqp_polish(h0, n, t_limit):
    def obj(h):
        c = compute_corr(h, n)
        return float(np.max(c) * (2.0 / n))

    def obj_grad(h):
        c = compute_corr(h, n)
        k_max = int(np.argmax(c))
        g = compute_subgradient_at_lags(h, n, [k_max - (n - 1)], [1.0])
        return g

    def sum_fun(h):
        return np.sum(h) - n / 2.0

    def sum_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(
            fun=obj, x0=h0, jac=obj_grad,
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 600, 'ftol': 1e-14}
        )
        if time.time() > t_limit:
            return h0, compute_c5(h0, n)
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        c5_new = compute_c5(h_new, n)
        if c5_new < compute_c5(h0, n):
            return h_new, c5_new
    except Exception:
        pass
    return h0, compute_c5(h0, n)

    x_old = np.linspace(0, 2, n_old, endpoint=False) + dx_old / 2.0

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
