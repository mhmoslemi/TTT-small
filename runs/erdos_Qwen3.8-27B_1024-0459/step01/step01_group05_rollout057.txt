def smooth_max_grad_analytical(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full')  # length 2n-1, symmetric
    mx = np.max(corr)
    shifted = beta * (corr - mx)
    shifted = np.clip(shifted, -80, 80)
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    
    grad = np.zeros(n)
    
    # k=0 term (middle of corr array, index n-1)
    w0 = w[n-1]
    if w0 > 1e-12:
        grad += w0 * (1.0 - 2.0 * h)
    
    # k>0 terms (both +k and -k contribute equally)
    for k in range(1, n):
        wk = 2.0 * w[n-1+k]  # factor of 2 for ±k symmetry
        if wk < 1e-12:
            continue
        # grad component for lag k:
        # g[t] = [t >= k] - [t >= k]*h[t-k] - [t <= n-1-k]*h[t+k]
        g = np.zeros(n)
        g[k:] += 1.0
        g[k:] -= h[:n-k]      # h[t-k] for t in [k, n-1]
        g[:n-k] -= h[k:]      # h[t+k] for t in [0, n-1-k]
        grad += wk * g
    
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) == 0:
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = beta * (corr - mx)
    s = np.clip(s, -80, 80)
    return mx + np.log(np.sum(np.exp(s))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = beta * (corr - mx)

    h0 = project_sum(h0, target)

    bounds = [(0, 1)] * n

            lambda h: smooth_max_val(h, beta),
            h0,

            constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if res.x is not None:

    except:
        pass
    return h0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = int(np.sum(free))
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            free = np.where(h < 1.0)[0]
        else:
            free = np.where(h > 0.0)[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = beta * (corr - mx)
    s = np.clip(s, -80.0, 80.0)
    return mx + np.log(np.sum(np.exp(s))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    s = beta * (corr - mx)
    s = np.clip(s, -80.0, 80.0)
    ev = np.exp(s)
    w = ev / np.sum(ev)
    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = 2.0 * w[n - 1 + k]
        if wk < 1e-14:
            continue
        g = np.zeros(n)
        g[k:] += 1.0
        g[k:] -= h[:n - k]
        g[:n - k] -= h[k:]
        grad += wk * g
    return grad

def constraint_sum(h, target):
    return np.sum(h) - target

def constraint_jac(h, target):
    return np.ones(len(h))

def run_slsqp(h0, n, beta, maxiter, ftol):
    target = n / 2.0
    h0 = project_sum(h0, target)
    cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, target),
             'jac': lambda x: constraint_jac(x, target)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_max_val(x, beta),
            h0,
            jac=lambda x: smooth_max_grad(x, beta),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        if res.x is not None:
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0

def run_projected_grad(h0, n, beta, maxiter, lr_init, lr_decay):
    target = n / 2.0
    h = project_sum(h0, target)
    lr = lr_init
    for it in range(maxiter):
        f_cur = smooth_max_val(h, beta)
        g = smooth_max_grad(h, beta)
        h_new = h - lr * g
        h_new = project_sum(h_new, target)
        f_new = smooth_max_val(h_new, beta)
