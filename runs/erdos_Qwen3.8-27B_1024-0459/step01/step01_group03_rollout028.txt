def proj(x, target):
    x = np.clip(np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    n = x.size

        if s < target:
            lo = mid
        else:
            hi = mid

            out[idx] = float(np.clip(out[idx] + res, 0.0, 1.0))
    return out

def upsample(p, n):
    p = np.asarray(p, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 < 2:
        return proj(np.full(n, 0.5), n * 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p, left=p[0], right=p[-1])
    return proj(h, n * 0.5)

def corr_grad_sum(h, indices):
    """Sum of gradients of correlation at given output indices."""
    n = h.size
    G = np.zeros(n)
    center = n - 1
    for t in indices:
        k = int(t) - center
        if k == 0:
            G += (1.0 - 2.0 * h)
        elif k > 0:
            G[:n-k] += (1.0 - h[k:])
            G[k:] -= h[:n-k]
        else:
            m = -k
            G[m:] += (1.0 - h[:n-m])
            G[:n-m] -= h[m:]
    return G

def smooth_obj(x):
    n = _G_STATE['n']
    alpha = _G_STATE['alpha']
    h = np.asarray(x, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    shifted = np.clip(alpha * (corr - mx), -700, 0)
    ev = np.exp(shifted)
    s = np.sum(ev)
    if s < 1e-300:
        return 1e100
    return (2.0/n) * (mx + np.log(s)/alpha)
