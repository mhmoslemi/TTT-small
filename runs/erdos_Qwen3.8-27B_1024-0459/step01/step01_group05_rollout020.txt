def analytic_gradient(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # smooth max weights
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)  # length 2n-1
    
    # W[k] = weights[n-1+k] for k = -(n-1), ..., n-1
    # W has length 2n-1, indexed from -(n-1) to n-1
    
    # First term: convolve(W, 1-h) evaluated at j=0,...,n-1
    # (W * v)[j] = sum_k W[k] * v[j-k] for valid k and j-k
    # W[k] is defined for k in [-(n-1), n-1]
    # v = 1-h, defined for indices 0 to n-1
    # (W * v)[j] = sum_k W[k] * v[j-k] where j-k in [0,n-1] and k in [-(n-1),n-1]
    
    # This is a standard convolution. We can compute it as:
    # full_conv = np.correlate(W_extended, (1-h), mode='full')
    # where W_extended is W placed at the right position.
    
    # Actually, let me just do it directly:
    W = weights  # W[k-(n-1)] = weights[k], so W is indexed 0 to 2n-2
    # W[k] in my "centered" indexing = weights[k+(n-1)]
    
    # First term at position j: sum_{p=0}^{n-1} W_centered[j-p] * (1-h[p])
    # where W_centered[k] = weights[n-1+k]
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    h_vec = h
    
    for j in range(n):
        # First term: sum_{p=0}^{n-1} weights[n-1+j-p] * (1-h[p])
        idx1 = n - 1 + j - np.arange(n)  # indices into weights
        # idx1 ranges from n-1+j-(n-1) = j to n-1+j
        # All in [0, 2n-2] since j in [0,n-1]
        grad[j] += np.sum(weights[idx1] * one_minus_h)
        
        # Second term: sum_{q=0}^{n-1} weights[n-1+q-j] * h[q]
        idx2 = n - 1 + np.arange(n) - j  # indices into weights
        # idx2 ranges from n-1-j to 2n-2-j

def smooth_max_fn(x, beta):
    m = np.max(x)

    shifted = np.clip(shifted, -60.0, 60.0)
    val = m + np.logaddexp.reduce(shifted) / beta  # Actually this isn't quite right

def get_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -60.0, 60.0)
    exp_s = np.exp(shifted)
    return exp_s / np.sum(exp_s)

def smooth_max_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_fn(corr, beta)

def smooth_max_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = get_weights(corr, beta)  # length 2n-1
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for j in range(n):
        # First term: sum_{p=0}^{n-1} weights[n-1+j-p] * (1-h[p])
        idx1 = n - 1 + j - np.arange(n)
        grad[j] += np.dot(weights[idx1], one_minus_h)
        # Second term: sum_{q=0}^{n-1} weights[n-1+q-j] * h[q]
        idx2 = n - 1 + np.arange(n) - j
        grad[j] -= np.dot(weights[idx2], h)
    
    return grad

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
        else:
            free = np.where(h > 0.0)[0]
            if len(free) > 0:
                per = diff / len(free)
                h[free] = np.clip(h[free] + per, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n, target, beta, max_iter, ftol):
    def obj(h):
        return smooth_max_obj(h, beta)
    
    def grad(h):
        return smooth_max_grad(h, beta)
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': max_iter, 'ftol': ftol})
    
    if result is not None and result.x is not None:

    return h0
