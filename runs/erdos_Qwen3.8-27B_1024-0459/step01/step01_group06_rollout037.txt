import numpy as np
from scipy.optimize import minimize
import time

def get_c5_values(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, n):
    c = get_c5_values(h, n)
    return float(np.max(c) * (2.0 / n))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
        else:
            mask = h < 1.0 - 1e-15
        if mask.sum() == 0:
            break
        h[mask] -= diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, n, beta):
    c = get_c5_values(h, n) * (2.0 / n)
    m = np.max(c)
    return float(m + np.log(np.sum(np.exp(c - m))) / beta)

def soft_max_grad(h, n, beta):
    c = get_c5_values(h, n) * (2.0 / n)
    m = np.max(c)
    w = np.exp(c - m)
    w /= w.sum()
    dx = 2.0 / n
    one_m_h = 1.0 - h
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wi = w[lag + (n-1)]
        if wi < 1e-15:
            continue
        if lag >= 0:
            grad[:n-lag] += wi * one_m_h[lag:]
            grad[lag:] -= wi * h[:n-lag]
        else:
            lp = -lag
            grad[lp:] += wi * one_m_h[:n-lp]
            grad[:n-lp] -= wi * h[lp:]
    return grad * dx

def fista_optimize(h0, n, beta_schedule, time_limit, rng):
    """FISTA optimization with soft-max objective"""
    t_start = time.time()
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_obj = soft_max_obj(h0, n, beta_schedule[0])
    
    for beta_idx, beta in enumerate(beta_schedule):
        # FISTA iterations for this beta
        z_prev = h.copy()
        t = 1.0
        y = h.copy()
        L = 10.0  # initial Lipschitz estimate
        
        max_iter = 500
        for k in range(max_iter):
            if time.time() - t_start > time_limit:
                break
            
            g = soft_max_grad(y, n, beta)
            # Backtracking line search
            step = 1.0 / L
            x_new = y - step * g
            x_proj = project_feasible(x_new, n)
            
            # Check descent
            f_new = soft_max_obj(x_proj, n, beta)
            f_quad = soft_max_obj(y, n, beta) + g @ (x_proj - y) + (L/2) * np.sum((x_proj - y)**2)
            
            if f_new > f_quad + 1e-10:
                L *= 2
                continue
            
            z_new = x_proj
            t_new = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
            y_new = z_new + ((t-1)/t_new) * (z_new - z_prev)
            y_new = project_feasible(y_new, n)
            z_prev = z_new
            t = t_new
            y = y_new
            h = z_new
            
            if soft_max_obj(h, n, beta) < best_obj:
                best_obj = soft_max_obj(h, n, beta)
                best_h = h.copy()
        
        # After FISTA, do a few SLSQP iterations for polish
        if time.time() - t_start < time_limit:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
            try:
                res = minimize(
                    fun=soft_max_obj,
                    x0=h,
                    jac=soft_max_grad,
                    args=(n, beta),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-12}
                )
                h_new = project_feasible(np.clip(res.x, 0, 1), n)
                if compute_c5(h_new, n) < compute_c5(best_h, n):
                    best_h = h_new
            except Exception:
                pass
    
    return best_h, compute_c5(best_h, n)

def get_c5_values(h, n):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, n):
    c = get_c5_values(h, n)
    return float(np.max(c) * (2.0 / n))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = h.sum() - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-10
        else:
            mask = h < 1.0 - 1e-10
        if mask.sum() == 0:
            break
        step = diff / mask.sum()
        # Don't overshoot
        if diff > 0:
            max_step = h[mask].min() - 1e-10
            step = min(step, max_step)
        else:
            max_step = 1.0 - h[mask].max() - 1e-10
            step = min(-step, max_step)
            step = -step
        h[mask] -= step
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq_fun(h, n):
    return np.sum(h) - n / 2.0

def sum_eq_jac(h, n):
    return np.ones(n)

def soft_max_fun(h, n, beta):
    c = get_c5_values(h, n) * (2.0 / n)
    m = np.max(c)
    return float(m + np.log(np.sum(np.exp(c - m))) / beta)

def soft_max_grad(h, n, beta):
    c = get_c5_values(h, n) * (2.0 / n)
    m = np.max(c)
    w = np.exp(c - m)
    w /= w.sum()
    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    for lag in range(-(n-1), n):
        wi = w[lag + n - 1]
        if wi < 1e-14:
            continue
        if lag >= 0:
            grad[:n-lag] += wi * omh[lag:]
            grad[lag:] -= wi * h[:n-lag]
        else:
            lp = -lag
            grad[lp:] += wi * omh[:n-lp]
            grad[:n-lp] -= wi * h[lp:]
    return grad * dx

def hard_max_fun(h, n):
    return compute_c5(h, n)

    c = get_c5_values(h, n)
    k_max = np.argmax(c)

    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n-lag] += omh[lag:]
        grad[lag:] -= h[:n-lag]
    else:
        lp = -lag
        grad[lp:] += omh[:n-lp]
        grad[:n-lp] -= h[lp:]
    return grad * dx

    """Run SLSQP optimization, returning the best solution found."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]

        fun = lambda x: soft_max_fun(x, n, beta)
        jac = lambda x: soft_max_grad(x, n, beta)

        jac = lambda x: hard_max_grad(x, n)

        return h_new, compute_c5(h_new, n)
