        scores[2*d - 1] = float(np.dot(hf[:n-d], 1.0 - hf[d:]))
        scores[2*d] = float(np.dot(hf[d:], 1.0 - hf[:n-d]))

    """Update scores after flipping position idx to new_val"""
    i = int(idx)
    nv = int(new_val)
    for d in range(1, n):
        pidx = 2*d - 1
        midx = 2*d
        if i - d >= 0:
            hj = int(h[i-d])
            if nv == 1:
                scores[pidx] -= hj
                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += hj
                scores[midx] -= 1.0 - hj

            if nv == 1:
                scores[pidx] += 1.0 - hj
                scores[midx] -= hj
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += hj
    h[i] = nv

        n -= 1
    if t0 is None:
        t0 = 3.0
    if cooling is None:
        cooling = 0.9995
    
    h = make_binary(start_h, n, rng)
    scores = binary_scores(h, n)

    best = cur

            break

        if len(zeros) == 0 or len(ones) == 0:
            break
        i = int(zeros[rng.integers(len(zeros))])
        j = int(ones[rng.integers(len(ones))])
        
        # Apply swap
        binary_flip_update(scores, h, i, 1, n)
        binary_flip_update(scores, h, j, 0, n)
        new_cur = float(np.max(scores))
        delta = new_cur - cur

                best_h = h.copy()
        else:
            # Revert
            binary_flip_update(scores, h, i, 0, n)
            binary_flip_update(scores, h, j, 1, n)

        if T < 0.01:
            T = 0.01  # Keep a minimum temperature for多样性

        j = int(ones[rng.integers(len(ones))])
        binary_flip_update(scores, h, i, 1, n)
        binary_flip_update(scores, h, j, 0, n)
        new_cur = float(np.max(scores))
        delta = new_cur - cur
        if delta <= 0 or (T > 0.01 and rng.random() < math.exp(-delta / T)):
            cur = new_cur
            zeros = np.flatnonzero(h == 0)
            ones = np.flatnonzero(h == 1)
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            binary_flip_update(scores, h, i, 0, n)
            binary_flip_update(scores, h, j, 1, n)
        T *= cooling
    return best_h.astype(float), best * (2.0 / n)

def score_and_grad(h, n, alpha):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, h, mode="full")
    ac = corr[n - 1:n - 1 + n]
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2 * n - 1, dtype=float)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2 * d - 1] = prefix[n - d] - ac[d]
        scores[2 * d] = total - prefix[d] - ac[d]
    m = float(np.max(scores))
    grad = np.zeros(n, dtype=float)
    if alpha <= 1e-14:
        tol = max(1e-8, 1e-7 * max(1.0, m))
        active = np.flatnonzero(scores >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(scores))])
        coef = 1.0 / float(active.size)
        for k in active:
            _add_grad_component(grad, int(k), coef, h, n)
    else:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if math.isfinite(A) and A > 0:
            thresh = 1e-8
            for k in range(len(w)):
                ck = w[k] / A
                if ck >= thresh:
                    _add_grad_component(grad, k, ck, h, n)
    grad *= 2.0 / n
    if alpha <= 1e-14:
        obj = m * (2.0 / n)
    else:
        obj = (m + alpha * math.log(float(np.sum(w)))) * (2.0 / n)
    return obj, grad

def _add_grad_component(grad, idx, coef, h, n):
    if coef == 0.0:
        return
    if idx == 0:
        grad += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
    else:
        d = idx // 2
    m = n - d
    if m <= 0:
        return
    if idx % 2 == 1:
        grad[:m] += coef * (1.0 - h[d:])
        grad[d:] += coef * (-h[:m])
    else:
        grad[:m] += coef * (-h[d:])
        grad[d:] += coef * (1.0 - h[:m])

def constraint_eq(x, *args):
    n = max(1, int(x.size))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)

def constraint_jac(x, *args):
    n = max(1, int(x.size))
    return np.full((1, n), 1.0 / float(n), dtype=float)

def slsqp_obj(x, alpha):
    check_deadline()
    h = np.clip(x, 0.0, 1.0)
    n = h.size
    obj, _ = score_and_grad(h, n, alpha)
    return obj if math.isfinite(obj) else 1e6

def slsqp_grad(x, alpha):
    check_deadline()
    h = np.clip(x, 0.0, 1.0)
    n = h.size
    _, grad = score_and_grad(h, n, alpha)
    return grad if np.all(np.isfinite(grad)) else np.zeros(n)

def run_slsqp(h, n, time_limit):
    h_start = as_feasible(h, n)
    start_t = time.time()
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]
    best_h = h_start.copy()
    best_c = compute_c5(h_start)
    alphas = [max(0.3, 0.003 * n), max(0.1, 0.001 * n), max(0.03, 0.0003 * n), 0.0]
    fb = float(time_limit)
    per_alpha = fb / len(alphas)
    old_dl = _RUN_DEADLINE
    for ai, alpha in enumerate(alphas):
        rem = _RUN_DEADLINE - time.time() if _RUN_DEADLINE is not None else fb
        stage_t = min(per_alpha, rem - 10.0)
        if stage_t < 10.0:
            break
        _RUN_DEADLINE = time.time() + stage_t
        maxiter = max(30, int(200.0 * (80.0 / float(n))))
        maxiter = min(maxiter, 500)
        try:
            res = minimize(
                slsqp_obj, h_start, args=(alpha,),
                jac=slsqp_grad, method="SLSQP",
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0}
            )
            if res is not None and getattr(res, "x", None) is not None:
                h_new = as_feasible(res.x, n)
                c_new = compute_c5(h_new)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new
                    h_start = h_new
                else:
                    h_start = as_feasible(res.x, n)
        except BudgetError:
            pass
        except Exception:
            pass
        _RUN_DEADLINE = old_dl
    return best_h, best_c

def upsample_linear(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent).astype(float)

def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 20.0 if budget_s > 60.0 else max(2.0, 0.1 * budget_s)
    deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = deadline
    rng = np.random.default_rng(seed)

    parent = get_parent()
    if parent is not None:
        n0 = int(parent.size)
        h0 = parent.copy()
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = thue_morse_binary(n0, rng).astype(float)
        as_feasible(h0, n0)
        c0 = compute_c5(h0)

    if n0 % 2 != 0:
        n0 -= 1
        h0 = as_feasible(h0[:n0], n0)
        c0 = compute_c5(h0)

    best_h = h0.copy()
    best_c = c0
    best_n = n0

    def update_best(h, c, n):
        nonlocal best_h, best_c, best_n
        if math.isfinite(c) and c < best_c - 1e-15:
            best_h = h.copy()
            best_c = float(c)
            best_n = int(n)

    # Phase 1: Binary SA with diverse starts at n0
    phase1_time = min(250.0, max(30.0, 0.2 * (deadline - t0)))
    phase1_deadline = time.time() + phase1_time
    starts = [h0]
    starts.append(thue_morse_binary(n0, rng).astype(float))
    starts.append(random_binary(n0, rng).astype(float))
    # Add a "block" start: first half 1s, second half 0s
    block = np.zeros(n0, dtype=float)
    block[:n0//2] = 1.0
    starts.append(block)
    # Add reverse-block
    rblock = np.zeros(n0, dtype=float)
    rblock[n0//2:] = 1.0
    starts.append(rblock)

    sa_time = phase1_time / len(starts)
    for si, sh in enumerate(starts):
        if time.time() > phase1_deadline:
            break
        rem = phase1_deadline - time.time()
        if rem < 5.0:
            break
        tl = min(sa_time, rem)
        try:
            bh, bc = sa_binary(sh, n0, rng, tl, t0=3.0, cooling=0.9997, target_active=0.85)
            if bh is not None and math.isfinite(bc):
                bhf = as_feasible(bh, n0)
                bc2 = compute_c5(bhf)
                update_best(bhf, bc2, n0)
        except Exception:
            pass

    # Phase 2: Continuous SLSQP at n0
    phase2_time = min(300.0, max(50.0, 0.25 * (deadline - t0)))
    if time.time() < deadline - phase2_time - 50.0:
        try:
            h_start = best_h.copy() if best_n == n0 else upsample_linear(best_h, n0)
            h_start = as_feasible(h_start, n0)
            c_start = compute_c5(h_start)
            sh, sc = run_slsqp(h_start, n0, phase2_time)
            if sh is not None and math.isfinite(sc):
                update_best(sh, sc, n0)
        except Exception:
            pass

    # Phase 3: Binary SA on rounded continuous result
    if time.time() < deadline - 30.0:
        phase3_time = min(100.0, max(10.0, 0.08 * (deadline - t0)))
        rem = deadline - time.time() - 30.0
        phase3_time = min(phase3_time, rem)
        if phase3_time > 10.0:
            try:
                rounded = (best_h > 0.5).astype(float)
                rounded = as_feasible(rounded, best_n)
                bh, bc = sa_binary(rounded, best_n, rng, phase3_time, t0=2.0, cooling=0.9999, target_active=0.9)
                if bh is not None and math.isfinite(bc):
                    bhf = as_feasible(bh, best_n)
                    bc2 = compute_c5(bhf)
                    update_best(bhf, bc2, best_n)
            except Exception:
                pass

    # Phase 4: Try higher resolution n2 = 2*n0
    n2 = 2 * n0
    if time.time() < deadline - 80.0 and n2 <= 400:
        phase4_time = min(300.0, max(30.0, 0.25 * (deadline - t0)))
        rem = deadline - time.time() - 60.0
        phase4_time = min(phase4_time, rem)
        if phase4_time > 30.0:
            # Binary SA at n2
            if n2 % 2 == 0:
                up = upsample_linear(best_h, n2)
                up_bin = (up > 0.5).astype(float)
                up_bin = as_feasible(up_bin, n2)
                sa_t = min(120.0, phase4_time * 0.4)
                if sa_t > 10.0:
                    try:
                        bh2, bc2 = sa_binary(up_bin, n2, rng, sa_t, t0=2.5, cooling=0.9998, target_active=0.8)
                        if bh2 is not None and math.isfinite(bc2):
                            bhf2 = as_feasible(bh2, n2)
                            bc2v = compute_c5(bhf2)
                            update_best(bhf2, bc2v, n2)
                    except Exception:
                        pass
            # SLSQP at n2
            slsqp_t = phase4_time - (120.0 if n2 % 2 == 0 else 0)
            slsqp_t = max(30.0, slsqp_t * 0.5)
            rem = deadline - time.time() - 50.0
            slsqp_t = min(slsqp_t, rem)
            if slsqp_t > 20.0:
                try:
                    h2_start = upsample_linear(best_h, n2)
                    h2_start = as_feasible(h2_start, n2)
                    sh2, sc2 = run_slsqp(h2_start, n2, slsqp_t)
                    if sh2 is not None and math.isfinite(sc2):
                        update_best(sh2, sc2, n2)
                except Exception:
                    pass

    # Phase 5: Final polish - SLSQP with alpha=0 on best solution
    if _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 20.0:
        final_time = float(_RUN_DEADLINE - time.time() - 15.0)
        if final_time > 30.0:
            try:
                hn = best_n
                h_polish = as_feasible(best_h, hn)
                c_polish = compute_c5(h_polish)
                # Run SLSQP with alpha=0
                bounds = [(0.0, 1.0)] * hn
                constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]
                old_dl = _RUN_DEADLINE
                _RUN_DEADLINE = time.time() + final_time * 0.9
                maxiter = max(50, int(150.0 * (80.0 / float(hn))))
                res = minimize(
                    slsqp_obj, h_polish, args=(0.0,),
                    jac=slsqp_grad, method="SLSQP",
                    bounds=bounds, constraints=constraints,
                    options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0}
                )
                _RUN_DEADLINE = old_dl
                if res is not None and getattr(res, "x", None) is not None:
                    h_new = as_feasible(res.x, hn)
                    c_new = compute_c5(h_new)
                    update_best(h_new, c_new, hn)
            except Exception:
                pass

    # Final validation
    h_final = as_feasible(best_h, best_n)
    c_final = compute_c5(h_final)
    if not math.isfinite(c_final):
        c_final = c0
        h_final = h0
        best_n = n0
    return h_final, float(c_final), int(best_n)
