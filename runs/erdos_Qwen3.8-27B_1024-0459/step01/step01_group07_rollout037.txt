import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            count = np.sum(free)
            if count == 0:
                break
            h[free] = np.minimum(h[free] + diff / count, 1.0)
        else:
            free = h > 1e-15
            count = np.sum(free)
            if count == 0:
                break
            h[free] = np.maximum(h[free] + diff / count, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, T, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    c = conv / T
    c_max = np.max(c)
    return T * (c_max + np.log(np.sum(np.exp(c - c_max))))

def soft_max_grad(h, T, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    c = conv / T
    c_max = np.max(c)
    e = np.exp(c - c_max)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-16:
            continue
        k = m - (n - 1)
        p1s = max(0, k)
        p1e = min(n, k + n)
        if p1s < p1e:
            grad[p1s:p1e] += wm * dx * (1.0 - h[p1s - k:p1e - k])
        p2s = max(0, -k)
        p2e = min(n, n - k)
        if p2s < p2e:
            grad[p2s:p2e] -= wm * dx * h[p2s + k:p2e + k]
    return grad

    try:
        res = minimize(
            lambda x: soft_max_obj(x, T, n),
            h0.copy(),
            jac=lambda x: soft_max_grad(x, T, n),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq',
                          'fun': lambda x: np.sum(x) - n / 2.0,
                          'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_opt = project_feasible(res.x, n)
    except Exception:
        h_opt = h0
    return h_opt, compute_c5(h_opt, n)

def upsample(h, n_old, n_new):
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=n_old)
    return project_feasible(h_new, n_new)

def projected_grad_descent_sm(h0, n, T, n_iter=300, lr=0.03):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    best_obj = soft_max_obj(h, T, n)
    for it in range(n_iter):
        obj, grad = soft_max_obj(h, T, n), soft_max_grad(h, T, n)
        h_new = project_feasible(h - lr * grad, n)
        obj_new = soft_max_obj(h_new, T, n)
        if obj_new < obj:
            h = h_new
            c5 = compute_c5(h, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            lr *= 0.9

    return best_h, best_c5

def local_search(h0, n, n_trials=200, scale=0.05):

    return best_h, best_c5
