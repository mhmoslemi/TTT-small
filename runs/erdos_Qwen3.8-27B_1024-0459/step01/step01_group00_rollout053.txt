def projected_gradient_step(h, beta, lr, n_iter=500):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_obj = compute_c5(h_cur)
    
    for it in range(n_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, 'full')
        obj, weights = smooth_max_and_weights(corr, beta)
        
        grad = compute_gradient(h_cur, weights, beta)
        
        # Project gradient to tangent space (remove mean)
        grad_tan = grad - np.mean(grad)
        
        # Line search
        accepted = False
        for trial in range(20):
            h_new = h_cur - lr * grad_tan
            h_new = project_constraint(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_obj - 1e-14:
                best_obj = c5_new
                best_h = h_new.copy()
            if c5_new <= compute_c5(h_cur) + 1e-12:
                h_cur = h_new
                accepted = True
                break
            lr_trial = lr * (0.5 ** (trial + 1))
            h_new = h_cur - lr_trial * grad_tan
            h_new = project_constraint(h_new, target)
        
        if not accepted:
            break
        
        # Adaptively adjust learning rate
        if it % 50 == 0 and it > 0:
            # Could adjust here
            pass
    
    return best_h, best_obj

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def local_search(h, n, n_trials=3000, step_size=0.03):
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for _ in range(n_trials):
        h_try = h_cur.copy()
        n_mod = np.random.randint(1, 8)
        idxs = np.random.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += np.random.uniform(-step_size, step_size)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_constraint(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_cur = h_try.copy()
    
    return best_h, best_c5
