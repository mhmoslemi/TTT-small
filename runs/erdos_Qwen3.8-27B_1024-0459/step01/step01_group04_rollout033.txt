import numpy as np
import time
import math
from scipy.optimize import minimize

_RUN_DEADLINE = None

class BudgetError(Exception):
    pass

def check_deadline():
    if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
        raise BudgetError

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv) * (2.0 / n))
    return c if math.isfinite(c) else 1e6

def compute_scores(h, n):
    corr = np.correlate(h, h, mode="full")
    ac = corr[n-1:n-1+n]
    prefix = np.empty(n+1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2*n-1, dtype=float)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2*d-1] = prefix[n-d] - ac[d]
        scores[2*d] = total - prefix[d] - ac[d]
    return scores

def fix_sum(h, target):
    n = h.size
    s = float(h.sum())
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Bisection to find shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        s_mid = float(np.sum(np.clip(h+mid, 0.0, 1.0)))
        if s_mid < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    # Fine-tune
    diff = target - float(h.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            adj = h.copy()
            adj[mask] += diff / cnt
            h = np.clip(adj, 0.0, 1.0)
    return h

    grad = np.zeros(n, dtype=float)
    grad += w[0] * (1.0 - 2.0 * h)
    for d in range(1, n):
        m = n - d
        wp = w[2*d-1]
        wn = w[2*d]
        sw = wp + wn
        if sw < 1e-14:
            continue
        grad[:m] += wp - sw * h[d:d+m]
        grad[d:] += wn - sw * h[:m]
    return grad

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    if n <= 0:
        return np.zeros(0)
    t = np.zeros(n, dtype=float)
    t[0] = 0.0
    for i in range(1, n):
        t[i] = t[i // 2] + (i % 2)
    t = np.where(t % 2 == 0, 0.0, 1.0)
    # Ensure exactly n/2 ones (for even n)
    if n % 2 == 0:
        target = n // 2
        current = int(t.sum())
        if current > target:
            # Remove some 1s
            ones = np.where(t == 1.0)[0]
            for i in range(current - target):
                t[ones[i]] = 0.0
        elif current < target:
            zeros = np.where(t == 0.0)[0]
            for i in range(target - current):
                t[zeros[i]] = 1.0
    return t

def upsample(parent, n):
    n0 = parent.size
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0/n0)
    x = (np.arange(n) + 0.5) * (2.0/n)
    return np.interp(x, x0, parent)

def proj_grad_nesterov(h0, n, iters, deadline, alpha0=2.0, alpha1=0.01):
    """Nesterov accelerated projected gradient for smoothmax objective."""
    h = h0.copy().astype(float)
    h_prev = h0.copy().astype(float)
    best_h = h0.copy()
    best_c = compute_c5(h0)
    dx = 2.0/n
    
    momentum = 0.0
    
    for it in range(iters):
        if time.time() > deadline:
            break
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        
        # Nesterov extrapolation
        beta = (1.0 + math.sqrt(1.0 + 4.0*momentum*momentum)) / 2.0
        # Simplified: use fixed momentum
        t = it + 1
        beta = (1.0 + math.sqrt(1.0 + 4.0*(it+1))) / 2.0  # not quite right
        
        # Actually, let's use a simpler momentum:
        # y = h + (it)/(it+2) * (h - h_prev)
        if it > 0:
            y = h + (float(it)/(it+2.0)) * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
        else:
            y = h
        
        # Compute scores and gradient
        scores = compute_scores(y, n)
        m = float(scores.max())
        
        # Alpha (temperature)
        alpha = alpha0 * (alpha1/alpha0) ** (float(it)/max(1, iters-1))
        
        if alpha < 1e-10:
            tol = max(1e-6, 1e-5*m)
            w = np.zeros(2*n-1)
            active = scores >= m - tol
            w[active] = 1.0
            sw = w.sum()
            if sw > 0:
                w /= sw
        else:
            w = np.exp((scores - m)/alpha)
            sw = w.sum()
            if sw > 1e-30:
                w /= sw
            else:
                w = np.zeros(2*n-1)
                w[np.argmax(scores)] = 1.0
        
        grad = weighted_gradient(w, y, n)
        
        # Step size: Polyak-like
        lr = 2.0 / (float(it) + 1.0 + 3.0)
        # Scale by n to get proper magnitude
        lr *= 0.5
        
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        h_new = fix_sum(h_new, 0.5*n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
    
    return best_h, best_c

def binary_hill(h0, n, deadline, max_ops=50000, rng=None):
    """Binary hill climbing with swap moves."""
    if n % 2 != 0:
        return None, 1e9
    if rng is None:
        rng = np.random.default_rng(123)
    
    # Convert to binary
    target = n // 2
    h = np.zeros(n, dtype=np.int8)
    vals = h0.copy()
    order = np.argsort(-vals)
    h[order[:target]] = 1
    
    # Compute initial scores
    scores = compute_scores(h.astype(float), n)
    cur_max = float(scores.max())
    best = cur_max
    best_h = h.copy()
    dx = 2.0/n
    
    ops = 0
    stall = 0
    stall_limit = 5000
    
    while ops < max_ops and time.time() < deadline:
        ops += 1
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        
        # Propose a swap
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Try to improve: swap a 1 at position j with a 0 at position i
        # such that the max score decreases
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]
        
        # Apply swap
        h[i] = 1
        h[j] = 0
        
        new_scores = compute_scores(h.astype(float), n)
        new_max = float(new_scores.max())
        
        if new_max < cur_max - 1e-12:
            cur_max = new_max
            stall = 0
            if cur_max < best:
                best = cur_max
                best_h = h.copy()
        else:
            # Revert
            h[i] = 0
            h[j] = 1
            stall += 1
        
        if stall > stall_limit:
            break
    
    return best_h.astype(float), best*dx
