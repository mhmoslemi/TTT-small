    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def _grad_single(h, s):
    n = h.size
    g = np.zeros(n, dtype=float)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(s)
        g[lag:] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g


def _smooth_obj(x):
    n = _g_n
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e6
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = min(_g_topk, corr.size)
    idx = np.argpartition(corr, -topk)[-topk:]
    vals = corr[idx]
    m = float(np.max(vals))
    a = _g_alpha
    y = np.clip(a * (vals - m), -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z < 1e-300 or not np.isfinite(z):
        return 1e6
    return (2.0 / n) * (m + np.log(z) / a)


def _smooth_grad(x):
    n = _g_n
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    topk = min(_g_topk, corr.size)
    idx = np.argpartition(corr, -topk)[-topk:]
    vals = corr[idx]
    m = float(np.max(vals))
    a = _g_alpha
    y = np.clip(a * (vals - m), -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z < 1e-300 or not np.isfinite(z):
        return np.zeros(n, dtype=float)
    w = e / z
    center = n - 1
    g = np.zeros(n, dtype=float)
    for i in range(len(idx)):
        if w[i] < 1e-12:
            continue
        s = int(idx[i]) - center
        g += w[i] * _grad_single(h, s)
    return (2.0 / n) * g


def _sum_eq(x):
    return float(np.sum(x)) - _g_target


def _sum_eq_jac(x):
    return np.ones_like(x, dtype=float)


def upsample(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return project_density(p.copy(), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def local_optimize(h, deadline, topk=5, max_steps=400):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    center = n - 1
    stall = 0

    for step_i in range(max_steps):
        if time.perf_counter() >= deadline:
            break

        corr = np.correlate(best, 1.0 - best, mode='full')
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_idx = np.sort(top_idx[-topk:])

        g = np.zeros(n, dtype=float)
        for t in top_idx:
            s = int(t) - center
            g += _grad_single(best, s)
        g /= float(len(top_idx))

        improved = False

        # Gradient step
        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm > 1e-14:
            d /= norm
            for step in (0.03, 0.07, 0.15, 0.25, 0.4):
                cand = project_density(best + step * d, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-12:
                    best, best_c, improved = cand, c, True
                    break

        # Pair search
        if not improved:
            neg_idx = np.argsort(g)[:6]
            pos_idx = np.argsort(-g)[:6]
            found = False
            for ii in neg_idx:
                ii = int(ii)
                for jj in pos_idx:
                    jj = int(jj)
                    if ii == jj:
                        continue
                    cap = min(best[ii], 1.0 - best[jj])
                    if cap < 1e-10:
                        continue
                    for frac in (1.0, 0.7, 0.4, 0.2):
                        d = frac * cap
                        if best[ii] + d > 1.0 or best[jj] - d < 0.0:
                            continue
                        cand = best.copy()
                        cand[ii] += d
                        cand[jj] -= d
                        c = evaluate_erdos(cand)
                        if c < best_c - 1e-12:
                            best, best_c, found = cand, c, True
                            break
                    if found:
                        break
                if found:
                    break

        # Inverse pair search (move mass the other way)
        if not improved:
            for ii in pos_idx:
                ii = int(ii)
                for jj in neg_idx:
                    jj = int(jj)
                    if ii == jj:
                        continue
                    cap = min(best[jj], 1.0 - best[ii])
                    if cap < 1e-10:
                        continue
                    for frac in (1.0, 0.5, 0.25):
                        d = frac * cap
                        if best[jj] - d < 0.0 or best[ii] + d > 1.0:
                            continue
                        cand = best.copy()
                        cand[jj] -= d
                        cand[ii] += d
                        c = evaluate_erdos(cand)
                        if c < best_c - 1e-12:
                            best, best_c, improved = cand, c, True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Block move: shift a short contiguous segment
        if not improved and step_i % 4 == 0 and n >= 6:
            for bstart in range(0, n - 3, 3):
                for blen in (2, 3, 4):
                    if bstart + blen >= n:
                        continue
                    for shift in (1, -1, 2, -2):
                        if bstart + shift < 0 or bstart + blen + shift >= n:
                            continue
                        cand = best.copy()
                        block = best[bstart:bstart + blen].copy()
                        cand[bstart:bstart + blen] = cand[bstart + shift:bstart + blen + shift]
                        cand[bstart + shift:bstart + blen + shift] = block
                        c = evaluate_erdos(cand)
                        if c < best_c - 1e-12:
                            best, best_c, improved = cand, c, True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 15:
                break
        else:
            stall = 0

    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def coordinate_descent(h, deadline, max_passes=60):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(77)

    for _pass in range(max_passes):
        if time.perf_counter() >= deadline:
            break
        idx_perm = rng.permutation(n)
        improved_any = False
        for i in idx_perm:
            if time.perf_counter() >= deadline:
                break
            base = best[i]
            for delta in (0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01):
                new_v = base + delta
                if new_v < 0.0 or new_v > 1.0:
                    continue
                cand = best.copy()
                cand[i] = new_v
                cand = project_density(cand, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best, best_c, improved_any = cand, c, True
                    break
        if not improved_any:
            break

    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def slsqp_phase(h_init, n, deadline, alphas=None):
    if alphas is None:
        if n <= 120:
            alphas = [10.0, 30.0, 80.0, 200.0]
        elif n <= 250:
            alphas = [10.0, 30.0, 100.0]
        else:
            alphas = [10.0, 25.0, 70.0]

    target = 0.5 * n
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": _sum_eq, "jac": _sum_eq_jac},)
    h_cur = h_init.copy()
    c_cur = evaluate_erdos(h_cur)

    maxiter_per = max(40, int(200 * (120.0 / max(n, 1))))
    maxiter_per = min(maxiter_per, 300)

    for alpha in alphas:
        if time.perf_counter() >= deadline:
            break
        _g_alpha = alpha
        _g_topk = min(5, 2 * n - 1)
        try:
            res = minimize(
                _smooth_obj, h_cur,
                method='SLSQP', jac=_smooth_grad,
                bounds=bounds, constraints=constraints,
                options={'maxiter': maxiter_per, 'ftol': 1e-11, 'disp': False}
            )
            if hasattr(res, 'x') and res.x is not None:
                xg = np.asarray(res.x, dtype=float)
                if np.all(np.isfinite(xg)):
                    cand = project_density(xg, target)
                    c = evaluate_erdos(cand)
                    if c < c_cur:
                        h_cur = cand
                        c_cur = c
                    elif alpha < 50:
                        h_cur = cand
        except Exception:
            pass
    return project_density(h_cur, target), evaluate_erdos(h_cur)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = t0 + max(20.0, budget - 10.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        pass

    if parent is not None:
        np_ = int(parent.size)
        pp = project_density(parent, 0.5 * np_)
        cp = evaluate_erdos(pp)
        if cp < best_c:
            best_h, best_c, best_n = pp.copy(), cp, np_

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x), 0, 1)
        best_h = project_density(h, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Resolution plan: focus on 2-3 resolutions with deep optimization
    base_n = best_n if best_n >= 2 else 90
    nlist = sorted(set([
        base_n,
        min(base_n * 2, 360),
        min(base_n * 4, 360),
        180,
        360,
    ]))
    nlist = [n for n in nlist if 2 <= n <= 360]

    # Phase 1: SLSQP at each resolution
    for ni in nlist:
        if time.perf_counter() >= deadline - 80.0:
            break

        starts = []
        if best_h is not None:
            if best_h.size == ni:
                starts.append(best_h.copy())
            else:
                starts.append(upsample(best_h, ni))
        if parent is not None and parent.size != ni:
            starts.append(upsample(parent, ni))
        # Random structured start
        x = (np.arange(ni) + 0.5) * (2.0 / ni)
        h_r = 0.5 + 0.25 * np.sin(2 * np.pi * int(rng.integers(1, 5)) * x + rng.uniform(0, 6.28))
        h_r += 0.15 * np.sin(2 * np.pi * int(rng.integers(2, 8)) * x + rng.uniform(0, 6.28))
        h_r = np.clip(h_r, 0, 1)
        starts.append(project_density(h_r, 0.5 * ni))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if float(np.max(np.abs(s - u))) < 1e-7:
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        for s in uniq:
            if time.perf_counter() >= deadline - 80.0:
                break
            sub_dl = min(deadline - 80.0, time.perf_counter() + 250.0)
            h_out, c_out = slsqp_phase(s, ni, sub_dl)
            if c_out < best_c:
                best_h, best_c, best_n = h_out.copy(), c_out, ni

    # Phase 2: Local optimization at best resolution
    if best_h is not None and time.perf_counter() < deadline - 40.0:
        sub_dl = min(deadline - 40.0, time.perf_counter() + 200.0)
        h_loc, c_loc = local_optimize(best_h, sub_dl, topk=5, max_steps=500)
        if c_loc < best_c:
            best_h, best_c = h_loc, c_loc

    # Phase 3: Try higher resolution if time allows
    if best_h is not None and best_n < 360 and time.perf_counter() < deadline - 120.0:
        n_hi = min(best_n * 2, 360)
        if n_hi > best_n:
            h_hi = upsample(best_h, n_hi)
            sub_dl = min(deadline - 120.0, time.perf_counter() + 300.0)
            h_out, c_out = slsqp_phase(h_hi, n_hi, sub_dl)
            if time.perf_counter() < deadline - 60.0:
                h_loc2, c_loc2 = local_optimize(h_out, min(deadline - 60.0, time.perf_counter() + 150.0), topk=5, max_steps=300)
                if c_loc2 < c_out:
                    h_out, c_out = h_loc2, c_loc2
            if c_out < best_c:
                best_h, best_c, best_n = h_out.copy(), c_out, n_hi

    # Phase 4: Coordinate descent polish
    if best_h is not None and time.perf_counter() < deadline - 10.0:
        sub_dl = min(deadline - 10.0, time.perf_counter() + 150.0)
        h_cd, c_cd = coordinate_descent(best_h, sub_dl, max_passes=80)
        if c_cd < best_c:
            best_h, best_c = h_cd, c_cd

    # Phase 5: Final local polish
    if best_h is not None and time.perf_counter() < deadline - 5.0:
        sub_dl = deadline - 5.0
        h_fin, c_fin = local_optimize(best_h, sub_dl, topk=7, max_steps=300)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin

    if best_h is None:
        fb = project_density(np.full(45, 0.5, dtype=float), 22.5)
        return fb, evaluate_erdos(fb), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
