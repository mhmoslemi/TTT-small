import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    
    def sum_clipped(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    
    if sum_clipped(0.0) == target and np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12):
        return np.clip(h, 0.0, 1.0)
    
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = sum_clipped(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def gradient_of_corr_k(h, k):
    n = len(h)
    d = n - 1 - k
    grad = np.zeros(n)
    i_start1 = max(0, -d)
    i_end1 = min(n, n - d)
    if i_end1 > i_start1:
        grad[i_start1:i_end1] += (1.0 - h[i_start1 + d:i_end1 + d])
    i_start2 = max(0, d)
    i_end2 = min(n, d + n)
    if i_end2 > i_start2:
        grad[i_start2:i_end2] -= h[i_start2 - d:i_end2 - d]
    return grad

def gradient_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    y = corr * dx
    m = np.max(y)
    exps = np.exp(beta * (y - m))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-12:
            continue
        w = weights[k] * dx
        d = n - 1 - k
        i_start1 = max(0, -d)
        i_end1 = min(n, n - d)
        if i_end1 > i_start1:
            grad[i_start1:i_end1] += w * (1.0 - h[i_start1 + d:i_end1 + d])
        i_start2 = max(0, d)
        i_end2 = min(n, d + n)
        if i_end2 > i_start2:
            grad[i_start2:i_end2] -= w * h[i_start2 - d:i_end2 - d]
    return grad, y

def gradient_descent_smooth(h, n, beta, max_iters, time_limit):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    lr = 0.5
    momentum = 0.9
    v = np.zeros(n)
    prev_obj = None
    
    for it in range(max_iters):
        if time.time() - time_limit > time_limit - time_limit:  # placeholder
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        y = corr * (2.0 / n)
        m = np.max(y)
        exps = np.exp(beta * (y - m))
        weights = exps / np.sum(exps)
        
        obj = m + np.log(np.sum(exps)) / beta
        if obj > 1e6:
            break
        
        grad = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] < 1e-12:
                continue
            w = weights[k] * (2.0 / n)
            d = n - 1 - k
            i_start1 = max(0, -d)
            i_end1 = min(n, n - d)
            if i_end1 > i_start1:
                grad[i_start1:i_end1] += w * (1.0 - h[i_start1 + d:i_end1 + d])
            i_start2 = max(0, d)
            i_end2 = min(n, d + n)
            if i_end2 > i_start2:
                grad[i_start2:i_end2] -= w * h[i_start2 - d:i_end2 - d]
        
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Adaptive lr (barzilai-borwein style)
        if prev_obj is not None and abs(prev_obj - obj) > 1e-14:
            step = h_new - h
            s_norm = np.dot(step, step)
            y_dot = prev_obj - obj
            if s_norm > 1e-14 and abs(y_dot) > 1e-14:
                lr_new = s_norm / abs(y_dot)
                lr = 0.5 * lr + 0.5 * np.clip(lr_new, 0.01, 2.0)
        prev_obj = obj
        
        h = h_new
        lr *= 0.999
    
    return best_h, best_c5

def subgradient_polish(h, n, max_iters, time_limit):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    t = 0
    
    for it in range(max_iters):
        t += 1
        if time.time() > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        y = corr * dx
        k_star = np.argmax(y)
        
        grad = gradient_of_corr_k(h, k_star) * dx
        
        # Polyak-style step size
        lr = 0.5 / (np.linalg.norm(grad) + 1e-10)
        lr = min(lr, 0.5)
        
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def coordinate_descent_polish(h, n, max_rounds, time_limit):
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        y = corr * dx
        max_y = np.max(y)
        
        for i in range(n):
            if time.time() > time_limit:
                break
            for delta in [0.1, 0.05, 0.01, -0.1, -0.05, -0.01]:
                h_trial = h.copy()
                h_trial[i] += delta
                if h_trial[i] < 0 or h_trial[i] > 1:
                    continue
                # Adjust another coordinate to maintain sum
                diff = (n / 2.0) - np.sum(h_trial)
                if abs(diff) < 1e-10:
                    pass
                else:
                    # spread the difference to a random other coordinate
                    others = [j for j in range(n) if j != i]
                    j = others[rnd % len(others)]
                    h_trial[j] += diff
                    if h_trial[j] < -1e-10 or h_trial[j] > 1.0 + 1e-10:

def binary_round_optimize(h, n):
    """Try binary rounding with local search"""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Start with threshold rounding
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    
    if n_ones != n_want:
        # Adjust
        if n_ones > n_want:
            idx_ones = np.where(h_bin == 1.0)[0]
            order = np.argsort(h[idx_ones])[:n_ones - n_want]
            h_bin[idx_ones[order]] = 0.0
        elif n_ones < n_want:
            idx_zeros = np.where(h_bin == 0.0)[0]
            order = np.argsort(-h[idx_zeros])[:n_want - n_ones]
            h_bin[idx_zeros[order]] = 1.0
    
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
