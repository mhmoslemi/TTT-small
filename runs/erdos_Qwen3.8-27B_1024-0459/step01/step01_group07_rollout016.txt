from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_grad_single(h, n_points, m):
    """Subgradient of the max at shift m."""
    n = n_points
    grad = np.zeros(n)
    if m >= 0:
        if 0 < m < n:
            grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] += (-h[m:])
    else:

def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = (h < 1.0 - 1e-12)
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = (h > 1e-12)
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Distribute equally among non-boundary points
        interior = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(interior) > 0:
            h[interior] += (target - s) / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
        else:
            h += (target - s) / n
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n_points, maxiter=500, ftol=1e-12):
    n = n_points
    target = n / 2.0
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return compute_c5(x, n)
    
    def jac(x):
        return compute_grad_hard(x, n)
    
    res = minimize(obj, h0.copy(), jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.fun

def lse_anneal_optimize(h0, n_points, temps, maxiter_per_temp=200):
    """Optimize using LSE smoothing with temperature annealing."""
    n = n_points
    h = h0.copy()
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    for temp in temps:
        for it in range(maxiter_per_temp):
            val, grad = compute_lse_val_and_grad(h, n, temp)
            # Project gradient to be orthogonal to the equality constraint
            # (i.e., subtract the mean)
            grad_centered = grad - np.mean(grad)
            # Gradient descent step with adaptive step size
            # Use backtracking line search
            lr = 0.01
            for _ in range(30):
                h_new = h - lr * grad_centered
                h_new = project_feasible(h_new, n)
                val_new = compute_c5(h_new, n)
                if val_new < val:
                    break
                lr *= 0.5
            else:
                break  # no improvement
            h = h_new
            val_hard = compute_c5(h, n)
            if val_hard < best_val:
                best_val = val_hard
                best_h = h.copy()
    
    return best_h, best_val

def grad_descent_projected(h0, n_points, n_iter=500, lr_init=0.02):

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_grad_single(h, n, m):
    n = n
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - h
        grad[:] -= h
    elif m >= 0:
        if m < n:
            grad[m:] += (1.0 - h[:n - m])
        grad[:n - m] += (-h[m:])
    else:
        p = -m
        if p < n:
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] += (-h[:n - p])
    return grad


def compute_grad_hard(h, n_points):
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(conv))
    m = k_max - (n - 1)
    return compute_grad_single(h, n, m) * dx


def compute_lse_val_grad(h, n_points, temp):
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    vals = conv * dx
    c_max = np.max(vals)
    shifted = (vals - c_max) / temp
    shifted = np.clip(shifted, -40.0, 40.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    val = c_max + temp * np.log(sum_exp)
    weights = exp_vals / sum_exp

    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-6:
            continue
        m = idx - (n - 1)
        g = compute_grad_single(h, n, m)
        grad += w * g
    return val, grad * dx


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = (h < 1.0 - 1e-14)
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = (h > 1e-14)
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(interior) > 0:
            h[interior] += (target - s) / np.sum(interior)
        else:
            h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_optimize(h0, n_points, maxiter=400, ftol=1e-12):
    n = n_points
    target = n / 2.0
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return compute_c5(x, n)

    def jac(x):
        return compute_grad_hard(x, n)

    res = minimize(obj, h0.copy(), jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.fun


def lse_anneal(h0, n_points, temps, iters_per_temp=150):
    n = n_points
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    prev_val = compute_lse_val_grad(h, n, temps[0])[0]

    for temp in temps:
        lr = 0.05
        for it in range(iters_per_temp):
            val, grad = compute_lse_val_grad(h, n, temp)
            grad_centered = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_centered)
            if gnorm < 1e-12:
                break

            improved = False
            step = lr
            for _ in range(40):
                h_new = h - step * grad_centered
                h_new = project_feasible(h_new, n)
                new_hard = compute_c5(h_new, n)
                new_lse = compute_lse_val_grad(h_new, n, temp)[0]
                if new_lse < val or new_hard < best_val:
                    h = h_new
                    val = new_lse
                    if new_hard < best_val:
                        best_val = new_hard
                        best_h = h.copy()
                    improved = True
                    break
                step *= 0.5
            if not improved:
                lr *= 0.7
                if lr < 1e-10:
                    break

    return best_h, best_val


def proj_subgrad(h0, n_points, n_iter=400, lr_init=0.03):
    n = n_points
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_val = compute_c5(h, n)
    lr = lr_init

    for it in range(n_iter):
        grad = compute_grad_hard(h, n)
        grad_c = grad - np.mean(grad)
        gn = np.linalg.norm(grad_c)
        if gn < 1e-12:
            break

        step = lr
        improved = False
        for _ in range(40):
            h_new = h - step * grad_c
            h_new = project_feasible(h_new, n)
            v = compute_c5(h_new, n)
            if v < best_val - 1e-15:
                h = h_new
                best_val = v
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.85
            if lr < 1e-10:
                break

    return best_h, best_val


def make_structured(n, rng, kind):
    if kind == 0:
        return np.full(n, 0.5)
    elif kind == 1:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        return h
    elif kind == 2:
        h = np.zeros(n)
        bs = max(2, n // 20)
        for i in range(0, n, 2 * bs):
            h[i:i + bs] = 1.0
        return h
    elif kind == 3:
        x = np.linspace(0, 2, n, endpoint=False)
        return 0.5 + 0.4 * np.sin(2 * np.pi * x)
    elif kind == 4:
        x = np.linspace(0, 2, n, endpoint=False)
        f = rng.randint(1, 5)
        a = rng.uniform(0.2, 0.45)
        return 0.5 + a * np.sin(2 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
    else:
        return rng.uniform(0, 1, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Get parent
    h_base = None
    n_base = 140
    try:
        h_base = np.array(initial_h_values, dtype=float)
        if len(h_base) > 0:
            n_base = len(h_base)
    except Exception:
        h_base = None

    if h_base is None:
        h_base = np.full(n_base, 0.5)

    h_base = project_feasible(h_base, n_base)
    best_h = h_base.copy()
    best_val = compute_c5(best_h, n_base)
    best_n = n_base

    # Phase 1: Quick SLSQP polish of parent
    if time_left() > 250:
        try:
            h_try, v_try = slsqp_optimize(best_h, best_n, maxiter=500, ftol=1e-13)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 2: LSE annealing on base resolution
    if time_left() > 350:
        temps = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
        try:
            h_try, v_try = lse_anneal(best_h, best_n, temps, iters_per_temp=120)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 3: Subgradient descent polish
    if time_left() > 300:
        try:
            h_try, v_try = proj_subgrad(best_h, best_n, n_iter=500, lr_init=0.02)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 4: Random restarts at base resolution
    attempts = 0
    while time_left() > 450 and attempts < 300:
        r = rng.random()
        if r < 0.15:
            h_rand = make_structured(best_n, rng, rng.randint(5))
        elif r < 0.3:
            h_rand = rng.binomial(1, 0.5, best_n).astype(float)
        elif r < 0.5:
            h_rand = best_h + rng.normal(0, 0.35, best_n)
        elif r < 0.7:
            bs = rng.randint(2, max(3, best_n // 15))
            h_rand = np.zeros(best_n)
            for i in range(0, best_n, 2 * bs):
                h_rand[i:i + bs] = 1.0
            h_rand += rng.normal(0, 0.1, best_n)
        else:
            x = np.linspace(0, 2, best_n, endpoint=False)
            f = rng.randint(1, 6)
            a = rng.uniform(0.15, 0.45)
            ph = rng.uniform(0, 2 * np.pi)
            h_rand = 0.5 + a * np.sin(2 * np.pi * f * x / 2.0 + ph) + rng.normal(0, 0.08, best_n)

        h_rand = project_feasible(h_rand, best_n)
        attempts += 1

        try:
            h_try, v_try = slsqp_optimize(h_rand, best_n, maxiter=200, ftol=1e-10)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            try:
                h_try, v_try = proj_subgrad(h_rand, best_n, n_iter=200, lr_init=0.03)
                if v_try < best_val:
                    best_h, best_val = h_try, v_try
            except Exception:
                pass

    # Phase 5: Multi-resolution
    res_list = [100, 140, 180, 240]
    for n_fine in res_list:
        if time_left() < 250:
            break
        if n_fine <= best_n:
            continue

        x_coarse = np.linspace(0, 2, best_n, endpoint=False)
        x_fine = np.linspace(0, 2, n_fine, endpoint=False)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        h_fine = project_feasible(h_fine, n_fine)

        # SLSQP at higher res
        if time_left() > 200:
            try:
                h_try, v_try = slsqp_optimize(h_fine, n_fine, maxiter=300, ftol=1e-11)
                h_try = project_feasible(h_try, n_fine)
                v_try = compute_c5(h_try, n_fine)
                if v_try < best_val:
                    best_h, best_val, best_n = h_try, v_try, n_fine
            except Exception:
                pass

        # LSE anneal at higher res
        if time_left() > 200:
            try:
                temps = [0.04, 0.02, 0.01, 0.005, 0.002]
                h_try, v_try = lse_anneal(h_fine, n_fine, temps, iters_per_temp=80)
                h_try = project_feasible(h_try, n_fine)
                v_try = compute_c5(h_try, n_fine)
                if v_try < best_val:
                    best_h, best_val, best_n = h_try, v_try, n_fine
            except Exception:
                pass

        # Fresh random at higher res
        if time_left() > 150:
            try:
                h_rand = project_feasible(rng.uniform(0, 1, n_fine), n_fine)
                h_try, v_try = slsqp_optimize(h_rand, n_fine, maxiter=250, ftol=1e-10)
                h_try = project_feasible(h_try, n_fine)
                v_try = compute_c5(h_try, n_fine)
                if v_try < best_val:
                    best_h, best_val, best_n = h_try, v_try, n_fine
            except Exception:
                pass

    # Phase 6: Perturbation around best
    for i in range(40):
        if time_left() < 80:
            break
        for scale in [0.015, 0.03, 0.06, 0.1, 0.15]:
            if time_left() < 40:
                break
            r = rng.random()
            if r < 0.3:
                h_pert = best_h + rng.normal(0, scale, best_n)
            elif r < 0.6:
                idx = rng.choice(best_n, size=max(2, best_n // 8), replace=False)
                h_pert = best_h.copy()
                h_pert[idx] = np.clip(h_pert[idx] + rng.normal(0, scale * 3, len(idx)), 0, 1)
            elif r < 0.8:
                idx = rng.choice(best_n, size=max(2, best_n // 12), replace=False)
                h_pert = best_h.copy()
                h_pert[idx] = 1.0 - h_pert[idx]
            else:
                idx = rng.choice(best_n, size=max(3, best_n // 5), replace=False)
                h_pert = best_h.copy()
                h_pert[idx] = rng.uniform(0, 1, len(idx))

            h_pert = project_feasible(h_pert, best_n)
            try:
                h_try, v_try = slsqp_optimize(h_pert, best_n, maxiter=300, ftol=1e-11)
                h_try = project_feasible(h_try, best_n)
                v_try = compute_c5(h_try, best_n)
                if v_try < best_val:
                    best_h, best_val = h_try, v_try
            except Exception:
                try:
                    h_try, v_try = proj_subgrad(h_pert, best_n, n_iter=150, lr_init=0.02)
                    if v_try < best_val:
                        best_h, best_val = h_try, v_try
                except Exception:
                    pass

    # Phase 7: Final aggressive polish
    if time_left() > 20:
        for _ in range(15):
            if time_left() < 8:
                break
            try:
                h_try, v_try = slsqp_optimize(best_h, best_n, maxiter=600, ftol=1e-14)
                h_try = project_feasible(h_try, best_n)
                v_try = compute_c5(h_try, best_n)
                if v_try < best_val:
                    best_h, best_val = h_try, v_try
                else:
                    break
            except Exception:
                break

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
