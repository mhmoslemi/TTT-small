        t = t_start * (t_min / t_start) ** (it / n_iter)
        # Pick random swap
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

                best_c5 = c5
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]

        return h_old.astype(float).copy()

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def round_binary(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin


def smooth_max_fn(h, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return float((m + tau * np.log(np.sum(np.exp((corr - m) / tau)))) * dx)


def penalty_fn(h, tau, dx, mu, n):
    base = smooth_max_fn(h, tau, dx)
    s = h.sum() - n / 2.0
    return base + mu * s * s


def eq_con_fn(h, n):
    return np.array([h.sum() - n / 2.0])


def eq_con_jac(h, n):
    return np.ones((1, n))


def run_slsqp(x0, n, tau, maxiter, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        fun=smooth_max_fn, x0=x0, args=(tau, dx),
        method='SLSQP', bounds=bounds,
        constraints=[{'type': 'eq', 'fun': eq_con_fn, 'args': (n,),
                      'jac': eq_con_jac}],
        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
    )
    return res


def run_lbfgsb_penalty(x0, n, tau, mu, maxiter, ftol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        fun=penalty_fn, x0=x0, args=(tau, dx, mu, n),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
    )
    return res


def sa_search(h_bin, t_start, t_end, n_iter, rng, c5_current):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    c5 = c5_current
    best_h = h.copy()
    best_c5 = c5
    for it in range(n_iter):
        frac = float(it) / float(max(n_iter, 1))
        t = t_start * np.exp(frac * np.log(t_end / t_start))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)

            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5


def greedy_swap(h_bin, max_rounds=20):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nc = compute_c5(h)
                if nc < c5 - 1e-14:
                    c5 = nc
                    improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, c5
