def project_feasible(x, n):
    target = n / 2.0
    # Find lambda such that sum(clip(x - lambda, 0, 1)) = target
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x - (lo+hi)/2, 0, 1)
    # Fix up any numerical issues
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        # Distribute among free variables
        free = (result > 1e-12) & (result < 1 - 1e-12)
        if np.any(free):
            result[free] += diff / np.sum(free)
            result = np.clip(result, 0, 1)
    return result

import numpy as np

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_feasible(x, n):
    target = n / 2.0
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x - lam, 0.0, 1.0)
    diff = target - float(np.sum(result))
    if abs(diff) > 1e-12:
        free = (result > 1e-14) & (result < 1.0 - 1e-14)
        nf = int(np.sum(free))
        if nf > 0:
            result[free] += diff / nf
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_grad(h, n, temp):

    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    c = conv - np.max(conv)

    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[n - 1] = 1.0

    v = np.zeros(n)

    best_val = compute_c5(h, n)
    temp = temp_init
    patience = 0
    it = 0
    
    while time.time() < deadline:

        it += 1

        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
