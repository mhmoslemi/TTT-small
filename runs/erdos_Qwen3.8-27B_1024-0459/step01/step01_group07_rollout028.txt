import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = (h < 1.0 - 1e-12)
            nf = int(np.sum(free))
            if nf == 0:
                break
            step = min(diff / nf, 1.0 / nf)
            h[free] = np.minimum(h[free] + step, 1.0)
        else:
            free = (h > 1e-12)
            nf = int(np.sum(free))
            if nf == 0:
                break
            step = min(-diff / nf, h[free].min() / nf) if nf > 0 else 0
            h[free] = np.maximum(h[free] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        diff = (target - s) / n
        h = np.clip(h + diff, 0.0, 1.0)
    return h


def smooth_max_val_grad(h, n, T):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    values = conv * dx
    max_v = np.max(values)
    log_sum = T * (np.log(np.sum(np.exp((values - max_v) / T))) + max_v / T)
    if T < 1e-10:
        weights = np.zeros(len(values))
        weights[np.argmax(values)] = 1.0
    else:
        exp_vals = np.exp((values - max_v) / T)
        weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for m in range(n):
        w = weights[m] * dx
        if w < 1e-14:
            continue
        grad[:m + 1] += w * (1.0 - h[n - 1 - m:n])
        grad[n - 1 - m:] += w * (-h[:m + 1])
    for m in range(n, 2 * n - 1):
        w = weights[m] * dx
        if w < 1e-14:
            continue
        start1 = m - n + 1
        grad[start1:] += w * (1.0 - h[:n - start1])
        end2 = 2 * n - 2 - m
        if end2 >= 0:
            grad[:end2 + 1] += w * (-h[m - n + 1:])
    return log_sum, grad


def adam_smooth(h, n, T_list, lr, max_iters, time_limit, t_start):
    mmt = np.zeros(n)
    vel = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    best = h.copy()
    best_val = compute_c5(h, n)
    cur = h.copy()
    per_T = max(10, max_iters // len(T_list))
    for Ti in T_list:
        T = T_list[Ti]
        for it in range(per_T):
            if time.time() - t_start > time_limit:
                return best, best_val
            T_use = max(T, 1e-10)
            _, grad = smooth_max_val_grad(cur, n, T_use)
            t += 1
            mmt = beta1 * mmt + (1 - beta1) * grad
            vel = beta2 * vel + (1 - beta2) * grad * grad
            mmt_hat = mmt / (1 - beta1 ** t)
            vel_hat = vel / (1 - beta2 ** t)
            cur_new = cur - lr * mmt_hat / (np.sqrt(vel_hat) + eps)
            cur_new = project_feasible(cur_new, n)
            val = compute_c5(cur_new, n)
            if val < best_val:
                best_val = val
                best = cur_new.copy()
            cur = cur_new
    return best, best_val


def slsqp_polish(h0, n, maxiter=300):
    target = n / 2.0
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(_slsqp_obj, h0.copy(), args=(n,),
                   jac=_slsqp_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x, res.fun


def _slsqp_obj(x, n):
    return compute_c5(x, n)


def _slsqp_jac(x, n):
    target = n / 2.0
    return 0.0


def _slsqp_cons(x, n):
    target = n / 2.0
    return np.sum(x) - target


def _slsqp_grad(x, n):
    target = n / 2.0
    dx = 2.0 / n
    conv = np.correlate(x, 1.0 - x, mode='full')
    k_max = int(np.argmax(conv))
    grad = np.zeros(n)
    if k_max < n:
        s = k_max
        grad[:s + 1] += (1.0 - x[n - 1 - s:n]) * dx
        grad[n - 1 - s:] += (-x[:s + 1]) * dx
    else:
        s = k_max - (n - 1)
        start1 = s
        grad[start1:] += (1.0 - x[:n - start1]) * dx
        end2 = 2 * n - 2 - (n - 1 + s)
        if end2 >= 0:
            grad[:end2 + 1] += (-x[s:]) * dx
    return grad

        h = np.full(n, 0.5) + rng.normal(0, 0.15, n)
        h = np.clip(h, 0, 1)
    return project_feasible(h, n)


def interpolate_to(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Get initial solution from parent
    h0 = None
    n0 = 70
    try:
        if initial_h_values is not None and len(initial_h_values) > 0:
            h0 = np.array(initial_h_values, dtype=float)
            n0 = len(h0)
    except Exception:
        pass
    if h0 is None:
        h0 = np.full(n0, 0.5)
    h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_val = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Polish parent with smooth max Adam at n0
    if time_left() > 100:
        T_sch = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
        h_try, v_try = adam_smooth(h0, n0, T_sch, lr=0.3, max_iters=3000,
                                   time_limit=min(150, time_left() - 10), t_start=t0)
        h_try = project_feasible(h_try, n0)
        v_try = compute_c5(h_try, n0)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    # Phase 2: Multiple restarts at n0
    if time_left() > 200:
        for restart in range(30):
            if time_left() < 80:
                break
            kind = restart % 6
            h_rand = structured_init(n0, rng, kind)
            if kind >= 4:
                h_rand = 0.5 * h_rand + 0.5 * best_h
                h_rand = project_feasible(h_rand, n0)
            T_sch_r = [0.3, 0.1, 0.05, 0.02]
            h_try, v_try = adam_smooth(h_rand, n0, T_sch_r, lr=0.2,
                                       max_iters=1500, time_limit=min(30, time_left() - 60), t_start=time.time())
            h_try = project_feasible(h_try, n0)
            v_try = compute_c5(h_try, n0)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
                best_n = n0
            if time_left() < 100:
                break

    # Phase 3: Higher resolutions
    if time_left() > 250:
        for n_fine in [100, 140, 200]:
            if time_left() < 100:
                break
            h_fine = interpolate_to(best_h, best_n, n_fine)
            T_sch_f = [0.3, 0.15, 0.07, 0.03, 0.01]
            h_try, v_try = adam_smooth(h_fine, n_fine, T_sch_f, lr=0.15,
                                       max_iters=2000, time_limit=min(120, time_left() - 50), t_start=time.time())
            h_try = project_feasible(h_try, n_fine)
            v_try = compute_c5(h_try, n_fine)
            if v_try < best_val:
                best_h, best_val, best_n = h_try, v_try, n_fine
            # Also try a fresh random at this resolution
            if time_left() > 80:
                h_rand = structured_init(n_fine, rng, 1)
                h_try2, v_try2 = adam_smooth(h_rand, n_fine, [0.3, 0.1, 0.03], lr=0.15,
                                             max_iters=1000, time_limit=min(60, time_left() - 30), t_start=time.time())
                h_try2 = project_feasible(h_try2, n_fine)
                v_try2 = compute_c5(h_try2, n_fine)
                if v_try2 < best_val:
                    best_h, best_val, best_n = h_try2, v_try2, n_fine

    # Phase 4: SLSQP polish on best
    if time_left() > 60:
        try:
            h_try, v_try = slsqp_polish(best_h, best_n, maxiter=200)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 5: More Adam polish at best resolution
    if time_left() > 40:
        T_sch_final = [0.05, 0.02, 0.01, 0.005]
        h_try, v_try = adam_smooth(best_h, best_n, T_sch_final, lr=0.05,
                                   max_iters=3000, time_limit=min(40, time_left() - 10), t_start=time.time())
        h_try = project_feasible(h_try, best_n)
        v_try = compute_c5(h_try, best_n)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    # Phase 6: Final SLSQP
    if time_left() > 20:
        try:
            h_try, v_try = slsqp_polish(best_h, best_n, maxiter=500)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    assert np.all(best_h >= -1e-15) and np.all(best_h <= 1 + 1e-15)
    best_h = np.clip(best_h, 0, 1)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
