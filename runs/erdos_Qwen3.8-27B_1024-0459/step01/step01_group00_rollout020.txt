```python
def compute_smooth_max_and_grad(x, beta):
    n = len(x)
    one = 1.0 - x
    corr = np.correlate(x, one, mode='full')
    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    w = exps / np.sum(exps)
    
    sm_val = cmax + (1.0/beta) * np.log(np.sum(exps))
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2*n - 1):
        lag = m - (n - 1)
        wm = w[m]
        if wm < 1e-12:
            continue
        # For this lag, compute contribution to each x[i]
        # contrib[i] = (1-x[i+lag]) * valid1 + (-x[i-lag]) * valid2
        # valid1: 0 <= i+lag <= n-1, i.e., -lag <= i <= n-1-lag
        # valid2: 0 <= i-lag <= n-1, i.e., lag <= i <= n-1+lag
        
        i_min1 = max(0, -lag)
        i_max1 = min(n-1, n-1-lag)
        i_min2 = max(0, lag)
        i_max2 = min(n-1, n-1+lag)
        
        if i_min1 <= i_max1:
            idx1 = np.arange(i_min1, i_max1+1)
            grad[idx1] += wm * (1.0 - x[idx1 + lag])
        
        if i_min2 <= i_max2:
            idx2 = np.arange(i_min2, i_max2+1)
            grad[idx2] += wm * (-x[idx2 - lag])
    
    return sm_val, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final projection
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        # Redistribute proportionally
        pos_mask = (h > 0) & (h < 1)
        if np.any(pos_mask):
            h[pos_mask] += diff / np.count_nonzero(pos_mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def smooth_max_and_grad(x, beta, penalty=0.0, n_target=None):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    exp_sum = np.sum(exps)
    w = exps / exp_sum
    sm_val = cmax + (1.0/beta) * np.log(exp_sum)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        lag = m - (n - 1)
        # Contribution from x[i+l] term (first factor in correlation)
        # ∂/∂x[i] of x[i] * (1-x[i+lag]) = (1-x[i+lag])
        # valid when 0 <= i <= n-1 and 0 <= i+lag <= n-1
        i_min1 = max(0, -lag)
        i_max1 = min(n - 1, n - 1 - lag)
        if i_min1 <= i_max1:
            idx = np.arange(i_min1, i_max1 + 1)
            grad[idx] += wm * (1.0 - x[idx + lag])
        
        # Contribution from (1-x[i+lag]) term (second factor)
        # ∂/∂x[j] of x[j-lag] * (1-x[j]) = -x[j-lag]
        # valid when 0 <= j <= n-1 and 0 <= j-lag <= n-1
        j_min2 = max(0, lag)
        j_max2 = min(n - 1, n - 1 + lag)
        if j_min2 <= j_max2:
            idx = np.arange(j_min2, j_max2 + 1)
            grad[idx] += wm * (-x[idx - lag])
    
    if penalty > 0 and n_target is not None:
        s_viol = np.sum(x) - n_target
        sm_val += penalty * s_viol * s_viol
        grad += 2.0 * penalty * s_viol
    
    return sm_val, grad
```
