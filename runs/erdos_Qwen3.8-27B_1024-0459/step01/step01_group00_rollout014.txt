    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):

        diff = target - s
        if abs(diff) < 1e-14:
            break

        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_out[mask] += diff / cnt
            h_out = np.clip(h_out, 0, 1)
        c5 = compute_c5(h_out)
        if time.time() - start_t < time_limit:
            return h_out, c5
    except Exception:
        pass
    return None, None

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)
