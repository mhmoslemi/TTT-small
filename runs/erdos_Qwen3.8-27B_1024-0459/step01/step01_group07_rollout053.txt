    l = k_star - (n - 1)
    grad = np.zeros(n)
    i_lo = max(0, -l)
    i_hi = min(n, n - l)
    if i_lo < i_hi:
        grad[i_lo:i_hi] += (1.0 - h[i_lo + l:i_hi + l])
    i_lo2 = max(0, l)
    i_hi2 = min(n, n + l)
    if i_lo2 < i_hi2:
        grad[i_lo2:i_hi2] -= h[i_lo2 - l:i_hi2 - l]
    grad *= dx
    return np.max(conv) * dx, grad


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0
            cnt = np.count_nonzero(free)
            if cnt == 0:
                break
            h[free] = np.minimum(h[free] + diff / cnt, 1.0)
        else:
            free = h > 0.0
            cnt = np.count_nonzero(free)
            if cnt == 0:
                break
            h[free] = np.maximum(h[free] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def adam_smooth_max(h0, n, max_iters, deadline, temp_start, temp_end, lr_base, beta1=0.85, beta2=0.999):
    h = h0.copy()
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8
    t = 0
    no_improve = 0
    for i in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        frac = i / max(1, max_iters - 1)
        temp = temp_start * (temp_end / temp_start) ** frac
        obj, grad = compute_smooth_max_grad(h, n, temp)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr_base / (1.0 + 0.003 * t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        true_obj = compute_c5(h, n)
        if true_obj < best_obj - 1e-10:
            best_obj = true_obj
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break
    return best_h, best_obj


def subgrad_optimize(h0, n, max_iters, deadline, lr_base=0.04):
    h = h0.copy()
    best_h = h.copy()
    best_obj = compute_c5(h, n)
    for i in range(max_iters):
        if time.time() > deadline:
            break
        val, grad = subgradient_c5(h, n)
        lr = lr_base / (1.0 + 0.005 * i)
        h = h - lr * grad
        h = project_feasible(h, n)
        obj = compute_c5(h, n)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    return best_h, best_obj


def slsqp_polish(h0, n, maxiter=500):
    target = n / 2.0
    n_points_local = n

    def obj_func(x):
        return compute_c5(x, n_points_local)

    def obj_grad(x):
        _, g = subgradient_c5(x, n_points_local)
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n_points_local)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_func, h0.copy(), jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x, res.fun


def interpolate_h(h_source, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_source)


def binary_round(h, n):
    target = int(n / 2)
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Initialize
    h_init = None
    n_init = 70
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        if n_init == 0:
            h_init = None
    except Exception:
        pass
    if h_init is None:
        h_init = np.full(n_init, 0.5)
    h_init = project_feasible(h_init, n_init)

    best_h = h_init.copy()
    best_val = compute_c5(best_h, n_init)
    best_n = n_init

    # Phase 1: Adam smooth max at initial resolution (long run)
    deadline = t0 + remaining() * 0.25
    h_try, v_try = adam_smooth_max(h_init, n_init, 8000, deadline, 0.15, 0.015, 0.07)
    h_try = project_feasible(h_try, n_init)
    v_try = compute_c5(h_try, n_init)
    if v_try < best_val:
        best_h, best_val, best_n = h_try, v_try, n_init

    # Phase 2: Subgradient polish
    if remaining() > 100:
        deadline = t0 + remaining() * 0.1
        h_try, v_try = subgrad_optimize(best_h, best_n, 5000, deadline, 0.03)
        h_try = project_feasible(h_try, best_n)
        v_try = compute_c5(h_try, best_n)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    # Phase 3: SLSQP polish
    if remaining() > 80:
        try:
            h_try, v_try = slsqp_polish(best_h, best_n, maxiter=300)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 4: Multi-resolution search
    resolutions = [100, 140, 200, 280, 350]
    for res in resolutions:
        if remaining() < 120:
            break
        if res == best_n:
            continue
        # Interpolate
        h_interp = interpolate_h(best_h, best_n, res)
        h_interp = project_feasible(h_interp, res)
        # Randomized variant
        h_rand = h_interp + rng.normal(0, 0.08, res)
        h_rand = project_feasible(h_rand, res)

        deadline1 = t0 + min(remaining() * 0.35, 200)
        if deadline1 > t0 + 20:
            h_try1, v_try1 = adam_smooth_max(h_interp, res, 5000, deadline1, 0.12, 0.01, 0.06)
            h_try1 = project_feasible(h_try1, res)
            v_try1 = compute_c5(h_try1, res)
            if v_try1 < best_val:
                best_h, best_val, best_n = h_try1, v_try1, res

        if remaining() < 80:
            break
        deadline2 = t0 + min(remaining() * 0.3, 150)
        if deadline2 > t0 + 20:
            h_try2, v_try2 = adam_smooth_max(h_rand, res, 4000, deadline2, 0.12, 0.015, 0.06)
            h_try2 = project_feasible(h_try2, res)
            v_try2 = compute_c5(h_try2, res)
            if v_try2 < best_val:
                best_h, best_val, best_n = h_try2, v_try2, res

        # Subgradient at this resolution
        if remaining() > 60 and best_n == res:
            deadline3 = t0 + min(remaining() * 0.15, 60)
            h_try3, v_try3 = subgrad_optimize(best_h, best_n, 3000, deadline3, 0.02)
            h_try3 = project_feasible(h_try3, best_n)
            v_try3 = compute_c5(h_try3, best_n)
            if v_try3 < best_val:
                best_h, best_val = h_try3, v_try3

    # Phase 5: Random restarts at best resolution
    scales = [0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5]
    for r in range(len(scales) * 2):
        if remaining() < 40:
            break
        scale = scales[r % len(scales)]
        if rng.random() < 0.4:
            h_pert = best_h + rng.normal(0, scale, best_n)
        else:
            h_pert = best_h.copy()
            n_pert = max(3, best_n // 4)
            idx = rng.choice(best_n, size=n_pert, replace=False)
            h_pert[idx] = np.clip(h_pert[idx] + rng.normal(0, scale * 2, n_pert), 0, 1)
        h_pert = project_feasible(h_pert, best_n)
        deadline = t0 + min(remaining() * 0.2, 100)
        if deadline > t0 + 15:
            h_try, v_try = adam_smooth_max(h_pert, best_n, 2500, deadline, 0.1, 0.02, 0.05)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try

    # Phase 6: Binary push and re-optimize
    if remaining() > 60:
        h_bin = binary_round(best_h, best_n)
        h_bin = project_feasible(h_bin + rng.normal(0, 0.02, best_n), best_n)
        deadline = t0 + min(remaining() * 0.3, 120)
        h_try, v_try = adam_smooth_max(h_bin, best_n, 3000, deadline, 0.08, 0.01, 0.05)
        h_try = project_feasible(h_try, best_n)
        v_try = compute_c5(h_try, best_n)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    # Phase 7: Final subgradient + SLSQP polish
    if remaining() > 30:
        deadline = t0 + min(remaining() * 0.4, 80)
        h_try, v_try = subgrad_optimize(best_h, best_n, 4000, deadline, 0.02)
        h_try = project_feasible(h_try, best_n)
        v_try = compute_c5(h_try, best_n)
        if v_try < best_val:
            best_h, best_val = h_try, v_try

    if remaining() > 20:
        try:
            h_try, v_try = slsqp_polish(best_h, best_n, maxiter=200)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
