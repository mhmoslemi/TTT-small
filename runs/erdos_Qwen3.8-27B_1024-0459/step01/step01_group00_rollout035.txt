C[j] = sum_{k=j}^{j+n-1} w[k]

Now, A[j] with w-index (j+n-1-i):
- j=0: w[n-1-i] for i=0..n-1 → w[n-1], w[n-2], ..., w[0]
- j=n-1: w[2n-2-i] for i=0..n-1 → w[2n-2], w[2n-3], ..., w[n-1]

B[j] with w-index (n-1-j+i):
- j=0: w[n-1+i] for i=0..n-1 → w[n-1], w[n], ..., w[2n-2]
- j=n-1: w[i] for i=0..n-1 → w[0], w[1], ..., w[n-1]

These are both valid w indices (0 to 2n-2). ✓

Now for vectorization:
A is the matrix product: A = W1 @ h where W1[j,i] = w[j+n-1-i]
B = W2 @ h where W2[j,i] = w[n-1-j+i]

W1 and W2 are n×n matrices. For n=160, that's a 160×160 matrix multiply, very fast.

C[j] = cumw[j+n] - cumw[j] where cumw[m] = sum_{k=0}^{m-1} w[k], cumw[0]=0.
Actually: C[j] = sum_{k=j}^{j+n-1} w[k] = cumw[j+n] - cumw[j] where cumw is cumsum with an extra 0 at the start.

```python
def compute_smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')  # length 2n-1, indices 0..2n-2
    x = corr * dx
    m = np.max(x)
    exp_x = np.exp(x - m)
    w = exp_x / np.sum(exp_x)  # length 2n-1
    
    i = np.arange(n)
    j = np.arange(n)
    
    # A[j] = sum_i h[i] * w[j+n-1-i]
    # Matrix W1[j,i] = w[j+n-1-i]
    W1_idx = j[:, None] + n - 1 - i[None, :]  # shape (n,n), values in [0, 2n-2]
    A = (w[W1_idx] * h).sum(axis=0) if False else (w[W1_idx] @ h)  # (n,1) → need to be careful
    
    # Actually: A[j] = sum_i W1_idx_val * h[i] = row j of (w[W1_idx] @ h)
    # w[W1_idx] has shape (n,n), h has shape (n,)
    # (w[W1_idx] @ h) gives shape (n,)
    A = w[W1_idx] @ h  # shape (n,)
    
    W2_idx = n - 1 - j[:, None] + i[None, :]  # shape (n,n), values in [0, 2n-2]
    B = w[W2_idx] @ h  # shape (n,)
    
    cumw = np.concatenate([[0.0], np.cumsum(w)])  # length 2n
    C = cumw[j + n] - cumw[j]  # shape (n,)
    
    grad = (C - A - B) * dx
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = corr * dx
    m = np.max(x)
    return m + np.log(np.sum(np.exp(x - m))) / beta

def compute_smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = corr * dx
    m = np.max(x)
    exp_x = np.exp(x - m)
    w = exp_x / np.sum(exp_x)
    
    i = np.arange(n)
    j = np.arange(n)
    
    W1_idx = j[:, None] + n - 1 - i[None, :]
    A = w[W1_idx] @ h
    
    W2_idx = n - 1 - j[:, None] + i[None, :]
    B = w[W2_idx] @ h
    
    cumw = np.concatenate([[0.0], np.cumsum(w)])
    C = cumw[j + n] - cumw[j]
    
    grad = (C - A - B) * dx
    return grad

def project_onto_simplex(h_new, target):
    """Project h_new onto {h: 0<=h<=1, sum(h)=target} via bisection on offset."""
    # Find t such that sum(clip(h_new - t, 0, 1)) = target
    lo = h_new.min() - 1.0
    hi = h_new.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h_new - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(h_new - t, 0.0, 1.0)

def projected_gradient(h, n, beta, lr=0.01, n_iter=500):
    target = n / 2.0
    best_h = h.copy()
    best_obj = smooth_objective(h, beta)
    
    for it in range(n_iter):
        grad = compute_smooth_grad(h, beta)
        h_new = h - lr * grad
        h_proj = project_onto_simplex(h_new, target)
        
        obj = smooth_objective(h_proj, beta)
        if obj < best_obj:
            best_obj = obj
            best_h = h_proj.copy()
        
        h = h_proj
        # Adaptive learning rate
        if it > 50:
            lr *= 0.999
    
    return best_h, best_obj

def binary_local_search(h_init, n, n_trials=50000, T_start=0.1, T_end=0.001):
    """Simulated annealing on binary solutions with fixed weight."""
    n = len(h_init)
    target = n / 2.0
    
    # Round to binary
    h = h_init.copy()
    # Sort and threshold
    sorted_h, sorted_idx = np.sort(h), np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[target:]] = 1.0  # top n/2 values become 1
    # Wait, target = n/2, so indices target..n-1 become 1 (that's n/2 values)
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    for trial in range(n_trials):
        T = T_start * (T_end / T_start) ** (trial / n_trials)
        
        # Flip two bits: one 1→0 and one 0→1
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)
        
        if c5_new < best_c5 or (np.random.random() < np.exp(-(c5_new - best_c5) / T)):
            best_c5 = min(best_c5, c5_new)
            best_h = h_bin.copy()
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Load parent
    global initial_h_values
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    
    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Smooth gradient optimization on parent resolution
    if remaining() > 100:
        n = best_n
        h_work = make_feasible(best_h.copy(), n)
        
        # Anneal beta from low to high
        for beta in [5.0, 10.0, 20.0, 40.0, 80.0, 160.0]:
            if remaining() < 30:
                break
            h_work, obj = projected_gradient(h_work, n, beta, lr=0.5, n_iter=300)
            c5 = compute_c5(h_work)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_work.copy()
                best_n = n
    
    # Phase 2: Binary local search
    if remaining() > 200:
        h_bin = best_h.copy()
        n = best_n
        # Round
        idx = np.argsort(h_bin)
        h_bin[idx[:n//2]] = 0.0
        h_bin[idx[n//2:]] = 1.0
        h_bin = make_feasible(h_bin, n)
        
        h_bin_opt, c5_bin = binary_local_search(h_bin, n, n_trials=30000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin_opt.copy()
            best_n = n
    
    # Phase 3: Try higher resolution
    for n_try in [200, 240, 320]:
        if remaining() < 150:
            break
        
        n = n_try
        h_interp = interpolate_to_n(best_h, n)
        h_interp = make_feasible(h_interp, n)
        
        for beta in [20.0, 80.0]:
            if remaining() < 30:
                break
            h_interp, obj = projected_gradient(h_interp, n, beta, lr=0.3, n_iter=200)
            c5 = compute_c5(h_interp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_interp.copy()
                best_n = n
    
    # Phase 4: Final polish - high beta gradient
    if remaining() > 30:
        n = best_n
        h_work = make_feasible(best_h.copy(), n)
        for beta in [100.0, 200.0, 500.0]:
            if remaining() < 15:
                break
            h_work, obj = projected_gradient(h_work, n, beta, lr=0.1, n_iter=200)
            c5 = compute_c5(h_work)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_work.copy()
    
    # Final check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
```

```python
def optimize_free_variables(h_fixed, free_indices, n):
    """Optimize free variables while keeping others fixed."""
    n = len(h_fixed)
    dx = 2.0 / n
    target = n / 2.0
    
    # Fixed sum constraint: sum(free_vars) = target - sum(fixed)
    fixed_sum = np.sum(h_fixed[np.isin(np.arange(n), free_indices, inverse=True)])
    free_target = target - fixed_sum
    
    # Objective: maximize... no, minimize max_k corr(k)
    # corr(k) = sum_i h[i] - sum_i h[i]*h[i+s]
    # This is quadratic in the free variables
    
    from scipy.optimize import minimize
    
    def obj(free_vals):
        h = h_fixed.copy()
        h[free_indices] = free_vals
        return compute_c5(h)  # or smooth version
    
    # Use SLSQP or Nelder-Mead for the small problem
    x0 = h_fixed[free_indices].copy()
    bounds = [(0, 1)] * len(free_indices)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - free_target}]
    
    res = minimize(obj, x0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': 1000, 'ftol': 1e-15})
    
    h_out = h_fixed.copy()
    h_out[free_indices] = res.x
    return h_out, compute_c5(h_out)
```
