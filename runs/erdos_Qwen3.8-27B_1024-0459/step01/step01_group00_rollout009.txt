                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        T *= cool
    return best_h, best_c5

def generate_diverse_inits(n, rng):
    inits = []
    # All 0.5
    inits.append(np.full(n, 0.5))
    # Block patterns
    if n % 4 == 0:
        inits.append(np.tile([1, 1, 0, 0], n // 4).astype(float))
        inits.append(np.tile([1, 0, 0, 1], n // 4).astype(float))
        inits.append(np.tile([1, 0, 1, 0], n // 4).astype(float))
    # Quantile-based
    qs = np.linspace(0, 1, n)
    inits.append((qs * 2 < 1).astype(float))
    # Sinusoidal
    x = np.linspace(0, 2, n, endpoint=False)
    inits.append(0.5 + 0.4 * np.sin(np.pi * x)).clip(0, 1)
    inits.append(0.5 + 0.4 * np.sin(2 * np.pi * x)).clip(0, 1)
    # Thue-Morse
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    inits.append(tm)
    # Van der Corput
    vdc = np.zeros(n)
    for i in range(n):
        v = 0.0
        f = 1.0
        m = i + 1
        while m > 0:
            v += (m % 2) * f
            m //= 2
            f *= 0.5
        vdc[i] = v
    inits.append(vdc)
    # Random
    for _ in range(3):
        inits.append(rng.rand(n))
    return [make_feasible(hi, n) for hi in inits]
