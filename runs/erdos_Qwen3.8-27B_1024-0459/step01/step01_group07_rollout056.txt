```python
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_c5_all(h, n_points):
    """Returns all correlation values * dx"""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv * dx


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        # Distribute remaining among free variables
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += (target - s) / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

                   temp_init=1.0, temp_decay=0.999, time_limit=None, t_start=None):
    """
    Adam with projection for the smooth max objective.
    Uses log-sum-exp smoothing with adaptive temperature.
    """
    n = n_points
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    temp = temp_init
    
    conv_size = 2 * n - 1
    
    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        t += 1
        
        # Compute all f_s values
        f_all = compute_c5_all(h, n)
        
        # Soft max (log-sum-exp) for smoothness
        max_f = np.max(f_all)
        exp_f = np.exp((f_all - max_f) / temp)
        # Softmax weights
        w = exp_f / np.sum(exp_f)
        
        # Gradient of smooth max
        # For each shift s (index in conv array), compute df_s/dh
        # conv index m corresponds to shift s = m - (n-1)
        # f_s = sum_{i valid} h[i]*(1-h[i+s])*dx
        # df_s/dh[j] = (1-h[j+s])*dx [if valid] - h[j-s]*dx [if valid]

        for m_idx in range(conv_size):
            if w[m_idx] < 1e-10:
                continue
            s = m_idx - (n - 1)
            # Contribution from h[j] as the "first" variable: (1-h[j+s])*dx for valid j
            if s >= 0:
                # j+s < n => j < n-s
                j_max = n - s

                # Contribution from h[j] as the "second" variable: -h[j-s]*dx for valid j

                grad[:j_max] += w[m_idx] * (1.0 - h[p:p+j_max]) * dx
                # j-p >= 0 => j >= p
                grad[p:] += w[m_idx] * (-h[:j_max]) * dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_new, n)
        
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
        temp *= temp_decay
    
    return best_h, best_val
```

```python
    """Compute gradient of log-sum-exp smoothed objective.
    w_full: array of shape (2n-1,) with softmax weights for each shift
    """
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w_full[k]
        if wk < 1e-12:

        s = k - (n - 1)

            grad[s:] += wk * (1.0 - h[:n-s]) * dx

        else:
            p = -s
            # (1-h[j+p]) for j in [0, n-1-p]
            grad[:n-p] += wk * (1.0 - h[p:]) * dx
            # -h[j-p] for j in [p, n-1]
            grad[p:] += wk * (-h[:n-p]) * dx
    return grad
```

This is O(n²) per call but with small constants. For n=70, 2n-1=139 iterations with vectorized numpy ops of length ≤70. Should be very fast.

```python
def adam_smooth_optimize(h0, n, n_iter=2000, lr=0.01, beta1=0.9, beta2=0.999,
                         temp_init=0.5, temp_final=0.05, time_limit=None, t_start=None):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    dx = 2.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        t = it + 1
        temp = temp_init * (temp_final / temp_init) ** (it / max(n_iter-1, 1))
        
        # Compute smooth gradient
        grad = smooth_grad(h, n, temp, dx)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_new, n)
        
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_val
```
