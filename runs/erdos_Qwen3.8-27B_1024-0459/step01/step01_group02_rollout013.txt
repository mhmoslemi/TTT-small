_G_BETA = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(8):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    max_sh = np.max(shifted)
    if max_sh > 400:
        log_sum = max_sh + np.log(np.sum(np.exp(shifted - max_sh)))
    elif max_sh < -400:
        min_sh = np.min(shifted)
        log_sum = min_sh + np.log(np.sum(np.exp(shifted - min_sh)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    return (m + log_sum / beta) * dx
