This is a 1D convex minimization of a piecewise-linear function. The minimum is at a point where two of the linear functions cross below all others, or at t=0 or t=1.

Algorithm to find the minimum:
1. Compute all (a_k, g_k) pairs
2. The function f(t) = max_k (a_k + g_k*t)
3. f(0) = max(a_k), f(1) = max(a_k + g_k)
4. The minimum is at some t* where the "active" set changes
5. We can find it by: the subgradient of f at t is the set {g_k : k is active at t}
6. f is minimized when 0 is in the subgradient, i.e., there exist active k with g_k < 0 and g_k > 0 (or an active g_k = 0)

For a practical implementation:
- Evaluate f on a fine grid (e.g., 200 points in [0,1])
- Find the minimum grid point
- Optionally refine with a bisection on the subgradient

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            h = np.clip(h + diff/n, 0, 1)
            continue
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-15:
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0, 1)
    return h

    return np.correlate(h, 1.0 - h, mode='full')

def coord_descent_step(h, i, n, n_grid=200):
    """Optimize h[i] given all other coordinates fixed."""
    # Compute slopes g_k for each lag k
    # corr = np.correlate(h, 1-h, 'full') has length 2n-1
    # corr[k + n - 1] corresponds to lag k (from -(n-1) to n-1)

    # For lag k (>0): g = (1-h[i-k])*(i>=k) - h[i+k]*(i+k<n)
    # For lag k (=0): g = 1 - 2*h[i]
    # For lag k (<0): g = (1-h[i-k])*(i>=k) - h[i+k]*(i+k<n)  [same formula, k negative]
    
    # Actually let me just compute it directly.
    # corr[j] for j=0..2n-2 corresponds to lag k = j-(n-1)
    # slope_k(i) = (1-h[i-k]) if 0<=i-k<n else 0, minus h[i+k] if 0<=i+k<n else 0
    
    h_i = h[i]
    # For each k from -(n-1) to n-1:
    # slope = (1-h[i-k]) * [0<=i-k<n] - h[i+k] * [0<=i+k<n]
    
    k_vals = np.arange(-(n-1), n)
    # First term: (1-h[i-k]) valid when 0<=i-k<n
    term1_mask = (i - k_vals >= 0) & (i - k_vals < n)
    term1 = np.where(term1_mask, 1.0 - h[i - k_vals[term1_mask]], 0.0)
    
    # Second term: -h[i+k] valid when 0<=i+k<n
    term2_mask = (i + k_vals >= 0) & (i + k_vals < n)
    term2 = -np.where(term2_mask, h[i + k_vals[term2_mask]], 0.0)
    
    slopes = term1 + term2
    
    # Current correlation values
    corr = np.correlate(h, 1.0 - h, mode='full')
    # a_k = corr - slope * h_i (so that corr_k = a_k + slope_k * h_i)
    a_vals = corr - slopes * h_i
    
    # Find min of max_k (a_k + slope_k * t) for t in [0, 1]
    grid = np.linspace(0, 1, n_grid)
    # For each t, compute max_k (a_k + slope_k * t)
    # This is a vectorized operation: (n_grid, 2n-1) matrix, take max along axis 1
    vals = a_vals[None, :] + slopes[None, :] * grid[:, None]
    f_vals = np.max(vals, axis=1)
    
    best_idx = np.argmin(f_vals)
    best_t = grid[best_idx]
    
    # Refine around best
    t_min = grid[max(0, best_idx-1)]
    t_max = grid[min(n_grid-1, best_idx+1)]
    grid_fine = np.linspace(t_min, t_max, 50)
    vals_f = a_vals[None, :] + slopes[None, :] * grid_fine[:, None]
    f_vals_f = np.max(vals_f, axis=1)
    best_idx_f = np.argmin(f_vals_f)
    best_t = grid_fine[best_idx_f]
    
    return best_t

def coord_descent(h, n, n_sweeps=100, n_grid=200):
    h = h.copy()
    for sweep in range(n_sweeps):
        # Random order for diversity
        order = np.random.permutation(n)
        for i in order:
            new_t = coord_descent_step(h, i, n, n_grid)
            h[i] = new_t
        # Project to sum constraint
        h = make_feasible(h, n)
    return h

def coord_descent_fast(h, n, n_sweeps=500, n_grid=500):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * 2.0 / n
    n_corr = len(corr)  # 2n-1
    
    for sweep in range(n_sweeps):
        order = np.random.permutation(n)
        improved = 0
        for i in order:
            h_i = h[i]
            # Compute slopes
            k_vals = np.arange(-(n-1), n)
            term1_mask = (i - k_vals >= 0) & (i - k_vals < n)
            term1 = np.zeros(n_corr)
            term1[term1_mask] = 1.0 - h[i - k_vals[term1_mask]]
            term2_mask = (i + k_vals >= 0) & (i + k_vals < n)
            term2 = np.zeros(n_corr)
            term2[term2_mask] = -h[i + k_vals[term2_mask]]
            slopes = term1 + term2
            
            a_vals = corr - slopes * h_i
            
            # Minimize max_k(a_k + slopes_k * t) over t in [0,1]
            # Fine grid
            grid = np.linspace(0, 1, n_grid)
            vals = a_vals + slopes * grid[0]  # just t=0 first
            # Actually vectorize:
            # (n_grid, n_corr) matrix
            f_at_grid = np.max(a_vals[None,:] + slopes[None,:] * grid[:,None], axis=1)
            best_j = np.argmin(f_at_grid)
            new_t = grid[best_j]
            
            if new_t != h_i:
                h[i] = new_t
                corr += slopes * (new_t - h_i)
                improved += 1
        
        # Project
        h = make_feasible(h, n)
        # Recompute corr after projection
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr_new) * 2.0 / n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        corr = corr_new
        
        if improved == 0:
            break
    
    return best_h, best_c5

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            h = np.clip(h + diff / n, 0, 1)
        else:
            h[mask] += diff / cnt
            h = np.clip(h, 0, 1)

        else:
            idx = int(np.argmax(h))
