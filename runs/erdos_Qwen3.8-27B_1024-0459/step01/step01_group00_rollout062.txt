def sg_step(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    ks = int(np.argmax(corr))
    s = ks - (n - 1)
    g = np.zeros(n)
    js = np.arange(n)
    for j in range(n):
        v = 0.0
        jp = j + s
        if 0 <= jp < n:
            v += 1.0 - h[jp]
        if s != 0:
            jm = j - s
            if 0 <= jm < n:
                v -= h[jm]
        g[j] = v
    g -= g.mean()
    hn = h - alpha * g
    hn = np.clip(hn, 0.0, 1.0)

                    best_c5, best_h, best_n = cb2, hb2, best_n

    # --- Phase 3: Diverse initializations at best_n ---
    if rem() > 60:
        inits = gen_structured(best_n, rng)
        inits.append(fix_sum(interp_to(best_h, best_n), best_n))
        for h0 in inits[:4]:
            if rem() < 20:
                break
            h0 = fix_sum(h0, best_n)
            hb, cb = run_lbfgs(h0, best_n, min(rem(), 80))
            if cb < best_c5:
                best_c5, best_h, best_n = cb, hb, best_n

    # --- Phase 4: Subgradient optimization ---
    if rem() > 40:
        hb, cb = sg_opt(best_h, best_n, min(rem(), 150), a0=0.03)
        if cb < best_c5:
            best_c5, best_h, best_n = cb, hb, best_n

    # --- Phase 5: Binary annealing ---
    if rem() > 40:
        hbin = fix_sum((best_h > 0.5).astype(float), best_n)
        hb, cb = bin_swap(hbin, best_n, min(rem(), 150))
        if cb < best_c5:
            best_c5, best_h, best_n = cb, hb, best_n

    # --- Phase 6: Local perturbation polish ---
    if rem() > 20:
        hb, cb = local_pol(best_h, best_n, min(rem(), 100), ntr=8000, st=0.01)
        if cb < best_c5:
            best_c5, best_h, best_n = cb, hb, best_n

    # --- Phase 7: Final SLSQP + L-BFGS polish ---
    if rem() > 20:
        hb, cb = slsq_opt(best_h, best_n, min(rem(), 80))
        if cb < best_c5:
            best_c5, best_h, best_n = cb, hb, best_n
    if rem() > 15:
        hb, cb = run_lbfgs(best_h, best_n, min(rem(), 60))
        if cb < best_c5:
            best_c5, best_h, best_n = cb, hb, best_n

    # --- Phase 8: Tiny local search ---
    if rem() > 10:
        hb, cb = local_pol(best_h, best_n, min(rem(), 40), ntr=5000, st=0.005)
        if cb < best_c5:
            best_c5, best_h, best_n = cb, hb, best_n

    # Final feasibility
    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
