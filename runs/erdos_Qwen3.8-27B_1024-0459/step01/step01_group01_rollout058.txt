    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    c_max = np.max(corr)
    exps = np.exp((corr - c_max) / tau)
    weights = exps / np.sum(exps)
    
    # weights[s] corresponds to shift s = (n-1) - (len(corr)-1)/2 + ... 
    # Actually, corr has length 2n-1, indexed 0 to 2n-2
    # corr[k] corresponds to shift s = k - (n-1)
    
    grad = np.zeros(n)
    for i in range(n):
        # Part 1: -h[i+s] for valid s where i+s in [0, n-1]
        # s ranges from -(n-1) to (n-1), i+s in [0, n-1] means s in [-i, n-1-i]
        # s index in corr: k = s + (n-1), so k in [n-1-i, 2n-2-i]
        s_min = max(-(n-1), -i)
        s_max = min(n-1, n-1-i)
        for s in range(s_min, s_max + 1):
            k = s + (n-1)
            grad[i] -= weights[k] * h[i+s]
        
        # Part 2: (1-h[i-s]) for valid s where i-s in [0, n-1]
        # i-s in [0, n-1] means s in [i-(n-1), i]

        for s in range(s_min2, s_max2 + 1):
            k = s + (n-1)
            grad[i] += weights[k] * (1-h[i-s])
    
    return grad

import numpy as np
from scipy.optimize import minimize
import functools
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum(h, n, target=None):
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def upsample_to(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def smooth_max_val(corr, tau):
    m = np.max(corr)
    if tau < 1e-10:
        return m
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -100.0, 0.0)
    return m + tau * np.log(np.sum(np.exp(shifted)))


def obj_func(x, n, dx, tau):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(smooth_max_val(corr, tau) * dx)


def con_func(x, n):
    return x.sum() - n / 2.0


def con_jac(x, n):
    return np.ones(n)


def do_slsqp(x0, n, dx, tau, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    obj_p = functools.partial(obj_func, n=n, dx=dx, tau=tau)
    con_p = functools.partial(con_func, n=n)
    jac_p = functools.partial(con_jac, n=n)
    try:
        res = minimize(
            fun=obj_p,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
        )
        h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return None, None


def lbfgsb_obj(x, n, dx, tau, mu):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max_val(corr, tau)
    pen = mu * (h.sum() - n / 2.0) ** 2
    return sm * dx + pen


def lbfgsb_grade(x, n, dx, tau, mu):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c_max = np.max(corr)
    exps = np.exp(np.clip((corr - c_max) / tau, -100.0, 0.0))
    weights = exps / np.sum(exps)

    grad = np.zeros(n)
    n_corr = len(corr)
    center = (n - 1)
    for i in range(n):
        s1 = max(0, -(i - (n - 1)))
        s2 = min(n_corr - 1, (n - 1) - i + i + n - 1 - 1)
        # Part 1: -sum h[i+s] * w[s] for valid s (i+s in [0,n-1])
        # s in [-i, n-1-i], k = s + center in [center-i, center+n-1-i]
        k_lo = max(0, center - i)
        k_hi = min(n_corr - 1, center + n - 1 - i)
        if k_hi >= k_lo:
            s_vals = np.arange(k_lo, k_hi + 1) - center
            j_vals = i + s_vals
            grad[i] -= np.dot(weights[k_lo:k_hi + 1], h[j_vals])
        # Part 2: +sum (1-h[i-s]) * w[s] for valid s (i-s in [0,n-1])
        # s in [i-(n-1), i], k = s + center in [center+i-n+1, center+i]
        k_lo2 = max(0, center + i - (n - 1))
        k_hi2 = min(n_corr - 1, center + i)
        if k_hi2 >= k_lo2:
            s_vals2 = np.arange(k_lo2, k_hi2 + 1) - center
            j_vals2 = i - s_vals2
            grad[i] += np.dot(weights[k_lo2:k_hi2 + 1], 1.0 - h[j_vals2])

    # Add penalty gradient
    pen_coeff = 2.0 * mu * (h.sum() - n / 2.0)
    grad += pen_coeff
    grad *= dx

    return dx * grad + 0.0


def do_lbfgsb(x0, n, dx, tau, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    obj_p = functools.partial(lbfgsb_obj, n=n, dx=dx, tau=tau, mu=mu)
    try:
        res = minimize(
            fun=obj_p,
            x0=x0,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
        )
        h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return None, None
