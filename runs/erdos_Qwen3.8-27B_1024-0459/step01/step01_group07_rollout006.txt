TARGET_C5 = 0.38085


def project_box_sum(x, n):
    target = 0.5 * float(n)
    x = np.asarray(x, dtype=float)
    if x.size != n:
        x = np.full(n, 0.5)
    if not np.all(np.isfinite(x)):
        x = np.full(n, 0.5)

    lo = -2.0 - max(0.0, float(np.max(x)))
    hi = 2.0 + max(0.0, target - float(np.sum(x)))
    if hi <= lo:
        hi = lo + 1.0

    for _ in range(20):
        if float(np.sum(np.clip(x + lo, 0.0, 1.0))) >= target:
            lo -= max(1.0, hi - lo)
        else:
            break
    for _ in range(20):
        if float(np.sum(np.clip(x + hi, 0.0, 1.0))) <= target:
            hi += max(1.0, hi - lo)
        else:
            break

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.clip(x + mid, 0.0, 1.0))) < target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    y = np.clip(x + lam, 0.0, 1.0)

    diff = target - float(np.sum(y))
    free = (y > 1e-12) & (y < 1.0 - 1e-12)
    if np.any(free):
        y[free] += diff / float(np.sum(free))
        y = np.clip(y, 0.0, 1.0)

    diff = target - float(np.sum(y))
    if abs(diff) > 1e-13:
        y += diff / float(n)
        y = np.clip(y, 0.0, 1.0)
    return y


def erdos_value(x, n):
    y = np.asarray(x, dtype=float)
    if y.size != n:
        y = np.full(n, 0.5)
    y = np.clip(y, 0.0, 1.0)
    dx = 2.0 / float(n)
    conv = np.correlate(y, 1.0 - y, mode="full")
    return float(np.max(conv) * dx)


def binary_feasible(x, n):
    x = np.asarray(x, dtype=float)
    if x.size != n:
        x = np.full(n, 0.5)
    if n % 2 == 0:
        k = n // 2
    else:
        k = int(np.int64(np.round(0.5 * n)))
    idx = np.argsort(-x)[:k]
    y = np.zeros(n, dtype=float)
    y[idx] = 1.0
    return y


def smooth_value_only(x, n, temp):
    y = np.asarray(x, dtype=float)
    if y.size != n:
        y = np.full(n, 0.5)
    y = np.clip(y, 0.0, 1.0)
    dx = 2.0 / float(n)
    conv = np.correlate(y, 1.0 - y, mode="full") * dx
    mx = float(np.max(conv))
    shifted = (conv - mx) / max(1e-12, float(temp))
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        return mx
    return mx + float(temp) * math.log(s)


def smooth_obj_grad(x, n, temp):
    y = np.asarray(x, dtype=float)
    if y.size != n:
        y = np.full(n, 0.5)
    if not np.all(np.isfinite(y)):
        y = np.full(n, 0.5)
    y = np.clip(y, 0.0, 1.0)
    dx = 2.0 / float(n)

    conv = np.correlate(y, 1.0 - y, mode="full") * dx
    mx = float(np.max(conv))
    temp = max(1e-12, float(temp))
    shifted = (conv - mx) / temp
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        idx0 = int(np.argmax(conv))
        w[idx0] = 1.0
        s = 1.0

    val = mx + temp * math.log(s)
    grad = np.zeros(n, dtype=float)

    active = np.where(w > max(s * 1e-8, 1e-300))[0]
    k_max = min(len(conv), max(12, int(3.0 * math.sqrt(max(1, n)))))
    if len(active) > k_max:
        active = np.argsort(-w)[:k_max]

    for idx in active:
        idx = int(idx)
        r = idx - (n - 1)
        st = max(0, -r)
        en = min(n - 1, n - 1 - r)
        sl_base = slice(st, en + 1)
        sl_shift = slice(st + r, en + r + 1)
        coeff = (float(w[idx]) / s) * dx
        grad[sl_shift] += (1.0 - y[sl_base]) * coeff
        grad[sl_base] += (-y[sl_shift]) * coeff

    return val, grad


def sum_constraint(x, target):
    return float(np.sum(x) - target)


def sum_constraint_grad(x, target):
    return np.ones_like(x, dtype=float)


def choose_slsqp_maxiter(n, factor):
    if n <= 80:
        base = 260
    elif n <= 130:
        base = 170
    elif n <= 200:
        base = 110
    else:
        base = 70
    return max(20, int(float(base) * float(factor)))


def slsqp_smooth(h0, n, temp, maxiter):
    x0 = project_box_sum(h0, n)
    target = 0.5 * float(n)
    cons = [
        {
            "type": "eq",
            "fun": sum_constraint,
            "jac": sum_constraint_grad,
            "args": (target,),
        }
    ]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            smooth_obj_grad,
            x0,
            args=(n, temp),
            method="SLSQP",
            jac=True,
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-11},
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            xp = project_box_sum(res.x, n)
            return xp, smooth_value_only(xp, n, temp)
    except Exception:
        pass
    return x0, smooth_value_only(x0, n, temp)


def pgl_polish(h0, n, temp, max_iter, lr_init, deadline):
    h = project_box_sum(h0, n)
    best = erdos_value(h, n)
    h_best = h.copy()
    f, g = smooth_obj_grad(h, n, temp)
    lr = float(lr_init)
    fails = 0

    for _ in range(int(max_iter)):
        if time.time() >= float(deadline):
            break
        if np.linalg.norm(g) < 1e-12:
            break

        gt = g - np.mean(g)
        accepted = False
        lr_cur = lr

        for _ in range(50):
            y = project_box_sum(h - lr_cur * gt, n)
            fy = smooth_value_only(y, n, temp)
            if fy <= f + 1e-13 or lr_cur < 1e-12:
                accepted = True
                break
            lr_cur *= 0.5

        if not accepted:
            fails += 1
            if fails > 10:
                break
            lr = max(lr_cur, 1e-4)
            continue

        h = y
        f, g = smooth_obj_grad(h, n, temp)
        lr = min(0.5, lr_cur * 1.2)
        val = erdos_value(h, n)
        if val < best:
            best = val
            h_best = h.copy()

    return h_best, best


def optimize_start(h0, n, temps, deadline, factor=1.0):
    h_best = project_box_sum(h0, n)
    best_val = erdos_value(h_best, n)
    h_cur = h_best.copy()
    cur_val = best_val

    maxiter = choose_slsqp_maxiter(n, factor)
    for temp in temps:
        if time.time() >= float(deadline):
            break
        h_try, _ = slsqp_smooth(h_cur, n, temp, maxiter)
        v_try = erdos_value(h_try, n)
        if v_try < best_val:
            best_val = v_try
            h_best = h_try.copy()

        if v_try < cur_val + 0.004:
            h_cur = h_try
            cur_val = v_try

    if time.time() < float(deadline) - 5.0:
        h_try, v_try = pgl_polish(h_best, n, 0.003, 90, 0.02, deadline)
        if v_try < best_val:
            best_val = v_try
            h_best = h_try

    return h_best, best_val


def pick_extreme(candidates, vals, mode, k, rng):
    candidates = np.asarray(candidates, dtype=int)
    vals = np.asarray(vals, dtype=float)
    k = int(min(int(k), len(candidates)))
    if k <= 0:
        return int(candidates[0])

    if mode > 0:
        order = np.argsort(-vals)
    else:
        order = np.argsort(vals)
    order = order[:k]
    order = order[:max(1, len(order))]
    choice = int(order[int(rng.randint(int(len(order))))])
    return int(candidates[choice])


def binary_local_search(h0, n, max_iters, rng, deadline):
    h = binary_feasible(h0, n)
    best = erdos_value(h, n)
    h_best = h.copy()
    cur = best
    temp = 0.07

    for _ in range(int(max_iters)):
        if time.time() >= float(deadline):
            break

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        _, g = smooth_obj_grad(h, n, 0.02)
        g = g - np.mean(g)

        neg = ones[g[ones] < -1e-10]
        pos = zeros[g[zeros] > 1e-10]
        if len(neg) == 0:
            neg = ones
        if len(pos) == 0:
            pos = zeros

        if rng.random() < 0.82:
            i = pick_extreme(neg, g[neg], -1, 4, rng)
            j = pick_extreme(pos, g[pos], 1, 4, rng)
        else:
            i = int(rng.choice(neg))
            j = int(rng.choice(pos))

        if i == j:
            continue

        old_i = h[i]
        old_j = h[j]
        h[i] = old_j
        h[j] = old_i
        val = erdos_value(h, n)

        if val <= cur:
            cur = val
            if val < best:
                best = val
                h_best = h.copy()
                temp = max(1e-4, temp * 0.999)
            sigma = 0.999
        else:
            delta = val - cur
            if rng.random() < math.exp(-max(0.0, delta) / max(1e-4, temp)):
                cur = val
                temp = max(1e-4, temp * 0.999)
            else:
                h[i] = old_i
                h[j] = old_j
                temp = max(1e-4, temp * 0.998)
                if rng.random() < 0.01:
                    temp = max(1e-4, temp * 0.98)

    return h_best, best


def anneal_continuous(h0, n, max_iters, rng, deadline):
    h = project_box_sum(h0, n)
    best = erdos_value(h, n)
    h_best = h.copy()
    cur = best
    sigma = 0.18
    temp = 0.035

    for _ in range(int(max_iters)):
        if time.time() >= float(deadline):
            break

        y = project_box_sum(h + sigma * rng.normal(size=n), n)
        val = erdos_value(y, n)

        if val <= cur:
            h = y
            cur = val
            if val < best:
                best = val
                h_best = y.copy()
                sigma = min(0.45, sigma * 1.02)
            else:
                sigma = max(0.02, sigma * 0.999)
            temp = max(1e-4, temp * 0.999)
        else:
            delta = val - cur
            if rng.random() < math.exp(-max(0.0, delta) / max(1e-4, temp)):
                h = y
                cur = val
                temp = max(1e-4, temp * 0.999)
            else:
                if rng.random() < 0.02:
                    sigma = max(0.02, sigma * 0.92)
                temp = max(1e-4, temp * 0.9995)

    return h_best, best


def resample_nearest(h, n_new):
    h = np.asarray(h, dtype=float)
    n_old = len(h)
    if n_old == 0:
        return project_box_sum(np.full(n_new, 0.5), n_new)
    idx = np.rint(np.linspace(0.0, float(n_old - 1), int(n_new))).astype(int)
    idx = np.clip(idx, 0, n_old - 1)
    return project_box_sum(h[idx], n_new)


def resample_linear(h, n_new):
    h = np.asarray(h, dtype=float)
    n_old = len(h)
    if n_old == 0:
        return project_box_sum(np.full(n_new, 0.5), n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, int(n_new), endpoint=False)
    return project_box_sum(np.interp(x_new, x_old, h), n_new)


def interval_h(n, m, rng, sym=False):
    n = int(n)
    m = max(1, int(m))
    total = 0.5 * float(n)

    one = rng.dirichlet(np.ones(m, dtype=float)) * total
    gap = rng.dirichlet(np.ones(m + 1, dtype=float)) * total

    h = np.zeros(n, dtype=float)
    pos = 0.0
    for j in range(m):
        pos += float(gap[j])
        a = int(np.floor(pos - 1e-12))
        pos += float(one[j])
        b = int(np.ceil(pos - 1e-12))
        a = max(0, min(n, a))
        b = max(0, min(n, b))
        if b > a:
            h[a:b] = 1.0

    h = project_box_sum(h, n)
    if sym:
        h = 0.5 * (h + h[::-1])
        h = project_box_sum(h, n)
    return h


def fourier_h(n, max_freq, rng):
    n = int(n)
    max_freq = max(1, int(max_freq))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    y = 0.5 * np.ones(n, dtype=float)
    for k in range(1, max_freq + 1):
        amp = rng.uniform(0.04, 0.22) / (float(k) ** 0.8)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        y += amp * np.sin(np.pi * float(k) * x + phase)
        y -= amp * 0.5 * np.cos(np.pi * float(k) * x - phase) ** 2 * 0.05
    return project_box_sum(y, n)


def generate_starts(n, base_h, rng, limit):
    n = int(n)
    starts = []

    if base_h is not None and len(base_h) > 0:
        if len(base_h) != n:
            starts.append(resample_nearest(base_h, n))
            starts.append(resample_linear(base_h, n))
        else:
            starts.append(project_box_sum(base_h, n))

    for m in (2, 3, 4, 5, 6):
        starts.append(interval_h(n, m, rng, False))
        starts.append(interval_h(n, m, rng, True))

    starts.append(fourier_h(n, 3, rng))
    starts.append(fourier_h(n, 5, rng))
    starts.append(project_box_sum(rng.uniform(0.0, 1.0, n), n))
    starts.append(binary_feasible(rng.uniform(0.0, 1.0, n), n))

    vals = []
    for s in starts:
        s = project_box_sum(s, n)
        vals.append(erdos_value(s, n))

    vals = np.asarray(vals, dtype=float)
    order = np.argsort(vals)
    limit = max(1, int(min(int(limit), len(order))))
    out = []
    for idx in order[:limit]:
        out.append(starts[int(idx)])
    return out


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = max(30.0, float(budget_s))
    deadline = t0 + budget - 5.0
    rng = np.random.RandomState(int(seed) % (2**31 - 1))

    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        n0 = 70
        h0 = np.full(n0, 0.5)
    else:
        h0 = parent.astype(float, copy=True).ravel()
        n0 = len(h0)

    best_h = project_box_sum(h0, n0)
    best_val = erdos_value(best_h, n0)
    best_n = int(n0)

    if budget < 60.0:
        best_h = project_box_sum(best_h, best_n)
        best_val = erdos_value(best_h, best_n)
        return (best_h, best_val, best_n)

    initial_deadline = min(deadline, t0 + min(140.0, 0.15 * budget))
    if time.time() < initial_deadline:
        h_try, v_try = optimize_start(
            best_h,
            best_n,
            (0.08, 0.04, 0.02, 0.01, 0.005, 0.002),
            initial_deadline,
            1.3,
        )
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()
            best_n = best_n

    binary_deadline = min(deadline - 300.0, t0 + min(200.0, 0.25 * budget))
    if time.time() < binary_deadline:
        for nb in (50, 70, 90, 100):
            if time.time() >= binary_deadline:
                break
            base_bin = resample_nearest(best_h, nb) if best_n != nb else best_h
            starts_bin = []
            starts_bin.append(binary_feasible(base_bin, nb))
            starts_bin.append(binary_feasible(interval_h(nb, 4, rng, False), nb))
            starts_bin.append(binary_feasible(interval_h(nb, 6, rng, True), nb))
            starts_bin.append(binary_feasible(interval_h(nb, 3, rng, False), nb))

            iters = max(2000, min(30000, int(1500000 / max(1, nb))))
            for s in starts_bin:
                if time.time() >= binary_deadline:
                    break
                h_try, v_try = binary_local_search(s, nb, iters, rng, binary_deadline)
                if v_try < best_val:
                    best_val = v_try
                    best_h = h_try.copy()
                    best_n = nb

                if time.time() < binary_deadline - 5.0:
                    cont_start = project_box_sum(h_try + 0.03 * rng.normal(size=nb), nb)
                    h_c, v_c = optimize_start(
                        cont_start,
                        nb,
                        (0.03, 0.012, 0.005),
                        binary_deadline,
                        0.7,
                    )
                    if v_c < best_val:
                        best_val = v_c
                        best_h = h_c.copy()
                        best_n = nb

    sa_deadline = min(deadline - 260.0, binary_deadline + min(130.0, 0.15 * budget))
    if time.time() < sa_deadline:
        sa_iters = max(2000, min(60000, int(700000 / max(1, best_n))))
        h_try, v_try = anneal_continuous(best_h, best_n, sa_iters, rng, sa_deadline)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()
            best_n = best_n

        if time.time() < sa_deadline - 10.0:
            h_alt = binary_feasible(resample_nearest(best_h, best_n), best_n)
            h_alt = project_box_sum(h_alt + 0.05 * rng.normal(size=best_n), best_n)
            h_try, v_try = anneal_continuous(h_alt, best_n, sa_iters, rng, sa_deadline)
            if v_try < best_val:
                best_val = v_try
                best_h = h_try.copy()
                best_n = best_n

    multi_deadline = deadline - 80.0
    n_list = [best_n]
    for cand in (70, 100, 120, 140, 180, 220):
        if cand not in n_list and cand <= max(90, best_n + 80):
            n_list.append(cand)
    n_list = sorted(set(n_list))[:5]

    for n_try in n_list:
        if time.time() >= multi_deadline:
            break
        n_try = int(n_try)
        if n_try < 10:
            continue

        starts = generate_starts(n_try, best_h, rng, 7)
        for s in starts[:4]:
            if time.time() >= multi_deadline:
                break
            factor = 0.9 if n_try <= 120 else 0.7
            temps = (0.05, 0.02, 0.008, 0.003)
            if n_try > 150:
                temps = (0.05, 0.02, 0.008)
            h_try, v_try = optimize_start(s, n_try, temps, multi_deadline, factor)
            if v_try < best_val:
                best_val = v_try
                best_h = h_try.copy()
                best_n = n_try

    if time.time() < deadline - 150.0 and best_n < 160:
        nf = 220
        if time.time() < deadline - 150.0 and best_n >= 180:
            nf = 280
        if time.time() < deadline - 120.0:
            h_f = resample_linear(best_h, nf)
            h_try, v_try = optimize_start(h_f, nf, (0.05, 0.02, 0.008), deadline - 30.0, 0.6)
            if v_try < best_val:
                best_val = v_try
                best_h = h_try.copy()
                best_n = nf

    if time.time() < deadline - 8.0:
        h_try, v_try = optimize_start(
            best_h,
            best_n,
            (0.012, 0.006, 0.003, 0.0015),
            deadline,
            1.2,
        )
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()
            best_n = best_n

    best_h = project_box_sum(best_h, best_n)
    best_val = erdos_value(best_h, best_n)
    return (best_h, best_val, int(best_n))
