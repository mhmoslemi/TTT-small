    return h_bin

def kronecker_init(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        if ((i + 0.5) * alpha) % 1.0 < 0.5:
            h[i] = 1.0
    target = n // 2
    s = int(np.sum(h))
    if s < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - s]] = 1.0
    elif s > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:s - target]] = 0.0
    return h

def binary_ils(h_init, n, rng, time_limit=30.0):
    t0 = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    while time.time() - t0 < time_limit:
        # Local search: try single swaps
        no_improve_count = 0
        while no_improve_count < 80 and time.time() - t0 < time_limit:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            improved = False
            n_trials = 200
            best_trial_val = best_val
            best_pair = None
            
            for _ in range(n_trials):
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                h[a], h[b] = h[b], h[a]
                val = compute_c5(h, n)
                if val < best_trial_val - 1e-15:
                    best_trial_val = val
                    best_pair = (a, b)
                h[a], h[b] = h[b], h[a]
            
            if best_pair is not None:
                a, b = best_pair
                h[a], h[b] = h[b], h[a]
                if best_trial_val < best_val:
                    best_val = best_trial_val
                    best_h = h.copy()
                improved = True
            
            if not improved:
                no_improve_count += 1
                # Try 2-swaps for escape
                if no_improve_count == 30:
                    tried = 0
                    best_2swap_val = best_val
                    best_2swap = None
                    while tried < 100 and best_2swap is None:
                        tried += 1
                        o1 = ones[rng.randint(len(ones))]
                        o2 = ones[rng.randint(len(ones))]
                        z1 = zeros[rng.randint(len(zeros))]
                        z2 = zeros[rng.randint(len(zeros))]
                        if o1 == o2 or z1 == z2:
                            continue
                        h[o1], h[z1] = h[z1], h[o1]
                        h[o2], h[z2] = h[z2], h[o2]
                        val = compute_c5(h, n)
                        if val < best_2swap_val - 1e-15:
                            best_2swap_val = val
                            best_2swap = (o1, o2, z1, z2)
                        h[z1], h[o1] = h[o1], h[z1]
                        h[z2], h[o2] = h[o2], h[z2]
                    if best_2swap is not None:
                        o1, o2, z1, z2 = best_2swap
                        h[o1], h[z1] = h[z1], h[o1]
                        h[o2], h[z2] = h[z2], h[o2]
                        best_val = best_2swap_val
                        best_h = h.copy()
                        no_improve_count = 0
                continue
        
        # Perturbation for ILS
        k = rng.choice([2, 3, 4, 5, 6, 8])
        for _ in range(k):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a], h[b] = h[b], h[a]
    
    return best_h, best_val

def soft_grad_descent(h, n, rng, time_limit=30.0):
    """Projected gradient descent with soft-max objective."""
    t0 = time.time()
    h = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.1
    
    for it in range(10000):
        if time.time() - t0 > time_limit:
            break
        
        conv = np.correlate(h, 1.0 - h, mode='full') * dx
        temp = 0.02 + 0.03 * np.exp(-it / 500.0)
        m = np.max(conv)
        w = np.exp((conv - m) / temp)
        wsum = np.sum(w)
        if wsum < 1e-10:
            break
        w /= wsum
        
        # Compute gradient analytically
        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            if w[k_idx] < 1e-4:
                continue
            lag = k_idx - (n - 1)
            wk = w[k_idx] * dx
            if lag > 0:
                # c = sum_{j=0}^{n-1-lag} h[j+lag]*(1-h[j])
                # dc/dh[j] = (1-h[j-lag])*[j>=lag] - h[j+lag]*[j<n-lag]
                if n - lag > 0:
                    grad[:n-lag] += -h[lag:] * wk
                if lag < n:
                    grad[lag:] += (1.0 - h[:n-lag]) * wk
            elif lag < 0:
                p = -lag
                # c = sum_{j=0}^{n-1-p} h[j]*(1-h[j+p])
                # dc/dh[j] = (1-h[j+p])*[j<n-p] - h[j-p]*[j>=p]
                if n - p > 0:
                    grad[:n-p] += (1.0 - h[p:]) * wk
                if p < n:
                    grad[p:] += -h[:n-p] * wk
            else:
                # c = sum h[j]*(1-h[j]), dc/dh[j] = 1-2h[j]
                grad += (1.0 - 2.0 * h) * wk
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        lr *= 0.998
    
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    def tl():
        return budget_s - (time.time() - t_start)
    
    # Initialize
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except:
        pass
    
    if h_init is not None and len(h_init) > 0:
        n0 = len(h_init)
    else:
        n0 = 70
        h_init = np.full(n0, 0.5)
    
    h_init = project_feasible(h_init, n0)
    best_h = h_init.copy()
    best_val = compute_c5(h_init, n0)
    best_n = n0
    
    # Phase 1: Binary ILS on parent
    h_bin = make_binary(h_init, n0)
    h_b, v_b = binary_ils(h_bin, n0, rng, time_limit=min(150, tl() * 0.12))
    if v_b < best_val:
        best_h, best_val = h_b, v_b
    
    # Phase 2: Kronecker initializations
    alphas = [(1+5**0.5)/4, 1/3, 2/5, 3/7, 4/9, 5/12, 0.382, 0.618,
              1/4, 3/11, 5/18, 7/25, 0.36, 0.44, 0.33, 0.47, 0.29, 0.51]
    
    resolutions = [n0, 60, 80, 100, 120]
    for n_r in resolutions:
        if tl() < 45:
            break
        if n_r < 10:
            continue
        if n_r % 2 != 0:
            n_r += 1
        for alpha in alphas:
            if tl() < 20:
                break
            h_kr = kronecker_init(n_r, alpha)
            h_kr_b, v_kr = binary_ils(h_kr, n_r, rng, time_limit=min(12, tl() * 0.04))
            if v_kr < best_val:
                best_h, best_val, best_n = h_kr_b, v_kr, n_r
    
    # Phase 3: Soft gradient descent on best
    if tl() > 40:
        h_soft = best_h + rng.normal(0, 0.03, best_n)
        h_soft = project_feasible(h_soft, best_n)
        h_s, v_s = soft_grad_descent(h_soft, best_n, rng, time_limit=min(50, tl() * 0.1))
        if v_s < best_val:
            best_h, best_val = h_s, v_s
    
    # Phase 4: More ILS with different perturbation strengths
    h_bin2 = make_binary(best_h, best_n)
    h_b2, v_b2 = binary_ils(h_bin2, best_n, rng, time_limit=min(200, tl() * 0.2))
    if v_b2 < best_val:
        best_h, best_val = h_b2, v_b2
    
    # Phase 5: Higher resolutions
    if tl() > 100:
        for n_f in [140, 200, 280, 350]:
            if tl() < 50:
                break
            if n_f % 2 != 0:
                n_f += 1
            x_c = np.linspace(0, 2, best_n, endpoint=False)
            x_f = np.linspace(0, 2, n_f, endpoint=False)
            h_f = np.interp(x_f, x_c, best_h)
            h_f = project_feasible(h_f, n_f)
            h_fb = make_binary(h_f, n_f)
            h_fres, v_fres = binary_ils(h_fb, n_f, rng, time_limit=min(30, tl() * 0.1))
            if v_fres < best_val:
                best_h, best_val, best_n = h_fres, v_fres, n_f
            if tl() > 30:
                h_sf, v_sf = soft_grad_descent(h_f, n_f, rng, time_limit=min(20, tl() * 0.05))
                if v_sf < best_val:
                    best_h, best_val, best_n = h_sf, v_sf, n_f
    
    # Phase 6: Final SLSQP polish
    if tl() > 25:
        try:
            n = best_n
            target = n / 2.0
            def obj(x):
                return compute_c5(x, n)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                     'jac': lambda x: np.ones(n)}]
            bounds = [(0.0, 1.0)] * n
            res = minimize(obj, best_h.copy(), method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
            h_try = project_feasible(res.x, n)
            v_try = compute_c5(h_try, n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except:
            pass
    
    # Final
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)
    return (best_h, best_val, best_n)
