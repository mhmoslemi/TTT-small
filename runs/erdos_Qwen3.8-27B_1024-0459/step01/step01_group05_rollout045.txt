    upper = 1.0

    for _ in range(100):

    lo, hi = 0.0, 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        vals = np.clip(h - mid, 0.0, 1.0)

        else:

    """Gradient of correlation at a specific offset (lag = offset) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    # corr at this offset: sum_i h[i]*(1-h[i+offset]) for valid indices

        if 0 <= j + offset < n:

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    comp = 1.0 - h
    return np.correlate(h, comp, mode='full')


def project_onto_constraints(h, target):
    h = h.astype(float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        vals = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(vals)
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            free = h_proj < 0.9999999
        else:
            free = h_proj > 1e-7
        if np.any(free):
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def grad_corr_single(h, offset):
    n = len(h)
    g = np.zeros(n)
    for j in range(n):
        jp = j + offset
        jm = j - offset
        if 0 <= jp < n:
            g[j] += (1.0 - h[jp])
        if 0 <= jm < n:
            g[j] -= h[jm]
    return g


def smooth_max_value_grad(h, beta):
    corr = compute_corr(h)
    n = len(h)
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    weights = exp_vals / s
    val = cmax + np.log(s) / beta
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-14:
            continue
        offset = k - (n - 1)
        g += weights[k] * beta * grad_corr_single(h, offset)
    return val, g

def subgradient_refine(h0, target, n_iter, step_init):

    n = len(h)
    best_h = h.copy()

    x_old = np.linspace(0, 2, n_old, endpoint=False)

    return np.interp(x_new, x_old, h_old)
