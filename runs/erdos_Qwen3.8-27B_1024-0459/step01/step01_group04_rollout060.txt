                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += float(hj)
                scores[midx] -= 1.0 - hj
        if i + d < n:
            hj = int(h[i + d])
            if new_val == 1:
                scores[pidx] += 1.0 - hj
                scores[midx] -= float(hj)
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += float(hj)
    h[i] = new_val

def binary_local_search(start_h, n, deadline, seed):
    rng = np.random.default_rng(seed)
    target = n // 2
    h = np.zeros(n, dtype=np.int8)
    if start_h is not None and start_h.size == n:
        order = np.argsort(-start_h)
        h[order[:target]] = 1
    else:
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1
    scores = binary_scores(h, n)
    cur = float(np.max(scores))
    best = cur
    best_h = h.copy()
    stall = 0
    max_stall = 5000
    it = 0
    while it < max_stall and time.time() < deadline:
        it += 1
        if it % 10000 == 0:
            check_deadline()
        zeros = np.flatnonzero(h == 0)
        ones = np.flatnonzero(h == 1)
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = int(zeros[rng.integers(len(zeros))])
        j = int(ones[rng.integers(len(ones))])
        hi = int(h[i])
        hj = int(h[j])
        if hi == hj:
            continue
        binary_flip(scores, h, i, 1 - hi, n)
        binary_flip(scores, h, j, 1 - hj, n)
        new_max = float(np.max(scores))
        if new_max < cur - 1e-12:
            cur = new_max
            stall = 0
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            stale = (new_max <= cur + 1e-12) and (rng.random() < 0.05 and stall > 100)
            if stale:
                cur = new_max
                stall += 1
            else:
                binary_flip(scores, h, i, hi, n)
                binary_flip(scores, h, j, hj, n)
                stall += 1
        if stall > max_stall:
            break
    return best_h.astype(float), best * (2.0 / n)

def slsqp_obj(x, alpha):
    n = x.size
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    if alpha > 1e-14:
        w = np.exp((corr - m) / alpha)
        A = float(np.sum(w))
        if math.isfinite(A) and A > 0:
            val = m + alpha * math.log(A)
        else:
            val = m
    else:
        val = m
    return float(val) * (2.0 / n)

def slsqp_grad(x, alpha):
    n = x.size
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    pw, _ = compute_smooth_weights(corr, alpha)
    return compute_grad(h, n, pw)

def slsqp_c_eq(x, *args):
    n = max(1, len(x))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)

def slsqp_c_jac(x, *args):
    n = max(1, len(x))
    return np.full((1, n), 1.0 / float(n))

def run_slsqp_polish(h_init, n, deadline):
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": slsqp_c_eq, "jac": slsqp_c_jac}]
    alpha = 0.01
    maxiter = int(max(30, min(200, 4000.0 / n)))
    old_dl = _RUN_DEADLINE
    _RUN_DEADLINE = min(deadline, time.time() + max(10.0, (deadline - time.time())))
    try:
        res = minimize(slsqp_obj, h, args=(alpha,), jac=slsqp_grad,
                       method="SLSQP", bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0})
    except BudgetError:
        res = None
    except Exception:
        res = None
    finally:
        _RUN_DEADLINE = old_dl
    if res is not None and res.x is not None:
        h = project_feasible(res.x, n)
    return h

def upsample(parent, n):
    n0 = parent.size
    if n0 <= 0:
        return np.full(n, 0.5)
    if n == n0:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent).astype(float)

def gen_perturbation(h, n, rng, scale):
    pert = h + rng.normal(0, scale, size=n)
    return project_feasible(pert, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    global_deadline = t0 + budget_s - 15.0
    _RUN_DEADLINE = global_deadline
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None and parent.size >= 4:
        n = int(parent.size)
        h0 = project_feasible(parent, n)
    else:
        n = 70
        h0 = np.full(n, 0.5)
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = 0.5 + 0.3 * np.cos(math.pi * x)
        h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c = compute_c5(h0)
    best_n = n

    remaining = global_deadline - time.time()
    if remaining < 30:
        return best_h, best_c, best_n

    phase1_budget = 0.55 * remaining
    phase2_budget = 0.20 * remaining
    phase3_budget = 0.15 * remaining
    phase4_budget = 0.10 * remaining

    # Phase 1: Smooth PGD from multiple starts
    t_phase1_end = time.time() + phase1_budget
    starts = [h0.copy()]
    for sc in (0.03, 0.08, 0.15):
        starts.append(gen_perturbation(h0, n, rng, sc))

    for si, hs in enumerate(starts):
        if time.time() > t_phase1_end:
            break
        seg_time = (t_phase1_end - time.time()) / max(1, len(starts) - si)
        seg_dl = time.time() + seg_time
        hb, cb = run_smooth_pgd(hs, n, best_c, seg_dl)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
            best_n = n

    # Phase 2: Exact subgradient refinement
    t_phase2_end = time.time() + phase2_budget
    if time.time() < t_phase2_end:
        hb2, cb2 = run_exact_subgrad(best_h, n, best_c, t_phase2_end)
        if cb2 < best_c:
            best_c = cb2
            best_h = hb2.copy()

    # Phase 3: SLSQP polish
    t_phase3_end = time.time() + phase3_budget
    if time.time() < t_phase3_end - 5:
        hp = run_slsqp_polish(best_h, n, t_phase3_end)
        cp = compute_c5(hp)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Phase 4: Binary local search + try higher n
    t_phase4_end = global_deadline
    if time.time() < t_phase4_end - 10 and n % 2 == 0:
        # Binary search at current n
        t_bin = min(30.0, (t_phase4_end - time.time()) * 0.4)
        bh, bc = binary_local_search(best_h, n, time.time() + t_bin, seed + 1)
        if bc < best_c:
            best_c = bc
            best_h = bh.copy()

        # Try higher resolution
        n2 = 2 * n
        if time.time() < t_phase4_end - 30:
            h_up = upsample(best_h, n2)
            h_up = project_feasible(h_up, n2)
            c_up = compute_c5(h_up)
            if c_up < best_c:
                best_c = c_up
                best_h = h_up.copy()
                best_n = n2
            # Quick PGD at n2
            t_rem = max(10.0, t_phase4_end - time.time() - 5)
            if t_rem > 10:
                hb2, cb2 = run_smooth_pgd(h_up, n2, best_c, time.time() + t_rem)
                if cb2 < best_c:
                    best_c = cb2
                    best_h = hb2.copy()
                    best_n = n2

    # Final validation
    h_final = project_feasible(best_h, best_n)
    c_final = compute_c5(h_final)
    if c_final > best_c:
        best_c = c_final
    return h_final, float(best_c), int(best_n)
