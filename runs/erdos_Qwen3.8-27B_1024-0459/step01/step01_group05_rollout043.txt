2. For those K shifts, compute the breakpoints
3. Check the function value at each breakpoint and at the boundaries
4. Pick the best

    n = len(h)

    # Using the formula: ∂c_s/∂h[p] = (1-h[p+s])*I[valid1] + (-h[p-s])*I[valid2]

    d = np.zeros(2*n - 1)

        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)

        if lo <= i <= hi:
            ps = i + s
            if 0 <= ps < n:
                gi += (1.0 - h[ps])
        if lo <= i - s <= hi:
            ps = i - s
            if 0 <= ps < n:
                gi += (-h[ps])
        
        gj = 0.0
        if lo <= j <= hi:
            ps = j + s
            if 0 <= ps < n:
                gj += (1.0 - h[ps])
        if lo <= j - s <= hi:
            ps = j - s
            if 0 <= ps < n:
                gj += (-h[ps])
        
        d[m] = gi - gj
    
    # δ bounds
    lo_d = max(-h[i], h[j] - 1.0)
    hi_d = min(1.0 - h[i], h[j])
    
    if lo_d >= hi_d - 1e-15:
        return 0.0
    
    # Find the minimum of f(δ) = max_s(corr[s] + δ*d[s]) on [lo_d, hi_d]
    # Use top-K approach
    # Find top K indices by corr value
    top_k = np.argsort(corr)[-K:]
    
    # Evaluate at boundaries and breakpoints among top-K
    candidates = [lo_d, hi_d, 0.0]

        for b in top_k:

                delta = (corr[b] - corr[a]) / (d[a] - d[b])
                if lo_d <= delta <= hi_d:
                    candidates.append(delta)
    
    best_delta = 0.0
    best_val = np.max(corr)
    for delta in candidates:
        val = np.max(corr + delta * d)
        if val < best_val:
            best_val = val
            best_delta = delta
    
    return best_delta

    G = np.zeros((n, 2*n - 1))
    for m in range(2*n - 1):
        s = m - (n - 1)
        lo = max(0, -s)
        hi = min(n-1, n-1-s)
        # First term: (1-h[p+s]) for p in [lo, hi]
        for p in range(lo, hi + 1):
            G[p, m] += (1.0 - h[p + s])
        # Second term: (-h[p-s]) for p-s in [lo, hi], i.e., p in [lo+s, hi+s]
        lo2 = lo + s
        hi2 = hi + s
        lo2 = max(0, lo2)
        hi2 = min(n-1, hi2)
        for p in range(lo2, hi2 + 1):
            G[p, m] += (-h[p - s])
    return G

def optimize_swap_delta(corr, d, lo_d, hi_d, K=20):
    """Minimize max(corr + δ*d) for δ in [lo_d, hi_d]."""
    if lo_d >= hi_d - 1e-14:
        return 0.0
    
    # Top-K breakpoint method
    idx = np.argsort(corr)[-K:]
    
    candidates = set()
    candidates.add(lo_d)
    candidates.add(hi_d)
    
    for a in idx:
        for b in idx:
            if a != b:
                dd = d[a] - d[b]
                if abs(dd) > 1e-12:
                    delta = (corr[b] - corr[a]) / dd
                    if lo_d - 1e-10 <= delta <= hi_d + 1e-10:
                        candidates.add(np.clip(delta, lo_d, hi_d))
    
    best_delta = 0.0
    best_val = np.max(corr)
    for delta in candidates:
        val = np.max(corr + delta * d)
        if val < best_val:
            best_val = val
            best_delta = delta
    
    return best_delta

def project_sum(h, target):
    """Project h to satisfy sum(h) = target, 0 <= h <= 1."""
    h = h.copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 0) & (h < 1)
        if np.sum(free) == 0:
            break
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0, 1)
    return h
