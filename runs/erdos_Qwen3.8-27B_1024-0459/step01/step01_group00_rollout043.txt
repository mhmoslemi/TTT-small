    f_vals = compute_f(h, n)  # f(k) for all k
    max_val = np.max(f_vals)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(12345)

    for _ in range(n_trials):
        if time.time() - start_time > time_limit - 1.0:
            break
        h_try = best_h.copy()
        n_mod = rng.randint(1, 5)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step_size, step_size)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()

    return best_h, best_c5

def generate_structured_inits(n, seed):
    rng = np.random.RandomState(seed)
    inits = []
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h1.copy())
        h2 = np.tile([1.0, 0.0, 1.0, 0.0], n // 4)
        inits.append(h2.copy())
    if n % 2 == 0:
        h3 = np.zeros(n)
        h3[:n // 4] = 1.0
        h3[n // 2:3 * n // 4] = 1.0
        inits.append(h3.copy())
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm, n))
    h5 = np.zeros(n)
    positions = sorted(range(n), key=lambda i: (i * 7919) % n)
    for j in range(n // 2):
        h5[positions[j]] = 1.0
    inits.append(make_feasible(h5, n))
    inits.append(make_feasible(rng.rand(n) * 1.5, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent.copy(), best_n)
    else:
        best_n = 160
        best_h = make_feasible(np.random.RandomState(seed).rand(best_n) * 1.5, best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution
    if remaining() > 30:
        h_opt, c5_opt = coord_descent(best_h, best_n, mu=5.0,
                                      max_sweeps=500,
                                      time_limit=min(60, remaining() - 10),
                                      start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Try subgradient at parent resolution for diversity
    if remaining() > 30:
        h_sg, c5_sg = subgradient_descent(best_h, best_n,
                                          n_iters=300,
                                          time_limit=min(20, remaining() - 10),
                                          start_time=start_time,
                                          eta_start=0.05)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: Multi-resolution optimization
    resolutions = [best_n]
    for nr in [240, 320, 400, 480, 640]:
        if nr != best_n and remaining() > 60:
            resolutions.append(nr)
    resolutions = sorted(set(resolutions))

    for n in resolutions:
        if remaining() < 40:
            break
        if n == best_n:
            continue

        h_interp = make_feasible(interpolate_to_n(best_h, n), n)
        c5_interp = compute_c5(h_interp)

        inits = [h_interp]
        for hs in generate_structured_inits(n, seed + n):
            c5_hs = compute_c5(hs)
            if c5_hs < c5_interp + 0.05:
                inits.append(hs)

        best_at_n = h_interp.copy()
        best_c5_at_n = c5_interp

        for h0 in inits[:4]:
            if remaining() < 30:
                break
            tl = min(remaining() * 0.25, 120)
            if tl < 15:
                break
            h_cd, c5_cd = coord_descent(h0, n, mu=3.0,
                                        max_sweeps=500,
                                        time_limit=tl,
                                        start_time=time.time())
            if c5_cd < best_c5_at_n:
                best_c5_at_n = c5_cd
                best_at_n = h_cd.copy()

        if best_c5_at_n < best_c5:
            best_c5 = best_c5_at_n
            best_h = best_at_n.copy()
            best_n = n

    # Phase 4: Local perturbation polish on the best solution
    if remaining() > 20:
        tl = min(remaining() * 0.4, 120)
        if tl > 10:
            best_h, best_c5 = local_perturbation(best_h, best_n,
                                                 n_trials=10000,
                                                 step_size=0.02,
                                                 time_limit=tl,
                                                 start_time=time.time())

    # Phase 5: Final coordinate descent polish
    if remaining() > 15:
        tl = min(remaining() - 5, 60)
        if tl > 10:
            h_fin, c5_fin = coord_descent(best_h, best_n, mu=10.0,
                                          max_sweeps=300,
                                          time_limit=tl,
                                          start_time=time.time())
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Phase 6: Final local perturbation
    if remaining() > 10:
        tl = min(remaining() - 3, 30)
        if tl > 5:
            best_h, best_c5 = local_perturbation(best_h, best_n,
                                                 n_trials=5000,
                                                 step_size=0.01,
                                                 time_limit=tl,
                                                 start_time=time.time())

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
