def compute_c5(h, n_points):

    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_c5_full(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')

def smooth_max(vals, temp):
    """Smooth approximation of max using log-sum-exp."""
    m = np.max(vals)
    if m == 0 and np.all(vals == 0):
        return 0.0, np.zeros_like(vals)
    shifted = vals / temp
    shifted -= np.max(shifted)
    logsum = np.log(np.sum(np.exp(shifted)))
    value = m + temp * logsum
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    return value, weights

def compute_smooth_c5(h, n_points, temp=0.05):
    """Smoothed max of correlation with gradient."""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    shifted = (conv - m) / temp
    exp_shifted = np.exp(shifted)
    logsum = np.log(np.sum(exp_shifted))
    value = m + temp * logsum
    weights = exp_shifted / np.sum(exp_shifted)
    return value, weights, conv

def binary_local_search(h, n_points, n_trials=1000):

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_c5_and_weights(h, n_points, temp):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    shifted = (conv - m) / temp
    max_shifted = np.max(shifted)
    exp_s = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_s)
    value = m + temp * (max_shifted + np.log(sum_exp))
    weights = exp_s / sum_exp
    return value, weights, conv

def compute_smooth_gradient(h, n_points, weights, dx):
    n = n_points
    grad = np.zeros(n)
    # weights has length 2n-1, corresponding to lags -(n-1) to n-1
    for j in range(2 * n - 1):
        if weights[j] < 1e-12:
            continue
        k = j - (n - 1)  # actual lag
        w = weights[j]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def compute_c5_weights(h, n_points, temp):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(conv)
    shifted = (conv - m) / temp
    mx = np.max(shifted)
    exp_s = np.exp(shifted - mx)
    sum_exp = np.sum(exp_s)
    value = m + temp * (mx + np.log(sum_exp))
    weights = exp_s / sum_exp
    return value, weights, conv


def compute_smooth_grad(h, n_points, weights, dx):
    n = n_points
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-12:
            continue
        k = j - (n - 1)
        if k >= 0:
            # d/dh[t]: (1-h[t-k]) for k<=t<=n-1, minus h[t+k] for 0<=t<=n-1-k
            lo1, hi1 = k, n - 1
            if lo1 <= hi1:
                idx = np.arange(lo1, hi1 + 1)
                grad[idx] += w * (1.0 - h[idx - k])
            lo2, hi2 = 0, n - 1 - k
            if lo2 <= hi2:
                idx = np.arange(lo2, hi2 + 1)
                grad[idx] -= w * h[idx + k]
        else:
            p = -k
            lo1, hi1 = 0, n - 1 - p
            if lo1 <= hi1:
                idx = np.arange(lo1, hi1 + 1)
                grad[idx] += w * (1.0 - h[idx + p])
            lo2, hi2 = p, n - 1
            if lo2 <= hi2:
                idx = np.arange(lo2, hi2 + 1)
                grad[idx] -= w * h[idx - p]
    return grad * dx


def project_feasible(h, n_points):
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = h < 1.0 - 1e-15
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] += diff / cnt
            np.minimum(h, 1.0, out=h)
        else:
            free = h > 1e-15
            cnt = np.sum(free)
            if cnt == 0:
                break
            h[free] += diff / cnt
            np.maximum(h, 0.0, out=h)
    np.clip(h, 0.0, 1.0, out=h)
    return h


def adam_smooth_max(h0, n_points, max_iter, temps, time_check, base_lr=0.12):
    n = n_points
    dx = 2.0 / n
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_h = h.copy()
    best_val = compute_c5(h, n)
    t_iter = 0
    no_improve = 0
    patience = 300

    for temp in temps:
        n_it = max_iter // len(temps)
        for it in range(n_it):
            t_iter += 1
            if time_check() < 5:
                return best_h, best_val
            lr_t = base_lr / (1.0 + 0.003 * it)
            val, weights, conv = compute_c5_weights(h, n, temp)
            grad = compute_smooth_grad(h, n, weights, dx)
            m_hat = beta1 * m_hat + (1 - beta1) * grad
            v_hat = beta2 * v_hat + (1 - beta2) * grad * grad
            h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_feasible(h_new, n)
            actual = compute_c5(h_new, n)
            h = h_new
            if actual < best_val:
                best_val = actual
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > patience:
                break

    return best_h, best_val


def binary_hill_climb(h, n_points, rng, n_trials=5000):
    n = n_points
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    s = int(np.sum(h_bin))
    if s > target:
        ones = np.where(h_bin == 1)[0]
        idx = rng.choice(len(ones), s - target, replace=False)
        h_bin[ones[idx]] = 0
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        idx = rng.choice(len(zeros), target - s, replace=False)
        h_bin[zeros[idx]] = 1
    best = h_bin.copy()
    best_val = compute_c5(best, n)
    for _ in range(n_trials):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        new = best.copy()
        new[ones[i]] = 0.0
        new[zeros[j]] = 1.0
        v = compute_c5(new, n)
        if v < best_val - 1e-15:
            best = new
            best_val = v
    return best, best_val


def thue_morse_init(n, rng):
    tm = np.zeros(1 << 20, dtype=int)
    for i in range(1, len(tm)):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm[:n].astype(float)
    h += rng.normal(0, 0.15, n)
    h = np.clip(h, 0, 1)
    return h


def beatty_init(n, alpha, rng):
    h = np.zeros(n)
    ones_count = n // 2
    positions = set()
    for k in range(1, 400):
        pos = int(np.floor(k * alpha))
        if 0 <= pos < n and pos not in positions:
            positions.add(pos)
            if len(positions) == ones_count:
                break
    for pos in positions:
        h[pos] = 1.0
    h += rng.normal(0, 0.1, n)
    h = np.clip(h, 0, 1)
    return h


def slsqp_polish(h0, n_points, maxiter=500):
    n = n_points
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda x: compute_c5(x, n),
                   h0.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x, res.fun
