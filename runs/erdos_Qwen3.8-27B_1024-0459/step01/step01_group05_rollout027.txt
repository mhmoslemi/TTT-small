def thue_morse(n):
    # Generate Thue-Morse sequence of length n
    t = np.zeros(n)
    for i in range(n):
        # Count trailing ones in binary
        count = 0
        x = i
        while x & 1:
            count += 1
            x >>= 1
        t[i] = count % 2
    return t

For n=128: sum(t) = 64 = n/2. ✓ (Since exactly half of 0..127 have even popcount.)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def project_onto_box_sum(h, target):
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_test)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            free = np.where(h_proj < 1.0)[0]
        else:
            free = np.where(h_proj > 0.0)[0]
        if len(free) > 0:
            h_proj[free] += diff / len(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = corr * dx
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = corr * dx
    m = np.max(x)

    e = np.exp(shifted)

    term2 = np.correlate(p, h, mode='full')[n-1:2*n-1][::-1]
    
    return dx * (term1 - term2)

    def fun(h):
        return np.sum(h) - target

        return np.ones(len(h))
    return {'type': 'eq', 'fun': fun, 'jac': jac}
