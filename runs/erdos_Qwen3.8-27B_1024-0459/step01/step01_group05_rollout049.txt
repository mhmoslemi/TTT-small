    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    shifted = np.clip(shifted, -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    shifted = beta * (x - m)
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
