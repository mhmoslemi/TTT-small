            if cur < best:
                best = cur
                best_h = h.copy()
        elif new_max <= cur + 1e-12:
            if rng.random() < 0.05:
                stall += 1
            else:
                # revert
                binary_flip_update(scores, h, i, vi, n)
                binary_flip_update(scores, h, j, vj, n)
                scores = cur_scores
                stall += 1
        else:
            if rng.random() < math.exp(-(new_max - cur) / temp):
                cur = new_max
                stall = 0
            else:
                binary_flip_update(scores, h, i, vi, n)
                binary_flip_update(scores, h, j, vj, n)
                scores = cur_scores
                stall += 1
        if stall > stall_limit:
            break
    final_c = cur * (2.0 / n)
    if final_c < best + 1e-12:
        best = final_c
    return best_h.astype(float), best

def upsample(h_parent, n_target):
    n0 = h_parent.size
    if n0 == n_target:
        return h_parent.copy()
    if n0 <= 1:
        return np.full(n_target, 0.5)
    if n0 > 0 and n_target % n0 == 0:
        return np.repeat(h_parent, n_target // n0).astype(float)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x, x0, h_parent).astype(float)

class State:
    def __init__(self, h, c, n):
        self.h = np.array(h, dtype=float)
        self.c = float(c)
        self.n = int(n)
    def update(self, h, c, n):
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size == 0:
            return
        n = int(n)
        if h.size != n:
            if h.size > n:
                h = h[:n]
            else:
                return
        h2 = project_feasible(h, n)
        c2 = compute_c5(h2)
        if math.isfinite(c2) and c2 < self.c - 1e-15:
            self.h = h2
            self.c = c2
            self.n = n

def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE
    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0:
        budget_s = 30.0
    reserve = 20.0 if budget_s > 60 else max(2.0, 0.1 * budget_s)
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _RUN_DEADLINE = global_deadline
    
    rng = np.random.default_rng(seed)
    
    # Get parent
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim != 1 or parent.size < 2:
            parent = None
    except Exception:
        parent = None
    
    if parent is not None:
        n0 = int(parent.size)
        h0 = project_feasible(parent.copy(), n0)
        c0 = compute_c5(h0)
    else:
        n0 = 70
        h0 = np.full(n0, 0.5)
        # Create a sawtooth-like initial
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h0 = np.clip(0.5 + 0.4 * np.sin(np.pi * x), 0, 1)
        h0 = project_feasible(h0, n0)
        c0 = compute_c5(h0)
    
    state = State(h0, c0, n0)
    
    # Phase 1: Binary local search at n=n0
    if n0 % 2 == 0:
        blim = min(80.0, max(10.0, 0.12 * (global_deadline - t0)))
        blead = t0 + blim
        bh, bc = binary_local_search(h0, n0, rng, blead, 80000, 5000)
        if math.isfinite(bc):
            state.update(bh, bc, n0)
    
    # Phase 2: Nesterov subgradient at n0
    if time.time() < global_deadline - 200:
        nelim = min(250.0, 0.35 * (global_deadline - t0))
        n_lead = t0 + nelim
        n_iters = int(300000.0 * (80.0 / float(n0)))
        n_iters = min(n_iters, 500000)
        nh, nc = nestorov_subgrad(state.h, state.n, n_iters, n_lead)
        if math.isfinite(nc):
            state.update(nh, nc, state.n)
    
    # Phase 3: Multi-stage SLSQP at n0
    if time.time() < global_deadline - 150:
        slim = min(300.0, 0.35 * (global_deadline - t0))
        run_slsqp_multi(state.h, state.n, state, slim)
    
    # Phase 4: Active-set polish at n0
    if time.time() < global_deadline - 80:
        polim = min(120.0, 0.15 * (global_deadline - t0))
        active_set_polish(state.h, state.n, state, polim)
    
    # Phase 5: Try higher resolution
    n1 = 2 * n0
    if n1 % 2 == 0 and time.time() < global_deadline - 100:
        h_up = upsample(state.h, n1)
        c_up = compute_c5(h_up)
        state_up = State(h_up, c_up, n1)
        
        # Quick SLSQP at n1
        if time.time() < global_deadline - 60:
            slim1 = min(150.0, 0.2 * (global_deadline - t0))
            alphas1 = [max(1.0, 0.005 * n1), max(0.3, 0.001 * n1), 0.0]
            bounds1 = [(0.0, 1.0)] * n1
            cons1 = [{"type": "eq", "fun": con_eq, "jac": con_jac}]
            for alpha in alphas1:
                if time.time() > global_deadline - 30:
                    break
                maxiter = max(30, int(150.0 * (80.0 / float(n1))))
                maxiter = min(maxiter, 300)
                sub_dl = min(time.time() + 50.0, global_deadline - 10)
                old_dl = _RUN_DEADLINE
                _RUN_DEADLINE = sub_dl
                try:
                    res = minimize(slsqp_obj, state_up.h, args=(alpha,),
                                   jac=slsqp_grad, method="SLSQP",
                                   bounds=bounds1, constraints=cons1,
                                   options={"maxiter": maxiter, "ftol": 1e-10, "disp": 0})
                except Exception:
                    res = None
                finally:
                    _RUN_DEADLINE = old_dl
                if res is not None and getattr(res, "x", None) is not None:
                    hn = project_feasible(res.x, n1)
                    cn = compute_c5(hn)
                    if math.isfinite(cn) and cn < state_up.c:
                        state_up.h = hn
                        state_up.c = cn
        
        # Compare: is n1 solution better than n0?
        if state_up.c < state.c - 1e-8:
            state = state_up
    
    # Phase 6: Multiple SLSQP restarts at best resolution
    if time.time() < global_deadline - 50:
        best_n = state.n
        for restart in range(3):
            if time.time() > global_deadline - 30:
                break
            # Perturb
            if rng.random() < 0.5:
                hp = project_feasible(state.h + 0.05 * rng.standard_normal(best_n), best_n)
            else:
                hp = project_feasible(state.h + 0.02 * rng.standard_normal(best_n), best_n)
            cp = compute_c5(hp)
            if cp < state.c:
                state.h = hp
                state.c = cp
            # Quick SLSQP
            sub_dl = min(time.time() + 60.0, global_deadline - 15)
            old_dl = _RUN_DEADLINE
            _RUN_DEADLINE = sub_dl
            try:
                bounds_r = [(0.0, 1.0)] * best_n
                cons_r = [{"type": "eq", "fun": con_eq, "jac": con_jac}]
                res = minimize(slsqp_obj, state.h, args=(0.0,),
                               jac=slsqp_grad, method="SLSQP",
                               bounds=bounds_r, constraints=cons_r,
                               options={"maxiter": 200, "ftol": 1e-11, "disp": 0})
            except Exception:
                res = None
            finally:
                _RUN_DEADLINE = old_dl
            if res is not None and getattr(res, "x", None) is not None:
                hn = project_feasible(res.x, best_n)
                cn = compute_c5(hn)
                if math.isfinite(cn) and cn < state.c:
                    state.h = hn
                    state.c = cn
    
    # Final: ensure validity
    h_final = project_feasible(state.h, state.n)
    c_final = compute_c5(h_final)
    n_final = int(state.n)
    
    if not is_valid(h_final):
        h_final = project_feasible(h0, n0)
        c_final = compute_c5(h_final)
        n_final = n0
    
    return h_final, float(c_final), int(n_final)
