```python
def project_to_constraint(y, n):
    """Project y onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    
    def sum_at_tau(tau):
        return np.sum(np.clip(y - tau, 0.0, 1.0))
    
    # Bisection
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_at_tau(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0.0, 1.0)
```

def compute_c5(h):

def smooth_max(corr, beta):
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    return cmax + np.log(np.sum(exp_vals)) / beta

def smooth_max_grad(h, n, beta):
    """Compute smooth max of correlations and its gradient."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    softmax = exp_vals / np.sum(exp_vals)
    obj = cmax + np.log(np.sum(exp_vals)) / beta
