    # Let lag = m - (n-1), so c[m] = sum_i a[i] * v[i - lag] for valid i
    lag = m_idx - (n - 1)
    grad = np.zeros(n)
    # c[m] = sum_i a[i] * v[i - lag] where i in [max(0,lag), min(n-1, n-1+lag)] (for lag>=0)
    # d(c)/da[j]: v[j-lag] if j-lag valid (from a[j]*v[j-lag] term)
    # d(c)/da[j] from v side: -a[j+lag] if j+lag valid (from a[j+lag]*v[j] = a[j+lag]*(1-h[j]))
    # Wait: v = 1-h, so v[j] = 1-h[j]
    # c = sum_i a[i]*(1-h[i-lag]) = sum_i a[i] - sum_i a[i]*h[i-lag]
    # d/dh[j]: from first sum: 1 (since a[j]=h[j])
    #          from second sum: -a[j]*h[j-lag] (i=j term) - a[j+lag]*h[j] (i-lag=j term, i=j+lag)
    # But a[i]=h[i] and a[j+lag]=h[j+lag]
    # So d/dh[j] = 1 - h[j]*h[j-lag] - h[j+lag]*h[j]  (valid indices only)
    for j in range(n):
        val = 0.0
        # From sum_i h[i]: contributes 1 always
        val += 1.0
        # From -sum_i h[i]*h[i-lag]:
        # Term where i=j: -h[j]*h[j-lag] if 0<=j-lag<n
        if 0 <= j - lag < n:
            val -= h[j] * h[j - lag]
        # Term where i-lag=j, i=j+lag: -h[j+lag]*h[j] if 0<=j+lag<n
        if 0 <= j + lag < n:
            val -= h[j + lag] * h[j]
        # But we also need: the sum over i is only over valid i where both i and i-lag are in [0,n-1]
        # This is already handled by the conditions above
        grad[j] = val
    return grad


def active_set_optimize(h, n, n_iter=200, step_decay=0.98, base_step=0.5):
    """Subgradient method focusing on worst lags."""
    h = h.copy()
    dx = 2.0 / n
    step = base_step
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Find top 3 lags
        top3 = np.argsort(corr)[-3:]
        # Average gradient of top 3
        grad = np.zeros(n)
        for k_idx in top3:
            grad += np.sign(corr[k_idx] - corr[k_idx]) * 1  # placeholder
            g = grad_c_at_lag(h, n, k_idx)
            grad += g
        grad /= 3.0
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        # Step in negative gradient direction
        h_new = h - step * grad / grad_norm
        # Project onto [0,1]^n
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum
        h_new = fix_sum(h_new, n)
        c_old = np.max(corr) * dx
        c_new = compute_c5(h_new)
        if c_new < c_old:
            h = h_new
        else:
            step *= step_decay
            if step < 1e-6:
                break
    return h


def nm_obj(x, n, dx):
    """Objective for Nelder-Mead (actual non-smooth C5)."""
    h = np.clip(x, 0.0, 1.0)
    h = fix_sum(h, n)
    return compute_c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    # Get parent
    if 'initial_h_values' in dir() and initial_h_values is not None and np.size(initial_h_values) > 0:
        parent_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        parent_h = np.full(180, 0.5)

    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    n_parent = len(parent_h)

    # Phase 1: Nelder-Mead at lower resolution (n=60 or 90) for global exploration
    n_nm = min(60, n_parent)
    x0_nm = upsample_to(parent_h, n_nm)
    x0_nm = fix_sum(x0_nm, n_nm)
    dx_nm = 2.0 / n_nm
    elapsed = time.time() - t0

    if elapsed < budget_s * 0.15:
        # Multiple Nelder-Mead restarts at n_nm
        starts = [x0_nm]
        # Add periodic patterns
        for pf, df in [(1/3, 0.5), (1/4, 0.5), (2/5, 0.5), (1/2, 0.5), (3/7, 0.5), (5/9, 0.5)]:
            starts.append(periodic_init(n_nm, pf, df))
        # Add a few random
        for _ in range(3):
            hr = rng.random(n_nm)
            hr = fix_sum(hr, n_nm)
            starts.append(hr)

        for start in starts:
            if time.time() - t0 > budget_s * 0.15:
                break
            start = fix_sum(start, n_nm)
            try:
                res = minimize(
                    nm_obj, start, args=(n_nm, dx_nm),
                    method='Nelder-Mead',
                    options={'maxiter': 500, 'xatol': 1e-6, 'fatol': 1e-9}
                )
                h_nm = fix_sum(np.clip(res.x, 0, 1), n_nm)
                c_nm = compute_c5(h_nm)
                if c_nm < best_c5:
                    # Upsample to parent resolution
                    h_up = upsample_to(h_nm, n_parent)
                    h_up = fix_sum(h_up, n_parent)
                    c_up = compute_c5(h_up)
                    if c_up < best_c5:
                        best_c5 = c_up
                        best_h = h_up
                    else:
                        best_c5 = c_nm
                        best_h = h_nm
            except Exception:
                pass

    # Phase 2: SLSQP refinement at parent resolution
    n2 = n_parent
    dx2 = 2.0 / n2
    x0_2 = fix_sum(best_h, n2) if len(best_h) == n2 else fix_sum(upsample_to(best_h, n2), n2)
    bounds2 = [(0.0, 1.0)] * n2
    elapsed = time.time() - t0

    if elapsed < budget_s * 0.55:
        tau_list = [0.1, 0.05, 0.03, 0.015, 0.01, 0.005, 0.003]
        maxiter_list = [2000, 2000, 2000, 2000, 2500, 2500, 3000]
        for tau, maxit in zip(tau_list, maxiter_list):
            if time.time() - t0 > budget_s * 0.55:
                break
            obj_p = functools.partial(smooth_max_obj, n=n2, dx=dx2, tau=tau)
            con_p = functools.partial(eq_con, n=n2)
            jac_p = functools.partial(eq_jac, n=n2)
            try:
                res = minimize(
                    fun=obj_p, x0=x0_2, method='SLSQP',
                    bounds=bounds2,
                    constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
                    options={'maxiter': maxit, 'ftol': 1e-13, 'disp': False}
                )
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n2)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    x0_2 = h_new
                else:
                    x0_2 = h_new  # still use as next start even if not better
            except Exception:
                pass

        # Subgradient refinement at n2
        if time.time() - t0 < budget_s * 0.60:
            h_sg = active_set_optimize(best_h, n2, n_iter=100, step_decay=0.95, base_step=0.3)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg
            # SLSQP from subgradient result
            if time.time() - t0 < budget_s * 0.65:
                tau = 0.005
                obj_p = functools.partial(smooth_max_obj, n=n2, dx=dx2, tau=tau)
                con_p = functools.partial(eq_con, n=n2)
                jac_p = functools.partial(eq_jac, n=n2)
                try:
                    res = minimize(
                        fun=obj_p, x0=h_sg, method='SLSQP',
                        bounds=bounds2,
                        constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
                        options={'maxiter': 2000, 'ftol': 1e-13, 'disp': False}
                    )
                    h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n2)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                except Exception:
                    pass

    # Phase 3: Higher resolution (2x parent)
    n3 = 2 * n2
    dx3 = 2.0 / n3
    x0_3 = fix_sum(upsample_to(best_h, n3), n3)
    bounds3 = [(0.0, 1.0)] * n3
    elapsed = time.time() - t0

    if elapsed < budget_s * 0.85:
        tau_list3 = [0.05, 0.02, 0.01, 0.005, 0.002]
        maxiter3 = [2000, 2000, 2500, 2500, 3000]
        for tau, maxit in zip(tau_list3, maxiter3):
            if time.time() - t0 > budget_s * 0.85:
                break
            obj_p = functools.partial(smooth_max_obj, n=n3, dx=dx3, tau=tau)
            con_p = functools.partial(eq_con, n=n3)
            jac_p = functools.partial(eq_jac, n=n3)
            try:
                res = minimize(
                    fun=obj_p, x0=x0_3, method='SLSQP',
                    bounds=bounds3,
                    constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
                    options={'maxiter': maxit, 'ftol': 1e-13, 'disp': False}
                )
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n3)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    x0_3 = h_new
                else:
                    x0_3 = h_new
            except Exception:
                pass

        # Perturbation restarts at higher resolution
        n_pert = 3
        for r in range(n_pert):
            if time.time() - t0 > budget_s * 0.88:
                break
            noise = rng.normal(0, 0.05, n3)
            x_pert = fix_sum(np.clip(x0_3 + noise, 0.0, 1.0), n3)
            tau = 0.01
            obj_p = functools.partial(smooth_max_obj, n=n3, dx=dx3, tau=tau)
            con_p = functools.partial(eq_con, n=n3)
            jac_p = functools.partial(eq_jac, n=n3)
            try:
                res = minimize(
                    fun=obj_p, x0=x_pert, method='SLSQP',
                    bounds=bounds3,
                    constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
                    options={'maxiter': 1500, 'ftol': 1e-12, 'disp': False}
                )
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n3)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    x0_3 = h_new
            except Exception:
                pass

    # Phase 4: Final polish with very small tau and noise
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    bounds_best = [(0.0, 1.0)] * n_best
    elapsed = time.time() - t0

    if elapsed < budget_s * 0.96:
        # Multiple tiny perturbations + SLSQP
        for r in range(20):
            if time.time() - t0 > budget_s * 0.97:
                break
            sigma = 0.01 if r < 10 else 0.005
            small = rng.normal(0, sigma, n_best)
            x_pert = fix_sum(np.clip(best_h + small, 0.0, 1.0), n_best)
            tau = 0.002
            obj_p = functools.partial(smooth_max_obj, n=n_best, dx=dx_best, tau=tau)
            con_p = functools.partial(eq_con, n=n_best)
            jac_p = functools.partial(eq_jac, n=n_best)
            try:
                res = minimize(
                    fun=obj_p, x0=x_pert, method='SLSQP',
                    bounds=bounds_best,
                    constraints=[{'type': 'eq', 'fun': con_p, 'jac': jac_p}],
                    options={'maxiter': 1000, 'ftol': 1e-14, 'disp': False}
                )
                h_new = fix_sum(np.clip(res.x, 0.0, 1.0), n_best)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

        # Binary rounding attempts
        for r in range(5):
            if time.time() - t0 > budget_s * 0.98:
                break
            h_rounded = (best_h > 0.5).astype(float)
            h_rounded = fix_sum(h_rounded, n_best)
            c5_rounded = compute_c5(h_rounded)
            if c5_rounded < best_c5:
                best_c5 = c5_rounded
                best_h = h_rounded.copy()
            # Soft rounding: push values toward 0 or 1
            h_pushed = np.where(best_h > 0.3, 1.0, 0.0)
            h_pushed = fix_sum(h_pushed, n_best)
            c5_pushed = compute_c5(h_pushed)
            if c5_pushed < best_c5:
                best_c5 = c5_pushed
                best_h = h_pushed.copy()

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
