import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    e = np.exp(shifted)
    return e / np.sum(e)

def project_constraint(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}"""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    # Bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix numerical issues
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if np.any(free):
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    obj = smooth_max(corr, beta)
    weights = smooth_max_weights(corr, beta)
    
    # Gradient: grad[i] = sum_k weights[k] * d(corr[k]*dx)/dh[i]
    # corr[k] = dx * sum_j h[j] * (1-h[j+k-(n-1)])
    # d(corr[k])/dh[i] = dx * [(1-h[i+k-(n-1)]) - h[i-k+(n-1)]] (with bounds)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue

        j_hi = min(n - 1, n - 1 + k)

        h_idx = j_idx + k - (n - 1)

        # d(corr[k])/dh[i] for i in h_idx: -h[i-k+(n-1)]
        grad[h_idx] -= w * dx * h[j_idx]
    
    return obj, grad

            n_best = len(h)
            h = best_h + np.random.randn(n_best) * 0.05
            h = project_constraint(h, target)
    
    return best_h, best_obj

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max_vals(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    e = np.exp(shifted)
    se = np.sum(e)
    return m + np.log(se) / beta, e / se

def project_constraint(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    # Bisection on shift lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_p = np.clip(h - lam, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-10:

        if np.any(free):

            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

    n = len(h)
    dx = 2.0 / n

    corr_raw = np.correlate(h, h_comp, mode='full')

        h_idx = j_idx + k - (n - 1)
        grad[j_idx] += w * dx * (1.0 - h[h_idx])
        grad[h_idx] -= w * dx * h[j_idx]
    
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return np.max(corr) * dx

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    e = np.exp(shifted)
    se = np.sum(e)
    return m + np.log(se) / beta, e / se

def project_constraint(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-10:
        free = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        if np.any(free):
            h_p[free] += diff / float(np.sum(free))
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    h_comp = 1.0 - h
    corr_raw = np.correlate(h, h_comp, mode='full')

    grad = np.zeros(n)

        j_lo = max(0, n - 1 - k)
        j_hi = min(n - 1, n - 1 + k)

        grad[j_idx] += w * dx * (1.0 - h[h_idx])
