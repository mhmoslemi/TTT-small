def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    """Project x to [0,1]^n with sum = target using bisection."""
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(h))
    if abs(res) > 1e-15:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
    # Final fix
    res = target - float(np.sum(h))
    if abs(res) > 1e-15:
        for idx in range(n):
            if 0.0 < h[idx] < 1.0:
                h[idx] = np.clip(h[idx] + res, 0.0, 1.0)
                break
    return h

    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

            best_h, best_c, best_n = h2, c2, n1
    
    # Phase 2: Multi-resolution optimization
    resolutions = []
    for mult in (2, 4, 6, 8, 12, 16):
        nn = int(n1 * mult)
        if nn <= 800:
            resolutions.append(nn)
    for extra in (180, 360, 540, 720):
        if n1 <= extra <= 800:
            if extra not in resolutions:
                resolutions.append(extra)
    resolutions = sorted(set(resolutions))
    
    for ni in resolutions:
        rem = deadline - time.perf_counter()
        if rem < 40.0:
            break
        
        # Determine time allocation
        n_res = len([r for r in resolutions if r >= ni and (deadline - time.perf_counter()) > 40])
        per_res = min(rem / max(1, n_res), 350.0)
        per_res = max(25.0, per_res)
        sub_dl = time.perf_counter() + min(per_res, rem - 5.0)
        
        # Generate starts
        starts = []
        starts.append(upsample_h(best_h, ni))
        if parent is not None:
            starts.append(upsample_h(parent, ni))
        for mm in range(2):
            starts.append(make_random_start(ni, rng, mm))
        
        seen = []
        uniq_starts = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in seen:
                if np.allclose(s, u, atol=0.01):
                    dup = True
                    break
            if not dup:
                uniq_starts.append(s)
                seen.append(s)
        
        if not uniq_starts:
            continue
        
        for si, sv in enumerate(uniq_starts):
            rem2 = deadline - time.perf_counter()
            if rem2 < 20.0:
                break
            left = max(1, len(uniq_starts) - si)
            allot = min(rem2 / left, 200.0)
            if allot < 12.0:
                break
            ssub_dl = time.perf_counter() + allot
            
            cur_h = sv.copy()
            cur_c = evaluate_erdos(cur_h)
            
            # Momentum gradient (fast for high n)
            if ni > 100 and ssub_dl - time.perf_counter() > 15.0:
                mh, mc = momentum_optimize(cur_h, ni, ssub_dl - 15.0, k_active=max(3, ni // 50), max_iters=30000)
                if mc < cur_c:
                    cur_h, cur_c = mh, mc
            
            # Local search
            if ssub_dl - time.perf_counter() > 10.0:
                lh, lc = pair_local_search(cur_h, ni, ssub_dl, max_iters=5000, k_active=5, use_random=0.2)
                if lc < cur_c:
                    cur_h, cur_c = lh, lc
            
            # Triple moves for finer tuning
            if ni <= 200 and ssub_dl - time.perf_counter() > 8.0:
                corr_tmp = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                kt = min(5, 2 * ni - 1)
                idxs_t = np.argsort(corr_tmp)[-kt:]
                g_t = corr_gradient(cur_h, idxs_t) / float(kt)
                g_t -= np.mean(g_t)
                th, tc, tf = interpolated_triple_move(cur_h, ni, g_t, corr_tmp, kt, ssub_dl)
                if tf and tc < cur_c:
                    cur_h, cur_c = th, tc
            
            cur_h = project_density(cur_h, 0.5 * ni)
            cur_c = evaluate_erdos(cur_h)
            if cur_c < best_c:
                best_h, best_c, best_n = cur_h, cur_c, ni
    
    # Phase 3: Final intensive polish at best resolution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 30.0:
        nb = int(best_h.size)
        if rem > 60.0:
            h3, c3 = pair_local_search(best_h, nb, deadline - 20.0, max_iters=20000, k_active=8, use_random=0.5)
            if c3 < best_c:
                best_h, best_c, best_n = h3, c3, nb
        if deadline - time.perf_counter() > 15.0:
            h4, c4 = momentum_optimize(best_h, nb, deadline - 10.0, k_active=6, max_iters=15000)
            if c4 < best_c:
                best_h, best_c, best_n = h4, c4, nb
        if deadline - time.perf_counter() > 10.0:
            h5, c5 = pair_local_search(best_h, nb, deadline - 5.0, max_iters=5000, k_active=6, use_random=0.1)
            if c5 < best_c:
                best_h, best_c, best_n = h5, c5, nb
    
    # Final validation
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
