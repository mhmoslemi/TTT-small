import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h > 1e-12
        else:
            mask = h < 1.0 - 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] -= diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) > 1e-13:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() > 0:
            h[mask] -= diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    val = (m + np.log(np.sum(np.exp(scaled - m)))) / beta
    return float(val * (2.0 / n))


def smooth_grad(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-13:
            continue
        lag = abs(k - (n - 1))
        if lag == 0:
            continue
        lim = n - lag
        grad[:lim] += wk * (1.0 - h[lag:lag + lim])
        grad[lag:] -= wk * h[:lim]
    return grad * dx


def hard_obj(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / n))


def hard_grad(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(c)
    lag = abs(k_max - (n - 1))
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        lim = n - lag
        grad[:lim] += (1.0 - h[lag:lag + lim])
        grad[lag:] -= h[:lim]
    return grad * dx


def multi_lag_grad(h, n, top_k=5):
    c = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(c)[::-1][:top_k]
    top_vals = c[top_idx]
    beta_local = 20.0
    exp_vals = beta_local * (top_vals - top_vals[0])
    weights = np.exp(exp_vals)
    weights /= np.sum(weights)
    dx = 2.0 / n
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        lag = abs(int(ki) - (n - 1))
        if lag == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        else:
            lim = n - lag
            grad[:lim] += wi * (1.0 - h[lag:lag + lim])
            grad[lag:] -= wi * h[:lim]
    return grad * dx


def sum_eq_fun(h, n):
    return np.sum(h) - n / 2.0


def sum_eq_jac(h, n):
    return np.ones(n)


def slsqp_smooth(h, n, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(
            fun=smooth_obj, x0=h, jac=smooth_grad, args=(n, beta),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-13}
        )
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        return h_new
    except Exception:
        return h.copy()


def slsqp_hard(h, n, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(
            fun=hard_obj, x0=h, jac=hard_grad, args=(n,),
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h_new = project_feasible(np.clip(res.x, 0.0, 1.0), n)
        return h_new
    except Exception:
        return h.copy()


def projected_subgradient(h, n, n_iters=2000, top_k=3, t_start=None):
    best_h = h.copy()
    best_c5 = hard_obj(h, n)
    vel = np.zeros(n)
    lr = 0.005
    gamma = 0.9
    dx = 2.0 / n

    for it in range(n_iters):
        if t_start is not None and time.time() - t_start > 1280:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        c5_val = np.max(c) * dx
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()

        active_thresh = c[-1] + 0.03 * (c[0] - c[-1] + 1e-15) if c[0] != c[-1] else c[0]
        active_mask = c > active_thresh
        if not active_mask.any():
            active_mask[np.argmax(c)] = True
        n_act = active_mask.sum()

        grad = np.zeros(n)
        for k in np.where(active_mask)[0]:
            lag = abs(int(k) - (n - 1))
            if lag == 0:
                grad[:] += (1.0 - 2.0 * h)
            else:
                lim = n - lag
                grad[:lim] += (1.0 - h[lag:lag + lim])
                grad[lag:] -= h[:lim]
        grad *= dx / n_act

        vel = gamma * vel + lr * grad
        h_trial = h - vel
        h_trial = project_feasible(h_trial, n)

        trial_c5 = hard_obj(h_trial, n)

            h = h_trial
            lr = min(lr * 1.005, 0.05)
        else:
            lr *= 0.85
            vel *= 0.5

    return best_h, best_c5

    best_c5 = compute_c5(best_h)

    # Verify with evaluator
    try:
        val = evaluate_erdos_solution(best_h, best_n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
