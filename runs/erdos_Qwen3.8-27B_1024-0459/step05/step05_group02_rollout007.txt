def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def gradient_directed_binary_search(h_bin, n, max_iters=20000, top_k=5, rng=None):
    """Binary search guided by gradient of the max correlation."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_bin.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        
        # Compute gradient
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h, ki, n)
        g -= np.mean(g)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Greedy: find best swap
        # Want to move 1 to where g is most negative (among zeros)
        # and move 0 to where g is most positive (among ones)
        best_delta = 0
        best_swap = None
        
        # Top candidates
        n_try = min(20, len(ones))
        n_try_z = min(20, len(zeros))
        
        # Score: for ones, we want to move them to low-g zeros
        # For zeros, we want to move them to high-g ones
        top_ones = ones[np.argsort(g[ones])[:n_try]]  # most negative g
        top_zeros = zeros[np.argsort(-g[zeros])[:n_try_z]]  # most positive g
        
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                h[i] = 1
                h[j] = 0
                delta = nc - cur_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_swap = (i, j)
        
        # If no improvement found, try random swaps
        if best_swap is None:
            found = False

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TOPK = 5
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 100.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 200.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=5, maxiter=800):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 10})
        return proj(res.x, n)
    except Exception:
        return h
