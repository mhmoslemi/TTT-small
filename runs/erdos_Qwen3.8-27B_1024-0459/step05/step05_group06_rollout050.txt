def subgrad_step(h, n, n_active=5):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.5)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

    trial_resolutions = sorted(set(trial_resolutions))

    for rn in trial_resolutions:
        if remaining_time() < 50:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        
        # Quick SLSQP
        for beta in [5000, 50000]:
            if remaining_time() < 15:
                break
            h_r = slsqp_optimize(h_r, rn, beta, 250)
        
        # Quick coordinate descent
        if remaining_time() > 15:
            h_r, c5_r = coord_descent_exact(h_r, rn, n_sweeps=15)
        
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 7: Subgradient + SLSQP interleaved
    n = best_n
    h = best_h.copy()
    for it in range(300):
        if remaining_time() < 10:
            break
        grad = subgrad_step(h, n, 5)
        if grad is None:
            break
        base_step = 0.03 / (1.0 + it * 0.003)
        improved = False
        for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 0.1, 3.0, 0.05]:
            step = base_step * sm
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved:
            if it % 50 == 25 and remaining_time() > 25:
                h_new = slsqp_optimize(h, n, 80000, 250)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                if it > 200:
                    break

    # Phase 8: Random restarts with SLSQP
    n = best_n
    n_perturb = max(5, n // 50)
    for pert in range(20):
        if remaining_time() < 20:
            break
        h_pert = best_h.copy()
        idx = rng.choice(n, n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.1, 0.1, n_pert)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.003:
            h_new = slsqp_optimize(h_pert, n, 50000, 250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if remaining_time() > 15:
                h_new2, c5_new2 = coord_descent_exact(h_new if c5_new < best_c5 else h_pert, n, n_sweeps=10)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2.copy()

    # Phase 9: Final intensive coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = coord_descent_exact(best_h, best_n, n_sweeps=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 8:
        h_new = slsqp_optimize(best_h, best_n, 500000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: One more coordinate descent
    if remaining_time() > 5:
        h_new, c5_new = coord_descent_exact(best_h, best_n, n_sweeps=30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint is met exactly
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    # Final recompute
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
