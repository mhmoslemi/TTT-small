            n_scan = 12
            ts_scan = np.linspace(0.0, 1.0, n_scan)
            f_scan = np.array([eval_f(t) for t in ts_scan])
            idx_min = int(np.argmin(f_scan))

            # Determine bracket for golden section
            if idx_min == 0:
                lo, hi = 0.0, ts_scan[1]
            elif idx_min == n_scan - 1:
                lo, hi = ts_scan[-2], 1.0
            else:
                lo = ts_scan[max(0, idx_min - 1)]
                hi = ts_scan[min(n_scan - 1, idx_min + 1)]

            # Golden section refinement
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_f(c1)
            f2 = eval_f(c2)
            for _ in range(45):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = eval_f(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_f(c2)

            t_gs = (lo + hi) / 2.0
            f_gs = eval_f(t_gs)

            # Check key candidates
            t_best, f_best = t_gs, f_gs
            f0 = eval_f(0.0)
            if f0 < f_best:
                t_best, f_best = 0.0, f0
            f1b = eval_f(1.0)
            if f1b < f_best:
                t_best, f_best = 1.0, f1b
            f05 = eval_f(0.5)
            if f05 < f_best:
                t_best, f_best = 0.5, f05

            if f_best < f_old - 1e-15:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                improved = True

        if not improved:
            break

    return best, compute_c5(best)


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def _g_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]


def _g_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]


def _g_con_fun(x):
    return np.sum(x) - _N_WORK / 2.0


def _g_con_jac(x):
    return np.ones(_N_WORK)


def slsqp_refine(h, n, beta, maxiter=500):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = float(beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_con_fun, 'jac': _g_con_jac}]
    try:
        res = minimize(_g_obj, h, jac=_g_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-28, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def binarize_h(h, n):
    """Project h to binary: keep n/2 largest values as 1."""
    h = np.array(h, dtype=float).copy()
    target_ones = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target_ones]] = 1.0
    return h_bin


def active_set_slsqp(h, n, max_iters=20, n_active=10):
    """Active-set method: minimize max correlation by adding constraints iteratively."""
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n

    for iteration in range(max_iters):
        if remaining_time() < 10:
            break

        corr = compute_corr(best)
        active = np.argsort(corr)[::-1][:n_active]
        active_set = set(int(k) for k in active)

        # Build constraints for active lags
        def make_con_fun(k_idx):
            def con_fun(x):
                c = np.correlate(x, 1.0 - x, mode='full')
                return c[k_idx] * dx - c.max()
            return con_fun

        # Alternative: minimize t with constraints corr_k * dx <= t
        # Use a simpler approach: smooth max with high beta on just active set
        beta_active = 50000.0

        def active_obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            c_active = c[list(active)]
            m = np.max(c_active)
            shifted = np.minimum(beta_active * (c_active - m), 0.0)
            exp_vals = np.exp(shifted)
            wsum = exp_vals.sum()
            return (m + np.log(wsum) / beta_active) * dx

        def active_obj_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            c_active = c[list(active)]
            m = np.max(c_active)
            shifted = np.minimum(beta_active * (c_active - m), 0.0)
            exp_vals = np.exp(shifted)
            wsum = exp_vals.sum()
            weights = exp_vals / wsum
            grad = np.zeros(n)
            for idx, k in enumerate(active):
                w = weights[idx]
                if w < 1e-15:
                    continue
                offset = int(k) - (n - 1)
                if offset >= 0:
                    npairs = n - offset
                    if npairs > 0:
                        grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                        grad[:npairs] -= w * x[offset:offset + npairs]
                else:
                    aoff = -offset
                    npairs = n - aoff
                    if npairs > 0:
                        grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                        grad[aoff:aoff + npairs] -= w * x[:npairs]
            grad *= dx
            return grad

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]

        try:
            res = minimize(active_obj, best, jac=active_obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-25, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                trial = project_feasible(res.x, n)
                c5_trial = compute_c5(trial)
                if c5_trial < best_c5 - 1e-15:
                    best = trial
                    best_c5 = c5_trial
                    n_active = min(n_active + 2, 20)
                else:
                    break
        except Exception:
            break

    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            best_h[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Coordinate descent polish at original resolution
    h = best_h.copy()
    h, c5 = fast_coord_descent(h, n, max_sweeps=40)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP at original resolution
    h = best_h.copy()
    for beta in [2000, 10000, 50000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_refine(h, n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 3: Active-set SLSQP at original resolution
    if remaining_time() > 30:
        h_new, c5_new = active_set_slsqp(best_h, best_n, max_iters=10, n_active=8)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Upsample to n=1280
    if remaining_time() > 120:
        high_n = 1280
        h_high = interp_to(best_h, best_n, high_n)
        h_high = project_feasible(h_high, high_n)
        c5_high = compute_c5(h_high)

        # Coordinate descent at high resolution
        h_new, c5_new = fast_coord_descent(h_high, high_n, max_sweeps=80)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = high_n
        else:
            h_new = h_high

        # SLSQP at high resolution
        if remaining_time() > 60:
            h_slq = slsqp_refine(h_new, high_n, 5000, 500)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()
                best_n = high_n
                h_new = h_slq

        # More coordinate descent
        if remaining_time() > 40:
            h_new, c5_new = fast_coord_descent(h_new, high_n, max_sweeps=60)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = high_n

        # Active-set at high resolution
        if remaining_time() > 40:
            h_new, c5_new = active_set_slsqp(best_h, best_n, max_iters=8, n_active=10)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: High beta SLSQP refinement
    if remaining_time() > 50:
        h = best_h.copy()
        for beta in [20000, 100000, 500000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_refine(h, best_n, beta, 600)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new

    # Phase 6: Perturbation-based restarts
    if remaining_time() > 80:
        for p in range(12):
            if remaining_time() < 25:
                break
            h_pert = best_h.copy()
            noise_scale = 0.08 / (1.0 + p * 0.4)
            mask = rng.random(best_n) < 0.25
            n_mask = np.sum(mask)
            if n_mask > 0:
                h_pert[mask] += rng.randn(n_mask) * noise_scale
            h_pert = project_feasible(h_pert, best_n)
            h_new, c5_new = fast_coord_descent(h_pert, best_n, max_sweeps=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 30:
                    h_slq = slsqp_refine(h_new, best_n, 20000, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()

            # Also try structured perturbation: flip a contiguous block
            if remaining_time() > 30:
                block_len = rng.randint(best_n // 8, best_n // 3)
                start = rng.randint(0, max(1, best_n - block_len))
                h_struct = best_h.copy()
                h_struct[start:start + block_len] = 1.0 - h_struct[start:start + block_len]
                h_struct = project_feasible(h_struct, best_n)
                h_new, c5_new = fast_coord_descent(h_struct, best_n, max_sweeps=40)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 7: Try n=2560 if time permits
    if remaining_time() > 100 and best_n < 2560:
        ultra_n = 2560
        h_ultra = interp_to(best_h, best_n, ultra_n)
        h_ultra = project_feasible(h_ultra, ultra_n)
        h_new, c5_new = fast_coord_descent(h_ultra, ultra_n, max_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = ultra_n
            if remaining_time() > 40:
                h_slq = slsqp_refine(h_new, ultra_n, 10000, 400)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    best_n = ultra_n
                    h_new, c5_new = fast_coord_descent(h_new, ultra_n, max_sweeps=40)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = ultra_n

    # Phase 8: Final coordinate descent
    if remaining_time() > 10:
        h_new, c5_new = fast_coord_descent(best_h, best_n, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 8:
        h_new = slsqp_refine(best_h, best_n, 200000, 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One more coord descent
    if remaining_time() > 5:
        h_new, c5_new = fast_coord_descent(best_h, best_n, max_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
