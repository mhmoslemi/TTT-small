                best_c5 = c5_t
                best_h = h.copy()

        if rem() > 10:
            t_tgt = best_c5 * 0.992
            h = penalty_opt(h, n, t_tgt, maxiter=600)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        if rem() > 10:
            h, c5_t = equalize_shifts(h, n, K=10, max_iters=200)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

    # Phase 10: Fine-tuned subgradient
    if rem() > 10:
        for it in range(200):
            if rem() < 3:
                break
            corr = compute_corr(best_h)
            idx = np.argsort(-corr)[:12]
            ct = corr[idx]
            w = np.exp(100.0 * (ct - ct.max()))
            w /= w.sum()
            g = np.zeros(best_n)
            for i in range(12):
                if w[i] > 1e-12:
                    g += w[i] * grad_corr_k(best_h, idx[i], best_n)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            for step in [0.005, 0.01, 0.002, 0.008, 0.015]:
                ht = project_feasible(best_h - step * g / gn, best_n)
                ct2 = compute_c5(ht)
                if ct2 < best_c5:
                    best_c5 = ct2
                    best_h = ht.copy()
                    break
                elif it % 20 == 19:
                    best_h = project_feasible(best_h - step * 0.5 * g / gn, best_n)
                    best_c5 = compute_c5(best_h)

    # Final SLSQP polish
    for beta in [200000, 500000]:
        if rem() < 5:
            break
        h_fin = slsqp_smax(best_h, best_n, beta, maxiter=500)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
