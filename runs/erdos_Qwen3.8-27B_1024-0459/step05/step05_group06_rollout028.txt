            break

    return best, best_c5

def coord_descent_vectorized(h, n, n_sweeps=100):
    """A slightly different CD that processes all coordinates, updating corr incrementally."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    grid = np.linspace(0.0, 1.0, 9)

    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        improved_any = False

        for i in range(n):
            t0 = best[i]
            if t0 < 1e-12 or t0 > 1.0 - 1e-12:
                continue

            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            a[n - 1] += 1.0
            b[n - 1] += -1.0
            lo_off = max(1, n - i)
            hi_off = min(n - 1, i + n - 1)
            for off in range(1, n):
                if i + off < n:
                    a[n - 1 + off] += (1.0 - best[i + off])
                if i - off >= 0:
                    a[n - 1 - off] += (-best[i - off])

            base = best_corr - a * t0 - b * t0 * t0

            best_t = t0
            best_f = np.max(base + a * t0 + b * t0 * t0)
            for t in grid:
                f = np.max(base + a * t + b * t * t)
                if f < best_f - 1e-15:
                    best_f = f
                    best_t = t

            if best_t != t0:
                lo = max(0.0, best_t - 0.12)
                hi = min(1.0, best_t + 0.12)
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1)
                f2 = np.max(base + a * c2 + b * c2 * c2)
                for _ in range(40):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2)
                t_opt = (lo + hi) / 2.0
                f_opt = np.max(base + a * t_opt + b * t_opt * t_opt)
                if f_opt < best_f:
                    best_f = f_opt
                    best_t = t_opt

            if f_opt * dx < best_c5 - 1e-14:
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved_any = True

        if not improved_any:
            break

    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgrad_step(h, n, n_active=5):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.5)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

def random_restart(h, n, rng, pert_size=0.1, n_pert=None):
    if n_pert is None:
        n_pert = max(5, n // 50)
    h_pert = h.copy()
    idx = rng.choice(n, n_pert, replace=False)
    h_pert[idx] += rng.uniform(-pert_size, pert_size, n_pert)
    h_pert = project_feasible(h_pert, n)
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Intensive coordinate descent at initial resolution
    # Use many sweeps since this is the main workhorse
    h_new, c5_new = coord_descent(h, n, n_sweeps=1500)
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h_new.copy()
        h = h_new.copy()

    # Phase 2: SLSQP with moderate beta
    if remaining_time() > 30:
        h_slsqp = slsqp_optimize(h, n, 5000, 300)
        c5_slsqp = compute_c5(h_slsqp)
        if c5_slsqp < best_c5:
            best_c5 = c5_slsqp
            best_h = h_slsqp.copy()
            h = h_slsqp.copy()
        # Post-SLSQP coordinate descent
        h_new, c5_new = coord_descent(h, n, n_sweeps=500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: SLSQP with higher beta
    if remaining_time() > 30:
        h_slsqp = slsqp_optimize(h, n, 50000, 300)
        c5_slsqp = compute_c5(h_slsqp)
        if c5_slsqp < best_c5:
            best_c5 = c5_slsqp
            best_h = h_slsqp.copy()
            h = h_slsqp.copy()
        h_new, c5_new = coord_descent(h, n, n_sweeps=500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Interpolate to higher resolution and optimize
    n2 = n * 2
    if n2 % 2 != 0:
        n2 += 1
    if n2 <= 4096 and remaining_time() > 60:
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)

        # Coordinate descent at higher resolution
        h2, c5_2 = coord_descent(h2, n2, n_sweeps=800)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

        # SLSQP at higher resolution
        if remaining_time() > 40:
            h2 = slsqp_optimize(h2, n2, 20000, 200)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

        # More coordinate descent
        if remaining_time() > 30:
            h2, c5_2 = coord_descent(h2, n2, n_sweeps=500)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

    # Phase 5: Try another resolution level
    n3 = best_n * 2
    if n3 <= 6000 and remaining_time() > 80:
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        h3, c5_3 = coord_descent(h3, n3, n_sweeps=500)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3
        if remaining_time() > 40:
            h3 = slsqp_optimize(h3, n3, 30000, 150)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n3
        if remaining_time() > 20:
            h3, c5_3 = coord_descent(h3, n3, n_sweeps=300)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n3

    # Phase 6: Random restarts at best resolution
    n = best_n
    h = best_h.copy()
    n_restarts = 8
    for r in range(n_restarts):
        if remaining_time() < 40:
            break
        h_pert = random_restart(h, n, rng, pert_size=0.08, n_pert=max(5, n // 40))
        h_pert, c5_pert = coord_descent(h_pert, n, n_sweeps=300)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            h = h_pert.copy()

    # Phase 7: Subgradient + coordinate descent interplay
    if remaining_time() > 60:
        h = best_h.copy()
        n = best_n
        for it in range(200):
            if remaining_time() < 15:
                break
            grad = subgrad_step(h, n, 5)
            if grad is None:
                break
            base_step = 0.03 / (1.0 + it * 0.01)
            improved = False
            for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 0.1]:
                step = base_step * sm
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                if it % 50 == 25 and remaining_time() > 40:
                    h, c5_new = coord_descent(h, n, n_sweeps=200)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                elif it > 100:
                    break

    # Phase 8: Final intensive coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 10:
        h_new = slsqp_optimize(best_h, best_n, 100000, 150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final coordinate descent after SLSQP
    if remaining_time() > 8:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint is met exactly
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)
            final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
