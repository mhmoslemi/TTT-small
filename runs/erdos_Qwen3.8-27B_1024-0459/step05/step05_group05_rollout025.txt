```python
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:

        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)

def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v

    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
```

```python
def swap_delta(h, i, j, n):
    """Compute delta for swap (i: 1->0, j: 0->1) at all lags."""
    # Change at position i: was 1, now 0
    # Change at position j: was 0, now 1
    
    delta = np.zeros(2*n - 1)
    
    # Position i went from 1 to 0
    # As "first" at lag d >= 0: old=1*(1-h[i+d]), new=0, delta=-(1-h[i+d])
    # valid for d = 0, 1, ..., n-1-i
    if i + 0 < n:
        d_vals = np.arange(0, min(n-i, n))
        delta[n-1+d_vals] -= (1 - h[i:d_vals[-1]+i+1])  # h[i], h[i+1], ..., h[i+d_vals[-1]]
    
    # As "second" at lag d >= 0: old=h[i-d]*(1-1)=0, new=h[i-d]*(1-0)=h[i-d], delta=+h[i-d]
    # valid for d = 0, 1, ..., i
    if i >= 0:
        d_vals = np.arange(0, min(i+1, n))
        delta[n-1+d_vals] += h[i-d_vals[::-1]]  # h[i], h[i-1], ..., h[i-d_vals[-1]]
    
    # Position j went from 0 to 1
    # As "first" at lag d >= 0: old=0, new=1*(1-h[j+d]), delta=+(1-h[j+d])
    if j + 0 < n:
        d_vals = np.arange(0, min(n-j, n))
        delta[n-1+d_vals] += (1 - h[j:j+d_vals[-1]+1])
    
    # As "second" at lag d >= 0: old=h[j-d]*(1-0)=h[j-d], new=h[j-d]*(1-1)=0, delta=-h[j-d]
    if j >= 0:
        d_vals = np.arange(0, min(j+1, n))
        delta[n-1+d_vals] -= h[j-d_vals[::-1]]
    
    # Now handle negative lags
    # For lag -a (a>0): corr = sum_{i=0}^{n-1-a} h[i+a]*(1-h[i])
    # Position p appears as h[i+a] with i=p-a: old=1*(1-h[p-a]), new=0, delta=-(1-h[p-a])
    # valid if p-a >= 0, i.e., a <= p
    
    # Position i (1->0) as h[i+a] (first in the pair for negative lag -a):
    # delta = -(1-h[i-a]) for a = 1, ..., i
    if i >= 1:
        a_vals = np.arange(1, i+1)
        delta[n-1-a_vals] -= (1 - h[i-a_vals])
    
    # Position i (1->0) as 1-h[i] (second in the pair for negative lag -a):
    # old = h[i+a]*(1-1) = 0, new = h[i+a]*(1-0) = h[i+a], delta = +h[i+a]

    if i + 1 < n:
        a_vals = np.arange(1, n-i)
        delta[n-1-a_vals] += h[i+a_vals]
    
    # Position j (0->1) as h[j+a] (first):
    # old = 0*(1-h[j-a]) = 0, new = 1*(1-h[j-a]) = 1-h[j-a], delta = +(1-h[j-a])
    # valid if j-a >= 0, i.e., a <= j
    if j >= 1:
        a_vals = np.arange(1, j+1)
        delta[n-1-a_vals] += (1 - h[j-a_vals])
```
