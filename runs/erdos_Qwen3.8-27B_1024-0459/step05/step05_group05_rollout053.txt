            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute Δc for all offsets efficiently
        # Δc[off] = h[i-off] + h[i+off] - h[j+off] - h[j-off] (out of bounds = 0)
        # offset ranges from -(n-1) to (n-1), stored as index off + (n-1) in c
        
        # For h[i-off]: valid when 0 ≤ i-off < n, i.e., i-n+1 ≤ off ≤ i
        # For h[i+off]: valid when 0 ≤ i+off < n, i.e., -i ≤ off ≤ n-1-i
        # For h[j+off]: valid when 0 ≤ j+off < n, i.e., -j ≤ off ≤ n-1-j
        # For h[j-off]: valid when 0 ≤ j-off < n, i.e., j-n+1 ≤ off ≤ j
        
        # Build Δc array
        dc = np.zeros(2*n-1)
        
        # Contribution from h[i] going 1→0: -(1-h[i+off]) for valid i+off, +h[i-off] for valid i-off
        # i.e., Δc[off] += -(1-h[i+off]) + h[i-off] = h[i+off] - 1 + h[i-off]
        
        # h[i+off] for valid i+off: off from -i to n-1-i
        lo = max(-(n-1), -i)  # also ≥ -(n-1)
        hi = min(n-1, n-1-i)  # also ≤ n-1
        if lo <= hi:
            off_range = np.arange(lo, hi+1)
            dc[off_range + (n-1)] += h[i + off_range] - 1.0
