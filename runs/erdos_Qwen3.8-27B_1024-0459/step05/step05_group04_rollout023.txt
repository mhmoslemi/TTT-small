def make_swap_delta(h, i, j, n):
    """Compute correlation change for swap h[i]:1->0, h[j]:0->1. O(n)."""
    r = 2 * n - 1
    dr = np.zeros(r)

    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)

    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)

            j = rng.randint(n)

            s_i = np.zeros(2*n-1)
            s_j = np.zeros(2*n-1)
            for k in range(2*n-1):

                idx_i2 = i + off
                idx_i1 = i - off
                idx_j2 = j + off
                idx_j1 = j - off

                sj = 0.0

                if 0 <= idx_i1 < n:
                    si -= h[idx_i1]
                if 0 <= idx_j2 < n:
                    sj += (1 - h[idx_j2])
                if 0 <= idx_j1 < n:
                    sj -= h[idx_j1]
                s_i[k] = si
                s_j[k] = sj

            # The new corr is corr + δ*d
            # We want to minimize max(corr + δ*d) over valid δ

            # i.e., -h[i] ≤ δ ≤ 1-h[i] and h[j]-1 ≤ δ ≤ h[j]
            delta_lo = max(-h[i], h[j]-1)
            delta_hi = min(1-h[i], h[j])
            
            if delta_lo >= delta_hi:
                continue
            
            # Find optimal δ
            new_corr = corr + delta_hi * d
            best_val = float(np.max(new_corr))
            new_corr = corr + delta_lo * d
            if np.max(new_corr) < best_val:
                best_val = float(np.max(new_corr))
                best_delta = delta_lo
            else:
                best_delta = delta_hi
            
            # The optimal is where the top 2 lines cross
            # Lines: corr[k] + δ*d[k]
            # Find the δ that minimizes max of these lines
            # This is the point where the two highest lines intersect
            # At the optimum, at least two lines are at the max
            # So: corr[a] + δ*d[a] = corr[b] + δ*d[b] = max
            # δ = (corr[b] - corr[a]) / (d[a] - d[b])
            
            # Find the two lines that form the max at the current best_delta
            vals = corr + best_delta * d
            top2 = np.argsort(-vals)[:3]
            for a in top2:
                for b in range(len(d)):
                    if a == b:
                        continue
                    if d[a] == d[b]:
                        continue

                            best_delta = opt_delta
            # Also check crossings with other lines

            # upper envelope has a "valley" - this happens at a crossing
            # of two lines that are both on the upper envelope
            
            # Simpler: just do a ternary search or check all breakpoints
            # Breakpoints: δ_ab = (corr[b]-corr[a])/(d[a]-d[b]) for all a,b
            # But that's O(n²) which is too slow

            breakpoints = []
            for a in range(len(d)):

                    if d[a] != d[b]:
                        bp = (corr[b] - corr[a]) / (d[a] - d[b])
                        if delta_lo < bp < delta_hi:
                            breakpoints.append(bp)

                    if new_max < best_val:
                        best_val = new_max
                        best_delta = bp
            
            if best_val < float(np.max(corr)) - 1e-12:
                h[i] += best_delta
                h[j] -= best_delta
                corr += best_delta * d
                improved = True
        
        if not improved:
            break
    
    return h, float(np.max(corr)) * (2.0/n)

→ i-k+n-1 ≤ n-1 → k ≥ i
So k ∈ [i, i+n-1], clamped to [0, 2n-2]
Value: h[m] * δ where δ is the change in (1-h[i])
For h[i]: 1→0: (1-h[i]) changes from 0 to 1, δ=+1. So Δ = +h[m] = +h[i-k+(n-1)]
For h[j]: 0→1: (1-h[j]) changes from 1 to 0, δ=-1. So Δ = -h[m] = -h[j-k+(n-1)]
