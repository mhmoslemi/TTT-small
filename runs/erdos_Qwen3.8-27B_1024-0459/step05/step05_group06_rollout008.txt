            best_h = h.copy()

    # Phase 3: Pair swap descent
    if remaining_time() > 60:
        h = best_h.copy()
        h, c5_ps = pair_swap_descent(h, n, n_sweeps=15)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h.copy()

    # Phase 4: SLSQP with higher beta
    if remaining_time() > 30:
        h = slsqp_optimize(best_h, best_n, 10000, 300)
        c5_check = compute_c5(h)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h.copy()

    # Phase 5: More coordinate descent
    if remaining_time() > 100:
        h, c5_cd = coord_descent_fast(best_h, best_n, n_sweeps=400)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

    # Phase 6: Weighted subgradient + project
    if remaining_time() > 80:
        h = best_h.copy()
        n = best_n
        base_step = 0.03
        for it in range(200):
            if remaining_time() < 10:
                break
            grad = weighted_subgrad_step(h, n, 80)
            if grad is None:
                break
            step = base_step / (1.0 + it * 0.003)
            improved = False
            for sm in [1.0, 0.5, 0.25, 2.0, 0.1]:
                h_trial = h - step * sm * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved and it % 50 == 40:
                if remaining_time() > 30:
                    h = slsqp_optimize(h, n, 50000, 200)
                    c5_check = compute_c5(h)
                    if c5_check < best_c5:
                        best_c5 = c5_check
                        best_h = h.copy()

    # Phase 7: More coordinate descent after subgradient
    if remaining_time() > 80:
        h, c5_cd = coord_descent_fast(best_h, best_n, n_sweeps=300)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

    # Phase 8: Try different resolution
    for trial_n in [best_n + 128, best_n - 128, 800, 768]:
        if remaining_time() < 80:
            break
        rn = trial_n
        if rn % 2 != 0:
            rn += 1
        if rn < 128 or rn > 2000:
            continue
        if rn == best_n:
            continue
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        h_r = slsqp_optimize(h_r, rn, 5000, 200)
        if remaining_time() > 30:
            h_r, c5_r = coord_descent_fast(h_r, rn, n_sweeps=100)

        c5_r = compute_c5(h_r)
        if c5_r < best_c5 - 1e-15:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 9: Back to best resolution, final SLSQP
    if remaining_time() > 25:
        h = slsqp_optimize(best_h, best_n, 100000, 300)
        c5_check = compute_c5(h)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h.copy()

    # Phase 10: Final intensive coordinate descent
    if remaining_time() > 40:
        h, c5_cd = coord_descent_fast(best_h, best_n, n_sweeps=300)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

    # Phase 11: Pair swap final
    if remaining_time() > 20:
        h, c5_ps = pair_swap_descent(best_h, best_n, n_sweeps=10)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h.copy()

    # Phase 12: Final coord descent
    if remaining_time() > 15:
        h, c5_cd = coord_descent_fast(best_h, best_n, n_sweeps=100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
