def project_feasible(h, n):

def smooth_max_and_grad(h, n, beta):

            print("n=1024 not better: {:.8f} vs {:.8f}".format(c5_1024, best_c5))

    # Random restarts from best solution
    n_best = best_n
    h_best = best_h.copy()
    restart_count = 0
    while rem() > 40 and restart_count < 12:
        # Create perturbation with decreasing scale
        scale = 0.08 * (1.0 - 0.06 * restart_count)
        if scale < 0.01:
            scale = 0.01
        h_pert = h_best + scale * rng.randn(n_best)
        h_pert = project_feasible(h_pert, n_best)
        c5_pert = compute_c5(h_pert)

        # Quick optimization
        h_rt = slsqp_step(h_pert, n_best, 5000, maxiter=400)
        c5_rt = compute_c5(h_rt)

        if c5_rt < best_c5 + 0.0005:
            # Worth further refinement
            h_rt = slsqp_step(h_rt, n_best, 20000, maxiter=600)
            c5_rt = compute_c5(h_rt)
            if c5_rt < best_c5:
                best_c5 = c5_rt
                best_h = h_rt.copy()
                h_best = h_rt.copy()
                print("Restart {} improved: C5={:.8f}".format(restart_count, best_c5))
        restart_count += 1

    # Final intensive polish at best resolution
    n_fin = best_n
    h_fin = best_h.copy()

    # SLSQP with very high beta
    for beta in [30000, 80000, 150000, 300000]:
        if rem() < 5:
            break
        h_fin = slsqp_step(h_fin, n_fin, beta, maxiter=1000)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Fine coordinate descent
    if rem() > 10:
        h_cd, c5_cd = coord_descent(best_h, best_n, max_sweeps=6,
                                    time_limit=min(20.0, rem() * 0.5))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final SLSQP
    if rem() > 5:
        h_fin = slsqp_step(best_h, best_n, 200000, maxiter=1200)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print("Final: n={}, C5={:.8f}".format(best_n, best_c5))

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
        print("Evaluated C5: {:.8f}".format(best_c5))
    except Exception:
        pass

    return (best_h, best_c5, best_n)
