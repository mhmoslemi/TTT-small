def project_binary(h, n):
    """Round to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones == n_half:
        return (h > 0.5).astype(float)
    if n_ones > n_half:
        ones = np.where(h > 0.5)[0]
        # Remove the ones with lowest h values
        order = np.argsort(h[ones])
        remove = order[:n_ones - n_half]
        h[ones[remove]] = 0
    else:

        # Add the zeros with highest h values
        order = np.argsort(-h[zeros])
        add = order[:n_half - n_ones]

def swap_delta(h, p, q, n):
    """O(n) correlation change for swap h[p]:1->0, h[q]:0->1."""
    r = 2 * n - 1
    dr = np.zeros(r)
    # h[p] decreases by 1
    lo = max(p, 0); hi = min(p + n - 1, r - 1)
    if lo <= hi:

    if lo2 <= hi2:

    # h[q] increases by 1
