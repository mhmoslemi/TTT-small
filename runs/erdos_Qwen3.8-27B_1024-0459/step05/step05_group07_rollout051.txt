import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(interior)
        if cnt > 0:
            h[interior] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)


def coord_descent(h, n, max_sweeps=200):
    """Efficient single-coordinate descent with exact 1D minimization."""
    best = h.copy().astype(np.float64)
    corr = compute_corr(best).astype(np.float64).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    gr = 0.6180339887

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        old_max = np.max(corr)
        any_improve = False

        for i in range(n):
            old = best[i]

            i_mk = i - offsets
            vmk = (i_mk >= 0) & (i_mk < n)
            i_pk = i + offsets
            vpk = (i_pk >= 0) & (i_pk < n)

            A = np.zeros(ncorr)
            if np.any(vmk):
                A[vmk] += 1.0 - best[i_mk[vmk]]
            if np.any(vpk):
                A[vpk] -= best[i_pk[vpk]]
            A[n - 1] = 0.0

            f_old = np.max(corr)
            best_t = old
            best_f = f_old

            # Candidate: t=0
            lin0 = corr - A * old
            q0 = corr[n - 1] - old * (1.0 - old)
            f0 = max(np.max(lin0), q0)
            if f0 < best_f - 1e-14:
                best_f = f0
                best_t = 0.0

            # Candidate: t=1
            lin1 = corr + A * (1.0 - old)
            q1 = corr[n - 1] - old * (1.0 - old)
            f1 = max(np.max(lin1), q1)
            if f1 < best_f - 1e-14:
                best_f = f1
                best_t = 1.0

            # Candidate: t=0.5
            lin05 = corr + A * (0.5 - old)
            q05 = corr[n - 1] - old * (1.0 - old) + 0.25
            f05 = max(np.max(lin05), q05)
            if f05 < best_f - 1e-14:
                best_f = f05
                best_t = 0.5

            # Golden section refinement
            if best_t != old:
                if best_t > old:
                    lo = old
                    hi = min(1.0, best_t + 0.25)
                else:
                    lo = max(0.0, best_t - 0.25)
                    hi = old
                if hi - lo < 1e-10:
                    hi = min(1.0, lo + 1e-5)
                    lo = max(0.0, hi - 1e-5)

                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                def _fv(t):
                    lm = np.max(corr + A * (t - old))
                    qv = corr[n - 1] - old * (1.0 - old) + t * (1.0 - t)
                    return lm if lm > qv else qv

                f1v = _fv(c1)
                f2v = _fv(c2)
                for _ in range(28):
                    if f1v < f2v:
                        hi = c2
                        c2 = c1
                        f2v = f1v
                        c1 = hi - gr * (hi - lo)
                        f1v = _fv(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1v = f2v
                        c2 = lo + gr * (hi - lo)
                        f2v = _fv(c2)

                t_opt = (lo + hi) * 0.5
                f_opt = _fv(t_opt)
                if f_opt < best_f - 1e-13:
                    best_t = t_opt
                    best_f = f_opt

            if best_t != old and best_f < f_old - 1e-14:
                best[i] = best_t
                d = best_t - old
                corr += A * d
                corr[n - 1] += best_t * (1.0 - best_t) - old * (1.0 - old)
                any_improve = True

        if not any_improve:
            break
        if np.max(corr) >= old_max - 1e-15:
            break

def subgrad_step(h, n, step_size, n_active=8):
    corr = compute_corr(h)
    max_c = np.max(corr)
    if max_c < 1e-15:
        return h
    threshold = max_c - 0.15
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    if len(active) > n_active:
        idx = np.argsort(corr[active])[::-1][:n_active]
        active = active[idx]

    grad = np.zeros(n)
    for k_idx in active:
        k = int(k_idx) - (n - 1)
        if k >= 0:
            npairs = n - k
            if npairs > 0:
                grad[k:k + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[k:k + npairs]
        else:
            m = -k
            npairs = n - m
            if npairs > 0:
                grad[:npairs] += (1.0 - h[m:m + npairs])
                grad[m:m + npairs] -= h[:npairs]

    grad /= float(len(active))
    h_new = h - step_size * grad
    return project_feasible(h_new, n)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def sinusoidal_init(n, freq, amp):
    x = np.arange(n) * 2.0 / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def binary_random_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def slsqp_refine(h, n, beta, maxiter=300):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        ev = np.exp(shifted)
        ws = np.sum(ev)
        if ws < 1e-300:
            ws = 1.0
        return (m + np.log(ws) / beta) * dx

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        ev = np.exp(shifted)
        ws = np.sum(ev)
        if ws < 1e-300:
            ws = 1.0
        w = ev / ws
        grad = np.zeros(n)
        for ki in range(len(corr)):
            wi = w[ki]
            if wi < 1e-14:
                continue
            off = ki - (n - 1)
            if off >= 0:
                npairs = n - off
                if npairs > 0:
                    grad[off:off + npairs] += wi * (1.0 - x[:npairs])
                    grad[:npairs] -= wi * x[off:off + npairs]
            else:
                aoff = -off
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += wi * (1.0 - x[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= wi * x[:npairs]
        grad *= dx
        return grad

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5t = compute_c5(trial)
            c5o = compute_c5(h)
            if c5t <= c5o + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()
