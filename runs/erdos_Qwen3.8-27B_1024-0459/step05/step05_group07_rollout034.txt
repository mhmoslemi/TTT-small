def compute_corr(h):

def project_feasible(h, n):

    target = n / 2.0

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:

            valid_minus = (i_minus >= 0) & (i_minus < n)

            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

def subgradient_descent(h, n, max_iters=200, base_step=0.05):
    """Subgradient descent with projection."""
    best = h.copy()
    best_c5 = compute_c5(best)

    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(best)
        m = np.max(corr)
        # Find active set
        threshold = m - 0.1 * m
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 10:
            active = active[np.argsort(corr[active])[::-1][:10]]

        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in active:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += (1.0 - best[:npairs])
                    grad[:npairs] -= best[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= best[:npairs]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = base_step / (1.0 + 0.05 * it)
        new_h = project_feasible(best - step * grad / gnorm * np.max(gnorm), n)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best = new_h
            best_c5 = new_c5

    return best, best_c5

        if n_high <= 2560:
            h_high = interp_to(best_h, n_cur, n_high)
            h_high = project_feasible(h_high, n_high)
            c5_high = compute_c5(h_high)

            h_high, c5_high = coord_descent(h_high, n_high, max_sweeps=100)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high
                n_cur = n_high

            if remaining_time() > 60:
                h_high = best_h.copy()
                h_high = slsqp_refine(h_high, n_cur, 2000, 400)
                c5_high = compute_c5(h_high)
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_high.copy()

            if remaining_time() > 40:
                h_high = best_h.copy()
                h_high, c5_high = coord_descent(h_high, n_cur, max_sweeps=100)
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_high.copy()

            if remaining_time() > 40:
                h_high = best_h.copy()
                h_high, c5_high = pair_flip_local_search(h_high, n_cur, max_iters=100, rng=rng, max_pairs_per_iter=200)
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_high.copy()
                if remaining_time() > 20:
                    h_high, c5_high = coord_descent(best_h, n_cur, max_sweeps=100)
                    if c5_high < best_c5:
                        best_c5 = c5_high
                        best_h = h_high.copy()

    # Phase 6: Try even higher resolution if time allows
    if remaining_time() > 100:
        n_high2 = n_cur * 2
        if n_high2 <= 4000:
            h_high2 = interp_to(best_h, n_cur, n_high2)
            h_high2 = project_feasible(h_high2, n_high2)
            h_high2, c5_high2 = coord_descent(h_high2, n_high2, max_sweeps=50)
            if c5_high2 < best_c5:
                best_c5 = c5_high2
                best_h = h_high2.copy()
                best_n = n_high2
                n_cur = n_high2
            if remaining_time() > 30:
                h_high2, c5_high2 = coord_descent(best_h, n_cur, max_sweeps=30)
                if c5_high2 < best_c5:
                    best_c5 = c5_high2
                    best_h = h_high2.copy()

    # Phase 7: Subgradient descent for escaping local minima
    if remaining_time() > 40:
        h = best_h.copy()
        h, c5 = subgradient_descent(h, n_cur, max_iters=100, base_step=0.03)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Update A_mat and do coordinate descent
        if remaining_time() > 20:
            h, c5 = coord_descent(best_h, n_cur, max_sweeps=150)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 8: Multiple random restarts with coordinate descent
    if remaining_time() > 60:
        n_restarts = 4
        for r in range(n_restarts):
            if remaining_time() < 20:
                break
            # Perturb best solution
            h_pert = best_h.copy()
            noise_scale = 0.1 / (1.0 + r * 0.5)
            mask = rng.random(n_cur) < 0.2
            h_pert[mask] += rng.randn(int(np.sum(mask))) * noise_scale
            h_pert = project_feasible(h_pert, n_cur)

            h_pert, c5_pert = coord_descent(h_pert, n_cur, max_sweeps=100)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

            # Try SLSQP on the best restart
            if remaining_time() > 30:
                h_slq = slsqp_refine(best_h, n_cur, 5000, 300)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()

    # Phase 9: Thue-Morse based restart
    if remaining_time() > 60:
        h_tm = thue_morse_init(n_cur)
        h_tm, c5_tm = coord_descent(h_tm, n_cur, max_sweeps=200)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
        if remaining_time() > 30:
            h_tm = slsqp_refine(best_h, n_cur, 10000, 400)
            c5_tm = compute_c5(h_tm)
            if c5_tm < best_c5:
                best_c5 = c5_tm
                best_h = h_tm.copy()

    # Phase 10: Final SLSQP polishing with high beta
    if remaining_time() > 15:
        h = best_h.copy()
        for beta in [20000, 80000, 200000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(h, n_cur, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 11: Final coordinate descent
    if remaining_time() > 10:
        h, c5 = coord_descent(best_h, n_cur, max_sweeps=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 12: One more pair-flip if time
    if remaining_time() > 20:
        h, c5 = pair_flip_local_search(best_h, n_cur, max_iters=100, rng=rng, max_pairs_per_iter=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if remaining_time() > 10:
            h, c5 = coord_descent(best_h, n_cur, max_sweeps=100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, n_cur)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_cur)
