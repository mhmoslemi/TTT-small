def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g


def smooth_max_val_grad(h, n, beta, topK):
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:topK]
    c_top = corr[top_idx]
    m = c_top.max()
    sh = beta * (c_top - m)
    cl = np.clip(sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = ex.sum()
    w = ex / s
    val = (m + np.log(s) / beta) * (2.0 / n)
    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        wm = w[i] * (2.0 / n)
        if wm > 1e-15:
            g += wm * grad_corr_k(h, k, n)
    return val, g


def slsqp_smooth(h, n, beta, topK, maxiter):
    def obj(x):
        v, _ = smooth_max_val_grad(x, n, beta, topK)
        pen = 5000.0 * (np.sum(x) - n / 2.0) ** 2
        return v + pen

    def obj_grad(x):
        _, g = smooth_max_val_grad(x, n, beta, topK)
        g = g + 10000.0 * (np.sum(x) - n / 2.0) * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h


def lbfgs_smooth(h, n, beta, topK, maxiter):
    def obj(x):
        v, _ = smooth_max_val_grad(x, n, beta, topK)
        pen = 10000.0 * (np.sum(x) - n / 2.0) ** 2
        return v + pen

    def obj_grad(x):
        _, g = smooth_max_val_grad(x, n, beta, topK)
        g = g + 20000.0 * (np.sum(x) - n / 2.0) * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                               'gtol': 1e-14, 'maxls': 50})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h


def remez_equalize(h, n, K, max_rounds):
    h = h.copy()
    best_c5 = compute_c5(h)
    for rd in range(max_rounds):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:K]
        c_top = corr[top_idx]
        c_max = c_top.max()
        c_min = c_top[-1]
        spread = c_max - c_min
        if spread < 1e-12:
            break
        # Weighted gradient: push down max, push up min
        g_dir = np.zeros(n)
        for k in top_idx:
            nick = (c_top - c_min) / (spread + 1e-30)
            weight = nick - 0.5
            g_dir += weight * grad_corr_k(h, k, n)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.01, 0.02, 0.005, 0.03, 0.015, 0.008, 0.025, 0.003]:
            ht = project_feasible(h - step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def subgradient_step(h, n, K, beta_s):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta_s * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, k, n)
    return g, np.max(corr)


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_push(h, n, push_strength):
    h = h.copy()
    mask_mid = (h > 0.05) & (h < 0.95)
    h[mask_mid] = h[mask_mid] * (1.0 - push_strength) + np.where(h[mask_mid] > 0.5, 1.0, 0.0) * push_strength
    return project_feasible(h, n)


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def optimize_resolution(h_init, n, time_alloc, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Step 1: Quick L-BFGS polish
    t_start = time.time()
    for beta in [200, 800, 3000]:
        if time.time() - t_start > time_alloc * 0.15:
            break
        h_new = lbfgs_smooth(h, n, beta, topK=30, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Step 2: Remez equalize
    if time.time() - t_start < time_alloc * 0.5:
        h_eq, c5_eq = remez_equalize(h, n, K=8, max_rounds=100)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Step 3: SLSQP with higher beta
    for beta in [5000, 15000, 40000]:
        if time.time() - t_start > time_alloc * 0.85:
            break
        h_new = slsqp_smooth(h, n, beta, topK=20, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Step 4: Final Remez
    h_eq, c5_eq = remez_equalize(h, n, K=10, max_rounds=80)
    if c5_eq < best_c5:
        best_c5 = c5_eq
        best_h = h_eq.copy()

    # Step 5: Fine subgradient
    for it in range(60):
        if time.time() - t_start > time_alloc:
            break
        g, cmax = subgradient_step(h, n, K=20, beta_s=80.0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        for step in [0.005, 0.01, 0.003, 0.008, 0.002]:
            ht = project_feasible(h - step * g / gn, n)
            cn = compute_c5(ht)
            if cn < best_c5:
                best_c5 = cn
                best_h = ht.copy()
                h = ht.copy()
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # === Phase 1: Intensive optimization at parent resolution ===
    t_start = time.time()
    h = best_h.copy()
    n = best_n

    # 1a: L-BFGS-B with increasing beta (fast convergence)
    for beta in [100, 400, 1500, 5000, 15000]:
        h_new = lbfgs_smooth(h, n, beta, topK=35, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        if remaining_time() < 30:
            break

    # 1b: Remez equalization
    h_eq, c5_eq = remez_equalize(h, n, K=8, max_rounds=150)
    if c5_eq < best_c5:
        best_c5 = c5_eq
        best_h = h_eq.copy()
        h = h_eq.copy()

    # 1c: SLSQP high-beta polish
    for beta in [10000, 30000, 80000]:
        h_new = slsqp_smooth(h, n, beta, topK=15, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        if remaining_time() < 30:
            break

    # 1d: Fine Remez + subgradient
    h_eq, c5_eq = remez_equalize(h, n, K=12, max_rounds=100)
    if c5_eq < best_c5:
        best_c5 = c5_eq
        best_h = h_eq.copy()
        h = h_eq.copy()

    for it in range(80):
        if remaining_time() < 10:
            break
        g, cmax = subgradient_step(h, n, K=25, beta_s=100.0)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        for step in [0.003, 0.006, 0.001, 0.004, 0.002]:
            ht = project_feasible(h - step * g / gn, n)
            cn = compute_c5(ht)
            if cn < best_c5:
                best_c5 = cn
                best_h = ht.copy()
                h = ht.copy()
                break

    # === Phase 2: Try n=640 resolution ===
    if remaining_time() > 60:
        n640 = 640
        h640 = interp_to(best_h, best_n, n640)
        h640 = project_feasible(h640, n640)
        c5_640_init = compute_c5(h640)

        # Optimize at 640
        h_640_opt, c5_640 = optimize_resolution(h640, n640, min(120.0, remaining_time() * 0.35), rng)
        if c5_640 < best_c5:
            best_c5 = c5_640
            best_h = h_640_opt.copy()
            best_n = n640

    # === Phase 3: Try n=768 resolution ===
    if remaining_time() > 50:
        n768 = 768
        h768 = interp_to(best_h, best_n, n768)
        h768 = project_feasible(h768, n768)

        h_768_opt, c5_768 = optimize_resolution(h768, n768, min(100.0, remaining_time() * 0.30), rng)
        if c5_768 < best_c5:
            best_c5 = c5_768
            best_h = h_768_opt.copy()
            best_n = n768

    # === Phase 4: Re-optimize at best resolution with remaining time ===
    if remaining_time() > 30:
        n = best_n
        h = best_h.copy()

        # L-BFGS with very high beta (near-exact max)
        for beta in [50000, 100000, 200000]:
            h_new = lbfgs_smooth(h, n, beta, topK=20, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            if remaining_time() < 15:
                break

        # Final Remez
        h_eq, c5_eq = remez_equalize(h, n, K=10, max_rounds=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

        # Ultra-fine subgradient
        for it in range(100):
            if remaining_time() < 8:
                break
            g, cmax = subgradient_step(h, n, K=15, beta_s=150.0)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            for step in [0.001, 0.002, 0.0005, 0.0015, 0.0008]:
                ht = project_feasible(h - step * g / gn, n)
                cn = compute_c5(ht)
                if cn < best_c5:
                    best_c5 = cn
                    best_h = ht.copy()
                    h = ht.copy()
                    break

    # === Phase 5: Binary push exploration ===
    if remaining_time() > 40:
        n = best_n
        h_base = best_h.copy()
        for push in [0.3, 0.5, 0.7]:
            if remaining_time() < 20:
                break
            h_push = binary_push(h_base, n, push)
            # Quick optimize from pushed start
            h_opt = lbfgs_smooth(h_push, n, 2000, topK=25, maxiter=200)
            c5_try = compute_c5(h_opt)
            if c5_try < best_c5:
                h_eq, c5_eq = remez_equalize(h_opt, n, K=8, max_rounds=80)
                if c5_eq < best_c5:
                    best_c5 = c5_eq
                    best_h = h_eq.copy()
                    best_n = n

    # === Final feasibility and evaluation ===
    best_n = best_n
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
