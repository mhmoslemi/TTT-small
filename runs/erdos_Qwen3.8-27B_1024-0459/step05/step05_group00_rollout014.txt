def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def perturb_and_optimize(h, n, num_tries, rng, beta=5000, maxiter=200):
    global _T0, _BUDGET
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(num_tries):
        if _BUDGET - (time.time() - _T0) < 20:
            break
        scale = rng.uniform(0.02, 0.08)
        h_pert = project_feasible(h + scale * rng.randn(n), n)
        h_opt = slsqps_solve(h_pert, n, beta=beta, maxiter=maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Progressive SLSQP at parent resolution
    for beta in [50, 200, 1000, 3000, 8000, 15000]:
        if remaining() < 8:
            break
        h = slsqps_solve(h, n, beta=beta, maxiter=300)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Phase 2: Short subgradient burst
    if remaining() > 15:
        h_sg = h.copy()
        h_sg, c5_sg = subgradient_loop(h_sg, n, time_limit=15, rng=rng, K=3, beta_g=25)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: Block coordinate descent
    if remaining() > 30:
        h_bd = h.copy()
        h_bd, c5_bd = block_descent(h_bd, n, block_size=32, num_sweeps=3, beta=200)
        if c5_bd < best_c5:
            best_c5 = c5_bd
            best_h = h_bd.copy()
            h = h_bd.copy()

    # Phase 4: More SLSQP after block descent
    for beta in [20000, 40000]:
        if remaining() < 8:
            break
        h = slsqps_solve(h, n, beta=beta, maxiter=350)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Phase 5: Random perturbations + SLSQP
    if remaining() > 40:
        h_rp = best_h.copy()
        h_rp, c5_rp = perturb_and_optimize(h_rp, n, num_tries=4, rng=rng,
                                           beta=8000, maxiter=200)
        if c5_rp < best_c5:
            best_c5 = c5_rp
            best_h = h_rp.copy()
            h = h_rp.copy()

    # Phase 6: Higher resolution n=768
    if remaining() > 80:
        n_t = 768
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        for beta in [1000, 5000, 15000, 30000]:
            if remaining() < 10:
                break
            h_feas = slsqps_solve(h_feas, n_t, beta=beta, maxiter=250)
        c5_new = compute_c5(h_feas)
        if remaining() > 15:
            h_feas, c5_new = subgradient_loop(h_feas, n_t, time_limit=15, rng=rng, K=3, beta_g=25)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_feas.copy()
            best_n = n_t
            h = h_feas.copy()
            n = n_t

    # Phase 7: Block descent at best resolution
    if remaining() > 40:
        h_bd = h.copy()
        bs = 32 if n <= 768 else 64
        h_bd, c5_bd = block_descent(h_bd, n, block_size=bs, num_sweeps=2, beta=300)
        if c5_bd < best_c5:
            best_c5 = c5_bd
            best_h = h_bd.copy()
            h = h_bd.copy()

    # Phase 8: Final high-beta SLSQP
    for beta in [50000, 100000, 200000]:
        if remaining() < 6:
            break
        h = slsqps_solve(h, n, beta=beta, maxiter=400)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Phase 9: One more perturbation round
    if remaining() > 30:
        h_rp = best_h.copy()
        h_rp, c5_rp = perturb_and_optimize(h_rp, n, num_tries=3, rng=rng,
                                           beta=30000, maxiter=250)
        if c5_rp < best_c5:
            best_c5 = c5_rp
            best_h = h_rp.copy()

    # Phase 10: Final micro-subgradient
    if remaining() > 10:
        h_sg = best_h.copy()
        h_sg, c5_sg = subgradient_loop(h_sg, n, time_limit=min(10, remaining()),
                                       rng=rng, K=2, beta_g=50)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final polish SLSQP
    if remaining() > 5:
        h = slsqps_solve(best_h, n, beta=100000, maxiter=300)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
