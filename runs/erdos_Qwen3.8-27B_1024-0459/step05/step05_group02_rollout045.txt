    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_fun, h, jac=_lse_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 10})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def nesterov_grad(h, n, top_k=20, max_iters=5000, step_init=0.02):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.9
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        m = c_top[0]
        # Softmax weights
        sh = 500.0 * (c_top - m)
        sh = np.clip(sh, -100.0, 100.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.001:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.1)
            else:
                step *= 0.95
        else:
            h = best_h.copy()
            v *= 0.3
            step *= 0.6

        if it % 2000 == 0 and it > 0:
            step *= 0.8

    return best_h, best_c5


def binary_search_2swap(h, n, max_iters=3000, top_k=5):
    """Binary local search with both 1-swap and 2-swap moves."""
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]

        # Compute combined gradient for top lags
        g = np.zeros(n)
        c_top = corr[idx_top]
        m = c_top[0]
        sh = 500.0 * (c_top - m)
        sh = np.clip(sh, -100.0, 100.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        for ki, wi in zip(idx_top, w):
            g += wi * corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0.0
        best_move = None
        best_type = 0

        # Score for 1-swap
        score_one = -g[ones]
        score_zero = g[zeros]

        n_try1 = min(10, len(ones))
        n_try0 = min(10, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try1]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try0]]

        # Try 1-swaps
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                nc = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = ((i, j),)
                    best_type = 1

        # Try 2-swaps (top candidates)
        n_try2 = min(5, len(ones))
        n_try2z = min(4, len(zeros))
        top_ones2 = ones[np.argsort(-score_one)[:n_try2]]
        top_zeros2 = zeros[np.argsort(-score_zero)[:n_try2z]]

        for i1 in top_ones2:
            for i2 in top_ones2:
                if i2 <= i1:
                    continue
                for j1 in top_zeros2:
                    for j2 in top_zeros2:
                        if j2 <= j1:
                            continue
                        if i1 in (j1, j2) or i2 in (j1, j2):
                            continue
                        h_bin[i1] = 0.0
                        h_bin[i2] = 0.0
                        h_bin[j1] = 1.0
                        h_bin[j2] = 1.0
                        nc = compute_c5(h_bin)
                        h_bin[i1] = 1.0
                        h_bin[i2] = 1.0
                        h_bin[j1] = 0.0
                        h_bin[j2] = 0.0
                        delta = nc - cur_c5
                        if delta < best_delta:
                            best_delta = delta
                            best_move = ((i1, i2, j1, j2),)
                            best_type = 2

        if best_move is None or best_delta >= 0:
            # Random escape
            found = False
            for _ in range(8):
                i = int(ones[np.random.randint(len(ones))])
                j = int(zeros[np.random.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                nc = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = ((i, j),)
                    best_type = 1
                    found = True
            if not found:
                break

        mv = best_move[0]
        if best_type == 1:
            i, j = mv
            h_bin[i] = 0.0
            h_bin[j] = 1.0
        else:
            i1, i2, j1, j2 = mv
            h_bin[i1] = 0.0
            h_bin[i2] = 0.0
            h_bin[j1] = 1.0
            h_bin[j2] = 1.0

        cur_c5 = compute_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def slsqp_pass(h, n, beta, top_k=20, maxiter=200):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)

    def _sc(x):
        return np.sum(x) - _TGT

    def _scj(x):
        return np.ones(_N)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_lse_fun, h, jac=_lse_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return proj(res.x, n)
    except Exception:
        return h


def top_equalize(h, n, top_k=10, max_iters=500):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-15:
            break
        d /= dn

        improved = False
        for scale in [3, 1, 2, 0.5, 5, 0.1, 1.5, 0.05, 4]:
            alpha = 0.01 * scale
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for scale in [2, 1, 3, 0.5]:
                alpha = 0.005 * scale
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        rng_h = np.random.RandomState(seed).rand(parent_n)
        parent_h = (rang_h > 0.5).astype(float)
        # Fix sum
        n_ones = int(np.sum(parent_h))
        if n_ones > parent_n // 2:
            idx = np.where(parent_h == 1)[0]
            parent_h[idx[parent_n//2:]] = 0
            idx0 = np.where(parent_h == 0)[0]
            parent_h[idx0[:parent_n//2 - n_ones + (n_ones - parent_n//2)]] = 1
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Optimize at parent resolution with increased top_k
    n = best_n
    h = best_h.copy()

    for beta in [500, 2000, 5000, 10000]:
        if remaining() < 15:
            break
        for tk in [10, 20]:
            if remaining() < 8:
                break
            h_new = lbfgs_pass(h, n, beta, top_k=tk, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 2: Nesterov gradient at parent resolution
    if remaining() > 30:
        h_ng, c5_ng = nesterov_grad(h, n, top_k=20, max_iters=6000, step_init=0.02)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()
            h = h_ng.copy()

    # Phase 3: Top equalize
    if remaining() > 15:
        h_eq, c5_eq = top_equalize(h, n, top_k=15, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Binary local search with 2-swap at parent resolution
    if remaining() > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_b2s, c5_b2s = binary_search_2swap(h_bin, best_n, max_iters=2000, top_k=5)
        # Refine continuously
        h_ref = h_b2s.copy()
        for beta in [5000, 15000]:
            if remaining() < 5:
                break
            h_ref = lbfgs_pass(h_ref, best_n, beta, top_k=15, maxiter=300)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
        elif c5_b2s < best_c5:
            best_c5 = c5_b2s
            best_h = h_b2s.copy()

    # Phase 5: Multi-resolution - try higher resolutions
    for n_t in [best_n * 2, best_n + 64, int(best_n * 1.5)]:
        if remaining() < 60:
            break
        if n_t <= 128 or n_t > 2048 or n_t % 2 != 0:
            continue
        if n_t == best_n:
            continue

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # Optimize at new resolution
        for beta in [2000, 8000, 20000]:
            if remaining() < 15:
                break
            h_feas = lbfgs_pass(h_feas, n_t, beta, top_k=20, maxiter=300)

        # Nesterov at new resolution
        if remaining() > 20:
            h_feat, c5_feat = nesterov_grad(h_feas, n_t, top_k=15, max_iters=3000, step_init=0.015)
            if c5_feat < compute_c5(h_feas):
                h_feas = h_feat

        # Binary 2-swap at new resolution
        h_bin_t = (h_feas > 0.5).astype(float)
        h_bt, c5_bt = binary_search_2swap(h_bin_t, n_t, max_iters=1500, top_k=5)
        h_ref_t = h_bt.copy()
