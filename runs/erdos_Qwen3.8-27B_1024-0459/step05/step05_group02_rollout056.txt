def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float).copy()

def corr_grad_single(h, k, n):
    """Gradient of corr[k] with respect to h."""
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

def remaining():

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):

    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):

    dx = 2.0 / n
    corr = compute_corr(x)

    c = corr[idx]
    m = c[0]

    g = np.zeros(n)

def binary_sa(h_init, n, max_iters=4000, t_init=0.4, t_min=0.003, cool=0.99985):

    best_c5 = compute_c5(h_bin)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_refine(h, n, top_k=6, max_iters=5000, step_init=0.03):

    best_h = h.copy()

    momentum = 0.9
    step = step_init

        if remaining() < 3:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx

            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - np.mean(c_top)) + 0.1
        w = np.maximum(w, 0.01)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.003:
            h = h_new
            if c5_new <= c5:

        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.7

        if it % 1000 == 0 and it > 0:
            step *= 0.85
            momentum = min(momentum + 0.005, 0.95)

    # Phase 1: Intensive optimization at parent resolution
    h_parent = optimize_at_n(best_h, best_n, time_alloc=min(240, remaining()), rng=rng)
    if h_parent[1] < best_c5:
        best_c5 = h_parent[1]
        best_h = h_parent[0].copy()

    # Phase 2: Try multiple nearby resolutions
    # Generate candidate N values
    n_cands = set()
    for d in range(-6, 13, 2):
        nc = best_n + d
        if 120 <= nc <= 600 and nc % 2 == 0:
            n_cands.add(nc)
    for nc in [200, 224, 256, 288, 320, 352, 400, 416, 448, 480, 512, 544, 560]:
        if 120 <= nc <= 600 and nc % 2 == 0:
            n_cands.add(nc)

    n_cands = sorted(n_cands)
    n_cands = [nc for nc in n_cands if nc != best_n]
    n_cands.sort(key=lambda x: abs(x - best_n))

    # Allocate time per resolution based on proximity
    total_remaining = remaining()
    # Give more time to closer N values
    for idx, n_t in enumerate(n_cands[:14]):
        if remaining() < 30:
            break
        # Allocate time: closer N gets more time
        dist = abs(n_t - best_n)
        if dist <= 4:
            alloc = min(120, remaining() - 30)
        elif dist <= 10:
            alloc = min(90, remaining() - 30)
        elif dist <= 30:
            alloc = min(70, remaining() - 30)
        else:
            alloc = min(50, remaining() - 30)

        if alloc < 10:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        h_opt, c5_opt = optimize_at_n(h_feas, n_t, time_alloc=alloc, rng=rng)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 3: Focused binary search on the best solution
    if remaining() > 40:
        h_bin_b = (best_h > 0.5).astype(float)

        # Multiple greedy passes with different top_k
        for tk in [3, 5, 7, 10]:
            if remaining() < 15:
                break
            h_g, c5_g = binary_greedy(h_bin_b, best_n, top_k=tk, max_iters=3000,
                                       n_cand_o=20, n_cand_z=20)
            c5_gc = compute_c5(h_g)
            if c5_gc < best_c5:
                best_c5 = c5_gc
                best_h = h_g.copy()
                h_bin_b = h_g.copy()

        # SA from best binary
        if remaining() > 20:
            h_sa3, c5_sa3 = binary_sa(h_bin_b, best_n, max_iters=3000,
                                       t_init=0.2, t_min=0.003, cool=0.99993)
            if c5_sa3 < best_c5:
                best_c5 = c5_sa3
                best_h = h_sa3.copy()
                h_bin_b = h_sa3.copy()

        # Another greedy after SA
        if remaining() > 15:
            h_g2, c5_g2 = binary_greedy(h_bin_b, best_n, top_k=5, max_iters=2000,
                                         n_cand_o=20, n_cand_z=20)
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()

    # Phase 4: Final continuous refinement
    if remaining() > 15:
        h_final = (best_h > 0.5).astype(float).copy()
        for beta in [20000, 60000, 150000]:
            if remaining() < 8:
                break
            h_final = lbfgs_opt(h_final, best_n, beta, top_k=6, maxiter=300)
            c5_f = compute_c5(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    # Phase 5: If still have time, try a few more N values near the best
    if remaining() > 30:
        n_final = best_n
        extra_ns = []
        for d in range(-4, 5, 2):
            nc = n_final + d
            if 120 <= nc <= 600 and nc % 2 == 0 and nc != n_final:
                extra_ns.append(nc)
        for n_e in extra_ns[:4]:
            if remaining() < 20:
                break
            h_ie = interp_to(best_h, best_n, n_e)
            h_fe = proj(h_ie, n_e)
            h_oe, c5_e = optimize_at_n(h_fe, n_e, time_alloc=min(40, remaining() - 10), rng=rng)
            if c5_e < best_c5:
                best_c5 = c5_e
                best_h = h_oe.copy()
                best_n = n_e

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5
        best_h = proj(best_h, best_n)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
