    # Find tau such that sum(clip(h - tau, 0, 1)) = target

        else:

            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr)
        
        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > 400:
                for _ in range(4):
                    i = ones_i[rng.randint(len(ones_i))]
                    j = zeros_i[rng.randint(len(zeros_i))]
                    if h[i] == 1 and h[j] == 0:
                        dr = make_swap_delta(h, i, j, n)
                        h[i] = 0
                        h[j] = 1
                        corr = corr + dr
                ones_i = np.where(h == 1)[0]
                zeros_i = np.where(h == 0)[0]
                cur_max = float(np.max(corr))
                stall = 0
    
    return best_h, best_max * dx

def top_shifts_reduce(h, n, top_k=8, max_iters=300):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        c5_cur = float(np.max(corr)) * (2.0 / n)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        
        idx = np.argsort(-corr)[:top_k]
        c_top_max = corr[idx[0]]
        c_top_min = corr[idx[-1]]
        if c_top_max - c_top_min < 1e-12:
            break
        
        g_avg = np.zeros(n)
        for k in idx:
            o = k - (n - 1)
            if o >= 0:
                m = n - o
                g_avg[o:n] += (1.0 - h[:m])
                g_avg[:m] -= h[o:n]
            else:
                p = -o
                m = n - p
                g_avg[:m] += (1.0 - h[p:])
                g_avg[p:] -= h[:m]
        g_avg /= top_k
        gn = np.linalg.norm(g_avg)
        if gn < 1e-14:
            break
        
        improved = False
        for step in [0.02, 0.01, 0.005, 0.05, 0.002, 0.03, 0.08, 0.015, 0.04]:
            h_new = project_feasible(h - step * g_avg / gn, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def directed_hill_climb(h, n, max_iters, rng):
    """Steepest descent on exact max using subgradient."""
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        cmax = float(np.max(corr))
        c5_cur = cmax * dx
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Subgradient: average over all k at (or near) the max
        thresh = cmax - 0.05 * cmax
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        g = np.zeros(n)
        for k in active:
            o = k - (n - 1)
            if o >= 0:
                m = n - o
                g[o:n] += (1.0 - h[:m])
                g[:m] -= h[o:n]
            else:
                p = -o
                m = n - p
                g[:m] += (1.0 - h[p:])
                g[p:] -= h[:m]
        g /= len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        improved = False
        for step in [0.03, 0.015, 0.008, 0.06, 0.004, 0.02, 0.01, 0.05, 0.025]:
            h_new = project_feasible(h - step * g / gn, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                h = h_new
                best_c5 = c5_new
                best_h = h_new.copy()
                improved = True
                break
        if not improved:
            # Try random perturbation
            for _ in range(5):
                mask = rng.rand(n) < 0.3
                pert = (rng.rand(n) - 0.5) * 0.1 * mask
                h_new = project_feasible(h + pert, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: Binary local search at parent resolution
    n1 = best_n
    h_bin1 = (best_h > 0.5).astype(float)
    n1_half = n1 // 2
    n1_ones = int(np.sum(h_bin1))
    if n1_ones != n1_half:
        sort_idx = np.argsort(-best_h)
        h_bin1 = np.zeros(n1)
        h_bin1[sort_idx[:n1_half]] = 1.0
    
    t1 = min(150, int(remaining() * 0.12))
    if t1 > 20:
        h_b1, c5_b1 = binary_local_search(h_bin1, n1, time_limit=t1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n1
    
    # Also try Thue-Morse binary search at parent res
    h_tm = thue_morse(n1)
    c5_tm = compute_c5(h_tm)
    if c5_tm < best_c5 + 0.01:
        t1b = min(80, int(remaining() * 0.06))
        if t1b > 20:
            h_tm2, c5_tm2 = binary_local_search(h_tm, n1, time_limit=t1b, rng=np.random.RandomState(seed+7))
            if c5_tm2 < best_c5:
                best_c5 = c5_tm2
                best_h = h_tm2.copy()
                best_n = n1
    
    # Phase 2: Upsample to higher resolution + continuous optimization
    n2 = min(best_n + 64, 448)
    h2 = interp_to(best_h, best_n, n2)
    h2 = project_feasible(h2, n2)
    
    for beta in [30, 80, 200, 500, 1200, 3000, 7000, 15000]:
        if remaining() < 25:
            break
        max_iter = min(600, int(remaining() * 0.06))
        if max_iter < 30:
            break
        h2, _ = projected_gradient(h2, n2, beta, max_iter=max_iter, step_init=0.04)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2
    
    # Phase 3: Directed hill climb (subgradient on exact max)
    if remaining() > 40:
        h3, c5_3 = directed_hill_climb(best_h, best_n, max_iters=500, rng=rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = best_n
    
    # Phase 4: Top shifts reduction
    if remaining() > 30:
        h4, c5_4 = top_shifts_reduce(best_h, best_n, top_k=6, max_iters=250)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Phase 5: Try higher resolution
    n5 = min(best_n + 64, 576)
    if remaining() > 180 and n5 != best_n and n5 <= 576:
        h5 = interp_to(best_h, best_n, n5)
        h5 = project_feasible(h5, n5)
        for beta in [100, 400, 1000, 3000, 8000, 20000]:
            if remaining() < 20:
                break
            max_iter = min(500, int(remaining() * 0.05))
            if max_iter < 30:
                break
            h5, _ = projected_gradient(h5, n5, beta, max_iter=max_iter, step_init=0.03)
            c5_5 = compute_c5(h5)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()
                best_n = n5
        if remaining() > 30:
            h5, c5_5b = top_shifts_reduce(h5, n5, top_k=8, max_iters=200)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5.copy()
                best_n = n5
        if remaining() > 30:
            h5, c5_5c = directed_hill_climb(h5, n5, max_iters=300, rng=rng)
            if c5_5c < best_c5:
                best_c5 = c5_5c
                best_h = h5.copy()
                best_n = n5
    
    # Phase 6: Binary local search at best resolution
    n6 = best_n
    h_bin6 = (best_h > 0.5).astype(float)
    n6_half = n6 // 2
    n6_ones = int(np.sum(h_bin6))
    if n6_ones != n6_half:
        sort_idx = np.argsort(-best_h)
        h_bin6 = np.zeros(n6)
        h_bin6[sort_idx[:n6_half]] = 1.0
    t6 = min(120, int(remaining() * 0.12))
    if t6 > 20:
        h_b6, c5_b6 = binary_local_search(h_bin6, n6, time_limit=t6, rng=np.random.RandomState(seed+13))
        if c5_b6 < best_c5:
            best_c5 = c5_b6
            best_h = h_b6.copy()
            best_n = n6
    
    # Phase 7: Final continuous polish
    n7 = best_n
    h7 = best_h.copy()
    for beta in [5000, 15000, 40000]:
        if remaining() < 10:
            break
        max_iter = min(400, int(remaining() * 0.06))
        if max_iter < 20:
            break
        h7, _ = projected_gradient(h7, n7, beta, max_iter=max_iter, step_init=0.02)
        c5_7 = compute_c5(h7)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
            best_n = n7
    
    if remaining() > 20:
        h7, c5_7b = top_shifts_reduce(h7, n7, top_k=10, max_iters=200)
        if c5_7b < best_c5:
            best_c5 = c5_7b
            best_h = h7.copy()
            best_n = n7
    
    if remaining() > 15:
        h7, c5_7c = directed_hill_climb(h7, n7, max_iters=200, rng=np.random.RandomState(seed+99))
        if c5_7c < best_c5:
            best_c5 = c5_7c
            best_h = h7.copy()
            best_n = n7
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
