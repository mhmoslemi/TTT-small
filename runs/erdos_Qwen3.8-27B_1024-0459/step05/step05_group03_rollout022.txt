import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def thue_morse_binary(n):
    tm = np.zeros(n, dtype=float)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        tm[i] = float(c & 1)
    return tm

def fix_binary_count(h, n, rng):
    h = h.copy()
    ones_count = int(np.round(np.sum(h)))
    target = n // 2
    if ones_count > target:
        idx = np.where(h == 1)[0]
        to_flip = rng.choice(idx, size=ones_count - target, replace=False)
        h[to_flip] = 0.0
    elif ones_count < target:
        idx = np.where(h == 0)[0]
        to_flip = rng.choice(idx, size=target - ones_count, replace=False)
        h[to_flip] = 1.0
    return h

def binary_sa(h0, n, n_iter=30000, T0=0.5, T1=0.005, rng=None, callback=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = (h0 > 0.5).astype(float).copy()
    h = fix_binary_count(h, n, rng)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    best_h = h.copy()
    best_max = c_max
    ones = np.where(h == 1)[0].astype(int)
    zeros = np.where(h == 0)[0].astype(int)
    dx = 2.0 / n
    for it in range(n_iter):
        if remaining() < 3:
            break
        T = T0 * (T1 / T0) ** (it / float(n_iter))
        pa = int(ones[rng.randint(len(ones))])
        pb = int(zeros[rng.randint(len(zeros))])
        h[pa] = 0.0
        h[pb] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - c_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c_max = new_max
            corr = new_corr
            ones = np.where(h == 1)[0].astype(int)
            zeros = np.where(h == 0)[0].astype(int)
            if c_max < best_max:
                best_max = c_max
                best_h = h.copy()
        else:
            h[pb] = 0.0
            h[pa] = 1.0
        if callback and it % 5000 == 0:

    dx = 2.0 / n

        if remaining() < 5:

def continuous_refine(h0, n, max_iters=2000, beta=2000.0, rng=None):
    """Smooth-max projected gradient with adaptive step."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = project_feasible(h0.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.7
    step = 0.02
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        w = w * dx
        # Gradient of weighted sum of correlations
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:a + np_])
                g[a:a + np_] -= wk * h[:np_]
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            step *= 1.01
        else:
            step *= 0.95
            if it % 20 == 0:
                h = project_feasible(h - 0.5 * momentum, n)
        step = min(step, 0.1)
        if it % 200 == 0:
            mu = min(0.9, mu + 0.02)
    return best_h, best_c5

def slsqp_smooth(h0, n, beta, maxiter=300):
    """SLSQP with smooth max objective."""
    h = project_feasible(h0.copy(), n)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        w = w * dx
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:a + np_])
                g[a:a + np_] -= wk * x[:np_]
        return g

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj(h) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):

        # Quick greedy
        g_rng2 = np.random.RandomState(seed * 500 + n_try)
        h_g2, c5_g2 = greedy_binary_fast(h_bin_t, n_try, max_rounds=10, n_samples=6000, rng=g_rng2)
        if c5_g2 < c5_bin_t:
            c5_bin_t = c5_g2
            h_bin_t = h_g2

        # Continuous refinement
        if remaining() > 20:
            h_c2 = project_feasible(h_bin_t + rng.uniform(-0.03, 0.03, n_try), n_try)
            h_r2, c5_r2 = continuous_refine(h_c2, n_try, max_iters=800, beta=2000.0)
            if c5_r2 < c5_bin_t:
                c5_bin_t = c5_r2
                h_bin_t = h_r2

        if c5_bin_t < best_c5:
            best_c5 = c5_bin_t
            best_h = h_bin_t.copy()
            best_n = n_try

    # === Phase 4: Final intensive polish ===
    if remaining() > 15:
        for beta in [30000, 80000, 200000]:
            if remaining() < 8:
                break
            h_p = slsqp_smooth(best_h, best_n, beta, maxiter=300)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Final gradient descent
    if remaining() > 10:
        h_f, c5_f = continuous_refine(best_h, best_n, max_iters=1000, beta=5000.0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final binary check
    if remaining() > 10:
        h_bf = round_to_binary(best_h, best_n)
        c5_bf = compute_c5(h_bf)
        if c5_bf < best_c5:
            best_h = h_bf.copy()
            best_c5 = c5_bf

    # Validate and finalize
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
