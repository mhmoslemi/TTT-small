To reduce corr[k*], I want to choose j and δ such that δ * g_k*[j] < 0 while not increasing other top lags too much.

This is essentially a coordinate descent / Frank-Wolfe type approach. Let me implement a simple version:

        # Increasing h[j] when g_max[j] < 0 helps
        # Decreasing h[j] when g_max[j] > 0 helps

        best_delta = 0

            A_rows.append(np.append(-I[i], 0.0))
            b_vals.append(-hc[i] + delta)
        Aeq = np.zeros((1, n + 1))
        Aeq[0, :n] = 1.0
        beq = np.array([n / 2.0])
        A_arr = np.array(A_rows)
        b_arr = np.array(b_vals)
        try:
            r = linprog(co, A_ub=A_arr, b_ub=b_arr, A_eq=Aeq, b_eq=beq,
                        bounds=[(0, 1)] * n + [(None, None)], method='highs')
            if r.success:
                hc2 = project_feasible(r.x[:n], n)
                c52 = compute_c5(hc2)
                if c52 < c5c - 1e-12:
                    hc = hc2
                    c5c = c52
                else:
                    ba, bc = 0.0, c5c
                    for a in [0.3, 0.5, 0.7, 1.0]:
                        ht = project_feasible(h + a * (hc2 - h), n)
                        ct = compute_c5(ht)
                        if ct < bc:
                            bc, ba = ct, a
                    if ba > 0:
                        hc = project_feasible(h + ba * (hc2 - h), n)
                        c5c = compute_c5(hc)
            else:
                break
        except Exception:
            break
    return hc, c5c

def interp(src, sn, dn):
    xp = (np.arange(sn) + 0.5) * 2.0 / sn
    xn = (np.arange(dn) + 0.5) * 2.0 / dn
    return np.interp(xn, xp, src)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    bh = None
    bc5 = float('inf')
    bn = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        ph = project_feasible(ph, pn)
        bh, bc5, bn = ph.copy(), compute_c5(ph), pn
    except Exception:
        pn = 512
        ph = np.zeros(pn)
        ph[:pn // 2] = 1.0
        ph = project_feasible(ph, pn)
        bh, bc5, bn = ph.copy(), compute_c5(ph), pn

    # Phase 1: quick subgradient + SLSQP at parent resolution
    if remaining() > 20:
        hsg, c5sg = subgrad(bh, bn, K=25, bw=30, max_iter=100)
        if c5sg < bc5:
            bc5, bh = c5sg, hsg.copy()
    if remaining() > 15:
        hn = slsqp(bh, bn, 1000, 200, K=40)
        cn = compute_c5(hn)
        if cn < bc5:
            bc5, bh = cn, hn.copy()

    # Phase 2: interpolate to 1024 and optimize
    n2 = 1024
    h2 = project_feasible(interp(bh, bn, n2), n2)
    c52 = compute_c5(h2)
    hcur = h2.copy()

    if remaining() > 40:
        hcur, c52s = subgrad(hcur, n2, K=25, bw=30, max_iter=80)
        if c52s < c52:
            c52 = c52s
            hcur = hcur.copy()

    for beta in [500, 1500, 5000, 15000, 40000]:
        if remaining() < 15:
            break
        hcur = slsqp(hcur, n2, beta, 300)
        c52 = compute_c5(hcur)
        if c52 < bc5:
            bc5, bh, bn = c52, hcur.copy(), n2

    # Phase 3: cutting plane at best resolution
    if remaining() > 30:
        hc, c5c = cutplane(bh, bn, K=30, delta=0.03, max_iters=80)
        if c5c < bc5:
            bc5, bh = c5c, hc.copy()

    # SLSQP after cutting plane
    for beta in [10000, 30000, 80000]:
        if remaining() < 10:
            break
        hn = slsqp(bh, bn, beta, 400)
        cn = compute_c5(hn)
        if cn < bc5:
            bc5, bh = cn, hn.copy()

    # Phase 4: try 2048 if time allows
    if remaining() > 120:
        n3 = 2048
        h3 = project_feasible(interp(bh, bn, n3), n3)
        c53 = compute_c5(h3)
        hcur3 = h3.copy()

        for beta in [1000, 5000, 15000]:
            if remaining() < 20:
                break
            hcur3 = slsqp(hcur3, n3, beta, 250)
            c53 = compute_c5(hcur3)
            if c53 < bc5:
                bc5, bh, bn = c53, hcur3.copy(), n3

        if remaining() > 30:
            hc3, c5c3 = cutplane(bh, bn, K=25, delta=0.02, max_iters=40)
            if c5c3 < bc5:
                bc5, bh = c5c3, hc3.copy()
            if remaining() > 10:
                hn3 = slsqp(bh, bn, 20000, 300)
                cn3 = compute_c5(hn3)
                if cn3 < bc5:
                    bc5, bh = cn3, hn3.copy()

    # Phase 5: random restarts
    if remaining() > 50:
        for rep in range(6):
            if remaining() < 30:
                break
            sigma = 0.02 + 0.01 * rep
            hp = project_feasible(bh + rng.randn(bn) * sigma, bn)
            hp = subgrad(hp, bn, K=20, bw=30, max_iter=50)
            hp = slsqp(hp, bn, 3000, 200, K=30)
            hp = slsqp(hp, bn, 10000, 200, K=15)
            c5p = compute_c5(hp)
            if c5p < bc5:
                bc5, bh = c5p, hp.copy()

    # Phase 6: final intensive polish
    for beta in [20000, 60000, 120000]:
        if remaining() < 8:
            break
        hn = slsqp(bh, bn, beta, 500)
        cn = compute_c5(hn)
        if cn < bc5:
            bc5, bh = cn, hn.copy()

    # Final subgradient micro-adjustments
    if remaining() > 8:
        hf, c5f = subgrad(bh, bn, K=15, bw=80, max_iter=100)
        if c5f < bc5:
            bc5, bh = c5f, hf.copy()

    for beta in [50000, 100000]:
        if remaining() < 5:
            break
        hn = slsqp(bh, bn, beta, 500)
        cn = compute_c5(hn)
        if cn < bc5:
            bc5, bh = cn, hn.copy()

    # Final feasibility
    bh = project_feasible(bh, bn)
    fc5 = compute_c5(bh)
    bc5 = min(bc5, fc5)

    try:
        ec = evaluate_erdos_solution(bh)
        if isinstance(ec, (int, float, np.floating)) and ec < bc5:
            bc5 = float(ec)
    except Exception:
        pass

    return (bh, bc5, bn)
