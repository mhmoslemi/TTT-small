import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def corr_gradient_at(h, k_off, n):
    g = np.zeros(n)
    if k_off >= 0:
        np_ = n - k_off
        if np_ > 0:
            g[k_off:k_off + np_] += (1.0 - h[:np_])
            g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        np_ = n - ao
        if np_ > 0:
            g[:np_] += (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= h[:np_]
    return g

def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v

def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

def _sc(x):
    return np.sum(x) - _TGT

def _scj(x):
    return np.ones(_N)

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]

def peak_shave(h, n, top_k=8, max_rounds=50, target_reductions=0.001):
    """Iteratively reduce the top correlations by solving focused subproblems."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for round_ in range(max_rounds):
        if remaining() < 5:
            break
        
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        max_val = corr[top_idx[0]]
        current_max_c5 = max_val * dx
        
        # Create an objective that penalizes the top correlations
        # More weight on the very top ones
        weights = np.zeros(len(corr))
        for rank, idx in enumerate(top_idx):
            weights[idx] = 1.0 / (rank + 1)
        weights /= weights.sum()
        
        # Define a weighted sum objective
        def peak_obj(x):
            c = compute_corr(x)
            return float(np.dot(weights, c) * dx)
        
        def peak_grad(x):
            c = compute_corr(x)
            g = np.zeros(n)
            for k_idx in range(len(c)):
                w = weights[k_idx] * dx
                if w < 1e-15:
                    continue
                off = k_idx - (n - 1)
                g += w * corr_gradient_at(x, off, n)
            return g
        
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
        
        try:
            res = minimize(peak_obj, h, jac=peak_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
            if res.fun < peak_obj(h):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    h = h_new.copy()
                else:
                    # Try to use the result even if max didn't improve
                    # (other correlations may have decreased)
                    h = h_new.copy()
        except Exception:
            pass
        
        if current_max_c5 - best_c5 < 1e-8:
            break
    
    return h, best_c5

    if off >= 0:
        np_ = n - off
        if np_ > 0:
            g[off:off+np_] += (1.0 - h[:np_])
            g[:np_] -= h[off:off+np_]
    else:
        ao = -off
        np_ = n - ao
        if np_ > 0:
            g[:np_] += (1.0 - h[ao:ao+np_])
            g[ao:ao+np_] -= h[:np_]

    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v

        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            for idx in perm[:n_ones - n_half]:
                h[ones_idx[idx]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            for idx in perm[:n_half - n_ones]:
                h[zeros_idx[idx]] = 1
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def focused_binary_search(h_bin, n, rng, time_limit=200):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones > n_half:
        ones_idx = np.where(h == 1)[0]
        perm = rng.permutation(len(ones_idx))
        for idx in perm[:n_ones - n_half]:
            h[ones_idx[idx]] = 0
    elif n_ones < n_half:
        zeros_idx = np.where(h == 0)[0]
        perm = rng.permutation(len(zeros_idx))
        for idx in perm[:n_half - n_ones]:
            h[zeros_idx[idx]] = 1

    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = int(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start_time = time.time()
    stagnation = 0
    improve_count = 0

    while time.time() - start_time < time_limit and remaining() > 5:
        corr = compute_corr(h)
        max_val = int(np.max(corr))
        n_top = min(15, 2 * n - 1)
        top_idx = np.argsort(-corr)[:n_top]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_swap = None
        n_samples = 60

        for _ in range(n_samples):
            r = rng.random()
            if r < 0.55:
                # Targeted: pick from hot correlation region
                k_hot = top_idx[rng.randint(min(6, len(top_idx)))]
                off = k_hot - (n - 1)
                if off >= 0:
                    hot_mask = (h[:n-off] == 1) & (h[off:] == 0)
                    candidates = np.where(hot_mask)[0]
                    if len(candidates) > 0:
                        i = candidates[rng.randint(len(candidates))]
                    else:
                        i = ones[rng.randint(len(ones))]
                else:
                    ao = -off
                    hot_mask = (h[ao:] == 1) & (h[:n-ao] == 0)
                    candidates = np.where(hot_mask)[0]
                    if len(candidates) > 0:
                        i = candidates[rng.randint(len(candidates))]
                    else:
                        i = ones[rng.randint(len(ones))]
            elif r < 0.8:
                # Semi-targeted: pick 1 that participates in multiple top correlations
                i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]

            # For j, try to pick a zero in a "cold" region
            if rng.random() < 0.4:
                # Pick zero near positions where h is already 1 (to cluster)
                if len(ones) > 0:
                    anchor = ones[rng.randint(len(ones))]
                    offset = rng.randint(-5, 6)
                    j_candidate = (anchor + offset) % n
                    if h[j_candidate] == 0 and j_candidate != i:
                        j = j_candidate
                    else:
                        j = zeros[rng.randint(len(zeros))]
                else:
                    j = zeros[rng.randint(len(zeros))]
            else:
                j = zeros[rng.randint(len(zeros))]

            if i == j or h[i] != 1 or h[j] != 0:
                continue

            h[i] = 0.0
            h[j] = 1.0
            new_max = int(np.max(compute_corr(h)))
            delta = new_max - max_val
            h[i] = 1.0
            h[j] = 0.0

            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)

        # Try a few double-swap moves
        for _ in range(8):
            i1 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            if j1 == i1:
                continue
            i2 = ones[rng.randint(len(ones))]
            j2 = zeros[rng.randint(len(zeros))]
            if i2 == i1 or j2 == j1 or j2 == i1 or i2 == j1:
                continue
            if h[i1] != 1 or h[j1] != 0 or h[i2] != 1 or h[j2] != 0:
                continue

            h[i1] = 0.0; h[j1] = 1.0
            h[i2] = 0.0; h[j2] = 1.0
            new_max = int(np.max(compute_corr(h)))
            delta = new_max - max_val
            h[i1] = 1.0; h[j1] = 0.0
            h[i2] = 1.0; h[j2] = 0.0
            if delta < best_delta:
                best_delta = delta
                best_swap = (i1, j1, i2, j2)

        if best_swap is not None and best_delta < 0:
            if len(best_swap) == 2:
                h[best_swap[0]] = 0.0
                h[best_swap[1]] = 1.0
            else:
                h[best_swap[0]] = 0.0; h[best_swap[1]] = 1.0
                h[best_swap[2]] = 0.0; h[best_swap[3]] = 1.0
            stagnation = 0
            improve_count += 1
            cur_max += best_delta
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stagnation += 1
            if stagnation > 80:
                n_pert = rng.randint(2, min(5, n_samples // 10 + 2))
                for _ in range(n_pert):
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    a = ones_now[rng.randint(len(ones_now))]
                    b = zeros_now[rng.randint(len(zeros_now))]
                    h[a] = 0.0
                    h[b] = 1.0
                cur_max = int(np.max(compute_corr(h)))
                stagnation = 0

    return best_h, best_max * dx_v

def coordinate_descent_binary(h, n, rng, n_rounds=200):
    """Try flipping each coordinate individually."""
    h = h.copy()
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = int(np.max(corr))
    best_h = h.copy()
    best_max = cur_max

    for rd in range(n_rounds):
        if remaining() < 5:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            new_val = 1.0 - old_val
            h[i] = new_val
            nm = int(np.max(compute_corr(h)))
            if nm < cur_max:
                cur_max = nm
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            else:
                h[i] = old_val
        if not improved:
            break
    return best_h, best_max * dx_v

def peak_shave_iter(h, n, top_k=8, max_rounds=30):
    """Iteratively focus SLSQP on the top correlations."""
    global _N, _TGT
    _N = n
    _TGT = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_v = 2.0 / n

    for round_ in range(max_rounds):
        if remaining() < 8:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        max_val = corr[top_idx[0]]

        # Build weighted objective focusing on top correlations
        weights = np.zeros(2 * n - 1)
        for rank in range(top_k):
            weights[top_idx[rank]] = 1.0 / (rank + 1.0)
        weights /= weights.sum()

        def wobj(x):
            c = compute_corr(x)
            return float(np.dot(weights, c) * dx_v)

        def wgrad(x):
            c = compute_corr(x)
            g = np.zeros(n)
            for kk in range(len(c)):
                wk = weights[kk] * dx_v
                if wk < 1e-15:
                    continue
                off = kk - (n - 1)
                g += wk * corr_grad_at(x, off, n)
            return g

        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sumc, 'jac': _sumcj}]
        try:
            res = minimize(wobj, h, jac=wgrad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-16, 'iprint': 0})
            if res.fun < wobj(h):
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                h = h_new.copy()
        except Exception:
            pass

    return h, best_c5

def top_shifts_dir(h, n, top_k=5, max_iters=200):
    """Directional search to equalize top correlations."""
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-9:
            break
        k_top = idx[0]
        k_bot = idx[-1]
        off_t = k_top - (n - 1)
        off_b = k_bot - (n - 1)
        g_t = corr_grad_at(h, off_t, n)
        g_b = corr_grad_at(h, off_b, n)
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015, 0.5, 0.2]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse_seq(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP refinement on parent
    for beta in [200, 1000, 5000, 20000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-shifts directional search
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_dir(h, n, top_k=5, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: INTENSIVE BINARY LOCAL SEARCH
    # This is the main workhorse - try to get max_corr <= 73 for n=384
    if remaining() > 50:
        h_bin = binarize(h, n)
        c5_bin_target = compute_c5(h_bin)

        # Multiple rounds of binary search with restarts
        sa_time1 = min(500, max(60, int(remaining() * 0.45)))
        h_b, c5_b = focused_binary_search(h_bin, n, rng, time_limit=sa_time1)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()

        # Second pass from randomized perturbation
        if remaining() > 100:
            h_b2 = h_b + rng.randn(n) * 0.3
            h_bin2 = binarize(h_b2, n)
            sa_time2 = min(300, max(30, int(remaining() * 0.25)))
            h_b2r, c5_b2 = focused_binary_search(h_bin2, n, rng, time_limit=sa_time2)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2r.copy()
                h = h_b2r.copy()

        # Third pass from Thue-Morse
        if remaining() > 80:
            h_tm = thue_morse_seq(n, rng)
            sa_time3 = min(200, max(20, int(remaining() * 0.2)))
            h_tm_r, c5_tm = focused_binary_search(h_tm, n, rng, time_limit=sa_time3)
            if c5_tm < best_c5:
                best_c5 = c5_tm
                best_h = h_tm_r.copy()
                h = h_tm_r.copy()

    # Phase 4: Coordinate descent on best binary
    if remaining() > 30:
        h_bin_best = (best_h > 0.5).astype(float).copy()
        h_cd, c5_cd = coordinate_descent_binary(h_bin_best, n, rng, n_rounds=100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 5: Continuous refinement from best
    if remaining() > 20:
        h_cont = project_feasible(h, n)
        for beta in [5000, 20000, 50000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(h_cont, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cont = h_new.copy()
        if remaining() > 10:
            h_eq, c5_eq = top_shifts_dir(h_cont, n, top_k=8, max_iters=200)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()

    # Phase 6: Peak shaving
    if remaining() > 30:
        h_ps, c5_ps = peak_shave_iter(best_h, n, top_k=10, max_rounds=20)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()

    # Phase 7: Try different resolutions
    if remaining() > 60:
        n_base = best_n
        n_candidates = []
        for delta in [-4, -2, 2, 4, 8, 16, 32, -8, -16]:
            nt = n_base + delta
            if 128 <= nt <= 600 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [256, 320, 384, 448, 512]:
            if 128 <= nt <= 600 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates.sort()

        for n_t in n_candidates[:6]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)

            # Quick SLSQP
            for beta in [2000, 10000]:
                if remaining() < 8:
                    break
                h_t = slsqp_smooth(h_t, n_t, beta, maxiter=200)

            # Binary search at this resolution
            h_tb = binarize(h_t, n_t)
            t_limit = min(60, max(10, int(remaining() * 0.15)))
            h_tb, c5_tb = focused_binary_search(h_tb, n_t, rng, time_limit=t_limit)

            c5_t = compute_c5(h_tb)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_tb.copy()
                best_n = n_t

    # Phase 8: Final intensive binary polish on best
    if remaining() > 40:
        h_final_bin = (best_h > 0.5).astype(float).copy()
        n_fin = best_n
        t_final = min(200, max(20, int(remaining() * 0.4)))
        h_fin, c5_fin = focused_binary_search(h_final_bin, n_fin, rng, time_limit=t_final)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

        # Final continuous touch
        if remaining() > 10:
            h_fin_c = slsqp_smooth(h_fin, n_fin, 30000, maxiter=300)
            c5_fin_c = compute_c5(h_fin_c)
            if c5_fin_c < best_c5:
                best_c5 = c5_fin_c
                best_h = h_fin_c.copy()

    # Phase 9: Final peak shave + directional
    if remaining() > 15:
        h_ps2, c5_ps2 = peak_shave_iter(best_h, best_n, top_k=12, max_rounds=10)
        if c5_ps2 < best_c5:
            best_c5 = c5_ps2
            best_h = h_ps2.copy()

    if remaining() > 10:
        h_eq2, c5_eq2 = top_shifts_dir(best_h, best_n, top_k=6, max_iters=150)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
