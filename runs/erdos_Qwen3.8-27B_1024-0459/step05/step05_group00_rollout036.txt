def project_feasible(h, n):

def grad_corr_k(h, k, n):

def cutting_plane_iterative(h, n, time_limit, K=30):

            fk = corr[k]

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        cnt = np.count_nonzero(mask)

            h[mask] += diff / cnt
        else:

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:offset + npairs] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:offset + npairs]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:])
            g[a:] -= h[:npairs]
    return g

                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                v *= 0.5
    return best_h, best_c5


def cutting_plane_small(h, n, time_limit, K=25, delta=0.015):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stuck = 0
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub_list = []
        b_ub_list = []
        for k in idx:
            g = grad_corr_k(h, k, n)
            fk = corr[k]
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(-fk + np.dot(g, h))
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        I = np.eye(n)
        A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                             np.hstack([-I, np.zeros((n, 1))])])
        b_trust = np.concatenate([h + delta, h + delta])
        A_ub = np.vstack([A_ub, A_trust])
        b_ub = np.concatenate([b_ub, b_trust])
        bounds = [(0, 1)] * n + [(None, None)]
        improved = False
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                h_lp = res.x[:n]
                best_val = compute_c5(h)
                for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        h = h_try
                        improved = True
                        if c_try < best_c5:
                            best_c5 = c_try
                            best_h = h.copy()
                        break
        except Exception:
            pass
        if not improved:
            stuck += 1
            if stuck > 30:
                break
        else:
            stuck = 0
    return best_h, best_c5


def equalize_precise(h, n, K=4, time_limit=60.0):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for it in range(1000):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        c_max_v = float(c_top[0])
        c_min_v = float(c_top[-1])
        if c_max_v - c_min_v < 1e-13:
            break
        g = np.zeros(n)
        for k in idx:
            w = (corr[k] - c_min_v + 1e-10) ** 2
            g += w * grad_corr_k(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g_dir = -g / gn
        improved = False
        for step in [0.04, 0.02, 0.01, 0.03, 0.005, 0.06, 0.015, 0.008]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            for step in [0.001, 0.002, 0.0005, 0.003]:
                ht = project_feasible(h + step * g_dir, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    break
            else:
                if it > 50:
                    break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def micro_optimize(h, n, time_limit=10.0):
    """Very fine gradient steps on the top 2-3 lags."""
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for it in range(500):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 2:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:3]
        g = np.zeros(n)
        for k in top_idx:
            g += grad_corr_k(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        for step in [0.004, 0.002, 0.001, 0.003, 0.006, 0.0015, 0.0008]:
            h_try = project_feasible(h - step * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-16:
                best_c5 = c_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
                break
        if not improved and it > 100:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.random.rand(parent_n)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

    n = best_n
    h = best_h.copy()

    # Phase 1: Momentum subgradient (25s)
    if remaining() > 25:
        h_sg, c5_sg = momentum_subgrad(h, n, time_limit=min(25.0, remaining()), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: SLSQP with escalating beta (up to 150s)
    beta_list = [200, 500, 1500, 5000, 15000, 50000, 150000]
    for beta in beta_list:
        if remaining() < 12:
            break
        t_start = time.time()
        h_new = slsqp_smooth_max(h, n, beta, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        elapsed = time.time() - t_start
        if elapsed > 25:
            break

    # Phase 3: Equalize top lags (45s)
    if remaining() > 30:
        h_eq, c5_eq = equalize_precise(h, n, K=5, time_limit=min(45.0, remaining()))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Cutting plane small steps (100s)
    if remaining() > 50:
        h_cp, c5_cp = cutting_plane_small(h, n, time_limit=min(100.0, remaining()), K=30, delta=0.012)
        if c5_cp < best_c5:
            best_c5 = c5_cp
            best_h = h_cp.copy()
            h = h_cp.copy()

    # Phase 5: SLSQP again with high beta (60s)
    for beta in [50000, 200000, 500000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth_max(h, n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Equalize + micro (40s)
    if remaining() > 20:
        h_eq, c5_eq = equalize_precise(h, n, K=4, time_limit=min(25.0, remaining()))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
        if remaining() > 10:
            h_mc, c5_mc = micro_optimize(h, n, time_limit=min(15.0, remaining()))
            if c5_mc < best_c5:
                best_c5 = c5_mc
                best_h = h_mc.copy()
                h = h_mc.copy()

    # Phase 7: Try higher resolution if time allows
    if remaining() > 100:
        n_t = min(1024, best_n * 2)
        if n_t > best_n:
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)
            c5_int = compute_c5(h_feas)
            if remaining() > 50:
                h_sg2, c5_sg2 = momentum_subgrad(h_feas, n_t, time_limit=min(25.0, remaining() * 0.3), rng)
                if c5_sg2 < c5_int:
                    h_feas = h_sg2.copy()
                    c5_int = c5_sg2
            if remaining() > 30:
                h_cp2, c5_cp2 = cutting_plane_small(h_feas, n_t, time_limit=min(35.0, remaining() * 0.35), K=20, delta=0.01)
                if c5_cp2 < c5_int:
                    h_feas = h_cp2.copy()
                    c5_int = c5_cp2
            if remaining() > 15:
                h_feas = slsqp_smooth_max(h_feas, n_t, 20000, maxiter=800)
                c5_int = compute_c5(h_feas)
            if c5_int < best_c5:
                best_c5 = c5_int
                best_h = h_feas.copy()
                best_n = n_t

    # Phase 8: Final intensive polishing at best resolution
    n = best_n
    h = best_h.copy()

    for beta in [200000, 800000]:
        if remaining() < 8:
            break
        h = slsqp_smooth_max(h, n, beta, maxiter=2000)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    if remaining() > 15:
        h_eq, c5_eq = equalize_precise(h, n, K=6, time_limit=min(20.0, remaining()))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining() > 8:
        h_mc, c5_mc = micro_optimize(h, n, time_limit=min(10.0, remaining()))
        if c5_mc < best_c5:
            best_c5 = c5_mc
            best_h = h_mc.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
