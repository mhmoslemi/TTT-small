        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 100.0 * (np.sum(x) - _TGT) ** 2

def _sm_grad(x):

    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 200.0 * (np.sum(x) - _TGT)
    return g

def binary_greedy(h, n, top_k=5, max_iters=5000):

    n_half = n // 2
    if int(np.sum(h_bin)) != n_half:
        idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)

    cur_c5 = best_c5
    
    ones = np.where(h_bin == 1)[0].copy()
    zeros = np.where(h_bin == 0)[0].copy()
    
    it = 0
    while it < max_iters and remaining() > 3:

        cur_max = float(np.max(corr))
        idx_top = np.argsort(-corr)[:top_k]
        
        # Compute gradient direction

        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        
        # Find best swap
        # Want to decrease h where g is negative, increase where g is positive
        # So: flip 1→0 where g is most negative, flip 0→1 where g is most positive
        score_remove = -g[ones]  # higher = more beneficial to remove
        score_add = g[zeros]  # higher = more beneficial to add
        
        n_try_r = min(15, len(ones))
        n_try_a = min(15, len(zeros))
        top_remove = ones[np.argsort(-score_remove)[:n_try_r]]
        top_add = zeros[np.argsort(-score_add)[:n_try_a]]
        
        best_delta = -1e-9  # need improvement

                if i == j:

                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0

def binary_sa(h, n, max_iters=3000, T_init=0.3, T_final=0.005):

    zeros = np.where(h_bin == 0)[0].copy()

        if remaining() < 3:
            break
        T = T_init * (T_final / T_init) ** (float(it) / max_iters)

            ones = np.delete(ones, np.where(ones == i)[0][0])

            h_bin[j] = 0

def interp_to(src_h, src_n, dst_n):

                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                    found = True
            if not found:
                break

        i, j = best_swap
        h_bin[i] = 0
        h_bin[j] = 1
        cur_c5 = compute_c5(h_bin)

        oi = np.where(ones == i)[0]
        if len(oi) > 0:
            ones = np.delete(ones, oi[0])
        oj = np.where(zeros == j)[0]
        if len(oj) > 0:
            zeros = np.delete(zeros, oj[0])
        ones = np.append(ones, j)
        zeros = np.append(zeros, i)

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def binary_2swap(h, n, top_k=5, max_iters=1000, n_cand=6):
    """2-swap: exchange two 1s with two 0s simultaneously."""
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if int(np.sum(h_bin)) != n_half:
        idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[idx[:n_half]] = 1.0

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    ones = np.where(h_bin == 1)[0].copy()
    zeros = np.where(h_bin == 0)[0].copy()

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]

        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)

        score_r = -g[ones]
        score_a = g[zeros]
        nc_r = min(n_cand, len(ones))
        nc_a = min(n_cand, len(zeros))
        top_r = ones[np.argsort(-score_r)[:nc_r]]
        top_a = zeros[np.argsort(-score_a)[:nc_a]]

        best_delta = -1e-9
        best_2swap = None

        for i1 in top_r:
            for i2 in top_r:
                if i1 >= i2:
                    continue
                for j1 in top_a:
                    for j2 in top_a:
                        if j1 >= j2:
                            continue
                        if i1 in (j1, j2) or i2 in (j1, j2):
                            continue
                        h_bin[i1] = 0
                        h_bin[i2] = 0
                        h_bin[j1] = 1
                        h_bin[j2] = 1
                        nc = compute_c5(h_bin)
                        h_bin[i1] = 1
                        h_bin[i2] = 1
                        h_bin[j1] = 0
                        h_bin[j2] = 0
                        delta = nc - cur_c5
                        if delta < best_delta:
                            best_delta = delta
                            best_2swap = (i1, i2, j1, j2)

        if best_2swap is None:
            break

        i1, i2, j1, j2 = best_2swap
        h_bin[i1] = 0
        h_bin[i2] = 0
        h_bin[j1] = 1
        h_bin[j2] = 1
        cur_c5 = compute_c5(h_bin)

        ones = np.setdiff1d(ones, [i1, i2])
        zeros = np.setdiff1d(zeros, [j1, j2])
        ones = np.union1d(ones, [j1, j2])
        zeros = np.union1d(zeros, [i1, i2])

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def binary_sa(h, n, max_iters=2000, T_init=0.15, T_final=0.003):
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if int(np.sum(h_bin)) != n_half:
        idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    ones = np.where(h_bin == 1)[0].copy()
    zeros = np.where(h_bin == 0)[0].copy()

    for it in range(max_iters):
        if remaining() < 3:
            break
        T = T_init * (T_final / T_init) ** (float(it) / max_iters)

        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]

        h_bin[i] = 0
        h_bin[j] = 1
        nc = compute_c5(h_bin)
        delta = nc - cur_c5

        accept = delta < 0
        if not accept and T > 1e-6:
            log_prob = -delta * n / (2.0 * T)
            if log_prob > 50:
                accept = True
            else:
                accept = np.random.random() < np.exp(min(log_prob, 50))

        if accept:
            cur_c5 = nc
            oi = np.where(ones == i)[0]
            oj = np.where(zeros == j)[0]
            if len(oi) > 0:
                ones = np.delete(ones, oi[0])
            if len(oj) > 0:
                zeros = np.delete(zeros, oj[0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coordinate_descent(h, n, max_sweeps=10, top_k=5):
    """Coordinate descent: optimize one variable at a time."""
    h = proj(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for sweep in range(max_sweeps):
        if remaining() < 3:
            break
        improved = False
        for i in range(n):
            if remaining() < 1:
                break
            old_val = h[i]
            # Try a few candidate values
            candidates = [0.0, 1.0, 0.25, 0.5, 0.75, old_val + 0.1, old_val - 0.1]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            best_v = old_val
            best_v_c5 = compute_c5(h)

            for v in candidates:
                if abs(v - old_val) < 1e-6:
                    continue
                h[i] = v
                # Adjust to maintain sum constraint approximately
                diff = h.sum() - n / 2.0
                if abs(diff) > 1e-10:
                    mask = np.arange(n) != i
                    mask = mask & (h > 1e-6) & (h < 1.0 - 1e-6)
                    cnt = np.count_nonzero(mask)
                    if cnt > 0:
                        h[mask] -= diff / cnt
                        h = np.clip(h, 0, 1)
                c5_try = compute_c5(h)
                if c5_try < best_v_c5:
                    best_v_c5 = c5_try
                    best_v = v
            h[i] = best_v
            if best_v != old_val:
                improved = True

        h = proj(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if not improved:
            break

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Continuous optimization at parent resolution
    h = best_h.copy()
    for beta in [100, 500, 2000, 10000, 30000]:
        if remaining() < 5:
            break
        h_new = lbfgs_opt(h, best_n, beta, top_k=5, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Binary greedy at parent resolution
    if remaining() > 15:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_greedy(h_bin, best_n, top_k=5, max_iters=3000, n_cand=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        h_ref = lbfgs_opt(h_b, best_n, beta=10000, top_k=5, maxiter=400)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # 2-swap at parent
    if remaining() > 15:
        h_b2, c5_b2 = binary_2swap(best_h, best_n, top_k=5, max_iters=500, n_cand=5)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
        h_ref2 = lbfgs_opt(h_b2, best_n, beta=15000, top_k=5, maxiter=300)
        c5_ref2 = compute_c5(h_ref2)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2.copy()

    # Phase 3: Multi-resolution search
    base_n = best_n
    res_list = []
    for mult in [1.5, 2, 2.5, 3, 4, 5, 6]:
        nr = int(base_n * mult)
        nr = 2 * (nr // 2)
        if 128 <= nr <= 3072 and nr != base_n:
            res_list.append(nr)
    res_list = sorted(set(res_list))

    for n_r in res_list:
        if remaining() < 40:
            break

        h_interp = interp_to(best_h, best_n, n_r)
        h_feas = proj(h_interp, n_r)

        # Continuous optimization
        for beta in [1000, 5000, 20000]:
            if remaining() < 10:
                break
            h_feas = lbfgs_opt(h_feas, n_r, beta, top_k=6, maxiter=300)
        c5_f = compute_c5(h_feas)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_feas.copy()
            best_n = n_r

        # Binary greedy
        if remaining() > 20:
            h_bin = (h_feas > 0.5).astype(float)
            h_b, c5_b = binary_greedy(h_bin, n_r, top_k=6, max_iters=1500, n_cand=8)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_r
            h_ref = lbfgs_opt(h_b, n_r, beta=15000, top_k=6, maxiter=300)
            c5_r = compute_c5(h_ref)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_ref.copy()
                best_n = n_r

        # 2-swap
        if remaining() > 15:
            h_b2, c5_b2 = binary_2swap(h_feas, n_r, top_k=5, max_iters=400, n_cand=4)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n_r
            h_r2 = lbfgs_opt(h_b2, n_r, beta=20000, top_k=5, maxiter=250)
            c5_r2 = compute_c5(h_r2)
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
                best_n = n_r

    # Phase 4: Simulated annealing at best resolution
    if remaining() > 25:
        h_bin = (best_h > 0.5).astype(float)
        h_sa, c5_sa = binary_sa(h_bin, best_n, max_iters=2000, T_init=0.1, T_final=0.002)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        h_ref = lbfgs_opt(h_sa, best_n, beta=20000, top_k=6, maxiter=400)
        c5_r = compute_c5(h_ref)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_ref.copy()

    # Phase 5: Random restarts at best resolution
    n_r = best_n
    for restart in range(3):
        if remaining() < 20:
            break
        # Start from a perturbed version
        h_pert = best_h.copy()
        n_half = n_r // 2
        frac = (h_pert > 0.5).astype(float)
        n_flip = max(1, n_r // 50)
        flip_idx = np.random.choice(n_half, size=min(n_flip, n_half), replace=False)
        ones_idx = np.where(frac == 1)[0]
        zeros_idx = np.where(frac == 0)[0]
        for fi in flip_idx:
            i = ones_idx[fi]
            j = zeros_idx[np.random.randint(len(zeros_idx))]
            if i != j:
                frac[i] = 0
                frac[j] = 1
        h_pert = frac.astype(float)

        h_g, c5_g = binary_greedy(h_pert, n_r, top_k=6, max_iters=1000, n_cand=8)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        h_ref = lbfgs_opt(h_g, n_r, beta=15000, top_k=6, maxiter=300)
        c5_r = compute_c5(h_ref)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_ref.copy()

    # Phase 6: Final intensive L-BFGS at best resolution
    n_final = best_n
    h = best_h.copy()
    for beta in [5000, 15000, 40000, 80000]:
        if remaining() < 5:
            break
        h_new = lbfgs_opt(h, n_final, beta, top_k=8, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Final binary polish
    if remaining() > 15:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_greedy(h_bin, n_final, top_k=6, max_iters=2000, n_cand=10)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        h_r = lbfgs_opt(h_b, n_final, beta=50000, top_k=8, maxiter=500)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 8: Coordinate descent refinement
    if remaining() > 15:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_sweeps=3, top_k=6)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final: ensure feasibility and validate
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
