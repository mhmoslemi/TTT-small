    return (m + np.log(np.sum(ez)) / beta) * dx

    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g

        h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        h_rand_c = project_feasible(h_rand, n)
        for beta in [2000, 20000]:
            if remaining_time() < 10:
                break
            h_rand_c = slsqp_opt(h_rand_c, n, beta, maxiter=300)
        c5_rand = compute_c5(h_rand_c)
        if c5_rand < best_c5 - 1e-14:
            best_c5 = c5_rand
            best_h = h_rand_c.copy()

    # Phase 7: Second round of binary search from best continuous solution (100s)
    if remaining_time() > 60:
        h_bin2 = to_binary(best_h, n)
        sa_time2 = min(80, int(remaining_time() * 0.4))
        h_b2, c5_b2 = binary_local_search(h_bin2, n, sa_time2, rng)
        if c5_b2 < best_c5 - 1e-14:
            best_c5 = c5_b2
            best_h = h_b2.copy()
        # Refine
        if remaining_time() > 15:
            h_b2c = project_feasible(h_b2, n)
            for beta in [5000, 50000]:
                if remaining_time() < 8:
                    break
                h_b2c = slsqp_opt(h_b2c, n, beta, maxiter=300)
            c5_b2c = compute_c5(h_b2c)
            if c5_b2c < best_c5 - 1e-14:
                best_c5 = c5_b2c
                best_h = h_b2c.copy()

    # Phase 8: Try higher resolution binary search (80s)
    if remaining_time() > 60:
        n_high = best_n + 128
        if n_high % 2 != 0:
            n_high += 1
        if n_high <= 1024:
            h_hi = interp_to(best_h, best_n, n_high)
            h_hi_bin = to_binary(h_hi, n_high)
            sa_t = min(50, int(remaining_time() * 0.5))
            h_hi_b, c5_hi_b = binary_local_search(h_hi_bin, n_high, sa_t, rng)
            if c5_hi_b < best_c5 - 1e-14:
                best_c5 = c5_hi_b
                best_h = h_hi_b.copy()
                best_n = n_high
                n = n_high
            # SLSQP on the high-res result
            if remaining_time() > 15:
                h_hi_c = project_feasible(interp_to(best_h, best_n, n_high), n_high)
                h_hi_c = slsqp_opt(h_hi_c, n_high, 20000, maxiter=300)
                c5_hi_c = compute_c5(h_hi_c)
                if c5_hi_c < best_c5 - 1e-14:
                    best_c5 = c5_hi_c
                    best_h = h_hi_c.copy()
                    best_n = n_high
                    n = n_high

    # Phase 9: Final intensive polish (remaining time)
    n = best_n
    h = best_h.copy()
    for beta in [100000, 300000, 1000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=7, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=3, max_iters=300)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Coordinate descent polish
    if remaining_time() > 15:
        h_cr, c5_cr = coordinate_descent(best_h, n, max_sweeps=10)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 500000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    n = best_n
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
