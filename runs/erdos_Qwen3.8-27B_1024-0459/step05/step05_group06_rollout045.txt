```python
def projected_cg(h, n, beta, maxiter=200):
    dx = 2.0 / n
    h = h.copy()
    d = np.zeros(n)  # search direction
    # Initial gradient
    f, g = smooth_value_grad(h, n, beta)
    # Project gradient onto feasible direction (sum=0)
    g_proj = g - np.mean(g)
    d = -g_proj
    d = project_direction(d, h, n)  # ensure 0<=h+alpha*d<=1 for some alpha>0
    for it in range(maxiter):
        # Line search
        alpha = backtracking_line_search(h, d, n, beta, f)
        h_new = h + alpha * d
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        f_new, g_new = smooth_value_grad(h_new, n, beta)
        if abs(f_new - f) < 1e-14:
            break
        g_new_proj = g_new - np.mean(g_new)
        beta_cg = np.dot(g_new_proj - g_proj, g_new_proj - g_proj) / max(np.dot(d, g_new_proj - g_proj), 1e-16)
        d_new = -g_new_proj + beta_cg * d
        d_new = project_direction(d_new, h_new, n)
        if np.linalg.norm(d_new) < 1e-14:
            d_new = -g_new_proj
        d = d_new
        h = h_new
        f = f_new
        g_proj = g_new_proj
    return h
```
