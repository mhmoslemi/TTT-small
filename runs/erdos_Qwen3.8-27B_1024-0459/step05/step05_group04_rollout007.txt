def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()

            hi = mid

        cl = np.clip(beta * corr, -50.0, 50.0)
        e = np.exp(cl)
        w = e / np.sum(e)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                m = n - off
                g[off:off+m] += wk * (1.0 - x[:m])
                g[:m] -= wk * x[off:off+m]
            else:
                ao = -off
                m = n - ao
                g[:m] += wk * (1.0 - x[ao:ao+m])
                g[ao:ao+m] -= wk * x[:m]
        return g
    
    def constraint_fun(x):
        return np.sum(x) - tgt
    
    def constraint_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k=6, max_iters=500):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                m = n - off
                g_avg[off:off+m] += (1.0 - h[:m])
                g_avg[:m] -= h[off:off+m]
            else:
                ao = -off
                m = n - ao
                g_avg[:m] += (1.0 - h[ao:ao+m])
                g_avg[ao:ao+m] -= h[:m]
        
        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                m = n - off
                g_bot[off:off+m] += (1.0 - h[:m])
                g_bot[:m] -= h[off:off+m]
            else:
                ao = -off
                m = n - ao
                g_bot[:m] += (1.0 - h[ao:ao+m])
                g_bot[ao:ao+m] -= h[:m]
        
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.02, 0.05, 0.01, 0.03, 0.08, 0.005, 0.1, 0.015, 0.04, 0.025]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5
