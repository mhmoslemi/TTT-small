    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty

def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g

def lbfgs_pass(h, n, beta, top_k, maxiter):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-11, 'maxfun': maxiter * 12})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h

def slsqp_pass(h, n, beta, top_k, maxiter):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, iprint=0,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return proj(res.x, n)
    except Exception:
        return h

def binary_sa_search(h_bin, n, max_iters, rng, top_k=4, n_cand=25):
    """Binary local search with gradient direction and simulated annealing."""
    h_bin = h_bin.copy()
    n_half = n // 2
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]

    dx = 2.0 / n
    corr = compute_corr(h_bin)
    cur_max = float(np.max(corr))
    best_h = h_bin.copy()
    best_c5 = cur_max * dx
    best_max = cur_max

    temp = 2.0
    temp_min = 0.05
    no_improve = 0
    recompute_every = 4

    for it in range(max_iters):
        if remaining() < 3:
            break

        if it % recompute_every == 0:
            corr = compute_corr(h_bin)
            cur_max = float(np.max(corr))
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_c5 = cur_max * dx
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1

        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)

        g_ones = g[ones]
        g_zeros = g[zeros]
        n_o = len(ones)
        n_z = len(zeros)
        if n_o == 0 or n_z == 0:
            break

        nc = min(n_cand, n_o)
        nc_z = min(n_cand, n_z)
        top_ones = np.argsort(-g_ones)[:nc]
        top_zeros = np.argsort(g_zeros)[:nc_z]

        best_delta = 0.5
        best_pair = None
        for ii in top_ones:
            i_pos = ones[ii]
            for jj in top_zeros:
                j_pos = zeros[jj]
                if i_pos == j_pos:
                    continue
                delta_est = -(g_ones[ii]) + g_zeros[jj]
                if delta_est < best_delta:
                    best_delta = delta_est
                    best_pair = (i_pos, j_pos)

        if best_pair is None:
            i_pos = int(ones[rng.randint(n_o)])
            j_pos = int(zeros[rng.randint(n_z)])
            if i_pos == j_pos:
                i_pos = int(ones[rng.randint(n_o)])
                if i_pos == j_pos:
                    continue
            best_pair = (i_pos, j_pos)
            best_delta = 0.0

        i_pos, j_pos = best_pair
        h_bin[i_pos] = 0
        h_bin[j_pos] = 1

        if it % recompute_every == 0:
            corr = compute_corr(h_bin)
            new_max = float(np.max(corr))
        else:
            new_max = cur_max + best_delta * 0.6

        if new_max <= cur_max + 1e-12:
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if new_max < cur_max:
                cur_max = new_max
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-8)):
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
                cur_max = new_max
            else:
                h_bin[i_pos] = 1
                h_bin[j_pos] = 0

        if it % 500 == 0 and it > 0:
            temp = max(temp * 0.96, temp_min)
        if no_improve > 2000 and temp > temp_min:
            temp = max(temp * 0.5, temp_min)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def binary_minmax_search(h_bin, n, max_iters, rng, top_k=5, n_cand=35):
    """Binary search minimizing the max over top-k lags using exact evaluation of candidates."""
    h_bin = h_bin.copy()
    n_half = n // 2
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]

    dx = 2.0 / n
    corr = compute_corr(h_bin)
    cur_max = float(np.max(corr))
    best_h = h_bin.copy()
    best_c5 = cur_max * dx
    best_max = cur_max

    temp = 1.5
    temp_min = 0.03

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h_bin)
        cur_max = float(np.max(corr))
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            best_c5 = cur_max * dx
            best_h = h_bin.copy()

        idx_top = np.argsort(-corr)[:top_k]
        c_top = corr[idx_top]

        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)

        g_ones = g[ones]
        g_zeros = g[zeros]
        nc = min(n_cand, len(ones))
        nc_z = min(n_cand, len(zeros))
        top_ones = np.argsort(-g_ones)[:nc]
        top_zeros = np.argsort(g_zeros)[:nc_z]

        best_delta = cur_max
        best_pair = None
        for ii in range(nc):
            i_pos = ones[top_ones[ii]]
            for jj in range(nc_z):
                j_pos = zeros[top_zeros[jj]]
                if i_pos == j_pos:
                    continue
                delta_est = -(g_ones[top_ones[ii]]) + g_zeros[top_zeros[jj]]
                if delta_est < best_delta:
                    best_delta = delta_est
                    best_pair = (i_pos, j_pos)

        if best_pair is None:
            i_pos = int(ones[rng.randint(len(ones))])
            j_pos = int(zeros[rng.randint(len(zeros))])
            if i_pos == j_pos:
                continue
            best_pair = (i_pos, j_pos)
            best_delta = 0.0

        i_pos, j_pos = best_pair
        h_bin[i_pos] = 0
        h_bin[j_pos] = 1

        corr_new = compute_corr(h_bin)
        new_max = float(np.max(corr_new))

        if new_max <= cur_max + 1e-14:
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            cur_max = new_max
            corr = corr_new
        else:
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-8)):
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
                cur_max = new_max
                corr = corr_new
            else:
                h_bin[i_pos] = 1
                h_bin[j_pos] = 0

        if it % 400 == 0 and it > 0:
            temp = max(temp * 0.97, temp_min)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def nesterov_gradient(h, n, top_k, max_iters, step_init):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - np.min(c_top) + 0.5)
        w = np.maximum(w, 0.01)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)
        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.12)
        else:
            v *= 0.5
            step *= 0.75

        if it % 2000 == 0 and it > 0:
            step *= 0.9
            momentum = min(momentum + 0.003, 0.93)

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def top_equalize(h, n, top_k, max_iters):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-6:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            break
        d /= dn

        improved = False
        for scale in [4, 2, 1, 0.5, 8, 0.25, 6, 0.1, 3, 0.15]:
            alpha = 0.015 * scale
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for scale in [2, 1, 4, 0.5]:
                alpha = 0.01 * scale
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def fourier_perturb(h, n, rng, num_modes=3, amp=0.1):
    h_pert = h.copy()
    x = np.arange(n)
    for _ in range(num_modes):
        freq = rng.choice([1, 2, 3, 4, 5, 6, 7, 8])
        phase = rng.uniform(0, 2 * np.pi)
        wave = amp * np.sin(2 * np.pi * freq * x / n + phase)
        h_pert = h_pert + wave
    return proj(h_pert, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: L-BFGS with increasing beta at parent resolution
    for beta in [200, 500, 1500, 4000, 10000, 25000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=10, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Nesterov gradient (top-8)
    if remaining() > 30:
        h_pg, c5_pg = nesterov_gradient(h, n, top_k=8, max_iters=6000, step_init=0.025)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 3: Top equalize
    if remaining() > 10:
        h_eq, c5_eq = top_equalize(h, n, top_k=6, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Binary SA search (extensive)
    if remaining() > 60:
        h_bin_init = (best_h > 0.5).astype(float)
        h_bl, c5_bl = binary_sa_search(h_bin_init, best_n, max_iters=150000,
                                        rng=rng, top_k=5, n_cand=30)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

        # Continuous refinement of binary result
        h_ref = h_bl.copy()
        for beta in [3000, 10000, 30000]:
            if remaining() < 5:
                break
            h_ref = lbfgs_pass(h_ref, best_n, beta, top_k=8, maxiter=350)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h = h_ref.copy()

    # Phase 5: Min-max binary search
    if remaining() > 60:
        h_bin2 = (best_h > 0.5).astype(float)
        h_bl2, c5_bl2 = binary_minmax_search(h_bin2, best_n, max_iters=100000,
                                              rng=rng, top_k=5, n_cand=35)
        if c5_bl2 < best_c5:
            best_c5 = c5_bl2
            best_h = h_bl2.copy()
        h_ref2 = h_bl2.copy()
        for beta in [5000, 15000, 40000]:
            if remaining() < 5:
                break
            h_ref2 = lbfgs_pass(h_ref2, best_n, beta, top_k=8, maxiter=300)
        c5_ref2 = compute_c5(h_ref2)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2.copy()

    # Phase 6: Another round of Nesterov + equalize on best
    h = best_h.copy()
    if remaining() > 20:
        h_pg2, c5_pg2 = nesterov_gradient(h, best_n, top_k=6, max_iters=4000, step_init=0.015)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()
            h = h_pg2.copy()

    if remaining() > 10:
        h_eq2, c5_eq2 = top_equalize(h, best_n, top_k=8, max_iters=250)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 7: Try different resolutions
    n_candidates = sorted(set([
        best_n, best_n - 4, best_n - 2, best_n + 2, best_n + 4,
        best_n + 8, best_n - 8,
        380, 382, 386, 388, 392,
    ]))
    n_candidates = [nn for nn in n_candidates if 256 <= nn <= 512 and nn % 2 == 0]
    n_candidates = [nn for nn in n_candidates if nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        for beta in [1000, 5000, 15000]:
            if remaining() < 10:
                break
            h_feas = lbfgs_pass(h_feas, n_t, beta, top_k=8, maxiter=300)

        h_feas, c5_feas = top_equalize(h_feas, n_t, top_k=5, max_iters=150)

        h_b2 = (h_feas > 0.5).astype(float)
        h_bl3, c5_bl3 = binary_sa_search(h_b2, n_t, max_iters=50000,
                                          rng=rng, top_k=4, n_cand=25)
        h_ref3 = h_bl3.copy()
        for beta in [5000, 20000]:
            if remaining() < 5:
                break
            h_ref3 = lbfgs_pass(h_ref3, n_t, beta, top_k=6, maxiter=250)
        c5_ref3 = compute_c5(h_ref3)

        c5_cand = min(c5_feas, c5_bl3, c5_ref3)
        if c5_ref3 <= c5_cand:
            h_cand = h_ref3
        elif c5_bl3 <= c5_cand:
            h_cand = h_bl3
        else:
            h_cand = h_feas

        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    if remaining() > 40:
        h_bin_f = (h > 0.5).astype(float)
        h_bf, c5_bf = binary_minmax_search(h_bin_f, n, max_iters=80000,
                                            rng=rng, top_k=6, n_cand=30)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
            h = h_bf.copy()

        h_rf = h.copy()
        for beta in [10000, 30000, 80000]:
            if remaining() < 5:
                break
            h_rf = lbfgs_pass(h_rf, n, beta, top_k=10, maxiter=300)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()

    # SLSQP final polish
    for beta in [30000, 80000]:
        if remaining() < 8:
            break
        h_new = slsqp_pass(best_h, best_n, beta, top_k=8, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final top equalize
    if remaining() > 8:
        h_eq3, c5_eq3 = top_equalize(best_h, best_n, top_k=10, max_iters=200)
        if c5_eq3 < best_c5:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()

    # Final binary micro-search
    if remaining() > 30:
        h_bin_final = (best_h > 0.5).astype(float)
        h_fin, c5_fin = binary_sa_search(h_bin_final, best_n, max_iters=50000,
                                          rng=rng, top_k=6, n_cand=20)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        h_ref_fin = h_fin.copy()
        for beta in [20000, 50000]:
            if remaining() < 5:
                break
            h_ref_fin = lbfgs_pass(h_ref_fin, best_n, beta, top_k=10, maxiter=200)
        c5_ref_fin = compute_c5(h_ref_fin)
        if c5_ref_fin < best_c5:
            best_c5 = c5_ref_fin
            best_h = h_ref_fin.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        # Try the slightly-adjusted version
        best_c5 = final_c5

    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, float(best_c5), best_n)
