import numpy as np

_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    diff = target - np.sum(h)
    if abs(diff) > 1e-10 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            for j in range(n):
                if 1e-14 < h[j] < 1.0 - 1e-14:
                    h[j] = np.clip(h[j] + diff2, 0.0, 1.0)
                    break
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    result = np.interp(x_new, xp, src_h)
    return project_feasible(result, dst_n)


def coord_descent(h0, n, max_sweeps=200, verbose=False):
    best = h0.copy().astype(float)
    corr = compute_corr(best).astype(float)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    mid = n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    best_c5 = np.max(corr) * dx

    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        improved = False

        for i in range(n):
            old = best[i]

            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[mid] = 0.0

            f_old_val = np.max(corr)

            best_t = old
            best_f = f_old_val

            # Check t=0
            f0_lin = np.max(corr - A * old)
            f0_quad = corr[mid] - old * (1.0 - old)
            f0 = max(f0_lin, f0_quad)
            if f0 < best_f - 1e-14:
                best_t, best_f = 0.0, f0

def active_set_step(h, n, n_active=30, step_size=0.01):
    """Project subgradient step focusing on top lags."""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    mid = n - 1
    sorted_idx = np.argsort(corr)[::-1]

    # Use lags within 30% of max
    threshold = max_corr - 0.30 * (max_corr - np.percentile(corr, 5))
    active = sorted_idx[corr[sorted_idx] >= threshold]
    if len(active) > n_active:
        active = sorted_idx[:n_active]
    if len(active) == 0:
        active = sorted_idx[:n_active]

    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - mid
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h.copy()
    grad /= gnorm
    h_new = h - step_size * grad
    return project_feasible(h_new, n)


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def _g_smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]


def _g_smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]


def _g_constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0


def _g_constraint_sum_jac(x):
    return np.ones(_N_WORK)


def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    try:
        res = minimize(_g_smooth_obj, h, jac=_g_smooth_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)
