def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def eval_smooth_max(h, beta):
    corr = compute_corr(h)
    m = corr.max()
    shifted = beta * (corr - m)
    # Clip to avoid overflow/underflow
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    sm = m + np.log(exps.sum()) / beta
    return sm


def eval_smooth_max_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = corr.max()
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    
    g = np.zeros(n)
    js = np.arange(n)
    
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-15:
            continue
        d = k - (n - 1)  # shift
        # g[j] += wk * [1(j-d valid) * (1-h[j-d]) - 1(j+d valid) * h[j+d]]

        g[mask1] += wk * (1.0 - h[js[mask1] - d])
        g[mask2] -= wk * h[js[mask2] + d]
    
    return g, m

    """SLSQP optimization of smooth max"""
    def obj(x):
        return eval_smooth_max_scaled(x, beta)
    
    def obj_grad(x):
        return eval_smooth_max_grad_scaled(x, beta)
    
    def con_eq(x):
        return np.array([np.sum(x) - n / 2.0])
    
    def con_eq_jac(x):
        return np.array([[1.0] * n])
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,

        if res.x is not None:

def top_k_weighted_grad(h, n, K=20, beta=50.0):

    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]

    weights = np.exp(beta * (c_top - c_top.max()))
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= wsum

        g[mask2] -= weights[i] * h[js[mask2] + d]
