    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = len(corr)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def _g_opt_obj(x):
    return smooth_obj_grad(x, _OPT_N, _OPT_BETA)[0]


def _g_opt_grad(x):
    return smooth_obj_grad(x, _OPT_N, _OPT_BETA)[1]


def _g_constraint_sum(x):
    return np.sum(x) - _OPT_N / 2.0


def _g_constraint_sum_jac(x):
    return np.ones(_OPT_N)


def lbfgsb_optimize(h, n, beta, maxiter=2000):
    global _OPT_N, _OPT_BETA
    _OPT_N = n
    _OPT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_g_opt_obj, h, jac=_g_opt_grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-30, 'gtol': 1e-18,
                               'maxls': 50})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def slsqp_refine(h, n, beta, maxiter=500):
    global _OPT_N, _OPT_BETA
    _OPT_N = n
    _OPT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    try:
        res = minimize(_g_opt_obj, h, jac=_g_opt_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-25, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def coord_obj(t, base, A, n):
    vals = base + A * t
    vals[n - 1] = base[n - 1] + t - t * t
    return np.max(vals)


def fast_coord_descent(h, n, max_sweeps=200):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = np.max(best_corr)

            t_best = old
            f_best = f_old
            for tc in [0.0, 0.5, 1.0]:
                fc = coord_obj(tc, base, A, n)
                if fc < f_best - 1e-14:
                    f_best = fc
                    t_best = tc

            if t_best != old:
                if t_best > old:
                    lo = old
                    hi = min(1.0, t_best + 0.3)
                else:
                    lo = max(0.0, t_best - 0.3)
                    hi = old
                if hi - lo < 1e-12:
                    lo = max(0.0, min(1.0, (lo + hi) / 2.0) - 1e-6)
                    hi = lo + 2e-6

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = coord_obj(c1, base, A, n)
                f2 = coord_obj(c2, base, A, n)

                for _ in range(60):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = coord_obj(c1, base, A, n)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = coord_obj(c2, base, A, n)

                t_best = (lo + hi) / 2.0
                f_best = coord_obj(t_best, base, A, n)

            if f_best < f_old - 1e-13:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break

    return best, best_c5


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def perturb_and_optimize(h_start, n, rng, max_sweeps_cd=60, n_perturbations=5):
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)

    for p in range(n_perturbations):
        if remaining_time() < 15:
            break
        h_pert = h_start.copy()
        scale = 0.2 / (1.0 + p * 0.5)
        mask = rng.random(n) < 0.25
        h_pert[mask] += rng.randn(int(np.sum(mask))) * scale
        h_pert = project_feasible(h_pert, n)

        h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=max_sweeps_cd)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5


def structured_perturbation(h, n, rng):
    """Try structured moves: block flips, shifts."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    trials = 200
    for _ in range(trials):
        if remaining_time() < 5:
            break
        h_trial = best_h.copy()
        move = rng.randint(3)
        if move == 0:
            # Swap two positions
            i, j = rng.choice(n, 2, replace=False)
            h_trial[i], h_trial[j] = h_trial[j], h_trial[i]
            h_trial = project_feasible(h_trial, n)
        elif move == 1:
            # Flip a small block
            bs = rng.randint(1, 6)
            start = rng.randint(0, max(1, n - bs))
            h_trial[start:start + bs] = 1.0 - h_trial[start:start + bs]
            h_trial = project_feasible(h_trial, n)
        else:
            # Gentle random noise on subset
            idx = rng.choice(n, n // 8, replace=False)
            h_trial[idx] += rng.randn(len(idx)) * 0.1
            h_trial = project_feasible(h_trial, n)

        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-14:
            best_c5 = c5_trial
            best_h = h_trial.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _RNG
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    _RNG = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Coordinate descent from parent (multiple sweeps)
    h = best_h.copy()
    for _ in range(3):
        if remaining_time() < 30:
            break
        h, c5 = fast_coord_descent(h, n, max_sweeps=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: L-BFGS-B with smoothed objective at various betas
    h = best_h.copy()
    for beta in [300, 1000, 3000, 10000]:
        if remaining_time() < 25:
            break
        h_new = lbfgsb_optimize(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        elif c5_new < compute_c5(h) + 1e-15:
            h = h_new

    # Phase 3: More coordinate descent after gradient optimization
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: SLSQP refinement
    if remaining_time() > 30:
        h = best_h.copy()
        for beta in [5000, 20000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_refine(h, n, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new

    # Phase 5: Structured perturbations + coordinate descent
    if remaining_time() > 40:
        h_sp = best_h.copy()
        h_sp, c5_sp = structured_perturbation(h_sp, n, _RNG)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()
            h_sp, c5_sp = fast_coord_descent(h_sp, n, max_sweeps=50)
            if c5_sp < best_c5:
                best_c5 = c5_sp
                best_h = h_sp.copy()

    # Phase 6: Random perturbations
    if remaining_time() > 50:
        h_pt = best_h.copy()
        h_pt, c5_pt = perturb_and_optimize(h_pt, n, _RNG, max_sweeps_cd=50, n_perturbations=6)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt.copy()

    # Phase 7: Try different resolutions
    resolutions_to_try = []
    if n == 640:
        resolutions_to_try = [768, 800, 896, 1024, 1280]
    elif n <= 800:
        resolutions_to_try = [n + 128, n + 256, 1280]
    else:
        resolutions_to_try = [n // 2 if n > 800 else 640]

    for res_n in resolutions_to_try:
        if remaining_time() < 60:
            break
        if res_n < 200 or res_n > 2000:
            continue
        h_high = interp_to(best_h, best_n, res_n)
        h_high = project_feasible(h_high, res_n)
        c5_high = compute_c5(h_high)

        h_new, c5_new = fast_coord_descent(h_high, res_n, max_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = res_n

        if remaining_time() > 40:
            h_lb = lbfgsb_optimize(best_h, best_n, 3000, maxiter=400)
            c5_lb = compute_c5(h_lb)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb.copy()
            h_new2, c5_new2 = fast_coord_descent(best_h, best_n, max_sweeps=40)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()

    # Phase 8: Thue-Morse and other structured inits at best resolution
    if remaining_time() > 60:
        h_tm = thue_morse_init(best_n)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 + 0.08:
            h_new, c5_new = fast_coord_descent(h_tm, best_n, max_sweeps=80)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            if remaining_time() > 30:
                h_lb = lbfgsb_optimize(best_h, best_n, 5000, maxiter=500)
                c5_lb = compute_c5(h_lb)
                if c5_lb < best_c5:
                    best_c5 = c5_lb
                    best_h = h_lb.copy()

    # Phase 9: Binary init exploration
    if remaining_time() > 40:
        bin_inits = 5
        for bi in range(bin_inits):
            if remaining_time() < 20:
                break
            h_bin = binary_init(best_n, _RNG)
            h_bin_proj = project_feasible(h_bin, best_n)
            h_new, c5_new = fast_coord_descent(h_bin_proj, best_n, max_sweeps=40)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 10: Final intensive coordinate descent
    if remaining_time() > 15:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, best_n, max_sweeps=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 11: Final L-BFGS-B polish with high beta
    if remaining_time() > 12:
        h = best_h.copy()
        for beta in [20000, 50000]:
            if remaining_time() < 8:
                break
            h_new = lbfgsb_optimize(h, best_n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new

    # Phase 12: Very final coordinate descent
    if remaining_time() > 8:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, best_n, max_sweeps=60)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 13: One more SLSQP
    if remaining_time() > 10:
        h_new = slsqp_refine(best_h, best_n, 30000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
