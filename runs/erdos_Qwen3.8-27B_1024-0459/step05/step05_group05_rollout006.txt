            new_maxes = np.max(cand, axis=1)
            deltas = new_maxes - cur_max
            jmin = int(np.argmin(deltas))
            if deltas[jmin] < best_delta - 1e-14:
                best_delta = deltas[jmin]
                best_ii = ii
                best_jj = jmin
        
        if best_ii >= 0:
            i = ones[best_ii]
            j = zeros[best_jj]
            h[i] = 0.0
            h[j] = 1.0
            stall = 0
        else:
            stall += 1
            if stall > 300:
                # Perturbation
                n_pert = rng.randint(1, 4)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_pert):
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0.0
                    h[j] = 1.0
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                stall = 0
    
    corr_final = compute_corr(best_h)
    return best_h, float(np.max(corr_final) * dx)
