def sa_binary_search(h_binary, n, n_iter, t_start, t_end, rng):
    """Simulated annealing on binary sequences with pair swaps."""
    h = h_binary.copy()
    corr = np.correlate(h, 1-h, mode='full').astype(float)
    current_energy = np.max(corr)
    best_energy = current_energy
    best_h = h.copy()
    best_corr = corr.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        # Temperature
        alpha = it / n_iter
        temp = t_start * (t_end / t_start) ** alpha
        
        # Propose swap
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        # Compute change in corr
        delta = np.zeros(2*n-1)
        # Contribution from a (1->0)
        for d in range(max(-a, -(n-1)), min(n-a, n)):
            # offset d, position a: was 1*(1-h[a+d]), now 0
            if 0 <= a+d < n:
                delta[n-1+d] -= (1-h[a+d])
        for d in range(max(a-n+1, -(n-1)), min(n-a, n)):

        new_energy = np.max(new_corr)
        
        # Accept/reject

```python
def sa_binary(h, N, n_iter, rng, t_start=5.0, t_end=0.1):
    """Simulated annealing on binary sequences."""
    h = h.copy()
    corr = np.correlate(h.astype(float), 1-h.astype(float), mode='full').astype(float)
    current_e = np.max(corr)
    best_e = current_e
    best_h = h.copy()
    
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    # Precompute
    h_rev = h[::-1].copy()  # h[0:N][::-1]
    
    for it in range(n_iter):
        if remaining_time() < 5:
            break
        alpha = (it + 1) / n_iter
        temp = t_start * (t_end / t_start) ** alpha
        
        # Propose swap
        a = ones_pos[rng.randint(len(ones_pos))]
        b = zeros_pos[rng.randint(len(zeros_pos))]
        if a == b:
            continue
        
        # Compute delta
        delta = np.zeros(2*N-1)
        delta[N-1-a : 2*N-1-a] += h
        delta[N-1-b : 2*N-1-b] -= h
        delta[a : a+N] += h_rev - 1
        delta[b : b+N] += 1 - h_rev
        delta[N-1] -= h[a]
        
        new_corr = corr + delta
        new_e = np.max(new_corr)
        
        if new_e <= current_e or rng.random() < np.exp((current_e - new_e) / max(temp, 1e-10)):
            # Accept
            h[a] = 0
            h[b] = 1
            corr = new_corr
            # Update h_rev
            h_rev = h[::-1].copy()
            # Update positions
            ones_pos = np.delete(ones_pos, np.where(ones_pos == a)[0][0])
            zeros_pos = np.delete(zeros_pos, np.where(zeros_pos == b)[0][0])
            ones_pos = np.append(ones_pos, b)
            zeros_pos = np.append(zeros_pos, a)
            current_e = new_e
            
            if new_e < best_e:
                best_e = new_e
                best_h = h.copy()
    
    return best_h, best_e
```
