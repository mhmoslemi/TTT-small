from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_GN = 0
_GBeta = 1.0
_GTGT = 0.0
_GMU = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def make_swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] -= (1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)
    hi3 = min(j + n - 1, r - 1)
    if lo3 <= hi3:
        ks = np.arange(lo3, hi3 + 1)
        dr[lo3:hi3+1] += (1.0 - h[j + n - 1 - ks])
    lo4 = max(n - 1 - j, 0)

        m = n - ao
        g[:m] += (1.0 - h[ao:ao+m])
        g[ao:ao+m] -= h[:m]

        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0

    xp = (np.arange(sn) + 0.5) * 2.0 / sn
