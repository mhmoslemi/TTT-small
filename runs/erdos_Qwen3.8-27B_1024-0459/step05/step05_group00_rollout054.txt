def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

        if remaining() > 10:
            h2, c5_2 = equalize_top(h2, n2, K=6, max_iters=80)

        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2
            h = h2.copy()
            n = n2

    # Phase 8: Final polish at best resolution
    h = best_h.copy()
    n = best_n

    for cycle in range(3):
        if remaining() < 15:
            break

        for beta in [80000, 200000, 400000]:
            if remaining() < 8:
                break
            h = slsqp_smooth_max(h, n, beta, maxiter=400)

        if remaining() > 10:
            h, c5_tmp = equalize_top(h, n, K=6, max_iters=80)

        if remaining() > 8:
            cur = compute_c5(h)
            tgt = cur - 0.0001
            h = target_descent(h, n, tgt, maxiter=150)

        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 9: Tiny-step fine tuning
    h = best_h.copy()
    n = best_n
    for it in range(80):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:5]
        g = np.zeros(n)
        for k in idx:
            g += grad_corr_k(h, k, n)
        g /= 5.0
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        improved = False
        for step in [0.003, 0.001, 0.005, 0.002, 0.0005, 0.004]:
            ht = project_feasible(h - (step * n / gn) * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                best_h = ht.copy()
                improved = True
                break
        if not improved and it > 40:
            break

    # Phase 10: One more SLSQP polish at very high beta
    if remaining() > 5:
        h_fin = slsqp_smooth_max(best_h, best_n, 500000, maxiter=300)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
