            #           = -(t-old) * h[i-d]  (if only i-d valid)
            #           = 0  (if neither valid)

                val = 0.0
                if 0 <= i+d < n:

                if 0 <= i-d < n:
                    val += best[i-d]
                if d == 0:
                    val = 0.0  # Handle separately (quadratic)
                coef[k] = val

            #        corr_new[n-1] = corr[n-1] + t*(1-t) - old*(1-old)
            #        = corr[n-1] + t - t² - old + old²

            # This is min_t max of (2n-2) linear functions + 1 quadratic

            gr = (np.sqrt(5) - 1) / 2
            lo = 0.0
            hi = 1.0

                base = corr + (t - old) * coef
                base[n-1] = corr[n-1] + t - t*t - old + old*old
                return np.max(base)
            
            c1 = hi - gr * (hi - lo)

            f2 = eval_t(c2)

                    c1 = hi - gr * (hi - lo)
                    f1 = eval_t(c1)
                else:
                    lo = c1; c1 = c2; f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_t(c2)
            
            t_best = (lo + hi) / 2
            f_best = eval_t(t_best)
            f_old = best_max
            
            if f_best < f_old - tol:

                corr += (t_best - old) * coef

                corr[n-1] += t_best - t_best*t_best - old + old*old
                best[i] = t_best
                best_max = np.max(corr)
                improved = True
        
        if not improved:
            break

def project_feasible(h, n):

def smooth_max_obj_grad(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * (2.0 / n)
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:

        offset = k - (n - 1)
        if offset >= 0:

            if npairs > 0:

                grad[:npairs] -= w * x[offset:offset + npairs]

                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= (2.0 / n)
    return val, grad
