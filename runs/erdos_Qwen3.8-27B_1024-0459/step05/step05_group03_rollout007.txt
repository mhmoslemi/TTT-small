def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def slsqp_opt(h, n, beta, maxiter=400, top_k=0):

    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

        corr = compute_corr(x)
        if top_k > 0 and top_k < len(corr):
            idx = np.argsort(corr)[-top_k:]
            top = corr[idx]
            m = top.max()
            sh = np.clip(beta * (top - m), -40.0, 40.0)
            ex = np.exp(sh)
            return (m + np.log(ex.sum()) / beta) * dx
        else:
            m = corr.max()
            sh = np.clip(beta * (corr - m), -40.0, 40.0)
            ex = np.exp(sh)
            return (m + np.log(ex.sum()) / beta) * dx

    def _grad(x):
        x = np.clip(x, 0, 1)
        corr = compute_corr(x)
        if top_k > 0 and top_k < len(corr):
            idx = np.argsort(corr)[-top_k:]

        for beta in [2000, 15000, 80000]:
            if remaining() < 20:
                break
            h_in = slsqp_opt(h_in, n_new, beta, maxiter=300)

        for tk in [3, 5, 8]:
            if remaining() < 15:
                break
            h_in = slsqp_opt(h_in, n_new, 30000, maxiter=250, top_k=tk)

        if remaining() > 25:
            h_in = lbfgsb_opt(h_in, n_new, 50000, maxiter=2000)

        c5_in = compute_c5(h_in)
        if c5_in < best_c5 - 1e-14:
            best_c5 = c5_in
            best_h = h_in.copy()
            best_n = n_new

    # Phase 5: Subgradient top-k equalization
    if remaining() > 30:
        for tk in [3, 5, 8, 12]:
            if remaining() < 15:
                break
            h_sg, c5_sg = subgrad_topk(best_h, best_n, top_k=tk, max_iters=400)
            if c5_sg < best_c5 - 1e-14:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Phase 6: Binary optimization
    if remaining() > 40:
        h_bin = round_to_binary(best_h, best_n)
        h_sa, c5_sa = sa_binary(h_bin, best_n, max_iters=4000, T0=0.002, T1=0.0002)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

        if remaining() > 30:
            h_gb, c5_gb = greedy_binary(best_h, best_n, max_rounds=15, sample_size=35)
            if c5_gb < best_c5:
                best_c5 = c5_gb
                best_h = h_gb.copy()

        # SLSQP from binary
        if remaining() > 20:
            h_from_bin = slsqp_opt(best_h if best_c5 < c5_sa else h_sa, best_n, 50000, maxiter=400)
            c5_fb = compute_c5(h_from_bin)
            if c5_fb < best_c5:
                best_c5 = c5_fb
                best_h = h_from_bin.copy()

    # Phase 7: Stochastic restarts at best resolution
    rng = np.random.RandomState(seed + 100)
    if remaining() > 60:
        for attempt in range(4):
            if remaining() < 30:
                break
            noise = rng.randn(best_n) * 0.05
            h_pert = project_feasible(best_h + noise, best_n)
            c5_pert = compute_c5(h_pert)
            if c5_pert > best_c5 + 0.002:
                continue
            h_res = slsqp_opt(h_pert, best_n, 20000, maxiter=400)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
            if remaining() > 15:
                h_res2 = slsqp_opt(h_res, best_n, 100000, maxiter=300, top_k=3)
                c5_res2 = compute_c5(h_res2)
                if c5_res2 < best_c5:
                    best_c5 = c5_res2
                    best_h = h_res2.copy()

    # Phase 8: Final intensive polish
    if remaining() > 20:
        for beta in [50000, 200000, 500000]:
            if remaining() < 12:
                break
            h_fin = slsqp_opt(best_h, best_n, beta, maxiter=500)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Phase 9: Ultra-fine top-2 equalization
    if remaining() > 15:
        for beta in [100000, 300000]:
            if remaining() < 10:
                break
            h_u2 = slsqp_opt(best_h, best_n, beta, maxiter=400, top_k=2)
            c5_u2 = compute_c5(h_u2)
            if c5_u2 < best_c5:
                best_c5 = c5_u2
                best_h = h_u2.copy()

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
