```python
    h_cur = h_bin.copy()
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    zeros = np.where(h_cur == 0)[0]

            break
        T = T_init * (ratio ** it)

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

                best_h = h_cur.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 200.0
_TOPK = 10
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    # L2-minimal projection via shift
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h + lam, 0.0, 1.0)
    # Fine correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idxs = np.argsort(np.abs(h - 0.5))
        for i in idxs[:30]:
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        if np_ > 0:
            g[offset:offset + np_] += (1.0 - h[:np_])
            g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        if np_ > 0:
            g[:np_] += (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 100.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j in range(len(idx)):
        g += w[j] * corr_grad_single(x, idx[j], n)
    g *= dx
    g += 200.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=10, maxiter=600):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 15})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def nesterov_opt(h, n, top_k=10, max_iters=10000, step_init=0.04):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        c_top = corr[idx]
        m = c_top[0]
        sh = 50.0 * (c_top - m)
        sh = np.clip(sh, -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)

        g = np.zeros(n)
        for j in range(len(idx)):
            g += w[j] * corr_grad_single(h, idx[j], n)
        g -= np.mean(g)
        g[(h >= 1.0 - 1e-10)] = 0.0
        g[(h <= 1e-10)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.005:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.015, 0.12)
                momentum = min(momentum + 0.001, 0.95)
        else:
            h = best_h.copy()
            v *= 0.3
            step *= 0.65

        if it % 500 == 0 and it > 0:
            step *= 0.88
            momentum = min(momentum + 0.003, 0.95)

    return best_h, best_c5


def top_equalize(h, n, top_k=8, max_iters=500):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 4:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-7:
            # Try to reduce the max directly
            g_max = corr_grad_single(h, idx[0], n)
            g_max -= np.mean(g_max)
            gn = np.linalg.norm(g_max)
            if gn < 1e-14:
                break
            direction = -g_max / gn
            improved = False
            for alpha in [0.01, 0.005, 0.02, 0.002, 0.008, 0.015, 0.003]:
                h_try = proj(h + alpha * direction, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if not improved:
                break
            continue

        # Equalize: move in direction that reduces c_max and increases c_min
        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_max
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn

        improved = False
        for scale in [1.0, 0.5, 2.0, 0.3, 3.0, 0.7, 0.1, 4.0, 0.4, 5.0]:
            alpha = 0.015 * scale
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for scale in [0.5, 1.0, 2.0, 0.3]:
                alpha = 0.01 * scale
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def binary_local_search(h, n, max_iters=3000, top_k=5):
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5
    rng = np.random.RandomState(123)

    ones_list = list(np.where(h_bin == 1)[0])
    zeros_list = list(np.where(h_bin == 0)[0])

    for it in range(max_iters):
        if remaining() < 4:
            break
        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]

        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        n_ones = len(ones_list)
        n_zeros = len(zeros_list)
        if n_ones == 0 or n_zeros == 0:
            break

        best_delta = 0
        best_swap = None

        # Scored candidates
        score_one = -g[ones_list]
        score_zero = g[zeros_list]
        n_try = min(15, n_ones)
        n_try_z = min(15, n_zeros)
        top_ones_idx = np.argsort(-score_one)[:n_try]
        top_zeros_idx = np.argsort(-score_zero)[:n_try_z]

        for ti in top_ones_idx:
            i = ones_list[ti]
            for tj in top_zeros_idx:
                j = zeros_list[tj]
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j, ti, tj)

        # Random candidates
        for _ in range(8):
            ti = rng.randint(n_ones)
            tj = rng.randint(n_zeros)
            i = ones_list[ti]
            j = zeros_list[tj]
            h_bin[i] = 0
            h_bin[j] = 1
            nc = compute_c5(h_bin)
            h_bin[i] = 1
            h_bin[j] = 0
            delta = nc - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j, ti, tj)

        if best_swap is None or best_delta >= 0:
            # 2-opt: try swapping two pairs
            found = False
            for _ in range(4):
                ti1 = rng.randint(n_ones)
                ti2 = rng.randint(n_ones)
                if ti1 == ti2:
                    continue
                tj1 = rng.randint(n_zeros)
                tj2 = rng.randint(n_zeros)
                if tj1 == tj2:
                    continue
                i1, i2 = ones_list[ti1], ones_list[ti2]
                j1, j2 = zeros_list[tj1], zeros_list[tj2]
                h_bin[i1] = 0
                h_bin[i2] = 0
                h_bin[j1] = 1
                h_bin[j2] = 1
                nc = compute_c5(h_bin)
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    h_bin[i1] = 1
                    h_bin[i2] = 1
                    h_bin[j1] = 0
                    h_bin[j2] = 0
                    best_swap = (i1, j1, ti1, tj1)
                    found = True
                    # Accept and update
                    h_bin[i1] = 0
                    h_bin[j1] = 1
                    h_bin[i2] = 0
                    h_bin[j2] = 1
                    ones_list[ti1] = j1
                    ones_list[ti2] = j2
                    zeros_list[tj1] = i1
                    zeros_list[tj2] = i2
                    cur_c5 = nc
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h_bin.copy()
                else:
                    h_bin[i1] = 1
                    h_bin[i2] = 1
                    h_bin[j1] = 0
                    h_bin[j2] = 0
            if not found:
                break
            continue

        i, j, ti, tj = best_swap
        h_bin[i] = 0
        h_bin[j] = 1
        ones_list[ti] = j
        zeros_list[tj] = i
        cur_c5 = compute_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def binary_sa(h_bin, n, max_iters=200000, T_init=0.5, T_min=0.003, seed=42):
    h_cur = h_bin.copy()
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5
    rng = np.random.RandomState(seed)

    ones_list = list(np.where(h_cur == 1)[0])
    zeros_list = list(np.where(h_cur == 0)[0])
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)

    ratio = (T_min / T_init) ** (1.0 / max(1, max_iters))
    T = T_init

    for it in range(max_iters):
        if remaining() < 5:
            break
        T *= ratio
        ri = rng.randint(n_ones)
        rj = rng.randint(n_zeros)
        i = ones_list[ri]
        j = zeros_list[rj]

        h_cur[i] = 0
        h_cur[j] = 1
        new_c5 = compute_c5(h_cur)
        delta = new_c5 - cur_c5

        if delta <= 0:
            cur_c5 = new_c5
            ones_list[ri] = j
            zeros_list[rj] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_cur.copy()
        else:
            if T > 1e-12:
                log_p = -delta / T
                if log_p > -40 and rng.random() < np.exp(log_p):
                    cur_c5 = new_c5
                    ones_list[ri] = j
                    zeros_list[rj] = i
                else:
                    h_cur[i] = 1
                    h_cur[j] = 0

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Quick continuous optimization at parent resolution (~80s)
    n1 = best_n
    h1 = best_h.copy()
    for beta in [200, 800, 3000]:
        if remaining() < 10:
            break
        h1 = lbfgs_pass(h1, n1, beta, top_k=10, maxiter=400)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n1

    h1, c5_1 = nesterov_opt(h1, n1, top_k=10, max_iters=4000, step_init=0.03)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Interpolate to n=768 and optimize (~500s)
    if remaining() > 100:
        n2 = 768
        h2 = interp_to(best_h, best_n, n2)
        h2 = proj(h2, n2)
        c5_2 = compute_c5(h2)

        # L-BFGS at higher resolution
        for beta in [300, 1000, 4000, 15000, 40000]:
            if remaining() < 15:
                break
            h2 = lbfgs_pass(h2, n2, beta, top_k=12, maxiter=500)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

        # Nesterov at n2
        if remaining() > 30:
            h2_nc, c5_2nc = nesterov_opt(h2, n2, top_k=12, max_iters=8000, step_init=0.03)
            if c5_2nc < best_c5:
                best_c5 = c5_2nc
                best_h = h2.copy() if False else h2_nc.copy()
                best_n = n2
                h2 = h2_nc.copy()

        # Top equalize at n2
        if remaining() > 15:
            h2_eq, c5_2eq = top_equalize(h2, n2, top_k=10, max_iters=300)
            if c5_2eq < best_c5:
                best_c5 = c5_2eq
                best_h = h2_eq.copy()
                best_n = n2
                h2 = h2_eq.copy()

        # More L-BFGS with very high beta
        for beta in [80000, 200000]:
            if remaining() < 10:
                break
            h2 = lbfgs_pass(h2, n2, beta, top_k=15, maxiter=400)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

    # Phase 3: Binary optimization at best_n (~200s)
    if remaining() > 60:
        n3 = best_n
        h3_bin = (best_h > 0.5).astype(float)
        c5_bin = compute_c5(h3_bin)

        # Binary local search
        h3_bl, c5_3bl = binary_local_search(h3_bin, n3, max_iters=2000, top_k=5)
        if c5_3bl < c5_bin:
            h3_bin = h3_bl
            c5_bin = c5_3bl

        # Binary SA
        h3_sa, c5_3sa = binary_sa(h3_bin, n3, max_iters=100000, T_init=0.3, T_min=0.002, seed=seed)
        if c5_3sa < c5_bin:
            h3_bin = h3_sa
            c5_bin = c5_3sa

        # Refine binary solution continuously
        h3_ref = h3_bin.copy()
        for beta in [2000, 10000, 50000]:
            if remaining() < 8:
                break
            h3_ref = lbfgs_pass(h3_ref, n3, beta, top_k=12, maxiter=400)
        c5_3ref = compute_c5(h3_ref)

        if c5_3ref < best_c5:
            best_c5 = c5_3ref
            best_h = h3_ref.copy()
            best_n = n3
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h3_bin.copy()
            best_n = n3

    # Phase 4: Try n=512 (~150s)
    if remaining() > 80:
        n4 = 512
        h4 = interp_to(best_h, best_n, n4)
        h4 = proj(h4, n4)
        for beta in [500, 2000, 8000, 30000]:
            if remaining() < 10:
                break
            h4 = lbfgs_pass(h4, n4, beta, top_k=10, maxiter=400)
        h4_nc, c5_4nc = nesterov_opt(h4, n4, top_k=10, max_iters=4000, step_init=0.03)
        h4_eq, c5_4eq = top_equalize(h4_nc if c5_4nc < compute_c5(h4) else h4, n4, top_k=8, max_iters=200)
        h4_bin = (c5_4eq < c5_4nc) and h4_eq or (h4_nc if c5_4nc < compute_c5(h4) else h4)
        h4_b = (h4_bin > 0.5).astype(float)
        h4_bl, c5_4bl = binary_local_search(h4_b, n4, max_iters=1000, top_k=4)
        h4_ref = h4_bl.copy()
        for beta in [5000, 20000]:
            if remaining() < 5:
                break
            h4_ref = lbfgs_pass(h4_ref, n4, beta, top_k=10, maxiter=300)
        c5_4 = min(compute_c5(h4), c5_4nc, c5_4eq, c5_4bl, compute_c5(h4_ref))
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_n = n4
            if compute_c5(h4_ref) == c5_4:
                best_h = h4_ref.copy()
            elif c5_4bl == c5_4:
                best_h = h4_bl.copy()
            elif c5_4eq == c5_4:
                best_h = h4_eq.copy()
            elif c5_4nc == c5_4:
                best_h = h4_nc.copy()
            else:
                best_h = h4.copy()

    # Phase 5: Try n=1024 if time allows (~150s)
    if remaining() > 100:
        n5 = 1024
        h5 = interp_to(best_h, best_n, n5)
        h5 = proj(h5, n5)
        for beta in [500, 2000, 8000]:
            if remaining() < 12:
                break
            h5 = lbfgs_pass(h5, n5, beta, top_k=12, maxiter=350)
        h5_nc, c5_5nc = nesterov_opt(h5, n5, top_k=12, max_iters=5000, step_init=0.025)
        c5_5 = min(compute_c5(h5), c5_5nc)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5_nc.copy() if c5_5 == c5_5nc else h5.copy()
            best_n = n5
        # Quick top equalize
        if remaining() > 15:
            h_best_5 = h5_nc if c5_5 == c5_5nc else h5
            h5_eq, c5_5eq = top_equalize(h_best_5, n5, top_k=10, max_iters=200)
            if c5_5eq < best_c5:
                best_c5 = c5_5eq
                best_h = h5_eq.copy()
                best_n = n5

    # Phase 6: Final intensive refinement at best resolution (~200s)
    n_f = best_n
    h_f = best_h.copy()

    # Multi-round L-BFGS with increasing beta
    for beta in [5000, 20000, 100000, 500000]:
        if remaining() < 8:
            break
        h_f = lbfgs_pass(h_f, n_f, beta, top_k=15, maxiter=400)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Nesterov final
    if remaining() > 20:
        h_fn, c5_fn = nesterov_opt(h_f, n_f, top_k=12, max_iters=6000, step_init=0.02)
        if c5_fn < best_c5:
            best_c5 = c5_fn
            best_h = h_fn.copy()
            h_f = h_fn.copy()

    # Top equalize final
    if remaining() > 10:
        h_feq, c5_feq = top_equalize(h_f, n_f, top_k=12, max_iters=300)
        if c5_feq < best_c5:
            best_c5 = c5_feq
            best_h = h_feq.copy()
            h_f = h_feq.copy()

    # Binary polish
    if remaining() > 30:
        h_fb = (best_h > 0.5).astype(float)
        h_fb_ls, c5_fb = binary_local_search(h_fb, n_f, max_iters=1500, top_k=5)
        h_fb_ref = h_fb_ls.copy()
        for beta in [10000, 50000]:
            if remaining() < 5:
                break
            h_fb_ref = lbfgs_pass(h_fb_ref, n_f, beta, top_k=12, maxiter=300)
        c5_fb_ref = compute_c5(h_fb_ref)
        if c5_fb_ref < best_c5:
            best_c5 = c5_fb_ref
            best_h = h_fb_ref.copy()
        elif c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb_ls.copy()

    # Final SLSQP-style refinement
    if remaining() > 10:
        h_f2 = best_h.copy()
        for beta in [200000, 1000000]:
            if remaining() < 5:
                break
            h_f2 = lbfgs_pass(h_f2, n_f, beta, top_k=20, maxiter=200)
        c5_f2 = compute_c5(h_f2)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
