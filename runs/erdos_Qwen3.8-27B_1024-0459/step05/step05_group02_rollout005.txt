def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_loop(h, n, top_k, max_iters, step_init, momentum):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    step = step_init
    stale = 0

    for it in range(max_iters):

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5 - 1e-14:

            best_h = h.copy()
            stale = 0
        else:
            stale += 1

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - np.mean(c_top)) + 0.05
        w = np.maximum(w, 0.01)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.001:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.1)

                step *= 0.95

            v *= 0.3
            step *= 0.7

            best_c5 = compute_c5(best_h)
            if stale > 1500:
                break

            step *= 0.88

    return best_h, best_c5
