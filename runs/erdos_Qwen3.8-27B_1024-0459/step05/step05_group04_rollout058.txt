    r = 2 * n - 1
    dr = np.zeros(r)

    # As h[m] factor: -(1-h[i+k-(n-1)]) for k where 0<=i+k-(n-1)<=n-1
    #   k in [n-1-i, 2(n-1)-i]

    if lo <= hi:

        dr[lo2:hi2+1] += h[m_vals]
    
    # Adding h[j]=1 (was 0):

    hi3 = min(r - 1, 2 * (n - 1) - j)

        m_vals4 = n - 1 + j - ks4
        dr[lo4:hi4+1] -= h[m_vals4]

from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def precompute_deltas(h, n):
    """Precompute delta vectors for removing each 1 and adding each 0."""
    r = 2 * n - 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    D_ones = np.zeros((len(ones), r))
    for idx, i in enumerate(ones):
        # Remove h[i] (1->0)
        dr = np.zeros(r)
        lo = max(0, n - 1 - i)
        hi = min(r - 1, 2 * (n - 1) - i)
        if lo <= hi:
            ks = np.arange(lo, hi + 1)
            h_idx = i + ks - (n - 1)
            dr[lo:hi+1] -= (1.0 - h[h_idx])
        lo2 = max(0, i)
        hi2 = min(r - 1, i + n - 1)
        if lo2 <= hi2:
            ks2 = np.arange(lo2, hi2 + 1)
            m_vals = n - 1 + i - ks2
            dr[lo2:hi2+1] += h[m_vals]
        D_ones[idx] = dr
    
    D_zeros = np.zeros((len(zeros), r))
    for idx, j in enumerate(zeros):
        # Add h[j] (0->1)
        dr = np.zeros(r)
        lo3 = max(0, n - 1 - j)
        hi3 = min(r - 1, 2 * (n - 1) - j)
        if lo3 <= hi3:
            ks3 = np.arange(lo3, hi3 + 1)
            h_idx3 = j + ks3 - (n - 1)
            dr[lo3:hi3+1] += (1.0 - h[h_idx3])
        lo4 = max(0, j)
        hi4 = min(r - 1, j + n - 1)
        if lo4 <= hi4:
            ks4 = np.arange(lo4, hi4 + 1)
            m_vals4 = n - 1 + j - ks4
            dr[lo4:hi4+1] -= h[m_vals4]
        D_zeros[idx] = dr
    
    return D_ones, D_zeros, ones, zeros

def best_swap_update(h, corr, D_ones, D_zeros, ones, zeros):
    """Find the best swap (one->zero, zero->one) that minimizes max correlation."""
    n = len(h)
    current_max = float(np.max(corr))
    
    best_val = current_max
    best_i = -1
    best_j = -1
    
    for ia in range(len(ones)):
        v = corr + D_ones[ia]  # (767,)
        # For each zero position: max(v + D_zeros[j])
        if len(zeros) > 0:
            maxes = np.max(v[None, :] + D_zeros, axis=1)  # (n_zeros,)
            j_best = int(np.argmin(maxes))
            val_best = maxes[j_best]
            if val_best < best_val:
                best_val = val_best
                best_i = ia
                best_j = j_best
    
    return best_i, best_j, best_val

def binary_local_search_all(h_bin, n, time_limit=300, rng=None):
    """All-swap binary local search."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_bin > 0.5).copy().astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            to_rm = rng.choice(ones, n_ones - n_half, replace=False)
            h[to_rm] = 0
        else:
            zeros = np.where(h == 0)[0]
            to_add = rng.choice(zeros, n_half - n_ones, replace=False)
            h[to_add] = 1
    
    corr = compute_corr(h).copy()
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    start = time.time()
    stall_count = 0
    max_stall = 20
    
    while remaining() > 10 and time.time() - start < time_limit:
        D_ones, D_zeros, ones, zeros = precompute_deltas(h, n)
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_i, best_j, best_val = best_swap_update(h, corr, D_ones, D_zeros, ones, zeros)
        
        if best_val < cur_max - 1e-12:
            i = ones[best_i]
            j = zeros[best_j]
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h).copy()  # Recompute to avoid drift
            cur_max = float(np.max(corr))
            stall_count = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall_count += 1
            if stall_count > max_stall:
                # Random perturbation: make 2-3 random swaps
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        h[i] = 0
                        h[j] = 1
                corr = compute_corr(h).copy()
                cur_max = float(np.max(corr))
                stall_count = 0
    
    return best_h, best_max * dx

def simulated_annealing(h_bin, n, time_limit=120, rng=None):
    """SA on binary solutions."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_bin > 0.5).copy().astype(float)
    n_half = n // 2
    # Ensure correct count
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, n_ones - n_half, replace=False)] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, n_half - n_ones, replace=False)] = 1
    
    corr = compute_corr(h).copy()
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    T = 5.0  # Initial temperature
    T_min = 0.1
    cooling = 0.995
    
    start = time.time()
    
    while remaining() > 10 and time.time() - start < time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        new_corr = compute_corr(h).copy()
        new_max = float(np.max(new_corr))
        
        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            corr = new_corr
            cur_max = new_max
        else:
            # Undo
            h[i] = 1
            h[j] = 0
            corr = corr  # unchanged
        
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
        
        T = max(T * cooling, T_min)
    
    return best_h, best_max * dx
