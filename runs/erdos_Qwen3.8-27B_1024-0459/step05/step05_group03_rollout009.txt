def coord_desc_1d(h, i, n):
    """Optimize h[i] by 1D convex minimization of max correlation."""
    dx = 2.0 / n
    # Compute f_k as linear function of t = h[i]
    # f_k(t) = a_k * t + b_k
    # where a_k = dx * [(1-h[i+m_k]) - h[i-m_k]] for the k that involve i
    # and b_k = f_k(0) = dx * sum_{j≠i} h[j](1-h[j+m_k])
    
    # For each k (lag m = k-(n-1)):
    # f_k involves h[i] if i is a valid index in the sum
    # Term 1: h[i] * (1-h[i+m]) contributes (1-h[i+m]) to a_k
    # Term 2: h[i-m] * (1-h[i]) contributes -h[i-m] to a_k (when i-m is valid)
    
    # Compute base correlations (with h[i] = 0)
    h[i] = 0.0
    base_corr = compute_corr(h)  # f_k with h[i]=0
    
    # Compute the "slope" a_k for each k
    a = np.zeros(2*n-1)
    m_range = np.arange(-n+1, n)  # lags
    for m in m_range:
        k = m + n - 1  # correlation index
        # Contribution to f_k from h[i]:
        # If 0 <= i < n and 0 <= i+m < n: term h[i]*(1-h[i+m]) → slope = (1-h[i+m])*dx
        # If 0 <= i-m < n and 0 <= i < n: term h[i-m]*(1-h[i]) → slope = -h[i-m]*dx
        slope = 0.0
        if 0 <= i+m < n:
            slope += (1.0 - h[i+m]) * dx
        if 0 <= i-m < n:
            slope -= h[i-m] * dx
        a[k] = slope
    
    # Now f_k(t) = a[k]*t + base_corr[k]
    # Minimize max_k (a[k]*t + base_corr[k]) subject to 0<=t<=1
    
    # This is a 1D convex (piecewise linear) minimization
    # The minimum is where the subgradient contains 0
    # Or at the boundary t=0 or t=1
    
    # Simple approach: dense sampling + refinement
    t_samples = np.linspace(0, 1, 200)
    vals = np.max(base_corr + t_samples[:, None] * a[None, :], axis=1)
    t_best = t_samples[np.argmin(vals)]
    
    # Refine: binary search around t_best
    lo = max(0, t_samples[max(0, np.argmin(vals)-1)])
    hi = min(1, t_samples[min(len(t_samples)-1, np.argmin(vals)+1)])
    for _ in range(50):
        mid = (lo + hi) / 2
        if np.max(base_corr + mid * a) <= np.max(base_corr + (mid + (hi-lo)*0.1) * a) if mid + (hi-lo)*0.1 <= 1 else True:
            hi = mid
        else:
            lo = mid
    t_opt = (lo + hi) / 2
    
    h[i] = t_opt
    return h

import numpy as np
from scipy.optimize import minimize, brent
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            # All at bounds, need to move some
            if diff > 0:
                idx = np.argmin(h)
                h[idx] += min(diff, 1.0 - h[idx])
            else:
                idx = np.argmax(h)
                h[idx] += max(diff, -(h[idx]))
        
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

# 1D coordinate descent: for coordinate i, find optimal t
def optimize_coord_1d(h, i, n, ngrid=500):
    """Optimize h[i] to minimize max correlation, ignoring sum constraint."""
    dx = 2.0 / n
    # Save current value
    t0 = h[i]
    h[i] = 0.0
    base = compute_corr(h)  # correlations with h[i]=0
    
    # Compute slope for each correlation
    # f_k(t) = base[k] + slope[k] * t
    slope = np.zeros(2*n-1)
    # h[i] appears in:
    # Term as "first": h[i] * (1-h[i+m]) for valid i+m
    # Term as "second": h[i-m] * (1-h[i]) for valid i-m
    m = np.arange(-n+1, n)  # lags
    # For each lag m (correlation index k = m + n - 1):
    # If i+m is valid (0<=i+m<n): contributes (1-h[i+m])*dx to slope
    # If i-m is valid (0<=i-m<n): contributes -h[i-m]*dx to slope
    valid1 = (i + m >= 0) & (i + m < n)
    valid2 = (i - m >= 0) & (i - m < n)
    slope[valid1] = (1.0 - h[i + m[valid1]]) * dx
    slope[valid2] -= h[i - m[valid2]] * dx
    
    # Grid search + refinement
    tg = np.linspace(0, 1, ngrid)
    # f_k(t) = base[k] + slope[k]*t for all k
    vals = base[None, :] + tg[:, None] * slope[None, :]
    maxvals = np.max(vals, axis=1)

    t_lo = tg[max(0, idx-1)]
    t_hi = tg[min(ngrid-1, idx+1)]
    
    # Bisection (convex function)
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2
        t_right = min(1.0, t_mid + (t_hi - t_lo) * 0.4)
        if np.max(base + t_mid * slope) <= np.max(base + t_right * slope):
            t_hi = t_right
        else:
            t_lo = t_mid
    t_opt = (t_lo + t_hi) / 2
    
    h[i] = t_opt
    return h

def coord_descent(h, n, n_sweeps=5, ngrid=500, sweep_order='forward'):
    """Multi-sweep coordinate descent."""

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
