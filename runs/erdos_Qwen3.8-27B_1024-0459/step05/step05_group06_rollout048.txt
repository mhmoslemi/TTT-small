def project_symmetric(h, n):

Let me also think about a completely different approach: **using the fact that the optimal solution is likely binary or near-binary**.

If h is binary (0 or 1), the problem becomes a combinatorial optimization:
- Choose exactly n/2 positions to be 1
- Minimize max_k (number of positions where h=1 and h shifted by k = 0, or vice versa)

This is equivalent to minimizing the maximum "asymmetric autocorrelation" of the binary sequence.

For this combinatorial problem, I could use:
- Simulated annealing
- Greedy + local search (swap two 1s)
- The coordinate descent I described (which for binary variables just tries 0 and 1)

Actually, for binary h, the coordinate descent becomes:
- For each position i, try changing h[i] from 0 to 1 (and h[j] from 1 to 0) or vice versa
- Check if C₅ decreases

This is a "swap" operation. Very fast to evaluate.

Let me implement a hybrid:
1. Start from the parent solution
2. Snap to binary (round to 0 or 1, fix sum)
3. Do swap-based local search
4. Then de-binary with coordinate descent (allow fractional values)
5. Polish with SLSQP

import numpy as np

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_array(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        mask = (h > 1e-14) & (h < 1.0 - 1e-14)

        else:

                # find closest to 1
                idx = np.argmax(h)

            else:
                idx = np.argmin(h)
                h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def slsqp_symmetric(h, n, beta=5000, maxiter=300):
    """SLSQP on the symmetric reduced problem."""
    half = n // 2
    dx = 2.0 / n
    
    # Variables: h[0], ..., h[half-1]
    # h[n-1-i] = 1 - h[i]
    # Constraint: sum(h) = n/2 is automatic due to symmetry
    # But we need 0 <= h[i] <= 1
    
    x0 = h[:half].copy()
    full_h = h.copy()
    
    def make_full(x):
        fh = np.empty(n)
        fh[:half] = x
        fh[n-1-half+1:] = 1.0 - x  # h[n-1-i] for i in 0..half-1
        # Wait: fh[n-1-i] = 1-x[i] for i in 0..half-1
        # So fh[n-1], fh[n-2], ..., fh[n-half] = 1-x[0], 1-x[1], ..., 1-x[half-1]
        # i.e., fh[n-half:] = 1 - x[::-1]
        fh[n-half:] = 1.0 - x[::-1]
        if n % 2 == 1:
            fh[half] = 0.5
        return fh
    
    def obj(x):
        fh = make_full(x)
        c = np.correlate(fh, 1.0 - fh, mode='full')
        m = np.max(c)
        # Smooth max
        shifted = np.minimum(beta * (c - m), 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    
    def grad_obj(x):
        fh = make_full(x)
        c = np.correlate(fh, 1.0 - fh, mode='full')
        m = np.max(c)
        shifted = np.minimum(beta * (c - m), 0.0)
        ew = np.exp(shifted)
        w = ew / ew.sum()
        
        # Gradient of c[k] w.r.t. fh[p]
        # c[k] = sum_n fh[n+k](1-fh[n])
        # dc[lag=d]/dfh[p] = (1-fh[p-d]) if p-d valid [from first factor at index p]
        #                   + (-fh[p+d]) if p+d valid [from second factor at index p]
        # In the 'full' array, index for lag d is d + (n-1)
        
        g = np.zeros(n)
        for k_idx in range(2*n-1):
            w_k = w[k_idx]
            if w_k < 1e-15:
                continue
            d = k_idx - (n-1)  # lag
            if d >= 0:
                # c = sum_{n=0}^{n-1-d} fh[n+d](1-fh[n])
                for n_idx in range(n - d):
                    g[n_idx + d] += w_k * (1.0 - fh[n_idx])
                    g[n_idx] -= w_k * fh[n_idx + d]
            else:
                dd = -d
                for n_idx in range(n - dd):
                    g[n_idx] += w_k * (1.0 - fh[n_idx + dd])
                    g[n_idx + dd] -= w_k * fh[n_idx]
        
        # Project gradient back to x (first half)
        gx = np.zeros(half)
        for i in range(half):
            gx[i] += g[i]  # from fh[i] = x[i]
            gx[i] -= g[n-1-i]  # from fh[n-1-i] = 1-x[i], dh[i]/dx[i] = -1
        return gx
    
    bounds = [(0.0, 1.0)] * half
    try:
        res = minimize(obj, x0, jac=grad_obj, method='SLSQP',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            result = make_full(res.x)
            result = np.clip(result, 0.0, 1.0)
            # Re-enforce symmetry
            for i in range(half):
                result[n-1-i] = 1.0 - result[i]
            return result
    except Exception:
        pass
    return h.copy()

def swap_local_search_binary(h, n, n_iterations=5000):
    """Local search on binary h via pairwise swaps."""
    h = h.copy()
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Fix sum to n/2
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Need to flip some 1s to 0s
        ones_idx = np.where(h_bin == 1.0)[0]
        # Flip the ones that are "most redundant"
        for idx in ones_idx[len(ones_idx)-(n_ones-target):]:
            h_bin[idx] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        for idx in zeros_idx[:target-n_ones]:
            h_bin[idx] = 1.0
    
    current_c5 = compute_c5(h_bin)
    current_corr = compute_corr_array(h_bin)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for it in range(n_iterations):
        if remaining_time() < 5:
            break
        # Pick random pair of 1 and 0 to swap
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Try swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_try = np.max(compute_corr_array(h_bin)) * dx
        if c5_try < current_c5 - 1e-15:
            current_c5 = c5_try
            current_corr = compute_corr_array(h_bin)
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    
    return h_bin, current_c5
