    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=1000):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < _sm_obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def grad_single_shift(h, k, n):
    """Gradient of corr[k] with respect to h."""
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def top_k_equalize(h, n, top_k=5, max_iters=800, time_limit=None):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        if time_limit is not None and time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-12:
            break
        g_dir = np.zeros(n)
        n_top = len(idx_top)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (n_top - 1))
            g_dir += w_k * grad_single_shift(h, k, n)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.002, 0.005, 0.01, 0.02, 0.05, 0.001, 0.003, 0.05, 0.1, 0.2]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def subgradient_zigzag(h, n, max_iters=5000, time_limit=90):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    k_curr = None
    k_prev = None
    stale_count = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        corr = compute_corr(h)
        k_curr = int(np.argmax(corr))

        if k_prev is not None and k_curr != k_prev:
            g = 0.5 * (grad_single_shift(h, k_curr, n) + grad_single_shift(h, k_prev, n))
            k_prev = k_curr
        else:
            g = grad_single_shift(h, k_curr, n)
            if k_prev is None:
                k_prev = k_curr

        g_norm = np.linalg.norm(g)
        if g_norm > 1e-14:
            g = g / g_norm

        improved = False
        base_step = max(0.001, 0.02 * (1.0 - 0.5 * it / max_iters))
        for step in [base_step * 0.2, base_step * 0.5, base_step, base_step * 2.0,
                     base_step * 5.0, 0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
            ht = project_feasible(h - step * g, n)
            c5_new = compute_c5(ht)
            if c5_new < best_c5 - 1e-15:
                h = ht
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                stale_count = 0
                break
        if not improved:
            stale_count += 1
            if stale_count > 200:
                break
            if stale_count % 50 == 0:
                rng = np.random.RandomState(stale_count)
                pert = rng.randn(n) * 0.005
                ht = project_feasible(h + pert, n)
                c5_new = compute_c5(ht)
                if c5_new < best_c5:
                    h = ht
                    best_c5 = c5_new
                    best_h = h.copy()
                    stale_count = 0

    return best_h, best_c5


def binary_2swap_sa(h, n, time_limit=120, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    curr_c5 = best_c5
    start = time.time()

    T_start = 5.0 / dx
    T_end = 0.001 / dx
    total_iters = 20000
    min_T_ratio = T_end / T_start

    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    no_improve = 0

    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        if remaining_time() < 5:
            break

        T = T_start * (min_T_ratio ** (it / max(1, total_iters)))

        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        if i == j:
            continue

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5

        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            ones[i_idx] = j
            zeros[j_idx] = i
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

        if no_improve > 5000:
            break

    return best_h, best_c5


def binary_2swap_greedy(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    no_improve_rounds = 0

    while time.time() - start < time_limit and remaining_time() > 5:
        improved = False
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        n_pairs = len(ones) * len(zeros)

        if n_pairs <= 50000:
            best_delta = 1e-10
            best_pair = None
            for ii in range(len(ones)):
                for jj in range(len(zeros)):
                    i = ones[ii]
                    j = zeros[jj]
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    c_new = compute_c5(h_bin)
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
                    delta = c_new - best_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                best_c5 = compute_c5(h_bin)
                best_h = h_bin.copy()
                improved = True
        else:
            n_trials = min(200, n_pairs)
            best_delta = 1e-10
            best_pair = None
            for _ in range(n_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c_new = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                delta = c_new - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                best_c5 = compute_c5(h_bin)
                best_h = h_bin.copy()
                improved = True

        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 20:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = compute_c5(h_bin)

    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def coordinate_refine(h, n, max_sweeps=30):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            if 0.01 < old < 0.99:
                cands = [old + 0.1, old - 0.1, old + 0.3, old - 0.3, 0.0, 1.0]
            else:
                cands = [0.0, 1.0, old + 0.05, old - 0.05]
            cands = [c for c in cands if 0.0 <= c <= 1.0 and abs(c - old) > 1e-6]
            best_cand_val = old
            best_cand_c5 = best_c5
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-13:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive SLSQP on parent resolution
    h = best_h.copy()
    for beta in [500, 2000, 10000, 50000, 200000, 1000000, 5000000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization (larger k)
    if remaining_time() > 20:
        for tk in [5, 10, 20]:
            if remaining_time() < 15:
                break
            h_eq, c5_eq = top_k_equalize(h, best_n, top_k=tk, max_iters=1000, time_limit=60)
            if c5_eq < best_c5 - 1e-15:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 3: Subgradient zigzag
    if remaining_time() > 30:
        h_sg, c5_sg = subgradient_zigzag(h, best_n, max_iters=8000, time_limit=90)
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: More SLSQP after all the above
    for beta in [500000, 2000000, 10000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Multi-resolution search
    n_candidates = [512, 640, 768, 448, 320]
    for n_t in n_candidates:
        if remaining_time() < 40:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 15:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=1000)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=6, max_iters=500, time_limit=40)
        h_opt, c5_opt = subgradient_zigzag(h_opt, n_t, max_iters=3000, time_limit=30)

        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 6: Binary 2-swap simulated annealing
    if remaining_time() > 60:
        sa_time = min(150, int(remaining_time() * 0.35))
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_2swap_sa(h_bin, best_n, time_limit=sa_time, rng=rng)
        # Refine continuously
        h_b_c = h_b.copy()
        for beta in [10000, 100000, 1000000]:
            if remaining_time() < 15:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=1500)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = top_k_equalize(h_b_c, best_n, top_k=8, max_iters=500, time_limit=30)
        h_b_c, c5_b_c = subgradient_zigzag(h_b_c, best_n, max_iters=3000, time_limit=30)

        if c5_b_c < best_c5 - 1e-15:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy().astype(float)

    # Phase 7: Binary 2-swap greedy
    if remaining_time() > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_g, c5_g = binary_2swap_greedy(h_bin, best_n, time_limit=min(60, int(remaining_time()*0.3)), rng=rng)
        h_g_c = h_g.copy()
        for beta in [50000, 500000]:
            if remaining_time() < 15:
                break
            h_g_c = slsqp_opt(h_g_c, best_n, beta, maxiter=1000)
        c5_g_c = compute_c5(h_g_c)
        h_g_c, c5_g_c = top_k_equalize(h_g_c, best_n, top_k=8, max_iters=400, time_limit=25)

        if c5_g_c < best_c5 - 1e-15:
            best_c5 = c5_g_c
            best_h = h_g_c.copy()
            h = h_g_c.copy()

    # Phase 8: Alternative starting points
    if remaining_time() > 50:
        alt_starts = [theta_morse(best_n)]
        # Random balanced
        h_rand = np.zeros(best_n)
        h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
        alt_starts.append(h_rand)
        # Shifted
        alt_starts.append(project_feasible(np.roll(best_h, 1), best_n))

        for h_alt in alt_starts:
            if remaining_time() < 30:
                break
            h_opt = h_alt
            for beta in [5000, 100000, 1000000]:
                if remaining_time() < 10:
                    break
                h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=1000)
            c5_opt = compute_c5(h_opt)
            h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=8, max_iters=400, time_limit=30)
            h_opt, c5_opt = subgradient_zigzag(h_opt, best_n, max_iters=2000, time_limit=25)
            if c5_opt < best_c5 - 1e-15:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 9: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [1000000, 5000000, 20000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=10, max_iters=800, time_limit=30)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 15:
        h_sg, c5_sg = subgradient_zigzag(best_h, n, max_iters=5000, time_limit=min(60, int(remaining_time()*0.5)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Coordinate refinement
    if remaining_time() > 15:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=20)
        if c5_cr < best_c5 - 1e-15:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 10000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, n)
