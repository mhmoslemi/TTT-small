corr'[k] = corr[k] + δ*L_k + δ²*Q_k

import numpy as np
from scipy.optimize import minimize
import time

_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def eval_paired_obj(corr, a, c_vec, delta):
    return np.max(corr + delta * a + delta * delta * c_vec)


def golden_section_paired(corr, a, c_vec, d_lo, d_hi, max_iter=45):
    if d_hi - d_lo < 1e-14:
        return d_lo, eval_paired_obj(corr, a, c_vec, d_lo)
    gr = 0.6180339887498949
    x1 = d_hi - gr * (d_hi - d_lo)
    x2 = d_lo + gr * (d_hi - d_lo)
    f1 = eval_paired_obj(corr, a, c_vec, x1)
    f2 = eval_paired_obj(corr, a, c_vec, x2)
    best_f = min(f1, f2)
    best_d = x1 if f1 <= f2 else x2
    f_lo = eval_paired_obj(corr, a, c_vec, d_lo)
    f_hi = eval_paired_obj(corr, a, c_vec, d_hi)
    if f_lo < best_f:
        best_f, best_d = f_lo, d_lo
    if f_hi < best_f:
        best_f, best_d = f_hi, d_hi
    for _ in range(max_iter):
        if d_hi - d_lo < 1e-13:
            break
        if f1 < f2:
            d_hi = x2
            x2 = x1
            f2 = f1
            x1 = d_hi - gr * (d_hi - d_lo)
            f1 = eval_paired_obj(corr, a, c_vec, x1)
        else:
            d_lo = x1
            x1 = x2
            f1 = f2
            x2 = d_lo + gr * (d_hi - d_lo)
            f2 = eval_paired_obj(corr, a, c_vec, x2)
        if f1 < best_f:
            best_f, best_d = f1, x1
        if f2 < best_f:
            best_f, best_d = f2, x2
    return best_d, best_f


def paired_cd_round(h, n, n_pairs, rng, max_gs=45):
    h = h.copy()
    corr = compute_corr(h)
    ks = np.arange(2 * n - 1)
    c5 = np.max(corr) * (2.0 / n)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        a = np.zeros(2 * n - 1)
        idx_i = i + ks - (n - 1)
        idx_j = j + ks - (n - 1)
        valid_i = (idx_i >= 0) & (idx_i < n)
        valid_j = (idx_j >= 0) & (idx_j < n)
        a[valid_i] -= h[idx_i[valid_i]]
        a[valid_j] += h[idx_j[valid_j]]
        a[n - 1] -= h[i]
        k2 = j - i + n - 1
        if 0 <= k2 < 2 * n - 1:
            a[k2] += h[i]
        k3 = i - j + n - 1
        if 0 <= k3 < 2 * n - 1:
            a[k3] += h[j]
        a[n - 1] -= h[j]
        c_vec = np.zeros(2 * n - 1)
        c_vec[n - 1] = -2.0
        if 0 <= k2 < 2 * n - 1:
            c_vec[k2] = 1.0
        if 0 <= k3 < 2 * n - 1:
            c_vec[k3] = 1.0
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-14:
            continue
        delta_opt, f_opt = golden_section_paired(corr, a, c_vec, d_lo, d_hi, max_gs)
        f_current = np.max(corr)
        if f_opt < f_current - 1e-14:
            h[i] += delta_opt
            h[j] -= delta_opt
            corr += delta_opt * a + delta_opt * delta_opt * c_vec
            new_c5 = np.max(corr) * (2.0 / n)
            if new_c5 < c5:
                c5 = new_c5
    return h, c5


def smooth_max_slsqp(h, n, beta, maxiter=400):
    dx = 2.0 / n

    def sm_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def sm_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.success or res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h


def subgradient_round(h, n, n_iters, rng, K=15, beta=50.0):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta * (c_top - c_top.max()))
        s = weights.sum()
        if s < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= s
        g = np.zeros(n)
        ks_arr = np.arange(2 * n - 1)
        for ii in range(K):
            if weights[ii] < 1e-12:
                continue
            k = idx[ii]
            w = weights[ii]
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += w * (1.0 - h[:np_])
                g[:np_] -= w * h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += w * (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= w * h[:np_]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.03 / (1.0 + 0.01 * it)
        h_new = project_feasible(h - step * g / gn, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g / gn, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    h = h_try
                    break
    return h, best_c5
