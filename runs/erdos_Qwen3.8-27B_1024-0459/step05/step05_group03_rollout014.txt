    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def projected_grad_desc(h, n, beta, max_iters=500):
    """Projected subgradient descent with momentum."""
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.8
    step = 0.03

    for it in range(max_iters):
        if remaining() < 5:
            break
        g = smooth_grad(h, beta)
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        elif it % 100 == 0:
            h_rollback = project_feasible(h - 0.3 * momentum, n)
            if compute_c5(h_rollback) < compute_c5(h):
                h = h_rollback
        step *= 0.999
        if it % 200 == 0:
            mu = min(0.95, mu + 0.02)

    return best_h, best_c5


def slsqp_refine(h, n, beta, maxiter=300):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.success or res.fun < smooth_obj(h0, beta) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def random_binary(n):
    h = np.zeros(n)
    idx = np.random.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_n = parent_n
        best_c5 = compute_c5(best_h)
    except Exception:
        parent_n = 384
        best_n = parent_n
        best_h = np.zeros(parent_n)
        best_h[:parent_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Phase 1: Round parent to binary and do targeted local search
    h_bin = round_to_binary(best_h, best_n)
    best_h_bin, best_c5_bin = swap_at_distance_ls(h_bin, best_n, max_rounds=300)
    if best_c5_bin < best_c5:
        best_c5 = best_c5_bin
        best_h = best_h_bin.copy()

    # Full pair search on best binary
    if remaining() > 60:
        h_fp, c5_fp = full_pair_ls(best_h_bin, best_n, max_rounds=30)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # 2-swap search
    if remaining() > 60:
        h_ts, c5_ts = two_swap_ls(best_h, best_n, max_rounds=40)
        if c5_ts < best_c5:
            best_c5 = c5_ts
            best_h = h_ts.copy()

    # Phase 2: Multi-resolution binary search
    res_list = [256, 512, 768]
    for n_t in res_list:
        if remaining() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = round_to_binary(h_interp, n_t)
        h_t, c5_t = swap_at_distance_ls(h_bin_t, n_t, max_rounds=150)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 3: Continuous refinement at best resolution
    if remaining() > 40:
        h_cont = best_h.astype(float).copy()
        h_cont = project_feasible(h_cont, best_n)
        for beta in [2000, 10000, 50000]:
            if remaining() < 15:
                break
            h_pg, c5_pg = projected_grad_desc(h_cont, best_n, beta, max_iters=300)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                h_cont = h_pg.copy()

    # Phase 4: SLSQP polish
    if remaining() > 30:
        h_opt = slsqp_refine(best_h, best_n, 20000, maxiter=200)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Round to binary again and do final local search
    if remaining() > 30:
        h_final_bin = round_to_binary(best_h, best_n)
        h_fb, c5_fb = swap_at_distance_ls(h_final_bin, best_n, max_rounds=200)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()

        if remaining() > 20:
            h_fp2, c5_fp2 = full_pair_ls(h_fb, best_n, max_rounds=20)
            if c5_fp2 < best_c5:
                best_c5 = c5_fp2
                best_h = h_fp2.copy()

    # Phase 6: Try additional resolutions
    if remaining() > 100:
        for n_t in [448, 640, 896]:
            if remaining() < 60:
                break
            h_interp = interp_to(best_h, best_n, n_t)
            h_bin_t = round_to_binary(h_interp, n_t)
            h_t, c5_t = swap_at_distance_ls(h_bin_t, n_t, max_rounds=100)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 7: Random restarts at best resolution
    if remaining() > 60:
        for restart in range(3):
            if remaining() < 30:
                break
            h_rand = random_binary(best_n)
            # Blend with current best
            blend = 0.7 * h_rand + 0.3 * best_h
            h_blend_bin = round_to_binary(blend, best_n)
            h_r, c5_r = swap_at_distance_ls(h_blend_bin, best_n, max_rounds=80)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 8: Final SLSQP + binary polish
    if remaining() > 15:
        h_opt2 = slsqp_refine(best_h, best_n, 50000, maxiter=150)
        c5_opt2 = compute_c5(h_opt2)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()

    if remaining() > 10:
        h_bin_f = round_to_binary(best_h, best_n)
        h_ff, c5_ff = swap_at_distance_ls(h_bin_f, best_n, max_rounds=100)
        if c5_ff < best_c5:
            best_c5 = c5_ff
            best_h = h_ff.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
