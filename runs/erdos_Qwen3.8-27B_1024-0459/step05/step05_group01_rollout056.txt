        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 1.0
    cool_rate = 0.9995
    no_improve = 0
    max_no_improve = 3000
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= cool_rate
        if no_improve > max_no_improve:
            break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at parent resolution
    h = best_h.copy()
    for beta in [200, 500, 2000, 10000, 50000, 200000, 1000000]:
        if remaining_time() < 40:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization
    if remaining_time() > 20:
        h_eq, c5_eq = topk_equalize(h, best_n, top_k=5, max_iters=600)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Coordinate descent (new approach)
    if remaining_time() > 60:
        h_cd, c5_cd = coord_descent(best_h.copy(), best_n, max_sweeps=50)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
        else:
            h = best_h.copy()

    # Phase 4: Subgradient top-k method
    if remaining_time() > 60:
        h_sg, c5_sg = subgradient_topk(h, best_n, top_k=5, max_iters=2000)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()
        else:
            h = best_h.copy()

    # Phase 5: More SLSQP after all the above
    for beta in [200000, 1000000, 5000000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Multi-resolution search
    resolutions = [576, 768, 1024, 512, 640]
    resolutions = [r for r in resolutions if r % 2 == 0 and r >= 256 and r <= 2048]
    resolutions = sorted(set(resolutions), key=lambda x: abs(x - best_n))

    for n_t in resolutions[:4]:
        if remaining_time() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [1000, 10000, 100000]:
            if remaining_time() < 30:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=500)

        h_opt, c5_opt = topk_equalize(h_opt, n_t, top_k=5, max_iters=400)

        if remaining_time() > 30:
            h_opt = slsqp_opt(h_opt, n_t, 500000, maxiter=600)
            c5_opt = compute_c5(h_opt)

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 7: Binary SA + continuous refinement
    if remaining_time() > 120:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0

        sa_time = min(150, max(20, int(remaining_time() * 0.2)))
        h_b, c5_b = binary_sa(h_bin, best_n, sa_time, rng)

        h_b_c = project_feasible(h_b, best_n)
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 15:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=600)
        c5_bc = compute_c5(h_b_c)
        h_b_c, c5_bc = topk_equalize(h_b_c, best_n, top_k=5, max_iters=400)

        if c5_bc < best_c5 - 1e-14:
            best_c5 = c5_bc
            best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 8: Try alternative starts
    if remaining_time() > 80:
        h_tm = theta_morse(best_n)
        h_tm_opt = h_tm
        for beta in [5000, 50000]:
            if remaining_time() < 20:
                break
            h_tm_opt = slsqp_opt(h_tm_opt, best_n, beta, maxiter=500)
        c5_tm = compute_c5(h_tm_opt)
        h_tm_opt, c5_tm = topk_equalize(h_tm_opt, best_n, top_k=5, max_iters=300)
        if c5_tm < best_c5 - 1e-14:
            best_c5 = c5_tm
            best_h = h_tm_opt.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 30:
        h_cd2, c5_cd2 = coord_descent(best_h.copy(), best_n, max_sweeps=40)
        if c5_cd2 < best_c5 - 1e-14:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 15:
        h_final = slsqp_opt(best_h, best_n, 2000000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 10:
        h_eq2, c5_eq2 = topk_equalize(best_h, best_n, top_k=7, max_iters=400)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
