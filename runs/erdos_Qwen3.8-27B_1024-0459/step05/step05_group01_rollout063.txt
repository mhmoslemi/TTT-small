- There exist multipliers λ_k ≥ 0 for each constraint f_k(h) ≤ t
- λ_k * (f_k(h) - t) = 0 (complementary slackness)

    # Random balanced

    h_rnd[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
    alt_starts.append(h_rnd)
    # Shifted
    h_shift = np.roll(best_h, best_n // 3)
    alt_starts.append(project_feasible(h_shift, best_n))
    # Anti-symmetric perturbation
    h_pert = best_h.copy()
    h_pert[rng.choice(best_n, best_n // 24, replace=False)] = 1.0 - best_h[rng.choice(best_n, best_n // 24, replace=False)]
    alt_starts.append(project_feasible(h_pert, best_n))

    for h_alt in alt_starts:
        if remaining_time() < 30:
            break
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()
        if remaining_time() > 25:
            h_alt_c, c5_alt_c = continuous_refine(h_alt, best_n, time_limit=25)
            if c5_alt_c < best_c5 - 1e-15:
                best_c5 = c5_alt_c
                best_h = h_alt_c

    # Phase 7: Second round of SA from best
    if remaining_time() > 40:
        sa_time2 = min(80, int(remaining_time() * 0.25))
        h_bin2 = (best_h > 0.5).astype(float)
        nh2 = best_n // 2
        if int(np.sum(h_bin2)) != nh2:
            idx2 = np.argsort(-h_bin2)[:nh2]
            h_bin2 = np.zeros(best_n)
            h_bin2[idx2] = 1.0
        h_sa2, c5_sa2 = binary_sa(h_bin2, best_n, sa_time2, rng, max_iters=2000000)
        if c5_sa2 < best_c5 - 1e-15:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        # Refine
        if remaining_time() > 30:
            h_r2, c5_r2 = continuous_refine(best_h, best_n, time_limit=40)
            if c5_r2 < best_c5 - 1e-15:
                best_c5 = c5_r2
                best_h = h_r2

    # Phase 8: Final intensive SLSQP polish
    if remaining_time() > 15:
        for beta in [100000, 500000, 1000000]:
            if remaining_time() < 8:
                break
            h_pol = slsqp_opt(best_h, best_n, beta, maxiter=600)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5 - 1e-15:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 9: Final active set + coordinate
    if remaining_time() > 10:
        h_fin, c5_fin = active_set_equalize(best_h, best_n, top_k=10, max_iters=1000, time_limit=30)
        if c5_fin < best_c5 - 1e-15:
            best_c5 = c5_fin
            best_h = h_fin

    # Final L-BFGS-B polish with very high beta
    if remaining_time() > 8:
        h_fin2 = lbfgs_opt(best_h, best_n, 500000, maxiter=400, pen=1e6)
        c5_fin2 = compute_c5(h_fin2)
        if c5_fin2 < best_c5 - 1e-15:
            best_c5 = c5_fin2
            best_h = h_fin2

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
