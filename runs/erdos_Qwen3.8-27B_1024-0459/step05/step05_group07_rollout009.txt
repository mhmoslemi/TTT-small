def softmax_obj_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

            h_new, c5_new = coord_descent(h_tm, best_n, max_sweeps=120)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 30:
                    h_slq = slsqp_refine(h_new, best_n, 20000, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()

    # Phase 10: Try higher resolution n=1280 if time allows
    if remaining_time() > 120:
        n3 = 1280
        h3 = interp_to(best_h, best_n, n3)
        h3 = project_feasible(h3, n3)
        h3, c5_3 = coord_descent(h3, n3, max_sweeps=80)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n3
        if remaining_time() > 40:
            h3_new = slsqp_refine(best_h, best_n, 3000, 250)
            c5_slq = compute_c5(h3_new)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h3_new.copy()
            if remaining_time() > 20:
                h3_new, c5_cd = coord_descent(best_h, best_n, max_sweeps=50)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h3_new.copy()

    # Phase 11: Active-set + CD alternating
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = active_set_search(h, best_n, max_rounds=50, top_k=5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 12: Final SLSQP polish
    if remaining_time() > 15:
        h = best_h.copy()
        for beta in [50000, 200000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_refine(h, best_n, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new

    # Phase 13: Final coordinate descent
    if remaining_time() > 10:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Recompute to be safe
        final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
