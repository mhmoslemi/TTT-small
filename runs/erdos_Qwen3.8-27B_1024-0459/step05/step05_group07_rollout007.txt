            # corr[k] = sum_{j: valid} h[j] * (1 - h[j + k - (n-1)])
            # Let offset = k - (n-1)
            # corr[k] = sum_{j: 0<=j<n, 0<=j+offset<n} h[j] * (1-h[j+offset])
            # d/dh[i]: if i is in first position (j=i, need 0<=i+offset<n): (1-h[i+offset])
            #          if i is in second position (j+offset=i, need 0<=i-offset<n): -h[i-offset]
            # For offset=0: both give (1-h[i]) - h[i] ... wait no.
            # If offset=0: corr = sum h[j]*(1-h[j])
            # d/dh[i] = (1-2*h[i]) ... actually:
            # d/dh[i] [h[i]*(1-h[i])] = (1-h[i]) - h[i] = 1-2*h[i]
            # But from the two-term formula: (1-h[i]) + (-h[i]) = 1-2*h[i]. Correct.

            A = np.zeros(ncorr)
            for k in range(ncorr):
                offset = k - (n - 1)
                # i in first position: need 0 <= i+offset < n
                if 0 <= i + offset < n:
                    A[k] += (1.0 - best[i + offset])
                # i in second position: j+offset = i, so j = i-offset, need 0 <= i-offset < n
                if 0 <= i - offset < n:
                    A[k] -= best[i - offset]

            # base_k = corr_k - A_k * old, except for offset 0 where it's quadratic
            base = best_corr - A * old
            # Fix the offset-0 term (index n-1):
            # corr[n-1] includes h[i]*(1-h[i]) from this coordinate
            # base[n-1] should be corr[n-1] - h[i]*(1-h[i]) + old*(1-old) ... 
            # Actually: corr[n-1] = sum_j h[j]*(1-h[j])
            # = (sum_{j≠i} h[j]*(1-h[j])) + old*(1-old)
            # base[n-1] = corr[n-1] - A[n-1]*old
            # A[n-1] = (1-best[i]) - best[i] = 1-2*old (since offset=0, both terms apply)
            # base[n-1] = corr[n-1] - (1-2*old)*old = corr[n-1] - old + 2*old²
            # But we want: sum_{j≠i} h[j](1-h[j]) = corr[n-1] - old*(1-old) = corr[n-1] - old + old²
            # Hmm, these don't match. Let me just directly set base[n-1]:
            base[base_idx] = best_corr[base_idx] - old * (1.0 - old)

            # Now optimize: min_t max(base[k] + A[k]*t for k≠base_idx, base[base_idx] + t - t²)
            t_opt, f_opt = golden_1d(base, A, old, ncorr, base_idx, n_loops=55)

            if t_opt != old:
                # Update
                best[i] = t_opt
                # Update correlation efficiently
                for k in range(ncorr):
                    if A[k] != 0.0:
                        best_corr[k] += A[k] * (t_opt - old)
                best_corr[base_idx] += t_opt * (1.0 - t_opt) - old * (1.0 - old)
                best_corr = np.clip(best_corr, 0.0, None)  # correlations should be non-negative
                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break
        if verbose and sweep % 50 == 0:
            print(f"  CD sweep {sweep}: C5={best_c5:.8f}")

    return best, best_c5


def vectorized_coord_descent(h, n, max_sweeps=200):
    """Faster coordinate descent using precomputed offset structure."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    base_idx = n - 1
    best_c5 = np.max(best_corr) * dx

    # Precompute: for each i, which k values are affected and what A[k] is
    # This is O(n * ncorr) which for n=640 is ~800K - doable per sweep if vectorized

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        # Process coordinates in random order to avoid bias
        order = np.arange(n)
        for i in order:
            old = best[i]
            if old <= 1e-15 or old >= 1.0 - 1e-15:
                old = np.clip(old, 0.0, 1.0)

            # Compute A vectorized
            offsets = np.arange(-(n - 1), n)  # shape (2n-1,)
            ip = i + offsets  # i + offset for each k
            im = i - offsets  # i - offset for each k
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)

            A = np.zeros(ncorr)
            A[vp] += (1.0 - best[ip[vp]])
            A[vm] -= best[im[vm]]

            base = best_corr - A * old
            base[base_idx] = best_corr[base_idx] - old * (1.0 - old)

            # Golden section on [0,1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0

            def fv(t):
                mid = base + A * t
                mid[base_idx] = base[base_idx] + t - t * t
                return np.max(mid)

            f_old = fv(old)
            lo = 0.0
            hi = 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = fv(c1)
            f2 = fv(c2)
            if f1 > f2:
                f1, f2 = f2, f1
                c1, c2 = c2, c1
            for _ in range(50):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = fv(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = fv(c2)
            t_opt = (lo + hi) / 2.0
            f_opt = fv(t_opt)

            if f_opt < f_old - 1e-13:
                best[i] = t_opt
                best_corr += A * (t_opt - old)
                best_corr[base_idx] += t_opt * (1.0 - t_opt) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break

    return best, best_c5


def binary_push(h, n, n_iters=200):
    """Try pushing values toward 0 or 1 to exploit the non-uniformity."""
    best = project_feasible(h, n)
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    base_idx = n - 1
    best_c5 = np.max(best_corr) * dx

    for it in range(n_iters):
        if remaining_time() < 5:
            break
        improved = False
        # Try small perturbations toward 0 or 1
        for i in range(n):
            old = best[i]
            if old <= 0.01 or old >= 0.99:
                continue
            # Try pushing toward 0 or 1 by a small amount
            delta = 0.02
            for direction in [-1, 1]:
                t = np.clip(old + direction * delta, 0.0, 1.0)
                if t == old:
                    continue
                # Evaluate
                offsets = np.arange(-(n - 1), n)
                ip = i + offsets
                im = i - offsets
                vp = (ip >= 0) & (ip < n)
                vm = (im >= 0) & (im < n)
                A = np.zeros(ncorr)
                A[vp] += (1.0 - best[ip[vp]])
                A[vm] -= best[im[vm]]
                new_corr = best_corr + A * (t - old)
                new_corr[base_idx] += t * (1 - t) - old * (1 - old)
                c5_new = np.max(new_corr) * dx
                if c5_new < best_c5 - 1e-14:
                    best[i] = t
                    best_corr = new_corr.copy()
                    best_c5 = c5_new
                    improved = True
                    break

        if not improved:
            break

    return project_feasible(best, n), best_c5


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def active_set_slsqp(h, n, max_active=20, maxiter=500):
    """SLSQP focusing on active constraints."""
    corr = compute_corr(h)
    dx = 2.0 / n
    max_c = np.max(corr)
    # Get active set (top constraints)
    active = np.argsort(corr)[::-1][:min(max_active, len(corr))]
    beta = 100000.0 / max_active

    _N_WORK_local = n
    _BETA_LOCAL = beta
    dx_local = dx

    def obj_local(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = np.minimum(beta * (c - m), 0.0)
        ev = np.exp(sh)
        ws = np.sum(ev)
        return (m + np.log(ws) / beta) * dx_local

    def grad_local(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = np.minimum(beta * (c - m), 0.0)
        ev = np.exp(sh)
        w = ev / np.sum(ev)
        g = np.zeros(n)
        for k in range(len(c)):
            if w[k] < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    g[off:off+np_] += w[k] * (1.0 - x[:np_])
                    g[:np_] -= w[k] * x[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ > 0:
                    g[:np_] += w[k] * (1.0 - x[ao:ao+np_])
                    g[ao:ao+np_] -= w[k] * x[:np_]
        return g * dx_local

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_local, h, jac=grad_local, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-28, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-13:
                return trial
    except Exception:
        pass
    return h.copy()


def multi_scale_init(n, rng):
    """Create a multi-scale structured initialization."""
    h = np.zeros(n)
    # Start with a few blocks
    n_blocks = rng.choice([4, 8, 16, 32])
    block_size = n // n_blocks
    for b in range(n_blocks):
        if b % 2 == 0:
            h[b*block_size:(b+1)*block_size] = 1.0
    h = project_feasible(h, n)
    # Add noise
    h += rng.randn(n) * 0.1
    h = np.clip(h, 0, 1)
    h = project_feasible(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Thorough coordinate descent from parent
    if remaining_time() > 60:
        h = best_h.copy()
        h, c5 = vectorized_coord_descent(h, n, max_sweeps=250)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: SLSQP refinement
    if remaining_time() > 40:
        h = best_h.copy()
        h_new = active_set_slsqp(h, n, max_active=15, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: More coordinate descent
    if remaining_time() > 40:
        h = best_h.copy()
        h, c5 = vectorized_coord_descent(h, n, max_sweeps=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Binary push to encourage non-uniform values
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = binary_push(h, n, n_iters=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Follow up with CD
        if remaining_time() > 20:
            h, c5 = vectorized_coord_descent(h, n, max_sweeps=100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 5: Perturbation + re-optimization
    if remaining_time() > 60:
        for p in range(6):
            if remaining_time() < 30:
                break
            h_pert = best_h.copy()
            scale = 0.1 / (1.0 + p * 0.5)
            mask = rng.random(n) < 0.2
            h_pert[mask] += rng.randn(np.sum(mask)) * scale
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = vectorized_coord_descent(h_pert, n, max_sweeps=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 30:
                    h_slq = active_set_slsqp(h_new, n, 15, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()

    # Phase 6: Try Thue-Morse and structured inits
    if remaining_time() > 60:
        h_tm = thue_morse_init(n)
        h_new, c5_new = vectorized_coord_descent(h_tm, n, max_sweeps=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            if remaining_time() > 30:
                h_slq = active_set_slsqp(h_new, n, 15, 400)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    h_cd, c5_cd = vectorized_coord_descent(h_slq, n, max_sweeps=80)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd.copy()

        # Multi-scale init
        if remaining_time() > 40:
            h_ms = multi_scale_init(n, rng)
            h_new, c5_new = vectorized_coord_descent(h_ms, n, max_sweeps=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: Higher resolution
    if remaining_time() > 150:
        high_n = 2 * n
        if high_n <= 2048:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            c5_high = compute_c5(h_high)
            if c5_high < best_c5 + 0.001:  # Only proceed if not much worse
                h_new, c5_new = vectorized_coord_descent(h_high, high_n, max_sweeps=100)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = high_n
                    if remaining_time() > 60:
                        h_slq = active_set_slsqp(h_new, high_n, 15, 300)
                        c5_slq = compute_c5(h_slq)
                        if c5_slq < best_c5:
                            best_c5 = c5_slq
                            best_h = h_slq.copy()
                            best_n = high_n
                        h_cd, c5_cd = vectorized_coord_descent(best_h, best_n, max_sweeps=60)
                        if c5_cd < best_c5:
                            best_c5 = c5_cd
                            best_h = h_cd.copy()
                            best_n = best_n

    # Phase 8: Very fine coordinate descent (many sweeps, small improvements)
    if remaining_time() > 30:
        n_cur = best_n
        h = best_h.copy()
        h, c5 = vectorized_coord_descent(h, n_cur, max_sweeps=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 9: Final SLSQP with very high beta
    if remaining_time() > 20:
        n_cur = best_n
        h = best_h.copy()
        for beta_try in [50000, 200000]:
            if remaining_time() < 10:
                break
            _N_WORK = n_cur
            _BETA_WORK = beta_try
            h_new = slsqp_refine(h, n_cur, beta_try, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 10: Final coordinate descent polish
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = vectorized_coord_descent(best_h, n_cur, max_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
