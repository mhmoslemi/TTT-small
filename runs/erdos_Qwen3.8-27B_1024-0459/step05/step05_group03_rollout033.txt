def fast_swap(h, a, b):
    """Apply swap h[a]: 1->0, h[b]: 0->1 and return the delta in correlation."""
    n = len(h)
    delta = np.zeros(2*n-1)
    
    # h[a]: 1 -> 0
    # Contribution as first factor: -(1-h[a-shift]) for k in [a+1, a+n-1]
    k_start = a + 1
    k_end = a + n - 1
    shifts = np.arange(k_start, k_end+1) - (n-1)  # shift values
    # a - shift should be in [0, n-1]
    idx = a - shifts  # = a - (k - (n-1)) = a + n - 1 - k
    delta[k_start:k_end+1] -= (1 - h[idx])
    
    # Contribution as second factor: -h[a+shift] for k in [n-1-a, 2n-2-a]
    k_start = n - 1 - a
    k_end = 2*n - 2 - a
    shifts = np.arange(k_start, k_end+1) - (n-1)
    idx = a + shifts  # = a + k - (n-1)
    delta[k_start:k_end+1] -= h[idx]
