            # Accept
            if new_corr < best_corr_max:
                best_corr_max = new_corr
                best_h = h.copy()
        else:
            # Reject
            h[pa] = 1.0
            h[pb] = 0.0
    
    return best_h, best_corr_max * dx

def binary_ls_2swap_sampled(h_bin, n, max_rounds=30, samples_per_round=500):
    """Binary local search with sampled 2-swap moves."""
    h = (h_bin > 0.5).astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(123)
    
    for rnd in range(max_rounds):
        if remaining() < 10:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        c_max = np.max(compute_corr(h))
        best_improve = 1e-15
        best_move = None
        
        for _ in range(samples_per_round):
            ia = rng.randint(len(ones))
            ib = (ia + 1 + rng.randint(len(ones) - 1)) % len(ones)
            ic = rng.randint(len(zeros))
            id_ = (ic + 1 + rng.randint(len(zeros) - 1)) % len(zeros)
            
            pa, pb = ones[ia], ones[ib]
            pc, pd = zeros[ic], zeros[id_]
            
            h[pa] = 0; h[pb] = 0; h[pc] = 1; h[pd] = 1
            nc = np.max(compute_corr(h))
            h[pa] = 1; h[pb] = 1; h[pc] = 0; h[pd] = 0
            
            if c_max - nc > best_improve:
                best_improve = c_max - nc
                best_move = (pa, pb, pc, pd)
        
        if best_move is None:
            break
        pa, pb, pc, pd = best_move
        h[pa] = 0; h[pb] = 0; h[pc] = 1; h[pd] = 1
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def nesterov_smooth(h, n, beta, max_iters=2000):
    h = project_feasible(h, n)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        y_new = y + alpha * (y - h)
        g = smooth_grad(h
, beta)
        lr = 1.0 / (beta * (it + 10))
        h_new = project_feasible(y_new - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new
    return best_h, best_c5

def subgrad_topk(h, n, top_k=5, max_iters=800):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.85
    step = 0.05
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w_top = c_top - c_top.min() + 1e-10
        w_top = w_top ** 2
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= dx
        g = corr_grad_weighted(h, weights)
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        elif it % 50 == 0:
            h_rollback = project_feasible(h - 0.5 * momentum, n)
            cr = compute_c5(h_rollback)
            if cr < compute_c5(h):
                h = h_rollback
        step *= 0.9995
        if it % 100 == 0:
            mu = min(0.95, mu + 0.01)
    return best_h, best_c5

def sturmian_pattern(n, alpha):
    """Generate a Sturmian-like binary pattern."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (int((i + 1) * alpha) - int(i * alpha)) else 0.0
    # Adjust to have exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        # Remove excess ones
        idx = np.where(h == 1)[0]
        for i in range(n_ones - n // 2):
            h[idx[len(idx) - 1 - i]] = 0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        for i in range(n // 2 - n_ones):
            h[idx[len(idx) - 1 - i]] = 1.0
    return h
