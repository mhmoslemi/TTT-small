import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 1000.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def _g_smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]


def _g_smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]


def _g_constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0


def _g_constraint_sum_jac(x):
    return np.ones(_N_WORK)


def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    c5_orig = compute_c5(h)
    try:
        res = minimize(_g_smooth_obj, h, jac=_g_smooth_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-28, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n, dtype=float) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n, dtype=float) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def fast_coord_descent(h, n, max_sweeps=200, rng=None):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        if rng is not None and sweep % 5 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))

            # Evaluate at a few key points
            t_candidates = [0.0, 1.0, 0.5, old]
            # Add points where linear terms cross the quadratic
            for k_idx in range(ncorr):
                if k_idx == n - 1:
                    continue
                if abs(A[k_idx]) > 1e-14:
                    # base[k] + A[k]*t = base[n-1] + t - t^2
                    # t^2 + (A[k]-1)*t + (base[k]-base[n-1]) = 0
                    aq = 1.0
                    bq = A[k_idx] - 1.0
                    cq = base[k_idx] - base[n - 1]
                    disc = bq * bq - 4.0 * aq * cq
                    if disc >= 0:
                        sq = np.sqrt(disc)
                        t1 = (-bq + sq) / 2.0
                        t2 = (-bq - sq) / 2.0
                        if 0.0 <= t1 <= 1.0:
                            t_candidates.append(t1)
                        if 0.0 <= t2 <= 1.0:
                            t_candidates.append(t2)

            # Evaluate f at all candidates
            t_best = old
            f_best = f_old
            for t in t_candidates:
                if t < 0 or t > 1:
                    continue
                mid = base + A * t
                mid[n - 1] = base[n - 1] + t - t * t
                f_val = np.max(mid)
                if f_val < f_best - 1e-15:
                    f_best = f_val
                    t_best = t

            # Golden section refinement around t_best
            if t_best != old and f_best < f_old - 1e-13:
                if t_best > old:
                    lo = max(0.0, min(old, t_best) - 0.05)
                    hi = min(1.0, max(old, t_best) + 0.05)
                else:
                    lo = max(0.0, min(old, t_best) - 0.05)
                    hi = min(1.0, max(old, t_best) + 0.05)
                if hi - lo < 1e-12:
                    hi = lo + 1e-6

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                def _eval_t(t):
                    mid = base + A * t
                    mid[n - 1] = base[n - 1] + t - t * t
                    return np.max(mid)

                f1 = _eval_t(c1)
                f2 = _eval_t(c2)
                for _ in range(30):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _eval_t(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = _eval_t(c2)
                t_best = (lo + hi) / 2.0
                f_best = _eval_t(t_best)

            if f_best < f_old - 1e-13:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                improved = True

        if not improved:
            break

    best_c5 = np.max(best_corr) * dx
    return best, best_c5


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def subgradient_descent(h, n, n_steps, rng):
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = compute_corr(best)
    max_corr = np.max(corr)

    for it in range(n_steps):
        if remaining_time() < 5:
            break
        step = 0.03 / (1.0 + it * 0.02)

        threshold = max_corr - 0.3
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 10:
            idx = active[np.argsort(corr[active])[::-1][:10]]
            active = idx

        grad = np.zeros(n)
        for k_idx in active:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += (1.0 - best[:npairs])
                    grad[:npairs] -= best[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= best[:npairs]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad /= gnorm

        h_new = best - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best = h_new
            corr = compute_corr(best)
            max_corr = np.max(corr)
            best_c5 = c5_new

    return best, best_c5
