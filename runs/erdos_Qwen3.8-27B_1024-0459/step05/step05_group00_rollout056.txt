    return g

        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def smooth_max_topk(h, n, beta, top_K):
    corr = compute_corr(h)
    n_corr = len(corr)
    K = min(top_K, n_corr)
    idx = np.argpartition(-corr, K)[:K]
    c_sub = corr[idx]
    
    m = c_sub.max()
    cl = np.clip(beta * (c_sub - m), -50.0, 50.0)
    ex = np.exp(cl)
    ex_sum = ex.sum()
    w = ex / ex_sum
    
    sm_val = (m + np.log(ex_sum) / beta) * (2.0 / n)
    
    g = np.zeros(n)
    for i in range(K):
        if w[i] > 1e-15:
            g += w[i] * grad_corr_k(h, idx[i], n)
    g *= (2.0 / n)
    
    return sm_val, g


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def optimize_lgbfgs(h, n, beta, top_K=50, maxiter=1500):
    mu = 1000.0
    
    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        val, _ = smooth_max_topk(hx, n, beta, top_K)
        pen = mu * (np.sum(hx) - n / 2.0) ** 2
        return val + pen
    
    def grad(x):
        hx = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_topk(hx, n, beta, top_K)
        g += 2.0 * mu * (np.sum(hx) - n / 2.0)
        at_lo = x < 1e-15
        at_hi = x > 1.0 - 1e-15
        if np.any(at_lo):
            g[at_lo] = 0.0
        if np.any(at_hi):
            g[at_hi] = 0.0
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxls': 50, 'maxcor': 20})
        return project_feasible(res.x, n)
    except Exception:
        return h


def optimize_slsqp(h, n, beta, top_K=50, maxiter=500):
    def sm_obj(x):
        val, _ = smooth_max_topk(x, n, beta, top_K)
        return val
    
    def sm_grad(x):
        _, g = smooth_max_topk(x, n, beta, top_K)
        return g
    
    def c_eq(x):
        return np.sum(x) - n / 2.0
    
    def c_eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def subgrad_descent(h, n, time_limit, rng, K=25, beta_w=30.0):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    stall = 0
    
    while time.time() - start < time_limit and remaining_time() > 5:
        it += 1
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        w = np.exp(beta_w * (c_top - c_top.max()))
        w /= w.sum()
        
        g = np.zeros(n)
        for i in range(K):
            if w[i] > 1e-12:
                g += w[i] * grad_corr_k(h, idx[i], n)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        step = 0.08 / (1.0 + 0.03 * it)
        h_new = project_feasible(h - step * g / gn, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            stall = 0
        else:
            stall += 1
            for f in [0.5, 0.2, 0.1]:
                h_try = project_feasible(h - step * f * g / gn, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    stall = 0
                    break
            if stall > 80:
                break
    
    return best_h, best_c5


def local_search(h, n, rng, n_tries=5000):
    h = h.copy()
    best_c5 = compute_c5(h)
    improved_total = 0
    
    for t in range(n_tries):
        if remaining_time() < 5:
            break
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        
        for d in [0.005, -0.005, 0.01, -0.01, 0.002, -0.002]:
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            old_i, old_j = h[i], h[j]
            h[i], h[j] = ni, nj
            c5_try = compute_c5(h)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                improved_total += 1
            else:
                h[i], h[j] = old_i, old_j
                break  # try different d for same pair
    
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
