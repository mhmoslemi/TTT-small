    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = target_sum}."""
    n = len(h)
    # Bisection on lambda
    lo = h.min() - target_sum  # ensures sum(clip(h-lo,0,1)) ≥ target_sum
    hi = h.max() + target_sum  # ensures sum(clip(h-hi,0,1)) ≤ target_sum
    
    # Actually, let's think about this more carefully
    # We want sum(clip(h_i - lambda, 0, 1)) = target_sum
    # clip(h_i - lambda, 0, 1) is a non-increasing function of lambda
    # At lambda -> -inf, clip -> 1, sum -> n
    # At lambda -> +inf, clip -> 0, sum -> 0
    # So we need lambda in [h.min()-1, h.max()]
    
    lo = h.min() - 1.0
    hi = h.max()
    
    for _ in range(60):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

```python
def slp_iteration(h, n, t, dx):
    """One SLP iteration: linearize bilinear terms and solve LP."""
    r = 2*n - 1
    # For each shift k, the constraint is:
    # sum_i h[i]*(1-h[i+k])*dx ≤ t
    # Linearized around h: 
    # sum_i [h0[i]*(1-h0[i+k]) + (1-h0[i+k])*(h[i]-h0[i]) - h0[i]*(h[i+k]-h0[i+k])] * dx ≤ t
    
    # This is: constant_k + sum_i A_k[i] * h[i] * dx ≤ t
    # where A_k[i] = (1-h0[i+k]) - h0[i-k] (with boundary handling)
    
    # Using cvxpy:
    import cvxpy as cp
    h_var = cp.Variable(n)
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n/2
    ]
    
    for k in range(r):
        off = k - (n-1)
        if off >= 0:
            np_ = n - off
            # c_k = sum_{i=0}^{np_-1} h[i+off]*(1-h[i])
            # Linearized: sum h0[i+off]*(1-h0[i]) + (1-h0[i])*(h[i+off]-h0[i+off]) - h0[i+off]*(h[i]-h0[i])
            const = np.sum(h[off:off+np_] * (1-h[:np_]))
            A = np.zeros(n)
            A[off:off+np_] += (1 - h[:np_])
            A[:np_] -= h[off:off+np_]
        else:
            ao = -off
            np_ = n - ao
            # c_k = sum_{i=0}^{np_-1} h[i]*(1-h[i+ao])
            const = np.sum(h[:np_] * (1-h[ao:ao+np_]))
            A = np.zeros(n)
            A[:np_] += (1 - h[ao:ao+np_])
            A[ao:ao+np_] -= h[:np_]
        
        constraints.append(const + A @ h_var) * dx <= t
    
    obj = cp.Minimize(t)  # Actually we're checking feasibility for given t
    prob = cp.Problem(cp.Minimize(0), constraints)  # Feasibility
    prob.solve()
    
    if h_var.value is not None:
        return h_var.value
    return h
```

The idea is to identify the top K shifts and find h that equalizes their correlation values (subject to constraints). This is a small optimization problem.

For K active shifts k₁, ..., k_K, we want:
minimize max(c_{k₁}, ..., c_{k_K})
subject to sum(h) = n/2, 0 ≤ h ≤ 1

This can be solved by:
1. Setting all c_{k_i} = t (equalization)
2. Solving the resulting system
3. Or using a small QP

Actually, for the equalization approach, we can use the following:
- Compute the gradient of each active correlation: g_k(i) = (1-h[i+k]) - h[i-k]
- We want to find a direction d such that g_k · d ≈ 0 for all active k
- This is a least-squares problem: min ||G @ d||² where G is the matrix of gradients
- Subject to: 1ᵀd = 0 (preserve sum), and d should decrease the max

Actually, a simpler approach: 
- Compute the average gradient of top K shifts: g_avg = (1/K) sum_{k in top} g_k
- Take a step in the -g_avg direction
- This pushes down the average of the top K correlations
- But it might not equalize them

A better approach:
- Find d that minimizes max_k (c_k + g_k · d) over k in top K
- This is a small LP in d (K + 2 variables, K + 3 constraints)

        g_avg -= g_avg.mean()  # preserve sum
        
        gn = np.linalg.norm(g_avg)
        if gn < 1e-14:
            break
        g_avg /= gn
        
        improved = False
        for step_size in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08, 0.015]:
            h_trial = project_box_simplex(h - step_size * g_avg, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c5_trial
                improved = True
                break
        
        if not improved:
            break
    
    return h, best_c5
