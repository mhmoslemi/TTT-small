            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
            else:
                h[i], h[j] = h[j], h[i]
                h = project_feasible(h, n)
                break
        else:
            break
    return h, best_c5


def multi_resolve_optimize(h_parent, n_parent, n_target, budget_frac, rng):
    """Upsample and optimize at target resolution."""
    h = interp_to(h_parent, n_parent, n_target)
    h = project_feasible(h, n_target)
    c5 = compute_c5(h)

    # SLSQP stages
    for beta in [500, 5000, 50000, 300000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n_target, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-14:
            c5 = c5_new
            h = h_new.copy()

    # Top-k equalization
    h, c5 = top_k_equalize(h, n_target, top_k=6, max_iters=300)

    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP improvement on parent
    h = best_h.copy()
    for beta in [5000, 50000, 200000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization on parent
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=6, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 3: Binary local search / simulated annealing on parent
    if remaining_time() > 40:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0

        sa_time = min(100, max(20, int(remaining_time() * 0.25)))
        h_b, c5_b = binary_sa(h_bin, best_n, sa_time, rng, use_multi_move=True)

        # Polish binary result continuously
        if c5_b < best_c5 + 0.01:
            h_b_c = h_b.copy()
            for beta in [5000, 50000, 300000]:
                if remaining_time() < 10:
                    break
                h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=400)
            c5_b_c = compute_c5(h_b_c)
            h_b_c, c5_b_c = top_k_equalize(h_b_c, best_n, top_k=6, max_iters=300)
            if c5_b_c < best_c5 - 1e-14:
                best_c5 = c5_b_c
                best_h = h_b_c.copy()

            # Lag-targeted refinement
            if remaining_time() > 15:
                h_lt, c5_lt = lag_targeted_refine(best_h, best_n, n_lags=10, max_rounds=100)
                if c5_lt < best_c5 - 1e-14:
                    best_c5 = c5_lt
                    best_h = h_lt.copy()

    # Phase 4: Multi-resolution: try n=512
    if remaining_time() > 60:
        n_t = 512
        h_512 = interp_to(best_h, best_n, n_t)
        h_512 = project_feasible(h_512, n_t)
        c5_512 = compute_c5(h_512)

        for beta in [2000, 20000, 200000]:
            if remaining_time() < 10:
                break
            h_512 = slsqp_opt(h_512, n_t, beta, maxiter=500)
        c5_512 = compute_c5(h_512)
        h_512, c5_512 = top_k_equalize(h_512, n_t, top_k=6, max_iters=400)

        if remaining_time() > 20:

            n_half_512 = n_t // 2
            n_ones_512 = int(np.sum(h_512_bin))
            if n_ones_512 != n_half_512:
                sort_idx = np.argsort(-h_512_bin)
                h_512_bin = np.zeros(n_t)
                h_512_bin[sort_idx[:n_half_512]] = 1.0
            sa_time_512 = min(80, max(15, int(remaining_time() * 0.3)))
            h_512_b, c5_512_b = binary_sa(h_512_bin, n_t, sa_time_512, rng, use_multi_move=True)
            h_512_b = slsqp_opt(h_512_b, n_t, 100000, maxiter=400)
            c5_512_b = compute_c5(h_512_b)
            h_512_b, c5_512_b = top_k_equalize(h_512_b, n_t, top_k=6, max_iters=300)

                h_512 = h_512_b.copy()

            best_h = h_512.copy()

    if remaining_time() > 80:
        n_t = 768
        h_768 = interp_to(best_h, best_n, n_t)
        h_768 = project_feasible(h_768, n_t)
        c5_768 = compute_c5(h_768)

        for beta in [5000, 50000]:
            if remaining_time() < 10:
                break
            h_768 = slsqp_opt(h_768, n_t, beta, maxiter=400)
        c5_768 = compute_c5(h_768)
        h_768, c5_768 = top_k_equalize(h_768, n_t, top_k=6, max_iters=250)

        if c5_768 < best_c5 - 1e-14:
            best_c5 = c5_768
            best_h = h_768.copy()
            best_n = n_t

    # Phase 6: Lag-targeted refinement on best
    if remaining_time() > 20:
        h_lt, c5_lt = lag_targeted_refine(best_h, best_n, n_lags=10, max_rounds=200)
        if c5_lt < best_c5 - 1e-14:
            best_c5 = c5_lt
            best_h = h_lt.copy()

    # Phase 7: Final intensive SLSQP
    n = best_n
    h = best_h.copy()
    for beta in [100000, 500000, 1500000, 3000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Final top-k equalization
    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(best_h, n, top_k=8, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 9: One more binary nudge if we're close
    if remaining_time() > 30 and best_c5 > 0.38085:
        h_bin = (best_h > 0.5).astype(float)
        n_half = n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:

            h_bin = np.zeros(n)

        h_b = slsqp_opt(h_b, n, 500000, maxiter=500)
        c5_b = compute_c5(h_b)
        h_b, c5_b = top_k_equalize(h_b, n, top_k=8, max_iters=200)
        if c5_b < best_c5 - 1e-14:

            best_h = h_b.copy()

    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 2000000, maxiter=600)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:

        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
