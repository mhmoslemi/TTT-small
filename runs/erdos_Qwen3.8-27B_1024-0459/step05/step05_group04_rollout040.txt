                contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for ci in contrib[:10]:
                    candidates.append((ci + off, ci))
            else:
                ao = -off
                np_ = n - ao
                contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for ci in contrib[:10]:
                    candidates.append((ci, ci + ao))
        for _ in range(60):
            candidates.append((ones[rng.randint(n_ones)], zeros[rng.randint(n_zeros)]))
        
        best_move = None
        best_new_max = cur_max + 1e-15
        seen = set()
        
        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            
            dr = swap_delta(h, i, j, n)
            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr, new_max)
        
        if best_move is None:
            for attempt in range(30):
                i = ones[rng.randint(n_ones)]
                j = zeros[rng.randint(n_zeros)]
                if h[i] != 1 or h[j] != 0:
                    continue
                dr = swap_delta(h, i, j, n)
                new_corr = corr + dr
                new_max = float(np.max(new_corr))
                delta = new_max - cur_max
                if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-12)):
                    best_move = (i, j, dr, new_max)
                    break
        
        if best_move is not None:
            i, j, dr, new_max = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 800:
            for _ in range(8):
                i = ones[rng.randint(n_ones)]
                j = zeros[rng.randint(n_zeros)]
                if h[i] == 1 and h[j] == 0:
                    dr = swap_delta(h, i, j, n)
                    h[i] = 0
                    h[j] = 1
                    corr = corr + dr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_max = float(np.max(corr))
            stall = 0
            t = max(t, 0.05)
    
    return best_h, best_max * dx

def ilsa_binary(h_bin, n, time_limit=200, rng=None):
    """Iterated Local Search with Annealing."""
    if rng is None:
        rng = np.random.RandomState(42)
    best_h, best_c5 = binary_sa(h_bin, n, time_limit=min(60, time_limit * 0.3), rng=rng)
    
    while remaining() > 30 and (time_limit - (time.time() - time.time())) < time_limit:
        if remaining() < 30:
            break
        elapsed_so_far = 60
        rem_time = time_limit - elapsed_so_far
        if rem_time < 10:
            break
        
        # Perturb best solution
        h_pert = best_h.copy()
        n_pert = rng.randint(3, 8)
        for _ in range(n_pert):
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0
            h_pert[j] = 1
        
        t_sub = min(40, int(remaining() * 0.25))
        if t_sub < 10:
            break
        
        h_new, c5_new = binary_sa(h_pert, n, time_limit=t_sub, rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def lse_obj(x, beta, n):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = beta * (corr - m)
    cl = np.clip(sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        return m * (2.0 / n)
    return (m + np.log(s) / beta) * (2.0 / n)

def lse_grad(x, beta, n):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = beta * (corr - m)
    cl = np.clip(sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w = ex / s
    g = np.zeros(n)
    wc = w * (2.0 / n)
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

def slsqp_refine(h, n, beta, maxiter=300):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    _n = n
    _beta = beta
    _tgt = n / 2.0
    
    def obj_fn(x):
        return lse_obj(x, _beta, _n)
    
    def grad_fn(x):
        return lse_grad(x, _beta, _n)
    
    def cons_fn(x):
        return np.sum(x) - _tgt
    
    def cons_jac(x):
        return np.ones(_n)
    
    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}]
    
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.success or (hasattr(res, 'fun') and res.fun < obj_fn(h)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    
    # Fallback: L-BFGS-B with penalty
    try:
        mu = 5000.0
        def obj_pen(x):
            base = lse_obj(x, _beta, _n)
            penalty = mu * (np.sum(x) - _tgt) ** 2
            return base + penalty
        
        def grad_pen(x):
            base_g = lse_grad(x, _beta, _n)
            pen_g = 2.0 * mu * (np.sum(x) - _tgt) * np.ones(_n)
            return base_g + pen_g
        
        res2 = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-8})
        h_res = project_feasible(res2.x, n)
        if compute_c5(h_res) < compute_c5(h):
            return h_res
    except Exception:
        pass
    
    return h

def projected_grad_descent(h, n, beta, max_iters=500, lr=0.01):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        g = lse_grad(h, beta, n)
        # Project gradient to satisfy sum constraint: subtract mean
        g = g - np.mean(g)
        
        # Line search
        improved = False
        for step in [lr, lr*0.5, lr*0.1, lr*2, lr*0.05]:
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.001, 0.005, 0.002, 0.0005]:
                h_new = h - step * g
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                break
        # Adaptive learning rate
        if improved:
            lr = min(lr * 1.1, 0.1)
        else:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    
    # Phase 1: ILS binary search at parent resolution
    h_bin = binary_round(best_h, n)
    t1 = min(350, int(remaining() * 0.35))
    if t1 > 30:
        h_b, c5_b = ilsa_binary(h_bin, n, time_limit=t1, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
    
    # Phase 2: Other starting points at same resolution
    starts = [
        ('tm', thue_morse(n)),
        ('spec', spectral_init(n, rng)),
        ('rand', random_binary(n, rng)),
    ]
    for name, sp in starts:
        if remaining() < 50:
            break
        t2 = min(150, int(remaining() * 0.15))
        if t2 < 20:
            break
        h_b, c5_b = ilsa_binary(sp, n, time_limit=t2, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
    
    # Phase 3: Try other resolutions
    resolutions = []
    for mult in [0.75, 0.875, 1.125, 1.25, 1.5]:
        r = int(n * mult)
        if r % 2 != 0:
            r += 1
        if 192 <= r <= 768:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))
    resolutions.sort(key=lambda x: abs(x - n))
    
    for n_t in resolutions[:5]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = binary_round(h_interp, n_t)
        t3 = min(80, int(remaining() * 0.1))
        if t3 < 15:
            break
        h_b, c5_b = binary_sa(h_bin_t, n_t, time_limit=t3, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_t
    
    # Phase 4: Continuous refinement at best resolution
    n = best_n
    h = project_feasible(best_h, n)
    
    for beta in [50, 200, 1000, 5000, 20000]:
        if remaining() < 20:
            break
        h_new = slsqp_refine(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Projected gradient descent polish
    if remaining() > 30:
        h_pg, c5_pg = projected_grad_descent(h, n, beta=20000, max_iters=1000, lr=0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    
    # Phase 6: Higher beta SLSQP
    for beta in [50000, 100000, 200000]:
        if remaining() < 12:
            break
        h_new = slsqp_refine(best_h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 7: Try finer resolution
    if remaining() > 80:
        n_fine = n + 64
        if n_fine % 2 != 0:
            n_fine += 1
        h_fine = interp_to(best_h, n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        for beta in [2000, 10000, 50000]:
            if remaining() < 20:
                break
            h_fine = slsqp_refine(h_fine, n_fine, beta, maxiter=150)
        c5_fine = compute_c5(h_fine)
        if remaining() > 20:
            h_fine_pg, c5_fine_pg = projected_grad_descent(h_fine, n_fine, beta=50000, max_iters=300, lr=0.01)
            c5_fine = min(c5_fine, c5_fine_pg)
            if c5_fine_pg < c5_fine:
                h_fine = h_fine_pg
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine
    
    # Phase 8: Final SLSQP polish
    n = best_n
    h = project_feasible(best_h, n)
    for beta in [100000, 300000]:
        if remaining() < 8:
            break
        h_new = slsqp_refine(h, n, beta, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Final: ensure feasibility and compute exact C5
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
