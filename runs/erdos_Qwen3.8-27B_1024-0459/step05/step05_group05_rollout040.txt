def compute_c5(h):

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse(n):

def m_sequence(n):
    """Generate an m-sequence (maximal length LFSR) of length n."""
    # Find a primitive polynomial of degree d where 2^d - 1 >= n
    # For simplicity, use a known set
    primitive_polys = {
        8: 0x47, 9: 0x11D, 10: 0x12D, 11: 0x169, 12: 0x1D9,
        13: 0x1BD, 14: 0x19D, 15: 0x121, 16: 0x1C4, 17: 0x169,
        18: 0x1F5, 19: 0x1C3, 20: 0x165, 21: 0x18D, 22: 0x163,
        23: 0x189, 24: 0x15C, 25: 0x153, 26: 0x16E, 27: 0x16B,
        28: 0x12F, 29: 0x133, 30: 0x11F, 31: 0x183
    }
    d = max(2, int(np.ceil(np.log2(n + 1))))
    if d > 31:
        d = 31
    if d not in primitive_polys:
        # Fallback to random
        h = np.zeros(n)
        h[:n // 2] = 1.0
        np.random.shuffle(h)
        return h
    ppoly = primitive_polys[d]
    reg = 1  # initial state
    seq = []
    mlen = (1 << d) - 1
    for _ in range(mlen):
        seq.append(1 if reg & 1 else 0)
        lsb = reg & 1
        reg >>= 1
        if lsb:
            reg ^= (ppoly >> 1)
    h = np.array(seq[:n], dtype=float)
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            h[ones_idx[:n_ones - n_half]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            h[zeros_idx[:n_half - n_ones]] = 1
    return h


def interp_to(src_h, src_n, dst_n):

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n

                    i = ones[rng.randint(len(ones))]

                    h[i] = 0
                    h[j] = 1

                if best_swap is not None:
                    i, j = best_swap
                    h[i] = 0
                    h[j] = 1

                    no_improve = 0
                    if cur_max < best_max:

                        best_h = h.copy()
                else:

        # Single random swap with SA acceptance

            cur_max = nc

def slsqp_smooth(h, n, beta, maxiter=400):
    """Smoothed max optimization with SLSQP."""
    h = project_feasible(h, n)
    dx_v = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx_v
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

        h_r = rng.uniform(0, 1, n_base)
        h_r = project_feasible(h_r, n_base)
        h_rb = binarize(h_r, n_base)
        h_sd4, c5_sd4 = steepest_binary_descent(h_rb, n_base, rng, time_limit=min(15, remaining() * 0.08))
        if c5_sd4 < best_c5:
            best_c5 = c5_sd4
            best_h = h_sd4.copy()
    
    # Phase 2: Multi-resolution binary search
    # Try nearby resolutions, looking for ones where the binary grid gives better C5
    n_candidates = list(range(max(380, n_base - 10), min(520, n_base + 80), 2))
    # Ensure n_base is included
    if n_base % 2 != 0:
        n_candidates.append(n_base)
    n_candidates = sorted(set(n_candidates))
    
    # Sort by potential (smaller n gives coarser grid but might have better integer ratio)
    # Priority: resolutions where the "natural" max_correlation * dx is close to or below target
    for n_t in n_candidates:
        if remaining() < 8:
            break
        if n_t <= 0 or n_t % 2 != 0:
            continue
        
        # Interpolate best solution to new resolution and binarize
        h_t = interp_to(best_h, best_n, n_t)
        h_tb = binarize(h_t, n_t)
        c5_t0 = compute_c5(h_tb)
        
        # Quick steepest descent
        h_td, c5_td = steepest_binary_descent(h_tb, n_t, rng, time_limit=min(15, remaining() * 0.05))
        c5_t = min(c5_td, c5_t0)
        h_best_t = h_td if c5_td <= c5_t0 else h_tb
        
        # If promising, do SA
        if c5_t < best_c5 + 0.001 and remaining() > 10:
            h_ts, c5_ts = binary_sa_search(h_best_t, n_t, rng, time_limit=min(25, remaining() * 0.08))
            if c5_ts < c5_t:
                c5_t = c5_ts
                h_best_t = h_ts
        
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_best_t.copy()
            best_n = n_t
    
    # Phase 3: Continuous refinement of best binary solution
    n = best_n
    h = best_h.copy()
    
    if remaining() > 15:
        # Try SLSQP from the binary solution
        for beta in [2000, 10000, 30000]:
            if remaining() < 8:
                break
            h_new = slsqp_smooth(h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    # Phase 4: Further binary refinement at best resolution
    if remaining() > 20:
        h_bin2 = binarize(best_h, best_n)
        # Multiple SA restarts
        for restart in range(3):
            if remaining() < 8:
                break
            perturbed = h_bin2.copy()
            o = np.where(perturbed == 1)[0]
            z = np.where(perturbed == 0)[0]
            for _ in range(rng.randint(1, 4)):
                a = o[rng.randint(len(o))]
                b = z[rng.randint(len(z))]
                perturbed[a] = 0
                perturbed[b] = 1
                o = np.where(perturbed == 1)[0]
                z = np.where(perturbed == 0)[0]
            h_r2, c5_r2 = binary_sa_search(perturbed, best_n, rng, time_limit=min(20, remaining() * 0.1))
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
            h_sd_r, c5_sd_r = steepest_binary_descent(best_h, best_n, rng, time_limit=min(10, remaining() * 0.05))
            if c5_sd_r < best_c5:
                best_c5 = c5_sd_r
                best_h = h_sd_r.copy()
    
    # Phase 5: Final continuous polish
    if remaining() > 10:
        for beta in [5000, 20000, 50000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(best_h, best_n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 6: Try a few more resolutions with the best solution
    if remaining() > 20:
        for delta in [-4, -2, 2, 4, 6, 8]:
            if remaining() < 6:
                break
            n_t = best_n + delta
            if n_t < 128 or n_t > 700 or n_t % 2 != 0:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_tb = binarize(h_t, n_t)
            c5_t0 = compute_c5(h_tb)
            h_td, c5_td = steepest_binary_descent(h_tb, n_t, rng, time_limit=min(10, remaining() * 0.04))
            c5_t = min(c5_td, c5_t0)
            h_best_t = h_td if c5_td <= c5_t0 else h_tb
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_best_t.copy()
                best_n = n_t
                # Quick continuous refinement
                if remaining() > 5:
                    h_c = slsqp_smooth(best_h, best_n, 10000, 150)
                    c5_c = compute_c5(h_c)
                    if c5_c < best_c5:
                        best_c5 = c5_c
                        best_h = h_c.copy()
    
    # Final validation
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h
    
    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
