def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Fix rounding
    total = np.sum(h)
    if abs(total - target) > 1e-10:
        # Force exact
        h = np.round(h * n) / n  # This might not work well
        # Better: adjust one element
        diff = target - np.sum(h)
        if diff > 0:
            # Need to add diff to some element
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    return corr_grad_weighted(h, w * dx)

def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g

def lbfgs_run(h, n, beta, maxiter=1000):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        return smooth_obj(x, beta)
    
    def _grad(x):
        return smooth_grad(x, beta)
    
    def _sc(x):
        return np.sum(x) - n / 2.0
    
    def _scj(x):
        return np.ones(n)
    
    try:
        res = minimize(_obj, h0, jac=_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-10})
        # L-BFGS-B doesn't handle equality constraints well
        # Project back
        h_res = project_feasible(res.x, n)
        if compute_c5(h_res) < compute_c5(h0):
            return h_res
    except Exception:
        pass
    return h0

def slsqp_run(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        return smooth_obj(x, beta)
    
    def _grad(x):
        return smooth_grad(x, beta)
    
    def _sc(x):
        return np.sum(x) - n / 2.0
    
    def _scj(x):
        return np.ones(n)
    
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < smooth_obj(h0, beta) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

Wait, the targeted binary LS above is O(n * 20 * 20 * n) = O(400n²) per round, which for n=384 is about 59 million operations per round. That's too slow for multiple rounds.

Let me optimize. The key insight is that for a given swap (pa, pb), the delta vector has a very specific structure:
delta[d] = (h[pa-d] if valid else 0) - (h[pa+d] if valid else 0) - (h[pb-d] if valid else 0) + (h[pb+d] if valid else 0)

This is the sum of at most 4 "spike" vectors. Specifically:
- If pa-d is valid, delta[pa-d] += h[pa-d] = 1 (since h is binary and we're evaluating at specific indices)
- Wait, no. delta[d] = h[pa-d] - h[pa+d] - h[pb-d] + h[pb+d]

This means delta is non-zero only at 4 specific positions:
- d = pa - (pa-d) ... no, that's not right.

Let me re-derive. The correlation is:
corr[d] = Σ_i h[i] * (1 - h[i-d])

When we change h[pa] from 1 to 0 and h[pb] from 0 to 1, the change in corr[d] is:
Δcorr[d] = Σ_i Δh[i] * (1-h[i-d]) + h[i] * Δ(1-h[i-d])

Where Δh[pa] = -1, Δh[pb] = +1, and Δ(1-h[i-d]) is non-zero when i-d = pa or i-d = pb, i.e., i = pa+d or i = pb+d.

So:
Δcorr[d] = -1 * (1-h[pa-d]) + 1 * (1-h[pb-d]) + h[pa+d] * (-1) + h[pb+d] * 1
= -(1-h[pa-d]) + (1-h[pb-d]) - h[pa+d] + h[pb+d]

For binary h:
= -1 + h[pa-d] + 1 - h[pb-d] - h[pa+d] + h[pb+d]
= h[pa-d] - h[pb-d] - h[pa+d] + h[pb+d]

Now, this is a function of d. For a fixed (pa, pb), this is:
- h[pa-d]: non-zero (equals 1) when pa-d is a valid index where h=1
- -h[pb-d]: non-zero (equals -1) when pb-d is a valid index where h=1
- -h[pa+d]: non-zero (equals -1) when pa+d is a valid index where h=1
- h[pb+d]: non-zero (equals 1) when pb+d is a valid index where h=1

So Δcorr[d] is non-zero only at d values where one of these conditions holds. Since h is binary with n/2 ones, there are about n/2 such d values for each term, so Δcorr has about 2n non-zero entries.

But computing it for all d is still O(n) per swap. The question is whether we can do it faster.

Alternative: precompute for each position i the vector v_i where v_i[d] = h[i-d] - h[i+d] (with boundary handling). Then Δcorr = v_pa - v_pb (element-wise).

But storing all v_i would be O(n²) memory, which for n=384 is about 147K floats = 1.2 MB. That's fine!

Actually, v_i[d] = h[i-d] - h[i+d] depends on i and d, so it's a n×(2n-1) matrix. For n=384, that's 384 × 767 ≈ 295K floats ≈ 2.4 MB. Fine.

But computing it is O(n²) which is fine.

Then for any swap (pa, pb): Δcorr = v_pa - v_pb (element-wise, O(n) operation).

So the algorithm becomes:
1. Precompute v[i] for all i (O(n²) total)
2. For each candidate swap (pa, pb): compute Δcorr = v[pa] - v[pb] (O(n))
3. new_max = max(corr + Δcorr) (O(n))

So each swap evaluation is O(n), and with O(n²) swaps, total is O(n³).

For n=384: 384³ ≈ 56 million. At ~100M operations/second in Python with numpy, that's about 0.5 seconds per full pass. That's too slow for a full O(n²) search.

But we can be smarter:
- Only consider O(n) candidate swaps (the top ones by some heuristic)
- Or use a "greedy" approach: at each step, find the single best swap and apply it

For the greedy approach:
1. Compute v[i] for all i
2. For the current worst shift d*, find the pair (pa, pb) that most reduces corr[d*]
   - We want to minimize v[pa][d*_idx] - v[pb][d*_idx] (where d*_idx is the index in the correlation array)
   - Wait, that's just (v[pa] - v[pb])[d*_idx], which we can compute as v[pa][d*_idx] - v[pb][d*_idx]
   - So we want to minimize v[pa][d*_idx] (minimize over pa in ones) and minimize -v[pb][d*_idx] (i.e., maximize v[pb][d*_idx] over pb in zeros)
   
   Wait: Δcorr[d*] = (v[pa] - v[pb])[d*_idx] = v[pa][d*_idx] - v[pb][d*_idx]
   To minimize this: minimize v[pa][d*_idx] and maximize v[pb][d*_idx]
   
   So: best_pa = argmin_{pa in ones} v[pa][d*_idx]
        best_pb = argmax_{pb in zeros} v[pb][d*_idx]

3. Apply the swap, update corr, repeat.

This is O(n) per swap (to compute v[pa][d*_idx] for all pa, and v[pb][d*_idx] for all pb), and we do O(n) swaps, so total O(n²). 

But wait, after each swap, v changes (because h changed). So we need to update v.

When we swap pa and pb:

- That is, v[i] changes only if i is within distance n-1 of pa or pb, which is always true.

  - i+d = pb → d = pb-i: v[i][pb-i] changes

So for each i, at most 4 entries of v[i] change per swap. Updating v after a swap is O(n) (for all i, update at most 4 positions each).

Key optimization: For a given shift d, the delta at that shift for swap (pa, pb) is:
Δ(d) = h[pa-d] - h[pb-d] - h[pa+d] + h[pb+d]

I can precompute for each position i:
- left[i] = h[i-d] if 0≤i-d<n else 0
- right[i] = h[i+d] if 0≤i+d<n else 0

Then Δ(d) = left[pa] - right[pa] - left[pb] + right[pb] = (left[pa] - right[pa]) + (right[pb] - left[pb])

So for a given d:
- Score for pa (to minimize Δ): minimize (left[pa] - right[pa])
- Score for pb (to minimize Δ): minimize (right[pb] - left[pb]) = maximize (left[pb] - right[pb])

This is O(n) to compute for all positions given d.

                val = 0.0
                if 0 <= pb - d < n:
                    val -= h[pb - d]
                if 0 <= pb + d < n:
                    val += h[pb + d]
                pb_score[ii] = val
            
            # We want to minimize pa_score[pa_idx] + pb_score[pb_idx]
            # Get top candidates
            best_pa_idx = np.argsort(pa_score)[:min(candidates, len(ones))]
            best_pb_idx = np.argsort(pb_score)[:min(candidates, len(zeros))]
            
            for pa_i in best_pa_idx:
                pa = ones[pa_i]
                for pb_i in best_pb_idx:
                    pb = zeros[pb_i]
                    # Compute full delta
                    delta = np.zeros(2 * n - 1)
                    for ds in range(-(n-1), n):
                        val = 0.0
                        if 0 <= pa - ds < n:
                            val += h[pa - ds]
                        if 0 <= pa + ds < n:
                            val -= h[pa + ds]
                        if 0 <= pb - ds < n:
                            val -= h[pb - ds]
                        if 0 <= pb + ds < n:
                            val += h[pb + ds]
                        delta[ds + n - 1] = val
                    
                    new_corr = corr + delta
                    new_max = np.max(new_corr)
                    improve = np.max(corr) - new_max
                    
                    if improve > best_improve:
                        best_improve = improve
                        best_pa = pa
                        best_pb = pb
                        best_new_corr = new_corr
        
        if best_pa < 0:
            break
        
        h[best_pa] = 0.0
        h[best_pb] = 1.0
        corr = best_new_corr.copy()
        curr_c5 = float(np.max(corr)) * 2.0 / n
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
    
    return best_h, best_c5
