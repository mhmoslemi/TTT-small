import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g


def smooth_max_grad(h, n, beta, n_top=80):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    total = np.sum(ex)
    w = ex / total
    sm_val = (m + np.log(total) / beta) * (2.0 / n)
    g = np.zeros(n)
    thresh = 1e-10
    active = w > thresh
    for kidx in np.where(active)[0]:
        wk = w[kidx] * (2.0 / n)
        if wk < 1e-14:
            continue
        g += wk * grad_corr_k(h, kidx, n)
    return sm_val, g, m


def nesterov_sgd(h, n, time_limit, rng, lr_init=0.015, decay=0.9997, mom_init=0.9):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    lr = lr_init
    mom = mom_init
    start = time.time()
    stuck = 0

    n_iter = max(200, int(time_limit * 40))
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        progress = min(1.0, it / max(n_iter, 1))
        beta = 30.0 + 200.0 * progress ** 2

        _, g, _ = smooth_max_grad(h, n, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn

        # Nesterov: evaluate at y = h + mom * v
        y = h + mom * v
        _, g_y, _ = smooth_max_grad(y, n, beta)
        g_yn = np.linalg.norm(g_y)
        if g_yn < 1e-15:
            break
        g_y = g_y / g_yn

        v = mom * v - lr * g_y

def slsqp_smooth(h, n, beta, maxiter=400):
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        thresh = 1e-12
        for kidx in np.where(w > thresh)[0]:
            wk = w[kidx] * (2.0 / n)
            if wk < 1e-14:
                continue
            g += wk * grad_corr_k(x, kidx, n)
        return g

    def c_fun(x):
        return np.sum(x) - n / 2.0

    def c_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and not np.any(np.isnan(res.x)):
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h

            break

        idx = rng.choice(n, size=ncoord, replace=False)
        improved = False

        for j in idx:
            old_val = h[j]
            candidates = [0.0, 1.0, 0.5]
            for d in [0.08, -0.08, 0.04, -0.04, 0.02, -0.02, 0.01, -0.01]:
                candidates.append(old_val + d)
            candidates = [c for c in candidates if 0.0 <= c <= 1.0]

            best_local = old_val

                    best_local = cv

def structured_perturb(h, n, rng, scale=0.04):
