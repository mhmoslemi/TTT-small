        for step in [0.05, 0.02, 0.1, 0.01, 0.2, 0.03, 0.005, 0.15]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            stall += 1
            if stall > 100:
                break
        else:
            stall = 0
    return best_h, best_c5

def binary_sa(h_bin, n, time_limit=100, rng=None, T_init=5.0, T_final=0.01):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_bin > 0.5).copy().astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            to_rm = rng.choice(ones, n_ones - n_half, replace=False)
            h[to_rm] = 0
        else:
            zeros = np.where(h == 0)[0]
            to_add = rng.choice(zeros, n_half - n_ones, replace=False)
            h[to_add] = 1
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    r = 2 * n - 1
    start = time.time()
    total_iters = 5000
    iter_count = 0
    stall = 0
    max_stall = 200
    while time.time() - start < time_limit and remaining() > 10 and iter_count < total_iters:
        iter_count += 1
        T = T_init * (T_final / T_init) ** (iter_count / max(1, total_iters))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Generate candidates
        candidates = []
        top_k = np.argsort(-corr)[:10]
        for s in top_k[:6]:
            off = s - (n - 1)
            if off >= 0:
                np_ = n - off
                valid = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for ci in valid[:20]:
                    candidates.append((ci + off, ci))
            else:
                ao = -off
                np_ = n - ao
                valid = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for ci in valid[:20]:
                    candidates.append((ci, ci + ao))
        # Random candidates
        for _ in range(40):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        best_move = None
        best_new_max = cur_max
        seen = set()
        accepted_lose = None
        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            # Compute delta corr
            dr = np.zeros(r)
            lo_i = i
            hi_i = min(i + n - 1, r - 1)
            if lo_i <= hi_i:
                idx = np.arange(0, hi_i - lo_i + 1)
                dr[lo_i:hi_i+1] += -(1.0 - 2.0 * h[idx])
            lo_j = j
            hi_j = min(j + n - 1, r - 1)
            if lo_j <= hi_j:
                idx2 = np.arange(0, hi_j - lo_j + 1)
                dr[lo_j:hi_j+1] += (1.0 - 2.0 * h[idx2])
            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr)
            elif new_max > best_new_max and T > 0.02:
                delta = new_max - best_new_max
                if rng.rand() < np.exp(-delta / max(T, 1e-10)):
                    if accepted_lose is None or new_max < accepted_lose[1]:
                        accepted_lose = (i, j, dr, new_max)
        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        elif accepted_lose is not None:
            i, j, dr, nm = accepted_lose
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = nm
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                # Kick: 3 random swaps
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        dr = np.zeros(r)
                        hi_i = min(i + n - 1, r - 1)
                        idx = np.arange(0, hi_i - i + 1)
                        dr[i:hi_i+1] += -(1.0 - 2.0 * h[idx])
                        hi_j = min(j + n - 1, r - 1)
                        idx2 = np.arange(0, hi_j - j + 1)
                        dr[j:hi_j+1] += (1.0 - 2.0 * h[idx2])
                        h[i] = 0
                        h[j] = 1
                        corr = corr + dr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = float(np.max(corr))
                stall = 0
    return best_h, best_max * dx

def slsqp_smooth(h, n, beta, maxiter=500):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sq_fun, 'jac': _sq_jac, 'args': (n,)}]

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        g = 1.0 - 2.0 * x
        return np.correlate(w, g, mode='valid') * (2.0 / n)

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def _sq_fun(x, n):
    return np.sum(x) - n / 2.0

def _sq_jac(x, n):
    return np.ones(n)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        if n_ones > n_half:
            idx = np.argsort(-h)
            h_bin = np.zeros(n)
            h_bin[idx[:n_half]] = 1.0
        else:
            # Use fractional parts to decide
            sorted_idx = np.argsort(-h)
            h_bin = np.zeros(n)
            h_bin[sorted_idx[:n_half]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        # Generate Thue-Morse based initial
        parent_h = np.zeros(parent_n)
        for i in range(parent_n):
            c = bin(i).count('1') % 2
            parent_h[i] = 1.0 if c == 0 else 0.0
        n_o = int(np.sum(parent_h))
        if n_o != parent_n // 2:
            sort_idx = np.argsort(-parent_h)
            parent_h = np.zeros(parent_n)
            parent_h[sort_idx[:parent_n // 2]] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary SA at parent resolution
    n = best_n
    h_bin = make_binary(best_h, n)
    t1 = min(250, int(remaining() * 0.25))
    if t1 > 20:
        h_b1, c5_b1 = binary_sa(h_bin, n, time_limit=t1, rng=rng, T_init=8.0, T_final=0.005)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n
            n = n

    # Phase 2: Projected gradient at parent resolution
    if remaining() > 60:
        t2 = min(120, int(remaining() * 0.15))
        h_pg, c5_pg = proj_gradient_opt(best_h, best_n, beta_start=30.0, beta_end=30000.0, time_limit=t2)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: Top-K descent
    if remaining() > 40:
        t3 = min(60, int(remaining() * 0.1))
        h_top, c5_top = top_k_descent(best_h, best_n, K=10, time_limit=t3)
        if c5_top < best_c5:
            best_c5 = c5_top
            best_h = h_top.copy()

    # Phase 4: More binary SA (second round with different T schedule)
    if remaining() > 80:
        h_bin2 = make_binary(best_h, best_n)
        rng2 = np.random.RandomState(seed + 1000)
        t4 = min(200, int(remaining() * 0.2))
        h_b2, c5_b2 = binary_sa(h_bin2, best_n, time_limit=t4, rng=rng2, T_init=3.0, T_final=0.002)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 5: More projected gradient (higher beta range)
    if remaining() > 50:
        t5 = min(100, int(remaining() * 0.12))
        h_pg2, c5_pg2 = proj_gradient_opt(best_h, best_n, beta_start=100.0, beta_end=100000.0, time_limit=t5)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()

    # Phase 6: Try higher resolution
    if remaining() > 100:
        n_fine = best_n + 128
        h_fine = interp_to(best_h, best_n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        h_fin_bin = make_binary(h_fine, n_fine)
        rng3 = np.random.RandomState(seed + 2000)
        t6a = min(120, int(remaining() * 0.12))
        h_fb, c5_fb = binary_sa(h_fin_bin, n_fine, time_limit=t6a, rng=rng3, T_init=5.0, T_final=0.01)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()
            best_n = n_fine
        if remaining() > 50:
            t6b = min(60, int(remaining() * 0.08))
            h_fp, c5_fp = proj_gradient_opt(best_h, best_n, beta_start=50.0, beta_end=50000.0, time_limit=t6b)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Phase 7: SLSQP polish at best resolution
    if remaining() > 60:
        n = best_n
        for beta in [5000, 20000, 80000]:
            if remaining() < 30:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 8: Final top-K descent
    if remaining() > 30:
        h_top2, c5_top2 = top_k_descent(best_h, best_n, K=12, time_limit=min(50, int(remaining() * 0.08)))
        if c5_top2 < best_c5:
            best_c5 = c5_top2
            best_h = h_top2.copy()

    # Phase 9: Final projected gradient
    if remaining() > 20:
        h_fpg, c5_fpg = proj_gradient_opt(best_h, best_n, beta_start=500.0, beta_end=200000.0, time_limit=min(40, int(remaining() * 0.06)))
        if c5_fpg < best_c5:
            best_c5 = c5_fpg
            best_h = h_fpg.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
