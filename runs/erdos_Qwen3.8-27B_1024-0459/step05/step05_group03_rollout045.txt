    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = np.array(h, float)
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:n//2]] = 1.0
    return out

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    w = ex / np.sum(ex)
    dx = 2.0 / n
    w = w * dx
    # Compute gradient of weighted sum of correlations
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            # corr[k] = sum_{i=0}^{np_-1} h[i] * (1 - h[i+off])
            # d/dh[i] for i < np_: (1 - h[i+off])
            # d/dh[i] for i >= off: -h[i-off]
            g[:np_] += wk * (1.0 - h[off:off + np_])
            g[off:off + np_] -= wk * h[:np_]
        else:
            a = -off
            np_ = n - a
            # corr[k] = sum_{i=0}^{np_-1} h[i+a] * (1 - h[i])
            g[a:a+np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[a:a+np_]
    return g

def slsqp_run(h0, n, beta, maxiter=400):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        return smooth_obj(x, beta)
    
    def _grad(x):
        return smooth_grad(x, beta)
    
    def _sc(x):
        return np.sum(x) - n / 2.0
    
    def _scj(x):
        return np.ones(n)
    
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def sim_anneal_binary(h_bin, n, max_iters=50000, T0=1.0, T1=0.01):
    """Simulated annealing on binary sequences with pair-swap moves."""
    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    curr_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    n_ones = n // 2
    n_zeros = n - n_ones
    
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if remaining() < 10:
            break
        T = T0 * (T1 / T0) ** (it / max_iters)
        
        # Choose a random 1 and a random 0 to swap
        ia = rng.randint(n_ones)
        ib = rng.randint(n_zeros)
        pa = ones[ia]
        pb = zeros[ib]
        
        # Perform swap
        h[pa] = 0.0
        h[pb] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / (T * dx)):
            # Accept
            curr_c5 = new_c5
            # Update ones and zeros arrays
            ones[ia] = pb
            zeros[ib] = pa
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Reject, undo
            h[pa] = 1.
