def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Focused coordinate descent (active lags)
    if remaining_time() > 40:
        h_new, c5_new = coord_descent_focused(h, n, n_sweeps=150, focus_radius=0.01)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: More SLSQP
    if remaining_time() > 30:
        h_new = slsqp_optimize(h, n, 500000, 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Heavy coordinate descent
    if remaining_time() > 50:
        h_new, c5_new = coord_descent(h, n, n_sweeps=120)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Try nearby resolutions
    nearby_res = []
    for delta in [-4, -2, 2, 4, 6, 8, 12, 16]:
        rn = best_n + delta
        if 64 <= rn <= 2000 and rn % 2 == 0:
            nearby_res.append(rn)
    nearby_res.sort(key=lambda x: abs(x - best_n))

    for rn in nearby_res[:4]:
        if remaining_time() < 50:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        h_r = slsqp_optimize(h_r, rn, 50000, 250)
        if remaining_time() > 20:
            h_r, c5_r = coord_descent(h_r, rn, n_sweeps=40)

        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 8: Final SLSQP polish
    n = best_n
    h = best_h.copy()
    if remaining_time() > 20:
        h_new = slsqp_optimize(h, n, 2000000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Final coordinate descent rounds
    while remaining_time() > 15:
        h_new, c5_new = coord_descent(h, best_n, n_sweeps=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Try focused descent
            h_new, c5_new = coord_descent_focused(h, best_n, n_sweeps=80, focus_radius=0.005)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            else:
                break

    # Phase 10: Random micro-perturbations
    n = best_n
    h = best_h.copy()
    n_perturb = max(3, n // 100)
    for pert in range(20):
        if remaining_time() < 8:
            break
        h_pert = h.copy()
        idx = rng.choice(n, n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.02, 0.02, n_pert)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.001:
            h_new, c5_new = coord_descent(h_pert, n, n_sweeps=30)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Final validation and cleanup
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
            final_c5 = compute_c5(best_h)

    # One more coordinate descent pass if time allows
    if remaining_time() > 5:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=20)
        if c5_new < final_c5:
            best_h = h_new
            final_c5 = c5_new

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
