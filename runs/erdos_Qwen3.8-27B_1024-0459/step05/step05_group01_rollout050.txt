    """Find the minimum norm convex combination of gradients of active lags."""
    grads = []
    for k in active_lags:
        off = k - (n - 1)
        g = np.zeros(n)
        if off >= 0:
            np_ = n - off
            g[off:] += (1.0 - h[:np_])
            g[:np_] -= h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += (1.0 - h[a:])
            g[a:] -= h[:np_]
        grads.append(g)
    G = np.array(grads)  # shape: (m, n)
    # Find weights w >= 0, sum w = 1, minimizing ||G^T w||^2
    # This is: min ||G^T w||^2 s.t. sum w = 1, w >= 0
    # Using scipy: this is a QP
    # ||G^T w||^2 = w^T G G^T w
    # So we minimize w^T (G G^T) w s.t. sum w = 1, w >= 0
    m = len(active_lags)
    if m == 1:
        return -grads[0]
    GGt = G @ G.T
    # Solve the QP using a simple projected gradient or scipy
    w = np.ones(m) / m  # uniform weights
    for _ in range(100):
