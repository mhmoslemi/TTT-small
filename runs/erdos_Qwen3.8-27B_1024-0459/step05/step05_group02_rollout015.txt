    """Generate binary h using golden ratio fencing."""
    h = np.zeros(n)
    phi = (1 + 5**0.5) / 2
    alpha = 1.0 / (phi * phi)  # 1/phi^2 ≈ 0.382

def proj_cont(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def make_golden(n, alpha=None):
    if alpha is None:
        alpha = 1.0 / (1.0 + 5**0.5) ** 2
    pos = np.arange(n)
    frac = (pos * alpha) % 1.0
    h = (frac < 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(h))
    if diff > 0:
        zeros = np.where(h == 0)[0]
        for k in range(diff):
            idx = zeros[np.argmax(frac[zeros[k:]] - 0.5) + k]
            # Actually, pick the zero with frac closest to 0.5 from above
            pass
        # Simpler: add ones at positions with largest frac among zeros
        zero_idx = np.where(h == 0)[0]
        add_order = zero_idx[np.argsort(-frac[zero_idx])]
        h[add_order[:diff]] = 1.0
    elif diff < 0:
        one_idx = np.where(h == 1)[0]
        rem_order = one_idx[np.argsort(frac[one_idx])]
        h[rem_order[:(-diff)]] = 0.0
    return h

def interpolate_h(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def _sm_obj(x):
    n = _N_G
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(-corr)[:_TOPK_G]
    c = corr[idx]
    m = c[0]
    s = _BETA_G * (c - m)
    s = np.clip(s, -50, 50)
    e = np.exp(s)
    return (m + np.log(e.sum()) / _BETA_G) * dx + 50.0 * (x.sum() - _TGT_G)**2

def _sm_grad(x):
    n = _N_G
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(-corr)[:_TOPK_G]
    c = corr[idx]
    m = c[0]
    s = _BETA_G * (c - m)
    s = np.clip(s, -50, 50)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        lag = ki - (n - 1)
        if lag >= 0:
            L = n - lag
            g[:L] += w[j] * (1.0 - x[lag:lag+L])
            g[lag:lag+L] -= w[j] * x[:L]
        else:
            al = -lag
            L = n - al
            g[al:al+L] += w[j] * (1.0 - x[:L])
            g[:L] -= w[j] * x[al:al+L]
    g *= dx
    g += 100.0 * (x.sum() - _TGT_G)
    return g

def lbfgs_opt(h, n, beta=1000, topk=5, maxiter=500):
    global _N_G, _BETA_G, _TOPK_G, _TGT_G
    _N_G, _BETA_G, _TOPK_G, _TGT_G = n, beta, topk, n/2.0
    h = proj_cont(h, n)
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return proj_cont(res.x, n)
    except:
        return h

def binary_hc(h_init, n, max_iters, rng, top_k=5):
    h = (h_init > 0.5).astype(float)
    # Ensure exactly n/2 ones
    if np.sum(h) != n//2:
        order = np.argsort(-h_init)
        h = np.zeros(n); h[order[:n//2]] = 1.0
    
    corr = np.correlate(h, 1.0-h, mode='full')
    cur_mc = float(np.max(corr))
    best_h = h.copy()
    best_mc = cur_mc
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Score candidates
        ti = np.argsort(-corr)[:top_k]
        sf1 = np.zeros(n)
        sf0 = np.zeros(n)
        for mi in ti:
            lag = mi - (n-1)
            if lag >= 0:
                L = n - lag
                if L > 0:
                    sf1[:L] += (1.0 - h[lag:lag+L])
                    sf0[lag:] += h[:L]
            else:
                al = -lag; L = n - al
                if L > 0:
                    sf1[al:al+L] += (1.0 - h[:L])
                    sf0[:L] += h[al:al+L]
        
        nt1 = min(20, len(ones))
        nt0 = min(20, len(zeros))
        t1 = ones[np.argsort(-sf1[ones])[:nt1]]
        t0 = zeros[np.argsort(-sf0[zeros])[:nt0]]
        
        best_nm = cur_mc
        best_p = None
        for i in t1:
            for j in t0:
                if i == j: continue
                h[i]=0; h[j]=1
                nm = float(np.max(np.correlate(h,1.-h,mode='full')))
                h[i]=1; h[j]=0
                if nm < best_nm:
                    best_nm = nm; best_p = (i,j)
        
        # Random candidates
        for _ in range(8):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i==j: continue
            h[i]=0; h[j]=1
            nm = float(np.max(np.correlate(h,1.-h,mode='full')))
            h[i]=1; h[j]=0
            if nm < best_nm:
                best_nm = nm; best_p = (i,j)
        
        if best_p:
            i,j = best_p
            h[i]=0; h[j]=1
            corr = np.correlate(h,1.-h,mode='full')
            cur_mc = best_nm
            if cur_mc < best_mc:
                best_mc = cur_mc; best_h = h.copy()
        else:
            if it % 800 == 700:
                # Perturb
                for _ in range(3):
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    if i != j:
                        h[i]=0; h[j]=1
                corr = np.correlate(h,1.-h,mode='full')
                cur_mc = float(np.max(corr))
            else:
                # Try harder random
                found = False
                for _ in range(30):
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    if i==j: continue
                    h[i]=0; h[j]=1
                    nm = float(np.max(np.correlate(h,1.-h,mode='full')))
                    h[i]=1; h[j]=0
                    if nm < cur_mc:
                        i_s, j_s = i, j
                        nm_s = nm
                        found = True
                        break
                if found:
                    h[i_s]=0; h[j_s]=1
                    corr = np.correlate(h,1.-h,mode='full')
                    cur_mc = nm_s
                    if cur_mc < best_mc:
                        best_mc = cur_mc; best_h = h.copy()
                else:
                    break
    
    return best_h, best_mc * (2.0/n)
