def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)

    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP on parent
    h = best_h.copy()
    for beta in [500, 5000, 50000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Heavy binary SA at parent resolution
    # Convert to binary
    h_bin = (best_h > 0.5).astype(float)
    n_half = best_n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(best_n)
        h_bin[sort_idx[:n_half]] = 1.0

    # Do multiple SA rounds with different temps
    sa_time_first = min(300, max(60, int(remaining_time() * 0.35)))
    h_sa, c5_sa = binary_sa(h_bin, best_n, sa_time_first, rng,
                            t_init=0.01, t_final=0.0001, cooldown=0.998, restart_gap=5000)

    # Also from other starts
    h_tm = theta_morse(best_n)
    h_tm_bin = (h_tm > 0.5).astype(float)
    h_tm_bin = h_tm_bin  # already binary-ish
    if int(np.sum(h_tm_bin)) != n_half:
        sort_idx = np.argsort(-h_tm_bin)
        h_tm_bin = np.zeros(best_n)
        h_tm_bin[sort_idx[:n_half]] = 1.0

    h_rand = np.zeros(best_n)
    h_rand[rng.choice(best_n, n_half, replace=False)] = 1.0

    sa_time_alt = min(150, max(30, int(remaining_time() * 0.15)))
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    h_sa2, c5_sa2 = binary_sa(h_tm_bin, best_n, sa_time_alt, rng,
                              t_init=0.01, t_final=0.0002, cooldown=0.997, restart_gap=3000)
    if c5_sa2 < best_c5:
        best_c5 = c5_sa2
        best_h = h_sa2.copy()

    h_sa3, c5_sa3 = binary_sa(h_rand, best_n, sa_time_alt, rng,
                              t_init=0.012, t_final=0.0002, cooldown=0.996, restart_gap=3000)
    if c5_sa3 < best_c5:
        best_c5 = c5_sa3
        best_h = h_sa3.copy()

    # Phase 3: More SA refinement on best binary
    h_best_bin = (best_h > 0.5).astype(float)
    if int(np.sum(h_best_bin)) != n_half:
        sort_idx = np.argsort(-h_best_bin)
        h_best_bin = np.zeros(best_n)
        h_best_bin[sort_idx[:n_half]] = 1.0

    sa_time_refine = min(200, max(40, int(remaining_time() * 0.2)))
    h_ref, c5_ref = binary_sa(h_best_bin, best_n, sa_time_refine, rng,
                              t_init=0.004, t_final=0.00003, cooldown=0.9985, restart_gap=6000)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref.copy()

    # Phase 4: Try other resolutions
    other_ns = [256, 320, 448, 512]
    for n_t in other_ns:
        if remaining_time() < 30:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        # Quick SLSQP
        h_opt = h_feas
        for beta in [5000, 50000]:
            if remaining_time() < 15:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=300)
        c5_opt = compute_c5(h_opt)

        # Binary SA on this resolution
        h_b = (h_opt > 0.5).astype(float)
        nh = n_t // 2
        if int(np.sum(h_b)) != nh:
            sort_idx = np.argsort(-h_b)
            h_b = np.zeros(n_t)
            h_b[sort_idx[:nh]] = 1.0
        sa_t = min(40, max(10, int(remaining_time() * 0.05)))
        h_b2, c5_b2 = binary_sa(h_b, n_t, sa_t, rng,
                                t_init=0.008, t_final=0.0002, cooldown=0.997, restart_gap=3000)
        if c5_b2 < c5_opt:
            c5_opt = c5_b2
            h_opt = h_b2.copy()

        # Continuous refinement
        h_cont = h_opt.copy()
        for beta in [20000, 100000]:
            if remaining_time() < 10:
                break
            h_cont = slsqp_opt(h_cont, n_t, beta, maxiter=400)
        c5_cont = compute_c5(h_cont)
        if c5_cont < c5_opt:
            c5_opt = c5_cont
            h_opt = h_cont.copy()

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 5: Top-k equalization on best
    n = best_n
    h = best_h.copy()
    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=5, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Final SLSQP polish
    for beta in [100000, 500000, 1000000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Final top-k
    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=4, max_iters=300)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
