If I change h[i0] from 1 to 1-ε (where h[i0+k*]=0), this reduces corr[k*] by ε (approximately).
If I change h[j0] from 0 to ε (where h[j0+k*]=1), this reduces corr[k*] by ε (approximately).

The optimal strategy is to find a set of positions where:
- Reducing some 1s to 1-ε reduces the peak correlations
- Increasing some 0s to ε reduces the peak correlations
- The net sum change is 0
- Other correlations don't increase too much

This is a linear programming problem in ε! Let me formulate it:

Variables: ε_i ≥ 0 for positions where h[i]=1 (set h[i] = 1-ε_i)

- Σ ε_i = Σ δ_j (sum conservation)
- For each peak shift k: Σ_{contributing i} ε_i + Σ_{contributing j} δ_j ≥ M - target
- ε_i + δ_j ≤ 1 (bounded perturbation)
- 0 ≤ ε_i, δ_j ≤ 1

Objective: minimize Σ ε_i (or minimize ε_max)

            else:
                a = -off
                nsum = n - a
                mask = (h[:nsum] == 1) & (h[a:a+nsum] == 0)
                one_score[:nsum] += mask
        
        # Prioritize high-score ones
        one_idx = ones[np.argsort(-one_score[ones])]
        
        best_swap = None
        best_new_max = c_max
        
        # Try swaps from high-priority ones with random zeros
        tried = 0
        max_tried = 400
        for i in one_idx[:min(40, len(one_idx))]:
            if tried >= max_tried:
                break
            for _ in range(min(10, len(zeros))):
                j = zeros[rng.randint(len(zeros))]
                key = (min(i,j), max(i,j))
                if key in tabu and time.time() - tabu[key] < tabu_wall:
                    continue
                tried += 1
                h[i] = 0
                h[j] = 1
                nc = compute_corr(h)
                new_max = float(np.max(nc))
                h[i] = 1
                h[j] = 0
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            tabu[(min(i,j), max(i,j))] = time.time()
            no_improve = 0
        else:
            no_improve += 1
    
    return best_h, best_max_corr * dx_v

def targeted_softening(h_bin, n, num_free=8, iterations=100):
    """Allow small continuous perturbations at selected positions to reduce peak."""
    h = h_bin.astype(float).copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    
    for _ in range(iterations):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_k = np.argsort(-corr)[:3]
        
        # Find positions to soften: ones that contribute to top correlations
        candidates = []
        for kt in top_k:
            off = kt - (n - 1)
            if off >= 0:
                nsum = n - off
                for i in range(nsum):
                    if h[i + off] > 0.8 and h[i] < 0.2:
                        candidates.append(i + off)
                    elif h[i] > 0.8 and h[i + off] < 0.2:
                        candidates.append(i)
            else:
                a = -off
                nsum = n - a
                for i in range(nsum):
                    if h[i] > 0.8 and h[i + a] < 0.2:
                        candidates.append(i)
                    elif h[i + a] > 0.8 and h[i] < 0.2:
                        candidates.append(i + a)
        
        if not candidates:
            break
        
        # Select top candidates (most contributing)
        candidates = list(set(candidates))
        rng_local = np.random.RandomState(len(candidates))
        if len(candidates) > num_free:
            candidates = list(rng_local.choice(candidates, num_free, replace=False))
        
        # Try small perturbations
        improved = False
        for pos in candidates:
            if remaining() < 3:
                break
            if h[pos] > 0.5:
                # Turn down: h[pos] -> h[pos] - eps
                for eps in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    ht = h.copy()
                    ht[pos] -= eps
                    # Compensate: add eps to a zero that doesn't hurt
                    zeros = np.where(h < 0.1)[0]
                    if len(zeros) == 0:
                        break
                    best_comp = None
                    best_c = compute_c5(project_feasible(ht, n))
                    for z in rng_local.choice(zeros, min(20, len(zeros)), replace=False):
                        ht2 = ht.copy()
                        ht2[z] += eps
                        ht2 = project_feasible(ht2, n)
                        c2 = compute_c5(ht2)
                        if best_comp is None or c2 < best_comp:
                            best_comp = c2
                            best_z = z
                    if best_comp < best_c5 - 1e-10:
                        h = best_h_sol = h.copy()
                        h[pos] -= eps
                        h[best_z] += eps
                        h = project_feasible(h, n)
                        best_c5 = compute_c5(h)
                        improved = True
                        break
            else:
                # Turn up: h[pos] -> h[pos] + eps
                for eps in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    ht = h.copy()
                    ht[pos] += eps
                    zeros = np.where(h > 0.9)[0]
                    if len(zeros) == 0:
                        break
                    best_comp = None
                    for z in rng_local.choice(zeros, min(20, len(zeros)), replace=False):
                        ht2 = ht.copy()
                        ht2[z] -= eps
                        ht2 = project_feasible(ht2, n)
                        c2 = compute_c5(ht2)
                        if best_comp is None or c2 < best_comp:
                            best_comp = c2
                            best_z = z
                    if best_comp < best_c5 - 1e-10:
                        h[pos] += eps
                        h[best_z] -= eps
                        h = project_feasible(h, n)
                        best_c5 = compute_c5(h)
                        improved = True
                        break
        if not improved:
            break
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: Quick SLSQP warmup
    for beta in [100, 500, 2000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 2: Top-shifts equalization
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=5, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 3: Smart binary search from best continuous
    if remaining() > 60:
        h_bin = binarize(h, n)
        c5_bin = compute_c5(h_bin)
        sa_time = min(200, max(30, int(remaining() * 0.4)))
        h_b, c5_b = smart_binary_search(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()
    
    # Phase 4: Continuous refinement from best binary
    h_bin_best = binarize(best_h, best_n)
    if remaining() > 40:
        # Targeted softening
        h_soft, c5_soft = targeted_softening(h_bin_best, best_n, num_free=10, iterations=80)
        if c5_soft < best_c5:
            best_c5 = c5_soft
            best_h = h_soft.copy()
            h = h_soft.copy()
    
    # Continued from best
    h = best_h.copy()
    if remaining() > 40:
        # L-BFGS-B smooth max
        h_lb = lbfgs_smooth(h, best_n, 5000, maxiter=100)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()
            h = h_lb.copy()
    
    if remaining() > 30:
        h_eq2, c5_eq2 = top_shifts_focus(h, best_n, top_k=4, max_iters=150)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()
    
    # Phase 5: More SLSQP with high beta
    for beta in [5000, 15000, 40000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(h, best_n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 6: Diverse starts
    if remaining() > 80:
        # Thue-Morse at current n
        h_tm = thue_morse(n, rng)
        h_tm = slsqp_smooth(h_tm, n, 3000, 200)
        h_tm, c5_tm = top_shifts_focus(h_tm, n, 4, 100)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
        
        # Binary TM search
        h_bm, c5_bm = smart_binary_search(thue_morse(n, rng), n, rng, time_limit=min(60, int(remaining() * 0.3)))
        if c5_bm < best_c5:
            best_c5 = c5_bm
            best_h = h_bm.copy()
            # Continuous refine
            h_ref = slsqp_smooth(h_bm, n, 5000, 200)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
            else:
                h_soft2, c5_s2 = targeted_softening(h_bm, n, 8, 50)
                if c5_s2 < best_c5:
                    best_c5 = c5_s2
                    best_h = h_soft2.copy()
    
    # Phase 7: Resolution sweep
    if remaining() > 60:
        n_base = best_n
        n_candidates = []
        for delta in [-4, -2, 2, 4, 6, 8, 12, 16, -8, -12, 24, -24]:
            nt = n_base + delta
            if 128 <= nt <= 700 and nt % 2 == 0:
                n_candidates.append(nt)
        n_candidates = list(dict.fromkeys(n_candidates))  # unique, preserve order
        
        for n_t in n_candidates[:6]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            
            # Quick optimize
            h_t = slsqp_smooth(h_t, n_t, 3000, 200)
            h_t, c5_t = top_shifts_focus(h_t, n_t, 4, 100)
            
            # Binary search at this resolution
            if remaining() > 30:
                h_tb = binarize(h_t, n_t)
                h_tb, c5_tb = smart_binary_search(h_tb, n_t, rng, time_limit=
                     min(40, int(remaining() * 0.3)))
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb.copy()
                    # Continuous refine from binary
                    h_ref = slsqp_smooth(h_t, n_t, 5000, 200)
                    c5_ref = compute_c5(h_ref)
                    if c5_ref < c5_t:
                        c5_t = c5_ref
                        h_t = h_ref.copy()
                    else:
                        h_s, c5_s = targeted_softening(h_t, n_t, 8, 40)
                        if c5_s < c5_t:
                            c5_t = c5_s
                            h_t = h_s.copy()
            
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t
    
    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    
    for beta in [10000, 30000, 80000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    for tk in [3, 5, 7]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Final binary + continuous cycle
    if remaining() > 30:
        h_bin_final = binarize(h, n)
        h_bf, c5_bf = smart_binary_search(h_bin_final, n, rng, time_limit=min(50, int(remaining() * 0.4)))
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
            h_soft, c5_s = targeted_softening(h_bf, n, 12, 60)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_soft.copy()
            h_ref = slsqp_smooth(best_h, n, 10000, 300)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
    
    # Final top-shifts polish
    if remaining() > 10:
        h_eq, c5_eq = top_shifts_focus(best_h, best_n, top_k=5, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < best_c5:
                best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
