_G_LAM = 0.0

# === Core evaluation ===
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

# === Projection ===
def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

# === Smooth max (log-sum-exp) ===
def lse_obj(x, beta):
    corr = corr_array(x)
    m = corr.max()
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(x)
    return (m + np.log(ex.sum()) / beta) * dx

def lse_grad(x, beta):
    corr = corr_array(x)
    m = corr.max()
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    w *= (2.0 / len(x))
    return _cgw(x, w)

def _cgw(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-18:
            continue
        off = k - (n - 1)
        if off >= 0:
            m = n - off
            g[off:off + m] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[off:off + m]
        else:
            a = -off
            m = n - a
            g[:m] += wk * (1.0 - h[a:a + m])
            g[a:a + m] -= wk * h[:m]
    return g

# === SLSQP wrappers ===

    return lse_obj(x, _G_BETA)

def _sg(x):
    return lse_grad(x, _G_BETA)

def _sc(x):
    return x.sum() - _G_N / 2.0

def _scg(x):
    return np.ones(_G_N)

def do_slsqp(h, n, beta, maxiter=500):
    global _G_BETA, _G_N
    _G_BETA = beta
    _G_N = n
    h0 = project_feasible(h, n)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sf, h0, jac=_sg, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scg}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < lse_obj(h0, beta) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def _alf(x):
    obj = lse_obj(x, _G_BETA)
    c = x.sum() - _G_N / 2.0
    return obj + _G_MU * c * c / 2.0 + _G_LAM * c

def _alg(x):
    g = lse_grad(x, _G_BETA)
    c = x.sum() - _G_N / 2.0
    g += (_G_MU * c + _G_LAM) * np.ones(len(x))
    return g

    _G_N = n
    _G_LAM = 0.0

        except Exception:
            break
        c = h.sum() - n / 2.0

            break
        _G_LAM += _G_MU * c
        _G_MU *= 1.5
    return project_feasible(h, n)

def do_subgrad(h, n, top_k, max_iters):
    h = project_feasible(h, n)
    dx = 2.0 / n
    bh = h.copy()
    bc = compute_c5(h)
    mom = np.zeros(n)
    mu = 0.85
    step = 0.05
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = corr_array(h)
        ti = np.argsort(-corr)[:top_k]
        ct = corr[ti]
        wt = (ct - ct.min() + 1e-10) ** 2
        wt /= wt.sum()
        w = np.zeros(2 * n - 1)
        w[ti] = wt
        w *= dx
        g = _cgw(h, w)
        g -= g.mean()
        mom = mu * mom - step * g
        hn = project_feasible(h + mom, n)
        c5n = compute_c5(hn)
        if c5n < bc - 1e-15:
            bc = c5n
            bh = hn.copy()
            h = hn.copy()
        step *= 0.9995
    return bh, bc

    y = h.copy()
    bh = h.copy()

    tv = 1.0
    for it in range(max_iters):
        if remaining() < 5:
            break
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tv * tv)) / 2.0
        alpha = (tv - 1.0) / tn

        g = lse_grad(y_new, beta)
        lr = 2.0 / (beta * (it + 10) + 1.0)

        c5n = compute_c5(hn)
        if c5n < bc:
            bc = c5n
            bh = hn.copy()
        y = h
        h = hn
        tv = tn
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_G_BETA = 1.0
_G_N = 0
_G_LAM = 0.0
_G_MU = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def lse_obj(x, beta):
    corr = corr_array(x)
    m = corr.max()
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(x)
    return (m + np.log(ex.sum()) / beta) * dx

def lse_grad(x, beta):
    corr = corr_array(x)
    m = corr.max()
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / ex.sum()
    w *= (2.0 / len(x))
    return _cgw(x, w)

def _cgw(h, w):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-18:
            continue
        off = k - (n - 1)
        if off >= 0:
            m = n - off
            g[off:off + m] += wk * (1.0 - h[:m])
            g[:m] -= wk * h[off:off + m]
        else:
            a = -off
            m = n - a
            g[:m] += wk * (1.0 - h[a:a + m])
            g[a:a + m] -= wk * h[:m]
    return g

# SLSQP wrappers
def _sf(x):
    return lse_obj(x, _G_BETA)
def _sg(x):
    return lse_grad(x, _G_BETA)
def _sc(x):
    return x.sum() - _G_N / 2.0
def _scg(x):
    return np.ones(_G_N)

def do_slsqp(h, n, beta, maxiter=500):
    global _G_BETA, _G_N
    _G_BETA = beta
    _G_N = n
    h0 = project_feasible(h, n)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sf, h0, jac=_sg, method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scg}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < lse_obj(h0, beta) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

# Aug Lag L-BFGS-B
def _alf(x):
    obj = lse_obj(x, _G_BETA)
    c = x.sum() - _G_N / 2.0
    return obj + _G_MU * c * c / 2.0 + _G_LAM * c

def _alg(x):
    g = lse_grad(x, _G_BETA)
    c = x.sum() - _G_N / 2.0
    g += (_G_MU * c + _G_LAM) * np.ones(len(x))
    return g

def do_auglag(h, n, beta, outer=30, mu_init=1.0):
    global _G_BETA, _G_N, _G_LAM, _G_MU
    _G_BETA = beta
    _G_N = n
    _G_LAM = 0.0
    _G_MU = mu_init
    h = project_feasible(h, n)
    bnds = [(0.0, 1.0)] * n
    for _ in range(outer):
        if remaining() < 10:
            break
        try:
            res = minimize(_alf, h, jac=_alg, method='L-BFGS-B',
                           bounds=bnds, options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
            h = project_feasible(res.x, n)
        except Exception:
            break
        c = h.sum() - n / 2.0
        if abs(c) < 1e-10:
            break
        _G_LAM += _G_MU * c
        _G_MU *= 1.5
    return project_feasible(h, n)

# Subgradient
def do_subgrad(h, n, top_k, max_iters):
    h = project_feasible(h, n)
    dx = 2.0 / n
    bh = h.copy()
    bc = compute_c5(h)
    mom = np.zeros(n)
    mu = 0.85
    step = 0.05
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = corr_array(h)
        ti = np.argsort(-corr)[:top_k]
        ct = corr[ti]
        wt = (ct - ct.min() + 1e-10) ** 2
        wt /= wt.sum()
        w = np.zeros(2 * n - 1)
        w[ti] = wt
        w *= dx
        g = _cgw(h, w)
        g -= g.mean()
        mom = mu * mom - step * g
        hn = project_feasible(h + mom, n)
        c5n = compute_c5(hn)
        if c5n < bc - 1e-15:
            bc = c5n
            bh = hn.copy()
            h = hn.copy()
        step *= 0.9995
    return bh, bc

def do_nesterov(h, n, beta, max_iters):

    y = h.copy()
    bh = h.copy()
    bc = compute_c5(h)
    tv = 1.0
    for it in range(max_iters):
        if remaining() < 5:
            break
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tv * tv)) / 2.0
        alpha = (tv - 1.0) / tn

        g = lse_grad(yn, beta)
        lr = 2.0 / (beta * (it + 10) + 1.0)

        c5n = compute_c5(hn)

            bh = hn.copy()
        y = h
        h = hn
        tv = tn
    return bh, bc
