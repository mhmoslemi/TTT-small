def project_feasible(h, n):

def grad_k(h, k, n):
    """Gradient of corr[k] with respect to h"""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g

    dx = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        s = np.clip(beta * (corr - m), -80, 80)
        e = np.exp(s)
        return (m + np.log(e.sum()) / beta) * dx
    
    def grad(x):

        m = np.max(corr)
        s = np.clip(beta * (corr - m), -80, 80)

def coord_pair_descent(h, n, max_sweeps=10):
    """Coordinate pair descent: change h[i] and h[j] to maintain sum"""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Find a good reservoir: a coordinate far from i and not at a bound
            candidates = [j for j in range(n) if j != i and 0.1 < h[j] < 0.9]
            if not candidates:
                continue
            # Pick the one that gives most improvement
            j = candidates[i % len(candidates)]  # Rotating reservoir
            
            # Binary search for optimal delta
            # f(δ) = max_k corr_k(h + δ*(e_i - e_j))
            # This is convex in δ, so we can use ternary search or binary search on derivative
            lo, hi = -0.5, 0.5
            best_delta = 0
            best_val = compute_c5(h)
            
            for _ in range(30):
                mid = (lo + hi) / 2
                h_test = h.copy()
                h_test[i] = np.clip(h[i] + mid, 0, 1)
                h_test[j] = np.clip(h[j] - mid, 0, 1)
                val = compute_c5(h_test)
                if val < best_val:
                    best_val = val
                    best_delta = mid
                # For convex f, if f(mid) < f(lo) then minimum is to the right
                h_test2 = h.copy()
                h_test2[i] = np.clip(h[i] + lo, 0, 1)
                h_test2[j] = np.clip(h[j] - lo, 0, 1)
                if compute_c5(h_test2) > val:
                    lo = mid
                else:
                    hi = mid
            
            if best_delta != 0:
                h[i] = np.clip(h[i] + best_delta, 0, 1)
                h[j] = np.clip(h[j] - best_delta, 0, 1)
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_c5:
                    best_c5 = c
                    improved = True
        
        if not improved:
            break

def sigmoid_optimize(h_init, n, beta, maxiter=1000):
    """Optimize using sigmoid parameterization and L-BFGS-B"""
    # Initialize a from h: a = logit(h) = log(h/(1-h))
    h_safe = np.clip(h_init, 1e-6, 1 - 1e-6)
    a_init = np.log(h_safe / (1 - h_safe))
    
    penalty_coeff = 1e8
    
    def obj_and_grad(a):
        h = 1.0 / (1.0 + np.exp(-2.0 * a))  # sigmoid(2a) for wider range
        # Actually, let's use h = sigmoid(a) = 1/(1+exp(-a))
        # Then a is unconstrained and h ∈ (0,1)
        # But the range of h is limited by the range of a
        # For h to cover [0.01, 0.99], a needs to be in [-4.6, 4.6]
        
        corr = compute_corr(h)
        m = np.max(corr)
        s = np.clip(beta * (corr - m), -80, 80)
        e = np.exp(s)
        obj = (m + np.log(e.sum()) / beta) * (2.0 / n)

        p = e / e.sum()

        g_h = np.zeros(n)

                npairs = n - offset
                g_h[offset:] += wk * (1.0 - h[:npairs])

            else:
                a_off = -offset
                npairs = n - a_off
                g_h[:npairs] += wk * (1.0 - h[a_off:])
                g_h[a_off:] -= wk * h[:npairs]

        g_a = g_h * h * (1.0 - h)
        
        # Add penalty for sum constraint
        sum_dev = np.sum(h) - n / 2.0
        obj_pen = penalty_coeff * sum_dev ** 2
        g_a_pen = 2 * penalty_coeff * sum_dev * h * (1.0 - h)
        
        return obj + obj_pen, g_a + g_a_pen
    
    # Use L-BFGS-B
    res = minimize(lambda a: obj_and_grad(a)[0], a_init,

                   options={'maxiter': maxiter, 'ftol': 1e-18})

from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = float(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h[:] += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = float(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_smooth_max(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -80.0, 80.0)
    e = np.exp(s)
    p = e / e.sum()
    g = np.zeros(n)
    for k in np.where(p > 1e-10)[0]:
        wk = p[k] * dx
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[offset:]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:npairs]

def smooth_max_slsqp(h, n, beta, maxiter=800):
    dx = 2.0 / n

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        s = np.clip(beta * (corr - m), -80.0, 80.0)
        e = np.exp(s)
        return (m + np.log(e.sum()) / beta) * dx

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        s = np.clip(beta * (corr - m), -80.0, 80.0)
        e = np.exp(s)
        p = e / e.sum()
        g = np.zeros(n)
        for k in np.where(p > 1e-10)[0]:
            wk = p[k] * dx
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                g[offset:] += wk * (1.0 - x[:npairs])
                g[:npairs] -= wk * x[offset:]
            else:
                a = -offset
                npairs = n - a
                g[:npairs] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:npairs]
        return g

    def c_eq(x):
        return np.sum(x) - n / 2.0

    def c_eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}]
    try:

def cutting_plane_refine(h, n, K=20, delta=0.04, max_iter=100):
