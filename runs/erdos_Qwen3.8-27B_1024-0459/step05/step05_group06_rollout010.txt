def block_init(n, M, seed=0):
    """Create block-structured initialization with M blocks."""
    rng = np.random.RandomState(seed)
    ones_needed = n // 2
    block_len = ones_needed // M
    gap = (n - M * block_len) // (M + 1)  # gap between and around blocks
    h = np.zeros(n)
    pos = gap
    for m in range(M):
        end = pos + block_len
        if end > n:
            end = n
        h[pos:end] = 1.0
        pos = end + gap
    # Adjust to get exactly n/2 ones
    ones = int(np.sum(h))
    if ones < ones_needed:
        zero_idx = np.where(h == 0)[0]
        extra = rng.choice(zero_idx, ones_needed - ones, replace=False)
        h[extra] = 1.0
    elif ones > ones_needed:
        one_idx = np.where(h == 1)[0]
        extra = rng.choice(one_idx, ones - ones_needed, replace=False)
        h[extra] = 0.0
    return h

def smooth_obj_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def _obj_fn(x):
    return smooth_obj_grad(x, _N_OPT, _BETA_OPT)[0]


def _grad_fn(x):
    return smooth_obj_grad(x, _N_OPT, _BETA_OPT)[1]


def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0


def _con_jac(x):
    return np.ones(_N_OPT)


def slsqp_opt(h, n, beta, maxiter=300):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()


def coord_descent_fast(h, n, n_sweeps=100, top_k=30):
    """Fast coordinate descent using top-k active lags."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    center = n - 1
    tgrid = np.linspace(0.0, 1.0, 81)

    for sweep in range(n_sweeps):
        if remaining_time() < 2:
            break
        improved = False
        max_corr_val = np.max(best_corr)
        active_lags = np.where(best_corr >= max_corr_val - 2.0)[0]
        if len(active_lags) > top_k:
            active_lags = np.argsort(best_corr)[::-1][:top_k]
        active_lags = list(active_lags)
        if center not in active_lags:
            active_lags.append(center)

        for i in range(n):
            if remaining_time() < 1:
                break
            t0 = best[i]

            # Compute delta coefficients for active lags
            # For lag m: delta = (t - t0) * coeff_m  (linear)
            # Plus quadratic at center
            coeffs = np.zeros(len(active_lags))
            has_quadratic = False

            for idx_m, m in enumerate(active_lags):
                j1 = i - m + center
                j2 = i + m - center
                c = 0.0
                if m == center:
                    has_quadratic = True
                else:
                    if 0 <= j1 < n:
                        c += (1.0 - best[j1])
                    if 0 <= j2 < n:
                        c += -best[j2]
                coeffs[idx_m] = c

            # Evaluate on grid
            best_t = t0
            best_f = max_corr_val * dx
            for t in tgrid:
                if abs(t - t0) < 1e-14:
                    continue
                new_vals = best_corr[active_lags] + (t - t0) * coeffs
                if has_quadratic:
                    ci = list(active_lags).index(center)
                    new_vals[ci] += t * (1 - t) - t0 * (1 - t0)
                # Also check center if not in active
                ctr_val = best_corr[center] + (t - t0) * 0.0
                if has_quadratic:
                    ctr_val = best_corr[center] + t * (1 - t) - t0 * (1 - t0)
                if center in active_lags:
                    pass
                else:
                    new_vals = np.append(new_vals, ctr_val)
                else:
                    pass
                cand_max = np.max(new_vals)
                # Safety: also check center
                if center not in active_lags:
                    ctr_v = best_corr[center] + t * (1.0 - t) - t0 * (1.0 - t0)
                    cand_max = max(cand_max, ctr_v)
                f_val = cand_max * dx
                if f_val < best_f - 1e-15:
                    best_f = f_val
                    best_t = t

            if best_t != t0 and best_f < best_c5 - 1e-14:
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                max_corr_val = np.max(best_corr)
                active_lags = np.where(best_corr >= max_corr_val - 2.0)[0]
                if len(active_lags) > top_k:
                    active_lags = np.argsort(best_corr)[::-1][:top_k]
                active_lags = list(active_lags)
                if center not in active_lags:
                    active_lags.append(center)
                improved = True

        if not improved:
            break

    best = project_feasible(best, n)
    final_c5 = compute_c5(best)
    return best, final_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def block_init(n, M, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones_per_block = (n // 2) // M
    gap = (n - M * ones_per_block) // (M + 1)
    pos = gap
    for m in range(M):
        end = pos + ones_per_block
        if end > n:
            end = n
        h[pos:min(end, n)] = 1.0
        pos = min(end, n) + gap
    ones = int(np.sum(h))
    target = n // 2
    if ones < target:
        zeros = np.where(h == 0)[0]
        extra = rng.choice(zeros, target - ones, replace=False)
        h[extra] = 1.0
    elif ones > target:
        ones_idx = np.where(h == 1)[0]
        extra = rng.choice(ones_idx, ones - target, replace=False)
        h[extra] = 0.0
    return h


def thue_morse_init(n, seed=0):
    t = np.zeros(n, dtype=np.int64)
    for i in range(n):
        t[i] = bin(i).count('1') & 1
    h = (1 - t).astype(float)
    ones = int(np.sum(h))
    target = n // 2
    rng = np.random.RandomState(seed)
    if ones > target:
        idx = np.where(h == 1)[0]
        chg = rng.choice(idx, ones - target, replace=False)
        h[chg] = 0.0
    elif ones < target:
        idx = np.where(h == 0)[0]
        chg = rng.choice(idx, target - ones, replace=False)
        h[chg] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n, seed)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP at current resolution
    for beta in [2000, 10000, 50000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: Intensive coordinate descent
    if remaining_time() > 30:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=150, top_k=40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: SLSQP with high beta for precision
    for beta in [100000, 300000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, n, beta, 350)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: More coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=100, top_k=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Try different initializations at same resolution
    inits = []
    for M in [4, 8, 16, 32]:
        inits.append(block_init(n, M, seed=M))
    inits.append(thue_morse_init(n, seed=1))
    inits.append(thue_morse_init(n, seed=42))

    for init_h in inits:
        if remaining_time() < 60:
            break
        init_h = project_feasible(init_h, n)
        c5_init = compute_c5(init_h)
        if c5_init < best_c5 + 0.02:  # Only pursue if reasonable
            h_try = init_h.copy()
            h_try = slsqp_opt(h_try, n, 5000, 300)
            h_try, c5_try = coord_descent_fast(h_try, n, n_sweeps=80, top_k=40)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                n = best_n

    # Phase 6: Up-sample to higher resolution and optimize
    higher_res_list = []
    for mult in [1.5, 2.0]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if 64 <= rn <= 2560:
            higher_res_list.append(rn)
    for extra in [256, 512]:
        rn = best_n + extra
        if rn % 2 != 0:
            rn += 1
        if 64 <= rn <= 2560:
            if rn not in higher_res_list:
                higher_res_list.append(rn)
    higher_res_list = sorted(set(higher_res_list))

    for rn in higher_res_list:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        # Optimize at higher resolution
        for beta in [3000, 20000, 100000]:
            if remaining_time() < 25:
                break
            h_r = slsqp_opt(h_r, rn, beta, 300)

        if remaining_time() > 20:
            h_r, c5_r = coord_descent_fast(h_r, rn, n_sweeps=60, top_k=30)

        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 7: Random perturbations around best
    n = best_n
    h = best_h.copy()
    n_perturb = max(5, n // 40)
    for pert in range(20):
        if remaining_time() < 20:
            break
        h_pert = h.copy()
        idx = rng.choice(n, n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.1, 0.1, n_pert)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.01:
            h_new = slsqp_opt(h_pert, n, 20000, 250)
            h_new, c5_new = coord_descent_fast(h_new, n, n_sweeps=50, top_k=40)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Final intensive polish
    n = best_n
    h = best_h.copy()
    if remaining_time() > 15:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=200, top_k=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 10:
        h_new = slsqp_opt(best_h, best_n, 500000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 8:
        h_new, c5_new = coord_descent_fast(best_h, best_n, n_sweeps=100, top_k=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
