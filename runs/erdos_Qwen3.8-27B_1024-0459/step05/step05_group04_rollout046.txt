4. **Lag targeting**: Identify the top-k worst lags and specifically try to reduce them
5. **Resolution diversity**: Try n=384, n=380, n=388, n=392, n=400, etc.

import numpy as np
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2*(n-1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)
    hi3 = min(j + n - 1, r - 1)
    if lo3 <= hi3:
        ks = np.arange(lo3, hi3 + 1)
        dr[lo3:hi3+1] += (1.0 - h[j + n - 1 - ks])
    lo4 = max(n - 1 - j, 0)
    hi4 = min(2*(n-1) - j, r - 1)
    if lo4 <= hi4:
        ks = np.arange(lo4, hi4 + 1)
        dr[lo4:hi4+1] -= h[j + ks - (n - 1)]
    dr[n - 1] -= 2.0
    return dr

def ensure_count(h, n):
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            to_rm = np.random.RandomState(0).choice(ones, n_ones - n_half, replace=False)
            h[to_rm] = 0
        else:
            zeros = np.where(h == 0)[0]
            to_add = np.random.RandomState(0).choice(zeros, n_half - n_ones, replace=False)
            h[to_add] = 1
    return h

def binary_ls_targeted(h_init, n, time_limit, rng):
    h = (h_init > 0.5).copy().astype(float)
    ensure_count(h, n)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    start = time.time()
    stall = 0
    max_stall = 800

    while time.time() - start < time_limit and remaining() > 10:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        top_m = min(16, 2*n-1)
        top_idx = np.argsort(-corr)[:top_m]
        candidates = []

        for s in top_idx[:10]:
            off = s - (n - 1)
            if off >= 0:
                np_ = n - off
                ci = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for idx in ci[:20]:
                    candidates.append((int(idx + off), int(idx)))
            else:
                ao = -off
                np_ = n - ao
                ci = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for idx in ci[:20]:
                    candidates.append((int(idx), int(idx + ao)))

        for _ in range(40):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))

        best_move = None
        best_nm = cur_max + 1e-15
        seen = set()

        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (min(i,j), max(i,j))
            if key in seen:
                continue
            seen.add(key)
            dr = swap_delta(h, i, j, n)
            nm = float(np.max(corr + dr))
            if nm < best_nm:
                best_nm = nm
                best_move = (i, j, dr)

        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_nm
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                n_pert = rng.randint(3, 8)
                for _ in range(n_pert):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        h[i] = 0
                        h[j] = 1
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0

    return best_h, best_max * dx

def binary_sa(h_init, n, time_limit, rng, T0=0.05):
    h = (h_init > 0.5).copy().astype(float)
    ensure_count(h, n)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    T = T0
    T_min = 1e-5
    cool = 0.9998
    start = time.time()

    while time.time() - start < time_limit and remaining() > 5:
        if T < T_min:
            T = T0
        T *= cool
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        top_idx = int(np.argmax(corr))
        off = top_idx - (n - 1)
        targeted = False
        if off >= 0:
            np_ = n - off
            ci = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
            if len(ci) > 0 and rng.rand() < 0.4:
                idx = ci[rng.randint(len(ci))]
                i = int(idx + off)
                j = int(idx)
                targeted = True
        if not targeted:
            off2 = -off
            if off2 > 0:
                np_ = n - off2
                ci = np.where((h[:np_] == 1) & (h[off2:off2+np_] == 0))[0]
                if len(ci) > 0 and rng.rand() < 0.4:
                    idx = ci[rng.randint(len(ci))]
                    i = int(idx)
                    j = int(idx + off2)
                    targeted = True
        if not targeted:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        dr = swap_delta(h, i, j, n)
        nm = float(np.max(corr + dr))
        delta = nm - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-15)):
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = nm
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

    return best_h, best_max * dx

def block_ls(h_init, n, time_limit, rng):
    h = (h_init > 0.5).copy().astype(float)
    ensure_count(h, n)
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining() > 10:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_move = None
        best_nm = cur_max + 1e-15

        for L in [1, 2, 3, 4, 5]:
            n_cands = 15 if L == 1 else 8
            for _ in range(n_cands):
                i = rng.randint(0, n - L)
                j = rng.randint(0, n - L)
                if abs(i - j) < L:
                    continue
                bi = h[i:i+L]
                bj = h[j:j+L]
                if np.sum(bi) != np.sum(bj):
                    continue
                dr = np.zeros(2*n - 1)
                for off in range(L):
                    ii = i + off
                    jj = j + off
                    if h[ii] != h[jj]:
                        if h[ii] == 1:
                            dr += swap_delta(h, ii, jj, n)
                        else:
                            dr += swap_delta(h, jj, ii, n)
                nm = float(np.max(corr + dr))
                if nm < best_nm:
                    best_nm = nm
                    best_move = (i, j, L, dr)

        if best_move is not None:
            i, j, L, dr = best_move
            for off in range(L):
                h[i + off], h[j + off] = h[j + off], h[i + off]
            corr = corr + dr
            cur_max = best_nm
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > 300:
                n_pert = rng.randint(2, 6)
                for _ in range(n_pert):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        h[i] = 0
                        h[j] = 1
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0

    return best_h, best_max * dx

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def continuous_refine(h, n, beta, maxiter):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    from scipy.optimize import minimize

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac,
             'args': (n,)}]

    res = minimize(_obj_fun, h, jac=_grad_fun, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   args=(n, beta),
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
    if res.fun < _obj_fun(h, n, beta):
        return project_feasible(res.x, n)
    return h

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def _obj_fun(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

def _grad_fun(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * (2.0 / n)
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

def _con_jac_static(x):
    return np.ones(len(x))

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def random_binary(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            h[rng.choice(idx, n_ones - target, replace=False)] = 0
        else:
            idx = np.where(h == 0)[0]
            h[rng.choice(idx, target - n_ones, replace=False)] = 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)

        best_c5 = compute_c5(best_h)

    # Phase 1: Targeted binary local search
    t1 = min(250, int(remaining() * 0.2))
    if t1 > 20:
        h1, c5_1 = binary_ls_targeted(h0, n, t1, rng)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n

    # Phase 2: Block local search
    t2 = min(200, int(remaining() * 0.15))
    if t2 > 20:
        h2, c5_2 = block_ls(best_h, best_n, t2, rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n

    # Phase 3: Simulated annealing
    t3 = min(200, int(remaining() * 0.15))
    if t3 > 20:
        h3, c5_3 = binary_sa(best_h, best_n, t3, rng, T0=0.08)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            best_n = n

    # Phase 4: Multiple restarts with diverse starts
    starts = []
    starts.append(('tm', thue_morse(n)))
    starts.append(('rand1', random_binary(n, np.random.RandomState(seed+1))))
    starts.append(('rand2', random_binary(n, np.random.RandomState(seed+2))))

    for name, sp in starts:
        if remaining() < 60:
            break
        sp_c5 = compute_c5(sp)
        if sp_c5 < best_c5 * 1.01:
            t_as = min(100, int(remaining() * 0.08))
            if t_as > 15:
                h_a, c5_a = binary_ls_targeted(sp, n, t_as, rng)
                if c5_a < best_c5:
                    best_c5 = c5_a
                    best_h = h_a.copy()
                    best_n = n
                if remaining() > 30:
                    t_sa = min(80, int(remaining() * 0.06))
                    h_s, c5_s = binary_sa(h_a, n, t_sa, rng, T0=0.06)
                    if c5_s < best_c5:
                        best_c5 = c5_s
                        best_h = h_s.copy()
                        best_n = n

    # Phase 5: Try different resolutions
    res_list = sorted(set([n, n-12, n-8, n-4, n+4, n+8, n+12, n+16,
                            int(n*0.8), int(n*1.2)]))
    res_list = [r for r in res_list if 200 <= r <= 600 and r % 2 == 0 and r != n]
    res_list.sort(key=lambda x: abs(x - n))

    for n_t in res_list[:5]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin = (h_interp > 0.5).astype(float)
        ensure_count(h_bin, n_t)
        t_bi = min(80, int(remaining() * 0.06))
        if t_bi > 15:
            h_b, c5_b = binary_ls_targeted(h_bin, n_t, t_bi, rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_t

    # Phase 6: SA at best resolution
    if remaining() > 60:
        n = best_n
        t_sa2 = min(150, int(remaining() * 0.12))
        h_sa2, c5_sa2 = binary_sa(best_h, n, t_sa2, rng, T0=0.04)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            best_n = n

    # Phase 7: Second round of targeted LS + block LS
    if remaining() > 80:
        n = best_n
        t_l2 = min(120, int(remaining() * 0.1))
        h_l2, c5_l2 = binary_ls_targeted(best_h, n, t_l2, rng)
        if c5_l2 < best_c5:
            best_c5 = c5_l2
            best_h = h_l2.copy()
            best_n = n
        if remaining() > 40:
            t_b2 = min(80, int(remaining() * 0.06))
            h_b2, c5_b2 = block_ls(best_h, best_n, t_b2, rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n

    # Phase 8: Continuous refinement
    n = best_n
    h_cont = best_h.copy()
    for beta in [5000, 20000, 50000]:
        if remaining() < 15:
            break
        try:
            from scipy.optimize import minimize
            h_new = project_feasible(h_cont, n)
            bounds = [(0.0, 1.0)] * n
            constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
            res = minimize(_obj_fun, h_new, jac=_grad_fun, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          args=(n, beta),
                          options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
            h_candidate = project_feasible(res.x, n)
            c5_cand = compute_c5(h_candidate)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_candidate.copy()
                best_n = n
                h_cont = h_candidate.copy()
        except Exception:
            pass

    # Phase 9: Final binary polish
    if remaining() > 30:
        n = best_n
        h_bin_final = (best_h > 0.5).astype(float)
        ensure_count(h_bin_final, n)
        t_final = min(60, int(remaining() * 0.05))
        h_f, c5_f = binary_ls_targeted(h_bin_final, n, t_final, rng)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
