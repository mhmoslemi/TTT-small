from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_OPT = 0
_BETA_OPT = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-13 and n > 0:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _grad_fn(x):

    return np.sum(x) - _N_OPT / 2.0

def _con_jac(x):
    return np.ones(_N_OPT)

def slsqp_optimize(h, n, beta, maxiter=400):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n

    c5_before = compute_c5(h)
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-13:
                return trial
    except Exception:
        pass
    return h.copy()

def coord_descent(h, n, n_sweeps=30):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        improved_any = False
        interior = np.where((best > 1e-10) & (best < 1.0 - 1e-10))[0]
        if len(interior) == 0:
            interior = np.arange(n)

        for i in interior:
            if remaining_time() < 1:
                break
            t0 = best[i]
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n - 1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            base = best_corr - a * t0 - b * t0 * t0

            cands = [0.0, 1.0, t0,
                     max(0.0, t0 - 0.2), min(1.0, t0 + 0.2),
                     max(0.0, t0 - 0.1), min(1.0, t0 + 0.1),
                     max(0.0, t0 - 0.05), min(1.0, t0 + 0.05),
                     1.0 - t0]
            best_t = t0
            best_f = np.max(best_corr) * dx
            for t in cands:
                if abs(t - t0) < 1e-12:
                    continue
                f = np.max(base + a * t + b * t * t) * dx
                if f < best_f - 1e-15:
                    best_f = f
                    best_t = t

            if best_t != t0:
                margin = 0.2
                lo = max(0.0, best_t - margin)
                hi = min(1.0, best_t + margin)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                for _ in range(60):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                t_opt = (lo + hi) / 2.0
                f_opt = np.max(base + a * t_opt + b * t_opt * t_opt) * dx
                if f_opt < best_f:
                    best_f = f_opt
                    best_t = t_opt

                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx

            if s < 1e-10:
                return 10.0
            x_norm = x / s * (n_blocks / 2.0)
            x_norm = np.clip(x_norm, 0, 1)
            h = np.repeat(x_norm, pts_per_block)
            if len(h) < target_sum:
                h = np.concatenate([h, np.zeros(target_sum - len(h))])
            h = h[:target_sum]
            dx = 2.0 / target_sum
            integral = np.sum(h) * dx
            if abs(integral - 1.0) > 0.05:
                return 10.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            return np.max(corr) * dx

        try:
            res = minimize(obj, vals, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-5, 'fatol': 1e-7})
            if res.fun < best_c5:
                best_c5 = res.fun
                best_vals = res.x
        except Exception:
            pass

    if best_vals is None:
        return None
    best_vals = np.clip(best_vals, 0, 1)
    h = np.repeat(best_vals, pts_per_block)
    if len(h) < 640:
        h = interp_to(h, len(h), 640)
    elif len(h) > 640:
        h = interp_to(h, len(h), 640)
    h = project_feasible(h, 640)
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_init = np.zeros(best_n)
        idx = np.arange(0, best_n, 2)
        h_init[idx] = 1.0
        best_h = project_feasible(h_init, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP at increasing beta
    for beta in [500, 2000, 10000, 50000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_optimize(h, n, beta, 500)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: Aggressive coordinate descent
    if remaining_time() > 60:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: SLSQP polish
    for beta in [100000, 500000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_optimize(h, n, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 4: More coordinate descent
    if remaining_time() > 40:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Two-interval parametric optimization (different starting point)
    if remaining_time() > 30:
        ti_result = two_interval_optimize(seed + 1)
        if ti_result is not None:
            ti_h, ti_c5 = ti_result
            ti_c5_actual = compute_c5(ti_h)
            if ti_c5_actual < best_c5:
                best_c5 = ti_c5_actual
                best_h = ti_h.copy()
                best_n = len(ti_h)
                h = ti_h.copy()
                n = best_n

    # Phase 6: Multi-block optimization
    if remaining_time() > 30:
        mb_result = multi_block_optimize(seed + 2, n_blocks=16)
        if mb_result is not None:
            mb_h, mb_c5 = mb_result
            mb_c5_actual = compute_c5(mb_h)
            if mb_c5_actual < best_c5:
                best_c5 = mb_c5_actual
                best_h = mb_h.copy()
                best_n = len(mb_h)
                h = mb_h.copy()
                n = best_n

    # Phase 7: Multi-resolution optimization
    higher_res = []
    for mult in [1.5, 2.0, 1.33]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if 128 <= rn <= 2000:
            higher_res.append(rn)
    for offset in [128, 256, 384]:
        rn = best_n + offset
        if rn % 2 != 0:
            rn += 1
        if 128 <= rn <= 2000:
            higher_res.append(rn)
    higher_res = sorted(set(higher_res))

    for rn in higher_res:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)

        for beta in [5000, 20000]:
            if remaining_time() < 25:
                break
            h_r = slsqp_optimize(h_r, rn, beta, 300)

        if remaining_time() > 30:
            h_r, c5_r = coord_descent_fast(h_r, rn, n_sweeps=50)

        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn
            h = h_r.copy()
            n = rn

        # Also try at lower resolution
        if rn > best_n:
            h_lo = interp_to(h_r, rn, best_n)
            h_lo = project_feasible(h_lo, best_n)
            c5_lo = compute_c5(h_lo)
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = best_n
                h = h_lo.copy()
                n = best_n

    # Phase 8: Random perturbations
    n = best_n
    h = best_h.copy()
    n_perturb = max(8, n // 40)
    for pert in range(20):
        if remaining_time() < 20:
            break
        h_pert = h.copy()
        idx = rng.choice(n, n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.2, 0.2, n_pert)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.003:
            h_new = coord_descent_fast(h_pert, n, n_sweeps=50)
            c5_new = compute_c5(h_new[0])
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new[0].copy()
                h = h_new[0].copy()

    # Phase 9: Final intensive coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = coord_descent_fast(best_h, best_n, n_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 10:
        h_new = slsqp_optimize(best_h, best_n, 1000000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # One more CD round
        if remaining_time() > 10:
            h_new2, c5_new2 = coord_descent_fast(best_h, best_n, n_sweeps=50)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
