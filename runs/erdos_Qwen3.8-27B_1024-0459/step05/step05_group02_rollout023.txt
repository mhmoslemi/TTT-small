def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def set_global(n, beta, topk):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _BETA = beta
    _TOPK = topk
    _TGT = n / 2.0


def sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 200.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 400.0 * (np.sum(x) - _TGT)
    return g


def lp_fun(x, p=20, topk=8):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:topk]
    c = corr[idx]
    c_clipped = np.maximum(c, 1e-12)
    val = (np.sum(c_clipped ** p)) ** (1.0 / p) * dx
    penalty = 200.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def lp_grad(x, p=20, topk=8):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:topk]
    c = corr[idx]
    c_clipped = np.maximum(c, 1e-12)
    cp = c_clipped ** (p - 1)
    norm = (np.sum(c_clipped ** p)) ** ((1.0 / p) - 1.0)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += cp[j] * corr_grad_single(x, ki, n)
    g *= norm * dx
    g += 400.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=6, maxiter=500):
    set_global(n, beta, top_k)
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sm_fun, h, jac=sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxfun': maxiter * 15})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h

            break
        g_max = corr_grad_single(h, idx[0], n)

        d = g_min - g_max
        d -= np.mean(d)

        if dn < 1e-14:

        improved = False

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_grad(h, n, top_k=5, max_iters=3000, step_init=0.02):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx:
            g += corr_grad_single(h, ki, n)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        if c5_new < best_c5 + 0.002:

            if c5_new <= c5:

            step *= 0.75
