        g[:np_] -= h[offset:offset + np_]
    else:

        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]

                        stuck_count = 0
                        break
    
    return best_h, best_max * (2.0 / n)

def binary_sa_search(h, n, max_iters=50000, seed=43, T_init=1.0):
    """Simulated annealing for binary search."""
    rng = np.random.RandomState(seed)
    h = proj_binary(h, n).copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    
    T = T_init
    T_min = 0.05
    cooling = 0.99995
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(new_corr))
        h[i] = 1.0
        h[j] = 0.0
        
        delta = new_max - cur_max
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        
        T = max(T * cooling, T_min)
        if it % 20000 == 19999 and T <= T_min:
            T = T_init * 0.5
    
    return best_h, best_max * (2.0 / n)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        # Count 1s in binary representation
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 0 else 0.0
    # Ensure exactly n/2 ones
    if int(np.sum(h)) != n // 2:
        h = proj_binary(h, n)
    return h

def make_beatty(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        val = (i * alpha) % 1.0
        h[i] = 1.0 if val < 0.5 else 0.0
    h = proj_binary(h, n)
    return h

def make_quasiperiodic(n, freq, phase):
    x = np.arange(n) * 2.0 / n
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x + phase)
    h = proj_binary(h, n)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return np.array(src_h, dtype=float).copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def multi_start_binary_search(h_init, n, seeds, max_iters_each):
    """Try multiple restarts of binary local search."""
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    for s in seeds:
        if remaining() < 10:
            break
        h_bin = proj_binary(h_init, n)
        # Perturb initial
        if s > 42:
            rng_p = np.random.RandomState(s + 100)
            n_ones = n // 2
            for _ in range(min(10, n // 30)):
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
                i = int(ones[rng_p.randint(len(ones))])
                j = int(zeros[rng_p.randint(len(zeros))])
                if i != j:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
        
        h_res, c5_res = binary_local_search(h_bin, n, max_iters=max_iters_each, seed=s, top_k=5)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except (NameError, Exception):
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: Continuous refinement at parent resolution
    n_p = best_n
    h_cont = best_h.copy()
    for beta in [200, 1000, 5000, 20000]:
        if remaining() < 15:
            break
        h_new = lbfgs_pass(h_cont, n_p, beta, top_k=6, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cont = h_new.copy()
    
    # Phase 2: Binary local search at parent resolution
    h_bin_p = proj_binary(best_h, best_n)
    c5_bp = compute_c5(h_bin_p)
    
    # Multi-restart binary search
    seeds_bs = [42, 43, 44, 45, 46]
    iters_each = 30000
    for s in seeds_bs:
        if remaining() < 15:
            break
        h_try = h_bin_p.copy()
        if s > 42:
            rng_p = np.random.RandomState(s * 7 + 3)
            n_ones = best_n // 2
            n_perturb = min(5, best_n // 80)
            for _ in range(n_perturb):
                ones = np.where(h_try == 1)[0]
                zeros = np.where(h_try == 0)[0]
                i = int(ones[rng_p.randint(len(ones))])
                j = int(zeros[rng_p.randint(len(zeros))])
                if i != j:
                    h_try[i] = 0.0
                    h_try[j] = 1.0
        
        h_res, c5_res = binary_local_search(h_try, best_n, max_iters=iters_each, seed=s, top_k=5)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            best_n = best_n
    
    # Phase 3: Try alternative initializations at parent resolution
    alt_inits = []
    # Thue-Morse
    tm = make_thue_morse(best_n)
    alt_inits.append(('tm', tm))
    # Beatty sequences
    for alpha in [0.5 + 1.0/50.0, 0.5 + 1.0/70.0, 1.0/3.0, 2.0/5.0, 0.5 - 1.0/40.0]:
        bt = make_beatty(best_n, alpha)
        alt_inits.append(('beatty', bt))
    # Quasiperiodic
    for f, ph in [(1.618, 0.0), (1.618, 1.5), (2.414, 0.7), (0.764, 0.3), (1.272, 2.1)]:
        qp = make_quasiperiodic(best_n, f, ph)
        alt_inits.append(('qp', qp))
    
    for name, h_alt in alt_inits:
        if remaining() < 20:
            break
        h_alt_bin = proj_binary(h_alt, best_n)
        h_res, c5_res = binary_local_search(h_alt_bin, best_n, max_iters=15000, seed=seed+100+len(alt_inits), top_k=5)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    
    # Phase 4: Try different resolutions
    n_candidates = sorted(set([384, 512, 768, 576, 448, 640]))
    n_candidates = [nn for nn in n_candidates if nn >= 128 and nn % 2 == 0]
    
    for n_t in n_candidates:
        if remaining() < 30:
            break
        
        # Interpolate best solution to new resolution
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = proj_binary(h_interp, n_t)
        
        # Binary local search at this resolution
        h_res_t, c5_res_t = binary_local_search(h_bin_t, n_t, max_iters=20000, seed=seed+n_t, top_k=5)
        
        # Try a few more restarts
        for rs in range(3):
            if remaining() < 10:
                break
            h_pert = h_res_t.copy()
            rng_r = np.random.RandomState(seed + n_t + rs * 31)
            n_perturb = min(5, n_t // 80)
            for _ in range(n_perturb):
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
                i = int(ones[rng_r.randint(len(ones))])
                j = int(zeros[rng_r.randint(len(zeros))])
                if i != j:
                    h_pert[i] = 0.0
                    h_pert[j] = 1.0
            h_r2, c5_r2 = binary_local_search(h_pert, n_t, max_iters=10000, seed=seed+n_t+rs*100, top_k=5)
            if c5_r2 < c5_res_t:
                c5_res_t = c5_r2
                h_res_t = h_r2.copy()
        
        if c5_res_t < best_c5:
            best_c5 = c5_res_t
            best_h = h_res_t.copy()
            best_n = n_t
    
    # Phase 5: Continue binary search on best solution found
    h_bin_best = proj_binary(best_h, best_n)
    if remaining() > 20:
        h_res_b, c5_res_b = binary_local_search(h_bin_best, best_n, max_iters=50000, seed=seed+999, top_k=6)
        if c5_res_b < best_c5:
            best_c5 = c5_res_b
            best_h = h_res_b.copy()
    
    # Phase 6: Simulated annealing on best
    if remaining() > 30:
        h_sa, c5_sa = binary_sa_search(best_h, best_n, max_iters=30000, seed=seed+55, T_init=0.8)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 7: One more directed local search + SA kick cycle
    for cycle in range(3):
        if remaining() < 15:
            break
        # Directed search
        h_dir, c5_dir = binary_local_search(best_h, best_n, max_iters=20000, seed=seed+1000+cycle*7, top_k=7)
        if c5_dir < best_c5:
            best_c5 = c5_dir
            best_h = h_dir.copy()
        # Small SA to escape
        if remaining() > 15:
            h_sa2, c5_sa2 = binary_sa_search(best_h, best_n, max_iters=10000, seed=seed+2000+cycle*11, T_init=1.5)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
    
    # Phase 8: Continuous refinement of best binary solution
    if remaining() > 20:
        h_cont_ref = proj_continuous(best_h, best_n)
        for beta in [5000, 20000, 50000]:
            if remaining() < 10:
                break
            h_cr = lbfgs_pass(h_cont_ref, best_n, beta, top_k=5, maxiter=300)
            c5_cr = compute_c5(h_cr)
            if c5_cr < best_c5:
                best_c5 = c5_cr
                best_h = h_cr.copy()
                h_cont_ref = h_cr.copy()
        # Try binary again after continuous refinement
        h_bin_final = proj_binary(best_h, best_n)
        c5_bin_final = compute_c5(h_bin_final)
        if c5_bin_final < best_c5:
            best_c5 = c5_bin_final
            best_h = h_bin_final.copy()
    
    # Phase 9: Final intensive binary polish
    if remaining() > 15:
        h_final_bin = proj_binary(best_h, best_n)
        h_polish, c5_polish = binary_local_search(h_final_bin, best_n, max_iters=30000, seed=seed+9999, top_k=8)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish.copy()
    
    # Ensure feasibility
    best_h = proj_continuous(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    
    # Also check binary version
    h_chk = proj_binary(best_h, best_n)
    c5_chk = compute_c5(h_chk)
    if c5_chk < best_c5:
        best_c5 = c5_chk
        best_h = h_chk
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
