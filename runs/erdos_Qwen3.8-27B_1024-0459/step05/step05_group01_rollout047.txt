def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0

            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)

        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Snaggle: ensure exact sum
    while abs(np.sum(h) - target) > 1e-14:
        diff = target - np.sum(h)
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        step = min(abs(diff), abs(h[idx] - (0.0 if diff > 0 else 1.0)))
        h[idx] += np.sign(diff) * step
        h = np.clip(h, 0.0, 1.0)

def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)

    S = np.sum(ez)

            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a

            g[a:] -= wk * x[:np_]
