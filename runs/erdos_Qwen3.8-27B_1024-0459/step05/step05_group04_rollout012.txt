def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def binarize(h, n):
    """Round to binary with exactly n/2 ones."""
    n_half = n // 2
    h = np.array(h, dtype=float)
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n_half]] = 1.0
    return hb

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def greedy_construct(n, rng):
    h = np.zeros(n)
    n_half = n // 2
    positions = list(range(n))
    rng.shuffle(positions)

        best_max = float('inf')

def gradient_refine(h, n, time_limit=100, rng=None):
    """Simple gradient descent on continuous relaxation with top-max focus."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start = time.time()
    lr = 0.05
    
    while time.time() - start < time_limit and remaining() > 10:
        corr = compute_corr(h)
        top_k = np.argsort(-corr)[:5]
        
        # Gradient of max(corr[top_k]) w.r.t. h
        g = np.zeros(n)
        for k in top_k:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(top_k)
        
        # Line search
        improved = False
        for step in [lr, lr*0.5, lr*2, lr*0.25]:
            ht = project_feasible(h - step * g, n)
            c5 = compute_c5(ht)
            if c5 < best_c5 - 1e-14:
                h = ht
                best_c5 = c5
                best_h = ht.copy()
                improved = True
                break
        
        if not improved:
            lr *= 0.8
            if lr < 1e-5:
                break
    
    return best_h, best_c5

    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.exp(beta * (corr - m))
    w = shifted / np.sum(shifted)
    
    grad = np.zeros(n)
    # For each x: grad[x] = 1 - sum_k w[k]*h[x-(k-(n-1))] - sum_k w[k]*h[x+(k-(n-1))]
    # Let k' = k - (n-1), so k' in [-(n-1), (n-1)]
    # w[k] = w[k'+(n-1)]
    
    for x in range(n):
        s1 = 0.0
        s2 = 0.0
        for kp in range(-(n-1), n):  # k' from -(n-1) to n-1
            wk = w[kp + (n-1)]
            if wk < 1e-15:
                continue
            idx1 = x - kp  # x - k'
            if 0 <= idx1 < n:
                s1 += wk * h[idx1]
            idx2 = x + kp  # x + k'
            if 0 <= idx2 < n:
                s2 += wk * h[idx2]
        grad[x] = 1.0 - s1 - s2
    
    # Scale by (2/n) for the actual C5 objective
    return grad * (2.0/n), m * (2.0/n)
