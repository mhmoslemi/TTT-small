def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with log-sum-exp smooth max."""
    def _obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * 2.0 / n

    def _grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex) * 2.0 / n
        # Gradient of corr wrt h
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:a + np_])
                g[a:a + np_] -= wk * x[:np_]
        return g

    def _con(x):
        return np.sum(x) - n / 2.0

    def _con_jac(x):
        return np.ones(n)

    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con, 'jac': _con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

            h_t = binarize(h_t, n_t)
            h_t, c5_t = local_search_binary(h_t, n_t, max_iters=50000, use_sa=False, rng=rng)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t
            if remaining() > 20:
                h_t2 = h_t.copy()
                h_t2 = slsqp_smooth(h_t2, n_t, 5000, maxiter=200)
                c5_t2 = compute_c5(h_t2)
                if c5_t2 < best_c5:
                    best_c5 = c5_t2
                    best_h = h_t2.copy()
                    best_n = n_t

    # Phase 7: Random restarts at best resolution
    if remaining() > 60:
        for restart in range(5):
            if remaining() < 20:
                break
            # Perturb best solution
            h_r = best_h.copy()
            n_flip = rng.randint(3, 12)
            ones_r = np.where(h_r > 0.5)[0]
            zeros_r = np.where(h_r <= 0.5)[0]
            if len(ones_r) >= n_flip and len(zeros_r) >= n_flip:
                flip_ps = rng.choice(ones_r, size=n_flip, replace=False)
                flip_qs = rng.choice(zeros_r, size=n_flip, replace=False)
                h_r[flip_ps] = 0.0
                h_r[flip_qs] = 1.0
            h_r = binarize(h_r, best_n)
            h_r, c5_r = local_search_binary(h_r, best_n, max_iters=30000, use_sa=False, rng=rng)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 8: Final polish - intensive local search
    if remaining() > 20:
        h_final = binarize(best_h, best_n)
        h_final, c5_final = local_search_binary(h_final, best_n, max_iters=200000, use_sa=False, rng=rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 9: Final SLSQP polish
    if remaining() > 15:
        h_pol = best_h.astype(float).copy()
        for beta in [10000, 50000, 200000]:
            if remaining() < 8:
                break
            h_pol = slsqp_smooth(h_pol, best_n, beta, maxiter=200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
            else:
                break

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
