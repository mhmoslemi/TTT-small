    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx_v

    dx_v = 2.0 / n

    m = np.max(corr)
    sh = corr - m

    ex = np.exp(cl)

        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])

        return sm_grad(x, n, beta)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def top_shifts_refine(h, n, top_k=5, max_iters=300):
    """Iteratively reduce the top-k worst shifts using gradient direction."""
    h = project_feasible(h, n).copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-10:
            break
        # Compute gradient of the top shift minus gradient of the bottom-of-top shift
        k_top = idx[0]
        k_bot = idx[-1]
        start_shift = n - 1
        off_t = k_top - start_shift
        off_b = k_bot - start_shift

        g_t = np.zeros(n)
        g_b = np.zeros(n)
        if off_t >= 0:
            np_ = n - off_t
            g_t[off_t:off_t + np_] += (1.0 - h[:np_])
            g_t[:np_] -= h[off_t:off_t + np_]
        else:
            ao_t = -off_t
            np_ = n - ao_t
            g_t[:np_] += (1.0 - h[ao_t:ao_t + np_])
            g_t[ao_t:ao_t + np_] -= h[:np_]
        if off_b >= 0:
            np_ = n - off_b
            g_b[off_b:off_b + np_] += (1.0 - h[:np_])
            g_b[:np_] -= h[off_b:off_b + np_]
        else:
            ao_b = -off_b
            np_ = n - ao_b
            g_b[:np_] += (1.0 - h[ao_b:ao_b + np_])
            g_b[ao_b:ao_b + np_] -= h[:np_]

        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        d -= np.mean(d)  # project to tangent of sum constraint

        improved = False
        for step in [0.03, 0.06, 0.015, 0.01, 0.05, 0.08, 0.02, 0.12]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try negative direction
            for step in [0.03, 0.015, 0.06]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

                best_n = n

        # Inverse Thue-Morse
        if remaining() > 30:
            h_tm2 = 1.0 - thue_morse(n, rng)
            h_tm2 = project_feasible(h_tm2, n)
            h_tm2 = slsqp_smooth(h_tm2, n, 3000, 300)
            c5_tm2 = compute_c5(h_tm2)
            if c5_tm2 < best_c5:
                best_c5 = c5_tm2
                best_h = h_tm2.copy()
                best_n = n

        # Random starts
        for rs in range(3):
            if remaining() < 20:
                break
            h_r = rng.uniform(0, 1, n)
            h_r = project_feasible(h_r, n)
            h_r = slsqp_smooth(h_r, n, 5000, 400)
            h_r, c5_r = top_shifts_refine(h_r, n, 4, 100)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n

    # Phase 6: Resolution sweep
    if remaining() > 80:
        n_base = best_n
        n_candidates = []
        for delta in [-4, -2, 2, 4, 6, 8, 12, 16, 24, -8, -12, -16, 32, -32]:
            nt = n_base + delta
            if 200 <= nt <= 600 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [256, 320, 384, 448, 512]:
            if 200 <= nt <= 600 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:6]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t
            h_t = slsqp_smooth(h_t, n_t, 5000, 300)
            h_t, c5_t = top_shifts_refine(h_t, n_t, 4, 100)
            if remaining() > 20 and c5_t < best_c5 + 0.002:
                h_t_bin = binarize(h_t, n_t)
                h_tb, c5_tb = directed_binary_search(h_t_bin, n_t, rng, time_limit=15)
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb.copy()
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for beta in [8000, 20000, 50000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    for tk in [3, 5, 7]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_refine(h, n, top_k=tk, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final directed binary search from best
    if remaining() > 30:
        h_bin_final = binarize(h, n)
        c5_fb = compute_c5(h_bin_final)
        h_df, c5_df = directed_binary_search(h_bin_final, n, rng,
                                              time_limit=min(60, int(remaining() * 0.5)))
        if c5_df < best_c5:
            best_c5 = c5_df
            best_h = h_df.copy()
            best_n = n
            # Quick continuous refinement
            h_cf = slsqp_smooth(h_df, best_n, 10000, 300)
            c5_cf = compute_c5(h_cf)
            if c5_cf < best_c5:
                best_c5 = c5_cf
                best_h = h_cf.copy()

    # Final SLSQP polish
    if remaining() > 10:
        h_final = slsqp_smooth(best_h, best_n, 30000, 400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
