def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[:npairs] += (1.0 - h[offset:offset + npairs])
        g[offset:offset + npairs] -= h[:npairs]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] -= h[a:a + npairs]
        g[a:a + npairs] += (1.0 - h[:npairs])
    return g


def smooth_max_and_grad(h, n, beta, K):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx].copy()
    m = c_top.max()
    shifted = np.clip(beta * (c_top - m), -50.0, 50.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    obj = (m + np.log(exp_vals.sum()) / beta) * (2.0 / n)
    g = np.zeros(n)
    dx = 2.0 / n
    for i in range(K):
        if w[i] < 1e-15:
            continue
        gk = grad_corr_k(h, idx[i], n)
        g += (w[i] * dx) * gk
    return obj, g


def lbfgsb_optimize(h, n, beta, K, maxiter=300):

def slsqp_optimize(h, n, beta, K, maxiter=500):
    def sm_obj(x):
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:K]
        ct = corr[idx]
        m = ct.max()
        sh = np.clip(beta * (ct - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(ex.sum()) / beta) * (2.0 / n)

    def sm_grad(x):
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:K]
        ct = corr[idx]
        m = ct.max()
        sh = np.clip(beta * (ct - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / ex.sum()
        g = np.zeros(n)
        for i in range(K):
            if w[i] < 1e-15:
                continue
            g += (w[i] * (2.0 / n)) * grad_corr_k(x, idx[i], n)
        return g

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,

    except Exception:
        pass
    return h

def cutting_plane_iteration(h, n, K=25, delta=0.06):

    c_max = float(np.max(corr))
    idx = np.argsort(-corr)[:K]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub_list = []
    b_ub_list = []

    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g, h))

    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, h + delta])

    A_ub = np.array(A_ub_list + list(A_trust))
    b_ub = np.array(b_ub_list + list(b_trust))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0, 1)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        pass

    except Exception:

            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
