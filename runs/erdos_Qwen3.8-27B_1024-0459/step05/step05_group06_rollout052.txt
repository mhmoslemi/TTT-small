For each lag m (0 to 2n-2), with off = m - (n-1):
- Let valid1 = (0 <= i+off < n): if true, the term h[i]*(1-h[i+off]) contributes
- Let valid2 = (0 <= i-off < n): if true, the term h[i-off]*(1-h[i]) contributes

When h[i] = t:
- Term1 (if valid1): t * (1 - h[i+off])
- Term2 (if valid2): h[i-off] * (1-t) = h[i-off] - h[i-off]*t

So corr[m] = other_terms + [valid1 * t*(1-h[i+off])] + [valid2 * (h[i-off] - h[i-off]*t)]
           = (other_terms + valid2 * h[i-off]) + t * [valid1*(1-h[i+off]) - valid2*h[i-off]]

Let A[m] = valid1*(1-h[i+off]) - valid2*h[i-off]
Let B[m] = other_terms + valid2*h[i-off]

Then corr[m] = B[m] + A[m]*t

And B[m] = current_corr[m] - A[m]*t0 (since current h[i]=t0)

So: obj(t) = max_m (B[m] + A[m]*t) * dx

To find the minimum of this convex piecewise-linear function over [0,1]:
- Evaluate at t=0: f0 = max(B) * dx
- Evaluate at t=1: f1 = max(B+A) * dx
- The minimum is at t=0, t=1, or at a breakpoint

For the breakpoint: find t where the argmax changes. We can do this with a "convex 1D" approach:
- The derivative (subgradient) at t is A[argmax_m(B[m]+A[m]*t)]
- Find t* where the subgradient crosses 0

- At t=0: slope = A[argsort(B)[-1]] (the line with max B)

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12 and n > 0:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff2 / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def slsqp_optimize(h, n, beta, maxiter=400):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:

                      bounds=bounds, constraints=cons,

        order = rng_cd.permutation(n)
        
        for i in order:
            if remaining_time() < 0.3:
                break
            t0 = best[i]
            
            # Compute A for each lag
            A = np.zeros(ncorr)
            for m in range(ncorr):
                off = m - (n - 1)
                a_val = 0.0
                if 0 <= i + off < n:
                    a_val += (1.0 - best[i + off])
                if 0 <= i - off < n:
                    a_val += (-best[i - off])
                A[m] = a_val
            B = corr - A * t0
            
            # Quick check: evaluate at 0, 1, t0
            f0 = np.max(B) * dx
            f1 = np.max(B + A) * dx
            ft0 = np.max(B + A * t0) * dx
            
            best_t = t0
            best_f = ft0
            
            if f0 < best_f - 1e-15:
                best_f = f0
                best_t = 0.0
            if f1 < best_f - 1e-15:
                best_f = f1
                best_t = 1.0
            
            # Golden section if not at boundary
            if best_t > 0.01 and best_t < 0.99:
                # Determine search window based on where the max is
                active_at_0 = B > np.max(B) - 0.01 * np.std(B)
                active_at_1 = (B + A) > np.max(B + A) - 0.01 * np.std(B + A)
                
                # Just do a moderate golden section
                lo = max(0.0, best_t - 0.3)
                hi = min(1.0, best_t + 0.3)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                fc1 = np.max(B + A * c1) * dx
                fc2 = np.max(B + A * c2) * dx
                for _ in range(50):
                    if hi - lo < 1e-13:
                        break
                    if fc1 < fc2:
                        hi = c2
                        c2 = c1
                        fc2 = fc1
                        c1 = hi - gr * (hi - lo)
                        fc1 = np.max(B + A * c1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        fc1 = fc2
                        c2 = lo + gr * (hi - lo)
                        fc2 = np.max(B + A * c2) * dx
                t_gold = (lo + hi) / 2.0
                f_gold = np.max(B + A * t_gold) * dx
                if f_gold < best_f - 1e-15:
                    best_f = f_gold
                    best_t = t_gold
            
            if best_f < best_c5 - 1e-14:
                best[i] = best_t
                corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved_any = True
        
        if not improved_any:
            break
    
    return best, best_c5

def binary_snap(h, n):
    """Snap h to nearest binary sequence with correct count."""
    h_bin = (h > 0.5).astype(float)
    cnt = int(np.sum(h_bin))
    target = n // 2
    if cnt > target:
        ones = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones])
        for idx in order[:cnt - target]:
            h_bin[ones[idx]] = 0.0
    elif cnt < target:
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(h[zeros])[::-1]
        for idx in order[:target - cnt]:
            h_bin[zeros[idx]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, max_iters=3000, seed=789):
    """Local search on binary sequences by flipping pairs."""
    rng = np.random.RandomState(seed)
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    
    ones = np.where(best == 1.0)[0]
    zeros = np.where(best == 0.0)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    stall = 0
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        
        best_improve = best_c5
        best_pair = None
        n_trials = min(300, n_ones * n_zeros)
        
        for _ in range(n_trials):
            ii = rng.randint(n_ones)
            jj = rng.randint(n_zeros)
            i = ones[ii]
            j = zeros[jj]
            if i == j:
                continue
            trial = best.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            c = compute_c5(trial)
            if c < best_improve - 1e-16:
                best_improve = c
                best_pair = (i, j)
        
        if best_pair is not None:
            best[best_pair[0]] = 0.0
            best[best_pair[1]] = 1.0
            best_c5 = best_improve
            ones = np.where(best == 1.0)[0]
            zeros = np.where(best == 0.0)[0]
            n_ones = len(ones)
            n_zeros = len(zeros)
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break
    
    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_OPT, _BETA_OPT
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

        parent_c5 = compute_c5(parent_h)

        parent_h = np.zeros(parent_n)
        idx = np.random.RandomState(0).choice(parent_n, parent_n // 2, replace=False)
        parent_h[idx] = 1.0
        parent_c5 = compute_c5(parent_h)
    
    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n
    n = best_n
    h = best_h.copy()
    
    # Phase 1: SLSQP at parent resolution
    for beta in [2000, 10000, 50000, 200000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_optimize(h, n, beta, 400)
        c5_new = compute_c5(h_new)

            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    if remaining_time() > 100:
        h_new, c5_new = fast_coord_descent(h, n, n_sweeps=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 3: SLSQP polish
    for beta in [100000, 500000, 2000000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_optimize(h, n, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    # Phase 4: More coordinate descent
    if remaining_time() > 80:
        h_new, c5_new = fast_coord_descent(h, n, n_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Binary snap and local search
    if remaining_time() > 200:
        h_bin = binary_snap(best_h, best_n)
        h_bin, c5_bin = binary_local_search(h_bin, best_n, max_iters=5000, seed=1)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h = h_bin.copy()
            # Re-optimize continuously from binary
            h_new = slsqp_optimize(h, best_n, 100000, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            if remaining_time() > 60:
                h_new, c5_new = fast_coord_descent(h, best_n, n_sweeps=100)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
    
    # Phase 6: Higher resolution
    for target_n in [960, 1280]:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, target_n)
        h_r = project_feasible(h_r, target_n)
        c5_r = compute_c5(h_r)
        
        for beta in [5000, 50000, 200000]:
            if remaining_time() < 20:
                break
            h_r = slsqp_optimize(h_r, target_n, beta, 300)

            h_r, c5_r = fast_coord_descent(h_r, target_n, n_sweeps=50)
        
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = target_n
    
    # Phase 7: Random perturbations
    n = best_n
    h = best_h.copy()
    n_perturb = max(8, n // 40)
    for pert in range(20):
        if remaining_time() < 30:
            break
        h_pert = h.copy()
        idx = rng.choice(n, n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.1, 0.1, n_pert)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.003:
            h_new = slsqp_optimize(h_pert, n, 50000, 200)
            c5_new = compute_c5(h_new)
            if remaining_time() > 20:
                h_new, c5_new = fast_coord_descent(h_new, n, n_sweeps=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    if remaining_time() > 30:
        h_new, c5_new = fast_coord_descent(best_h, best_n, n_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 9: Final SLSQP
    if remaining_time() > 10:
        h_new = slsqp_optimize(best_h, best_n, 1000000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 10: One more CD pass
    if remaining_time() > 8:
        h_new, c5_new = fast_coord_descent(best_h, best_n, n_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Ensure exact sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
