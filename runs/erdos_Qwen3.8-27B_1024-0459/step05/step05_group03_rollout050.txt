    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue

        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]

            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    w = w * dx
    return corr_grad_weighted(h, w)


def slsqp_opt(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    best_x = h0.copy()
    best_f = _obj(h0)
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < best_f - 1e-15:
            best_x = project_feasible(res.x, n)
    except Exception:
        pass
    return best_x


def sa_binary(h_bin, n, n_iter=30000, T0=3.0, T1=0.05, focus=True):
    """Simulated annealing on binary h with informed swap moves."""
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    rng_local = np.random.RandomState(int(time.time() * 1000) % 100000)

    curr = T0
    alpha = (T1 / T0) ** (1.0 / max(n_iter, 1))

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    for it in range(n_iter):
        if remaining() < 10:
            break
        if it % 5000 == 0:
            curr = max(T1, curr * (alpha ** 5000))

        if focus and it < n_iter * 0.5:
            # Informed: focus on positions near max-correlation shifts
            corr = compute_corr(h)
            top_shifts = np.argsort(-corr)[:15]
            # Gather positions involved in top correlations
            involved = set()
            for ks in top_shifts:
                off = ks - (n - 1)
                if off >= 0:
                    for i in range(min(30, n - off)):
                        involved.add(i)
                        involved.add(i + off)
                else:
                    a = -off
                    for i in range(min(30, n - a)):
                        involved.add(i + a)
                        involved.add(i)
            inv_arr = np.array(list(involved) + [0])
            pa = inv_arr[rng_local.randint(len(inv_arr))]
            if h[pa] == 0:
                pb = inv_arr[rng_local.randint(len(inv_arr))]
                while h[pb] == 1 or pb == pa:
                    pb = inv_arr[rng_local.randint(len(inv_arr))]
            else:
                pb = inv_arr[rng_local.randint(len(inv_arr))]
                while h[pb] == 0 or pb == pa:
                    pb = inv_arr[rng_local.randint(len(inv_arr))]
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            pa = ones[rng_local.randint(len(ones))]
            pb = zeros[rng_local.randint(len(zeros))]

        h[pa] = 0.0
        h[pb] = 1.0
        new_c5 = compute_c5(h)

        delta = new_c5 - cur_c5
        if delta < 0 or rng_local.rand() < np.exp(-delta / max(curr, 1e-10)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[pa] = 1.0
            h[pb] = 0.0

    return best_h, best_c5


def binary_ls_multi(h_bin, n, max_rounds=30):
    """Binary local search with swap + block moves."""
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if remaining() < 15:
            break
        corr_cur = compute_corr(h)
        c_max = np.max(corr_cur)
        best_improve = 1e-15
        best_action = None

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        # Find boundary positions (transitions)
        trans_1 = []
        trans_0 = []
        for i in range(n):
            if h[i] == 1 and (i == 0 or h[i - 1] == 0):
                trans_1.append(i)
            if h[i] == 1 and (i == n - 1 or h[i + 1] == 0):
                trans_1.append(i)
            if h[i] == 0 and (i == 0 or h[i - 1] == 1):
                trans_0.append(i)
            if h[i] == 0 and (i == n - 1 or h[i + 1] == 1):
                trans_0.append(i)

        cands_1 = list(set(trans_1))[:40]
        cands_0 = list(set(trans_0))[:40]

        # Also add some random positions for diversity
        if len(cands_1) < 20:
            cands_1 += list(rng_global.randint(0, n, 10 - len(cands_1) // 2))
        if len(cands_0) < 20:
            cands_0 += list(rng_global.randint(0, n, 10 - len(cands_0) // 2))

        # Try swaps
        for pa in cands_1:
            if h[pa] != 1:
                continue
            h[pa] = 0.0
            for pb in cands_0:
                if pb == pa or h[pb] != 0:
                    continue
                h[pb] = 1.0
                nc = np.max(compute_corr(h))
                h[pb] = 0.0
                imp = c_max - nc
                if imp > best_improve:
                    best_improve = imp
                    best_action = ('swap', pa, pb)
            h[pa] = 1.0

        # Try block moves (shift small blocks of 1s)
        blocks = []
        i = 0
        while i < n:
            if h[i] == 1:
                j = i
                while j + 1 < n and h[j + 1] == 1:
                    j += 1
                if j - i + 1 <= 4:
                    blocks.append((i, j))
                i = j + 1
            else:
                i += 1

        for (bi, bj) in blocks[:30]:
            blen = bj - bi + 1
            for shift in [-4, -3, -2, -1, 1, 2, 3, 4]:
                ni, nj = bi + shift, bj + shift
                if ni < 0 or nj >= n:
                    continue
                if np.any(h[ni:nj + 1] == 1):
                    continue
                if ni <= bj and nj >= bi:
                    continue
                h[bi:bj + 1] = 0.0
                h[ni:nj + 1] = 1.0
                nc = np.max(compute_corr(h))
                h[ni:nj + 1] = 0.0
                h[bi:bj + 1] = 1.0
                imp = c_max - nc
                if imp > best_improve:
                    best_improve = imp
                    best_action = ('block', bi, bj, shift)

        if best_action is not None:
            if best_action[0] == 'swap':
                _, pa, pb = best_action
                h[pa] = 0.0
                h[pb] = 1.0
            else:
                _, bi, bj, shift = best_action
                h[bi:bj + 1] = 0.0
                h[bi + shift:bj + shift + 1] = 1.0

            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_smooth(h, n, beta, max_iters=2000):
    h = project_feasible(h, n)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0

    for it in range(max_iters):
        if remaining() < 5:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha_n = (t_val - 1.0) / t_new
        y_new = y + alpha_n * (y - h)
        g = smooth_grad(h, beta)
        lr = 1.0 / (beta * 0.1 * (it + 10))
        lr = min(lr, 0.5)
        h_new = project_feasible(y_new - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new

    return best_h, best_c5
