def corr_grad_at_k(h, k, n):
    """Gradient of corr[k] with respect to h."""
    g = np.zeros(n)

    if off >= 0:

        g[off:off+m] += (1.0 - h[:m])

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def corr_grad_k(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        m = n - off
        g[off:off+m] += (1.0 - h[:m])
        g[:m] -= h[off:off+m]
    else:
        a = -off
        m = n - a
        g[:m] += (1.0 - h[a:a+m])
        g[a:a+m] -= h[:m]
    return g

def slsqp_opt(h, n, beta, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0
    
    def _obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        s = corr - m
        c = np.clip(beta * s, -50.0, 50.0)
        return (m + np.log(np.sum(np.exp(c))) / beta) * dx
    
    def _grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        s = corr - m
        c = np.clip(beta * s, -50.0, 50.0)
        ex = np.exp(c)
        w = ex / np.sum(ex) * dx
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk > 1e-14:
                g += wk * corr_grad_k(x, k, n)
        return g
    
    def _cfun(x):
        return np.sum(x) - target
    
    def _cjac(x):
        return np.ones(n)
    
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:

                       bounds=bounds, constraints=cons,

        T = max(T * 0.9998, T_min)
        corr = compute_corr(h)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

            else:
