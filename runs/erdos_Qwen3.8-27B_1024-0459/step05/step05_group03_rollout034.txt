def project_feasible(h, n):

            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:

        h = np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)
    ex = np.exp(shifted)
    sx = np.sum(ex)
    w = ex / sx
    dx = 2.0 / n
    obj_val = (m + np.log(sx) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for j in range(2*n - 1):
        wk = w[j] * dx
        if wk < 1e-16:
            continue
        s = j - (n - 1)
        # d(corr_s)/d(h[i]) = (1-h[i-s]) * [0≤i-s<n] - h[i+s] * [0≤i+s<n]
        # First part: (1-h[i-s]) for i in [max(0,s), min(n, s+n))
        i1 = max(0, s)
        i2 = min(n, s + n)
        if i1 < i2:
            idx = np.arange(i1, i2)
            grad[idx] += wk * (1.0 - h[idx - s])
        # Second part: -h[i+s] for i in [max(0,-s), min(n, n-s))
        i3 = max(0, -s)
        i4 = min(n, n - s)
        if i3 < i4:
            idx2 = np.arange(i3, i4)
            grad[idx2] -= wk * h[idx2 + s]
    return obj_val, grad

        _, g = lse_obj_and_grad(x, beta)
        return g

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def lse_obj_and_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    sx = np.sum(ex)
    w = ex / sx
    dx = 2.0 / n
    obj_val = (m + np.log(sx) / beta) * dx
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        wk = w[j] * dx
        if wk < 1e-16:
            continue
        s = j - (n - 1)
        i1 = max(0, s)
        i2 = min(n, s + n)
        if i1 < i2:
            idx = np.arange(i1, i2)
            grad[idx] += wk * (1.0 - h[idx - s])
        i3 = max(0, -s)
        i4 = min(n, n - s)
        if i3 < i4:
            idx2 = np.arange(i3, i4)
            grad[idx2] -= wk * h[idx2 + s]
    return obj_val, grad


def slsqp_opt(h, n, beta, maxiter=300):
    h0 = project_feasible(h, n)
    obj_at_0 = lse_obj_and_grad(h0, beta)[0]

    def _obj(x):
        return lse_obj_and_grad(x, beta)[0]

    def _grad(x):
        return lse_obj_and_grad(x, beta)[1]

    def _con_fun(x):
        return np.sum(x) - n / 2.0

    def _con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj_at_0 - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def paired_cd_stochastic(h, n, n_rounds=80, pairs_per_round=2000, grid_size=20, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stagnation = 0

    for rnd in range(n_rounds):
        if remaining() < 10:
            break
        round_improved = False
        all_indices = np.arange(n)
        for _ in range(pairs_per_round):
            if remaining() < 5:
                break
            i, j = rng.choice(all_indices, size=2, replace=False)
            S = h[i] + h[j]
            if S < 1e-10 or S > 2.0 - 1e-10:
                continue
            lo = max(0.0, S - 1.0)
            hi = min(1.0, S)
            if hi - lo < 1e-10:
                continue
            grid = np.linspace(lo, hi, grid_size)
            cur_val = compute_c5(h)
            best_t = h[i]
            best_grid_val = cur_val
            for t in grid:
                h[i] = t
                h[j] = S - t
                val = compute_c5(h)
                if val < best_grid_val:
                    best_grid_val = val
                    best_t = t
            h[i] = best_t
            h[j] = S - best_t
            if best_grid_val < best_c5 - 1e-14:
                best_c5 = best_grid_val
                best_h = h.copy()
                round_improved = True

        if round_improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 15:
                # Try random perturbation to escape
                for _ in range(20):
                    i, j = rng.choice(all_indices, size=2, replace=False)
                    S = h[i] + h[j]
                    lo = max(0.0, S - 1.0)
                    hi = min(1.0, S)
                    t = rng.uniform(lo, hi)
                    h[i] = t
                    h[j] = S - t
                    val = compute_c5(h)
                    if val < best_c5:
                        best_c5 = val
                        best_h = h.copy()
                        stagnation = 0
                        break
                if stagnation > 25:
                    break

    return best_h, best_c5


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin
