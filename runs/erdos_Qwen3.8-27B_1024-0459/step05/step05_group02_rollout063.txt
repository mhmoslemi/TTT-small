def proj(h, n):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=n/2}"""
    h = np.array(h, dtype=float).copy()

        diff = target - np.sum(h)

            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

        h = np.clip(h, 0.0, 1.0)

    """Gradient of corr[k] (full correlate index k) w.r.t. h."""

    else:
        ao = -offset

def fourier_perturb(h, n, rng, num_modes=3, max_freq=5):

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n

    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def sm_objective(x):

    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 200.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def sm_gradient(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]

    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)

    g = np.zeros(n)

        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 400.0 * (np.sum(x) - _TGT) * np.ones(n)

def threshold_optimize(h, n, target_c5, max_iters=3000, lr=0.002):

        viol_idx = np.where(viol > 1e-8)[0]

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse_init(n):
    """Generate Thue-Morse based binary initialization."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    h = (1 - tm).astype(float)
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones > n_half:
        excess = n_ones - n_half
        one_idx = np.where(h == 1)[0]
        to_flip = one_idx[:excess]
        h[to_flip] = 0
    elif n_ones < n_half:
        deficit = n_half - n_ones
        zero_idx = np.where(h == 0)[0]
        to_flip = zero_idx[:deficit]
        h[to_flip] = 1
    return proj(h, n)


def fourier_perturb(h, n, rng, num_modes=3, max_amp=0.04):
    h_out = h.copy()
    x = (np.arange(n) + 0.5) * 2.0 / n
    for _ in range(num_modes):
        f = rng.randint(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.005, max_amp)
        h_out += amp * np.cos(2 * np.pi * f * x + phase)
    return proj(h_out, n)
