def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -80.0, 80.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        return m * dx_v
    return (m + np.log(s) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -80.0, 80.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-300:
        w = np.full(len(corr), 1.0 / len(corr))
    else:
        w = ex / s
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

    return np.sum(x) - _TGT

    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=500):

    return h


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def binarize(h, n):

def corr_gradient_at(h, k_off, n):
    g = np.zeros(n)
    if k_off >= 0:
        np_ = n - k_off
        g[k_off:k_off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def top_shifts_smooth(h, n, top_k=5, max_iters=400):
    """Smooth direction: reduce top correlations via projected gradient."""
    h = project_feasible(h, n).copy()

    step = 0.03
    for it in range(max_iters):

        for k in top_idx:
            off = k - (n - 1)
            g += corr_gradient_at(h, off, n)
        g /= len(top_idx)
        g -= np.mean(g)  # project off sum constraint direction
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        improved = False
        for s in [step, step * 0.5, step * 2, step * 0.25, step * 0.1]:

                best_c5 = c_new
                improved = True
                break

    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

            c5_t = compute_c5(h_t)
            c5_tm = compute_c5(h_tm)
            if c5_tm < c5_t:
                h_t = h_tm
                c5_t = c5_tm
            if c5_t < best_c5 + 0.005:
                h_opt, c5_opt = optimize_block(h_t, n_try, rng, time_limit=min(50, int(remaining() * 0.3)))
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_try

    # Final polish
    if remaining() > 10:
        h_final = best_h.copy()
        n_final = best_n
        for beta in [30000, 80000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(h_final, n_final, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
