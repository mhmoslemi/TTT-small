One idea: **convex-concave procedure (CCP)**.
The objective max_k ∑_i h[i]*(1-h[i+k])*dx can be written as:
max_k [∑_i h[i]*dx - ∑_i h[i]*h[i+k]*dx]
= 1 - max_k ∑_i h[i]*h[i+k]*dx (since ∑_i h[i]*dx = 1)

Actually, one more key insight: I should try to **directly compute the exact minimizer** for each coordinate rather than using golden section. 

For coordinate i, f(t) = max(max_{k≠n-1} [b_k + A_k * t], b_{n-1} + t - t^2)

The minimum of this over t ∈ [0,1] occurs at one of:
1. t = 0 or t = 1
2. The point where the quadratic term equals the maximum affine term
3. A point where two affine terms cross (but this is a maximum, so it's a "peak" not a "valley")

Actually, f(t) = max of several piecewise-linear (affine) functions and one concave quadratic. The minimum of f over [0,1] occurs at:
- An endpoint (0 or 1)
- A point where the quadratic tg[t] = b_{n-1} + t - t^2 equals some affine function b_k + A_k * t
- Possibly where the maximum affine function changes identity (but at such a point, f is continuous and the derivative changes sign from negative to positive only if the quadratic is not active)

The key observation: f(t) is minimized where the "tightest" constraint is balanced. In practice, the optimum is often where the quadratic term equals the maximum of the affine terms.

The equation t - t^2 = b_k + A_k * t - b_{n-1} is:
t^2 + (A_k - 1) * t + (b_k - b_{n-1}) = 0

This gives t = [-(A_k-1) ± sqrt((A_k-1)^2 - 4(b_k-b_{n-1}))] / 2

We could check all such candidate points plus the endpoints. But there are O(n) affine terms, so O(n) candidates, and evaluating f at each is O(n), giving O(n^2) per coordinate. That's too slow.

Better approach: Use ternary/golden section but with fewer iterations (20-30 should be enough for 1e-8 precision on [0,1]).

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 1000.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def _g_smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]


def _g_smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]


def _g_constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0


def _g_constraint_sum_jac(x):
    return np.ones(_N_WORK)


def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    try:
        res = minimize(_g_smooth_obj, h, jac=_g_smooth_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coord_descent(h, n, max_sweeps=200):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 5.0:
            break
        improved = False
        for i in range(n):
            old = best[i]
            if old < 1e-15 or old > 1.0 - 1e-15:
                pass

            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)
            f_old = np.max(best_corr)

            # Evaluate at key candidates
            t_best = old
            f_best = f_old

            # t=0
            f0 = max(base[n - 1], np.max(base))
            if f0 < f_best - 1e-14:
                f_best = f0
                t_best = 0.0

            # t=1
            f1 = max(base[n - 1], np.max(base + A))
            if f1 < f_best - 1e-14:
                f_best = f1
                t_best = 1.0

            # t=0.5
            fmid = max(base[n - 1] + 0.25, np.max(base + 0.5 * A))
            if fmid < f_best - 1e-14:
                f_best = fmid
                t_best = 0.5

            # Golden section search converging toward t_best
            if t_best != old:
                if t_best > old:
                    lo = max(0.0, old - 0.1)
                    hi = min(1.0, t_best + 0.3)
                else:
                    lo = max(0.0, t_best - 0.3)
                    hi = min(1.0, old + 0.1)
                if hi - lo < 1e-10:
                    hi = lo + 1e-6

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                mid1 = base + A * c1
                mid1[n - 1] = base[n - 1] + c1 - c1 * c1
                f1v = np.max(mid1)

                mid2 = base + A * c2
                mid2[n - 1] = base[n - 1] + c2 - c2 * c2
                f2v = np.max(mid2)

                for _ in range(35):
                    if f1v < f2v:
                        hi = c2
                        c2 = c1
                        f2v = f1v
                        c1 = hi - gr * (hi - lo)
                        mid1 = base + A * c1
                        mid1[n - 1] = base[n - 1] + c1 - c1 * c1
                        f1v = np.max(mid1)
                    else:
                        lo = c1
                        c1 = c2
                        f1v = f2v
                        c2 = lo + gr * (hi - lo)
                        mid2 = base + A * c2
                        mid2[n - 1] = base[n - 1] + c2 - c2 * c2
                        f2v = np.max(mid2)

                t_best = (lo + hi) / 2.0
                mid = base + A * t_best
                mid[n - 1] = base[n - 1] + t_best - t_best * t_best
                f_best = np.max(mid)

            if f_best < f_old - 1e-13:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break

    return best, best_c5


def block_coord_descent(h, n, max_sweeps=50, block_size=2):
    """Optimize blocks of coordinates with grid + refine."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 5.0:
            break
        improved = False
        for i in range(0, n - block_size + 1, block_size):
            idx = list(range(i, i + block_size))
            old_vals = best[idx].copy()
            f_old = np.max(best_corr)

            # Compute A for each coordinate in block
            A_list = []
            base_list = []
            for ii in idx:
                i_plus = ii + offsets
                i_minus = ii - offsets
                vp = (i_plus >= 0) & (i_plus < n)
                vm = (i_minus >= 0) & (i_minus < n)
                A = np.zeros(ncorr)
                if np.any(vm):
                    A[vm] += (1.0 - best[i_minus[vm]])
                if np.any(vp):
                    A[vp] -= best[i_plus[vp]]
                A[n - 1] = 0.0
                A_list.append(A)

            # Grid search over [0,1]^block_size with coarse grid
            grid = [0.0, old_vals[0], 1.0]
            best_f = f_old
            best_t = old_vals.copy()

            for t0 in grid:
                mid = best_corr.copy()
                for j, ii in enumerate(idx):
                    t = t0 if j == 0 else old_vals[j]
                    mid += A_list[j] * (t - old_vals[j])
                    mid[n - 1] += t * (1 - t) - old_vals[j] * (1 - old_vals[j])
                f_new = np.max(mid)
                if f_new < best_f - 1e-14:
                    best_f = f_new
                    best_t = np.array(idx_vals_copy(old_vals, t0, 0))

            # Refine with small perturbations
            for _ in range(20):
                new_best = False
                for di in [-0.1, 0.1, -0.05, 0.05]:
                    t_new = best_t + di
                    if 0 <= t_new[0] <= 1:
                        mid = best_corr.copy()
                        for j, ii in enumerate(idx):
                            t = t_new[j]
                            mid += A_list[j] * (t - old_vals[j])
                            mid[n - 1] += t * (1 - t) - old_vals[j] * (1 - old_vals[j])
                        f_new = np.max(mid)
                        if f_new < best_f - 1e-14:
                            best_f = f_new
                            best_t = t_new.copy()
                            new_best = True
                if not new_best:
                    break

            if best_f < f_old - 1e-13:
                for j, ii in enumerate(idx):
                    best[ii] = best_t[j]
                    best_corr += A_list[j] * (best_t[j] - old_vals[j])
                    best_corr[n - 1] += best_t[j] * (1 - best_t[j]) - old_vals[j] * (1 - old_vals[j])
                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break

    return best, best_c5


def idx_vals_copy(old_vals, new_val, pos):
    out = old_vals.copy()
    out[pos] = new_val
    return out


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def subgradient_step(h, n, step_size, n_active=10):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    threshold = max_corr - 0.3
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        idx = active[np.argsort(corr[active])[::-1][:n_active]]
        active = idx
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h
    grad /= gnorm
    h_new = h - step_size * grad
    return project_feasible(h_new, n)
