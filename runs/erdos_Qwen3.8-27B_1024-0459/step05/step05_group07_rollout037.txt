```python
    return np.sum(x) - _N_WORK / 2.0

    return np.ones(_N_WORK)

def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    try:
        res = minimize(_g_smooth_obj, h, jac=_g_smooth_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def fast_coord_descent(h, n, max_sweeps=200):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))

            # Candidates
            t_candidate = 0.5
            mid = base + A * t_candidate
            mid[n - 1] = base[n - 1] + t_candidate - t_candidate * t_candidate
            f_cand = np.max(mid)
            
            t_best = old
            f_best = f_old
            if f_cand < f_best - 1e-14:
                t_best = t_candidate
                f_best = f_cand

            f_0 = max(base[n - 1], np.max(base))
            if f_0 < f_best - 1e-14:
                t_best = 0.0
                f_best = f_0
            f_1 = max(base[n - 1], np.max(base + A))
            if f_1 < f_best - 1e-14:
                t_best = 1.0
                f_best = f_1

            # Golden section
            if t_best != old:
                if t_best > old:
                    lo = old
                    hi = min(1.0, t_best + 0.3)
                else:
                    lo = max(0.0, t_best - 0.3)
                    hi = old

                if hi - lo < 1e-12:
                    hi = lo + 1e-8

                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                f1 = max(base + A * c1, base + A * c1) # won't work this way
                # Need to handle the parabolic term
                mid1 = base + A * c1
                mid1[n - 1] = base[n - 1] + c1 - c1 * c1
                f1 = np.max(mid1)
                
                mid2 = base + A * c2
                mid2[n - 1] = base[n - 1] + c2 - c2 * c2
                f2 = np.max(mid2)

                for _ in range(50):
                    if hi - lo < 1e-12:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        mid1 = base + A * c1
                        mid1[n - 1] = base[n - 1] + c1 - c1 * c1
                        f1 = np.max(mid1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        mid2 = base + A * c2
                        mid2[n - 1] = base[n - 1] + c2 - c2 * c2
                        f2 = np.max(mid2)
                
                t_best = (lo + hi) / 2.0
                mid_t = base + A * t_best
                mid_t[n - 1] = base[n - 1] + t_best - t_best * t_best
                f_best = np.max(mid_t)

            if f_best < f_old - 1e-13:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break

    return best, best_c5

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)
```

```python
    grad *= dx
    return grad
```

Actually, I realize I should also try a completely different approach: using `scipy.optimize.linprog` or `cvxpy` for a relaxed version of the problem. But since the problem is non-convex, this won't directly work. However, I could use a linearization approach:

At the current solution h*, linearize the correlation:
c_k ≈ c_k* + sum_i (dc_k/dh[i]) * (h[i] - h*[i])

Then solve:
min t
s.t. c_k* + sum_i (dc_k/dh[i]) * (h[i] - h*[i]) <= t for all k
     sum h[i] = n/2
     0 <= h[i] <= 1

This is a linear program! I can solve it with linprog or cvxpy.

The LP gives a direction to move. I then take a line search in that direction and repeat (this is essentially a cutting-plane or outer approximation method for the non-convex problem).

This could be much faster than SLSQP because each LP is solved in seconds (or less) using a dedicated LP solver.

Let me implement this:

```python
def lp_step(h, n, max_slacks=100):
    """One linear programming step: linearize and solve LP."""
    from scipy.optimize import linprog
    dx = 2.0 / n
    corr = compute_corr(h)
    ncorr = len(corr)
    
    # Compute gradient of each correlation w.r.t. h
    # This is a (ncorr x n) matrix
    # G[k, i] = dc_k / dh[i]
    # From the formula: G[k, i] = (1-h[i+offset_k]) * I(valid) - h[i-offset_k] * I(valid)
    # where offset_k = k - (n-1)
    
    # Build the LP:
    # min t
    # s.t. t - sum_i G[k,i] * h[i] >= c_k - sum_i G[k,i] * h[i] (for all k)
    #      sum h[i] = n/2
    #      0 <= h[i] <= 1
    
    # Actually, let me reformulate:
    # min t
    # s.t. sum_i G[k,i] * h[i] - t <= c_k^lin(k) for all k
    #      where c_k^lin(k) = c_k - sum_i G[k,i] * h[i] + sum_i G[k,i] * h[i]
    #      = c_k (the current value)
    #      Wait, the linearization is: c_k(h) ≈ c_k(h*) + sum_i G[k,i] * (h[i] - h*[i])
    #      = c_k + sum_i G[k,i] * h[i] - sum_i G[k,i] * h*[i]
    #      So the constraint is: c_k + sum_i G[k,i] * h[i] - sum_i G[k,i] * h*[i] <= t
    #      i.e., sum_i G[k,i] * h[i] - t <= sum_i G[k,i] * h*[i] - c_k
    
    # Let me define: rhs_k = sum_i G[k,i] * h*[i] - c_k
    # Then: A @ [h; t] <= rhs
    # where A has rows [G[k,:], -1] for k = 0, ..., ncorr-1
    # And the objective is to minimize t, i.e., c_obj = [0, ..., 0, 1] (last component)
    
    # But we also need: sum h = n/2 (equality constraint)
    # And: 0 <= h <= 1, t >= 0 (or t >= some lower bound)
    
    # For linprog: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    # x = [h[0], ..., h[n-1], t]
    
    # This LP has n+1 variables and ncorr inequality constraints + 1 equality constraint.
    # For n=640, ncorr=1279, that's 641 variables and 1280 constraints.
    # This should be solvable quickly by an LP solver.
    
    # But computing G (the 1279 x 640 matrix) is O(n^2) which is fine.
    
    # Actually, let me not build the full matrix explicitly. I'll use the structure.
    
    # For now, let's just compute G explicitly:
    G = np.zeros((ncorr, n))
    for k in range(ncorr):
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            G[k, offset:offset+npairs] += (1.0 - h[:npairs])
            G[k, :npairs] -= h[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            G[k, :npairs] += (1.0 - h[aoff:aoff+npairs])
            G[k, aoff:aoff+npairs] -= h[:npairs]
    
    # rhs_k = G[k,:] @ h - c_k = G[k,:] @ h - corr[k]
    rhs = G @ h - corr
    
    # A_ub: each row is [G[k,:], -1], b_ub[k] = rhs[k]
    A_ub = np.zeros((ncorr, n + 1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = rhs
    
    # A_eq: sum h = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # c_obj: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        t_new = res.x[n]
        return h_new, t_new
    return None, None
```

This is interesting! The LP step gives a direction to decrease the maximum correlation. If I follow this direction (with a line search), I should be able to make progress.

However, there's a catch: the linearization is only valid locally. If the LP solution is far from h*, the linearization might be poor. So I need a line search:

```python
def lp_with_linesearch(h, n, max_steps=10):
    for step in range(max_steps):
        h_lp, t_lp = lp_step(h, n)
        if h_lp is None:
            break
        # Line search: find alpha in (0,1] that minimizes C5(h + alpha*(h_lp - h))
        best_alpha = 0
        best_c5 = compute_c5(h)
        for alpha in [0.1, 0.2, 0.5, 0.8, 1.0]:
            h_trial = h + alpha * (h_lp - h)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                best_c5 = c5_trial
                best_alpha = alpha
        if best_alpha > 0:
            h = h + best_alpha * (h_lp - h)
            h = project_feasible(h, n)
        else:
            break
    return h, compute_c5(h)
```
