def _swap_delta(h, i, j, n):
    """Compute correlation change for swap h[i]:1->0, h[j]:0->1."""
    r = 2 * n - 1
    dr = np.zeros(r)
    # Effect of h[i] decreasing by 1
    for k in range(max(i, 0), min(i + n, r)):
        idx = i + n - 1 - k
        if 0 <= idx < n:
            dr[k] += -(1.0 - h[idx])
    for k in range(max(n - 1 - i, 0), min(2*(n-1) - i + 1, r)):
        idx = i + k - (n - 1)
        if 0 <= idx < n:
            dr[k] += h[idx]
    # Effect of h[j] increasing by 1
    for k in range(max(j, 0), min(j + n, r)):
        idx = j + n - 1 - k
        if 0 <= idx < n:
            dr[k] += (1.0 - h[idx])
    for k in range(max(n - 1 - j, 0), min(2*(n-1) - j + 1, r)):
        idx = j + k - (n - 1)
        if 0 <= idx < n:
            dr[k] -= h[idx]
    # Correction at middle
    dr[n - 1] -= 2.0
    return dr

def slsqp_smooth(h, n, beta, maxiter=300):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                    'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_eq(h, n, top_k=5, max_iters=300):
    h = h.copy()

                h_fb, c5_fb = binary_ls(h_fine_bin, n_fine, time_limit=t_fine, rng=rng)
                if c5_fb < best_c5:
                    best_c5 = c5_fb
                    best_h = h_fb.copy()
                    best_n = n_fine

    # Phase 5: Continuous SLSQP refinement
    n = best_n
    h = project_feasible(best_h, n)
    for beta in [200, 1000, 5000, 20000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Top-shifts equalization
    if remaining() > 30:
        h_eq, c5_eq = top_shifts_eq(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 7: More SLSQP with higher beta
    for beta in [10000, 30000, 80000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Final top-shifts equalization
    if remaining() > 15:
        h_eq, c5_eq = top_shifts_eq(h, n, top_k=8, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
