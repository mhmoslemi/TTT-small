    """Delta of corr_+(d) when swapping h[i](1) and h[j](0)."""
    # Affected m values: {i, j, i-d, j-d} ∩ [0, n-1-d]

        hmd = 0 if (m+d) == i else (1 if (m+d) == j else h[m+d])
        new = hmd * (1 - hm)

def swap_delta_full(h, i, j, n):
    """Compute delta for all lags when swapping h[i](1) and h[j](0). Returns array of length 2n-1."""
    delta = np.zeros(2*n-1)
    
    for d in range(1, n):
        # Positive lag d (index n-1+d):
        affected = set()
        for m in [i, j, i-d, j-d]:
            if 0 <= m <= n-1-d:
                affected.add(m)
        
        d_plus = 0.0
        for m in affected:
            hm = 0 if m == i else (1 if m == j else h[m])
            hmd = 0 if (m+d) == i else (1 if (m+d) == j else h[m+d])
            d_plus += hmd * (1 - hm) - h[m+d] * (1 - h[m])
        delta[n-1+d] = d_plus
        
        # Negative lag -d (index n-1-d):
        d_minus = 0.0
        for m in affected:
            hm = 0 if m == i else (1 if m == j else h[m])
            hmd = 0 if (m+d) == i else (1 if (m+d) == j else h[m+d])
            d_minus += hm * (1 - hmd) - h[m] * (1 - h[m+d])
        delta[n-1-d] = d_minus
