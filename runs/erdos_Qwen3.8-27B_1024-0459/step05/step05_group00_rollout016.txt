import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def smooth_max_slsqp(h, n, beta, maxiter=500, ftol=1e-15):
    dx = 2.0 / n
    def sm_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def sm_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h

def top_k_subgradient(h, n, K=15, beta=50.0):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, k, n)
    return g

def cutting_plane_lp(h, n, K=25, delta=0.05):

    c_max = np.max(corr)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    b_ub_list = []
    
    for k in idx:

        fk = corr[k]

        row[-1] = -1.0

def structured_perturbation(h, n, rng):

    candidates = []

    h2 = project_feasible(h + pert, n)
    candidates.append(h2)
    
    # 3. Shift perturbation

    # Blend back to maintain sum

    candidates.append(h4)
    
    return candidates

def project_feasible(h, n):

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset

        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    n = best_n
    h = best_h.copy()

    # Phase 1: Warm-up SLSQP
    if remaining() > 20:
        h = run_slsqp(h, n, beta=1000, maxiter=400)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 2: Subgradient descent to escape local min
    if remaining() > 40:
        h_sg, c5_sg = subgradient_descent(h, n, min(50.0, remaining() * 0.15), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: Progressive SLSQP with increasing sharpness
    for beta in [5000, 15000, 40000]:
        if remaining() < 15:
            break
        h = run_slsqp(h, n, beta=beta, maxiter=600, ftol=1e-16)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 4: Binary snap + re-optimize
    if remaining() > 40:
        h_snap, c5_snap = binary_snap_and_optimize(h, n, beta=30000, maxiter=500)
        if c5_snap < best_c5:
            best_c5 = c5_snap
            best_h = h_snap.copy()
            h = h_snap.copy()

    # Phase 5: Equalize top constraints
    if remaining() > 15:
        h_eq, c5_eq = equalize_top(h, n, K=8, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Cutting plane LP
    if remaining() > 40:
        for it in range(300):
            if remaining() < 5:
                break
            h_lp, t_lp = cutting_plane_lp(h, n, K=30, delta=0.03)
            if t_lp < compute_c5(h) - 1e-10:
                best_val = compute_c5(h)
                improved = False
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.8]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        h = h_try
                        improved = True
                    else:
                        break
                if improved and compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
            elif it > 150:
                break

    # Phase 7: Post-cutting-plane SLSQP
    for beta in [50000, 100000]:
        if remaining() < 10:
            break
        h = run_slsqp(h, n, beta=beta, maxiter=600, ftol=1e-16)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 8: Second binary snap
    if remaining() > 30:
        h_snap2, c5_snap2 = binary_snap_and_optimize(h, n, beta=50000, maxiter=500)
        if c5_snap2 < best_c5:
            best_c5 = c5_snap2
            best_h = h_snap2.copy()
            h = h_snap2.copy()

    # Phase 9: Equalize again
    if remaining() > 15:
        h_eq2, c5_eq2 = equalize_top(h, n, K=10, max_iters=400)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    # Phase 10: Binary simulated annealing
    if remaining() > 40:
        h_bin = best_h.copy()
        h_bin[h_bin > 0.5] = 1.0
        h_bin[h_bin <= 0.5] = 0.0
        h_bin = project_feasible(h_bin, n)
        h_sa, c5_sa = binary_simulated_annealing(h_bin, n, min(40.0, remaining() * 0.25), rng)
        # Try to improve the SA result with SLSQP
        if c5_sa < compute_c5(best_h) + 0.01:
            h_sa_opt = run_slsqp(h_sa, n, beta=20000, maxiter=400)
            c5_sa_opt = compute_c5(h_sa_opt)
            if c5_sa_opt < best_c5:
                best_c5 = c5_sa_opt
                best_h = h_sa_opt.copy()
                h = h_sa_opt.copy()
        else:
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h = h_sa.copy()

    # Phase 11: SLSQP on potentially improved solution
    for beta in [60000, 120000]:
        if remaining() < 8:
            break
        h = run_slsqp(h, n, beta=beta, maxiter=700, ftol=1e-16)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Phase 12: Try higher resolution
    for mult in [2.0, 1.5]:
        if remaining() < 50:
            break
        n_t = int(round(best_n * mult / 32) * 32)
        if n_t <= best_n or n_t > 2048:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        if remaining() > 20:
            h_cur, c5_cur = subgradient_descent(h_feas, n_t, min(15.0, remaining() * 0.2), rng)
            if c5_cur > c5_int:
                h_cur = h_feas.copy()
                c5_cur = c5_int

        for beta in [10000, 40000]:
            if remaining() < 8:
                break
            h_cur = run_slsqp(h_cur, n_t, beta=beta, maxiter=400)
            c5_cur = compute_c5(h_cur)

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_t
            h = h_cur.copy()

    # Phase 13: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for cycle in range(3):
        if remaining() < 20:
            break

        if remaining() > 15:
            h, c5_tmp = subgradient_descent(h, n, min(15.0, remaining() * 0.2), rng)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

        for beta in [50000, 100000, 200000]:
            if remaining() < 6:
                break
            h = run_slsqp(h, n, beta=beta, maxiter=600, ftol=1e-16)
            c5_tmp = compute_c5(h)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

        if remaining() > 10:
            h_eq3, c5_eq3 = equalize_top(h, n, K=12, max_iters=300)
            if c5_eq3 < best_c5:
                best_c5 = c5_eq3
                best_h = h_eq3.copy()
                h = h_eq3.copy()

        if remaining() > 20:
            h_s, c5_s = binary_snap_and_optimize(h, n, beta=80000, maxiter=500)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                h = h_s.copy()

    # Phase 14: Ultra-fine gradient steps
    if remaining() > 5:
        for it in range(200):
            if remaining() < 3:
                break
            corr = compute_corr(best_h)
            idx = np.argsort(-corr)[:15]
            g = np.zeros(best_n)
            for k in idx:
                g += grad_corr_k(best_h, k, best_n)
            g /= len(idx)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            improved = False
            for step in [0.001, 0.0005, 0.002, 0.0003, 0.003, 0.0001]:
                h_try = project_feasible(best_h - step * g / gn, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    improved = True
                    break
            if not improved and it > 100:
                break

    # Final feasibility enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
