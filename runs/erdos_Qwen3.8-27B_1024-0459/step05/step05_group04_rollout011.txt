def project_onto_set(y, target):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    p = np.clip(y - t, 0.0, 1.0)
    diff = target - p.sum()
    if abs(diff) > 1e-10:
        mask = (p > 1e-12) & (p < 1.0 - 1e-12)
        if mask.sum() > 0:
            p[mask] += diff / mask.sum()
    p = np.clip(p, 0.0, 1.0)
