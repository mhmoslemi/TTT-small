    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-13:

        # LP: min C s.t. A_g @ h - C <= f_vals - A_g @ h, sum(h) = n/2, 0<=h<=1
        A_ub = np.hstack([A_g, -np.ones((M, 1))])
        b_ub = f_vals - A_g @ h
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')
        
        if not result.success:
            break
        
        h_lp = result.x[:n]
        h_lp = project_feasible(h_lp, n)
        
        # Try accepting the LP solution with damping
        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_try = (1 - alpha) * h + alpha * h_lp
            h_try = project_feasible(h_try, n)
            corr_try = compute_corr(h_try)
            max_try = np.max(corr_try)
            if max_try < current_max - 1e-12:
                h = h_try
                corr = corr_try
                current_max = max_try
                accepted = True
                break
        
        if accepted:
            best_c5 = current_max * dx
            no_improve = 0
            
            # Update active set
            top_lags = np.argsort(corr)[::-1]
            # Keep lags that are close to max, add new ones
            threshold = current_max - 0.05  # relative threshold
            new_active = top_lags[corr[top_lags] > threshold]
            M = len(new_active)
            if M > M_max:
                new_active = new_active[:M_max]
            if M < M_min:
                new_active = top_lags[:M_min]
            active = new_active
        else:
            no_improve += 1
            if no_improve > 20:
                # Try a smaller active set or different approach
                active = np.argsort(corr)[::-1][:max(M_min, M // 2)]
                no_improve = 0
                if no_improve > 50:
                    break
        
        it += 1
    
    return h, best_c5

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)


def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += diff / free.sum()
        else:
            if diff > 0:
                h[int(h.argmin())] = min(1.0, h[int(h.argmin())] + diff)
            else:
                h[int(h.argmax())] = max(0.0, h[int(h.argmax())] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():

def get_grads(h, n, indices):
    grads = np.zeros((len(indices), n))
    for idx, ci in enumerate(indices):
        offset = ci - (n - 1)
        if offset >= 0:
            grads[idx, :n - offset] += (1.0 - h[offset:])
            if offset > 0:
                grads[idx, offset:] += (-h[:n - offset])
            else:
                grads[idx, :] += (-h)
        else:
            a = -offset
            grads[idx, a:] += (1.0 - h[:n - a])
            grads[idx, :n - a] += (-h[a:])
    return grads


def active_set_lp_iter(h, corr, n, active, damping_list=None):

        damping_list = [1.0, 0.7, 0.5, 0.33, 0.2, 0.1]
    M = len(active)
    A_g = get_grads(h, n, active)
    f_vals = corr[active]
    A_ub = np.empty((M, n + 1))

def smooth_obj_eval(x, n, beta):
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = float(np.sum(exp_vals))
    return (m + np.log(wsum) / beta) * (2.0 / n)


def smooth_obj_grad(x, n, beta):
    corr = compute_corr(x)
    m = float(np.max(corr))
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = float(np.sum(exp_vals))
    weights = exp_vals / wsum
    grad_arr = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad_arr[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad_arr[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad_arr[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad_arr[aoff:aoff + npairs] -= w * x[:npairs]
    grad_arr *= (2.0 / n)
    return grad_arr


def slsqp_refine(h, n, beta, maxiter=300):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj_eval(x, n, beta)

    def obj_grad(x):
        return smooth_obj_grad(x, n, beta)

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-30, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

    if src_n == dst_n:

    return np.interp(x_new, xp, src_h)
