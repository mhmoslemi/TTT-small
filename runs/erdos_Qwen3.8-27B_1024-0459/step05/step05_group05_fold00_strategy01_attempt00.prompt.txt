def corr_gradient_at(h, k_off, n):
    """Gradient of correlation at a specific offset."""
    g = np.zeros(n)
    if k_off >= 0:
        np_ = n - k_off
        g[k_off:k_off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgradient_max(h, n, n_iter=200):
    """Direct subgradient descent on the max correlation."""
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        # Subgradient: average gradient of top correlations
        threshold = m - 0.5
        top_mask = corr >= threshold
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]

        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            g += corr_gradient_at(h, off, n)
        g /= len(top_idx)

        # Project gradient to tangent space of sum constraint
        g -= np.mean(g)
        # Also clip to [0,1] direction
        g = np.where(h > 0.99, -np.abs(g), g)
        g = np.where(h < 0.01, np.abs(g), g)

        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.9
            if step < 0.001:
                break

    return best_h, best_c5
