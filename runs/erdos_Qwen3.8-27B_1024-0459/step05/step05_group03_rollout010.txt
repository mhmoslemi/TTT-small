from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / n
    w = w * dx
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g

def slsqp_optimize(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        return smooth_obj(x, beta)
    
    def _grad(x):
        return smooth_grad(x, beta)
    
    def _sc(x):
        return np.sum(x) - n / 2.0
    
    def _scj(x):
        return np.ones(n)
    
    try:

                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.success or res.fun < smooth_obj(h0, beta) - 1e-15:

    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0

        else:
