        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao

    return g

                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                if nc < cur_c5 + 0.004:
                    cur_c5 = nc
                else:
                    h[i] = 1
                    h[j] = 0
            else:
                break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_from_frac(h, n):
    n_half = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:n_half]] = 1.0
    return h_bin


def fourier_init(n, rng, n_modes=4):
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = np.ones(n) * 0.5
    for m in range(1, n_modes + 1):
        freq = rng.randint(1, 3) * m
        phase = rng.uniform(0, 2 * np.pi)
        amp = 0.25 / np.sqrt(m)
        h += amp * np.cos(2 * np.pi * freq * x + phase)
    h += 0.05 * rng.randn(n)
    h = np.clip(h, 0, 1)
    return proj(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT, _PENMODE
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: L-BFGS-B refinement at parent resolution
    for beta, tk in [(200, 5), (500, 5), (1000, 5), (3000, 4), (8000, 4), (20000, 3)]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=tk, maxiter=500, penmode=0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Penalty-mode L-BFGS
    for beta in [50, 200, 500]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=400, penmode=1)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Subgradient descent
    if remaining() > 30:
        h_sg, c5_sg = subgradient_descent(h, n, max_iters=2500, step_init=0.07, decay=0.9995)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: More L-BFGS with high beta
    for beta, tk in [(15000, 5), (30000, 4), (50000, 3), (100000, 3)]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=tk, maxiter=400, penmode=0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Binary local search with 2-swaps
    if remaining() > 40:
        h_bin = binary_from_frac(h, n)
        h_bl, c5_bl = binary_search_v2(h_bin, n, max_iters=6000, top_k=4)

        h_ref = h_bl.copy()
        for beta in [3000, 10000, 30000]:
            if remaining() < 5:
                break
            h_ref = lbfgs_pass(h_ref, n, beta, top_k=5, maxiter=300, penmode=0)
        c5_ref = compute_c5(h_ref)

        c5_cand = min(c5_bl, c5_ref)
        h_cand = h_ref if c5_ref <= c5_bl else h_bl
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            h = h_cand.copy()

    # Phase 6: SLSQP for constraint precision
    for beta in [5000, 20000]:
        if remaining() < 8:
            break
        h_new = slsqp_pass(h, n, beta, top_k=5, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Higher resolution (double N)
    if remaining() > 100:
        n_high = best_n * 2
        if n_high % 2 == 0 and n_high <= 1024:
            h_high = interp_to(h, best_n, n_high)
            h_high = proj(h_high, n_high)

            for beta in [500, 2000, 8000]:
                if remaining() < 20:
                    break
                h_high = lbfgs_pass(h_high, n_high, beta, top_k=5, maxiter=300, penmode=0)

            if remaining() > 40:
                h_high_bin = binary_from_frac(h_high, n_high)
                h_hbl, c5_hbl = binary_search_v2(h_high_bin, n_high, max_iters=2000, top_k=4)
                h_hr = h_hbl.copy()
                for beta in [5000, 20000]:
                    if remaining() < 10:
                        break
                    h_hr = lbfgs_pass(h_hr, n_high, beta, top_k=4, maxiter=200, penmode=0)
                c5_hr = compute_c5(h_hr)

                c5_high = min(c5_hbl, c5_hr, compute_c5(h_high))
                h_hb = h_hr if c5_hr <= c5_hbl else h_hbl
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_hb.copy()
                    best_n = n_high

    # Phase 8: Multi-start with Fourier init at best_n
    if remaining() > 60:
        h_f = fourier_init(best_n, rng, n_modes=5)
        for beta in [500, 3000]:
            if remaining() < 10:
                break
            h_f = lbfgs_pass(h_f, best_n, beta, top_k=5, maxiter=300, penmode=0)
        h_f_bin = binary_from_frac(h_f, best_n)
        h_fbl, c5_fbl = binary_search_v2(h_f_bin, best_n, max_iters=2500, top_k=4)
        h_fr = h_fbl.copy()
        for beta in [5000, 20000]:
            if remaining() < 8:
                break
            h_fr = lbfgs_pass(h_fr, best_n, beta, top_k=4, maxiter=200, penmode=0)
        c5_fr = compute_c5(h_fr)
        c5_f = min(c5_fbl, c5_fr, compute_c5(h_f))
        h_fb = h_fr if c5_fr <= c5_fbl else h_fbl
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_fb.copy()

    # Phase 9: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    if remaining() > 25:
        h_sg2, c5_sg2 = subgradient_descent(h, n, max_iters=2000, step_init=0.05, decay=0.9997)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            h = h_sg2.copy()

    for beta, tk in [(30000, 5), (60000, 4), (100000, 3)]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=tk, maxiter=300, penmode=0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = binary_from_frac(h, n)
        h_bf, c5_bf = binary_search_v2(h_bin_f, n, max_iters=4000, top_k=5)
        h_rf = h_bf.copy()
        for beta in [15000, 50000]:
            if remaining() < 5:
                break
            h_rf = lbfgs_pass(h_rf, n, beta, top_k=5, maxiter=200, penmode=0)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
