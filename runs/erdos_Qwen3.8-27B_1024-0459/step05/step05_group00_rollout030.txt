import numpy as np
from scipy.optimize import minimize, linprog
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g


def full_gradient(h, n, top_idx, weights):
    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        if weights[i] > 1e-15:
            g += weights[i] * grad_corr_k(h, k, n)
    return g

        c5_new = compute_c5(h_new)
        
        v = h_new - h
        h = h_new
        t += 1
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5


def vanilla_subgradient(h, n, time_limit, rng, K=15, beta=80.0):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        w = np.exp(beta * (c_top - c_top.max()))
        s = w.sum()
        if s < 1e-30:
            w = np.ones(K) / K
        else:
            w /= s
        
        g = full_gradient(h, n, idx, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        elapsed = time.time() - start
        step = min(0.3 * n / gn, 0.5 / (1.0 + 0.05 * elapsed))
        
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    return best_h, best_c5


def binary_local_search(h, n, time_limit, rng, max_sweeps=50):
    """Binary local search: threshold, then 2-flip moves."""
    start = time.time()
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]
    
    best_bin = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    dx = 2.0 / n
    best_corr_max = np.max(compute_corr(h_bin))
    
    patience = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 10:
            break
        
        improved = False
        # Try random pairs of flips
        n_tries = min(2000, len(ones) * len(zeros))
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            # Flip i from 1 to 0, j from 0 to 1
            h_test = best_bin.copy()
            h_test[i] = 0.0
            h_test[j] = 1.0
            
            corr_test = compute_corr(h_test)
            max_test = np.max(corr_test)
            
            if max_test < best_corr_max - 1e-10:
                best_bin = h_test
                best_corr_max = max_test
                best_c5 = max_test * dx
                ones = np.where(best_bin > 0.5)[0]
                zeros = np.where(best_bin < 0.5)[0]
                improved = True
                patience = 0
            else:
                patience += 1
        
        if not improved and patience > 100:
            break
    
    return best_bin, best_c5


def relaxed_binary_search(h, n, time_limit, rng):
    """Binary search followed by gradient relaxation."""
    start = time.time()
    
    h_bin, c5_bin = binary_local_search(h, n, min(time_limit * 0.5, 60), rng)
    
    if time.time() - start > time_limit:
        return h_bin, c5_bin
    
    # Relax: use binary as init, do gradient descent
    h_rel = h_bin.copy()
    rem = time_limit - (time.time() - start)
    if rem > 5:
        h_rel, c5_rel = vanilla_subgradient(h_rel, n, rem * 0.7, rng, K=10, beta=100.0)
        if c5_rel > c5_bin:
            return h_bin, c5_bin
        return h_rel, c5_rel
    
    return h_bin, c5_bin


def smooth_max_slsqp(h, n, beta, maxiter=300):
    def sm_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def sm_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h


def cutting_plane_lp(h, n, K=30, delta=0.05):
    corr = compute_corr(h)
    c_max = np.max(corr)
    idx = np.argsort(-corr)[:K]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub_list = []
    b_ub_list = []
    
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g, h))
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    I = np.eye(n)
    A_trust = np.vstack([
        np.hstack([I, np.zeros((n, 1))]),
        np.hstack([-I, np.zeros((n, 1))]
    ])
    b_trust = np.concatenate([h + delta, h + delta])
    
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    
    return h.copy(), c_max


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def targeted_gradient_descent(h, n, time_limit, rng):
    """Fine-grained gradient descent with adaptive step and top-K focus."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    stuck = 0
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:8]
        c_top = corr[idx]
        
        # Focus on top few with high weight
        w = np.exp(100.0 * (c_top - c_top.max()))
        s = w.sum()
        if s < 1e-30:
            w = np.ones(len(idx)) / len(idx)
        else:
            w /= s
        
        g = full_gradient(h, n, idx, w)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Try multiple step sizes
        improved = False
        for step in [0.02, 0.01, 0.005, 0.03, 0.015, 0.008]:
            h_try = project_feasible(h - step * g / gn, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                best_c5 = c_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                stuck = 0
                break
        if not improved:
            stuck += 1
            if stuck > 20:
                # Perturb to escape
                h = project_feasible(h + 0.01 * rng.randn(n), n)
                stuck = 0
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Load parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick Nesterov subgradient at parent resolution
    if remaining() > 30:
        h_sg, c5_sg = nesterov_subgradient(h, n, min(30.0, remaining() * 0.1), rng, K=20, beta=50.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: Binary local search at parent resolution
    if remaining() > 60:
        h_bin, c5_bin = binary_local_search(h, n, min(50.0, remaining() * 0.2), rng, max_sweeps=100)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h = h_bin.copy()
        # Relax from binary
        if remaining() > 40:
            h_rel, c5_rel = vanilla_subgradient(h_bin, n, min(30.0, remaining() * 0.2), rng, K=10, beta=100.0)
            if c5_rel < best_c5:
                best_c5 = c5_rel
                best_h = h_rel.copy()
                h = h_rel.copy()

    # Phase 3: SLSQP polish at parent resolution
    for beta in [200, 1000, 5000, 15000]:
        if remaining() < 10:
            break
        h_new = smooth_max_slsqp(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Cutting plane iterations at parent resolution
    if remaining() > 40:
        for it in range(80):
            if remaining() < 5:
                break
            h_lp, t_lp = cutting_plane_lp(h, n, K=30, delta=0.04)
            if t_lp < compute_c5(h) - 1e-10:
                cur_c5 = compute_c5(h)
                for alpha in [1.0, 0.5, 0.3, 0.7]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < cur_c5:
                        cur_c5 = c_try
                        h = h_try
                    else:
                        break
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            elif it > 40:
                break

    # Phase 5: Targeted gradient descent
    if remaining() > 30:
        h_tg, c5_tg = targeted_gradient_descent(h, n, min(40.0, remaining() * 0.2), rng)
        if c5_tg < best_c5:
            best_c5 = c5_tg
            best_h = h_tg.copy()
            h = h_tg.copy()

    # Phase 6: Multi-resolution expansion
    res_candidates = []
    for mult in [1.25, 1.5, 1.75, 2.0, 2.5, 3.0]:
        res = int(round(best_n * mult / 32) * 32)
        if best_n < res <= 3072 and res % 2 == 0:
            res_candidates.append(res)
    res_candidates = sorted(set(res_candidates), key=lambda x: abs(x - best_n))

    for n_t in res_candidates[:6]:
        if remaining() < 60:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_cur = h_feas.copy()
        c5_cur = min(c5_int, best_c5)

        # Nesterov at new resolution
        if remaining() > 40:
            h_sg, c5_sg = nesterov_subgradient(h_cur, n_t, min(25.0, remaining() * 0.2), rng, K=20, beta=50.0)
            if c5_sg < c5_cur:
                c5_cur = c5_sg
                h_cur = h_sg

        # Binary search at new resolution
        if remaining() > 50:
            h_bin, c5_bin = binary_local_search(h_cur, n_t, min(35.0, remaining() * 0.25), rng, max_sweeps=80)
            if c5_bin < c5_cur:
                c5_cur = c5_bin
                h_cur = h_bin
            # Quick relaxation
            if remaining() > 20:
                h_rel, c5_rel = vanilla_subgradient(h_bin, n_t, min(15.0, remaining() * 0.15), rng, K=10, beta=80.0)
                if c5_rel < c5_cur:
                    c5_cur = c5_rel
                    h_cur = h_rel

        # SLSQP at new resolution
        for beta in [500, 3000, 10000, 30000]:
            if remaining() < 15:
                break
            h_cur = smooth_max_slsqp(h_cur, n_t, beta, maxiter=250)
            c5_cur = compute_c5(h_cur)

        # Cutting plane at new resolution
        if remaining() > 30:
            for it in range(50):
                if remaining() < 5:
                    break
                h_lp, t_lp = cutting_plane_lp(h_cur, n_t, K=25, delta=0.03)
                if t_lp < compute_c5(h_cur) - 1e-10:
                    cur_c5 = compute_c5(h_cur)
                    for alpha in [1.0, 0.5, 0.3]:
                        h_try = project_feasible(h_cur + alpha * (h_lp - h_cur), n_t)
                        c_try = compute_c5(h_try)
                        if c_try < cur_c5:
                            cur_c5 = c_try
                            h_cur = h_try
                        else:
                            break
                    c5_cur = cur_c5
                elif it > 30:
                    break

        # Targeted gradient
        if remaining() > 20:
            h_tg, c5_tg = targeted_gradient_descent(h_cur, n_t, min(20.0, remaining() * 0.15), rng)
            if c5_tg < c5_cur:
                c5_cur = c5_tg
                h_cur = h_tg

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_t

    # Phase 7: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    for cycle in range(3):
        if remaining() < 25:
            break

        # Nesterov
        if remaining() > 15:
            h, c5_tmp = nesterov_subgradient(h, n, min(20.0, remaining() * 0.2), rng, K=15, beta=100.0)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

        # SLSQP with high beta
        if remaining() > 15:
            for beta in [20000, 50000, 100000]:
                if remaining() < 8:
                    break
                h = smooth_max_slsqp(h, n, beta, maxiter=400)
            c5_tmp = compute_c5(h)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

        # Cutting plane
        if remaining() > 20:
            for it in range(60):
                if remaining() < 5:
                    break
                h_lp, t_lp = cutting_plane_lp(h, n, K=35, delta=0.02)
                if t_lp < compute_c5(h) - 1e-10:
                    cur_c5 = compute_c5(h)
                    for alpha in [1.0, 0.5, 0.7, 0.3]:
                        h_try = project_feasible(h + alpha * (h_lp - h), n)
                        c_try = compute_c5(h_try)
                        if c_try < cur_c5:
                            cur_c5 = c_try
                            h = h_try
                        else:
                            break
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                elif it > 40:
                    break

        # Binary search
        if remaining() > 40:
            h_bin, c5_bin = binary_local_search(h, n, min(30.0, remaining() * 0.2), rng, max_sweeps=100)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                h = h_bin.copy()
            if remaining() > 20:
                h_rel, c5_rel = vanilla_subgradient(h, n, min(20.0, remaining() * 0.15), rng, K=10, beta=150.0)
                if c5_rel < best_c5:
                    best_c5 = c5_rel
                    best_h = h_rel.copy()
                    h = h_rel.copy()

    # Phase 8: Final polish
    if remaining() > 15:
        h = smooth_max_slsqp(best_h, best_n, 100000, maxiter=500)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Very fine targeted steps
    if remaining() > 10:
        h = best_h.copy()
        for it in range(30):
            if remaining() < 3:
                break
            corr = compute_corr(h)
            idx = np.argsort(-corr)[:5]
            c_top = corr[idx]
            w = np.exp(200.0 * (c_top - c_top.max()))
            s = w.sum()
            if s < 1e-30:
                w = np.ones(len(idx)) / len(idx)
            else:
                w /= s
            g = full_gradient(h, best_n, idx, w)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            improved = False
            for step in [0.005, 0.002, 0.001, 0.01, 0.003]:
                h_try = project_feasible(h - step * g / gn, best_n)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-14:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
            if not improved and it > 15:
                break

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
