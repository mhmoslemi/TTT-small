import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0

_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def binarize(h, n):

    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for i in range(n_ones - target):
            h_bin[ones[order[i]]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(target - n_ones):
            h_bin[zeros[order[i]]] = 1

def perturb_binary(h_bin, n, n_flips=3, rng=None):

def smooth_max_obj(x, n, beta, top_k):

        val = (m + np.log(np.sum(ex)) / beta) * dx

    penalty = 100.0 * (np.sum(x) - n/2.0) ** 2

def smooth_max_grad(x, n, beta, top_k):
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:top_k]
    c = corr[idx]
    m = c[0]
    
    if beta > 0.01:
        sh = beta * (c - m)
        sh = np.clip(sh, -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
    else:
        w = np.zeros(top_k)
        w[0] = 1.0
    
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        # Compute gradient of corr[ki] with respect to x
        shift = ki - (n - 1)
        if shift >= 0:
            np_ = n - shift
            g[shift:shift+np_] += w[j] * (1.0 - x[:np_])
            g[:np_] -= w[j] * x[shift:shift+np_]
        else:
            ao = -shift
            np_ = n - ao
            g[:np_] += w[j] * (1.0 - x[ao:ao+np_])
            g[ao:ao+np_] -= w[j] * x[:np_]
    g *= dx
    g += 200.0 * (np.sum(x) - n/2.0)
    return g
