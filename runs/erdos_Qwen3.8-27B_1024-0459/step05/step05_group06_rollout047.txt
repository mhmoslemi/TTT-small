def subgrad_focus_step(h, n, n_active=5, tol=0.1):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = np.argsort(corr, kind='stable')[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    # Subtract mean to maintain sum constraint
    grad = grad - np.mean(grad)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5 + 0.02:
            for beta in [10000, 50000]:
                if remaining_time() < 25:
                    break
                h_r = slsqp_optimize(h_r, rn, beta, 250)
            if remaining_time() > 25:
                h_r, c5_r = coord_descent_v2(h_r, rn, n_sweeps=40, tol=1e-15)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = rn

    # Phase 7: Random perturbations to escape local optima
    n = best_n
    h = best_h.copy()
    for pert_size in [8, 16, 4]:
        n_perturb = max(3, n // pert_size)
        for pert in range(25):
            if remaining_time() < 30:
                break
            h_pert = h.copy()
            idx = _RNG.choice(n, n_perturb, replace=False)
            magnitude = 0.08 if pert_size == 4 else (0.12 if pert_size == 8 else 0.18)
            h_pert[idx] += _RNG.uniform(-magnitude, magnitude, n_perturb)
            h_pert = project_feasible(h_pert, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 + 0.008:
                h_new, c5_new = coord_descent_v2(h_pert, n, n_sweeps=30, tol=1e-15)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 8: Final intensive coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = coord_descent_v2(best_h, best_n, n_sweeps=60, tol=1e-16)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 12:
        h_new = slsqp_optimize(best_h, best_n, 1000000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 8:
        h_new, c5_new = coord_descent_v2(best_h, best_n, n_sweeps=30, tol=1e-16)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation and constraint enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
