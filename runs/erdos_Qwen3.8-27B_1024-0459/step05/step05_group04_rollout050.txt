            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_new[mask] += diff / cnt
                h_new = np.clip(h_new, 0.0, 1.0)
            else:
                h_new[:] += diff / n
                h_new = np.clip(h_new, 0.0, 1.0)

        h = h_new
        lr *= decay

    return best_h, best_c5

def top_shift_equalize(h, n, time_limit, top_k=6):
    h = h.copy().astype(float)
    dx = 2.0 / n
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    for _ in range(10000):
        if time.time() - start < time_limit and remaining() > 3:
            pass
        else:
            break

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        if c_max <= 0:
            break

        # Gradient of top-k average
        g_top = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_top[off:off + np_] += (1.0 - h[:np_])
                g_top[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_top[:np_] += (1.0 - h[ao:ao + np_])
                g_top[ao:ao + np_] -= h[:np_]
        g_top /= top_k

        # Also pull up the bottom shifts
        bot_k = 3
        bot_idx = np.argsort(corr)[:bot_k]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        g_bot /= bot_k

        d = g_bot - g_top
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn

        improved = False
        for step in [0.01, 0.02, 0.005, 0.03, 0.008, 0.015, 0.003, 0.04, 0.002]:
            ht = h + step * d
            ht = np.clip(ht, 0.0, 1.0)
            diff_s = target - np.sum(ht)
            if abs(diff_s) > 1e-10:
                mask = (ht > 1e-6) & (ht < 1.0 - 1e-6)
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    ht[mask] += diff_s / cnt
                    ht = np.clip(ht, 0.0, 1.0)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def round_to_binary(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    no = int(np.sum(h_bin))
    if no != target:
        if no > target:
            # Remove from positions closest to 0.5
            vals = h[~(h > 0.5)]
            sort_idx = np.argsort(h[(h > 0.5)])
            ones_mask = np.zeros(n, dtype=bool)
            ones_mask[h_bin == 1] = True
            ones_positions = np.where(ones_mask)[0]
            # Sort ones by how close to 0.5 (least committed)
            commit = np.abs(h[ones_positions] - 0.5)
            order = np.argsort(commit)
            for idx in order[:no - target]:
                h_bin[ones_positions[idx]] = 0
        else:
            zeros_mask = np.zeros(n, dtype=bool)
            zeros_mask[h_bin == 0] = True
            zeros_positions = np.where(zeros_mask)[0]
            commit = np.abs(h[zeros_positions] - 0.5)
            order = np.argsort(commit)
            for idx in order[:target - no]:
                h_bin[zeros_positions[idx]] = 1
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _SLN, _SLB, _SLTGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)

    parent_h = project_feasible(parent_h, parent_n)
    parent_c5 = compute_c5(parent_h)
    best_h = parent_h.copy()
    best_c5 = parent_c5
    best_n = parent_n
    n = best_n

    # Phase 1: SA on binary version
    h_bin = round_to_binary(best_h, n, rng)
    t1 = min(280, int(remaining() * 0.28))
    if t1 > 30:
        h_b1, c5_b1 = binary_sa(h_bin, n, t1, rng, T_start=0.05, T_end=0.0008)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n

    # Phase 2: Diverse starts with SA
    starts = []
    # Thue-Morse
    starts.append(thue_morse(n))
    # Greedy
    if remaining() > 120:
        starts.append(greedy_construct(n, np.random.RandomState(seed + 7)))
    # Random
    for _ in range(2):
        starts.append(random_binary_h(n, rng))
    # Perturbed current best
    h_pert = best_h.copy()
    n_flip = max(5, n // 40)
    flip_positions = rng.choice(n, n_flip, replace=False)
    h_pert[flip_positions] = 1.0 - h_pert[flip_positions]
    h_pert = project_feasible(h_pert, n)
    starts.append(h_pert)

    for sp in starts:
        if remaining() < 35:
            break
        sp_c5 = compute_c5(sp)
        if sp_c5 < best_c5 + 0.025:
            sp_bin = round_to_binary(sp, n, rng)
            t2 = min(100, int(remaining() * 0.08))
            if t2 > 15:
                h_sa, c5_sa = binary_sa(sp_bin, n, t2, rng, T_start=0.04, T_end=0.001)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n

    # Phase 3: Resolution sweep
    n = best_n
    resolutions = sorted(set([
        n - 16, n - 8, n + 8, n + 16, n - 32, n + 32,
        n - 48, n + 48,
        int(n * 0.8), int(n * 1.2), int(n * 1.4),
    ]))
    resolutions = [r for r in resolutions if 128 <= r <= 800 and r % 2 == 0 and r != n]
    resolutions.sort(key=lambda x: abs(x - n))

    for n_t in resolutions[:5]:
        if remaining() < 35:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = round_to_binary(h_interp, n_t, rng)

        t3 = min(70, int(remaining() * 0.06))
        if t3 > 12:
            h_b3, c5_b3 = binary_sa(h_bin_t, n_t, t3, rng, T_start=0.04, T_end=0.001)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t

    # Phase 4: Continuous subgradient refinement
    n = best_n
    h = best_h.copy().astype(float)
    h = project_feasible(h, n)

    t4 = min(100, int(remaining() * 0.08))
    if t4 > 10:
        h_sg, c5_sg = subgradient_solve(h, n, t4, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 5: Top-shift equalization
    if remaining() > 20:
        t5 = min(60, int(remaining() * 0.05))
        if t5 > 5:
            h_eq, c5_eq = top_shift_equalize(h, n, t5, top_k=6)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 6: SLSQP refinement
    if remaining() > 30:
        n = best_n
        h = best_h.copy().astype(float)
        h = project_feasible(h, n)
        _SLN = n
        _SLTGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]

        for beta in [3000, 8000, 20000, 50000]:
            if remaining() < 12:
                break
            _SLB = float(beta)
            try:
                res = minimize(slsqp_fun, h, jac=slsqp_jac, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-16, 'iprint': 0})
                if res.fun < best_c5:
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
            except Exception:
                pass

    # Phase 7: Final SA polish
    n = best_n
    h_bin = round_to_binary(best_h, n, rng)
    t7 = min(80, int(remaining() * 0.08))
    if t7 > 10:
        h_b7, c5_b7 = binary_sa(h_bin, n, t7, rng, T_start=0.015, T_end=0.0003)
        if c5_b7 < best_c5:
            best_c5 = c5_b7
            best_h = h_b7.copy()
            best_n = n

    # Phase 8: Final continuous polish
    if remaining() > 10:
        n = best_n
        h = best_h.copy().astype(float)
        h = project_feasible(h, n)
        t8 = min(40, int(remaining() * 0.04))
        if t8 > 5:
            h_sg8, c5_sg8 = subgradient_solve(h, n, t8, rng)
            if c5_sg8 < best_c5:
                best_c5 = c5_sg8
                best_h = h_sg8.copy()
                best_n = n

    # Final: ensure feasibility and verify
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
