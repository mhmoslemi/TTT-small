            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def corr_grad_weighted(h, weights):
    """Compute gradient of sum_k weights[k]*corr[k] with respect to h."""
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g

def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    w = w * dx
    return corr_grad_weighted(h, w)

def slsqp_optimize(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        return smooth_obj(x, beta)
    def _grad(x):
        return smooth_grad(x, beta)
    def _con(x):
        return np.sum(x) - n / 2.0
    def _conj(x):
        return np.ones(n)
    
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con, 'jac': _conj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < smooth_obj(h0, beta) - 1e-16:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def simulated_annealing_binary(h, n, max_iters=50000, T0=0.5, T_end=0.01):
    """Simulated annealing for binary h with swap moves."""
    h = (h > 0.5).astype(float).copy()
    # Ensure exactly n/2 ones
    idx = np.where(h == 1)[0]
    if len(idx) != n // 2:
        # Fix by random reassignment
        h = np.zeros(n)
        perm = np.random.permutation(n)
        h[perm[:n//2]] = 1.0
    
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    
    T = T0
    cool_rate = (T_end / T0) ** (1.0 / max(max_iters, 1))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        if it % 1000 == 0:
            T *= cool_rate ** 1000
        
        # Propose a swap: pick one 1-position and one 0-position
        ia = np.random.randint(len(ones))
        ib = np.random.randint(len(zeros))
        pa = ones[ia]
        pb = zeros[ib]
        
        h[pa] = 0.0
        h[pb] = 1.0
        new_c5 = compute_c5(h)
        
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            delta = new_c5 - cur_c5
            if delta < 1e-15:
                accept = True
            elif T > 1e-8 and np.random.random() < np.exp(-delta / T):
                accept = True
        
        if accept:
            cur_c5 = new_c5
            ones[ia] = pb
            zeros[ib] = pa
            # Keep lists sorted for efficiency (optional)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[pa] = 1.0
            h[pb] = 0.0
        
        if it % 5000 == 0:
            # Sort the index lists
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    
    return best_h, best_c5

def topk_subgradient(h, n, top_k=5, max_iters=500):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.9
    step = 0.03
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w_top = (c_top - c_top.min() + 1e-10) ** 2
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= dx
        g = corr_grad_weighted(h, weights)
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        step *= 0.999
    return best_h, best_c5

def swap_local_search_binary(h, n, max_rounds=30):
    """Greedy swap local search: only accept improving moves."""
    h = (h > 0.5).astype(float).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr_cur = compute_corr(h)
        c_max = np.max(corr_cur)
        best_improve = 1e-15
        best_pair = None
        
        for ia in range(len(ones)):
            pa = ones[ia]
            h[pa] = 0.0
            for ib in range(len(zeros)):
                pb = zeros[ib]
                h[pb] = 1.0
                nc = np.max(compute_corr(h))
                h[pb] = 0.0
                imp = c_max - nc
                if imp > best_improve:
                    best_improve = imp
                    best_pair = (pa, pb)
            h[pa] = 1.0
        
        if best_pair is None:
            break
        pa, pb = best_pair
        h[pa] = 0.0
        h[pb] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin
