import numpy as np

_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 0.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def balance_binary(h, n):
    h = (h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones > n_half:
        ones = np.where(h == 1.0)[0]
        excess = ones[:n_ones - n_half]
        h[excess] = 0.0
    elif n_ones < n_half:
        zeros = np.where(h == 0.0)[0]
        deficit = zeros[:n_half - n_ones]
        h[deficit] = 1.0
    return h


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[np_:])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):

    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:

def theta_morse(n):

    best_c5 = compute_c5(best_h)
    best_n = parent_n
    n_ones_target = best_n // 2
    
    # Phase 1: Quick improvement at parent resolution
    if remaining_time() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = balance_binary(h_bin, best_n)
        h_g, c5_g = greedy_local_search(h_bin, best_n, time_limit=20, rng=rng)
        if c5_g < best_c5 - 1e-14:
            best_c5 = c5_g
            best_h = h_g.copy()
    
    # Phase 2: Try multiple resolutions with SA + polish
    # Focus on higher resolutions for discretization improvement
    resolutions = []
    for mult in [1.0, 1.25, 1.5, 1.75, 2.0, 2.5]:
        nn = int(parent_n * mult)
        if nn % 2 != 0:
            nn += 1
        if 256 <= nn <= 1200:
            resolutions.append(nn)
    # Also add specific candidates
    for nn in [448, 512, 640, 768, 896, 1024]:
        if 256 <= nn <= 1200:
            resolutions.append(nn)
    resolutions = sorted(set(resolutions))
    
    # Allocate time: more time for higher resolutions
    total_sa_time = min(700, remaining_time() * 0.6)
    # Weight by resolution (larger gets more time)
    weights = [np.sqrt(r) for r in resolutions]
    total_w = sum(weights)
    
    for idx, n in enumerate(resolutions):
        time_left = remaining_time() - 60  # Reserve 60s for final polish
        if time_left < 30:
            break
        t_alloc = min(120, total_sa_time * weights[idx] / total_w)
        t_alloc = min(t_alloc, time_left)
        
        if t_alloc < 15:
            continue
        
        # Get initial solution at this resolution
        if n == parent_n:
            h0 = best_h.copy()
        else:
            h0 = interp_to(best_h, best_n, n)
        
        # Binary SA
        h_bin = (h0 > 0.5).astype(float)
        h_bin = balance_binary(h_bin, n)
        
        # Determine how much time for SA vs local search
        sa_time = t_alloc * 0.7
        ls_time = t_alloc * 0.3
        
        h_sa, c5_sa = simulated_annealing(h_bin, n, sa_time, rng)
        
        # Greedy local search on SA result
        if remaining_time() > 30:
            h_ls, c5_ls = greedy_local_search(h_sa, n, ls_time, rng)
            if c5_ls < c5_sa:
                h_sa = h_ls
                c5_sa = c5_ls
        
        # Continuous polish
        if remaining_time() > 40:
            polish_time = min(60, remaining_time() * 0.15)
            h_pol, c5_pol = continuous_polish(h_sa, n, polish_time)
            if c5_pol < c5_sa:
                h_sa = h_pol
                c5_sa = c5_pol
        
        # Top-k directed refinement
        if remaining_time() > 30:
            h_dir, c5_dir = top_k_directed(h_sa, n, top_k=4, max_iters=200,
                                           time_limit=30)
            if c5_dir < c5_sa:
                h_sa = h_dir
                c5_sa = c5_dir
        
        if c5_sa < best_c5 - 1e-14:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n
    
    # Phase 3: Try alternative starting points at best resolution
    if remaining_time() > 60:
        n = best_n
        alt_starts = []
        
        # Thue-Morse
        tm = theta_morse(n)
        alt_starts.append(tm)
        
        # Random balanced
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
        alt_starts.append(h_rand)
        
        # Shifted best
        alt_starts.append(np.roll(best_h, 1))
        
        for h_alt in alt_starts:
            if remaining_time() < 30:
                break
            t_sa = min(40, remaining_time() * 0.4)
            if t_sa < 10:
                continue
            h_s, c5_s = simulated_annealing(h_alt, n, t_sa, rng)
            if remaining_time() > 20:
                h_p, c5_p = continuous_polish(h_s, n, min(30, remaining_time() * 0.3))
                if c5_p < c5_s:
                    h_s, c5_s = h_p, c5_p
            if c5_s < best_c5 - 1e-14:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n
    
    # Phase 4: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    if remaining_time() > 20:
        h_eq, c5_eq = top_k_directed(h, n, top_k=6, max_iters=300,
                                     time_limit=min(30, remaining_time() * 0.3))
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    if remaining_time() > 15:
        h2 = continuous_polish(h, n, min(40, remaining_time() * 0.4))
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5 - 1e-14:
            best_c5 = c5_2
            best_h = h2.copy()
    
    if remaining_time() > 10:
        h3 = slsqp_opt(best_h, n, 500000, maxiter=300)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5 - 1e-14:
            best_c5 = c5_3
            best_h = h3.copy()
    
    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
