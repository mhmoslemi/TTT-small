                m = n - d
                g_dir[d:] += (1.0 - h[:m])
                g_dir[:m] -= h[d:]
            else:
                a = -d
                m = n - a
                g_dir[:m] += (1.0 - h[a:])
                g_dir[a:] -= h[:m]
        
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        
        improved = False
        for step in [0.001, 0.002, 0.005, 0.01, 0.003, 0.008, 0.02, 0.04]:
            ht = project_bisect(h + step * d, target)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.0005, 0.0002, 0.001]:
                ht = project_bisect(h + step * d, target)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def coordinate_refine(h, n, max_sweeps=20, time_limit=60):
    """Refine each coordinate to its best value."""
    h = project_bisect(h, n / 2.0)
    target = n / 2.0
    best_c5 = compute_c5(h)
    start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit - 2 or remaining_time() < 3:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0]
            if 0.0 < old < 1.0:
                cands.extend([old - 0.1, old - 0.05, old + 0.05, old + 0.1,
                              old * 0.5, old + (1.0 - old) * 0.5])
            cands = [c for c in cands if 0.0 <= c <= 1.0]
            
            best_cand_c5 = best_c5
            best_cand_val = old
            for c in cands:
                h[i] = c
                h = project_bisect(h, target)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-13:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_bisect(h, target)
        
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_bisect(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP at parent resolution (aim for ~15% of budget)
    p1_time = budget_s * 0.12
    h = best_h.copy()
    n = best_n
    for beta in [200, 1000, 5000, 20000]:
        if remaining_time() < p1_time * 0.3:
            break
        h_new = slsqp_run(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Top-k equalization on parent
    if remaining_time() > budget_s * 0.5:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=5, max_iters=200,
                                       time_limit=min(40, remaining_time() * 0.1))
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 2: Higher resolution optimization (main effort, ~50% of budget)
    # Try n=768
    n_high = best_n * 2  # 768
    if remaining_time() > budget_s * 0.3:
        h_interp = interp_to(best_h, best_n, n_high)
        h_high = project_bisect(h_interp, n_high / 2.0)
        c5_high_init = compute_c5(h_high)
        
        # Projected gradient with progressive beta
        p2_time = remaining_time() * 0.5
        h_high, c5_high = multi_start_projected_grad(
            h_high, n_high, n_iters=2000, momentum=0.9,
            time_limit=min(p2_time, budget_s * 0.35), rng=rng)
        
        if c5_high < best_c5 - 1e-14:
            best_c5 = c5_high
            best_h = h_high.copy()
            best_n = n_high
        
        # Active set equalization at high res
        if remaining_time() > budget_s * 0.15:
            h_as, c5_as = active_set_equalize(best_h, best_n,
                                               time_limit=min(60, remaining_time() * 0.15))
            if c5_as < best_c5 - 1e-14:
                best_c5 = c5_as
                best_h = h_as.copy()
        
        # More projected gradient at high res
        if remaining_time() > budget_s * 0.15:
            h_pg, c5_pg = project_grad_opt(best_h, best_n, 50000, 1000,
                                            momentum=0.95,
                                            time_limit=min(120, remaining_time() * 0.2))
            if c5_pg < best_c5 - 1e-14:
                best_c5 = c5_pg
                best_h = h_pg.copy()
        
        # Top-k equalization at high res
        if remaining_time() > budget_s * 0.1:
            h_eq2, c5_eq2 = top_k_equalize(best_h, best_n, top_k=6, max_iters=400,
                                             time_limit=min(80, remaining_time() * 0.2))
            if c5_eq2 < best_c5 - 1e-14:
                best_c5 = c5_eq2
                best_h = h_eq2.copy()
    
    # Phase 3: Try n=512 as well
    n_mid = 512
    if remaining_time() > budget_s * 0.2 and best_n != n_mid:
        h_mid = interp_to(best_h, best_n, n_mid)
        h_mid = project_bisect(h_mid, n_mid / 2.0)
        p3_time = min(120, remaining_time() * 0.2)
        h_mid, c5_mid = multi_start_projected_grad(
            h_mid, n_mid, n_iters=1500, momentum=0.9,
            time_limit=p3_time, rng=rng)
        if c5_mid < best_c5 - 1e-14:
            best_c5 = c5_mid
            best_h = h_mid.copy()
            best_n = n_mid
    
    # Phase 4: Binary local search (diversification)
    if remaining_time() > budget_s * 0.15:
        n_bs = min(best_n, 384)  # Binary search is expensive; cap at 384
        if best_n > n_bs:
            h_bs_start = interp_to(best_h, best_n, n_bs)
        else:
            h_bs_start = best_h.copy()
        
        h_bin, c5_bin = binary_local_search(h_bs_start, n_bs,
                                             time_limit=min(80, remaining_time() * 0.2),
                                             rng=rng)
        if c5_bin < best_c5 - 1e-14:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n_bs
        
        # Refine binary result with projected gradient
        if remaining_time() > budget_s * 0.1:
            h_ref, c5_ref = project_grad_opt(h_bin, n_bs, 5000, 500,
                                              momentum=0.9,
                                              time_limit=min(60, remaining_time() * 0.15))
            if c5_ref < best_c5 - 1e-14:
                best_c5 = c5_ref
                best_h = h_ref.copy()
    
    # Phase 5: Final refinement at best resolution
    n_final = best_n
    h_final = best_h.copy()
    
    if remaining_time() > 20:
        # More projected gradient with high beta
        h_pg2, c5_pg2 = project_grad_opt(h_final, n_final, 100000, 2000,
                                          momentum=0.95,
                                          time_limit=min(80, remaining_time() * 0.3))
        if c5_pg2 < best_c5 - 1e-14:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()
            h_final = h_pg2.copy()
    
    if remaining_time() > 15:
        h_eq3, c5_eq3 = top_k_equalize(h_final, n_final, top_k=5, max_iters=300,
                                         time_limit=min(40, remaining_time() * 0.2))
        if c5_eq3 < best_c5 - 1e-14:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()
            h_final = h_eq3.copy()
    
    if remaining_time() > 15:
        h_as2, c5_as2 = active_set_equalize(h_final, n_final,
                                             time_limit=min(30, remaining_time() * 0.2))
        if c5_as2 < best_c5 - 1e-14:
            best_c5 = c5_as2
            best_h = h_as2.copy()
    
    # Final SLSQP polish if at n=384 (feasible)
    if best_n <= 400 and remaining_time() > 15:
        h_sls = slsqp_run(best_h, best_n, 100000, maxiter=200)
        c5_sls = compute_c5(h_sls)
        if c5_sls < best_c5 - 1e-14:
            best_c5 = c5_sls
            best_h = h_sls.copy()
    
    # Coordinate refinement
    if remaining_time() > 10:
        h_cr, c5_cr = coordinate_refine(best_h, best_n, max_sweeps=10,
                                         time_limit=min(30, remaining_time() * 0.5))
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()
    
    # Final projection and evaluation
    best_h = project_bisect(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
