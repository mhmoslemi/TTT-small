        delta = c_new - best_c5
        
        if delta < -1e-12 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        elapsed = time.time() - start
        progress = min(1.0, elapsed / max(time_limit, 1))
        T = T_min + (T - T_min) * (1.0 - progress) ** 2
    
    # Final local search from best
    best_h, best_c5 = binary_steepest_descent(best_h, n, time_limit=30, rng=rng)
    return best_h, best_c5


def two_swap_search(h, n, n_trials=5000, time_limit=30, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    h = to_binary(h).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        found = False
        for _ in range(n_trials):
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            a, b = int(ones[i1]), int(ones[i2])
            c, d = int(zeros[j1]), int(zeros[j2])
            
            h[a] = 0.0
            h[b] = 0.0
            h[c] = 1.0
            h[d] = 1.0
            c_new = compute_c5(h)
            
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h.copy()
                found = True
            else:
                h[a] = 1.0
                h[b] = 1.0
                h[c] = 0.0
                h[d] = 0.0
        
        if not found:
            break
    
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _sm_obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def top_k_equalize(h, n, top_k=5, max_iters=300):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.005, 0.01, 0.02, 0.05, 0.002, 0.008, 0.03, 0.08, 0.15]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = None
    
    if parent_h is not None:
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
    else:
        best_n = 384
        best_h = theta_morse(best_n)
    
    best_c5 = compute_c5(best_h)
    
    # Generate starting points
    starts = {}
    starts['parent'] = to_binary(best_h)
    starts['thue'] = theta_morse(best_n)
    starts['random1'] = random_binary(best_n, rng)
    starts['random2'] = random_binary(best_n, np.random.RandomState(seed + 1))
    
    # Phase 1: Binary steepest descent on each starting point
    # Allocate time based on number of starts
    n_starts = len(starts)
    time_per_start = min(150, max(30, int(remaining_time() * 0.5 / n_starts)))
    
    for name, h_start in starts.items():
        if remaining_time() < 10:
            break
        h_opt, c5_opt = binary_steepest_descent(h_start, best_n, time_limit=time_per_start, rng=rng)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: Simulated annealing from best
    if remaining_time() > 30:
        sa_time = min(90, max(20, int(remaining_time() * 0.15)))
        h_sa, c5_sa = simulated_annealing_binary(best_h, best_n, time_limit=sa_time, rng=rng)
        if c5_sa < best_c5 - 1e-14:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 3: Two-swap search
    if remaining_time() > 20:
        tw_time = min(60, max(10, int(remaining_time() * 0.1)))
        h_2s, c5_2s = two_swap_search(best_h, best_n, n_trials=3000, time_limit=tw_time, rng=rng)
        if c5_2s < best_c5 - 1e-14:
            best_c5 = c5_2s
            best_h = h_2s.copy()
    
    # Phase 4: More binary local search after 2-swaps
    if remaining_time() > 20:
        h_opt, c5_opt = binary_steepest_descent(best_h, best_n, time_limit=min(120, int(remaining_time() * 0.2)), rng=rng)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 5: Try different resolutions
    n_try_list = []
    for nn in [256, 320, 384, 448, 512, 640, 768]:
        if nn != best_n and 128 <= nn <= 1024 and nn % 2 == 0:
            n_try_list.append(nn)
    n_try_list.sort(key=lambda x: abs(x - best_n))
    
    for n_t in n_try_list[:4]:
        if remaining_time() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin = to_binary(h_interp)
        alloc = min(120, max(15, int(remaining_time() * 0.15)))
        h_opt, c5_opt = binary_steepest_descent(h_bin, n_t, time_limit=alloc, rng=rng)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
    
    # Phase 6: Try Thue-Morse at other resolutions
    for n_t in [256, 512]:
        if remaining_time() < 20:
            break
        h_tm = theta_morse(n_t)
        alloc = min(80, max(10, int(remaining_time() * 0.1)))
        h_opt, c5_opt = binary_steepest_descent(h_tm, n_t, time_limit=alloc, rng=rng)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
    
    # Phase 7: Continuous refinement of best binary solution
    n = best_n
    h_cont = best_h.copy()
    
    for beta in [500, 5000, 50000, 200000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h_cont, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cont = h_new.copy()
    
    # Top-k equalization
    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h_cont, n, top_k=5, max_iters=200)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h_cont = h_eq.copy()
    
    # Phase 8: Final SLSQP polish with high beta
    for beta in [500000, 1000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h_cont, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cont = h_new.copy()
    
    # Phase 9: One more round of binary search at the final resolution
    if remaining_time() > 20:
        h_bin_final = to_binary(best_h)
        alloc = min(60, max(10, int(remaining_time() * 0.25)))
        h_opt, c5_opt = binary_steepest_descent(h_bin_final, n, time_limit=alloc, rng=rng)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        # Also try continuous from this
        if remaining_time() > 15:
            h_c = slsqp_opt(h_opt, n, 100000, maxiter=300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5 - 1e-14:
                best_c5 = c5_c
                best_h = h_c.copy()
    
    # Final: compare binary vs continuous, pick best
    h_bin_check = to_binary(best_h)
    c5_bin_check = compute_c5(h_bin_check)
    if c5_bin_check < best_c5 - 1e-14:
        best_c5 = c5_bin_check
        best_h = h_bin_check.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, n)
