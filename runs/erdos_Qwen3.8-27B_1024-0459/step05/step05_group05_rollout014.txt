def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    while abs(np.sum(h) - target) > 1e-14:
        for i in range(n):
            if 0 < h[i] < 1:
                h[i] += target - np.sum(h)
                h[i] = np.clip(h[i], 0, 1)
                break
        else:
            # If all are 0 or 1, we can't fix (shouldn't happen with n/2 constraint)
            break
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    wc = w * dx_v

        wk = wc[k]

        off = k - (n - 1)

            g[off:off + np_] += wk * (1.0 - x[:np_])

        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])

    return g

    return np.sum(x) - _TGT

    return np.ones(_N)


def lbfgs_smooth(h, n, beta, maxiter=500):

    _N = n
    _TGT = n / 2.0

    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:

def adam_smooth(h, n, beta, n_iter=2000, lr=0.02):
    """ADAM optimizer for the smooth max objective."""
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n).copy()
    dx_v = 2.0 / n
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_epoch = 0
    
    for it in range(n_iter):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m_val = np.max(corr)
        sh = corr - m_val
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        w = ex / s
        wc = w * dx_v
        
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= wk * h[:np_]
        
        # Project gradient to sum=0 subspace
        g -= np.mean(g)
        
        # ADAM update
        it1 = it + 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        m_hat = m / (1 - b1 ** it1)
        v_hat = v / (1 - b2 ** it1)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            best_epoch = it
            h = h_new.copy()
        else:
            h = h_new.copy()  # still move forward
        
        # Adaptive learning rate
        if it > 100 and best_epoch < it - 200:
            lr *= 0.7
            if lr < 0.0001:
                break
    
    return best_h, best_c5

def multi_scale_optimize(h, n, rng, total_time=100):
    """Optimize at multiple scales."""
    start = time.time()
    dx = 2.0 / n
    
    # Downsample to lower resolution
    for factor in [4, 2]:
        if time.time() - start > total_time * 0.3 or remaining() < 10:
            break
        n_low = n // factor
        if n_low < 32 or n_low % 2 != 0:
            continue
        # Downsample by averaging
        h_low = np.zeros(n_low)
        for i in range(n_low):
            h_low[i] = np.mean(h[i*factor:(i+1)*factor])
        h_low = project_feasible(h_low, n_low)
        
        # Optimize at low resolution
        _, c5_low = adam_smooth(h_low, n_low, 5000, n_iter=500, lr=0.05)
        
        # Upsample back
        h_up = np.interp((np.arange(n) + 0.5) * dx, 
                        (np.arange(n_low) + 0.5) * dx, h_low)
        h_up = project_feasible(h_up, n)
        c5_up = compute_c5(h_up)
        
        # Refine at full resolution
        _, c5_full = adam_smooth(h_up, n, 5000, n_iter=300, lr=0.02)
        if c5_full < compute_c5(h):
            h = h  # keep (already refined)
    
    return h

def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n

    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g


def adam_optimize(h, n, beta, n_iter=2000, lr=0.02):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n).copy()
    dx_v = 2.0 / n
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_iter = 0
    stuck_count = 0

    for it in range(n_iter):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        m_val = np.max(corr)
        sh = corr - m_val
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        w = ex / s
        wc = w * dx_v

        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= wk * h[:np_]

        g -= np.mean(g)
        it1 = it + 1
        m_v = b1 * m_v + (1 - b1) * g
        v_v = b2 * v_v + (1 - b2) * g * g
        m_hat = m_v / (1 - b1 ** it1)
        v_hat = v_v / (1 - b2 ** it1)
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        h = h_new
        c_new = compute_c5(h)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h.copy()
            best_iter = it
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 300:
                lr *= 0.7
                stuck_count = 0
                if lr < 0.0005:
                    break
    return best_h, best_c5


def equil_slsqp(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)

    def cfun(x):
        return np.sum(x) - _TGT

    def cjac(x):
        return np.ones(_N)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h):
            return h_new
    except Exception:
        pass
    return h


def binarize(h, n):
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones > n_half:
        ones_idx = np.where(h == 1)[0]
        h[ones_idx[:n_ones - n_half]] = 0
    elif n_ones < n_half:
        zeros_idx = np.where(h == 0)[0]
        h[zeros_idx[:n_half - n_ones]] = 1
    return h


def corr_grad_at(h, off, n):
    g = np.zeros(n)
    if off >= 0:
        np_ = n - off
        g[off:off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[off:off + np_]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def top_shifts_equalize(h, n, top_k=5, max_iters=400):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        if corr[top_idx[0]] - corr[top_idx[-1]] < 1e-10:
            break
        g_top = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            g_top += corr_grad_at(h, off, n)
        g_top /= len(top_idx)
        g_top -= np.mean(g_top)
        dn = np.linalg.norm(g_top)
        if dn < 1e-14:
            break
        d = g_top / dn
        improved = False
        for step in [0.03, 0.06, 0.01, 0.1, 0.02, 0.04, 0.08, 0.15]:
            ht = project_feasible(h - step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def binary_focused_search(h_bin, n, rng, time_limit=60):
    h = (h_bin > 0.5).astype(float).copy()
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    tabu = {}
    tabu_len = 50
    no_improve = 0
    max_no_improve = 8000

    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > max_no_improve:
            break

        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        top_k = 8
        top_idx = np.argsort(-corr)[:top_k]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_swap = None
        n_ones = len(ones)
        n_zeros = len(zeros)

        # Phase A: Focused on worst correlations
        for _ in range(4):
            k_hot = top_idx[rng.randint(len(top_idx))]
            off = k_hot - (n - 1)
            if off >= 0:
                faults = np.where((h[:n - off] == 1) & (h[off:] == 0))[0]
            else:
                ao = -off
                faults = np.where((h[ao:] == 1) & (h[:n - ao] == 0))[0]
            if len(faults) == 0:
                continue
            n_check = min(15, len(faults))
            check_idx = rng.choice(len(faults), n_check, replace=False) if len(faults) > n_check else np.arange(len(faults))
            for fi in check_idx:
                if off >= 0:
                    i = int(fi)
                else:
                    i = int(fi) + (-off)
                if i < 0 or i >= n or h[i] != 1:
                    continue
                # Try swapping h[i] with a few zeros
                n_try = min(5, n_zeros)
                try_idx = rng.choice(n_zeros, n_try, replace=False) if n_zeros > n_try else np.arange(n_zeros)
                for j_idx in try_idx:
                    j = int(zeros[j_idx])
                    if j == i:
                        continue
                    key = (i, j)
                    if key in tabu and tabu[key] > _T0 - 2:
                        continue
                    h[i] = 0
                    h[j] = 1
                    nc = float(np.max(compute_corr(h)))
                    h[i] = 1
                    h[j] = 0
                    delta = nc - cur_max
                    if delta < best_delta - 1e-14:
                        best_delta = delta
                        best_swap = (i, j)
                        if delta < -2.0:
                            break
                if best_delta < -2.0:
                    break
            if best_delta < -2.0:
                break

        # Phase B: Random swaps
        n_rand = 8
        r_idx = rng.randint(0, n_ones, size=n_rand) if n_ones >= n_rand else np.arange(n_ones)
        s_idx = rng.randint(0, n_zeros, size=n_rand) if n_zeros >= n_rand else np.arange(n_zeros)
        for t in range(min(len(r_idx), len(s_idx))):
            i = int(ones[r_idx[t]])
            j = int(zeros[s_idx[t]])
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            h[i] = 1
            h[j] = 0
            delta = nc - cur_max
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        # Phase C: Weighted random targeting hot regions
        if no_improve > 100:
            hot_positions = []
            for k in top_idx[:3]:
                off = k - (n - 1)
                if off >= 0:
                    hot_positions.extend(range(0, n - off))
                    hot_positions.extend(range(off, n))
                else:
                    ao = -off
                    hot_positions.extend(range(ao, n))
                    hot_positions.extend(range(0, n - ao))
            if hot_positions:
                hp = np.array(hot_positions[:200])
                for _ in range(4):
                    i = hp[rng.randint(len(hp))]
                    if h[i] != 1:
                        continue
                    j = zeros[rng.randint(n_zeros)]
                    if j == i:
                        continue
                    h[i] = 0
                    h[j] = 1
                    nc = float(np.max(compute_corr(h)))
                    h[i] = 1
                    h[j] = 0
                    delta = nc - cur_max
                    if delta < best_delta - 1e-14:
                        best_delta = delta
                        best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            tabu[(i, j)] = time.time()
            tabu[(j, i)] = time.time()
            if len(tabu) > tabu_len * 2:
                now = time.time()
                tabu = {k2: v2 for k2, v2 in tabu.items() if now - v2 < 10}
            no_improve = 0
            new_max = float(np.max(compute_corr(h)))
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            if no_improve > 300:
                # Escape: random 2-3 flip
                n_flip = rng.randint(2, 4)
                ones_now = np.where(h == 1)[0].copy()
                zeros_now = np.where(h == 0)[0].copy()
                for _ in range(n_flip):
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0
                    h[j] = 1
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                no_improve = 0

    return best_h, best_max * dx_v


def brolocal_search(h_bin, n, rng, time_limit=30):
    """Broader local search with 2-swap moves."""
    h = (h_bin > 0.5).astype(float).copy()
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()

    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        worst_k = np.argmax(corr)
        off = worst_k - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Find faults at worst lag
        if off >= 0:
            faults_1 = np.where((h[:n - off] == 1) & (h[off:] == 0))[0]
        else:
            ao = -off
            faults_1 = np.where((h[ao:] == 1) & (h[:n - ao] == 0))[0]

        best_delta = 0
        best_move = None

        # Try 1-swap moves from faults
        if len(faults_1) > 0:
            n_check = min(10, len(faults_1))
            check = rng.choice(len(faults_1), n_check, replace=False) if len(faults_1) > n_check else np.arange(len(faults_1))
            for fi in check:
                if off >= 0:
                    i = int(fi)
                else:
                    i = int(fi) + (-off)
                if h[i] != 1:
                    continue
                j_candidates = rng.choice(len(zeros), min(4, len(zeros)), replace=False)
                for ji in j_candidates:
                    j = int(zeros[ji])
                    if j == i:
                        continue
                    h[i] = 0
                    h[j] = 1
                    nc = float(np.max(compute_corr(h)))
                    h[i] = 1
                    h[j] = 0
                    delta = nc - cur_max
                    if delta < best_delta - 1e-14:
                        best_delta = delta
                        best_move = ('1swap', i, j)

        # Try 2-swap moves (simultaneous)
        if len(ones) >= 2 and len(zeros) >= 2:
            n_two = rng.randint(0, 5)
            for _ in range(n_two):
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if i1 == i2 or j1 == j2 or i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                    continue
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
                nc = float(np.max(compute_corr(h)))
                delta = nc - cur_max
                h[j2] = 0
                h[j1] = 0
                h[i2] = 1
                h[i1] = 1
                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_move = ('2swap', int(i1), int(i2), int(j1), int(j2))

        if best_move is not None:
            if best_move[0] == '1swap':
                _, i, j = best_move
                h[i] = 0
                h[j] = 1
            else:
                _, i1, i2, j1, j2 = best_move
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
            new_max = float(np.max(compute_corr(h)))
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        elif rng.rand() < 0.05:
            # Small random perturbation
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1

    return best_h, best_max * dx_v


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT, _RNG
    _T0 = time.time()
    _BUDGET = budget_s
    _RNG = np.random.RandomState(seed)
    rng = _RNG

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: ADAM continuous optimization from parent
    if remaining() > 40:
        h_a, c5_a = adam_optimize(h, n, 5000, n_iter=1500, lr=0.015)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    # Phase 2: ADAM with higher beta
    if remaining() > 30:
        h_a2, c5_a2 = adam_optimize(best_h, best_n, 20000, n_iter=1000, lr=0.008)
        if c5_a2 < best_c5:
            best_c5 = c5_a2
            best_h = h_a2.copy()

    # Phase 3: Top-shifts equalization (continuous)
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_equalize(best_h, best_n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
        h_eq2, c5_eq2 = top_shifts_equalize(best_h, best_n, top_k=10, max_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 4: Binary focused search (main target: max_corr <= 73 at n=384)
    if remaining() > 40:
        h_bin = binarize(best_h, best_n)
        t_limit = min(80, max(20, int(remaining() * 0.2)))
        h_b, c5_b = binary_focused_search(h_bin, best_n, rng, time_limit=t_limit)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 5: Broader binary search from multiple starts
    if remaining() > 30:
        h_bin_tm = thue_morse(best_n)
        h_b2, c5_b2 = binary_focused_search(h_bin_tm, best_n, rng, time_limit=30)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

        if remaining() > 15:
            h_b3, c5_b3 = brolocal_search(best_h, best_n, rng, time_limit=20)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()

    # Phase 6: Resolution sweep
    if remaining() > 60:
        n_candidates = []
        for delta in [-32, -16, -8, -4, -2, 2, 4, 8, 16, 32, 64, 96]:
            nt = best_n + delta
            if 128 <= nt <= 700 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [256, 320, 384, 448, 512, 576, 640]:
            if 128 <= nt <= 700 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:6]:
            if remaining() < 20:
                break
            if n_t == best_n:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t = compute_c5(h_t)

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

            if remaining() > 15:
                h_t, c5_t2 = adam_optimize(h_t, n_t, 5000, n_iter=600, lr=0.02)
                if c5_t2 < best_c5:
                    best_c5 = c5_t2
                    best_h = h_t.copy()
                    best_n = n_t

            if remaining() > 15:
                h_bin_t = binarize(h_t, n_t)
                h_bt, c5_bt = binary_focused_search(h_bin_t, n_t, rng, time_limit=15)
                if c5_bt < best_c5:
                    best_c5 = c5_bt
                    best_h = h_bt.copy()
                    best_n = n_t

    # Phase 7: Diverse continuous starts
    if remaining() > 40:
        # Thue-Morse continuous
        h_tm = thue_morse(best_n).astype(float)
        h_tm, c5_tm = adam_optimize(h_tm, best_n, 3000, n_iter=800, lr=0.02)
        h_tm, c5_tm = top_shifts_equalize(h_tm, best_n, 4, 150)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()

        # Random start
        if remaining() > 20:
            h_r = rng.uniform(0, 1, best_n)
            h_r = project_feasible(h_r, best_n)
            h_r, c5_r = adam_optimize(h_r, best_n, 3000, n_iter=600, lr=0.02)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

        # Perturbed best
        if remaining() > 15:
            h_p = best_h + rng.randn(best_n) * 0.05
            h_p = project_feasible(h_p, best_n)
            h_p, c5_p = adam_optimize(h_p, best_n, 8000, n_iter=800, lr=0.01)
            h_p, c5_p = top_shifts_equalize(h_p, best_n, 5, 150)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 8: SLSQP refinement with very high beta
    if remaining() > 15:
        h_s = equil_slsqp(best_h, best_n, 50000, maxiter=300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best
