    w = w * (p / sm) * (c / sm)**(p-1) if sm > 0 else w
    # Actually the gradient of (sum c_i^p)^(1/p) w.r.t. c_j is c_j^(p-1) / (sum c_i^p)^(1/p - 1)
    # = c_j^(p-1) * (sum c_i^p)^(1/p) / (sum c_i^p)
    # = c_j^(p-1) / (sum c_i^p)^(1 - 1/p)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        coef = (c[j]**(p-1)) / (np.sum(c_p) ** (1.0 - 1.0/p))
        g += coef * corr_grad_at(h, ki, n)
    return g * dx


def lbfgs_p_max(h, n, p=16, topk=8, maxiter=400):
    state = {'n': n, 'p': p, 'topk': topk, 'tgt': n/2.0}

    def fun(x):
        nn = state['n']
        dx = 2.0/nn
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:state['topk']]
        c = corr[idx].astype(float)
        cp = c**state['p']
        sm = np.sum(cp)**(1.0/state['p']) * dx
        pen = 100.0 * (np.sum(x) - state['tgt'])**2
        return sm + pen

    def grad(x):
        nn = state['n']
        dx = 2.0/nn
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:state['topk']]
        c = corr[idx].astype(float)
        cp = c**state['p']
        sp = np.sum(cp)
        g = np.zeros(nn)
        for j, ki in enumerate(idx):
            coef = (c[j]**(state['p']-1)) / (sp ** (1.0 - 1.0/state['p']))
            g += coef * corr_grad_at(x, ki, nn)
        g *= dx
        g += 200.0 * (np.sum(x) - state['tgt'])
        return g

    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxfun': maxiter*10})
        return proj(res.x, n)
    except Exception:
        return h


def binary_search_1swap(h, n, max_iters=5000, topk=5):
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    stagnation = 0

    for it in range(max_iters):
        if _remaining() < 2:
            break

        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:topk]
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_at(h, ki, n)
        g -= np.mean(g)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        score_one = -g[ones]
        score_zero = g[zeros]

        n_try_o = min(20, len(ones))
        n_try_z = min(20, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try_o]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        best_delta = 0
        best_swap = None
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                h[i] = 1
                h[j] = 0
                delta = nc - cur_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is None or best_delta >= 0:
            # Random kicks
            found = False
            n_rand = 10
            for _ in range(n_rand):
                i = int(ones[np.random.randint(len(ones))])
                j = int(zeros[np.random.randint(len(zeros))])
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                h[i] = 1
                h[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                    found = True
            if not found:
                stagnation += 1
                if stagnation > 50:
                    break
                # Random restart: perturb current best
                for _ in range(3):
                    ii = int(ones[np.random.randint(len(ones))])
                    jj = int(zeros[np.random.randint(len(zeros))])
                    h[ii] = 0
                    h[jj] = 1
                cur_c5 = compute_c5(h)
                stagnation = 0
                continue

        i, j = best_swap
        h[i] = 0
        h[j] = 1
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

    return best_h, best_c5


def binary_search_2swap(h, n, max_iters=2000, topk=4):
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    stagnation = 0

    for it in range(max_iters):
        if _remaining() < 2:
            break

        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:topk]
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_at(h, ki, n)
        g -= np.mean(g)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        # 2-swap: flip two 1s to 0 and two 0s to 1
        # Score: we want to remove 1s where g is very negative, add 1s where g is very positive
        score_one = -g[ones]  # want high (negative g)
        score_zero = g[zeros]  # want high (positive g)

        n_try = min(8, len(ones))
        n_try_z = min(8, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        best_delta = 0
        best_swap = None

        # Try pairs
        for ii in range(len(top_ones)):
            for jj in range(ii+1, len(top_ones)):
                i1, i2 = top_ones[ii], top_ones[jj]
                for kk in range(len(top_zeros)):
                    for ll in range(kk+1, len(top_zeros)):
                        j1, j2 = top_zeros[kk], top_zeros[ll]
                        if len({i1, i2, j1, j2}) < 4:
                            continue
                        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                        nc = compute_c5(h)
                        h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
                        delta = nc - cur_c5
                        if delta < best_delta - 1e-15:
                            best_delta = delta
                            best_swap = (i1, i2, j1, j2)

        if best_swap is not None and best_delta < 0:
            i1, i2, j1, j2 = best_swap
            h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            # Try 1-swaps as fallback
            n_try_o = min(10, len(ones))
            n_try_z = min(10, len(zeros))
            top_ones = ones[np.argsort(-score_one)[:n_try_o]]
            top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]
            found = False
            for i in top_ones:
                for j in top_zeros:
                    if i == j:
                        continue
                    h[i] = 0; h[j] = 1
                    nc = compute_c5(h)
                    h[i] = 1; h[j] = 0
                    delta = nc - cur_c5
                    if delta < 0:
                        h[i] = 0; h[j] = 1
                        cur_c5 = nc
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h.copy()
                        stagnation = 0
                        found = True
                        break
                if found:
                    break
            if not found:
                stagnation += 1
                if stagnation > 80:
                    break

    return best_h, best_c5


def nesterov_gradient(h, n, max_iters=5000, step_init=0.04, topk=5):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.9
    step = step_init

    for it in range(max_iters):
        if _remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:topk]
        g = np.zeros(n)
        c_top = corr[idx]
        w = c_top - c_top.min() + 0.05
        w = np.maximum(w, 0.01)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_at(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        t = (1.0 + np.sqrt(1.0 + 4.0 * (it+1)**2)) / 2.0
        mu = (it + 1) / (t + 1)

        h_prev = h.copy()
        v = mu * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new <= c5 + 1e-12:
            h = h_new
            step = min(step * 1.01, 0.12)
        else:
            h = h_prev.copy()
            v *= 0.5
            step *= 0.8

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def random_binary_init(n, rng):
    h = np.zeros(n)
    half = n // 2
    idx = rng.permutation(n)[:half]
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick continuous refinement of parent (~80s)
    if _remaining() > 80:
        h = lbfgs_p_max(h, n, p=16, topk=6, maxiter=300)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

        h = lbfgs_p_max(h, n, p=40, topk=5, maxiter=200)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: Binary local search at parent resolution (~200s)
    if _remaining() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_rs, c5_rs = binary_search_1swap(h_bin, best_n, max_iters=8000, topk=5)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()

    if _remaining() > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_rs2, c5_rs2 = binary_search_2swap(h_bin, best_n, max_iters=3000, topk=4)
        if c5_rs2 < best_c5:
            best_c5 = c5_rs2
            best_h = h_rs2.copy()

    # Phase 3: Try higher resolutions (~400s)
    resolutions = [512, 640, 448]
    # Add some based on best_n
    for res in [best_n + 64, best_n + 128, best_n + 96]:
        if res not in resolutions and 128 <= res <= 768:
            resolutions.append(res)

    resolutions = sorted(set(r for r in resolutions if 128 <= r <= 768 and r % 2 == 0),
                          key=lambda x: abs(x - 512))
    resolutions = resolutions[:3]

    for n_t in resolutions:
        if _remaining() < 100:
            break

        # Interpolate current best to new resolution
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # Also try a random binary init
        h_rand = random_binary_init(n_t, rng)

        # Quick L-BFGS on interpolated
        h_t = lbfgs_p_max(h_feas, n_t, p=16, topk=6, maxiter=200)
        c5_t = compute_c5(h_t)
        candidates = [(h_t, c5_t)]

        # Binary search on interpolated
        h_b = (h_t > 0.5).astype(float)
        h_b1, c5_b1 = binary_search_1swap(h_b, n_t, max_iters=4000, topk=5)
        candidates.append((h_b1, c5_b1))

        # 2-swap polish
        h_b2, c5_b2 = binary_search_2swap(h_b1, n_t, max_iters=2000, topk=4)
        candidates.append((h_b2, c5_b2))

        # Continuous refine of best binary
        h_ref = lbfgs_p_max(h_b2, n_t, p=32, topk=5, maxiter=250)
        c5_ref = compute_c5(h_ref)
        candidates.append((h_ref, c5_ref))

        # More binary after continuous
        h_b3 = (h_ref > 0.5).astype(float)
        h_b3, c5_b3 = binary_search_1swap(h_b3, n_t, max_iters=2000, topk=4)
        candidates.append((h_b3, c5_b3))
        h_ref2 = lbfgs_p_max(h_b3, n_t, p=64, topk=4, maxiter=200)
        c5_ref2 = compute_c5(h_ref2)
        candidates.append((h_ref2, c5_ref2))

        # Select best candidate
        for hc, c5c in candidates:
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc.copy()
                best_n = n_t

    # Phase 4: Intensive refinement at best resolution (~300s)
    n = best_n
    h = best_h.copy()

    if _remaining() > 40:
        h_pg, c5_pg = nesterov_gradient(h, n, max_iters=4000, step_init=0.03, topk=5)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    if _remaining() > 60:
        # More binary search
        h_bin = (best_h > 0.5).astype(float)
        h_rs, c5_rs = binary_search_1swap(h_bin, n, max_iters=8000, topk=4)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()

        h_rs2, c5_rs2 = binary_search_2swap(h_rs, n, max_iters=3000, topk=4)
        if c5_rs2 < best_c5:
            best_c5 = c5_rs2
            best_h = h_rs2.copy()

        # Continuous refine
        h_ref = lbfgs_p_max(best_h, n, p=64, topk=5, maxiter=300)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 5: Final polish (~150s)
    n = best_n
    h = best_h.copy()

    if _remaining() > 60:
        # Very tight smooth max
        h = lbfgs_p_max(h, n, p=100, topk=4, maxiter=300)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

        # Final binary polish
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_search_1swap(h_bin_f, n, max_iters=5000, topk=3)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

        # One more continuous pass
        h_ref_f = lbfgs_p_max(best_h, n, p=200, topk=3, maxiter=200)
        c5_rf = compute_c5(h_ref_f)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_ref_f.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
