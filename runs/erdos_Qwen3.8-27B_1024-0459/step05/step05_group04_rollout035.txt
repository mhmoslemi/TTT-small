```python
def subgradient_step(h, n, step, top_k=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k shifts
    idx = np.argsort(-corr)[:top_k]
    # Compute gradient of average of top-k correlations
    grad = np.zeros(n)
    for k in idx:
        off = k - (n - 1)
        if off >= 0:
            m = n - off
            grad[off:off+m] += (1.0 - h[:m])
            grad[:m] -= h[off:off+m]
        else:
            ao = -off
            m = n - ao
            grad[:m] += (1.0 - h[ao:ao+m])
            grad[ao:ao+m] -= h[:m]
    grad /= top_k
    # Update
    h_new = h - step * grad
    # Project
    h_new = np.clip(h_new, 0.0, 1.0)
    # Adjust sum
    target = n / 2.0
    diff = target - np.sum(h_new)
    # Distribute adjustment
    mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
    if np.sum(mask) > 0:
        h_new[mask] += diff / np.sum(mask)
        h_new = np.clip(h_new, 0.0, 1.0)
    else:
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new
```

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

        idx = np.argsort(-corr)[:top_k * 2]
        # Use top-k for gradient
        idx_k = idx[:top_k]
        
        # Compute subgradient (average gradient of top-k correlations)
        grad = np.zeros(n)
        for k in idx_k:
            off = k - (n - 1)
            if off >= 0:
                m = n - off
                grad[off:off+m] += (1.0 - h[:m])
                grad[:m] -= h[off:off+m]
            else:
                ao = -off
                m = n - ao
                grad[:m] += (1.0 - h[ao:ao+m])
                grad[ao:ao+m] -= h[:m]
        grad /= top_k
        
        # Try different step sizes
        improved = False
        for factor in [1.0, 0.5, 2.0, 0.25, 4.0, 0.1]:

            h_new = project_feasible(h - s * grad, n)

                improved = True
                break

            step *= 0.8
            patience += 1

    return best_h, best_c5

            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        best_move = None
        best_new_max = cur_max + 1e-15
        
        seen = set()
        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            
            new_max = eval_swap_fast(h, corr, i, j, n)
            
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j)
        
        if best_move is not None:
            i, j = best_move
            dr = make_swap_delta(h, i, j, n)
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                # Aggressive perturbation
                n_pert = rng.randint(3, 8)
                for _ in range(n_pert):
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    dr = make_swap_delta(h, i, j, n)
                    h[i] = 0
                    h[j] = 1
                    corr = corr + dr
                cur_max = float(np.max(corr))
                stall = 0
                T = max(T_min, T * 0.7)
            else:
                T = max(T_min, T * cooling)
    
    return best_h, best_max * dx
