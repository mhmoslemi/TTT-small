import cvxpy as cp

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

def remaining():

def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g

def smooth_obj(h, beta):

    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    w = w * dx

    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

        return np.sum(x) - n / 2.0

        return np.ones(n)

def subgrad_topk(h, n, top_k=5, max_iters=800):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.85
    step = 0.05

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w_top = c_top - c_top.min() + 1e-10
        w_top = w_top ** 2
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= dx
        g = corr_grad_weighted(h, weights)
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        step *= 0.9995
        if it % 100 == 0:
            mu = min(0.95, mu + 0.01)
    return best_h, best_c5

For the LP:
minimize t
subject to: A @ [delta; t] <= b
            delta >= -h
            delta <= 1-h
            sum(delta) = 0

The A matrix has shape (K + n, n+1) where K is the number of active constraints.
- First K rows: [coeff_k, -1] for each active k, with b_k = -corr(k)
- Next n rows: identity for delta lower bound: [I, 0] @ [delta; t] >= -h → [-I, 0] @ [delta; t] <= h
- Next n rows: [I, 0] @ [delta; t] <= 1-h
- Equality: [ones(1,n), 0] @ [delta; t] = 0

This is a standard LP that linprog can handle.

```python
from scipy.optimize import linprog

def lp_step_linprog(h, n, top_k=10):
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:top_k]
    
    # Variables: x = [delta_0, ..., delta_{n-1}, t]
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active k: sum_i coeff_k[i] * delta[i] - t <= -corr[k]
    rows = []
    b_ub = []
    for idx in top_idx:
        off = idx - (n - 1)
        coeff = np.zeros(n)
        if off >= 0:
            np_ = n - off
            coeff[:np_] += (1.0 - h[off:off+np_])
            coeff[off:] -= h[:np_]
        else:
            a = -off
            np_ = n - a
            coeff[a:a+np_] += (1.0 - h[:np_])
            coeff[:np_] -= h[a:a+np_]
        row = np.zeros(n + 1)
        row[:n] = coeff
        row[n] = -1.0
        rows.append(row)
        b_ub.append(-corr[idx])
    
    # delta[i] >= -h[i] → -delta[i] <= h[i]
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        rows.append(row)
        b_ub.append(h[i])
    
    # delta[i] <= 1 - h[i]
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        rows.append(row)
        b_ub.append(1.0 - h[i])
    
    A_ub = np.array(rows)
    b_ub = np.array(b_ub)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: delta[i] in [-h[i], 1-h[i]], t free
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)] + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        delta = res.x[:n]
        return delta
    return None
```

import numpy as np
from scipy.optimize import minimize, linprog
import time

_T0 = 0.0
_BUDGET = 1300.0
_rng = np.random.RandomState(42)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    w = w * dx
    return corr_grad_weighted(h, w)


def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def lp_solve_step(h, n, top_k=10):
    """Solve LP to find best first-order descent direction."""
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:top_k]

    c = np.zeros(n + 1)
    c[n] = 1.0

    rows = []
    b_ub = []

    for idx in top_idx:
        off = idx - (n - 1)
        coeff = np.zeros(n)
        if off >= 0:
            np_ = n - off
            coeff[:np_] += (1.0 - h[off:off + np_])
            coeff[off:] -= h[:np_]
        else:
            a = -off
            np_ = n - a
            coeff[a:a + np_] += (1.0 - h[:np_])
            coeff[:np_] -= h[a:a + np_]
        row = np.zeros(n + 1)
        row[:n] = coeff
        row[n] = -1.0
        rows.append(row)
        b_ub.append(-corr[idx])

    # -delta[i] <= h[i] (i.e., delta[i] >= -h[i])
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        rows.append(row)
        b_ub.append(h[i] + 1e-12)

    # delta[i] <= 1 - h[i]
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        rows.append(row)
        b_ub.append(1.0 - h[i] - 1e-12)

    A_ub = np.array(rows)
    b_ub = np.array(b_ub)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    bounds = [(-h[i] - 1e-12, 1.0 - h[i] - 1e-12) for i in range(n)] + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n].copy()
    except Exception:
        pass
    return None


def lp_optimize(h, n, top_k_start=5, max_rounds=300):
    """Iterative LP-based optimization with adaptive active set."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    top_k = top_k_start
    stall_count = 0

    for it in range(max_rounds):
        if remaining() < 5:
            break

        delta = lp_solve_step(h, n, top_k=top_k)
        if delta is None:
            break

        # Check if meaningful improvement direction
        norm_d = np.linalg.norm(delta)
        if norm_d < 1e-8:
            stall_count += 1
            if stall_count > 5:
                top_k = min(top_k + 3, 30)
                stall_count = 0
                if top_k > 30:
                    break
            continue

        # Try full step, then damped steps
        found = False
        for alpha in [1.0, 0.5, 0.25, 0.1]:
            h_try = project_feasible(h + alpha * delta, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                found = True
                stall_count = 0
                break

        if not found:
            stall_count += 1
            if stall_count > 8:
                top_k = min(top_k + 2, 30)
                stall_count = 0
                if top_k > 30:
                    break

    return best_h, best_c5


def slsqp_run(h, n, beta, maxiter=400):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.success or res.fun < _obj(h0):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def subgrad_topk(h, n, top_k=5, max_iters=600):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.85
    step = 0.05

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w_top = c_top - c_top.min() + 1e-10
        w_top = w_top ** 2
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= dx
        g = corr_grad_weighted(h, weights)
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        step *= 0.9997
        if it % 150 == 0:
            mu = min(0.95, mu + 0.005)
    return best_h, best_c5


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def binary_greedy_ls(h_bin, n, max_rounds=30):
    """Greedy binary local search with single swaps."""
    h = (h_bin > 0.5).astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improve = 1e-12
        best_pair = None

        # Targeted search: focus on positions contributing to top correlations
        top_corr_idx = np.argsort(-corr)[:5]

        # Build priority lists
        ones_priority = np.zeros(len(ones))
        zeros_priority = np.zeros(len(zeros))

        for ti in top_corr_idx:
            off = ti - (n - 1)
            if off >= 0:
                np_ = n - off
                for i in range(np_):
                    if h[i] > 0.5 and h[i + off] < 0.5:
                        ia = np.searchsorted(ones, i)
                        if ia < len(ones) and ones[ia] == i:
                            ones_priority[ia] += 1
                    if h[i + off] < 0.5 and h[i] > 0.5:
                        ia = np.searchsorted(zeros, i + off)
                        if ia < len(zeros) and zeros[ia] == i + off:
                            zeros_priority[ia] += 1
            elif off < 0:
                a = -off
                np_ = n - a
                for i in range(np_):
                    if h[i + a] > 0.5 and h[i] < 0.5:
                        ia = np.searchsorted(ones, i + a)
                        if ia < len(ones) and ones[ia] == i + a:
                            ones_priority[ia] += 1
                    if h[i] < 0.5 and h[i + a] > 0.5:
                        ia = np.searchsorted(zeros, i)
                        if ia < len(zeros) and zeros[ia] == i:
                            zeros_priority[ia] += 1

        # Try top-priority swaps
        top_ones = ones[np.argsort(-ones_priority)[:40]] if len(ones) <= 40 else \
            np.concatenate([ones[np.argsort(-ones_priority)[:40]], ones[np.random.choice(len(ones), min(40, len(ones)), replace=False)]])
        top_zeros = zeros[np.argsort(-zeros_priority)[:40]] if len(zeros) <= 40 else \
            np.concatenate([zeros[np.argsort(-zeros_priority)[:40]], zeros[np.random.choice(len(zeros), min(40, len(zeros)), replace=False)]])

        # Deduplicate
        top_ones = np.unique(top_ones[:60])
        top_zeros = np.unique(top_zeros[:60])

        for pa in top_ones:
            h[pa] = 0.0
            for pb in top_zeros:
                if pb == pa:
                    continue
                h[pb] = 1.0
                nc = np.max(compute_corr(h))
                h[pb] = 0.0
                imp = c_max - nc
                if imp > best_improve:
                    best_improve = imp
                    best_pair = (pa, pb)
            h[pa] = 1.0

        if best_pair is None:
            break
        pa, pb = best_pair
        h[pa] = 0.0
        h[pb] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    return best_h, best_c5


def simulated_annealing_binary(h_bin, n, max_iters=3000, T0=0.3, T_end=0.005):
    """Simulated annealing for binary solutions."""
    global _rng
    h = (h_bin > 0.5).astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    for it in range(max_iters):
        if remaining() < 3:
            break
        T = T0 * (T_end / T0) ** (float(it) / max(float(max_iters), 1.0))

        pa = int(ones[_rng.randint(len(ones))])
        pb = int(zeros[_rng.randint(len(zeros))])

        h[pa] = 0.0
        h[pb] = 1.0

        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5

        if delta < 0 or _rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[pa] = 1.0
            h[pb] = 0.0

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_smooth(h, n, beta, max_iters=1500):
    h = project_feasible(h, n)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0

    for it in range(max_iters):
        if remaining() < 3:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        y_new = y + alpha * (y - h)
        g = smooth_grad(h, beta)
        lr = 1.0 / (beta * (it + 10))
        h_new = project_feasible(y_new - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new
    return best_h, best_c5


def perturb_and_refine(h, n, num_trials=8):
    """Add random perturbations and re-optimize to escape local optima."""
    global _rng
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for trial in range(num_trials):
        if remaining() < 10:
            break
        # Random perturbation
        noise = _rng.randn(n) * 0.02
        h_pert = project_feasible(h + noise, n)
        c5_pert = compute_c5(h_pert)

        # Quick SLSQP on perturbed
        h_opt = slsqp_run(h_pert, n, 2000, maxiter=200)
        c5_opt = compute_c5(h_opt)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5
