    zeros = list(np.where(h < 0.5)[0])

    for it in range(max_iters):
        if remaining() < 10:
            break
        i = ones[_RNG.randint(len(ones))]
        j = zeros[_RNG.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        nc = compute_c5(h)
        delta = nc - cur_c5

        if delta < 0 or _RNG.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = nc
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if nc < best_c5:
                best_c5 = nc
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T = max(T * cooling, T_min)
    return best_h, best_c5


def sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 200.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        offset = ki - (n - 1)
        if offset >= 0:
            np_ = n - offset
            tmp = np.zeros(n)
            tmp[offset:offset + np_] += (1.0 - x[:np_])
            tmp[:np_] -= x[offset:offset + np_]
            g += w[j] * tmp
        else:
            ao = -offset
            np_ = n - ao
            tmp = np.zeros(n)
            tmp[:np_] += (1.0 - x[ao:ao + np_])
            tmp[ao:ao + np_] -= x[:np_]
            g += w[j] * tmp
    g *= dx
    g += 400.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=5, maxiter=1000):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sm_fun, h, jac=sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxiter * 15})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def nesterov_gradient(h, n, top_k=5, max_iters=5000, step_init=0.02):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = np.maximum(c_top - c_top[-1] + 0.05, 0.01)
        w /= w.sum()
        for j, ki in enumerate(idx):
            offset = ki - (n - 1)
            if offset >= 0:
                np_ = n - offset
                tmp = np.zeros(n)
                tmp[offset:offset + np_] += (1.0 - h[:np_])
                tmp[:np_] -= h[offset:offset + np_]
                g += w[j] * tmp
            else:
                ao = -offset
                np_ = n - ao
                tmp = np.zeros(n)
                tmp[:np_] += (1.0 - h[ao:ao + np_])
                tmp[ao:ao + np_] -= h[:np_]
                g += w[j] * tmp
        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.12)
        else:
            h = best_h.copy()
            v *= 0.3
            step *= 0.6
        if it % 2000 == 0 and it > 0:
            momentum = min(momentum + 0.005, 0.93)
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def equalize_top(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-7:
            break

        # Build direction: reduce max, increase min
        g = np.zeros(n)
        for ki in idx:
            offset = ki - (n - 1)
            if offset >= 0:
                np_ = n - offset
                tmp = np.zeros(n)
                tmp[offset:offset + np_] += (1.0 - h[:np_])
                tmp[:np_] -= h[offset:offset + np_]
            else:
                ao = -offset
                np_ = n - ao
                tmp = np.zeros(n)
                tmp[:np_] += (1.0 - h[ao:ao + np_])
                tmp[ao:ao + np_] -= h[:np_]
            if ki == idx[0]:
                g -= tmp
            elif ki == idx[-1]:
                g += tmp
        g -= np.mean(g)
        dn = np.linalg.norm(g)
        if dn < 1e-14:
            break
        g /= dn

        improved = False
        for scale in [3, 1, 2, 0.5, 5, 0.2, 8, 1.5, 0.1]:
            alpha = 0.015 * scale
            h_try = proj(h + alpha * g, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT, _RNG
    _T0 = time.time()
    _BUDGET = budget_s
    _RNG = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = make_sturmian(parent_n, 0.6180339887)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h_bin_init = (best_h > 0.5).astype(float)

    # Phase 1: Thorough binary ILS at parent resolution
    # Use multiple diverse starting points
    alphas = [0.618, 0.707, 0.555, 0.733, 0.667, 0.75, 0.58, 0.4142, 0.685, 0.727]
    for ai, alpha in enumerate(alphas):
        if remaining() < 60:
            break
        h_start = h_bin_init.copy() if ai == 0 else make_sturmian(n, alpha)
        # Quick directed greedy
        h_g, c5_g = directed_greedy(h_start, n, max_rounds=80, top_peaks=5, n_candidates=10)
        # ILS
        h_il, c5_il = ils_binary(h_g, n, max_rounds=30, perturb_size=(2, 6))
        if c5_il < best_c5:
            best_c5 = c5_il
            best_h = h_il.copy()
            best_n = n
            h_bin_init = h_il.copy()

    # Phase 2: SA on best binary
    if remaining() > 120:
        sa_time_frac = 0.35
        h_sa, c5_sa = sa_binary(best_h, best_n, T_init=1.5, T_min=0.005,
                                cooling=0.9998, max_iters=300000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Continuous L-BFGS-B refinement
    h = best_h.copy()
    for beta in [500, 2000, 8000, 20000, 50000]:
        if remaining() < 15:
            break
        h_ref = lbfgs_pass(h, best_n, beta, top_k=6, maxiter=800)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h = h_ref.copy()

    # Phase 4: Nesterov gradient
    if remaining() > 40:
        h_ng, c5_ng = nesterov_gradient(h, best_n, top_k=5, max_iters=4000, step_init=0.015)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()
            h = h_ng.copy()

    # Phase 5: Equalize top correlations
    if remaining() > 20:
        h_eq, c5_eq = equalize_top(h, best_n, top_k=6, max_iters=250)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Binary polish + continuous refine
    if remaining() > 40:
        h_bin2 = (best_h > 0.5).astype(float)
        h_bp, c5_bp = directed_greedy(h_bin2, best_n, max_rounds=150, top_peaks=6, n_candidates=15)
        h_bp, c5_bp = random_greedy(h_bp, best_n, max_iters=30000)
        # Continuous from this binary
        h_cr = lbfgs_pass(h_bp, best_n, 10000, top_k=6, maxiter=600)
        c5_cr = compute_c5(h_cr)
        c5_cr2 = compute_c5(h_bp)
        if min(c5_cr, c5_cr2) < best_c5:
            best_c5 = min(c5_cr, c5_cr2)
            best_h = (h_cr if c5_cr <= c5_cr2 else h_bp).copy()

    # Phase 7: Multi-resolution exploration
    n_candidates = sorted(set([
        best_n, max(128, best_n - 4), best_n + 4, best_n + 8,
        best_n - 8, int(best_n * 1.25), int(best_n * 1.1),
    ]))
    n_candidates = [nn for nn in n_candidates if 200 <= nn <= 640 and nn % 2 == 0]
    n_candidates = [nn for nn in n_candidates if nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:4]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)
        h_b = (h_feas > 0.5).astype(float)

        # Quick binary optimization
        h_b, c5_b = directed_greedy(h_b, n_t, max_rounds=60, top_peaks=5, n_candidates=10)
        h_b, c5_b = random_greedy(h_b, n_t, max_iters=15000)

        # Continuous
        h_c = lbfgs_pass(h_b, n_t, 5000, top_k=5, maxiter=500)
        c5_c = compute_c5(h_c)
        h_c2 = lbfgs_pass(h_c, n_t, 20000, top_k=5, maxiter=400)
        c5_c2 = compute_c5(h_c2)

        cand_c5 = min(c5_b, c5_c, c5_c2)
        if c5_c2 <= cand_c5:
            cand_h = h_c2
        elif c5_c <= cand_c5:
            cand_h = h_c
        else:
            cand_h = h_b

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    # Multiple L-BFGS-B passes with increasing sharpness
    for beta in [15000, 30000, 60000, 100000]:
        if remaining() < 10:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Nesterov from current best
    if remaining() > 25:
        h_ng2, c5_ng2 = nesterov_gradient(best_h, best_n, top_k=4, max_iters=3000, step_init=0.01)
        if c5_ng2 < best_c5:
            best_c5 = c5_ng2
            best_h = h_ng2.copy()

    # Final equalize
    if remaining() > 10:
        h_eq2, c5_eq2 = equalize_top(best_h, best_n, top_k=7, max_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final binary + continuous
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = directed_greedy(h_bin_f, best_n, max_rounds=100, top_peaks=6, n_candidates=12)
        h_f = lbfgs_pass(h_bf, best_n, 20000, top_k=5, maxiter=400)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        elif c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5
        best_h = proj(best_h, best_n)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
