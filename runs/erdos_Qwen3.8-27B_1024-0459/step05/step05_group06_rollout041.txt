def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)

            h[mask] += diff / cnt
        elif diff > 0:

            order = np.argsort(h)
            for idx in order:
                room = 1.0 - h[idx]
                add = min(room, diff)
                h[idx] += add
                diff -= add
                if diff < 1e-15:

                sub = min(room, -diff)
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-15:
                    break

    return h
