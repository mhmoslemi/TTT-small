def alternating_lp(h, n, max_iters=50, time_limit=120):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    start = time.time()
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() - start > time_limit or remaining_time() < 10:
            break
        
        # Compute correlation coefficients for each shift
        # corr(k) = sum_i h[i] * (1-h[i+k]) for valid (i, i+k)
        # For shift k (0 to 2n-2), the offset is k-(n-1)
        
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        
        for k in range(2*n - 1):
            off = k - (n - 1)
            if off >= 0:
