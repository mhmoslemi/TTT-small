    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA

    S = np.sum(ez)

            a = -off
            np_ = n - a

def subgradient_step(h, n, step_size=0.01):
    """One step of projected subgradient on the max correlation."""
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:5]
    m = corr[top_idx[0]]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / len(top_idx)
    # Gradient of weighted sum of correlations
    g = np.zeros(n)
    for k in top_idx:
        wk = 1.0 / len(top_idx)
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:])
            g[a:] -= wk * h[:np_]
    h_new = h - step_size * g
    h_new = project_feasible(h_new, n)
    return h_new

        h_sa, c5_sa = simulated_annealing(best_h, n, time_limit=t_alloc3, rng=rng)
        if c5_sa < best_c5 - 1e-14:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Post-SA: quick local search
    if remaining_time() > 30:
        h_l2, c5_l2 = binary_2swap_deterministic(best_h, n, time_limit=min(60, remaining_time() * 0.15))
        if c5_l2 < best_c5 - 1e-14:
            best_c5 = c5_l2
            best_h = h_l2.copy()

    # Phase 4: Continuous refinement with subgradient (15% time)
    if remaining_time() > 30:
        h_cont = best_h.copy()
        h_cont = project_feasible(h_cont, n)
        sg_time = min(150, remaining_time() * 0.15)
        sg_start = time.time()
        for i in range(20000):
            if time.time() - sg_start > sg_time or remaining_time() < 10:
                break
            step_sz = 0.02 * (1.0 - (time.time() - sg_start) / sg_time * 0.8)
            h_new = subgradient_step(h_cont, n, step_sz)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            if i % 500 == 0 and remaining_time() < 15:
                break
        h_cont = best_h.copy()

    # Phase 5: SLSQP continuous optimization (10% time)
    for beta in [500, 5000, 50000, 200000]:
        if remaining_time() < 15:
            break
        h_opt = slsqp_opt(best_h, n, beta, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Top-k equalization
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(best_h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 7: Multi-resolution attempt
    if remaining_time() > 60:
        for n_trial in [n + 24, n - 24, n + 48, n + 96]:
            if n_trial < 100 or n_trial > 1000 or n_trial % 2 != 0:
                continue
            if remaining_time() < 30:
                break
            h_interp = interp_to(best_h, n, n_trial)
            h_feas = project_feasible(h_interp, n_trial)
            c5_int = compute_c5(h_feas)
            # Quick SLSQP
            h_opt = slsqp_opt(h_feas, n_trial, 5000, maxiter=300)
            c5_opt = compute_c5(h_opt)
            h_opt, c5_opt = top_k_equalize(h_opt, n_trial, top_k=5, max_iters=200)
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                n = n_trial

    # Phase 8: Final intensive polish
    n_final = len(best_h)
    if remaining_time() > 15:
        h_final = slsqp_opt(best_h, n_final, 500000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 8:
        h_eq2, c5_eq2 = top_k_equalize(best_h, n_final, top_k=8, max_iters=300)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final subgradient sweep
    if remaining_time() > 10:
        h_sg = best_h.copy()
        for i in range(500):
            if remaining_time() < 5:
                break
            step_sz = 0.005
            h_sg = subgradient_step(h_sg, n_final, step_sz)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5 - 1e-14:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n_final)
