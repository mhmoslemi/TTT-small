        w = ex / np.sum(ex)
        
        # g[i] = dx * sum_k w[k] * ((1-h[(i-k) mod n]) - h[(i+k) mod n])
        # = dx * (sum(w) - (w*circconv*x)[i] - (w_rev*circconv*x)[i])
        W = np.fft.fft(w)
        X = np.fft.fft(x)
        w_rev = w[::-1].copy()
        w_rev[0] = w[0]  # keep DC
        W_rev = np.fft.fft(w_rev)
        
        term2 = np.fft.ifft(W * X).real
        term3 = np.fft.ifft(W_rev * X).real
        g = dx * (np.sum(w) - term2 - term3)
        return g
    
    def constraint_fun(x):
        return np.sum(x) - target
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(objective, h, jac=gradient, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < objective(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def nlsqp_smooth(h, n, beta, maxiter=2000):
    """L-BFGS-B with smooth max using FFT for gradient."""
    dx = 2.0 / n
    h = project_feasible(h, n)
    
    target = n / 2.0
    
    def objective(x):
        corr = circular_corr_fft(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx
    
    def gradient(x):
        corr = circular_corr_fft(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        
        W = np.fft.fft(w)
        X = np.fft.fft(x)
        w_rev = np.zeros(n)
        w_rev[0] = w[0]
        w_rev[1:] = w[n-1:0:-1]
        W_rev = np.fft.fft(w_rev)
        
        term2 = np.fft.ifft(W * X).real
        term3 = np.fft.ifft(W_rev * X).real
        g = dx * (np.sum(w) - term2 - term3)
        return g
    
    # Use penalty for the sum constraint
    def penalty_obj(x):
        base = objective(x)
        s = np.sum(x) - target
        penalty = 100.0 * s * s
        return base + penalty
    
    def penalty_grad(x):
        base_g = gradient(x)
        s = np.sum(x) - target
        pen_g = base_g + 200.0 * s * np.ones(n)
        return pen_g
    
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(penalty_obj, h, jac=penalty_grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0})
        h_out = project_feasible(res.x, n)
        return h_out
    except Exception:
        pass
    return h

def top_k_equalize(h, n, top_k=5, max_iters=200):
    """Directly push down the top-k correlations using projected subgradient."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    step_sizes = [0.02, 0.05, 0.01, 0.08, 0.03, 0.015, 0.1]
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        
        # Build subgradient for top-k
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        
        g /= top_k
        g -= np.mean(g)  # projection onto sum=const hyperplane
        
        improved = False
        for step in step_sizes:
            h_new = project_feasible(h - step * g, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c_new
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.005, 0.002, 0.001]:
                h_new = project_feasible(h - step * g, n)
                c_new = compute_c5(h_new)
                if c_new < best_c5 - 1e-14:
                    h = h_new
                    best_c5 = c_new
                    improved = True
                    break
            if not improved:
                break
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def spectral_optimize(h, n, n_modes=30, maxiter=200, beta=1000):
    """Optimize in Fourier space (first n_modes coefficients)."""
    # Get initial Fourier coefficients
    H = np.fft.fft(h)
    
    # Parameterize: h = IFFT(c), where c[0] is fixed, c[k] and c[n-k] are conjugate
    # We optimize c[1:n_modes] (and conjugates)
    
    def params_to_h(params):
        c = np.zeros(n, dtype=complex)
        c[0] = H[0].real  # DC: n/2 for our constraint
        for k in range(1, n_modes + 1):
            c[k] = params[2*(k-1)] + 1j * params[2*(k-1)+1]
            c[n-k] = np.conj(c[k])
        # Fill remaining with initial values
        for k in range(n_modes + 1, n // 2 + 1):
            c[k] = H[k]
        h_out = np.real(np.fft.ifft(c))
        return h_out
    
    def objective(params):
        h_out = params_to_h(params)
        h_out = np.clip(h_out, 0.0, 1.0)
        h_out = project_feasible(h_out, n)
        corr = circular_corr_fft(h_out)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    # Initial params from H
    params0 = np.zeros(2 * n_modes)
    for k in range(1, n_modes + 1):
        params0[2*(k-1)] = H[k].real
        params0[2*(k-1)+1] = H[k].imag
    
    bounds = [(-1.0, 1.0)] * len(params0)
    
    try:
        res = minimize(objective, params0, method='Nelder-Mead',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
        h_opt = params_to_h(res.x)
        h_opt = project_feasible(h_opt, n)
        return h_opt
    except Exception:
        return h
