    top_k = np.argsort(combined_no_diag)[::-1][:12]

    candidates = set([0.0, 1.0])
    for k1 in top_k:
        for k2 in top_k:
            if k1 == k2:
                continue
            b_diff = A[k1] - A[k2]
            if abs(b_diff) > 1e-14:
                t_cross = (base[k2] - base[k1]) / b_diff
                if 0.0 < t_cross < 1.0:
                    candidates.add(t_cross)
        # Cross with diagonal: base[k] + A[k]*t = base[diag] + t - t^2
        # t^2 + (A[k]-1)*t + (base[k] - base[diag]) = 0
        a_coef = 1.0
        b_coef = A[k1] - 1.0
        c_coef = base[k1] - base[diag_idx]
        disc = b_coef * b_coef - 4.0 * a_coef * c_coef
        if disc >= 0:
            sqrt_disc = np.sqrt(disc)
            t1 = (-b_coef + sqrt_disc) / (2.0 * a_coef)
            t2 = (-b_coef - sqrt_disc) / (2.0 * a_coef)
            if 0.0 < t1 < 1.0:
                candidates.add(t1)
            if 0.0 < t2 < 1.0:
                candidates.add(t2)

    for t in candidates:
        t = max(0.0, min(1.0, t))
        mid = base + A * t
        mid[diag_idx] = base[diag_idx] + t - t * t
        f_val = np.max(mid)
        if f_val < best_f - 1e-15:
            best_f = f_val
            best_t = t

    # Golden section refinement around best_t
    if 0.0 < best_t < 1.0:
        lo = max(0.0, best_t - 0.05)
        hi = min(1.0, best_t + 0.05)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)

        def _eval_fn(tt):
            mid = base + A * tt
            mid[diag_idx] = base[diag_idx] + tt - tt * tt
            return np.max(mid)

        f1 = _eval_fn(c1)
        f2 = _eval_fn(c2)
        for _ in range(45):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = _eval_fn(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = _eval_fn(c2)
        t_ref = (lo + hi) / 2.0
        f_ref = _eval_fn(t_ref)
        if f_ref < best_f - 1e-15:
            best_f = f_ref
            best_t = t_ref

    return best_t, best_f


def coord_descent_v2(h, n, max_sweeps=200):
    """Improved coordinate descent with exact 1D minimization."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = np.max(best_corr)
            t_best, f_best = exact_1d_min(base, A, ncorr)

            if f_best < f_old - 1e-14:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break

    return best, best_c5


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def structured_init(n, rng):
    """Alternating blocks with some randomness."""
    h = np.zeros(n)
    block_size = max(2, n // 128)
    for i in range(n):
        block = i // block_size
        if block % 2 == 0:
            h[i] = 1.0 if rng.random() < 0.55 else 0.0
        else:
            h[i] = 1.0 if rng.random() < 0.45 else 0.0
    return project_feasible(h, n)


def binary_local_search(h, n, max_flips=2000):
    """Try flipping groups of binary-valued coordinates."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    best_c5 = np.max(best_corr) * dx
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    # Identify near-binary coordinates
    binary_mask = (best < 0.15) | (best > 0.85)
    binary_idx = np.where(binary_mask)[0]

    for flip_iter in range(max_flips):
        if remaining_time() < 5:
            break
        if len(binary_idx) < 2:
            break

        # Try flipping 1-3 random binary coords
        n_flip = 1 + (flip_iter % 3)
        if len(binary_idx) < n_flip:
            n_flip = len(binary_idx)
        if n_flip < 1:
            break

        pick = rng_choice(binary_idx, min(n_flip, len(binary_idx)), None)
        old_vals = best[pick].copy()

        # Flip: 0→1, 1→0 (approximately)
        new_vals = np.where(old_vals < 0.5, 1.0, 0.0)

        # Compute new correlation
        delta = new_vals - old_vals
        new_corr = best_corr.copy()
        for idx in pick:
            d = delta[np.where(pick == idx)[0][0]]
            i_plus = idx + offsets
            i_minus = idx - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)
            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0
            new_corr += A * d
            new_corr[n - 1] += new_vals[np.where(pick == idx)[0][0]] * (1 - new_vals[np.where(pick == idx)[0][0]]) - old_vals[np.where(pick == idx)[0][0]] * (1 - old_vals[np.where(pick == idx)[0][0]])

        f_new = np.max(new_corr)
        if f_new < np.max(best_corr) - 1e-14:
            best[pick] = new_vals
            best_corr = new_corr.copy()
            best_c5 = f_new * dx
            # Renormalize
            best = project_feasible(best, n)
            best_corr = compute_corr(best).astype(float)
            best_c5 = np.max(best_corr) * dx

    return best, best_c5


def rng_choice(arr, k, rng_obj):
    idx = np.random.permutation(len(arr))[:k]
    return arr[idx]


def targeted_pressure(h, n, n_active=10, n_steps=100):
    """Targeted optimization: focus on the worst k values."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for step in range(n_steps):
        if remaining_time() < 5:
            break

        # Find active (worst) k values
        max_corr = np.max(best_corr)
        threshold = max_corr - 0.1 * max_corr
        active_k = np.where(best_corr >= threshold)[0]
        if len(active_k) == 0:
            active_k = np.array([np.argmax(best_corr)])
        if len(active_k) > n_active:
            active_k = active_k[np.argsort(best_corr[active_k])[::-1][:n_active]]

        # For each active k, find the coordinate that most reduces it
        best_improvement = 0.0
        best_i = -1
        best_new_val = 0.0

        # Sample some coordinates to try
        trial_indices = np.random.permutation(n)[:min(50, n)]

        for i in trial_indices:
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            # Try to reduce the max by adjusting h[i]
            for t_try in [0.0, 1.0, 0.5, 1.0 - old]:
                if t_try == old or t_try < 0 or t_try > 1:
                    continue
                new_max = 0.0
                for k_idx in active_k:
                    val = best_corr[k_idx] + A[k_idx] * (t_try - old)
                    if k_idx == n - 1:
                        val = best_corr[k_idx] + t_try * (1 - t_try) - old * (1 - old)
                    new_max = max(new_max, val)
                # Also check a few random non-active k
                for k_idx in np.random.choice(ncorr, min(20, ncorr), replace=False):
                    val = best_corr[k_idx] + A[k_idx] * (t_try - old)
                    if k_idx == n - 1:
                        val = best_corr[k_idx] + t_try * (1 - t_try) - old * (1 - old)
                    new_max = max(new_max, val)

                improvement = max_corr - new_max
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_new_val = t_try

        if best_i >= 0 and best_improvement > 1e-10:
            old = best[best_i]
            i = best_i
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)
            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0
            best[i] = best_new_val
            best_corr += A * (best_new_val - old)
            best_corr[n - 1] += best_new_val * (1 - best_new_val) - old * (1 - old)
            best_c5 = np.max(best_corr) * dx

    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Quick coordinate descent at original resolution
    h = best_h.copy()
    h, c5 = coord_descent_v2(h, best_n, max_sweeps=80)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP at original resolution
    if remaining_time() > 60:
        h = best_h.copy()
        for beta in [2000, 10000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_refine(h, best_n, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new

        # Coordinate descent after SLSQP
        h = best_h.copy()
        h, c5 = coord_descent_v2(h, best_n, max_sweeps=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Interpolate to higher resolution (n=1280)
    if remaining_time() > 100:
        n_high = 2 * best_n
        if n_high <= 1920:
            h_high = interp_to(best_h, best_n, n_high)
            h_high = project_feasible(h_high, n_high)
            c5_high = compute_c5(h_high)

            # Coordinate descent at higher resolution
            h_high, c5_high = coord_descent_v2(h_high, n_high, max_sweeps=100)

            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

            # SLSQP at higher resolution
            if remaining_time() > 80:
                h_slq = slsqp_refine(best_h, best_n, 5000, 400)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()

                # More coordinate descent
                h = best_h.copy()
                h, c5 = coord_descent_v2(h, best_n, max_sweeps=80)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    # Phase 4: Even higher resolution (n=1920 or 2560)
    if remaining_time() > 150:
        n_target = min(2560, 4 * best_n)
        if n_target > best_n:
            h_high2 = interp_to(best_h, best_n, n_target)
            h_high2 = project_feasible(h_high2, n_target)
            h_high2, c5_high2 = coord_descent_v2(h_high2, n_target, max_sweeps=60)

            if c5_high2 < best_c5:
                best_c5 = c5_high2
                best_h = h_high2.copy()
                best_n = n_target

            if remaining_time() > 60:
                h_slq = slsqp_refine(best_h, best_n, 8000, 300)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()

                h = best_h.copy()
                h, c5 = coord_descent_v2(h, best_n, max_sweeps=50)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    # Phase 5: Targeted pressure optimization
    if remaining_time() > 60:
        h = best_h.copy()
        n_cur = best_n
        h, c5 = targeted_pressure(h, n_cur, n_active=8, n_steps=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Coordinate descent after pressure
        h = best_h.copy()
        h, c5 = coord_descent_v2(h, n_cur, max_sweeps=60)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Multiple perturbation restarts
    if remaining_time() > 120:
        n_cur = best_n
        n_restart = 6
        for p in range(n_restart):
            if remaining_time() < 30:
                break
            h_pert = best_h.copy()
            noise_scale = 0.1 / (1 + p * 0.4)
            mask = rng.random(n_cur) < (0.25 - p * 0.02)
            h_pert[mask] += rng.randn(int(np.sum(mask))) * noise_scale
            h_pert = project_feasible(h_pert, n_cur)

            h_new, c5_new = coord_descent_v2(h_pert, n_cur, max_sweeps=60)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

            if remaining_time() > 40:
                h_slq = slsqp_refine(h_new, n_cur, 5000, 200)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()

    # Phase 7: Try different structured initializations at current resolution
    if remaining_time() > 80:
        n_cur = best_n
        h_tm = thue_morse_init(n_cur)
        h_tm, c5_tm = coord_descent_v2(h_tm, n_cur, max_sweeps=60)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            if remaining_time() > 30:
                h_slq = slsqp_refine(best_h, n_cur, 10000, 300)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()

    # Phase 8: Binary local search
    if remaining_time() > 40:
        h = best_h.copy()
        n_cur = best_n
        h, c5 = binary_local_search(h, n_cur, max_flips=500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 9: Final SLSQP polish with very high beta
    if remaining_time() > 15:
        n_cur = best_n
        h = best_h.copy()
        for beta in [20000, 100000, 500000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(h, n_cur, beta, 500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new

    # Phase 10: Final coordinate descent
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = coord_descent_v2(best_h, n_cur, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: One more targeted pressure
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = targeted_pressure(best_h, n_cur, n_active=6, n_steps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final: coordinate descent + SLSQP
    if remaining_time() > 8:
        n_cur = best_n
        h_new, c5_new = coord_descent_v2(best_h, n_cur, max_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 5:
        n_cur = best_n
        h_slq = slsqp_refine(best_h, n_cur, 200000, 200)
        c5_slq = compute_c5(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
