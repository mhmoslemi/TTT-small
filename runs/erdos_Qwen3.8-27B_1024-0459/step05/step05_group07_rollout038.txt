            v = max(v, corr_0 + t - t*t - old + old*old)
            return v
        
        f_cur = eval_f(old)
        if f_cur <= max_corr + 1e-12:
            continue
        
        lo = 0.0
        hi = 1.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = eval_f(c1)
        f2 = eval_f(c2)
        for _ in range(50):
            if f1 < f2:
                hi = c2; c2 = c1; f2 = f1
                c1 = hi - gr * (hi - lo); f1 = eval_f(c1)
            else:
                lo = c1; c1 = c2; f1 = f2
                c2 = lo + gr * (hi - lo); f2 = eval_f(c2)
        
        t_best = (lo + hi) / 2.0
        f_best = eval_f(t_best)
        
        if f_best < f_cur - 1e-12:
            delta = t_best - old
            if pos_max > 0:
                h_corr[n:n+pos_max] += delta * (1.0 - h_pos)
            if neg_max > 0:
                h_corr[n-1-i:n-1] -= delta * h_neg
            h_corr[n-1] += t_best - t_best*t_best - old + old*old
            h_refined[i] = t_best
            max_corr = np.max(h_corr)
    
    h_refined = project_feasible(h_refined, n)
    c5_final = compute_c5(h_refined)
    return h_refined, c5_final


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def slsqp_refine(h, n, beta, maxiter=300):
    """SLSQP refinement using log-sum-exp smoothing."""
    N_WORK = n
    B_WORK = beta
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(B_WORK * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        return (m + np.log(wsum) / B_WORK) * (2.0 / N_WORK)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(B_WORK * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        grad = np.zeros(N_WORK)
        for k in range(2 * N_WORK - 1):
            w = weights[k]
            if w < 1e-14:
                continue
            offset = k - (N_WORK - 1)
            if offset >= 0:
                npairs = N_WORK - offset
                if npairs > 0:
                    grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[offset:offset+npairs]
            else:
                aoff = -offset
                npairs = N_WORK - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                    grad[aoff:aoff+npairs] -= w * x[:npairs]
        grad *= (2.0 / N_WORK)
        return grad
    
    def cons_fun(x):
        return np.sum(x) - N_WORK / 2.0
    
    def cons_jac(x):
        return np.ones(N_WORK)
    
    bounds = [(0.0, 1.0)] * N_WORK
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        # Thue-Morse initialization
        h_init = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            h_init[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(h_init, best_n)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    
    # Phase 1: Thorough coordinate descent from parent
    h = best_h.copy()
    h, c5 = fast_coord_descent(h, n, max_sweeps=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Phase 2: SLSQP with moderate beta
    if remaining_time() > 60:
        h = best_h.copy()
        h_new = slsqp_refine(h, n, 1000, 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h, c5 = fast_coord_descent(h_new, n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 3: Binary snap + swap search + refine back
    if remaining_time() > 60:
        h_snap, c5_snap = snap_to_binary_and_refine(best_h, n)
        if c5_snap < best_c5:
            best_c5 = c5_snap
            best_h = h_snap.copy()
            # Further coordinate descent from snapped solution
            h_cd, c5_cd = fast_coord_descent(h_snap, n, max_sweeps=100)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    
    # Phase 4: SLSQP with higher beta
    if remaining_time() > 60:
        h = best_h.copy()
        h_new = slsqp_refine(h, n, 5000, 500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h, c5 = fast_coord_descent(h_new, n, max_sweeps=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 5: Perturbation + coordinate descent
    if remaining_time() > 90:
        for p in range(5):
            if remaining_time() < 30:
                break
            h_pert = best_h.copy()
            noise_scale = 0.1 / (1.0 + p * 0.5)
            mask = rng.random(n) < 0.2
            n_masked = int(np.sum(mask))
            if n_masked > 0:
                h_pert[mask] += rng.randn(n_masked) * noise_scale
            h_pert = project_feasible(h_pert, n)
            
            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=80)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 6: Try higher resolution
    if remaining_time() > 120:
        high_n = 2 * n
        if high_n <= 1280:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            
            h_new, c5_new = fast_coord_descent(h_high, high_n, max_sweeps=60)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = high_n
                
                if remaining_time() > 60:
                    h_slq = slsqp_refine(h_new, high_n, 3000, 400)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        best_n = high_n
                        
                    h_cd, c5_cd = fast_coord_descent(h_new, high_n, max_sweeps=50)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd.copy()
                        best_n = high_n
    
    # Phase 7: Final SLSQP polish
    if remaining_time() > 30:
        n_cur = best_n
        h = best_h.copy()
        for beta in [10000, 50000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_refine(h, n_cur, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    # Phase 8: Final coordinate descent
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 9: One more binary snap attempt
    if remaining_time() > 20:
        h_snap, c5_snap = snap_to_binary_and_refine(best_h, best_n)
        if c5_snap < best_c5:
            best_c5 = c5_snap
            best_h = h_snap.copy()
    
    # Phase 10: Final polish
    if remaining_time() > 10:
        h_new, c5_new = fast_coord_descent(best_h, best_n, max_sweeps=80)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
