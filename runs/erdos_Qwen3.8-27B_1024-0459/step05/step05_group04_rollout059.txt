    best_val = cur_val
    dx = 2.0 / n
    
    T = 0.03
    T_min = 0.0005
    cooling = 0.9995
    
    start = time.time()
    no_improve = 0
    max_no_improve = 3000
    
    while time.time() - start < time_limit and remaining() > 10:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Generate candidates from top shifts
        top_shifts = np.argsort(-corr)[:14]
        candidates = []
        
        for s in top_shifts[:10]:
            off = s - (n - 1)
            if off >= 0:
                m = n - off
                mask = (h[:m] == 1) & (h[off:off+m] == 0)
                idx = np.where(mask)[0]
                if len(idx) > 0:
                    for ci in idx[:25]:
                        candidates.append((ci + off, ci))
            else:
                ao = -off
                m = n - ao
                mask = (h[:m] == 1) & (h[ao:ao+m] == 0)
                idx = np.where(mask)[0]
                if len(idx) > 0:
                    for ci in idx[:25]:
                        candidates.append((ci, ci + ao))
        
        # Random candidates
        n_rand = 50
        for _ in range(n_rand):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        # Best move selection
        best_move = None
        best_delta = 0.0
        accept_delta = 0.0
        
        seen = set()
        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            
            dr = swap_delta_v2(h, i, j, n)
            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            delta = new_max - cur_val
            
            if delta < best_delta:
                best_delta = delta
                best_move = (i, j, dr)
            elif T > T_min and delta < 0.08 and rng.random() < np.exp(-delta / max(T, 1e-6)):
                if best_move is None or delta < accept_delta:
                    accept_delta = delta
                    if best_move is None:
                        best_move = (i, j, dr)
        
        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_val = float(np.max(corr))
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cooling
        
        if no_improve > max_no_improve:
            # Perturbation: 5 random swaps
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(5):
                if len(ones) > 0 and len(zeros) > 0:
                    ii = ones[rng.randint(len(ones))]
                    jj = zeros[rng.randint(len(zeros))]
                    dr2 = swap_delta_v2(h, ii, jj, n)
                    h[ii] = 0
                    h[jj] = 1
                    corr = corr + dr2
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
            cur_val = float(np.max(corr))
            no_improve = 0
    
    return best_h, best_val * dx

def projected_gradient_max(h_init, n, time_limit=60):
    """Projected gradient with momentum to minimize max correlation."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    momentum = np.zeros(n)
    lr = 0.002
    top_k = 8
    
    start = time.time()
    
    while time.time() - start < time_limit and remaining() > 10:
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        top_idx = np.argsort(-corr)[:top_k]
        
        # Subgradient of max (use top-k average for smoother gradient)
        g = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                g[:m] += (1.0 - h[s:s+m])
                g[s:] -= h[:m]
            else:
                ao = -s
                m = n - ao
                g[:m] += (1.0 - h[ao:ao+m])
                g[ao:ao+m] -= h[:m]
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Nesterov momentum
        momentum = 0.9 * momentum - lr * g
        h_new = h + momentum
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.02)
        else:
            h = h_new  # still move (momentum carries through)
            lr *= 0.9
    
    return best_h, best_c5

def slsqp_refine(h_init, n, beta, maxiter=300):
    """SLSQP with log-sum-exp smoothing."""
    h = project_feasible(h_init, n)
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = beta * (corr - m)
        sh = np.clip(sh, -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = beta * (corr - m)
        sh = np.clip(sh, -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-12:
                continue
            off = k - (n - 1)
            if off >= 0:
                m2 = n - off
                g[off:off+m2] += wk * (1.0 - x[:m2])
                g[:m2] -= wk * x[off:off+m2]
            else:
                ao = -off
                m2 = n - ao
                g[:m2] += wk * (1.0 - x[ao:ao+m2])
                g[ao:ao+m2] -= wk * x[:m2]
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h_init, n, top_k=6, max_iters=300):
    """Iteratively equalize top correlation shifts."""
    h = project_feasible(h_init, n)
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_max = corr[top_idx[0]]
        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        
        # Gradient to push down top, pull up bottom
        g_top = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                m = n - off
                g_top[off:off+m] += (1.0 - h[:m])
                g_top[:m] -= h[off:off+m]
            else:
                ao = -off
                m = n - ao
                g_top[:m] += (1.0 - h[ao:ao+m])
                g_top[ao:ao+m] -= h[:m]
        
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                m = n - off
                g_bot[off:off+m] += (1.0 - h[:m])
                g_bot[:m] -= h[off:off+m]
            else:
                ao = -off
                m = n - ao
                g_bot[:m] += (1.0 - h[ao:ao+m])
                g_bot[ao:ao+m] -= h[:m]
        
        d = g_bot - g_top
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.005, 0.01, 0.02, 0.03, 0.008, 0.015, 0.05, 0.003, 0.04, 0.007]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 384
        parent_h = thue_morse_bin(parent_n)
    
    best_h = project_feasible(parent_h, parent_n)
    best_c5 = compute_c5(best_h)
    best_n = parent_n
    
    # Phase 1: Binary optimization at parent resolution
    h_bin = (best_h > 0.5).astype(float)
    t1 = min(250, int(remaining() * 0.35))
    if t1 > 30:
        h_b1, c5_b1 = binary_sa(h_bin, parent_n, t1, rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = parent_n
    
    # Phase 2: Thue-Morse start at parent resolution
    if remaining() > 40:
        tm = thue_morse_bin(parent_n)
        h_b2, c5_b2 = binary_sa(tm, parent_n, min(120, int(remaining() * 0.15)), rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = parent_n
    
    # Phase 3: Try other resolutions
    resolutions = []
    for delta in [-48, -32, -16, -8, 8, 16, 32, 48, 64, 96]:
        r = parent_n + delta
        if 128 <= r <= 1024 and r % 2 == 0:
            resolutions.append(r)
    for mult in [0.8, 1.2, 1.5, 0.67]:
        r = int(parent_n * mult // 2 * 2)
        if 128 <= r <= 1024 and r not in resolutions:
            resolutions.append(r)
    
    resolutions.sort(key=lambda x: abs(x - parent_n))
    
    for n_r in resolutions[:5]:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_r)
        h_bin_r = (h_interp > 0.5).astype(float)
        # Ensure correct count
        target_r = n_r // 2
        n_ones = int(np.sum(h_bin_r))
        if n_ones != target_r:
            if n_ones > target_r:
                idx = np.where(h_bin_r == 1)[0]
                h_bin_r[idx[:n_ones - target_r]] = 0
            else:
                idx = np.where(h_bin_r == 0)[0]
                h_bin_r[idx[:target_r - n_ones]] = 1
        
        t_r = min(80, int(remaining() * 0.1))
        if t_r > 15:
            h_br, c5_br = binary_sa(h_bin_r, n_r, t_r, rng)
            if c5_br < best_c5:
                best_c5 = c5_br
                best_h = h_br.copy()
                best_n = n_r
        
        # Also try Thue-Morse at this resolution
        if remaining() > 25 and n_r != parent_n:
            tm_r = thue_morse_bin(n_r)
            h_br2, c5_br2 = binary_sa(tm_r, n_r, min(50, int(remaining() * 0.06)), rng)
            if c5_br2 < best_c5:
                best_c5 = c5_br2
                best_h = h_br2.copy()
                best_n = n_r
    
    # Phase 4: Continuous refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    # Small perturbation
    h_pert = h + rng.uniform(-0.005, 0.005, n)
    h_pert = project_feasible(h_pert, n)
    
    # Projected gradient
    t_pg = min(60, int(remaining() * 0.15))
    if t_pg > 10:
        h_pg, c5_pg = projected_gradient_max(h_pert, n, t_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg
    
    # SLSQP with increasing beta
    for beta in [200, 1000, 5000, 20000]:
        if remaining() < 15:
            break
        h_s = slsqp_refine(h, n, beta, maxiter=400)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            h = h_s
    
    # Phase 5: Top-shifts equalization
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq
    
    # Phase 6: More SLSQP with very high beta
    for beta in [50000, 100000, 200000]:
        if remaining() < 10:
            break
        h_s = slsqp_refine(h, n, beta, maxiter=300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            h = h_s
    
    # Phase 7: Try slightly higher resolution
    if remaining() > 80:
        n_fine = best_n + 64
        h_fine = interp_to(best_h, best_n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        
        h_fine, c5_f = projected_gradient_max(h_fine, n_fine, min(40, int(remaining() * 0.2)))
        if remaining() > 15:
            h_fine = slsqp_refine(h_fine, n_fine, 10000, maxiter=300)
            c5_f = compute_c5(h_fine)
        if remaining() > 10:
            h_fine, c5_f2 = top_shifts_equalize(h_fine, n_fine, top_k=6, max_iters=200)
            c5_f = min(c5_f, c5_f2)
            h_fine = project_feasible(h_fine, n_fine)
            c5_f = compute_c5(h_fine)
        
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_fine.copy()
            best_n = n_fine
    
    # Phase 8: Final binary refinement at best resolution
    if remaining() > 30:
        n_f = best_n
        h_bf = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_sa(h_bf, n_f, min(60, int(remaining() * 0.3)), rng)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
            best_n = n_f
    
    # Phase 9: Final continuous polish
    if remaining() > 15:
        n_f = best_n
        h_f = best_h + rng.uniform(-0.003, 0.003, n_f)
        h_f = project_feasible(h_f, n_f)
        h_f, c5_f = projected_gradient_max(h_f, n_f, min(30, int(remaining() * 0.3)))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = n_f
    
    # Ensure feasibility and final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
