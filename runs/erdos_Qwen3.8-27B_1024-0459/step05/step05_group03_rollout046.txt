_T0 = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.count_nonzero(free) > 0:
            h[free] += diff / np.count_nonzero(free)
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

# BB-step projected gradient on smooth max
def bb_gradient_optimize(h, n, beta, max_iters=2000):
    h = project_feasible(h, n)
    dx = 2.0 / n
    h_prev = h.copy()
    g_prev = None
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex) * dx
        
        # Compute gradient of weighted correlation sum
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:a + np_])
                g[a:a + np_] -= wk * h[:np_]
        
        # BB step size
        if g_prev is not None:
            s = h - h_prev
            y = g - g_prev
            yy = np.dot(y, y)
            if yy > 1e-14:

            best_h = h_new.copy()

For a swap (i: 1→0, j: 0→1), the change in corr[k] for the full correlation:

corr_new[k] = corr_old[k] + Δ_i[k] + Δ_j[k]

where Δ_i[k] is the change due to h[i] going from 1 to 0, and Δ_j[k] is the change due to h[j] going from 0 to 1.

For Δ_i[k] (h[i]: 1→0):
- In the term h[t]*(1-h[t+k-(n-1)]), if t=i: old contribution is 1*(1-h[i+k-(n-1)]), new is 0
  So Δ_i_from_first[k] = -(1-h[i+k-(n-1)]) if i+k-(n-1) is valid
- In the term where i appears in the second array: we need t+k-(n-1)=i, so t=i-k+(n-1)
  Old: h[i-k+(n-1)]*(1-1) = 0, New: h[i-k+(n-1)]*(1-0) = h[i-k+(n-1)]
  So Δ_i_from_second[k] = h[i-k+(n-1)] if i-k+(n-1) is valid

Similarly for Δ_j[k] (h[j]: 0→1):
- First array (t=j): old: 0*(1-h[j+k-(n-1)])=0, new: 1*(1-h[j+k-(n-1)])
  Δ_j_from_first[k] = (1-h[j+k-(n-1)]) if valid
- Second array (t+k-(n-1)=j, t=j-k+(n-1)): old: h[j-k+(n-1)]*(1-0)=h[j-k+(n-1)], new: h[j-k+(n-1)]*(1-1)=0

        g_prev = g

        step *= 0.999

def slsqp_optimize(h, n, beta, maxiter=300):
    """SLSQP optimization for continuous relaxation."""
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        dx = 2.0 / len(x)
        return (m + np.log(np.sum(ex)) / beta) * dx
    
    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        dx = 2.0 / len(x)
        w = w * dx
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    g[off:off + np_] += wk * (1.0 - x[:np_])
                    g[:np_] -= wk * x[off:off + np_]
            else:
                a = -off
                np_ = n - a
                if np_ > 0:
                    g[:np_] += wk * (1.0 - x[a:a + np_])
                    g[a:a + np_] -= wk * x[:np_]
        return g
    
    def _con_fun(x):
        return np.sum(x) - n / 2.0
    
    def _con_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:

        h_alt_bin, c5_alt = binary_directed_search(h_alt_bin, n_main, max_rounds=40)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt_bin.copy()
            best_n = n_main
    
    # Phase 5: Final intensive polish on best solution
    n_best = best_n
    h_best = best_h.copy()
    
    # Continuous + binary cycle
    for cycle in range(3):
        if remaining() < 30:
            break
        
        # Continuous refine
        h_cont, c5_cont = smooth_max_optimize(h_best, n_best, 30000, max_iters=200)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
            h_best = h_cont.copy()
        
        # Binary refine
        h_bin2 = round_to_binary(h_best, n_best)
        h_bin2, c5_bin2 = binary_directed_search(h_bin2, n_best, max_rounds=50)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
            h_best = h_bin2.copy()
        
        # SA refine
        if remaining() > 40:
            h_sa2, c5_sa2 = binary_sa_search(h_bin2, n_best, max_iters=20000, t_start=0.1, t_end=0.002)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                h_best = h_sa2.copy()
    
    # Phase 6: Final SLSQP on continuous
    if remaining() > 30:
        h_final = slsqp_optimize(best_h, best_n, 50000, maxiter=300)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        
        # And binary from that
        h_final_bin = round_to_binary(best_h, best_n)
        h_final_bin, c5_fb = binary_directed_search(h_final_bin, best_n, max_rounds=30)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_final_bin.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
