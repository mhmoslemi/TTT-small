```python
def swap_effect(h, a, b, n):
    # a: position to go 1→0, b: position to go 0→1
    k = np.arange(1, n)  # lags 1 to n-1
    
    # pos_corr changes
    dp = np.zeros(n-1)
    # [a <= n-1-k] => k <= n-1-a
    mask_a_first = k <= (n-1-a)
    dp[mask_a_first] += -(1 - h[a+k[mask_a_first]])
    # [b <= n-1-k] => k <= n-1-b
    mask_b_first = k <= (n-1-b)
    dp[mask_b_first] += (1 - h[b+k[mask_b_first]])
    # [a >= k] => k <= a
    mask_a_second = k <= a
    dp[mask_a_second] += h[a-k[mask_a_second]]
    # [b >= k] => k <= b
    mask_b_second = k <= b
    dp[mask_b_second] -= h[b-k[mask_b_second]]
    
    # neg_corr changes
    dn = np.zeros(n-1)
    # [a >= k] => k <= a
    mask_a_first_n = k <= a
    dn[mask_a_first_n] += -(1 - h[a-k[mask_a_first_n]])
    # [b >= k] => k <= b
    mask_b_first_n = k <= b
    dn[mask_b_first_n] += (1 - h[b-k[mask_b_first_n]])
    # [a <= n-1-k] => k <= n-1-a
    mask_a_second_n = k <= (n-1-a)
    dn[mask_a_second_n] += h[a+k[mask_a_second_n]]
    # [b <= n-1-k] => k <= n-1-b
    mask_b_second_n = k <= (n-1-b)
    dn[mask_b_second_n] -= h[b+k[mask_b_second_n]]
    
    return dp, dn
```

```python
    best_max = current_max
```

```python
        h[a] = 0.0
        h[b] = 1.0
        new_corr = np.correlate(h, 1.0-h, mode='full')
        new_max = np.max(new_corr)
        
        delta = new_max - current_max
        if delta < 0 or rng.random() < np.exp(-delta * n / T):
            # Accept
            current_max = new_max
            corr = new_corr
            # Update ones/zeros
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            # Undo
            h[a] = 1.0
            h[b] = 0.0
```

```python
        wi = w[i]
        if wi == 0:
            continue

        if off >= 0:
```
