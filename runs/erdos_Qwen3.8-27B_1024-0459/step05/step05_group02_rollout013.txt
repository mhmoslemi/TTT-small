    return best_h, best_c5


def greedy_binary(h_bin, n, max_iters=50000):
    """Greedy local search - only accept improvements."""
    dx = 2.0 / n
    h = h_bin.copy()
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr)) * dx
    cur_c5 = best_c5
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 1e10
        best_swap = None
        n_try_o = min(25, len(ones))
        n_try_z = min(25, len(zeros))
        
        # Use gradient-based selection
        g = np.zeros(n)
        idx_top = np.argsort(-corr)[:3]
        for ki in idx_top:
            # gradient contribution
            offset = ki - (n - 1)
            if offset >= 0:
                np_ = n - offset
                g[offset:offset + np_] += (1.0 - h[:np_])
                g[:np_] -= h[offset:offset + np_]
            else:
                ao = -offset
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        
        g -= np.mean(g)
        score_one = -g[ones]
        score_zero = g[zeros]
        top_ones = ones[np.argsort(-score_one)[:n_try_o]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]
        
        for i in top_ones:
            for j in top_zeros:
                h[i] = 0; h[j] = 1
                nc = np.max(compute_corr(h)) * dx
                h[i] = 1; h[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
        
        if best_swap is None or best_delta >= -1e-12:
            # Random swap to escape
            for _ in range(10):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h[i] = 0; h[j] = 1
                nc = np.max(compute_corr(h)) * dx
                h[i] = 1; h[j] = 0
                if nc < cur_c5 - 1e-12:
                    best_delta = nc - cur_c5
                    best_swap = (i, j)
                    break
            if best_swap is None:
                break
        
        i, j = best_swap
        h[i] = 0; h[j] = 1
        cur_c5 = float(np.max(compute_corr(h))) * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5


def lbfgs_smooth(h, n, beta, top_k=5, maxiter=500):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _BETA = beta
    _TOPK = top_k
    _TGT = n / 2.0
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 15})
        return proj(res.x, n)
    except Exception:
        return h


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 100.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * _corr_grad_single(x, ki, n)
    g *= dx
    g += 200.0 * (np.sum(x) - _TGT)
    return g


def _corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def top_equalize(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-9:
            break
        
        g_max = _corr_grad_single(h, idx[0], n)
        g_min = _corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            break
        d /= dn
        
        improved = False
        for alpha in [0.01, 0.03, 0.005, 0.05, 0.02, 0.008, 0.04, 0.015]:
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for alpha in [0.01, 0.02, 0.005, 0.03]:
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    target_c5 = 0.38085
    
    # Phase 1: Binary SA at parent resolution (main optimization)
    h_bin = binarize(best_h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Multi-round SA with decreasing temperature
    sa_time = _BUDGET * 0.45
    sa_deadline = _T0 + sa_time
    
    for sa_round in range(4):
        if time.time() > sa_deadline:
            break
        T_start = 0.4 - 0.05 * sa_round
        T_end = 0.001
        max_iters = 500000
        h_sa, c5_sa = sa_binary(h_bin, n, rng, max_iters, T_start, T_end,
                                two_swap_prob=0.12)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_bin = h_sa.copy()
    
    # Phase 2: Greedy binary polish
    if remaining() > 20:
        h_g, c5_g = greedy_binary(best_h if len(best_h) == n else h_bin, n, max_iters=30000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            h_bin = h_g.copy()
    
    # Phase 3: Continuous refinement of best binary
    if remaining() > 30:
        h_cont = h_bin.copy()
        for beta in [5000, 20000, 80000]:
            if remaining() < 8:
                break
            h_cont = lbfgs_smooth(h_cont, n, beta, top_k=5, maxiter=400)
            c5 = compute_c5(h_cont)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cont.copy()
    
    # Top equalize
    if remaining() > 15:
        h_eq, c5_eq = top_equalize(best_h, n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Phase 4: Try alternative binary initializations + SA
    if remaining() > 60:
        # structured initialization: blocks
        n_blocks = 8
        block_size = n // n_blocks
        h_struct = np.zeros(n)
        for b in range(n_blocks):
            start = b * block_size
            half = block_size // 2
            h_struct[start:start + half] = 1.0
        
        h_struct_bin = binarize(h_struct, n)
        h_sa2, c5_sa2 = sa_binary(h_struct_bin, n, rng, 200000, 0.35, 0.002, two_swap_prob=0.1)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
    
    # Phase 5: Resolution change (try higher n)
    n_hi = n + 128  # 512 if n=384
    if n_hi % 2 == 0 and n_hi <= 768 and remaining() > 80:
        h_interp = interp_to(best_h, n, n_hi)
        h_hi = binarize(h_interp, n_hi)
        h_sa3, c5_sa3 = sa_binary(h_hi, n_hi, rng, 300000, 0.35, 0.002, two_swap_prob=0.1)
        
        if c5_sa3 < best_c5:
            # Refine continuously
            h_ref = h_sa3.copy()
            for beta in [10000, 50000]:
                if remaining() < 8:
                    break
                h_ref = lbfgs_smooth(h_ref, n_hi, beta, top_k=5, maxiter=300)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n_hi
            elif c5_sa3 < best_c5:
                best_c5 = c5_sa3
                best_h = h_sa3.copy()
                best_n = n_hi
    
    # Also try n+64
    n_mid = n + 64
    if n_mid % 2 == 0 and n_mid <= 768 and n_mid != n_hi and remaining() > 60:
        h_interp = interp_to(best_h, best_n, n_mid)
        h_mid = binarize(h_interp, n_mid)
        h_sa4, c5_sa4 = sa_binary(h_mid, n_mid, rng, 200000, 0.3, 0.002, two_swap_prob=0.1)
        if c5_sa4 < best_c5:
            h_ref = h_sa4.copy()
            for beta in [10000, 40000]:
                if remaining() < 8:
                    break
                h_ref = lbfgs_smooth(h_ref, n_mid, beta, top_k=5, maxiter=300)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n_mid
            elif c5_sa4 < best_c5:
                best_c5 = c5_sa4
                best_h = h_sa4.copy()
                best_n = n_mid
    
    # Phase 6: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    # Very high beta smooth max
    for beta in [100000, 200000]:
        if remaining() < 8:
            break
        h_new = lbfgs_smooth(h, n, beta, top_k=4, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Final greedy binary
    if remaining() > 15:
        h_fin_bin = binarize(best_h, n)
        h_gf, c5_gf = greedy_binary(h_fin_bin, n, max_iters=20000)
        if c5_gf < best_c5:
            best_c5 = c5_gf
            best_h = h_gf.copy()
    
    # Final top equalize
    if remaining() > 8:
        h_eq2, c5_eq2 = top_equalize(best_h, n, top_k=6, max_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
    
    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        # Use projected value
        best_c5 = final_c5
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
