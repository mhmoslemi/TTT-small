                best_swap_delta = delta
                best_swap = (i, j)

        # Strategy 3: Swap that specifically reduces the worst correlation
        k_worst = top_idx[0]
        off_w = k_worst - (n - 1)
        if off_w >= 0:
            # corr = sum h[i]*(1-h[i+off]), faults are h[i]=1, h[i+off]=0
            faults = np.where((h[:n - off_w] == 1) & (h[off_w:] == 0))[0]
            if len(faults) > 0:
                fi = faults[rng.randint(len(faults))]
                i = fi
                j = zeros[rng.randint(len(zeros))]
                if j != i and 0 <= j < n and j < n - off_w or j >= off_w:
                    h[i] = 0
                    h[j] = 1
                    nc = float(np.max(compute_corr(h)))
                    delta = nc - max_val
                    h[i] = 1
                    h[j] = 0
                    if delta < best_swap_delta:
                        best_swap_delta = delta
                        best_swap = (i, j)
        else:
            ao = -off_w
            faults = np.where((h[ao:] == 1) & (h[:n - ao] == 0))[0]
            if len(faults) > 0:
                fi = faults[rng.randint(len(faults))]
                i = fi + ao
                j = zeros[rng.randint(len(zeros))]
                if j != i:
                    h[i] = 0
                    h[j] = 1
                    nc = float(np.max(compute_corr(h)))
                    delta = nc - max_val
                    h[i] = 1
                    h[j] = 0
                    if delta < best_swap_delta:
                        best_swap_delta = delta
                        best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val

            # Accept if improving or with probability exp(-delta/temp)
            if delta <= 0 or rng.random() < np.exp(-delta / max(temp, 0.01)):
                cur_max = nc
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                    no_improve = 0
                temp *= cool_rate
            else:
                h[i] = 1
                h[j] = 0
                temp *= cool_rate

    return best_h, best_max * dx_v


def binary_greedy_search(h_bin, n, rng, time_limit=60):
    """Greedy binary search with multi-swap lookahead."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            rem = ones_idx[perm[:n_ones - n_half]]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            add = zeros_idx[perm[:n_half - n_ones]]
            h[add] = 1
    dx_v = 2.0 / n
    cur_max = float(np.max(compute_corr(h)))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:5]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0.0
        best_swap = None
        # Evaluate many candidates
        n_try = 30
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - cur_max
            h[i] = 1
            h[j] = 0
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)

        # Also targeted
        for _ in range(15):
            k_hot = top_idx[rng.randint(len(top_idx))]
            off = k_hot - (n - 1)
            if off >= 0:
                faults = np.where((h[:n - off] == 1) & (h[off:] == 0))[0]
            else:
                ao = -off
                faults = np.where((h[ao:] == 1) & (h[:n - ao] == 0))[0]
            if len(faults) == 0:
                continue
            fi = faults[rng.randint(len(faults))]
            if off >= 0:
                i = fi
            else:
                i = fi + (-off)
            if i >= n or h[i] != 1:
                continue
            j = zeros[rng.randint(len(zeros))]
            if j == i:
                continue
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - cur_max
            h[i] = 1
            h[j] = 0
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)

        if best_swap is not None and best_delta < -1e-14:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            cur_max = float(np.max(compute_corr(h)))
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > 500:
                # Escape: random 2-swap
                if len(ones) >= 2 and len(zeros) >= 2:
                    i1, i2 = ones[rng.choice(len(ones), 2, replace=False)]
                    j1, j2 = zeros[rng.choice(len(zeros), 2, replace=False)]
                    h[i1] = 0
                    h[j1] = 1
                    h[i2] = 0
                    h[j2] = 1
                    cur_max = float(np.max(compute_corr(h)))
                stall = 0

    return best_h, best_max * dx_v


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgradient_max(h, n, n_iter=200):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 0.3
        top_mask = corr >= threshold
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(corr)])

        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            g += corr_gradient_at(h, off, n)
        g /= len(top_idx)
        g -= np.mean(g)

        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            step = min(step * 1.05, 0.3)
        else:
            step *= 0.85
            if step < 0.001:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT, _P_VEC
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Game-theoretic minimax (new approach)
    if remaining() > 60:
        h_gt, c5_gt = game_theoretic(h, n, max_rounds=30, slsqp_maxiter=150)
        if c5_gt < best_c5:
            best_c5 = c5_gt
            best_h = h_gt.copy()
            h = h_gt.copy()

    # Phase 2: SLSQP smooth max refinement
    for beta in [1000, 5000, 20000, 50000, 100000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Top-shifts equalization (multiple passes)
    for tk in [3, 5, 7, 10]:
        if remaining() < 10:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Binary SA search (intensive)
    if remaining() > 40:
        h_bin = binarize(h, n)
        sa_time = min(100, max(20, int(remaining() * 0.25)))
        h_b, c5_b = binary_sa_search(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            # Continuous refinement from binary
            h_c = slsqp_smooth(h_b, n, 20000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                h = h_c.copy()
            else:
                h = h_b.copy()
        else:
            h = best_h.copy()

    # Phase 5: Binary greedy search (different heuristic)
    if remaining() > 40:
        h_bin2 = binarize(h, n)
        g_time = min(80, max(15, int(remaining() * 0.2)))
        h_g, c5_g = binary_greedy_search(h_bin2, n, rng, time_limit=g_time)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            h_c = slsqp_smooth(h_g, n, 20000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                h = h_c.copy()
            else:
                h = h_g.copy()

    # Phase 6: Subgradient polish
    if remaining() > 20:
        h_sg, c5_sg = subgradient_max(h, n, n_iter=200)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 7: Diverse starts
    if remaining() > 60:
        h_tm = thue_morse(n, rng)
        h_tm = slsqp_smooth(h_tm, n, 5000, 300)
        h_tm, c5_tm = top_shifts_focus(h_tm, n, 4, 200)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            h = h_tm.copy()

        # Inverse
        h_tm2 = project_feasible(1.0 - h_tm, n)
        h_tm2 = slsqp_smooth(h_tm2, n, 5000, 300)
        c5_tm2 = compute_c5(h_tm2)
        if c5_tm2 < best_c5:
            best_c5 = c5_tm2
            best_h = h_tm2.copy()
            h = h_tm2.copy()

        # Random
        for rs in range(3):
            if remaining() < 15:
                break
            h_r = rng.uniform(0, 1, n)
            h_r = project_feasible(h_r, n)
            h_r = slsqp_smooth(h_r, n, 5000, 300)
            h_r, c5_r = top_shifts_focus(h_r, n, 4, 150)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                h = h_r.copy()

    # Phase 8: Resolution sweep
    if remaining() > 60:
        n_base = best_n
        n_candidates = []
        for delta in [-24, -16, -12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 24, 32, 48, 64]:
            nt = n_base + delta
            if 160 <= nt <= 700 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [256, 320, 384, 448, 512, 576, 640]:
            if 160 <= nt <= 700 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:6]:
            if remaining() < 25:
                break
            if n_t <= 0:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

            # Game-theoretic at new resolution
            h_t, c5_t = game_theoretic(h_t, n_t, max_rounds=15, slsqp_maxiter=100)
            h_t = slsqp_smooth(h_t, n_t, 20000, 200)
            c5_t = compute_c5(h_t)
            h_t, c5_t = top_shifts_focus(h_t, n_t, 4, 150)

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

            # Quick binary at new resolution
            if remaining() > 20 and c5_t < best_c5 + 0.002:
                h_t_bin = binarize(h_t, n_t)
                h_tb, c5_tb = binary_sa_search(h_t_bin, n_t, rng, time_limit=15)
                if c5_tb < best_c5:
                    best_c5 = c5_tb
                    best_h = h_tb.copy()
                    best_n = n_t

    # Phase 9: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # Game-theoretic final
    if remaining() > 30:
        h_gt, c5_gt = game_theoretic(h, n, max_rounds=20, slsqp_maxiter=200)
        if c5_gt < best_c5:
            best_c5 = c5_gt
            best_h = h_gt.copy()
            h = h_gt.copy()

    for beta in [20000, 50000, 100000, 200000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    for tk in [3, 5, 8]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary SA
    if remaining() > 25:
        h_bin = binarize(h, n)
        h_b, c5_b = binary_sa_search(h_bin, n, rng, time_limit=min(40, int(remaining() * 0.5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = slsqp_smooth(h_b, n, 50000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    if remaining() > 10:
        h_sg, c5_sg = subgradient_max(best_h, best_n, n_iter=150)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final projection
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
