**Key idea: Use a "pairwise" coordinate descent that maintains feasibility**

In standard coordinate descent with a sum constraint, when you change h[i] by δ, you need to change another h[j] by -δ. The optimal pair update would be:

For a pair (i, j), change h[i] → h[i]+δ and h[j] → h[j]-δ. The new correlation is:

This is linear in δ for each k, so the new max is max_k (corr_old[k] + δ * d_k) where d_k = ∂corr[k]/∂h[i] - ∂corr[k]/∂h[j].

Minimizing max_k (a_k + d_k * δ) over δ is a 1D problem that can be solved exactly (it's piecewise linear, minimum at a breakpoint).

This is much more efficient than the single-coordinate approach and maintains feasibility!

    for k in range(ncorr):
        w = weights[k]
        if w < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

    return smooth_obj_grad(x, _N_OPT, _BETA_OPT)[0]

def _grad_fn(x):
    return smooth_obj_grad(x, _N_OPT, _BETA_OPT)[1]

def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _con_jac(x):

def slsqp_optimize(h, n, beta, maxiter=400):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h0 = h.copy()
    c5_before = compute_c5(h0)
    try:
        res = minimize(_obj_fn, h0, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before:
                return trial
    except Exception:
        pass
    return h0.copy()

def pairwise_cd(h, n, max_sweeps=200):
    """Pairwise coordinate descent maintaining sum constraint."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n-1), n)
    rng = np.random.RandomState(12345)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 15:
            break
        n_pairs = 3000
        improved_count = 0
        for _ in range(n_pairs):
            if remaining_time() < 3:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi = best[i]
            hj = best[j]
            # Bounds on delta: hi+d in [0,1], hj-d in [0,1]
            lo = max(-hi, hj - 1.0)
            hi_bound = min(1.0 - hi, 1.0 - hj)
            if lo >= hi_bound - 1e-15:
                continue
            
            # Compute d[k] = ∂corr[k]/∂h[i] - ∂corr[k]/∂h[j]
            d = np.zeros(ncorr)
            m1 = i + offsets
            v1 = (m1 >= 0) & (m1 < n)
            d[v1] += (1.0 - best[m1[v1]])
            m2 = i - offsets
            v2 = (m2 >= 0) & (m2 < n)
            d[v2] -= best[m2[v2]]
            m3 = j + offsets
            v3 = (m3 >= 0) & (m3 < n)
            d[v3] -= (1.0 - best[m3[v3]])
            m4 = j - offsets
            v4 = (m4 >= 0) & (m4 < n)
            d[v4] += best[m4[v4]]
            
            # Minimize max_k (best_corr[k] + delta * d[k]) over delta in [lo, hi_bound]
            # Convex piecewise-linear in delta
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi_bound - gr * (hi_bound - lo)
            c2 = lo + gr * (hi_bound - lo)
            f1 = np.max(best_corr + c1 * d)
            f2 = np.max(best_corr + c2 * d)
            for _ in range(30):
                if hi_bound - lo < 1e-13:
                    break
                if f1 < f2:
                    hi_bound = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi_bound - gr * (hi_bound - lo)
                    f1 = np.max(best_corr + c1 * d)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi_bound - lo)
                    f2 = np.max(best_corr + c2 * d)
            
            delta_opt = (lo + hi_bound) / 2.0
            if abs(delta_opt) < 1e-14:
                continue
            f_opt = np.max(best_corr + delta_opt * d)
            
            if f_opt * dx < best_c5 - 1e-15:
                best[i] += delta_opt
                best[j] -= delta_opt
                best = np.clip(best, 0.0, 1.0)
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved_count += 1
        
        if improved_count == 0:
            break
    
    return best, best_c5

def interp_to(src_h, src_n, dst_n):

    return np.interp(x_new, xp, src_h)
