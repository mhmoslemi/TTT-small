_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_all_f(h):
    """Compute all f(k) values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            # All at boundaries, adjust the ones closest to the needed direction
            if diff > 0:
                # Need to increase sum, prioritize increasing values closest to 1
                idx = np.argsort(h)[-max(1, int(abs(diff)) + 1):]
                for i in idx:
                    room = 1.0 - h[i]
                    adj = min(room, diff)
                    h[i] += adj
                    diff -= adj
                    if abs(diff) < 1e-15:

                    adj = min(room, -diff)
                    h[i] -= adj
                    diff += adj
                    if abs(diff) < 1e-15:
                        break
        h = np.clip(h, 0.0, 1.0)
    return h
