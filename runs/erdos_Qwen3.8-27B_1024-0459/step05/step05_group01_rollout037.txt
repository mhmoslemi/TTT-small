        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    if best_n % 2 != 0:
        best_h = np.zeros(best_n + 1)
        best_h[:(best_n + 1) // 2] = 1.0
        best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish on parent
    h = best_h.copy()
    for beta in [500, 2000, 10000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization
    if remaining_time() > 20:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=5, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Intensive binary local search at parent resolution
    if remaining_time() > 60:
        bs_time = min(300, max(60, int(remaining_time() * 0.35)))
        h_bin = to_binary(best_h, best_n)
        h_b, c5_b = binary_search_full(h_bin, best_n, bs_time, rng)
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()

    # Phase 4: Refine binary result continuously
    if remaining_time() > 30:
        h_c = best_h.copy()
        for beta in [2000, 20000, 100000]:
            if remaining_time() < 15:
                break
            h_c = slsqp_opt(h_c, best_n, beta, maxiter=400)
        c5_c = compute_c5(h_c)
        h_c, c5_c = top_k_equalize(h_c, best_n, top_k=5, max_iters=300)
        if c5_c < best_c5 - 1e-14:
            best_c5 = c5_c
            best_h = h_c.copy()
            h = h_c.copy()

    # Phase 5: Multi-resolution expansion to 768
    n_hi = best_n * 2
    if n_hi % 2 != 0:
        n_hi += 1
    if remaining_time() > 120 and 384 < n_hi <= 1024:
        h_interp = interp_to(best_h, best_n, n_hi)
        h_feas = project_feasible(h_interp, n_hi)
        c5_int = compute_c5(h_feas)
        h_hi = h_feas
        for beta in [500, 3000, 20000]:
            if remaining_time() < 20:
                break
            h_hi = slsqp_opt(h_hi, n_hi, beta, maxiter=400)
        c5_hi = compute_c5(h_hi)
        h_hi, c5_hi = top_k_equalize(h_hi, n_hi, top_k=5, max_iters=300)
        # Binary search at high res
        if remaining_time() > 60:
            bs_time_hi = min(150, max(30, int(remaining_time() * 0.25)))
            h_hi_bin = to_binary(h_hi, n_hi)
            h_hi_b, c5_hi_b = binary_search_full(h_hi_bin, n_hi, bs_time_hi, rng)
            if c5_hi_b < c5_hi:
                h_hi = h_hi_b
                c5_hi = c5_hi_b
        # Refine continuously
        if remaining_time() > 20:
            for beta in [5000, 50000]:
                if remaining_time() < 15:
                    break
                h_hi = slsqp_opt(h_hi, n_hi, beta, maxiter=300)
            h_hi, c5_hi = top_k_equalize(h_hi, n_hi, top_k=5, max_iters=200)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5 - 1e-14:
            best_c5 = c5_hi
            best_h = interp_to(h_hi, n_hi, best_n)
            best_h = project_feasible(best_h, best_n)
            h = best_h.copy()

    # Phase 6: Try n=512
    n_mid = 512
    if remaining_time() > 100 and abs(n_mid - best_n) > 10:
        h_mid = interp_to(best_h, best_n, n_mid)
        h_mid = project_feasible(h_mid, n_mid)
        for beta in [1000, 10000, 50000]:
            if remaining_time() < 15:
                break
            h_mid = slsqp_opt(h_mid, n_mid, beta, maxiter=400)
        h_mid, c5_mid = top_k_equalize(h_mid, n_mid, top_k=5, max_iters=300)
        if remaining_time() > 40:
            h_mid_b, c5_mid_b = binary_search_full(to_binary(h_mid, n_mid), n_mid,
                                                    min(100, int(remaining_time() * 0.2)), rng)
            if c5_mid_b < c5_mid:
                h_mid = h_mid_b
                c5_mid = c5_mid_b
        if c5_mid < best_c5 - 1e-14:
            best_c5 = c5_mid
            best_h = interp_to(h_mid, n_mid, best_n)
            best_h = project_feasible(best_h, best_n)
            h = best_h.copy()

    # Phase 7: SA on binary at parent resolution
    if remaining_time() > 60:
        sa_time = min(120, max(30, int(remaining_time() * 0.2)))
        h_sa, c5_sa = sa_binary(best_h, best_n, sa_time, rng)
        if c5_sa < best_c5 - 1e-14:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h = h_sa.copy()

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [10000, 50000, 200000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining_time() > 20:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=15)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final binary polish
    if remaining_time() > 30:
        h_fin_bin = to_binary(best_h, n)
        h_fin_b, c5_fin_b = binary_search_full(h_fin_bin, n,
                                                min(80, int(remaining_time() * 0.5)), rng)
        if c5_fin_b < best_c5 - 1e-14:
            best_c5 = c5_fin_b
            best_h = h_fin_b.copy()
        # Quick continuous refinement of best
        h_fin_c = slsqp_opt(best_h, n, 50000, maxiter=300)
        c5_fin_c = compute_c5(h_fin_c)
        h_fin_c, c5_fin_c = top_k_equalize(h_fin_c, n, top_k=5, max_iters=200)
        if c5_fin_c < best_c5 - 1e-14:
            best_c5 = c5_fin_c
            best_h = h_fin_c.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
