And if we flip h[j0] from 0 to 1:

So a pair swap (flip i from 1→0, flip j from 0→1) at shift k* ≥ 0:

    - If j ≥ k*: term at j-k*: was 0, now (1-h[j-k*]). Change: +(1-h[j-k*])

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    n_half = n // 2
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > n_half:
        ones_idx = np.where(h_bin == 1.0)[0]
        ones_sorted = ones_idx[np.argsort(h[ones_idx])]
        h_bin[ones_sorted[:n_ones - n_half]] = 0.0
    elif n_ones < n_half:
        zeros_idx = np.where(h_bin == 0.0)[0]
        zeros_sorted = zeros_idx[np.argsort(-h[zeros_idx])]
        h_bin[zeros_sorted[:n_half - n_ones]] = 1.0
    return h_bin


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        h = project_feasible(h, n)
    return h


def sa_binary(h_init, n, time_limit, rng, t_init=1.5):
    h = h_init.copy()
    n_half = n // 2
    if int(np.sum(h)) != n_half:
        h = to_binary(h, n)
    start = time.time()
    T = t_init
    curr_max = int(np.max(compute_corr(h)))
    best_c5 = curr_max * (2.0 / n)
    best_h = h.copy()
    no_improve = 0
    step = 0
    ones_list = np.where(h == 1.0)[0].copy()
    zeros_list = np.where(h == 0.0)[0].copy()

    while time.time() - start < time_limit and remaining_time() > 5:
        if step % 5000 == 0 and step > 0:
            T = max(0.05, t_init * (0.4 ** (step / 200000.0)))
            if no_improve > 80000 and T < 0.8:
                T = t_init * 0.6
                no_improve = 0

        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]

        h[i] = 0.0
        h[j] = 1.0
        new_max = int(np.max(compute_corr(h)))
        delta = new_max - curr_max

        if delta <= 0 or (T > 0.05 and rng.rand() < np.exp(-float(delta) / T)):
            # Accept
            curr_max = new_max
            idx_i = np.where(ones_list == i)[0]
            if len(idx_i) > 0:
                ones_list[idx_i[0]] = j
            idx_j = np.where(zeros_list == j)[0]

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

        step += 1
        T = max(0.05, T * 0.9998)

    return best_h, best_c5

def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _sm_obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def top_k_equalize(h, n, top_k=5, max_iters=300):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08, 0.15]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)
