_CUR_N = 0
_SM_BETA = 1000.0

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

# Smooth max objective and gradient
def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    return (m + np.log(S) / beta) * dx

def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g

    return np.sum(x) - _CUR_N / 2.0

    return np.ones(_CUR_N)

    _CUR_N = n
    _SM_BETA = beta

def multiplicative_weight_opt(h, n, max_iter=1000, eta=None):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Initialize weights
    n_lags = 2*n - 1
    w = np.ones(n_lags) / n_lags
    
    if eta is None:
        eta = 5.0  # learning rate
    
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        
        corr = compute_corr(h)
        
        # Update weights (multiplicative)
        w *= np.exp(eta * corr / np.max(corr))
        w /= np.sum(w)
        
        # Compute gradient of weighted sum
        g = np.zeros(n)
        for k in range(n_lags):
            wk = w[k] * dx
            if wk < 1e-10:
                continue
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:])
                g[a:] -= wk * h[:np_]
        
        # Gradient descent step with line search
        # Project gradient to satisfy sum constraint
        g_proj = g - np.mean(g)  # This ensures sum(h + step*g_proj) = sum(h)
        
        gn = np.linalg.norm(g_proj)
        if gn < 1e-14:
            break
        d = -g_proj / gn

        improved = False

    h = best_h.copy()
    for b in [500000, 2000000, 10000000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_opt(h, n, b, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=6, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 10:
        h_sd, c5_sd = steepest_descent_polish(best_h, n, max_iter=300)
        if c5_sd < best_c5 - 1e-14:
            best_c5 = c5_sd
            best_h = h_sd.copy()

    if remaining_time() > 10:
        h_cr, c5_cr = coordinate_scan(best_h, n, max_sweeps=10)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 5000000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
