def binary_local_search(h_bin, n, rng, time_limit=60):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    # Ensure exactly n/2 ones
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - n_half]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:n_half - n_ones]]] = 1
    
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_improve = 0
    
    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0
        best_swap = None
        swap_type = None
        
        # Try 1-swaps (11 candidates)
        for _ in range(11):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - cur_max
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)
                swap_type = 'one'
        
        # Try 2-swaps (7 candidates)
        if len(ones) >= 2 and len(zeros) >= 2:
            for _ in range(7):
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[(rng.randint(len(ones)) + 1) % len(ones)]
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[(rng.randint(len(zeros)) + 1) % len(zeros)]
                if i1 == i2 or j1 == j2:
                    continue
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                nc = float(np.max(compute_corr(h)))
                delta = nc - cur_max
                h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_swap = (i1, i2, j1, j2)
                    swap_type = 'two'
        
        if best_swap is not None:
            if swap_type == 'one':
                i, j = best_swap
                h[i] = 0; h[j] = 1
            else:
                i1, i2, j1, j2 = best_swap
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            cur_max += best_delta
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max

                cur_max = float(np.max(compute_corr(h)))
