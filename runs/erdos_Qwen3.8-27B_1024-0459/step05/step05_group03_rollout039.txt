def best_swap_search(h, n, max_rounds=100):
    """Find best single swap that reduces C5. Repeat until no improvement."""
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_imp = 1e-15
        best_pair = None
        
        # Evaluate all swaps
        corr_cur = np.correlate(h, 1.0 - h, mode='full')
        cc_max = float(np.max(corr_cur))
        
        for ia in range(len(ones)):
            p = ones[ia]
            for ib in range(len(zeros)):
                q = zeros[ib]
                # Compute delta for swap (p,q)
                # After swap: h[p]=0, h[q]=1
                # delta_cc[m] for each lag m
                # Use the formula: delta_cc = -delta_ac (since cc = n/2 - ac)
                # delta_ac[k] = -h[p-k]*valid - h[p+k]*valid + h[q-k]*valid*(q-k!=p) + h[q+k]*valid*(q+k!=p)
                
                # Vectorized computation:
                # Term 1: -h[p-k] for valid k (0 <= p-k < n)
                # In terms of correlation index m (lag = m - (n-1)):
                # k = m - (n-1), so p-k = p - m + n - 1
                # Valid when 0 <= p-m+n-1 < n, i.e., m in [p, p+n-1] (clamped)
                lo1 = max(0, p)
                hi1 = min(2*n-1, p + n - 1 + 1)  # exclusive
                if lo1 < hi1:
                    idx1 = p - np.arange(lo1, hi1) + n - 1
                    t1 = -h[idx1]
                    delta = np.zeros(2*n-1)
                    delta[lo1:hi1] += t1
                
                # Term 2: -h[p+k] for valid k (0 <= p+k < n)
                # p+k = p + m - (n-1), valid when 0 <= p+m-n+1 < n, i.e., m in [n-1-p, 2n-2-p]
                lo2 = max(0, n - 1 - p)
                hi2 = min(2*n-1, 2*n - 2 - p + 1)
                if lo2 < hi2:
                    idx2 = p + np.arange(lo2, hi2) - (n - 1)
                    t2 = -h[idx2]
                    delta[lo2:hi2] += t2
                
                # Term 3: h[q-k] for valid k, q-k != p
                # q-k = q - m + n - 1, valid when 0 <= q-m+n-1 < n, i.e., m in [q, q+n-1]
                lo3 = max(0, q)
                hi3 = min(2*n-1, q + n - 1 + 1)
                if lo3 < hi3:
                    idx3 = q - np.arange(lo3, hi3) + n - 1
                    t3 = h[idx3].copy()
                    # Zero out where idx3 == p (i.e., q - m + n - 1 = p, m = q - p + n - 1)
                    mp = q - p + n - 1
                    if 0 <= mp < 2*n-1:
                        t3[mp - lo3] = 0.0
                    delta[lo3:hi3] += t3
                
                # Term 4: h[q+k] for valid k, q+k != p
                # q+k = q + m - (n-1), valid when 0 <= q+m-n+1 < n, i.e., m in [n-1-q, 2n-2-q]
                lo4 = max(0, n - 1 - q)
                hi4 = min(2*n-1, 2*n - 2 - q + 1)
                if lo4 < hi4:
                    idx4 = q + np.arange(lo4, hi4) - (n - 1)
                    t4 = h[idx4].copy()
                    # Zero out where idx4 == p (q + m - n + 1 = p, m = p - q + n - 1)
                    mp = p - q + n - 1
                    if 0 <= mp < 2*n-1:
                        t4[mp - lo4] = 0.0
                    delta[lo4:hi4] += t4
                
                # delta_cc = -delta_ac (since cc = n/2 - ac, so cc' = n/2 - ac - delta_ac = cc - delta_ac)
                # new_cc = corr_cur - delta
                new_cc = corr_cur - delta
                new_max = float(np.max(new_cc))
                imp = cc_max - new_max
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (p, q)
        
        if best_pair is None:
            break
        p, q = best_pair
        h[p] = 0.0
        h[q] = 1.0
        c5_new = compute_c5(h)
        if c5_new < c5_best:
            c5_best = c5_new
            best_h = h.copy()
        c5_cur = c5_new
    
    return best_h, c5_best
