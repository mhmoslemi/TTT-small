def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            adj = diff / cnt
            # Don't push past bounds
            max_add = 1.0 - h[mask]
            min_add = h[mask]
            if adj > 0:
                adj = min(adj, np.max(max_add))
            else:
                adj = max(adj, -np.max(min_add))
            h[mask] += adj
        else:
            if diff > 0:
                h[np.argmin(h)] = min(1.0, h[np.argmin(h)] + diff)
            else:
                h[np.argmax(h)] = max(0.0, h[np.argmax(h)] + diff)
        h = np.clip(h, 0.0, 1.0)
    return h

# SLSQP helper (simpler version)
def _smooth_obj(x):
    n = _N_OPT
    dx = 2.0 / n
    beta = _BETA_OPT
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    return (m + np.log(wsum) / beta) * dx

def _smooth_grad(x):
    n = _N_OPT
    dx = 2.0 / n
    beta = _BETA_OPT
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * x[:npairs]
    grad *= dx
    return grad

def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _con_jac(x):
    return np.ones(_N_OPT)

def slsqp_optimize(h, n, beta, maxiter=300):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            return trial
    except Exception:
        pass
    return h.copy()
