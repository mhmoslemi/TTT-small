def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            adj = diff / cnt
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        # Spread remaining across all interior points
        mask = (h > 0) & (h < 1)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
            # If still off, fix the largest element
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-12:
                for j in range(n):
                    if h[j] + diff2 >= 0 and h[j] + diff2 <= 1:
                        h[j] += diff2
                        break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _g_smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]

def _g_smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]

def _g_constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0

def _g_constraint_sum_jac(x):
    return np.ones(_N_WORK)

def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    try:

                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):

            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def linearized_minimax_step(h, n, step_size=1.0):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    
    # Compute gradients of each corr_k at h
    # grad_k[i] = d(corr_k)/d(h[i])
    # For corr_k with offset d:
    #   d(corr_k)/d(h[i]) = (1-h[i-d]) if i-d valid (from the h[i] term)
    #                      + (-h[i+d]) if i+d valid (from the (1-h[i+d]) term in another position)
    # Actually, let me compute this more carefully.
    # corr_k = sum_j h[j] * (1-h[j+offset_k])
    # d(corr_k)/d(h[i]) = (1-h[i+offset_k]) [from h[i] factor]
    #                    + sum_{j: j+offset_k = i} h[j] * (-1) [from (1-h[j+offset_k]) factor]
    #                    = (1-h[i+offset_k]) - h[i-offset_k] (if both valid)
    
    # For the minimax, we want to minimize the maximum of (corr_k + grad_k^T (h-h0))
    # This is a LP: min t s.t. corr_k + grad_k^T (h-h0) <= t, sum(h)=n/2, 0<=h<=1
    
    # But solving 1279 LP constraints might be slow. Let's use only the top-k.
    top_k = 10
    k_indices = np.argsort(corr)[-top_k:]
    
    # Build the gradient matrix for top-k
    # G[i, l] = grad_{k_l}[i]
    G = np.zeros((n, top_k))
    for l, k_idx in enumerate(k_indices):
        offset = int(k_idx) - (n - 1)
        for i in range(n):
            g = 0.0
            # Term from h[i] being the first factor
            if 0 <= i + offset < n:
                g += (1.0 - h[i + offset])
            # Term from h[i] being in (1-h[j+offset]) where j+offset = i, so j = i-offset
            if 0 <= i - offset < n:
                g += (-h[i - offset])
            G[i, l] = g
    
    # Solve LP: min t s.t. corr_l + G[:,l]^T (h-h0) <= t for l=0..top_k-1
    #                          sum(h) = n/2, 0 <= h <= 1
    # Variables: h (n vars) + t (1 var) = n+1 vars
    # Constraints: top_k inequality + 1 equality + 2n bounds
    
    # Use scipy.optimize.linprog
    from scipy.optimize import linprog
    
    # Minimize: c^T x where x = [h, t], c = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each l: G[:,l]^T h - t <= -corr_l
    # => [G[:,l], -1] @ [h, t] <= -corr[l]
    A_ub = np.zeros((top_k, n + 1))
    for l in range(top_k):
        A_ub[l, :n] = G[:, l]
        A_ub[l, n] = -1.0
    b_ub = -corr[k_indices]
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        # Take a step
        h_result = (1 - step_size) * h + step_size * h_new
        h_result = project_feasible(h_result, n)
        return h_result
    
    return h.copy()

import numpy as np
from scipy.optimize import minimize, linprog
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 1000.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13 and n > 0:
        mask = (h > 0) & (h < 1)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - np.sum(h)
            if abs(diff2) > 1e-13:
                for j in range(n):
                    if 0 <= h[j] + diff2 <= 1:
                        h[j] += diff2
                        break
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def _g_smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]


def _g_smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]


def _g_constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0


def _g_constraint_sum_jac(x):
    return np.ones(_N_WORK)


def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    try:
        res = minimize(_g_smooth_obj, h, jac=_g_smooth_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-11:
                return trial
    except Exception:
        pass
    return h.copy()


def fast_coord_descent(h, n, max_sweeps=100):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        max_impr = 0.0

        for i in range(n):
            old = best[i]

            A = np.zeros(ncorr)
            if i + 1 < n:
                A[n : n + (n - 1 - i)] = 1.0 - best[i + 1:n]
            if i > 0:
                A[n - 1 - i : n - 1] = -best[0:i]

            base = best_corr - A * old
            base_n1 = best_corr[n - 1] - old * (1.0 - old)

            def eval_at(t):

                v0 = base_n1 + t * (1.0 - t)
                if v0 > vmax:
                    return v0
                return vmax

            f_best = f_old
            for tc in [0.0, 1.0, 0.5, 0.25, 0.75]:

                if fc < f_best - 1e-15:

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def linearized_minimax_step(h, n, step_size, top_k):
    dx = 2.0 / n
    corr = compute_corr(h)

    if top_k >= 2 * n - 1:
        k_indices = np.arange(2 * n - 1)
    else:
        k_indices = np.argsort(corr)[-top_k:]

    G = np.zeros((n, len(k_indices)))
    for l, k_idx in enumerate(k_indices):
        offset = int(k_idx) - (n - 1)
        if offset == 0:
            G[:, l] = 1.0 - 2.0 * h
        elif offset > 0:
            G[offset:, l] = 1.0 - h[:n - offset]
            G[:n - offset, l] = -h[offset:]
        else:
            ao = -offset
            G[:n - ao, l] = 1.0 - h[ao:]
            G[ao:, l] = -h[:n - ao]

    m_k = len(k_indices)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((m_k, n + 1))
    for l in range(m_k):
        A_ub[l, :n] = G[:, l]
        A_ub[l, n] = -1.0
    b_ub = -corr[k_indices].copy()

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_new = res.x[:n]
            h_result = (1.0 - step_size) * h + step_size * h_new
            h_result = project_feasible(h_result, n)
            return h_result
    except Exception:
        pass
    return h.copy()


def subgradient_directed_perturb(h, n, rng, scale):
    corr = compute_corr(h)
    m = np.max(corr)
    threshold = m - 0.3
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    if len(active) > 15:
        active = active[np.argsort(corr[active])[::-1][:15]]

    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset == 0:
            grad += 1.0 - 2.0 * h
        elif offset > 0:
            grad[offset:] += (1.0 - h[:n - offset])
            grad[:n - offset] -= h[offset:]
        else:
            ao = -offset
            grad[:n - ao] += (1.0 - h[ao:])
            grad[ao:] -= h[:n - ao]

    grad_norm = np.linalg.norm(grad)
    if grad_norm > 1e-14:
        h_pert = h - scale * grad / grad_norm
    else:
        h_pert = h + rng.randn(n) * scale
    h_pert += rng.randn(n) * scale * 0.1
    return project_feasible(h_pert, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Initial coordinate descent
    h, c5 = fast_coord_descent(best_h.copy(), n, max_sweeps=40)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h

    # Phase 2: Linearized minimax iterations
    if remaining_time() > 60:
        step_size = 0.7
        for it in range(800):
            if remaining_time() < 10:
                break
            h_new = linearized_minimax_step(best_h, n, step_size, top_k=30)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new
            else:
                step_size = max(0.1, step_size * 0.95)
                if step_size <= 0.1:
                    step_size = 0.7

    # Phase 3: Coordinate descent after linearized minimax
    if remaining_time() > 30:
        h, c5 = fast_coord_descent(best_h.copy(), n, max_sweeps=40)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 4: SLSQP with smooth max
    for beta in [2000, 10000, 50000]:
        if remaining_time() < 40:
            break
        h_new = slsqp_refine(best_h.copy(), n, beta, 3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        if remaining_time() > 30:
            h_cd, c5_cd = fast_coord_descent(best_h.copy(), n, max_sweeps=25)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Phase 5: More linearized minimax with larger top_k
    if remaining_time() > 60:
        for it in range(600):
            if remaining_time() < 10:
                break
            tk = 50 if it % 100 < 50 else 80
            h_new = linearized_minimax_step(best_h, n, 0.5, top_k=tk)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new

    # Phase 6: Coordinate descent
    if remaining_time() > 20:
        h, c5 = fast_coord_descent(best_h.copy(), n, max_sweeps=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 7: Subgradient-directed restarts
    if remaining_time() > 80:
        for restart in range(6):
            if remaining_time() < 30:
                break
            scale = 0.03 - restart * 0.003
            h_pert = subgradient_directed_perturb(best_h, n, rng, scale)
            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=40)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            if remaining_time() > 40:
                h_sl = slsqp_refine(best_h.copy(), n, 30000, 2000)
                c5_sl = compute_c5(h_sl)
                if c5_sl < best_c5:
                    best_c5 = c5_sl
                    best_h = h_sl

    # Phase 8: High SLSQP beta polish
    if remaining_time() > 40:
        for beta in [100000, 300000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_refine(best_h.copy(), n, beta, 4000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                if remaining_time() > 20:
                    h_cd, c5_cd = fast_coord_descent(best_h.copy(), n, max_sweeps=20)
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd

    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        h, c5 = fast_coord_descent(best_h.copy(), n, max_sweeps=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h

    # Phase 10: One final SLSQP
    if remaining_time() > 15:
        h_new = slsqp_refine(best_h.copy(), n, 200000, 3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 11: Try higher resolution if time permits
    if remaining_time() > 150:
        high_n = n * 2
        h_high = interp_to(best_h, n, high_n)
        h_high = project_feasible(h_high, high_n)
        h_hh, c5_hh = fast_coord_descent(h_high, high_n, max_sweeps=25)
        if remaining_time() > 80:
            h_hh = slsqp_refine(h_hh, high_n, 10000, 2000)
        c5_hh = compute_c5(h_hh)
        if remaining_time() > 40:
            h_hh, c5_hh2 = fast_coord_descent(h_hh, high_n, max_sweeps=20)
            if c5_hh2 < c5_hh:
                c5_hh = c5_hh2
        if c5_hh < best_c5:
            best_c5 = c5_hh
            best_h = h_hh
            best_n = high_n

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
