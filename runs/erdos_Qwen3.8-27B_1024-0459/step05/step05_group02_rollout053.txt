import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TOPK = 5
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=5, maxiter=800):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 10})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_pass(h, n, top_k=20, max_iters=10000, step_init=0.04):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.9
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        sh = 2.0 * (c_top - np.max(c_top))
        sh = np.clip(sh, -50.0, 50.0)
        w = np.exp(sh)
        sw = np.sum(w)
        if sw > 0:
            w = w / sw
        else:
            w = np.ones(top_k) / top_k
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.12)
        else:
            if it % 500 == 0:
                h = best_h.copy()
                v *= 0.3
                step *= 0.6
            else:
                v *= 0.5
                step *= 0.8

        if it % 2000 == 0 and it > 0:
            step *= 0.85
            momentum = min(momentum + 0.003, 0.94)

    return best_h, best_c5


def binary_sa(h_bin, n, max_iters=8000, top_k=5, t0=0.003, tcool=0.9997):
    """Binary local search with simulated annealing."""
    h = h_bin.copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx
    best_c5 = cur_c5
    rng = np.random.RandomState(42)

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        cur_c5 = float(np.max(corr)) * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        idx_top = np.argsort(-corr)[:top_k]

        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h, ki, n)
        g -= np.mean(g)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        score_one = -g[ones]
        score_zero = g[zeros]

        n_try_o = min(20, len(ones))
        n_try_z = min(20, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try_o]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        best_delta = 0
        best_swap = None
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                h[i] = 1
                h[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is None:
            for _ in range(8):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                h[i] = 1
                h[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is None:
            break

        i, j = best_swap
        accept = best_delta < 0
        if not accept:
            temp = t0 * (tcool ** it)
            if temp > 1e-8:
                r = rng.random()
                if r < np.exp(-best_delta / max(temp, 1e-10)):
                    accept = True

            h[i] = 0
            h[j] = 1
            cur_c5 = compute_c5(h)
        else:

                cur_c5 = best_c5

    return best_h, best_c5


def top_equalize(h, n, top_k=10, max_iters=300):

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            if it > 0:
                break
            d = -g_max
            dn = np.linalg.norm(d)
            if dn < 1e-13:
                break
        d /= dn

        improved = False
        for scale in [3, 1, 5, 0.5, 2, 0.1, 8, 0.2]:
            alpha = 0.02 * scale

            if c5_try < best_c5 - 1e-14:

                improved = True

                    improved = True

def coordinate_descent(h, n, max_sweeps=30, top_k=20):
    """Coordinate descent: cycle through positions, adjust each to min max-corr."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(7)

    for sweep in range(max_sweeps):
        if remaining() < 5:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            old_val = h[i]
            # Binary search for optimal h[i] in [max(0, old_val-0.3), min(1, old_val+0.3)]
            lo = max(0.0, old_val - 0.2)
            hi = min(1.0, old_val + 0.2)
            if hi - lo < 1e-10:
                continue
            best_v = old_val
            best_local = compute_c5(h)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                h[i] = mid
                v1 = proj(h, n)
                c5_1 = compute_c5(v1)
                h[i] = (lo + mid) / 2.0
                v2 = proj(h, n)
                c5_2 = compute_c5(v2)
                h[i] = (mid + hi) / 2.0
                v3 = proj(h, n)
                c5_3 = compute_c5(v3)
                if min(c5_1, c5_2, c5_3) < best_local:
                    mvals = [mid, (lo + mid) / 2.0, (mid + hi) / 2.0]
                    mdeps = [c5_1, c5_2, c5_3]
                    bi = np.argmin(mdeps)
                    best_v = mvals[bi]
                    best_local = mdeps[bi]
                    improved = True
                    if mdeps[bi] == c5_1:
                        pass
                    elif mdeps[bi] == c5_2:
                        hi = mid
                    else:
                        lo = mid
                else:
                    # Try midpoint
                    h[i] = mid
                    vm = proj(h, n)
                    c5_m = compute_c5(vm)
                    if c5_m < best_local:
                        best_v = mid
                        best_local = c5_m
                        improved = True
                        hi = mid
                    else:
                        lo = mid
                    break
            h[i] = best_v
        h = proj(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
        if not improved:
            break
    return h, best_c5
