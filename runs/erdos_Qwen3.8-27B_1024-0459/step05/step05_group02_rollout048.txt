        if best_move is not None:
            if best_delta < -1e-12:
                accept = True
            else:
                if rng.rand() < np.exp(min(0.0, best_delta) / (T * dx)):
                    accept = True
        else:
            # Random 1-swap as escape
            if it % 7 == 0:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i != j:
                    h_bin[i] = 0
                    h_bin[j] = 1
                    nc = compute_c5(h_bin)
                    delta = nc - cur_c5
                    if delta < 0 or rng.rand() < np.exp(delta / (T * dx * 10)):
                        accept = True
                        best_move = ('r', i, j, None, None)
                    else:
                        h_bin[i] = 1
                        h_bin[j] = 0

        if not accept:
            stale += 1
            if stale > 80:
                # Larger perturbation
                n_flip = rng.randint(3, 8)
                for _ in range(n_flip):
                    if len(ones) > n_ones_target + 1 and len(zeros) > 1:
                        i = int(ones[rng.randint(len(ones))])
                        j = int(zeros[rng.randint(len(zeros))])
                        h_bin[i] = 0
                        h_bin[j] = 1
                    elif len(zeros) > n_ones_target + 1 and len(ones) > 1:
                        i = int(zeros[rng.randint(len(zeros))])
                        j = int(ones[rng.randint(len(ones))])
                        h_bin[i] = 1
                        h_bin[j] = 0
                cur_c5 = compute_c5(h_bin)
                stale = 0
                continue
            continue

        # Apply move
        if best_move[0] == '1' or best_move[0] == 'r':
            h_bin[best_move[1]] = 0
            h_bin[best_move[2]] = 1
        else:
            h_bin[best_move[1]] = 0
            h_bin[best_move[2]] = 0
            h_bin[best_move[3]] = 1
            h_bin[best_move[4]] = 1

        cur_c5 = compute_c5(h_bin)
        stale = 0
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def gradient_continuous_optimize(h, n, top_k=8, max_iters=3000, step_init=0.05):
    """Nesterov momentum projected gradient descent on smooth max."""
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx:
            g += corr_grad_single(h, ki, n)
        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.12)
        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.75

        if it % 2000 == 0 and it > 0:
            step *= 0.8
            momentum = min(momentum + 0.005, 0.93)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = proj(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse_init(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n

    # Phase 1: Quick L-BFGS at parent resolution
    for beta in [500, 2000, 8000]:
        if remaining() < 15:
            break
        h_new = lbfgs_pass(best_h, n, beta, top_k=6, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Binarize and efficient binary search at parent n
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = proj(h_bin, n)
        h_bs, c5_bs = efficient_binary_search(h_bin, n, max_iters=3000, top_k=6, rng=rng)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()

    # Phase 3: Try higher resolutions
    higher_ns = []
    for factor in [1.5, 2.0, 2.5, 3.0]:
        nn = int(n * factor)
        nn = 2 * (nn // 2)
        if nn > n and nn <= 1536:
            higher_ns.append(nn)
    higher_ns = sorted(set(higher_ns))

    for n_t in higher_ns:
        if remaining() < 60:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # L-BFGS at new resolution
        for beta in [1000, 5000, 15000]:
            if remaining() < 20:
                break
            h_feas = lbfgs_pass(h_feas, n_t, beta, top_k=6, maxiter=350)
            c5_chk = compute_c5(h_feas)
            if c5_chk < best_c5:
                best_c5 = c5_chk
                best_h = h_feas.copy()
                best_n = n_t

        # Binary search at new resolution
        if remaining() > 30:
            h_b = (h_feas > 0.5).astype(float)
            h_b = proj(h_b, n_t)
            h_bs2, c5_bs2 = efficient_binary_search(h_b, n_t, max_iters=4000, top_k=6, rng=rng)
            if c5_bs2 < best_c5:
                best_c5 = c5_bs2
                best_h = h_bs2.copy()
                best_n = n_t

            # Continuous refinement after binary
            if remaining() > 20:
                h_ref = h_bs2.copy()
                h_ref2, c5_ref = gradient_continuous_optimize(h_ref, n_t, top_k=6, max_iters=2000, step_init=0.04)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref2.copy()
                    best_n = n_t

                h_ref3 = lbfgs_pass(h_ref2, n_t, 20000, top_k=5, maxiter=300)
                c5_r3 = compute_c5(h_ref3)
                if c5_r3 < best_c5:
                    best_c5 = c5_r3
                    best_h = h_ref3.copy()
                    best_n = n_t

    # Phase 4: Multiple restarts with perturbation at best resolution
    n_cur = best_n
    restart_count = 0
    max_restarts = 4
    while remaining() > 60 and restart_count < max_restarts:
        restart_count += 1
        h_rom = best_h.copy()
        # Random perturbation: flip 5-15 pairs
        n_flip = rng.randint(5, 16)
        ones = np.where(h_rom > 0.5)[0]
        zeros = np.where(h_rom < 0.5)[0]
        if len(ones) > 0 and len(zeros) > 0:
            flip_o = rng.choice(ones, size=min(n_flip, len(ones)), replace=False)
            flip_z = rng.choice(zeros, size=min(n_flip, len(zeros)), replace=False)
            n_f = min(len(flip_o), len(flip_z))
            h_rom[flip_o[:n_f]] = 0.0
            h_rom[flip_z[:n_f]] = 1.0
        h_rom = proj(h_rom, n_cur)

        # Quick optimization
        h_rom = lbfgs_pass(h_rom, n_cur, 5000, top_k=6, maxiter=300)
        h_rom_bin = (h_rom > 0.5).astype(float)
        h_rom_bin = proj(h_rom_bin, n_cur)
        h_rom_bs, c5_rom = efficient_binary_search(h_rom_bin, n_cur, max_iters=2000, top_k=6, rng=rng)
        if c5_rom < best_c5:
            best_c5 = c5_rom
            best_h = h_rom_bs.copy()
            best_n = n_cur
        else:
            c5_rom_c = compute_c5(h_rom)
            if c5_rom_c < best_c5:
                best_c5 = c5_rom_c
                best_h = h_rom.copy()
                best_n = n_cur

    # Phase 5: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # Gradient descent
    if remaining() > 20:
        h_pg, c5_pg = gradient_continuous_optimize(h, n, top_k=8, max_iters=4000, step_init=0.03)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # L-BFGS with very high beta (focus on max)
    for beta in [30000, 60000]:
        if remaining() < 10:
            break
        h_new = lbfgs_pass(best_h, n, beta, top_k=4, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bin_f = proj(h_bin_f, n)
        h_bf, c5_bf = efficient_binary_search(h_bin_f, n, max_iters=3000, top_k=5, rng=rng)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        # Continuous after binary
        if remaining() > 10:
            h_rf = lbfgs_pass(h_bf, n, 30000, top_k=4, maxiter=300)
            c5_rf = compute_c5(h_rf)
            if c5_rf < best_c5:
                best_c5 = c5_rf
                best_h = h_rf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    if final_c5 < best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, best_n)
