def compute_corr(h):

def project_feasible(h, n):

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:

        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g


def top_k_gradient(h, n, K=20, beta=50.0):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    cmax = c_top.max()
    weights = np.exp(beta * (c_top - cmax))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, k, n)
    return g, float(np.max(corr))

def smooth_max_slsqp(h, n, beta, maxiter=400):
    dx = 2.0 / n
    
    def sm_obj(x):

        m = np.max(corr)
        threshold = m - 6.0 / max(beta, 1.0)

        if len(active) == 0:
            active = np.array([m])
        sh = active - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        se = np.sum(ex)
        if se < 1e-300:
            return m * dx
        return (m + np.log(se) / beta) * dx
    
    def sm_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        threshold = m - 6.0 / max(beta, 1.0)
        active_idx = np.where(corr >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        
        sh = corr[active_idx] - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        se = np.sum(ex)
        if se < 1e-300:
            return np.zeros(n)
        w = ex / (se * dx)

        return np.ones(n)

                break
            h4 = smooth_max_slsqp(h4, n4, beta, maxiter=250)

        if remaining() > 15:
            h4, c5_4 = equalize_and_descend(h4, n4, max_iters=100, rng=rng)

        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
            best_n = n4

    # === Phase 5: Intensive final polish ===
    n = best_n
    h = best_h.copy()

    for cycle in range(3):
        if remaining() < 25:
            break

        # Nesterov
        if remaining() > 20:
            h, c5_t = nesterov_subgradient(h, n, min(18.0, remaining() * 0.25), rng)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # SLSQP high beta
        for beta in [80000, 200000, 500000]:
            if remaining() < 10:
                break
            h = smooth_max_slsqp(h, n, beta, maxiter=400)
        c5_t = compute_c5(h)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()

        # Equalize
        if remaining() > 15:
            h, c5_t = equalize_and_descend(h, n, max_iters=200, rng=rng)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

        # Micro
        if remaining() > 10:
            h, c5_t = micro_refine(h, n, max_iters=200)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

    # === Phase 6: Random restart perturbation (try to escape local min) ===
    if remaining() > 40:
        for restart in range(3):
            if remaining() < 20:
                break
            h_r = best_h.copy() + rng.randn(best_n) * 0.005
            h_r = project_feasible(h_r, best_n)
            c5_r = compute_c5(h_r)

            if remaining() > 15:
                h_r, c5_r = nesterov_subgradient(h_r, best_n, min(12.0, remaining() * 0.4), rng)

            if remaining() > 10:
                h_r = smooth_max_slsqp(h_r, best_n, 50000, maxiter=300)

            if remaining() > 10:
                h_r, c5_r = equalize_and_descend(h_r, best_n, max_iters=100, rng=rng)

            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # === Final micro-tuning ===
    if remaining() > 5:
        h, c5_t = micro_refine(best_h, best_n, max_iters=300)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h.copy()

    # Ensure final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
