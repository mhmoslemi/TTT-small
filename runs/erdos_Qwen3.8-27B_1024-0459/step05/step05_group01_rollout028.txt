    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.fun < _sm_obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def top_k_equalize(h, n, top_k=5, max_iters=500):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.003, 0.005, 0.01, 0.02, 0.002, 0.008, 0.03, 0.08, 0.15, 0.05]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def random_balanced(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def binary_local_search_v2(h, n, time_limit=120, rng=None):
    """Binary local search with single and double swaps + simulated annealing."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = 0.01
    T_min = 1e-6
    T_decay = 0.9995
    no_improve = 0
    max_no_improve = 200

    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        c_old = compute_c5(h)
        improved = False

        # Try single swap (deterministic best if small, random if large)
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= 50000:
            best_delta = 1e-10
            best_move = None
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c_new = compute_c5(h)
                    h[i] = 1.0
                    h[j] = 0.0
                    delta = c_new - c_old
                    if delta < best_delta:
                        best_delta = delta
                        best_move = (i, j)
            if best_move is not None:
                i, j = best_move
                h[i] = 0.0
                h[j] = 1.0
                c_new = compute_c5(h)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Random single swaps + occasional double swaps
            n_trials = 200
            best_delta = 1e-10
            best_move = None
            best_move_type = 0
            for _ in range(n_trials):
                if rng.random() < 0.7:
                    # Single swap
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                    c_new = compute_c5(h)
                    h[i] = 1.0
                    h[j] = 0.0
                    delta = c_new - c_old
                    if delta < best_delta:
                        best_delta = delta
                        best_move = (i, j)
                        best_move_type = 0
                else:
                    # Double swap
                    if len(ones) < 2 or len(zeros) < 2:
                        continue
                    i1 = ones[rng.randint(len(ones))]
                    i2 = ones[rng.randint(len(ones))]
                    j1 = zeros[rng.randint(len(zeros))]
                    j2 = zeros[rng.randint(len(zeros))]
                    if i1 == i2 or j1 == j2:
                        continue
                    h[i1] = 0.0
                    h[i2] = 0.0
                    h[j1] = 1.0
                    h[j2] = 1.0
                    c_new = compute_c5(h)
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
                    delta = c_new - c_old
                    if delta < best_delta:
                        best_delta = delta
                        best_move = (i1, i2, j1, j2)
                        best_move_type = 1

            if best_move is not None:
                if best_move_type == 0:
                    i, j = best_move
                    h[i] = 0.0
                    h[j] = 1.0
                else:
                    i1, i2, j1, j2 = best_move
                    h[i1] = 0.0
                    h[i2] = 0.0
                    h[j1] = 1.0
                    h[j2] = 1.0
                c_new = compute_c5(h)
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1

        # Simulated annealing: accept worse moves with probability exp(-delta/T)
        if not improved and no_improve > max_no_improve // 2:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - compute_c5(best_h)
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                no_improve = 0
            else:
                h[i] = 1.0
                h[j] = 0.0

        if no_improve > max_no_improve:
            # Restart: random perturbation
            for _ in range(10):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
            no_improve = 0

        T *= T_decay
        if T < T_min:
            T = T_min

    return best_h, best_c5


def subgradient_method(h, n, max_iters=2000, lr_start=0.02):
    """Projected subgradient method for the max correlation."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_start

    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5 = c_max * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Subgradient: gradient of the max active correlation
        active = np.where(np.abs(corr - c_max) < 1e-8)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        w = np.zeros(len(corr))
        w[active] = 1.0 / len(active)

        # Compute gradient
        g = np.zeros(n)
        for k in active:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += (1.0 - h[:np_]) / len(active)
                g[:np_] -= h[off:] / len(active)
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:]) / len(active)
                g[a:] -= h[:np_] / len(active)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = g / gn

        # Try several step sizes
        improved = False
        for alpha in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            ht = project_feasible(h - alpha * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-12:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            # Take a small step anyway (for exploration)
            h = project_feasible(h - lr * 0.1 * g, n)

        # Decay learning rate
        lr *= 0.999
        if lr < 1e-5:
            lr = 1e-5

    return best_h, best_c5


def coordinate_refine(h, n, max_sweeps=50):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0, old]
            if 0.0 < old < 1.0:
                cands.extend([old * 0.5, old + (1.0 - old) * 0.5,
                              max(0.0, old - 0.25), min(1.0, old + 0.25)])
            h[i] = old
            h = project_feasible(h, n)
            best_cand_c5 = compute_c5(h)
            best_cand_val = old
            for c in cands:
                h[i] = np.clip(c, 0.0, 1.0)
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-12:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
            if best_cand_c5 < best_c5:
                best_c5 = best_cand_c5
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive SLSQP at original resolution
    h = best_h.copy()
    betas = [200, 1000, 5000, 20000, 100000, 500000, 2000000, 10000000]
    for beta in betas:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization (fixed version)
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=6, max_iters=800)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Subgradient method
    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_method(h, best_n, max_iters=3000, lr_start=0.01)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: More SLSQP
    for beta in [500000, 5000000, 50000000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Multi-resolution search
    n_targets = [best_n]
    for mult in [0.75, 1.0, 1.25, 1.5, 2.0, 2.5]:
        nn = int(best_n * mult)
        if nn % 2 != 0:
            nn += 1
        if 200 <= nn <= 2048:
            n_targets.append(nn)
    n_targets = sorted(set(n_targets))
    n_targets.sort(key=lambda x: abs(x - best_n))

    for n_t in n_targets[:10]:
        if remaining_time() < 30:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [1000, 10000, 100000, 1000000]:
            if remaining_time() < 15:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=600)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=5, max_iters=400)

        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 6: Binary local search with v2 (double swaps + annealing)
    if remaining_time() > 40:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0

        sa_time = min(200, max(20, int(remaining_time() * 0.35)))
        h_b, c5_b = binary_local_search_v2(h_bin, best_n, time_limit=sa_time, rng=rng)

        # Refine binary result continuously
        h_b_c = h_b.copy()
        for beta in [5000, 50000, 500000, 5000000]:
            if remaining_time() < 10:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=500)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = top_k_equalize(h_b_c, best_n, top_k=5, max_iters=400)

        if c5_b_c < best_c5 - 1e-14:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()

        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 7: Alternative starting points
    alt_starts = []
    alt_starts.append(theta_morse(best_n))
    alt_starts.append(random_balanced(best_n, rng))
    # Antisymmetric: h[i] + h[n-1-i] = 1
    h_asym = np.zeros(best_n)
    half = best_n // 2
    for i in range(half):
        val = rng.random()
        h_asym[i] = val
        h_asym[best_n - 1 - i] = 1.0 - val
    h_asym[best_n - 1 - half] = 0.5 if best_n % 2 == 0 else 0.5
    # Make it binary-like
    h_asym = (h_asym > 0.5).astype(float)
    n_ones = int(np.sum(h_asym))
    if n_ones > best_n // 2:
        excess = n_ones - best_n // 2
        ones_idx = np.where(h_asym == 1.0)[0]
        for i in range(excess):
            h_asym[ones_idx[i]] = 0.0
    elif n_ones < best_n // 2:
        deficit = best_n // 2 - n_ones
        zeros_idx = np.where(h_asym == 0.0)[0]
        for i in range(deficit):
            h_asym[zeros_idx[i]] = 1.0
    alt_starts.append(h_asym)

    for h_alt in alt_starts:
        if remaining_time() < 30:
            break
        h_alt = project_feasible(h_alt, best_n)
        h_opt = h_alt
        for beta in [1000, 10000, 100000]:
            if remaining_time() < 15:
                break
            h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=500)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=5, max_iters=300)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [1000000, 10000000, 100000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=8, max_iters=600)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 10:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=40)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP polish
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 10000000, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
