def interp_to(src, src_n, dst_n):

        return src.copy()

    return np.interp(xn, xp, src)

def top_k_refine(h, n, top_k=5, n_iters=200, time_limit=30):

    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    start = time.time()
    step = 0.01
    for it in range(n_iters):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        # Direction: decrease max, increase min (among top k)
        g = np.zeros(n)
        for pos, k in enumerate(idx):
            if pos == 0:
                w_k = 1.0
            else:
                w_k = -1.0 / (top_k - 1)
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ <= 0: continue
                g[off:] += w_k * (1.0 - h[:np_])
                g[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                if np_ <= 0: continue
                g[:np_] += w_k * (1.0 - h[a:])
                g[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = g / gn
        improved = False
        for s in [step, step*0.5, step*2, step*0.1, step*5, 0.005, 0.02, 0.05, 0.1]:
            ht = project_feasible(h + s * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                step = s
                improved = True
                break
        if not improved:
            step *= 0.5
    return h, best_c5

    corr = compute_corr(x)
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    smooth = (m + np.log(np.sum(ez)) / beta) * dx
    viol = np.sum(x) - n / 2.0
    return smooth + pen * viol * viol

    corr = compute_corr(x)

        if time.time() - start > time_limit or remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ <= 0:
                    continue
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                if np_ <= 0:
                    continue
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g /= top_k
        dx_ = 2.0 / n
        for s in [step, step * 0.5, step * 0.2, 0.01]:
            ht = project_feasible(h - s * dx_ * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                step = s
                break
        else:
            step *= 0.7
    return h, best_c5


def binary_swap_search(h, n, time_limit=120, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_pairs = len(ones) * len(zeros)
        improved = False
        if n_pairs <= 60000:
            best_delta = 1e-10
            best_pair = None
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c_new = compute_c5(h)
                    h[i] = 1.0
                    h[j] = 0.0
                    delta = c_new - best_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h[i] = 0.0
                h[j] = 1.0
                best_c5 = compute_c5(h)
                best_h = h.copy()
                improved = True
        else:
            n_trials = min(600, n_pairs)
            best_delta = 1e-10
            best_pair = None
            for _ in range(n_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c_new = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = c_new - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h[i] = 0.0
                h[j] = 1.0
                best_c5 = compute_c5(h)
                best_h = h.copy()
                improved = True
        if not improved:
            no_improve += 1
            if no_improve > 25:
                break
            for _ in range(5):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
            best_c5 = compute_c5(h)
            if best_c5 < compute_c5(best_h):
                best_h = h.copy()
            else:
                h = best_h.copy()
                best_c5 = compute_c5(h)
        else:
            no_improve = 0
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive L-BFGS-B on parent resolution
    h = best_h.copy()
    betas_p1 = [100, 300, 1000, 3000, 10000, 30000, 100000, 300000, 1000000]
    for beta in betas_p1:
        if remaining_time() < 12:
            break
        h_new = lbfgs_opt(h, best_n, beta, maxiter=400, pen=1e4)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization
    if remaining_time() > 15:
        h_eq, c5_eq = top_k_refine(h, best_n, top_k=5, n_iters=400, time_limit=30)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: More L-BFGS-B with higher beta
    for beta in [500000, 2000000, 5000000]:
        if remaining_time() < 10:
            break
        h_new = lbfgs_opt(h, best_n, beta, maxiter=500, pen=1e5)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Subgradient descent on true max
    if remaining_time() > 15:
        h_sg, c5_sg = subgrad_descent(h, best_n, time_limit=15, top_k=3)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 5: Multi-resolution search
    n_tries = []
    for d in range(-20, 41, 2):
        nn = best_n + d
        if 180 <= nn <= 1200 and nn % 2 == 0:
            n_tries.append(nn)
    for nv in [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024]:
        if 180 <= nv <= 1200 and nv % 2 == 0:
            n_tries.append(nv)
    n_tries = sorted(set(n_tries), key=lambda x: abs(x - best_n))

    for n_t in n_tries[:12]:
        if remaining_time() < 18:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt = h_feas
        for beta in [500, 5000, 50000, 200000]:
            if remaining_time() < 8:
                break
            h_opt = lbfgs_opt(h_opt, n_t, beta, maxiter=250, pen=1e4)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_refine(h_opt, n_t, top_k=4, n_iters=200, time_limit=8)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 6: Binary search on best resolution
    if remaining_time() > 35:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        if int(np.sum(h_bin)) != n_half:
            idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[idx[:n_half]] = 1.0

        sa_time = min(200, max(20, int(remaining_time() * 0.35)))
        h_b, c5_b = binary_swap_search(h_bin, best_n, time_limit=sa_time, rng=rng)

        # Refine binary result continuously
        h_b_c = h_b.copy()
        for beta in [2000, 20000, 200000, 2000000]:
            if remaining_time() < 8:
                break
            h_b_c = lbfgs_opt(h_b_c, best_n, beta, maxiter=300, pen=1e4)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = top_k_refine(h_b_c, best_n, top_k=5, n_iters=200, time_limit=10)

        if c5_b_c < best_c5 - 1e-14:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 7: Alternative starting structures
    if remaining_time() > 40:
        h_tm = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt += v & 1
                v >>= 1
            h_tm[i] = 1.0 if cnt % 2 == 1 else 0.0
        h_tm = project_feasible(h_tm, best_n)

        h_rand = np.zeros(best_n)
        h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0

        h_pattern = np.zeros(best_n)
        for i in range(best_n):
            h_pattern[i] = 1.0 if (i % 3 == 0) or (i % 3 == 1 and i % 6 < 3) else 0.0
        h_pattern = project_feasible(h_pattern, best_n)

        for h_alt in [h_tm, h_rand, h_pattern]:
            if remaining_time() < 15:
                break
            h_opt = h_alt
            for beta in [500, 5000, 50000]:
                if remaining_time() < 6:
                    break
                h_opt = lbfgs_opt(h_opt, best_n, beta, maxiter=200, pen=1e4)
            c5_opt = compute_c5(h_opt)
            h_opt, c5_opt = top_k_refine(h_opt, best_n, top_k=5, n_iters=100, time_limit=8)
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [1000000, 5000000, 20000000]:
        if remaining_time() < 8:
            break
        h_new = lbfgs_opt(h, n, beta, maxiter=600, pen=1e5)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = top_k_refine(h, n, top_k=7, n_iters=400, time_limit=15)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 6:
        h_sg, c5_sg = subgrad_descent(best_h, n, time_limit=8, top_k=5)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final polish
    if remaining_time() > 5:
        h_new = lbfgs_opt(best_h, n, 10000000, maxiter=300, pen=1e6)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
