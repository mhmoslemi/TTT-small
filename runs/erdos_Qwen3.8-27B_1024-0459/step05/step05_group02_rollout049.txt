import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TOPK = 5
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):

    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]

    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    pen = 200.0 * (np.sum(x) - _TGT) ** 2
    return val + pen


def _sm_grad(x):

def nesterov_max(h, n, max_iters, step_init=0.02, top_k=3):

        if remaining() < 5:

        corr = compute_corr(h)

            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 2000:
            if step > 0.001:
                step *= 0.9
            stall_count = 0

        c_top = corr[idx]
        # Softmax weights emphasizing the max

        sh = bwc * (c_top - c_top[0])
        w = np.exp(np.clip(sh, -20, 0))
        w /= np.sum(w)

        g = np.zeros(n)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-10)] = 0.0
        g[(h <= 1e-10)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_try = h + v
        h_new = proj(h_try, n)
        c5_new = compute_c5(h_new)

        if c5_new <= c5 + 0.002:
            h = h_new
            if c5_new < c5:
                step = min(step * 1.005, 0.08)
        else:
            v *= 0.3
            step *= 0.85

        if it % 500 == 0 and it > 0:
            momentum = min(momentum + 0.003, 0.93)

    return best_h, best_c5


def subgrad_line_search(h, n, max_iters=500, step_init=0.01):

def binary_sa(h_init, n, n_iters, rng, t_init=0.003, t_final=0.00003):
    """Simulated annealing with swap moves for binary h."""
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            idx_ones = np.where(h == 1)[0]
            excess = n_ones - n_half
            remove = idx_ones[:excess]
            h[remove] = 0
        else:
            idx_zeros = np.where(h == 0)[0]
            deficit = n_half - n_ones
            add = idx_zeros[:deficit]
            h[add] = 1

    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    n_steps = max(1, n_iters)

    for it in range(n_iters):
        if remaining() < 5:
            break

        t = t_init * (t_final / t_init) ** (it / n_steps)

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5

        if delta < 0 or (t > 1e-8 and rng.random() < np.exp(-delta / t)):
            cur_c5 = new_c5
            # Update sets occasionally
            if it % 50 == 0 or it == n_iters - 1:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                pos_i = np.where(ones == i)[0]
                pos_j = np.where(zeros == j)[0]
                if len(pos_i) > 0:
                    ones = np.delete(ones, pos_i[0])
                if len(pos_j) > 0:
                    zeros = np.delete(zeros, pos_j[0])
                ones = np.append(ones, j)
                zeros = np.append(zeros, i)

            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5


def binary_local_search(h_init, n, max_iters=3000, top_k=4, rng=None):

            h[idx[:n_half - n_ones]] = 1

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0

    for it in range(max_iters):
        if remaining() < 5:
            break

        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]

        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h, ki, n)
        g -= np.mean(g)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        score_one = -g[ones]
        score_zero = g[zeros]

        n_try_o = min(16, len(ones))
        n_try_z = min(16, len(zeros))
        top_o = ones[np.argsort(-score_one)[:n_try_o]]
        top_z = zeros[np.argsort(-score_zero)[:n_try_z]]

        best_delta = 0
        best_swap = None
        for i in top_o:
            for j in top_z:
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                h[i] = 1
                h[j] = 0
                d = nc - cur_c5
                if d < best_delta:
                    best_delta = d
                    best_swap = (i, j)

        if best_swap is None or best_delta >= 0:
            # Random exploratory swaps
            found = False
            for _ in range(8):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                h[i] = 1
                h[j] = 0
                d = nc - cur_c5
                if d < best_delta:
                    best_delta = d
                    best_swap = (i, j)
                    found = True
            if not found:
                stall += 1
                if stall > 100:
                    break
                continue
        else:
            stall = 0

        i, j = best_swap
        h[i] = 0
        h[j] = 1
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def fourier_init(n, seed, num_modes=5):
    """Generate a random Fourier-based initialization."""
    rng = np.random.RandomState(seed)
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = np.zeros(n)
    for _ in range(num_modes):
        freq = rng.randint(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        h += np.cos(2 * np.pi * freq * x / 2.0 + phase)
    h = 0.5 + 0.3 * h / np.max(np.abs(h))
    h = np.clip(h, 0, 1)
    return proj(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Intensive L-BFGS-B at parent resolution with increasing beta
    for beta, tk, mi in [(500, 8, 500), (2000, 6, 800), (8000, 4, 1000),
                         (20000, 3, 800), (50000, 2, 600)]:
        if remaining() < 15:
            break
        h_new = lbfgs_pass(h, n, beta, tk, mi)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Nesterov gradient with true max
    if remaining() > 60:
        h_pg, c5_pg = nesterov_max(h, n, max_iters=8000, step_init=0.02, top_k=3)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 3: Subgradient line search on true max
    if remaining() > 30:
        h_sg, c5_sg = subgrad_line_search(h, n, max_iters=300, step_init=0.008)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: Binary local search from continuous solution
    if remaining() > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_bl, c5_bl = binary_local_search(h_bin, best_n, max_iters=2000, top_k=4, rng=rng)
        # Refine continuously from binary
        h_ref = h_bl.copy()
        for bt in [5000, 20000]:
            if remaining() < 8:
                break
            h_ref = lbfgs_pass(h_ref, best_n, bt, 3, 400)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
        elif c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Phase 5: Binary SA for diversification
    if remaining() > 120:
        # SA from current best
        h_sa, c5_sa = binary_sa(best_h, best_n, n_iters=500000, rng=rng,
                                t_init=0.002, t_final=0.00002)
        h_ref_sa = h_sa.copy()
        for bt in [5000, 20000, 80000]:
            if remaining() < 8:
                break
            h_ref_sa = lbfgs_pass(h_ref_sa, best_n, bt, 3, 300)
        c5_ref_sa = compute_c5(h_ref_sa)
        if c5_ref_sa < best_c5:
            best_c5 = c5_ref_sa
            best_h = h_ref_sa.copy()
        elif c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Try different resolutions
    n_candidates = [best_n + 32, best_n + 64, best_n + 128,
                    best_n - 32, int(best_n * 1.5), best_n * 2]
    n_candidates = sorted(set([nn for nn in n_candidates
                               if 128 <= nn <= 1200 and nn % 2 == 0
                               and nn != best_n]))
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        for bt in [2000, 10000, 30000]:
            if remaining() < 12:
                break
            h_feas = lbfgs_pass(h_feas, n_t, bt, 4, 500)

        h_pg2, c5_pg2 = nesterov_max(h_feas, n_t, max_iters=3000, step_init=0.015, top_k=3)
        c5_t = min(compute_c5(h_feas), c5_pg2)

        # Binary SA at this resolution
        if remaining() > 60:
            h_bin_t = (h_feas if c5_pg2 > compute_c5(h_feas) else h_pg2)
            h_bin_t = (h_bin_t > 0.5).astype(float)
            h_bl_t, c5_bl_t = binary_local_search(h_bin_t, n_t, max_iters=1000, top_k=3, rng=rng)
            h_ref_t = lbfgs_pass(h_bl_t, n_t, 20000, 3, 300)
            c5_ref_t = compute_c5(h_ref_t)
            c5_t = min(c5_t, c5_bl_t, c5_ref_t)
            if c5_ref_t <= c5_t:
                h_best_t = h_ref_t
            elif c5_bl_t <= c5_t:
                h_best_t = h_bl_t
            else:
                h_best_t = h_feas if c5_pg2 > compute_c5(h_feas) else h_pg2
        else:
            h_best_t = h_feas if c5_pg2 > compute_c5(h_feas) else h_pg2

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_best_t.copy()
            best_n = n_t

    # Phase 7: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    # More L-BFGS-B with very high beta
    for beta, tk, mi in [(50000, 2, 500), (100000, 2, 400), (200000, 1, 300)]:
        if remaining() < 10:
            break
        h_new = lbfgs_pass(h, n, beta, tk, mi)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Nesterov final
    if remaining() > 20:
        h_pg3, c5_pg3 = nesterov_max(h, n, max_iters=5000, step_init=0.01, top_k=2)
        if c5_pg3 < best_c5:
            best_c5 = c5_pg3
            best_h = h_pg3.copy()

    # Subgradient final
    if remaining() > 15:
        h_sg2, c5_sg2 = subgrad_line_search(best_h, n, max_iters=200, step_init=0.005)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_local_search(h_bin_f, n, max_iters=1500, top_k=4, rng=rng)
        h_rf = lbfgs_pass(h_bf, n, 50000, 2, 400)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()
        elif c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Phase 8: Multi-start perturbation
    if remaining() > 60:
        for pert_seed in [123, 456, 789]:
            if remaining() < 20:
                break
            rng_p = np.random.RandomState(pert_seed)
            h_pert = best_h + rng_p.normal(0, 0.05, n)
            h_pert = proj(h_pert, n)
            h_pert = lbfgs_pass(h_pert, n, 10000, 3, 500)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    # Final Nesterov squeeze
    if remaining() > 15:
        h_ns, c5_ns = nesterov_max(best_h, n, max_iters=3000, step_init=0.005, top_k=2)
        if c5_ns < best_c5:
            best_c5 = c5_ns
            best_h = h_ns.copy()

    # Ensure feasibility
    best_n = len(best_h)
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
