        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)
    hi3 = min(j + n - 1, r - 1)
    if lo3 <= hi3:
        ks = np.arange(lo3, hi3 + 1)
        dr[lo3:hi3+1] += (1.0 - h[j + n - 1 - ks])
    lo4 = max(n - 1 - j, 0)
    hi4 = min(2 * (n - 1) - j, r - 1)
    if lo4 <= hi4:
        ks = np.arange(lo4, hi4 + 1)
        dr[lo4:hi4+1] -= h[j + ks - (n - 1)]
    dr[n - 1] -= 2.0
    return dr

def soft_max_energy(corr_arr, beta):
    top_k = min(12, len(corr_arr))
    top = np.partition(corr_arr, -top_k)[-top_k:]
    smax = top.max()
    cl = np.clip(beta * (top - smax), -50.0, 50.0)
    ex = np.exp(cl)
    return smax + np.log(np.sum(ex)) / beta

def ensure_count(h, n):
    h = h.copy()
    n_half = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h > 0.5)[0]
            h[ones[:n_ones - n_half]] = 0
        else:
            zeros = np.where(h <= 0.5)[0]
            h[zeros[:n_half - n_ones]] = 1
    return (h > 0.5).astype(float)

def sa_binary_search(h_bin, n, time_limit, rng, t_init=8.0, cooling=0.9997, t_min=0.005,
                     n_top_shifts=10, n_contrib=20, n_random=50, restart_every=300):
    h = ensure_count(h_bin, n)
    n_half = n // 2
    dx = 2.0 / n
    r = 2 * n - 1
    
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    cur_energy = soft_max_energy(corr, 500.0)
    best_h = h.copy()
    best_max = cur_max
    best_energy = cur_energy
    
    T = t_init
    stall = 0
    start = time.time()
    
    while time.time() - start < time_limit and remaining() > 10:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        candidates = []
        top_idx = np.argsort(-corr)[:n_top_shifts]
        for s in top_idx:
            off = s - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0 and off < np_:
                    contrib = np.where((h[off:np_min] == 1) & (h[:np_min] == 0))[0] if False else np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                    for ci in contrib[:n_contrib]:
                        candidates.append((ci + off, ci))
            else:
                ao = -off
                np_ = n - ao
                if np_ > 0 and ao < np_:
                    contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                    for ci in contrib[:n_contrib]:
                        candidates.append((ci, ci + ao))
        
        for _ in range(n_random):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        best_move = None
        best_new_energy = cur_energy + 1e-12
        seen = set()
        
        for idx in range(min(200, len(candidates))):
            i, j = candidates[idx]
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            
            dr = make_swap_delta(h, i, j, n)
            new_corr_here = corr + dr
            new_max = float(np.max(new_corr_here))
            new_energy = soft_max_energy(new_corr_here, 500.0)
            
            if new_energy < best_new_energy:
                best_new_energy = new_energy
                best_new_max = new_max
                best_move = (i, j, dr, new_max)
        
        if best_move is not None:
            i, j, dr, new_max = best_move
            delta = best_new_energy - cur_energy
            if delta < 0 or (T > t_min and rng.rand() < np.exp(-delta / (T + 1e-12))):
                h[i] = 0
                h[j] = 1
                corr = corr + dr
                cur_max = new_max
                cur_energy = best_new_energy
                stall = 0
                if cur_energy < best_energy or cur_max < best_max:
                    best_max = min(best_max, cur_max)
                    best_energy = min(best_energy, cur_energy)
                    if cur_max < best_max + 1e-15:
                        best_h = h.copy()
                        best_max = cur_max
                    else:
                        best_energy = cur_energy
            else:
                stall += 1
        else:
            stall += 1
        
        T *= cooling
        if T < t_min:
            T = t_min
        
        if stall > restart_every:
            n_perturb = max(3, stall // 100)
            for _ in range(n_perturb):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if h[i] == 1 and h[j] == 0:
                    dr = make_swap_delta(h, i, j, n)
                    h[i] = 0
                    h[j] = 1
                    corr = corr + dr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            cur_max = float(np.max(corr))
            cur_energy = soft_max_energy(corr, 500.0)
            stall = 0
            T = max(T, 3.0)
    
    return best_h, best_max * dx

def block_flip_move(h, n, rng, block_len):
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return None, None
    start = rng.randint(0, n - block_len + 1)
    block_h = h[start:start+block_len].copy()
    ones_in_block = np.sum(block_h)
    zeros_in_block = block_len - ones_in_block
    if ones_in_block == zeros_in_block:
        flipped = 1.0 - block_h
        return start, flipped
    return None, None

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def legendre_seq(n):
    p = n + 1
    h = np.zeros(n)
    for i in range(n):
        a = (i + 1) % p
        leg = 1.0
        base = a
        exp_val = (p - 1) // 2
        result = 1.0
        b = base
        e = exp_val
        while e > 0:
            if e % 2 == 1:
                result = (result * b) % p
            b = (b * b) % p
            e //= 2
        if result == 0:
            h[i] = 0.5
        else:
            h[i] = 1.0 if result * 2 <= p else 0.0
    n_ones = int(np.sum(h > 0.5))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            sort_idx = np.argsort(h)
            h[sort_idx[:n_ones - target]] = 0.0
        else:
            sort_idx = np.argsort(-h)
            h[sort_idx[:target - n_ones]] = 1.0
    return h

def m_sequence(n):
    # Find a good primitive polynomial for m-sequence
    # For n = 2^m - 1 type lengths, but we'll just use a pseudo-random pattern
    # Actually, let's use a simpler approach: golden ratio based
    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    vals = np.zeros(n)
    val = 0.5
    for i in range(n):
        val = (val + phi) % 1.0
        vals[i] = val
    threshold = np.median(vals)
    h = (vals > threshold).astype(float)
    # Force exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def continuous_refine(h, n, time_limit, rng):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    N = n
    TGT = n / 2.0
    
    def corr_of(x):
        return np.correlate(x, 1.0 - x, mode='full')
    
    def obj_smooth(x, beta):
        corr = corr_of(x)
        top_k = min(15, len(corr))
        top = np.partition(corr, -top_k)[-top_k:]
        smax = top.max()
        cl = np.clip(beta * (top - smax), -50.0, 50.0)
        ex = np.exp(cl)
        return (smax + np.log(np.sum(ex)) / beta) * (2.0 / N)
    
    def grad_smooth(x, beta):
        corr = corr_of(x)
        top_k = min(15, len(corr))
        top_idx = np.argpartition(-corr, top_k - 1)[:top_k]
        top = corr[top_idx]
        smax = top.max()
        cl = np.clip(beta * (top - smax), -50.0, 50.0)
        ex = np.exp(cl)
        w_top = ex / np.sum(ex)
        wc = w_top * (2.0 / N)
        
        g = np.zeros(N)
        for idx_pos in range(len(top_idx)):
            k = top_idx[idx_pos]
            wk = wc[idx_pos]
            if wk < 1e-15:
                continue
            off = k - (N - 1)
            if off >= 0:
                np_ = N - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = N - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    bounds = [(0.0, 1.0)] * N
    cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - TGT,
                'jac': lambda x: np.ones(N)}
    
    alphas = [50, 200, 500, 1000, 3000, 10000, 30000, 100000, 300000]
    
    for beta in alphas:
        if remaining() < 10:
            break
        try:
            res = minimize(obj_smooth, h, jac=lambda x: grad_smooth(x, beta),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-10})
            if res.success or res.fun < obj_smooth(h, beta):
                h_new = project_feasible(res.x, N)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def top_shifts_gradient(h, n, max_iters=500):
    h = h.copy()
    best_c5 = compute_c5(h)
    TGT = n / 2.0
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_k = min(8, len(corr))
        top_idx = np.argsort(-corr)[:top_k]
        
        g_top = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_top[off:off + np_] += (1.0 - h[:np_])
                g_top[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_top[:np_] += (1.0 - h[ao:ao + np_])
                g_top[ao:ao + np_] -= h[:np_]
        
        # Add regularization toward balance
        g_balance = h - TGT / n
        d = -g_top + 0.1 * g_balance
        
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.015, 0.04, 0.008, 0.12]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try negative direction
            for step in [0.005, 0.01, 0.02]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: Binary SA at parent resolution (main workhorse)
    n = parent_n
    h_bin = (best_h > 0.5).astype(float)
    h_bin = ensure_count(h_bin, n)
    
    sa_time = min(350, int(remaining() * 0.3))
    if sa_time > 30:
        h_b1, c5_b1 = sa_binary_search(h_bin, n, sa_time, rng,
                                        t_init=10.0, cooling=0.9996, t_min=0.003,
                                        n_top_shifts=10, n_contrib=25, n_random=60)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
    
    # Phase 2: Second SA pass with different temperature schedule
    if remaining() > 60:
        h_sa2_time = min(150, int(remaining() * 0.15))
        if h_sa2_time > 20:
            h_b2, c5_b2 = sa_binary_search(best_h, best_n, h_sa2_time, rng,
                                            t_init=15.0, cooling=0.9993, t_min=0.002,
                                            n_top_shifts=12, n_contrib=30, n_random=40,
                                            restart_every=200)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
    
    # Phase 3: Try alternative initializations at same resolution
    inits = []
    tm = thue_morse(n)
    inits.append(tm)
    lg = legendre_seq(n)
    inits.append(lg)
    ms = m_sequence(n)
    inits.append(ms)
    
    for init_h in inits:
        if remaining() < 30:
            break
        sa_t = min(80, int(remaining() * 0.07))
        if sa_t > 10:
            h_t, c5_t = sa_binary_search(init_h, n, sa_t, rng,
                                          t_init=8.0, cooling=0.9997, t_min=0.005,
                                          n_top_shifts=8, n_contrib=20, n_random=40)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n
    
    # Phase 4: Multi-resolution binary search
    resolutions = []
    for delta in [-32, -16, 16, 32, 64, 96, 128]:
        rn = best_n + delta
        if 192 <= rn <= 800 and rn % 2 == 0 and rn != best_n:
            resolutions.append(rn)
    # Also try round numbers
    for rn in [256, 320, 384, 448, 512, 576, 640, 704, 768]:
        if 192 <= rn <= 800 and rn % 2 == 0 and rn not in resolutions:
            resolutions.append(rn)
    
    resolutions.sort(key=lambda x: abs(x - best_n))
    
    for n_t in resolutions[:5]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        h_bin_t = ensure_count(h_bin_t, n_t)
        
        sa_t = min(100, int(remaining() * 0.08))
        if sa_t > 15:
            h_t, c5_t = sa_binary_search(h_bin_t, n_t, sa_t, rng,
                                          t_init=8.0, cooling=0.9997, t_min=0.005,
                                          n_top_shifts=10, n_contrib=20, n_random=50)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t
    
    # Phase 5: Continuous refinement at best resolution
    n = best_n
    h = best_h.copy()
    cont_time = min(200, int(remaining() * 0.2))
    if cont_time > 20:
        h_nr, c5_nr = continuous_refine(h, n, cont_time, rng)
        if c5_nr < best_c5:
            best_c5 = c5_nr
            best_h = h_nr.copy()

    if remaining() > 30:
        h_eq, c5_eq = top_shifts_gradient(h, n, max_iters=1000)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Phase 7: Another continuous pass with the refined binary/continuous hybrid
    h = best_h.copy()
    if remaining() > 30:
        h_nr2, c5_nr2 = continuous_refine(h, n, min(100, int(remaining() * 0.08)), rng)
        if c5_nr2 < best_c5:
            best_c5 = c5_nr2
            best_h = h_nr2.copy()
    
    # Phase 8: Try going to higher resolution with the best solution
    if remaining() > 80:
        n_fine = best_n + 64
        if n_fine <= 900 and n_fine % 2 == 0:
            h_fine = interp_to(best_h, best_n, n_fine)
            h_fine_bin = (h_fine > 0.5).astype(float)

                                               t_init=5.0, cooling=0.9998, t_min=0.01,

                if c5_fb < best_c5:

                    best_h = h_fb.copy()
                    best_n = n_fine
            
            if remaining() > 40:
                h_fx, c5_fx = continuous_refine(best_h, best_n,
                                                min(60, int(remaining() * 0.04)), rng)
                if c5_fx < best_c5:
                    best_c5 = c5_fx
                    best_h = h_fx.copy()
    
    # Phase 9: Final polish
    n = best_n
    h = best_h.copy()
    if remaining() > 15:
        h_fin, c5_fin = top_shifts_gradient(h, n, max_iters=500)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    if remaining() > 15:
        h_fin2, c5_fin2 = continuous_refine(best_h, best_n,
                                            min(50, int(remaining() * 0.03)), rng)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
