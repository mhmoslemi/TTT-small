            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

def _sc(x):
    return np.sum(x) - _TGT

def _scj(x):
    return np.ones(_N)

def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def binary_exhaustive_search(h_bin, n, rng, max_time=90):
    """Exhaustive single-swap local search with 2-swap escape."""
    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    best_c5 = float(np.max(compute_corr(h)) * dx)
    best_h = h.copy()
    start = time.time()
    stall = 0
    
    while time.time() - start < max_time and remaining() > 20:
        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0.0
        best_move = None
        # Exhaustive: try every (one_i, zero_j) pair
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                nm = float(np.max(compute_corr(h)))
                h[i] = 1.0
                h[j] = 0.0
                d = nm - cur_max
                if d < best_delta - 1e-14:
                    best_delta = d
                    best_move = (i, j)
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            stall = 0
            nc5 = float(np.max(compute_corr(h)) * dx)
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            stall += 1
            if stall >= 8:
                # 2-swap escape: try many random 2-bit moves
                found = False
                for _ in range(300):
                    idx_o = rng.choice(ones, 2, replace=False)
                    idx_z = rng.choice(zeros, 2, replace=False)
                    i1, i2 = int(idx_o[0]), int(idx_o[1])
                    j1, j2 = int(idx_z[0]), int(idx_z[1])
                    h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                    nm = float(np.max(compute_corr(h)))
                    h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
                    if nm < cur_max - 1e-14:
                        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                        stall = 0
                        nc5 = float(np.max(compute_corr(h)) * dx)
                        if nc5 < best_c5:
                            best_c5 = nc5
                            best_h = h.copy()
                        found = True
                        break
                if not found and stall > 20:
                    break
            elif stall >= 3:
                # Small random perturbation
                n_flips = rng.randint(2, 5)
                for _ in range(n_flips):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = int(rng.choice(ones))
                    j = int(rng.choice(zeros))
                    h[i] = 0; h[j] = 1
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
    
    return h, best_c5

def genetic_binary(h_init, n, rng, max_time=60):
    """Genetic algorithm on binary vectors with n/2 ones."""
    dx = 2.0 / n
    n_half = n // 2
    pop_size = 40
    
    def make_pop():
        pop = np.zeros((pop_size, n))
        # Seed with initial and variations
        pop[0] = h_init
        for p in range(1, pop_size):
            if p < 6:
                pop[p] = h_init.copy()
                # Mutate
                n_mut = rng.randint(3, 12)
                ones = np.where(pop[p] == 1)[0]
                zeros = np.where(pop[p] == 0)[0]
                for _ in range(n_mut):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    oi = int(rng.choice(ones))
                    zi = int(rng.choice(zeros))
                    pop[p, oi] = 0
                    pop[p, zi] = 1
                    ones = np.where(pop[p] == 1)[0]
                    zeros = np.where(pop[p] == 0)[0]
            else:
                # Random balanced binary
                perm = rng.permutation(n)
                pop[p, perm[:n_half]] = 1.0
        return pop
    
    pop = make_pop()
    fitness = np.array([float(np.max(compute_corr(pop[p]))) for p in range(pop_size)])
    best_fitness = fitness.min()
    best_individual = pop[np.argmin(fitness)].copy()
    start = time.time()
    gen = 0
    
    while time.time() - start < max_time and remaining() > 30:
        gen += 1
        # Tournament selection
        def select():
            for _ in range(20):
                a, b = rng.randint(0, pop_size), rng.randint(0, pop_size)
                if a == b:
                    continue
                return a if fitness[a] < fitness[b] else b
            return rng.randint(0, pop_size)
        
        new_pop = np.zeros_like(pop)
        new_pop[0] = best_individual  # Elitism
        for p in range(1, pop_size):
            pa = select()
            pb = select()
            # Uniform crossover
            mask = rng.rand(n) < 0.5
            child = np.where(mask, pop[pa], pop[pb])
            # Fix sum to n_half
            s = int(np.sum(child))
            if s != n_half:
                if s > n_half:
                    ones = np.where(child == 1)[0]
                    rem = rng.choice(ones, s - n_half, replace=False)
                    child[rem] = 0
                else:
                    zeros = np.where(child == 0)[0]
                    add = rng.choice(zeros, n_half - s, replace=False)
                    child[add] = 1
            # Mutation
            if rng.rand() < 0.7:
                n_mut = rng.randint(1, 4)
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                for _ in range(n_mut):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    oi = int(rng.choice(ones))
                    zi = int(rng.choice(zeros))
                    child[oi] = 0
                    child[zi] = 1
                    ones = np.where(child == 1)[0]
                    zeros = np.where(child == 0)[0]
            new_pop[p] = child
        
        pop = new_pop
        fitness = np.array([float(np.max(compute_corr(pop[p]))) for p in range(pop_size)])
        if fitness.min() < best_fitness:
            best_fitness = fitness.min()
            best_individual = pop[np.argmin(fitness)].copy()
        # Local search on best
        bi = pop[np.argmin(fitness)].copy()
        # Quick local search: 20 random swaps
        for _ in range(20):
            corr_b = compute_corr(bi)
            cm = float(np.max(corr_b))
            ones_b = np.where(bi == 1)[0]
            zeros_b = np.where(bi == 0)[0]
            if len(ones_b) == 0 or len(zeros_b) == 0:
                break
            best_d = 0.0
            best_mv = None
            for _ in range(30):
                oi = int(rng.choice(ones_b))
                zi = int(rng.choice(zeros_b))
                bi[oi] = 0; bi[zi] = 1
                nm = float(np.max(compute_corr(bi)))
                bi[oi] = 1; bi[zi] = 0
                d = nm - cm
                if d < best_d - 1e-14:
                    best_d = d
                    best_mv = (oi, zi)
            if best_mv is not None:
                bi[best_mv[0]] = 0; bi[best_mv[1]] = 1
                npf = float(np.max(compute_corr(bi)))
                if npf < best_fitness:
                    best_fitness = npf
                    best_individual = bi.copy()
                    pop[np.argmin(fitness)] = bi
                    fitness[np.argmin(fitness)] = npf
    
    return best_individual, best_fitness * dx

def top_shifts_focus(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-10:
            break
        # Subgradient direction: reduce top, increase bottom
        g = np.zeros(n)
        n_tot = 2 * n - 1
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(idx)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.01, 0.03, 0.05, 0.02, 0.08, 0.005, 0.04]:
            ht = project_feasible(h + step * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try negative direction too
            for step in [0.01, 0.03, 0.05]:
                ht = project_feasible(h - step * g, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5

def subgradient_max(h, n, n_iter=200):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.03
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 0.3
        top_mask = corr >= threshold
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]
        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(top_idx)
        g -= np.mean(g)
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.92
            if step < 0.001:
                break
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP to get a continuous baseline
    for beta in [100, 500, 2000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Exhaustive binary local search at n
    if remaining() > 60:
        h_bin = binarize(h, n)
        h_b, c5_b = binary_exhaustive_search(h_bin, n, rng, max_time=min(90, int(remaining() * 0.25)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()
            # Continuous refinement from binary
            h_c = slsqp_smooth(h_b, n, 5000, 400)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                h = h_c.copy()
        # Also try from parent binary directly
        h_bin2 = binarize(best_h, n)
        if not np.array_equal(h_bin2, h_bin):
            h_b2, c5_b2 = binary_exhaustive_search(h_bin2, n, rng, max_time=30)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                h = h_b2.copy()

    # Phase 3: Genetic algorithm for broader exploration
    if remaining() > 80:
        h_seed = binarize(h, n)
        h_ga, c5_ga = genetic_binary(h_seed, n, rng, max_time=min(50, int(remaining() * 0.2)))
        if c5_ga < best_c5:
            best_c5 = c5_ga
            best_h = h_ga.copy()
            # Refine
            h_ga_r = slsqp_smooth(h_ga, n, 5000, 400)
            c5_ga_r = compute_c5(h_ga_r)
            if c5_ga_r < best_c5:
                best_c5 = c5_ga_r
                best_h = h_ga_r.copy()
                h = h_ga_r.copy()
        # Another GA from Thue-Morse
        h_tm = thue_morse(n, rng)
        h_ga2, c5_ga2 = genetic_binary(h_tm, n, rng, max_time=min(40, int(remaining() * 0.15)))
        if c5_ga2 < best_c5:
            best_c5 = c5_ga2
            best_h = h_ga2.copy()
            h = h_ga2.copy()

    # Phase 4: More SLSQP refinement with higher betas
    for beta in [5000, 15000, 40000, 100000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Top-shifts equalization
    for tk in [3, 5, 8]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Subgradient polish
    if remaining() > 20:
        h_sg, c5_sg = subgradient_max(h, n, n_iter=200)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 7: Resolution sweep
    if remaining() > 100:
        n_candidates = []
        for nt in [380, 386, 392, 400, 448, 512, 576, 640, 768]:
            if 128 <= nt <= 1024 and nt % 2 == 0 and nt != n:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - n))
        
        for n_t in n_candidates[:6]:
            if remaining() < 30:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t
            
            for beta in [2000, 10000]:
                if remaining() < 10:
                    break
                h_t = slsqp_smooth(h_t, n_t, beta, 300)
            c5_t = compute_c5(h_t)
            
            # Binary search at this resolution
            if remaining() > 20 and n_t <= 512:
                h_t_bin = binarize(h_t, n_t)
                h_tb, c5_tb = binary_exhaustive_search(h_t_bin, n_t, rng, max_time=20)
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb.copy()
                    # Quick continuous refine
                    h_t = slsqp_smooth(h_t, n_t, 10000, 200)
                    c5_t = compute_c5(h_t)
            
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    
    for beta in [20000, 50000, 150000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final top-shifts
    if remaining() > 10:
        for tk in [3, 5]:
            h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=500)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Final binary search if close
    if remaining() > 25:
        h_bin = binarize(h, n)
        h_b, c5_b = binary_exhaustive_search(h_bin, n, rng, max_time=min(30, int(remaining() * 0.4)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = slsqp_smooth(h_b, n, 10000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Final subgradient
    if remaining() > 8:
        h_sg, c5_sg = subgradient_max(best_h, best_n, n_iter=100)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
