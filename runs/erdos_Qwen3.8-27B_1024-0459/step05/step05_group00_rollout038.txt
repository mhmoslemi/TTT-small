    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def top_k_gradient(h, n, K=20, beta=50.0):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, k, n)
    return g, np.max(corr)

= min_h Σ_k λ_k Σ_i h_i - Σ_k λ_k Σ_i h_i h_{i+off_k}
= min_h c^T h - h^T A h
where c is a constant vector (depending on λ) and A is a matrix (depending on λ)

The Hessian of the objective is -2A (where A is symmetric). If A is positive semi-definite, the objective is concave and we're minimizing a concave function over a polytope - the minimum is at a vertex. This means bang-bang solutions (h_i ∈ {0, 1}).

Let me check: A_{ij} = (1/2) Σ_k λ_k * [1 if |i-j| corresponds to off_k]. Since all λ_k ≥ 0, A is a weighted sum of "band matrices" (matrices with 1s on specific diagonals). These are typically positive semi-definite if the weights are "spread out" enough.

Actually, for a single lag with weight 1, the matrix is:
A_{ij} = 1/2 if j = i + off or j = i - off (within bounds), 0 otherwise.
This is a (symmetric) adjacency matrix of a path-like graph, which is not necessarily PSD.

For off = 1: A is the adjacency matrix of a path graph P_n, which has eigenvalues 2cos(πk/(n+1)) for k=0,...,n-1. These can be negative! So A is indefinite.

This means the Lagrangian dual approach doesn't directly give a convex problem.

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)

            break

        else:

        h = np.clip(h, 0.0, 1.0)

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def active_set_slsqp(h, n, K=10, maxiter=300, lambda_tol=1e-4):

    dx = 2.0 / n
    h = h.copy()

        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        
        # Build objective: minimize max of top K, approximated by weighted sum
        # Weights are chosen to push the highest ones down more
        c_max = top_vals[0]
        c_min_top = top_vals[-1]
        if c_max - c_min_top < 1e-10:
            weights = np.ones(K) / K
        else:
            # Exponential weighting: higher values get more weight
            t = (top_vals - c_min_top) / (c_max - c_min_top)  # in [0,1]
            weights = np.exp(10 * t)
            weights /= weights.sum()

            corr_x = compute_corr(x)
            return np.dot(weights, corr_x[top_idx]) * dx

            for i, k in enumerate(top_idx):
                if weights[i] > 1e-15:

            return g * dx

            return np.sum(x) - n / 2.0

            return np.ones(n)

            # g[i] > 0: increasing h[i] increases corr[k]

def full_gradient_weighted(h, n, weights):
    """Compute weighted gradient: sum_k weights[k] * grad_corr_k"""
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return g


def smooth_max_slsqp(h, n, beta, maxiter=500):
    dx = 2.0 / n

    def sm_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -100.0, 100.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def sm_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -100.0, 100.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = full_gradient_weighted(x, n, w * dx)
        return g

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h


def active_set_slsqp(h, n, K=10, outer_iters=8, maxiter_per=200):
    """Iteratively equalize top-K correlation values using reweighted SLSQP."""
    dx = 2.0 / n
    h = h.copy()

    for outer in range(outer_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 10:
            break

        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]

        c_max = top_vals[0]
        c_min_top = top_vals[-1]
        if c_max - c_min_top < 1e-12:
            weights = np.ones(K) / K
        else:
            t = (top_vals - c_min_top) / (c_max - c_min_top)
            weights = np.exp(15.0 * t)
            weights /= weights.sum()

        full_weights = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            full_weights[k] = weights[i]

        def obj(x):
            corr_x = compute_corr(x)
            return np.dot(full_weights, corr_x) * dx

        def grad_obj(x):
            return full_gradient_weighted(x, n, full_weights * dx)

        def sc(x):
            return np.sum(x) - n / 2.0

        def scj(x):
            return np.ones(n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]

        try:
            res = minimize(obj, h, jac=grad_obj, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per, 'ftol': 1e-18, 'iprint': 0})
            if res.x is not None:
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new)
                c_old = compute_c5(h)
                if c_new <= c_old + 1e-12:
                    h = h_new
                else:
                    h = project_feasible(0.5 * h + 0.5 * res.x, n)
            else:
                break
        except Exception:
            break

    return h


def subgradient_descent(h, n, time_limit, rng):
    global _T0, _BUDGET
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        progress = min(1.0, (time.time() - start) / max(time_limit, 1.0))
        K = 30 if progress < 0.2 else (20 if progress < 0.5 else 15 if progress < 0.8 else 8)
        beta_g = 15.0 if progress < 0.2 else (30.0 if progress < 0.5 else 60.0 if progress < 0.8 else 100.0)

        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta_g * (c_top - c_top.max()))
        s = weights.sum()
        if s < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= s

        g = np.zeros(n)
        for i, k in enumerate(idx):
            if weights[i] > 1e-12:
                g += weights[i] * grad_corr_k(h, k, n)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        base_step = 0.08 / (1.0 + 0.02 * it)
        step = min(base_step, 0.5)

        h_new = project_feasible(h - step * g / gn, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h - step * factor * g / gn, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break

        it += 1

    return best_h, best_c5


def coordinate_search(h, n, time_limit, rng):
    """Targeted pairwise coordinate search to reduce max correlation."""
    start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        corr = compute_corr(h)
        top_k = np.argsort(-corr)[:8]
        improved = False

        for k in top_k:
            g = grad_corr_k(h, k, n)

            # Find best source (decrease) and destination (increase)
            # g[i] > 0 means h[i] contributes positively to this correlation
            # We want to decrease h where g is large positive, increase where g is large negative

            # Try the extreme swap
            candidates_src = np.argsort(-g)[:10]
            candidates_dst = np.argsort(g)[:10]

            found = False
            for i_src in candidates_src:
                for i_dst in candidates_dst:
                    if i_src == i_dst:
                        continue
                    if h[i_src] < 1e-10 or h[i_dst] > 1.0 - 1e-10:
                        continue
                    max_delta = min(h[i_src], 1.0 - h[i_dst])
                    for frac in [0.5, 0.1, 0.05, 0.01, 0.2, 0.02]:
                        delta = frac * max_delta
                        if delta < 1e-8:
                            continue
                        h_try = h.copy()
                        h_try[i_dst] += delta
                        h_try[i_src] -= delta
                        h_try = np.clip(h_try, 0, 1)
                        c_try = compute_c5(h_try)
                        if c_try < best_c5 - 1e-14:
                            h = h_try
                            best_c5 = c_try
                            improved = True
                            found = True
                            break
                    if found:
                        break
                if found:
                    break

        if not improved:
            # Random perturbations
            n_random = 200
            for _ in range(n_random):
                if time.time() - start > time_limit:
                    break
                i, j = rng.choice(n, 2, replace=False)
                i_src, i_dst = (i, j) if h[i] > h[j] else (j, i)
                max_delta = min(h[i_src], 1.0 - h[i_dst])
                if max_delta < 1e-5:
                    continue
                delta = rng.uniform(0.001, 0.03) * max_delta
                h_try = h.copy()
                h_try[i_dst] += delta
                h_try[i_src] -= delta
                h_try = np.clip(h_try, 0, 1)
                c_try = compute_c5(h_try)
                if c_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c_try
                    improved = True

            if not improved:
                # Try small uniform perturbations
                for _ in range(50):
                    if time.time() - start > time_limit:
                        break
                    pert = rng.uniform(-0.01, 0.01, n)
                    h_try = project_feasible(h + pert, n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c_try
                        improved = True

            if not improved:
                break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def equalize_kkt(h, n, K=12, max_iters=100):
    """Try to satisfy KKT conditions: weighted gradient of active corrs aligns with 1-vector."""
    dx = 2.0 / n
    h = h.copy()

    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]

        # Check if top K are already equalized
        spread = top_vals[0] - top_vals[-1]
        if spread < 1e-10:
            break

        # Weights to push toward equalization
        c_center = np.mean(top_vals)
        weights = np.zeros(2 * n - 1)
        for i, k in enumerate(top_idx):
            # Weight proportional to how far above center
            w_i = max(0, (top_vals[i] - c_center) / spread + 0.1)
            weights[k] = w_i

        # Also add small weight to next few (to prevent them from exceeding)
        next_idx = np.argsort(-corr)[K:K + 5]
        for k in next_idx:
            if corr[k] > c_center - 0.01 * spread:
                weights[k] = 0.05

        total_w = weights.sum()
        if total_w < 1e-15:
            break
        weights /= total_w

        g = full_gradient_weighted(h, n, weights)
        # Remove component along 1-vector (to respect sum constraint)
        g_mean = np.mean(g)
        g_proj = g - g_mean

        gn = np.linalg.norm(g_proj)
        if gn < 1e-14:
            break

        # Line search
        improved = False
        for step in [0.05, 0.1, 0.02, 0.01, 0.08, 0.03, 0.005, 0.2]:
            h_try = project_feasible(h - step * g_proj / gn, n)
            c_try = compute_c5(h_try)
            c_cur = compute_c5(h)
            if c_try < c_cur - 1e-15:
                h = h_try
                improved = True
                break

        if not improved:
            break

    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick subgradient descent
    if remaining() > 30:
        h_sg, c5_sg = subgradient_descent(h, n, min(35.0, remaining() * 0.1), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: SLSQP smooth max with increasing beta
    for beta in [300, 1000, 3000, 10000]:
        if remaining() < 10:
            break
        h_new = smooth_max_slsqp(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Active set equalization
    if remaining() > 20:
        h_new = active_set_slsqp(h, n, K=10, outer_iters=6, maxiter_per=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: KKT equalization
    if remaining() > 15:
        h_new = equalize_kkt(h, n, K=12, max_iters=80)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: More SLSQP with higher beta
    for beta in [20000, 50000, 100000]:
        if remaining() < 8:
            break
        h_new = smooth_max_slsqp(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Coordinate search
    if remaining() > 25:
        h_cs, c5_cs = coordinate_search(h, n, min(50.0, remaining() * 0.2), rng)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs.copy()
            h = h_cs.copy()

    # Phase 7: Active set again
    if remaining() > 15:
        h_new = active_set_slsqp(h, n, K=15, outer_iters=5, maxiter_per=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Try resolution 768
    if remaining() > 60:
        n_t = 768
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_cur = h_feas
        c5_cur = c5_int
        local_best = min(best_c5, c5_int)

        if remaining() > 20:
            h_sg2, c5_sg2 = subgradient_descent(h_cur, n_t, min(15.0, remaining() * 0.2), rng)
            if c5_sg2 < c5_cur:
                h_cur = h_sg2
                c5_cur = c5_sg2

        for beta in [3000, 10000, 30000]:
            if remaining() < 8:
                break
            h_new = smooth_max_slsqp(h_cur, n_t, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h_cur = h_new
                c5_cur = c5_new

        if remaining() > 15:
            h_new = active_set_slsqp(h_cur, n_t, K=10, outer_iters=4, maxiter_per=150)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h_cur = h_new
                c5_cur = c5_new

        if remaining() > 15:
            h_cur, c5_cur = coordinate_search(h_cur, n_t, min(20.0, remaining() * 0.3), rng)

        if c5_cur < local_best:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_t

    # Phase 9: Cycle at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(3):
        if remaining() < 20:
            break

        # Subgradient
        if remaining() > 15:
            h, c5_tmp = subgradient_descent(h, n, min(20.0, remaining() * 0.2), rng)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

        # SLSQP
        for beta in [8000, 25000]:
            if remaining() < 8:
                break
            h_new = smooth_max_slsqp(h, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Active set
        if remaining() > 12:
            h_new = active_set_slsqp(h, n, K=12, outer_iters=4, maxiter_per=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Coordinate
        if remaining() > 15:
            h_cs, c5_cs = coordinate_search(h, n, min(25.0, remaining() * 0.2), rng)
            if c5_cs < best_c5:
                best_c5 = c5_cs
                best_h = h_cs.copy()
                h = h_cs.copy()

    # Phase 10: Final intensive polish
    h = best_h.copy()
    n = best_n

    # Very high beta SLSQP
    for beta in [100000, 200000]:
        if remaining() < 5:
            break
        h_new = smooth_max_slsqp(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final active set with narrow K
    if remaining() > 10:
        h_new = active_set_slsqp(h, n, K=8, outer_iters=6, maxiter_per=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final coordinate search
    if remaining() > 10:
        h_cs, c5_cs = coordinate_search(best_h, best_n, min(30.0, remaining() * 0.5), rng)
        if c5_cs < best_c5:
            best_c5 = c5_cs
            best_h = h_cs.copy()

    # KKT final
    if remaining() > 8:
        h_new = equalize_kkt(best_h, best_n, K=10, max_iters=50)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
