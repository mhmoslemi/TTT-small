            h[mask] += diff / np.sum(mask)
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        for j in range(n):
            if 1e-14 < h[j] < 1.0 - 1e-14:
                h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                break
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def coord_descent(h, n, max_sweeps=100):
    """Vectorized coordinate descent."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)
            
            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0
            
            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)
            
            f_old = np.max(best_corr)
            
            # Golden section on [0, 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            def _eval(t):
                mid = base + A * t
                mid[n - 1] = base[n - 1] + t - t * t
                return np.max(mid)
            
            f1 = _eval(c1)
            f2 = _eval(c2)
            for _ in range(50):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _eval(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = _eval(c2)
            t_best = (lo + hi) / 2.0
            f_best = _eval(t_best)
            
            if f_best < f_old - 1e-10:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                improved = True
        
        if not improved:
            break
    
    return best, np.max(best_corr) * dx
