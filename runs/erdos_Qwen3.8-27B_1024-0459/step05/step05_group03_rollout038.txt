```python
def sa_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)
    h[ones] = 1.0
    return h

def compute_overlap_binary(h, n):
    """Compute overlap[k] for k = -(n-1), ..., (n-1)."""
    # overlap at lag k = sum_i h[i]*(1-h[i+k]) for valid i
    # For binary h, this is an integer
    overlap = np.zeros(2*n - 1, dtype=np.int32)
    # For positive lags k = 0, ..., n-1:
    # overlap[n-1+k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    # For negative lags:
    # overlap[n-1-k] = sum_{i=k}^{n-1} h[i]*(1-h[i-k]) = sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
    
    # Use the fact that for binary h: h[i]*(1-h[i+k]) = h[i] - h[i]*h[i+k]
    # So overlap[n-1+k] = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i]*h[i+k]
    
    # Or just compute directly with numpy
    for k in range(n):
        overlap[n-1+k] = int(np.sum(h[:n-k] * (1-h[k:])))
    for k in range(1, n):
        overlap[n-1-k] = int(np.sum(h[k:] * (1-h[:n-k])))
    # overlap[n-1] = 0 for binary (h[i]*(1-h[i]) = 0)
    return overlap
```
