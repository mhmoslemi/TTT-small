    target = n / 2.0
    for _ in range(300):

        g[offset:offset + np_] += (1.0 - h[:np_])

        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])

    return g

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            new_c5 = compute_c5(h)
            delta = new_c5 - curr_c5
            
            if delta < 0 or (T > 1e-15 and rng.rand() < np.exp(-delta / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve_since = 0
                else:
                    no_improve_since += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                no_improve_since += 1
        
        # Check if we should restart
        if no_improve_since > 50000:
            # Perturb: flip 5 random pairs
            h = best_h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(5):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
            h = proj(h, n)
            no_improve_since = 0
    
    return best_h, best_c5

def binary_local_search(h, n, max_iters=5000, top_k=5):
    """Local search for binary h."""
    h_bin = (h > 0.5).astype(float).copy()
    n_ones = int(np.sum(h_bin))
    if n_ones != n // 2:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n//2]] = 1.0
    
    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        
        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]
        
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try top candidates
        score_one = -g[ones]
        score_zero = g[zeros]
        
        best_delta = 0
        best_swap = None
        n_try = min(15, len(ones))
        top_ones = ones[np.argsort(-score_one)[:n_try]]
        n_try_z = min(15, len(zeros))
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]
        
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
        
        # Random swaps
        for _ in range(8):
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
            if i == j:
                continue
            h_bin[i] = 0
            h_bin[j] = 1
            nc = compute_c5(h_bin)
            h_bin[i] = 1
            h_bin[j] = 0
            delta = nc - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)
        
        if best_swap is None or best_delta >= 0:
            break
        
        i, j = best_swap
        h_bin[i] = 0
        h_bin[j] = 1
        cur_c5 = compute_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
