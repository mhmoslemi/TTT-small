```python
def binary_greedy_ls(h, n, max_iters=10000, time_limit=100):
    """Greedy local search on binary h to minimize max correlation."""
    h = h.copy()
    start_time = time.time()
    best_c5 = compute_c5(h)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        c = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(c)
        max_val = c[max_idx]
        L_star = max_idx - (n - 1)  # the lag
        
        # Find best swap
        best_new_max = max_val
        best_pair = None
        
        for p in ones:
            # get(p+L_star), get(p-L_star)
            p_plus = p + L_star
            p_minus = p - L_star
            hp_plus = h[p_plus] if 0 <= p_plus < n else 0
            hp_minus = h[p_minus] if 0 <= p_minus < n else 0
            
            for q in zeros:
                q_plus = q + L_star
                q_minus = q - L_star
                hq_plus = h[q_plus] if 0 <= q_plus < n else 0
                hq_minus = h[q_minus] if 0 <= q_minus < n else 0
                
                delta = hp_plus - hq_plus + hq_minus - hp_minus
                if delta < 0:
                    # Compute new max
                    # c_new = c + delta_vector
                    # But delta depends on lag, so we need to compute for all lags
                    # This is O(n) per candidate, too slow
                    # Instead, just check the top few shifts
                    new_at_star = max_val + delta
                    if new_at_star >= best_new_max:
                        continue
                    # Check a few other top shifts
                    top_shifts = np.argsort(-c)[:5]
                    ok = True
                    for ts in top_shifts:
                        if ts == max_idx:
                            continue
                        ts_lag = ts - (n-1)
                        # Compute delta at this lag
                        pp2 = p + ts_lag
                        pm2 = p - ts_lag
                        qp2 = q + ts_lag
                        qm2 = q - ts_lag
                        hpp = h[pp2] if 0 <= pp2 < n else 0
                        hpm = h[pm2] if 0 <= pm2 < n else 0
                        hqp = h[qp2] if 0 <= qp2 < n else 0
                        hqm = h[qm2] if 0 <= qm2 < n else 0
                        d2 = hpp - hqp + hqm - hpm
                        if c[ts] + d2 > new_at_star and c[ts] + d2 > best_new_max:
                            ok = False
                            break
                    if ok:
                        # Full verification
                        h[p] = 0
                        h[q] = 1
                        new_max = np.max(np.correlate(h, 1.0-h, mode='full'))
                        h[p] = 1
                        h[q] = 0
                        if new_max < best_new_max:
                            best_new_max = new_max
                            best_pair = (p, q)
        
        if best_pair is None:
            break
        
        p, q = best_pair
        h[p] = 0
        h[q] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        new_c5 = best_new_max * 2.0 / n
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return h, best_c5
```

```python
def binary_sa(h, n, T_start=1.0, T_end=0.01, max_iters=50000, time_limit=60):
    """Simulated annealing on binary h."""
    h = h.copy()
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5 = best_c5
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        # Random swap
        pi = np.random.choice(len(ones))
        qi = np.random.choice(len(zeros))
        p = ones[pi]
        q = zeros[qi]
        
        h[p] = 0
        h[q] = 1
        new_c5 = compute_c5(h)
        
        if new_c5 < c5 or np.random.random() < np.exp(-(new_c5 - c5) / T):
            c5 = new_c5
            # Update ones and zeros
            ones[pi] = q
            zeros[qi] = p
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[p] = 1
            h[q] = 0
    
    return best_h, best_c5
```

```python
def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break

        if free.sum() > 0:

        else:
```

       = h[p + m - (n-1)] - h[q + m - (n-1)] + h[q - m + (n-1)] - h[p - m + (n-1)]

So Δc[m] = h[m + (p-n+1)] - h[m + (q-n+1)] + h[-m + (n-1+q)] - h[-m + (n-1-p)]

This is a combination of h at shifted positions. If I precompute the full correlation and then add this Δc vector, I get the new correlation without recomputing from scratch.

The Δc vector can be computed in O(n) for a given (p,q):
```python
def delta_corr(h, p, q, n):
    """Compute the change in the full correlation vector after swapping h[p](1->0) and h[q](0->1)."""
    L_range = np.arange(-(n-1), n)  # all possible lags
    # get(x) = h[x] if 0<=x<n else 0
    def _get(pos):
        valid = (pos >= 0) & (pos < n)
        result = np.zeros(len(pos))
        result[valid] = h[pos[valid]]
        return result
    
    dc = _get(p + L_range) - _get(q + L_range) + _get(q - L_range) - _get(p - L_range)
    return dc
```

This is O(n) per swap, and we can then compute new_max = np.max(c + dc) in O(n). So the full verification is O(n) instead of O(n²)!

This is a huge speedup. Now each candidate evaluation is O(n) = O(384) instead of O(n²) = O(147K). So 7500 candidates × 384 operations = 2.9M operations per iteration, which is about 3ms in numpy. We can do 1000+ iterations per second!
