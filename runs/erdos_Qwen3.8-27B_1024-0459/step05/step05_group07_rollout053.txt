                    # Solve base[ia] + A[ia]*t (+ t-t^2 if ia==mid) = base[ib] + A[ib]*t (+ t-t^2 if ib==mid)
                    if ia != mid and ib != mid:
                        if abs(A[ia] - A[ib]) > 1e-14:
                            t_eq = (base[ib] - base[ia]) / (A[ia] - A[ib])
                            if 0 <= t_eq <= 1:
                                candidates.append(t_eq)
                    elif ia == mid and ib != mid:
                        # base[mid] + t - t^2 = base[ib] + A[ib]*t
                        # t^2 - (1-A[ib])*t + (base[ib]-base[mid]) = 0
                        a_q = 1.0
                        b_q = -(1.0 - A[ib])
                        c_q = base[ib] - base[mid]
                        disc = b_q*b_q - 4*a_q*c_q
                        if disc >= 0:
                            t1 = (-b_q + np.sqrt(disc)) / 2
                            t2 = (-b_q - np.sqrt(disc)) / 2
                            for t_eq in [t1, t2]:
                                if 0 <= t_eq <= 1:
                                    candidates.append(t_eq)
                    elif ib == mid and ia != mid:
                        # base[ia] + A[ia]*t = base[mid] + t - t^2
                        # t^2 + (A[ia]-1)*t + (base[ia]-base[mid]) = 0
                        a_q = 1.0
                        b_q = A[ia] - 1.0
                        c_q = base[ia] - base[mid]
                        disc = b_q*b_q - 4*a_q*c_q
                        if disc >= 0:
                            t1 = (-b_q + np.sqrt(disc)) / 2
                            t2 = (-b_q - np.sqrt(disc)) / 2
                            for t_eq in [t1, t2]:
                                if 0 <= t_eq <= 1:
                                    candidates.append(t_eq)
            
            best_t = old
            best_f = f_old
            for t in candidates:
                t = max(0.0, min(1.0, t))
                vals = base + A * t
                vals[mid] = base[mid] + t - t * t
                f = np.max(vals)
                if f < best_f - 1e-13:
                    best_f = f
                    best_t = t
            
            # Golden section refinement around best_t
            if best_t != old:
                lo = max(0.0, best_t - 0.15)
                hi = min(1.0, best_t + 0.15)
                if hi - lo > 1e-10:
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    v1 = base.copy()
                    v1[mid] = base[mid]
                    v1 += A * c1
                    v1[mid] = base[mid] + c1 - c1*c1
                    f1 = np.max(v1)
                    v2 = base.copy()
                    v2 += A * c2
                    v2[mid] = base[mid] + c2 - c2*c2
                    f2 = np.max(v2)
                    for _ in range(20):
                        if f1 < f2:
                            hi, c2, f2 = c2, c1, f1
                            c1 = hi - gr * (hi - lo)
                            v1 = base.copy()
                            v1 += A * c1
                            v1[mid] = base[mid] + c1 - c1*c1
                            f1 = np.max(v1)
                        else:
                            lo, c1, f1 = c1, c2, f2
                            c2 = lo + gr * (hi - lo)
                            v2 = base.copy()
                            v2 += A * c2
                            v2[mid] = base[mid] + c2 - c2*c2
                            f2 = np.max(v2)
                    best_t = (lo + hi) / 2.0
                    vals = base + A * best_t
                    vals[mid] = base[mid] + best_t - best_t * best_t
                    best_f = np.max(vals)
            
            if best_f < f_old - 1e-13:
                best[i] = best_t
                best_corr += A * (best_t - old)
                best_corr[mid] += best_t * (1.0 - best_t) - old * (1.0 - old)
                improved = True
        
        if not improved:
            break
    
    best_c5 = np.max(best_corr) * dx
    return best, best_c5


def binary_snap(h, n, n_trials=2000):
    """Try to snap h to binary while maintaining feasibility and improving C5."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best).astype(float).copy()
    rng = np.random.RandomState(12345)
    
    for _ in range(n_trials):
        if remaining_time() < 5:
            break
        # Pick a random position
        i = rng.randint(n)
        # Try flipping to 0 or 1
        old = best[i]
        for new_val in [0.0, 1.0]:
            if abs(new_val - old) < 1e-10:
                continue
            # Compute new correlation
            # A_i is the effect vector
            ncorr = 2 * n - 1
            mid = n - 1
            offsets = np.arange(-(n-1), n)
            i_plus = i + offsets
            i_minus = i - offsets
            vp = (i_plus >= 0) & (i_plus < n)
            vm = (i_minus >= 0) & (i_minus < n)
            A = np.zeros(ncorr)
            if np.any(vm):
                A[vm] += (1.0 - best[i_minus[vm]])
            if np.any(vp):
                A[vp] -= best[i_plus[vp]]
            
            new_corr = best_corr + A * (new_val - old)
            new_corr[mid] += new_val * (1.0 - new_val) - old * (1.0 - old)
            
            # Check sum constraint
            sum_diff = new_val - old
            # We need sum to stay at n/2, so we might need to adjust another coord
            # For now, check if the C5 improves
            new_c5 = np.max(new_corr)
            if new_c5 < best_c5:
                # Apply
                best_corr = new_corr
                best[i] = new_val
                best_c5 = new_c5
                break
    
    # Project to feasibility
    best = project_feasible(best, n)
    best_c5 = compute_c5(best)
    return best, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
