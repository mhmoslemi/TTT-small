import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TOPK = 5
_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty

def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g

def lbfgs_pass(h, n, beta, top_k=5, maxiter=800):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 10})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h

The first term is h shifted by d (with zero-padding), and the second term is h shifted by -d.

So V[:, d+n-1] = roll(h, d, padded) + roll(h, -d, padded)

This can be computed using np.roll with padding, or using FFT.

Actually, for the initial build, O(n²) is fine since it's a one-time cost. The important thing is that the update is O(n) per position change.

When h[p] changes by δ, V[q, col] changes if h[p] is one of the terms in V[q, col].
V[q, col] = h[q-d]*1_{valid} + h[q+d]*1_{valid} where col = d+n-1.

- q+d = p (so d = p-q): V[q, p-q+n-1] includes h[p]

So: for each q, V[q, q-p+n-1] += δ and V[q, p-q+n-1] += δ.

This is correct and O(n) per position change.
