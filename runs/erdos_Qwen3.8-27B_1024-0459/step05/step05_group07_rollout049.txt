def cvxpy_solve_linearized(h_current, n, active_lags, trust_radius):
    import cvxpy as cp
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    corr = compute_corr(h_current)
    max_corr = np.max(corr)
    
    constraints = [
        h >= 0, h <= 1,
        cp.sum(h) == n / 2,
        cp.norm(h - h_current, 2) <= trust_radius
    ]
    
    # Add linearized constraints for active lags
    for k_idx in active_lags:
        offset = k_idx - (n - 1)
        # Linearize sum h[i+offset]*(1-h[i])
        # ≈ sum h_cur[i+offset]*(1-h[i]) + (1-h_cur[i])*(h[i+offset]-h_cur[i+offset])
        # = sum h_cur[i+offset] - sum h_cur[i+offset]*h[i] - sum h_cur[i+offset]*h_cur[i] + sum h[i+offset] - sum h_cur[i+offset]
        
        # Simpler: corr_k ≈ corr_k_current + grad_k · (h - h_current)
        # where grad_k[i] = (1-h_current[i]) if i is the "h" index in the pair
        #                    -h_current[i] if i is the "(1-h)" index
        
        grad_k = np.zeros(n)
        if offset >= 0:
            npairs = n - offset
            for i in range(npairs):
                grad_k[i] += -h_current[i + offset]  # from h[i+offset]*(1-h[i])
                grad_k[i + offset] += (1.0 - h_current[i])
        else:
            aoff = -offset
            npairs = n - aoff
            for i in range(npairs):
                grad_k[i] += -h_current[i + aoff]
                grad_k[i + aoff] += (1.0 - h_current[i])
        
        # corr_k(h) ≈ corr_k(h_cur) + grad_k · (h - h_cur)
        constraints.append(cp.sum(grad_k * h) - cp.sum(grad_k * h_current) + corr[k_idx] <= t)
    
    obj = cp.Maximize(t)  # We want to find the max feasible t... 
    # Actually we want minimize t subject to corr_k <= t for all active k
    # But we also need to ensure non-active lags don't exceed t
    # So: minimize t subject to corr_k_lin <= t for active, cor_i_lin <= t for non-active
    
    # Actually, for a valid upper bound, we need ALL lags:
    # Let's just add all of them
    
    problem = cp.Problem(cp.Minimize(t), constraints)
    problem.solve(solver=cp.GLOP)  # or CPLEX
    if h.value is not None:
        return h.value
    return None

                            c2 = lo + gr * (hi - lo)
                            v2 = base + A * c2
                            v2[mid] = base[mid] + c2 - c2 * c2
                            f2 = np.max(v2)

                    best_t = (lo + hi) / 2.0
                    vv = base + A * best_t
                    vv[mid] = base[mid] + best_t - best_t * best_t
                    best_f = np.max(vv)

                if best_f < cur_max - 1e-13:
                    best[i] = best_t
                    best_corr += A * (best_t - old)
                    best_corr[mid] += best_t * (1.0 - best_t) - old * (1.0 - old)
                    improved = True

        if not improved:
            break

    final_c5 = float(np.max(best_corr)) * dx
    return best, final_c5

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _g_smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]

def _g_smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]

def _g_constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0

def _g_constraint_sum_jac(x):
    return np.ones(_N_WORK)

def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    h0 = h.copy().astype(float)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    try:
        res = minimize(_g_smooth_obj, h0, jac=_g_smooth_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def binary_flip_sa(h_input, n, max_iters=50000, seed=42):
    rng = np.random.RandomState(seed)
    h = np.array(h_input, dtype=float)
    h = (h > 0.5).astype(float)
    h = fix_binary_sum(h, n)
    dx = 2.0 / n
    corr = compute_corr(h).astype(float)
    c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = c5
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])

    T_start = 0.005
    T_end = 0.0001
    total = max_iters

    for it in range(total):
        if remaining_time() < 10:
            break
        if it % 1000 == 0:
            T = T_start * (T_end / T_start) ** (it / total)
            if c5 - best_c5 > 0.002 and it > 5000:
                T = min(T * 5, 0.01)
        else:
            T = T_start * (T_end / T_start) ** (it / total)

        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]

        delta = np.zeros(ncorr)
        ip = i + offsets
        im = i - offsets
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        if np.any(vp):
            delta[vp] += h[ip[vp]]
        if np.any(vm):
            delta[vm] -= (1.0 - h[im[vm]])

        jp = j + offsets
        jm = j - offsets
        vpj = (jp >= 0) & (jp < n)
        vmj = (jm >= 0) & (jm < n)
        if np.any(vpj):
            delta[vpj] -= h[jp[vpj]]
        if np.any(vmj):
            delta[vmj] += (1.0 - h[jm[vmj]])

        new_corr = corr + delta
        new_c5 = float(np.max(new_corr)) * dx

        if new_c5 < c5 - 1e-15 or rng.random() < np.exp(-(new_c5 - c5) / max(T, 1e-10)):
            corr = new_corr
            c5 = new_c5
            h[i] = 0.0
            h[j] = 1.0
            ones_list[ones_list.index(i)] = j
            zeros_list[zeros_list.index(j)] = i
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Thorough coordinate descent from parent
    h, c5 = coord_descent(h, n, max_sweeps=400)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP refinement
    if remaining_time() > 60:
        h2 = slsqp_refine(h, n, 3000, 400)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        h, c5 = coord_descent(h2, n, max_sweeps=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Higher beta SLSQP
    if remaining_time() > 60:
        h2 = slsqp_refine(best_h, n, 20000, 500)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        h, c5 = coord_descent(best_h, n, max_sweeps=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Binary flip simulated annealing
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = fix_binary_sum(h_bin, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5 + 0.02:
            h_b, c5_b = binary_flip_sa(h_bin, n, max_iters=40000, seed=seed)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                h, c5 = coord_descent(h_b, n, max_sweeps=300)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    # Phase 5: Multiple perturbations + coordinate descent
    if remaining_time() > 120:
        for p in range(12):
            if remaining_time() < 30:
                break
            scale = 0.12 / (1.0 + 0.2 * p)
            mask = rng.random(n) < 0.25
            h_pert = best_h.copy()
            h_pert[mask] += rng.randn(np.sum(mask)) * scale
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = coord_descent(h_pert, n, max_sweeps=150)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 40:
                    h_slq = slsqp_refine(h_new, n, 10000, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        h_new2, c5_new2 = coord_descent(h_slq, n, max_sweeps=100)
                        if c5_new2 < best_c5:
                            best_c5 = c5_new2
                            best_h = h_new2.copy()

    # Phase 6: Try structured inits
    if remaining_time() > 120:
        h_tm = thue_morse_init(n)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 + 0.03:
            h_new, c5_new = coord_descent(h_tm, n, max_sweeps=300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        # Random binary
        h_rb = np.zeros(n)
        ones = rng.choice(n, n // 2, replace=False)
        h_rb[ones] = 1.0
        c5_rb = compute_c5(h_rb)
        if c5_rb < best_c5 + 0.05:
            h_b, c5_b = binary_flip_sa(h_rb, n, max_iters=30000, seed=seed + 1)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                h, c5 = coord_descent(h_b, n, max_sweeps=200)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    # Phase 7: Higher resolution
    if remaining_time() > 180:
        high_n = 2 * n
        if high_n <= 2560:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            c5_high = compute_c5(h_high)
            h_new, c5_new = coord_descent(h_high, high_n, max_sweeps=80)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = high_n
                if remaining_time() > 60:
                    h_slq = slsqp_refine(h_new, high_n, 5000, 200)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        best_n = high_n
                        h_new2, c5_new2 = coord_descent(h_slq, high_n, max_sweeps=40)
                        if c5_new2 < best_c5:
                            best_c5 = c5_new2
                            best_h = h_new2.copy()
                            best_n = high_n

    # Phase 8: Block coordinate descent on pairs
    if remaining_time() > 60:
        n_cur = best_n
        h_cur = best_h.copy()
        h_corr = compute_corr(h_cur).astype(float)
        dx_cur = 2.0 / n_cur
        ncorr_cur = 2 * n_cur - 1
        mid_cur = n_cur - 1
        offsets_cur = np.arange(-(n_cur - 1), n_cur, dtype=np.int64)

        for sweep in range(30):
            if remaining_time() < 10:
                break
            improved = False
            for i in range(0, n_cur - 1, 2):
                j = i + 1
                old_i = h_cur[i]
                old_j = h_cur[j]

                Ai = np.zeros(ncorr_cur)
                ip_i = i + offsets_cur
                im_i = i - offsets_cur
                vpi = (ip_i >= 0) & (ip_i < n_cur)
                vmi = (im_i >= 0) & (im_i < n_cur)
                if np.any(vmi):
                    Ai[vmi] += (1.0 - h_cur[im_i[vmi]])
                if np.any(vpi):
                    Ai[vpi] -= h_cur[ip_i[vpi]]
                Ai[mid_cur] = 0.0

                Aj = np.zeros(ncorr_cur)
                ip_j = j + offsets_cur
                im_j = j - offsets_cur
                vpj = (ip_j >= 0) & (ip_j < n_cur)
                vmj = (im_j >= 0) & (im_j < n_cur)
                if np.any(vmj):
                    Aj[vmj] += (1.0 - h_cur[im_j[vmj]])
                if np.any(vpj):
                    Aj[vpj] -= h_cur[ip_j[vpj]]
                Aj[mid_cur] = 0.0

                base = h_corr - Ai * old_i - Aj * old_j
                base[mid_cur] = h_corr[mid_cur] - old_i * (1 - old_i) - old_j * (1 - old_j)

                cur_max = np.max(h_corr)
                best_f_block = cur_max
                best_ti = old_i
                best_tj = old_j

                for ti_c in [0.0, old_i, 1.0]:
                    for tj_c in [0.0, old_j, 1.0]:
                        vv = base + Ai * ti_c + Aj * tj_c
                        vv[mid_cur] = base[mid_cur] + ti_c * (1 - ti_c) + tj_c * (1 - tj_c)
                        f_v = np.max(vv)
                        if f_v < best_f_block - 1e-13:
                            best_f_block = f_v
                            best_ti = ti_c
                            best_tj = tj_c

                if best_f_block < cur_max - 1e-13:
                    lo_i, hi_i = max(0.0, best_ti - 0.3), min(1.0, best_ti + 0.3)
                    lo_j, hi_j = max(0.0, best_tj - 0.3), min(1.0, best_tj + 0.3)
                    for _ in range(30):
                        new_best = False
                        for di in [-0.08, 0.08, -0.03, 0.03, -0.01, 0.01]:
                            for dj in [-0.08, 0.08, -0.03, 0.03, -0.01, 0.01]:
                                ti_t = best_ti + di
                                tj_t = best_tj + dj
                                if 0.0 <= ti_t <= 1.0 and 0.0 <= tj_t <= 1.0:
                                    vv = base + Ai * ti_t + Aj * tj_t
                                    vv[mid_cur] = base[mid_cur] + ti_t * (1 - ti_t) + tj_t * (1 - tj_t)
                                    f_v = np.max(vv)
                                    if f_v < best_f_block - 1e-13:
                                        best_f_block = f_v
                                        best_ti = ti_t
                                        best_tj = tj_t
                                        new_best = True
                        if not new_best:
                            break

                    h_cur[i] = best_ti
                    h_cur[j] = best_tj
                    h_corr += Ai * (best_ti - old_i) + Aj * (best_tj - old_j)
                    h_corr[mid_cur] += best_ti * (1 - best_ti) + best_tj * (1 - best_tj) - old_i * (1 - old_i) - old_j * (1 - old_j)
                    improved = True

            if not improved:
                break

        c5_block = float(np.max(h_corr)) * dx_cur
        if c5_block < best_c5:
            best_c5 = c5_block
            best_h = h_cur.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 15:
        n_cur = best_n
        h2 = slsqp_refine(best_h, n_cur, 50000, 300)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 10:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
