def project_feasible(h, n):

class SmoothMaxProblem:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
        self.dx = 2.0 / n

        m = np.max(corr)
        sh = corr - m
        cl = np.clip(self.beta * sh, -50.0, 50.0)

def projected_gradient_smooth(h, n, beta, steps=500, step_size=0.01):
    """Projected gradient descent on smooth max with sum constraint."""
    dx = 2.0 / n
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    ss = step_size
    for it in range(steps):
        if remaining() < 3:
            break
        # Compute objective gradient
        corr = compute_corr(h)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        
        # Gradient
        g = np.zeros(n)
        wc = w * dx
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= wk * h[:np_]
        
        # Project gradient to tangent space
        g -= np.mean(g)
        
        # Line search
        improved = False
        for s in [ss, ss*0.5, ss*0.25, ss*2, ss*1.5]:
            h_new = project_feasible(h - s * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
                break
        if not improved:
            ss *= 0.5
            if ss < 0.0001:
                break
    return best_h, best_c5

    """Tabu search on binary vectors."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, n, beta):
    """Smooth max of correlation * dx."""
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx

def smooth_max_grad(h, n, beta):
    """Gradient of smooth max correlation * dx."""
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    
    g = np.zeros(n)
    # Only process active weights
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-8:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return g * dx

            break

        g -= np.mean(g)
        
        improved = False
        for s in [ss, ss * 0.5, ss * 0.1, ss * 2.0]:
            h_new = project_feasible(h - s * g, n)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:

def would_reduce_top(h, i, j, top_offsets, n):
    """Check if swapping h[i]:1→0, h[j]:0→1 reduces all top offsets."""
    for d in top_offsets:
        off = d - (n - 1)
        if off >= 0:
            # corr for offset off: sum h[pos]*(1-h[pos+off])
            # Position i contribution: h[i]*(1-h[i+off]) = 1*(1-h[i+off]) if i+off < n
            # After swap: 0
            # Position i-off contribution: h[i-off]*(1-h[i]) = h[i-off]*0 = 0 (if i-off >= 0)
            # After swap: h[i-off]*(1-0) = h[i-off]
            # Position j contribution: h[j]*(1-h[j+off]) = 0 (if j+off < n)
            # After swap: 1*(1-h[j+off])
            # Position j-off contribution: h[j-off]*(1-h[j]) = h[j-off]*1 (if j-off >= 0)
            # After swap: h[j-off]*0 = 0
            
            old_i = (1 - h[i + off]) if (i + off < n) else 0
            new_i = 0
            old_im = h[i - off] if (i - off >= 0) else 0  # was 0 since h[i]=1
            new_im = h[i - off] if (i - off >= 0) else 0
            old_j = 0  # h[j] = 0
            new_j = (1 - h[j + off]) if (j + off < n) else 0
            old_jm = h[j - off] if (j - off >= 0) else 0  # h[j-off]*(1-h[j]) = h[j-off]
            new_jm = 0
            
            # Change = new - old
            change = (new_i - old_i) + (new_im - old_im) + (new_j - old_j) + (new_jm - old_jm)
            if change > 0:
                return False
        else:
            ao = -off
            # corr for offset -ao: sum (1-h[pos])*h[pos+ao]
            # = sum h[pos+ao]*(1-h[pos])
            # Position i: appears as h[i] in the "h[pos+ao]" role when pos = i-ao >= 0
            #   contribution: (1-h[i-ao])*h[i] = (1-h[i-ao])*1 = 1-h[i-ao] → 0
            # Position i: appears in "1-h[pos]" role when pos = i
            #   contribution: (1-h[i])*h[i+ao] = 0*h[i+ao] = 0 → (1-0)*h[i+ao] = h[i+ao]
            # Position j: appears as h[j] in "h[pos+ao]" role when pos = j-ao >= 0
            #   contribution: (1-h[j-ao])*h[j] = (1-h[j-ao])*0 = 0 → (1-h[j-ao])*1 = 1-h[j-ao]
            # Position j: appears in "1-h[pos]" role when pos = j
            #   contribution: (1-h[j])*h[j+ao] = 1*h[j+ao] = h[j+ao] → 0*h[j+ao] = 0
            
            change = 0
            if i - ao >= 0:
                change += 0 - (1 - h[i - ao])  # i in h[pos+ao] role
            if i + ao < n:
                change += h[i + ao] - 0  # i in 1-h[pos] role
            if j - ao >= 0:
                change += (1 - h[j - ao]) - 0  # j in h[pos+ao] role
            if j + ao < n:
                change += 0 - h[j + ao]  # j in 1-h[pos] role
            
            if change > 0:
                return False
    
    return True

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():

def smooth_max_val(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_max_gradient(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for k in active:
        wk = w[k]
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]

def swap_change_at_offset(h, i, j, d, n):

    if d == 0:

    if d > 0:

        if i + d < n:

            change += h[i + m]
        if j >= m:
            change += (1.0 - h[j - m])

            change += (-h[j + m])

            else:
                faults = np.array([], dtype=int)

            if len(faults) == 0:
                continue
            n_sample = min(len(faults), 10)
            sample = rng.choice(len(faults), n_sample, replace=False)

            for si in sample:
                fp = int(faults[si])
                if off > 0:
                    i_pos = fp
                    j_pos = fp + off
                elif off < 0:
                    i_pos = fp + ao
                    j_pos = fp
                else:
                    continue

                if h[i_pos] != 1 or h[j_pos] != 0:
                    continue

                key = (i_pos, j_pos)
                if key in tabu and tabu[key] > it:
                    continue

                if not swap_reduces_top(h, i_pos, j_pos, top_offsets, n):
                    continue

                h[i_pos] = 0
                h[j_pos] = 1
                new_max = float(np.max(compute_corr(h)))
                h[i_pos] = 1
                h[j_pos] = 0

                delta = new_max - max_val
                if delta < best_delta:
                    best_delta = delta
                    best_swap = key

        # Random candidates
        for _ in range(20):
            i_pos = int(ones[rng.randint(len(ones))])
            j_pos = int(zeros[rng.randint(len(zeros))])
            key = (i_pos, j_pos)
            if key in tabu and tabu[key] > it:
                continue
            if not swap_reduces_top(h, i_pos, j_pos, top_offsets, n):
                continue
            h[i_pos] = 0
            h[j_pos] = 1
            new_max = float(np.max(compute_corr(h)))
            h[i_pos] = 1
            h[j_pos] = 0
            delta = new_max - max_val
            if delta < best_delta:
                best_delta = delta
                best_swap = key

        if best_swap is not None:
            i_pos, j_pos = best_swap
            h[i_pos] = 0
            h[j_pos] = 1
            tabu[(i_pos, j_pos)] = it + tabu_len
            tabu[(j_pos, i_pos)] = it + tabu_len
            new_max = float(np.max(compute_corr(h)))
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                no_improve = 0
        else:
            cur_max = max_val

    return best_h, best_max * dx


def binarize(h, n):
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse_init(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def multistart_search(h_start, n, rng, time_limit=30):
    """Try multiple small perturbations and keep the best."""
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    start = time.time()

    for trial in range(20):
        if time.time() - start > time_limit or remaining() < 5:
            break
        h_p = h_start + rng.randn(n) * 0.05
        h_p = project_feasible(h_p, n)
        h_p, c5_p = centered_descent(h_p, n, 5000, 100, 0.008)
        h_p, c5_p = eq_top_corr(h_p, n, 5, 150, 0.008)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse_init(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Continuous optimization with various smoothness parameters
    for beta in [50, 200, 1000, 5000, 20000, 100000]:
        if remaining() < 8:
            break
        h_new, c5_new = centered_descent(h, n, beta, max_iter=300, init_step=0.01)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-correlation equalization
    for tk in [3, 5, 7, 10, 15]:
        if remaining() < 8:
            break
        h_eq, c5_eq = eq_top_corr(h, n, top_k=tk, max_iter=400, init_step=0.012)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Binary tabu search at parent resolution
    if remaining() > 40:
        h_bin = binarize(h, n)
        tl = min(150, int(remaining() * 0.3))
        h_b, c5_b = binary_tabu_search(h_bin, n, rng, time_limit=tl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c, c5_c = centered_descent(h_b, n, 10000, 200, 0.008)
            h_c, c5_c = eq_top_corr(h_c, n, 5, 200, 0.008)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
            h = best_h.copy()
        else:
            h = best_h.copy()

    # Phase 4: Structured starts (Thue-Morse, etc.)
    if remaining() > 60:
        h_tm = thue_morse_init(n, rng)
        h_tm, c5_tm = centered_descent(h_tm, n, 5000, 200, 0.01)
        h_tm, c5_tm = eq_top_corr(h_tm, n, 5, 200, 0.01)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            h = h_tm.copy()

        if remaining() > 30:
            h_bi = binarize(h_tm, n)
            h_bi, c5_bi = binary_tabu_search(h_bi, n, rng, time_limit=min(50, int(remaining() * 0.25)))
            if c5_bi < best_c5:
                best_c5 = c5_bi
                best_h = h_bi.copy()
                h_c, c5_c = centered_descent(h_bi, n, 10000, 150, 0.008)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c.copy()

        # Inverse Thue-Morse
        if remaining() > 30:
            h_itm = 1.0 - h_tm
            h_itm = project_feasible(h_itm, n)
            h_itm, c5_itm = centered_descent(h_itm, n, 5000, 150, 0.01)
            if c5_itm < best_c5:
                best_c5 = c5_itm
                best_h = h_itm.copy()

    # Phase 5: Multi-resolution exploration
    if remaining() > 80:
        n_candidates = []
        for delta in [16, -16, 32, -32, 48, -48, 64, -64, 80, -80, 96]:
            nt = best_n + delta
            if 128 <= nt <= 800 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [256, 320, 384, 448, 512, 576, 640, 720, 768]:
            if 128 <= nt <= 800 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:6]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)

            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

            h_t, c5_t = centered_descent(h_t, n_t, 3000, 200, 0.01)
            h_t, c5_t = eq_top_corr(h_t, n_t, 5, 150, 0.01)

            if remaining() > 15 and c5_t < best_c5 + 0.002:
                h_bi = binarize(h_t, n_t)
                h_bi, c5_bi = binary_tabu_search(h_bi, n_t, rng, time_limit=25)
                if c5_bi < c5_t:
                    c5_t = c5_bi
                    h_t = h_bi.copy()
                    h_c, c5_c = centered_descent(h_t, n_t, 10000, 100, 0.005)
                    if c5_c < c5_t:
                        c5_t = c5_c
                        h_t = h_c

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 6: Multistart local search from best
    if remaining() > 40:
        h_ms, c5_ms = multistart_search(best_h, best_n, rng, time_limit=min(50, int(remaining() * 0.3)))
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for beta in [10000, 50000, 200000]:
        if remaining() < 5:
            break
        h_new, c5_new = centered_descent(h, n, beta, 200, 0.005)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    for tk in [3, 5, 8, 12, 20]:
        if remaining() < 5:
            break
        h_eq, c5_eq = eq_top_corr(h, n, top_k=tk, max_iter=400, init_step=0.005)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary refinement
    if remaining() > 20:
        h_bin = binarize(h, n)
        tl = min(40, int(remaining() * 0.5))
        h_b, c5_b = binary_tabu_search(h_bin, n, rng, time_limit=tl)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c, c5_c = centered_descent(h_b, n, 50000, 150, 0.005)
            h_c, c5_c = eq_top_corr(h_c, n, 5, 150, 0.005)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Try evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
