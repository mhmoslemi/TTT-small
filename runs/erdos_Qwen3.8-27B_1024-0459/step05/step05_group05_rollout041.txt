def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    corr = compute_corr(h)
    dx = 2.0 / len(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    # Gradient of smooth max w.r.t. h
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # d/corr[k]/dh[p] = (1-h[p+d]) - h[p-d] (with boundary handling)
    g = np.zeros(n)
    for k in range(2*n-1):

        if wk < 1e-15:

        d = k - (n-1)
        if d >= 0:

            g[d:d+np_] += wk * (1.0 - h[:np_])

            g[:np_] += wk * (1.0 - h[ao:ao+np_])

def swap_delta(h, i, j, n):
    """Compute change in correlation when h[i]: 1->0, h[j]: 0->1"""
    # d_k = k - (n-1) for k in 0..2n-2
    d = np.arange(-(n-1), n)  # offsets
    # corr[k] = sum valid h[p] * (1-h[p+d_k])
    # Change when h[i] 1->0: -1 * [(1-h[i+d]) - h[i-d]] for valid
    # Change when h[j] 0->1: +1 * [(1-h[j+d]) - h[j-d]] for valid
    
    # For offset d:
    # h[i+d] is valid when 0 <= i+d < n
    # h[i-d] is valid when 0 <= i-d < n
    
    delta = np.zeros(2*n-1)
    for k_idx, d_val in enumerate(d):
        term_i = 0.0
        term_j = 0.0
        if 0 <= i + d_val < n:
            term_i += (1.0 - h[i + d_val])
        if 0 <= i - d_val < n:
            term_i -= h[i - d_val]
        if 0 <= j + d_val < n:
            term_j += (1.0 - h[j + d_val])
        if 0 <= j - d_val < n:
            term_j -= h[j - d_val]
        delta[k_idx] = -term_i + term_j
    return delta

    d = np.arange(-(n-1), n)
    # h[i+d] where valid
    i_plus = i + d
    i_minus = i - d
    j_plus = j + d
    j_minus = j - d
    
    valid_ip = (i_plus >= 0) & (i_plus < n)
    valid_im = (i_minus >= 0) & (i_minus < n)
    valid_jp = (j_plus >= 0) & (j_plus < n)
    valid_jm = (j_minus >= 0) & (j_minus < n)
    
    h_ip = np.where(valid_ip, h[np.clip(i_plus, 0, n-1)], 0.0)
    h_im = np.where(valid_im, h[np.clip(i_minus, 0, n-1)], 0.0)
    h_jp = np.where(valid_jp, h[np.clip(j_plus, 0, n-1)], 0.0)
    h_jm = np.where(valid_jm, h[np.clip(j_minus, 0, n-1)], 0.0)
    
    term_i = np.where(valid_ip, 1.0 - h_ip, 0.0) - np.where(valid_im, h_im, 0.0)
    term_j = np.where(valid_jp, 1.0 - h_jp, 0.0) - np.where(valid_jm, h_jm, 0.0)
    
    return -term_i + term_j

def swap_delta(h, i, j, n):
    """Change in corr when h[i]:1->0, h[j]:0->1. Returns array of length 2n-1."""
    d = np.arange(-(n-1), n)
    ip, im = i + d, i - d
    jp, jm = j + d, j - d

    vim = (im >= 0) & (im < n)
    vjp = (jp >= 0) & (jp < n)
    vjm = (jm >= 0) & (jm < n)
    
    hip = np.where(vip, h[np.maximum(np.minimum(ip, n-1), 0)], 0.0)
    him = np.where(vim, h[np.maximum(np.minimum(im, n-1), 0)], 0.0)
    hjp = np.where(vjp, h[np.maximum(np.minimum(jp, n-1), 0)], 0.0)
    hjm = np.where(vjm, h[np.maximum(np.minimum(jm, n-1), 0)], 0.0)

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def sm_obj(h, beta):
    corr = compute_corr(h)
    dx = 2.0 / len(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50, 50)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * dx


def sm_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50, 50)
    w = np.exp(sh) / np.sum(np.exp(sh))
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            np_ = n - d
            g[d:d + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[d:d + np_]
        else:
            ao = -d
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return g


def slsqp_optimize(h, n, beta, maxiter=500):
    def obj(x):
        return sm_obj(x, beta)

    def grad(x):
        return sm_grad(x, beta)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.fun < obj(h0):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def swap_delta(h, i, j, n):
    d = np.arange(-(n - 1), n)
    ip = i + d
    im = i - d
    jp = j + d
    jm = j - d
    vip = (ip >= 0) & (ip < n)
    vim = (im >= 0) & (im < n)
    vjp = (jp >= 0) & (jp < n)
    vjm = (jm >= 0) & (jm < n)
    hip = np.where(vip, h[np.clip(ip, 0, n - 1)], 0.0)
    him = np.where(vim, h[np.clip(im, 0, n - 1)], 0.0)
    hjp = np.where(vjp, h[np.clip(jp, 0, n - 1)], 0.0)
    hjm = np.where(vjm, h[np.clip(jm, 0, n - 1)], 0.0)
    term_i = np.where(vip, 1.0 - hip, 0.0) - np.where(vim, him, 0.0)
    term_j = np.where(vjp, 1.0 - hjp, 0.0) - np.where(vjm, hjm, 0.0)
    return -term_i + term_j


def new_max_after_delta(corr, delta, cur_max):
    nz = delta != 0
    if not np.any(nz):
        return cur_max
    changed = corr[nz] + delta[nz]
    mask_max = np.max(changed)
    if np.any(~nz):
        unmask_max = np.max(corr[~nz])
    else:
        unmask_max = 0.0
    return max(mask_max, unmask_max)


def binary_local_search(h_bin, n, rng, time_limit=60):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            idx = rng.choice(ones, n_ones - n_half, replace=False)
            h[idx] = 0
        else:
            zeros = np.where(h == 0)[0]
            idx = rng.choice(zeros, n_half - n_ones, replace=False)
            h[idx] = 1

    corr = compute_corr(h).copy()
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n

    start = time.time()
    no_improve = 0
    max_no_improve = 800

    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > max_no_improve:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0.0
        best_swap = None
        n_try = 40 + int(20 * no_improve / 100)

        top_idx = np.argsort(-corr)[:8]
        for _ in range(n_try):
            if rng.random() < 0.4:
                k_hot = top_idx[rng.randint(len(top_idx))]
                off = k_hot - (n - 1)
                if off >= 0:
                    np_ = n - off
                    faults = np.where((h[:np_] == 1) & (h[off:off + np_] == 0))[0]
                else:
                    ao = -off
                    np_ = n - ao
                    faults = np.where((h[ao:ao + np_] == 1) & (h[:np_] == 0))[0]
                if len(faults) == 0:
                    continue
                fi = faults[rng.randint(len(faults))]
                if off >= 0:
                    i = fi
                else:
                    i = fi + ao
                j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            if i == j or h[i] != 1 or h[j] != 0:
                continue
            delta = swap_delta(h, i, j, n)
            nm = new_max_after_delta(corr, delta, cur_max)
            delta_max = nm - cur_max
            if delta_max < best_delta - 1e-10:
                best_delta = delta_max
                best_swap = (i, j, delta)

        if best_swap is not None:
            i, j, delta = best_swap
            corr += delta
            h[i] = 0
            h[j] = 1
            cur_max += best_delta
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

    corr_final = compute_corr(best_h)
    return best_h, float(np.max(corr_final)) * dx


def binary_sa(h_bin, n, rng, time_limit=100):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            idx = rng.choice(ones, n_ones - n_half, replace=False)
            h[idx] = 0
        else:
            zeros = np.where(h == 0)[0]
            idx = rng.choice(zeros, n_half - n_ones, replace=False)
            h[idx] = 1

    corr = compute_corr(h).copy()
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n

    start = time.time()
    T = 3.0
    T_min = 0.005
    cool_rate = 0.9995

    while time.time() - start < time_limit and remaining() > 5:
        if T < T_min:
            T = T_min
        T *= cool_rate

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        delta = swap_delta(h, i, j, n)
        nm = new_max_after_delta(corr, delta, cur_max)

        if nm < cur_max - 1e-10:
            corr += delta
            h[i] = 0
            h[j] = 1
            cur_max = nm
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        elif rng.random() < np.exp(-(nm - cur_max) / T):
            corr += delta
            h[i] = 0
            h[j] = 1
            cur_max = nm

    corr_final = compute_corr(best_h)
    return best_h, float(np.max(corr_final)) * dx


def binarize(h, rng=None):
    n = len(h)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def thue_morse_seq(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def interleaved_seq(n, rng=None):
    h = np.zeros(n)
    h[::2] = 1.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = ones_idx[:n_ones - n_half] if rng is None else rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = zeros_idx[:n_half - n_ones] if rng is None else rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    return h


def projected_gradient(h, n, beta, n_iter=300):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if remaining() < 3:
            break
        g = sm_grad(h, beta)
        g -= np.mean(g)
        g = np.where(h > 0.99, -np.abs(g), g)
        g = np.where(h < 0.01, np.abs(g), g)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            step = min(step * 1.1, 0.2)
        else:
            step *= 0.8
            if step < 0.001:
                break
    return best_h, best_c5
