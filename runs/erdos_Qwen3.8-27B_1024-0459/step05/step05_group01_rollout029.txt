- t = 1
- A point where two or more f_k(t) are equal and all others are ≤ that value

```python
def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        # Count bits
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2

        # Remove excess ones
        ones = np.where(h == 1)[0]
        excess = n_ones - target
        h[ones[:excess]] = 0
    elif n_ones < target:
        # Add needed ones
        zeros = np.where(h == 0)[0]
        needed = target - n_ones
        h[zeros[:needed]] = 1
    return h
```

```python
def flip_delta(h, a, b, n):
    """Compute the change in corr when flipping h[a]:1->0 and h[b]:0->1"""
    corr_delta = np.zeros(2*n-1)
    # Effect of h[a]: 1 -> 0
    # Left factor: h[a] * (1-h[a+k-(n-1)])
    # Old: 1 * (1-h[a+k-(n-1)]), New: 0
    # Delta: -(1-h[a+k-(n-1)])
    lo_a = max(0, n-1-a)
    hi_a = min(2*n-2, n-1+a)
    if lo_a <= hi_a:
        idx = np.arange(lo_a, hi_a+1)
        right_idx = a + idx - (n-1)  # = a+k-(n-1), ranges 0..2a
        right_idx = right_idx[(right_idx >= 0) & (right_idx < n)]
        idx_valid = idx[(a + idx - (n-1) >= 0) & (a + idx - (n-1) < n)]
        if len(idx_valid) > 0:
            ri = a + idx_valid - (n-1)
            corr_delta[idx_valid] -= (1.0 - h[ri])
    
    # Right factor: h[j] * (1-h[a]) where j+k-(n-1) = a, so k = a-j+(n-1)
    # Old: h[j] * (1-1) = 0, New: h[j] * (1-0) = h[j]
    # Delta: +h[j]
    lo_a2 = max(0, a)
    hi_a2 = min(2*n-2, a+n-1)
    if lo_a2 <= hi_a2:
        idx = np.arange(lo_a2, hi_a2+1)
        left_idx = a - idx + (n-1)  # = j, ranges 0..n-1
        valid = (left_idx >= 0) & (left_idx < n)
        if np.any(valid):
            corr_delta[idx[valid]] += h[left_idx[valid]]
    
    # Effect of h[b]: 0 -> 1
    # Left factor: h[b] * (1-h[b+k-(n-1)])
    # Old: 0, New: (1-h[b+k-(n-1)])
    # Delta: +(1-h[b+k-(n-1)])
    lo_b = max(0, n-1-b)
    hi_b = min(2*n-2, n-1+b)
    if lo_b <= hi_b:
        idx = np.arange(lo_b, hi_b+1)
        valid = (b + idx - (n-1) >= 0) & (b + idx - (n-1) < n)
        if np.any(valid):
            ri = b + idx[valid] - (n-1)
            corr_delta[idx[valid]] += (1.0 - h[ri])
    
    # Right factor: h[j] * (1-h[b]) where j+k-(n-1) = b
    # Old: h[j] * (1-0) = h[j], New: h[j] * (1-1) = 0
    # Delta: -h[j]
    lo_b2 = max(0, b)
    hi_b2 = min(2*n-2, b+n-1)
    if lo_b2 <= hi_b2:
        idx = np.arange(lo_b2, hi_b2+1)
        left_idx = b - idx + (n-1)
        valid = (left_idx >= 0) & (left_idx < n)
        if np.any(valid):
            corr_delta[idx[valid]] -= h[left_idx[valid]]
    
    return corr_delta
```

```python
def flip_corr_delta(h, a, b, n):
    """
    Compute the change to corr when h[a]: 1->0 and h[b]: 0->1.
    Returns the delta array (length 2n-1).
    """
    delta = np.zeros(2*n-1)
    
    # h[a]: 1 -> 0
    # As left factor: h[a]*(1-h[a+k-(n-1)])  for valid a+k-(n-1)
    # Old contribution: 1*(1-h[a+k-(n-1)]) = 1-h[a+k-(n-1)]
    # New contribution: 0
    # Delta: -(1-h[a+k-(n-1)]) = h[a+k-(n-1)] - 1
    # Valid k: 0 <= a+k-(n-1) <= n-1, i.e., n-1-a <= k <= n-1+a
    lo = max(0, n-1-a)
    hi = min(2*n-2, n-1+a)
    if lo <= hi:
        ks = np.arange(lo, hi+1)
        ri = a + ks - (n-1)  # right index
        # ri should be in [0, n-1] by construction
        delta[ks] += (h[ri] - 1.0)  # = -(1-h[ri])
    
    # As right factor: h[j]*(1-h[a]) where j+k-(n-1)=a, i.e., k = a-j+(n-1)
    # Old: h[j]*(1-1) = 0
    # New: h[j]*(1-0) = h[j]
    # Delta: +h[j]
    # Valid: 0 <= j <= n-1 and 0 <= k <= 2n-2
    # k = a-j+(n-1), j from 0 to n-1
    # k ranges from a+(n-1) down to a
    # So k in [a, a+n-1], clipped to [0, 2n-2]
    lo2 = max(0, a)
    hi2 = min(2*n-2, a+n-1)
    if lo2 <= hi2:
```
