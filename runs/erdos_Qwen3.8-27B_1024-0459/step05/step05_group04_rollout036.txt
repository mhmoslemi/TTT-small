def project_feasible(h, n):

def make_swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)

def subgradient_minimax(h, n, time_limit=100, rng=None):
    """Minimize max_k c_k(h) using subgradient method with projection."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    start = time.time()
    lr = 0.02
    n_active = 10
    patience = 200
    no_improve = 0
    best_lr = lr

    while time.time() - start < time_limit and remaining() > 10:
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        cur_c5 = c_max * dx

        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1

        if no_improve > patience:
            lr = best_lr * 0.5
            if lr < 1e-10:
                break
            no_improve = 0

        idx = np.argsort(-corr)[:n_active]
        weights = corr[idx] - corr[idx[-1]] + 1.0
        weights = weights / np.sum(weights)

        g = np.zeros(n)
        for w, k in zip(weights, idx):
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += w * (1.0 - h[:np_])
                g[:np_] -= w * h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += w * (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= w * h[:np_]

        g_sum = np.sum(g)
        g -= g_sum / n

        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break

        accepted = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            step = lr * factor / g_norm
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                h = h_new
                accepted = True
                break

        if not accepted:
            lr *= 0.7

    return best_h, best_c5

def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _TGT,
             'jac': lambda x: np.ones(_N)}]

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / _BETA) * (2.0 / n)

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]

        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]

        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn

        improved = False
        for step in [0.02, 0.01, 0.05, 0.03, 0.08, 0.005, 0.015, 0.04, 0.06]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            corr = compute_corr(h)
            ones = np.where(h == 1)[0]
            order = np.argsort([np.max(corr) for _ in range(len(ones))])
            # Remove ones that are at high-correlation positions
            ones_sorted = ones[np.argsort(-h[ones])]
            h[ones_sorted[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def binary_init(h_cont, n):
    """Convert continuous to binary respecting sum constraint."""
    h = (h_cont > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            order = np.argsort(h_cont[ones])
            h[ones[order[:n_ones - n_half]]] = 0
        else:
            zeros = np.where(h == 0)[0]
            order = np.argsort(-h_cont[zeros])
            h[zeros[order[:n_half - n_ones]]] = 1
    return h

def kernel_smooth(h, n, sigma=0.5):
    """Apply Gaussian kernel smoothing."""
    x = np.arange(n) - n // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= np.sum(kernel)
    h_smooth = np.convolve(h, kernel, mode='same')
    return project_feasible(h_smooth, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n

    # Phase 1: Binary local search with SA at parent resolution
    h_bin = binary_init(best_h, n)
    t1 = min(350, int(remaining() * 0.35))
    if t1 > 30:
        h_b1, c5_b1 = binary_local_search(h_bin, n, time_limit=t1, rng=np.random.RandomState(seed))
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()

    # Phase 1b: Multi-swap refinement
    if remaining() > 60:
        h_b1b = binary_init(best_h, n)
        h_b1b, c5_1b = multi_swap_search(h_b1b, n, time_limit=min(60, int(remaining() * 0.1)), rng=np.random.RandomState(seed + 7))
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h_b1b.copy()

    # Phase 2: Alternative binary starts
    if remaining() > 100:
        alt_starts = [thue_morse(n)]
        # Also try shifted Thue-Morse
        tm = thue_morse(n)
        for shift in [1, 2, 3]:
            alt_starts.append(np.roll(tm, shift))
        # Random start
        rand_h = (rng.rand(n) > 0.5).astype(float)
        n_half = n // 2
        n_o = int(np.sum(rand_h))
        if n_o != n_half:
            if n_o > n_half:
                idx = np.where(rand_h == 1)[0]
                rand_h[idx[:n_o - n_half]] = 0
            else:
                idx = np.where(rand_h == 0)[0]
                rand_h[idx[:n_half - n_o]] = 1
        alt_starts.append(rand_h)

        for sp in alt_starts:
            if remaining() < 40:
                break
            sp_c5 = compute_c5(sp)
            t_alt = min(80, int(remaining() * 0.08))
            if t_alt > 15:
                h_alt, c5_alt = binary_local_search(sp, n, time_limit=t_alt, rng=np.random.RandomState(seed + 100))
                if c5_alt < best_c5:
                    best_c5 = c5_alt
                    best_h = h_alt.copy()

    # Phase 3: Continuous subgradient minimax
    h_cont = project_feasible(best_h.copy(), n)
    t_sg = min(150, int(remaining() * 0.15))
    if t_sg > 30:
        h_sg, c5_sg = subgradient_minimax(h_cont, n, time_limit=t_sg, rng=np.random.RandomState(seed + 200))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 4: SLSQP smooth optimization with increasing beta
    h_work = project_feasible(best_h.copy(), n)
    for beta in [500, 2000, 10000, 30000, 80000]:
        if remaining() < 15:
            break
        t_slsqp = min(60, int(remaining() * 0.06))
        h_new = slsqp_smooth(h_work, n, beta, maxiter=min(300, t_slsqp * 2))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new.copy()
        else:
            h_work = h_new.copy()  # Continue from here even if not better

    # Phase 5: Top-shifts equalization
    if remaining() > 20:
        h_work = project_feasible(best_h.copy(), n)
        h_eq, c5_eq = top_shifts_equalize(h_work, n, top_k=8, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 6: Smoothing trick - smooth the binary solution slightly
    if remaining() > 30:
        h_bin_best = binary_init(best_h, n)
        for sigma in [0.3, 0.5, 0.8, 1.0]:
            h_smoothed = kernel_smooth(h_bin_best, n, sigma)
            h_sm, c5_sm = subgradient_minimax(h_smoothed, n, time_limit=min(30, int(remaining() * 0.05)), rng=np.random.RandomState(seed + 300))
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()
            if remaining() < 20:
                break

    # Phase 7: Multi-resolution exploration
    resolutions = sorted(set([n - 8, n - 4, n + 4, n + 8, n + 16, n + 32]))
    resolutions = [r for r in resolutions if 128 <= r <= 1024 and r % 2 == 0 and r != n]
    resolutions.sort(key=lambda x: abs(x - n))

    for n_t in resolutions[:4]:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = binary_init(h_interp, n_t)
        t_bi = min(50, int(remaining() * 0.06))
        if t_bi > 10:
            h_b3, c5_b3 = binary_local_search(h_bin_t, n_t, time_limit=t_bi, rng=np.random.RandomState(seed + 400))
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t

        # Also try continuous at this resolution
        if remaining() > 30:
            h_cont_t = project_feasible(interp_to(best_h, best_n, n_t), n_t)
            h_sg_t, c5_sg_t = subgradient_minimax(h_cont_t, n_t, time_limit=min(30, int(remaining() * 0.05)), rng=np.random.RandomState(seed + 500))
            if c5_sg_t < best_c5:
                best_c5 = c5_sg_t
                best_h = h_sg_t.copy()
                best_n = n_t

    # Phase 8: Try a larger resolution for continuous optimization
    if remaining() > 80:
        n_fine = max(n + 64, 448)
        n_fine = ((n_fine // 2) * 2)  # ensure even
        h_fine = project_feasible(interp_to(best_h, best_n, n_fine), n_fine)
        h_fine, c5_fine = subgradient_minimax(h_fine, n_fine, time_limit=min(60, int(remaining() * 0.15)), rng=np.random.RandomState(seed + 600))
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine
        if remaining() > 30:
            h_eq_f, c5_eq_f = top_shifts_equalize(best_h, best_n, top_k=8, max_iters=300)
            if c5_eq_f < best_c5:
                best_c5 = c5_eq_f
                best_h = h_eq_f.copy()

    # Phase 9: Final binary polish from best continuous
    n_final = best_n
    h_bin_final = binary_init(best_h, n_final)
    t_final = min(80, int(remaining() * 0.2))
    if t_final > 15:
        h_fp, c5_fp = binary_local_search(h_bin_final, n_final, time_limit=t_final, rng=np.random.RandomState(seed + 700))
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()
            best_n = n_final

    # Phase 10: Final SLSQP polish
    n = best_n
    h = project_feasible(best_h.copy(), n)
    for beta in [50000, 100000, 200000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 10:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=10, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
