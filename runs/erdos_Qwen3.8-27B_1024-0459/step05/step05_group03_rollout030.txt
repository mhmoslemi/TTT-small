    n = len(h_bin)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return np.max(corr), corr

def binary_swap_energy(h_bin, remove_pos, add_pos):
    """Compute the change in max correlation when moving a 1 from remove_pos to add_pos."""
    # This is expensive to do exactly, so we approximate
    n = len(h_bin)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    old_max = np.max(corr)
    
    h_bin[remove_pos] = 0
    h_bin[add_pos] = 1
    new_corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    new_max = np.max(new_corr)
    
    h_bin[remove_pos] = 1
    h_bin[add_pos] = 0
    return new_max - old_max, old_max

def simulated_annealing_binary(h_bin, n, max_iters=50000, t_start=1.0, t_end=0.01, cooling=0.9997):
    """Simulated annealing on binary vectors with single-swap moves."""
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    curr_max, curr_corr = binary_energy(h_bin)
    best_h = h_bin.copy()
    best_max = curr_max

            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        j = rng.randint(len(zeros))

            h_bin[p2] = 0

def sa_binary(h_bin, n, max_iters, t_start, t_end, cooling, seed=42):
    """Simulated annealing on binary vectors."""
    rng = np.random.RandomState(seed)
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()

    best_max = curr_max
    t = t_start

    for it in range(max_iters):
        if remaining() < 10:
            break
        
        i = rng.randint(n_ones)
        j = rng.randint(n_ones)
        p1 = ones[i]
        p2 = zeros[j]
        
        # Make the swap
        h_bin[p1] = 0.0
        h_bin[p2] = 1.0
        new_corr = compute_corr(h_bin)
        new_max = float(np.max(new_corr))

            curr_corr = new_corr
            ones[i] = p2
            zeros[j] = p1
            if curr_max < best_max:
                best_max = curr_max
                best_h = h_bin.copy()
        else:
            h_bin[p1] = 1.0
            h_bin[p2] = 0.0
        
        t = max(t_end, t * cooling)
    
    return best_h, best_max * (2.0 / n)

3. The objective is a quadratic function of h (since each corr[k] is bilinear in h)
4. Use a QP solver or gradient method

Actually, the objective sum_{k∈K} corr[k] is a quadratic form in h. Let me write it out:
corr[k] = sum_i h[i]*(1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]

So sum_{k∈K} corr[k] = |K| * sum_i h[i] - sum_{k∈K} sum_i h[i]*h[i+k]
= |K| * (n/2) - sum_{k∈K} sum_i h[i]*h[i+k]

The first term is constant. So minimizing the sum of correlations is equivalent to maximizing:
sum_{k∈K} sum_i h[i]*h[i+k]

This is a quadratic form: h^T Q h where Q is a symmetric matrix (the sum over k∈K of the circulant matrix for lag k).

So the continuous problem (with the top-K smoothing) is:
maximize h^T Q h
subject to: 0 ≤ h ≤ 1, sum(h) = n/2

This is a non-convex QP (maximizing a quadratic with potential indefinite Q). However, we can solve it using:
- Coordinate descent
- Alternating direction method
- Or treat it as a max-cut problem (which is NP-hard in general, but has good heuristics)

= Q[j,j]*t² + 2*(Q_{-j}^T h_{-j})*t where t = h[j]
