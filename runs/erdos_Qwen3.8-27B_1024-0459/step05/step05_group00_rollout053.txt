import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] -= h[a:n]
        g[a:n] += (1.0 - h[:npairs])
    return g


def smooth_max_obj(x, beta, n):
    corr = compute_corr(x)
    dx = 2.0 / n
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_max_grad(x, beta, n):
    corr = compute_corr(x)
    dx = 2.0 / n
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    total = np.sum(ex)
    w = ex / total
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] -= wk * x[a:]
            g[a:] += wk * (1.0 - x[:np_])
    return g


def slsqp_optimize(h, n, beta, maxiter=500):
    def obj(x):
        return smooth_max_obj(x, beta, n)

    def grad(x):
        return smooth_max_grad(x, beta, n)

    def cons_f(x):
        return np.sum(x) - n / 2.0

    def cons_j(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def subgradient_optimize(h, n, time_limit, rng, K=30, beta_start=40, beta_end=150):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    n_iter = max(100, int(time_limit * 40))
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        progress = it / max(n_iter, 1)
        beta = beta_start + (beta_end - beta_start) * progress
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_top = corr[idx]
        weights = np.exp(beta * (c_top - c_top.max()))
        s = weights.sum()
        if s < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= s
        g = np.zeros(n)
        for i, k in enumerate(idx):
            if weights[i] > 1e-12:
                g += weights[i] * grad_corr_k(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.04 * n / gn
        step *= 0.3 + 0.7 * (0.5 + 0.5 * np.cos(np.pi * progress))
        step = max(step, 0.002)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)

            best_h = h_new.copy()
            h = h_new.copy()

                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
    return best_h, best_c5
