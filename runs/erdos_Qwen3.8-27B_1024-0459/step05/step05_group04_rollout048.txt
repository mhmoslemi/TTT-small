def equalize_gradient(h, n, top_k=5, step_lr=0.01):
    """Gradient step to equalize top-k correlations."""
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:top_k]
    top_vals = corr[top_idx]
    mean_top = np.mean(top_vals)
    
    # Gradient of sum((corr[top_i] - mean_top)^2) w.r.t. h
    # = 2 * sum_i (corr[top_i] - mean_top) * d(corr[top_i])/dh
    
    g = np.zeros(n)
    for k in top_idx:
        w = corr[k] - mean_top
        if abs(w) < 1e-10:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off+np_] += w * (1.0 - h[:np_])
            g[:np_] += w * (-h[off:off+np_])
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += w * (1.0 - h[ao:ao+np_])
            g[ao:ao+np_] += w * (-h[:np_])
    
    # Also add constraint gradient (keep sum constant)
    g = g - np.mean(g)  # project to zero-sum space
    
    return h - step_lr * g

                        if h[i] == 1 and h[j] == 0:
                            dr = swap_delta(h, i, j, n)
                            nm = fast_new_max(corr, dr, cur_max)
                            delta_t = nm - cur_max
                            if delta_t < 0 or rng.rand() < np.exp(-delta_t / T):
                                h[i] = 0
                                h[j] = 1
                                corr = corr + dr
                                cur_max = nm
                T *= cooling
                if no_improve > 500 and restart_count < 10:
                    # Restart: perturb and reset
                    restart_count += 1
                    n_flip = rng.randint(3, 8)
                    fi = rng.choice(ones, n_flip, replace=False)
                    fj = rng.choice(zeros, n_flip, replace=False)
                    h[fi] = 0
                    h[fj] = 1
                    corr = compute_corr(h)
                    cur_max = float(np.max(corr))
                    no_improve = 0
                    T = 0.003

    return best_h, best_max * dx

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def make_periodic_pattern(n, period, duty=0.5):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < int(period * duty):
            h[i] = 1.0
    return h

def make_interleaved(n, block_size):
    h = np.zeros(n)
    for i in range(n):
        block = i // block_size
        if block % 2 == 0:
            h[i] = 1.0
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def projected_gradient_smooth(h, n, beta, n_iters=200, lr_init=0.05):
    """Project gradient descent on smooth max of correlations."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init

    for it in range(n_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        sh = corr - m
        cl = np.clip(beta * sh, -40.0, 40.0)
        ex = np.exp(cl)
        sx = np.sum(ex)
        w = ex / sx

        # Gradient of smooth_max(h)'s contribution to C5
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-10:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:ao+np_])
                g[ao:ao+np_] -= wk * h[:np_]

        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

    return best_h, best_c5

def top_shifts_equalize(h, n, top_k=6, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        top_vals = corr[idx]
        mean_top = np.mean(top_vals)
        if top_vals[0] - top_vals[-1] < 1e-8:
            break

        g = np.zeros(n)
        for k in idx:
            w = corr[k] - mean_top
            if abs(w) < 1e-12:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += w * (1.0 - h[:np_])
                g[:np_] -= w * h[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += w * (1.0 - h[ao:ao+np_])
                g[ao:ao+np_] -= w * h[:np_]

        g = g - np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        improved = False
        for step in [0.005, 0.01, 0.02, 0.003, 0.015, 0.03, 0.008, 0.002]:
            ht = project_feasible(h + step * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n

    # Phase 1: Binary SA at parent resolution (main optimization)
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        h_bin = np.zeros(n)
        sort_idx = np.argsort(-best_h)
        h_bin[sort_idx[:n_half]] = 1.0

    t1 = min(350, int(remaining() * 0.35))
    if t1 > 30:
        h_b1, c5_b1 = binary_sa_optimize(h_bin, n, t1, rng, top_shifts=25, max_cand=2500)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()

    # Phase 2: Try alternative starting points at same resolution
    starts = []
    tm = thue_morse(n)
    starts.append(('tm', tm))
    # Interleaved pattern
    for bs in [3, 5, 7]:
        if bs * 2 <= n:
            starts.append((f'inter{bs}', make_interleaved(n, bs)))
    # Periodic with different periods
    for p in [6, 8, 10, 12]:
        if p * 2 <= n:
            starts.append((f'per{p}', make_periodic_pattern(n, p)))

    for name, sp in starts:
        if remaining() < 40:
            break
        # Project to have exactly n/2 ones
        sp = (sp > 0.5).astype(float)
        so = int(np.sum(sp))
        if so != n_half:
            if so > n_half:
                idx = np.where(sp == 1)[0]
                sp[idx[rng.choice(so, so - n_half, replace=False)]] = 0
            else:
                idx = np.where(sp == 0)[0]
                sp[idx[rng.choice(n - so, n_half - so, replace=False)]] = 1

        t2 = min(60, int(remaining() * 0.08))
        if t2 > 15:
            h_b2, c5_b2 = binary_sa_optimize(sp, n, t2, rng, top_shifts=20, max_cand=1500)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()

    # Phase 3: Multi-resolution binary search
    n_parent = best_n
    resolutions = sorted(set([
        n_parent - 32, n_parent - 16, n_parent + 16, n_parent + 32,
        n_parent + 64, 256, 320, 448, 512
    ]))
    resolutions = [r for r in resolutions if 128 <= r <= 768 and r % 2 == 0 and r != n_parent]
    resolutions.sort(key=lambda x: abs(x - n_parent))

    for n_t in resolutions[:3]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1

        t3 = min(80, int(remaining() * 0.07))
        if t3 > 15:
            h_b3, c5_b3 = binary_sa_optimize(h_bin_t, n_t, t3, rng, top_shifts=20, max_cand=1500)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t

    # Phase 4: Continuous refinement on best binary solution
    n = best_n
    h = best_h.copy()

    for beta in [200, 1000, 5000, 20000]:
        if remaining() < 15:
            break
        h_new, c5_new = projected_gradient_smooth(h, n, beta, n_iters=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Top-shifts equalization
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=6, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Another round of binary search from continuous solution
    if remaining() > 60:
        h_bin_now = (h > 0.5).astype(float)
        nh = int(np.sum(h_bin_now))
        if nh != n // 2:
            h_bin_now = np.zeros(n)
            h_bin_now[np.argsort(-h)[:n//2]] = 1.0
        t6 = min(80, int(remaining() * 0.1))
        if t6 > 20:
            h_b6, c5_b6 = binary_sa_optimize(h_bin_now, n, t6, rng, top_shifts=30, max_cand=3000)
            if c5_b6 < best_c5:
                best_c5 = c5_b6
                best_h = h_b6.copy()

    # Phase 7: Final continuous polish
    if remaining() > 15:
        h = best_h.copy()
        for beta in [5000, 20000, 80000]:
            if remaining() < 10:
                break
            h_new, c5_new = projected_gradient_smooth(h, n, beta, n_iters=200)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Final equalization
    if remaining() > 10:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=8, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
