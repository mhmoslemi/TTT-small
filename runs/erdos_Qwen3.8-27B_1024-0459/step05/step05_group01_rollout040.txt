def subgradient_step(h, n, top_k=10, alpha=0.01):
    """One step of subgradient descent on max(corr)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k lags
    idx_top = np.argsort(corr)[-top_k:]
    
    # Weights: proportional to (corr[k] - corr.min)^2 or similar
    c_vals = corr[idx_top]
    c_max = c_vals.max()
    c_min = c_vals.min()
    if c_max - c_min < 1e-15:
        w = np.ones(top_k) / top_k
    else:
        w = ((c_vals - c_min) / (c_max - c_min)) ** 2
        w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    n_half = n - 1
    for pos, k in enumerate(idx_top):
        if w[pos] < 1e-15:
            continue
        d = k - n_half  # lag
        ad = abs(d)
        if ad == 0:
            # corr[n-1] = sum(h[i] * (1-h[i])) = sum(h[i]) - sum(h[i]^2)
            # grad[i] = 1 - 2*h[i]
            grad += w[pos] * (1.0 - 2.0 * h)
        elif d > 0:
            # corr[n-1+d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # grad[i] = (1-h[i+d]) for i <= n-1-d
            # grad[i+d] = -h[i] for i <= n-1-d, i.e., grad[j] = -h[j-d] for j >= d
            valid = n - d
            grad[:valid] += w[pos] * (1.0 - h[d:d+valid])  # wait, h[d:d+valid] has length valid
            # h[d:d+valid] = h[d], h[d+1], ..., h[d+valid-1] = h[d]..h[n-1]
            # So (1-h[d:d+valid]) has the right length for the first 'valid' entries
            grad[d:] -= w[pos] * h[:valid]
        else:
            # d < 0, let a = -d > 0
            # corr[n-1-a] = sum_{i=0}^{n-1-a} h[i+a]*(1-h[i])
            # grad[i+a] += (1-h[i]) for i <= n-1-a
            # grad[i] += -h[i+a] for i <= n-1-a
            a = -d
            valid = n - a
            grad[a:] += w[pos] * (1.0 - h[:valid])
            grad[:valid] -= w[pos] * h[a:]
    
    # Take step and project
    h_new = project_feasible(h - alpha * grad, n)
    return h_new, corr

def project_feasible(h, n):
    h = np.array(h, dtype=float).flatten()
    target = n / 2.0
    # Bisection on lambda
    s = np.sum(np.clip(h, 0.0, 1.0))
    if abs(s - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def thue_morse(n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

def subgradient_descent(h, n, max_iter=500, time_limit=60, top_k=20):
    """Subgradient descent on the max correlation"""
    start = time.time()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.02
    n_half = n - 1
    
    for it in range(max_iter):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        
        corr = compute_corr(h)
        c_max = corr.max()
        
        # Get top-k lags with exponential weighting
        idx_top = np.argsort(corr)[-top_k:]
        c_vals = corr[idx_top]
        c_min_top = c_vals.min()
        
        if c_max - c_min_top < 1e-12:
            w = np.ones(top_k) / top_k
        else:
            # Exponential weighting
            beta = 20.0
            w = np.exp((c_vals - c_min_top) * beta / max(c_max - c_min_top, 1e-15))
            w /= w.sum()

        grad = np.zeros(n)
        for pos, k in enumerate(idx_top):
            wp = w[pos]
            if wp < 1e-12:
                continue
            d = k - n_half
            if d == 0:
                grad += wp * (1.0 - 2.0 * h)
            elif d > 0:
                valid = n - d
                grad[:valid] += wp * (1.0 - h[d:d+valid])
                grad[d:] -= wp * h[:valid]
            else:
                a = -d
                valid = n - a
                grad[a:] += wp * (1.0 - h[:valid])
                grad[:valid] -= wp * h[a:a+valid]

                best_c5 = c5_new

                improved = True
                alpha = min(alpha * 1.1, 0.1)
                break
            alpha *= 0.6

            alpha *= 0.5

    start = time.time()

    dx = 2.0 / n

        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        beta = 500.0
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        return (m + np.log(np.sum(np.exp(z))) / beta) * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if time.time() - start < time_limit:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h0) - 1e-14:
                return h_new, c5_new
    except Exception:
        pass
    return h0, compute_c5(h0)

def compute_c5(h):
    n = len(h)

def project_feasible(h, n):
    h = np.array(h, dtype=float).flatten()
    target = n / 2.0
    s = np.sum(np.clip(h, 0.0, 1.0))
    if abs(s - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def thue_morse(n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

def slsqp_refine(h, n, maxiter=400, time_limit=40):
    global _slq_n
    start = time.time()
    h0 = project_feasible(h, n)
    _slq_n = n
    dx = 2.0 / n

    def sobj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        beta = 1000.0
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        return (m + np.log(np.sum(np.exp(z))) / beta) * dx

    def sgrad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        beta = 1000.0
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        g = np.zeros(n)
        n_half = n - 1
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-14:
                continue
            off = k - n_half
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:np_]
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_eq_fun, 'jac': _slq_eq_jac}]

    try:
        res = minimize(sobj, h0, jac=sgrad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() - start < time_limit:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h0)
            if c5_new < c5_old - 1e-14:
                return h_new, c5_new
    except Exception:
        pass
    return h0, compute_c5(h0)

                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]

        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:

        d = -g_dir / gn
