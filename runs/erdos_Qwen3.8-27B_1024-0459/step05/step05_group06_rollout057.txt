            if np_ > 0:
                grad[:np_] += wk * (1.0 - h[ao:ao+np_])

            else:
                h[np.argmax(h)] = max(0.0, h[np.argmax(h)] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(free.sum())
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)


def smooth_max_val(x):
    n = _OPT_N
    beta = _OPT_BETA
    dx = 2.0 / n
    corr = compute_corr(x)
    m = corr.max()
    sh = np.minimum(beta * (corr - m), 0.0)
    ev = np.exp(sh)
    ws = ev.sum()
    return (m + np.log(ws) / beta) * dx


def smooth_max_grad(x):
    n = _OPT_N
    beta = _OPT_BETA
    dx = 2.0 / n
    corr = compute_corr(x)
    m = corr.max()
    sh = np.minimum(beta * (corr - m), 0.0)
    ev = np.exp(sh)
    w = ev / ev.sum()
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        wk = w[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            if np_ > 0:
                grad[off:off + np_] += wk * (1.0 - x[:np_])
                grad[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            if np_ > 0:
                grad[:np_] += wk * (1.0 - x[ao:ao + np_])
                grad[ao:ao + np_] -= wk * x[:np_]
    return grad * dx


def _con_fun(x):
    return x.sum() - _OPT_N * 0.5


def _con_jac(x):
    return np.ones(len(x))


def slsqp_opt(h, n, beta, maxiter=400):
    global _OPT_N, _OPT_BETA
    _OPT_N = n
    _OPT_BETA = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(smooth_max_val, h, jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def coord_desc(h, n, n_sweeps=80):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        for i in range(n):
            if remaining_time() < 1:
                break
            t0 = best[i]
            a = np.zeros(ncorr)
            b_arr = np.zeros(ncorr)
            for off in range(-(n - 1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] = 1.0
                    b_arr[k] = -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            base = best_corr - a * t0 - b_arr * t0 * t0

            cands = [0.0, 1.0, t0]
            for d in [-0.5, -0.3, -0.15, -0.05, 0.05, 0.15, 0.3, 0.5]:
                cands.append(max(0.0, min(1.0, t0 + d)))
            cands.append(min(1.0, max(0.0, 1.0 - t0)))

            best_f = (base + a * t0 + b_arr * t0 * t0).max() * dx
            best_t = t0
            for t in cands:
                if t < 0.0 or t > 1.0:
                    continue
                f = (base + a * t + b_arr * t * t).max() * dx
                if f < best_f:
                    best_f = f
                    best_t = t

            if best_t != t0:
                lo = max(0.0, best_t - 0.5)
                hi = min(1.0, best_t + 0.5)
                if hi - lo > 1e-12:
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = (base + a * c1 + b_arr * c1 * c1).max() * dx
                    f2 = (base + a * c2 + b_arr * c2 * c2).max() * dx
                    for _ in range(80):
                        if hi - lo < 1e-14:
                            break
                        if f1 < f2:
                            hi, c2, f2 = c2, c1, f1
                            c1 = hi - gr * (hi - lo)
                            f1 = (base + a * c1 + b_arr * c1 * c1).max() * dx
                        else:
                            lo, c1, f1 = c1, c2, f2
                            c2 = lo + gr * (hi - lo)
                            f2 = (base + a * c2 + b_arr * c2 * c2).max() * dx
                    t_opt = (lo + hi) * 0.5
                    f_opt = (base + a * t_opt + b_arr * t_opt * t_opt).max() * dx
                    if f_opt < best_f:
                        best_f = f_opt
                        best_t = t_opt

            if best_f < best_c5 - 1e-15:
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True

        if not improved:
            break

    return best, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    fp = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(fp, xp, src)


def minimax_iter(h, n, n_iters=300):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    lam = np.ones(ncorr) / ncorr
    eta = 0.3
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    for it in range(n_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        losses = corr * dx

        grad = np.zeros(n)
        for k in range(ncorr):
            lk = lam[k]
            if lk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    grad[off:off + np_] += lk * (1.0 - h[:np_])
                    grad[:np_] -= lk * h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ > 0:
                    grad[:np_] += lk * (1.0 - h[ao:ao + np_])
                    grad[ao:ao + np_] -= lk * h[:np_]
        grad *= dx

        step = 0.05 / (1.0 + 0.02 * it)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        lam = lam * np.exp(-eta * losses)
        lam_sum = lam.sum()
        if lam_sum > 0:
            lam /= lam_sum
        else:
            lam = np.ones(ncorr) / ncorr
        h = h_new

    return best_h, best_c5


def binary_local_search(h, n, max_iters=3000):
    best = h.copy()
    best_c5 = compute_c5(best)
    rng_ls = np.random.RandomState(777)
    n_ones = int(n * 0.5)

    for it in range(max_iters):
        if remaining_time() < 3:
            break
        # Random pair of one and zero positions
        ones = np.where(best > 0.5)[0]
        zeros = np.where(best < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng_ls.randint(len(ones))]
        j = zeros[rng_ls.randint(len(zeros))]

        trial = best.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        trial = project_feasible(trial, n)
        c5_t = compute_c5(trial)
        if c5_t < best_c5 - 1e-14:
            best = trial
            best_c5 = c5_t

    return best, best_c5


def push_binary(h, n):
    """Push h toward binary (0/1) while maintaining sum constraint."""
    b = (h > 0.5).astype(np.float64)
    cnt = int(b.sum())
    target = n * 0.5
    if cnt > target:
        # Need to turn some 1s to 0s
        excess = cnt - target
        idx = np.where(b == 1.0)[0]
        order = idx[np.argsort(h[idx])[:excess]]
        b[order] = 0.0
    elif cnt < target:
        deficit = target - cnt
        idx = np.where(b == 0.0)[0]
        order = idx[np.argsort(-h[idx])[:deficit]]
        b[order] = 1.0
    return b


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s

    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Get initial solution
    try:
        parent_h = np.asarray(initial_h_values, dtype=np.float64)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        rng0 = np.random.RandomState(0)
        h0 = np.zeros(best_n)
        h0[rng0.choice(best_n, best_n // 2, replace=False)] = 1.0
        best_h = project_feasible(h0, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with increasing beta
    for beta in [2000, 10000, 50000, 200000]:
        if remaining_time() < 20:
            break
        h = slsqp_opt(h, n, beta, 500)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Intensive coordinate descent
    if remaining_time() > 60:
        h, c5 = coord_desc(h, n, n_sweeps=120)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Minimax iteration
    if remaining_time() > 60:
        h_mm, c5_mm = minimax_iter(best_h, best_n, 400)
        if remaining_time() > 10:
            h_mm = slsqp_opt(h_mm, best_n, 100000, 200)
            c5_mm = compute_c5(h_mm)
            h_mm, _ = coord_desc(h_mm, best_n, n_sweeps=40)
            c5_mm = compute_c5(h_mm)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()

    # Phase 4: More SLSQP at very high beta
    for beta in [1000000, 5000000]:
        if remaining_time() < 12:
            break
        h = slsqp_opt(best_h, best_n, beta, 300)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Coordinate descent again
    if remaining_time() > 30:
        h, c5 = coord_desc(best_h, best_n, n_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Try different resolutions
    resolutions_to_try = []
    base_n = best_n
    for mult in [0.7, 0.8, 0.9, 1.1, 1.25, 1.4, 1.6, 2.0]:
        rn = int(base_n * mult)
        if rn % 2 != 0:
            rn += 1
        if 128 <= rn <= 2000:
            resolutions_to_try.append(rn)
    for extra in [64, 128, 256]:
        rn = base_n + extra
        if rn % 2 != 0:
            rn += 1
        if 128 <= rn <= 2000:
            resolutions_to_try.append(rn)
    resolutions_to_try = sorted(set(resolutions_to_try))

    for rn in resolutions_to_try:
        if remaining_time() < 55:
            break
        if rn == best_n:
            continue
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        if remaining_time() > 15:
            h_r = slsqp_opt(h_r, rn, 20000, 300)
        if remaining_time() > 15:
            h_r = slsqp_opt(h_r, rn, 200000, 200)
        if remaining_time() > 20:
            h_r, c5_r = coord_desc(h_r, rn, n_sweeps=30)
        else:
            c5_r = compute_c5(h_r)

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 7: Binary push + re-optimize
    if remaining_time() > 50:
        h_bin = push_binary(best_h, best_n)
        h_bin = project_feasible(h_bin, best_n)
        h_bin = slsqp_opt(h_bin, best_n, 50000, 300)
        h_bin, c5_bin = coord_desc(h_bin, best_n, n_sweeps=40)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 8: Binary local search
    if remaining_time() > 30:
        h_bl, c5_bl = binary_local_search(best_h, best_n, 2000)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
        # Refine after binary search
        if remaining_time() > 15:
            h_ref = slsqp_opt(best_h, best_n, 200000, 200)
            h_ref, c5_ref = coord_desc(h_ref, best_n, n_sweeps=40)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Phase 9: Another minimax round
    if remaining_time() > 40:
        h_mm2, c5_mm2 = minimax_iter(best_h, best_n, 200)
        if remaining_time() > 10:
            h_mm2 = slsqp_opt(h_mm2, best_n, 500000, 200)
            h_mm2, c5_mm2 = coord_desc(h_mm2, best_n, n_sweeps=30)
            c5_mm2 = compute_c5(h_mm2)
        if c5_mm2 < best_c5:
            best_c5 = c5_mm2
            best_h = h_mm2.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 15:
        h_f, c5_f = coord_desc(best_h, best_n, n_sweeps=80)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 11: Final SLSQP polish
    if remaining_time() > 8:
        h_p = slsqp_opt(best_h, best_n, 10000000, 150)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Final coordinate descent
    if remaining_time() > 5:
        h_f2, c5_f2 = coord_desc(best_h, best_n, n_sweeps=30)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint
    target = best_n * 0.5
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        free = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        cnt = int(free.sum())
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
