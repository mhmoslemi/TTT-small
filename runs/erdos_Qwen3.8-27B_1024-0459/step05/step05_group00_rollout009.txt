def nesterov_descent(h, n, time_limit, K=25, beta=80.0):
    """Nesterov accelerated projected gradient descent."""
    h = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    t = 1.0
    lr = 0.1
    vel = np.zeros(n)
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break
        
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        momentum = (t - 1) / t_new
        
        g, c_max, _ = top_k_gradient(y, n, K=K, beta=beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Adaptive step
        step = lr / (1 + 0.01 * t)
        y_new = y - step * g / gn
        h_new = project_feasible(y_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        y = momentum * y + (1 - momentum) * h_new
        t = t_new
        lr *= 0.999
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:

def coordinate_optimize(h, n, time_limit, K=20):
    """Cyclic coordinate descent - for each i, optimize h[i]."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break
        
        improved = False
        # Get gradient once
        g, c_max, idx = top_k_gradient(h, n, K=K, beta=100.0)
        
        for i in range(0, n, max(1, n // 512)):  # Sample some coordinates
            remaining = time_limit - (time.time() - start)
            if remaining < 1:
                break
            
            gi = g[i]
            if abs(gi) < 1e-12:
                continue
            
            # Try small moves
            for delta_i in [-0.01, 0.01, -0.005, 0.005, -0.02, 0.02]:
                if h[i] + delta_i < 0 or h[i] + delta_i > 1:
                    continue
                h_old = h[i]
                h[i] += delta_i
                # Adjust sum: distribute -delta_i among other indices
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = np.where(mask & (h > 1e-8) & (h < 1 - 1e-8))[0]
                if len(others) == 0:
                    h[i] = h_old
                    continue
                adj = -delta_i / len(others)
                h[others] += adj
                h = np.clip(h, 0, 1)
                c5_try = compute_c5(h)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] = h_old
                    h[others] -= adj
            
        if not improved:
            break
    
    return h, best_c5
