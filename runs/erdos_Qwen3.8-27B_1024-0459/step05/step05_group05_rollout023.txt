def interp_to(src_h, src_n, dst_n):

    s = np.sum(ex)

    return h_bin

def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rng.shuffle(ones_idx)
            h[ones_idx[:n_ones - n_half]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                rng.shuffle(zeros_idx)
            h[zeros_idx[:n_half - n_ones]] = 1
    return h

def corr_gradient_at(h, k_off, n):
    g = np.zeros(n)
    if k_off >= 0:
        np_ = n - k_off
        g[k_off:k_off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def binary_sa(h_bin, n, rng, time_limit=60, T_init=8.0):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    T = T_init
    T_min = 0.05
    cooling = 0.9992
    no_improve = 0
    max_no_improve = 30000
    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > max_no_improve:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0
        h[j] = 1
        new_max = float(np.max(compute_corr(h)))
        delta = new_max - cur_max
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            h[i] = 0
            h[j] = 1
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                no_improve = 0
        else:
            h[i] = 1
            h[j] = 0
        T = max(T * cooling, T_min)
    return best_h, best_max * dx_v

def binary_multiswap(h_bin, n, rng, time_limit=40):
    h = (h_bin > 0.5).astype(float).copy()
    dx_v = 2.0 / n
    cur_max = float(np.max(compute_corr(h)))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    stall = 0
    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h)
        max_val = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        top_idx = np.argsort(-corr)[:10]
        best_delta = 0.0
        best_move = None
        for ki in top_idx[:6]:
            off = ki - (n - 1)
            if off >= 0:
                np_ = n - off
                faults = np.where((h[:np_] == 1) & (h[off:off+np_] == 0))[0]
            else:
                ao = -off
                np_ = n - ao
                faults = np.where((h[ao:ao+np_] == 1) & (h[:np_] == 0))[0]
            for fi in faults[:min(len(faults), 15)]:
                if off >= 0:
                    i_pos = fi
                else:
                    i_pos = fi + ao
                if i_pos < 0 or i_pos >= n or h[i_pos] != 1:
                    continue
                for _ in range(4):
                    j_pos = zeros[rng.randint(len(zeros))]
                    if j_pos == i_pos:
                        continue
                    h[i_pos] = 0
                    h[j_pos] = 1
                    nc = float(np.max(compute_corr(h)))
                    delta = nc - max_val
                    h[i_pos] = 1
                    h[j_pos] = 0
                    if delta < best_delta - 1e-12:
                        best_delta = delta
                        best_move = (i_pos, j_pos, 1)
        for _ in range(30):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-12:
                best_delta = delta
                best_move = (i, j, 1)
        # Try 2-swap (move two ones)
        if len(ones) > 1 and len(zeros) > 1:
            for _ in range(15):
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                if i1 == i2:
                    continue
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if j1 == j2 or j1 == i1 or j2 == i2:
                    continue
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
                nc = float(np.max(compute_corr(h)))
                delta = nc - max_val
                h[i1] = 1
                h[i2] = 1
                h[j1] = 0
                h[j2] = 0
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_move = (i1, i2, 2, j1, j2)
        if best_move is not None:
            if best_move[2] == 1:
                h[best_move[0]] = 0
                h[best_move[1]] = 1
            else:
                h[best_move[0]] = 0
                h[best_move[1]] = 0
                h[best_move[3]] = 1
                h[best_move[4]] = 1
            cur_max = float(np.max(compute_corr(h)))
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > 800:
                nf = rng.randint(2, 5)
                for _ in range(nf):
                    ones2 = np.where(h == 1)[0]
                    zeros2 = np.where(h == 0)[0]
                    if len(ones2) == 0 or len(zeros2) == 0:
                        break
                    a = ones2[rng.randint(len(ones2))]
                    b = zeros2[rng.randint(len(zeros2))]
                    h[a] = 0
                    h[b] = 1
                cur_max = float(np.max(compute_corr(h)))
                stall = 0
    return best_h, best_max * dx_v

def top_shifts_focus(h, n, top_k=5, max_iters=400):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_comb = np.zeros(n)
        for ki in idx:
            off = ki - (n - 1)
            g_comb += corr_gradient_at(h, off, n)
        # Subtract gradient of min-top to equalize
        off_bot = idx[-1] - (n - 1)
        g_bot = corr_gradient_at(h, off_bot, n)
        d = g_comb - top_k * g_bot
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_comb
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for step in [0.01, 0.03, 0.05, 0.02, 0.005, 0.08, 0.1, 0.015]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try opposite direction
            for step in [0.01, 0.03, 0.05, 0.02]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def subgradient_max(h, n, n_iter=200):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 0.5
        top_idx = np.where(corr >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(corr)])
        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            g += corr_gradient_at(h, off, n)
        g /= len(top_idx)
        g -= np.mean(g)
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.88
            if step < 0.002:
                break
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Progressive SLSQP with increasing beta
    for beta in [30, 100, 500, 2000, 10000, 50000, 200000, 800000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=700)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h = h_new.copy()

    # Phase 2: L-BFGS-B refinement
    if remaining() > 40:
        h_lb = lbfgs_smooth(h, n, 30000, maxiter=5000)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()
            h = h_lb.copy()

    # Phase 3: Top-shifts equalization (multiple configs)
    for tk in [3, 5, 8, 12]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Binary SA (multiple temperatures)
    if remaining() > 50:
        h_bin = binarize(h, n)
        for T_init in [8.0, 4.0, 2.0]:
            if remaining() < 15:
                break
            sa_time = min(80, int(remaining() * 0.3))
            h_b, c5_b = binary_sa(h_bin, n, rng, time_limit=sa_time, T_init=T_init)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n
                h_b2, c5_b2 = binary_sa(h_b, n, rng, time_limit=min(30, int(remaining()*0.2)), T_init=1.0)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_b2.copy()
                h = best_h.copy()
            else:
                h = best_h.copy()

    # Phase 5: Multi-swap binary search
    if remaining() > 40:
        h_bin = binarize(best_h, best_n)
        h_ms, c5_ms = binary_multiswap(h_bin, best_n, rng, time_limit=min(50, int(remaining()*0.4)))
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()
            best_n = best_n
            # Refine continuously
            h_c = slsqp_smooth(h_ms, best_n, 50000, 600)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Phase 6: Subgradient
    if remaining() > 15:
        h_sg, c5_sg = subgradient_max(best_h, best_n, n_iter=250)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 7: Diverse starts
    if remaining() > 100:
        # Thue-Morse
        h_tm = thue_morse(n, rng)
        h_tm = slsqp_smooth(h_tm, n, 5000, 500)
        h_tm = slsqp_smooth(h_tm, n, 50000, 500)
        h_tm, c5_tm = top_shifts_focus(h_tm, n, 5, 250)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            best_n = n

        # Inverse Thue-Morse
        h_tm2 = 1.0 - h_tm
        h_tm2 = project_feasible(h_tm2, n)
        h_tm2 = slsqp_smooth(h_tm2, n, 50000, 500)
        c5_tm2 = compute_c5(h_tm2)
        if c5_tm2 < best_c5:
            best_c5 = c5_tm2
            best_h = h_tm2.copy()
            best_n = n

        # Perturbed best
        h_p = best_h + rng.randn(n) * 0.2
        h_p = project_feasible(h_p, n)
        h_p = slsqp_smooth(h_p, n, 20000, 600)
        h_p, c5_p = top_shifts_focus(h_p, n, 5, 200)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = n

        # Random starts
        for rs in range(3):
            if remaining() < 20:
                break
            h_r = rng.dirichlet(np.ones(n)) * (n / n)
            h_r = np.clip(h_r * n / np.sum(h_r) * (n/n), 0, 1)
            h_r = project_feasible(rng.uniform(0, 1, n), n)
            h_r = slsqp_smooth(h_r, n, 10000, 500)
            h_r, c5_r = top_shifts_focus(h_r, n, 5, 200)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n

    # Phase 8: Resolution sweep
    if remaining() > 80:
        n_base = best_n
        n_candidates = []
        for nt in [n_base-32, n_base-16, n_base-8, n_base, n_base+8, n_base+16, n_base+32, n_base+64]:
            if 128 <= nt <= 800 and nt % 2 == 0:
                if nt not in n_candidates:
                    n_candidates.append(nt)
        for nt in [256, 384, 512, 640]:
            if 128 <= nt <= 800 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:8]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            if compute_c5(h_t) < best_c5:
                best_c5 = compute_c5(h_t)
                best_h = h_t.copy()
                best_n = n_t
            for beta in [5000, 50000, 200000]:
                if remaining() < 10:
                    break
                h_t = slsqp_smooth(h_t, n_t, beta, 500)
            c5_t = compute_c5(h_t)
            h_t, c5_t = top_shifts_focus(h_t, n_t, 5, 200)
            if remaining() > 20:
                h_t_bin = binarize(h_t, n_t)
                h_tb, c5_tb = binary_sa(h_t_bin, n_t, rng, time_limit=min(25, int(remaining()*0.3)), T_init=4.0)
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb.copy()
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 9: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for beta in [50000, 200000, 800000, 2000000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    for tk in [3, 5, 7, 10, 15]:
        if remaining() < 5:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary SA
    if remaining() > 30:
        h_bin = binarize(best_h, best_n)
        h_b, c5_b = binary_sa(h_bin, best_n, rng, time_limit=min(60, int(remaining()*0.5)), T_init=3.0)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = slsqp_smooth(h_b, best_n, 200000, 700)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Final multi-swap
    if remaining() > 20:
        h_bin = binarize(best_h, best_n)
        h_ms, c5_ms = binary_multiswap(h_bin, best_n, rng, time_limit=min(25, int(remaining()*0.4)))
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Final subgradient
    if remaining() > 10:
        h_sg, c5_sg = subgradient_max(best_h, best_n, n_iter=200)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final top-shifts
    if remaining() > 8:
        h_eq, c5_eq = top_shifts_focus(best_h, best_n, top_k=5, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
