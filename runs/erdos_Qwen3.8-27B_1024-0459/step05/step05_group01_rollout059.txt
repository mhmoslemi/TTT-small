        return np.correlate(h, 1.0 - h, mode='full')

    H = np.fft.rfft(h, size=size)

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = corr_full(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = corr_full(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):

        if wk < 1e-15:

        else:

            # Try anti-grad direction
            for step in [0.003, 0.006, 0.01, 0.02, 0.04]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def random_balanced(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def soft_max_opt(h, n, target_c5, max_iter=200):
    """Use penalty/hinge approach to push below target"""
    global _CUR_N, _SM_BETA
    _CUR_N = n
    dx = 2.0 / n
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

    def hinge_obj(x):
        corr = corr_full(x)
        t = target_c5 / dx
        violations = np.maximum(corr - t, 0.0)
        return np.sum(violations) + 0.001 * np.sum(violations**2)

    def hinge_grad(x):
        corr = corr_full(x)
        t = target_c5 / dx
        viol = (corr > t).astype(float)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if viol[k] == 0:
                continue
            w = 1.0 + 0.002 * (corr[k] - t)
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += w * (1.0 - x[:np_])
                g[:np_] -= w * x[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += w * (1.0 - x[a:])
                g[a:] -= w * x[:np_]
        return g

    try:
        res = minimize(hinge_obj, h, jac=hinge_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 50})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h


def coordinate_refine(h, n, max_sweeps=20):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = h[i]
            if old <= 0.001 or old >= 0.999:
                continue
            cands = [0.0, 1.0, old*0.5, 1.0-old*0.5, max(0,old-0.25), min(1,old+0.25)]
            best_val = old
            best_c = best_c5
            for c in cands:
                if c < 0 or c > 1:
                    continue
                h[i] = c
                h = project_feasible(h, n)
                cv = compute_c5(h)
                if cv < best_c - 1e-14:
                    best_c = cv
                    best_val = c
                    improved = True
            h[i] = best_val
            h = project_feasible(h, n)
            if improved and best_c < best_c5:
                best_c5 = best_c
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP on parent (fast convergence)
    h = best_h.copy()
    for beta in [500, 2000, 8000, 30000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Binary local search with incremental updates
    if remaining_time() > 60:
        h_bin = (h > 0.5).astype(float)
        sa_time = min(120, int(remaining_time() * 0.25))
        h_b, c5_b = binary_local_search_fast(h_bin, best_n, sa_time, rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Continuous refinement after binary
    h = best_h.copy()
    for beta in [10000, 50000, 200000, 800000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Top-k equalization
    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=6, max_iters=1000)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 5: Hinge/penalty approach
    if remaining_time() > 40:
        target = best_c5 - 0.0001
        h_hinge = soft_max_opt(h, best_n, target, max_iter=300)
        c5_h = compute_c5(h_hinge)
        if c5_h < best_c5 - 1e-15:
            best_c5 = c5_h
            best_h = h_hinge.copy()
            h = h_hinge.copy()

    # Phase 6: Multi-resolution
    n_trials = sorted(set([best_n, best_n+64, best_n+128, best_n+256,
                           512, 640, 768, 1024]),
                      key=lambda x: abs(x - best_n))
    n_trials = [x for x in n_trials if 200 <= x <= 1200 and x % 2 == 0]

    for n_t in n_trials:
        if remaining_time() < 30:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_opt = h_feas
        for beta in [2000, 10000, 50000]:
            if remaining_time() < 12:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if remaining_time() > 10:
            h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=5, max_iters=400)

        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 7: Alternative starting points
    alts = [theta_morse(best_n), random_balanced(best_n, rng)]
    for h_alt in alts:
        if remaining_time() < 35:
            break
        h_opt = h_alt
        for beta in [3000, 20000, 100000]:
            if remaining_time() < 12:
                break
            h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=400)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=5, max_iters=400)
        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 8: More binary search from current best
    if remaining_time() > 50:
        h_bin = (h > 0.5).astype(float)
        sa_time = min(100, int(remaining_time() * 0.2))
        h_b, c5_b = binary_local_search_fast(h_bin, best_n, sa_time, rng)
        # Refine continuously
        h_br = project_feasible(h_b, best_n)
        for beta in [10000, 100000, 500000]:
            if remaining_time() < 10:
                break
            h_br = slsqp_opt(h_br, best_n, beta, maxiter=500)
        c5_br = compute_c5(h_br)
        h_br, c5_br = top_k_equalize(h_br, best_n, top_k=6, max_iters=500)
        if c5_br < best_c5 - 1e-15:
            best_c5 = c5_br
            best_h = h_br.copy()
            h = h_br.copy()
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 9: Final intensive SLSQP polish
    n = best_n
    h = best_h.copy()
    for beta in [200000, 1000000, 5000000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=8, max_iters=1000)
        if c5_eq < best_c5 - 1e-16:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 10: Coordinate refinement
    if remaining_time() > 12:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=15)
        if c5_cr < best_c5 - 1e-16:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 3000000, maxiter=2000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-16:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
