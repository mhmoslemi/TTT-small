import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def smooth_obj(h, beta):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / beta) * dx

def smooth_grad(h, beta):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        np_ = n - abs(off)
        if off >= 0:
            g[:np_] += wk * (1.0 - h[off:off + np_])
            g[off:] -= wk * h[:np_]
        else:
            ao = -off
            g[ao:] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[ao:ao + np_]
    return g

def lbfgsb_optimize(h, n, beta, max_iter=400, penalty=5000.0):
    global _N, _BETA
    _N = n
    _BETA = beta
    h0 = project_feasible(h, n)
    
    def obj(x):
        v = smooth_obj(x, beta)
        s = np.sum(x) - n / 2.0
        return v + penalty * s * s
    
    def grad(x):
        g = smooth_grad(x, beta)
        g += 2.0 * penalty * (np.sum(x) - n / 2.0) * np.ones(n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                                'maxls': 30, 'maxfun': max_iter * 3})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA
    _N = n
    _BETA = beta
    h0 = project_feasible(h, n)
    
    def obj(x):
        return smooth_obj(x, beta)
    
    def grad(x):
        return smooth_grad(x, beta)
    
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.success or (res.x is not None and res.fun < obj(h0)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def top_shifts_focus(h, n, top_k=5, max_iters=300):
    h = project_feasible(h, n).copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        if c_max < 1e-10:
            break
        
        k_top = idx[0]
        k_bot = idx[-1]
        off_t = k_top - (n - 1)
        off_b = k_bot - (n - 1)
        
        def corr_g(off):
            g = np.zeros(n)
            np_ = n - abs(off)
            if off >= 0:
                g[:np_] += (1.0 - h[off:off+np_])
                g[off:] -= h[:np_]
            else:
                ao = -off
                g[ao:] += (1.0 - h[:np_])
                g[:np_] -= h[ao:ao+np_]
            return g
        
        g_t = corr_g(off_t)
        g_b = corr_g(off_b)
        d = g_b - g_t
        d -= np.mean(d)  # project to tangent
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        
        improved = False
        for step in [0.03, 0.05, 0.01, 0.08, 0.02, 0.1, 0.005]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                best_h = ht.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgradient_max(h, n, n_iter=200):

    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.04
    
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 0.3
        top_mask = corr >= threshold
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = np.array([np.argmax(corr)])
        
        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            np_ = n - abs(off)
            if off >= 0:
                g[:np_] += (1.0 - h[off:off+np_])
                g[off:] -= h[:np_]
            else:
                ao = -off
                g[ao:] += (1.0 - h[:np_])
                g[:np_] -= h[ao:ao+np_]
        g /= len(top_idx)
        g -= np.mean(g)
        
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-16:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            step = min(step * 1.1, 0.15)
        else:
            step *= 0.85
            if step < 0.002:
                break
    return best_h, best_c5

def binarize(h, n):

def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):

        h_p = project_feasible(h_p, n_cur)
        h_p = lbfgsb_optimize(h_p, n_cur, 20000, max_iter=400, penalty=10000.0)
        h_p, c5_p = top_shifts_focus(h_p, n_cur, top_k=5, max_iters=150)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = n_cur
    
    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    
    for beta in [20000, 80000, 200000, 500000]:
        if remaining() < 8:
            break
        h_new = lbfgsb_optimize(h, n, beta, max_iter=600, penalty=50000.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    for beta in [30000, 100000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Final top-shifts
    for tk in [3, 5, 7, 10]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Final binary search
    if remaining() > 30:
        h_bin = binarize(h, n)
        h_b, c5_b = binary_local_search(h_bin, n, rng, time_limit=min(50, int(remaining() * 0.5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = lbfgsb_optimize(h_b, n, 50000, max_iter=400, penalty=50000.0)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
    
    # Final subgradient
    if remaining() > 12:
        h_sg, c5_sg = subgradient_max(best_h, best_n, n_iter=150)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < best_c5:
                best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
