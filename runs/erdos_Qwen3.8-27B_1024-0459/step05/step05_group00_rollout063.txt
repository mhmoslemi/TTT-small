        lag_info.append((k, n_w_total, cnt))
        n_w_total += cnt

    if n_w_total == 0:
        return h.copy(), compute_c5(h), False

    nv = n + n_w_total + 1
    dx = 2.0 / n
    t_idx = nv - 1

    c_obj = np.zeros(nv)
    c_obj[t_idx] = 1.0

    ar = []
    ac = []
    av = []
    b_ub_list = []

    # Lag constraints: sum(w_{i,k}) * dx <= t
    for (k, w0, cnt) in lag_info:
        ri = len(b_ub_list)
        for i in range(cnt):
            ar.append(ri)
            ac.append(n + w0 + i)
            av.append(dx)
        ar.append(ri)
        ac.append(t_idx)
        av.append(-1.0)
        b_ub_list.append(0.0)

    # McCormick constraints for each w
    for (k, w0, cnt) in lag_info:
        m = abs(k)
        for i in range(cnt):
            wi = n + w0 + i
            if k > 0:
                ui = i
                vi = i + k
            else:
                ui = i + m
                vi = i
            # w - h[u] <= 0
            ri = len(b_ub_list)
            ar.append(ri); ac.append(wi); av.append(1.0)
            ar.append(ri); ac.append(ui); av.append(-1.0)
            b_ub_list.append(0.0)
            # w + h[v] <= 1
            ri = len(b_ub_list)
            ar.append(ri); ac.append(wi); av.append(1.0)
            ar.append(ri); ac.append(vi); av.append(1.0)
            b_ub_list.append(1.0)
            # -w + h[u] - h[v] <= 0
            ri = len(b_ub_list)
            ar.append(ri); ac.append(wi); av.append(-1.0)
            ar.append(ri); ac.append(ui); av.append(1.0)
            ar.append(ri); ac.append(vi); av.append(-1.0)
            b_ub_list.append(0.0)

    A_ub = csr_matrix((av, (ar, ac)), shape=(len(b_ub_list), nv))
    b_ub = np.array(b_ub_list)

    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0.0, 1.0)] * n + [(0.0, None)] * n_w_total + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy(), float(res.x[t_idx]), True
    except Exception:
        pass

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='revised simplex')
        if res.success:
            return res.x[:n].copy(), float(res.x[t_idx]), True
    except Exception:
        pass

    return h.copy(), compute_c5(h), False


def _sm_obj(x):
    n = _SM_N
    beta = _SM_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = beta * (corr - m)
    cl = np.clip(sh, -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(cl))) / beta) * (2.0 / n)


def _sm_grad(x):
    n = _SM_N
    beta = _SM_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = beta * (corr - m)
    cl = np.clip(sh, -50.0, 50.0)
    ex = np.exp(cl)
    s = ex.sum()
    if s < 1e-30:
        return np.zeros(n)
    wc = (ex / s) * (2.0 / n)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:])
            g[ao:ao + np_] -= wk * x[:np_]
    return g


def _sm_eq(x):
    return np.sum(x) - _SM_N / 2.0


def _sm_eq_jac(x):
    return np.ones(_SM_N)


def slsqp_sm(h, n, beta, maxiter):
    global _SM_N, _SM_BETA
    _SM_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sm_eq, 'jac': _sm_eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def subgrad(h, n, time_limit, rng):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 10:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:20]
        c_top = corr[idx]
        wts = np.exp(50.0 * (c_top - c_top.max()))
        wts = wts / wts.sum()
        g = np.zeros(n)
        for i in range(len(idx)):
            if wts[i] > 1e-10:
                g += wts[i] * grad_corr_k(h, int(idx[i]), n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        for step in [0.1, 0.05, 0.02, 0.01, 0.005]:
            h_new = project_feasible(h - step * g / gn, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Get parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = int(len(parent_h))
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    h = best_h.copy()
    n = best_n

    # Phase 1: Quick subgradient
    if rem() > 30:
        h_sg, c5_sg = subgrad(h, n, min(20.0, rem() * 0.1), rng)
        if c5_sg < best_c5:
            best_c5, best_h, h = c5_sg, h_sg.copy(), h_sg.copy()

    # Phase 2: Initial SLSQP
    for beta in [500, 2000]:
        if rem() < 10:
            break
        h = slsqp_sm(h, n, beta, 300)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 3: LP + SLSQP iterations at parent resolution
    for iteration in range(50):
        if rem() < 40:
            break

        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:20]
        k_set = set()
        for ix in top_idx:
            k = int(ix) - (n - 1)
            if k != 0 and abs(k) < n:
                K_set_add(k_set, k)

        if len(k_set) > 0:
            h_lp, t_lp, ok = mcmorrick_lp(h, n, k_set)
            if ok:
                h_lp = project_feasible(h_lp, n)
                c5_lp = compute_c5(h_lp)
                if c5_lp < best_c5:
                    best_c5, best_h = c5_lp, h_lp.copy()
                    h = h_lp.copy()

                # Also try binary projection
                h_bp = binary_project(h_lp, n)
                c5_bp = compute_c5(h_bp)
                if c5_bp < best_c5:
                    best_c5, best_h = c5_bp, h_bp.copy()
                    h = h_bp.copy()

        if rem() > 15:
            h = slsqp_sm(h, n, 5000, 300)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()

    # Phase 4: Multi-resolution
    for mult in [0.75, 1.25, 1.5, 2.0]:
        if rem() < 60:
            break
        n_t = int(round(best_n * mult / 32) * 32)
        if n_t < 128 or n_t > 2048 or n_t == best_n:
            continue

        h_t = interp_to(best_h, best_n, n_t)
        h_t = project_feasible(h_t, n_t)
        c5_t = compute_c5(h_t)

        # Subgradient at new resolution
        if rem() > 30:
            h_t, c5_t = subgrad(h_t, n_t, min(15.0, rem() * 0.2), rng)

        # LP at new resolution
        K_lp = 12 if n_t > 768 else 16
        for lp_iter in range(10):
            if rem() < 20:
                break
            corr_t = compute_corr(h_t)
            top_t = np.argsort(-corr_t)[:K_lp]
            kt = set()
            for ix in top_t:
                k = int(ix) - (n_t - 1)
                if k != 0 and abs(k) < n_t:
                    K_set_add(kt, k)
            if len(kt) > 0:
                h_lp_t, t_lp_t, ok_t = mcmorrick_lp(h_t, n_t, kt)
                if ok_t:
                    h_lp_t = project_feasible(h_lp_t, n_t)
                    c5_lp_t = compute_c5(h_lp_t)
                    if c5_lp_t < c5_t:
                        c5_t = c5_lp_t
                        h_t = h_lp_t.copy()
                    h_bp_t = binary_project(h_lp_t, n_t)
                    c5_bp_t = compute_c5(h_bp_t)
                    if c5_bp_t < c5_t:
                        c5_t = c5_bp_t
                        h_t = h_bp_t.copy()

            if rem() > 10:
                h_t = slsqp_sm(h_t, n_t, 5000, 300)
                c5_t = compute_c5(h_t)

        if c5_t < best_c5:
            best_c5, best_h, best_n = c5_t, h_t.copy(), n_t
            h = h_t.copy()
            n = n_t

    # Phase 5: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    K_lp = 16 if n <= 512 else 12

    for cycle in range(30):
        if rem() < 30:
            break

        corr = compute_corr(h)
        top = np.argsort(-corr)[:K_lp]
        kc = set()
        for ix in top:
            k = int(ix) - (n - 1)
            if k != 0 and abs(k) < n:
                K_set_add(kc, k)

        if len(kc) > 0:
            h_lp, t_lp, ok = mcmorrick_lp(h, n, kc)
            if ok:
                h_lp = project_feasible(h_lp, n)
                c5_lp = compute_c5(h_lp)
                if c5_lp < best_c5:
                    best_c5, best_h = c5_lp, h_lp.copy()
                    h = h_lp.copy()
                h_bp = binary_project(h_lp, n)
                c5_bp = compute_c5(h_bp)
