def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def coord_optimize_single(h, n, i, corr_cur):
    """Optimize coordinate h[i] given current correlation array."""
    dx = 2.0 / n
    ncorr = 2 * n - 1
    t0 = h[i]
    a = np.zeros(ncorr)
    b = np.zeros(ncorr)
    for off in range(-(n - 1), n):
        k = n - 1 + off
        if off == 0:
            a[k] += 1.0
            b[k] += -1.0
        else:
            if 0 <= i + off < n:
                a[k] += (1.0 - h[i + off])
            if 0 <= i - off < n:
                a[k] += (-h[i - off])
    base = corr_cur - a * t0
    def eval_t(t):
        return np.max(base + a * t + b * t * t) * dx

    # Strategic initial candidates
    cands = [0.0, 1.0, t0, 1.0 - t0,
             max(0.0, t0 - 0.3), min(1.0, t0 + 0.3),
             max(0.0, t0 - 0.1), min(1.0, t0 + 0.1),
             max(0.0, t0 - 0.05), min(1.0, t0 + 0.05),
             max(0.0, t0 - 0.01), min(1.0, t0 + 0.01)]
    best_t = t0
    best_f = eval_t(t0)
    for t in cands:
        if abs(t - t0) < 1e-14:
            continue
        t = np.clip(t, 0.0, 1.0)
        f = eval_t(t)
        if f < best_f - 1e-16:
            best_f = f
            best_t = t

    # Golden section refinement
    if best_t != t0:
        margin = 0.3
        lo = max(0.0, best_t - margin)
        hi = min(1.0, best_t + margin)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        f1 = eval_t(c1)
        f2 = eval_t(c2)
        for _ in range(80):
            if hi - lo < 1e-14:
                break
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                f1 = eval_t(c1)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                f2 = eval_t(c2)
        t_opt = (lo + hi) / 2.0
        f_opt = eval_t(t_opt)
        if f_opt < best_f:
            best_f = f_opt
            best_t = t_opt

    # Also try endpoints
    for t_ep in [0.0, 1.0]:
        f_ep = eval_t(t_ep)
        if f_ep < best_f:
            best_f = f_ep
            best_t = t_ep

    return best_t, best_f

def active_gradient_step(h, n, n_top=10, step_size=0.02):

    if len(active) == 0:

        active = np.argsort(corr)[::-1][:n_top]

    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)

            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]

    # Normalize
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h.copy()
    grad = grad / gnorm

            h_cd, c5_cd = coord_descent(h_new, n, n_sweeps=15)
            if c5_cd < best_c5 - 1e-16:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                h = h_cd.copy()

    # Phase 7: Try slightly different resolutions
    for rn in [best_n - 4, best_n + 4, best_n + 8]:
        if remaining_time() < 50:
            break
        if rn < 64 or rn % 2 != 0:
            continue
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        h_r = slsqp_optimize(h_r, rn, 20000, 300)
        if remaining_time() > 15:
            h_r, c5_r = coord_descent(h_r, rn, n_sweeps=25)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5 - 1e-16:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 8: SLSQP with very high beta for final precision
    n = best_n
    h = best_h.copy()
    for beta in [500000, 2000000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_optimize(h, n, beta, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 9: Final intensive coordinate descent
    if remaining_time() > 10:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=50)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One more SLSQP polish
    if remaining_time() > 6:
        h_new = slsqp_optimize(best_h, best_n, 5000000, 100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: Final coord descent
    if remaining_time() > 4:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=20)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-12:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)
        final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, final_c5, best_n)
