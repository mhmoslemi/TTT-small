            f_vals[idx] = np.max(corr_t)

        t_best = t_vals[best_idx]

            lo = t_vals[best_idx - 1]
            hi = t_vals[best_idx + 1]
            for _ in range(50):

                m2 = hi - (hi - lo) / 3
                f1 = np.max(intercepts + slopes * m1 + quad_coeff * m1 * m1)

                    lo = m1

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):

            h[mask] += diff / cnt
        elif diff > 0:
            order = np.argsort(h)
            for idx in order:
                space = 1.0 - h[idx]
                add = min(diff, space)
                h[idx] += add
                diff -= add
                if diff < 1e-15:

                sub = min(-diff, space)
                h[idx] -= sub
                diff += sub
                if diff > -1e-15:
                    break

    return h

def topm_obj_grad(h, n, m):
    """Compute top-m mean of correlations and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ncorr = len(corr)
    
    if m >= ncorr:
        top_mask = np.ones(ncorr, dtype=bool)
        m_eff = ncorr
    else:
        # Get top m
        part = np.partition(corr, ncorr - m)
        threshold = part[ncorr - m]
        top_mask = corr >= threshold - 1e-12
        m_eff = int(np.sum(top_mask))
        if m_eff == 0:
            idx = np.argmax(corr)
            top_mask = np.zeros(ncorr, dtype=bool)
            top_mask[idx] = True
            m_eff = 1
    
    val = np.sum(corr[top_mask]) / m_eff * dx
    
    grad = np.zeros(n)
    for k in np.where(top_mask)[0]:
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= h[:npairs]
    
    grad *= 1.0 / m_eff * dx
    
    # Penalty for sum constraint
    s = np.sum(h) - n / 2.0
    val += 0.5 * 100.0 * s * s
    grad += 100.0 * s
    
    return val, grad

def lbfgs_optimize(h, n, m, maxiter=500):
    """Optimize using L-BFGS-B with top-m mean objective."""
    c0 = compute_c5(h)
    
    def obj(x):
        return topm_obj_grad(x, n, m)[0]
    
    def grad(x):
        return topm_obj_grad(x, n, m)[1]
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial < c0:
                return trial
    except Exception:
        pass
    return h.copy()

def simulated_annealing_01(h, n, n_iters=5000, T_start=0.01, T_end=0.0001, seed=42):

    rng = np.random.RandomState(seed)
    best = h.copy()
    # Round to 0/1
    best[best >= 0.5] = 1.0
    best[best < 0.5] = 0.0
    best = project_feasible(best, n)
    
    current_c5 = compute_c5(best)
    best_c5 = current_c5
    best_h = best.copy()
    
    ones = np.where(best == 1.0)[0]
    zeros = np.where(best == 0.0)[0]
    
    for it in range(n_iters):
        if remaining_time() < 5:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, n_iters))
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap: h[i] 1->0, h[j] 0->1
        old_corr = compute_corr(best)
        best[i] = 0.0
        best[j] = 1.0
        new_corr = compute_corr(best)
        
        old_max = np.max(old_corr)
        new_max = np.max(new_corr)
        delta = (new_max - old_max) * (2.0 / n)
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            ones = np.where(best == 1.0)[0]
            zeros = np.where(best == 0.0)[0]
            current_c5 = new_max * (2.0 / n)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best.copy()
        else:
            # Revert
            best[i] = 1.0
            best[j] = 0.0
    
    return best_h, best_c5

def slsqp_smooth(h, n, beta, maxiter=300):

                    g[offset:offset+npairs] += w * (1.0 - x[:npairs])
                    g[:npairs] -= w * x[offset:offset+npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    g[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                    g[aoff:aoff+npairs] -= w * x[:npairs]
        g *= dx
        s = np.sum(x) - n / 2.0
        g += 2.0 * s
        return g
    
    c0 = compute_c5(h)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5t = compute_c5(trial)
            if c5t < c0:
                return trial
    except Exception:
        pass
    return h.copy()

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = random_01_init(best_n, 0)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: Exact coordinate descent from parent
    if remaining_time() > 60:
        h_cd, c5_cd = exact_coord_descent(h, n, max_sweeps=80, seed=seed)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
    
    # Phase 2: L-BFGS-B with top-m mean
    for m in [3, 5, 10, 20]:
        if remaining_time() < 30:
            break
        h_new = lbfgs_optimize(h, n, m, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 3: More coordinate descent
    if remaining_time() > 60:
        h_cd, c5_cd = exact_coord_descent(h, n, max_sweeps=100, seed=seed+1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
    
    # Phase 4: SLSQP with softmax
    for beta in [3000, 10000, 50000]:
        if remaining_time() < 25:
            break
        h_new = slsqp_smooth(h, n, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Coordinate descent again
    if remaining_time() > 40:
        h_cd, c5_cd = exact_coord_descent(h, n, max_sweeps=80, seed=seed+2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
    
    # Phase 6: Simulated annealing for 0/1 structures
    if remaining_time() > 200:
        h_sa, c5_sa = simulated_annealing_01(best_h, best_n, n_iters=8000, seed=seed+3)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h = h_sa.copy()
        # Refine the 0/1 solution with coordinate descent
        if c5_sa < best_c5 + 0.01:
            h_cd, c5_cd = exact_coord_descent(h_sa, best_n, max_sweeps=60, seed=seed+4)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                h = h_cd.copy()
    
    # Phase 7: Random restarts
    if remaining_time() > 100:
        for restart in range(8):
            if remaining_time() < 30:
                break
            h_rand = random_01_init(n, seed + 100 + restart)
            h_rand, c5_r = exact_coord_descent(h_rand, n, max_sweeps=40, seed=seed+200+restart)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_rand.copy()
                h = h_rand.copy()
            elif c5_r < best_c5 + 0.01:
                # Close to best, try more optimization
                h_new = lbfgs_optimize(h_rand, n, 5, maxiter=300)
                c5_new = compute_c5(h_new)
                h_cd, c5_cd = exact_coord_descent(h_new, n, max_sweeps=60, seed=seed+300+restart)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    h = h_cd.copy()
    
    # Phase 8: Try different resolutions
    for mult in [0.5, 1.5, 2.0]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if rn < 64 or rn > 2000:
            continue
        if remaining_time() < 80:
            break
        # Interpolate
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        x_new = (np.arange(rn) + 0.5) * 2.0 / rn
        h_r = np.interp(x_new, xp, best_h)
        h_r = project_feasible(h_r, rn)
        
        h_cd, c5_cd = exact_coord_descent(h_r, rn, max_sweeps=50, seed=seed+400)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = rn
    
    # Phase 9: Final intensive coordinate descent
    if remaining_time() > 30:
        h_cd, c5_cd = exact_coord_descent(best_h, best_n, max_sweeps=100, seed=seed+500)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 10: Final L-BFGS polish
    if remaining_time() > 15:
        h_new = lbfgs_optimize(best_h, best_n, 3, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final coordinate descent
    if remaining_time() > 10:
        h_cd, c5_cd = exact_coord_descent(best_h, best_n, max_sweeps=50, seed=seed+600)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Ensure constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)
    
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
