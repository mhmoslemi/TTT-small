def project_feasible(h, n):

def slsqp_opt(h, n, beta, maxiter=300):
    dx = 2.0 / n
    ncorr = 2*n - 1
    
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        return (m + np.log(wsum) / beta) * dx
    
    def smooth_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        
        grad = np.zeros(n)
        for k in range(ncorr):
            w = weights[k]
            if w < 1e-14:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[offset:offset+npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                    grad[aoff:aoff+npairs] -= w * x[:npairs]
        grad *= dx
        return grad
    
    def sum_con(x):
        return np.sum(x) - n / 2.0
    
    def sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac}]
    
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-24, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= compute_c5(h) + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            step = diff / cnt
            h[mask] += step
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-11:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 1:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)


def coord_descent(h, n, max_sweeps=100):
    best = h.astype(float).copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    converged_sweeps = 0

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_minus = i - offsets
            i_plus = i + offsets
            vm = (i_minus >= 0) & (i_minus < n)
            vp = (i_plus >= 0) & (i_plus < n)

            A = np.zeros(ncorr)
            if np.any(vm):
                A[vm] += (1.0 - best[i_minus[vm]])
            if np.any(vp):
                A[vp] -= best[i_plus[vp]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))

            best_t = old
            best_f = f_old

            for tc in (0.0, 1.0, 0.5):
                if abs(tc - old) < 1e-12:
                    continue
                mid = base + A * tc
                mid[n - 1] = base[n - 1] + tc - tc * tc
                f_c = float(np.max(mid))
                if f_c < best_f - 1e-12:
                    best_f = f_c
                    best_t = tc

            if best_t != old:
                if best_t > old:
                    lo, hi = old, min(1.0, best_t + 0.3)
                else:
                    lo, hi = max(0.0, best_t - 0.3), old

                if hi - lo > 1e-10:
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    mid1 = base + A * c1
                    mid1[n - 1] = base[n - 1] + c1 - c1 * c1
                    f1 = float(np.max(mid1))
                    mid2 = base + A * c2
                    mid2[n - 1] = base[n - 1] + c2 - c2 * c2
                    f2 = float(np.max(mid2))
                    for _ in range(28):
                        if f1 < f2:
                            hi = c2
                            c2 = c1
                            f2 = f1
                            c1 = hi - gr * (hi - lo)
                            mid1 = base + A * c1
                            mid1[n - 1] = base[n - 1] + c1 - c1 * c1
                            f1 = float(np.max(mid1))
                        else:
                            lo = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo + gr * (hi - lo)
                            mid2 = base + A * c2
                            mid2[n - 1] = base[n - 1] + c2 - c2 * c2
                            f2 = float(np.max(mid2))
                    best_t = (lo + hi) / 2.0
                    mid_f = base + A * best_t
                    mid_f[n - 1] = base[n - 1] + best_t - best_t * best_t
                    best_f = float(np.max(mid_f))

            if best_f < f_old - 1e-12:
                diff_t = best_t - old
                best[i] = best_t
                best_corr += A * diff_t
                best_corr[n - 1] += best_t * (1.0 - best_t) - old * (1.0 - old)
                improved = True

        if not improved:
            converged_sweeps += 1
            if converged_sweeps >= 3:
                break
        else:
            converged_sweeps = 0

    return best, float(np.max(best_corr) * dx)


def pair_flip_optimize(h, n, max_rounds=30):
    best = h.astype(float).copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    rng_local = np.random.RandomState(12345)

    for rnd in range(max_rounds):
        if remaining_time() < 10:
            break
        improved = False
        n_try = min(200, n // 3)
        for _ in range(n_try):
            i = rng_local.randint(0, n)
            j = rng_local.randint(0, n)
            if i == j:
                continue
            oi, oj = best[i], best[j]
            ti, tj = 1.0 - oi, 1.0 - oj

            Ai = np.zeros(ncorr)
            mi = (i - offsets >= 0) & (i - offsets < n)
            pi = (i + offsets >= 0) & (i + offsets < n)
            if np.any(mi):
                Ai[mi] += (1.0 - best[i - offsets[mi]])
            if np.any(pi):
                Ai[pi] -= best[i + offsets[pi]]
            Ai[n - 1] = 0.0

            Aj = np.zeros(ncorr)
            mj = (j - offsets >= 0) & (j - offsets < n)
            pj = (j + offsets >= 0) & (j + offsets < n)
            if np.any(mj):
                Aj[mj] += (1.0 - best[j - offsets[mj]])
            if np.any(pj):
                Aj[pj] -= best[j + offsets[pj]]
            Aj[n - 1] = 0.0

            new_corr = best_corr + Ai * (ti - oi) + Aj * (tj - oj)
            new_corr[n - 1] += ti * (1 - ti) - oi * (1 - oi) + tj * (1 - tj) - oj * (1 - oj)
            f_new = float(np.max(new_corr))
            f_old = float(np.max(best_corr))

            if f_new < f_old - 1e-12:
                best[i] = ti
                best[j] = tj
                best_corr = new_corr
                improved = True

        if not improved:
            break

    return best, float(np.max(best_corr) * dx)


def block_perturb(h, n, rng, block_size=4):
    h_new = h.copy()
    n_blocks = n // (block_size * 4) + 1
    for _ in range(n_blocks):
        start = rng.randint(0, max(1, n - block_size))
        h_new[start:start + block_size] = 1.0 - h_new[start:start + block_size]
    return project_feasible(h_new, n)


def slsqp_refine(h, n, beta, maxiter=300):
    dx = 2.0 / n
    ncorr = 2 * n - 1

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        ev = np.exp(shifted)
        ws = np.sum(ev)
        return (m + np.log(ws) / beta) * dx

    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        ev = np.exp(shifted)
        ws = np.sum(ev)
        wts = ev / ws
        grad = np.zeros(n)
        for k in range(ncorr):
            w = wts[k]
            if w < 1e-13:
                continue
            off = k - (n - 1)
            if off >= 0:
                npa = n - off
                if npa > 0:
                    grad[off:off + npa] += w * (1.0 - x[:npa])
                    grad[:npa] -= w * x[off:off + npa]
            else:
                ao = -off
                npa = n - ao
                if npa > 0:
                    grad[:npa] += w * (1.0 - x[ao:ao + npa])
                    grad[ao:ao + npa] -= w * x[:npa]
        grad *= dx
        return grad

    def _con(x):
        return np.sum(x) - n / 2.0

    def _con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-11:
                return trial
    except Exception:
        pass
    return h.copy()


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def fractional_tune(h, n, max_iters=200):
    best = h.astype(float).copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for it in range(max_iters):
        if remaining_time() < 3:
            break
        sorted_idx = np.argsort(best_corr)[::-1]
        top_k = sorted_idx[:4]
        top_vals = best_corr[top_k]
        max_val = top_vals[0]
        min_top = top_vals[-1]

        if max_val - min_top < 0.1:
            break

        best_improvement = 0.0
        best_idx = -1
        best_frac = 0.0

        candidates = list(range(0, n, 2))
        np.random.RandomState(it).shuffle(candidates[:min(100, len(candidates))])

        for i in candidates[:80]:
            old = best[i]
            if old < 0.01 or old > 0.99:
                continue
            i_minus = i - offsets
            i_plus = i + offsets
            vm = (i_minus >= 0) & (i_minus < n)
            vp = (i_plus >= 0) & (i_plus < n)
            A = np.zeros(ncorr)
            if np.any(vm):
                A[vm] += (1.0 - best[i_minus[vm]])
            if np.any(vp):
                A[vp] -= best[i_plus[vp]]
            A[n - 1] = 0.0
            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            for frac in [0.3, 0.7, 0.2, 0.8, 0.4, 0.6]:
                if abs(frac - old) < 0.01:
                    continue
                mid = base + A * frac
                mid[n - 1] = base[n - 1] + frac - frac * frac
                f_new = float(np.max(mid))
                improvement = float(np.max(best_corr)) - f_new
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_idx = i
                    best_frac = frac

        if best_idx >= 0 and best_improvement > 1e-11:
            old = best[best_idx]
            i = best_idx
            i_minus = i - offsets
            i_plus = i + offsets
            vm = (i_minus >= 0) & (i_minus < n)
            vp = (i_plus >= 0) & (i_plus < n)
            A = np.zeros(ncorr)
            if np.any(vm):
                A[vm] += (1.0 - best[i_minus[vm]])
            if np.any(vp):
                A[vp] -= best[i_plus[vp]]
            A[n - 1] = 0.0
            best_corr += A * (best_frac - old)
            best_corr[n - 1] += best_frac * (1 - best_frac) - old * (1 - old)
            best[best_idx] = best_frac
        else:
            break

    return best, float(np.max(best_corr) * dx)


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_h = parent_h.copy()
        best_n = len(parent_h)
        best_c5 = compute_c5(best_h)
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Coordinate descent from parent
    curr_h = best_h.copy()
    curr_h, c5 = coord_descent(curr_h, n, max_sweeps=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = curr_h.copy()
        n = len(best_h)

    # Phase 2: Pair flips
    if remaining_time() > 80:
        curr_h = best_h.copy()
        curr_h, c5 = pair_flip_optimize(curr_h, n, max_rounds=20)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()
        curr_h, c5 = coord_descent(curr_h, n, max_sweeps=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()

    # Phase 3: Diverse restarts
    n_restarts = 16
    for r in range(n_restarts):
        if remaining_time() < 45:
            break

        if r % 4 == 0:
            h_r = block_perturb(best_h, n, rng, block_size=3)
        elif r % 4 == 1:
            h_r = best_h.copy()
            flip_idx = rng.choice(n, size=n // 8, replace=False)
            h_r[flip_idx] = 1.0 - h_r[flip_idx]
            h_r = project_feasible(h_r, n)
        elif r % 4 == 2:
            h_r = best_h.copy()
            mask = rng.random(n) < 0.15
            h_r[mask] += rng.randn(int(np.sum(mask))) * 0.2
            h_r = project_feasible(h_r, n)
        else:
            h_r = best_h.copy()
            for _ in range(5):
                start = rng.randint(0, max(1, n - 10))
                length = rng.randint(3, 12)
                h_r[start:min(start + length, n)] = 1.0 - h_r[start:min(start + length, n)]
            h_r = project_feasible(h_r, n)

        curr_h, c5 = coord_descent(h_r, n, max_sweeps=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()

        if remaining_time() < 45:
            break
        curr_h, c5 = pair_flip_optimize(curr_h, n, max_rounds=10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()
        curr_h, c5 = coord_descent(curr_h, n, max_sweeps=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()

    # Phase 4: SLSQP refinement
    if remaining_time() > 120:
        curr_h = best_h.copy()
        for beta in [2000, 10000]:
            if remaining_time() < 40:
                break
            curr_h = slsqp_refine(curr_h, n, beta, maxiter=300)
            c5 = compute_c5(curr_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = curr_h.copy()
        curr_h, c5 = coord_descent(best_h, n, max_sweeps=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()

    # Phase 5: Fractional tuning
    if remaining_time() > 30:
        curr_h, c5 = fractional_tune(best_h, n, max_iters=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()
        curr_h, c5 = coord_descent(curr_h, n, max_sweeps=60)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()

    # Phase 6: Try higher resolutions
    target_resolutions = [960, 1280]
    for tn in target_resolutions:
        if remaining_time() < 120 or tn > 2000:
            continue
        h_t = interp_to(best_h, best_n, tn)
        h_t = project_feasible(h_t, tn)
        c5_t = compute_c5(h_t)

        curr_h, c5 = coord_descent(h_t, tn, max_sweeps=70)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()
            best_n = tn

        if remaining_time() > 60:
            curr_h, c5 = pair_flip_optimize(curr_h, tn, max_rounds=10)
            curr_h, c5 = coord_descent(curr_h, tn, max_sweeps=50)
            if c5 < best_c5:
                best_c5 = c5
                best_h = curr_h.copy()
                best_n = tn

        if remaining_time() > 40:
            curr_h, c5 = fractional_tune(curr_h, tn, max_iters=50)
            if c5 < best_c5:
                best_c5 = c5
                best_h = curr_h.copy()
                best_n = tn

    # Phase 7: Try lower resolution (sometimes works better)
    if remaining_time() > 80 and best_n > 512:
        for tn in [512, 448]:
            if remaining_time() < 40:
                break
            h_t = interp_to(best_h, best_n, tn)
            h_t = project_feasible(h_t, tn)
            curr_h, c5 = coord_descent(h_t, tn, max_sweeps=80)
            if c5 < best_c5:
                best_c5 = c5
                best_h = curr_h.copy()
                best_n = tn

    # Phase 8: Final SLSQP
    if remaining_time() > 30:
        curr_h = slsqp_refine(best_h, best_n, 50000, maxiter=250)
        c5 = compute_c5(curr_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()
            best_n = len(best_h)
        curr_h, c5 = coord_descent(best_h, best_n, max_sweeps=60)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()

    # Phase 9: Final coordinate descent polish
    if remaining_time() > 15:
        curr_h, c5 = coord_descent(best_h, best_n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()

    if remaining_time() > 8:
        curr_h, c5 = fractional_tune(best_h, best_n, max_iters=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = curr_h.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
