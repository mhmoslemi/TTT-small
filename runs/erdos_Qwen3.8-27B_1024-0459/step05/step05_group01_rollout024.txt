def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0

def _eq_jac(x):
    return np.ones(_CUR_N)

def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < _sm_obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def subgradient_opt(h, n, time_limit, top_k=5, alpha0=0.5):
    """Projected subgradient descent for continuous refinement."""
    global _CUR_N
    _CUR_N = n
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    t = 0
    
    while time.time() - start < time_limit and remaining_time() > 5:
        t += 1
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        avg_corr = corr[top_idx].mean()
        
        # Compute average subgradient
        grad = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                grad[off:] += (1.0 - h[:n - off])
                grad[:n - off] -= h[off:]
            else:
                a = -off
                grad[:n - a] += (1.0 - h[a:])
                grad[a:] -= h[:n - a]
        grad /= len(top_idx)
        
        # Step size: diminishing
        alpha = alpha0 / (1.0 + 0.01 * t)
        
        h_new = h - alpha * grad * dx
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def top_k_equalize(h, n, top_k=5, max_iters=300):
    """Equalize the top-k correlations via directional optimization."""
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08, 0.15, 0.002]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try negative direction
            for step in [0.01, 0.02, 0.05, 0.005, 0.03]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-13:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    
    return h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def random_balanced(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = theta_morse(384)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Intensive binary steepest descent at parent resolution
    # Use ~50% of budget
    t1_limit = budget_s * 0.45
    h_bin, c5_bin = binary_steepest_descent(best_h, best_n, t1_limit, rng)
    if c5_bin < best_c5 - 1e-14:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Phase 2: Multi-swap escape attempts
    if remaining_time() > 60:
        t2_limit = min(60, remaining_time() * 0.2)
        h_ms, c5_ms = binary_multi_swap_search(best_h, best_n, t2_limit, rng, n_swaps=2)
        if c5_ms < best_c5 - 1e-14:
            best_c5 = c5_ms
            best_h = h_ms.copy()
    
    # Phase 3: Continuous refinement via SLSQP
    if remaining_time() > 30:
        t3_limit = min(120, remaining_time() * 0.2)
        start_t = time.time()
        for beta in [500, 2000, 10000, 50000, 200000]:
            if time.time() - start_t > t3_limit or remaining_time() < 15:
                break
            h_new = slsqp_opt(best_h, best_n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 4: Top-k equalization
    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(best_h, best_n, top_k=5, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Phase 5: More binary search after continuous refinement
    if remaining_time() > 30:
        h_bin2, c5_bin2 = binary_steepest_descent(best_h, best_n,
            min(120, remaining_time() * 0.2), rng)
        if c5_bin2 < best_c5 - 1e-14:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
    
    # Phase 6: Try higher resolution
    if remaining_time() > 80:
        n_high = best_n * 2
        if n_high % 2 != 0:
            n_high += 1
        h_interp = interp_to(best_h, best_n, n_high)
        h_feas = project_feasible(h_interp, n_high)
        # Binarize
        h_bin_h = (h_feas > 0.5).astype(float)
        nh = n_high // 2
        if int(np.sum(h_bin_h)) != nh:
            idx = np.argsort(-h_bin_h)
            h_bin_h = np.zeros(n_high)
            h_bin_h[idx[:nh]] = 1.0
        
        t6_limit = min(remaining_time() * 0.25, 180)
        h_hi, c5_hi = binary_steepest_descent(h_bin_h, n_high, t6_limit, rng)
        # Continuous refinement at higher res
        if remaining_time() > 20:
            h_hi = slsqp_opt(h_hi, n_high, 10000, maxiter=300)
            c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5 - 1e-14:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_high
    
    # Phase 7: Try Thue-Morse starting point at best_n
    if remaining_time() > 60:
        h_tm = theta_morse(best_n)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5:
            h_tm_opt, c5_tm_opt = binary_steepest_descent(h_tm, best_n,
                min(60, remaining_time() * 0.3), rng)
            if c5_tm_opt < best_c5:
                best_c5 = c5_tm_opt
                best_h = h_tm_opt.copy()
    
    # Phase 8: Final continuous polish
    if remaining_time() > 15:
        for beta in [100000, 500000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_opt(best_h, best_n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(best_h, best_n, top_k=6, max_iters=200)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Final binary polish
    if remaining_time() > 20:
        h_bp, c5_bp = binary_steepest_descent(best_h, best_n,
            min(remaining_time() - 5, 60), rng)
        if c5_bp < best_c5 - 1e-14:
            best_c5 = c5_bp
            best_h = h_bp.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
