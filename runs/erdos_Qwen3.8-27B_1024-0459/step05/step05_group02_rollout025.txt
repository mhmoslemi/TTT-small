def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=5, maxiter=800):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxiter * 10,
                                'maxls': 50})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def nesterov_gradient(h, n, top_k=5, max_iters=10000, step_init=0.025):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        c_min_top = c_top[-1]
        spread = c_top[0] - c_min_top
        if spread > 1e-12:
            w = (c_top - c_min_top) / spread + 0.05
        else:
            w = np.ones(top_k) / top_k
        w = np.maximum(w, 0.01)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.12)
            else:
                step *= 0.95
        else:
            h = best_h.copy()
            v *= 0.3
            step *= 0.6

        if it % 2000 == 0 and it > 0:
            step *= 0.8
            momentum = min(momentum + 0.02, 0.95)

    return best_h, best_c5


def simulated_anneal_binary(h_bin, n, max_iters=5000, T_init=0.3, T_min=0.005, seed=42):
    rng = np.random.RandomState(seed)
    h_bin = h_bin.copy()
    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    for it in range(max_iters):
        if remaining() < 2:
            break

        T = T_init * (T_min / T_init) ** (it / float(max_iters))

        h_bin[rng.randint(n)] = 1.0 - h_bin[rng.randint(n)]
        if h_bin.sum() != n // 2:
            h_bin[rng.randint(n)] = 1.0 - h_bin[rng.randint(n)]

        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()

    return best_h, best_c5


def directed_binary_search(h_bin, n, max_iters=5000, top_k=4, seed=42):
    rng = np.random.RandomState(seed)
    h_bin = h_bin.copy()
    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    for it in range(max_iters):
        if remaining() < 2:
            break

        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]

        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        score_one = -g[ones]
        score_zero = g[zeros]

        n_try_o = min(15, len(ones))
        n_try_z = min(15, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try_o]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        best_delta = 0
        best_swap = None
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is None:
            for _ in range(8):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is None:
            break

        i, j = best_swap
        h_bin[i] = 0
        h_bin[j] = 1
        cur_c5 = compute_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def full_pipeline(h, n, seed=42):
    """Run a full optimization pipeline at a given resolution."""
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # L-BFGS passes with increasing beta
    for beta in [200, 500, 1500, 5000, 15000, 40000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Nesterov gradient
    if remaining() > 20:
        h_pg, c5_pg = nesterov_gradient(h, n, top_k=5, max_iters=6000, step_init=0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Binary local search
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = directed_binary_search(h_bin, n, max_iters=4000, top_k=4, seed=seed)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

        # Continuous refinement of binary
        h_ref = h_b.copy()
        for beta in [5000, 20000]:
            if remaining() < 5:
                break
            h_ref = lbfgs_pass(h_ref, n, beta, top_k=4, maxiter=400)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Simulated annealing on binary
    if remaining() > 30:
        h_bin2 = (best_h > 0.5).astype(float)
        h_sa, c5_sa = simulated_anneal_binary(h_bin2, n, max_iters=3000,
                                               T_init=0.2, T_min=0.003, seed=seed + 1)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

        # Then directed search from SA result
        h_b2, c5_b2 = directed_binary_search(h_sa, n, max_iters=2000, top_k=4, seed=seed + 2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Final L-BFGS polish
    for beta in [30000, 80000]:
        if remaining() < 5:
            break
        h_new = lbfgs_pass(best_h, n, beta, top_k=3, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = proj(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5


def perturb_and_reoptimize(h, n, seed=42, n_perturb=10):
    """Perturb the solution and re-optimize to escape local optima."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for trial in range(n_perturb):
        if remaining() < 15:
            break

        h_pert = h.copy()
        # Random Gaussian perturbation
        noise = rng.randn(n) * 0.1
        h_pert = proj(h_pert + noise, n)

        # Quick L-BFGS
        h_new = lbfgs_pass(h_pert, n, beta=2000, top_k=5, maxiter=300)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Continue refining
            h_new2 = lbfgs_pass(h_new, n, beta=10000, top_k=4, maxiter=300)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()

        # Also try binary perturbation
        h_bin = (best_h > 0.5).astype(float)
        # Swap a random pair
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) > 0 and len(zeros) > 0:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h_bin[i] = 0
            h_bin[j] = 1
            c5_b = compute_c5(h_bin)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_bin.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Full pipeline at parent resolution
    h1, c5_1 = full_pipeline(best_h.copy(), best_n, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: Try higher resolutions
    n_options = []
    base = best_n
    for mult in [1.0, 1.333, 1.5, 2.0, 2.667]:
        nn = int(round(base * mult / 2)) * 2
        if nn >= 128 and nn <= 2048 and nn not in n_options:
            n_options.append(nn)
    n_options.sort()
    # Remove the base resolution (already done)
    n_options = [nn for nn in n_options if nn != best_n]

    for n_t in n_options[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)
        h_t, c5_t = full_pipeline(h_feas, n_t, seed=seed + len(n_options) - n_options.index(n_t))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 3: Perturb and re-optimize at best resolution
    if remaining() > 30:
        h_p, c5_p = perturb_and_reoptimize(best_h, best_n, seed=seed + 100, n_perturb=8)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 4: More aggressive binary search at best resolution
    if remaining() > 25:
        h_bin = (best_h > 0.5).astype(float)
        # Multiple SA runs with different temps
        for T_init in [0.1, 0.05, 0.15]:
            if remaining() < 10:
                break
            h_sa, c5_sa = simulated_anneal_binary(h_bin, best_n, max_iters=2000,
                                                   T_init=T_init, T_min=0.002,
                                                   seed=seed + 200)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h_bin = h_sa.copy()

        # Final directed search
        if remaining() > 10:
            h_b, c5_b = directed_binary_search(best_h, best_n, max_iters=3000,
                                                top_k=5, seed=seed + 300)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 5: Try slightly different resolutions around best
    if remaining() > 30:
        nearby = [best_n - 2, best_n + 2, best_n - 4, best_n + 4, best_n + 6, best_n - 6]
        nearby = [nn for nn in nearby if 128 <= nn <= 2048 and nn % 2 == 0 and nn != best_n]
        for n_t in nearby[:3]:
            if remaining() < 20:
                break
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = proj(h_interp, n_t)
            # Quick optimization
            h_new = lbfgs_pass(h_feas, n_t, beta=5000, top_k=5, maxiter=400)
            c5_new = compute_c5(h_new)
            h_new = lbfgs_pass(h_new, n_t, beta=20000, top_k=4, maxiter=300)
            c5_new2 = compute_c5(h_new)
            h_bin_t = (h_new > 0.5).astype(float)
            h_b, c5_b = directed_binary_search(h_bin_t, n_t, max_iters=2000, top_k=4, seed=seed + 400)
            c5_best_t = min(c5_new, c5_new2, c5_b)
            if c5_b <= c5_best_t:
                h_candidate = h_b
            elif c5_new2 <= c5_best_t:
                h_candidate = h_new
            else:
                h_candidate = h_new
            if c5_best_t < best_c5:
                best_c5 = c5_best_t
                best_h = h_candidate.copy()
                best_n = n_t

    # Phase 6: Final refinement
    if remaining() > 15:
        # Final Nesterov
        h_nf, c5_nf = nesterov_gradient(best_h, best_n, top_k=4, max_iters=3000, step_init=0.015)
        if c5_nf < best_c5:
            best_c5 = c5_nf
            best_h = h_nf.copy()

        # Final L-BFGS with very high beta
        if remaining() > 8:
            h_f = lbfgs_pass(best_h, best_n, beta=100000, top_k=3, maxiter=300)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
