def active_set_equate(h, n, top_k=8, max_iters=1000, tol=1e-12):
    """Equalize the top_k lags using projected gradient / Newton steps."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)

        c_vals = corr[idx_top]

            break
        
        # Direction: increase worst lags, decrease best
        # Build gradient direction
        g_dir = np.zeros(n)
        weights = c_vals - c_vals.mean()  # negative for high, positive for low
        norm_w = np.sqrt(np.sum(weights**2))
        if norm_w < 1e-15:

        weights = weights / norm_w

            else:

                np_ = n - a
                g_dir[:np_] += wk * (1.0 - h[a:])
                g_dir[a:] -= wk * h[:np_]
        
        gn = np.linalg.norm(g_dir)
        if gn < 1e-15:
            break
        d = g_dir / gn
        
        # Project onto feasible direction (maintain sum)
        d = d - np.mean(d)
        gn2 = np.linalg.norm(d)
        if gn2 < 1e-15:
            break
        d = d / gn2
        
        # Line search
        improved = False
        steps = [0.001, 0.003, 0.01, 0.03, 0.005, 0.02, 0.05, 0.1]

                                best_h = h.copy()
                    else:
                        h[i] = 1.0
                        h[j] = 0.0
                    continue
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                c_new = compute_c5(h)
                delta = c_new - current_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    if delta < 0:
                        current_c5 = c_new
                        if c_new < best_c5:
                            best_c5 = c_new
                            best_h = h.copy()
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    h[j1] = 0.0
                    h[j2] = 0.0
    
    return best_h, best_c5


def binary_local_search(h_bin, n, time_limit=60, rng=None):
    """Steepest-descent binary local search with 1-1 and 2-2 swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    n_half = n // 2
    h = (h_bin > 0.5).astype(float).copy()
    if int(np.sum(h)) != n_half:
        idx = np.argsort(-h_bin)
        h = np.zeros(n)
        h[idx[:n_half]] = 1.0
    
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    no_improve = 0
    
    while time.time() - t_start < time_limit and remaining_time() > 5:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # Try 1-1 swaps (sample if too many)
        n_pairs_1 = len(ones) * len(zeros)
        if n_pairs_1 <= 40000:
            best_delta = 1e-10
            best_move = None
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c = compute_c5(h)
                    h[i] = 1.0
                    h[j] = 0.0
                    delta = c - best_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_move = ('11', i, j)
            
            if best_move is not None:
                _, i, j = best_move
                h[i] = 0.0
                h[j] = 1.0
                best_c5 = compute_c5(h)
                best_h = h.copy()
                improved = True
        else:
            # Sample swaps
            n_trials = 500
            best_delta = 1e-10
            best_move = None
            for _ in range(n_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = c - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = ('11', i, j)
            
            if best_move is not None:
                _, i, j = best_move
                h[i] = 0.0
                h[j] = 1.0
                best_c5 = compute_c5(h)
                best_h = h.copy()
                improved = True
        
        # Try 2-2 swaps (sample)
        if len(ones) >= 2 and len(zeros) >= 2:
            n_trials = 200
            best_delta_2 = 1e-10
            best_move_2 = None
            for _ in range(n_trials):
                i1, i2 = rng.choice(ones, 2, replace=False)
                j1, j2 = rng.choice(zeros, 2, replace=False)
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                c = compute_c5(h)
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
                delta = c - best_c5
                if delta < best_delta_2:
                    best_delta_2 = delta
                    best_move_2 = (i1, i2, j1, j2)
            
            if best_move_2 is not None:
                i1, i2, j1, j2 = best_move_2
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                best_c5 = compute_c5(h)
                best_h = h.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
    
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def coordinate_refine(h, n, max_sweeps=40):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0]
            if 0.0 < old < 1.0:
                cands.extend([old - 0.1, old + 0.1, old - 0.3, old + 0.3,
                              0.5 * old, 0.5 * (2.0 - old)])
            cands = [min(1.0, max(0.0, c)) for c in cands]
            cands.append(old)
            
            base_c5 = compute_c5(h)
            best_val = old
            best_val_c5 = base_c5
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_val_c5 - 1e-13:
                    best_val_c5 = c5_v
                    best_val = c
                    improved_sweep = True
            h[i] = best_val
            h = project_feasible(h, n)
            if best_val_c5 < best_c5:
                best_c5 = best_val_c5
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP on parent with increasing beta
    h = best_h.copy()
    for beta in [200, 1000, 5000, 20000, 100000, 500000, 2000000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 2: Active-set equalization (top 6-10 lags)
    for top_k in [4, 6, 8, 10]:
        if remaining_time() < 10:
            break
        h_eq, c5_eq = active_set_equate(h, best_n, top_k=top_k, max_iters=2000)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 3: More SLSQP after equalization
    for beta in [500000, 2000000, 10000000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 4: Binary search + smooth refinement
    h_bin = (best_h > 0.5).astype(float)
    n_half = best_n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        idx = np.argsort(-best_h)
        h_bin = np.zeros(best_n)
        h_bin[idx[:n_half]] = 1.0
    
    # SA for binary
    sa_time = min(180, max(20, int(remaining_time() * 0.2)))
    if sa_time > 10:
        h_sa, c5_sa = sa_binary_search(h_bin, best_n, time_limit=sa_time, rng=rng, n_restarts=4)
    else:
        h_sa, c5_sa = h_bin.copy(), compute_c5(h_bin)
    
    # Local search on best of (parent-bin, SA)
    best_bin = h_sa if c5_sa < compute_c5(h_bin) else h_bin
    ls_time = min(60, max(10, int(remaining_time() * 0.08)))
    if ls_time > 5:
        h_ls, c5_ls = binary_local_search(best_bin, best_n, time_limit=ls_time, rng=rng)
        if c5_ls < c5_sa:
            best_bin = h_ls
            c5_sa = c5_ls
    
    # Smooth refine the best binary
    h_smooth = best_bin.copy()
    for beta in [5000, 50000, 500000, 5000000]:
        if remaining_time() < 10:
            break
        h_smooth = slsqp_opt(h_smooth, best_n, beta, maxiter=800)
    c5_smooth = compute_c5(h_smooth)
    h_smooth, c5_smooth = active_set_equate(h_smooth, best_n, top_k=6, max_iters=1000)
    
    if c5_smooth < best_c5 - 1e-14:
        best_c5 = c5_smooth
        best_h = h_smooth.copy()
        h = h_smooth.copy()
    
    # Also check raw binary
    c5_bin = compute_c5(best_bin)
    if c5_bin < best_c5 - 1e-14:
        best_c5 = c5_bin
        best_h = best_bin.copy()
    
    # Phase 5: Multi-resolution
    n_trials = [best_n]
    for mult in [0.9, 1.1, 1.25, 1.5, 0.8]:
        nn = int(best_n * mult)
        if nn >= 200 and nn <= 1200 and nn % 2 == 0:
            n_trials.append(nn)
    n_trials = sorted(set(n_trials))
    # Sort by proximity to best_n
    n_trials.sort(key=lambda x: abs(x - best_n))
    
    for n_t in n_trials[1:6]:
        if remaining_time() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        
        h_opt = h_feas
        for beta in [2000, 20000, 200000]:
            if remaining_time() < 12:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=600)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = active_set_equate(h_opt, n_t, top_k=6, max_iters=800)
        
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()
    
    # Phase 6: Alternative starting points
    # Thue-Morse based
    def theta_morse(n):
        h = np.zeros(n)
        for i in range(n):
            v = i
            cnt = 0
            while v:
                cnt += v & 1
                v >>= 1
            h[i] = 1.0 if cnt % 2 == 1 else 0.0
        return project_feasible(h, n)
    
    alt_starts = [theta_morse(best_n)]
    # Random
    h_rand = np.zeros(best_n)
    h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
    alt_starts.append(h_rand)
    # Shifted
    alt_starts.append(project_feasible(np.roll(best_h, 1), best_n))
    alt_starts.append(project_feasible(np.roll(best_h, -1), best_n))
    
    for h_alt in alt_starts:
        if remaining_time() < 25:
            break
        h_opt = h_alt.copy()
        for beta in [1000, 20000, 200000]:
            if remaining_time() < 10:
                break
            h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=500)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = active_set_equate(h_opt, best_n, top_k=6, max_iters=600)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
    
    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [1000000, 5000000, 20000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    if remaining_time() > 8:
        h_eq, c5_eq = active_set_equate(h, n, top_k=8, max_iters=2000)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Phase 8: Coordinate refinement
    if remaining_time() > 15:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=50)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()
    
    # Final polish
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 10000000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    if remaining_time() > 5:
        h_eq, c5_eq = active_set_equate(best_h, n, top_k=10, max_iters=1000)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, n)
