def smooth_max_and_grad(corr, beta):
    """Compute smooth max and gradient of smooth max w.r.t. corr"""
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.minimum(shifted, 0)  # Ensure numerical stability
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = m + np.log(wsum) / beta
    return val, weights

def compute_obj_grad_h(h, n, beta):
    """Compute smooth max objective and gradient w.r.t. h"""
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.minimum(shifted, 0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)

    val = (m + np.log(wsum) / beta) * dx

    ncorr = 2 * n - 1

                grad[offset:offset+npairs] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                # corr[k] = sum_{i=0}^{npairs-1} h[i] * (1-h[i+aoff])
                grad[:npairs] += w * (1.0 - h[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * h[:npairs]
    grad *= dx
    return val, grad
