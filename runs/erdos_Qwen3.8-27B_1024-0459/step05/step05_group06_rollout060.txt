    try:
        res = minimize(obj, h[interior].copy(), jac=grad, method='SLSQP',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = h.copy()
            trial[interior] = res.x
            trial = project_feasible(trial, n)
            if compute_c5(trial) <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def coord_descent(h, n, n_sweeps=50, tol=1e-15):
    """Coordinate descent with golden section refinement."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        improved_any = False
        interior = np.where((best > 1e-12) & (best < 1.0 - 1e-12))[0]
        if len(interior) == 0:
            interior = np.arange(n)
        
        for i in interior:
            if remaining_time() < 1:
                break
            t0 = best[i]
            # Compute coefficients for corr as function of h[i]
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n-1), n):
                ki = n - 1 + off
                if off == 0:
                    # h[i]*(1-h[i]) = h[i] - h[i]^2 => a+=1, b+=-1
                    a[ki] += 1.0
                    b[ki] += -1.0
                else:
                    if 0 <= i + off < n:
                        # h[i] appears as source: h[i]*(1-h[i+off])
                        a[ki] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        # h[i] appears as target: h[i-off]*(1-h[i])
                        a[ki] += (-best[i - off])
            
            base = best_corr - a * t0 - b * t0 * t0
            
            # Evaluate at candidates
            cands = [0.0, 1.0, t0]
            for delta in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
                cands.append(max(0.0, t0 - delta))
                cands.append(min(1.0, t0 + delta))
            
            best_t = t0
            best_f = np.max(base + a * t0 + b * t0 * t0) * dx
            for t in cands:
                if abs(t - t0) < 1e-14:
                    continue
                f = np.max(base + a * t + b * t * t) * dx
                if f < best_f - 1e-16:
                    best_f = f
                    best_t = t
            
            # Golden section if we found improvement
            if best_t != t0:
                margin = 0.3
                lo = max(0.0, best_t - margin)
                hi = min(1.0, best_t + margin)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                for _ in range(80):
                    if hi - lo < 1e-14:
                        break
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                t_opt = (lo + hi) / 2.0
                f_opt = np.max(base + a * t_opt + b * t_opt * t_opt) * dx
                if f_opt < best_f:
                    best_f = f_opt
                    best_t = t_opt
            
            if best_f < best_c5 - tol:
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved_any = True
        
        if not improved_any:
            break
    
    return best, best_c5

def pair_descent(h, n, n_pairs=200, n_outer=20):
    """Pair coordinate descent: change h[i]+=d, h[j]-=d preserving sum."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    rng = np.random.RandomState(123)
    
    for outer in range(n_outer):
        if remaining_time() < 3:
            break
        improved_any = False
        for _ in range(n_pairs):
            if remaining_time() < 1:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            ti = best[i]
            tj = best[j]
            # Determine range of delta
            d_lo = max(-ti, -(1.0 - tj))
            d_hi = min(1.0 - ti, tj)
            if d_hi - d_lo < 1e-10:
                continue
            
            # corr[k] + delta*(g_i[k] - g_j[k]) for linear part
            # But at lag 0 there's a quadratic term: (ti+d)(1-ti-d) vs ti(1-ti)
            # Let's just evaluate at a few points
            mid = (d_lo + d_hi) / 2
            best_d = 0.0
            best_f = best_c5
            
            # Try several candidates
            cands = [0.0, mid, d_lo, d_hi,
                     d_lo + 0.25*(d_hi-d_lo), d_lo + 0.75*(d_hi-d_lo),
                     0.01, -0.01, 0.05, -0.05, 0.1, -0.1]
            for d in cands:
                d = max(d_lo, min(d_hi, d))
                if abs(d) < 1e-14:
                    continue
                trial = best.copy()
                trial[i] = ti + d
                trial[j] = tj - d
                c = compute_c5(trial)
                if c < best_f - 1e-15:
                    best_f = c
                    best_d = d
            
            if best_d != 0.0:
                # Golden section
                lo = max(d_lo, best_d - 0.15)
                hi = min(d_hi, best_d + 0.15)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                def eval_d(d):
                    trial = best.copy()
                    trial[i] = ti + d
                    trial[j] = tj - d
                    return compute_c5(trial)
                f1 = eval_d(c1)
                f2 = eval_d(c2)
                for _ in range(50):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = eval_d(c1)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = eval_d(c2)
                d_opt = (lo + hi) / 2.0
                f_opt = eval_d(d_opt)
                if f_opt < best_f:
                    best_f = f_opt
                    best_d = d_opt
                
                if best_f < best_c5 - 1e-15:
                    best[i] = ti + best_d
                    best[j] = tj - best_d
                    best_c5 = best_f
                    best_corr = compute_corr(best)
                    improved_any = True
        
        if not improved_any:
            break
    
    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgrad_descent(h, n, max_iter=200, n_active_lags=4):
    """Subgradient descent with projection."""
    best = h.copy()
    best_c5 = compute_c5(best)
    
    for it in range(max_iter):
        if remaining_time() < 3:
            break
        corr = compute_corr(best)
        max_corr = np.max(corr)
        active = np.where(corr >= max_corr - 0.3)[0]
        if len(active) > n_active_lags:
            active = np.argsort(corr)[::-1][:n_active_lags]
        
        grad = np.zeros(n)
        for k_idx in active:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset+npairs] += (1.0 - best[:npairs])
                    grad[:npairs] -= best[offset:offset+npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[aoff:aoff+npairs])
                    grad[aoff:aoff+npairs] -= best[:npairs]
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        grad_n = grad / gnorm
        base_step = 0.05 / (1.0 + it * 0.008)
        improved = False
        for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 0.125]:
            step = base_step * sm
            trial = best - step * grad_n
            trial = project_feasible(trial, n)
            c5_trial = compute_c5(trial)
            if c5_trial < best_c5 - 1e-16:
                best_c5 = c5_trial
                best = trial.copy()
                improved = True
                break
        if not improved and it > 100:
            break
    
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: SLSQP at current resolution with various betas
    for beta in [200, 1000, 5000, 20000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_full(h, n, beta, maxiter=300, n_active=8)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    # Phase 2: Coordinate descent (many sweeps)
    if remaining_time() > 60:
        h_new, c5_new = coord_descent(h, n, n_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 3: SLSQP with higher beta
    for beta in [50000, 200000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_full(h, n, beta, maxiter=250, n_active=6)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    # Phase 4: Pair descent
    if remaining_time() > 40:
        h_new, c5_new = pair_descent(h, n, n_pairs=500, n_outer=15)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: More coordinate descent
    if remaining_time() > 40:
        h_new, c5_new = coord_descent(h, n, n_sweeps=40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 6: Subgradient with SLSQP interleaving
    if remaining_time() > 60:
        for sub_it in range(3):
            if remaining_time() < 20:
                break
            h_new, c5_new = subgrad_descent(h, n, max_iter=100, n_active_lags=5)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            if remaining_time() > 20:
                h_new2 = slsqp_interior(h, n, 10000, maxiter=200, n_active=8)
                c5_new2 = compute_c5(h_new2)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2.copy()
                    h = h_new2.copy()
    
    # Phase 7: Try higher resolution
    for mult in [1.5, 2.0]:
        if remaining_time() < 80:
            break
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if rn > 2048:
            rn = 2048
        if rn <= best_n:
            continue
        
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        
        # Optimize at higher resolution
        for beta in [2000, 10000, 50000]:
            if remaining_time() < 20:
                break
            h_r = slsqp_full(h_r, rn, beta, maxiter=200, n_active=8)
        
        if remaining_time() > 30:
            h_r, c5_r = coord_descent(h_r, rn, n_sweeps=30)
        
        if remaining_time() > 15:
            h_r, c5_r = pair_descent(h_r, rn, n_pairs=300, n_outer=8)
        
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn
    
    # Phase 8: Final intensive coordinate descent on best
    n = best_n
    h = best_h.copy()
    if remaining_time() > 20:
        h_new, c5_new = coord_descent(h, n, n_sweeps=40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 9: Final SLSQP polish
    if remaining_time() > 10:
        h_new = slsqp_full(best_h, best_n, 1000000, maxiter=150, n_active=5)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 10: One last coordinate descent
    if remaining_time() > 8:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=20)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 11: Random perturbations to find better local optima
    n = best_n
    h = best_h.copy()
    for pert in range(15):
        if remaining_time() < 15:
            break
        n_perturb = max(5, n // 50)
        h_pert = h.copy()
        idx_p = rng.choice(n, n_perturb, replace=False)
        h_pert[idx_p] += rng.uniform(-0.08, 0.08, n_perturb)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.002:
            h_new = slsqp_interior(h_pert, n, 20000, maxiter=200, n_active=8)
            c5_new = compute_c5(h_new)
            if remaining_time() > 5:
                h_new2, c5_new2 = coord_descent(h_new, n, n_sweeps=15)
                if c5_new2 < c5_new:
                    h_new = h_new2
                    c5_new = c5_new2
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Ensure exact sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if isinstance(eval_c5, (int, float)) and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
