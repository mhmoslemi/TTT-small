def lor_loss_and_grad(h, n, beta):
    """Log-sum-exp loss and its gradient (scaled by dx)."""
    corr = compute_corr(h)
    m = float(np.max(corr))
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_vals = np.exp(shifted)
    denom = float(np.sum(exp_vals))
    lse_val = (m + np.log(denom) / beta) * (2.0 / n)
    w = exp_vals / denom
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off+np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off+np_]
        else:
            ao = -off
            np_ = n - ao
            g[ao:ao+np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[ao:ao+np_]
    return lse_val, g

    target = n / 2.0

    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    start = time.time()
    stall = 0
    
    while time.time() - start < time_limit and remaining_time() > 10:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        top_k = np.argsort(-corr)[:10]
        candidates = set()
        
        for s in top_k:
            off = s - (n - 1)
            if off >= 0:
                np_ = n - off
                contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for ci in contrib[:25]:
                    candidates.add((ci + off, ci))
            else:
                ao = -off
                np_ = n - ao
                contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for ci in contrib[:25]:
                    candidates.add((ci, ci + ao))
        
        for _ in range(40):
            candidates.add((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        best_move = None
        best_new_max = cur_max + 1e-15
        
        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            h[i] = 0
            h[j] = 1
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            h[i] = 1
            h[j] = 0
            
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j)
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > 400:
                for _ in range(5):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        h[i] = 0
                        h[j] = 1
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
    
    return best_h, best_max * dx

def top_shifts_equalize(h, n, top_k, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-12:
            break
        
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off+np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[ao:ao+np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[ao:ao+np_]
        
        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off+np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[ao:ao+np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[ao:ao+np_]
        
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.005, 0.01, 0.02, 0.03, 0.05, 0.002, 0.08, 0.015, 0.04, 0.001]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def greedy_construct(n, rng):
    h = np.zeros(n)
    n_half = n // 2
    used = set()
    positions = list(range(n))
    rng.shuffle(positions)
    
    for _ in range(n_half):
        best_pos = -1
        best_max = float('inf')
        for p in positions:
            if p in used:
                continue
            h[p] = 1
            corr = compute_corr(h)
            mx = float(np.max(corr))
            h[p] = 0
            if mx < best_max:
                best_max = mx
                best_pos = p
        if best_pos >= 0:
            h[best_pos] = 1
            used.add(best_pos)
        else:
            for p in positions:
                if p not in used:
                    h[p] = 1
                    used.add(p)
                    break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 384
        best_h = thue_morse(best_n)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary local search at parent resolution
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        if n_ones > n_half:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - n_half]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:n_half - n_ones]] = 1
    
    t1 = min(50, int(remaining_time() * 0.04))
    if t1 > 10:
        h_b1, c5_b1 = binary_local_search(h_bin, n, t1, rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
    
    # Phase 2: Continuous L-BFGS-B at parent resolution with increasing beta
    n = best_n
    h = best_h.copy()
    for beta in [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]:
        if remaining_time() < 30:
            break
        t_phase = min(35, int(remaining_time() * 0.03))
        if t_phase < 10:
            break
        h_new = lbfgs_optimize(h, n, beta, 50000.0, maxiter=800, time_limit=t_phase)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 3: Try alternate starting points at same resolution
    if remaining_time() > 120:
        starts = []
        starts.append(('thue', thue_morse(n)))
        starts.append(('greedy', greedy_construct(n, np.random.RandomState(seed + 200))))
        rand_h = (rng.rand(n) > 0.5).astype(float)
        ro = int(np.sum(rand_h))
        if ro != n // 2:
            if ro > n // 2:
                idx = np.where(rand_h == 1)[0]
                rand_h[idx[:ro - n // 2]] = 0
            else:
                idx = np.where(rand_h == 0)[0]
                rand_h[idx[:n // 2 - ro]] = 1
        starts.append(('rand', rand_h))
        
        for sname, s_h in starts:
            if remaining_time() < 60:
                break
            s_c5 = compute_c5(s_h)
            if s_c5 < best_c5 + 0.05:
                for beta in [100, 1000, 5000]:
                    if remaining_time() < 30:
                        break
                    h_new = lbfgs_optimize(s_h, n, beta, 50000.0, maxiter=500, time_limit=20)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
    
    # Phase 4: Higher resolution
    for n_t in [n + 64, n + 128, n + 256, n + 384]:
        if remaining_time() < 80:
            break
        if n_t > 1200:
            continue
        h_t = interp_to(best_h, best_n, n_t)
        h_t = project_feasible(h_t, n_t)
        for beta in [100, 500, 2000, 10000]:
            if remaining_time() < 25:
                break
            h_t = lbfgs_optimize(h_t, n_t, beta, 50000.0, maxiter=500, time_limit=20)
        c5_t = compute_c5(h_t)
        if remaining_time() > 15:
            h_t_eq, c5_t_eq = top_shifts_equalize(h_t, n_t, top_k=6, max_iters=200)
            if c5_t_eq < c5_t:
                c5_t = c5_t_eq
                h_t = h_t_eq
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t
    
    # Phase 5: Binary search at best resolution
    if remaining_time() > 40:
        n = best_n
        h_bin2 = (best_h > 0.5).astype(float)
        n2_half = n // 2
        n2_ones = int(np.sum(h_bin2))
        if n2_ones != n2_half:
            if n2_ones > n2_half:
                idx = np.where(h_bin2 == 1)[0]
                h_bin2[idx[:n2_ones - n2_half]] = 0
            else:
                idx = np.where(h_bin2 == 0)[0]
                h_bin2[idx[:n2_half - n2_ones]] = 1
        t5 = min(50, int(remaining_time() * 0.04))
        if t5 > 10:
            h_b2, c5_b2 = binary_local_search(h_bin2, n, t5, rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
    
    # Phase 6: Final continuous polish with high beta
    n = best_n
    h = best_h.copy()
    for beta in [5000, 10000, 20000, 50000, 100000]:
        if remaining_time() < 15:
            break
        t6 = min(25, int(remaining_time() * 0.02))
        if t6 < 5:
            break
        h_new = lbfgs_optimize(h, n, beta, 100000.0, maxiter=600, time_limit=t6)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 7: Top-shift equalization
    if remaining_time() > 15:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=8, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 8: Iterated local search (random perturbation + re-optimize)
    if remaining_time() > 30:
        nps = [5, 10, 20, 30]
        for npert in nps:
            if remaining_time() < 15:
                break
            h_pert = h.copy()
            idx_pert = rng.choice(n, min(npert, n), replace=False)
            h_pert[idx_pert] += rng.uniform(-0.1, 0.1, len(idx_pert))
            h_pert = project_feasible(h_pert, n)
            for beta in [10000, 30000]:
                if remaining_time() < 10:
                    break
                h_pert = lbfgs_optimize(h_pert, n, beta, 100000.0, maxiter=300, time_limit=10)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()
    
    # Phase 9: More top-shift equalization
    if remaining_time() > 10:
        h_eq2, c5_eq2 = top_shifts_equalize(h, n, top_k=5, max_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
    
    # Final: ensure feasibility and evaluate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
