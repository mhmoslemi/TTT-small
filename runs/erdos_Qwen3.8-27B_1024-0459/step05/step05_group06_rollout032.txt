def _obj_fn(x):
    return smooth_value_grad(x, _N_OPT, _BETA_OPT)[0]

def _grad_fn(x):
    return smooth_value_grad(x, _N_OPT, _BETA_OPT)[1]

def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _con_jac(x):
    return np.ones(_N_OPT)

def slsqp_optimize(h, n, beta, maxiter=400):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-13:
                return trial
    except Exception:
        pass
    return h.copy()

def solve_1d_minimax(A, B, lo=0.0, hi=1.0):
    """Solve min_t max_d (A[d] + B[d]*t) for t in [lo, hi]."""
    n = len(A)
    pos = B > 1e-15
    neg = B < -1e-15
    zero = ~(pos | neg)
    
    # Check if any zero-B component exceeds everything
    if np.any(zero):
        max_zero = np.max(A[zero])
    else:
        max_zero = -np.inf
    
    # Lower bound on v: at least max_zero
    v_lo = max(max_zero, np.max(np.minimum(A, A + B * hi)))
    # Upper bound on v: max over endpoints
    v_hi = max(np.max(A), np.max(A + B * hi))
    
    # If v_lo >= v_hi, just return the endpoint
    if v_lo >= v_hi - 1e-15:
        # Check which endpoint is better
        f0 = np.max(A)
        f1 = np.max(A + B * hi)
        if f0 <= f1:
            return lo, f0
        else:
            return hi, f1
    
    # Binary search on v
    for _ in range(60):
        v = (v_lo + v_hi) / 2.0
        # Check feasibility: exists t in [lo, hi] with A[d]+B[d]*t <= v for all d
        t_lower = lo
        t_upper = hi
        feasible = True
        if np.any(neg):
            candidates = (v - A[neg]) / B[neg]
            t_lower = max(t_lower, np.max(candidates))
        if np.any(pos):
            candidates = (v - A[pos]) / B[pos]
            t_upper = min(t_upper, np.min(candidates))
        if np.any(zero) and np.max(A[zero]) > v:
            feasible = False
        if t_lower > t_upper + 1e-15:
            feasible = False
        if feasible:
            v_hi = v
        else:
            v_lo = v
    
    v_opt = (v_lo + v_hi) / 2.0
    # Find optimal t
    t_lower = lo
    t_upper = hi
    if np.any(neg):
        candidates = (v_opt - A[neg]) / B[neg]
        t_lower = max(t_lower, np.max(candidates))
    if np.any(pos):
        candidates = (v_opt - A[pos]) / B[pos]
        t_upper = min(t_upper, np.min(candidates))
    
    # Clamp
    t_lower = max(t_lower, lo)
    t_upper = min(t_upper, hi)
    if t_lower > t_upper:
        t_opt = (lo + hi) / 2.0
    else:
        t_opt = (t_lower + t_upper) / 2.0
    
    return t_opt, np.max(A + B * t_opt)

def coord_descent_fast(h, n, n_sweeps=50, seed=0):
    """Fast coordinate descent using exact 1D minimax solution."""
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = compute_corr(best).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    rng = np.random.RandomState(seed)
    
    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        order = rng.permutation(n)
        improved_any = False
        for i in order:
            if remaining_time() < 1:
                break
            old_val = best[i]
            
            # Compute B[d] for this coordinate
            B = np.zeros(ncorr)
            # B[d_offset]: coefficient of h[i] in corr[d_offset]
            # d_offset = k - (n-1), where k is the index in the full correlation
            # For offset d (corresponding to shift d):
            #   h[i] appears as first factor when i+d is valid: contributes (1-h[i+d])
            #   h[i] appears as second factor when i-d is valid: contributes -h[i-d]
            # In the full correlation array (index k), offset = k - (n-1)
            # For offset d: valid range is j from max(0,-d) to min(n-1, n-1-d)
            
            # Simpler: for each k in [0, 2n-2], offset = k - (n-1)
            # h[i] contributes to corr[k] if:
            #   As h[j] with j=i: need i + offset in [0,n-1], i.e., offset in [-(i), n-1-i]
            #     contribution: (1 - h[i+offset])
            #   As h[j+offset] with j+offset=i, j=i-offset: need i-offset in [0,n-1], i.e., offset in [i-(n-1), i]
            #     contribution: -h[i-offset]
            
            for off in range(-(n-1), n):
                k = n - 1 + off
                b_val = 0.0
                if 0 <= i + off < n:
                    b_val += (1.0 - best[i + off])
                if 0 <= i - off < n:
                    b_val -= best[i - off]
                B[k] = b_val
            
            # A[k] = corr[k] - h[i] * B[k]
            A = corr - old_val * B
            
            # Solve min_t max_k (A[k] + B[k]*t) for t in [0,1]
            t_opt, val_opt = solve_1d_minimax(A, B, 0.0, 1.0)
            
            if abs(t_opt - old_val) > 1e-12:
                # Update corr incrementally
                delta = t_opt - old_val
                for off in range(-(n-1), n):
                    k = n - 1 + off
                    d_val = 0.0
                    if 0 <= i + off < n:
                        d_val += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        d_val -= best[i - off]
                    corr[k] += delta * d_val
                best[i] = t_opt
                
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    improved_any = True
        
        if not improved_any:
            break
    
    # Recompute to ensure accuracy
    best = project_feasible(best, n)
    final_c5 = compute_c5(best)
    return best, final_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_harmonic_init(n, a=0.3, b=0.1):
    """Create a harmonic initialization."""
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = 0.5 + a * np.cos(np.pi * (x - 1.0)) + b * np.cos(2.0 * np.pi * (x - 1.0))
    return project_feasible(h, n)

def make_bump_init(n, center=1.0, width=0.5, height=1.0):
    """Create a bump-like initialization."""
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = 0.5 + height * np.exp(-((x - center) / width) ** 2)
    return project_feasible(h, n)

def subgrad_step(h, n, n_active=5):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    threshold = max_corr - 0.5
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)
    
    n = best_n
    h = best_h.copy()
    
    # Phase 1: SLSQP with moderate beta for quick improvement (budget: ~40s)
    for beta in [2000, 10000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_optimize(h, n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    # Phase 2: Fast coordinate descent (budget: ~80s)
    if remaining_time() > 60:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=80, seed=seed)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 3: SLSQP with high beta (budget: ~60s)
    for beta in [50000, 200000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_optimize(h, n, beta, 350)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 4: More coordinate descent (budget: ~60s)
    if remaining_time() > 40:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=60, seed=seed+1)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Try diverse starting points (budget: ~150s)
    diverse_starts = []
    
    # Harmonic inits
    for a, b in [(0.3, 0.1), (0.4, -0.1), (0.25, 0.15), (0.35, 0.05), (0.45, 0.0)]:
        diverse_starts.append(make_harmonic_init(n, a, b))
    
    # Bump inits
    diverse_starts.append(make_bump_init(n, 1.0, 0.4, 0.8))
    diverse_starts.append(make_bump_init(n, 0.7, 0.3, 0.6))
    diverse_starts.append(make_bump_init(n, 1.3, 0.3, 0.6))
    
    # Random perturbations of best
    for scale in [0.1, 0.2, 0.35]:
        h_pert = best_h + rng.normal(0, scale, n)
        diverse_starts.append(project_feasible(h_pert, n))
    
    # Random 0/1 with smoothing
    for _ in range(3):
        h_rand = (rng.rand(n) < 0.5).astype(float)
        # Smooth slightly
        from scipy.ndimage import uniform_filter1d
        h_smooth = uniform_filter1d(h_rand, size=5)
        diverse_starts.append(project_feasible(h_smooth, n))
    
    for start in diverse_starts:
        if remaining_time() < 35:
            break
        # Quick SLSQP
        h_trial = slsqp_optimize(start, n, 5000, 200)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 + 0.01:
            # Promising, do more optimization
            h_trial = slsqp_optimize(h_trial, n, 50000, 300)
            c5_trial = compute_c5(h_trial)
            if remaining_time() > 15:
                h_trial, c5_trial = coord_descent_fast(h_trial, n, n_sweeps=30, seed=seed+100)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
    
    # Phase 6: Subgradient + SLSQP interleave (budget: ~100s)
    n = best_n
    h = best_h.copy()
    for it in range(300):
        if remaining_time() < 10:
            break
        grad = subgrad_step(h, n, 5)
        if grad is None:
            break
        base_step = 0.03 / (1.0 + it * 0.008)
        improved = False
        for sm in [1.0, 0.5, 2.0, 0.25, 0.75]:
            step = base_step * sm
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved:
            if it % 50 == 25 and remaining_time() > 30:
                h_new = slsqp_optimize(h, n, 50000, 250)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                if it > 200:
                    break
    
    # Phase 7: Try higher resolution (budget: ~120s)
    higher_res = set()
    for mult in [1.25, 1.5, 2.0]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if 64 <= rn <= 2000:
            higher_res.add(rn)
    higher_res = sorted(higher_res)
    
    for rn in higher_res:
        if remaining_time() < 50:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        
        for beta in [5000, 50000]:
            if remaining_time() < 15:
                break
            h_r = slsqp_optimize(h_r, rn, beta, 250)
            c5_r = compute_c5(h_r)
        
        if remaining_time() > 15:
            h_r, c5_r = coord_descent_fast(h_r, rn, n_sweeps=30, seed=seed+200)
        
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn
    
    # Phase 8: Final intensive coordinate descent (budget: ~80s)
    if remaining_time() > 40:
        h_new, c5_new = coord_descent_fast(best_h, best_n, n_sweeps=100, seed=seed+300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 9: Final SLSQP polish at very high beta
    if remaining_time() > 10:
        h_new = slsqp_optimize(best_h, best_n, 500000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 10: One more coordinate descent pass
    if remaining_time() > 10:
        h_new, c5_new = coord_descent_fast(best_h, best_n, n_sweeps=50, seed=seed+400)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Ensure sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
