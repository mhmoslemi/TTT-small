def slsqp_smooth(h, n, beta, maxiter=500):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    tgt = n / 2.0
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    cons = [{'type': 'eq', 'fun': lambda x, t=tgt: np.sum(x) - t,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k=6, max_iters=500):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]
        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        improved = False
        for step in [0.01, 0.03, 0.05, 0.02, 0.08, 0.005, 0.1, 0.015, 0.04, 0.06]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary SA at parent resolution
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    sa_time1 = min(400, int(remaining() * 0.35))
    if sa_time1 > 30:
        h_b1, c5_b1 = binary_sa_local_search(h_bin, n, time_limit=sa_time1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n

    # Phase 2: Restart binary SA from different starts at same n
    starts = [thue_morse(n), (thue_morse(n) > 0.5).astype(float) * 0 + 1.0]
    starts[1] = 1.0 - starts[1]
    # Adjust counts
    for sp in starts:
        sp_ones = int(np.sum(sp))
        if sp_ones != n_half:
            if sp_ones > n_half:
                idx = np.where(sp == 1)[0]
                sp[idx[:sp_ones - n_half]] = 0
            else:
                idx = np.where(sp == 0)[0]
                sp[idx[:n_half - sp_ones]] = 1
    for sp in starts:
        if remaining() < 40:
            break
        sp_c5 = compute_c5(sp)
        t_sa = min(150, int(remaining() * 0.12))
        if t_sa < 15:
            break
        h_fn, c5_fn = binary_sa_local_search(sp, n, time_limit=t_sa, rng=rng)
        if c5_fn < best_c5:
            best_c5 = c5_fn
            best_h = h_fn.copy()
            best_n = n

    # Phase 3: Resolution sweep with binary SA
    try_resolutions = []
    base = best_n
    for dn in [-64, -48, -32, -16, -12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 24, 32, 48, 64]:
        rn = base + dn
        if 128 <= rn <= 640 and rn % 2 == 0 and rn != base:
            try_resolutions.append(rn)
    try_resolutions = sorted(set(try_resolutions), key=lambda x: abs(x - base))

    for n_t in try_resolutions[:8]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        t_bi = min(60, int(remaining() * 0.06))
        if t_bi < 10:
            break
        h_b3, c5_b3 = binary_sa_local_search(h_bin_t, n_t, time_limit=t_bi, rng=rng)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()
            best_n = n_t

    # Phase 4: Continuous refinement at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [500, 2000, 10000, 50000, 200000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Top-shifts equalization
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=6, max_iters=800)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: More continuous refinement
    for beta in [50000, 200000, 500000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Try one finer resolution
    if remaining() > 80:
        n_fine = best_n + 32
        if n_fine % 2 != 0:
            n_fine += 1
        h_fine = interp_to(best_h, best_n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        for beta in [5000, 50000]:
            if remaining() < 15:
                break
            h_fine = slsqp_smooth(h_fine, n_fine, beta, maxiter=500)
        c5_fine = compute_c5(h_fine)
        if remaining() > 20:
            h_fine_e, c5_fine_e = top_shifts_equalize(h_fine, n_fine, top_k=6, max_iters=400)
            if c5_fine_e < c5_fine:
                h_fine = h_fine_e
                c5_fine = c5_fine_e
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine

    # Phase 8: Final polish at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [100000, 300000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    if remaining() > 10:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=8, max_iters=600)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
