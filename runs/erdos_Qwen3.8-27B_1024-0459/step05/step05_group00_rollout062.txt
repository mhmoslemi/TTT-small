        # Second position: j in [0, min(a, n_off)]: dc/dh[j] += -h[j+a]
        # (valid when 0 ≤ j ≤ n-1-a, i.e., j ≤ n_off-1)
        j2_hi = min(a, n_off)  # j in [0, j2_hi)
        if j2_hi > 0:
            g[0:j2_hi] -= h[a:a + j2_hi]
        # First position: j in [a, n): dc/dh[j] += (1-h[j-a])
        # (valid when a ≤ j ≤ n-1)
        if a < n:
            g[a:n] += (1.0 - h[0:n - a])
        # Overlap correction: in the overlap region j in [a, n_off), both positions contribute.
        # But I already added both, so the formula (1-h[j-a]) - h[j+a] is correct.
        # Wait, I added -h[j+a] for j in [0, j2_hi) and (1-h[j-a]) for j in [a, n).
        # In the overlap [a, n_off): I have (1-h[j-a]) from first position, and -h[j+a] from second.
        # But I only added -h[j+a] for j in [0, j2_hi). If j2_hi = min(a, n_off):
        #   If a ≤ n_off: j2_hi = a, so I added -h[j+a] for j in [0,a). NOT in [a, n_off).
        #   I need to also add -h[j+a] for j in [a, n_off) (the overlap region).
        
        # Let me just redo from scratch with the correct formula:
        g = np.zeros(n)
        # For offset < 0 (a = -offset > 0):
        # c_k = sum_{i=0}^{n-1-a} h[i+a]*(1-h[i])
        # 
        # h[j] as h[i+a] (i=j-a): valid if 0≤j-a≤n-1-a, i.e., a≤j≤n-1
        #   contribution: (1-h[j-a])
        # h[j] as h[i] (i=j): valid if 0≤j≤n-1-a, i.e., 0≤j≤n-1-a  
        #   contribution: -h[j+a]  (from d/dh[j] of (1-h[j])*h[j+a] = -h[j+a])
        
        # Region 1: j in [0, a-1] (only second position, if j ≤ n-1-a)
        r1_hi = min(a, n_off)  # j in [0, r1_hi)
        if r1_hi > 0:
            g[0:r1_hi] -= h[a:a + r1_hi]
        # Region 2: j in [a, n_off-1] (both positions, if a ≤ n_off-1)
        if a < n_off:
            g[a:n_off] += (1.0 - h[0:n_off - a]) - h[n_off:n_off + (n_off - a)]
            # h[j-a] for j in [a, n_off) → h[0, n_off-a)
            # h[j+a] for j in [a, n_off) → h[n_off, n_off + n_off - a) = h[n_off, 2*n_off)
            # But 2*n_off = 2*(n-a). Need 2*(n-a) ≤ n, i.e., n ≤ 2a, i.e., a ≥ n/2.
            # If a < n/2: 2*(n-a) > n, so h[j+a] goes out of bounds!
            # This means for j in [a, n_off) with j+a ≥ n, the first position is the only valid one.
            # j+a ≥ n → j ≥ n-a = n_off. But j < n_off, so j+a < 2*n_off.
            # Actually j+a for j in [a, n_off-1]: j+a in [2a, n_off-1+a) = [2a, n-1).
            # So j+a is always < n. Good, no out-of-bounds.
        # Region 3: j in [n_off, n-1] (only first position)
        if n_off < n:
            g[n_off:n] += (1.0 - h[0:a])
            # h[j-a] for j in [n_off, n) → h[n_off-a, n-a) = h[a, n-a)... 
            # n_off = n-a, so n_off-a = n-2a. Hmm.
            # j-a for j in [n_off, n-1] → [n_off-a, n-1-a] = [n-2a, n-1-a]
            # Need n-2a ≥ 0, i.e., a ≤ n/2.
            # If a > n/2: n-2a < 0, so for j in [n_off, 2a-1], j-a < 0 (invalid).
            # In that case, those j only have the "second position" but j > n_off-1 = n-1-a,
            # so second position is also invalid. These j have dc/dh[j] = 0.
        
    return g


def smooth_max_slsqp(h, n, beta, maxiter=500):
    def sm_obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-30:
            return m * (2.0 / n)
        return (m + np.log(s) / beta) * (2.0 / n)

    def sm_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-30:
            return np.zeros(n)
        w = ex / s
        g = np.zeros(n)
        # Only process lags with significant weight
        thresh = 1e-10
        active = w > thresh
        for k in range(2 * n - 1):
            if not active[k]:
                continue
            wk = w[k] * (2.0 / n)
            g += wk * corr_grad_component(x, k, n)
        return g

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(project_feasible(h, n)) + 1e-9:
                return h_new
    except Exception:
        pass
    return project_feasible(h, n)


def pairwise_cd(h, n, n_sweeps, n_evals, rng, time_budget_fn):
    """Pairwise coordinate descent maintaining sum constraint."""
    best_c5 = compute_c5(h)
    h = h.copy()
    no_improve = 0
    for sweep in range(n_sweeps):
        if time_budget_fn() < 5:
            break
        improved = False
        pairs = list(range(0, n - 1, 2))
        rng.shuffle(pairs)
        for i in pairs:
            j = i + 1
            C = h[i] + h[j]
            lo = max(0.0, C - 1.0)
            hi = min(1.0, C)
            if hi - lo < 1e-10:
                continue
            old_i = h[i]
            old_j = h[j]
            base_val = compute_c5(h)
            best_val = base_val
            best_x = old_i
            grid = np.linspace(lo, hi, n_evals)
            for t in grid:
                if abs(t - old_i) < 1e-10:
                    continue
                h[i] = t
                h[j] = C - t
                val = compute_c5(h)
                if val < best_val:
                    best_val = val
                    best_x = t
            if best_val < best_c5 - 1e-13:
                h[i] = best_x
                h[j] = C - best_x
                best_c5 = best_val
                improved = True
            else:
                h[i] = old_i
                h[j] = old_j
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 4:
                # Perturbation to escape local minimum
                scale = 0.03 + 0.02 * no_improve
                pert = rng.randn(n) * scale
                h_try = project_feasible(h + pert, n)
                if compute_c5(h_try) <= best_c5 + 0.002:
                    h = h_try
                no_improve = 0
                if sweep > n_sweeps - 3:
                    break
    return h, best_c5


def subgradient_step(h, n, K, beta, rng):
    """One subgradient descent step on top-K lags."""
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(50.0 * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for idx_k, w_k in zip(idx, weights):
        if w_k > 1e-12:
            g += w_k * corr_grad_component(h, idx_k, n)
    return g, np.max(corr)


def subgradient_descent(h, n, time_limit, budget_fn, rng):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    it = 0
    while time.time() - start < time_limit and budget_fn() > 5:
        it += 1
        g, c_max = subgradient_step(h, n, K=20, beta=50, rng=rng)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        base_step = 0.03 * n / gn
        step = base_step / (1.0 + it * 0.05)
        step = min(step, 0.2)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
    return best_h, best_c5


def cutting_plane_lp(h, n, K, delta):
    """LP with active set constraints + trust region."""
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g_k = corr_grad_component(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g_k
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g_k, h))
    # Trust region
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, delta - h + 1.0])
    # Fix: h_new - h <= delta AND h - h_new <= delta
    # h_new <= h + delta → row: h_new - delta*1 <= h → done above
    # h_new >= h - delta → -h_new <= delta - h → done above
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], float(res.x[-1])
    except Exception:
        pass
    return h.copy(), np.max(corr)


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def active_set_lp_solve(h, n, max_lags, time_budget_fn):
    """Iteratively grow active set and solve LP."""
    h = h.copy()
    best_c5 = compute_c5(h)
    active_lags = set()
    for iteration in range(50):
        if time_budget_fn() < 5:
            break
        corr = compute_corr(h)
        # Identify top lags
        top_idx = np.argsort(-corr)[:max_lags]
        c_max = np.max(corr)
        # Solve LP with these lags
        h_lp, t_lp = cutting_plane_lp(h, n, len(top_idx), 0.1)
        if t_lp < best_c5 - 1e-8:
            # Accept and line search
            best_val = compute_c5(h)
            for alpha in [1.0, 0.7, 0.5, 0.3]:
                h_try = project_feasible(h + alpha * (h_lp - h), n)
                c_try = compute_c5(h_try)
                if c_try < best_val:
                    h = h_try
                    best_val = c_try
                    break
            if best_val < best_c5:
                best_c5 = best_val
        else:
            break
    return h, best_c5


def make_random_init(n, rng):
    h = rng.rand(n)
    h = project_feasible(h, n)
    return h


def make_block_init(n, rng, n_blocks=8):
    """Create a block-structured initial point."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for b in range(n_blocks):
        val = 0.0 if b % 2 == 0 else 1.0
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        h[start:end] = val
    # Add random perturbation
    h += rng.randn(n) * 0.2
    h = project_feasible(h, n)
    return h


def make_sine_init(n, rng):
    """Sine-based initial point."""
    freq = rng.randint(2, 6)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    h = np.clip(h, 0, 1)
    h = project_feasible(h, n)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Get parent
    n = 512
    h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        n = len(parent_h)
        h = project_feasible(parent_h, n)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        best_n = n
    except Exception:
        h = make_block_init(n, rng, n_blocks=8)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        best_n = n

    # === Phase 1: Pairwise Coordinate Descent at parent resolution ===
    if remaining() > 30:
        t_limit = min(200.0, remaining() * 0.35)
        h, c5 = pairwise_cd(h, n, n_sweeps=80, n_evals=10, rng=rng,
                            time_budget_fn=lambda: _BUDGET - remaining() - 3.0)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # === Phase 2: Subgradient + SLSQP at parent resolution ===
    if remaining() > 40:
        h, c5 = subgradient_descent(h, n, time_limit=min(30, remaining() * 0.1),
                                    budget_fn=remaining, rng=rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        for beta in [500, 2000, 5000]:
            if remaining() < 10:
                break
            h = smooth_max_slsqp(h, n, beta, maxiter=400)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # === Phase 3: Active Set LP at parent resolution ===
    if remaining() > 30:
        h, c5 = active_set_lp_solve(h, n, max_lags=20, time_budget_fn=remaining)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # === Phase 4: Try higher resolutions ===
    res_candidates = []
    for mult in [1.33, 1.5, 1.75, 2.0, 2.5]:
        res = int(round(best_n * mult / 32) * 32)
        if 256 <= res <= 2048 and res % 2 == 0:
            res_candidates.append(res)
    res_candidates = sorted(set(res_candidates), key=lambda x: abs(x - best_n))

    for n_t in res_candidates[:4]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        h_cur = h_feas.copy()

        # Pairwise CD at new resolution
        if remaining() > 30:
            t_limit = min(60.0, remaining() * 0.3)
            h_cur, c5_cur = pairwise_cd(h_cur, n_t, n_sweeps=40, n_evals=8,
                                        rng=rng,
                                        time_budget_fn=lambda: _BUDGET - remaining() - 3.0)
            h_cur = project_feasible(h_cur, n_t)
            c5_cur = compute_c5(h_cur)
            if c5_cur > c5_int:
                h_cur = h_feas.copy()
                c5_cur = c5_int

        # SLSQP polish
        for beta in [3000, 10000]:
            if remaining() < 10:
                break
            h_cur = smooth_max_slsqp(h_cur, n_t, beta, maxiter=300)
            c5_cur = compute_c5(h_cur)

        # Active set LP
        if remaining() > 20:
            h_cur, c5_cur = active_set_lp_solve(h_cur, n_t, max_lags=15,
                                                time_budget_fn=remaining)

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_t

    # === Phase 5: Try different initial structures at parent resolution ===
    n_work = min(best_n, 512)  # Work at reasonable resolution
    inits = []
    inits.append(make_block_init(n_work, rng, n_blocks=6))
    inits.append(make_block_init(n_work, rng, n_blocks=12))
    inits.append(make_sine_init(n_work, rng))
    inits.append(make_random_init(n_work, rng))

    for h_init in inits[:3]:
        if remaining() < 40:
            break
        h_cur = h_init.copy()
        c5_cur = compute_c5(h_cur)
        if c5_cur >= best_c5:
            # Only worth trying if close
            c5_cur = compute_c5(h_cur)
            if c5_cur > best_c5 + 0.02:
                continue

        # Quick pairwise CD
        if remaining() > 20:
            h_cur, c5_cur = pairwise_cd(h_cur, n_work, n_sweeps=30, n_evals=8,
                                        rng=rng,
                                        time_budget_fn=lambda: _BUDGET - remaining() - 3.0)
            h_cur = project_feasible(h_cur, n_work)
            c5_cur = compute_c5(h_cur)

        # Quick SLSQP
        if remaining() > 10:
            h_cur = smooth_max_slsqp(h_cur, n_work, 5000, maxiter=200)
            c5_cur = compute_c5(h_cur)

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_work

    # === Phase 6: Intensive refinement at best resolution ===
    n_work = best_n
    h_work = best_h.copy()

    # Alternating CD + SLSQP
    for cycle in range(3):
        if remaining() < 20:
            break
        # Pairwise CD
        if remaining() > 15:
            h_work, c5_tmp = pairwise_cd(h_work, n_work, n_sweeps=30, n_evals=12,
                                         rng=rng,
                                         time_budget_fn=lambda: _BUDGET - remaining() - 3.0)
            h_work = project_feasible(h_work, n_work)
            c5_tmp = compute_c5(h_work)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h_work.copy()

        # SLSQP
        if remaining() > 10:
            for beta in [10000, 30000]:
                if remaining() < 5:
                    break
                h_work = smooth_max_slsqp(h_work, n_work, beta, maxiter=500)
                c5_tmp = compute_c5(h_work)
                if c5_tmp < best_c5:
                    best_c5 = c5_tmp
                    best_h = h_work.copy()

        # LP
        if remaining() > 15:
            h_work, c5_tmp = active_set_lp_solve(h_work, n_work, max_lags=25,
                                                 time_budget_fn=remaining)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h_work.copy()
                n_work = best_n
                h_work = best_h.copy()

    # === Phase 7: Final fine-tuning ===
    if remaining() > 10:
        # Very fine pairwise CD
        h_fine, c5_fine = pairwise_cd(best_h, best_n, n_sweeps=40, n_evals=20,
                                      rng=rng,
                                      time_budget_fn=lambda: _BUDGET - remaining() - 3.0)
        h_fine = project_feasible(h_fine, best_n)
        c5_fine = compute_c5(h_fine)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Final SLSQP polish
    if remaining() > 5:
        best_h = smooth_max_slsqp(best_h, best_n, 50000, maxiter=600)
        c5_final = compute_c5(best_h)
        if c5_final < best_c5:
            best_c5 = c5_final

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
