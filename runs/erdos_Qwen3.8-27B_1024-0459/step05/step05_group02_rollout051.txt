```python
def swap_delta_binary(h, i, j, k, n):
    """Compute change in corr at full-correlation index k after swapping h[i]:1→0, h[j]:0→1. Both are binary."""
    offset = k - (n - 1)
    delta = 0.0
    
    # i as first index
    if 0 <= i + offset < n:
        delta -= (1 - h[i + offset])
    # j as first index
    if 0 <= j + offset < n:
        delta += (1 - h[j + offset])
    # i as second index (i' = i - offset)
    if 0 <= i - offset < n:
        delta += h[i - offset]
    # j as second index (j' = j - offset)
    if 0 <= j - offset < n:
        delta -= h[j - offset]
    
    return delta
```

```python
def simulated_annealing(h_bin, n, iters=50000, t_init=0.5, t_final=0.01):
    """Simulated annealing on binary swaps."""
    n_half = n // 2
    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_h = h_bin.copy()
    cur_c5 = best_c5

    for it in range(iters):
        if remaining() < 3:
            break

        T = t_init * (t_final / t_init) ** (it / float(iters))

        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = int(ones[np.random.randint(len(ones))])
        j = int(zeros[np.random.randint(len(zeros))])
        if i == j:
            continue

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5

        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_c5


def continuous_refine(h, n, max_beta=20000):
    """Quick continuous refinement using L-BFGS-B (for fractional coordinate)."""

    target = n / 2.0

        idx = np.argsort(-corr)[:5]
        c = corr[idx]
        m = c[0]
        sh = _gB * (c - m)
        sh = np.clip(sh, -50.0, 50.0)
        ex = np.exp(sh)
        val = (m + np.log(np.sum(ex)) / _gB) * dx
        penalty = 100.0 * (np.sum(x) - target) ** 2
        return val + penalty

    def sm_grad(x):
        corr = compute_corr(x)
        idx = np.argsort(-corr)[:5]
        c = corr[idx]
        m = c[0]
        sh = _gB * (c - m)
        sh = np.clip(sh, -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for jj, ki in enumerate(idx):
            offset = ki - (n - 1)
            if offset >= 0:
                L = n - offset
                g[offset:offset + L] += (1.0 - x[:L])
                g[:L] -= x[offset:offset + L]
            else:
                ao = -offset
                L = n - ao
                g[:L] += (1.0 - x[ao:ao + L])
                g[ao:ao + L] -= x[:L]
            g *= 1.0  # per-lag gradient
        # Weight combination
        g_final = np.zeros(n)
        for jj, ki in enumerate(idx):
            offset = ki - (n - 1)
            tmp = np.zeros(n)
            if offset >= 0:
                L = n - offset
                tmp[offset:offset + L] += (1.0 - x[:L])
                tmp[:L] -= x[offset:offset + L]
            else:
                ao = -offset
                L = n - ao
                tmp[:L] += (1.0 - x[ao:ao + L])
                tmp[ao:ao + L] -= x[:L]
            g_final += w[jj] * tmp
        g_final *= dx
        g_final += 200.0 * (np.sum(x) - target) * np.ones(n) / n
        return g_final

    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n

    for beta in [5000, 20000]:
        if remaining() < 5:
            break
        _gN = n
        _gB = beta
        _gK = 5
        _gT = target
        try:

                           bounds=bounds,

            if compute_c5(h_new) < compute_c5(h):

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def random_binary(n, rng):
    """Generate random binary vector with n/2 ones."""
    n_half = n // 2
    h = np.zeros(n)
    idx = rng.choice(n, n_half, replace=False)
    h[idx] = 1.0
    return h


def structured_binary(n, rng, n_blocks=8):
    """Generate structured binary with alternating blocks."""
    n_half = n // 2
    h = np.zeros(n)
    block_size = n // n_blocks
    positions = rng.choice(n, 2 * n_blocks, replace=False)
    positions = np.sort(positions)
    for k in range(0, 2 * n_blocks, 2):
        start = positions[k]
        end = min(start + block_size, n)
        h[start:end] = 1.0
    # Fix to exactly n/2 ones
    h_bin = (h > 0.3).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > n_half:
        one_idx = np.where(h_bin == 1)[0]
        order = rng.permutation(len(one_idx))
        h_bin[one_idx[order[:n_ones - n_half]]] = 0.0
    elif n_ones < n_half:
        zero_idx = np.where(h_bin == 0)[0]
        order = rng.permutation(len(zero_idx))
        h_bin[zero_idx[order[:n_half - n_ones]]] = 1.0
    return h_bin

            best_h = h_cont.copy()

    # Phase 7: Final intensive binary polish
    if remaining() > 40:
        h_final_bin = to_binary(best_h, best_n)
        h_fp, c5_fp = wide_binary_ls(h_final_bin, best_n, max_iters=15000, n_candidates=50)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()
        # One more ILS
        if remaining() > 20:
            h_fm, c5_fm = ily_local_search(h_fp, best_n, n_restarts=5, perturb_size=4, max_iters_per_restart=3000)
            if c5_fm < best_c5:
                best_c5 = c5_fm
                best_h = h_fm.copy()

    # Phase 8: Final continuous micro-adjustment
    if remaining() > 15:
        h_micro = best_h.copy()
        h_micro = proj(h_micro, best_n)
        # Try adjusting individual coordinates
        n_pts = best_n
        dx_pts = 2.0 / n_pts
        for _ in range(3):
            if remaining() < 5:
                break
            improved = False
            for trial in range(200):
                if remaining() < 3:
                    break
                idx = rng.randint(n_pts)
                old_val = h_micro[idx]
                for new_val in [0.0, 1.0, min(1.0, old_val + 0.1), max(0.0, old_val - 0.1),
                                min(1.0, old_val + 0.05), max(0.0, old_val - 0.05)]:
                    h_micro[idx] = new_val
                    h_micro = proj(h_micro, n_pts)
                    c5_try = compute_c5(h_micro)
                    if c5_try < best_c5 - 1e-10:
                        best_c5 = c5_try
                        improved = True
                        break
                    h_micro[idx] = old_val
            if not improved:
                break

    # Final validation
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5  # use the actual value

    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
