import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:

def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin

def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(len(ones_idx), n_ones - n_half, replace=False)
            else:
                rem = np.arange(n_ones - n_half)
            h[ones_idx[rem]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(len(zeros_idx), n_half - n_ones, replace=False)
            else:
                add = np.arange(n_half - n_ones)
            h[zeros_idx[add]] = 1
    return h

def smooth_objective(x, n, beta):
    """Smooth approximation to max correlation."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    # Softmax
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (m + log_sum / beta) * (2.0 / n)

def smooth_gradient(x, n, beta):
    """Gradient of smooth objective."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    g = np.zeros(n)
    wc = w * dx
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off+np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off+np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao+np_])
            g[ao:ao+np_] -= wk * x[:np_]
    return g

def slsqp_optimize(h, n, beta, maxiter=300):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_objective_cur, h, jac=smooth_gradient_cur,
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.success or res.fun < smooth_objective_cur(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def smooth_objective_cur(x):
    return smooth_objective(x, _N_cur, _BETA_cur)
