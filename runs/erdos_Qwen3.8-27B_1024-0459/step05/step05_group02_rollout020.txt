def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        # Distribute among a few elements
        idx = np.argsort(np.abs(h - 0.5))[:5]
        for i in idx:
            adj = min(max(diff, -h[i]), 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        if np_ <= 0:
            return g
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        if np_ <= 0:
            return g
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

    zeros = np.where(h_bin == 0)[0]

            h_bin[i] = 1

def projected_gradient_nesterov(h, n, top_k=5, max_iters=10000, step_init=0.03):

    best_h = h.copy()

    v = np.zeros(n)

    step = step_init

            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - np.mean(c_top)) + 0.1
        w = np.maximum(w, 0.01)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.003:
            h = h_new
            if c5_new <= c5:

        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.7

        if it % 1000 == 0 and it > 0:
            step *= 0.85
            momentum = min(momentum + 0.005, 0.95)

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):

    return np.interp(x_new, xp, src_h)

        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - np.mean(c_top)) + 0.1
        w = np.maximum(w, 0.01)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.003:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.02, 0.15)
        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.7

        if it % 1000 == 0 and it > 0:
            step *= 0.85
            momentum = min(momentum + 0.005, 0.95)

    return best_h, best_c5


def top_equalize(h, n, top_k=5, max_iters=500):
    h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-8:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            d = -g_max
            dn = np.linalg.norm(d)
            if dn < 1e-13:
                break
        d /= dn

        improved = False
        for scale in [5, 3, 1, 2, 0.5, 8, 1.5, 0.1, 4, 0.2, 0.05, 0.8]:
            alpha = 0.02 * scale
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for scale in [3, 1, 5, 2, 0.3, 0.1]:
                alpha = 0.01 * scale
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def optimize_at_resolution(h_init, n, time_limit_frac):
    """Run a full optimization pipeline at a given resolution, respecting a fraction of remaining time."""
    t_limit = time_limit_frac * remaining()
    h = proj(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()

    def elapsed_frac():
        return (time.time() - t_start) / max(t_limit, 1.0)

    # L-BFGS passes
    for beta in [500, 3000, 15000]:
        if elapsed_frac() > 0.25 or remaining() < 10:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=700)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Nesterov gradient
    if elapsed_frac() < 0.4 and remaining() > 15:
        h_pg, c5_pg = projected_gradient_nesterov(h, n, top_k=5, max_iters=6000, step_init=0.025)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Top equalize
    if elapsed_frac() < 0.5 and remaining() > 10:
        h_eq, c5_eq = top_equalize(h, n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # More L-BFGS
    for beta in [50000, 120000]:
        if elapsed_frac() > 0.7 or remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=4, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n0 = best_n
    h0 = best_h.copy()

    # === Phase 1: Optimize at parent resolution ===
    h_opt, c5_opt = optimize_at_resolution(h0, n0, 0.18)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
        best_n = n0

    # === Phase 2: Aggressive binary search at parent resolution ===
    if remaining() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_bl, c5_bl = binary_search_aggressive(h_bin, best_n, max_iters=20000, top_k=4, seed=seed)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Refine binary result continuously
    if remaining() > 25:
        h_ref = best_h.copy()
        for beta in [5000, 30000]:
            if remaining() < 8:
                break
            h_ref = lbfgs_pass(h_ref, best_n, beta, top_k=4, maxiter=400)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # === Phase 3: Simulated annealing on binary ===
    if remaining() > 80:
        h_bin2 = (best_h > 0.5).astype(float)
        h_sa, c5_sa = binary_sa_full(h_bin2, best_n, max_iters=50000, t_init=0.012, t_final=0.0002, seed=seed+1)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

        # Refine
        if remaining() > 15:
            h_ref2 = best_h.copy()
            h_ref2 = lbfgs_pass(h_ref2, best_n, 20000, top_k=4, maxiter=400)
            c5_ref2 = compute_c5(h_ref2)
            if c5_ref2 < best_c5:
                best_c5 = c5_ref2
                best_h = h_ref2.copy()

    # === Phase 4: Higher resolution (n=512) ===
    n_h = 512
    if n_h != best_n and remaining() > 120:
        h_h = interp_to(best_h, best_n, n_h)
        h_h = proj(h_h, n_h)
        h_h_opt, c5_h_opt = optimize_at_resolution(h_h, n_h, 0.25)
        if c5_h_opt < best_c5:
            best_c5 = c5_h_opt
            best_h = h_h_opt.copy()
            best_n = n_h

        # Binary search at n=512
        if remaining() > 60:
            h_bin_h = (best_h > 0.5).astype(float)
            h_bl_h, c5_bl_h = binary_search_aggressive(h_bin_h, n_h, max_iters=15000, top_k=4, seed=seed+10)
            if c5_bl_h < best_c5:
                best_c5 = c5_bl_h
                best_h = h_bl_h.copy()

            # Refine
            if remaining() > 15:
                h_ref_h = best_h.copy()
                h_ref_h = lbfgs_pass(h_ref_h, n_h, 20000, top_k=4, maxiter=400)
                c5_ref_h = compute_c5(h_ref_h)
                if c5_ref_h < best_c5:
                    best_c5 = c5_ref_h
                    best_h = h_ref_h.copy()

    # === Phase 5: Even higher resolution (n=768) ===
    n_hh = 768
    if n_hh != best_n and remaining() > 100:
        h_hh = interp_to(best_h, best_n, n_hh)
        h_hh = proj(h_hh, n_hh)
        h_hh_opt, c5_hh_opt = optimize_at_resolution(h_hh, n_hh, 0.2)
        if c5_hh_opt < best_c5:
            best_c5 = c5_hh_opt
            best_h = h_hh_opt.copy()
            best_n = n_hh

        # Binary search at n=768
        if remaining() > 50:
            h_bin_hh = (best_h > 0.5).astype(float)
            h_bl_hh, c5_bl_hh = binary_search_aggressive(h_bin_hh, n_hh, max_iters=12000, top_k=4, seed=seed+20)
            if c5_bl_hh < best_c5:
                best_c5 = c5_bl_hh
                best_h = h_bl_hh.copy()

            # Refine
            if remaining() > 12:
                h_ref_hh = best_h.copy()
                h_ref_hh = lbfgs_pass(h_ref_hh, n_hh, 20000, top_k=4, maxiter=300)
                c5_ref_hh = compute_c5(h_ref_hh)
                if c5_ref_hh < best_c5:
                    best_c5 = c5_ref_hh
                    best_h = h_ref_hh.copy()

    # === Phase 6: Final intensive refinement at best resolution ===
    n = best_n
    h = best_h.copy()

    # Multiple L-BFGS with increasing beta
    for beta in [10000, 50000, 150000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=3, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final Nesterov
    if remaining() > 15:
        h_pg2, c5_pg2 = projected_gradient_nesterov(h, n, top_k=4, max_iters=4000, step_init=0.015)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()

    # Final top equalize
    if remaining() > 8:
        h_eq2, c5_eq2 = top_equalize(best_h, best_n, top_k=6, max_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final binary polish
    if remaining() > 20:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_search_aggressive(h_bin_f, best_n, max_iters=10000, top_k=3, seed=seed+100)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

        # Quick continuous refine
        if remaining() > 10:
            h_rf = best_h.copy()
            h_rf = lbfgs_pass(h_rf, best_n, 50000, top_k=3, maxiter=300)
            c5_rf = compute_c5(h_rf)
            if c5_rf < best_c5:
                best_c5 = c5_rf
                best_h = h_rf.copy()

    # Final SA kick
    if remaining() > 30:
        h_bin_k = (best_h > 0.5).astype(float)
        h_sk, c5_sk = binary_sa_full(h_bin_k, best_n, max_iters=20000, t_init=0.005, t_final=0.0001, seed=seed+200)
        if c5_sk < best_c5:
            best_c5 = c5_sk
            best_h = h_sk.copy()
        if remaining() > 8:
            h_fk = best_h.copy()
            h_fk = lbfgs_pass(h_fk, best_n, 30000, top_k=4, maxiter=300)
            c5_fk = compute_c5(h_fk)
            if c5_fk < best_c5:
                best_c5 = c5_fk
                best_h = h_fk.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    # Validate with harness function
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
