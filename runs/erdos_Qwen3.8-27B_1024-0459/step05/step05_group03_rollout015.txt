        if no_improve > patience:
            T = max(T * 1.5, 0.1)
            no_improve = 0

        T = max(T * decay, T_end)

    return best_h, best_c5


def greedy_binary_improve(h_bin, n, max_rounds=50):
    h = h_bin.astype(float).copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    for rnd in range(max_rounds):
        if remaining() < 10:
            break
        corr = compute_corr_full(h)
        c_max = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        best_imp = 1e-15
        best_pair = None

        for ia in range(len(ones)):
            pi = ones[ia]
            h[pi] = 0.0
            for ib in range(len(zeros)):
                pj = zeros[ib]
                if pj == pi:
                    continue
                h[pj] = 1.0
                c_new = float(np.max(compute_corr_full(h)))
                h[pj] = 0.0
                imp = c_max - c_new
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (pi, pj)
            h[pi] = 1.0

        if best_pair is None:
            break
        pi, pj = best_pair
        h[pi] = 0.0
        h[pj] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse(n):
    seq = np.zeros(n, dtype=float)
    for i in range(n):
        c = bin(i).count('1')
        seq[i] = float(c % 2)
    ones = np.where(seq == 1)[0]
    zeros = np.where(seq == 0)[0]
    if len(ones) < n // 2:
        for _ in range(n // 2 - len(ones)):
            z = zeros[0]
            seq[z] = 1.0
            zeros = zeros[1:]
    elif len(ones) > n // 2:
        for _ in range(len(ones) - n // 2):
            o = ones[0]
            seq[o] = 0.0
            ones = ones[1:]
    return seq


def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h


def subgrad_equalize(h, n, top_k=5, max_iters=500):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.9
    step = 0.03

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr_full(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w_top = c_top - c_top.min() + 1e-10
        w_top = w_top ** 2
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= dx

        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if abs(wk) < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:a + np_])
                g[a:a + np_] -= wk * h[:np_]

        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        step *= 0.9998
        if it % 100 == 0:
            mu = min(0.97, mu + 0.005)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial values
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    # Phase 1: SLSQP optimization at parent resolution
    n = parent_n
    h = project_feasible(parent_h, n)
    c5_init = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_init
    best_n = n

    for beta in [50, 200, 800, 3000, 10000, 30000]:
        if remaining() < 20:
            break
        h_opt = slsqp_optimize(h, n, beta, maxiter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 2: Round to binary and greedy improve
    h_bin = round_to_binary(best_h, best_n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    h_g, c5_g = greedy_binary_improve(h_bin, best_n, max_rounds=30)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
        h_bin = h_g.copy()

    # Phase 3: Simulated annealing on binary (main search)
    if remaining() > 60:
        h_sa, c5_sa = sim_anneal_binary(h_bin, best_n, max_iters=1500000,
                                        T_start=0.3, T_end=0.0003, seed=seed)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_bin = h_sa.copy()

    # Phase 4: More SA with different parameters
    if remaining() > 60:
        h_sa2, c5_sa2 = sim_anneal_binary(h_bin, best_n, max_iters=1000000,
                                          T_start=0.15, T_end=0.0001, seed=seed + 1)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            h_bin = h_sa2.copy()

    # Phase 5: Greedy after SA
    if remaining() > 30:
        h_g2, c5_g2 = greedy_binary_improve(h_bin, best_n, max_rounds=20)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()
            h_bin = h_g2.copy()

    # Phase 6: Subgradient equalization on continuous relaxation
    if remaining() > 30:
        h_cont = h_bin.astype(float)
        for beta in [2000, 8000, 20000]:
            if remaining() < 15:
                break
            h_cont = slsqp_optimize(h_cont, best_n, beta, maxiter=200)
        h_sg, c5_sg = subgrad_equalize(h_cont, best_n, top_k=5, max_iters=400)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 7: Multi-resolution exploration
    n_candidates = set()
    for scale in [1.33, 1.5, 2.0, 0.75]:
        nn = int(best_n * scale)
        if nn >= 128 and nn <= 1024 and nn % 2 == 0:
            n_candidates.add(nn)
    n_candidates.discard(best_n)
    n_candidates = sorted(n_candidates)

    for n_t in n_candidates[:3]:
        if remaining() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_interp = project_feasible(h_interp, n_t)

        # SLSQP at new resolution
        for beta in [500, 3000, 15000]:
            if remaining() < 20:
                break
            h_interp = slsqp_optimize(h_interp, n_t, beta, maxiter=250)
        c5_int = compute_c5(h_interp)

        # Round and SA
        h_bin_t = round_to_binary(h_interp, n_t)
        c5_bin_t = compute_c5(h_bin_t)
        h_g_t, c5_g_t = greedy_binary_improve(h_bin_t, n_t, max_rounds=15)
        c5_g_t = min(c5_bin_t, c5_g_t)
        h_best_t = h_bin_t if c5_bin_t <= c5_g_t else h_g_t

        if remaining() > 40:
            h_sa_t, c5_sa_t = sim_anneal_binary(h_best_t, n_t, max_iters=800000,
                                                T_start=0.25, T_end=0.0003, seed=seed + 10)
            if c5_sa_t < c5_g_t:
                c5_g_t = c5_sa_t
                h_best_t = h_sa_t

        c5_t = min(c5_int, c5_g_t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_n = n_t
            if c5_int <= c5_g_t:
                best_h = h_interp.copy()
            else:
                best_h = h_best_t

    # Phase 8: Try alternative starting points
    if remaining() > 80:
        n_t = best_n
        # Thue-Morse
        h_tm = thue_morse(n_t)
        h_tm_g, c5_tm = greedy_binary_improve(h_tm, n_t, max_rounds=10)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm_g.copy()

        if remaining() > 40:
            h_tm_sa, c5_tm_sa = sim_anneal_binary(h_tm_g, n_t, max_iters=600000,
                                                  T_start=0.2, T_end=0.0005, seed=seed + 20)
            if c5_tm_sa < best_c5:
                best_c5 = c5_tm_sa
                best_h = h_tm_sa.copy()

    # Phase 9: Final intensive SA from best
    if remaining() > 50:
        h_bin_final = round_to_binary(best_h, best_n)
        h_final_sa, c5_final_sa = sim_anneal_binary(h_bin_final, best_n,
                                                    max_iters=1000000,
                                                    T_start=0.1, T_end=0.0001,
                                                    seed=seed + 30)
        if c5_final_sa < best_c5:
            best_c5 = c5_final_sa
            best_h = h_final_sa.copy()

    # Phase 10: Final greedy + SLSQP polish
    if remaining() > 20:
        h_bin_fin = round_to_binary(best_h, best_n)
        h_fin_g, c5_fin_g = greedy_binary_improve(h_bin_fin, best_n, max_rounds=15)
        if c5_fin_g < best_c5:
            best_c5 = c5_fin_g
            best_h = h_fin_g.copy()

    if remaining() > 15:
        h_fin = project_feasible(best_h, best_n)
        h_fin = slsqp_optimize(h_fin, best_n, 50000, maxiter=200)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
