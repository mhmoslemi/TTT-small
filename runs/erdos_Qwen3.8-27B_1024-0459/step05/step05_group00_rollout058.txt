def active_slsqp(h, n, active_set, beta, maxiter=400):
    def obj(x):
        c = compute_corr(x)
        ca = c[active_set]
        m = np.max(ca)
        cl = np.clip(beta * (ca - m), -80.0, 80.0)
        ex = np.exp(cl)
        return (m + np.log(ex.sum()) / beta) * (2.0 / n)

    def grad(x):
        c = compute_corr(x)
        ca = c[active_set]
        m = np.max(ca)
        cl = np.clip(beta * (ca - m), -80.0, 80.0)
        ex = np.exp(cl)
        wa = ex / ex.sum()
        g = np.zeros(n)
        for j, k in enumerate(active_set):
            if wa[j] < 1e-10:
                continue
            ak = (2.0 / n) * wa[j]
            off = k - (n - 1)
            if off >= 0:
                a = off
                if a == 0:
                    g += ak * (1.0 - 2.0 * x)
                else:
                    g[:n - a] += ak * (1.0 - x[a:])
                    g[a:] -= ak * x[:n - a]
            else:
                a = -off
                g[a:] += ak * (1.0 - x[:n - a])
                g[:n - a] -= ak * x[a:]
        return g

    def ef(x):
        return np.sum(x) - n / 2.0

    def ej(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ef, 'jac': ej}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def subgrad_true_max(h, n, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step_base = 0.02
    for it in range(2000):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        active = np.where(np.abs(corr - c_max) < 1e-8)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grads = []
        for k in active:
            g = np.zeros(n)
            off = k - (n - 1)
            if off >= 0:
                a = off
                if a == 0:
                    g = (1.0 - 2.0 * h) * (2.0 / n)
                else:
                    g[:n - a] = (2.0 / n) * (1.0 - h[a:])
                    g[a:] = -(2.0 / n) * h[:n - a]
            else:
                a = -off
                g[a:] = (2.0 / n) * (1.0 - h[:n - a])
                g[:n - a] = -(2.0 / n) * h[a:]
            grads.append(g)

        if len(active) == 1:
            g_min = grads[0]
        elif len(active) > 1:
            A = np.array(grads)
            m = len(active)
            G = A.T @ A
            try:
                def qf(w):
                    return w @ G @ w
                def qf_g(w):
                    return 2.0 * G @ w
                w0 = np.ones(m) / m
                res_q = minimize(qf, w0, jac=qf_g, method='SLSQP',
                                 bounds=[(0, 1)] * m,
                                 constraints=[{'type': 'eq', 'fun': lambda w: np.sum(w) - 1,
                                               'jac': lambda w: np.ones(m)}],
                                 options={'maxiter': 100})
                w_opt = np.clip(res_q.x, 0, None)
                w_opt /= w_opt.sum()
            except Exception:
                w_opt = np.ones(m) / m
            g_min = w_opt @ A
        else:
            break

        gn = np.linalg.norm(g_min)
        if gn < 1e-14:
            break

        decay = 1.0 / (1.0 + 0.02 * it)
        step = step_base * decay
        h_try = project_feasible(h - step * g_min / gn * gn, n)
        h_try = project_feasible(h - step * g_min, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            h = h_try
        else:
            for f in [0.5, 0.25, 0.1]:
                h_t2 = project_feasible(h - step * f * g_min, n)
                c5_t2 = compute_c5(h_t2)
                if c5_t2 < best_c5:
                    best_c5 = c5_t2
                    h = h_t2
                    break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def pick_shift_pair(h, n, idx):
    a = idx[0]
    b = idx[-1]
    diff = h.max() - h.min()
    if diff < 1e-10:
        return a, b
    target_a = 0.3
    h_shifted = h - h.mean()
    corr_ab = np.correlate(h_shifted, h_shifted, mode='full')
    peak = np.argmax(corr_ab)
    shift = peak - (n - 1)
    return a, b


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 512
        h = np.zeros(n)
        h[:n // 2] = 1.0

    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_n = n

    # Phase 1: SLSQP with increasing beta
    for beta in [2000, 8000, 30000, 80000, 150000]:
        if remaining() < 15:
            break
        h_new = slsqp_run(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            break

    # Phase 2: True max subgradient descent
    if remaining() > 30:
        sg_time = min(50.0, remaining() * 0.08)
        h_sg, c5_sg = subgrad_true_max(h, n, sg_time)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: Active set SLSQP
    if remaining() > 20:
        for K in [3, 5, 8, 12]:
            if remaining() < 10:
                break
            corr = compute_corr(h)
            active_idx = np.argsort(-corr)[:K]
            for beta in [10000, 50000]:
                if remaining() < 8:
                    break
                h_new = active_slsqp(h, n, active_idx, beta, maxiter=300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 4: SLSQP again after subgradient
    for beta in [50000, 100000, 200000]:
        if remaining() < 10:
            break
        h_new = slsqp_run(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: More subgradient
    if remaining() > 20:
        sg_time = min(40.0, remaining() * 0.06)
        h_sg, c5_sg = subgrad_true_max(h, n, sg_time)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 6: Cutting plane LP for direction
    if remaining() > 30:
        for it_cp in range(200):
            if remaining() < 5:
                break
            corr = compute_corr(h)
            c_max = np.max(corr)
            idx_top = np.argsort(-corr)[:20]

            c_obj = np.zeros(n + 1)
            c_obj[-1] = 1.0
            A_ub_list = []
            b_ub_list = []

            for k in idx_top:
                g = np.zeros(n)
                off = k - (n - 1)
                if off >= 0:
                    a = off
                    if a == 0:
                        g = 1.0 - 2.0 * h
                    else:
                        g[:n - a] = (1.0 - h[a:])
                        g[a:] = -h[:n - a]
                else:
                    a = -off
                    g[a:] = (1.0 - h[:n - a])
                    g[:n - a] = -h[a:]

                row = np.zeros(n + 1)
                row[:n] = g
                row[-1] = -1.0
                A_ub_list.append(row)
                b_ub_list.append(-corr[k] + np.dot(g, h))

            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([n / 2.0])

            delta = 0.03
            I = np.eye(n)
            A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                                 np.hstack([-I, np.zeros((n, 1))])])
            b_trust = np.concatenate([h + delta, h + delta])

            A_ub = np.vstack([np.array(A_ub_list), A_trust])
            b_ub = np.concatenate([np.array(b_ub_list), b_trust])

            bounds_lp = [(0, 1)] * n + [(None, None)]
            try:
                res_lp = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds_lp, method='highs')
                if res_lp.success:
                    t_lp = res_lp.x[-1]
                    if t_lp < c_max - 1e-10:
                        h_lp = res_lp.x[:n]
                        best_alpha = 0
                        best_val = compute_c5(h)
                        for alpha in [1.0, 0.5, 0.3, 0.7, 0.2]:
                            h_try = project_feasible(h + alpha * (h_lp - h), n)
                            c_try = compute_c5(h_try)
                            if c_try < best_val:
                                best_val = c_try
                                best_alpha = alpha
                        if best_alpha > 0:
                            h = project_feasible(h + best_alpha * (h_lp - h), n)
                            if compute_c5(h) < best_c5:
                                best_c5 = compute_c5(h)
                                best_h = h.copy()
            except Exception:
                pass

    # SLSQP after cutting plane
    for beta in [80000, 150000]:
        if remaining() < 8:
            break
        h_new = slsqp_run(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Try higher resolution
    for mult in [1.5, 2.0]:
        if remaining() < 60:
            break
        n_t = int(round(best_n * mult / 16) * 16)
        if n_t == best_n or n_t > 2048:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_t = project_feasible(h_interp, n_t)
        c5_t = compute_c5(h_t)

        for beta in [5000, 30000, 80000]:
            if remaining() < 15:
                break
            h_t = slsqp_run(h_t, n_t, beta, maxiter=500)
            c5_t = compute_c5(h_t)

        if remaining() > 15:
            h_t, c5_t = subgrad_true_max(h_t, n_t, min(30.0, remaining() * 0.3))

        for beta in [50000, 100000]:
            if remaining() < 10:
                break
            h_t = slsqp_run(h_t, n_t, beta, maxiter=400)
            c5_t = compute_c5(h_t)

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 8: Final intensive polish at best resolution
    n = best_n
    h = best_h.copy()

    for cycle in range(3):
        if remaining() < 20:
            break
        h = slsqp_run(h, n, 100000, maxiter=500)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

        if remaining() > 15:
            h_sg, c5_sg = subgrad_true_max(h, n, min(20.0, remaining() * 0.2))
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

        if remaining() > 10:
            h = slsqp_run(h, n, 200000, maxiter=400)
            c5_tmp = compute_c5(h)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

    # Phase 9: Micro-steps via random perturbation + SLSQP
    if remaining() > 30:
        for attempt in range(20):
            if remaining() < 10:
                break
            pert = rng.randn(n) * 0.005
            h_pert = project_feasible(h + pert, n)
            h_pert = slsqp_run(h_pert, n, 50000, maxiter=200)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
