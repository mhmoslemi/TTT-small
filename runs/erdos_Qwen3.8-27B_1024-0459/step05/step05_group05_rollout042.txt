import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    # Project to [0,1]^n with sum = n/2
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute the difference among free variables
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            # All at bounds, can't adjust
            break
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def softmax_obj(h, beta, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    clipped = np.clip(beta * shifted, -50, 50)
    exps = np.exp(clipped)
    return (m + np.log(np.sum(exps)) / beta) * dx

def softmax_grad(h, beta, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    clipped = np.clip(beta * shifted, -50, 50)
    exps = np.exp(clipped)
    w = exps / np.sum(exps)
    g = np.zeros(n)
    wc = w * dx
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return g

def projected_gradient(h, n, beta, max_iter, tol=1e-10):
    """Projected gradient descent with backtracking line search."""
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_f = softmax_obj(h, beta, n)
    step = 1.0
    
    for it in range(max_iter):
        if remaining() < 3:
            break
        
        f = softmax_obj(h, beta, n)
        if f < best_f:
            best_f = f
            best_h = h.copy()
        
        g = softmax_grad(h, beta, n)
        
        # Project gradient to feasible direction (tangent to sum constraint)
        g = g - np.mean(g)
        
        # Backtracking line search
        accepted = False
        for _ in range(30):
            h_new = project_feasible(h - step * g, n)
            f_new = softmax_obj(h_new, beta, n)
            if f_new < f - 1e-4 * step * np.sum(g * (h_new - h)):
                h = h_new
                f = f_new
                accepted = True
                break
            step *= 0.5
        
        if not accepted:
            break
        
        # Occasionally increase step
        if it % 50 == 0:
            step = min(step * 1.5, 1.0)
    
    return best_h, best_f
