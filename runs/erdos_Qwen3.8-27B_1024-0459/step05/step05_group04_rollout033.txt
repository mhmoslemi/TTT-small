                continue
            seen.add(key)
            dr = make_swap_delta(h, i, j, n)
            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr)
        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                for _ in range(4):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        dr = make_swap_delta(h, i, j, n)
                        h[i] = 0
                        h[j] = 1
                        corr = corr + dr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = float(np.max(corr))
                stall = 0
    return best_h, best_max * dx

def soft_min_obj_grad(h, n, beta):
    corr = compute_corr(h)
    m = float(np.max(corr))
    sh = beta * (corr - m)
    sh = np.clip(sh, -500.0, 500.0)
    ex = np.exp(sh)
    w = ex / float(np.sum(ex))
    obj = (m + np.log(float(np.sum(ex))) / beta) * (2.0 / n)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * (2.0 / n)
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return obj, g, corr, m

def projected_gradient_opt(h, n, time_limit=200):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 30.0
    alpha = 0.3
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 10:
        obj, g, corr, m = soft_min_obj_grad(h, n, beta)
        improved = False
        for trial_alpha in [alpha, alpha*0.5, alpha*0.25, alpha*2.0, alpha*0.1, alpha*5.0, 0.05, 0.15]:
            h_new = project_box_sum(h - trial_alpha * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                alpha = trial_alpha
                improved = True
                break
        if improved:
            no_improve = 0
            if beta < 100000 and no_improve < 60:
                beta *= 1.05
        else:
            no_improve += 1
            beta *= 2.0
            alpha *= 0.5
            if alpha < 1e-10:
                break
            if no_improve > 300:
                break
    return best_h, best_c5

def top_k_gradient_step(h, n, top_k=6, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]
        bot_idx = np.argsort(corr)[:3]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015, 0.04, 0.06]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def greedy_construct(n, rng):
    h = np.zeros(n)
    n_half = n // 2
    used = set()
    positions = list(range(n))
    rng.shuffle(positions)
    for _ in range(n_half):
        best_pos = -1
        best_max = float('inf')
        for p in positions:
            if p in used:
                continue
            h[p] = 1
            corr = compute_corr(h)
            mx = float(np.max(corr))
            h[p] = 0
            if mx < best_max:
                best_max = mx
                best_pos = p
        if best_pos >= 0:
            h[best_pos] = 1
            used.add(best_pos)
        else:
            for p in positions:
                if p not in used:
                    h[p] = 1
                    used.add(p)
                    break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    # Phase 1: Binary local search at parent resolution
    t1 = min(250, int(remaining_time() * 0.28))
    if t1 > 20:
        h_b1, c5_b1 = binary_local_search(h_bin, n, time_limit=t1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n
    # Phase 2: Continuous projected gradient
    if remaining_time() > 60:
        h_cg, c5_cg = projected_gradient_opt(best_h, n, time_limit=min(180, int(remaining_time() * 0.22)))
        if c5_cg < best_c5:
            best_c5 = c5_cg
            best_h = h_cg.copy()
            best_n = n
    # Phase 3: Top-k equalization
    if remaining_time() > 30:
        h_eq, c5_eq = top_k_gradient_step(best_h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    # Phase 4: Binary snap from continuous + local search
    if remaining_time() > 60:
        h_bin4 = (best_h > 0.5).astype(float)
        n4_half = n // 2
        n4_ones = int(np.sum(h_bin4))
        if n4_ones != n4_half:
            if n4_ones > n4_half:
                idx = np.where(h_bin4 == 1)[0]
                h_bin4[idx[:n4_ones - n4_half]] = 0
            else:
                idx = np.where(h_bin4 == 0)[0]
                h_bin4[idx[:n4_half - n4_ones]] = 1
        t4 = min(120, int(remaining_time() * 0.15))
        if t4 > 20:
            h_b4, c5_b4 = binary_local_search(h_bin4, n, time_limit=t4, rng=rng)
            if c5_b4 < best_c5:
                best_c5 = c5_b4
                best_h = h_b4.copy()
                best_n = n
    # Phase 5: Try other resolutions
    resolutions = sorted(set([
        n, n + 32, n + 64, n + 96, n - 32, n - 64,
        int(n * 1.25), int(n * 1.5), int(n * 0.75),
    ]))
    resolutions = [r for r in resolutions if 128 <= r <= 1024 and r % 2 == 0 and r != n]
    resolutions.sort(key=lambda x: abs(x - n))
    for n_t in resolutions[:4]:
        if remaining_time() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        t_bi = min(60, int(remaining_time() * 0.12))
        if t_bi > 10:
            h_b3, c5_b3 = binary_local_search(h_bin_t, n_t, time_limit=t_bi, rng=rng)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t
        if remaining_time() > 40:
            h_cont = project_feasible(h_interp, n_t)
            h_cg2, c5_cg2 = projected_gradient_opt(h_cont, n_t, time_limit=min(80, int(remaining_time() * 0.1)))
            if c5_cg2 < best_c5:
                best_c5 = c5_cg2
                best_h = h_cg2.copy()
                best_n = n_t
    # Phase 6: More starting points at best resolution
    n = best_n
    start_points = [
        ('tm', thue_morse(n)),
    ]
    for name, sp in start_points:
        if remaining_time() < 40:
            break
        sp_c5 = compute_c5(sp)
        if sp_c5 < best_c5 + 0.02:
            t_sp = min(60, int(remaining_time() * 0.1))
            if t_sp > 10:
                h_bs, c5_bs = binary_local_search(sp, n, time_limit=t_sp, rng=rng)
                if c5_bs < best_c5:
                    best_c5 = c5_bs
                    best_h = h_bs.copy()
                    best_n = n
    # Phase 7: Final continuous refinement at best resolution
    n = best_n
    h = best_h.copy()
    for beta_start in [50, 200]:
        if remaining_time() < 20:
            break
        h_new = h.copy()
        start_time = time.time()
        beta = beta_start
        alpha = 0.2
        while time.time() - start_time < min(60, int(remaining_time() * 0.15)) and remaining_time() > 10:
            obj, g, corr, m = soft_min_obj_grad(h_new, n, beta)
            improved = False
            for trial_alpha in [alpha, alpha*0.5, alpha*0.25, alpha*2.0, 0.05, 0.1]:
                h_try = project_box_sum(h_new - trial_alpha * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h_new = h_try
                    best_c5 = c5_try
                    best_h = h_new.copy()
                    alpha = trial_alpha
                    improved = True
                    break
            if improved:
                beta *= 1.08
            else:
                beta *= 2.0
                alpha *= 0.4
                if alpha < 1e-9:
                    break
    # Phase 8: Top-k final
    if remaining_time() > 15:
        h_eq2, c5_eq2 = top_k_gradient_step(best_h, best_n, top_k=8, max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
    # Final: verify and return
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    return (best_h, best_c5, best_n)
