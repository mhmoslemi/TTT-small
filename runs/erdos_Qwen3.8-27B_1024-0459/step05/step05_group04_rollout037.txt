            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            delta = new_max - cur_max

            # SA acceptance
            if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10) / dx):
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j, dr, new_max)

        if best_move is not None:
            i, j, dr, new_max = best_move
            h[i] = 0
            h[j] = 1
            tabu_tab[i] = tabu_counter
            tabu_tab[j] = tabu_counter
            tabu_counter += 1
            corr = corr + dr
            cur_max = new_max

            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > max_no_improve:
                # Large perturbation to escape
                for _ in range(rng.randint(5, 15)):
                    oi = ones[rng.randint(len(ones))]
                    oz = zeros[rng.randint(len(zeros))]
                    if h[oi] == 1 and h[oz] == 0:
                        dr = make_swap_delta(h, oi, oz, n)
                        h[oi] = 0
                        h[oz] = 1
                        corr = corr + dr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = float(np.max(corr))
                no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Reverse last few moves or random perturbation
                for _ in range(rng.randint(3, 8)):
                    oi = rng.randint(n)
                    oz = rng.randint(n)
                    if h[oi] == 1 and h[oz] == 0:
                        dr = make_swap_delta(h, oi, oz, n)
                        h[oi] = 0
                        h[oz] = 1
                        corr = corr + dr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = float(np.max(corr))
                no_improve = 0

    return best_h, best_max * dx

def binary_greedy_refine(h_bin, n, time_limit, rng):
    """Pure greedy: always take the best improving move."""
    h = (h_bin > 0.5).copy().astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - n_half]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:n_half - n_ones]] = 1

    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    start = time.time()
    stall = 0
    max_stall = 800

    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Focus on top shifts
        top_k = np.argsort(-corr)[:12]
        candidates = []
        for s in top_k[:10]:
            off = s - (n - 1)
            if off >= 0:
                np_ = n - off
                contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for ci in contrib[:20]:
                    candidates.append((ci + off, ci))
            else:
                ao = -off
                np_ = n - ao
                contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for ci in contrib[:20]:
                    candidates.append((ci, ci + ao))
        # Random
        for _ in range(40):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))

        best_move = None
        best_new_max = cur_max + 1e-15
        seen = set()

        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            dr = make_swap_delta(h, i, j, n)
            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr)

        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                # Diversify
                for _ in range(5):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        dr = make_swap_delta(h, i, j, n)
                        h[i] = 0
                        h[j] = 1
                        corr = corr + dr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = float(np.max(corr))
                stall = 0

    return best_h, best_max * dx

def slsqp_refine(h, n, beta, maxiter=400):
    """SLSQP with soft-max objective."""
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        return (m + np.log(s) / beta) * (2.0 / n)

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def equalize_shifts(h, n, top_k=6, max_iters=400):
    """Directly equalize the top-k correlation shifts."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_top = corr[idx[0]]
        c_bot_idx = np.argsort(corr)[:3]
        c_bot = corr[c_bot_idx[0]]

        if c_top - c_bot < 1e-12:
            break

        # Gradient: push top down, push bottom up
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]

        g_bot = np.zeros(n)
        for k in c_bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]

        d = g_bot - g
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn

        improved = False
        for step in [0.02, 0.05, 0.01, 0.03, 0.08, 0.005, 0.04, 0.1, 0.015, 0.07]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break

    return h, best_c5

def freq_perturb(h, n, rng, strength=0.1):
    """Apply random phase perturbation in frequency domain."""
    H = np.fft.rfft(h)
    freqs = np.fft.rfftfreq(n)
    # Don't perturb DC (preserves sum constraint approximately)
    phase = rng.uniform(-np.pi, np.pi, len(H))
    scale = strength * np.exp(-freqs * 2.0)  # More perturbation at low freq
    phase[0] = 0.0
    scale[0] = 0.0
    H_new = H * (1.0 + scale * np.exp(1j * phase))
    h_new = np.fft.irfft(H_new, n)
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    # Binary version of parent
    h_bin = np.zeros(n)
    sort_idx = np.argsort(-best_h)
    h_bin[sort_idx[:n // 2]] = 1.0

    # Phase 1: Binary SA (main exploration)
    t1 = min(250, int(remaining() * 0.25))
    if t1 > 20:
        h_b1, c5_b1 = binary_sa(h_bin, n, time_limit=t1, rng=rng, tabu_len=80, n_top=10)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n

    # Phase 2: Binary greedy refine (exploitation)
    t2 = min(200, int(remaining() * 0.2))
    if t2 > 15:
        h_b2, c5_b2 = binary_greedy_refine(best_h, n, time_limit=t2, rng=rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = n

    # Phase 3: Multi-resolution binary search
    resolutions = []
    for dn in [-32, -16, -8, 8, 16, 32, 48, 64]:
        rt = n + dn
        if 128 <= rt <= 800 and rt % 2 == 0 and rt != n:
            resolutions.append(rt)
    # Also try some proportional
    for frac in [0.8, 1.2, 1.5]:
        rt = int(n * frac)
        if rt % 2 != 0:
            rt += 1
        if 128 <= rt <= 800 and rt != n and rt not in resolutions:
            resolutions.append(rt)
    resolutions.sort(key=lambda x: abs(x - n))

    for n_t in resolutions[:5]:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = np.zeros(n_t)
        idx_t = np.argsort(-h_interp)
        h_bin_t[idx_t[:n_t // 2]] = 1.0

        t_bi = min(80, int(remaining() * 0.07))
        if t_bi > 10:
            h_b3, c5_b3 = binary_sa(h_bin_t, n_t, time_limit=t_bi, rng=rng, tabu_len=60, n_top=8)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t

    # Phase 4: Alternative starting points at best resolution
    n = best_n
    starts = []
    # Thue-Morse
    starts.append(('tm', thue_morse(n)))
    # Frequency-perturbed parent
    mp = interp_to(best_h, best_n, n)
    starts.append(('freq', freq_perturb(mp, n, rng, 0.15)))
    # Random (balanced)
    hr = (rng.rand(n) > 0.5).astype(float)
    nr = int(np.sum(hr))
    if nr != n // 2:
        if nr > n // 2:
            idx = np.where(hr == 1)[0]
            hr[idx[:nr - n // 2]] = 0
        else:
            idx = np.where(hr == 0)[0]
            hr[idx[:n // 2 - nr]] = 1
    starts.append(('rand', hr))

    for name, sp in starts:
        if remaining() < 25:
            break
        sp_c5 = compute_c5(sp)
        if sp_c5 < best_c5 + 0.01:  # Only pursue if reasonably close
            t_s = min(80, int(remaining() * 0.06))
            if t_s > 10:
                h_s, c5_s = binary_sa(sp, n, time_limit=t_s, rng=rng, tabu_len=60, n_top=8)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                    best_n = n

    # Phase 5: Continuous refinement
    n = best_n
    h = interp_to(best_h, best_n, n)
    h = project_feasible(h, n)

    for beta in [50, 200, 500, 2000, 5000, 10000, 20000, 50000]:
        if remaining() < 8:
            break
        h_new = slsqp_refine(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Equalization
    if remaining() > 15:
        h_eq, c5_eq = equalize_shifts(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 7: More continuous with higher beta
    for beta in [10000, 30000, 60000, 100000]:
        if remaining() < 6:
            break
        h_new = slsqp_refine(h, n, beta, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Finer resolution
    if remaining() > 80:
        n_fine = n + 32
        if n_fine % 2 != 0:
            n_fine += 1
        h_fine = interp_to(best_h, n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        for beta in [500, 2000, 10000, 30000]:
            if remaining() < 10:
                break
            h_fine = slsqp_refine(h_fine, n_fine, beta, maxiter=250)
        c5_fine = compute_c5(h_fine)
        if remaining() > 15:
            h_fine, c5_fine2 = equalize_shifts(h_fine, n_fine, top_k=8, max_iters=200)
            c5_fine = min(c5_fine, c5_fine2)
            if c5_fine2 < c5_fine:
                h_fine = h_fine  # already updated
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine

    # Phase 9: Final binary polish at best resolution
    n = best_n
    h_bin_final = np.zeros(n)
    idx_f = np.argsort(-best_h)
    h_bin_final[idx_f[:n // 2]] = 1.0
    t_final = min(60, int(remaining() * 0.15))
    if t_final > 10:
        h_p, c5_p = binary_greedy_refine(h_bin_final, n, time_limit=t_final, rng=rng)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = n

    # Phase 10: Final continuous polish
    n = best_n
    h = interp_to(best_h, best_n, n)
    h = project_feasible(h, n)
    for beta in [20000, 50000, 100000, 200000]:
        if remaining() < 5:
            break
        h_new = slsqp_refine(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 10:
        h_eq, c5_eq = equalize_shifts(h, n, top_k=8, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
