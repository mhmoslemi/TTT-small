def grad_corr_k(h, k, n):
    """Gradient of the k-th correlation term with respect to h."""
    g = np.zeros(n)

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g


def sm_max_obj(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -60.0, 60.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-30:
        return m * (2.0 / n)
    return (m + np.log(s) / beta) * (2.0 / n)


def sm_max_grad(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -60.0, 60.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    if s < 1e-30:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w = ex / s
    g = np.zeros(n)
    wc = w * (2.0 / n)
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:

            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]

            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

def cvxpy_cutting_plane(h, n, K=30, delta=0.04):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    objective = cp.Minimize(t_var)
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0,
        h_var <= 1,
        h_var <= h + delta,
        h_var >= h - delta
    ]
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        rhs = -fk + np.dot(g, h)
        constraints.append(cp.dot(g, h_var) - t_var <= rhs)
    prob = cp.Problem(objective, constraints)
    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if h_var.value is not None and t_var.value is not None:
                return np.asarray(h_var.value).flatten(), float(t_var.value)
        except Exception:
            continue
    return h.copy(), float(np.max(corr))

        if _BUDGET - (time.time() - _T0) > 30:
            for K_val in [10, 14]:
                if _BUDGET - (time.time() - _T0) < 10:
                    break
                h2_eq, c5_2_eq = remedz_exchange(h2, n2, K=K_val, max_iters=250)
                if c5_2_eq < best_c5:
                    best_c5 = c5_2_eq
                    best_h = h2_eq.copy()
                    best_n = n2
                    h2 = h2_eq.copy()

        for beta in [20000, 40000]:
            if _BUDGET - (time.time() - _T0) < 10:
                break
            h2 = smooth_max_slsqp(h2, n2, beta, maxiter=600)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

    # Phase 6: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    for cycle in range(3):
        if _BUDGET - (time.time() - _T0) < 40:
            break

        # Remez with adaptive K
        if _BUDGET - (time.time() - _T0) > 20:
            for K_val in [12, 8, 16]:
                if _BUDGET - (time.time() - _T0) < 10:
                    break
                h_r, c5_r = remedz_exchange(h, n, K=K_val, max_iters=250)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    h = h_r.copy()

        # SLSQP
        if _BUDGET - (time.time() - _T0) > 15:
            for beta in [30000, 60000]:
                if _BUDGET - (time.time() - _T0) < 8:
                    break
                h = smooth_max_slsqp(h, n, beta, maxiter=600)
            c5_tmp = compute_c5(h)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()

        # Cutting plane
        if _BUDGET - (time.time() - _T0) > 25:
            for it in range(80):
                if _BUDGET - (time.time() - _T0) < 10:
                    break
                h_lp, t_lp = cvxpy_cutting_plane(h, n, K=40, delta=0.03)
                c_cur = compute_c5(h)
                if c_cur < best_c5:
                    best_c5 = c_cur
                    best_h = h.copy()
                if t_lp < c_cur - 1e-10:
                    h_try = project_feasible(h + 0.5 * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                elif it > 40:
                    break

    # Phase 7: Perturbation restart to escape local optima
    if _BUDGET - (time.time() - _T0) > 60:
        for trial in range(3):
            if _BUDGET - (time.time() - _T0) < 30:
                break
            scale = 0.005 * (trial + 1)
            h_pert = h + scale * (2.0 * rng.rand(n) - 1.0)
            h_pert = project_feasible(h_pert, n)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5 + 0.01:
                h_pert = smooth_max_slsqp(h_pert, n, 10000, maxiter=400)
                c5_pert = compute_c5(h_pert)
                h_pert = smooth_max_slsqp(h_pert, n, 30000, maxiter=400)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_pert.copy()
                    h = h_pert.copy()

    # Phase 8: Coordinate descent polish
    if _BUDGET - (time.time() - _T0) > 20:
        h_cd, c5_cd = coordinate_polish(best_h, best_n, max_passes=3, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 9: Final SLSQP polish
    for beta in [50000, 100000]:
        if _BUDGET - (time.time() - _T0) < 8:
            break
        h_final = smooth_max_slsqp(best_h, best_n, beta, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final Remez
    if _BUDGET - (time.time() - _T0) > 15:
        h_f, c5_f = remedz_exchange(best_h, best_n, K=10, max_iters=300)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
