def fista_smooth(h0, n, beta, max_iters=2000):
    """FISTA for smooth max with projection."""
    h = project_feasible(h0, n)
    h_prev = h.copy()
    t_val = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        y = h + alpha * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Gradient step
        _, grad = smooth_obj_and_grad(y, beta)
        # Step size for FISTA: 1/L where L is Lipschitz constant
        # L = beta * (2 * max_k |d(corr_k)/dh||_2^2) * dx ... complex
        # Use adaptive step
        lr = 1.0 / (beta * 0.5 * (it + 1))
        h_new = project_feasible(y - lr * grad, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
        t_val = t_new
    
    return best_h, best_c5

def binary_sa(h_bin, n, seed=42, max_iters=50000, T0=1.0, T1=0.01):
    """Simulated annealing for binary solution with 2-exchange moves."""
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr_now = compute_corr(h)
    c_max = np.max(corr_now)
    
    T = T0
    cooling = (T1 / T0) ** (1.0 / max_iters)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        T *= cooling
        
        # Choose move: 1-exchange or 2-exchange
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if rng.random() < 0.7:
            # 1-exchange: swap one 1 and one 0
            i = rng.choice(list(ones))
            j = rng.choice(list(zeros))
            h[i] = 0; h[j] = 1
            ones.discard(i); ones.add(j)
            zeros.discard(j); zeros.add(i)
            
            corr_new = compute_corr(h)
            c_new = np.max(corr_new)
            delta = c_new - c_max
            
            if delta < 0 or rng.random() < np.exp(-abs(delta) * n / 2.0 / max(T, 1e-10)):
                if c_new < best_c5 / (2.0/n):  # Compare in raw units
                    best_c5 = float(c_new * 2.0 / n)
                    best_h = h.copy()
                c_max = c_new
            else:
                # Revert
                h[i] = 1; h[j] = 0
                ones.discard(j); ones.add(i)
                zeros.discard(i); zeros.add(j)
        else:
            # 2-exchange: swap two 1s and two 0s
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(list(ones), size=2, replace=False)
            j1, j2 = rng.choice(list(zeros), size=2, replace=False)
            h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            ones.discard(i1); ones.discard(i2); ones.add(j1); ones.add(j2)
            zeros.discard(j1); zeros.discard(j2); zeros.add(i1); zeros.add(i2)
            
            corr_new = compute_corr(h)
            c_new = np.max(corr_new)
            delta = c_new - c_max

                zeros.discard(i1); zeros.discard(i2); zeros.add(j1); zeros.add(j2)
    
    return best_h, best_c5

def sturmian_init(n, alpha=None):
    """Sturmian (mechanical) word initialization."""
    if alpha is None:
        alpha = (np.sqrt(5) - 1) / 2.0  # golden ratio conjugate
    h = np.zeros(n)
    h[np.where(np.array([ (i * alpha) % 1.0 for i in range(n) ]) < 0.5)[0]] = 1.0
    return h

def thue_morse_init(n):
    """Thue-Morse sequence."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    # Ensure exactly n/2 ones
    if n % 2 == 0:
        ones_count = np.sum(h)
        if ones_count != n // 2:
            # Adjust
            if ones_count > n // 2:
                excess = ones_count - n // 2
                one_idx = np.where(h == 1)[0]
                for idx in one_idx[-int(excess):]:
                    h[idx] = 0
            else:
                deficit = n // 2 - ones_count
                zero_idx = np.where(h == 0)[0]
                for idx in zero_idx[-int(deficit):]:
                    h[idx] = 1
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    h_bin = np.zeros(n)
