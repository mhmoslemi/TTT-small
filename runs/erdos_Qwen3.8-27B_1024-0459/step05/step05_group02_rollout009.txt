```python
def compute_swap_delta(h, i, j, n):
    offsets = np.arange(-(n-1), n)  # -[n-1], ..., n-1
    # h[i]: 1 -> 0
    ip = i + offsets  # i + offset
    im = i - offsets  # i - offset
    vp = (ip >= 0) & (ip < n)
    vm = (im >= 0) & (im < n)
    di = np.zeros(2*n-1)
    di[vp] = h[ip[vp]]
    di[vm] -= (1.0 - h[im[vm]])
    
    # h[j]: 0 -> 1
    jp = j + offsets
    jm = j - offsets
    vpj = (jp >= 0) & (jp < n)
    vmj = (jm >= 0) & (jm < n)
    dj = np.zeros(2*n-1)
    dj[vmj] = (1.0 - h[jm[vmj]])
    dj[vpj] -= h[jp[vpj]]
    
    return di + dj
```

```python
        vmj = (jm >= 0) & (jm < n)
        delta[vmj] += (1.0 - h[jm[vmj]])
        delta[vpj] -= h[jp[vpj]]
        
        new_c5_raw = np.max(corr + delta)
        new_c5 = new_c5_raw * dx
        
        delta_c = new_c5 - cur_c5
        accept = (delta_c < 0) or (rng.random() < np.exp(-delta_c / max(t, 1e-12)))
        
        if accept:
            h[i] = 0
            h[j] = 1
            corr += delta
            cur_c5 = new_c5
            
            ones_list[ii] = j
            zeros_list[jj] = i
            
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Kick: if stuck, do random perturbations
        if no_improve > kick_interval and it % 1000 == 0:
            for _ in range(10):
                ii2 = rng.randint(n_ones)
                jj2 = rng.randint(n_ones)
                i2 = ones_list[ii2]
                j2 = zeros_list[jj2]
                h[i2] = 0
                h[j2] = 1
                ones_list[ii2] = j2
                zeros_list[jj2] = i2
            corr = compute_corr(h)
            cur_c5 = float(np.max(corr)) * dx
            no_improve = 0
    
    return best_h, best_c5

def smooth_max_obj_and_grad(h, n, beta, top_k=10):
    """Compute smooth max objective and gradient."""
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:top_k]
    c = corr[idx]
    m = c[0]
    sh = beta * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / beta) * dx
    
    # Gradient
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        offset = ki - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:offset+np_] += w[j] * (1.0 - h[:np_])
            g[:np_] -= w[j] * h[offset:offset+np_]
        else:
            ao = -offset
            np_ = n - ao
            g[:np_] += w[j] * (1.0 - h[ao:ao+np_])
            g[ao:ao+np_] -= w[j] * h[:np_]
    g *= dx
    
    return val, g

def lbfgs_smooth(h_init, n, beta, top_k=10, maxiter=500):
    """L-BFGS-B optimization with smooth max."""
    h0 = proj(h_init.copy(), n)
    target = n / 2.0
    
    def func(x):
        v, g = smooth_max_obj_and_grad(x, n, beta, top_k)
        return v + 50.0 * (np.sum(x) - target) ** 2
    
    def grad(x):
        v, g = smooth_max_obj_and_grad(x, n, beta, top_k)
        g += 100.0 * (np.sum(x) - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(func, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 15})
        return proj(res.x, n)
    except Exception:
        return h0

def greedy_descent(h_init, n, max_iters=50000):
    """Pure greedy descent on binary: always accept improvements."""
    rng = np.random.RandomState(123)
    h = h_init.copy()
    corr = compute_corr(h).astype(float)
    dx = 2.0 / n
    cur_c5 = float(np.max(corr)) * dx
    best_c5 = cur_c5
    best_h = h.copy()
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    n_ones = len(ones_list)
    
    offsets = np.arange(-(n-1), n, dtype=np.int32)
    nc = 2 * n - 1
    stuck = 0
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        
        ii = rng.randint(n_ones)
        jj = rng.randint(n_ones)
        i = ones_list[ii]
        j = zeros_list[jj]
        
        ip = i + offsets
        im = i - offsets
        delta = np.zeros(nc)
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        delta[vp] = h[ip[vp]]
        delta[vm] -= (1.0 - h[im[vm]])
        
        jp = j + offsets
        jm = j - offsets
        vpj = (jp >= 0) & (jp < n)
        vmj = (jm >= 0) & (jm < n)
        delta[vmj] += (1.0 - h[jm[vmj]])
        delta[vpj] -= h[jp[vpj]]
        
        new_c5 = float(np.max(corr + delta)) * dx
        if new_c5 < cur_c5 - 1e-12:
            h[i] = 0
            h[j] = 1
            corr += delta
            cur_c5 = new_c5
            ones_list[ii] = j
            zeros_list[jj] = i
            stuck = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            stuck += 1
        
        if stuck > 10000:
            break
    
    return best_h, best_c5

def targeted_repair(h_init, n, max_iters=5000):
    """Target moves that reduce the worst offset."""
    rng = np.random.RandomState(77)
    h = h_init.copy()
    corr = compute_corr(h).astype(float)
    dx = 2.0 / n
    cur_c5 = float(np.max(corr)) * dx
    best_c5 = cur_c5
    best_h = h.copy()
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    n_ones = len(ones_list)
    
    offsets_arr = np.arange(-(n-1), n, dtype=np.int32)
    nc = 2 * n - 1
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        
        # Find worst offset
        worst_k = int(np.argmax(corr))
        worst_offset = worst_k - (n - 1)
        
        # Find contributing positions at worst offset
        # corr[worst_k] = sum h[m+offset]*(1-h[m]) for valid m+offset, m
        # Want to find m where h[m+worst_offset]=1 and h[m]=0
        if worst_offset >= 0:
            m_range = range(n - worst_offset)
        else:
            m_range = range(n + worst_offset)
        
        contributing = []
        for m in m_range:
            mp = m + worst_offset
            if h[mp] == 1 and h[m] == 0:
                contributing.append(m)
        
        if not contributing:
            # No contributing pairs, try random swap
            ii = rng.randint(n_ones)
            jj = rng.randint(n_ones)
            i = ones_list[ii]
            j = zeros_list[jj]
        else:
            # Pick a contributing m, try to fix it
            m = contributing[rng.randint(len(contributing))]
            # Option A: flip h[m+worst_offset] from 1 to 0
            #   Need to flip some h[q] from 0 to 1
            #   Choose q where h[q+worst_offset]=1 (not contributing) to minimize damage
            p = m + worst_offset
            # Find good q: h[q]=0, h[q+worst_offset]=1 (so flipping q to 1 doesn't add new contribution)
            candidates_q = []
            for q in range(n):
                if h[q] == 0 and q != m:
                    qp = q + worst_offset
                    if 0 <= qp < n and h[qp] == 1:
                        candidates_q.append(q)
            
            if candidates_q:
                q = candidates_q[rng.randint(len(candidates_q))]
                i, j = p, q  # flip h[p]: 1->0, h[q]: 0->1
            else:
                # Just do random swap
                ii = rng.randint(n_ones)
                jj = rng.randint(n_ones)
                i = ones_list[ii]
                j = zeros_list[jj]
        
        # Evaluate swap
        ip = i + offsets_arr
        im = i - offsets_arr
        delta = np.zeros(nc)
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        delta[vp] = h[ip[vp]]
        delta[vm] -= (1.0 - h[im[vm]])
        
        jp = j + offsets_arr
        jm = j - offsets_arr
        vpj = (jp >= 0) & (jp < n)
        vmj = (jm >= 0) & (jm < n)
        delta[vmj] += (1.0 - h[jm[vmj]])
        delta[vpj] -= h[jp[vpj]]
        
        new_c5 = float(np.max(corr + delta)) * dx
        if new_c5 < cur_c5 - 1e-12:
            h[i] = 0
            h[j] = 1
            corr += delta
            cur_c5 = new_c5
            
            # Update lists
            for idx_l in range(len(ones_list)):
                if ones_list[idx_l] == i:
                    ones_list[idx_l] = j
                    break
            for idx_l in range(len(zeros_list)):
                if zeros_list[idx_l] == j:
                    zeros_list[idx_l] = i
                    break
            
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return best_h, best_c5
```
