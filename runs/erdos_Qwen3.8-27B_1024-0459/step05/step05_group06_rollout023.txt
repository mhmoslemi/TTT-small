def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def subgrad_projected(h, n, n_active=8):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 0.3)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        order = np.argsort(corr[active])[::-1][:n_active]
        active = active[order]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

    if lo_j == hi_j:
        lo_j, hi_j = 0.0, 1.0

    grid_i = np.linspace(lo_i, hi_i, 12)
    grid_j = np.linspace(lo_j, hi_j, 12)

    for ti in grid_i:
        for tj in grid_j:
            h_trial = h.copy()
            h_trial[i] = ti
            h_trial[j] = tj
            c5_t = compute_c5(h_trial)
            if c5_t < best_f * (2.0/n):
                best_f = c5_t * n / 2.0
                best_ti, best_tj = ti, tj
            elif c5_t * n / 2.0 < best_f:
                best_f = c5_t * n / 2.0
                best_ti, best_tj = ti, tj

    # Refine with golden section on each coordinate
    for coord in [0, 1]:
        for _ in range(30):
            if coord == 0:
                lo, hi = max(0.0, best_ti - 0.1), min(1.0, best_ti + 0.1)
            else:
                lo, hi = max(0.0, best_tj - 0.1), min(1.0, best_tj + 0.1)
            if hi - lo < 1e-10:
                break
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            h1 = h.copy()
            h2 = h.copy()
            if coord == 0:
                h1[i], h1[j] = c1, best_tj
                h2[i], h2[j] = c2, best_tj
            else:
                h1[i], h1[j] = best_ti, c1
                h2[i], h2[j] = best_ti, c2
            f1 = np.max(compute_corr(h1))
            f2 = np.max(compute_corr(h2))
            if f1 < f2:
                hi = c2
            else:
                lo = c1
        mid = (lo + hi) / 2.0
        if coord == 0:
            h_t = h.copy()
            h_t[i] = mid
            h_t[j] = best_tj
            if np.max(compute_corr(h_t)) < np.max(compute_corr(h)):
                best_ti = mid
        else:
            h_t = h.copy()
            h_t[i] = best_ti
            h_t[j] = mid
            if np.max(compute_corr(h_t)) < np.max(compute_corr(h)):
                best_tj = mid

    if abs(best_ti - h[i]) > 1e-10 or abs(best_tj - h[j]) > 1e-10:
        h[i] = best_ti
        h[j] = best_tj
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP warm-up
    for beta in [2000, 8000, 30000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_optimize(h, n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: Intensive coordinate descent with incremental updates
    if remaining_time() > 60:
        h_new, c5_new = coord_descent_v2(h, n, n_sweeps=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: SLSQP with higher beta
    for beta in [80000, 200000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_optimize(h, n, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: More coordinate descent
    if remaining_time() > 50:
        h_new, c5_new = coord_descent_v2(h, n, n_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Block coordinate optimization (pairs of adjacent coordinates)
    if remaining_time() > 60:
        h = best_h.copy()
        for _ in range(3):
            if remaining_time() < 15:
                break
            for i in range(0, n - 1, 2):
                if remaining_time() < 5:
                    break
                h = block_optimize(h, n, i, i + 1)
            # Also try some random pairs
            for _ in range(20):
                if remaining_time() < 3:
                    break
                p1, p2 = rng.choice(n, 2, replace=False)
                h = block_optimize(h, n, p1, p2)
        c5_blk = compute_c5(project_feasible(h, n))
        h = project_feasible(h, n)
        if c5_blk < best_c5:
            best_c5 = c5_blk
            best_h = h.copy()

    # Phase 6: Subgradient + SLSQP interleaving
    n = best_n
    h = best_h.copy()
    for it in range(400):
        if remaining_time() < 10:
            break
        grad = subgrad_projected(h, n, 8)
        if grad is None:
            break
        base_step = 0.06 / (1.0 + it * 0.008)
        improved = False
        for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 0.15, 0.4]:
            step = base_step * sm
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved and it % 80 == 40 and remaining_time() > 40:
            h_new = slsqp_optimize(h, n, 50000, 250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            if it > 300:
                break

    # Phase 7: Coordinate descent after subgradient
    if remaining_time() > 30:
        h_new, c5_new = coord_descent_v2(best_h, best_n, n_sweeps=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Try higher resolutions
    higher_res = sorted(set())
    for mult in [1.2, 1.33, 1.5, 1.75]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if 320 <= rn <= 1600:
            higher_res.append(rn)
    for extra in [64, 128, 192]:
        rn = best_n + extra
        if rn % 2 != 0:
            rn += 1
        if 320 <= rn <= 1600:
            higher_res.append(rn)
    # Deduplicate
    seen = set()
    unique_res = []
    for rn in higher_res:
        if rn not in seen:
            seen.add(rn)
            unique_res.append(rn)
    higher_res = unique_res

    for rn in higher_res:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        for beta in [5000, 30000]:
            if remaining_time() < 25:
                break
            h_r = slsqp_optimize(h_r, rn, beta, 250)

        if remaining_time() > 30:
            h_r, c5_r = coord_descent_v2(h_r, rn, n_sweeps=100)

        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 9: Random perturbations with structured restarts
    n = best_n
    h = best_h.copy()
    n_perturb = max(8, n // 40)
    for pert in range(40):
        if remaining_time() < 20:
            break
        h_pert = h.copy()
        if pert % 3 == 0:
            # Random small perturbations
            idx = rng.choice(n, n_pert, replace=False)
            h_pert[idx] += rng.uniform(-0.12, 0.12, n_pert)
        elif pert % 3 == 1:
            # Block shift: shift a block of values
            bsz = max(4, n // 50)
            start = rng.randint(0, n - bsz)
            h_pert[start:start+bsz] += rng.uniform(-0.08, 0.08)
        else:
            # Mirror perturbation
            idx = rng.randint(0, n // 2)
            h_pert[idx] += rng.uniform(-0.15, 0.15)
            h_pert[n - 1 - idx] += rng.uniform(-0.15, 0.15)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.003:
            h_new = slsqp_optimize(h_pert, n, 20000, 200)
            c5_new = compute_c5(h_new)
            if remaining_time() > 20:
                h_new, c5_new = coord_descent_v2(h_new, n, n_sweeps=80)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 10: Final intensive coordinate descent
    if remaining_time() > 15:
        h_new, c5_new = coord_descent_v2(best_h, best_n, n_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: Final SLSQP polish
    if remaining_time() > 10:
        h_new = slsqp_optimize(best_h, best_n, 500000, 150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 12: One last coordinate descent
    if remaining_time() > 8:
        h_new, c5_new = coord_descent_v2(best_h, best_n, n_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation and constraint enforcement
    best_h = project_feasible(best_h, best_n)
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
