def binary_sa_binary(h_bin, n, max_iters, T_initial, T_final, rng):
    """Simulated annealing on binary pattern with exactly n/2 ones."""
    h = h_bin.astype(float).copy()
    best_h = h.copy()
    corr = np.correlate(h, 1-h, mode='full')
    best_c5 = np.max(corr)
    c5 = best_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        T = T_initial * (T_final / T_initial) ** (it / max_iters)
        
        # Propose swap
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a, b = ones[ia], zeros[ib]
        
        # Perform swap
        h[a] = 0; h[b] = 1
        corr_new = np.correlate(h, 1-h, mode='full')
        c5_new = np.max(corr_new)
        
        delta = c5_new - best_c5
        if delta < 0 or rng.rand() < np.exp(-delta / (T + 1e-10)):
            # Accept
            ones[ia], zeros[ib] = b, a
            c5 = c5_new
            corr = corr_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # Reject
            h[a] = 1; h[b] = 0
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def beatty_init(n, alpha, seed_val):
    h = np.zeros(n)

    frac = idx - np.floor(idx)
    thresh = 1.0 - 2.0 * frac
    order = np.argsort(thresh)
    h[order[:n // 2]] = 1.0
    return h


def sturmian_init(n, seed_val):

def smooth_corr_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    return m + np.log(np.sum(ex)) / beta


def smooth_corr_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            np_ = n - lag
            g[lag:lag + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[lag:lag + np_]
        else:
            a = -lag
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def projected_gradient_smooth(h, n, beta, max_iters, lr_init, lr_decay):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_val = smooth_corr_obj(h, beta)
    y = h.copy()
    t_val = 1.0

    for it in range(max_iters):
        if remaining() < 5.0:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha_nesterov = (t_val - 1.0) / t_new
        y_new = h + alpha_nesterov * (h - y)
        g = smooth_corr_grad(y_new, beta)
        lr = lr_init / (1.0 + lr_decay * it)
        h_new = project_feasible(y_new - lr * g, n)
        val_new = smooth_corr_obj(h_new, beta)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new

    return best_h


def subgrad_equalize(h, n, max_iters, top_k=5, step_init=0.03):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5_val = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.8

    for it in range(max_iters):
        if remaining() < 5.0:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        w = np.zeros(len(corr))
        vals = corr[top_idx]
        wv = vals - vals.min() + 1e-10
        wv = wv ** 3
        wv /= wv.sum()
        w[top_idx] = wv

        g = np.zeros(n)
        for k in range(len(corr)):
            wk = w[k]
            if wk < 1e-14:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                np_ = n - lag
                g[lag:lag + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[lag:lag + np_]
            else:
                a = -lag
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:a + np_])
                g[a:a + np_] -= wk * h[:np_]

        g -= np.mean(g)
        step = step_init / (1.0 + 0.01 * it)

        h_new = project_feasible(h + momentum, n)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
