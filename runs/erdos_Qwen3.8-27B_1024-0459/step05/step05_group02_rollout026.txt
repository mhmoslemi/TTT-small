        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _ps_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    c = np.maximum(c, 1e-10)
    val = np.sum(c ** _P) ** (1.0 / _P) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _ps_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    c = np.maximum(c, 1e-10)
    S = np.sum(c ** _P)
    base = S ** (1.0 / _P - 1.0)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        w = base * (c[j] ** (_P - 1))
        g += w * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    beta = _SM_BETA
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = beta * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / beta) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    beta = _SM_BETA
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = beta * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_power(h, n, p, top_k, maxiter=500):
    global _N, _P, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _TOPK = top_k
    _P = p
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_ps_fun, h, jac=_ps_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxiter * 15})
        return proj(res.x, n)
    except Exception:
        return h


def lbfgs_softmax(h, n, beta, top_k, maxiter=500):
    global _N, _TOPK, _TGT, _SM_BETA
    _N = n
    _TGT = n / 2.0
    _TOPK = top_k
    _SM_BETA = beta
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxiter * 15})
        return proj(res.x, n)
    except Exception:
        return h


def binary_anneal(h_init, n, max_iters=50000, T0=0.01, T_end=0.0005, rng=None):
    """Simulated annealing on binary solutions."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    corr = compute_corr(h_bin)
    cur_c5 = float(np.max(corr) * dx)
    best_c5 = cur_c5
    cur_corr = corr.copy()

    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

    for it in range(max_iters):
        if remaining() < 5:
            break
        T = T0 * (T_end / T0) ** (it / max_iters)
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        # Compute delta in C5
        h_bin[i] = 0
        h_bin[j] = 1
        new_corr = compute_corr(h_bin)
        new_c5 = float(np.max(new_corr) * dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c5 = new_c5
            cur_corr = new_corr
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            # Update ones/zeros
            mask_i = (ones == i)
            mask_j = (zeros == j)
            ones = np.delete(ones, mask_i.nonzero()[0][0])
            ones = np.append(ones, j)
            zeros = np.delete(zeros, mask_j.nonzero()[0][0])
            zeros = np.append(zeros, i)
        else:
            h_bin[i] = 1
            h_bin[j] = 0

        if it % 2000 == 0 and it > 0:
            # Periodic re-optimization: try to reduce C5 by moving bits
            # toward gradient direction
            idx_top = np.argsort(-cur_corr)[:5]
            g = np.zeros(n)
            for ki in idx_top:
                g += corr_grad_single(h_bin, ki, n)
            g -= np.mean(g)
            # Find best single swap in gradient direction
            best_delta = 0
            best_swap = None
            top_ones = ones[np.argsort(-g[ones])[:8]]
            top_zeros = zeros[np.argsort(-g[zeros])[:8]]
            for io in top_ones:
                for iz in top_zeros:
                    if io == iz:
                        continue
                    h_bin[io] = 0
                    h_bin[iz] = 1
                    nc = compute_c5(h_bin)
                    h_bin[io] = 1
                    h_bin[iz] = 0
                    d = nc - cur_c5
                    if d < best_delta:
                        best_delta = d
                        best_swap = (io, iz)
            if best_swap is not None:
                io, iz = best_swap
                h_bin[io] = 0
                h_bin[iz] = 1
                cur_c5 = compute_c5(h_bin)
                cur_corr = compute_corr(h_bin)
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
                mask_i = (ones == io)
                mask_j = (zeros == iz)
                ones = np.delete(ones, mask_i.nonzero()[0][0])
                ones = np.append(ones, iz)
                zeros = np.delete(zeros, mask_j.nonzero()[0][0])
                zeros = np.append(zeros, io)

    return best_h, best_c5


def binary_hill_climb(h_init, n, max_iters=5000, top_k=5, rng=None):
    """Steepest-descent binary local search with multi-swap."""
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    cur_c5 = compute_c5(h_bin)
    best_c5 = cur_c5
    cur_corr = compute_corr(h_bin)

    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

    stall = 0
    for it in range(max_iters):
        if remaining() < 5:
            break
        idx_top = np.argsort(-cur_corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        best_delta = 0
        best_swap = None
        n_try_o = min(20, len(ones))
        n_try_z = min(20, len(zeros))
        top_ones = ones[np.argsort(-g[ones])[:n_try_o]]
        top_zeros = zeros[np.argsort(-g[zeros])[:n_try_z]]

        for io in top_ones:
            for iz in top_zeros:
                if io == iz:
                    continue
                h_bin[io] = 0
                h_bin[iz] = 1
                nc = compute_c5(h_bin)
                h_bin[io] = 1
                h_bin[iz] = 0
                d = nc - cur_c5
                if d < best_delta:
                    best_delta = d
                    best_swap = (io, iz)

        if best_swap is not None:
            io, iz = best_swap
            h_bin[io] = 0
            h_bin[iz] = 1
            cur_c5 = compute_c5(h_bin)
            cur_corr = compute_corr(h_bin)
            if cur_c5 < best_c5 - 1e-14:
                best_c5 = cur_c5
                best_h = h_bin.copy()
            mask_i = np.where(ones == io)[0]
            mask_j = np.where(zeros == iz)[0]
            if len(mask_i) > 0 and len(mask_j) > 0:
                ones = np.delete(ones, mask_i[0])
                ones = np.append(ones, iz)
                zeros = np.delete(zeros, mask_j[0])
                zeros = np.append(zeros, io)
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Random perturbation to escape
                for _ in range(10):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if i == j:
                        continue
                    h_bin[i] = 0
                    h_bin[j] = 1
                    nc = compute_c5(h_bin)
                    if nc < cur_c5:
                        cur_c5 = nc
                        cur_corr = compute_corr(h_bin)
                        if cur_c5 < best_c5 - 1e-14:
                            best_c5 = cur_c5
                            best_h = h_bin.copy()
                        mask_i = np.where(ones == i)[0]
                        mask_j = np.where(zeros == j)[0]
                        if len(mask_i) > 0 and len(mask_j) > 0:
                            ones = np.delete(ones, mask_i[0])
                            ones = np.append(ones, j)
                            zeros = np.delete(zeros, mask_j[0])
                            zeros = np.append(zeros, i)
                        stall = 0
                        break
                    else:
                        h_bin[i] = 1
                        h_bin[j] = 0
                else:
                    break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_gradient(h, n, top_k=10, max_iters=10000, step_init=0.05):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init
    stall = 0

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        # Weight by correlation value (larger corr gets more weight)
        w = c_top - c_top.min() + 0.01
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)
        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.12)
            else:
                step *= 0.95
        else:
            h = best_h.copy()
            v *= 0.3
            step *= 0.6

        if it % 500 == 0 and it > 0:
            step *= 0.9
            momentum = min(momentum + 0.003, 0.93)

        if stall > 2000:
            # Restart from best with small perturbation
            h = best_h + 0.01 * np.random.randn(n)
            h = proj(h, n)
            v = np.zeros(n)
            step = step_init * 0.5
            stall = 0

    return best_h, best_c5


def top_equalize(h, n, top_k=5, max_iters=500):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 4:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_max
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for scale in [3, 1, 5, 2, 0.5, 8, 0.1, 4, 0.2, 1.5]:
            alpha = 0.02 * scale
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for scale in [2, 1, 3, 0.5]:
                alpha = 0.01 * scale
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _P, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = proj(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_h = proj(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Power-sum L-BFGS-B at parent resolution with increasing p
    n = best_n
    h = best_h.copy()
    for p in [4, 8, 16, 32, 64, 128, 256]:
        if remaining() < 15:
            break
        h_new = lbfgs_power(h, n, p, top_k=20, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Soft-max L-BFGS-B with various betas
    for beta in [200, 500, 1000, 3000]:
        if remaining() < 12:
            break
        h_new = lbfgs_softmax(h, n, beta, top_k=10, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Nesterov gradient at parent resolution
    if remaining() > 30:
        h_pg, c5_pg = nesterov_gradient(h, n, top_k=10, max_iters=10000, step_init=0.04)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 4: Top equalization
    if remaining() > 15:
        h_eq, c5_eq = top_equalize(h, n, top_k=5, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 5: Binary hill climb at parent resolution
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_bc, c5_bc = binary_hill_climb(h_bin, n, max_iters=3000, top_k=5, rng=rng)
        if c5_bc < best_c5:
            best_c5 = c5_bc
            best_h = h_bc.copy()
        # Refine binary solution continuously
        h_ref = h_bc.copy()
        for p in [16, 64, 256]:
            if remaining() < 10:
                break
            h_ref = lbfgs_power(h_ref, n, p, top_k=10, maxiter=300)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 6: Multi-resolution exploration
    n_candidates = sorted(set([
        best_n + 2, best_n + 4, best_n + 8, best_n + 16,
        int(best_n * 1.15) if int(best_n * 1.15) % 2 == 0 else int(best_n * 1.15) + 1,
        int(best_n * 1.33) if int(best_n * 1.33) % 2 == 0 else int(best_n * 1.33) + 1,
        int(best_n * 1.5) if int(best_n * 1.5) % 2 == 0 else int(best_n * 1.5) + 1,
    ]))
    n_candidates = [nn for nn in n_candidates if 200 <= nn <= 800 and nn % 2 == 0 and nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # Power-sum optimization at new resolution
        for p in [8, 32, 128]:
            if remaining() < 15:
                break
            h_feas = lbfgs_power(h_feas, n_t, p, top_k=15, maxiter=350)

        # Soft-max refinement
        if remaining() > 15:
            h_feas = lbfgs_softmax(h_feas, n_t, 1000, top_k=8, maxiter=300)

        # Nesterov
        if remaining() > 20:
            h_pg2, c5_pg2 = nesterov_gradient(h_feas, n_t, top_k=8, max_iters=5000, step_init=0.03)
            if c5_pg2 < compute_c5(h_feas):
                h_feas = h_pg2

        # Top equalize
        if remaining() > 10:
            h_eq2, c5_eq2 = top_equalize(h_feas, n_t, top_k=5, max_iters=300)
            if c5_eq2 < compute_c5(h_feas):
                h_feas = h_eq2

        c5_feas = compute_c5(h_feas)

        # Binary search at this resolution
        h_b2 = (h_feas > 0.5).astype(float)
        if remaining() > 20:
            h_bl2, c5_bl2 = binary_hill_climb(h_b2, n_t, max_iters=2000, top_k=4, rng=rng)
            c5_bl2 = min(c5_bl2, c5_feas)
        else:
            h_bl2 = h_feas
            c5_bl2 = c5_feas

        # Refine best binary/continuous
        h_cand = h_bl2.copy()
        for p in [64, 256]:
            if remaining() < 10:
                break
            h_cand = lbfgs_power(h_cand, n_t, p, top_k=8, maxiter=250)
        c5_cand = compute_c5(h_cand)
        c5_cand = min(c5_cand, c5_bl2, c5_feas)

        if c5_cand < best_c5:
            # Determine which is best
            if c5_cand == compute_c5(h_cand) or c5_cand <= c5_bl2:
                best_c5 = c5_cand
                best_h = h_cand.copy() if c5_cand == compute_c5(h_cand) else h_bl2.copy()
            else:
                best_c5 = c5_bl2
                best_h = h_bl2.copy()
            best_n = n_t

    # Phase 7: Binary simulated annealing at best resolution
    if remaining() > 40:
        h_bin_a = (best_h > 0.5).astype(float)
        h_an, c5_an = binary_anneal(h_bin_a, best_n, max_iters=30000, T0=0.015, T_end=0.001, rng=rng)
        if c5_an < best_c5:
            best_c5 = c5_an
            best_h = h_an.copy()
        # Refine
        h_ref2 = h_an.copy()
        for p in [32, 128, 512]:
            if remaining() < 10:
                break
            h_ref2 = lbfgs_power(h_ref2, best_n, p, top_k=10, maxiter=300)
        c5_ref2 = compute_c5(h_ref2)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2.copy()

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # High-p power-sum
    for p in [256, 1024, 4096]:
        if remaining() < 8:
            break
        h_new = lbfgs_power(h, n, p, top_k=5, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # High-beta soft-max
    for beta in [5000, 10000, 20000]:
        if remaining() < 8:
            break
        h_new = lbfgs_softmax(h, n, beta, top_k=4, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final top equalize with larger k
    if remaining() > 10:
        h_eq3, c5_eq3 = top_equalize(h, n, top_k=8, max_iters=500)
        if c5_eq3 < best_c5:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_hill_climb(h_bin_f, best_n, max_iters=3000, top_k=4, rng=rng)
        h_rf = h_bf.copy()
        for p in [256, 1024]:
            if remaining() < 8:
                break
            h_rf = lbfgs_power(h_rf, best_n, p, top_k=5, maxiter=300)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()
        elif c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Final Nesterov polish
    if remaining() > 15:
        h_nf, c5_nf = nesterov_gradient(best_h, best_n, top_k=5, max_iters=5000, step_init=0.01)
        if c5_nf < best_c5:
            best_c5 = c5_nf
            best_h = h_nf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
