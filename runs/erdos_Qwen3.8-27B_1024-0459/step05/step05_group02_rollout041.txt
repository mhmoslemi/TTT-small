def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):

            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        # Find the element closest to 0.5 and adjust
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def _sm_fun(x):

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def projected_subgradient(h, n, top_k=5, max_iters=5000, step_init=0.01):

        c_top = corr[idx]
        
        # Softmax weights
        beta_w = 50.0
        sh = beta_w * (c_top - c_top.max())
        w = np.exp(sh) / np.exp(sh).sum()
        
        g = np.zeros(n)
        for j, ki in enumerate(idx):

        g -= np.mean(g)

        g[(h <= 1e-9)] = 0.0

        g /= gn

def compute_corr(h):

def proj(h, n):

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

    # Phase 4: Try intermediate resolutions
    candidates = []
    for mult in [1.2, 1.4, 1.6, 1.75, 1.5]:
        nc = int(best_n * mult)
        nc = nc if nc % 2 == 0 else nc + 1
        if 200 <= nc <= 1536 and nc != best_n:
            candidates.append(nc)
    candidates = sorted(set(candidates))

    for n_t in candidates[:4]:
        if remaining() < 40:
            break
        h_t = interp_to(best_h, best_n, n_t)
        h_t = proj(h_t, n_t)

        for beta, tk in [(1000, 5), (5000, 4), (20000, 3)]:
            if remaining() < 6:
                break
            h_t = lbfgs_pass(h_t, n_t, beta, top_k=tk, maxiter=250)

        if remaining() > 15:
            h_t, c5_t = top_equalize(h_t, n_t, top_k=4, max_iters=150)
        else:
            c5_t = compute_c5(h_t)

        # Binary ILS
        if remaining() > 25:
            h_t_bin = (h_t > 0.5).astype(float)
            h_t_il, c5_t_il = binary_ilsearch(h_t_bin, n_t, n_restarts=8, top_k=4, max_iters=2000, seed=seed + 10)
            if c5_t_il < c5_t:
                h_t = h_t_il
                c5_t = c5_t_il

        # Refine
        if remaining() > 10:
            h_t_ref = h_t.copy()
            for beta in [10000, 40000]:
                if remaining() < 4:
                    break
                h_t_ref = lbfgs_pass(h_t_ref, n_t, beta, top_k=3, maxiter=200)
            c5_t_ref = compute_c5(h_t_ref)
            if c5_t_ref < c5_t:
                h_t = h_t_ref
                c5_t = c5_t_ref

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 5: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    # Subgradient
    if remaining() > 30:
        h_pg, c5_pg = projected_subgradient(h, n, top_k=5, max_iters=4000, step_init=0.01)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Very high beta L-BFGS
    for beta, tk in [(50000, 3), (100000, 3), (200000, 2)]:
        if remaining() < 6:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=tk, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Top equalize
    if remaining() > 15:
        h_eq, c5_eq = top_equalize(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_ilsearch(h_bin_f, n, n_restarts=8, top_k=4, max_iters=2500, seed=seed + 20)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

        # Sliding window
        if remaining() > 15:
            h_sw, c5_sw = sliding_window_local_search(best_h, n, window=3, max_iters=2000)
            if c5_sw < best_c5:
                best_c5 = c5_sw
                best_h = h_sw.copy()

        # Refine
        h_rf = best_h.copy()
        for beta, tk in [(20000, 3), (80000, 2)]:
            if remaining() < 4:
                break
            h_rf = lbfgs_pass(h_rf, n, beta, top_k=tk, maxiter=250)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()

    # Final top equalize
    if remaining() > 10:
        h_eq2, c5_eq2 = top_equalize(best_h, best_n, top_k=8, max_iters=250)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
