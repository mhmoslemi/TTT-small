def _slsqp_single(h, n, beta, maxiter, dx):
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        return (m + np.log(np.sum(ez)) / beta) * dx

    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:np_]
        return g

    def _eq_fun(x):
        return np.sum(x) - n / 2.0

    def _eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,

        pass
    return h
