def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        mask = (h_proj > 1e-14) & (h_proj < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
        else:
            idx = int(np.argmin(h_proj))
            h_proj[idx] = np.clip(h_proj[idx] + diff, 0.0, 1.0)
    return h_proj

def smooth_obj_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        aoff = abs(offset)
        npairs = n - aoff
        if npairs > 0:
            grad[aoff:aoff + npairs] += w * (1.0 - h[:npairs])
            grad[:npairs] -= w * h[aoff:aoff + npairs]
    # Fix the offset=0 term (quadratic)
    # At offset 0, the derivative is 1-2h[i], but the above gives (1-h[i])-h[i] = 1-2h[i]. Correct.
    grad *= dx
    return val, grad

            break

        y = np.clip(y, 0.0, 1.0)
        f_y, grad_y = smooth_obj_and_grad(y, n, beta)

def coord_descent(h, n, max_sweeps=200, verbose=False):

    ncorr = 2 * n - 1

            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)
            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0
            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)
            f_old = float(np.max(best_corr))
            # Try candidates
            t_best = old
            f_best = f_old
            # Grid of candidates
            candidates = [0.0, 1.0, 0.5, old + 0.1, old - 0.1, old + 0.05, old - 0.05]
            for tc in candidates:
                if tc < 0 or tc > 1 or tc == old:
                    continue
                mid = base + A * tc
                mid[n - 1] = base[n - 1] + tc - tc * tc

                if fc < f_best - 1e-14:

                    t_best = tc
            # Golden section refine if improved

                if hi_gs - lo_gs < 1e-10:
                    hi_gs = lo_gs + 1e-6
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi_gs - gr * (hi_gs - lo_gs)
                c2 = lo_gs + gr * (hi_gs - lo_gs)

                def _eval_gs(t):
                    mid = base + A * t
                    mid[n - 1] = base[n - 1] + t - t * t
                    return float(np.max(mid))

                f1 = _eval_gs(c1)
                f2 = _eval_gs(c2)
                for _ in range(50):

                        c2 = c1
                        f2 = f1
                        c1 = hi_gs - gr * (hi_gs - lo_gs)

                        c1 = c2
                        f1 = f2
                        c2 = lo_gs + gr * (hi_gs - lo_gs)
                        f2 = _eval_gs(c2)

                f_best = _eval_gs(t_best)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def random_binary_init(n, rng):

    ones = rng.choice(n, n // 2, replace=False)

    return h


def periodic_init(n, period):
    h = np.zeros(n)

                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 8: SLSQP polish
    if remaining_time() > 30:
        for beta in [5000, 20000, 80000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_refine(best_h, best_n, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final SLSQP
    if remaining_time() > 8:
        h_new = slsqp_refine(best_h, best_n, 100000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
