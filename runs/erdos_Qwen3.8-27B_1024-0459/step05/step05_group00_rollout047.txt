        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def smooth_max_slsqp(h, n, beta, maxiter=300):
    def sm_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def sm_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-30:
            return np.zeros(n)
        w = ex / s
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            g += wk * grad_corr_k(x, k, n)
        return g * (2.0 / n)

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h.copy()

def coordinate_descent(h, n, n_sweeps=5, n_samples=64):
    """Coordinate descent: for each j, optimize h[j] in 1D."""
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        
        indices = np.arange(n)
        for j in indices:
            rem = _BUDGET - (time.time() - _T0)
            if rem < 5:
                    break
            
            # Compute B_s = d corr[s] / d h[j] for all s
            # For lag offset = s - (n-1), the shift is offset
            # h[j] appears in:
            #   First position: for shifts offset where 0 <= j < n and 0 <= j+offset < n
            #     → -j <= offset < n-j, i.e., shift s = offset in [-j, n-1-j]
            #   Second position: for shifts where 0 <= j-offset < n
            #     → j-n < offset <= j, i.e., shift s in [j-n+1, j]
            
            # B_s for first position: (1 - h[j+offset]) for valid offsets
            # B_s for second position: -h[j-offset] for valid offsets
            
            # corr[s] = sum of h[i]*(1-h[i+offset]) over valid i
            # d corr[s] / d h[j]:
            #   If j is in "first" position: i=j, contributes (1-h[j+offset])
            #   If j is in "second" position: i+offset=j, i=j-offset, contributes -h[j-offset]
            
            B = np.zeros(2*n - 1)  # indexed by s = 0..2n-2 (corresponding to offset = s-(n-1))
            for s in range(2*n-1):
                offset = s - (n-1)
                # First position: valid if 0 <= j < n and 0 <= j+offset < n
                if 0 <= j + offset < n:
                    B[s] += (1.0 - h[j + offset])
                # Second position: valid if 0 <= j-offset < n
                if 0 <= j - offset < n:
                    B[s] -= h[j - offset]
            
            # corr_base[s] = corr[s] - B[s]*h[j]
            corr = np.correlate(h, 1.0 - h, mode='full')
            corr_base = corr - B * h[j]
            
            # Minimize max_s (corr_base[s] + B[s]*t) for t in [0,1]
            # Use sampling + local refinement
            t_vals = np.linspace(0, 1, n_samples)
            f_vals = np.max(corr_base[None, :] + B[None, :] * t_vals[:, None], axis=1)
            t_best = t_vals[np.argmin(f_vals)]
            
            # Local refinement
            if 0 < t_best < 1:
                # Fine grid around t_best
                t_fine = np.linspace(t_best - 1.0/n_samples, t_best + 1.0/n_samples, n_samples)
                t_fine = np.clip(t_fine, 0, 1)
                f_fine = np.max(corr_base[None, :] + B[None, :] * t_fine[:, None], axis=1)
                t_best = t_fine[np.argmin(f_fine)]
            
            h[j] = t_best
        
        # Project to fix sum
        h = project_feasible(h, n)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
        else:

            pass
    
    return h, best_c5
