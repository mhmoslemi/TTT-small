def smooth_max_val_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + (max_shifted + np.log(wsum)) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    active_k = np.where(weights > 1e-16)[0]
    for k in active_k:
        w = weights[k]
        off = int(k) - (n - 1)
        if off == 0:
            grad += w * (1.0 - 2.0 * h)
        elif off > 0:
            npairs = n - off
            if npairs > 0:
                grad[:npairs] += w * (1.0 - h[off:off + npairs])
                grad[off:] -= w * h[:npairs]
        else:
            aoff = -off
            npairs = n - aoff
            if npairs > 0:
                grad[aoff:] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[aoff:aoff + npairs]
    grad *= dx
    return val, grad

def make_obj(name_beta):
    n, beta = name_beta

    def _obj(x):
        return smooth_max_val_grad(x, n, beta)[0]

    def _grad(x):
        return smooth_max_val_grad(x, n, beta)[1]

    return _obj, _grad

def slsqp_optimize(h, n, beta, maxiter=400, ftol=1e-18):
    n, b = (n, beta)

    def _obj(x):
        return smooth_max_val_grad(x, n, b)[0]

    def _grad(x):
        return smooth_max_val_grad(x, n, b)[1]

    def _con(x):
        return np.sum(x) - n / 2.0

    def _sjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj, h.copy(), jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=[{'type': 'eq', 'fun': _con, 'jac': _sjac}],
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-14:
                return trial
    except Exception:
        pass
    return h.copy()

def coord_descent_fast(h, n, n_sweeps=40):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(n_sweeps):
        if remaining_time() < 3:
            break
        improved_any = False
        interior = np.where((best > 1e-8) & (best < 1.0 - 1e-8))[0]
        if len(interior) == 0:
            interior = np.arange(n)

        for i in interior:
            if remaining_time() < 1:
                break
            t0 = best[i]
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n - 1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0 - 2.0 * t0
                    b[k] += 0.0
                elif off > 0:
                    if i < n - off:
                        a[k] += (1.0 - best[i + off])
                    if i >= off:
                        a[k] += (-best[i - off])
                else:
                    aoff = -off
                    if i >= aoff:
                        a[k] += (1.0 - best[i - aoff])
                    if i < n - aoff:
                        a[k] += (-best[i + aoff])
            base = best_corr - a * t0 - b * t0 * t0
            cands = [0.0, 1.0, max(0.0, t0 - 0.15), min(1.0, t0 + 0.15),
                     max(0.0, t0 - 0.05), min(1.0, t0 + 0.05), 1.0 - t0]
            best_t = t0
            best_f = np.max(best_corr) * dx
            for t in cands:
                if abs(t - t0) < 1e-12:
                    continue
                f = np.max(base + a * t + b * t * t) * dx
                if f < best_f - 1e-15:
                    best_f = f
                    best_t = t
            if best_t != t0:
                margin = 0.15
                lo = max(0.0, best_t - margin)
                hi = min(1.0, best_t + margin)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                for _ in range(50):
                    if hi - lo < 1e-14:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                t_opt = (lo + hi) / 2.0
                f_opt = np.max(base + a * t_opt + b * t_opt * t_opt) * dx
                if f_opt < best_f:
                    best_f = f_opt
                    best_t = t_opt
            if best_f < best_c5 - 1e-14:
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved_any = True
        if not improved_any:
            break
    return best, best_c5

def binary_local_search(h, n, max_sweeps=30):
    h_bin = np.where(h >= 0.5, 1.0, 0.0).astype(float)
    h_bin = project_feasible(h_bin, n)
    c5 = compute_c5(h_bin)
    corr = compute_corr(h_bin).copy()
    ones = list(np.where(h_bin > 0.5)[0])
    zeros = list(np.where(h_bin < 0.5)[0])

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in ones[:]:
            if remaining_time() < 2:
                break
            for j in zeros[:]:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                new_corr = compute_corr(h_bin)
                new_c5 = np.max(new_corr) * (2.0 / n)
                if new_c5 < c5 - 1e-15:
                    c5 = new_c5
                    corr = new_corr
                    ones.remove(i)
                    zeros.remove(j)
                    ones.append(j)
                    zeros.append(i)
                    improved = True
                else:
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
        if not improved:
            break
    return h_bin, c5

def try_resolution(h_best, n_best, target_n, budget_check):
    if target_n == n_best:
        return h_best.copy(), compute_c5(h_best)
    xp = (np.arange(n_best) + 0.5) * 2.0 / n_best
    x_new = (np.arange(target_n) + 0.5) * 2.0 / target_n
    h_r = np.interp(x_new, xp, h_best)
    h_r = project_feasible(h_r, target_n)
    c5_r = compute_c5(h_r)
    for beta in [5000, 50000, 200000]:
        if budget_check() < 30:
            break
        h_r = slsqp_optimize(h_r, target_n, beta, 300)
    if budget_check() > 20:
        h_r, c5_r = coord_descent_fast(h_r, target_n, n_sweeps=25)
    h_r = project_feasible(h_r, target_n)
    c5_r = compute_c5(h_r)
    return h_r, c5_r

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP with increasing beta
    for beta in [2000, 10000, 50000, 200000, 800000]:
        if remaining_time() < 40:
            break
        h_new = slsqp_optimize(h, n, beta, 500)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Coordinate descent (intensive)
    if remaining_time() > 60:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=80)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: More SLSQP at very high beta
    for beta in [500000, 2000000, 10000000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_optimize(h, n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Binary projection + local search
    if remaining_time() > 120:
        h_bin, c5_bin = binary_local_search(best_h, best_n, max_sweeps=20)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h = h_bin.copy()
        # Refine from binary
        for beta in [10000, 100000, 500000]:
            if remaining_time() < 30:
                break
            h_ref = slsqp_optimize(h_bin, best_n, beta, 300)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                h = h_ref.copy()

    # Phase 5: Another coordinate descent round
    if remaining_time() > 40:
        h_new, c5_new = coord_descent_fast(h, n, n_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Subgradient-style descent
    h = best_h.copy()
    n_sgd = max(10, n // 20)
    for it in range(300):
        if remaining_time() < 10:
            break
        corr = compute_corr(h)
        max_corr = np.max(corr)
        active = np.where(corr >= max_corr * 0.99)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 8:
            active = active[np.argsort(corr[active])[::-1]][:8]
        grad = np.zeros(n)
        for k_idx in active:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += (1.0 - h[:npairs])
                    grad[:npairs] -= h[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= h[:npairs]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad /= gnorm
        base_step = 0.06 / (1.0 + it * 0.003)
        improved = False
        for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 0.1, 3.0]:
            step = base_step * sm
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved:
            if it % 100 == 80 and remaining_time() > 40:
                h_new = slsqp_optimize(h, n, 100000, 250)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                if it > 200:
                    break

    # Phase 7: Try different resolutions
    n = best_n
    higher_res = set()
    for mult in [1.25, 1.5, 1.33, 2.0]:
        rn = int(round(n * mult))
        if rn % 2 != 0:
            rn += 1
        if 80 <= rn <= 1280:
            higher_res.add(rn)
    for extra in [128, 256, -128]:
        rn = n + extra
        if rn % 2 != 0:
            rn += 1
        if 80 <= rn <= 1280:
            higher_res.add(rn)
    higher_res = sorted(higher_res)

    for rn in higher_res:
        if remaining_time() < 80:
            break
        h_r, c5_r = try_resolution(best_h, best_n, rn, remaining_time)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 8: Random perturbations + SLSQP
    n = best_n
    h = best_h.copy()
    n_perturb = max(8, n // 40)
    for pert in range(40):
        if remaining_time() < 30:
            break
        h_pert = h.copy()
        idx = rng.choice(n, n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.1, 0.1, n_pert)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.01:
            h_new = slsqp_optimize(h_pert, n, 50000, 250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 9: Final intensive coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = coord_descent_fast(best_h, best_n, n_sweeps=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 10:
        h_new = slsqp_optimize(best_h, best_n, 5000000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: Final coordinate descent
    if remaining_time() > 8:
        h_new, c5_new = coord_descent_fast(best_h, best_n, n_sweeps=20)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
