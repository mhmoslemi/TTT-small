                if 0.0 < fc < 1.0:
                    candidates.append(fc)
            
            f_old = np.max(corr)
            t_best = old
            f_best = f_old
            
            for t in candidates:
                t = min(1.0, max(0.0, t))
                mid = base + A * t
                mid[n - 1] = base[n - 1] + t - t * t
                f_val = np.max(mid)
                if f_val < f_best - 1e-14:
                    f_best = f_val
                    t_best = t
            
            # Golden section around t_best
            if t_best != old:
                # Find a bracket
                if t_best > old:
                    lo = old
                    hi = min(1.0, t_best + 0.3)
                else:
                    lo = max(0.0, t_best - 0.3)
                    hi = old
                if hi - lo < 1e-8:
                    hi = min(1.0, lo + 0.1)
                    lo = max(0.0, hi - 0.1)
                
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                
                def _f_eval(t):
                    mid = base + A * t
                    mid[n - 1] = base[n - 1] + t - t * t
                    return np.max(mid)
                
                f1 = _f_eval(c1)
                f2 = _f_eval(c2)
                for _ in range(50):
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _f_eval(c1)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = _f_eval(c2)
                t_best = (lo + hi) / 2.0
                f_best = _f_eval(t_best)
            
            if f_best < f_old - 1e-13:
                best[i] = t_best
                corr += A * (t_best - old)
                corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                improved = True
        
        if not improved:
            break
    
    return best, np.max(compute_corr(best)) * dx

def slsqp_opt(h, n, beta, maxiter=500):
    global _N_W, _BETA_W
    _N_W = n
    _BETA_W = beta
    
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        vals = np.exp(shifted)
        ws = np.sum(vals)
        return (m + np.log(ws) / beta) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        vals = np.exp(shifted)
        w = vals / np.sum(vals)
        g = np.zeros(n)
        for k in range(ncorr):
            wk = w[k]
            if wk < 1e-12:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao+np_])
                g[ao:ao+np_] -= wk * x[:np_]
        g *= dx
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        c = 0
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    return project_feasible(h, n)

def interp(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(xn, xp, h_src)
