            i_minus = i - offsets
            valid_p = (i_plus >= 0) & (i_plus < n)
            valid_m = (i_minus >= 0) & (i_minus < n)
            
            A = np.zeros(ncorr)
            if np.any(valid_m):
                A[valid_m] += (1.0 - best[i_minus[valid_m]])
            if np.any(valid_p):
                A[valid_p] -= best[i_plus[valid_p]]
            A[n-1] = 0.0
            
            base = best_corr - A * old
            base[n-1] = best_corr[n-1] - old * (1.0 - old)
            
            # Minimize max(base[k] + A[k]*t) over t in [0,1], with base[n-1]+t(1-t)
            f_old = max(base[n-1] + old * (1.0 - old), np.max(base + A * old))
            
            # Try a few candidates
            t_cands = [0.0, 1.0, 0.5, old]
            # Also try the "balancing" point
            # The zero-lag term is base[n-1] + t - t², which is concave in t
            # So the max is likely at a boundary or where another term equals it
            
            best_t = old
            best_f = f_old
            
            for t in t_cands:
                f = max(base[n-1] + t - t*t, np.max(base + A * t))
                if f < best_f - 1e-14:
                    best_f = f
                    best_t = t
            
            # Golden section if we found an improvement
            if best_t != old:
                if best_t > old:
                    lo, hi = old, min(1.0, best_t + 0.1)
                else:
                    lo, hi = max(0.0, best_t - 0.1), old
                hi = max(hi, lo + 1e-8)
                
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(35):
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = max(base[n-1] + c1 - c1*c1, np.max(base + A * c1))
                    f2 = max(base[n-1] + c2 - c2*c2, np.max(base + A * c2))
                    if f1 < f2:
                        hi = c2
                    else:
                        lo = c1
                t_mid = (lo + hi) / 2.0
                f_mid = max(base[n-1] + t_mid - t_mid*t_mid, np.max(base + A * t_mid))
                if f_mid < best_f - 1e-13:
                    best_f = f_mid
                    best_t = t_mid
            
            if best_f < f_old - 1e-12:
                best[i] = best_t
                best_corr += A * (best_t - old)
                best_corr[n-1] += best_t * (1.0 - best_t) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True
        
        if not improved:
            break
    
    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
