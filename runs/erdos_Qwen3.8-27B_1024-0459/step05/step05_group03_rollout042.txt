    corr = np.correlate(x, 1.0 - x, mode='full')

    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(x)
    w = w * dx
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:a + np_])
            g[a:a + np_] -= wk * x[:np_]
    return g


def slsqp_run(h, n, beta, maxiter=300):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def projected_subgrad(h, n, top_k=5, max_iters=2000, seed=42):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.8
    step = 0.03
    rng = np.random.RandomState(seed)

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w_top = c_top - c_top.min() + 1e-10
        w_top = w_top ** 2
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= 2.0 / n
        g = smooth_grad(h, 1.0)
        # Use weighted gradient
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:a + np_])
                g[a:a + np_] -= wk * h[:np_]
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        step *= 0.9997
        if it % 100 == 0:
            mu = min(0.95, mu + 0.01)
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Generate candidates
    candidates = []

    # Try Thue-Morse at various n
    for n_try in [256, 384, 512, 768, 1024]:
        if n_try % 2 != 0:
            continue
        tm = thue_morse(n_try)
        # Check if it has exactly n/2 ones
        if int(np.sum(tm)) == n_try // 2:
            c5_tm = compute_c5(tm)
            candidates.append((c5_tm, tm, n_try))
        else:
            # Adjust: use first n_try of Thue-Morse, then fix count
            # Actually for powers of 2 it's always balanced. For others, skip.
            pass

    # Also try n=384 Thue-Morse (should be balanced as shown)
    if 384 not in [c[2] for c in candidates]:
        tm384 = thue_morse(384)
        if int(np.sum(tm384)) == 192:
            c5_tm384 = compute_c5(tm384)
            candidates.append((c5_tm384, tm384, 384))

    # Try parent solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        c5_parent = compute_c5(parent_h)
        candidates.append((c5_parent, parent_h.copy(), parent_n))
    except Exception:
        pass

    # Try a few random binary sequences
    for n_rand in [384, 512]:
        if remaining() < 10:
            break
        for trial in range(3):
            h_rand = np.zeros(n_rand)
            idx = rng.permutation(n_rand)
            h_rand[idx[:n_rand // 2]] = 1.0
            c5_rand = compute_c5(h_rand)
            candidates.append((c5_rand, h_rand, n_rand))

    # Sort by C5 and process best ones
    candidates.sort(key=lambda x: x[0])

    # Process top candidates with local search
    for c5_init, h_init, n_init in candidates[:5]:
        if remaining() < 30:
            break

        # Already good?
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_init.copy()
            best_n = n_init

        h_bin = (h_init > 0.5).astype(float)

        # Phase A: Exhaustive binary local search
        if remaining() > 20 and n_init <= 512:
            h_ls, c5_ls = exhaustive_binary_ls(h_bin, n_init, max_rounds=30, seed=seed)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n_init

            # Phase B: Double swap local search
            if remaining() > 20:
                h_ds, c5_ds = double_swap_ls(h_ls, n_init, max_rounds=20, n_trials=3000, seed=seed + 1)
                if c5_ds < best_c5:
                    best_c5 = c5_ds
                    best_h = h_ds.copy()
                    best_n = n_init
                h_current = h_ds
            else:
                h_current = h_ls

            # Phase C: Simulated annealing
            if remaining() > 60:
                n_sa = min(2000000, int(remaining() * 15000))
                h_sa, c5_sa = simulated_annealing_binary(h_current, n_init, n_steps=n_sa, T0=0.006, T1=0.00003, seed=seed + 2)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n_init

                # After SA, do more local search
                if remaining() > 20:
                    h_ls2, c5_ls2 = exhaustive_binary_ls(h_sa, n_init, max_rounds=20, seed=seed + 3)
                    if c5_ls2 < best_c5:
                        best_c5 = c5_ls2
                        best_h = h_ls2.copy()
                        best_n = n_init

        elif remaining() > 30 and n_init > 512:
            # For larger n, use SA directly
            n_sa = min(1500000, int(remaining() * 15000))
            h_sa, c5_sa = simulated_annealing_binary(h_bin, n_init, n_steps=n_sa, T0=0.005, T1=0.00002, seed=seed + 4)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_init
            if remaining() > 20:
                h_ls, c5_ls = exhaustive_binary_ls(h_sa, n_init, max_rounds=15, seed=seed + 5)
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()
                    best_n = n_init

    # Phase D: Continuous relaxation from best binary
    if remaining() > 30 and best_h is not None:
        n_cont = best_n
        h_cont = best_h.copy()
        for beta in [1000, 5000, 20000]:
            if remaining() < 15:
                break
            h_opt = slsqp_run(h_cont, n_cont, beta, maxiter=300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_cont = h_opt.copy()

    # Phase E: Subgradient equalization
    if remaining() > 30 and best_h is not None:
        h_eq, c5_eq = projected_subgrad(best_h, best_n, top_k=5, max_iters=1500, seed=seed + 10)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase F: Multi-resolution
    if remaining() > 60 and best_h is not None:
        n_candidates = [256, 384, 512, 640, 768]
        n_candidates = [x for x in n_candidates if x != best_n and x % 2 == 0]
        n_candidates.sort(key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:3]:
            if remaining() < 40:
                break
            h_interp = interp_to(best_h, best_n, n_t)
            h_bin_t = (h_interp > 0.5).astype(float)
            # Fix the count
            n_ones = int(np.sum(h_bin_t))
            if n_ones > n_t // 2:
                idx_ones = np.where(h_bin_t == 1)[0]
                # Remove excess ones (those with lowest h_interp value)
                excess = n_ones - n_t // 2
                vals = h_interp[idx_ones]
                remove_idx = idx_ones[np.argsort(vals)[:excess]]
                h_bin_t[remove_idx] = 0.0
            elif n_ones < n_t // 2:
                idx_zeros = np.where(h_bin_t == 0)[0]
                deficit = n_t // 2 - n_ones
                vals = h_interp[idx_zeros]
                add_idx = idx_zeros[np.argsort(-vals)[:deficit]]
                h_bin_t[add_idx] = 1.0

            c5_t = compute_c5(h_bin_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_bin_t.copy()
                best_n = n_t

            if remaining() > 20:
                h_ls_t, c5_ls_t = exhaustive_binary_ls(h_bin_t, n_t, max_rounds=15, seed=seed + 20)
                if c5_ls_t < best_c5:
                    best_c5 = c5_ls_t
                    best_h = h_ls_t.copy()
                    best_n = n_t

    # Phase G: Final intensive polish
    if remaining() > 20 and best_h is not None:
        h_final = (best_h > 0.5).astype(float)
        if int(np.sum(h_final)) != best_n // 2:
            h_final = project_feasible(best_h, best_n)
            h_final = np.where(h_final > 0.5, 1.0, 0.0)
            # Fix count
            s = int(np.sum(h_final))
            if s > best_n // 2:
                ones_idx = np.where(h_final == 1)[0]
                h_final[ones_idx[-(s - best_n // 2):]] = 0.0
            elif s < best_n // 2:
                zeros_idx = np.where(h_final == 0)[0]
                h_final[zeros_idx[:best_n // 2 - s]] = 1.0

        h_fin, c5_fin = exhaustive_binary_ls(h_final, best_n, max_rounds=25, seed=seed + 30)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

        # SA from best
        if remaining() > 30:
            n_sa = min(1000000, int(remaining() * 12000))
            h_sa2, c5_sa2 = simulated_annealing_binary(h_fin, best_n, n_steps=n_sa2, T0=0.004, T1=0.00001, seed=seed + 40)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()

        # SLSQP from best
        if remaining() > 20:
            h_pol = slsqp_run(best_h, best_n, 10000, maxiter=200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final validation
    if best_h is None:
        best_n = 384
        best_h = thue_morse(best_n)
        best_c5 = compute_c5(best_h)

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
