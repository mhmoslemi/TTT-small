                            np_ = n - ao
                            if idx < np_:
                                h_trial[idx] = np.clip(h_trial[idx] - trial_step, 0, 1)
                            elif idx >= ao:
                                h_trial[idx] = np.clip(h_trial[idx] + trial_step, 0, 1)
                    h_trial = project_feasible(h_trial, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5 - 1e-14:
                        h = h_trial
                        best_c5 = c5_trial
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5


def binary_local_search(h_bin, n, rng, time_limit=60):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - n_half]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:n_half - n_ones]]] = 1
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > 8000:
            break
        corr = compute_corr(h)
        max_val = float(np.max(corr))
        top_k = min(8, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_k]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0
        best_swap = None
        for _ in range(25):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)
        for _ in range(10):
            k_hot = top_idx[rng.randint(len(top_idx))]
            off = k_hot - (n - 1)
            if off >= 0:
                bad = np.where((h[:n - off] == 1) & (h[off:] == 0))[0]
            else:
                ao = -off
                bad = np.where((h[ao:] == 1) & (h[:n - ao] == 0))[0]
            if len(bad) == 0:
                continue
            if off >= 0:
                i = bad[rng.randint(len(bad))]
            else:
                i = bad[rng.randint(len(bad))] + ao
            j = zeros[rng.randint(len(zeros))]
            if j == i:
                continue
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            no_improve = 0
            cur_max = float(np.max(compute_corr(h)))
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            if no_improve > 400:
                n_flip = rng.randint(2, 5)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_flip):
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0
                    h[j] = 1
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                cur_max = float(np.max(compute_corr(h)))
    return best_h, best_max * dx


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def make_sinusoidal(n, freq=4, amp=0.3):
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x)
    return project_feasible(h, n)


def make_block(n, n_blocks=4):
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        if i % 2 == 0:
            h[i * block_size:(i + 1) * block_size] = 1.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = make_sinusoidal(parent_n, 4, 0.3)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: FISTA progressive at parent resolution
    h_new = fista_progressive(h, n,
                              betas=[50, 200, 800, 3000, 10000, 30000, 80000],
                              max_iter_per=3000,
                              max_lags_list=[None, None, None, 300, 150, 80, 40])
    c5_new = compute_c5(h_new)
    if c5_new < best_c5:
        best_c5 = c5_new
        best_h = h_new.copy()
        h = h_new.copy()

    # Phase 2: Coordinate descent polish
    if remaining() > 15:
        h_cd, c5_cd = coord_descent_polish(h, n, max_rounds=300)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 3: Binary local search
    if remaining() > 30:
        h_bin = (h > 0.5).astype(float)
        n_half = n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            if n_ones > n_half:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[:n_ones - n_half]] = 0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[:n_half - n_ones]] = 1
        sa_time = min(80, max(15, int(remaining() * 0.2)))
        h_b, c5_b = binary_local_search(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            # Refine from binary
            h_ref = fista_progressive(h_b, n,
                                      betas=[1000, 5000, 20000, 60000],
                                      max_iter_per=2000,
                                      max_lags_list=[200, 100, 50, 30])
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
            h = best_h.copy()
        else:
            h = best_h.copy()

    # Phase 4: Multi-resolution exploration
    if remaining() > 60:
        n_base = best_n
        n_candidates = [n_base - 32, n_base - 16, n_base - 8, n_base + 8,
                        n_base + 16, n_base + 32, n_base + 64, n_base + 96]
        # Add some round numbers
        for nt in [256, 320, 448, 512, 576, 640, 768]:
            if nt not in n_candidates and 128 <= nt <= 900:
                n_candidates.append(nt)
        n_candidates = [nt for nt in n_candidates if 128 <= nt <= 900 and nt % 2 == 0]
        n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:6]:
            if remaining() < 30:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            h_t = fista_progressive(h_t, n_t,
                                    betas=[200, 1000, 5000, 20000, 60000],
                                    max_iter_per=2000,
                                    max_lags_list=[None, 200, 100, 50, 30])
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t
            # Try binary at this resolution if close
            if c5_t < best_c5 + 0.002 and remaining() > 25:
                h_t_bin = (h_t > 0.5).astype(float)
                n_h = n_t // 2
                n_o = int(np.sum(h_t_bin))
                if n_o > n_h:
                    idx = np.where(h_t_bin == 1)[0]
                    perm = rng.permutation(len(idx))
                    h_t_bin[idx[perm[:n_o - n_h]]] = 0
                elif n_o < n_h:
                    idx = np.where(h_t_bin == 0)[0]
                    perm = rng.permutation(len(idx))
                    h_t_bin[idx[perm[:n_h - n_o]]] = 1
                h_tb, c5_tb = binary_local_search(h_t_bin, n_t, rng, time_limit=20)
                if c5_tb < best_c5:
                    best_c5 = c5_tb
                    best_h = h_tb.copy()
                    best_n = n_t
                    h_t = fista_progressive(h_tb, n_t,
                                            betas=[2000, 10000, 40000],
                                            max_iter_per=1500,
                                            max_lags_list=[100, 50, 30])
                    c5_tr = compute_c5(h_t)
                    if c5_tr < best_c5:
                        best_c5 = c5_tr
                        best_h = h_t.copy()

    # Phase 5: Diverse starts at best resolution
    if remaining() > 60:
        n_cur = best_n
        diverse_starts = [
            make_sinusoidal(n_cur, 3, 0.25),
            make_sinusoidal(n_cur, 5, 0.35),
            make_sinusoidal(n_cur, 7, 0.3),
            thue_morse(n_cur, rng),
            make_block(n_cur, 6),
            make_block(n_cur, 10),
        ]
        for ds in diverse_starts:
            if remaining() < 20:
                break
            ds = project_feasible(ds, n_cur)
            h_ds = fista_progressive(ds, n_cur,
                                     betas=[100, 500, 2000, 8000, 30000],
                                     max_iter_per=1500,
                                     max_lags_list=[None, None, 150, 80, 40])
            c5_ds = compute_c5(h_ds)
            if c5_ds < best_c5:
                best_c5 = c5_ds
                best_h = h_ds.copy()
                best_n = n_cur

    # Phase 6: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [20000, 50000, 100000, 200000, 400000]:
        if remaining() < 5:
            break
        h_new = fista_optimize(h, n, beta, max_iter=3000, max_lags=30)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final coordinate descent
    if remaining() > 10:
        h_cd, c5_cd = coord_descent_polish(best_h, best_n, max_rounds=200)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final binary polish
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        n_h = best_n // 2
        n_o = int(np.sum(h_bin))
        if n_o > n_h:
            idx = np.where(h_bin == 1)[0]
            perm = rng.permutation(len(idx))
            h_bin[idx[perm[:n_o - n_h]]] = 0
        elif n_o < n_h:
            idx = np.where(h_bin == 0)[0]
            perm = rng.permutation(len(idx))
            h_bin[idx[perm[:n_h - n_o]]] = 1
        h_b, c5_b = binary_local_search(h_bin, best_n, rng,
                                        time_limit=min(40, int(remaining() * 0.4)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = fista_optimize(h_b, best_n, 50000, max_iter=2000, max_lags=40)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
