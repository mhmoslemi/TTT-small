        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def binarize(h, n):
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n//2]] = 1.0
    return h_bin

# Key: effective binary local search
def binary_local_search_v2(h_init, n, rng, time_limit=600):
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    
    # Ensure exactly n_half ones
    if n_ones > n_half:
        ones_idx = np.where(h == 1)[0]
        perm = rng.permutation(len(ones_idx))
        h[ones_idx[perm[:n_ones - n_half]]] = 0
    elif n_ones < n_half:
        zeros_idx = np.where(h == 0)[0]
        perm = rng.permutation(len(zeros_idx))
        h[zeros_idx[perm[:n_half - n_ones]]] = 1
    
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    
    start = time.time()
    T = 2.0  # initial temperature for SA
    cooling = 0.9997
    no_improve = 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() - start < time_limit and remaining() > 5:
        T *= cooling
        
        corr = compute_corr(h)
        max_val = np.max(corr)
        top_idx = np.argsort(-corr)[:8]
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0
        best_swap = None
        n_trials = 30
        
        # Strategy 1: Random swaps
        for _ in range(n_trials // 2):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0; h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1; h[j] = 0
            if delta < best_delta - 1e-12:
                best_delta = delta
                best_swap = (i, j)
        
        # Strategy 2: Swaps targeting top correlations
        for _ in range(n_trials // 2):
            k_t = top_idx[rng.randint(len(top_idx))]
            off = k_t - (n - 1)
            if off >= 0:
                # corr[off] = sum h[i]*(1-h[i+off])
                # Want to reduce: find i where h[i]=1, h[i+off]=0
                valid = np.arange(n - off)
                faults = valid[(h[valid] == 1) & (h[valid+off] == 0)]
                if len(faults) == 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                else:
                    fi = faults[rng.randint(len(faults))]
                    # Either flip fi (1→0) or flip fi+off (0→1)
                    if rng.rand() < 0.5:
                        i = fi
                        j = zeros[rng.randint(len(zeros))]
                    else:
                        i = ones[rng.randint(len(ones))]
                        j = fi + off
                    
            else:
                ao = -off
                valid = np.arange(n - ao)
                # corr[-ao]: h[i+ao]*(1-h[i])... actually for negative offset
                # In full mode, index k = n-1+(-ao) = n-1-ao
                # This corresponds to sum over h[i]*(1-h[i-ao]) for valid i
                # = sum over h[i+ao]*(1-h[i]) for i in [0, n-ao)
                faults = valid[(h[valid+ao] == 1) & (h[valid] == 0)]
                if len(faults) == 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                else:
                    fi = faults[rng.randint(len(faults))]
                    if rng.rand() < 0.5:
                        i = fi + ao
                        j = zeros[rng.randint(len(zeros))]
                    else:
                        i = ones[rng.randint(len(ones))]
                        j = fi
            
            if i == j or h[i] != 1 or h[j] != 0:
                continue
            h[i] = 0; h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1; h[j] = 0
            if delta < best_delta - 1e-12:
                best_delta = delta
                best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0; h[j] = 1
            cur_max = max_val + best_delta
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1
            # Simulated annealing: accept worse move with prob exp(-delta/T)
            if T > 0.01 and no_improve % 50 == 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0; h[j] = 1
                nc = float(np.max(compute_corr(h)))
                delta = nc - max_val
                h[i] = 1; h[j] = 0
                if delta < 0 or rng.rand() < np.exp(-max(0, delta) / max(T, 0.001)):
                    h[i] = 0; h[j] = 1
                    cur_max = nc
    
    return best_h, best_max * dx

# Convex-based continuous optimization
def convex_max_corr(h_init, n, time_limit=120):
    """Minimize max_k c_k(h) using iterative thresholding."""
    dx = 2.0 / n
    h = project_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start = time.time()
    t = best_c5 / dx  # current threshold (in correlation units)
    
    for outer in range(20):
        if time.time() - start > time_limit or remaining() < 5:
            break
        
        # For threshold t, minimize sum max(0, c_k - t)^2
        # Use projected gradient descent
        t_inner = t
        h_work = h.copy()
        step_size = 0.01
        
        for it in range(500):
            if time.time() - start > time_limit or remaining() < 5:
                break
            corr = compute_corr(h_work)
            excess = np.maximum(corr - t_inner, 0)
            
            if np.all(excess < 1e-10):
                break
            
            # Gradient of sum max(0, c_k - t)^2 w.r.t. h
            g = np.zeros(n)
            for k in range(2*n-1):
                if excess[k] > 0:
                    off = k - (n-1)
                    if off >= 0:
                        np_ = n - off
                        g[off:off+np_] += 2*excess[k] * (1 - h_work[:np_])
                        g[:np_] -= 2*excess[k] * h_work[off:off+np_]
                    else:
                        ao = -off
                        np_ = n - ao
                        g[:np_] += 2*excess[k] * (1 - h_work[ao:ao+np_])
                        g[ao:ao+np_] -= 2*excess[k] * h_work[:np_]
            
            # Add constraint gradient
            g += 0.1 * (np.mean(h_work) - 0.5)
            
            h_work = project_feasible(h_work - step_size * g, n)
            
            # Adaptive step
            c_new = compute_c5(h_work)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_work.copy()
            
            step_size = max(0.001, min(0.05, step_size * 0.999))
        
        t = max(0.001, t * 0.9)
        h = best_h.copy()
    
    return best_h, best_c5

# SLSQP smooth optimization (similar to before but refined)
_N_global = 0
_BETA_global = 1.0
_TGT_global = 0.0

def _sm_obj(x):
    n = _N_global
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA_global * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA_global) * dx_v

def _sm_grad(x):
    n = _N_global
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA_global * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2*n-1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n-1)
        if off >= 0:
            np_ = n - off
            g[off:off+np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off+np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao+np_])
            g[ao:ao+np_] -= wk * x[:np_]
    return g

def _sc(x):
    return np.sum(x) - _TGT_global

def _scj(x):
    return np.ones(_N_global)

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N_global, _BETA_global, _TGT_global
    _N_global = n
    _TGT_global = n / 2.0
    _BETA_global = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_focus(h, n, top_k=5, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx_v = 2.0 / n
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-9:
            break
        k_top = idx[0]
        k_bot = idx[-1]
        off_t = k_top - (n-1)
        off_b = k_bot - (n-1)
        
        g_t = np.zeros(n)
        if off_t >= 0:
            np_ = n - off_t
            g_t[off_t:off_t+np_] += (1.0 - h[:np_])
            g_t[:np_] -= h[off_t:off_t+np_]
        else:
            ao = -off_t
            np_ = n - ao
            g_t[:np_] += (1.0 - h[ao:ao+np_])
            g_t[ao:ao+np_] -= h[:np_]
        
        g_b = np.zeros(n)
        if off_b >= 0:
            np_ = n - off_b
            g_b[off_b:off_b+np_] += (1.0 - h[:np_])
            g_b[:np_] -= h[off_b:off_b+np_]
        else:
            ao = -off_b
            np_ = n - ao
            g_b[:np_] += (1.0 - h[ao:ao+np_])
            g_b[ao:ao+np_] -= h[:np_]
        
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.5, 0.2, 0.015]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
