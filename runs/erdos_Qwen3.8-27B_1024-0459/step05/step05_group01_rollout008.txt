        if len(idx) > 0:
            per = diff / len(idx)
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h


def corr_grad_weighted(h, corr, weights, n):
    """Gradient of weighted sum of correlations."""
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if abs(wk) < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            grad[:np_] += wk * (1.0 - h[off:])
            grad[off:] -= wk * h[:np_]
        else:
            a = -off
            np_ = n - a
            grad[a:] += wk * (1.0 - h[:np_])
            grad[:np_] -= wk * h[a:]
    return grad


def smooth_max_opt(h, n, beta, n_steps=2000, ftol=1e-16):
    """Projected gradient on smooth max with backtracking."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_val = compute_c5(h)
    h_best = h.copy()
    
    for step in range(n_steps):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        shift = corr - np.max(corr)
        z = beta * shift
        z = np.clip(z, -500, 500)
        ez = np.exp(z)
        S = np.sum(ez)
        if S < 1e-300:
            break
        p = ez / S
        
        obj_val = (np.max(corr) + np.log(S) / beta) * dx
        if obj_val < best_val - ftol:
            best_val = obj_val
            h_best = h.copy()
        
        grad = corr_grad_weighted(h, corr, p, n) * dx
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        d = -grad / gn
        # Backtracking line search
        alpha = 1.0
        accepted = False
        for _ in range(30):
            alpha *= 0.55
            h_trial = project_feasible(h + alpha * d, n)
            corr_t = compute_corr(h_trial)
            shift_t = corr_t - np.max(corr_t)
            z_t = np.clip(beta * shift_t, -500, 500)
            ez_t = np.exp(z_t)
            S_t = np.sum(ez_t)
            obj_t = (np.max(corr_t) + np.log(S_t) / beta) * dx
            if obj_t < obj_val - 1e-16 * alpha:
                h = h_trial
                accepted = True
                break
        if not accepted:
            break
    
    # Return best found
    c5_best = compute_c5(h_best)
    if c5_best < compute_c5(h):
        return h_best, c5_best
    return h, compute_c5(h)


def topk_directed(h, n, top_k=8, max_iters=1000):
    """Directed descent on top-k correlations with careful step control."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr[:top_k*3])[:top_k]
        c_max = corr[idx_top[0]]
        c_range = c_max - corr[idx_top[-1]]
        if c_range < 1e-11:
            break
        
        # Direction: reduce top correlation, increase others in top-k
        direction = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                direction[:np_] += w_k * (1.0 - h[off:])
                direction[off:] -= w_k * h[:np_]
            else:
                a = -off
                np_ = n - a
                direction[a:] += w_k * (1.0 - h[:np_])
                direction[:np_] -= w_k * h[a:]
        
        gn = np.linalg.norm(direction)
        if gn < 1e-14:
            break
        d = direction / gn
        
        improved = False
        steps = [0.003, 0.007, 0.015, 0.03, 0.06, 0.012, 0.025, 0.05, 0.1, 0.005]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.001, 0.002, 0.0005]:
                ht = project_feasible(h + step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def binary_local_search(h, n, time_limit=80, rng=None):
    """Binary local search with single and paired swaps."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    
    dx = 2.0 / n
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    while time.time() - start < time_limit and remaining_time() > 5:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Strategy: try random single swaps, and occasionally paired swaps
        n_tries_single = 200
        n_tries_paired = 30 if n <= 500 else 15
        
        # Single swaps
        best_delta = 1e-12
        best_move = None
        for _ in range(n_tries_single):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            delta = c_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_move = ('swap', i, j)
        
        # Paired swaps (swap two 1s with two 0s)
        if n > 256:
            for _ in range(n_tries_paired):
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                a, b = ones[i1], ones[i2]
                c, d = zeros[j1], zeros[j2]
                h[a] = 0.0; h[b] = 0.0; h[c] = 1.0; h[d] = 1.0
                c_new = compute_c5(h)
                h[a] = 1.0; h[b] = 1.0; h[c] = 0.0; h[d] = 0.0
                delta = c_new - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = ('pswap', a, b, c, d)
        
        if best_move is not None:
            if best_move[0] == 'swap':
                h[best_move[1]] = 0.0
                h[best_move[2]] = 1.0
            else:
                h[best_move[1]] = 0.0
                h[best_move[2]] = 0.0
                h[best_move[3]] = 1.0
                h[best_move[4]] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
            improved = True
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break
            # Random kick
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
    
    return best_h, best_c5


def msequence(n_target):
    """Generate an m-sequence of target length (pad/truncate as needed)."""
    # Find m such that 2^m - 1 is close to n_target
    m = max(4, int(np.log2(n_target)) + 1)
    while (1 << m) - 1 < 128:
        m += 1
    
    # Generate m-sequence using LFSR (primitive polynomial)
    # Use polynomial x^m - x - 1 (works for many m)
    taps = [0, m - 1]  # taps at positions 0 and m-1
    state = 1
    seq = []
    for _ in range(2**m - 1):
        seq.append(state & 1)
        feedback = state & 1
        state >>= 1
        if (seq[-1] ^ (state >> (m-2) & 1) if m > 1 else seq[-1]) != feedback:
            pass
        # Simple LFSR
        if m <= 32:
            new_bit = ((state >> (m-2)) ^ (state >> (m-1))) & 1 if m > 1 else 1
        else:
            new_bit = 1
        # Actually let's use a known primitive polynomial
        # For simplicity, use x^m + x^(m-3) + 1 for small m
        break
    
    # Simpler approach: use bit-reversal or known sequence
    h = np.zeros(n_target)
    # Use a simpler construction: alternate with perturbation
    for i in range(n_target):
        # Thue-Morse as fallback
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    h = project_feasible(h, n_target)
    return h


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def golay_complementary(n):
    """Golay-like complementary sequence construction."""
    # Start with a base pattern and replicate
    base = np.array([1, 1, 1, -1, 1, 1, -1, -1], dtype=float)
    h = np.zeros(n)
    for i in range(n):
        h[i] = 0.5 + 0.5 * base[i % len(base)]
    return project_feasible(h, n)


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def slsqp_round(h, n, beta, maxiter=400):
    """One round of SLSQP with smooth max."""
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        z = np.clip(_SM_BETA * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        return (m + np.log(np.sum(ez)) / _SM_BETA) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        z = np.clip(_SM_BETA * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        p = ez / S
        g = corr_grad_weighted(x, corr, p, n)
        return g * (2.0 / n)
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or res.fun < obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


_CUR_N = 0
_SM_BETA = 0.0


def coordinate_descent(h, n, max_sweeps=20):
    """Per-coordinate optimization of the max correlation."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        for i in range(n):
            old = h[i]
            # Try: 0, 1, 0.5, old±0.25, old±0.1
            cands = [0.0, 1.0, 0.5, max(0, old - 0.25), min(1, old + 0.25),
                     max(0, old - 0.1), min(1, old + 0.1)]
            cands = [c for c in cands if 0.0 <= c <= 1.0 and abs(c - old) > 1e-6]
            best_c = old
            best_cv = compute_c5(h)
            for c in cands:
                h[i] = c
                h_p = project_feasible(h, n)
                cv = compute_c5(h_p)
                if cv < best_cv - 1e-12:
                    best_cv = cv
                    best_c = c
            if best_c != old:
                h[i] = best_c
                h = project_feasible(h, n)
                improved = True
                if compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = thue_morse(384)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # ---- Phase 1: Quick optimization at parent resolution ----
    h = best_h.copy()
    
    # Smooth max gradient descent (fast, no SLSQP overhead)
    h, c1 = smooth_max_opt(h, best_n, beta=100, n_steps=800)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h.copy()
    
    h, c1 = smooth_max_opt(h, best_n, beta=1000, n_steps=1500)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h.copy()
    
    # SLSQP polish
    for beta in [5000, 50000, 300000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_round(best_h, best_n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Top-k directed descent
    if remaining_time() > 15:
        h_eq, c5_eq = topk_directed(best_h, best_n, top_k=8, max_iters=800)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    h, c1 = smooth_max_opt(best_h, best_n, beta=10000, n_steps=2000)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h.copy()
    
    # ---- Phase 2: Multiple starting points at parent resolution ----
    starts = []
    starts.append(thue_morse(best_n))
    
    # Random balanced
    h_rand = np.zeros(best_n)
    h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
    starts.append(h_rand)
    
    # Shifted best
    starts.append(project_feasible(np.roll(best_h, 2), best_n))
    
    # Inverted best
    starts.append(project_feasible(1.0 - best_h, best_n))
    
    for h_s in starts:
        if remaining_time() < 40:
            break
        h_opt = h_s.copy()
        h_opt, c = smooth_max_opt(h_opt, best_n, beta=200, n_steps=600)
        h_opt = slsqp_round(h_opt, best_n, 10000, maxiter=400)
        c = compute_c5(h_opt)
        h_opt, c = topk_directed(h_opt, best_n, top_k=6, max_iters=400)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = h_opt.copy()
    
    # ---- Phase 3: Binary local search ----
    if remaining_time() > 50:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        if abs(int(np.sum(h_bin)) - n_half) > 0:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0
        
        sa_time = min(100, max(20, int(remaining_time() * 0.2)))
        h_b, c5_b = binary_local_search(h_bin, best_n, time_limit=sa_time, rng=rng)
        
        # Refine binary result continuously
        h_b_c = h_b.copy()
        for beta in [500, 5000, 50000]:
            if remaining_time() < 10:
                break
            h_b_c, c = smooth_max_opt(h_b_c, best_n, beta=beta, n_steps=800)
        h_b_c = slsqp_round(h_b_c, best_n, 100000, maxiter=400)
        c_b_c = compute_c5(h_b_c)
        h_b_c, c_b_c = topk_directed(h_b_c, best_n, top_k=8, max_iters=500)
        
        if c_b_c < best_c5 - 1e-14:
            best_c5 = c_b_c
            best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # ---- Phase 4: Multi-resolution search ----
    # Try resolutions around and above parent
    res_list = sorted(set([
        best_n - 64, best_n - 32, best_n, best_n + 32, best_n + 64,
        best_n + 128, best_n + 256,
        max(128, best_n // 2),
    ]))
    # Add multiples
    for mult in [1.15, 1.3, 1.5, 1.75, 2.0, 2.5]:
        nn = int(best_n * mult)
        if nn % 2 == 0 and 192 <= nn <= 1200:
            res_list.append(nn)
    res_list = sorted(set(r for r in res_list if r >= 192 and r <= 1200 and r % 2 == 0))
    res_list.sort(key=lambda x: abs(x - best_n))
    
    for n_t in res_list[:6]:
        if remaining_time() < 30:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        
        h_opt = h_feas
        for beta in [100, 1000]:
            if remaining_time() < 10:
                break
            h_opt, _ = smooth_max_opt(h_opt, n_t, beta=beta, n_steps=800)
        if remaining_time() > 15:
            h_opt = slsqp_round(h_opt, n_t, 20000, maxiter=400)
        if remaining_time() > 10:
            h_opt, c = topk_directed(h_opt, n_t, top_k=6, max_iters=400)
        else:
            c = compute_c5(h_opt)
        
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = h_opt.copy()
            best_n = n_t
    
    # ---- Phase 5: Intensive refinement at best resolution ----
    n = best_n
    h = best_h.copy()
    
    # Progressive smooth max
    for beta in [500, 5000, 50000, 200000, 1000000]:
        if remaining_time() < 8:
            break
        h, c = smooth_max_opt(h, n, beta=beta, n_steps=min(2000, int(remaining_time() * 20)))
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    
    # Top-k with larger k
    if remaining_time() > 10:
        h_eq, c5_eq = topk_directed(best_h, n, top_k=10, max_iters=600)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # SLSQP final
    for beta in [100000, 500000, 2000000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_round(best_h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Coordinate descent on a subset (full n might be too slow)
    if remaining_time() > 15 and n <= 500:
        h_cr, c5_cr = coordinate_descent(best_h, n, max_sweeps=10)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()
    
    # Final smooth max push
    if remaining_time() > 5:
        h, c = smooth_max_opt(best_h, n, beta=500000, n_steps=int(remaining_time() * 15))
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    
    # Final top-k
    if remaining_time() > 5:
        h_eq, c5_eq = topk_directed(best_h, n, top_k=12, max_iters=300)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, n)
