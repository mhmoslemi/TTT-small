    return smooth_max_value_grad(x, _N_OPT, _BETA_OPT)[1]

def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _con_jac(x):
    return np.ones(_N_OPT)

def lbfgs_lagrangian(h, n, beta, n_outer=15, maxiter_inner=300):
    global _N_OPT, _BETA_OPT, _MU_OPT
    _N_OPT = n
    _BETA_OPT = beta
    mu = 0.0
    current = h.copy()
    for outer in range(n_outer):
        if remaining_time() < 5:
            break
        _MU_OPT = mu
        try:
            res = minimize(_pen_obj, current, jac=_pen_grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter_inner, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                current = res.x
        except Exception:
            break
        s = np.sum(current) - n / 2.0
        if abs(s) < 1e-12:
            break
        mu -= 0.5 * s
        mu = np.clip(mu, -10.0, 10.0)
    return project_feasible(current, n)

def slsqp_optimize(h, n, beta, maxiter=300):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    c5_before = compute_c5(h)
    try:
        res = minimize(_eq_obj, h, jac=_eq_grad, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def fast_coord_descent(h, n, n_sweeps=50, seed=42):
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_corr = compute_corr(best)
    best_c5 = np.max(best_corr) * (2.0 / n)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(n_sweeps):
        if remaining_time() < 3:
            break
        improved_any = False
        order = rng.permutation(n)

        for idx in order:
            i = int(idx)
            t0 = best[i]
            if t0 < 1e-15 or t0 > 1.0 - 1e-15:
                # Still try to move boundary points
                pass

            # Compute coefficients a[k], b[k] for parsimonious evaluation
            # corr[k] as function of t_i: base + a[k]*t + b[k]*t^2
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for d in range(-(n - 1), n):
                k = n - 1 + d
                if d == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + d < n:
                        a[k] += (1.0 - best[i + d])
                    if 0 <= i - d < n:
                        a[k] += (-best[i - d])
            base = best_corr - a * t0 - b * t0 * t0

            # Evaluate at strategic candidates
            cands = np.array([0.0, 1.0, t0,
                             max(0.0, t0 - 0.2), min(1.0, t0 + 0.2),
                             max(0.0, t0 - 0.05), min(1.0, t0 + 0.05),
                             1.0 - t0])
            cands = cands[(cands >= -1e-15) & (cands <= 1.0 + 1e-15)]
            cands = np.clip(np.unique(cands), 0.0, 1.0)

            best_t = t0
            best_f = best_c5
            for t in cands:
                if abs(t - t0) < 1e-13:
                    continue
                f = np.max(base + a * t + b * t * t) * dx
                if f < best_f - 1e-15:
                    best_f = f
                    best_t = t

            # Golden section refinement
            if best_t != t0:
                margin = 0.1
                lo = max(0.0, best_t - margin)
                hi = min(1.0, best_t + margin)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                for _ in range(50):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                t_opt = (lo + hi) / 2.0
                f_opt = np.max(base + a * t_opt + b * t_opt * t_opt) * dx
                if f_opt < best_f:
                    best_f = f_opt
                    best_t = t_opt

            if best_f < best_c5 - 1e-14:
                t1 = best_t
                # Fast correlation update (O(n) instead of O(n^2))
                delta = t1 - t0
                best[i] = t1
                # Update corr for all lags
                for d in range(-(n - 1), n):
                    k = n - 1 + d
                    if d == 0:
                        best_corr[k] += t1 * (1.0 - t1) - t0 * (1.0 - t0)
                    else:
                        dc = 0.0
                        if 0 <= i + d < n:
                            dc += delta * (1.0 - best[i + d])
                        if 0 <= i - d < n:
                            dc += -delta * best[i - d]
                        best_corr[k] += dc
                best_c5 = np.max(best_corr) * dx
                improved_any = True

        if not improved_any:
            break

    # Verify with full correlation
    final_corr = compute_corr(best)
    final_c5 = np.max(final_corr) * dx
    return best, final_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def try_interval_structure(n, k_intervals, seed=42):
    """Try solutions that are union of k_intervals intervals."""
    rng = np.random.RandomState(seed)
    # Each interval: (start, length), total measure = 1
    best_h = None
    best_c5 = float('inf')
    for trial in range(20):
        if remaining_time() < 5:
            break
        if k_intervals == 1:
            starts = rng.uniform(0, 1, 1)
            lengths = np.array([1.0])
        elif k_intervals == 2:
            lengths = rng.dirichlet([1, 1])
            starts = rng.uniform(0, 2 - lengths, k_intervals)
            # Sort and check non-overlap
            order = np.argsort(starts)
            starts = starts[order]
            lengths = lengths[order]
            if starts[0] + lengths[0] > starts[1] + 0.01:
                continue
        else:
            lengths = rng.dirichlet(np.ones(k_intervals))
            starts = sorted(rng.uniform(0, 2, k_intervals))
            # Check non-overlap
            valid = True
            for i in range(k_intervals - 1):
                if starts[i] + lengths[i] > starts[i + 1] + 0.01:
                    valid = False
                    break
            if not valid:
                continue
            # Ensure within [0,2]
            for i in range(k_intervals):
                if starts[i] + lengths[i] > 2.0:
                    valid = False
                    break
            if not valid:
                continue

        # Build h
        h = np.zeros(n)
        for i in range(k_intervals):
            s = starts[i]
            l = lengths[i]
            idx_s = int(s // (2.0 / n))
            idx_e = int((s + l) // (2.0 / n))
            h[idx_s:idx_e] = 1.0
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Fast coordinate descent on parent resolution
    if remaining_time() > 60:
        h_new, c5_new = fast_coord_descent(h, n, n_sweeps=60, seed=seed)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Lagrangian L-BFGS-B with increasing beta
    for beta in [200, 1000, 5000, 20000]:
        if remaining_time() < 30:
            break
        h_new = lbfgs_lagrangian(h, n, beta, n_outer=10, maxiter_inner=250)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 3: SLSQP with higher beta
    for beta in [8000, 30000, 100000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_optimize(h, n, beta, 300)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 4: More coordinate descent
    if remaining_time() > 40:
        h_new, c5_new = fast_coord_descent(h, n, n_sweeps=80, seed=seed + 1)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Multi-resolution upsampling
    for rn in [int(best_n * 1.5), int(best_n * 2), int(best_n * 2.5), int(best_n * 3)]:
        if rn % 2 != 0:
            rn += 1
        if rn > 5000 or remaining_time() < 70:
            continue
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)

        for beta in [500, 3000, 15000]:
            if remaining_time() < 20:
                break
            h_r = lbfgs_lagrangian(h_r, rn, beta, n_outer=8, maxiter_inner=200)

        if remaining_time() > 30:
            h_r, c5_r_cd = fast_coord_descent(h_r, rn, n_sweeps=30, seed=seed + 10)
            c5_r = c5_r_cd
        else:
            c5_r = compute_c5(h_r)

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 6: Random perturbations + re-optimization
    n = best_n
    h = best_h.copy()
    for pert in range(50):
        if remaining_time() < 15:
            break
        n_pert = max(3, n // 30)
        h_pert = h.copy()
        idx = rng.choice(n, min(n_pert, n), replace=False)
        h_pert[idx] += rng.uniform(-0.1, 0.1, len(idx))
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.004:
            h_new = lbfgs_lagrangian(h_pert, n, 10000, n_outer=8, maxiter_inner=200)
            h_new, c5_new = fast_coord_descent(h_new, n, n_sweeps=20, seed=seed + 100 + pert)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 7: Try different structural initializations
    for k_int in [2, 3, 4]:
        if remaining_time() < 30:
            break
        h_struct, c5_struct = try_interval_structure(best_n, k_int, seed=seed + k_int)
        if h_struct is not None and c5_struct < best_c5 + 0.02:
            h_refined, c5_refined = fast_coord_descent(h_struct, best_n, n_sweeps=30, seed=seed + 200)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()

    # Phase 8: Final intensive coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = fast_coord_descent(best_h, best_n, n_sweeps=100, seed=seed + 500)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 12:
        h_new = slsqp_optimize(best_h, best_n, 200000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One more coordinate descent pass
    if remaining_time() > 8:
        h_new, c5_new = fast_coord_descent(best_h, best_n, n_sweeps=30, seed=seed + 999)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation and projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure exact sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
