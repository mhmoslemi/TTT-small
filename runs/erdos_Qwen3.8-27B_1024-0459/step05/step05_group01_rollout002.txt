```python
def swap_delta(h, i, j, n):
    """Compute change in correlation vector when swapping h[i]:1->0, h[j]:0->1"""
    delta = np.zeros(2*n-1)
    # i changes 1->0:
    # Term as h[n+k]: n = i-k, valid if 0<=i-k<n => k in [i-n+1, i]
    #   delta[k + n-1] += -(1-h[i-k])
    # Term in (1-h[n]): n=i, valid if 0<=i+k<n => k in [-i, n-1-i]
    #   delta[k + n-1] += h[i+k]
    # j changes 0->1:
    # Term as h[n+k]: n = j-k, valid if 0<=j-k<n => k in [j-n+1, j]
    #   delta[k + n-1] += (1-h[j-k])
    # Term in (1-h[n]): n=j, valid if 0<=j+k<n => k in [-j, n-1-j]
    #   delta[k + n-1] += -h[j+k]
    
    # For i (1->0):
    # As first factor h[n+k]: k ranges from i-(n-1) to i, but also k from -(n-1) to n-1
    k_lo = max(i-(n-1), -(n-1))
    k_hi = min(i, n-1)
    ks = np.arange(k_lo, k_hi+1)
    ns = i - ks  # n = i-k
    delta[ks + n-1] += -(1.0 - h[ns])
    
    k_lo2 = max(-i, -(n-1))
    k_hi2 = min(n-1-i, n-1)
    ks2 = np.arange(k_lo2, k_hi2+1)
    ns2 = i + ks2  # n+k = i+k
    delta[ks2 + n-1] += h[ns2]
    
    # For j (0->1):
    k_lo3 = max(j-(n-1), -(n-1))
    k_hi3 = min(j, n-1)
    ks3 = np.arange(k_lo3, k_hi3+1)
    ns3 = j - ks3
    delta[ks3 + n-1] += (1.0 - h[ns3])
    
    k_lo4 = max(-j, -(n-1))
    k_hi4 = min(n-1-j, n-1)
    ks4 = np.arange(k_lo4, k_hi4+1)
    ns4 = j + ks4
    delta[ks4 + n-1] += -h[ns4]
    
    return delta
```

```python
    # i: 1->0

        ks = np.arange(k_lo, k_hi + 1)
        ns = i - ks

    k_hi2 = min(n-1-i, n-1)

        ns2 = i + ks2
```
