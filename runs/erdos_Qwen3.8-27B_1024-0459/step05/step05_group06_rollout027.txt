- f is convex, so f' is monotonically increasing (in the sense of the subgradient)

- The "left derivative" at t is: min{a_k : c_k + a_k t = f(t)} (the minimum slope among active functions)

def compute_c5(h):

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13 and n > 0:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(free))

            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        wsum = 1.0
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset

            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def coord_descent_fast(h, n, n_sweeps=100, tol=1e-15):
    """Fast coordinate descent using exact 1D minimization of convex piecewise-linear f(t)."""
    best = h.copy()

        max_improvement = 0.0

        for i in range(n):
            if remaining_time() < 1.0:
                break
            t0 = best[i]

            # Compute coefficient vector a[k] such that:
            # new_corr[k] = best_corr[k] + (t - t0) * a[k]
            a = np.zeros(ncorr)
            for offset in range(-(n - 1), n):
                k = n - 1 + offset
                coef = 0.0
                # h[i] appears as h[j+offset] when j = i - offset
                if 0 <= i - offset < n:
                    coef += (1.0 - best[i - offset])
                # h[i] appears in (1-h[j]) when j = i, so h[i+offset] * (1-h[i])
                if 0 <= i + offset < n:
                    coef -= best[i + offset]
                a[k] = coef

            # f(t) = max_k (best_corr[k] + a[k]*(t - t0)) = max_k (c[k] + a[k]*t)
            c = best_corr - a * t0

            lo, hi = 0.0, 1.0

            c2 = lo + gr * (hi - lo)
            f1 = np.max(c + a * c1)

                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = np.max(c + a * c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = np.max(c + a * c2)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgrad_step(h, n, n_active=8):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    active = np.where(corr >= max_corr - 1.0)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return None
    return grad / gnorm

            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-16:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                break
        if not improved and it % 20 == 10:
            # Interleave with coordinate descent
            if remaining_time() > 30:
                h_new, c5_new = coord_descent_fast(h, n, n_sweeps=20)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 7: Random restarts / perturbations
    n = best_n
    h = best_h.copy()
    for pert in range(20):
        if remaining_time() < 30:
            break
        n_pert = max(5, n // 40)
        h_pert = h.copy()
        idx = rng.choice(n, n_pert, replace=False)
        h_pert[idx] += rng.uniform(-0.2, 0.2, n_pert)
        h_pert = project_feasible(h_pert, n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.002:
            h_new, c5_new = coord_descent_fast(h_pert, n, n_sweeps=30)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 8: Final intensive coordinate descent
    if remaining_time() > 20:
        h_new, c5_new = coord_descent_fast(best_h, best_n, n_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP at very high beta
    if remaining_time() > 15:
        h_new = slsqp_optimize(best_h, best_n, 500000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One more coordinate descent pass
    if remaining_time() > 10:
        h_new, c5_new = coord_descent_fast(best_h, best_n, n_sweeps=30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation and constraint enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Extra precision: one more tiny coordinate descent
    if remaining_time() > 5 and final_c5 > best_c5:
        final_c5 = best_c5
        best_h = best_h.copy()

    # Try evaluate_erdos_solution for validation
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
