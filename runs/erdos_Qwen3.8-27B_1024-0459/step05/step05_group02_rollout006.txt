        if len(active) > 12:
            active = np.argsort(-corr)[:12]

        if len(active) < 2:
            break

        c_vals = corr[active]
        c_mean = np.mean(c_vals)
        c_spread = c_vals.max() - c_vals.min()
        if c_spread < 1e-10:
            # All active lags are equal; try to reduce them
            g = np.zeros(n)
            for ki in active[:4]:
                g += corr_grad_single(h, ki, n)
            g *= dx
            g -= np.mean(g)
            g[(h >= 1.0 - 1e-10)] = 0.0
            g[(h <= 1e-10)] = 0.0
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g /= gn
            h = proj(h - 0.005 * g, n)
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
            else:
                break
            continue

        # Build equalization direction: push down high, push up low (relative)
        g_eq = np.zeros(n)
        for ki in active:
            ci = corr[ki]
            weight = (ci - c_mean) / c_spread  # positive for above-mean
            g_eq += weight * corr_grad_single(h, ki, n)

        g_eq *= dx
        g_eq -= np.mean(g_eq)
        g_eq[(h >= 1.0 - 1e-10)] = 0.0
        g_eq[(h <= 1e-10)] = 0.0
        gnorm = np.linalg.norm(g_eq)
        if gnorm < 1e-14:
            break
        g_eq /= gnorm

        # Line search
        improved = False
        for alpha in [0.08, 0.04, 0.02, 0.01, 0.005, 0.06, 0.03]:
            h_try = proj(h + alpha * g_eq, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            # Try reducing max directly
            k_max = int(np.argmax(corr))
            g_max = corr_grad_single(h, k_max, n)
            g_max *= dx
            g_max -= np.mean(g_max)
            g_max[(h >= 1.0 - 1e-10)] = 0.0
            g_max[(h <= 1e-10)] = 0.0
            gnorm = np.linalg.norm(g_max)
            if gnorm < 1e-14:
                break
            g_max /= gnorm
            for alpha in [0.04, 0.02, 0.01, 0.005]:
                h_try = proj(h - alpha * g_max, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if not improved:
                break

    return h, best_c5


def binary_local_search(h, n, max_iters=5000, top_k=4):
    """Enhanced binary local search with wider neighborhood."""
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    rng = np.random.RandomState(12345)

    for it in range(max_iters):
        if remaining() < 4:
            break

        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]
        cur_c5 = float(np.max(corr)) * dx

        # Compute gradient for weighting
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Score candidates
        score_one = -g[ones]
        score_zero = g[zeros]

        found_improvement = False
        best_delta = 0.0
        best_swap = None

        # Try top candidates systematically
        n_try_o = min(16, len(ones))
        n_try_z = min(16, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try_o]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                nc = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        # Random trials to escape
        if best_delta >= 0:
            n_rand = min(10, len(ones) * len(zeros))
            for _ in range(n_rand):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                nc = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is None or best_delta >= 0:
            # Stall: try larger perturbation
            if it % 200 == 199 and len(ones) > 4 and len(zeros) > 4:
                # Swap 2 pairs
                idx_o = rng.choice(len(ones), size=2, replace=False)
                idx_z = rng.choice(len(zeros), size=2, replace=False)
                i1, i2 = int(ones[idx_o[0]]), int(ones[idx_o[1]])
                j1, j2 = int(zeros[idx_z[0]]), int(zeros[idx_z[1]])
                h_bin[i1] = 0.0; h_bin[i2] = 0.0
                h_bin[j1] = 1.0; h_bin[j2] = 1.0
                nc = compute_c5(h_bin)
                if nc > cur_c5 + 0.001:
                    h_bin[i1] = 1.0; h_bin[i2] = 1.0
                    h_bin[j1] = 0.0; h_bin[j2] = 0.0
                    break
                else:
                    cur_c5 = nc
                    if nc < best_c5:
                        best_c5 = nc
                        best_h = h_bin.copy()
                    continue
            else:
                break

        i, j = best_swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        cur_c5 = compute_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()
        found_improvement = True

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def spectral_refine(h, n, n_modes=16, max_iters=300):
    """Refine in a low-dimensional Fourier subspace."""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Fourier basis (real)
    x = np.arange(n)
    cos_basis = np.array([np.cos(2.0 * np.pi * k * x / n) for k in range(1, n_modes + 1)])
    sin_basis = np.array([np.sin(2.0 * np.pi * k * x / n) for k in range(1, n_modes + 1)])

    # Project current h onto the subspace (centered)
    h_centered = h - np.mean(h)
    coeffs_c = np.array([np.dot(h_centered, cos_basis[k]) / n for k in range(n_modes)])
    coeffs_s = np.array([np.dot(h_centered, sin_basis[k]) / n for k in range(n_modes)])

    def eval_coeffs(coeffs):
        nonlocal_h = np.mean(h)
        for k in range(n_modes):
            nonlocal_h += coeffs[2*k] * cos_basis[k] / n + coeffs[2*k+1] * sin_basis[k] / n
        return nonlocal_h

    def fun_and_grad(coeffs):
        nonlocal_h = proj(np.mean(h) + np.zeros(n), n)
        for k in range(n_modes):
            nonlocal_h += coeffs[2*k] * cos_basis[k] + coeffs[2*k+1] * sin_basis[k]
        nonlocal_h = proj(nonlocal_h, n)
        c5 = compute_c5(nonlocal_h)
        # Finite difference gradient
        dim = 2 * n_modes
        grad = np.zeros(dim)
        eps = 0.001
        for i in range(dim):
            cp = coeffs.copy()
            cm = coeffs.copy()
            cp[i] += eps
            cm[i] -= eps
            hp = np.mean(h) + np.zeros(n)
            hm = np.mean(h) + np.zeros(n)
            for k in range(n_modes):
                hp += cp[2*k] * cos_basis[k] + cp[2*k+1] * sin_basis[k]
                hm += cm[2*k] * cos_basis[k] + cm[2*k+1] * sin_basis[k]
            hp = proj(hp, n)
            hm = proj(hm, n)
            grad[i] = (compute_c5(hp) - compute_c5(hm)) / (2 * eps)
        return c5, grad

    # Simple gradient descent in coefficient space
    coeffs = np.concatenate([coeffs_c, coeffs_s])
    step = 0.5
    for it in range(max_iters):
        if remaining() < 5:
            break
        val, grad = fun_and_grad(coeffs)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        coeffs -= step * grad / gnorm
        # Check
        new_h = np.mean(h) + np.zeros(n)
        for k in range(n_modes):
            new_h += coeffs[2*k] * cos_basis[k] + coeffs[2*k+1] * sin_basis[k]
        new_h = proj(new_h, n)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        else:
            step *= 0.8
        step = min(step * 1.005, 2.0)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = proj(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: L-BFGS with escalating beta at parent resolution
    for beta in [200, 500, 1500, 4000, 10000, 30000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=6, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Subgradient with Nesterov
    if remaining() > 30:
        h_sg, c5_sg = subgradient_descent(h, n, max_iters=6000, top_k=5, step_init=0.015)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: Lag equalization
    if remaining() > 20:
        h_eq, c5_eq = equalize_lags(h, n, max_iters=600)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: More L-BFGS
    for beta in [20000, 50000, 100000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=4, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Binary local search
    if remaining() > 40:
        h_bin_init = (best_h > 0.5).astype(float)
        h_bl, c5_bl = binary_local_search(h_bin_init, best_n, max_iters=5000, top_k=4)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
        # Refine binary solution continuously
        h_ref = h_bl.copy()
        for beta in [10000, 30000]:
            if remaining() < 5:
                break
            h_ref = lbfgs_pass(h_ref, best_n, beta, top_k=4, maxiter=300)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 6: Spectral refinement
    if remaining() > 30:
        h_spec, c5_spec = spectral_refine(best_h, best_n, n_modes=12, max_iters=200)
        if c5_spec < best_c5:
            best_c5 = c5_spec
            best_h = h_spec.copy()

    # Phase 7: Multi-resolution search
    n_candidates = sorted(set([
        best_n - 4, best_n - 2, best_n + 2, best_n + 4, best_n + 6,
        best_n + 8, best_n + 12, best_n - 6, best_n - 8,
        int(best_n * 1.1), int(best_n * 1.3), int(best_n * 0.9),
    ]))
    n_candidates = [nn for nn in n_candidates if 160 <= nn <= 800 and nn % 2 == 0]
    n_candidates = [nn for nn in n_candidates if nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:6]:
        if remaining() < 30:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        for beta in [1000, 5000, 20000]:
            if remaining() < 10:
                break
            h_feas = lbfgs_pass(h_feas, n_t, beta, top_k=6, maxiter=350)

        h_feas, c5_feas = equalize_lags(h_feas, n_t, max_iters=300)

        h_b2 = (h_feas > 0.5).astype(float)
        h_bl2, c5_bl2 = binary_local_search(h_b2, n_t, max_iters=3000, top_k=4)
        h_ref2 = h_bl2.copy()
        for beta in [10000, 30000]:
            if remaining() < 5:
                break
            h_ref2 = lbfgs_pass(h_ref2, n_t, beta, top_k=4, maxiter=250)
        c5_ref2 = compute_c5(h_ref2)

        c5_candidate = min(c5_feas, c5_bl2, c5_ref2)
        if c5_ref2 <= c5_candidate:
            h_candidate = h_ref2
        elif c5_bl2 <= c5_candidate:
            h_candidate = h_bl2
        else:
            h_candidate = h_feas

        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    if remaining() > 20:
        h_pg2, c5_pg2 = subgradient_descent(h, n, max_iters=4000, top_k=3, step_init=0.01)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()
            h = h_pg2.copy()

    if remaining() > 15:
        h_eq2, c5_eq2 = equalize_lags(h, n, max_iters=400)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    for beta in [50000, 150000]:
        if remaining() < 6:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=3, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_local_search(h_bin_f, best_n, max_iters=3000, top_k=3)
        h_rf = h_bf.copy()
        for beta in [30000, 100000]:
            if remaining() < 5:
                break
            h_rf = lbfgs_pass(h_rf, best_n, beta, top_k=3, maxiter=300)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()
        elif c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Final equalize
    if remaining() > 10:
        h_eqf, c5_eqf = equalize_lags(best_h, best_n, max_iters=200)
        if c5_eqf < best_c5:
            best_c5 = c5_eqf
            best_h = h_eqf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
