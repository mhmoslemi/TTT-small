import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_single(h, k):
    n = len(h)
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[off:off + np_]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:a + np_])
        g[a:a + np_] -= h[:np_]
    return g


def smooth_max_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_max_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    w = ex / np.sum(ex)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        g += w[k] * dx * corr_grad_single(h, k)
    return g


def slsqp_optimize(h, n, beta, maxiter=400):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj(x, beta)

    def grad(x):
        return smooth_max_grad(x, beta)

    def constr(x):
        return np.sum(x) - n / 2.0

    def constr_jac(x):
        return np.ones(n)

    obj0 = obj(h0)
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constr, 'jac': constr_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj0 - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def subgradient_equalize(h, n, top_k=10, max_iters=2000, verbose=False):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.82
    step = 0.08
    it_no_imp = 0
    dx = 2.0 / n

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)

        sorted_idx = np.argsort(-corr)
        top_idx = sorted_idx[:top_k]
        c_top = corr[top_idx]

        # Power-weighted concentration on widest gap
        gap = c_top - c_top.min()
        if gap < 1e-12:
            w_top = np.ones(top_k) / top_k
        else:
            w_top = np.exp(3.0 * (c_top - c_top[0]) / (c_top.max() - c_top.min() + 1e-15))
            w_top /= w_top.sum()

        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= dx

        g = np.zeros(n)
        for k in top_idx:
            g += weights[k] * corr_grad_single(h, k)
        g -= np.mean(g)

        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            it_no_imp = 0
        else:
            it_no_imp += 1
            if it_no_imp > 300:
                step = max(0.01, step * 0.6)
                mu = min(0.96, mu + 0.08)
                h = best_h.copy()
                momentum = np.zeros(n)
                it_no_imp = 0

        step *= 0.9992
        mu = min(0.96, 0.82 + 0.0004 * it)

    return best_h, best_c5


def binary_simulated_annealing(h_bin, n, max_iters=5000, T_init=0.5, T_min=0.01):
    h = (h_bin > 0.5).astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(42)
    T = T_init
    cool = (T_min / T_init) ** (1.0 / max_iters)

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    for it in range(max_iters):
        if remaining() < 10:
            break
        T *= cool

        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        i, j = ones[ia], zeros[ib]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5

        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            h[j] = 1.0
            h[i] = 0.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5


def binary_greedy_swap(h_bin, n, max_rounds=60):
    h = (h_bin > 0.5).astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(123)

    for rnd in range(max_rounds):
        if remaining() < 10:
            break
        corr = compute_corr(h)
        top_lags = np.argsort(-corr)[:8]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        best_improve = 1e-14
        best_swap = None
        tried = set()

        for lag in top_lags:
            off = lag - (n - 1)
            if abs(off) >= n:
                continue
            a = abs(off)
            # Pairs at this lag: (i, i+a) or (i+a, i) depending on sign
            if off >= 0:
                valid_i = ones[(ones + off) < n]
            else:
                valid_i = ones[(ones - a) >= 0]

            # Also consider reverse: zeros where the partner is 1
            if off >= 0:
                valid_j = zeros[(zeros - off) >= 0]
            else:
                valid_j = zeros[(zeros + a) < n]

            candidates = []
            for i in valid_i[:80]:
                j = i + off if off >= 0 else i - a
                if 0 <= j < n and h[j] == 0:
                    candidates.append((i, j))
            for j in valid_j[:80]:
                i = j + off if off >= 0 else j + a
                if 0 <= i < n and h[i] == 1:
                    candidates.append((i, j))

            for (i, j) in candidates[:120]:
                key = (min(i, j), max(i, j))
                if key in tried:
                    continue
                tried.add(key)
                h[i] = 0.0
                h[j] = 1.0
                new_c = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if new_c < best_c5 - best_improve:
                    best_improve = best_c5 - new_c
                    best_swap = (i, j)

        # Also try random swaps
        for _ in range(50):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            key = (min(i, j), max(i, j))
            if key in tried:
                continue
            tried.add(key)
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            if new_c < best_c5 - best_improve:
                best_improve = best_c5 - new_c
                best_swap = (i, j)

        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        best_c5 = compute_c5(h)
        best_h = h.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
