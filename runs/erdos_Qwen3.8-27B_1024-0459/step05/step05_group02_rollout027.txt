import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        for _ in range(n):
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] += diff
            h = np.clip(h, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def thue_morse(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n]


def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def shifted_tm(n, shift, rng):
    """Thue-Morse with random shift and optional complement."""
    t = thue_morse(n + 32)[shift:shift + n]
    if rng.rand() < 0.5:
        t = 1.0 - t
    # Fix count
    n_ones = int(np.sum(t))
    n_half = n // 2
    if n_ones > n_half:
        excess = n_ones - n_half
        ones_idx = np.where(t == 1)[0]
        to_flip = rng.choice(ones_idx, excess, replace=False)
        t[to_flip] = 0.0
    elif n_ones < n_half:
        deficit = n_half - n_ones
        zeros_idx = np.where(t == 0)[0]
        to_flip = rng.choice(zeros_idx, deficit, replace=False)
        t[to_flip] = 1.0
    return t


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=5, maxiter=500):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 8})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def binary_local_search(h_bin, n, max_iters=3000, top_k=3, rng=None):
    """Efficient binary local search with gradient-guided swaps."""
    if rng is None:
        rng = np.random.RandomState(777)
    h_bin = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    corr = compute_corr(h_bin)
    cur_c5 = float(np.max(corr)) * dx if (dx := 2.0 / n) else 0

    stuck_count = 0
    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h_bin)
        cur_c5 = float(np.max(corr)) * (2.0 / n)
        idx_top = np.argsort(-corr)[:top_k]

        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0.0
        best_swap = None

        score_one = -g[ones]
        score_zero = g[zeros]
        n_try = min(20, len(ones))
        top_ones = ones[np.argsort(-score_one)[:n_try]]
        n_try_z = min(20, len(zeros))
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is not None and best_delta < -1e-12:
            i, j = best_swap
            h_bin[i] = 0
            h_bin[j] = 1
            cur_c5 = compute_c5(h_bin)
            stuck_count = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            stuck_count += 1
            # Random perturbation to escape
            esc = False
            n_rand = 15
            for _ in range(n_rand):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                if nc < cur_c5:
                    h_bin[i] = 0
                    h_bin[j] = 1
                    cur_c5 = nc
                    esc = True
                    break
            if not esc:
                # Larger perturbation: swap 2 pairs
                for _ in range(8):
                    i1 = int(ones[rng.randint(len(ones))])
                    i2 = int(ones[rng.randint(len(ones))])
                    j1 = int(zeros[rng.randint(len(zeros))])
                    j2 = int(zeros[rng.randint(len(zeros))])
                    if len({i1, i2, j1, j2}) < 4:
                        continue
                    h_bin[i1] = 0
                    h_bin[i2] = 0
                    h_bin[j1] = 1
                    h_bin[j2] = 1
                    nc = compute_c5(h_bin)
                    h_bin[i1] = 1
                    h_bin[i2] = 1
                    h_bin[j1] = 0
                    h_bin[j2] = 0
                    if nc < cur_c5:
                        h_bin[i1] = 0
                        h_bin[i2] = 0
                        h_bin[j1] = 1
                        h_bin[j2] = 1
                        cur_c5 = nc
                        esc = True
                        break
            if not esc:
                break
            stuck_count = 0
            if stuck_count > 50:
                break

    return best_h, best_c5


def top_equalize(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn

        improved = False
        for scale in [3, 1, 0.5, 5, 2, 0.2, 8, 0.1, 4, 1.5]:
            alpha = 0.02 * scale
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for scale in [2, 1, 3, 0.5]:
                alpha = 0.01 * scale
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def nesterov_gradient(h, n, top_k=5, max_iters=5000, step_init=0.03):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.9
    step = step_init

    for it in range(max_iters):
        if remaining() < 5:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - np.mean(c_top)) + 0.1
        w = np.maximum(w, 0.01)
        w /= w.sum()
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.12)
        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.7

        if it % 2000 == 0 and it > 0:
            step *= 0.8
            momentum = min(momentum + 0.005, 0.95)

    return best_h, best_c5
