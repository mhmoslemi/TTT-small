def compute_corr(h):

def project_feasible(h, n):

def _sm_obj(x):

def corr_grad_at_offset(h, off, n):

def coord_descent(h, n, n_passes=50, rng=None):

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0

            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h

    for beta in [200000, 500000, 2000000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active-set subgradient
    if remaining() > 10:
        h_es, c5_es = exact_subgrad(h, n, n_iter=500, step0=0.04)
        if c5_es < best_c5:
            best_c5 = c5_es
            best_h = h_es.copy()
            h = h_es.copy()

    # Projected gradient fine-tune
    if remaining() > 10:
        h_pg, c5_pg = proj_grad_smooth(h, n, beta=100000, n_iter=500, step0=0.01)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Binary polish
    if remaining() > 25:
        h_bin = binarize(h, n)
        h_b, c5_b = binary_local_search(h_bin, n, rng, time_limit=min(30, int(remaining() * 0.4)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = slsqp_smooth(h_b, n, 200000, 400)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Coordinate descent final touch
    if remaining() > 10:
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=20, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final feasibility projection
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
