import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def swap_delta(h, a, b, n):
    """Compute delta in full correlation when swapping h[a] 1->0 and h[b] 0->1."""
    # h[a] is 1, h[b] is 0
    # For each shift s (0-indexed in full corr, where shift = s - (n-1)):
    # c[s] = sum_i h[i] * (1-h[i+shift]) for valid i
    # 
    # After swap: h'[a] = 0, h'[b] = 1
    # Delta at full-corr index s (shift = s - (n-1)):
    #   From h[a]: -(1 - h[a+shift]) if 0<=a+shift<n
    #   From h[b]: +(1 - h[b+shift]) if 0<=b+shift<n  
    #   From h[a] as target: -h[a-shift] if 0<=a-shift<n
    #   From h[b] as target: +h[b-shift] if 0<=b-shift<n
    
    shifts = np.arange(2*n-1) - (n-1)  # range from -(n-1) to (n-1)
    
    delta = np.zeros(2*n-1)
    
    # Term: -(1-h[a+shift]) for valid a+shift
    va = a + shifts  # positions of a+shift
    valid_a = (va >= 0) & (va < n)
    delta[valid_a] -= (1.0 - h[va[valid_a]])
    
    # Term: +(1-h[b+shift]) for valid b+shift
    vb = b + shifts
    valid_b = (vb >= 0) & (vb < n)
    delta[valid_b] += (1.0 - h[vb[valid_b]])
    
    # Term: -h[a-shift] for valid a-shift
    wa = a - shifts
    valid_wa = (wa >= 0) & (wa < n)
    delta[valid_wa] -= h[wa[valid_wa]]
    
    # Term: +h[b-shift] for valid b-shift
    wb = b - shifts
    valid_wb = (wb >= 0) & (wb < n)
    delta[valid_wb] += h[wb[valid_wb]]
    
    return delta

def binary_sa(h, n, t_start, t_end, n_iters, rng, verbose=False):
    """Simulated annealing on binary solution."""
    dx = 2.0 / n
    h = h.copy()
    corr = compute_full_corr(h)
    c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5
    best_corr = corr.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iters):
        if remaining() < 10:
            break
        
        T = t_start * (t_end / t_start) ** (it / max(1, n_iters - 1))
        
        # Pick random 1 and 0
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        
        if a == b:
            continue
        
        # Compute delta
        delta = swap_delta(h, a, b, n)
        new_corr = corr + delta
        new_c5 = np.max(new_corr) * dx
        
        # Accept?
        if new_c5 < c5 or rng.random() < np.exp(-(new_c5 - c5) / max(T, 1e-15)):
            # Accept
            h[a] = 0
            h[b] = 1
            corr = new_corr
            c5 = new_c5
            
            ones[ia] = b
            zeros[ib] = a
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_corr = corr.copy()
    
    return best_h, best_c5

def binary_hill_climb(h, n, max_passes=20, rng=None):
    """Hill climbing: find best improving swap each time."""
    dx = 2.0 / n
    h = h.copy()
    corr = compute_full_corr(h)
    c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for p in range(max_passes):
        if remaining() < 30:
            break
        
        best_improve = 1e-15
        best_swap = None
        best_new_corr = None
        
        # Try all pairs (expensive but thorough)
        # For efficiency, try a subset
        if len(ones) * len(zeros) > 50000:
            # Too many pairs, sample
            n_tries = 20000
            pairs = []
            for _ in range(n_tries):
                ia = rng.randint(len(ones)) if rng else np.random.randint(len(ones))
                ib = rng.randint(len(zeros)) if rng else np.random.randint(len(zeros))
                pairs.append((ia, ib))
        else:
            pairs = [(ia, ib) for ia in range(len(ones)) for ib in range(len(zeros))]
        
        for (ia, ib) in pairs:
            a = ones[ia]
            b = zeros[ib]
            delta = swap_delta(h, a, b, n)
            new_c5 = np.max(corr + delta) * dx
            improve = c5 - new_c5
            if improve > best_improve:
                best_improve = improve
                best_swap = (ia, ib, a, b)
                best_new_corr = corr + delta
        
        if best_swap is None:
            break
        
        ia, ib, a, b = best_swap
        h[a] = 0
        h[b] = 1
        corr = best_new_corr
        c5 = c5 - best_improve
        ones[ia] = b
        zeros[ib] = a
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

    beta = _SLSQP_BETA

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_SLSQP_N = 0
_SLSQP_BETA = 0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        sm = np.sum(proj)
        if sm > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            result[free] += diff / nf
            result = np.clip(result, 0.0, 1.0)
    return result


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def binarize(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def make_shifts(n):
    return np.arange(2 * n - 1) - (n - 1)


def swap_delta(h, a, b, n, shifts, n_total):
    delta = np.zeros(n_total)
    va = a + shifts
    m = (va >= 0) & (va < n)
    delta[m] -= (1.0 - h[va[m]])
    vb = b + shifts
    m = (vb >= 0) & (vb < n)
    delta[m] += (1.0 - h[vb[m]])
    wa = a - shifts
    m = (wa >= 0) & (wa < n)
    delta[m] -= h[wa[m]]
    wb = b - shifts
    m = (wb >= 0) & (wb < n)
    delta[m] += h[wb[m]]
    return delta


def binary_sa(h, n, shifts, n_total, t_start, t_end, max_iters, rng):
    dx = 2.0 / n
    h = h.copy()
    corr = compute_full_corr(h).astype(float)
    c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    n1 = len(ones)
    nz = len(zeros)
    for it in range(max_iters):
        if remaining() < 10:
            break
        T = t_start * (t_end / t_start) ** (it / max(1, max_iters - 1))
        ia = rng.randint(n1)
        ib = rng.randint(nz)
        a = ones[ia]
        b = zeros[ib]
        delta = swap_delta(h, a, b, n, shifts, n_total)
        new_c5 = float(np.max(corr + delta) * dx)
        accept = new_c5 < c5
        if not accept and T > 1e-8:
            ratio = -min((new_c5 - c5) / max(T, 1e-15), 50.0)
            if np.log(rng.random() + 1e-30) < ratio:
                accept = True
        if accept:
            h[a] = 0.0
            h[b] = 1.0
            corr = corr + delta
            c5 = new_c5
            ones[ia] = b
            zeros[ib] = a
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5


def guided_binary_sa(h, n, shifts, n_total, t_start, t_end, max_iters, rng):
    dx = 2.0 / n
    h = h.copy()
    corr = compute_full_corr(h).astype(float)
    c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    n1 = len(ones)
    nz = len(zeros)
    for it in range(max_iters):
        if remaining() < 10:
            break
        T = t_start * (t_end / t_start) ** (it / max(1, max_iters - 1))
        if rng.random() < 0.6:
            corr_max = np.max(corr)
            active = np.where(corr >= corr_max - 3.0)[0]
            if len(active) > 0:
                ak = active[rng.randint(len(active))]
                s_val = int(shifts[ak])
                a = ones[rng.randint(n1)]
                b = -1
                for dp in [s_val, -s_val]:
                    bt = a + dp
                    if 0 <= bt < n and h[bt] == 0:
                        b = bt
                        break
                if b < 0:
                    b = zeros[rng.randint(nz)]
            else:
                a = ones[rng.randint(n1)]
                b = zeros[rng.randint(nz)]
        else:
            a = ones[rng.randint(n1)]
            b = zeros[rng.randint(nz)]
        delta = swap_delta(h, a, b, n, shifts, n_total)
        new_c5 = float(np.max(corr + delta) * dx)
        accept = new_c5 < c5
        if not accept and T > 1e-8:
            ratio = -min((new_c5 - c5) / max(T, 1e-15), 50.0)
            if np.log(rng.random() + 1e-30) < ratio:
                accept = True
        if accept:
            h[a] = 0.0
            h[b] = 1.0
            corr = corr + delta
            c5 = new_c5
            ia = ones.index(a)
            ib = zeros.index(b)
            ones[ia] = b
            zeros[ib] = a
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5


def binary_hill_climb(h, n, shifts, n_total, max_passes, n_tries, rng):
    dx = 2.0 / n
    h = h.copy()
    corr = compute_full_corr(h).astype(float)
    c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    n1 = len(ones)
    nz = len(zeros)
    for p in range(max_passes):
        if remaining() < 30:
            break
        best_d = 1e-14
        best_a = -1
        best_b = -1
        best_delta = None
        for _ in range(n_tries):
            ia = rng.randint(n1)
            ib = rng.randint(nz)
            a = ones[ia]
            b = zeros[ib]
            delta = swap_delta(h, a, b, n, shifts, n_total)
            nc5 = float(np.max(corr + delta) * dx)
            d = c5 - nc5
            if d > best_d:
                best_d = d
                best_a = a
                best_b = b
                best_delta = delta
        if best_delta is None:
            break
        h[best_a] = 0.0
        h[best_b] = 1.0
        corr = corr + best_delta
        c5 = float(np.max(corr) * dx)
        ia = ones.index(best_a)
        ib = zeros.index(best_b)
        ones[ia] = best_b
        zeros[ib] = best_a
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def _slsqp_obj(x):
    n = _SLSQP_N
    beta = _SLSQP_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)


def _slsqp_grad(x):
    n = _SLSQP_N
    beta = _SLSQP_BETA
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex) * (2.0 / n)
    g = np.zeros(n)
    n_total = 2 * n - 1
    shifts = np.arange(n_total) - (n - 1)
    for k in range(n_total):
        wk = w[k]
        if wk < 1e-18:
            continue
        s = shifts[k]
        if s >= 0:
            np_ = n - s
            g[s:s + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[s:s + np_]
        else:
            a = -s
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:a + np_])
            g[a:a + np_] -= wk * x[:np_]
    return g


def _slsqp_con(x):
    return np.sum(x) - _SLSQP_N / 2.0


def _slsqp_con_jac(x):
    return np.ones(_SLSQP_N)


def slsqp_simax(h, n, beta, maxiter=300):
    global _SLSQP_N, _SLSQP_BETA
    _SLSQP_N = n
    _SLSQP_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h0, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _slsqp_obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def continuous_refine(h, n, max_iters, beta):
    dx = 2.0 / n
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    y = h.copy()
    t_val = 1.0
    n_total = 2 * n - 1
    shifts = np.arange(n_total) - (n - 1)
    for it in range(max_iters):
        if remaining() < 10:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        corr = compute_full_corr(h)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex) * dx
        g = np.zeros(n)
        for k in range(n_total):
            wk = w[k]
            if wk < 1e-18:
                continue
            s = shifts[k]
            if s >= 0:
                np_ = n - s
                g[s:s + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[s:s + np_]
            else:
                a = -s
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:a + np_])
                g[a:a + np_] -= wk * h[:np_]
        lr = 1.0 / (beta * (n * dx + 1.0) + 1e-10)
        y_new = y + alpha * (y - h)
        h_new = project_feasible(y_new - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

    # Phase 2: SA rounds (guided + regular)

        (0.010, 0.002, 150000, False),
