def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = len(corr)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff

                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]

    return val, grad

            # Grid search on a small set
            candidates_i = [0.0, old_i, 1.0, 0.3, 0.7]
            candidates_j = [0.0, old_j, 1.0, 0.3, 0.7]
            for ti in candidates_i:
                if ti < -1e-10 or ti > 1.0 + 1e-10:
                    continue
                ti = np.clip(ti, 0.0, 1.0)
                for tj in candidates_j:
                    if tj < -1e-10 or tj > 1.0 + 1e-10:
                        continue
                    tj = np.clip(tj, 0.0, 1.0)
                    new_corr = base + A_i * (ti - old_i) + A_j * (tj - old_j)
                    new_corr[n - 1] += ti * (1.0 - ti) + tj * (1.0 - tj)
                    f_new = np.max(new_corr)
                    if f_new < best_f - 1e-14:
                        best_f = f_new
                        best_ti = ti
                        best_tj = tj

            if best_ti != old_i or best_tj != old_j:
                # Local refinement with smaller steps
                for _ in range(20):
                    found = False
                    for di in [-0.1, 0.1, -0.03, 0.03, -0.01, 0.01]:
                        for dj in [-0.1, 0.1, -0.03, 0.03, -0.01, 0.01]:
                            ti = np.clip(best_ti + di, 0.0, 1.0)
                            tj = np.clip(best_tj + dj, 0.0, 1.0)
                            new_corr = base + A_i * (ti - old_i) + A_j * (tj - old_j)
                            new_corr[n - 1] += ti * (1.0 - ti) + tj * (1.0 - tj)
                            f_new = np.max(new_corr)
                            if f_new < best_f - 1e-14:
                                best_f = f_new
                                best_ti = ti
                                best_tj = tj
                                found = True
                    if not found:
                        break

                di = best_ti - old_i
                dj = best_tj - old_j
                if di != 0.0 or dj != 0.0:
                    best[i] = best_ti
                    best[j] = best_tj
                    best_corr += A_i * di + A_j * dj
                    best_corr[n - 1] += best_ti * (1.0 - best_ti) + best_tj * (1.0 - best_tj)
                    improved = True
                    best_c5 = np.max(best_corr) * dx

        if not improved:
            break

    best_c5 = np.max(best_corr) * dx
    return best, best_c5


def subgradient_step(h, n, step_size, n_active=10):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    threshold = max_corr - 0.3
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        idx = active[np.argsort(corr[active])[::-1][:n_active]]
        active = idx

    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h.copy()
    grad /= gnorm
    h_new = h - step_size * grad
    return project_feasible(h_new, n)


def targeted_perturb(h, n, rng, scale=0.1):
    """Perturb coordinates that contribute most to the top correlations."""
    corr = compute_corr(h)
    top_k = np.argsort(corr)[::-1][:5]
    h_pert = h.copy()
    for k_idx in top_k:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                idxs = np.random.choice(npairs, min(npairs, 20), replace=False)
                h_pert[offset + idxs] -= rng.uniform(0, scale)
                h_pert[idxs] += rng.uniform(0, scale)
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                idxs = np.random.choice(nppairs, min(npairs, 20), replace=False)
                h_pert[idxs] -= rng.uniform(0, scale)
                h_pert[aoff + idxs] += rng.uniform(0, scale)
    return project_feasible(h_pert, n)


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Thorough coordinate descent at parent resolution
    if remaining_time() > 100:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=400)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Block coordinate descent
    if remaining_time() > 80:
        h = best_h.copy()
        h, c5 = block_coord_descent(h, n, max_sweeps=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Follow with single CD
        if remaining_time() > 40:
            h, c5 = fast_coord_descent(h, n, max_sweeps=200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 3: SLSQP continuation
    if remaining_time() > 60:
        h = best_h.copy()
        for beta in [1000, 5000, 20000, 80000, 200000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_refine(h, n, beta, 500)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-15:
                h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    # Phase 4: More coordinate descent
    if remaining_time() > 40:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Upsample to higher resolution
    if remaining_time() > 150:
        high_n = 2 * n
        if high_n <= 2560:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            c5_high = compute_c5(h_high)

            if remaining_time() > 100:
                h_high, c5_high = fast_coord_descent(h_high, high_n, max_sweeps=300)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = high_n

            if remaining_time() > 40:
                h_high = best_h.copy()
                h_high, c5_high = block_coord_descent(h_high, best_n, max_sweeps=30)
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_high.copy()

            if remaining_time() > 30:
                h_slq = slsqp_refine(best_h, best_n, 50000, 400)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()

            if remaining_time() > 20:
                h_high, c5_high = fast_coord_descent(best_h, best_n, max_sweeps=200)
                if c5_high < best_c5:
                    best_c5 = c5_high
                    best_h = h_high.copy()

    # Phase 6: Try even higher resolution
    if remaining_time() > 120 and best_n < 2000:
        very_high_n = min(2 * best_n, 3200)
        h_vh = interp_to(best_h, best_n, very_high_n)
        h_vh = project_feasible(h_vh, very_high_n)
        c5_vh = compute_c5(h_vh)

        if remaining_time() > 80:
            h_vh, c5_vh = fast_coord_descent(h_vh, very_high_n, max_sweeps=200)
        if c5_vh < best_c5:
            best_c5 = c5_vh
            best_h = h_vh.copy()
            best_n = very_high_n

        if remaining_time() > 30:
            h_slq = slsqp_refine(best_h, best_n, 30000, 300)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()

    # Phase 7: Perturbation + restart cycles
    if remaining_time() > 80:
        n_perturb = 12
        n_cur = best_n
        for p in range(n_perturb):
            if remaining_time() < 25:
                break

            if p < 4:
                h_pert = targeted_perturb(best_h, n_cur, rng, scale=0.12 / (1 + p * 0.2))
            elif p < 8:
                h_pert = best_h.copy()
                noise_scale = 0.12 / (1 + (p - 4) * 0.3)
                mask = rng.random(n_cur) < 0.4
                h_pert[mask] += rng.randn(int(np.sum(mask))) * noise_scale
                h_pert = project_feasible(h_pert, n_cur)
            else:
                # Binary-like perturbation
                h_pert = best_h.copy()
                num_flips = n_cur // 10
                flip_idx = rng.choice(n_cur, num_flips, replace=False)
                for fi in flip_idx:
                    h_pert[fi] = 1.0 - h_pert[fi]
                h_pert = project_feasible(h_pert, n_cur)

            h_new, c5_new = fast_coord_descent(h_pert, n_cur, max_sweeps=200)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

            if remaining_time() > 20:
                h_fl = slsqp_refine(h_new, n_cur, 30000, 300)
                c5_fl = compute_c5(h_fl)
                if c5_fl < best_c5:
                    best_c5 = c5_fl
                    best_h = h_fl.copy()
                    if remaining_time() > 15:
                        h_new2, c5_new2 = fast_coord_descent(h_fl, n_cur, max_sweeps=100)
                        if c5_new2 < best_c5:
                            best_c5 = c5_new2
                            best_h = h_new2.copy()

    # Phase 8: Subgradient + CD alternation
    if remaining_time() > 40:
        n_cur = best_n
        h = best_h.copy()
        for it in range(800):
            if remaining_time() < 5:
                break
            step = 0.015 / (1.0 + it * 0.005)
            h_sg = subgradient_step(h, n_cur, step, n_active=8)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5 - 1e-15:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()
            elif it % 100 == 50 and remaining_time() > 15:
                h_new, c5_new = fast_coord_descent(h, n_cur, max_sweeps=30)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    h = best_h.copy()

    # Phase 9: Final SLSQP polish with very high beta
    if remaining_time() > 12:
        n_cur = best_n
        h = best_h.copy()
        for beta in [100000, 500000, 2000000]:
            if remaining_time() < 6:
                break
            h_new = slsqp_refine(h, n_cur, beta, 600)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 8:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: One more SLSQP
    if remaining_time() > 6:
        n_cur = best_n
        h_new = slsqp_refine(best_h, n_cur, 1000000, 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 12: Very final CD
    if remaining_time() > 4:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
