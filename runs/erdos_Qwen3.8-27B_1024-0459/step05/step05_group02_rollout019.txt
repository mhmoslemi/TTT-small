The first term varies with o (it's the sum of h over a shorter range for large |o|).
The second term is the "autocorrelation" at lag o.

def compute_effect_matrix(h, n):
    """eff[p, o] = (1-h[p+o]) - h[p-o] for valid indices, 0 otherwise.
    o ranges from -(n-1) to (n-1), indexed by o + n - 1."""
    corr_len = 2 * n - 1
    eff = np.zeros((n, corr_len))
    offsets = np.arange(corr_len) - (n - 1)

        for o_idx in range(corr_len):
            o = offsets[o_idx]
            p1 = i + o
            p2 = i - o
            val = 0.0
            if 0 <= p1 < n:
                val += (1.0 - h[p1])
            if 0 <= p2 < n:
                val -= h[p2]
            eff[i, o_idx] = val
    return eff

def compute_effect_matrix(h, n):
    corr_len = 2 * n - 1
    offsets = np.arange(corr_len) - (n - 1)
    # For each offset o, compute eff[:, o_idx]
    # eff[i, o_idx] = (1-h[i+o]) - h[i-o] with 0 for OOB
    eff = np.zeros((n, corr_len))
    for o_idx, o in enumerate(offsets):
        if o >= 0:

            idx2 = np.arange(o, n)

            # i-o = i+ao valid for i in [0, n-1-ao]

        worst_lag = int(np.argmax(corr))

        # Score: we want eff[j, worst_lag] - eff[i, worst_lag] to be as negative as possible
        score_i = eff[ones, worst_lag]
        score_j = eff[zeros, worst_lag]

        n_try_i = min(top_candidates, len(ones))
        n_try_j = min(top_candidates, len(zeros))
        top_i = ones[np.argsort(-score_i)[:n_try_i]]
        top_j = zeros[np.argsort(score_j)[:n_try_j]]

        best_delta = 0.0
        best_swap = None

        for i in top_i:
            for j in top_j:
                new_corr_fast = corr + eff[j] - eff[i]
                new_max = float(np.max(new_corr_fast))
                delta = new_max - cur_max
                if delta < best_delta - 1e-12:
                    best_delta = delta
                    best_swap = (int(i), int(j))

        # Also try some random swaps for diversity
        n_random = min(4, len(ones), len(zeros))
        for _ in range(n_random):
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
            new_corr_fast = corr + eff[j] - eff[i]
            new_max = float(np.max(new_corr_fast))
            delta = new_max - cur_max
            if delta < best_delta - 1e-12:
                best_delta = delta
                best_swap = (i, j)

        if best_swap is None:
            # Try to reduce second-worst lag
            second_lag = int(np.argsort(-corr)[1])
            score_i2 = eff[ones, second_lag]
            score_j2 = eff[zeros, second_lag]
            n_try2 = min(5, len(ones))
            top_i2 = ones[np.argsort(-score_i2)[:n_try2]]
            top_j2 = zeros[np.argsort(score_j2)[:n_try2]]
            for i in top_i2:
                for j in top_j2:
                    new_corr_fast = corr + eff[j] - eff[i]
                    new_max = float(np.max(new_corr_fast))
                    delta = new_max - cur_max
                    if delta < best_delta - 1e-12:
                        best_delta = delta
                        best_swap = (int(i), int(j))

        if best_swap is None:
            # Simulated annealing: accept a slightly worse move
            if np.random.rand() < 0.1 and no_improve > 200:
                i = int(ones[np.random.randint(len(ones))])
                j = int(zeros[np.random.randint(len(zeros))])
                new_corr_fast = corr + eff[j] - eff[i]
                new_max = float(np.max(new_corr_fast))
                delta = new_max - cur_max
                if delta < 0.02 / dx:  # Accept if not too much worse
                    h[i] = 0
                    h[j] = 1
                    continue
            break

        i, j = best_swap
        h[i] = 0
        h[j] = 1

    return best_h, best_c5


def nesterov_gradient(h, n, top_k=5, max_iters=5000, step_init=0.03):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init
    stall = 0

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            h = best_h + np.random.randn(n) * 0.005
            h = proj(h, n)
            v = np.zeros(n)
            step = step_init
            stall = 0
            continue

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = np.exp(50.0 * (c_top - c_top[0]))
        w /= w.sum()
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < c5 + 1e-14:
            h = h_new
            if c5_new < c5:
                step = min(step * 1.01, 0.15)
        else:
            v *= 0.5
            step *= 0.85

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick continuous optimization
    for beta in [500, 2000, 8000]:
        if remaining() < 15:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Nesterov gradient
    if remaining() > 30:
        h_pg, c5_pg = nesterov_gradient(h, n, top_k=5, max_iters=4000, step_init=0.025)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 3: Intensive binary local search
    if remaining() > 50:
        h_bin = (best_h > 0.5).astype(float)
        h_bin2, c5_bin2 = binary_local_search_v2(h_bin, n, max_iters=15000, top_candidates=12)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Phase 4: Continuous refinement of best binary solution
    if remaining() > 30:
        h_ref = (best_h > 0.3).astype(float) * 1.0  # threshold slightly lower
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = proj(h_ref, n)
        for beta in [3000, 10000, 30000]:
            if remaining() < 10:
                break
            h_ref = lbfgs_pass(h_ref, n, beta, top_k=4, maxiter=400)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 5: Try alternative resolutions
    n_candidates = sorted(set([
        n, n - 2, n + 2, n + 4, n + 6, n + 8,
        int(n * 1.1), int(n * 1.15), int(n * 1.2),
    ]))
    n_candidates = [nn for nn in n_candidates if 192 <= nn <= 768 and nn % 2 == 0]
    n_candidates = [nn for nn in n_candidates if nn != n]
    n_candidates.sort(key=lambda x: abs(x - n))

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # Quick continuous optimization
        for beta in [1000, 5000]:
            if remaining() < 15:
                break
            h_feas = lbfgs_pass(h_feas, n_t, beta, top_k=5, maxiter=300)

        # Binary local search at this resolution
        h_b = (h_feas > 0.5).astype(float)
        h_bl, c5_bl = binary_local_search_v2(h_b, n_t, max_iters=8000, top_candidates=10)
        c5_bl_full = min(compute_c5(h_bl), compute_c5(h_feas))

        # Continuous refinement
        h_rf = lbfgs_pass(h_bl, n_t, 8000, top_k=4, maxiter=300)
        c5_rf = compute_c5(h_rf)

        c5_cand = min(c5_bl, c5_rf, compute_c5(h_feas))
        if c5_cand < best_c5:
            best_c5 = c5_cand
            if c5_rf == c5_cand:
                best_h = h_rf.copy()
            elif c5_bl == c5_cand:
                best_h = h_bl.copy()
            else:
                best_h = h_feas.copy()
            best_n = n_t

    # Phase 6: Final intensive binary search at best resolution
    if remaining() > 40:
        h_fin_bin = (best_h > 0.5).astype(float)
        h_fin, c5_fin = binary_local_search_v2(h_fin_bin, best_n, max_iters=15000, top_candidates=15)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 7: Final continuous polish
    if remaining() > 20:
        h_pol = best_h.copy()
        for beta in [15000, 50000]:
            if remaining() < 8:
                break
            h_pol = lbfgs_pass(h_pol, best_n, beta, top_k=4, maxiter=300)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 8: One more binary pass if time allows
    if remaining() > 25:
        h_last_bin = (best_h > 0.5).astype(float)
        h_last, c5_last = binary_local_search_v2(h_last_bin, best_n, max_iters=8000, top_candidates=10)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()
        # Quick continuous refinement
        if remaining() > 10:
            h_last_ref = lbfgs_pass(h_last, best_n, 20000, top_k=4, maxiter=200)
            c5_last_ref = compute_c5(h_last_ref)
            if c5_last_ref < best_c5:
                best_c5 = c5_last_ref
                best_h = h_last_ref.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
