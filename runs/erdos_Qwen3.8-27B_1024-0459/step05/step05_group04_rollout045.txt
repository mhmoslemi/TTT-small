        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)
    hi3 = min(j + n - 1, r - 1)
    if lo3 <= hi3:
        ks = np.arange(lo3, hi3 + 1)
        dr[lo3:hi3+1] += (1.0 - h[j + n - 1 - ks])
    lo4 = max(n - 1 - j, 0)
    hi4 = min(2 * (n - 1) - j, r - 1)
    if lo4 <= hi4:
        ks = np.arange(lo4, hi4 + 1)
        dr[lo4:hi4+1] -= h[j + ks - (n - 1)]
    dr[n - 1] -= 2.0
    return dr

def corr_gradient(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:off+np_] += (1.0 - h[:np_])
        g[:np_] -= h[off:off+np_]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao+np_])
        g[ao:ao+np_] -= h[:np_]
    return g

def binary_local_search(h_bin, n, time_limit, rng):
    h = (h_bin > 0.5).copy().astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    start = time.time()
    stall = 0
    max_stall = 300
    
    while time.time() - start < time_limit and remaining() > 8:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        top_shifts = np.argsort(-corr)[:14]
        candidates = []
        for s in top_shifts:
            off = s - (n - 1)
            if off >= 0:
                np_ = n - off
                contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for ci in contrib[:25]:
                    candidates.append((ci + off, ci))
            else:
                ao = -off
                np_ = n - ao
                contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for ci in contrib[:25]:
                    candidates.append((ci, ci + ao))
        for _ in range(50):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        best_move = None
        best_new_max = cur_max + 1e-15
        seen = set()
        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            dr = make_swap_delta(h, i, j, n)
            new_max = float(np.max(corr + dr))
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr)
        
        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                for _ in range(5):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        dr = make_swap_delta(h, i, j, n)
                        h[i] = 0
                        h[j] = 1
                        corr = corr + dr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                cur_max = float(np.max(corr))
                stall = 0
    
    return best_h, best_max * dx

            else:

        # Perturbation: swap multiple pairs
        if time.time() - start < time_limit and remaining() > 10:

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / _BETA) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def con_fun(x):
        return np.sum(x) - _TGT
    
    def con_jac(x):
        return np.ones(_N)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def grad_desc_top(h, n, max_iters=1000, top_k=5):
    h = h.copy()
    best_c5 = compute_c5(h)
    step_sizes = [0.02, 0.05, 0.03, 0.01, 0.08, 0.015, 0.04, 0.1, 0.06, 0.025]
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        g_avg = np.zeros(n)
        for k in top_idx:
            g_avg += corr_gradient(h, k, n)
        g_avg /= top_k
        d = -g_avg
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in step_sizes:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try larger steps
            for step in [0.15, 0.2, 0.3, 0.5]:
                ht = project_feasible(h + step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def grad_asc_bottom(h, n, max_iters=500, top_k=5, bot_k=3):
    h = h.copy()
    best_c5 = compute_c5(h)
    step_sizes = [0.01, 0.03, 0.05, 0.02, 0.08, 0.015, 0.04, 0.06]
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        bot_idx = np.argsort(corr)[:bot_k]
        
        g_top = np.zeros(n)
        for k in top_idx:
            g_top += corr_gradient(h, k, n)
        g_top /= top_k
        
        g_bot = np.zeros(n)
        for k in bot_idx:
            g_bot += corr_gradient(h, k, n)
        g_bot /= bot_k
        
        d = g_bot - g_top
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in step_sizes:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def van_der_corput(n):
    h = np.zeros(n)
    for i in range(n):
        x = i + 1
        c = 0
        d = 1
        while x > 0:
            c += (x % 2) / d
            x //= 2
            d *= 2
        h[i] = 1.0 if c < 1.0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def binary_from_continuous(h_cont, n):
    n_half = n // 2
    sort_idx = np.argsort(-h_cont)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):

    try:
        parent_h = np.array(initial_h_values, dtype=float)

        parent_c5 = compute_c5(parent_h)

    # Phase 1: Binary ILS at parent resolution (main optimization)
    n = best_n

    if t1 > 30:

    ]
    # Add a frequency-informed start
    n_fft = n
    freq_init = np.zeros(n_fft)
    # Emphasize low frequencies to create smooth structure
    for f in [1, 2, 3, 5, 8]:
        freq_init[f] += 0.5
        freq_init[n_fft - f] += 0.5
    h_freq = np.real(np.fft.ifft(freq_init))
    h_freq = (h_freq - h_freq.min()) / (h_freq.max() - h_freq.min() + 1e-10)
    h_freq = binary_from_continuous(h_freq, n)
    start_points.append(('freq', h_freq))
    
    for name, sp in start_points:
        if remaining() < 40:
            break
        sp_c5 = compute_c5(sp)

            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n
    
    # Phase 3: Try other resolutions
    resolutions = sorted(set([
        n - 16, n - 8, n + 8, n + 16, n + 32, n + 48,

    ]))
    resolutions = [r for r in resolutions if 128 <= r <= 800 and r % 2 == 0 and r != n]

    for n_t in resolutions[:5]:

            break
        h_interp = interp_to(best_h, best_n, n_t)
