    """Precompute delta vectors for each position.
    delta_remove[i] = change in corr when h[i] goes 1->0
    delta_add[j] = change in corr when h[j] goes 0->1
    For binary h: delta_add[j] = -delta_remove[j] (by linearity)

            for idx in range(n-1-o+1):

        # Focus on top shifts for fault-based candidates
        top_count = min(8, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_count]

        # 1-swap: random candidates
        n_rand = 120
        for _ in range(n_rand):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            h[i] = 0
            h[j] = 1
            nc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            h[i] = 1
            h[j] = 0
            if nc < current_max - 1e-12 and (current_max - nc) > best_improvement:
                best_improvement = current_max - nc
                best_move = (1, i, j)

        # 1-swap: fault-focused candidates
        n_fault = 80
        for _ in range(n_fault):
            k_idx = top_idx[rng.randint(min(5, len(top_idx)))]
            off = int(k_idx) - (n - 1)
            if off >= 0:
                if n - off > 0:
                    faults = np.where((h[:n - off] == 1) & (h[off:] == 0))[0]
                else:
                    continue
            else:
                a = -off
                if n - a > 0:
                    faults = np.where((h[a:] == 1) & (h[:n - a] == 0))[0]
                else:
                    continue
            if len(faults) == 0:
                continue
            fi = faults[rng.randint(len(faults))]
            i = fi if off >= 0 else fi + (-off)
            j = zeros[rng.randint(n_zeros)]
            if i == j or i < 0 or i >= n or j < 0 or j >= n:
                continue
            if h[i] != 1 or h[j] != 0:
                continue
            h[i] = 0
            h[j] = 1
            nc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            h[i] = 1
            h[j] = 0
            if nc < current_max - 1e-12 and (current_max - nc) > best_improvement:
                best_improvement = current_max - nc
                best_move = (1, i, j)

        # 2-swap candidates
        if n_ones >= 2 and n_zeros >= 2:
            n_two = 60
            for _ in range(n_two):
                i1 = ones[rng.randint(n_ones)]
                i2 = ones[rng.randint(n_ones)]
                j1 = zeros[rng.randint(n_zeros)]
                j2 = zeros[rng.randint(n_zeros)]
                if len(set([i1, i2, j1, j2])) < 4:
                    continue
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
                nc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                h[i1] = 1
                h[i2] = 1
                h[j1] = 0
                h[j2] = 0
                if nc < current_max - 1e-12 and (current_max - nc) > best_improvement:
                    best_improvement = current_max - nc
                    best_move = (2, i1, i2, j1, j2)

        # 3-swap candidates (occasionally)
        if n_ones >= 3 and n_zeros >= 3 and rng.randint(4) == 0:
            for _ in range(10):
                i1 = ones[rng.randint(n_ones)]
                i2 = ones[rng.randint(n_ones)]
                i3 = ones[rng.randint(n_ones)]
                j1 = zeros[rng.randint(n_zeros)]
                j2 = zeros[rng.randint(n_zeros)]
                j3 = zeros[rng.randint(n_zeros)]
                all_idx = [i1, i2, i3, j1, j2, j3]
                if len(set(all_idx)) < 6:
                    continue
                h[i1] = 0; h[i2] = 0; h[i3] = 0
                h[j1] = 1; h[j2] = 1; h[j3] = 1
                nc = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                h[i1] = 1; h[i2] = 1; h[i3] = 1
                h[j1] = 0; h[j2] = 0; h[j3] = 0
                if nc < current_max - 1e-12 and (current_max - nc) > best_improvement:
                    best_improvement = current_max - nc
                    best_move = (3, i1, i2, i3, j1, j2, j3)

        if best_move is not None:
            mtype = best_move[0]
            if mtype == 1:
                i, j = best_move[1], best_move[2]
                h[i] = 0
                h[j] = 1
            elif mtype == 2:
                i1, i2, j1, j2 = best_move[1], best_move[2], best_move[3], best_move[4]
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            else:
                i1, i2, i3, j1, j2, j3 = best_move[1:7]
                h[i1] = 0; h[i2] = 0; h[i3] = 0
                h[j1] = 1; h[j2] = 1; h[j3] = 1
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            current_max = float(np.max(corr))
            no_improve = 0
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            # Try perturbation to escape
            if no_improve > 500 and n_ones >= 2 and n_zeros >= 2:
                n_flip = rng.randint(2, 5)
                for _ in range(n_flip):
                    if len(ones) < 1 or len(zeros) < 1:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0
                    h[j] = 1
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
                current_max = float(np.max(corr))

    return best_h, best_max

def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with smooth max approximation."""
    h = project_feasible(h, n)
    dx_v = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx_v
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def cons_eq(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n

    # Phase 1: Binary local search on parent
    if remaining_time() > 30:
        h_bin = make_binary(best_h, n)
        h_b, c5_raw_b = binary_local_search(h_bin, n, rng, time_limit=min(80, int(remaining_time() * 0.25)))
        c5_b = c5_raw_b * (2.0 / n)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n

    # Phase 2: Diverse initializations at current resolution
    if remaining_time() > 50:
        starts = []
        # Thue-Morse
        h_tm = thue_morse(n, rng)
        starts.append(('tm', h_tm))
        # Inverse Thue-Morse
        h_itm = 1.0 - h_tm
        starts.append(('itm', h_itm))
        # Golden ratio
        h_gr = golden_ratio_seq(n, rng)
        starts.append(('gr', h_gr))
        # Random
        h_rng = rng.uniform(0, 1, n)
        h_rng = project_feasible(h_rng, n)
        starts.append(('rng', h_rng))
        # Shifted parent binary
        h_shift = np.roll(make_binary(best_h, n), n // 4)
        starts.append(('shift', h_shift))

        for name, h_s in starts:
            if remaining_time() < 15:
                break
            h_s_bin = make_binary(h_s, n)
            h_lb, raw_max = binary_local_search(h_s_bin, n, rng, time_limit=min(25, int(remaining_time() * 0.15)))
            c5_s = raw_max * (2.0 / n)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_lb.copy()
                best_n = n

    # Phase 3: Resolution sweep
    if remaining_time() > 100:
        n_candidates = []
        for delta in [-32, -24, -16, -12, -8, -4, -2, 2, 4, 6, 8, 12, 16, 24, 32, 48, 64]:
            nt = best_n + delta
            if 128 <= nt <= 700 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [256, 320, 384, 448, 512, 576, 640]:
            if 128 <= nt <= 700 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:6]:
            if remaining_time() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            # Binary local search at this resolution
            h_t_bin = make_binary(h_t, n_t)
            h_tb, raw_max_tb = binary_local_search(h_t_bin, n_t, rng, time_limit=min(25, int(remaining_time() * 0.15)))
            c5_tb = raw_max_tb * (2.0 / n_t)
            if c5_tb < best_c5:
                best_c5 = c5_tb
                best_h = h_tb.copy()
                best_n = n_t

    # Phase 4: SLSQP continuous refinement on best binary
    if remaining_time() > 30:
        h_cont = best_h.copy()
        for beta in [2000, 10000, 50000]:
            if remaining_time() < 10:
                break
            h_cont = slsqp_smooth(h_cont, best_n, beta, maxiter=200)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 5: Another round of binary search from best
    if remaining_time() > 30:
        h_bin2 = make_binary(best_h, best_n)
        h_b2, raw_max2 = binary_local_search(h_bin2, best_n, rng, time_limit=min(40, int(remaining_time() * 0.25)))
        c5_b2 = raw_max2 * (2.0 / best_n)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 6: Final SLSQP polish
    if remaining_time() > 15:
        h_final = slsqp_smooth(best_h, best_n, 20000, maxiter=200)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: Final binary polish
    if remaining_time() > 10:
        h_bin3 = make_binary(best_h, best_n)
        h_b3, raw_max3 = binary_local_search(h_bin3, best_n, rng, time_limit=min(20, int(remaining_time() * 0.4)))
        c5_b3 = raw_max3 * (2.0 / best_n)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
