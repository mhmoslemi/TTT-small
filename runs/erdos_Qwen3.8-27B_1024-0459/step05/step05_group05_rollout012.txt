def project_feasible(h, n):

def swap_corr_delta(h, i, j, n):
    """Compute the change in each correlation value when swapping h[i]:1->0, h[j]:0->1.
    Returns the delta array (length 2n-1) and the max correlation after swap."""
    # For offset m, corr = sum_{p valid} h[p]*(1-h[p+m])
    # After swap: h[i]=0, h[j]=1
    # Affected terms for each offset m:
    #   (i as first): old=1*(1-h[i+m]), new=0 -> delta = -(1-h[i+m]) = h[i+m]-1
    #   (i as second, i-m as first): old=h[i-m]*0=0, new=h_new[i-m]*1 -> delta = h_new[i-m]
    #   (j as first): old=0, new=1*(1-h_new[j+m]) -> delta = 1-h_new[j+m]
    #   (j as second, j-m as first): old=h[j-m]*1, new=h_new[j-m]*0 -> delta = -h_new[j-m]
    
    # Be careful about overlaps when i+m==j or j+m==i etc.
    # Actually since we're computing per-offset, let's just handle it:
    
    h_new = h  # we just track i->0, j->1
    
    def hval(p):
        if p == i:
            return 0.0
        if p == j:
            return 1.0
        return h[p]
    
    deltas = np.zeros(2 * n - 1)
    for m in range(-(n - 1), n - 1):
        k = m + (n - 1)  # index in corr array
        delta = 0.0
        # i as first index: p=i, q=i+m
        if 0 <= i < n and 0 <= i + m < n:
            delta += hval(i + m) - 1.0  # old was 1*(1-h[i+m]), new is 0
        # i as second index: p=i-m, q=i
        if 0 <= i - m < n and 0 <= i < n:
            delta += hval(i - m)  # old was h[i-m]*0=0, new is h_new[i-m]*1
        # j as first index: p=j, q=j+m
        if 0 <= j < n and 0 <= j + m < n:
            delta += 1.0 - hval(j + m)  # old was 0, new is 1*(1-h_new[j+m])
        # j as second index: p=j-m, q=j
        if 0 <= j - m < n and 0 <= j < n:
            delta += -hval(j - m)  # old was h[j-m]*1, new is 0
        deltas[k] = delta
    return deltas

def slsqp_refine(h, n, beta, maxiter=300):
    """SLSQP with smooth max for continuous refinement."""
    dx_v = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_v
    
    def obj_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx_v
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    h0 = project_feasible(h, n)
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h0):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

        h_new = slsqp_refine(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final binary SA
    if remaining() > 30:
        h_bin = binarize(best_h, best_n)
        h_b, c5_b = binary_sa_search(h_bin, best_n, rng, time_limit=min(60, int(remaining() * 0.4)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final subgradient
    if remaining() > 15:
        h_sg, c5_sg = subgradient_refine(best_h, best_n, n_iter=80)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final top-shift
    if remaining() > 10:
        h_eq, c5_eq = top_shift_refine(best_h, best_n, top_k=5, max_iters=100)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
