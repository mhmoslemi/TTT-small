def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_gradient(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        if np_ > 0:
            g[offset:offset + np_] += (1.0 - h[:np_])
            g[:np_] -= h[offset:offset + np_]
    else:
        a = -offset
        np_ = n - a
        if np_ > 0:
            g[:np_] += (1.0 - h[a:a + np_])
            g[a:a + np_] -= h[:np_]
    return g


def smooth_max_val(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)


def smooth_max_grad(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = np.sum(ex)
    w = ex / s
    g = np.zeros(n)
    wc = w * (2.0 / n)
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            if np_ > 0:
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            if np_ > 0:
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
    return g


def slsqp_solve(h, n, beta, maxiter):
    def obj_func(x):
        return smooth_max_val(x, n, beta)

    def grad_func(x):
        return smooth_max_grad(x, n, beta)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def trust_region_step(h, n, K, delta):
    corr = compute_corr(h)
    sorted_idx = np.argsort(-corr)[:K]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_rows = []
    b_vals = []

    for k in sorted_idx:
        g = corr_gradient(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_vals.append(float(np.dot(g, h)) - fk)

    z_col = np.zeros((n, 1))
    I_n = np.eye(n)
    A_rows.append(np.hstack([I_n, z_col]))
    b_vals.extend(h + delta)
    A_rows.append(np.hstack([-I_n, z_col]))
    b_vals.extend(-h + delta)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0.0, 1.0)] * n + [(None, None)]
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy(), float(res.x[-1])
    except Exception:
        pass
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='revised simplex')
        if res.success:
            return res.x[:n].copy(), float(res.x[-1])
    except Exception:
        pass
    return h.copy(), float(np.max(corr))


def trust_region_optimize(h, n, time_limit, K=150):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    delta = 0.15
    start = time.time()
    stall_count = 0

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        h_step, t_lp = trust_region_step(h, n, K=K, delta=delta)
        c5_step = compute_c5(h_step)
        c5_cur = compute_c5(h)

        if c5_step < best_c5 - 1e-14:
            best_c5 = c5_step
            best_h = h_step.copy()
            h = h_step.copy()
            delta = min(delta * 1.3, 0.35)
            stall_count = 0
        elif c5_step < c5_cur - 1e-14:
            best_h = h_step.copy()
            h = h_step.copy()
            delta = min(delta * 1.1, 0.35)
            stall_count = 0
        else:
            improved = False
            for alpha in [0.7, 0.5, 0.3, 0.15, 0.05]:
                h_try = project_feasible(h + alpha * (h_step - h), n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    delta = min(delta * 1.1, 0.35)
                    improved = True
                    break
                elif c5_try < c5_cur - 1e-14:
                    h = h_try.copy()
                    improved = True
                    break
            if not improved:
                delta *= 0.7
                stall_count += 1
                if stall_count > 30 and delta < 0.01:
                    break

    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h)


def subgradient_optimize(h, n, time_limit):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    prev_g = None
    prev_h = None
    it = 0

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        it += 1

        corr = compute_corr(h)
        K = 80 if it < 200 else 40
        sorted_idx = np.argsort(-corr)[:K]
        c_top = corr[sorted_idx]
        beta_g = 20.0 + 3.0 * min(it, 100)
        w = np.exp(beta_g * (c_top - c_top[0]))
        w_sum = np.sum(w)
        if w_sum < 1e-30:
            w = np.ones(K) / K
        else:
            w /= w_sum

        g = np.zeros(n)
        for i in range(K):
            if w[i] > 1e-12:
                g += w[i] * corr_gradient(h, int(sorted_idx[i]), n)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        if prev_g is not None and prev_h is not None:
            sv = h - prev_h
            yv = g - prev_g
            sy = np.dot(sv, yv)
            ss = np.dot(sv, sv)
            yy = np.dot(yv, yv)
            if yy > 1e-14 and sy > 1e-14:
                step = ss / sy
            elif ss > 1e-14 and sy > 1e-14:
                step = sy / ss
            else:
                step = 2.0 / (gn * (1.0 + 0.05 * it))
            step = np.clip(step, 0.001, 0.3)
        else:
            step = 2.0 / gn
            step = min(step, 0.1)

        best_new_val = best_c5
        best_new_h = h
        for alpha_mult in [1.0, 0.6, 0.4, 0.25, 0.1, 0.05]:
            a = alpha_mult * step
            h_try = project_feasible(h - a * g / gn, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_new_val - 1e-14:
                best_new_val = c5_try
                best_new_h = h_try
                break

        if best_new_val < best_c5:
            best_c5 = best_new_val
            best_h = best_new_h.copy()
            h = best_new_h.copy()
        elif best_new_h is not h:
            h = best_new_h.copy()

        prev_g = g.copy()
        prev_h = h.copy()

    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h)


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coordinate_descent(h, n, time_limit, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        deltas = [0.01, 0.005, 0.02, 0.003, 0.001, -0.005, -0.01, -0.02, 0.002, -0.002]
        for d in deltas:
            if d > 0 and (h[i] < d or h[j] > 1.0 - d):
                continue
            if d < 0 and (h[j] < -d or h[i] > 1.0 + d):
                continue
            h_try = h.copy()
            h_try[i] += d
            h_try[j] -= d
            if h_try[i] < -1e-10 or h_try[i] > 1.0 + 1e-10:
                continue
            if h_try[j] < -1e-10 or h_try[j] > 1.0 + 1e-10:
                continue
            h_try = np.clip(h_try, 0.0, 1.0)
            if abs(np.sum(h_try) - n / 2.0) > 1e-6:
                continue
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                break

    return h, best_c5


def equalize_shifts(h, n, K=8, max_iters=150):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-12:
            break

        g_avg = np.zeros(n)
        for k in idx:
            g_avg += corr_gradient(h, int(k), n)
        g_avg /= K
        gn = np.linalg.norm(g_avg)
        if gn < 1e-14:
            break

        improved = False
        for step in [0.02, 0.04, 0.01, 0.005, 0.03, 0.002, 0.015]:
            for sign in [1, -1]:
                h_try = project_feasible(h + sign * step * g_avg / gn, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    best_h = parent_h.copy()
    best_c5 = compute_c5(parent_h)
    best_n = parent_n

    # Phase 1: Quick SLSQP at parent resolution
    for beta in [300, 1500, 6000]:
        if remaining() < 10:
            break
        h_new = slsqp_solve(best_h, best_n, beta, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Trust-region LP (main optimization)
    if remaining() > 50:
        tr_time = min(300.0, remaining() * 0.35)
        h_tr, c5_tr = trust_region_optimize(best_h, best_n, time_limit=tr_time, K=150)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr

    # Phase 3: SLSQP polish after trust region
    for beta in [8000, 25000, 50000]:
        if remaining() < 8:
            break
        h_new = slsqp_solve(best_h, best_n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: More trust-region
    if remaining() > 40:
        tr_time = min(150.0, remaining() * 0.25)
        h_tr, c5_tr = trust_region_optimize(best_h, best_n, time_limit=tr_time, K=200)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr

    # Phase 5: Multi-start with perturbations
    for trial in range(4):
        if remaining() < 50:
            break

        if trial == 0:
            h_pert = best_h + 0.015 * rng.randn(best_n)
        elif trial == 1:
            freq = 5.0 + trial * 2.0
            h_pert = best_h + 0.02 * np.sin(2.0 * np.pi * freq * np.arange(best_n) / best_n)
        elif trial == 2:
            h_pert = best_h + 0.04 * rng.randn(best_n)
        else:
            freq = 12.0
            h_pert = best_h + 0.01 * np.cos(2.0 * np.pi * freq * np.arange(best_n) / best_n)
            h_pert += 0.01 * rng.randn(best_n)

        h_pert = project_feasible(h_pert, best_n)

        # Trust-region from perturbation
        if remaining() > 30:
            h_tr, c5_tr = trust_region_optimize(h_pert, best_n, time_limit=min(40.0, remaining() * 0.3), K=120)
            if compute_c5(h_tr) < best_c5:
                best_c5 = compute_c5(h_tr)
                best_h = h_tr.copy()
            h_pert = h_tr

        # SLSQP polish on perturbation
        for beta in [5000, 20000]:
            if remaining() < 8:
                break
            h_new = slsqp_solve(h_pert, best_n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_pert = h_new

    # Phase 6: Equalize top shifts
    if remaining() > 15:
        h_eq, c5_eq = equalize_shifts(best_h, best_n, K=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq

    # Phase 7: Try higher resolution
    n_hi = 1024
    if remaining() > 120:
        h_hi = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_hi)

        if remaining() > 40:
            h_sg, c5_sg = subgradient_optimize(h_hi, n_hi, time_limit=min(50.0, remaining() * 0.3))
            if c5_sg < c5_hi:
                c5_hi = c5_sg
                h_hi = h_sg

        for beta in [3000, 12000]:
            if remaining() < 20:
                break
            h_new = slsqp_solve(h_hi, n_hi, beta, maxiter=100)
            c5_new = compute_c5(h_new)
            if c5_new < c5_hi:
                c5_hi = c5_new
                h_hi = h_new

        if remaining() > 20:
            h_tr, c5_tr = trust_region_optimize(h_hi, n_hi, time_limit=min(40.0, remaining() * 0.3), K=100)
            if c5_tr < c5_hi:
                c5_hi = c5_tr
                h_hi = h_tr

        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

    elif remaining() > 60:
        n_hi = 768
        h_hi = project_feasible(interp_to(best_h, best_n, n_hi), n_hi)
        c5_hi = compute_c5(h_hi)

        if remaining() > 25:
            h_sg, c5_sg = subgradient_optimize(h_hi, n_hi, time_limit=min(30.0, remaining() * 0.3))
            if c5_sg < c5_hi:
                c5_hi = c5_sg
                h_hi = h_sg

        for beta in [2000, 8000]:
            if remaining() < 15:
                break
            h_new = slsqp_solve(h_hi, n_hi, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < c5_hi:
                c5_hi = c5_new
                h_hi = h_new

        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi

    # Phase 8: Final coordinate descent
    if remaining() > 10:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, time_limit=min(60.0, remaining() * 0.4), rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 9: Final SLSQP polish
    if best_n <= 800:
        for beta in [50000, 100000, 200000]:
            if remaining() < 5:
                break
            h_new = slsqp_solve(best_h, best_n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new

    # Phase 10: Final trust-region cleanup
    if remaining() > 15 and best_n <= 1024:
        h_tr, c5_tr = trust_region_optimize(best_h, best_n, time_limit=min(30.0, remaining() * 0.4), K=200)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
