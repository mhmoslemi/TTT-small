For L-BFGS-B, I need to handle the equality constraint sum(h) = n/2. I can either:
a) Use SLSQP (handles constraints natively)
b) Use L-BFGS-B with a penalty: f + μ(Σh - n/2)²
c) Parameterize: h[0] = n/2 - Σ_{i>0} h[i], optimize only h[1:n]

Option (c) is clean but the constraint 0 ≤ h[0] ≤ 1 becomes a nonlinear constraint on h[1:n].

Option (b) is simplest. Let me use that.

Actually, SLSQP should be fine for n=512. Let me just make it more effective:
- Use analytic gradient
- Multiple β values (coarse to fine)

import numpy as np
from scipy.optimize import minimize, linprog
import time

_T0 = 0.0
_BUDGET = 1300.0

def _remaining():
    return max(0.0, _BUDGET - (time.time() - _T0))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:offset + npairs] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:offset + npairs]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:a + npairs])
            g[a:a + npairs] -= h[:npairs]
    return g

def smax_func(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -80, 80)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)

def smax_grad(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -80, 80)
    ex = np.exp(sh)
    w = ex / ex.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    # Only sum over non-negligible weights
    thresh = 1e-10
    for k in range(len(w)):
        if w[k] < thresh:
            continue
        g += w[k] * grad_corr_k(x, k, n)
    return g * dx

def slsqp_optimize(h, n, beta, maxiter=300):
    def obj(x):
        return smax_func(x, n, beta)
    def grad(x):
        return smax_grad(x, n, beta)
    def eq_fun(x):
        return np.array([np.sum(x) - n / 2.0])
    def eq_jac(x):
        return np.array([np.ones(n)])
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    return h.copy(), compute_c5(h)

def cutting_plane_lp(h, n, K=30, delta=0.05):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_max = float(corr[idx[0]])
    
    # Variables: h (n) + t (1) = n+1
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_rows = []
    b_rows = []
    
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        # Linearization: c_k(h_new) ≈ fk + g·(h_new - h) ≤ t
        # => g·h_new - t ≤ fk - g·h
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(fk - np.dot(g, h))
    
    # Trust region: ||h_new - h||_∞ ≤ δ
    I = np.eye(n)
    A_rows.append(np.hstack([I, np.zeros((n, 1))]))
    b_rows.extend(h + delta)
    A_rows.append(np.hstack([-I, np.zeros((n, 1))]))
    b_rows.extend(h + delta)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    best_h = h.copy()
    best_t = c_max
    
    for method in ['highs', 'revised simplex', 'simplex']:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method=method)
            if res.success:
                h_lp = res.x[:n]
                t_lp = res.x[-1]
                if t_lp < best_t:
                    best_t = t_lp
                    best_h = h_lp.copy()
                break
        except Exception:
            continue
    
    return best_h, best_t

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgradient_step(h, n, K=20, beta=50.0, step_scale=0.1):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, k, n)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h.copy()
    dx = 2.0 / n
    step = step_scale * dx / gn
    h_new = project_feasible(h - step * g, n)
    return h_new

def topk_local_search(h, n, K=8, max_iters=500, lr=0.02):
    """Nelder-Mead-like search on top-K correlations."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if _remaining() < 5:
            break
        corr = compute_corr(best_h)
        top_K = np.argsort(-corr)[:K]
        
        # Compute average gradient of top K
        g = np.zeros(n)
        for k in top_K:
            g += grad_corr_k(best_h, k, n)
        g /= K
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        g_dir = g / g_norm
        
        # Try multiple step sizes
        improved = False
        for step in [lr * 0.1, lr * 0.3, lr, lr * 3, lr * 10]:
            h_try = project_feasible(best_h - step * g_dir, n)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try
                improved = True
                break
        
        if not improved:
            # Try opposite direction (in case we're at a boundary)
            for step in [lr * 0.1, lr * 0.3, lr]:
                h_try = project_feasible(best_h + step * g_dir, n)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_h = h_try
                    best_c5 = c_try
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_c5
