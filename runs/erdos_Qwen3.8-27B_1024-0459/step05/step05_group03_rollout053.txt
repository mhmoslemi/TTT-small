import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(150):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-18:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -60.0, 0.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -60.0, 0.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    return corr_grad_weighted(h, w * dx)


def slsqp_run(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        if compute_c5(h_res) < compute_c5(h0) + 1e-15:
            return h_res
    except Exception:
        pass
    return h0


def slsqp_trust(h, n, beta, maxiter=400):
    """Trust-constr method as alternative to SLSQP."""
    h0 = project_feasible(h, n)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n), 'hess': lambda x, v: np.zeros((n, n))}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: smooth_obj(x, beta), h0,
                       jac=lambda x: smooth_grad(x, beta),
                       method='trust-constr', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'verbose': 0})
        h_res = project_feasible(res.x, n)
        if compute_c5(h_res) < compute_c5(h0) + 1e-15:
            return h_res
    except Exception:
        pass
    return h0


def nesterov_projected(h, n, beta, max_iters=3000):
    """Nesterov accelerated projected gradient with log-sum-exp smooth max."""
    h = project_feasible(h, n)
    h_curr = h.copy()
    h_next = h.copy()
    v = np.zeros(n)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0

        lr = 1.0 / (beta * 0.5 * (it + 1))

        h_curr = h_curr_new
        t_val = t_new

        if it % 200 == 0 and it > 0:

def subgrad_topk(h, n, top_k=5, max_iters=1500):
    """Subgradient method focusing on top-k correlated lags."""
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.1
    momentum = np.zeros(n)
    mu = 0.8

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        c_min_top = c_top.min()
        w_top = np.maximum(c_top - c_min_top, 1e-10)
        w_top = w_top ** 2
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        dx = 2.0 / n
        weights *= dx
        g = corr_grad_weighted(h, weights)
        g -= np.mean(g)

        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

        step = max(0.001, step * 0.999)
        if it % 200 == 0:
            mu = min(0.92, mu + 0.02)

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):

        for beta in [5000, 20000, 80000]:
            if remaining() < 30:
                break
            h_nes, c5_nes = nesterov_projected(h, n, beta, max_iters=2000)
            if c5_nes < best_c5 - 1e-15:
                best_c5 = c5_nes
                best_h = h_nes.copy()
                h = h_nes.copy()

    # Phase 4: Subgradient top-k
    if remaining() > 40:
        for tk in [3, 5, 7, 10, 15]:
            if remaining() < 15:
                break
            h_sg, c5_sg = subgrad_topk(h, n, top_k=tk, max_iters=1200)
            if c5_sg < best_c5 - 1e-15:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    # Phase 5: Binary greedy + SA exploration
    if remaining() > 60:
        h_bin = round_to_binary(best_h, best_n)
        # Greedy first
        h_g, c5_g = greedy_binary_ls(h_bin, best_n, max_rounds=25)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

        # SA
        if remaining() > 50:
            n_sa = min(30000, int(remaining() * 4))
            h_sa, c5_sa = simulated_annealing_binary(h_g, best_n, T_start=0.8, T_end=0.002, max_iters=n_sa)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

        # Multi-adjacent swap refinement
        if remaining() > 20:
            h_ma, c5_ma = multi_adjacent_swap(best_h, best_n, max_rounds=15)
            if c5_ma < best_c5:
                best_c5 = c5_ma
                best_h = h_ma.copy()

    # Phase 6: SLSQP polish from best binary
    if remaining() > 30:
        h_pol = round_to_binary(best_h, best_n).astype(float)
        for beta in [20000, 80000, 200000, 500000]:
            if remaining() < 10:
                break
            h_new = slsqp_run(h_pol, best_n, beta, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_pol = h_new.copy()

    # Phase 7: Multi-resolution exploration
    n_candidates = [384, 448, 480, 512, 640, 768, 256, 320]
    n_candidates = sorted(set([x for x in n_candidates if x != best_n and 128 <= x <= 1024 and x % 2 == 0]),
                         key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:4]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_interp = project_feasible(h_interp, n_t)
        h_opt = h_interp

        for beta in [2000, 10000, 50000, 200000]:
            if remaining() < 15:
                break
            h_opt = slsqp_run(h_opt, n_t, beta, maxiter=min(400, int(remaining() // 8)))
        c5_opt = compute_c5(h_opt)

        if remaining() > 25:
            h_bin_t = round_to_binary(h_opt, n_t)
            h_b_t, c5_b_t = greedy_binary_ls(h_bin_t, n_t, max_rounds=12)
            if c5_b_t < c5_opt:
                c5_opt = c5_b_t
                h_opt = h_b_t.copy()

        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 8: Diverse restarts
    if remaining() > 80:
        starts = []
        # Random binary starts
        for s in range(4):
            starts.append(make_diverse_start(best_n, seed + 100 + s))

        # Block patterns
        for period in [4, 6, 8, 12, 16]:
            bp = np.zeros(best_n)
            for i in range(best_n):
                if (i % (2 * period)) < period:
                    bp[i] = 1.0
            ones_cnt = int(np.sum(bp))
            tgt = best_n // 2
            if ones_cnt != tgt:
                h_s = round_to_binary(bp, best_n)
                starts.append(h_s)

        for h_s in starts:
            if remaining() < 40:
                break
            h_s = project_feasible(h_s, best_n)
            # Quick optimization
            h_s = slsqp_run(h_s, best_n, 10000, maxiter=300)
            c5_s = compute_c5(h_s)
            if remaining() > 20:
                h_sb = round_to_binary(h_s, best_n)
                h_sb, c5_sb = greedy_binary_ls(h_sb, best_n, max_rounds=8)
                if c5_sb < c5_s:
                    c5_s = c5_sb
                    h_s = h_sb
            if c5_s < best_c5 - 1e-15:
                best_c5 = c5_s
                best_h = h_s.copy()

    # Phase 9: Final intensive polish at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [100000, 300000, 800000, 1500000]:
        if remaining() < 8:
            break
        h_new = slsqp_run(h, n, beta, maxiter=min(600, int(remaining() // 6)))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 10: Final binary micro-adjustments
    if remaining() > 25:
        h_bin_fin = round_to_binary(best_h, best_n)
        h_mf, c5_mf = multi_adjacent_swap(h_bin_fin, best_n, max_rounds=10)
        if c5_mf < best_c5:
            best_c5 = c5_mf
            best_h = h_mf.copy()

        if remaining() > 20:
            h_sf, c5_sf = simulated_annealing_binary(h_bin_fin, best_n, T_start=0.3, T_end=0.001, max_iters=8000)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()

        # Final SLSQP from best binary
        if remaining() > 15:
            h_pfin = round_to_binary(best_h, best_n).astype(float)
            for beta in [200000, 600000]:
                if remaining() < 8:
                    break
                h_pfin = slsqp_run(h_pfin, best_n, beta, maxiter=400)
            c5_pf = compute_c5(h_pfin)
            if c5_pf < best_c5 - 1e-15:
                best_c5 = c5_pf
                best_h = h_pfin.copy()

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
