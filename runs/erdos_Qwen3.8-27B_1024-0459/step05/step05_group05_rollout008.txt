        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

            no_improve = 0

                best_max = cur_max
                best_h = h.copy()
        else:
            break
    
    return best_h, best_max * dx_v

    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if hasattr(res, 'fun') and res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
        if res.success:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def binary_full_sweep(h, n):
    """One full best-improvement sweep over all single swaps."""
    corr = compute_corr(h)
    max_val = float(np.max(corr))
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, max_val, False
    
    best_delta = 1e10
    best_swap = None
    
    for a in ones:
        h[a] = 0
        for b in zeros:
            h[b] = 1
            nc = float(np.max(compute_corr(h)))
            h[b] = 0
            h[a] = 1
            delta = nc - max_val
            if delta < best_delta:
                best_delta = delta
                best_swap = (a, b)
    
    if best_swap is not None and best_delta < -1e-14:
        a, b = best_swap
        h[a] = 0
        h[b] = 1
        return h, best_delta + max_val, True
    return h, max_val, False


def binary_local_search(h_bin, n, rng, time_limit=60):
    """Intensive binary local search with full sweeps and SA escape."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - n_half]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:n_half - n_ones]]] = 1
    
    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    sweep_count = 0
    
    while time.time() - start < time_limit and remaining() > 5:
        sweep_count += 1
        if sweep_count > 40:
            break
        
        h, cur_max, improved = binary_full_sweep(h, n)
        if not improved:
            # Try SA escape: random 2-3 swaps
            if sweep_count > 5:
                n_flip = rng.randint(2, 4)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_flip):
                    if len(ones_now) < 2 or len(zeros_now) < 2:
                        break
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0
                    h[j] = 1
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                cur_max = float(np.max(compute_corr(h)))
            else:
                break
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    
    return best_h, best_max * dx_v


def active_set_descent(h, n, max_iters=300, top_k=6):
    """Subgradient-like descent on the active (top-k) correlations."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.02
    stall = 0
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = float(np.max(corr))
        
        # Adaptive top_k
        idx_sorted = np.argsort(-corr)
        adaptive_k = min(top_k, len(corr))
        # Only include correlations within 0.5 of max
        idx = idx_sorted[:adaptive_k]
        threshold = corr[idx[0]] - 0.5
        idx = [k for k in idx if corr[k] >= threshold]
        if len(idx) == 0:
            idx = [idx_sorted[0]]
        
        # Average gradient of active set
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            g += corr_gradient_at(h, off, n)
        g /= len(idx)
        
        # Project to tangent space (sum constraint)
        g -= np.mean(g)
        
        # Handle boundaries: push inward
        at_upper = h > 0.999
        at_lower = h < 0.001
        g[at_upper] = np.minimum(g[at_upper], 0.0)
        g[at_lower] = np.maximum(g[at_lower], 0.0)
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        g_dir = g / gnorm
        
        # Line search: try multiple step sizes
        improved = False
        for s in [step, step * 0.5, step * 0.75, step * 1.5, step * 0.25, step * 2.0]:
            h_new = project_feasible(h - s * g_dir, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c_new
                best_h = h.copy()
                improved = True
                step = min(s * 1.1, 0.05)
                stall = 0
                break
        
        if not improved:
            stall += 1
            step *= 0.7
            if step < 1e-7 or stall > 50:
                break
    
    return best_h, best_c5


def top_shifts_equalize(h, n, top_k=5, max_iters=200):
    """Level the top correlations while reducing the max."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    step = 0.01
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        
        if c_max - c_min_top < 1e-10:
            break
        
        gt = corr_gradient_at(h, idx[0] - (n - 1), n)
        gb = corr_gradient_at(h, idx[-1] - (n - 1), n)
        
        # Combine: reduce max + level
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += corr_gradient_at(h, k - (n - 1), n)
        g_avg /= len(idx)
        
        d = -0.7 * g_avg + 0.3 * (gb - gt)
        d -= np.mean(d)
        
        # Boundary handling
        d[h > 0.999] = np.minimum(d[h > 0.999], 0.0)
        d[h < 0.001] = np.maximum(d[h < 0.001], 0.0)
        
        dn = np.linalg.norm(d)
        if dn < 1e-15:
            break
        d /= dn
        
        improved = False
        for s in [step, step * 0.5, step * 2, step * 0.25]:
            ht = project_feasible(h + s * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 1e-7:
                break
    
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def partial_relax(h_bin, n, n_boundary=20):
    """Allow positions near the top of the sorted h to take fractional values."""
    h = h_bin.copy()
    # Sort by value (1s first)
    sort_idx = np.argsort(-h)
    # Allow the boundary positions to be fractional
    boundary = sort_idx[n // 2 - n_boundary:n // 2 + n_boundary]
    h[boundary] = 0.5
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Progressive SLSQP on parent resolution
    for beta in [200, 1000, 5000, 20000, 80000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Active set descent
    if remaining() > 20:
        h_as, c5_as = active_set_descent(h, n, max_iters=400, top_k=6)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as.copy()
            h = h_as.copy()
        # Different top_k
        h_as2, c5_as2 = active_set_descent(h, n, max_iters=300, top_k=10)
        if c5_as2 < best_c5:
            best_c5 = c5_as2
            best_h = h_as2.copy()
            h = h_as2.copy()

    # Phase 3: Top shifts equalization
    if remaining() > 15:
        for tk in [4, 6, 8, 12]:
            if remaining() < 10:
                break
            h_eq, c5_eq = top_shifts_equalize(h, n, top_k=tk, max_iters=200)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 4: Binary local search (full best-improvement)
    if remaining() > 40:
        h_bin = binarize(h, n)
        sa_time = min(120, max(15, int(remaining() * 0.3)))
        h_b, c5_b = binary_local_search(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
        # Partial relaxation from binary
        if remaining() > 20:
            h_pr = partial_relax(h_b, n, n_boundary=20)
            h_pr = slsqp_smooth(h_pr, n, 5000, 300)
            c5_pr = compute_c5(h_pr)
            if c5_pr < best_c5:
                best_c5 = c5_pr
                best_h = h_pr.copy()
            
            h_pr2 = partial_relax(h_b, n, n_boundary=40)
            h_pr2 = slsqp_smooth(h_pr2, n, 10000, 300)
            c5_pr2 = compute_c5(h_pr2)
            if c5_pr2 < best_c5:
                best_c5 = c5_pr2
                best_h = h_pr2.copy()

    # Phase 5: SLSQP from best with very high beta
    n = best_n
    h = best_h.copy()
    for beta in [20000, 80000, 200000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 6: Resolution sweep
    if remaining() > 50:
        n_candidates = []
        for delta in [-4, -2, 2, 4, 6, 8, -8, 12, -12]:
            nt = best_n + delta
            if 160 <= nt <= 700 and nt % 2 == 0:
                n_candidates.append(nt)
        n_candidates = list(dict.fromkeys(n_candidates))

        for n_t in n_candidates[:6]:
            if remaining() < 15:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t
            
            h_t = slsqp_smooth(h_t, n_t, 5000, 300)
            c5_t = compute_c5(h_t)
            h_t, c5_t2 = active_set_descent(h_t, n_t, max_iters=200, top_k=6)
            c5_t = min(c5_t, c5_t2)
            if c5_t2 < c5_t:
                h_t = h_t
            
            # Quick binary check
            if remaining() > 25 and c5_t < best_c5 + 0.002:
                h_tb = binarize(h_t, n_t)
                h_tb2, c5_tb = binary_local_search(h_tb, n_t, rng, time_limit=min(30, int(remaining() * 0.3)))
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb2.copy()
            
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 7: Diverse starts
    if remaining() > 60:
        # Thue-Morse start
        h_tm = np.zeros(best_n)
        for i in range(best_n):
            h_tm[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
        n_ones_tm = int(np.sum(h_tm))
        n_half = best_n // 2
        if n_ones_tm != n_half:
            if n_ones_tm > n_half:
                oi = np.where(h_tm == 1)[0]
                h_tm[oi[:n_ones_tm - n_half]] = 0
            else:
                zi = np.where(h_tm == 0)[0]
                h_tm[zi[:n_half - n_ones_tm]] = 1
        h_tm = slsqp_smooth(h_tm, best_n, 5000, 300)
        h_tm, c5_tm = active_set_descent(h_tm, best_n, 200, 6)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()

        # Random perturbation of best
        if remaining() > 30:
            h_p = best_h + rng.randn(best_n) * 0.05
            h_p = project_feasible(h_p, best_n)
            h_p = slsqp_smooth(h_p, best_n, 20000, 400)
            h_p, c5_p = active_set_descent(h_p, best_n, 200, 8)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

        # Another random start
        if remaining() > 25:
            h_r = rng.uniform(0, 1, best_n)
            h_r = project_feasible(h_r, best_n)
            h_r = slsqp_smooth(h_r, best_n, 10000, 400)
            h_r, c5_r = active_set_descent(h_r, best_n, 200, 6)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    
    for beta in [50000, 200000, 500000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final active set + equalization
    for tk in [4, 6, 8]:
        if remaining() < 5:
            break
        h_f, c5_f = active_set_descent(h, n, max_iters=300, top_k=tk)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            h = h_f.copy()
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=tk, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary check
    if remaining() > 15:
        h_bin = binarize(best_h, best_n)
        h_b, c5_b = binary_local_search(h_bin, best_n, rng, time_limit=min(30, int(remaining() * 0.5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            # Quick continuous polish
            h_c = slsqp_smooth(h_b, best_n, 100000, 400)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            eval_c5 = float(eval_c5)
            if eval_c5 < best_c5:
                best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
