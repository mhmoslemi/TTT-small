def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, n):
    """Project h onto {0 <= x <= 1, sum x = n/2}"""
    h = np.array(h, dtype=float).flatten()
    h = np.clip(h, 0.0, 1.0)

    lo, hi = -1.0, 1.0

        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fine adjustment
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top-k lags
    idx = np.argsort(-corr)[:top_k]
    # Average gradient of top-k lags
    g = np.zeros(n)
    for k in idx:

        if off >= 0:

            g[a:] -= h[:np_]

    # Project
    h_new = project_box_sum(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 500.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_grad_component(h, k, n):
    """Gradient of corr[k] w.r.t. h for a single lag k"""
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def project_box_sum(h, n):
    """Project h onto {0 <= x <= 1, sum x = n/2} using bisection"""
    h = np.array(h, dtype=float).flatten()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    # Fine adjustment on free variables
    diff = target - proj.sum()
    if abs(diff) > 1e-13:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    idx = np.argsort(-corr)[:top_k]
    g = np.zeros(n)
    for k in idx:
        g += corr_grad_component(h, k, n)
    g /= len(idx)
    return g, corr.copy()

def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        g += wk * corr_grad_component(x, k, n)
    return g

def slsqp_opt(h, n, beta, maxiter=800):

    _CUR_N = n
    _SM_BETA = beta
    h0 = project_box_sum(h, n)

def theta_morse(n):

        cnt = 0

        if remaining_time() < 15:
            break
        h_new = slsqp_opt(best_h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Equalize + SLSQP cycle
    for cyc in range(3):
        if remaining_time() < 20:
            break
        h_eq, c5_eq = equalize_topk(best_h, best_n, top_k=6, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
        for beta in [30000, 200000, 1000000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_opt(best_h, best_n, beta, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 4: Try different starting points
    starts = []
    # Thue-Morse
    starts.append(theta_morse(best_n))
    # Reversed parent
    starts.append(best_h[::-1].copy())
    # 1 - parent
    starts.append(1.0 - best_h)
    starts[-1] = project_box_sum(starts[-1], best_n)
    # Random balanced
    h_r = np.zeros(best_n)
    h_r[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
    starts.append(h_r)
    # Shifted
    starts.append(np.roll(best_h, best_n // 6))
    starts[-1] = project_box_sum(starts[-1], best_n)
    
    for h_s in starts:
        if remaining_time() < 25:
            break
        h_s = project_box_sum(h_s, best_n)
        h_s, c5_s = projected_subgradient(h_s, best_n, time_limit=20, top_k=5, gamma0=30.0)
        h_s, c5_s = slsqp_opt(h_s, best_n, 20000, maxiter=500)
        c5_s = compute_c5(h_s)
        h_s, c5_s = equalize_topk(h_s, best_n, top_k=5, max_iters=300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5 - 1e-14:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 5: Binary combinatorial search
    if remaining_time() > 40:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(h_bin.sum())
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0
        sa_time = min(90, max(15, int(remaining_time() * 0.25)))
        h_b, c5_b = binary_pair_search(h_bin, best_n, time_limit=sa_time, rng=rng)
        
        # Refine binary result to continuous
        h_b_c = h_b.copy()
        for beta in [20000, 100000, 500000]:
            if remaining_time() < 10:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=500)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = equalize_topk(h_b_c, best_n, top_k=5, max_iters=300)
        c5_b_c = compute_c5(h_b_c)
        
        if c5_b_c < best_c5 - 1e-14:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 6: Try adjacent resolutions with intensive optimization
    if remaining_time() > 60:
        for delta in [-32, -16, 16, 32, 64]:
            if remaining_time() < 30:
                break
            n_t = best_n + delta
            if n_t < 128 or n_t > 1024 or n_t % 2 != 0:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_box_sum(h_t, n_t)
            h_t, c5_t = slsqp_opt(h_t, n_t, 10000, maxiter=500)
            h_t, c5_t = equalize_topk(h_t, n_t, top_k=5, max_iters=300)
            h_t, c5_t = slsqp_opt(h_t, n_t, 200000, maxiter=500)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5 - 1e-14:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [200000, 1000000, 5000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = h_new.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = equalize_topk(h, n, top_k=8, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 8: Coordinate refinement
    if remaining_time() > 10:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=20)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP polish with very high beta
    if remaining_time() > 15:
        h_final = slsqp_opt(best_h, n, 5000000, maxiter=1000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Subgradient polish
    if remaining_time() > 15:
        h_sub, c5_sub = projected_subgradient(best_h, n, time_limit=min(30, remaining_time() - 5), top_k=8, gamma0=10.0)
        if c5_sub < best_c5 - 1e-14:
            best_c5 = c5_sub
            best_h = h_sub.copy()
        # Final SLSQP after subgradient
        if remaining_time() > 10:
            h_f = slsqp_opt(best_h, n, 1000000, maxiter=500)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5 - 1e-14:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Ensure feasibility
    n = best_n
    best_h = project_box_sum(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
