        return m, np.zeros_like(corr)

    shifted = np.minimum(shifted, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = m + np.log(wsum) / beta

    """Minimize max_k [base[k] + A[k]*t] over t in [0,1]."""

        max_val = np.max(vals)

        subgrad = np.mean(A[active])

        sorted_vals = np.sort(vals)[::-1]

            t = t_new
            break
        t = t_new
