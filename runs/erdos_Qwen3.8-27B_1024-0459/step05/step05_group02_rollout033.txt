def compute_corr(h):

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def tabu_search_binary(h_bin, n, max_iters=50000, tabu_len=100, seed=42):
    """Tabu search for binary sequences."""
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            to_remove = rng.choice(ones, n_ones - n_half, replace=False)
            h[to_remove] = 0
        else:
            zeros = np.where(h == 0)[0]
            to_add = rng.choice(zeros, n_half - n_ones, replace=False)
            h[to_add] = 1
    
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    # Tabu list: store (position, iteration_when_it_becomes_non_tabu)
    tabu = {}  # position -> iteration when it's no longer tabu
    frequency = np.zeros(n, dtype=int)  # how often each position has been flipped
    
    for it in range(max_iters):
        if remaining() < 2:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose swap: prefer non-tabu moves, with frequency-based tiebreaking
        # Limit candidates for speed
        n_candidates = min(50, len(ones))
        cand_ones = ones[rng.randint(len(ones), n_candidates)]
        cand_zeros = zeros[rng.randint(len(zeros), n_candidates)]
        
        best_delta = float('inf')
        best_i, best_j = -1, -1
        best_aspiration = False
        
        for i in cand_ones:
            for j in cand_zeros:
                i_tabu = tabu.get(i, -1) > it
                j_tabu = tabu.get(j, -1) > it
                
                # Aspiration: accept tabu move if it improves best
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                h[i] = 1
                h[j] = 0
                
                delta = new_c5 - cur_c5
                aspiration = new_c5 < best_c5
                
                if (not i_tabu and not j_tabu) or aspiration:
                    # Score: prefer lower delta, break ties by lower frequency
                    score = delta + 0.001 * (frequency[i] + frequency[j])
                    if score < best_delta:
                        best_delta = score
                        best_i, best_j = i, j
                        best_aspiration = aspiration
        
        if best_i < 0:
            # All moves are tabu, clear tabu list
            tabu.clear()
            continue
        
        # Execute best move
        h[best_i] = 0
        h[best_j] = 1
        frequency[best_i] += 1
        frequency[best_j] += 1
        tabu[best_i] = it + tabu_len
        tabu[best_j] = it + tabu_len
        
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5

def structured_init(n, pattern='thue_morse'):
    """Create structured initial binary sequences."""
    h = np.zeros(n)
    n_half = n // 2
    
    if pattern == 'thue_morse':
        # Thue-Morse sequence
        tm = np.zeros(n, dtype=int)
        for i in range(1, n):
            tm[i] = tm[i >> 1] ^ (i & 1)
        h = tm.astype(float)
        
    elif pattern == 'alternating':
        h[::2] = 1.0
        
    elif pattern == 'golden':
        # Golden ratio-based
        phi = (1 + np.sqrt(5)) / 2
        for i in range(n):
            if ((i + 1) * phi) % 1.0 < 0.5:
                h[i] = 1.0
    
    elif pattern == 'blocks':
        # Alternating blocks of size ~2
        block = max(1, n // (2 * n_half))
        for i in range(n):
            h[i] = 1.0 if (i // block) % 2 == 0 else 0.0
    
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h == 1)[0]
            rng = np.random.RandomState(0)
            to_remove = rng.choice(ones, n_ones - n_half, replace=False)
            h[to_remove] = 0
        else:
            zeros = np.where(h == 0)[0]
            rng = np.random.RandomState(0)
            to_add = rng.choice(zeros, n_half - n_ones, replace=False)
            h[to_add] = 1
    
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
