One more thought: for the continuous refinement, instead of SLSQP (which was slow in the previous program), I'll use a simpler projected gradient method with the subgradient of the max function.

The subgradient of f(h) = max_k corr[k] at h is: corr[h_i] for the index i that achieves the max. More precisely, if multiple indices achieve the max, the subgradient is the convex hull of their individual gradients.

The gradient of corr[k] with respect to h[i]:
corr[k] = sum_m h[m] * (1-h[m+k]) (for the appropriate range)
∂corr[k]/∂h[i] = (1-h[i+k]) - h[i-k] (with appropriate boundary handling, same as before)

So the subgradient of f at h is: sum_{k in argmax} w_k * grad_corr[k] where w_k ≥ 0, sum w_k = 1.

A simple approach: use the gradient of the single worst lag, take a projected gradient step, and repeat.

            for idx in ones[n_half:]:
                h[idx] = 0.0
        else:
            zeros = np.where(h == 0.0)[0]
            for idx in zeros[:n_half - n_ones]:
                h[idx] = 1.0
    return h


def thue_morse_binary(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return ensure_binary(h, n)


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def generate_candidates(h, n, top_t=5, n_random=3):
    """Generate swap candidates targeting worst lags."""
    corr = compute_corr(h)
    curr_max = np.max(corr)
    worst_idx = np.argsort(-corr)[:top_t]
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return [], curr_max
    
    candidates = []
    seen = set()
    
    for ki in worst_idx:
        k = int(ki - (n - 1))
        
        # Compute A[i] for i in ones: effect of h[i] 1->0 on corr[ki]
        A = np.zeros(len(ones))
        ones_k_plus = ones + k
        ones_k_minus = ones - k
        mask_p = (ones_k_plus >= 0) & (ones_k_plus < n)
        mask_m = (ones_k_minus >= 0) & (ones_k_minus < n)
        A[mask_p] -= (1.0 - h[ones_k_plus[mask_p]])
        A[mask_m] += h[ones_k_minus[mask_m]]
        
        # Compute B[j] for j in zeros: effect of h[j] 0->1 on corr[ki]
        B = np.zeros(len(zeros))
        zeros_k_plus = zeros + k
        zeros_k_minus = zeros - k
        mask_p2 = (zeros_k_plus >= 0) & (zeros_k_plus < n)
        mask_m2 = (zeros_k_minus >= 0) & (zeros_k_minus < n)
        B[mask_p2] += (1.0 - h[zeros_k_plus[mask_p2]])
        B[mask_m2] -= h[zeros_k_minus[mask_m2]]
        
        # Best swap for this lag: minimize A[i] + B[j]
        i_min = np.argmin(A)
        j_min = np.argmin(B)
        i = int(ones[i_min])
        j = int(zeros[j_min])
        key = (min(i, j), max(i, j))
        if key not in seen:
            seen.add(key)
            candidates.append((i, j))
        
        # Also try second best
        A_sorted = np.argsort(A)
        B_sorted = np.argsort(B)
        for a_idx in A_sorted[:2]:
            for b_idx in B_sorted[:2]:
                i = int(ones[a_idx])
                j = int(zeros[b_idx])
                key = (min(i, j), max(i, j))
                if key not in seen:
                    seen.add(key)
                    candidates.append((i, j))
    
    # Add some random candidates
    rng_local = np.random.RandomState(int(curr_max * 1000) % 2**31)
    for _ in range(n_random):
        i = int(ones[rng_local.randint(len(ones))])
        j = int(zeros[rng_local.randint(len(zeros))])
        key = (min(i, j), max(i, j))
        if key not in seen:
            seen.add(key)
            candidates.append((i, j))
    
    return candidates, curr_max


def binary_local_search(h_init, n, time_limit, rng, top_t=5):
    """Fast binary local search with SA escape."""
    h = ensure_binary(h_init.copy(), n)
    dx = 2.0 / n
    
    corr = compute_corr(h)
    best_max_corr = int(np.max(corr))
    best_h = h.copy()
    best_c5 = best_max_corr * dx
    
    start = time.time()
    no_improve = 0
    T_sa = 3.0
    sa_floor = 0.05
    top_t_current = top_t
    
    while time.time() - start < time_limit:
        corr = compute_corr(h)
        curr_max = int(np.max(corr))
        
        # Generate candidates
        candidates, _ = generate_candidates(h, n, top_t=top_t_current, n_random=3)
        
        if len(candidates) == 0:
            break
        
        # Evaluate candidates
        best_cand = None
        best_cand_max = curr_max
        for (i, j) in candidates:
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = int(np.max(new_corr))
            h[i] = 1.0
            h[j] = 0.0
            if new_max < best_cand_max:
                best_cand_max = new_max
                best_cand = (i, j)
        
        if best_cand is not None:
            i, j = best_cand
            h[i] = 0.0
            h[j] = 1.0
            if best_cand_max < best_max_corr:
                best_max_corr = best_cand_max
                best_h = h.copy()
                best_c5 = best_max_corr * dx
            no_improve = 0
        else:
            # No improvement - SA step
            no_improve += 1
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            
            if no_improve > 3:
                # Try SA
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h[i] = 0.0
                h[j] = 1.0
                new_corr = compute_corr(h)
                new_max = int(np.max(new_corr))
                delta = new_max - curr_max
                
                if delta <= 0 or (T_sa > sa_floor and rng.rand() < np.exp(-delta / T_sa)):
                    if new_max < best_max_corr:
                        best_max_corr = new_max
                        best_h = h.copy()
                        best_c5 = best_max_corr * dx
                else:
                    h[i] = 1.0
                    h[j] = 0.0
                
                T_sa = max(sa_floor, T_sa * 0.998)
            
            if no_improve > 100:
                # Perturb: random multi-flip
                n_flips = rng.randint(3, min(15, len(ones) // 2))
                for _ in range(n_flips):
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    h[i] = 0.0
                    h[j] = 1.0
                T_sa = 3.0
                no_improve = 0
    
    return best_h, best_c5


def binary_sa_restart(h_init, n, time_limit, rng, num_restarts=None):
    """Binary local search with multiple restarts."""
    h_best = ensure_binary(h_init.copy(), n)
    c5_best = compute_c5(h_best)
    dx = 2.0 / n
    start = time.time()
    
    if num_restarts is None:
        num_restarts = max(2, int(time_limit / 60))
    
    per_restart = time_limit / num_restarts
    
    for r in range(num_restarts):
        if time.time() - start >= time_limit - 2:
            break
        t_remaining = min(per_restart, time_limit - (time.time() - start) - 1)
        if t_remaining < 3:
            break
        
        if r == 0:
            h_cur = h_best.copy()
        else:
            # Perturb best solution
            h_cur = h_best.copy()
            n_flips = rng.randint(5, 20)
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            for _ in range(n_flips):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                h_cur[i] = 0.0
                h_cur[j] = 1.0
        
        h_res, c5_res = binary_local_search(h_cur, n, t_remaining, rng, top_t=5)
        if c5_res < c5_best:
            c5_best = c5_res
            h_best = h_res
    
    return h_best, c5_best


def continuous_refine(h_init, n, time_limit):
    """Projected subgradient method for continuous refinement."""
    h = project_feasible(h_init.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    lr = 0.002
    patience = 0
    
    while time.time() - start < time_limit - 1:
        corr = compute_corr(h)
        k_max = int(np.argmax(corr))
        k = k_max - (n - 1)
        
        # Gradient of corr[k_max] w.r.t. h
        g = np.zeros(n)
        for i in range(n):
            gi = 0.0
            if 0 <= i + k < n:
                gi += (1.0 - h[i + k])
            if 0 <= i - k < n:
                gi -= h[i - k]
            g[i] = gi
        
        # Try different step sizes
        improved = False
        for scale in [lr, lr * 0.5, lr * 2.0, lr * 0.25]:
            h_try = h - scale * g
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_h = h_try.copy()
                best_c5 = c5_try
                improved = True
                break
        
        if not improved:
            h = project_feasible(h - lr * 0.1 * g, n)
            patience += 1
            if patience > 50:
                h = best_h.copy()
                lr *= 0.8
                patience = 0
        else:
            patience = 0
    
    return best_h, best_c5


def slsqp_opt(h_init, n, beta, maxiter=300):
    """SLSQP with smooth max."""
    global _CUR_N, _SM_BETA
    h0 = project_feasible(h_init.copy(), n)
    
    obj_val = _sm_obj(h0)
    
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < obj_val - 1e-14:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h0


_CUR_N = 0
_SM_BETA = 1000.0


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except Exception:
        h_parent = thue_morse_binary(384)
    n_parent = len(h_parent)
    
    best_h = h_parent.copy()
    best_n = n_parent
    best_c5 = compute_c5(h_parent)
    
    # === Phase 1: Binary local search at parent resolution ===
    h_bin = ensure_binary(h_parent, n_parent)
    t1 = min(remaining_time() * 0.40, 400)
    h_b1, c5_b1 = binary_sa_restart(h_bin, n_parent, t1, rng, num_restarts=4)
    if c5_b1 < best_c5:
        best_c5 = c5_b1
        best_h = h_b1
        best_n = n_parent
    
    # === Phase 2: Thue-Morse start ===
    if remaining_time() > 60:
        h_tm = thue_morse_binary(n_parent)
        t2 = min(remaining_time() * 0.20, 150)
        h_b2, c5_b2 = binary_sa_restart(h_tm, n_parent, t2, rng, num_restarts=3)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2
            best_n = n_parent
    
    # === Phase 3: Try different resolutions ===
    res_candidates = []
    for mult in [0.75, 1.0, 1.25, 1.5, 2.0]:
        nr = int(n_parent * mult)
        if nr % 2 == 0 and 128 <= nr <= 1024:
            res_candidates.append(nr)
    # Add specific values
    for nr in [256, 320, 384, 448, 512, 640, 768]:
        if nr % 2 == 0 and nr not in res_candidates:
            res_candidates.append(nr)
    res_candidates = sorted(set(res_candidates))
    res_candidates.remove(n_parent) if n_parent in res_candidates else None
    res_candidates.sort(key=lambda x: abs(x - n_parent))
    
    for nr in res_candidates[:5]:
        if remaining_time() < 40:
            break
        # Interpolate best solution to new resolution
        h_interp = interp_to(best_h, best_n, nr)
        h_bin_r = ensure_binary(h_interp, nr)
        
        t_r = min(remaining_time() * 0.15, 120)
        h_br, c5_br = binary_sa_restart(h_bin_r, nr, t_r, rng, num_restarts=2)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br
            best_n = nr
    
    # === Phase 4: Continuous refinement on best binary ===
    if remaining_time() > 20:
        n = best_n
        h_cont = project_feasible(best_h, n)
        t4 = min(remaining_time() * 0.25, 120)
        h_c, c5_c = continuous_refine(h_cont, n, t4)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
    
    # === Phase 5: SLSQP polish ===
    if remaining_time() > 30:
        n = best_n
        for beta in [5000, 50000, 200000]:
            if remaining_time() < 15:
                break
            _CUR_N = n
            _SM_BETA = beta
            h_s = slsqp_opt(best_h, n, beta, maxiter=400)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5 - 1e-14:
                best_c5 = c5_s
                best_h = h_s
    
    # === Phase 6: Final binary search squeeze ===
    if remaining_time() > 30:
        n = best_n
        h_final_bin = ensure_binary(best_h, n)
        t6 = min(remaining_time() * 0.5, 200)
        h_fb, c5_fb = binary_local_search(h_final_bin, n, t6, rng, top_t=8)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb
            best_n = n
    
    # === Phase 7: Final continuous polish ===
    if remaining_time() > 15:
        n = best_n
        h_cp = project_feasible(best_h, n)
        t7 = min(remaining_time() * 0.4, 80)
        h_c7, c5_c7 = continuous_refine(h_cp, n, t7)
        if c5_c7 < best_c5:
            best_c5 = c5_c7
            best_h = h_c7
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
