Key improvements in binary search:
1. Precompute the set of ones and zeros
2. For each iteration, pick the top 3 worst lags
3. For each worst lag, generate targeted swap candidates
4. Also generate random candidates as exploration
5. Accept the best candidate (or use annealing)
6. Periodically do multi-swap perturbations

Let me also add a "segment flip" move: flip all bits in a random segment [a, b]. This preserves the number of 1s only if the segment has equal 1s and 0s, which is unlikely. Instead, do a "segment swap": pick two equal-length segments and swap them.

Actually, for simplicity and effectiveness, let me stick with:
- Single swaps (1→0 at i, 0→1 at j)
- Double swaps (two such swaps simultaneously)
- Random perturbations (flip 2-4 random pairs)

And the targeting:
- For the worst lag k, find positions where the "contribution" to that lag is large
- Suggest swaps that reduce the worst lag

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            lo, hi = -1.5, 1.5
            for _ in range(60):
                mid = (lo + hi) / 2.0
                new_h = np.clip(h + mid, 0.0, 1.0)
                if np.sum(new_h) > target:
                    hi = mid
                else:
                    lo = mid
            h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
        else:
            break
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g


def _sc(x):
    return np.sum(x) - _TGT


def _scj(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.success or res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if rng is None:
            rng = np.random.RandomState(0)
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    return h


def corr_gradient_at(h, k_off, n):
    g = np.zeros(n)
    if k_off >= 0:
        np_ = n - k_off
        g[k_off:k_off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def binary_local_search_aggressive(h_bin, n, rng, time_limit=60):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - n_half]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:n_half - n_ones]]] = 1
    dx_v = 2.0 / n
    corr = compute_corr(h)
    best_max = float(np.max(corr))
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    max_no_improve = 8000
    anneal_temp = 0.15

    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > max_no_improve:
            break

        corr = compute_corr(h)
        max_val = float(np.max(corr))
        top_k = min(8, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_k]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0.0
        best_swap = None
        n_random = min(16, max(8, int(remaining() / 0.05)))

        for _ in range(n_random):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        # Targeted swaps: for each of top 3 lags, find contributing positions
        for tl in range(min(3, len(top_idx))):
            k_hot = top_idx[tl]
            off = k_hot - (n - 1)
            if off >= 0 and off < n:
                np_ = n - off
                # Contribution: h[i+off]*(1-h[i]) for i in [0, np_)
                # To reduce, we want to find i where h[i+off]=1 and h[i]=0 (contributes)
                # Swap: set h[i+off]=0, h[i]=1
                contrib = (h[:np_] == 0) & (h[off:off + np_] == 1)
                cidx = np.where(contrib)[0]
                if len(cidx) > 0:
                    for _ in range(3):
                        fi = cidx[rng.randint(len(cidx))]
                        i = fi + off  # position of 1 to flip to 0
                        j = fi  # position of 0 to flip to 1
                        if i == j:
                            continue
                        h[i] = 0
                        h[j] = 1
                        nc = float(np.max(compute_corr(h)))
                        delta = nc - max_val
                        h[i] = 1
                        h[j] = 0
                        if delta < best_delta - 1e-14:
                            best_delta = delta
                            best_swap = (i, j)
            elif off < 0 and -off < n:
                ao = -off
                np_ = n - ao
                # Contribution: h[i]*(1-h[i+ao]) for i in [0, np_)
                contrib = (h[:np_] == 1) & (h[ao:ao + np_] == 0)
                cidx = np.where(contrib)[0]
                if len(cidx) > 0:
                    for _ in range(3):
                        fi = cidx[rng.randint(len(cidx))]
                        i = fi  # position of 1 to flip to 0
                        j = fi + ao  # position of 0 to flip to 1
                        if i == j:
                            continue
                        h[i] = 0
                        h[j] = 1
                        nc = float(np.max(compute_corr(h)))
                        delta = nc - max_val
                        h[i] = 1
                        h[j] = 0
                        if delta < best_delta - 1e-14:
                            best_delta = delta
                            best_swap = (i, j)

        # Double swap (2 pairs simultaneously)
        if no_improve > 200 and len(ones) >= 2 and len(zeros) >= 2:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones) - 1)]
            if i2 >= i1:
                i2 += 1
                if i2 >= len(ones):
                    i2 -= 1
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros) - 1)]
            if j2 >= j1:
                j2 += 1
                if j2 >= len(zeros):
                    j2 -= 1
            if i1 != j1 and i1 != j2 and i2 != j1 and i2 != j2:
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
                nc = float(np.max(compute_corr(h)))
                delta = nc - max_val
                h[i1] = 1
                h[i2] = 1
                h[j1] = 0
                h[j2] = 0
                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_swap = (i1, j1)  # mark as double swap with special encoding
                    # Store double swap separately
                    best_swap = (i1 * n + j1, i2 * n + j2)  # encode

        if best_swap is not None:
            a, b = best_swap
            # Check if double swap encoding
            if a >= n:
                i1 = a // n
                j1 = a % n
                i2 = b // n
                j2 = b % n
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
            else:
                i, j = a, b
                h[i] = 0
                h[j] = 1
            cur_max = max_val + best_delta
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            if max_val > best_max + 0.5:
                # Reset to best if we've gone too far
                h = best_h.copy()
                max_val = best_max
        else:
            # No improving move - perturb to escape
            if no_improve > 300:
                n_flip = rng.randint(2, 5)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_flip):

                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0
                    h[j] = 1
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                max_val = float(np.max(compute_corr(h)))
            no_improve = 0

    return best_h, best_max * dx_v


def projected_gradient_smooth(h, n, beta, n_iter=300):
    """Projected gradient descent on smooth max approximation."""
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.08

    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)

        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= wk * h[:np_]

        # Project gradient: remove mean (tangent to sum constraint)
        g -= np.mean(g)

        # Line search
        found = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.05]:
            h_new = project_feasible(h - s * g, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                h = h_new
                best_c5 = c_new
                best_h = h.copy()
                found = True
                break
            elif c_new < compute_c5(h):
                h = h_new
                found = True
                break
        if not found:
            step *= 0.7
            if step < 0.001:
                break
        else:
            step = min(step * 1.1, 0.15)

    return best_h, best_c5


def top_shifts_focus(h, n, top_k=5, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-10:
            break
        k_top = idx[0]
        k_bot = idx[-1]
        off_t = k_top - (n - 1)
        off_b = k_bot - (n - 1)
        g_t = corr_gradient_at(h, off_t, n)
        g_b = corr_gradient_at(h, off_b, n)
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015, 0.5, 0.2, 0.04, 0.06]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def golay-like(n, rng):
    """Generate a Golay-like low-correlation binary sequence."""
    # Start with Thue-Morse and do quick greedy optimization
    h = thue_morse(n, rng)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            h[ones_idx[:n_ones - n_half]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            h[zeros_idx[:n_half - n_ones]] = 1
    # Quick greedy: for each position, try flipping and keep if improves max corr
    changed = True
    while changed and remaining() > 5:
        changed = False
        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        for i in range(n):
            if remaining() < 3:
                break
            old = h[i]
            j = None
            if old == 1:
                zeros = np.where(h == 0)[0]
                if len(zeros) == 0:
                    break
                j = zeros[rng.randint(len(zeros))]
            else:
                ones = np.where(h == 1)[0]
                if len(ones) == 0:
                    break
                j = ones[rng.randint(len(ones))]
            h[i] = 1.0 - old
            h[j] = old
            nc = float(np.max(compute_corr(h)))
            if nc < cur_max:
                cur_max = nc
                changed = True
            else:
                h[i] = old
                h[j] = 1.0 - old
    return h

                if c5_tc < best_c5:

                    best_h = h_tc.copy()
                    best_n = n_t
                    h = h_tc.copy()

    # Phase 6: Diverse binary starts at best resolution
    n = best_n
    if remaining() > 60:
        # Thue-Morse
        h_tm = thue_morse(n, rng)
        alloc = min(int(remaining() * 0.12), 50)
        h_tm_b, c5_tm = binary_local_search_aggressive(h_tm, n, rng, time_limit=alloc)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm_b.copy()
            h = h_tm_b.copy()
        # Continuous refine
        if remaining() > 20:
            h_tm_c = slsqp_smooth(h_tm_b, n, 10000, 300)
            h_tm_c, c5_tm_c = top_shifts_focus(h_tm_c, n, 5, 200)
            if c5_tm_c < best_c5:
                best_c5 = c5_tm_c
                best_h = h_tm_c.copy()
                h = h_tm_c.copy()

        # Random starts
        for rs in range(3):
            if remaining() < 20:
                break
            h_r = rng.uniform(0, 1, n)
            h_r = project_feasible(h_r, n)
            h_r_bin = binarize(h_r, n)
            alloc = min(int(remaining() * 0.1), 40)
            h_r_b, c5_r = binary_local_search_aggressive(h_r_bin, n, rng, time_limit=alloc)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r_b.copy()
                h = h_r_b.copy()

    # Phase 7: Intensive SLSQP cascade at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [3000, 8000, 20000, 50000, 100000, 200000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Final top-shifts with various settings
    for tk in [3, 5, 7, 10, 15]:
        if remaining() < 5:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 9: Final binary search
    if remaining() > 15:
        h_bin = binarize(h, n)
        alloc = min(int(remaining() * 0.3), 60)
        h_b, c5_b = binary_local_search_aggressive(h_bin, n, rng, time_limit=alloc)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            # Quick continuous
            if remaining() > 10:
                h_c = slsqp_smooth(h_b, n, 50000, 400)
                c5_c = compute_c5(h_c)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c.copy()

    # Phase 10: Final projected gradient
    if remaining() > 10:
        h_pg, c5_pg = projected_gradient_smooth(best_h, best_n, 20000, n_iter=100)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    best_h = final_h
    best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
