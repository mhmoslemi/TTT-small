    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-13:
                return trial
    except Exception:
        pass
    return h.copy()

def coord_descent_full(h, n, n_sweeps=30):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        improved_any = False
        # Include ALL points, not just interior
        for i in range(n):
            if remaining_time() < 0.5:
                break
            t0 = best[i]
            # Compute coefficients a[k], b[k] for corr[k] = base[k] + a[k]*t + b[k]*t^2
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            # offset = 0 (self-correlation): h[i]*(1-h[i])
            a[n - 1] += 1.0
            b[n - 1] += -1.0
            # offset > 0
            for off in range(1, n):
                k_pos = n - 1 + off
                k_neg = n - 1 - off
                # For k_pos: h[i] as first factor if i+off < n; as (1-h[i]) second factor if i-off >= 0
                if i + off < n:
                    a[k_pos] += (1.0 - best[i + off])
                if i - off >= 0:
                    a[k_neg] += (-best[i - off])
            base = best_corr - a * t0 - b * t0 * t0

            # Evaluate at coarse grid first
            grid = np.linspace(0.0, 1.0, 21)
            best_t = t0
            best_f = np.max(base + a * t0 + b * t0 * t0) * dx
            for t in grid:
                f = np.max(base + a * t + b * t * t) * dx
                if f < best_f - 1e-15:
                    best_f = f
                    best_t = t

            # Golden section on full [0,1] for 1D minimization
            lo = 0.0
            hi = 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = np.max(base + a * c1 + b * c1 * c1) * dx
            f2 = np.max(base + a * c2 + b * c2 * c2) * dx
            for _ in range(70):
                if hi - lo < 1e-13:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = np.max(base + a * c2 + b * c2 * c2) * dx
            t_opt = (lo + hi) / 2.0
            f_opt = np.max(base + a * t_opt + b * t_opt * t_opt) * dx
            if f_opt < best_f - 1e-15:
                best_f = f_opt
                best_t = t_opt

            if best_f < best_c5 - 1e-13 and abs(best_t - t0) > 1e-10:
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved_any = True

        if not improved_any:
            break

    return best, best_c5

def pair_swap_descent(h, n, n_trials=2000):
    """Try swapping h[i] and h[j] for random pairs."""
    best = h.copy()
    best_c5 = compute_c5(best)
    rng = np.random.RandomState(12345)
    idx_ones = np.where(best > 0.5)[0]
    idx_zeros = np.where(best < 0.5)[0]

    for trial in range(n_trials):
        if remaining_time() < 1:
            break
        if len(idx_ones) == 0 or len(idx_zeros) == 0:
            break
        i = idx_ones[rng.randint(len(idx_ones))]
        j = idx_zeros[rng.randint(len(idx_zeros))]
        if i == j:
            continue
        # Swap: set h[i] -> 0-ish, h[j] -> 1-ish (or more generally swap values)
        ti, tj = best[i], best[j]
        best[i] = tj
        best[j] = ti
        best = project_feasible(best, n)
        c5_new = compute_c5(best)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            idx_ones = np.where(best > 0.5)[0]
            idx_zeros = np.where(best < 0.5)[0]
        else:
            best[i] = ti
            best[j] = tj

    return best, best_c5

def active_refine(h, n, n_iter=300, threshold=0.01):
    """Subgradient step focused on active lags."""
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for it in range(n_iter):
        if remaining_time() < 2:
            break
        corr = compute_corr(best)
        max_corr = np.max(corr)
        # Active set with adaptive threshold
        thresh = max_corr * (1.0 - threshold)
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        grad = np.zeros(n)
        for k_idx in active:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += (1.0 - best[:npairs])
                    grad[:npairs] -= best[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= best[:npairs]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        grad = grad / gnorm
        base_step = 0.05 / (1.0 + it * 0.02)
        improved = False
        for sm in [1.0, 0.5, 2.0, 0.25, 0.75, 0.1, 3.0, 0.5]:
            step = base_step * sm
            h_trial = best - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-15:
                best_c5 = c5_trial
                best = h_trial.copy()
                improved = True
                break

        if not improved and it > 50:
            break

    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_varied_start(n, seed, variant):
    rng = np.random.RandomState(seed)
    if variant == 0:
        # Random binary
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    elif variant == 1:
        # Two intervals
        h = np.zeros(n)
        gap = rng.uniform(0.2, 0.8)
        w1 = 0.5
        w2 = 0.5
        s1 = rng.uniform(0, 2 - w1 - gap - w2)
        s2 = s1 + w1 + gap
        i1s, i1e = int(s1 * n / 2), min(n, int((s1 + w1) * n / 2))
        i2s, i2e = int(s2 * n / 2), min(n, int((s2 + w2) * n / 2))
        h[i1s:i1e] = 1.0
        h[i2s:i2e] = 1.0
    elif variant == 2:
        # Three intervals
        h = np.zeros(n)
        gaps = rng.dirichlet(np.ones(4)) * 1.0
        lens = rng.dirichlet(np.ones(3)) * 1.0
        pos = gaps[0]
        for i in range(3):
            s = pos
            e = pos + lens[i]
            si, ei = int(s * n / 2), min(n, int(e * n / 2))
            h[si:ei] = 1.0
            pos = e + gaps[i + 1]
    elif variant == 3:
        # Periodic with period 1/3
        period = n // 6
        duty = period // 2
        for s in range(0, n, period):
            h[s:s + duty] = 1.0
    elif variant == 4:
        # Symmetric random
        half = n // 2
        h_half = np.zeros(half)
        idx = rng.choice(half, half // 2, replace=False)
        h_half[idx] = 1.0
        h = np.concatenate([h_half, h_half[::-1]])
    elif variant == 5:
        # Fractal-like (Cantor approximation)
        h = np.ones(n)
        step = n
        while step > 1:
            step = step // 3
            for s in range(0, n, step * 3):
                h[s + step:s + 2 * step] = 0.0
        # Make it sum to n/2
        n_ones = np.sum(h > 0.5)
        if n_ones > n // 2:
            to_remove = n_ones - n // 2
            ones_idx = np.where(h > 0.5)[0]
            to_remove = rng.choice(ones_idx, min(to_remove, len(ones_idx)), replace=False)
            h[to_remove] = 0.0
        elif n_ones < n // 2:
            to_add = n // 2 - n_ones
            zero_idx = np.where(h < 0.5)[0]
            to_add = rng.choice(zero_idx, min(to_add, len(zero_idx)), replace=False)
            h[to_add] = 1.0
    else:
        # Shifted random
        h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        shift = rng.randint(1, n // 4)
        h = np.roll(h, shift)

    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP at current resolution (increasing beta)
    for beta in [500, 2000, 8000, 30000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_optimize(h, n, beta, 500)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: Intensive coordinate descent (full range search)
    if remaining_time() > 50:
        h_new, c5_new = coord_descent_full(h, n, n_sweeps=50)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Active set refinement
    if remaining_time() > 60:
        h_new, c5_new = active_refine(h, n, n_iter=300, threshold=0.01)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Pair swap search
    if remaining_time() > 40:
        h_new, c5_new = pair_swap_descent(best_h, best_n, n_trials=3000)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Multi-start at lower resolution for global exploration
    if remaining_time() > 180:
        n_low = 160
        n_starts = 20
        for si in range(n_starts):
            if remaining_time() < 50:
                break
            variant = si % 6
            s_seed = seed + si * 7919
            start = make_varied_start(n_low, s_seed, variant)
            start = project_feasible(start, n_low)

            # Quick optimization at low res
            start = slsqp_optimize(start, n_low, 5000, 200)
            if remaining_time() > 20:
                start, _ = coord_descent_full(start, n_low, n_sweeps=15)

            c5_s = compute_c5(start)
            if c5_s < best_c5 - 0.002:
                # Promising: upscale and refine
                h_up = interp_to(start, n_low, best_n)
                h_up = project_feasible(h_up, best_n)
                h_up = slsqp_optimize(h_up, best_n, 10000, 300)
                if remaining_time() > 20:
                    h_up, c5_up = coord_descent_full(h_up, best_n, n_sweeps=25)
                else:
                    c5_up = compute_c5(h_up)
                if c5_up < best_c5 - 1e-14:
                    best_c5 = c5_up
                    best_h = h_up.copy()
                    h = h_up.copy()

        # Also try the parent downsampled to low res and back
        if remaining_time() > 60:
            h_par_low = interp_to(best_h, best_n, n_low)
            h_par_low = project_feasible(h_par_low, n_low)
            h_par_low = slsqp_optimize(h_par_low, n_low, 10000, 300)
            h_par_low, _ = coord_descent_full(h_par_low, n_low, n_sweeps=20)
            h_back = interp_to(h_par_low, n_low, best_n)
            h_back = project_feasible(h_back, best_n)
            h_back = slsqp_optimize(h_back, best_n, 20000, 300)
            h_back, c5_back = coord_descent_full(h_back, best_n, n_sweeps=25)
            if c5_back < best_c5 - 1e-14:
                best_c5 = c5_back
                best_h = h_back.copy()
                h = h_back.copy()

    # Phase 6: Higher resolution
    higher_res_list = []
    for mult in [1.25, 1.5, 2.0]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if rn > best_n and rn <= 2000:
            higher_res_list.append(rn)
    for extra in [160, 320, 640]:
        rn = best_n + extra
        if rn % 2 != 0:
            rn += 1
        if rn > best_n and rn <= 2048:
            higher_res_list.append(rn)
    higher_res_list = sorted(set(higher_res_list))

    for rn in higher_res_list:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)

        for beta in [3000, 15000, 60000]:
            if remaining_time() < 25:
                break
            h_r = slsqp_optimize(h_r, rn, beta, 250)

        if remaining_time() > 25:
            h_r, c5_r = coord_descent_full(h_r, rn, n_sweeps=20)
        else:
            c5_r = compute_c5(h_r)

        if c5_r < best_c5 - 1e-14:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 7: More coordinate descent rounds
    n = best_n
    h = best_h.copy()
    for rd in range(4):
        if remaining_time() < 20:
            break
        h_new, c5_new = coord_descent_full(h, n, n_sweeps=35)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            break

    # Phase 8: Active refinement again
    if remaining_time() > 30:
        h_new, c5_new = active_refine(best_h, best_n, n_iter=200, threshold=0.005)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Pair swap again
    if remaining_time() > 30:
        h_new, c5_new = pair_swap_descent(best_h, best_n, n_trials=5000)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 15:
        h_new = slsqp_optimize(best_h, best_n, 100000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 11: Final coordinate descent
    if remaining_time() > 10:
        h_new, c5_new = coord_descent_full(best_h, best_n, n_sweeps=25)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 12: One more pair swap + active refine
    if remaining_time() > 15:
        h_new, c5_new = pair_swap_descent(best_h, best_n, n_trials=3000)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        if remaining_time() > 8:
            h_new, c5_new = active_refine(best_h, best_n, n_iter=100, threshold=0.003)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint is met exactly
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
