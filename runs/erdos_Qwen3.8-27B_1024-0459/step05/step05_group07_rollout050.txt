def fast_coord_descent(h, n, max_sweeps=100, annealing_temp=0.0):

    best_c5 = np.max(best_corr) * dx

            break

        improved = False

        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets

            valid_minus = (i_minus >= 0) & (i_minus < n)

                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            t_cands = [0.0, 1.0, 0.5, old]
            # Add.interp-based candidates
            t_cands.append(np.clip(old + 0.1 * (1.0 - old), 0, 1))
            t_cands.append(np.clip(old - 0.1 * old, 0, 1))

            t_best = old
            f_best = f_old

            for tc in t_cands:
                mid = base + A * tc
                mid[n - 1] = base[n - 1] + tc - tc * tc
                fc = np.max(mid)
                if fc < f_best - 1e-14:
                    f_best = fc
                    t_best = tc

            # Golden section around best

                    hi = min(1.0, t_best + 0.3)
                else:

                    hi = old

                    hi = lo + 1e-5

                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                def _eval_t(t):
                    mid = base + A * t
                    mid[n - 1] = base[n - 1] + t - t * t
                    return np.max(mid)

                        f2 = f1

                        c1 = c2

                        f2 = _eval_t(c2)

    def obj_grad(x):
        return penalized_obj(x)[1]

    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'gtol': 1e-12, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Quick coordinate descent from parent
    if remaining_time() > 60:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=80)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: L-BFGS-B with soft-smooth objective
    if remaining_time() > 120:
        h = best_h.copy()
        for beta in [300, 1500, 6000]:
            if remaining_time() < 30:
                break
            h_new = lgbfgs_refine(h, n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            elif c5_new < compute_c5(h):
                h = h_new

    # Phase 3: Coordinate descent after L-BFGS
    if remaining_time() > 80:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=100)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: SLSQP with multiple betas
    if remaining_time() > 120:
        h = best_h.copy()
        for beta in [2000, 10000, 50000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_refine(h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 5: Random perturbations with annealed coordinate descent
    if remaining_time() > 200:
        n_perturb = 12
        for p in range(n_perturb):
            if remaining_time() < 40:
                break
            h_pert = best_h.copy()
            noise_scale = 0.12 / (1.0 + p * 0.25)
            n_flip = max(10, n // 8)
            indices = rng.choice(n, min(n_flip, n), replace=False)
            h_pert[indices] += rng.randn(len(indices)) * noise_scale
            h_pert = project_feasible(h_pert, n)

            # Annealed coordinate descent
            temp = 0.001 / (1.0 + p * 0.2)
            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=60, annealing_temp=temp)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 6: Binary rounding + re-optimization
    if remaining_time() > 100:
        for thr in [(0.35, 0.65), (0.4, 0.6), (0.5, 0.5)]:
            if remaining_time() < 30:
                break
            h_bin = binary_round_and_fix(best_h, n, thr[0], thr[1])
            c5_bin = compute_c5(h_bin)
            h_new, c5_new = fast_coord_descent(h_bin, n, max_sweeps=60)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: Pair swap search
    if remaining_time() > 80:
        h = best_h.copy()
        h, c5 = pair_swap_search(h, n, rng, n_swaps=5000)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()

    # Phase 8: Higher resolution attempt
    if remaining_time() > 200:
        high_n = 2 * n
        if high_n <= 1600:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            c5_high = compute_c5(h_high)

            h_new, c5_new = fast_coord_descent(h_high, high_n, max_sweeps=40)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = high_n

            if remaining_time() > 80 and best_n == high_n:
                h_slq = slsqp_refine(best_h, high_n, 5000, 200)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5 - 1e-12:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    h_new, c5_new = fast_coord_descent(best_h, high_n, max_sweeps=30)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        best_h = h_new.copy()

    # Phase 9: Thue-Morse structured init at high res
    if remaining_time() > 120:
        h_tm = thue_morse_init(n)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 + 0.02:
            h_new, c5_new = fast_coord_descent(h_tm, n, max_sweeps=80)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 40:
                    h_slq = slsqp_refine(best_h, n, 15000, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5 - 1e-12:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        h_new, c5_new = fast_coord_descent(best_h, n, max_sweeps=40)
                        if c5_new < best_c5 - 1e-12:
                            best_c5 = c5_new
                            best_h = h_new.copy()

    # Phase 10: Final aggressive coordinate descent
    if remaining_time() > 30:
        n_cur = best_n
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n_cur, max_sweeps=80)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()

    # Phase 11: Final SLSQP polish
    if remaining_time() > 15:
        n_cur = best_n
        for beta in [30000, 100000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_refine(best_h, n_cur, beta, 200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 12: One more coordinate descent pass
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=50)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
