import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = beta * (corr - m)
    z = np.clip(z, -50, 50)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

def smooth_max_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50, 50)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g

def eq_fun(x):
    return [np.sum(x) - _CUR_N / 2.0]

def eq_jac(x):
    return np.ones((_CUR_N, 1))

def slsqp(h, n, beta, maxiter=500, ftol=1e-15):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_max_obj, h0, jac=smooth_max_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < smooth_max_obj(h0):
            hx = project_feasible(res.x, n)
            if compute_c5(hx) <= compute_c5(h0) + 1e-12:
                return hx
    except Exception:
        pass
    return h0

def topk_descent(h, n, k=6, max_iters=1000, alpha0=0.01):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:k]
        # Compute average gradient of top-k, weighted by (corr[k] - corr_min_top)
        c_min = corr[idx[-1]]
        c_max = corr[idx[0]]
        if c_max - c_min < 1e-12:
            break
        # Weight: each active lag gets weight proportional to corr[k] - c_min
        weights = corr[idx] - c_min
        w_sum = np.sum(weights)
        if w_sum < 1e-15:
            break
        weights = weights / w_sum
        
        g = np.zeros(n)
        for pos, kk in enumerate(idx):
            wk = weights[pos] * dx
            if wk < 1e-15:
                continue
            off = kk - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - h[a:])
                g[a:] -= wk * h[:np_]
        
        # Subtract the mean to maintain sum constraint
        g -= np.mean(g)
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        
        # Backtracking line search
        alpha = alpha0 / (1.0 + 0.01 * it)
        improved = False
        for _ in range(30):
            ht = project_feasible(h + alpha * d, n)
            cn = compute_c5(ht)

                h = ht
                best_c5 = cn
                best_h = ht.copy()
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # Try smaller alphas

                ht = project_feasible(h + a_try * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    best_h = ht.copy()
                    improved = True
                    break
            if not improved:
                break
    return best_h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)
