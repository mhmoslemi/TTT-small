import time

_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (m + log_sum / beta) * (2.0 / n)

def smooth_max_grad(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    dx_factor = 2.0 / n
    
    g = np.zeros(n)
    for k in range(2 * n - 1):

        if wk < 1e-16:

            g[ao:ao+np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[ao:ao+np_]
    return g

                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            h = h_new.copy()
        else:
            # Nudge to escape
            nudge = rng.randn(n) * 0.003
            h_try = project_feasible(h + nudge, n)
            h_try = slsqp_opt(h_try, n, beta, maxiter=100)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                h = h_try.copy()
            else:
                break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Load parent
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        pc5 = compute_c5(ph)
        best_h = ph.copy()
        best_c5 = pc5
        best_n = pn
    except Exception:
        pn = 512
        ph = np.zeros(pn)
        ph[:pn // 2] = 1.0
        pc5 = compute_c5(ph)
        best_h = ph.copy()
        best_c5 = pc5
        best_n = pn

    h = best_h.copy()
    n = best_n

    # Phase 1: Coarse-to-fine SLSQP at parent resolution
    for beta in [1000, 5000, 20000, 80000, 300000]:
        if rem() < 12:
            break
        h = slsqp_opt(h, n, beta, maxiter=400)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            h = h.copy()

    # Phase 2: Perturbation exploration at parent resolution
    for trial in range(8):
        if rem() < 30:
            break
        scale = 0.01 * (1.0 + 0.5 * trial)
        noise = rng.randn(n) * scale
        hp = project_feasible(best_h + noise, n)
        for beta in [20000, 100000]:
            if rem() < 10:
                break
            hp = slsqp_opt(hp, n, beta, maxiter=250)
            c5 = compute_c5(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()
                h = hp.copy()

    # Phase 3: Super-high-beta polish at parent resolution
    for beta in [500000, 2000000]:
        if rem() < 10:
            break
        h = slsqp_opt(best_h, n, beta, maxiter=400)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Resolution bump to 1024
    n2 = 1024
    if rem() > 90:
        h2 = project_feasible(interp_to(best_h, best_n, n2), n2)
        for beta in [5000, 20000, 80000, 300000]:
            if rem() < 12:
                break
            h2 = slsqp_opt(h2, n2, beta, maxiter=350)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
                best_n = n2
                h = h2.copy()
        # Ultra-high beta at 1024
        for beta in [800000, 3000000]:
            if rem() < 10:
                break
            h2 = slsqp_opt(h2, n2, beta, maxiter=350)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
                best_n = n2
                h = h2.copy()

    # Phase 5: Try resolution 2048
    n3 = 2048
    if rem() > 160:
        h3 = project_feasible(interp_to(best_h, best_n, n3), n3)
        for beta in [20000, 100000, 400000]:
            if rem() < 15:
                break
            h3 = slsqp_opt(h3, n3, beta, maxiter=300)
            c5 = compute_c5(h3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3.copy()
                best_n = n3

    # Phase 6: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    # Cycles of SLSQP with increasing beta + perturbation restarts
    beta_schedule = [200000, 800000, 3000000, 10000000]
    for cycle in range(3):
        if rem() < 30:
            break
        for beta in beta_schedule:
            if rem() < 10:
                break
            h = slsqp_opt(h, n, beta, maxiter=500)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Tiny perturbation to explore neighborhood
        if rem() > 25:
            noise = rng.randn(n) * 0.004
            hp = project_feasible(best_h + noise, n)
            hp = slsqp_opt(hp, n, 500000, maxiter=300)
            c5 = compute_c5(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()
                h = hp.copy()

    # Phase 7: L-BFGS-B pass (different optimizer might find a different local min)
    if rem() > 20:
        h_lb = lbfgsb_opt(best_h, n, 500000, maxiter=500)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

        h_lb = lbfgsb_opt(best_h, n, 2000000, maxiter=400)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    # Phase 8: Final ultra-fine SLSQP
    for beta in [5000000, 15000000, 50000000]:
        if rem() < 8:
            break
        h = slsqp_opt(best_h, n, beta, maxiter=600)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final feasibility enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # External evaluation
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < best_c5:
            best_c5 = ec5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
