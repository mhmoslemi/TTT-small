Alternative: Use the **Augmented Lagrangian** method. This converts the constrained problem into a sequence of unconstrained problems, which can be solved with L-BFGS-B.

The augmented Lagrangian is:
L(x, λ, ρ) = f(x) + λ*(g(x)) + (ρ/2)*g(x)²

where g(x) = sum(x) - n/2 is the constraint.

The algorithm:
1. Start with λ=0, ρ=1
2. Solve min_x L(x, λ, ρ) using L-BFGS-B (unconstrained except bounds)
3. Update: λ_new = λ + ρ*g(x*)
4. If |g(x*)| is small enough, stop; else increase ρ and repeat

This is a standard method and works well.

def augmented_lagrangian(h, n, beta, maxiter_lag=20, iter_per_sub=200):
    h = project_feasible(h.copy(), n)
    lam = 0.0
    rho = 1.0
    
    def obj_al(x):
        val, g_corr = smooth_max_and_grad(x, beta)
        s = np.sum(x) - n/2.0
        obj = val + lam * s + 0.5 * rho * s * s
        grad = g_corr + lam * np.ones(n) + rho * s * np.ones(n)
        return obj, grad
    
    for it in range(maxiter_lag):
        if remaining_time() < 5:
            break
        
        result = minimize(
            lambda x: obj_al(x)[0],
            h,
            jac=lambda x: obj_al(x)[1],
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': iter_per_sub, 'ftol': 1e-14}

        h = np.clip(result.x, 0, 1)
        s = np.sum(h) - n/2.0
        
        if abs(s) < 1e-8 * n:
            break
        
        lam = lam + rho * s
        rho *= 2.0
    
    return project_feasible(h, n)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    clamped = np.clip(beta * shifted, -50.0, 50.0)
    exp_vals = np.exp(clamped)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = weights[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return val, g


def lbfgs_augmented(h, n, beta, max_outer=15, max_inner=300, time_limit=100):
    h = project_feasible(h.copy(), n)
    lam = 0.0
    rho = 1.0
    start = time.time()

    def obj_fn(x):
        val, g_corr = smooth_max_val_grad(x, beta)
        s = np.sum(x) - n / 2.0
        obj = val + lam * s + 0.5 * rho * s * s
        grad = g_corr + (lam + rho * s) * np.ones(n)
        return obj, grad

    for outer in range(max_outer):
        if remaining_time() < 5 or time.time() - start > time_limit:
            break
        try:
            res = minimize(
                lambda x: obj_fn(x)[0],
                h,
                jac=lambda x: obj_fn(x)[1],
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-10}
            )
            h = np.clip(res.x, 0.0, 1.0)
        except Exception:
            break
        s = np.sum(h) - n / 2.0
        if abs(s) < 1e-6 * n:
            break
        lam += rho * s
        rho *= 2.0
        if rho > 1e6:
            break
    return project_feasible(h, n)


def flip_delta_one(i, h, n, flip_to_zero=True):
    """Compute delta corr vector when flipping position i."""
    nc = 2 * n - 1
    di = np.zeros(nc)
    ad = np.arange(1, n)
    if flip_to_zero:
        # h[i] goes from 1 to 0
        # For |d|=ad>0: delta = h[i+ad]*[i+ad<n] - (1-h[i-ad])*[i>=ad]
        vr = i + ad < n
        vl = i >= ad
        vals = np.zeros(len(ad))
        vals[vr] += h[i + ad[vr]]
        vals[vl] -= (1.0 - h[i - ad[vl]])
        di[(n - 1) + ad] = vals
        di[(n - 1) - ad] = vals
    else:
        # h[i] goes from 0 to 1
        # delta = (1-h[i-ad])*[i>=ad] - h[i+ad]*[i+ad<n]
        vr = i + ad < n
        vl = i >= ad
        vals = np.zeros(len(ad))
        vals[vl] += (1.0 - h[i - ad[vl]])
        vals[vr] -= h[i + ad[vr]]
        di[(n - 1) + ad] = vals
        di[(n - 1) - ad] = vals
    return di


def binary_local_search_v2(h_bin, n, rng, time_limit=60):
    """Efficient binary local search using O(n) flip delta computation."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            rem = ones_idx[perm[:n_ones - n_half]]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            add = zeros_idx[perm[:n_half - n_ones]]
            h[add] = 1

    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx_v
    start = time.time()
    no_improve = 0
    max_no_improve = 3000

    while time.time() - start < time_limit and remaining_time() > 5:
        no_improve += 1
        if no_improve > max_no_improve:
            break

        corr = compute_corr(h)
        max_val = np.max(corr)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Identify hot offsets
        top_k = min(5, 2 * n - 1)
        hot_idx = np.argsort(-corr)[:top_k]

        best_delta = 0.0
        best_swap = None
        n_candidates = 8
        n_focused = 6

        # Random candidate swaps
        for _ in range(n_candidates):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            di = flip_delta_one(i, h, n, True)
            dj = flip_delta_one(j, h, n, False)
            new_corr = corr + di + dj
            nm = float(np.max(new_corr))
            delta = nm - max_val
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        # Focused swaps near hot correlations
        for _ in range(n_focused):
            k_hot = hot_idx[rng.randint(len(hot_idx))]
            off = k_hot - (n - 1)
            ad = abs(off)
            if ad == 0:
                continue
            if off >= 0:
                faults = np.where((h[:n-ad] == 1) & (h[ad:] == 0))[0]
            else:
                faults = np.where((h[ad:] == 1) & (h[:n-ad] == 0))[0]
            if len(faults) == 0:
                continue
            fi = faults[rng.randint(len(faults))]
            if off >= 0:
                i = fi
            else:
                i = fi + ad
            if i < 0 or i >= n or h[i] != 1:
                continue
            j = zeros[rng.randint(len(zeros))]
            if j == i:
                continue
            di = flip_delta_one(i, h, n, True)
            dj = flip_delta_one(j, h, n, False)
            new_corr = corr + di + dj
            nm = float(np.max(new_corr))
            delta = nm - max_val
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
            no_improve = 0
            c5_now = cur_max * dx_v
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
        else:
            if no_improve > 500:
                # Escape local minimum: random perturbation
                n_flip = rng.randint(2, 5)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_flip):
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0
                    h[j] = 1
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                no_improve = 0

    return best_h, best_c5


def systematic_binary_search(h_bin, n, rng, time_limit=60):
    """Systematic search: for each 1-position, find the best 0-position to swap with."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - n_half]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:n_half - n_ones]]] = 1

    dx_v = 2.0 / n
    corr = compute_corr(h)
    current_max = float(np.max(corr))
    best_c5 = current_max * dx_v
    best_h = h.copy()
    start = time.time()

    while time.time() - start < time_limit and remaining_time() > 5:
        corr = compute_corr(h)
        max_val = float(np.max(corr))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # For each 1-position, try the best swap
        best_global_delta = 0.0
        best_global_swap = None

        # Sample a subset of 1-positions for efficiency
        n_try = min(len(ones), 40)
        sampled_ones = ones[rng.choice(len(ones), n_try, replace=False)] if len(ones) > 40 else ones

        for i in sampled_ones:
            di = flip_delta_one(i, h, n, True)
            # For each 0-position j, compute new max
            # Vectorized: compute delta_j for all j at once
            ad = np.arange(1, n)
            # delta_j[j_idx] for positive offset (n-1)+ad:
            # (1-h[j-ad])*[j>=ad] - h[j+ad]*[j+ad<n]
            # Shape: (len(zeros), n-1)
            j_col = zeros[:, None]  # (n_zeros, 1)
            ad_row = ad[None, :]    # (1, n-1)

            # (1-h[j-ad]) where j-ad >= 0
            valid_left = j_col >= ad_row
            left_vals = np.where(valid_left, 1.0 - h[np.where(valid_left, j_col - ad_row, 0)], 0.0)
            # h[j+ad] where j+ad < n
            valid_right = j_col + ad_row < n
            right_vals = np.where(valid_right, h[np.where(valid_right, j_col + ad_row, 0)], 0.0)

            dj_pos = left_vals - right_vals  # (n_zeros, n-1)

            # Build full delta_j matrix: shape (n_zeros, 2n-1)
            full_dj = np.zeros((len(zeros), 2 * n - 1))
            full_dj[:, (n - 1) + ad] = dj_pos
            full_dj[:, (n - 1) - ad] = dj_pos

            # Total delta for each j: di (constant) + full_dj
            total_delta = full_dj + di[None, :]
            new_corr_matrix = corr[None, :] + total_delta
            new_maxes = np.max(new_corr_matrix, axis=1)
            best_j_idx = np.argmin(new_maxes)
            best_delta_j = float(new_maxes[best_j_idx] - max_val)

            if best_delta_j < best_global_delta - 1e-14:
                best_global_delta = best_delta_j
                best_global_swap = (i, int(zeros[best_j_idx]))

        if best_global_swap is not None:
            i, j = best_global_swap
            h[i] = 0
            h[j] = 1
            current_max += best_global_delta
            if current_max * dx_v < best_c5:
                best_c5 = current_max * dx_v
                best_h = h.copy()
        else:
            # No improvement found in this sweep, do a random perturbation
            if len(ones) > 0 and len(zeros) > 0:
                n_flip = rng.randint(1, 4)
                for _ in range(n_flip):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0
                    h[j] = 1
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]

    return best_h, best_c5


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def subgradient_polish(h, n, n_iter=100):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.03
    corr = compute_corr(h)
    for it in range(n_iter):
        if remaining_time() < 3:
            break
        m = np.max(corr)
        threshold = m - 0.3
        top_idx = np.where(corr >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]
        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(top_idx)
        g -= np.mean(g)
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            corr = compute_corr(h)
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.85
            if step < 0.002:
                break
    return best_h, best_c5

            h = h_new.copy()

    # Final binary search
    if remaining_time() > 20:
        h_bin = binarize(h, n)
        h_b, c5_b = binary_local_search_v2(h_bin, n, rng, time_limit=min(40, int(remaining_time() * 0.4)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = lbfgs_augmented(h_b, n, 20000, max_outer=6, max_inner=200, time_limit=30)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Final subgradient
    if remaining_time() > 10:
        h_sg, c5_sg = subgradient_polish(best_h, best_n, n_iter=60)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
