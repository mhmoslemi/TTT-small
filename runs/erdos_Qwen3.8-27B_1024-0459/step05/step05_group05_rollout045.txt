def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - h
            mask = slack > 1e-10
            if np.count_nonzero(mask) > 0:
                h[mask] += diff / np.count_nonzero(mask)
            else:
                h[:] += diff / n
        else:
            mask = h > 1e-10
            if np.count_nonzero(mask) > 0:
                h[mask] += diff / np.count_nonzero(mask)
            else:
                h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_grad_at(h, off, n):
    g = np.zeros(n)
    if off >= 0:
        np_ = n - off
        g[off:off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[off:off + np_]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def power_mean_obj(h, n, p, penalty=500.0):
    dx = 2.0 / n
    corr = compute_corr(h)
    s = np.mean(corr ** p)
    pm = s ** (1.0 / p)
    sum_diff = np.sum(h) - n / 2.0
    return (pm + penalty * sum_diff ** 2) * dx


def power_mean_grad(h, n, p, penalty=500.0):
    dx = 2.0 / n
    corr = compute_corr(h)
    corr_safe = np.maximum(corr, 1e-15)
    w = corr_safe ** (p - 1)
    w = w / np.sum(w)
    g = np.zeros(n)
    wc = w * dx
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    sum_diff = np.sum(h) - n / 2.0
    g += 2.0 * penalty * sum_diff * dx * np.ones(n)
    return g


def proj_grad_momentum(h0, n, p, penalty=500.0, max_iter=2000, step0=0.05, momentum=0.9):
    h = project_feasible(h0, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    step = step0
    n_iter = 0
    for it in range(max_iter):
        if remaining() < 5:
            break
        n_iter += 1
        obj = power_mean_obj(h, n, p, penalty)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        g = power_mean_grad(h, n, p, penalty)
        grad_norm = np.linalg.norm(g)
        if grad_norm < 1e-12:
            break
        g_norm = g / grad_norm
        vel = momentum * vel + g_norm
        vel_norm = np.linalg.norm(vel)
        if vel_norm < 1e-14:
            break
        d = -vel / vel_norm
        h_trial = project_feasible(h + step * d, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial < c5_cur - 1e-12:
            h = h_trial
            step = min(step * 1.05, 0.5)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
        else:
            step *= 0.7
            if step < 0.001:
                break
    return best_h, best_c5


def binary_search_tabu(h_init, n, rng, time_limit=60):
    h = (np.array(h_init, dtype=float) > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if n_ones > n_half:
            perm = rng.permutation(len(ones_idx))
            for i in perm[:n_ones - n_half]:
                h[ones_idx[i]] = 0
        else:
            perm = rng.permutation(len(zeros_idx))
            for i in perm[:n_half - n_ones]:
                h[zeros_idx[i]] = 1
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_improve = 0
    max_no_improve = 8000
    tabu = set()
    tabu_len = 100
    history = []
    temp = 0.01

    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > max_no_improve:
            break
        temp = max(0.001, 0.01 * (0.995 ** no_improve))
        corr = compute_corr(h)
        max_val = float(np.max(corr))
        top_k = min(6, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_k]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1.0
        best_swap = None
        n_cand = min(20, len(ones) * len(zeros))
        for _ in range(n_cand):
            if remaining() < 3:
                break
            mode = rng.randint(4)
            if mode == 0:
                i = ones[rng.randint(len(ones))]
            elif mode == 1:
                k_hot = top_idx[rng.randint(len(top_idx))]
                off = k_hot - (n - 1)
                if off >= 0:
                    np_ = n - off
                    faults = np.where((h[:np_] == 1) & (h[off:off + np_] == 0))[0]
                else:
                    ao = -off
                    np_ = n - ao
                    faults = np.where((h[ao:ao + np_] == 1) & (h[:np_] == 0))[0]
                if len(faults) == 0:
                    i = ones[rng.randint(len(ones))]
                else:
                    i = faults[rng.randint(len(faults))]
            elif mode == 2:
                k_hot = top_idx[rng.randint(len(top_idx))]
                off = k_hot - (n - 1)
                if off > 0:
                    cands = ones[(ones + off < n) & (h[np.minimum(ones + off, n - 1)] == 0)]
                elif off < 0:
                    cands = ones[(ones + off >= 0) & (h[np.maximum(ones + off, 0)] == 0)]
                else:
                    cands = ones
                if len(cands) == 0:
                    i = ones[rng.randint(len(ones))]
                else:
                    i = cands[rng.randint(len(cands))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            if (min(i, j), max(i, j)) in tabu:
                continue
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            h[j] = 0
            h[i] = 1
            delta = nc - max_val
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)
        if best_swap is not None:
            accept = (best_delta < 0) or (rng.random() < np.exp(-best_delta / max(temp, 1e-6)))
            if accept:
                i, j = best_swap
                h[i] = 0
                h[j] = 1
                cur_max += best_delta
                tabu.add((min(i, j), max(i, j)))
                history.append((i, j))
                if len(history) > tabu_len:
                    old_i, old_j = history.pop(0)
                    tabu.discard((min(old_i, old_j), max(old_i, old_j)))
                if best_delta < 0:
                    no_improve = 0
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            else:
                no_improve += 50
        else:
            if no_improve > 500:
                n_flip = rng.randint(2, 5)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_flip):
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0
                    h[j] = 1
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                cur_max = float(np.max(compute_corr(h)))
    return best_h, best_max * dx


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False) if rng else zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def fourier_start(n, rng):
    h = np.zeros(n)
    n_fourier = n // 2
    for k in range(1, min(20, n_fourier)):
        amp = rng.uniform(0.05, 0.3) / (k + 1)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    h = h - np.mean(h)
    h = 0.5 + h * 3.0
    return project_feasible(h, n)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def top_shifts_refine(h, n, top_k=5, max_iters=200):
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    step = 0.03
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            g += corr_grad_at(h, off, n)
        g /= top_k
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            step = min(step * 1.1, 0.3)
        else:
            step *= 0.8
            if step < 0.002:
                break
    return h, best_c5


def lbfgs_smooth(h0, n, p, penalty=1000.0, max_iter=500):
    h = project_feasible(h0, n).copy()

    def obj(x):
        return power_mean_obj(x, n, p, penalty)

    def grad(x):
        return power_mean_grad(x, n, p, penalty)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h_work = best_h.copy()

    # Phase 1: L-BFGS-B smooth optimization at parent resolution
    for p in [4, 8, 16, 32, 64, 128, 256]:
        if remaining() < 15:
            break
        h_new = lbfgs_smooth(h_work, n, p, penalty=2000.0, max_iter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h_work = h_new.copy()

    # Phase 2: Momentum projected gradient
    if remaining() > 30:
        h_mom, c5_mom = proj_grad_momentum(h_work, n, p=32, penalty=1000.0, max_iter=1500, step0=0.04, momentum=0.85)
        if c5_mom < best_c5:
            best_c5 = c5_mom
            best_h = h_mom.copy()
            best_n = n
            h_work = h_mom.copy()

    # Phase 3: Top-shifts equalization
    if remaining() > 20:
        for tk in [3, 5, 8, 12]:
            if remaining() < 8:
                break
            h_eq, c5_eq = top_shifts_refine(h_work, n, top_k=tk, max_iters=300)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                best_n = n
                h_work = h_eq.copy()

    # Phase 4: Binary local search with tabu + SA
    if remaining() > 40:
        sa_time = min(120, max(20, int(remaining() * 0.35)))
        h_bin = (h_work > 0.5).astype(float)
        h_b, c5_b = binary_search_tabu(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
            h_work = h_b.copy()
        # Continuous refinement from binary
        if remaining() > 20:
            h_ref = lbfgs_smooth(h_work, n, 64, penalty=5000.0, max_iter=400)
            h_ref, c5_ref = top_shifts_refine(h_ref, n, 5, 200)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n
                h_work = h_ref.copy()

    # Phase 5: Multi-resolution sweep
    if remaining() > 60:
        n_candidates = []
        for delta in [-32, -24, -16, -8, -4, 4, 8, 16, 24, 32, 48, 64]:
            nt = best_n + delta
            if 128 <= nt <= 800 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [192, 256, 320, 384, 448, 512, 576, 640]:
            if 128 <= nt <= 800 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:6]:
            if remaining() < 30:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

            if remaining() > 20:
                h_t = lbfgs_smooth(h_t, n_t, 32, penalty=3000.0, max_iter=300)
                h_t, c5_t2 = top_shifts_refine(h_t, n_t, 5, 200)
                if c5_t2 < best_c5:
                    best_c5 = c5_t2
                    best_h = h_t.copy()
                    best_n = n_t

                if remaining() > 15:
                    h_bin = (h_t > 0.5).astype(float)
                    h_b, c5_b = binary_search_tabu(h_bin, n_t, rng, time_limit=min(20, int(remaining() * 0.4)))
                    if c5_b < best_c5:
                        best_c5 = c5_b
                        best_h = h_b.copy()
                        best_n = n_t

    # Phase 6: Diverse starts
    if remaining() > 40:
        diverse_starts = []
        # Thue-Morse at best_n
        h_tm = thue_morse(best_n, rng)
        diverse_starts.append(h_tm)
        # Fourier-based
        h_f1 = fourier_start(best_n, rng)
        diverse_starts.append(h_f1)
        h_f2 = fourier_start(best_n, rng)
        diverse_starts.append(h_f2)
        # Random
        h_r = rng.uniform(0, 1, best_n)
        diverse_starts.append(project_feasible(h_r, best_n))
        # Perturbed best
        h_p = best_h + rng.randn(best_n) * 0.08
        diverse_starts.append(project_feasible(h_p, best_n))

        for ds in diverse_starts[:4]:
            if remaining() < 15:
                break
            h_d = lbfgs_smooth(ds, best_n, 16, penalty=2000.0, max_iter=250)
            h_d, c5_d = top_shifts_refine(h_d, best_n, 5, 150)
            if c5_d < best_c5:
                best_c5 = c5_d
                best_h = h_d.copy()
                best_n = best_n
                h_work = h_d.copy()

    # Phase 7: Final intensive binary search on best solution
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_search_tabu(h_bin, best_n, rng, time_limit=min(60, int(remaining() * 0.5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n

    # Phase 8: Final continuous polish
    n_final = best_n
    h_final = best_h.copy()
    for p in [16, 64, 256, 1024]:
        if remaining() < 5:
            break
        h_new = lbfgs_smooth(h_final, n_final, p, penalty=10000.0, max_iter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_final = h_new.copy()

    for tk in [3, 5, 8]:
        if remaining() < 5:
            break
        h_eq, c5_eq = top_shifts_refine(h_final, n_final, tk, 200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h_final = h_eq.copy()

    # Final binary polish
    if remaining() > 10:
        h_bin = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_search_tabu(h_bin, best_n, rng, time_limit=min(30, int(remaining() * 0.5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = best_n

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
