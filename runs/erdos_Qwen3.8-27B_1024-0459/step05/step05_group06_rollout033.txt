    h = h.copy()
    ncorr = 2 * n - 1
    dx = 2.0 / n
    
    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        # Precompute gradient matrix
        G = compute_grad_matrix(h, n)
        corr = compute_corr(h)
        c5 = np.max(corr)
        
        improved = False
        # Try all coordinates in random order
        order = np.random.RandomState(sweep % 10000).permutation(n)
        
        for j in order:
            if remaining_time() < 1:
                break
            t_old = h[j]
            # 1D problem: min_t max_k (corr[k] + (t - t_old) * G[k,j])
            # = min_t max_k ((corr[k] - t_old*G[k,j]) + t*G[k,j])
            a = corr - t_old * G[:, j]  # intercept
            b = G[:, j]  # slope
            
            # Find min of f(t) = max_k(a[k] + b[k]*t) over t in [0,1]
            # Convex piecewise linear
            # Bisection on subgradient
            lo, hi = 0.0, 1.0
            t_best = t_old
            f_best = np.max(corr)
            
            # Check boundaries
            f_lo = np.max(a)  # f(0)
            f_hi = np.max(a + b)  # f(1)
            
            if f_lo <= f_best - 1e-14:
                t_best = 0.0
            if f_hi <= f_best - 1e-14:
                t_best = 1.0
            
            # Bisection
            for _ in range(60):
                if hi - lo < 1e-14:
                    break
                mid = (lo + hi) / 2.0
                f_mid = np.max(a + b * mid)
                # Subgradient at mid
                active = np.where(a + b * mid >= f_mid - 1e-12)[0]
                if len(active) > 0:
                    min_b = np.min(b[active])
                    max_b = np.max(b[active])
                    if min_b >= -1e-12:
                        # f is non-decreasing at mid, minimum is to the left
                        hi = mid
                    elif max_b <= 1e-12:
                        # f is non-increasing at mid, minimum is to the right
                        lo = mid
                    else:
                        # 0 is in subgradient, we found the minimum
                        lo = mid
                        hi = mid
                        break
                else:
                    hi = mid
            
            t_opt = (lo + hi) / 2.0
            f_opt = np.max(a + b * t_opt)
            
            if f_opt < f_best - 1e-13:
                h[j] = t_opt
                # Update corr
                corr += (t_opt - t_old) * G[:, j]
                f_best = f_opt
                improved = True
        
        if not improved:
            break
    
    # Project to ensure feasibility
    h = project_feasible(h, n)
    return h, compute_c5(h)

def smooth_value_grad(x, n, beta):
    """Smooth approximation of max(corr) using log-sum-exp."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    # Use shifted log-sum-exp for numerical stability
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    # Gradient
    G = compute_grad_matrix(x, n)
    grad = (weights @ G) * dx
    return val, grad

def _obj_fn(x):
    return smooth_value_grad(x, _N_G, _BETA_G)[0]

def _grad_fn(x):
    return smooth_value_grad(x, _N_G, _BETA_G)[1]

def _con_fun(x):
    return np.sum(x) - _N_G / 2.0

def _con_jac(x):
    return np.ones(_N_G)

def slsqp_optimize(h, n, beta, maxiter=300):
    global _N_G, _BETA_G
    _N_G = n
    _BETA_G = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def lp_cutting_plane(h, n, max_iters=50, n_active=10):
    """LP-based cutting plane optimization."""
    h = h.copy()
    ncorr = 2 * n - 1
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 10:
            break
        
        corr = compute_corr(h)
        c5 = np.max(corr)
        
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
        
        # Identify active constraints
        threshold = c5 - 0.1 * (c5 - np.min(corr) + 1e-10)
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > n_active:
            active = np.argsort(corr)[::-1][:n_active]
        
        # Solve LP
        G = compute_grad_matrix(h, n)
        G_active = G[active, :]  # shape (len(active), n)
        corr_active = corr[active]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for i in range(len(active)):
            # Linearized: corr[k](h_new) ≈ corr[k](h) + G[row] @ (h_new - h)
            # Want: corr_active[i] + G_active[i] @ (h_new - h) <= t_var
            # => G_active[i] @ h_new <= t_var + G_active[i] @ h - corr_active[i]
            rhs = G_active[i, :] @ h - corr_active[i]
            constraints.append(G_active[i, :] @ h_var - t_var <= rhs)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = project_feasible(h_var.value, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5 - 1e-14:
            h = h_new
            best_c5 = min(best_c5, c5_new)
        elif c5_new < best_c5 + 1e-10:
            h = h_new
        else:
            # No improvement, break
            break
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
