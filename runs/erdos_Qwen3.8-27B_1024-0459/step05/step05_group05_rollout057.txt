  = (1/4) [(n-|d|) - sum(s[i]) + sum(s[i+d]) - sum(s[i]*s[i+d])]
  = (1/4) [(n-|d|) - 0 - 0 - R_offset(d)]

Wait, sum(s[i]) over the overlapping range is not necessarily 0 unless the range is the full sequence. For shift d, the overlap is n-|d| positions, and the sum of s over those positions is not generally 0.

    h = h.copy()
    n_half = n // 2

        zeros = np.where(h == 0)[0]
        n_ones = len(ones)

        best_j = -1

        for jj in range(n_zeros):
            j = zeros[jj]

            for d in range(-j, n - j):

        for ii in range(n_ones):

            # Compute rem effect at i

            for d in range(i-(n-1), i+1):

                    rem[n-1+d] += h[i-d]

            base = corr + rem  # (2n-1,)
            new_corr_mat = base[None, :] + add_mat  # (n_zeros, 2n-1)
            new_maxes = np.max(new_corr_mat, axis=1)  # (n_zeros,)
            
            jj_best = np.argmin(new_maxes)
            if new_maxes[jj_best] < best_new_max - 1e-10:
                best_new_max = new_maxes[jj_best]
                best_i = i
                best_j = zeros[jj_best]
        
        if best_i >= 0 and best_j >= 0:
            h[best_i] = 0
            h[best_j] = 1
            cur_max = best_new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            break  # No improving swap found
    
    return best_h, best_max * dx
