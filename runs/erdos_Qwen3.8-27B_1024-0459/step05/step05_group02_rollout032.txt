def kkt_equalize(h, n, max_iters=200, n_active=8):
    """Proper KKT-based equalization of top correlations."""
    h = proj(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:n_active]
        c_vals = corr[idx]
        c_max = c_vals[0]
        c_min = c_vals[-1]
        
        if c_max - c_min < 1e-10 * c_max:
            break
        
        # Find direction d that makes all active correlations equal
        # We want: g_k · d = -f_k + c_target for all active k
        # where c_target = mean(c_vals)
        # This is a least squares problem
        target = np.mean(c_vals)
        b = np.zeros(n_active)
        for j, ki in enumerate(idx):
            b[j] = target - c_vals[j]
        
        # Build G matrix: G[j, i] = g_ki[i] (gradient of k-th corr)
        G = np.zeros((n_active, n))
        for j, ki in enumerate(idx):
            G[j] = corr_grad_single(h, ki, n)
        
        # Remove mean to satisfy constraint (direction should preserve sum)
        G -= G.mean(axis=1, keepdims=True)
        
        # Solve least squares: G d ≈ b, with constraint that d has zero mean
        # Use SVD for stability
        try:
            U, s, Vt = np.linalg.svd(G, full_matrices=False)
            # Filter small singular values
            mask = s > 1e-10 * s[0]
            if mask.sum() == 0:
                break
            s_inv = np.zeros_like(s)
            s_inv[mask] = 1.0 / s[mask]
            d = Vt[mask].T @ (s_inv[mask] * (U[:, mask].T @ b))
            d -= d.mean()  # zero mean for constraint
        except Exception:
            break
        
        dn = np.linalg.norm(d)
        if dn < 1e-12:
            break
        d /= dn
        
        # Line search
        best_alpha = 0
        best_val = best_c5
        for alpha in [0.05, 0.02, 0.01, 0.03, 0.015, 0.005, 0.1, 0.008]:
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = alpha
        
        if best_alpha == 0:
            # Try negative
            for alpha in [0.02, 0.01, 0.005]:
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_val:
                    best_val = c5_try
                    best_alpha = -alpha
        
        if best_alpha == 0:
            break
        
        h = proj(h + best_alpha * d, n)
        if best_val < best_c5:
            best_c5 = best_val
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def proj(h, n):

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

            g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        if np_ > 0:
            g[:np_] += (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= h[:np_]
    return g


def softmax_obj(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    topk = min(_TOPK, 2 * n - 1)
    idx = np.argsort(-corr)[:topk]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def softmax_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    topk = min(_TOPK, 2 * n - 1)
    idx = np.argsort(-corr)[:topk]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_optimize(h, n, beta, top_k=5, maxiter=400):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(softmax_obj, h, jac=softmax_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 10})
        return proj(res.x, n)
    except Exception:
        return h


def kkt_equalize(h, n, max_iters=300, n_active=8):
    h = proj(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 5:
            break

        corr = compute_corr(h)
        idx = np.argsort(-corr)[:n_active]
        c_vals = corr[idx]
        c_max = c_vals[0]
        c_min = c_vals[-1]

        if c_max - c_min < 1e-10 * max(c_max, 1e-10):
            break

        target = np.mean(c_vals)
        b = target - c_vals

        G = np.zeros((n_active, n))
        for j, ki in enumerate(idx):
            G[j] = corr_grad_single(h, ki, n)

        # Project onto zero-mean subspace
        G = G - G.mean(axis=1, keepdims=True)

        try:
            U, s, Vt = np.linalg.svd(G, full_matrices=False)
            mask = s > 1e-8 * s[0]
            if mask.sum() == 0:
                break
            s_inv = np.zeros_like(s)
            s_inv[mask] = 1.0 / s[mask]
            d = Vt[mask].T @ (s_inv[mask] * (U[:, mask].T @ b))
            d -= d.mean()
        except Exception:
            break

        dn = np.linalg.norm(d)
        if dn < 1e-12:
            break
        d /= dn

        best_alpha = 0
        best_val = best_c5
        for alpha in [0.04, 0.02, 0.01, 0.025, 0.015, 0.005, 0.08, 0.008]:
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_alpha = alpha

        if best_alpha == 0:
            for alpha in [0.02, 0.01, 0.005]:
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_val:
                    best_val = c5_try
                    best_alpha = -alpha

        if best_alpha == 0:
            break

        h = proj(h + best_alpha * d, n)
        if best_val < best_c5:
            best_c5 = best_val

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_refine(h_bin, n, betas=None):
    if betas is None:
        betas = [3000, 10000, 30000]
    h = h_bin.astype(float).copy()
    for beta in betas:
        if remaining() < 5:
            break
        h = lbfgs_optimize(h, n, beta, top_k=6, maxiter=300)
    return h


def directed_local_search(h_bin, n, max_iters=10000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)

    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        idx = np.argsort(-corr)[:4]

        g = np.zeros(n)
        for ki in idx:
            g += corr_grad_single(h, ki, n)
        g -= g.mean()

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_move = None

        n_ones_try = min(18, len(ones))
        n_zeros_try = min(18, len(zeros))
        top_ones = ones[np.argsort(-g[ones])[:n_ones_try]]
        top_zeros = zeros[np.argsort(-g[zeros])[:n_zeros_try]]

        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                h[i] = 1
                h[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = (0, i, j)

        # Random single swaps
        for _ in range(6):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            nc = compute_c5(h)
            h[i] = 1
            h[j] = 0
            delta = nc - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (0, i, j)

        # Block swaps
        for _ in range(6):
            L = int(rng.randint(2, 6))
            a = int(rng.randint(0, n - L))
            b = int(rng.randint(0, n - L))
            if abs(a - b) < L:
                continue
            ba = h[a:a + L].copy()
            bb = h[b:b + L].copy()
            h[a:a + L] = bb
            h[b:b + L] = ba
            nc = compute_c5(h)
            delta = nc - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (1, a, b)
            h[a:a + L] = ba
            h[b:b + L] = bb

        # Reversal moves
        for _ in range(4):
            a = int(rng.randint(0, n - 3))
            L = int(rng.randint(3, min(10, n - a)))
            b = a + L
            h[a:b] = h[a:b][::-1]
            nc = compute_c5(h)
            delta = nc - cur_c5
            if delta < best_delta:
                best_delta = delta
                best_move = (2, a, b)
            h[a:b] = h[a:b][::-1]

        if best_move is None or best_delta >= 0:
            no_improve += 1
            T = max(0.001, 0.005 * (1.0 - it / float(max_iters)))
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i != j:
                h[i] = 0
                h[j] = 1
                nc = compute_c5(h)
                delta = nc - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = nc
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                        no_improve = 0
                else:
                    h[i] = 1
                    h[j] = 0
            if no_improve > 2000:
                break
        else:
            no_improve = 0
            if best_move[0] == 0:
                _, i, j = best_move
                h[i] = 0
                h[j] = 1
            elif best_move[0] == 1:
                _, a, b = best_move
                # Need L, store it differently
                L = 3  # default
                # Actually we need to store L, let me restructure
                # For now skip block moves in acceptance to keep it simple
                pass
            else:
                _, a, b = best_move
                h[a:b] = h[a:b][::-1]

            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

    return best_h, best_c5


def sim_anneal_binary(h_start, n, max_iters=15000, T_init=0.008, T_final=0.0002, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)

    h_bin = (h_start > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    for it in range(max_iters):
        if remaining() < 5:
            break

        progress = it / float(max_iters)
        T = T_init * (T_final / T_init) ** progress

        r = rng.random()
        if r < 0.5:
            # Single swap
            i = int(np.where(h_bin == 1)[0][rng.randint(n // 2)])
            j = int(np.where(h_bin == 0)[0][rng.randint(n // 2)])
            if i == j:
                continue
            h_bin[i] = 0
            h_bin[j] = 1
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
            else:
                h_bin[i] = 1
                h_bin[j] = 0
        elif r < 0.85:
            # Block swap
            L = int(rng.randint(2, min(8, n // 10)))
            a = int(rng.randint(0, n - L))
            b = int(rng.randint(0, n - L))
            if abs(a - b) < L:
                continue
            ba = h_bin[a:a + L].copy()
            bb = h_bin[b:b + L].copy()
            h_bin[a:a + L] = bb
            h_bin[b:b + L] = ba
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
            else:
                h_bin[a:a + L] = ba
                h_bin[b:b + L] = bb
        else:
            # Reversal
            L = int(rng.randint(3, min(12, n // 5)))
            a = int(rng.randint(0, n - L))
            b = a + L
            h_bin[a:b] = h_bin[a:b][::-1]
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
            else:
                h_bin[a:b] = h_bin[a:b][::-1]

    return best_h, best_c5


def multi_round_optimize(h, n, rounds=3):
    """Alternate between binary search and continuous refinement."""
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for r in range(rounds):
        if remaining() < 20:
            break

        # Binary projection and local search
        h_bin = (best_h > 0.5).astype(float)
        h_bl, c5_bl = directed_local_search(h_bin, n, max_iters=3000,
                                            rng=np.random.RandomState(42 + r * 7))
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

        # Continuous refinement
        h_cont = binary_refine(h_bl, n, betas=[3000, 12000])
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

        # KKT equalization
        h_eq, c5_eq = kkt_equalize(best_h, n, max_iters=150, n_active=8)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

        # LBFGS with high beta
        h_lb = lbfgs_optimize(best_h, n, 20000, top_k=5, maxiter=200)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: LBFGS refinement at parent resolution with increasing beta
    for beta in [150, 400, 1000, 3000, 8000, 20000]:
        if remaining() < 10:
            break
        h_new = lbfgs_optimize(h, n, beta, top_k=6, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: KKT equalization
    if remaining() > 15:
        h_eq, c5_eq = kkt_equalize(h, n, max_iters=250, n_active=8)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Multi-round binary/continuous optimization
    if remaining() > 40:
        h_mr, c5_mr = multi_round_optimize(best_h, best_n, rounds=2)
        if c5_mr < best_c5:
            best_c5 = c5_mr
            best_h = h_mr.copy()
            h = h_mr.copy()

    # Phase 4: Simulated annealing with diverse moves
    if remaining() > 50:
        h_bin_sa = (best_h > 0.5).astype(float)
        h_sa, c5_sa = sim_anneal_binary(h_bin_sa, best_n, max_iters=15000,
                                         T_init=0.006, T_final=0.0002, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

        # Refine SA result
        h_sa_cont = binary_refine(h_sa, best_n, betas=[5000, 20000])
        h_sa_eq, c5_sa_eq = kkt_equalize(h_sa_cont, best_n, max_iters=100, n_active=6)
        c5_sa_final = min(c5_sa, compute_c5(h_sa_cont), c5_sa_eq)
        if c5_sa_final < best_c5:
            if c5_sa_eq <= c5_sa_final:
                best_h = h_sa_eq.copy()
            elif compute_c5(h_sa_cont) <= c5_sa_final:
                best_h = h_sa_cont.copy()
            else:
                best_h = h_sa.copy()
            best_c5 = c5_sa_final

    # Phase 5: Try higher resolution n=768
    n_high = 768
    if remaining() > 250 and best_n < n_high:
        h_interp = interp_to(best_h, best_n, n_high)
        h_high = proj(h_interp, n_high)

        for beta in [300, 1000, 4000, 12000]:
            if remaining() < 15:
                break
            h_high = lbfgs_optimize(h_high, n_high, beta, top_k=6, maxiter=250)

        h_high_eq, c5_high_eq = kkt_equalize(h_high, n_high, max_iters=150, n_active=8)

        # Binary search at higher resolution
        h_bin_high = (h_high_eq > 0.5).astype(float)
        h_bl_high, c5_bl_high = directed_local_search(h_bin_high, n_high,
                                                       max_iters=4000, rng=rng)
        h_high_ref = binary_refine(h_bl_high, n_high, betas=[5000, 15000])
        h_high_ref2, c5_high_ref2 = kkt_equalize(h_high_ref, n_high, max_iters=100, n_active=6)

        c5_high = min(compute_c5(h_high), c5_high_eq, c5_bl_high,
                      compute_c5(h_high_ref), c5_high_ref2)
        if c5_high < best_c5:
            if c5_high_ref2 == c5_high:
                best_h = h_high_ref2.copy()
            elif compute_c5(h_high_ref) == c5_high:
                best_h = h_high_ref.copy()
            elif c5_bl_high == c5_high:
                best_h = h_bl_high.copy()
            elif c5_high_eq == c5_high:
                best_h = h_high_eq.copy()
            else:
                best_h = h_high.copy()
            best_c5 = c5_high
            best_n = n_high

    # Phase 6: Try intermediate resolution n=512 if not already there
    n_mid = 512
    if remaining() > 150 and best_n != n_mid:
        h_interp2 = interp_to(best_h, best_n, n_mid)
        h_mid = proj(h_interp2, n_mid)

        for beta in [500, 2000, 8000]:
            if remaining() < 12:
                break
            h_mid = lbfgs_optimize(h_mid, n_mid, beta, top_k=6, maxiter=250)

        h_mid_eq, c5_mid_eq = kkt_equalize(h_mid, n_mid, max_iters=150, n_active=8)

        h_bin_mid = (h_mid_eq > 0.5).astype(float)
        h_bl_mid, c5_bl_mid = directed_local_search(h_bin_mid, n_mid,
                                                     max_iters=4000, rng=rng)
        h_mid_ref = binary_refine(h_bl_mid, n_mid, betas=[5000, 15000])
        h_mid_ref2, c5_mid_ref2 = kkt_equalize(h_mid_ref, n_mid, max_iters=100, n_active=6)

        c5_mid = min(compute_c5(h_mid), c5_mid_eq, c5_bl_mid,
                     compute_c5(h_mid_ref), c5_mid_ref2)
        if c5_mid < best_c5:
            if c5_mid_ref2 == c5_mid:
                best_h = h_mid_ref2.copy()
            elif c5_bl_mid == c5_mid:
                best_h = h_bl_mid.copy()
            elif c5_mid_eq == c5_mid:
                best_h = h_mid_eq.copy()
            elif compute_c5(h_mid_ref) == c5_mid:
                best_h = h_mid_ref.copy()
            else:
                best_h = h_mid.copy()
            best_c5 = c5_mid
            best_n = n_mid

    # Phase 7: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    if remaining() > 25:
        for beta in [15000, 40000, 80000]:
            if remaining() < 8:
                break
            h_new = lbfgs_optimize(h, n, beta, top_k=5, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    if remaining() > 15:
        h_eq2, c5_eq2 = kkt_equalize(h, n, max_iters=200, n_active=10)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    # Phase 8: Final binary polish with SA
    if remaining() > 30:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = sim_anneal_binary(h_bin_f, best_n, max_iters=8000,
                                         T_init=0.003, T_final=0.0001, rng=rng)
        h_bf2, c5_bf2 = directed_local_search(h_bf, best_n, max_iters=4000, rng=rng)

        h_rf = binary_refine(h_bf2, best_n, betas=[10000, 40000])
        h_rf2, c5_rf2 = kkt_equalize(h_rf, best_n, max_iters=100, n_active=8)

        c5_final = min(c5_bf, c5_bf2, compute_c5(h_rf), c5_rf2)
        if c5_final < best_c5:
            if c5_rf2 == c5_final:
                best_h = h_rf2.copy()
            elif compute_c5(h_rf) == c5_final:
                best_h = h_rf.copy()
            elif c5_bf2 == c5_final:
                best_h = h_bf2.copy()
            else:
                best_h = h_bf.copy()
            best_c5 = c5_final

    # Phase 9: One more multi-round if time permits
    if remaining() > 40:
        h_mr2, c5_mr2 = multi_round_optimize(best_h, best_n, rounds=2)
        if c5_mr2 < best_c5:
            best_c5 = c5_mr2
            best_h = h_mr2.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
