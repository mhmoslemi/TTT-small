                if abs(c - old_val) < 1e-8:
                    continue
                h[i] = c
                # Fix sum by spreading adjustment to neighbor
                sum_diff = target - np.sum(h)
                if abs(sum_diff) < 1e-10:
                    c_cur = compute_c5(h)
                    if c_cur < best_c - 1e-12:
                        best_c = c_cur
                        best_cand = c
                elif abs(sum_diff) < 1.0:
                    # Try quick fix
                    j = (i + 1) % n
                    adj = -sum_diff
                    if 0 <= h[i] + adj_adj(h[i], adj) <= 1:
                        pass  # skip complex fix, just evaluate
                h[i] = old_val

            if best_cand != old_val:
                h[i] = best_cand
                h = project_feasible(h, n)
                c_new = compute_c5(h)
                if c_new < best_c5:
                    best_c5 = c_new
                    improved = True
        if not improved:
            break
    return h, best_c5


def adj_adj(val, adj):
    return val + adj


def subgradient_step(h, n, K, beta, step):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_vals = corr[idx]
    weights = np.exp(beta * (c_vals - c_vals.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, k, n)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h
    h_new = project_feasible(h - step * g / gn, n)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = np.zeros(512)
        parent_h[:256] = 1.0

    n = len(parent_h)
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n

    # Phase 1: SLSQP polish at parent resolution with increasing beta
    for beta in [200, 1000, 5000, 20000, 50000]:
        if remaining() < 15:
            break
        h_try = slsqp_smooth_max(best_h, n, beta, maxiter=400)
        c_try = compute_c5(h_try)
        if c_try < best_c5:
            best_c5 = c_try
            best_h = h_try.copy()

    # Phase 2: Structured perturbations + SLSQP at parent resolution
    for p in range(8):
        if remaining() < 30:
            break
        scale = 0.01 + p * 0.015
        h_pert = structured_perturbation(best_h, n, rng, scale=scale)
        c_pert = compute_c5(h_pert)
        # SLSQP from perturbed point
        for beta in [1000, 10000]:
            if remaining() < 15:
                break
            h_s = slsqp_smooth_max(h_pert, n, beta, maxiter=300)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 3: Upsample to higher resolutions
    res_list = [int(round(n * m)) for m in [1.5, 2.0, 2.5]]
    res_list = [r for r in res_list if r > n and r <= 2048]
    res_list = [r for r in res_list if r % 2 == 0]
    res_list = sorted(set(res_list))

    for n_t in res_list[:3]:
        if remaining() < 80:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c_inter = compute_c5(h_feas)

        # SLSQP at new resolution
        for beta in [500, 3000, 15000, 50000]:
            if remaining() < 20:
                break
            h_new = slsqp_smooth_max(h_feas, n_t, beta, maxiter=400)
            c_new = compute_c5(h_new)
            if c_new < c_inter:
                c_inter = c_new
                h_feas = h_new

        if c_inter < best_c5:
            best_c5 = c_inter
            best_h = h_feas.copy()
            best_n = n_t

        # Perturbation at new resolution
        if remaining() > 40:
            h_pert2 = structured_perturbation(h_feas, n_t, rng, scale=0.015)
            for beta in [5000, 20000]:
                if remaining() < 20:
                    break
                h_new2 = slsqp_smooth_max(h_pert2, n_t, beta, maxiter=300)
                c_new2 = compute_c5(h_new2)
                if c_new2 < best_c5:
                    best_c5 = c_new2
                    best_h = h_new2.copy()
                    best_n = n_t

    # Phase 4: Subgradient descent for broader exploration
    if remaining() > 40:
        h_sg = best_h.copy()
        n_sg = best_n
        for it in range(2000):
            if time.time() - _T0 > _BUDGET - 30:
                break
            progress = it / 2000.0
            K = 30 if progress < 0.3 else (20 if progress < 0.7 else 10)
            beta_g = 30.0 if progress < 0.3 else (60.0 if progress < 0.7 else 100.0)
            step = 0.04 / (1.0 + 0.05 * it)
            h_sg_new = subgradient_step(h_sg, n_sg, K, beta_g, step)
            c_sg = compute_c5(h_sg_new)
            if c_sg < compute_c5(h_sg) - 1e-12:
                h_sg = h_sg_new
            elif it > 500 and it % 200 == 0:
                # Occasionally SLSQP from subgradient result
                if remaining() > 20:
                    h_pol = slsqp_smooth_max(h_sg, n_sg, 10000, maxiter=200)
                    c_pol = compute_c5(h_pol)
                    if c_pol < best_c5:
                        best_c5 = c_pol
                        best_h = h_pol.copy()
                        best_n = n_sg
                    h_sg = h_pol

        c_sg_final = compute_c5(h_sg)
        if c_sg_final < best_c5:
            best_c5 = c_sg_final
            best_h = h_sg.copy()
            best_n = n_sg

    # Phase 5: Multiple SLSQP passes at best resolution with very high beta
    n_final = best_n
    h_final = best_h.copy()
    for beta in [100000, 200000, 500000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth_max(h_final, n_final, beta, maxiter=600)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h_final = h_new.copy()

    # Phase 6: Fine SLSQP with lowest possible ftol
    if remaining() > 10:
        h_new = slsqp_smooth_max(h_final, n_final, 100000, maxiter=1000, ftol=1e-20)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

    # Phase 7: Try a few more perturbations at very small scale
    for p in range(4):
        if remaining() < 25:
            break
        h_pert = structured_perturbation(best_h, best_n, rng, scale=0.003 + p * 0.002)
        h_pert = slsqp_smooth_max(h_pert, best_n, 50000, maxiter=400)
        c_pert = compute_c5(h_pert)
        if c_pert < best_c5:
            best_c5 = c_pert
            best_h = h_pert.copy()

    # Phase 8: Final SLSQP
    if remaining() > 10:
        h_final2 = slsqp_smooth_max(best_h, best_n, 200000, maxiter=800)
        c_final2 = compute_c5(h_final2)
        if c_final2 < best_c5:
            best_c5 = c_final2
            best_h = h_final2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
