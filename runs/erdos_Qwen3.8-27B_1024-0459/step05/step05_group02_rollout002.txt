def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def structured_init(n, pattern='alternating'):
    """Create structured initial patterns."""
    h = np.zeros(n)
    if pattern == 'alternating':
        h[::2] = 1.0
        # Fix count
        if int(np.sum(h)) != n // 2:
            h[:n//2] = 1.0
            h[n//2:] = 0.0
    elif pattern == 'block':
        h[:n//2] = 1.0
    elif pattern == 'three_quarter':
        # Try a quasi-periodic pattern
        for i in range(n):
            if (i % 4) < 2:
                h[i] = 1.0
    elif pattern == 'quasi':
        # Golden ratio quasi-periodic
        golden = (1 + np.sqrt(5)) / 2
        phase = np.arange(n) * golden
        h = (phase % 1.0) < 0.5
        h = h.astype(float)
        if int(np.sum(h)) != n // 2:
            # Adjust
            excess = int(np.sum(h)) - n // 2
            if excess > 0:
                idx = np.where(h == 1)[0]
                h[idx[:excess]] = 0
            elif excess < 0:
                idx = np.where(h == 0)[0]
                h[idx[:abs(excess)]] = 1
    return proj(h, n)
