            if t_best != old:
                if t_best > old:
                    lo_o = old
                    hi_o = min(1.0, t_best + 0.3)
                else:
                    lo_o = max(0.0, t_best - 0.3)
                    hi_o = old
                if hi_o - lo_o < 1e-12:
                    hi_o = lo_o + 1e-8

                c1 = hi_o - gr * (hi_o - lo_o)
                c2 = lo_o + gr * (hi_o - lo_o)

                def _fval(t):
                    mid = base + A * t
                    mid[n-1] = base[n-1] + t - t * t
                    return np.max(mid)

                f1 = _fval(c1)
                f2 = _fval(c2)
                for _ in range(50):
                    if f1 < f2:
                        hi_o = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi_o - gr * (hi_o - lo_o)
                        f1 = _fval(c1)
                    else:
                        lo_o = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo_o + gr * (hi_o - lo_o)
                        f2 = _fval(c2)
                t_best = (lo_o + hi_o) / 2.0
                f_best = _fval(t_best)

            if f_best < f_old - 1e-14:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break

    best = project_feasible(best, n)
    best_c5 = compute_c5(best)
    return best, best_c5


def coord_descent_with_sum(h, n, max_sweeps=100):
    """Coordinate descent that maintains sum constraint by adjusting a partner coordinate."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-((n - 1)), n, dtype=np.int64)
    best_c5 = np.max(best_corr) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = max(base[n-1] + old - old*old, np.max(base + A * old))

            t_best = old
            f_best = f_old

            # Try boundary values
            f_0 = max(base[n-1], np.max(base))
            if f_0 < f_best - 1e-14:
                t_best = 0.0
                f_best = f_0
            f_1 = max(base[n-1], np.max(base + A))
            if f_1 < f_best - 1e-14:
                t_best = 1.0
                f_best = f_1

            # Golden section
            if t_best != old:
                if t_best > old:
                    lo_b = old
                    hi_b = min(1.0, t_best + 0.25)
                else:
                    lo_b = max(0.0, t_best - 0.25)
                    hi_b = old
                if hi_b - lo_b < 1e-12:
                    hi_b = lo_b + 1e-8

                c1 = hi_b - gr * (hi_b - lo_b)
                c2 = lo_b + gr * (hi_b - lo_b)

                def _fv(t):
                    mid = base + A * t
                    mid[n-1] = base[n-1] + t - t*t
                    return np.max(mid)

                f1 = _fv(c1)
                f2 = _fv(c2)
                for _ in range(45):
                    if f1 < f2:
                        hi_b = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi_b - gr * (hi_b - lo_b)
                        f1 = _fv(c1)
                    else:
                        lo_b = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo_b + gr * (hi_b - lo_b)
                        f2 = _fv(c2)
                t_best = (lo_b + hi_b) / 2.0
                f_best = _fv(t_best)

            if f_best < f_old - 1e-14:
                delta = t_best - old
                best[i] = t_best
                best_corr += A * delta
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)

                # Maintain sum: adjust coordinate j (pick one with room)
                if abs(delta) > 1e-15:
                    if delta > 0:
                        # Need to decrease another coordinate by delta
                        j = int(np.argmax(best))
                    else:
                        # Need to increase another coordinate by -delta
                        j = int(np.argmin(best))
                    if j != i:
                        old_j = best[j]
                        new_j = old_j - delta
                        if 0.0 <= new_j <= 1.0:
                            best[j] = new_j
                            # Update corr for coordinate j
                            j_plus = j + offsets
                            j_minus = j - offsets
                            vp = (j_plus >= 0) & (j_plus < n)
                            vm = (j_minus >= 0) & (j_minus < n)
                            Aj = np.zeros(ncorr)
                            if np.any(vm):
                                Aj[vm] += (1.0 - best[j_minus[vm]])
                            if np.any(vp):
                                Aj[vp] -= best[j_plus[vp]]
                            Aj[n-1] = 0.0
                            best_corr += Aj * (new_j - old_j)
                            best_corr[n-1] += new_j*(1-new_j) - old_j*(1-old_j)

                best_c5 = np.max(best_corr) * dx
                improved = True

        if not improved:
            break

    return best, best_c5


def subgradient_refine(h, n, n_iter=200):
    """Subgradient-style optimization focusing on active lags."""
    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for it in range(n_iter):
        if remaining_time() < 5:
            break
        h_curr = best.copy()
        corr = compute_corr(h_curr)
        max_val = np.max(corr)
        
        # Active set: lags within tolerance of max
        tol = max_val * 0.05 + max_val * 0.1
        active = np.where(corr >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 12:
            idx = active[np.argsort(corr[active])[::-1][:12]]
            active = idx

        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in active:
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                grad[offset:offset+npairs] += (1.0 - h_curr[:npairs])
                grad[:npairs] -= h_curr[offset:offset+npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                grad[:npairs] += (1.0 - h_curr[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= h_curr[:npairs]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size
        step = 0.01 / (1.0 + 0.005 * it)
        h_new = h_curr - step * grad / gnorm
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best = h_new.copy()
            best_c5 = c5_new
        else:
            # Try smaller step
            h_new2 = h_curr - step * 0.3 * grad / gnorm
            h_new2 = project_feasible(h_new2, n)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < best_c5 - 1e-15:
                best = h_new2.copy()
                best_c5 = c5_new2
            elif it > 30 and it % 30 == 0:
                # Coordinate descent reset
                best, best_c5 = fast_coord_descent(best, n, max_sweeps=30)
                if best_c5 >= compute_c5(h_curr):
                    best = h_curr.copy()
                    best_c5 = compute_c5(h_curr)

    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        rng2 = np.random.RandomState(123)
        best_h = np.clip(0.5 + 0.3 * rng2.randn(best_n), 0, 1)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Thorough coordinate descent at parent resolution
    h = best_h.copy()
    h, c5 = fast_coord_descent(h, n, max_sweeps=150)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP continuation at n=640
    h = best_h.copy()
    for beta in [500, 2000, 8000, 30000]:
        if remaining_time() < 25:
            break
        h_new = slsqp_refine(h, n, beta, 600)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 3: More coordinate descent
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Subgradient + CD alternating
    if remaining_time() > 60:
        h = best_h.copy()
        h, c5 = subgradient_refine(h, n, n_iter=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Perturbations at n=640
    if remaining_time() > 80:
        for p in range(6):
            if remaining_time() < 30:
                break
            h_pert = best_h.copy()
            noise_scale = 0.12 / (1.0 + p * 0.4)
            mask = rng.random(n) < 0.25
            h_pert[mask] += rng.randn(np.sum(mask)) * noise_scale
            h_pert = project_feasible(h_pert, n)

            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=60)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 30:
                    h_slq = slsqp_refine(h_new, n, 8000, 400)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        h_new2, c5_new2 = fast_coord_descent(h_slq, n, max_sweeps=50)
                        if c5_new2 < best_c5:
                            best_c5 = c5_new2
                            best_h = h_new2.copy()

    # Phase 6: Move to higher resolution n=1280
    if remaining_time() > 150:
        high_n = 2 * n
        h_high = interp_to(best_h, n, high_n)
        h_high = project_feasible(h_high, high_n)
        c5_high = compute_c5(h_high)

        h_high, c5_high = fast_coord_descent(h_high, high_n, max_sweeps=80)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high.copy()
            best_n = high_n

        if remaining_time() > 80:
            h_high = best_h.copy()
            for beta_h in [3000, 12000, 50000]:
                if remaining_time() < 20:
                    break
                h_new = slsqp_refine(h_high, high_n, beta_h, 500)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h_high) - 1e-15:
                    h_high = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_high.copy()
                    best_n = high_n

        if remaining_time() > 50:
            h_high = best_h.copy()
            h_high, c5_high = fast_coord_descent(h_high, high_n, max_sweeps=60)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = high_n

    # Phase 7: Targeted perturbation based on active lags
    if remaining_time() > 60:
        n_cur = best_n
        h_cur = best_h.copy()
        corr_cur = compute_corr(h_cur)
        max_corr_val = np.max(corr_cur)
        # Lags with correlation near max
        active_lags = np.where(corr_cur >= max_corr_val * 0.98)[0]
        
        # Try to reduce specific lags by adjusting h at affected positions
        for lag_idx in active_lags[:5]:
            if remaining_time() < 20:
                break
            offset = int(lag_idx) - (n_cur - 1)
            h_mod = h_cur.copy()
            # Perturb in the direction that reduces this lag
            if offset >= 0:
                positions_a = np.arange(offset, n_cur)
                positions_b = np.arange(n_cur - offset)
            else:
                positions_a = np.arange(n_cur + offset)
                positions_b = np.arange(-offset, n_cur)
            
            # Slightly decrease h at positions where h is high
            for pos in positions_a[:min(10, len(positions_a))]:
                if h_mod[pos] > 0.4:
                    h_mod[pos] -= 0.01
            for pos in positions_b[:min(10, len(positions_b))]:
                if h_mod[pos] < 0.6:
                    h_mod[pos] += 0.01
            h_mod = project_feasible(h_mod, n_cur)
            
            h_new, c5_new = fast_coord_descent(h_mod, n_cur, max_sweeps=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_cur

    # Phase 8: Final SLSQP polish
    if remaining_time() > 15:
        n_cur = best_n
        h = best_h.copy()
        for beta_f in [20000, 80000, 300000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_refine(h, n_cur, beta_f, 500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One more SLSQP
    if remaining_time() > 8:
        n_cur = best_n
        h_new = slsqp_refine(best_h, n_cur, 150000, 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final: one last CD pass
    if remaining_time() > 5:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
