    corr = compute_corr(h)
    return float(np.max(corr) * dx)

    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)

        g[a:] -= h[:np_]

    """Gradient descent on C5 with projection"""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        
        # Find active lags (top k)
        sorted_idx = np.argsort(-corr)
        active = sorted_idx[:top_k]
        
        # Check convergence
        if c_max - corr[sorted_idx[-1-top_k]] < 1e-10:
            # All top lags are equal
            pass
        
        # Compute gradient of average of top-k correlations
        g = np.zeros(n)
        for k in active:
            g += corr_gradient(h, k, n)
        g /= len(active)
        
        # Project gradient onto sum-zero subspace
        g -= np.mean(g)
        
        # Line search
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        d = -g / gn
        
        best_step = 0
        best_c5_local = best_c5
        for step in [step_scale, step_scale*0.5, step_scale*0.1, step_scale*2, step_scale*5]:

            c5_val = compute_c5(ht)
            if c5_val < best_c5_local - 1e-14:
                best_c5_local = c5_val
                best_step = step
                h = ht
        if best_step == 0:

                    best_c5_local = c5_val
                    h = ht
                    break
            else:
                break
        best_c5 = best_c5_local
    
    return h, best_c5
