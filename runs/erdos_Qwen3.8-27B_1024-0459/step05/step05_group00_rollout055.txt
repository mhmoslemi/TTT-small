def compute_c5(h):

def project_feasible(y, n):
    target = n / 2.0
    y = np.array(y, dtype=float)
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(80):

def smooth_max_obj(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(ex.sum()) / beta) * (2.0 / n)


def smooth_max_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / ex.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[k]
        offset = k - (n - 1)
        if offset >= 0:
            npair = n - offset
            g[:npair] += wk * (1.0 - h[offset:offset + npair])
            g[offset:offset + npair] -= wk * h[:npair]
        else:
            a = -offset
            npair = n - a
            g[:npair] += wk * (1.0 - h[a:a + npair])

def slsqp_smooth(h, n, beta, maxiter=300):
    def obj(x):
        return smooth_max_obj(x, beta)

    def grad(x):
        g, _ = smooth_max_grad(x, beta)
        return g

    def cons_eq(x):
        return np.sum(x) - n / 2.0

    def cons_eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                return h_new
    except Exception:
        pass
    return h


def coordinate_swaps(h, n, n_iter=2000):
    """Pairwise coordinate swaps to reduce max correlation."""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() - _T0 > _BUDGET - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Find active lags
        c_max = np.max(corr)
        active_threshold = c_max - 0.05 * c_max
        active = np.where(corr >= active_threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])

        # Compute gradient-like scores for each coordinate
        scores = np.zeros(n)
        for k in active:
            offset = k - (n - 1)
            if offset >= 0:
                npair = n - offset
                scores[:npair] += (1.0 - h[offset:offset + npair])
                scores[offset:offset + npair] -= h[:npair]
            else:
                a = -offset
                npair = n - a
                scores[:npair] += (1.0 - h[a:a + npair])
                scores[a:a + npair] -= h[:npair]

        # Find best swap: decrease high-score, increase low-score
        i_star = np.argmax(scores)
        j_star = np.argmin(scores)
        if i_star == j_star:
            continue
        if h[i_star] <= h[j_star] + 1e-10:
            # Try next candidates

            order_j = np.argsort(scores)
            found = False
            for ii in order_i[:20]:
                for jj in order_j[:20]:
                    if ii != jj and h[ii] > h[jj] + 1e-10:
                        i_star, j_star = ii, jj
                        found = True
                        break
                if found:
                    break
            if not found:
                break

        eps = h[i_star] - h[j_star]
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            e = eps * frac
            h_try = h.copy()
            h_try[i_star] -= e
            h_try[j_star] += e
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c_try
                best_h = h.copy()
                improved = True
                break
        if not improved and it > 200:
            break
    return best_h, best_c5


def slsqp_direct(h, n, maxiter=500):
    """SLSQP on the true max (nondifferentiable) using subgradient at ties."""
    def obj(x):
        return compute_c5(x)

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        active = np.where(corr >= c_max - 1e-10)[0]
        g = np.zeros(n)
        for k in active:
            offset = k - (n - 1)
            if offset >= 0:
                npair = n - offset
                g[:npair] += (1.0 - x[offset:offset + npair])
                g[offset:offset + npair] -= x[:npair]
            else:
                a = -offset
                npair = n - a
                g[:npair] += (1.0 - x[a:a + npair])
                g[a:a + npair] -= x[:npair]
        g /= len(active)
        return g * (2.0 / n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                    'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) < compute_c5(h):
                return h_new
    except Exception:
        pass
    return h

    # === Phase 5: Multiple perturbations and re-optimize ===
    n_perturbations = 4
    perturb_mags = [0.05, 0.03, 0.02, 0.01]
    for pi in range(n_perturbations):
        if remaining() < 60:
            break
        mag = perturb_mags[pi] if pi < len(perturb_mags) else 0.02
        h_pert = try_perturbation(best_h, best_n, rng, mag)
        # Quick optimization from perturbation
        for beta in [200, 1000]:
            if remaining() < 15:
                break
            h_new = slsqp_smooth(h_pert, best_n, beta, maxiter=300)
            if compute_c5(h_new) < compute_c5(h_pert):
                h_pert = h_new
        h_new = slsqp_direct(h_pert, best_n, maxiter=400)
        if compute_c5(h_new) < compute_c5(h_pert):
            h_pert = h_new
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()

    # === Phase 6: Fine-tune at best solution ===
    if remaining() > 30:
        ht, ct = fine_tune(best_h, best_n, min(60.0, remaining() * 0.4))
        if ct < best_c5:
            best_c5 = ct
            best_h = ht.copy()

    # === Phase 7: Try higher resolution (n=1024) ===
    if remaining() > 120:
        n_hi = best_n * 2
        if n_hi <= 2048:
            h_hi = interp_to(best_h, best_n, n_hi)
            h_hi = project_feasible(h_hi, n_hi)
            c5_hi = compute_c5(h_hi)
            # Optimize at higher res
            for beta in [300, 1500]:
                if remaining() < 20:
                    break
                h_hi = slsqp_smooth(h_hi, n_hi, beta, maxiter=250)
            if remaining() > 20:
                h_hi = slsqp_direct(h_hi, n_hi, maxiter=300)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi

    # === Phase 8: Try lower resolution (n=256) for potentially better structure ===
    if remaining() > 80 and best_n > 256:
        n_lo = min(best_n, 256)
        h_lo = interp_to(best_h, best_n, n_lo)
        h_lo = project_feasible(h_lo, n_lo)
        for beta in [100, 500, 2000]:
            if remaining() < 15:
                break
            h_lo = slsqp_smooth(h_lo, n_lo, beta, maxiter=400)
        if remaining() > 15:
            h_lo = slsqp_direct(h_lo, n_lo, maxiter=500)
        c5_lo = compute_c5(h_lo)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_lo

    # === Phase 9: Final intensive polish ===
    if remaining() > 20:
        h = best_h.copy()
        n = best_n
        # Alternating SLSQP + swaps
        for cycle in range(3):
            if remaining() < 15:
                break
            h = slsqp_smooth(h, n, 5000, maxiter=400)
            c5_t = compute_c5(h)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()
            if remaining() < 10:
                break
            h_cs, c5_cs = coordinate_swaps(h, n, n_iter=2000)
            if c5_cs < best_c5:
                best_c5 = c5_cs
                best_h = h_cs.copy()
                h = h_cs.copy()

    # === Phase 10: Ultra-fine local search ===
    if remaining() > 10:
        ht, ct = fine_tune(best_h, best_n, min(30.0, remaining() * 0.5))
        if ct < best_c5:
            best_c5 = ct
            best_h = ht.copy()

    # Final feasibility check and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
