def project_feasible(h, n):

def coord_descent(h, n, max_sweeps=200):
    best = h.copy()
    corr = compute_corr(best).astype(float)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    c5 = float(np.max(corr) * dx)
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    phi = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if remaining_time() < 10:
            break
        improved = False
        for i in range(n):
            old = best[i]
            ip = i + offsets
            im = i - offsets
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)

            A = np.zeros(ncorr)
            if np.any(vm):
                A[vm] += (1.0 - best[im[vm]])
            if np.any(vp):
                A[vp] -= best[ip[vp]]

            base = corr - A * old
            base[n - 1] = corr[n - 1] - old * (1.0 - old)
            f_old = float(np.max(corr))

            best_f = f_old

                    best_t = tc

                    lo, hi = 0.0, 1.0
                c1 = hi - phi * (hi - lo)
                c2 = lo + phi * (hi - lo)

                m1[n - 1] = base[n - 1] + c1 - c1 * c1
                f1 = float(np.max(m1))
                m2 = base + A * c2
                m2[n - 1] = base[n - 1] + c2 - c2 * c2
                f2 = float(np.max(m2))

                for _ in range(45):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - phi * (hi - lo)
                        m1 = base + A * c1
                        m1[n - 1] = base[n - 1] + c1 - c1 * c1
                        f1 = float(np.max(m1))
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + phi * (hi - lo)
                        m2 = base + A * c2
                        m2[n - 1] = base[n - 1] + c2 - c2 * c2
                        f2 = float(np.max(m2))

def _smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + math.log(wsum) / beta) * dx

    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

    return _smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]


def _sg(x):
    return _smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]


def _sc(x):
    return float(np.sum(x)) - _N_WORK / 2.0


def _sj(x):
    return np.ones(_N_WORK)


def slsqp_refine(h, n, beta, maxiter=400):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = float(beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _sj}]
    try:
        res = minimize(_so, h, jac=_sg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_t = compute_c5(trial)
            c5_o = compute_c5(h)
            if c5_t <= c5_o + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

            ip1 = i + offsets
            im1 = i - offsets
            A1 = np.zeros(ncorr)
            if np.any((im1 >= 0) & (im1 < n)):
                m = (im1 >= 0) & (im1 < n)
                A1[m] += (1.0 - best[im1[m]])
            if np.any((ip1 >= 0) & (ip1 < n)):
                m = (ip1 >= 0) & (ip1 < n)
                A1[m] -= best[ip1[m]]
            A1[n - 1] = 0.0

            ip2 = j + offsets
            im2 = j - offsets
            A2 = np.zeros(ncorr)
            if np.any((im2 >= 0) & (im2 < n)):
                m = (im2 >= 0) & (im2 < n)
                A2[m] += (1.0 - best[im2[m]])
            if np.any((ip2 >= 0) & (ip2 < n)):
                m = (ip2 >= 0) & (ip2 < n)
                A2[m] -= best[ip2[m]]
            A2[n - 1] = 0.0

            base = corr.copy()
            base[n - 1] = corr[n - 1] - o1 * (1.0 - o1) - o2 * (1.0 - o2)

            f_old = float(np.max(corr))
            bt1, bt2, bf = o1, o2, f_old

            grid = [0.0, 0.25, 0.5, 0.75, 1.0]
            for t1v in grid:
                for t2v in grid:
                    if abs(t1v - o1) < 1e-9 and abs(t2v - o2) < 1e-9:
                        continue
                    nc = base + A1 * t1v + A2 * t2v
                    nc[n - 1] = base[n - 1] + t1v * (1 - t1v) + t2v * (1 - t2v)
                    fv = float(np.max(nc))
                    if fv < bf - 1e-14:
                        bf = fv
                        bt1, bt2 = t1v, t2v

            if (bt1 != o1 or bt2 != o2) and bf < f_old - 1e-14:
                best[i] = bt1
                best[j] = bt2
                corr += A1 * (bt1 - o1) + A2 * (bt2 - o2)
                corr[n - 1] += bt1 * (1 - bt1) + bt2 * (1 - bt2) - o1 * (1 - o1) - o2 * (1 - o2)
                c5 = float(np.max(corr) * dx)
                improved = True
        if not improved:
            break
    return best, c5


def binary_round(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    c5b = compute_c5(b)
    c5o = compute_c5(h)
    if c5b < c5o:
        return b.copy()
    return h


def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = make_thue_morse(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Intensive coordinate descent from parent
    if remaining_time() > 250:
        h = best_h.copy()
        h, c5 = coord_descent(h, n, max_sweeps=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: SLSQP continuation
    if remaining_time() > 200:
        h = best_h.copy()
        for beta in (1000, 5000, 20000, 100000):
            if remaining_time() < 25:
                break
            h_new = slsqp_refine(h, n, beta, 350)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-15:
                h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    # Phase 3: Pair coordinate descent
    if remaining_time() > 120:
        h = best_h.copy()
        h, c5 = pair_cd(h, n, max_sweeps=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        h, c5 = coord_descent(h, n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Stochastic search (simulated annealing)
    if remaining_time() > 200:
        h = best_h.copy()
        h, c5 = stoch_search(h, n, rng, max_iters=4000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if remaining_time() > 50:
            h = best_h.copy()
            h, c5 = coord_descent(h, n, max_sweeps=100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 5: Try different initializations
    if remaining_time() > 200:
        inits = []
        h_tm = make_thue_morse(n)
        inits.append(('tm', h_tm))
        for s in range(3):
            h_rb = random_binary(n, np.random.RandomState(seed + 1000 + s))
            inits.append(('rb' + str(s), h_rb))
        for name, h_init in inits:
            if remaining_time() < 40:
                break
            h_w = h_init.copy()
            h_w, c5_w = coord_descent(h_w, n, max_sweeps=150)
            if c5_w < best_c5:
                best_c5 = c5_w
                best_h = h_w.copy()
            if remaining_time() > 30:
                h_s = slsqp_refine(h_w, n, 10000, 300)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                    h_s, c5_s2 = coord_descent(h_s, n, max_sweeps=50)
                    if c5_s2 < best_c5:
                        best_c5 = c5_s2
                        best_h = h_s.copy()

    # Phase 6: Binary rounding attempt
    if remaining_time() > 30:
        h_br = binary_round(best_h, n)
        c5_br = compute_c5(h_br)
        if c5_br < best_c5:
            h_br, c5_br2 = coord_descent(h_br, n, max_sweeps=100)
            if c5_br2 < best_c5:
                best_c5 = c5_br2
                best_h = h_br.copy()

    # Phase 7: Higher resolution
    if remaining_time() > 200:
        n2 = min(2 * n, 1280)
        if n2 > n:
            h2 = interp_to(best_h, n, n2)
            h2 = project_feasible(h2, n2)
            c5_2 = compute_c5(h2)
            h2, c5_2 = coord_descent(h2, n2, max_sweeps=100)
            if remaining_time() > 40:
                h2 = slsqp_refine(h2, n2, 10000, 300)
                c5_2 = compute_c5(h2)
                h2, c5_2 = coord_descent(h2, n2, max_sweeps=60)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n2

    # Phase 8: Restart stochastic search on best
    if remaining_time() > 120:
        h = best_h.copy()
        h, c5 = stoch_search(h, best_n, rng, max_iters=3000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if remaining_time() > 40:
            h = best_h.copy()
            h, c5 = coord_descent(h, best_n, max_sweeps=100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 20:
        for beta in (50000, 200000):
            if remaining_time() < 8:
                break
            h = best_h.copy()
            h_new = slsqp_refine(h, best_n, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 10:
        h = best_h.copy()
        h, c5 = coord_descent(h, best_n, max_sweeps=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 11: Final SLSQP
    if remaining_time() > 8:
        h = slsqp_refine(best_h, best_n, 100000, 200)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 12: Final coord descent
    if remaining_time() > 5:
        h, c5 = coord_descent(best_h, best_n, max_sweeps=40)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, final_c5, n_final)
