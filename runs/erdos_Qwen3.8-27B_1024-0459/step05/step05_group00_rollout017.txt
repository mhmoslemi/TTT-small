def slsqp_smooth_max_active(h, n, active_idx, beta, maxiter=500):
    """SLSQP minimizing smooth max over active correlations."""
    def sm_obj(x):
        corr = compute_corr(x)
        c_act = corr[active_idx]
        m = np.max(c_act)
        sh = np.clip(beta * (c_act - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    def sm_grad(x):
        corr = compute_corr(x)
        c_act = corr[active_idx]
        m = np.max(c_act)
        sh = np.clip(beta * (c_act - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for i, k in enumerate(active_idx):
            if w[i] > 1e-14:
                g += w[i] * grad_corr_k(x, int(k), n)
        return g * (2.0 / n)
    
    def sc(x):
        return np.sum(x) - n / 2.0
    
    def scj(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) <= compute_c5(h) + 1e-12:
            return h_new
    except Exception:
        pass
    return h

def cutting_plane_lp_v2(h, n, cut_lags, delta):
    """LP with given cut lags and trust region."""
    corr = compute_corr(h)
    c_max = np.max(corr)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub_list = []
    b_ub_list = []
    
    for k in cut_lags:
        k = int(k)
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(np.dot(g, h) - fk)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, -(h - delta)])
    
    if A_ub_list:
        A_ub = np.vstack([np.array(A_ub_list), A_trust])
        b_ub = np.concatenate([np.array(b_ub_list), b_trust])
    else:
        A_ub = A_trust
        b_ub = b_trust
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            h_new = project_feasible(res.x[:n], n)
            return h_new, res.x[-1]
    except Exception:
        pass
    
    return h.copy(), c_max

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def subgradient_step(h, n, K, beta, step_mult):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, int(k), n)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h
    step = step_mult / gn
    h_new = project_feasible(h - step * g, n)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.arange(n)[(h > 1e-8) & (h < 1.0 - 1e-8)]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    d = abs(offset)
    if d >= n:
        return g
    g[d:] += (1.0 - h[:n - d])
    g[:n - d] -= h[d:]
    return g


def slsqp_active(h, n, active_idx, beta, maxiter=500):
    def sm_obj(x):
        corr = compute_corr(x)
        c_act = corr[active_idx]
        m = np.max(c_act)
        sh = np.clip(beta * (c_act - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def sm_grad(x):
        corr = compute_corr(x)
        c_act = corr[active_idx]
        m = np.max(c_act)
        sh = np.clip(beta * (c_act - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for i, k in enumerate(active_idx):
            if w[i] > 1e-14:
                g += w[i] * grad_corr_k(x, int(k), n)
        return g * (2.0 / n)

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj, 'jac_sparsity': _ones_sparsity(n)}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) <= compute_c5(h) + 1e-12:
            return h_new
    except Exception:
        pass
    return h


def _ones_sparsity(n):
    import scipy.sparse
    return (np.zeros(n), np.arange(n))


def cutting_plane_lp(h, n, cut_lags, delta):
    corr = compute_corr(h)
    c_max = np.max(corr)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in cut_lags:
        k = int(k)
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        A_ub_list.append(np.concatenate([g, [-1.0]]))
        b_ub_list.append(np.dot(g, h) - fk)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, -(h - delta)])
    if A_ub_list:
        A_ub = np.vstack([np.array(A_ub_list), A_trust])
        b_ub = np.concatenate([np.array(b_ub_list), b_trust])
    else:
        A_ub = A_trust
        b_ub = b_trust
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return project_feasible(res.x[:n], n), res.x[-1]
    except Exception:
        pass
    return h.copy(), c_max


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgradient_step(h, n, K, beta, step_mult):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_k(h, int(k), n)
    gn = np.linalg.norm(g)
    if gn < 1e-14:
        return h
    step = step_mult / gn
    return project_feasible(h - step * g, n)


def coordinate_refine(h, n, max_rounds=5, deltas=None):
    if deltas is None:
        deltas = [0.02, -0.02, 0.01, -0.01, 0.04, -0.04]
    h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        c5_cur = best_c5
        for i in range(n):
            h_save = h[i]
            for d in deltas:
                hi = h_save + d
                if hi < 0 or hi > 1:
                    continue
                h[i] = hi
                h = project_feasible(h, n)
                c5_try = compute_c5(h)
                if c5_try < c5_cur - 1e-14:
                    c5_cur = c5_try
                    improved = True
                    break
                else:
                    h[i] = h_save
                    h = project_feasible(h, n)
            if c5_cur < best_c5:
                best_c5 = c5_cur
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.ones(parent_n) * 0.5
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n

    h = best_h.copy()
    n = best_n

    # Phase 1: Quick SLSQP polish at parent resolution
    for beta in [300, 1000, 3000]:
        if remaining() < 10:
            break
        corr = compute_corr(h)
        topK = np.argsort(-corr)[:min(12, 2 * n - 1)]
        h = slsqp_active(h, n, topK, beta, maxiter=400)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Upgrade to n=768
    target_n = 768
    if remaining() > 150:
        h2 = interp_to(h, n, target_n)
        h2 = project_feasible(h2, target_n)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5 + 0.003:
            for beta in [500, 2000, 5000, 10000]:
                if remaining() < 12:
                    break
                corr = compute_corr(h2)
                topK = np.argsort(-corr)[:12]
                h2 = slsqp_active(h2, target_n, topK, beta, maxiter=500)
                c5_2 = compute_c5(h2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()
                    best_n = target_n
            final_c5_2 = compute_c5(h2)
            if final_c5_2 < best_c5:
                best_c5 = final_c5_2
                best_h = h2.copy()
                best_n = target_n
            h = h2.copy()
            n = target_n

    # Phase 3: Upgrade to n=1024
    target_n2 = 1024
    if remaining() > 250 and best_n < 1024:
        h3 = interp_to(best_h, best_n, target_n2)
        h3 = project_feasible(h3, target_n2)
        c5_3 = compute_c5(h3)
        for beta in [1000, 5000, 15000]:
            if remaining() < 12:
                break
            corr = compute_corr(h3)
            topK = np.argsort(-corr)[:10]
            h3 = slsqp_active(h3, target_n2, topK, beta, maxiter=500)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = target_n2
        final_c5_3 = compute_c5(h3)
        if final_c5_3 < best_c5:
            best_c5 = final_c5_3
            best_h = h3.copy()
            best_n = target_n2
        h = h3.copy()
        n = target_n2

    # Phase 4: Cutting plane at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 40:
        cuts = []
        corr = compute_corr(h)
        cuts = list(np.argsort(-corr)[:30])
        for it in range(300):
            if remaining() < 5:
                break
            delta_cp = 0.04 if it < 50 else (0.025 if it < 150 else 0.015)
            h_lp, t_lp = cutting_plane_lp(h, n, cuts, delta_cp)
            c5_cur = compute_c5(h)
            if t_lp < c5_cur - 1e-10:
                moved = False
                for alpha in [1.0, 0.6, 0.4, 0.25, 0.15]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < c5_cur - 1e-14:
                        h = h_try
                        c5_cur = c_try
                        moved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur
                            best_h = h.copy()
                        break
                if not moved:
                    h = project_feasible(h + 0.05 * (h_lp - h), n)
            else:
                corr = compute_corr(h)
                new_top = np.argsort(-corr)[:8]
                for k in new_top:
                    if int(k) not in cuts:
                        cuts.append(int(k))
                cuts = cut_sort(cuts, h, n)[:45]
            if it % 25 == 24 and remaining() > 15:
                corr = compute_corr(h)
                topK = np.argsort(-corr)[:10]
                h = slsqp_active(h, n, topK, 5000, maxiter=250)
                c5_chk = compute_c5(h)
                if c5_chk < best_c5:
                    best_c5 = c5_chk
                    best_h = h.copy()

    # Phase 5: High-beta SLSQP refinement
    n = best_n
    h = best_h.copy()
    for beta in [20000, 50000, 100000]:
        if remaining() < 8:
            break
        corr = compute_corr(h)
        topK = np.argsort(-corr)[:8]
        h = slsqp_active(h, n, topK, beta, maxiter=700)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: Alternating subgradient + SLSQP
    for cycle in range(8):
        if remaining() < 15:
            break
        for sg_it in range(30):
            if remaining() < 5:
                break
            K = max(6, int(12 - cycle))
            h = subgradient_step(h, n, K, 60.0, 0.015)
        if remaining() > 10:
            corr = compute_corr(h)
            topK = np.argsort(-corr)[:10]
            h = slsqp_active(h, n, topK, 20000, maxiter=400)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 7: Coordinate descent
    n = best_n
    h = best_h.copy()
    if remaining() > 30:
        h, c5_cd = coordinate_refine(h, n, max_rounds=3,
                                     deltas=[0.015, -0.015, 0.008, -0.008, 0.025, -0.025])
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h.copy()

    # Phase 8: Random perturbations to escape local optima
    n = best_n
    h = best_h.copy()
    for trial in range(12):
        if remaining() < 12:
            break
        perturbation = 0.003 * rng.randn(n)
        h_try = project_feasible(h + perturbation, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 + 0.002:
            corr = compute_corr(h_try)
            topK = np.argsort(-corr)[:10]
            h_try = slsqp_active(h_try, n, topK, 30000, maxiter=300)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()

    # Phase 9: Final intensive polish
    n = best_n
    h = best_h.copy()
    if remaining() > 10:
        corr = compute_corr(h)
        for span in [6, 8, 10]:
            topK = np.argsort(-corr)[:span]
            h = slsqp_active(h, n, topK, 200000, maxiter=800)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                corr = compute_corr(h)

    # Phase 10: Final coordinate fine-tune
    if remaining() > 8:
        h, c5_final_cd = coordinate_refine(best_h, best_n, max_rounds=2,
                                           deltas=[0.005, -0.005, 0.002, -0.002])
        if c5_final_cd < best_c5:
            best_c5 = c5_final_cd
            best_h = h.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)


def cut_sort(cuts, h, n, max_cuts=45):
    corr = compute_corr(h)
    return sorted(cuts, key=lambda x: -corr[int(x)])[:max_cuts]
