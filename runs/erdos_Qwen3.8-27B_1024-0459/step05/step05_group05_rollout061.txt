```python
def swap_delta(h, i, j, n):
    """Compute Δc for swapping h[i] 1→0 and h[j] 0→1. O(n)."""
    # dm = h[i-m] + h[i+m] - h[j-m] - h[j+m] for m = 1..n-1
    m = np.arange(1, n)
    # h[i-m]: valid for i-m >= 0, i.e., m <= i
    # h[i+m]: valid for i+m < n, i.e., m < n-i
    # h[j-m]: valid for j-m >= 0, i.e., m <= j
    # h[j+m]: valid for j+m < n, i.e., m < n-j
    
    hi_minus = np.zeros(n-1)
    valid = m <= i
    hi_minus[valid] = h[i - m[valid]]
    
    hi_plus = np.zeros(n-1)
    valid = m < n - i
    hi_plus[valid] = h[i + m[valid]]
    
    hj_minus = np.zeros(n-1)
    valid = m <= j
    hj_minus[valid] = h[j - m[valid]]
    
    hj_plus = np.zeros(n-1)
    valid = m < n - j
    hj_plus[valid] = h[j + m[valid]]
    
    dm = hi_minus + hi_plus - hj_minus - hj_plus
    # Build full delta vector
    delta_c = np.zeros(2*n-1)
    delta_c[n-1+1:] = dm  # positive offsets
    delta_c[:n-1] = dm[::-1]  # negative offsets (symmetric)
    # offset 0 is 0
    return delta_c
```

```python
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g


def _sc(x):
    return np.sum(x) - _TGT


def _scj(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def precompute_profile(h, n):
    """Precompute P[p, d] = h[p-d]+h[p+d] for d=1..n-1 (with 0 outside)."""
    P = np.zeros((n, n - 1))
    for p in range(n):
        for d in range(1, n):
            v = 0.0
            if p - d >= 0:
                v += h[p - d]
            if p + d < n:
                v += h[p + d]
            P[p, d - 1] = v
    return P


def precompute_profile_fast(h, n):
    """Vectorized profile computation."""
    P = np.zeros((n, n - 1))
    for d in range(1, n):
        # h[p-d] for p=d..n-1
        P[d:, d - 1] += h[:n - d]
        # h[p+d] for p=0..n-d-1
        P[:n - d, d - 1] += h[d:]
    return P


def fast_binary_search(h_bin, n, rng, time_limit=60):
    """Intensive binary local search using O(n) swap delta computation."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            rem = ones_idx[perm[:n_ones - n_half]]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            add = zeros_idx[perm[:n_half - n_ones]]
            h[add] = 1

    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx_v = 2.0 / n
    P = precompute_profile_fast(h, n)
    start = time.time()
    stall = 0
    batch_size = 8000
    two_swap_freq = 0

    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Evaluate batch of 1-swaps
        n_cand = min(batch_size, len(ones) * len(zeros))
        ri = rng.randint(len(ones), n_cand)
        rj = rng.randint(len(zeros), n_cand)
        i_arr = ones[ri]
        j_arr = zeros[rj]

        # delta_profile = P[i] - P[j], shape (n_cand, n-1)
        delta_prof = P[i_arr] - P[j_arr]
        # Build full delta_c: [reversed_delta, 0, delta]
        delta_c = np.zeros((n_cand, 2 * n - 1))
        delta_c[:, n:] = delta_prof  # positive offsets
        delta_c[:, :n - 1] = delta_prof[:, ::-1]  # negative offsets

        # New correlations
        new_corr = corr[None, :] + delta_c
        new_max = np.max(new_corr, axis=1)

        best_idx = np.argmin(new_max)
        best_val = new_max[best_idx]

        if best_val < cur_max - 1e-12:
            i_s, j_s = i_arr[best_idx], j_arr[best_idx]
            h[i_s] = 0
            h[j_s] = 1
            # Update profile
            P[i_s] = h[i_s - 1: i_s][::-1] if i_s > 0 else 0
            P[i_s, :] = 0
            P[j_s, :] = 0
            for d in range(1, n):
                v = 0.0
                if i_s - d >= 0:
                    v += h[i_s - d]
                if i_s + d < n:
                    v += h[i_s + d]
                P[i_s, d - 1] = v
            for d in range(1, n):
                v = 0.0
                if j_s - d >= 0:
                    v += h[j_s - d]
                if j_s + d < n:
                    v += h[j_s + d]
                P[j_s, d - 1] = v
            cur_max = float(best_val)
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > 200:
                # Try 2-swaps to escape
                if len(ones) >= 2 and len(zeros) >= 2:
                    n_2cand = min(3000, (len(ones) * (len(ones) - 1)) * (len(zeros) * (len(zeros) - 1)) // 4)
                    ri1 = rng.randint(len(ones), n_2cand)
                    ri2 = rng.randint(len(ones), n_2cand)
                    rj1 = rng.randint(len(zeros), n_2cand)
                    rj2 = rng.randint(len(zeros), n_2cand)
                    # Filter duplicates
                    mask = (ri1 != ri2) & (rj1 != rj2)
                    ri1, ri2 = ri1[mask], ri2[mask]
                    rj1, rj2 = rj1[mask], rj2[mask]
                    if len(ri1) > 0:
                        i1_arr, i2_arr = ones[ri1], ones[ri2]
                        j1_arr, j2_arr = zeros[rj1], zeros[rj2]
                        # delta = P[i1]+P[i2] - P[j1]-P[j2]
                        delta2 = P[i1_arr] + P[i2_arr] - P[j1_arr] - P[j2_arr]
                        delta_c2 = np.zeros((len(ri1), 2 * n - 1))
                        delta_c2[:, n:] = delta2
                        delta_c2[:, :n - 1] = delta2[:, ::-1]
                        new_corr2 = corr[None, :] + delta_c2
                        new_max2 = np.max(new_corr2, axis=1)
                        bidx2 = np.argmin(new_max2)
                        if new_max2[bidx2] < cur_max - 1e-12:
                            i1s, i2s = i1_arr[bidx2], i2_arr[bidx2]
                            j1s, j2s = j1_arr[bidx2], j2_arr[bidx2]
                            h[i1s] = 0
                            h[i2s] = 0
                            h[j1s] = 1
                            h[j2s] = 1
                            P[i1s, :] = 0
                            P[i2s, :] = 0
                            P[j1s, :] = 0
                            P[j2s, :] = 0
                            for pp in [i1s, i2s, j1s, j2s]:
                                for d in range(1, n):
                                    v = 0.0
                                    if pp - d >= 0:
                                        v += h[pp - d]
                                    if pp + d < n:
                                        v += h[pp + d]
                                    P[pp, d - 1] = v
                            cur_max = float(new_max2[bidx2])
                            stall = 0
                            if cur_max < best_max:
                                best_max = cur_max
                                best_h = h.copy()
                            corr = compute_corr(h)
                        else:
                            corr = compute_corr(h)
                # Random kick
                n_kick = rng.randint(2, 5)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_kick):
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    kk_i = ones_now[rng.randint(len(ones_now))]
                    kk_j = zeros_now[rng.randint(len(zeros_now))]
                    h[kk_i] = 0
                    h[kk_j] = 1
                    P[kk_i, :] = 0
                    P[kk_j, :] = 0
                    for pp in [kk_i, kk_j]:
                        for d in range(1, n):
                            v = 0.0
                            if pp - d >= 0:
                                v += h[pp - d]
                            if pp + d < n:
                                v += h[pp + d]
                            P[pp, d - 1] = v
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                cur_max = float(np.max(compute_corr(h)))
                corr = compute_corr(h)
                stall = 0
            if stall > 2000:
                break

        # Update corr occasionally (for safety after 2-swaps)
        two_swap_freq += 1
        if two_swap_freq % 100 == 0:
            corr = compute_corr(h)

    return best_h, best_max * dx_v


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def top_shifts_focus(h, n, top_k=5, max_iters=200):
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-9:
            break
        # Subgradient direction: reduce top, increase bottom
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            g -= corr_grad_single(h, off, n)
        g /= top_k
        g -= np.mean(g)
        improved = False
        for step in [0.02, 0.05, 0.01, 0.03, 0.08, 0.005, 0.1, 0.5, 0.2]:
            ht = project_feasible(h - step * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def corr_grad_single(h, off, n):
    """Gradient of correlation at specific offset."""
    g = np.zeros(n)
    if off >= 0:
        np_ = n - off
        g[off:off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[off:off + np_]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n, rng)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Continuous SLSQP optimization (progressive smooth max)
    for beta in [50, 200, 800, 3000, 10000, 30000, 100000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=350)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h = h_new.copy()

    # Phase 2: Top-shifts equalization
    if remaining() > 15:
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=5, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
        h_eq2, c5_eq2 = top_shifts_focus(h, n, top_k=8, max_iters=150)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    # Phase 3: INTENSIVE binary local search (key phase with O(n) swaps)
    if remaining() > 30:
        h_bin = binarize(h, n)
        sa_time = min(120, max(20, int(remaining() * 0.35)))
        h_b, c5_b = fast_binary_search(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
            # Continuous refinement from binary
            h_cont = slsqp_smooth(h_b, n, 5000, 300)
            c5_cont = compute_c5(h_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
            h = best_h.copy()
        else:
            h = best_h.copy()

        # Second round of binary search from current best
        if remaining() > 20:
            h_bin2 = binarize(h, n)
            h_b2, c5_b2 = fast_binary_search(h_bin2, n, rng, time_limit=min(60, int(remaining() * 0.2)))
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n

    # Phase 4: Diverse starting points
    if remaining() > 60:
        # Thue-Morse at current n
        h_tm = thue_morse(n, rng)
        h_tm = slsqp_smooth(h_tm, n, 3000, 300)
        h_tm, c5_tm = top_shifts_focus(h_tm, n, 4, 100)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            best_n = n
        # Binary search from Thue-Morse
        if remaining() > 30:
            h_tm_bin = binarize(h_tm, n)
            h_tmb, c5_tmb = fast_binary_search(h_tm_bin, n, rng, time_limit=min(40, int(remaining() * 0.2)))
            if c5_tmb < best_c5:
                best_c5 = c5_tmb
                best_h = h_tmb.copy()
                best_n = n

        # Complementary Thue-Morse
        h_ctm = 1.0 - h_tm
        h_ctm = project_feasible(h_ctm, n)
        h_ctm, c5_ctm = top_shifts_focus(h_ctm, n, 4, 80)
        if c5_ctm < best_c5:
            best_c5 = c5_ctm
            best_h = h_ctm.copy()
            best_n = n

        # Periodic patterns
        for period in [7, 11, 13, 17]:
            if remaining() < 15:
                break
            h_per = np.zeros(n)
            pattern = np.zeros(period)
            pattern[:period // 2] = 1.0
            for i in range(n):
                h_per[i] = pattern[i % period]
            h_per = project_feasible(h_per, n)
            h_per = slsqp_smooth(h_per, n, 2000, 200)
            c5_per = compute_c5(h_per)
            h_per, c5_per = top_shifts_focus(h_per, n, 4, 80)
            if c5_per < best_c5:
                best_c5 = c5_per
                best_h = h_per.copy()
                best_n = n

    # Phase 5: Resolution sweep
    if remaining() > 80:
        n_base = best_n
        n_candidates = set()
        for delta in list(range(-32, 33, 2)):
            nt = n_base + delta
            if 128 <= nt <= 768 and nt % 2 == 0:
                n_candidates.add(nt)
        for nt in [256, 320, 384, 448, 512, 576, 640, 704]:
            if 128 <= nt <= 768 and nt % 2 == 0:
                n_candidates.add(nt)
        n_candidates = sorted(n_candidates, key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:10]:
            if remaining() < 20:
                break
            if n_t <= 0 or n_t < 64:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

            # Quick SLSQP
            h_t = slsqp_smooth(h_t, n_t, 5000, 250)
            c5_t = compute_c5(h_t)
            h_t, c5_t = top_shifts_focus(h_t, n_t, 4, 100)

            # Binary search at this resolution
            if remaining() > 15:
                h_t_bin = binarize(h_t, n_t)
                h_tb, c5_tb = fast_binary_search(h_t_bin, n_t, rng, time_limit=min(30, int(remaining() * 0.2)))
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb.copy()

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 6: Final refinement at best resolution
    n = best_n
    h = best_h.copy()

    for beta in [10000, 30000, 80000, 200000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final top-shifts
    for tk in [3, 5, 10]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary polish
    if remaining() > 20:
        h_bin = binarize(h, n)
        h_b, c5_b = fast_binary_search(h_bin, n, rng, time_limit=min(50, int(remaining() * 0.5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
            h_c = slsqp_smooth(h_b, n, 20000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
