def slsqp_optimize(h, n, beta=5000, maxiter=300):
    """SLSQP with smooth max approximation."""
    N = n
    TGT = n / 2.0
    BETA = float(beta)
    
    def sm_obj(x):
        dx_v = 2.0 / N
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / BETA) * dx_v
    
    def sm_grad(x):
        dx_v = 2.0 / N
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(N)
        wc = w * dx_v
        for k in range(2 * N - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (N - 1)
            if off >= 0:
                np_ = N - off
                if np_ > 0:
                    g[off:off+np_] += wk * (1.0 - x[:np_])
                    g[:np_] -= wk * x[off:off+np_]
            else:
                ao = -off
                np_ = N - ao
                if np_ > 0:
                    g[:np_] += wk * (1.0 - x[ao:ao+np_])
                    g[ao:ao+np_] -= wk * x[:np_]
        return g
    
    def cons_fun(x):
        return np.sum(x) - TGT
    
    def cons_jac(x):
        return np.ones(N)
    
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.success or res.fun < sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
