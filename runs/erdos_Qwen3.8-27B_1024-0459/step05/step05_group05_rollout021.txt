def swap_delta(h, i, j, n):
    """
    Compute delta in correlation array when h[i] goes 1→0 and h[j] goes 0→1.
    Returns array of length 2n-1 with the deltas.
    Assumes h is binary (0 or 1).
    """
    # Correlation: for offset d (-(n-1) to n-1):
    # d >= 0: c(d) = sum_{t=0}^{n-1-d} h[t+d]*(1-h[t])
    # d < 0: c(d) = sum_{t=0}^{n-1-|d|} h[t]*(1-h[t+|d|])
    # In np.correlate('full') indexing: index k = (n-1) + d
    
    # For position i (1→0) in c(d), d >= 0:
    # - i as t+d: t=i-d, valid if d <= i. Delta: -(1-h[i-d]) = h[i-d]-1
    # - i as t: valid if i <= n-1-d, i.e., d <= n-1-i. Delta: +h[i+d]
    
    # For position j (0→1) in c(d), d >= 0:
    # - j as t+d: t=j-d, valid if d <= j. Delta: +(1-h[j-d])
    # - j as t: valid if j <= n-1-d, i.e., d <= n-1-j. Delta: -h[j+d]
    
    # For d < 0 (let m = -d): c(-m) = sum_{t=0}^{n-1-m} h[t]*(1-h[t+m])
    # Position i (1→0):
    # - i as t: valid if i <= n-1-m. Delta: -(1-h[i+m])
    # - i as t+m: t=i-m, valid if m <= i. Delta: +h[i-m]
    # Position j (0→1):
    # - j as t: valid if j <= n-1-m. Delta: +(1-h[j+m])
    # - j as t+m: t=j-m, valid if m <= j. Delta: -h[j-m]
    
    deltas = np.zeros(2*n - 1)
    # d >= 0 part (indices n-1 to 2n-2 in correlate array)
    # d ranges from 0 to n-1
    # For each d, only 4 terms possible
    
    # Position i, d>=0:
    # As t+d: d in [0, i], index = (n-1)+d
    for d in range(min(i+1, n)):
        idx = (n-1) + d
        deltas[idx] += (h[i-d] - 1.0)  # -(1-h[i-d])
    # As t: d in [0, n-1-i], index = (n-1)+d
    for d in range(min(n-i, n)):
        idx = (n-1) + d
        deltas[idx] += h[i+d]
    
    # Position j, d>=0:
    # As t+d: d in [0, j]
    for d in range(min(j+1, n)):
        idx = (n-1) + d
        deltas[idx] += (1.0 - h[j-d])
    # As t: d in [0, n-1-j]
    for d in range(min(n-j, n)):
        idx = (n-1) + d
        deltas[idx] -= h[j+d]
    
    # d < 0 part (indices 0 to n-2 in correlate array)
    # m = -d ranges from 1 to n-1
    # For each m, index = (n-1) - m
    # Position i (1→0):
    # As t: m in [1, n-1-i]
    for m in range(1, min(n-i, n)):
        idx = (n-1) - m
        deltas[idx] += -(1.0 - h[i+m])
    # As t+m: m in [1, i]
    for m in range(1, min(i+1, n)):
        idx = (n-1) - m
        deltas[idx] += h[i-m]
    
    # Position j (0→1):
    # As t: m in [1, n-1-j]
    for m in range(1, min(n-j, n)):
        idx = (n-1) - m
        deltas[idx] += (1.0 - h[j+m])
    # As t+m: m in [1, j]
    for m in range(1, min(j+1, n)):
        idx = (n-1) - m
        deltas[idx] -= h[j-m]
    
    return deltas

        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - max(1.0, 0.05 * m)
        top_idx = np.where(corr >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]

        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            g += corr_grad_at(h, off, n)
        g /= len(top_idx)
        g -= np.mean(g)
        g = np.where(h > 0.99, -np.abs(g), g)
        g = np.where(h < 0.01, np.abs(g), g)

        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.85
            if step < 0.001:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick continuous refinement with L-BFGS-B
    for beta in [100, 500, 2000, 8000, 30000]:
        if remaining() < 10:
            break
        h_new = lbg_smooth(h, n, beta, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: SLSQP for tighter constraint handling
    for beta in [2000, 10000, 50000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Top-shifts equalization
    if remaining() > 15:
        h_eq, c5_eq = top_shifts_refine(h, n, top_k=5, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
        h_eq2, c5_eq2 = top_shifts_refine(h, n, top_k=8, max_iters=150)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    # Phase 4: Binary local search (main workhorse)
    if remaining() > 30:
        h_bin = binarize(h, n)
        sa_time = min(100, max(20, int(remaining() * 0.3)))
        h_b, c5_b = binary_local_search(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
            # Continuous refinement from binary
            h_cont = lbg_smooth(h_b, n, 5000, 200)
            c5_cont = compute_c5(h_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
                h = h_cont.copy()
            h_eq3, c5_eq3 = top_shifts_refine(best_h, n, 5, 150)
            if c5_eq3 < best_c5:
                best_c5 = c5_eq3
                best_h = h_eq3.copy()
                h = h_eq3.copy()
        else:
            h = best_h.copy()

    # Phase 5: Subgradient refinement
    if remaining() > 20:
        h_sg, c5_sg = subgradient_refine(h, n, n_iter=100)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 6: Resolution sweep
    if remaining() > 60:
        n_base = best_n
        n_candidates = []
        for delta in [-16, -8, -4, 4, 8, 16, 32, -32, 48, -48]:
            nt = n_base + delta
            if 128 <= nt <= 640 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [256, 320, 384, 448, 512, 576, 640]:
            if 128 <= nt <= 640 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:6]:
            if remaining() < 25:
                break
            if n_t <= 0:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

            h_t = lbg_smooth(h_t, n_t, 5000, 150)
            h_t = slsqp_smooth(h_t, n_t, 10000, 100)
            c5_t = compute_c5(h_t)
            h_t, c5_t2 = top_shifts_refine(h_t, n_t, 5, 100)
            c5_t = min(c5_t, c5_t2)

            if remaining() > 15 and c5_t < best_c5 + 0.002:
                h_t_bin = binarize(h_t, n_t)
                h_tb, c5_tb = binary_local_search(h_t_bin, n_t, rng, time_limit=min(20, int(remaining() * 0.3)))
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb.copy()

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 7: Diverse starts
    if remaining() > 40:
        # Block pattern
        n_blk = best_n
        h_blk = np.zeros(n_blk)
        third = n_blk // 3
        h_blk[third:2 * third] = 1.0
        h_blk = project_feasible(h_blk, n_blk)
        h_blk = lbg_smooth(h_blk, n_blk, 3000, 150)
        h_blk, c5_blk = top_shifts_refine(h_blk, n_blk, 5, 100)
        if c5_blk < best_c5:
            best_c5 = c5_blk
            best_h = h_blk.copy()
            best_n = n_blk

        # Two-block pattern
        h_2b = np.zeros(n_blk)
        q = n_blk // 4
        h_blk2 = np.zeros(n_blk)
        h_blk2[:q] = 1.0
        h_blk2[2 * q:3 * q] = 1.0
        h_blk2 = project_feasible(h_blk2, n_blk)
        h_blk2 = lbg_smooth(h_blk2, n_blk, 3000, 150)
        h_blk2, c5_2b = top_shifts_refine(h_blk2, n_blk, 5, 100)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h_blk2.copy()
            best_n = n_blk

        # Random start
        h_r = rng.uniform(0, 1, n_blk)
        h_r = project_feasible(h_r, n_blk)
        h_r = lbg_smooth(h_r, n_blk, 5000, 200)
        h_r, c5_r = top_shifts_refine(h_r, n_blk, 5, 100)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_blk

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for beta in [5000, 20000, 80000]:
        if remaining() < 5:
            break
        h_new = lbg_smooth(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    for tk in [3, 5, 10]:
        if remaining() < 5:
            break
        h_eq, c5_eq = top_shifts_refine(h, n, top_k=tk, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary search
    if remaining() > 20:
        h_bin = binarize(h, n)
        h_b, c5_b = binary_local_search(h_bin, n, rng, time_limit=min(30, int(remaining() * 0.4)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = lbg_smooth(h_b, n, 15000, 200)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Final subgradient
    if remaining() > 8:
        h_sg, c5_sg = subgradient_refine(best_h, best_n, n_iter=80)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
