def compute_corr(h):

def project_feasible(h, n):

def smax_obj_grad(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -60.0, 60.0)
    ex = np.exp(cl)
