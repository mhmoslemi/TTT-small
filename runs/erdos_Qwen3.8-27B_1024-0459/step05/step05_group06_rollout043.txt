            curr[j] = 0.0
    
    return best, best_max, best_corr

def interp_binary(src_h, src_n, dst_n):
    """Interpolate binary solution to different resolution"""
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    h_interp = np.interp((np.arange(dst_n) + 0.5) * 2.0 / dst_n, xp, src_h)
    k = dst_n // 2
    h_bin = np.zeros(dst_n)
    order = np.argsort(h_interp)[::-1]
    h_bin[order[:k]] = 1.0
    return h_bin

def slsqp_polish(h, n, beta, maxiter=300):
    """SLSQP optimization with log-sum-exp smoothing"""
    global _N_POL, _B_POL
    _N_POL = n
    _B_POL = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_smooth, h, jac=_grad_smooth, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

_N_POL = 640
_B_POL = 1000.0

def _obj_smooth(x):
    return _smooth_val(x, _N_POL, _B_POL)[0]

def _grad_smooth(x):
    return _smooth_val(x, _N_POL, _B_POL)[1]

def _con_fun(x):
    return np.sum(x) - _N_POL / 2.0

def _con_jac(x):
    return np.ones(_N_POL)

def _smooth_val(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        wsum = 1e-300
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
            grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
            grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def coord_descent_frac(h, n, n_sweeps=30):
    """Coordinate descent for fractional values"""
    best = h.copy()
    corr = compute_corr(best)
    dx = 2.0 / n
    max_val = np.max(corr) * dx
    ncorr = 2 * n - 1
    
    for sweep in range(n_sweeps):
        if remaining() < 10:
            break
        improved = False
        for i in range(n):
            if remaining() < 5:
                break
            t0_val = best[i]
            # Compute quadratic coefficients for corr as function of h[i]
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n - 1), n):
                k_idx = n - 1 + off
                if off == 0:
                    a[k_idx] += 1.0
                    b[k_idx] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k_idx] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k_idx] += (-best[i - off])
            
            base = corr - a * t0_val - b * t0_val * t0_val
            
            # Try candidate values
            cands = [0.0, 1.0, t0_val, 0.1, 0.5, 0.9, 1.0 - t0_val]
            cands = [max(0.0, min(1.0, c)) for c in cands]
            cands = list(set([round(c, 4) for c in cands]))
            
            best_t = t0_val
            best_f = np.max(base + a * t0_val + b * t0_val * t0_val) * dx
            
            for t in cands:
                if abs(t - t0_val) < 1e-10:
                    continue
                f = np.max(base + a * t + b * t * t) * dx
                if f < best_f - 1e-14:
                    best_f = f
                    best_t = t
            
            # Golden section if improved
            if best_t != t0_val:
                lo = max(0.0, best_t - 0.3)
                hi = min(1.0, best_t + 0.3)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                for _ in range(50):
                    if hi - lo < 1e-12:
                        break
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                t_opt = (lo + hi) / 2.0
                f_opt = np.max(base + a * t_opt + b * t_opt * t_opt) * dx
                if f_opt < best_f:
                    best_t = t_opt
                    best_f = f_opt
            
            if best_f < max_val - 1e-13:
                # Update with sum constraint
                diff = best[i] - best_t
                best[i] = best_t
                others = np.where(np.arange(n) != i)[0]
                free_mask = (best[others] > 1e-10) & (best[others] < 1.0 - 1e-10)
                if np.sum(free_mask) > 0:
                    best[others[free_mask]] -= diff / np.sum(free_mask)
                else:
                    best[others] -= diff / len(others)
                best = np.clip(best, 0.0, 1.0)
                corr = compute_corr(best)
                max_val = np.max(corr) * dx
                improved = True
        
        if not improved:
            break
    
    return best, max_val

def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET
    _START = time.time()
    _BUDGET = budget_s
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get initial solution from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 640
        parent_h = generate_structured_binary(parent_n, parent_n // 2, 0)
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n
    
    n = best_n
    
    # Phase 1: Binary local search at original resolution
    h_bin = make_binary_from_parent(parent_h)
    h_bin, bin_max, bin_corr = binary_greedy_search(h_bin, n, time_limit=250)
    h_bin_test = project_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin_test)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin_test.copy()
    
    # Phase 2: SA to potentially find better binary
    if remaining() > 100:
        h_sa, sa_max, sa_corr = binary_sa(h_bin, n, time_limit=150, seed=42)
        h_sa_test = project_feasible(h_sa, n)
        c5_sa = compute_c5(h_sa_test)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa_test.copy()
    
    # Phase 3: Fractional polishing from best binary
    if remaining() > 50:
        h_start = best_h.copy()
        h_pol = slsqp_polish(h_start, n, 5000, 300)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    if remaining() > 30:
        h_cd, c5_cd = coord_descent_frac(best_h, n, n_sweeps=25)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 4: Try different resolutions
    for rn in [512, 768, 896, 1024, 1152, 1280]:
        if remaining() < 80:
            break
        if rn == n:
            continue
        
        # Interpolate to new resolution
        h_r = interp_binary(best_h, best_n, rn)
        # Binary local search
        h_r, _, _ = binary_greedy_search(h_r, rn, time_limit=80, max_no_improve=40)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        
        if remaining() > 40:
            h_r2 = slsqp_polish(h_r, rn, 5000, 200)
            c5_r2 = compute_c5(h_r2)
            if c5_r2 < c5_r:
                h_r = h_r2
                c5_r = c5_r2
        
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn
    
    # Phase 5: More SA from best
    if remaining() > 80:
        n_cur = best_n
        h_cur = (best_h > 0.5).astype(float)
        k = n_cur // 2
        cnt = int(np.sum(h_cur))
        if cnt > k:
            oi = np.where(h_cur == 1)[0]
            h_cur[oi[np.argsort(best_h[oi])[:cnt-k]]] = 0
        elif cnt < k:
            zi = np.where(h_cur == 0)[0]
            h_cur[zi[np.argsort(best_h[zi])[::-1][:k-cnt]]] = 1
        
        h_sa2, _, _ = binary_sa(h_cur, n_cur, time_limit=100, seed=99)
        h_sa2 = project_feasible(h_sa2, n_cur)
        c5_sa2 = compute_c5(h_sa2)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            best_n = n_cur
    
    # Phase 6: Final fractional polish
    if remaining() > 20:
        h_final = slsqp_polish(best_h, best_n, 20000, 200)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    if remaining() > 15:
        h_cd2, c5_cd2 = coord_descent_frac(best_h, best_n, n_sweeps=15)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    # Verify constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)
        final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
