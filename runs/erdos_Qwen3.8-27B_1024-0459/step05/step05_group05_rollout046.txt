def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v

def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        np_ = n - abs(off)
        if off >= 0:
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

def _sc(x):
    return np.sum(x) - _TGT

    return np.ones(_N)

def slsqp_smooth(h, n, beta, maxiter=500):

    _TGT = n / 2.0

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        np_ = n - abs(off)
        if off >= 0:
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g


def _sc(x):
    return np.sum(x) - _TGT


def _scj(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.success or res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def corr_gradient_at(h, k_off, n):
    g = np.zeros(n)
    np_ = n - abs(k_off)
    if k_off >= 0:
        g[k_off:k_off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def top_shifts_focus(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-9:
            break
        k_top = idx[0]
        k_bot = idx[-1]
        off_t = k_top - (n - 1)
        off_b = k_bot - (n - 1)
        g_t = corr_gradient_at(h, off_t, n)
        g_b = corr_gradient_at(h, off_b, n)
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015, 0.5, 0.2, 0.002, 0.008]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def binary_local_search_v2(h_bin, n, rng, time_limit=60):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            for idx in ones_idx[perm[:n_ones - n_half]]:
                h[idx] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            for idx in zeros_idx[perm[:n_half - n_ones]]:
                h[idx] = 1

    dx_v = 2.0 / n
    corr = compute_corr(h)
    best_max = float(np.max(corr))
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    top_k = 5
    best_delta_history = []

    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > 15000:
            break

        corr = compute_corr(h)
        max_val = float(np.max(corr))
        top_idx = np.argsort(-corr)[:top_k]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0.0
        best_swap = None
        n_try = 40

        for _ in range(n_try):
            if rng.random() < 0.6:
                k_hot = top_idx[rng.randint(len(top_idx))]
                off = k_hot - (n - 1)
                if off >= 0:
                    valid_n = n - off
                    faults = np.where((h[:valid_n] == 1) & (h[off:off + valid_n] == 0))[0]
                else:
                    ao = -off
                    valid_n = n - ao
                    faults = np.where((h[ao:ao + valid_n] == 1) & (h[:valid_n] == 0))[0]
                if len(faults) > 0:
                    fi = faults[rng.randint(len(faults))]
                    if off >= 0:
                        i = fi
                    else:
                        i = fi + ao
                    j = zeros[rng.randint(len(zeros))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        # Try 2-swap moves (swap two pairs simultaneously)
        if no_improve > 50:
            n_2swap = 5
            for _ in range(n_2swap):
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if len(set([i1, i2, j1, j2])) < 4:
                    continue
                prev = [h[i1], h[i2], h[j1], h[j2]]
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                nc = float(np.max(compute_corr(h)))
                delta = nc - max_val
                h[i1], h[i2], h[j1], h[j2] = prev
                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_swap = (i1, i2, j1, j2)

        if best_swap is not None:
            if len(best_swap) == 2:
                i, j = best_swap
                h[i] = 0
                h[j] = 1
            else:
                i1, i2, j1, j2 = best_swap
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            no_improve = 0
            new_max = float(np.max(compute_corr(h)))
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            if no_improve > 200:
                n_flip = rng.randint(2, 4)
                for _ in range(n_flip):
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0
                    h[j] = 1
                no_improve = 0

    return best_h, best_max * dx_v


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgradient_max(h, n, n_iter=200):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 1.0
        top_idx = np.where(corr >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]
        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            g += corr_gradient_at(h, off, n)
        g /= len(top_idx)
        g -= np.mean(g)
        g = np.where(h > 0.99, -np.abs(g), g)
        g = np.where(h < 0.01, np.abs(g), g)
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.9
            if step < 0.0005:
                break
    return best_h, best_c5


def coordinate_descent(h, n, max_iter=500):
    """Cyclic coordinate descent: update each variable greedily."""
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_half = n // 2
    for it in range(max_iter):
        if remaining() < 3:
            break
        improved = False
        for i in range(n):
            # Try moving h[i] up or down by small amounts
            orig = h[i]
            if orig < 1.0:
                for delta in [0.05, 0.1, 0.02, 0.2]:
                    new_val = orig + delta
                    if new_val > 1.0:
                        continue
                    h[i] = new_val
                    h = project_feasible(h, n)
                    c = compute_c5(h)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        improved = True
                        break
            if orig > 0.0:
                for delta in [0.05, 0.1, 0.02, 0.2]:
                    new_val = orig - delta
                    if new_val < 0.0:
                        continue
                    h[i] = new_val
                    h = project_feasible(h, n)
                    c = compute_c5(h)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        improved = True
                        break
            if h[i] != orig:
                h[i] = h[i]  # keep
            else:
                h[i] = orig
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        n = parent_n
        h = parent_h.copy()
    except Exception:
        n = 384
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h = project_feasible(h, n)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        best_n = n

    # Phase 1: Progressive SLSQP refinement at parent resolution
    for beta in [200, 1000, 5000, 20000, 100000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-shifts equalization (multiple passes)
    if remaining() > 20:
        for tk in [3, 5, 7, 10]:
            h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=400)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # Phase 3: Binary local search (aggressive)
    if remaining() > 40:
        h_bin = binarize(h, n)
        sa_time = min(120, max(20, int(remaining() * 0.30)))
        h_b, c5_b = binary_local_search_v2(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
            # Continuous refinement from binary
            h_c = slsqp_smooth(h_b, n, 15000, 500)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                h = h_c.copy()
            else:
                h = h_b.copy()
        else:
            h = best_h.copy()

    # Phase 4: Subgradient + top-shifts
    if remaining() > 20:
        h_sg, c5_sg = subgradient_max(h, n, n_iter=200)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 5: Diverse starts
    if remaining() > 60:
        # Thue-Morse
        h_tm = thue_morse(n, rng)
        h_tm = slsqp_smooth(h_tm, n, 5000, 400)
        h_tm, c5_tm = top_shifts_focus(h_tm, n, 5, 200)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            best_n = n
            h = h_tm.copy()

        # Complement of parent
        h_comp = 1.0 - best_h
        h_comp = project_feasible(h_comp, n)
        h_comp = slsqp_smooth(h_comp, n, 5000, 400)
        c5_comp = compute_c5(h_comp)
        if c5_comp < best_c5:
            best_c5 = c5_comp
            best_h = h_comp.copy()
            best_n = n
            h = h_comp.copy()

        # Random starts
        for rs in range(5):
            if remaining() < 20:
                break
            h_r = rng.uniform(0, 1, n)
            h_r = project_feasible(h_r, n)
            h_r = slsqp_smooth(h_r, n, 5000, 400)
            h_r, c5_r = top_shifts_focus(h_r, n, 5, 150)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = n
                h = h_r.copy()

        # Perturbed best
        h_p = best_h + rng.randn(n) * 0.08
        h_p = project_feasible(h_p, n)
        h_p = slsqp_smooth(h_p, n, 10000, 500)
        h_p, c5_p = top_shifts_focus(h_p, n, 5, 200)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = n
            h = h_p.copy()

    # Phase 6: Resolution sweep
    if remaining() > 80:
        n_base = best_n
        n_candidates = []
        for delta in range(-32, 33, 2):
            nt = n_base + delta
            if 160 <= nt <= 600 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [256, 320, 384, 448, 512, 576, 640]:
            if 160 <= nt <= 600 and nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates = list(set(n_candidates))
        n_candidates.sort(key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:10]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

            for beta in [5000, 20000]:
                if remaining() < 15:
                    break
                h_t = slsqp_smooth(h_t, n_t, beta, 400)
            h_t, c5_t = top_shifts_focus(h_t, n_t, 5, 200)

            if remaining() > 25 and c5_t < best_c5 + 0.002:
                h_t_bin = binarize(h_t, n_t)
                h_tb, c5_tb = binary_local_search_v2(h_t_bin, n_t, rng, time_limit=30)
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb.copy()
                    h_tc = slsqp_smooth(h_t, n_t, 20000, 400)
                    c5_tc = compute_c5(h_tc)
                    if c5_tc < c5_t:
                        c5_t = c5_tc
                        h_t = h_tc.copy()

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 7: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for beta in [10000, 30000, 100000, 300000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    for tk in [3, 5, 8, 12]:
        if remaining() < 10:
            break
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary burst
    if remaining() > 30:
        h_bin = binarize(h, n)
        h_b, c5_b = binary_local_search_v2(h_bin, n, rng, time_limit=min(60, int(remaining() * 0.5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = slsqp_smooth(h_b, n, 50000, 500)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Final subgradient + top-shifts polish
    if remaining() > 10:
        h_sg, c5_sg = subgradient_max(best_h, best_n, n_iter=150)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        h_eq, c5_eq = top_shifts_focus(best_h, best_n, top_k=5, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
