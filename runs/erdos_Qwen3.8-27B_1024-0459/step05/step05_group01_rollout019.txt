                h[i] = 1.0
                h[j] = 0.0
                delta = c_new - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
            if best_pair is not None:
                i, j = best_pair
                h[i] = 0.0
                h[j] = 1.0
                best_c5 = compute_c5(h)
                best_h = h.copy()
                improved = True
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
    return best_h, best_c5


def simulated_annealing_binary(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    T = 0.1
    T_min = 0.001
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if len(set([i1, i2, j1, j2])) < 4:
            continue
        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, T_min)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
            no_improve += 1
        T *= 0.9995
        if T < T_min:
            T = T_min
    return best_h, best_c5


def coordinate_refine(h, n, max_sweeps=40):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0, old]
            if 0.0 < old < 1.0:
                cands.extend([old * 0.5, old + (1.0 - old) * 0.5,
                              max(0.0, old - 0.1), min(1.0, old + 0.1)])
            cands = [c for c in cands if 0.0 <= c <= 1.0]
            cands = list(set(cands))
            best_c5_i = best_c5
            best_val_i = old
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_c5_i - 1e-13:
                    best_c5_i = c5_v
                    best_val_i = c
                    improved_sweep = True
            h[i] = best_val_i
            h = project_feasible(h, n)
            if best_c5_i < best_c5:
                best_c5 = best_c5_i
        if not improved_sweep:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP on parent
    h = best_h.copy()
    for beta in [200, 1000, 5000, 30000, 200000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: FISTA with smooth max
    if remaining_time() > 20:
        h_f, c5_f = fista_optimize(h, best_n, beta=5000, max_iter=3000)
        if c5_f < best_c5 - 1e-15:
            best_c5 = c5_f
            best_h = h_f.copy()
            h = h_f.copy()

    # Phase 3: Top-k equalization (aggressive)
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=3, max_iters=1000)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=5, max_iters=800)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: More SLSQP at high beta
    for beta in [100000, 500000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Multi-resolution search
    n_cands = []
    for d in [-16, -8, -4, -2, 2, 4, 8, 16, 32, 64, 96, 128, 192, 256, 384]:
        nn = best_n + d
        if 160 <= nn <= 1200 and nn % 2 == 0:
            n_cands.append(nn)
    for m in [0.8, 0.9, 1.1, 1.25, 1.5, 1.75, 2.0, 2.5]:
        nn = int(best_n * m)
        if nn % 2 != 0:
            nn += 1
        if 160 <= nn <= 1200:
            n_cands.append(nn)
    n_cands = sorted(set(n_cands), key=lambda x: abs(x - best_n))

    for n_t in n_cands[:10]:
        if remaining_time() < 20:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt = h_feas
        for beta in [2000, 20000, 100000]:
            if remaining_time() < 8:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=400)
        if remaining_time() > 8:
            h_opt, c5_tmp = top_k_equalize(h_opt, n_t, top_k=4, max_iters=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 6: Alternative starting points
    alt_starts = [theta_morse(best_n), random_balanced(best_n, rng)]
    h_shift = np.roll(best_h, 2)
    alt_starts.append(project_feasible(h_shift, best_n))
    h_shift2 = np.roll(best_h, best_n // 4)
    alt_starts.append(project_feasible(h_shift2, best_n))

    for h_alt in alt_starts:
        if remaining_time() < 25:
            break
        h_opt = h_alt
        for beta in [5000, 50000, 300000]:
            if remaining_time() < 8:
                break
            h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=500)
        h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=4, max_iters=400)
        if c5_opt < best_c5 - 1e-15:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 7: Binary search + SA
    if remaining_time() > 40:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        if int(np.sum(h_bin)) != n_half:
            si = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[si[:n_half]] = 1.0
        sa_t = min(120, max(15, int(remaining_time() * 0.25)))
        h_b, c5_b = simulated_annealing_binary(h_bin, best_n, time_limit=sa_t, rng=rng)
        h_b, c5_b2 = binary_local_search(h_b, best_n, time_limit=min(60, remaining_time() - 5), rng=rng)
        c5_b = min(c5_b, c5_b2)
        if c5_b < best_c5:
            best_h_bin = h_b.copy()
        h_b_c = project_feasible(h_b.copy(), best_n)
        for beta in [20000, 200000, 1000000]:
            if remaining_time() < 8:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=600)
        h_b_c, c5_b_c = top_k_equalize(h_b_c, best_n, top_k=5, max_iters=500)
        if c5_b_c < best_c5 - 1e-15:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()

    # Phase 8: Final intensive refinement on best solution
    n = best_n
    h = best_h.copy()
    for beta in [200000, 1000000, 3000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=1200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=3, max_iters=600)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
        h_eq, c5_eq = top_k_equalize(h, n, top_k=5, max_iters=400)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 10:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=30)
        if c5_cr < best_c5 - 1e-15:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 2000000, maxiter=600)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
