        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-10:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
        else:
            lr = max(lr * 0.7, 1e-6)
    
    return best_h, best_c5

def slsqp_smooth(h, n, beta, maxiter=500):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    tgt = n / 2.0
    
    def obj(x):
        corr = compute_corr_array(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr_array(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k=5, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr_array(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]
        
        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.01, 0.02, 0.03, 0.05, 0.005, 0.08, 0.015, 0.04, 0.1]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: Binary local search at parent resolution
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    
    sa_time1 = min(250, int(remaining() * 0.2))
    if sa_time1 > 30:
        h_b1, c5_b1 = binary_local_search(h_bin, n, time_limit=sa_time1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n
    
    # Phase 2: Try Thue-Morse at n=512
    if remaining() > 100:
        n512 = 512
        h_tm = thue_morse(n512)
        c5_tm = compute_c5(h_tm)
        t_bi = min(200, int(remaining() * 0.15))
        if t_bi > 20:
            h_b512, c5_b512 = binary_local_search(h_tm, n512, time_limit=t_bi, rng=np.random.RandomState(seed+1))
            if c5_b512 < best_c5:
                best_c5 = c5_b512
                best_h = h_b512.copy()
                best_n = n512
    
    # Phase 3: Try interpolation to higher resolution
    if remaining() > 80:
        for n_t in [n + 64, 512, 768]:
            if remaining() < 40:
                break
            if n_t == best_n:
                continue
            if n_t % 2 != 0:
                continue
            if n_t < 128 or n_t > 1024:
                continue
            h_interp = interp_to(best_h, best_n, n_t)
            h_bin_t = (h_interp > 0.5).astype(float)
            nt_half = n_t // 2
            nt_ones = int(np.sum(h_bin_t))
            if nt_ones != nt_half:
                if nt_ones > nt_half:
                    idx = np.where(h_bin_t == 1)[0]
                    h_bin_t[idx[:nt_ones - nt_half]] = 0
                else:
                    idx = np.where(h_bin_t == 0)[0]
                    h_bin_t[idx[:nt_half - nt_ones]] = 1
            t_bi = min(100, int(remaining() * 0.08))
            if t_bi > 15:
                h_b3, c5_b3 = binary_local_search(h_bin_t, n_t, time_limit=t_bi, rng=np.random.RandomState(seed+n_t))
                if c5_b3 < best_c5:
                    best_c5 = c5_b3
                    best_h = h_b3.copy()
                    best_n = n_t
    
    # Phase 4: Continuous refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    for beta in [50, 200, 500, 1000, 3000, 8000, 20000, 50000]:
        if remaining() < 15:
            break
        h_new, c5_new = projected_gradient(h, n, beta, max_iters=1000, lr_init=0.05)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: SLSQP refinement
    if remaining() > 30:
        for beta in [500, 2000, 10000]:
            if remaining() < 15:
                break
            h_new = slsqp_smooth(h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
    
    # Phase 6: Top-shifts equalization
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 7: Try finer resolution
    if remaining() > 60:
        n_fine = best_n + 64
        if n_fine % 2 == 0 and n_fine <= 1024:
            h_fine = interp_to(best_h, best_n, n_fine)
            h_fine = project_feasible(h_fine, n_fine)
            for beta in [500, 2000, 10000]:
                if remaining() < 15:
                    break
                h_fine, c5_f = projected_gradient(h_fine, n_fine, beta, max_iters=800, lr_init=0.03)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_fine.copy()
                    best_n = n_fine
                    h = h_fine.copy()
            if remaining() > 15:
                h_fine, c5_f2 = top_shifts_equalize(h_fine, n_fine, top_k=6, max_iters=200)
                if c5_f2 < best_c5:
                    best_c5 = c5_f2
                    best_h = h_fine.copy()
                    best_n = n_fine
    
    # Phase 8: Final polish
    n = best_n
    h = best_h.copy()
    for beta in [5000, 20000, 50000]:
        if remaining() < 8:
            break
        h_new, c5_new = projected_gradient(h, n, beta, max_iters=500, lr_init=0.01)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    if remaining() > 10:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=8, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
