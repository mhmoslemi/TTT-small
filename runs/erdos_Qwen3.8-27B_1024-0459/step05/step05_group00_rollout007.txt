def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        np_ = n - offset
        g[offset:] += (1.0 - h[:np_])
        g[:np_] -= h[offset:]
    else:
        a = -offset
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def smooth_max_full(h, n, beta, maxiter=500):

    def obj(x):
        corr = compute_corr(x)
        m = corr.max()
        cl = np.clip(beta * (corr - m), -60.0, 60.0)
        return (m + np.log(np.exp(cl).sum()) / beta) * (2.0/n)
    
    def objg(x):
        corr = compute_corr(x)
        m = corr.max()
        cl = np.clip(beta * (corr - m), -60.0, 60.0)
        ex = np.exp(cl)
        s = ex.sum()
        if s < 1e-300:
            imax = np.argmax(corr)
            return grad_corr_k(x, imax, n) * (2.0/n)
        w = ex / s
        g = np.zeros(n)
        for idx in range(len(corr)):
            wk = w[idx] * (2.0/n)
            if wk < 1e-15:
                continue
            g += wk * grad_corr_k(x, idx, n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None:

    except Exception:
        pass
    return h

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def cutting_plane(h, n, K=30, delta=0.1):
    """LP cutting plane: linearize top K correlations."""
    global _T0, _BUDGET
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    
    # LP: min t
    # s.t. grad_f_k(h)·h_new - t <= grad_f_k(h)·h - f_k(h)   for k in idx
    #      0 <= h_new <= 1
    #      sum(h_new) = n/2
    #      ||h_new - h||_inf <= delta
    
    n_var = n + 1  # h_new (n) + t (1)
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    
    A_ub = []
    b_ub = []
    
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(n_var)
        row[:n] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(np.dot(g, h) - fk)
    
    # Trust region
    I = np.eye(n)
    A_trust1 = np.hstack([I, np.zeros((n, 1))])
    A_trust2 = np.hstack([-I, np.zeros((n, 1))])
    A_ub.append(A_trust1)
    A_ub.append(A_trust2)
    b_ub.extend([delta, delta])
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    
    return h.copy(), corr[idx[0]]
