**Walsh/Hadamard construction:**
A Hadamard matrix of order n gives a binary sequence where the autocorrelation is bounded. For a Hadamard matrix H of order n, the rows are ±1 vectors with inner product 0 (for different rows). If we map +1→1, -1→0, we get a binary sequence with specific correlation properties.

For a Hadamard matrix of order n (must be divisible by 4, except n=1,2), the rows have:
- Sum of squares: n (always n, since entries are ±1)
- Inner product with all-ones row: 0
- Inner product with row 0: 0

The autocorrelation of a Hadamard row (shifted) is:
For a row v of a Hadamard matrix, v·(v shifted by d) is not necessarily 0 for all d≠0. It's 0 only for d=0 (orthogonality with other rows).

Actually, what we want is: for the binary sequence b (mapped from Hadamard row), the cross-correlation with its "complement" (1-b) at all lags should be small.

Hadamard row v = (v_1, ..., v_n) with v_i ∈ {±1}.
Binary: b_i = (v_i + 1)/2.
Complement: 1-b_i = (1-v_i)/2.

Cross-correlation at lag d: sum_i b_i * (1-b_{i+d})
= sum_i (v_i+1)/2 * (1-v_{i+d})/2
= 1/4 * sum_i (v_i+1)(1-v_{i+d})
= 1/4 * sum_i [v_i - v_i*v_{i+d} + 1 - v_{i+d}]
= 1/4 * [sum_i v_i - sum_i v_i*v_{i+d} + n - sum_i v_{i+d}]

For a Hadamard row, sum_i v_i = 0 (since it's orthogonal to the all-ones row).
So: = 1/4 * [-sum_i v_i*v_{i+d} + n]
= 1/4 * [n - (v, v shifted by d)]

For d=0: (v,v) = n, so cross-corr = 0. ✓
For d≠0: (v, v shifted by d) depends on the specific row and the cyclic shift.

cross-corr(d) = 1/4 * (n-(-1)) = (n+1)/4

        offset = int(lag) - (n - 1)
        if offset == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if best[i] == best[j]:
                continue
        else:
            if offset > 0:
                valid_i = np.where((best[:n - offset] == 1) & (best[offset:] == 0))[0]
                valid_j = np.where((best[:n - offset] == 0) & (best[offset:] == 1))[0]
            else:
                ao = -offset
                valid_i = np.where((best[ao:] == 1) & (best[:n - ao] == 0))[0] + ao
                valid_j = np.where((best[ao:] == 0) & (best[:n - ao] == 1))[0] + ao
            if len(valid_i) == 0 or len(valid_j) == 0:
                i = rng.randint(n)
                j = rng.randint(n)
                if best[i] == best[j]:
                    continue
            else:
                i = valid_i[rng.randint(len(valid_i))]
                j = valid_j[rng.randint(len(valid_j))]
        if i == j:
            continue
        best[i], best[j] = best[j], best[i]
        new_corr = compute_corr(best)
        new_c5 = np.max(new_corr) * dx
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            corr = new_corr
            stall = 0
        else:
            best[i], best[j] = best[j], best[i]
            stall += 1
        if stall > 5000:
            break
    return best, best_c5

def thue_morse_binary(n):
    tm = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x > 0:
            c += x & 1
            x >>= 1
        tm[i] = c % 2
    # Ensure exactly n//2 ones
    n_ones = int(np.sum(tm))
    target = n // 2
    if n_ones > target:
        ones = np.where(tm == 1)[0]
        rng = np.random.RandomState(7)
        perm = rng.permutation(ones)
        for idx in perm[:n_ones - target]:
            tm[idx] = 0
    elif n_ones < target:
        zeros = np.where(tm == 0)[0]
        rng = np.random.RandomState(7)
        perm = rng.permutation(zeros)
        for idx in perm[:target - n_ones]:
            tm[idx] = 1
    return tm

def even_spaced_binary(n):
    h = np.zeros(n)
    h[::2] = 1.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[ones[target:]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - n_ones]] = 1
    return h

def slsqp_smooth(h, n, beta, maxiter=300):
    dx = 2.0 / n
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        return (m + np.log(wsum) / beta) * dx
    def grad_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        ncorr = 2 * n - 1
        grad = np.zeros(n)
        for k in range(ncorr):
            w = weights[k]
            if w < 1e-15:
                continue
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= w * x[:npairs]
        grad *= dx
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            return trial
    except Exception:
        pass
    return h.copy()

_N_OPT = 0
_BETA_OPT = 1000.0

def interpolated_source(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = to_binary(|even_spaced_binary(best_n), best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Binary simulated annealing + hill climbing at parent resolution
    h_bin = to_binary(best_h, best_n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    if remaining_time() > 300:
        t1 = remaining_time() * 0.35
        h_sa, c5_sa = binary_sa_climb(h_bin, best_n, t1, seed=seed)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 2: Targeted local search on the best binary
    if remaining_time() > 120:
        t2 = remaining_time() * 0.15
        h_tl, c5_tl = targeted_local_search(best_h, best_n, t2, seed=seed+10)
        if c5_tl < best_c5:
            best_c5 = c5_tl
            best_h = h_tl.copy()

    # Phase 3: More SA with different seed / temperature
    if remaining_time() > 200:
        t3 = remaining_time() * 0.25
        h_sa2, c5_sa2 = binary_sa_climb(best_h, best_n, t3, seed=seed+99)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Try different resolutions
    res_list = [512, 768, 1024, 800]
    res_list = [r for r in res_list if r != best_n and r % 2 == 0 and 256 <= r <= 2048]
    for rn in res_list:
        if remaining_time() < 80:
            break
        h_r = interpolated_source(best_h, best_n, rn)
        h_r = to_binary(h_r, rn)
        c5_r = compute_c5(h_r)
        if remaining_time() > 50:
            h_r, c5_r = binary_sa_climb(h_r, rn, min(remaining_time() * 0.3, 60.0), seed=seed + rn)
        if remaining_time() > 20:
            h_r, c5_r = targeted_local_search(h_r, rn, min(remaining_time() * 0.15, 30.0), seed=seed + rn + 50)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 5: Thue-Morse initialization + optimization
    if remaining_time() > 100:
        for tn in [512, 640, 1024]:
            if remaining_time() < 50:
                break
            h_tm = thue_morse_binary(tn)
            c5_tm = compute_c5(h_tm)
            if remaining_time() > 40:
                h_tm, c5_tm = binary_sa_climb(h_tm, tn, min(remaining_time() * 0.3, 50.0), seed=seed + 1000 + tn)
            c5_tm = compute_c5(h_tm)
            if c5_tm < best_c5:
                best_c5 = c5_tm
                best_h = h_tm.copy()
                best_n = tn

    # Phase 6: Smooth SLSQP refinement (allow non-binary)
    if remaining_time() > 40:
        h_smooth = to_binary(best_h, best_n).astype(float) * 0.5 + 0.5
        h_smooth = project_feasible(h_smooth, best_n)
        for beta in [2000, 20000]:
            if remaining_time() < 20:
                break
            h_smooth = slsqp_smooth(h_smooth, best_n, beta, 250)
            c5_s = compute_c5(h_smooth)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_smooth.copy()

    # Phase 7: Final intensive targeted search
    if remaining_time() > 30:
        h_fin = to_binary(best_h, best_n)
        c5_fin = compute_c5(h_fin)
        if remaining_time() > 25:
            h_fin, c5_fin = targeted_local_search(h_fin, best_n, remaining_time() * 0.4, seed=seed + 2000)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 8: More one more SA round if time
    if remaining_time() > 60:
        h_more, c5_more = binary_sa_climb(best_h, best_n, remaining_time() * 0.5, seed=seed + 5000)
        if c5_more < best_c5:
            best_c5 = c5_more
            best_h = h_more.copy()

    # Final: project and validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        # Try to fix
        target = best_n / 2.0
        diff = target - np.sum(best_h)
        if abs(diff) > 1e-10:
            free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
            if np.sum(free) > 0:
                best_h[free] += diff / np.sum(free)
                best_h = np.clip(best_h, 0.0, 1.0)
                final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
