            # Total delta for k≠n1: (t-old)*[(1-h[i-off_k]) - h[i+off_k]] = (t-old)*A[k]
            # For k=n1 (off=0): j=i and j=off are the same, term is h[i]*(1-h[i])
            # Delta = t*(1-t) - old*(1-old)
            
            # So: base[k] = corr[k] - old*A[k] for k≠n1
            #     base[n1] = corr[n1] - old*(1-old) [NOTE: this is corr[n1] - old + old²]
            # corr_new[k] = base[k] + t*A[k] for k≠n1
            # corr_new[n1] = base[n1] + t*(1-t) [NOTE: t*(1-t) = t - t²]
            
            base = best_corr - old * A
            base[n1] = best_corr[n1] - old * (1.0 - old)
            
            # Minimize f(t) = max(max_{k≠n1}(base[k] + A[k]*t), base[n1] + t*(1-t))
            # over t ∈ [0, 1]
            
            # Fast evaluation
            def eval_f(t):
                vals = base + A * t
                vals[n1] = base[n1] + t * (1.0 - t)
                return np.max(vals)
            
            f_old = eval_f(old)
            
            # Quick check: try t=0, t=1, t=0.5
            t_best = old
            f_best = f_old
            
            for t_try in [0.0, 0.5, 1.0]:
                if abs(t_try - old) < 1e-8:
                    continue
                ft = eval_f(t_try)
                if ft < f_best - 1e-12:
                    f_best = ft
                    t_best = t_try
            
            # Ternary search refinement around old
            if t_best != old:
                if t_best > old:
                    lo = max(0.0, old - 0.01)
                    hi = min(1.0, t_best + 0.1)
                else:
                    lo = max(0.0, t_best - 0.1)
                    hi = min(1.0, old + 0.01)
                
                if hi - lo < 1e-8:
                    hi = lo + 1e-6
                
                # Ternary search
                for _ in range(40):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    f_m1 = eval_f(m1)
                    f_m2 = eval_f(m2)
                    if f_m1 < f_m2:
                        hi = m2
                    else:
                        lo = m1
                t_try = (lo + hi) / 2.0
                f_try = eval_f(t_try)
                if f_try < f_best - 1e-12:
                    t_best = t_try
                    f_best = f_try
            
            if f_best < f_old - 1e-12:
                best[i] = t_best
                delta = t_best - old
                best_corr += A * delta
                best_corr[n1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                improved = True
        
        sweeps_done = sweep + 1
        if not improved:
            break
    
    final_c5 = np.max(best_corr) * dx
    return best, final_c5

def pair_coord_descent(h, n, max_sweeps=100):
    """Optimize pairs of adjacent coordinates."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    n1 = n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 10:
            break
        improved = False
        for i in range(0, n - 1, 2):
            j = i + 1
            old_i = best[i]
            old_j = best[j]
            
            # Compute A for i and j
            im_i = i - offsets
            ip_i = i + offsets
            im_j = j - offsets
            ip_j = j + offsets
            
            hi_im = np.zeros(ncorr)
            hi_im[(im_i >= 0) & (im_i < n)] = best[im_i[((im_i >= 0) & (im_i < n))]]
            hi_ip = np.zeros(ncorr)
            hi_ip[(ip_i >= 0) & (ip_i < n)] = best[ip_i[((ip_i >= 0) & (ip_i < n))]]
            A_i = (1.0 - hi_im) - hi_ip
            A_i[n1] = 0.0
            
            hj_im = np.zeros(ncorr)
            hj_im[(im_j >= 0) & (im_j < n)] = best[im_j[((im_j >= 0) & (im_j < n))]]
            hj_ip = np.zeros(ncorr)
            hj_ip[(ip_j >= 0) & (ip_j < n)] = best[ip_j[((ip_j >= 0) & (ip_j < n))]]
            A_j = (1.0 - hj_im) - hj_ip
            A_j[n1] = 0.0
            
            base = best_corr - old_i * A_i - old_j * A_j
            base[n1] = best_corr[n1] - old_i*(1-old_i) - old_j*(1-old_j)
            
            # Minimize max_k [base[k] + A_i[k]*ti + A_j[k]*tj] with self-terms at n1
            # Grid search on a small grid
            f_old = np.max(best_corr)
            best_ti, best_tj, best_f = old_i, old_j, f_old
            
            # Candidate values
            candidates_i = [0.0, old_i, 1.0]
            candidates_j = [0.0, old_j, 1.0]
            
            for ti in candidates_i:
                for tj in candidates_j:
                    vals = base + A_i * ti + A_j * tj
                    vals[n1] += ti*(1-ti) + tj*(1-tj)
                    fv = np.max(vals)
                    if fv < best_f - 1e-12:
                        best_f = fv
                        best_ti, best_tj = ti, tj
            
            # Local refinement
            if best_ti != old_i or best_tj != old_j:
                # 2D ternary-ish: try small perturbations
                for _ in range(20):
                    improved_local = False
                    for di in [-0.05, 0.05, -0.02, 0.02, -0.01, 0.01]:
                        for dj in [-0.05, 0.05, -0.02, 0.02, -0.01, 0.01]:
                            ti = best_ti + di
                            tj = best_tj + dj
                            if 0 <= ti <= 1 and 0 <= tj <= 1:
                                vals = base + A_i * ti + A_j * tj
                                vals[n1] += ti*(1-ti) + tj*(1-tj)
                                fv = np.max(vals)
                                if fv < best_f - 1e-12:
                                    best_f = fv
                                    best_ti, best_tj = ti, tj
                                    improved_local = True
                    if not improved_local:
                        break
            
            if best_f < f_old - 1e-12:
                best[i] = best_ti
                best[j] = best_tj
                best_corr += A_i * (best_ti - old_i) + A_j * (best_tj - old_j)
                best_corr[n1] += best_ti*(1-best_ti) + best_tj*(1-best_tj) - old_i*(1-old_i) - old_j*(1-old_j)
                improved = True
        
        if not improved:
            break
    
    final_c5 = np.max(best_corr) * dx
    return best, final_c5

def slsqp_refine(h, n, beta, maxiter):
    """SLSQP with softmax-smoothed max."""
    dx = 2.0 / n
    n1 = n - 1
    
    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        # Softmax smoothing
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        return (m + np.log(wsum) / beta) * dx
    
    def smooth_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        weights = exp_vals / np.sum(exp_vals)
        grad = np.zeros(n)
        for k in range(len(corr)):
            w = weights[k]
            if w < 1e-14:
                continue
            off = k - n1
            if off >= 0:
                npairs = n - off
                if npairs > 0:
                    grad[off:off+npairs] += w * (1.0 - x[:npairs])
                    grad[:npairs] -= w * x[off:off+npairs]
            else:
                aoff = -off
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                    grad[aoff:aoff+npairs] -= w * x[:npairs]
        grad *= dx
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-10:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp,
