```python
def grad_corr_at(h, n, k):
    """Gradient of corr[k] w.r.t. h."""
    g = np.zeros(n)
    offset = k - (n-1)
    if offset >= 0:
        np_ = n - offset
        # Left: i in [0, np_-1], contribution (1-h[i+offset])
        g[:np_] += (1.0 - h[offset:offset+np_])
        # Right: i in [offset, n-1], contribution -h[i-offset]
        g[offset:] -= h[:np_]
    else:
        m = -offset
        np_ = n - m
        # Left: i in [m, n-1], contribution (1-h[i-m])
        g[m:] += (1.0 - h[:np_])
        # Right: i in [0, np_-1], contribution -h[i+m]
        g[:np_] -= h[m:m+np_]
    return g
```
