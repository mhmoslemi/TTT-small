def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and n > 0:
        # Distribute remaining difference to closest-to-0.5 elements
        order = np.argsort(np.abs(h - 0.5))
        for j in range(min(10, n)):
            idx = order[j]
            adj = diff
            if adj > 0:
                adj = min(adj, 1.0 - h[idx])
            else:
                adj = max(adj, -h[idx])
            h[idx] += adj
            diff -= adj
            if abs(diff) < 1e-15:
                break
    return h

def thue_morse_init(n, shift=0):
    h = np.zeros(n)
    for i in range(n):
        v = (i + shift) % n
        cnt = 0
        tmp = v
        while tmp:
            cnt ^= (tmp & 1)
            tmp >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def random_binary_init(n, rng):

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx

    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def coord_descent(h, n, max_sweeps=100, random_order=False, rng=None, gs_iters=30):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        order = list(range(n))
        if random_order and rng is not None:
            rng.shuffle(order)

        for i in order:
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))

            # Try a few quick candidates
            t_best = old
            f_best = f_old
            for t_cand in [0.0, 1.0, 0.5]:
                f_cand = max(base[n - 1] + t_cand - t_cand * t_cand,
                             np.max(base + A * t_cand))
                if f_cand < f_best - 1e-14:
                    t_best = t_cand
                    f_best = f_cand

            # Golden section if we found improvement
            if t_best != old:
                if t_best > old:
                    lo = old
                    hi = min(1.0, t_best + 0.3)
                else:
                    lo = max(0.0, t_best - 0.3)
                    hi = old
                if hi - lo < 1e-12:
                    hi = lo + 1e-6
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                def _ev(t):
                    mid = base + A * t
                    return max(mid[n - 1] + t - t * t, np.max(mid))

                f1 = _ev(c1)
                f2 = _ev(c2)
                for _ in range(gs_iters):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _ev(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = _ev(c2)
                t_best = (lo + hi) / 2.0
                f_best = _ev(t_best)

            if f_best < f_old - 1e-12:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                improved = True

        if not improved:
            break

    best_c5 = np.max(best_corr) * dx
    return best, best_c5


def subgradient_step(h, n, step_size, n_active=10):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    threshold = max_corr - 0.3
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        idx = active[np.argsort(corr[active])[::-1][:n_active]]
        active = idx

    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h
    grad /= gnorm
    h_new = h - step_size * grad
    return project_feasible(h_new, n)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def block_coord_descent(h, n, max_sweeps=30, block_size=2, gs_iters=20):

    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for start in range(0, n - block_size + 1, block_size):
            idxs = list(range(start, start + block_size))
            old_vals = np.array([best[i] for i in idxs])

            # Compute combined A matrix: A[b, k] = coefficient of t_b in corr[k]
            A = np.zeros((block_size, ncorr))
            for b, i in enumerate(idxs):
                i_plus = i + offsets
                i_minus = i - offsets
                valid_plus = (i_plus >= 0) & (i_plus < n)
                valid_minus = (i_minus >= 0) & (i_minus < n)
                if np.any(valid_minus):
                    A[b, valid_minus] += (1.0 - best[i_minus[valid_minus]])
                if np.any(valid_plus):
                    A[b, valid_plus] -= best[i_plus[valid_plus]]
                A[b, n - 1] = 0.0

            base = best_corr - A.dot(old_vals)
            # Handle quadratic term at offset 0
            quad = old_vals * (1.0 - old_vals)
            base[n - 1] = best_corr[n - 1] - quad.sum()

            def _eval_block(t):
                new_corr = base + A.dot(t)
                new_quad = t * (1.0 - t)
                new_corr[n - 1] += new_quad.sum()
                return np.max(new_corr)

            f_old = _eval_block(old_vals)
            t_best = old_vals.copy()
            f_best = f_old

            # Try boundary points and midpoint
            candidates = [np.zeros(block_size), np.ones(block_size),
                          np.full(block_size, 0.5), old_vals.copy()]
            for t_c in candidates:
                t_c_clipped = np.clip(t_c, 0, 1)
                f_c = _eval_block(t_c_clipped)
                if f_c < f_best - 1e-14:
                    f_best = f_c
                    t_best = t_c_clipped.copy()

            # Nelder-Mead-like local search
            if not np.allclose(t_best, old_vals):
                for iteration in range(30):
                    found = False
                    for dim in range(block_size):
                        for direction in [-0.1, 0.1, -0.03, 0.03]:
                            t_try = t_best.copy()
                            t_try[dim] += direction
                            t_try = np.clip(t_try, 0, 1)
                            f_try = _eval_block(t_try)
                            if f_try < f_best - 1e-14:
                                f_best = f_try
                                t_best = t_try
                                found = True
                    if not found:
                        break

            if f_best < f_old - 1e-12:
                for b, i in enumerate(idxs):
                    best[i] = t_best[b]
                best_corr += A.dot(t_best - old_vals)
                new_quad = t_best * (1.0 - t_best)
                best_corr[n - 1] += new_quad.sum() - quad.sum()
                improved = True

        if not improved:
            break

    best_c5 = np.max(best_corr) * dx
    return best, best_c5
