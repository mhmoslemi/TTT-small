def continuous_coord_descent(h, n, max_sweeps=80, rng=None):
    if rng is None:

    best = h.copy()

            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))

            gr = (np.sqrt(5.0) - 1.0) / 2.0

            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

                mid[n - 1] = base[n - 1] + t - t * t

            f2 = _eval_t(c2)

                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = _eval_t(c1)

                    c2 = lo + gr * (hi - lo)

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

        if res.x is not None and np.all(np.isfinite(res.x)):

    h_bin, M_bin = binary_coord_descent(h_bin, n, max_sweeps=150, rng=rng, top_k=1)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Phase 2: Continuous refinement from binary solution
    if remaining_time() > 60:
        h_cont = best_h.copy()
        h_cont, c5_cont = continuous_coord_descent(h_cont, n, max_sweeps=80, rng=rng)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 3: SLSQP polish
    if remaining_time() > 40:
        h_sl = slsqp_refine(best_h, n, 3000, 400)
        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            h2, c5_2 = continuous_coord_descent(h_sl, n, max_sweeps=40, rng=rng)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 4: Multi-start binary search
    if remaining_time() > 100:
        h_start_bin = round_to_binary(best_h, n)
        h_ms, M_ms = multi_start_binary(h_start_bin, n, n_starts=6, max_sweeps=80, rng=rng)
        c5_ms = compute_c5(h_ms)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()
            if remaining_time() > 40:
                h_cont, c5_cont = continuous_coord_descent(h_ms, n, max_sweeps=40, rng=rng)
                if c5_cont < best_c5:
                    best_c5 = c5_cont
                    best_h = h_cont.copy()

    # Phase 5: Try higher resolution n=960
    if remaining_time() > 180:
        n2 = 960
        h_up = interp_to(best_h, best_n, n2)
        h_up = project_feasible(h_up, n2)
        h_up_bin = round_to_binary(h_up, n2)
        h_up_bin, _ = binary_coord_descent(h_up_bin, n2, max_sweeps=100, rng=rng, top_k=1)
        c5_up = compute_c5(h_up_bin)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up_bin.copy()
            best_n = n2
            if remaining_time() > 60:
                h_c, c5_c = continuous_coord_descent(h_up_bin, n2, max_sweeps=50, rng=rng)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c.copy()
                    best_n = n2

    # Phase 6: Try higher resolution n=1280
    if remaining_time() > 200:
        n3 = 1280
        h_up3 = interp_to(best_h, best_n, n3)
        h_up3 = project_feasible(h_up3, n3)
        h_up3_bin = round_to_binary(h_up3, n3)
        h_up3_bin, _ = binary_coord_descent(h_up3_bin, n3, max_sweeps=80, rng=rng, top_k=1)
        c5_up3 = compute_c5(h_up3_bin)
        if c5_up3 < best_c5:
            best_c5 = c5_up3
            best_h = h_up3_bin.copy()
            best_n = n3
            if remaining_time() > 50:
                h_c3, c5_c3 = continuous_coord_descent(h_up3_bin, n3, max_sweeps=40, rng=rng)
                if c5_c3 < best_c5:
                    best_c5 = c5_c3
                    best_h = h_c3.copy()
                    best_n = n3

    # Phase 7: Top-k binary search (exploration)
    if remaining_time() > 60:
        n_cur = best_n
        h_cur_bin = round_to_binary(best_h, n_cur)
        h_exp, M_exp = binary_coord_descent(h_cur_bin, n_cur, max_sweeps=60, rng=rng, top_k=3)
        c5_exp = compute_c5(h_exp)
        if c5_exp < best_c5:
            best_c5 = c5_exp
            best_h = h_exp.copy()

    # Phase 8: Final continuous polish
    if remaining_time() > 30:
        n_cur = best_n
        h_pol, c5_pol = continuous_coord_descent(best_h, n_cur, max_sweeps=60, rng=rng)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 9: SLSQP final
    if remaining_time() > 15:
        n_cur = best_n
        h_final = slsqp_refine(best_h, n_cur, 10000, 300)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 10: One more coordinate descent
    if remaining_time() > 10:
        n_cur = best_n
        h_cd, c5_cd = continuous_coord_descent(best_h, n_cur, max_sweeps=40, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
