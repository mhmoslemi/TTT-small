def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
            h[idx] = min(max(h[idx] + diff, 0.0), 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def coord_descent(h, n, max_sweeps=200):
    best = h.copy()
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if _time_left() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)
            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0
            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)
            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))

            t_wall = [0.0, low if (low := min(1.0, max(0.0, (2.0 * (base + A * 0.5)[n-1] - 0.25) / (2.0 * (base + A * 0.5)[n-1]) if False else 0.5))) else 0.5]
            # Simplified: just try golden section
            candidates = [0.0, old, 1.0, 0.25, 0.5, 0.75]
            t_best = old
            f_best = f_old
            for tc in candidates:
                arr = base + A * tc
                arr[n - 1] = base[n - 1] + tc - tc * tc
                fc = np.max(arr)
                if fc < f_best - 1e-14:
                    f_best = fc
                    t_best = tc

            if t_best != old:
                if t_best > old:
                    lo = old
                    hi = min(1.0, t_best + 0.3)
                else:
                    lo = max(0.0, t_best - 0.3)
                    hi = old
                if hi - lo < 1e-12:
                    pass
                else:
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    arr1 = base + A * c1
                    arr1[n - 1] = base[n - 1] + c1 - c1 * c1
                    f1 = np.max(arr1)
                    arr2 = base + A * c2
                    arr2[n - 1] = base[n - 1] + c2 - c2 * c2
                    f2 = np.max(arr2)
                    for _ in range(50):
                        if f1 < f2:
                            hi = c2
                            c2 = c1
                            f2 = f1
                            c1 = hi - gr * (hi - lo)
                            arr1 = base + A * c1
                            arr1[n - 1] = base[n - 1] + c1 - c1 * c1
                            f1 = np.max(arr1)
                        else:
                            lo = c1
                            c1 = c2
                            f1 = f2
                            c2 = lo + gr * (hi - lo)
                            arr2 = base + A * c2
                            arr2[n - 1] = base[n - 1] + c2 - c2 * c2
                            f2 = np.max(arr2)
                    t_best = (lo + hi) / 2.0
                    arr_b = base + A * t_best
                    arr_b[n - 1] = base[n - 1] + t_best - t_best * t_best
                    f_best = np.max(arr_b)

            if f_best < f_old - 1e-13:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                improved = True
        if not improved:
            break
    return best, float(np.max(best_corr) * dx)

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(2 * n - 1)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _g_smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]

def _g_smooth_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]

def _g_cons_sum(x):
    return np.sum(x) - _N_WORK / 2.0

def _g_cons_sum_jac(x):
    return np.ones(_N_WORK)

def slsqp_refine(h, n, beta, maxiter=400):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_cons_sum, 'jac': _g_cons_sum_jac}]
    try:
        res = minimize(_g_smooth_obj, h, jac=_g_smooth_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-25, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def targeted_lp_adjust(h, n, top_m=5, step_frac=0.5, max_iters=200):
    """Use LP to find small adjustments that reduce the top correlations."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for it in range(max_iters):
        if _time_left() < 5:
            break
        max_val = np.max(best_corr)
        active_mask = best_corr >= max_val - 0.5
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(best_corr)])
        # Sort by correlation value, take top_m
        if len(active_idx) > top_m:
            order = np.argsort(best_corr[active_idx])[::-1][:top_m]
            active_idx = active_idx[order]

        # Build LP: min t s.t. L_k(delta) <= t, sum(delta)=0, bounds
        # L_k(delta) = best_corr[k] + sum_i (dcorr_k/dh_i)*delta_i
        # For each active k, compute gradient of corr_k w.r.t. h
        A_mat = np.zeros((len(active_idx), n))
        for ii, k_idx in enumerate(active_idx):
            offset = int(k_idx) - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    A_mat[ii, offset:offset + npairs] += (1.0 - best[:npairs])
                    A_mat[ii, :npairs] -= best[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    A_mat[ii, :npairs] += (1.0 - best[aoff:aoff + npairs])
                    A_mat[ii, aoff:aoff + npairs] -= best[:npairs]

        m_act = len(active_idx)
        # Variables: [delta_0, ..., delta_{n-1}, t] (n+1 vars)
        # Objective: minimize t -> c = [0,...,0,1]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        # Constraints: A_mat @ delta - t <= -best_corr[active] -> rearrange
        # -A_mat @ delta + t >= best_corr[active]
        # linprog form: A_ub @ x <= b_ub
        # A_mat @ delta - t <= 0 for each active (but we want corr_k + A_k@delta <= t)
        # -> A_mat @ delta - t <= best_corr[active]
        A_ub = np.zeros((m_act, n + 1))
        A_ub[:, :n] = A_mat
        A_ub[:, n] = -1.0
        b_ub = best_corr[active_idx].copy()

        # Equality: sum(delta) = 0
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # Bounds: -best[i] + eps <= delta_i <= 1 - best[i] - eps, t >= 0
        lb = np.maximum(best - 0.01, 0.0) * (-1.0)
        lb = -np.minimum(best, 0.01)
        ub = 1.0 - best
        ub = np.minimum(ub, 0.01)
        bounds_lp = list(zip(lb, ub)) + [(0.0, max_val)]

        try:
            from scipy.optimize import linprog as _lp
            res = _lp(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
            if res.success:
                delta = res.x[:n]
                # Take a fraction of the step
                step = step_frac * delta
                h_new = best + step
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-15:
                    best = h_new
                    best_c5 = c5_new
                    best_corr = compute_corr(best)
                else:
                    step_frac *= 0.5
                    if step_frac < 0.001:
                        break
            else:
                break
        except Exception:
            break

    return best, best_c5

def multi_start_optimize(h_init, n, max_sweeps_cd=200, n_starts=5):
    """Try multiple perturbations and keep the best."""
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    h = h_init.copy()
    h, c5 = coord_descent(h, n, max_sweeps=max_sweeps_cd)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    rng = np.random.RandomState(12345)
    for s in range(n_starts):
        if _time_left() < 30:
            break
        h_pert = best_h.copy()
        noise_scale = 0.1 / (1.0 + s * 0.5)
        mask = rng.random(n) < 0.2
        h_pert[mask] += rng.randn(np.sum(mask)) * noise_scale
        h_pert = project_feasible(h_pert, n)
        h_new, c5_new = coord_descent(h_pert, n, max_sweeps=max_sweeps_cd // 2)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        # Thue-Morse init
        best_h = np.zeros(best_n)
        for i in range(best_n):
            v = i
            cnt = 0
            while v:
                cnt ^= (v & 1)
                v >>= 1
            best_h[i] = 1.0 if cnt == 1 else 0.0
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Coordinate descent from parent at original resolution
    h = best_h.copy()
    h, c5 = coord_descent(h, n, max_sweeps=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP at original resolution
    h = best_h.copy()
    for beta in [2000, 8000, 30000]:
        if _time_left() < 20:
            break
        h_new = slsqp_refine(h, n, beta, 400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Targeted LP at original resolution
    if _time_left() > 60:
        h = best_h.copy()
        h, c5 = targeted_lp_adjust(h, n, top_m=4, step_frac=0.8, max_iters=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Move to higher resolution (n=1280)
    high_n = 1280
    h_high = interp_to(best_h, best_n, high_n)
    h_high = project_feasible(h_high, high_n)
    c5_high = compute_c5(h_high)

    # Phase 5: Coordinate descent at high resolution
    h_high, c5_high = coord_descent(h_high, high_n, max_sweeps=150)
    if c5_high < best_c5:
        best_c5 = c5_high
        best_h = h_high.copy()
        best_n = high_n

    # Phase 6: SLSQP at high resolution with increasing beta
    h = best_h.copy() if best_n == high_n else h_high
    if best_n != high_n:
        h = interp_to(best_h, best_n, high_n)
        h = project_feasible(h, high_n)
    for beta in [5000, 20000, 80000]:
        if _time_left() < 30:
            break
        h_new = slsqp_refine(h, high_n, beta, 500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = high_n
            h = h_new.copy()

    # Phase 7: Targeted LP at high resolution
    if _time_left() > 60:
        h = best_h.copy()
        if best_n != high_n:
            h = interp_to(best_h, best_n, high_n)
            h = project_feasible(h, high_n)
        h, c5 = targeted_lp_adjust(h, high_n, top_m=5, step_frac=0.7, max_iters=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = high_n

    # Phase 8: More coordinate descent at high res
    if _time_left() > 40:
        h = best_h.copy()
        h_new, c5_new = coord_descent(h, best_n, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final SLSQP polish
    if _time_left() > 20:
        for beta in [100000, 300000]:
            if _time_left() < 10:
                break
            h_new = slsqp_refine(best_h, best_n, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 10: Final targeted LP
    if _time_left() > 15:
        h, c5 = targeted_lp_adjust(best_h, best_n, top_m=3, step_frac=0.5, max_iters=80)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 11: Final coordinate descent sweep
    if _time_left() > 10:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, final_c5, best_n)
