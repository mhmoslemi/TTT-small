```python
def compute_grad_for_lag(h, k, n):
    """Gradient of corr[k] with respect to h."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        # (1-h[j+offset]) for j in [0, n-1-offset]
        j1 = np.arange(0, n - offset)
        grad[j1] += (1.0 - h[j1 + offset])
        # -h[j-offset] for j in [offset, n-1]
        j2 = np.arange(offset, n)
        grad[j2] -= h[j2 - offset]
    else:
        a = -offset
        # (1-h[j-a]) for j in [a, n-1]
        j1 = np.arange(a, n)
        grad[j1] += (1.0 - h[j1 - a])
        # -h[j+a] for j in [0, n-1-a]
        j2 = np.arange(0, n - a)
        grad[j2] -= h[j2 + a]
    return grad
```

The Thue-Morse sequence is designed to minimize autocorrelation. For a binary sequence of length n with n/2 ones and n/2 zeros, the Thue-Morse sequence minimizes the maximum autocorrelation.

For the connection to our problem: if h is binary, corr[k] counts 1→0 transitions at shift k. We want to minimize the maximum count over k.

For n=640, the best binary solution should have max corr[k] around n/4 = 160, giving C₅ = 160 * 2/640 = 0.5. That's much worse than 0.38.

So the optimal solution is NOT binary. It uses fractional values (between 0 and 1) to spread out the correlation more evenly.

This makes sense: with fractional values, we can fine-tune the correlation at each lag independently (to some extent).

```python
def compute_A(h, i, n):
    """Compute d(corr[k])/d(h[i]) for all k."""
    ncorr = 2 * n - 1
    A = np.zeros(ncorr)
    offsets = np.arange(-(n-1), n)  # offset[k] = k - (n-1)
    
    # For offset >= 0:
    #   (a) h[i] in "h" position: valid if offset <= n-1-i, contribution (1-h[i+offset])
    #   (b) h[i] in "(1-h)" position: valid if offset <= i, contribution -h[i-offset]
    
    pos = offsets >= 0
    idx_pos = np.where(pos)[0]
    off_pos = offsets[idx_pos]
    
    # (a): offset <= n-1-i
    mask_a = off_pos <= (n - 1 - i)
    if np.any(mask_a):
        other = i + off_pos[mask_a]
        A[idx_pos[mask_a]] += (1.0 - h[other])
    
    # (b): offset <= i (and i <= n-1, which is always true)
    mask_b = off_pos <= i
    if np.any(mask_b):
        other = i - off_pos[mask_b]
        A[idx_pos[mask_b]] -= h[other]
    
    # For offset < 0, a = -offset:
    #   (a) h[i] in "h[j+a]" position: valid if a <= i <= n-1, contribution (1-h[i-a])
    #   (b) h[i] in "(1-h[j])" position: valid if i <= n-1-a, i.e., i+a <= n-1, contribution -h[i+a]
    
    neg = offsets < 0
    idx_neg = np.where(neg)[0]
    off_neg = offsets[idx_neg]
    a_vals = -off_neg  # positive
    
    # (a): a <= i
    mask_a = a_vals <= i
    if np.any(mask_a):
        other = i - a_vals[mask_a]
        A[idx_neg[mask_a]] += (1.0 - h[other])
    
    # (b): i + a <= n-1
    mask_b = (i + a_vals) <= (n - 1)
    if np.any(mask_b):
        other = i + a_vals[mask_b]
        A[idx_neg[mask_b]] -= h[other]
    
    return A
```

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

_START_TIME = 0.0
_BUDGET_S = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def compute_A_vec(h, i, n):
    """Compute d(corr[k])/d(h[i]) for all k (vector of length 2n-1)."""
    ncorr = 2 * n - 1
    A = np.zeros(ncorr)
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    pos = offsets >= 0
    idx_pos = np.where(pos)[0]
    off_pos = offsets[idx_pos]

    mask_a = off_pos <= (n - 1 - i)
    if np.any(mask_a):
        other = i + off_pos[mask_a]
        A[idx_pos[mask_a]] += (1.0 - h[other])

    mask_b = off_pos <= i
    if np.any(mask_b):
        other = i - off_pos[mask_b]
        A[idx_pos[mask_b]] -= h[other]

    neg = offsets < 0
    idx_neg = np.where(neg)[0]
    off_neg = offsets[idx_neg]
    a_vals = -off_neg

    mask_a = a_vals <= i
    if np.any(mask_a):
        other = i - a_vals[mask_a]
        A[idx_neg[mask_a]] += (1.0 - h[other])

    mask_b = (i + a_vals) <= (n - 1)
    if np.any(mask_b):
        other = i + a_vals[mask_b]
        A[idx_neg[mask_b]] -= h[other]

    return A


def grad_corr_k(h, k, n):
    """Full gradient of corr[k] w.r.t. h (vector of length n)."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        j1 = np.arange(0, n - offset)
        grad[j1] += (1.0 - h[j1 + offset])
        j2 = np.arange(offset, n)
        grad[j2] -= h[j2 - offset]
    else:
        a = -offset
        j1 = np.arange(a, n)
        grad[j1] += (1.0 - h[j1 - a])
        j2 = np.arange(0, n - a)
        grad[j2] -= h[j2 + a]
    return grad


def coord_descent(h, n, max_sweeps=200):
    """Fast coordinate descent with 1D minimization."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        sweeps_start = time.time()
        for i in range(n):
            old = best[i]
            A = compute_A_vec(best, i, n)

            # 1D minimization: f(t) = max(f_lin(t), f_quad(t))
            t_grid = np.linspace(0.0, 1.0, 25)
            t_minus_old = t_grid - old

            # f_lin(t) = max_{k != n-1} (best_corr[k] + A[k]*(t-old))
            # f_quad(t) = best_corr[n-1] + (t-old)*(1-t-old)
            lin_part = best_corr[None, :] + A[None, :] * t_minus_old[:, None]
            lin_part[:, n - 1] = -np.inf
            f_lin = np.max(lin_part, axis=1)

            f_quad = best_corr[n - 1] + t_minus_old * (1.0 - t_grid - old)
            f_total = np.maximum(f_lin, f_quad)
            best_t_idx = int(np.argmin(f_total))
            f_best_grid = f_total[best_t_idx]

            # Golden section refinement
            t_lo = t_grid[max(0, best_t_idx - 1)]
            t_hi = t_grid[min(24, best_t_idx + 1)]
            if t_hi - t_lo > 1e-8:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(20):
                    c1 = t_hi - gr * (t_hi - t_lo)
                    c2 = t_lo + gr * (t_hi - t_lo)
                    u1 = c1 - old
                    nc1 = best_corr + A * u1
                    nc1[n - 1] = best_corr[n - 1] + u1 * (1.0 - c1 - old)
                    f1 = np.max(nc1)
                    u2 = c2 - old
                    nc2 = best_corr + A * u2
                    nc2[n - 1] = best_corr[n - 1] + u2 * (1.0 - c2 - old)
                    f2 = np.max(nc2)
                    if f1 < f2:
                        t_hi = c2
                    else:
                        t_lo = c1
                t_mid = (t_lo + t_hi) / 2.0
                u_mid = t_mid - old
                nc_mid = best_corr + A * u_mid
                nc_mid[n - 1] = best_corr[n - 1] + u_mid * (1.0 - t_mid - old)
                f_mid = np.max(nc_mid)
                if f_mid < f_best_grid:
                    f_best_grid = f_mid
                    best_t = t_mid
                else:
                    best_t = t_grid[best_t_idx]
            else:
                best_t = t_grid[best_t_idx]

            if f_best_grid < np.max(best_corr) - 1e-14:
                best[i] = best_t
                upt = best_t - old
                best_corr += A * upt
                best_corr[n - 1] += upt * (1.0 - best_t - old)
                improved = True

        if not improved:
            break

    best_c5 = np.max(best_corr) * dx
    return best, best_c5


def cutting_plane_lp(h, n, max_iters=40):
    """Kelley cutting-plane method."""
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    cuts_a = []
    cuts_g = []

    # Initial cuts for the top lags
    corr_cur = compute_corr(h_cur)
    top_lags = np.argsort(corr_cur)[::-1][:30]
    for k in top_lags:
        g = grad_corr_k(h_cur, int(k), n)
        a_val = corr_cur[int(k)] - g @ h_cur
        cuts_a.append(a_val)
        cuts_g.append(g.copy())

    for it in range(max_iters):
        if remaining_time() < 5:
            break

        # Solve LP: min C s.t. a_j + g_j^T h <= C, 0<=h<=1, sum(h)=n/2
        h_var = cp.Variable(n)
        C_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]

        for a_val, g_vec in zip(cuts_a, cuts_g):
            constraints.append(a_val + g_vec @ h_var <= C_var)

        # Also add bound: C <= c5_cur (must not get worse)
        constraints.append(C_var <= c5_cur + 0.001)

        objective = cp.Minimize(C_var)
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, abs_acc=1e-9, rel_acc=1e-9)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=5000, eps=1e-9)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.array(h_var.value).flatten(), 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < c5_cur - 1e-12:
            c5_cur = c5_new
            h_cur = h_new.copy()

        # Add new cuts for violated lags
        corr_new = compute_corr(h_cur)
        C_sol = C_var.value if C_var.value is not None else c5_cur / dx
        violated = np.where(corr_new > C_sol * dx + 1e-8)[0]

        if len(violated) == 0:
            # No violations, convergence
            break

        # Add top violated lags (limit to 10 new per iteration)
        viol_sorted = violated[np.argsort(corr_new[violated])[::-1]][:10]
        n_new = 0
        for k in viol_sorted:
            k = int(k)
            g = grad_corr_k(h_cur, k, n)
            a_val = corr_cur[k] - g @ h_cur  # Use h_cur for linearization
            cuts_a.append(a_val)
            cuts_g.append(g.copy())
            n_new += 1
            if n_new >= 10:
                break

        # Also refresh: recompute corr_cur
        corr_cur = corr_new

        # Limit total cuts to prevent LP from getting too large
        if len(cuts_a) > 500:
            # Keep only the most relevant cuts (those closest to being active)
            # Evaluate all cuts at current solution
            cut_vals = np.array([a + g @ h_cur for a, g in zip(cuts_a, cuts_g)])
            # Keep cuts that are within 2x the max
            threshold = np.max(cut_vals) * 0.95
            keep_mask = cut_vals >= threshold
            if keep_mask.sum() < 50:
                keep_mask = np.argsort(cut_vals)[::-1][:200]
                new_mask = np.zeros(len(cuts_a), dtype=bool)
                new_mask[keep_mask] = True
                keep_mask = new_mask
            cuts_a = [cuts_a[i] for i in range(len(cuts_a)) if keep_mask[i]]
            cuts_g = [cuts_g[i] for i in range(len(cuts_g)) if keep_mask[i]]

    return h_cur, c5_cur


def subgradient_step(h, n, step_size, n_active=10):
    """Projected subgradient step on the max correlation."""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    threshold = max_corr - 0.3
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        idx = active[np.argsort(corr[active])[::-1][:n_active]]
        active = idx

    grad = np.zeros(n)
    for k_idx in active:
        k_idx = int(k_idx)
        g = grad_corr_k(h, k_idx, n)
        grad += g

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h
    grad /= gnorm
    h_new = h - step_size * grad
    return project_feasible(h_new, n)


def block_perturb(h, n, rng, block_len=None):
    """Reverse a random contiguous block of h."""
    h_new = h.copy()
    if block_len is None:
        block_len = rng.randint(max(5, n // 20), max(10, n // 4))
    start = rng.randint(0, n - block_len + 1)
    h_new[start:start + block_len] = h_new[start:start + block_len][::-1]
    return h_new


def slsqp_refine(h, n, beta, maxiter):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    ncorr = 2 * n - 1

    def smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        wsum = np.sum(np.exp(shifted))
        return (m + np.log(wsum) / beta) * dx

    def smooth_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.minimum(beta * (corr - m), 0.0)
        exp_vals = np.exp(shifted)
        wsum = np.sum(exp_vals)
        weights = exp_vals / wsum
        grad = np.zeros(n)
        for k in range(ncorr):
            w = weights[k]
            if w < 1e-14:
                continue
            g = grad_corr_k(x, k, n)
            grad += w * g
        grad *= dx
        return grad

    def constr_fun(x):
        return np.sum(x) - n / 2.0

    def constr_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]

    try:
        res = minimize(smooth_obj, h, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)
