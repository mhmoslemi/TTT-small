def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

def make_swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _TGT,
             'jac': lambda x: np.ones(_N)}]
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return m * (2.0 / n)
        return (m + np.log(s) / _BETA) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return np.zeros(n)
        w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ <= 0:
                    continue
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ <= 0:
                    continue
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h):

    except Exception:

    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return m * (2.0 / n)
        return (m + np.log(s) / _BETA) * (2.0 / n)
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return np.zeros(n)
        w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ <= 0:
                    continue
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ <= 0:
                    continue
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h):
            return h_new
    except Exception:
        pass
    return h

def lbfgs_optimize(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return m * (2.0 / n)
        return (m + np.log(s) / _BETA) * (2.0 / n)
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return np.zeros(n)
        w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ <= 0:
                    continue
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ <= 0:
                    continue
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 5000})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h):
            return h_new
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k=5, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ <= 0:
                    continue
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ <= 0:
                    continue
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]
        bot_idx = np.argsort(corr)[:min(5, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ <= 0:
                    continue
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ <= 0:
                    continue
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        improved = False
        for step in [0.005, 0.01, 0.02, 0.03, 0.05, 0.008, 0.015, 0.04, 0.003, 0.07]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try negative direction
            for step in [0.005, 0.01, 0.02, 0.03, 0.008]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def fourier_construct(n, K, rng):
    """Construct h from Fourier modes with random amplitudes."""
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    return h

def robust_continuous(h, n, time_limit=100, rng=None):
    """Multiple rounds of continuous optimization with alternating methods."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    start = time.time()
    round_num = 0
    while remaining() > 15 and time.time() - start < time_limit:
        round_num += 1
        # SLSQP
        if remaining() - (time.time() - start) > 20:
            for beta in [500, 2000, 10000, 50000]:
                if remaining() - (time.time() - start) < 10:
                    break
                h_new = slsqp_smooth(h, n, beta, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    h = h_new.copy()
        # L-BFGS-B
        if remaining() - (time.time() - start) > 20:
            for beta in [1000, 10000, 50000]:
                if remaining() - (time.time() - start) < 10:
                    break
                h_new = lbfgs_optimize(h, n, beta, maxiter=300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    h = h_new.copy()
        # Top-shifts equalization
        if remaining() - (time.time() - start) > 10:
            h_eq, c5_eq = top_shifts_equalize(h, n, top_k=6, max_iters=200)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                h = h_eq.copy()
        # Check for early termination
        if round_num > 3 and compute_c5(h) >= best_c5 - 1e-12:
            # Small perturbation and try again
            h_pert = h + rng.normal(0, 0.01, n)
            h_pert = project_feasible(h_pert, n)
            h = h_pert
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    n = best_n
    # Phase 1: Targeted binary local search
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    t1 = min(280, int(remaining() * 0.22))
    if t1 > 20:
        h_b1, c5_b1 = targeted_local_search(h_bin, n, time_limit=t1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
    # Phase 2: Simulated annealing with multi-scale moves
    t2 = min(320, int(remaining() * 0.25))
    if t2 > 30:
        h_b2, c5_b2 = simulated_annealing(best_h, n, time_limit=t2, rng=rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
    # Phase 3: Second targeted search from improved solution
    h_bin2 = (best_h > 0.5).astype(float)
    t3 = min(120, int(remaining() * 0.1))
    if t3 > 15:
        h_b3, c5_b3 = targeted_local_search(h_bin2, n, time_limit=t3, rng=rng)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()
    # Phase 4: Continuous refinement
    n = best_n
    h = project_feasible(best_h, n)
    t_cont = min(350, int(remaining() * 0.3))
    if t_cont > 20:
        h, c5_cont = robust_continuous(h, n, time_limit=t_cont, rng=rng)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h.copy()
    # Phase 5: Try different resolutions
    n_ref = best_n
    candidate_ns = set()
    for delta in [-32, -16, -8, 8, 16, 32, 64, 128]:
        candidate_ns.add(n_ref + delta)
    for f in [0.75, 1.25, 1.5]:
        candidate_ns.add(int(n_ref * f))
    candidate_ns = sorted([x for x in candidate_ns if 192 <= x <= 1024 and x % 2 == 0 and x != n_ref])
    candidate_ns.sort(key=lambda x: abs(x - n_ref))
    for n_t in candidate_ns[:3]:
        if remaining() < 25:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_t = project_feasible(h_interp, n_t)
        h_bin_t = (h_t > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        t_bi = min(35, int(remaining() * 0.04))
        if t_bi > 10:
            h_b_t, c5_b_t = targeted_local_search(h_bin_t, n_t, time_limit=t_bi, rng=rng)
            if c5_b_t < best_c5:
                best_c5 = c5_b_t
                best_h = h_b_t.copy()
                best_n = n_t
        if remaining() > 20:
            h_t2 = project_feasible(best_h, best_n)
            h_t2 = slsqp_smooth(h_t2, best_n, 10000, maxiter=200)
            c5_t2 = compute_c5(h_t2)
            if c5_t2 < best_c5:
                best_c5 = c5_t2
                best_h = h_t2.copy()
    # Phase 6: Try Fourier-based constructions at best resolution
    if remaining() > 40:
        for K in [5, 10, 15]:
            if remaining() < 20:
                break
            h_four = fourier_construct(best_n, K, rng)
            c5_four = compute_c5(h_four)
            if c5_four < best_c5 + 0.02:
                h_four = slsqp_smooth(h_four, best_n, 5000, maxiter=200)
                c5_four = compute_c5(h_four)
                if c5_four < best_c5:
                    best_c5 = c5_four
                    best_h = h_four.copy()
    # Phase 7: Final aggressive polish
    n = best_n
    h = project_feasible(best_h, n)
    for beta in [50000, 100000, 200000, 500000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    if remaining() > 15:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=10, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    # Final binary search on the continuous solution
    if remaining() > 20:
        h_bin_final = (best_h > 0.5).astype(float)
        n_final = len(best_h)
        n_half_f = n_final // 2
        n_ones_f = int(np.sum(h_bin_final))
        if n_ones_f != n_half_f:
            if n_ones_f > n_half_f:
                idx = np.where(h_bin_final == 1)[0]
                h_bin_final[idx[:n_ones_f - n_half_f]] = 0
            else:
                idx = np.where(h_bin_final == 0)[0]
                h_bin_final[idx[:n_half_f - n_ones_f]] = 1
        t_final = min(30, int(remaining() * 0.15))
        if t_final > 10:
            h_bf, c5_bf = targeted_local_search(h_bin_final, n_final, time_limit=t_final, rng=rng)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    return (best_h, best_c5, best_n)
