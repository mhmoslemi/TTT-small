def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0

    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _top_lag_obj(x, idx_top, lam):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    return np.dot(corr[idx_top], lam) * dx


def _top_lag_grad(x, idx_top, lam):
    n = _CUR_N
    dx = 2.0 / n
    g = np.zeros(n)
    for pos, k in enumerate(idx_top):
        wk = lam[pos] * dx
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500, ftol=1e-15):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    obj_val = _sm_obj(h0)
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < obj_val - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def top_lag_opt(h, n, top_k=6, maxiter=300):
    global _CUR_N
    _CUR_N = n
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    for it in range(maxiter):
        if remaining_time() < 5:
            break
        corr = compute_corr(h0)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min = corr[idx_top[-1]]
        if c_max - c_min < 1e-10:
            break
        lam = np.ones(top_k) / top_k
        lam[0] = 2.0 / top_k
        h0_new = project_feasible(h0, n)
        try:
            res = minimize(_top_lag_obj, h0_new,
                           lambda x: _top_lag_grad(x, idx_top, lam),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-14})
            if res.fun < _top_lag_obj(h0_new, idx_top, lam) - 1e-14:
                h0 = project_feasible(res.x, n)
            else:
                break
        except Exception:
            break
    return h0


def top_k_equalize(h, n, top_k=5, max_iters=400):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.003, 0.008, 0.015, 0.03, 0.06, 0.01, 0.02, 0.05, 0.1, 0.005]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            steps = [0.001, 0.002, 0.004, 0.006, 0.01]
            for step in steps:
                ht = project_feasible(h + step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def random_balanced(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def structured_start(n, rng):
    h = np.zeros(n)
    period = max(2, n // 32)
    for i in range(n):
        block = i // period
        pos_in_block = i % period
        if block % 2 == 0:
            h[i] = 1.0 if pos_in_block < period // 2 else 0.0
        else:
            h[i] = 0.0 if pos_in_block < period // 2 else 1.0
    h[rng.choice(n, min(20, n // 10), replace=False)] = 1.0 - h[rng.choice(n, min(20, n // 10), replace=False)]
    return project_feasible(h, n)


def binary_local_search(h, n, time_limit=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    dx = 2.0 / n
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_pairs = len(ones) * len(zeros)
        best_delta = -1e-12
        best_pair = None
        if n_pairs <= 50000:
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c_new = compute_c5(h)
                    h[i] = 1.0
                    h[j] = 0.0
                    delta = c_new - best_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (i, j)
        else:
            n_trials = min(500, n_pairs)
            for _ in range(n_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                c_new = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = c_new - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
            improved = True
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
    return best_h, best_c5


def coordinate_refine(h, n, max_sweeps=20):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0]
            if 0.0 < old < 1.0:
                cands.extend([0.25, 0.5, 0.75, old + 0.1, old - 0.1])
            cands = [c for c in cands if 0.0 <= c <= 1.0]
            best_cand_c5 = best_c5
            best_cand_val = old
            for c in cands:
                if abs(c - old) < 1e-6:
                    continue
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-13:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
        if not improved_sweep:
            break
        best_c5 = compute_c5(h)
    return h, best_c5


def multi_beta_slsqp(h, n, betas, maxiter=600):
    curr = project_feasible(h, n)
    for beta in betas:
        if remaining_time() < 15:
            break
        curr = slsqp_opt(curr, n, beta, maxiter=maxiter)
    return curr


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(384)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive progressive SLSQP at parent resolution
    h = best_h.copy()
    betas_p1 = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 25000]
    for beta in betas_p1:
        if remaining_time() < 40:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization + top-lag targeted optimization
    if remaining_time() > 30:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=8, max_iters=600)
        if c5_eq < best_c5 - 1e-16:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining_time() > 25:
        h_tl = top_lag_opt(h, best_n, top_k=6, maxiter=200)
        c5_tl = compute_c5(h_tl)
        if c5_tl < best_c5 - 1e-16:
            best_c5 = c5_tl
            best_h = h_tl.copy()
            h = h_tl.copy()

    # Phase 3: Higher beta SLSQP
    if remaining_time() > 30:
        for beta in [50000, 100000, 250000, 500000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_opt(h, best_n, beta, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Multi-resolution search
    if remaining_time() > 180:
        for target_n in [512, 640, 768]:
            if remaining_time() < 130:
                break
            if target_n % 2 != 0:
                continue
            h_interp = interp_to(best_h, best_n, target_n)
            h_feas = project_feasible(h_interp, target_n)
            h_t = h_feas.copy()
            betas_t = [100, 500, 2000, 10000, 50000, 200000]
            for beta in betas_t:
                if remaining_time() < 30:
                    break
                h_t = slsqp_opt(h_t, target_n, beta, maxiter=500)
            c5_t = compute_c5(h_t)
            h_t, c5_t = top_k_equalize(h_t, target_n, top_k=6, max_iters=200)
            if c5_t < best_c5 - 1e-16:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = target_n
                h = h_t.copy()

    # Phase 5: Alternative starting points at best_n
    alt_starts = [theta_morse(best_n), random_balanced(best_n, rng)]
    rng2 = np.random.RandomState(seed + 123)
    alt_starts.append(random_balanced(best_n, rng2))
    alt_starts.append(structured_start(best_n, rng))

    for h_alt in alt_starts:
        if remaining_time() < 50:
            break
        h_a = project_feasible(h_alt, best_n)
        c5_a = compute_c5(h_a)
        if c5_a >= best_c5 + 0.01:
            continue
        h_a = multi_beta_slsqp(h_a, best_n, [200, 1000, 5000, 20000, 100000], maxiter=400)
        c5_a = compute_c5(h_a)
        h_a, c5_a = top_k_equalize(h_a, best_n, top_k=6, max_iters=200)
        if c5_a < best_c5 - 1e-16:
            best_c5 = c5_a
            best_h = h_a.copy()
            h = h_a.copy()

    # Phase 6: Binary local search then continuous refinement
    if remaining_time() > 80:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0

        sa_time = min(90, max(15, int(remaining_time() * 0.25)))
        h_b, c5_b = binary_local_search(h_bin, best_n, time_limit=sa_time, rng=rng)

        if c5_b < best_c5 + 0.005:
            h_b_c = h_b.copy()
            h_b_c = multi_beta_slsqp(h_b_c, best_n, [500, 2000, 10000, 50000], maxiter=400)
            c5_bc = compute_c5(h_b_c)
            h_b_c, c5_bc = top_k_equalize(h_b_c, best_n, top_k=6, max_iters=200)
            if c5_bc < best_c5 - 1e-16:
                best_c5 = c5_bc
                best_h = h_b_c.copy()

        if c5_b < best_c5 - 1e-16:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 7: Final intensive refinement at best_n
    n = best_n
    h = best_h.copy()
    for beta in [100000, 300000, 1000000, 3000000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=10, max_iters=400)
        if c5_eq < best_c5 - 1e-16:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 8: Coordinate refinement
    if remaining_time() > 15:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=15)
        if c5_cr < best_c5 - 1e-16:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 2000000, maxiter=400)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-16:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure strict feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
