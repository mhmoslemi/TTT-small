def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]

                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]

            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def swap_local_search(h, n, max_sweeps=30):
    """For near-binary solutions, try swapping 1s and 0s."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 10:
            break
        improved = False
        ones_mask = best > 0.5
        zeros_mask = best < 0.5
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        if len(ones_idx) < 2 or len(zeros_idx) < 2:
            break
        rng_local = np.random.RandomState(sweep)
        n_tries = min(len(ones_idx) * len(zeros_idx), 500)
        for _ in range(n_tries):
            i = ones_idx[rng_local.randint(len(ones_idx))]
            j = zeros_idx[rng_local.randint(len(zeros_idx))]
            # Effect of h[i]: 1->0, h[j]: 0->1
            i_plus = i + offsets
            i_minus = -offsets + i
            valid_i_plus = (i_plus >= 0) & (i_plus < n)
            valid_i_minus = (i_minus >= 0) & (i_minus < n)
            A_i = np.zeros(len(offsets))
            if np.any(valid_i_minus):
                A_i[valid_i_minus] += (1.0 - best[i_minus[valid_i_minus]])
            if np.any(valid_i_plus):
                A_i[valid_i_plus] -= best[i_plus[valid_i_plus]]
            A_i[n - 1] = 0.0

            j_plus = j + offsets
            j_minus = -offsets + j
            valid_j_plus = (j_plus >= 0) & (j_plus < n)
            valid_j_minus = (j_minus >= 0) & (j_minus < n)
            A_j = np.zeros(len(offsets))
            if np.any(valid_j_minus):
                A_j[valid_j_minus] += (1.0 - best[j_minus[valid_j_minus]])
            if np.any(valid_j_plus):
                A_j[valid_j_plus] -= best[j_plus[valid_j_plus]]
            A_j[n - 1] = 0.0

            # New corr: best_corr + A_i*(-1) + A_j*(+1), quad term change
            new_corr = best_corr + (-A_i + A_j)
            old_quad_i = best[i] * (1.0 - best[i])
            new_quad_i = 0.0 * 1.0
            old_quad_j = best[j] * (1.0 - best[j])
            new_quad_j = 1.0 * 0.0
            new_corr[n - 1] += new_quad_i - old_quad_i + new_quad_j - old_quad_j

            new_max = np.max(new_corr)
            if new_max < best_c5 / dx - 1e-12:
                best[i] = 0.0
                best[j] = 1.0
                best_corr = new_corr
                best_c5 = new_max * dx
                improved = True

        if not improved:
            break
    return best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Thorough coordinate descent from parent
    if remaining_time() > 60:
        h = best_h.copy()
        h, c5 = coord_descent(h, n, max_sweeps=400)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: SLSQP refinement with increasing beta
    if remaining_time() > 30:
        h = best_h.copy()
        for beta in [2000, 8000, 30000, 100000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_refine(h, n, beta, 800)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-14:
                h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    # Phase 3: Coordinate descent after SLSQP
    if remaining_time() > 40:
        h = best_h.copy()
        h, c5 = coord_descent(h, n, max_sweeps=300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Multiple perturbations + re-optimization
    if remaining_time() > 80:
        n_perturb = 12
        for p in range(n_perturb):
            if remaining_time() < 30:
                break
            h_pert = best_h.copy()
            pert_type = p % 4
            if pert_type == 0:
                # Gaussian noise
                mask = rng.random(n) < 0.4
                scale = 0.2 / (1 + p * 0.2)
                h_pert[mask] += rng.randn(np.sum(mask)) * scale
            elif pert_type == 1:
                # Swap pairs
                ones = np.where(h_pert > 0.5)[0]
                zeros = np.where(h_pert < 0.5)[0]
                nsw = min(10, len(ones), len(zeros))
                for _ in range(nsw):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
            elif pert_type == 2:
                # Block shift
                blen = rng.randint(5, 20)
                bstart = rng.randint(0, n - blen)
                block = h_pert[bstart:bstart + blen].copy()
                h_pert[bstart:bstart + blen] = h_pert[bstart + blen:bstart + blen + blen]
                h_pert[bstart + blen:bstart + blen + blen] = block
                if bstart + 2 * blen < n:
                    h_pert[bstart + blen + blen:n] = np.concatenate([
                        h_pert[bstart:bstart + blen], h_pert[bstart + blen + blen:n]
                    ])
            else:
                # Smooth perturbation
                freq = rng.randint(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                pert = 0.15 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                h_pert += pert

            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = coord_descent(h_pert, n, max_sweeps=150)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

            if remaining_time() > 40:
                h_slq = slsqp_refine(h_new, n, 5000, 500)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    h_new, c5_new = coord_descent(h_slq, n, max_sweeps=100)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()

    # Phase 5: Try different resolutions
    if remaining_time() > 150:
        for try_n in [1280, 512]:
            if remaining_time() < 80:
                break
            if try_n == n:
                continue
            h_interp = interp_to(best_h, best_n, try_n)
            h_interp = project_feasible(h_interp, try_n)
            c5_int = compute_c5(h_interp)

            if c5_int < best_c5 + 0.005:
                h_new, c5_new = coord_descent(h_interp, try_n, max_sweeps=200)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = try_n

                if remaining_time() > 50:
                    h_slq = slsqp_refine(h_new, try_n, 8000, 600)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        best_n = try_n
                        h_new, c5_new = coord_descent(h_slq, try_n, max_sweeps=150)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            best_n = try_n

    # Phase 6: Thue-Morse and structured starts
    if remaining_time() > 80:
        for init_fn in [thue_morse_init, lambda nn: binary_init(nn, np.random.RandomState(seed + 7))]:
            if remaining_time() < 40:
                break
            n_try = best_n
            h_init = init_fn(n_try)
            c5_init = compute_c5(h_init)
            if c5_init < best_c5 + 0.02:
                h_new, c5_new = coord_descent(h_init, n_try, max_sweeps=200)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    if remaining_time() > 30:
                        h_slq = slsqp_refine(h_new, n_try, 10000, 500)
                        c5_slq = compute_c5(h_slq)
                        if c5_slq < best_c5:
                            best_c5 = c5_slq
                            best_h = h_slq.copy()
                            h_new, c5_new = coord_descent(h_slq, n_try, max_sweeps=100)
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h_new.copy()

    # Phase 7: Subgradient-style steps + coordinate descent
    if remaining_time() > 60:
        n_cur = best_n
        h = best_h.copy()
        h_corr = compute_corr(h).astype(float)
        for it in range(300):
            if remaining_time() < 5:
                break
            # Find active set
            max_val = np.max(h_corr)
            threshold = max_val * 0.95
            active = np.where(h_corr >= threshold)[0]
            if len(active) == 0:
                active = np.array([np.argmax(h_corr)])

            # Compute subgradient
            grad = np.zeros(n_cur)
            for k_idx in active:
                offset = int(k_idx) - (n_cur - 1)
                if offset >= 0:
                    npairs = n_cur - offset
                    if npairs > 0:
                        grad[offset:offset + npairs] += (1.0 - h[:npairs])
                        grad[:npairs] -= h[offset:offset + npairs]
                else:
                    aoff = -offset
                    npairs = n_cur - aoff
                    if npairs > 0:
                        grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                        grad[aoff:aoff + npairs] -= h[:npairs]

            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-16:
                break
            step = 0.01 / (1.0 + it * 0.05)
            h_new = h - step * grad / gnorm
            h_new = project_feasible(h_new, n_cur)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                h_corr = compute_corr(h).astype(float)
            elif it % 50 == 25 and remaining_time() > 20:
                h_cd, c5_cd = coord_descent(h, n_cur, max_sweeps=30)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    h = h_cd.copy()
                    h_corr = compute_corr(h).astype(float)
                else:
                    break

    # Phase 8: Final SLSQP polish with very high beta
    if remaining_time() > 15:
        n_cur = best_n
        h = best_h.copy()
        for beta in [50000, 200000, 500000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_refine(h, n_cur, beta, 1000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = coord_descent(best_h, n_cur, max_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One final SLSQP
    if remaining_time() > 8:
        n_cur = best_n
        h_new = slsqp_refine(best_h, n_cur, 200000, 600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
