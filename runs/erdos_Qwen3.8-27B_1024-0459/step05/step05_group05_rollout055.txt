            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-shifts equalization
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_focus(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
        h_eq2, c5_eq2 = top_shifts_focus(h, n, top_k=10, max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    # Phase 3: Intensive binary SA from best continuous
    if remaining() > 50:
        h_bin = binarize(h, n)
        sa_time = min(200, max(30, int(remaining() * 0.3)))
        h_b, c5_b = binary_sa(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
            h = h_b.copy()
        # Continuous refinement from binary
        if remaining() > 20:
            h_c = lbfgs_smooth(h_b, n, 10000, 500)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
            h_eq3, c5_eq3 = top_shifts_focus(h_c, n, 6, 300)
            if c5_eq3 < best_c5:
                best_c5 = c5_eq3
                best_h = h_eq3.copy()
                h = h_eq3.copy()

    # Phase 4: Binary SA from Thue-Morse
    if remaining() > 100:
        h_tm = thue_morse(n)
        h_tm, c5_tm = binary_sa(h_tm, n, rng, time_limit=min(150, int(remaining() * 0.25)))
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            best_n = n
            h = h_tm.copy()
            if remaining() > 20:
                h_c = lbfgs_smooth(h_tm, n, 10000, 400)
                c5_c = compute_c5(h_c)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c.copy()

    # Phase 5: Multi-resolution search
    if remaining() > 120:
        n_list = [320, 352, 384, 416, 448, 480, 512, 256, 576, 640]
        n_list = [nt for nt in n_list if 128 <= nt <= 768 and nt % 2 == 0]
        n_list.sort()
        for n_t in n_list:
            if remaining() < 30:
                break
            if n_t == best_n:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            h_t = lbfgs_smooth(h_t, n_t, 5000, 400)
            h_t, c5_t = top_shifts_focus(h_t, n_t, 5, 200)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t
            if remaining() > 30 and c5_t < best_c5 + 0.005:
                h_t_bin = binarize(h_t, n_t)
                h_tb, c5_tb = binary_sa(h_t_bin, n_t, rng, time_limit=min(40, int(remaining() * 0.3)))
                if c5_tb < best_c5:
                    best_c5 = c5_tb
                    best_h = h_tb.copy()
                    best_n = n_t

    # Phase 6: More diverse binary SA restarts
    if remaining() > 80:
        n_s = best_n
        starts = []
        starts.append(binarize(best_h, n_s))
        starts.append(thue_morse(n_s))
        # Random starts
        for _ in range(3):
            if remaining() < 20:
                break
            h_r = (rng.rand(n_s) < 0.5).astype(float)
            h_r = project_feasible(h_r, n_s)
            starts.append(h_r)
        # Perturbed best
        h_p = best_h.copy()
        mask = rng.rand(n_s) < 0.15
            h_p[mask] = 1.0 - h_p[mask]
        h_p = project_feasible(h_p, n_s)
        starts.append(binarize(h_p, n_s))

        for h_s in starts:
            if remaining() < 20:
                break
            h_s2, c5_s2 = binary_sa(h_s, n_s, rng, time_limit=min(60, int(remaining() * 0.2)))
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2.copy()
                best_n = n_s
                if remaining() > 15:
                    h_c = lbfgs_smooth(h_s2, n_s, 8000, 300)
                    c5_c = compute_c5(h_c)
                    if c5_c < best_c5:
                        best_c5 = c5_c
                        best_h = h_c.copy()

    # Phase 7: Final intensive refinement
    n_f = best_n
    h_f = best_h.copy()
    for beta in [10000, 30000, 80000]:
        if remaining() < 8:
            break
        h_f = lbfgs_smooth(h_f, n_f, beta, 500)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    for tk in [4, 6, 8, 12]:
        if remaining() < 10:
            break
        h_eq, c5_eq = top_shifts_focus(best_h, best_n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final binary search
    if remaining() > 30:
        h_bin = binarize(best_h, best_n)
        h_b, c5_b = binary_sa(h_bin, best_n, rng, time_limit=min(60, int(remaining() * 0.5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            if remaining() > 10:
                h_c = lbfgs_smooth(h_b, best_n, 15000, 400)
                c5_c = compute_c5(h_c)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
