def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    n = len(h)
    dx = 2.0 / n
    w = w * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:

            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]

        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

    return np.interp(x_new, xp, src_h)

        valid_fwd = (idx_fwd >= 0) & (idx_fwd < n)
        term2 = np.zeros(2 * n - 1)
        term2[valid_fwd] = (1.0 - h[idx_fwd[valid_fwd]])
        idx_bwd = b - s_vals
        valid_bwd = (idx_bwd >= 0) & (idx_bwd < n)
        term4 = np.zeros(2 * n - 1)
        term4[valid_bwd] = -h[idx_bwd[valid_bwd]]
        flip01[b] = term2 + term4
    
    return flip10, flip01


def efficient_1swap(h, n, max_rounds=30):
    """Efficient 1-swap using precomputed flip vectors."""
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if remaining() < 10:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        
        flip10, flip01 = precompute_flip_vectors(h, n)
        ones = list(flip10.keys())
        zeros = list(flip01.keys())
        
        if not ones or not zeros:
            break
        
        best_imp = 1e-15
        best_pair = None
        
        for a in ones:
            if remaining() < 10:
                break
            base = corr + flip10[a]
            # For each b, check max(base + flip01[b])
            for b in zeros:
                new_max = np.max(base + flip01[b])
                imp = c_max - new_max
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (a, b)
        
        if best_pair is None:
            break
        a, b = best_pair
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # Restore
            h[a] = 1.0
            h[b] = 0.0
            break
    
    return h, best_c5


def simulated_annealing(h_init, n, t_start, t_end, n_iter, rng):
    """Simulated annealing for binary sequences with 1-swap moves."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    
    for it in range(n_iter):
        if remaining() < 8:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        t = max(t, 0.0005)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-8)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[b] = 0.0
            h[a] = 1.0
    
    return best_h, best_c5


def multi_swap_sa(h_init, n, t_start, t_end, n_iter, rng, swap_size=2):
    """Simulated annealing with multi-swap moves."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    
    for it in range(n_iter):
        if remaining() < 8:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        t = max(t, 0.0005)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < swap_size or len(zeros) < swap_size:
            break
        
        # Pick random positions to swap
        a_idx = rng.choice(len(ones), size=swap_size, replace=False)
        b_idx = rng.choice(len(zeros), size=swap_size, replace=False)
        a_pos = ones[a_idx]
        b_pos = zeros[b_idx]
        
        for a in a_pos:
            h[a] = 0.0
        for b in b_pos:
            h[b] = 1.0
        
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-8)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            for b in b_pos:
                h[b] = 0.0
            for a in a_pos:
                h[a] = 1.0
    
    return best_h, best_c5


def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = float(np.max(corr))
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = float(np.max(corr))
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    n = len(h)
    dx = 2.0 / n
    w = w * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def slsqp_run(h, n, beta, maxiter=400):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def targeted_reduction(h, n, max_rounds=50):
    """Focus on reducing the specific shift(s) that give the maximum."""
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(99)
    
    for rnd in range(max_rounds):
        if remaining() < 10:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        # Find shifts near the maximum
        threshold = c_max - 1.5
        hot_shifts = np.where(corr >= threshold)[0]
        
        flip10, flip01 = precompute_flip_vectors(h, n)
        ones = list(flip10.keys())
        zeros = list(flip01.keys())
        if not ones or not zeros:
            break
        
        best_imp = 1e-15
        best_pair = None
        
        # Only consider swaps that reduce hot shifts
        for a in ones:
            if remaining() < 10:
                break
            base = corr + flip10[a]
            for b in zeros:
                new_corr = base + flip01[b]
                # Check if hot shifts are reduced
                max_hot = np.max(new_corr[hot_shifts]) if len(hot_shifts) > 0 else 0
                new_max = np.max(new_corr)
                # We want new_max < c_max, especially reducing hot shifts
                imp = c_max - new_max
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (a, b)
        
        if best_pair is None:
            break
        a, b = best_pair
        h[a] = 0.0
        h[b] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            h[a] = 1.0
            h[b] = 0.0
            break
    
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_n = parent_n
        best_c5 = compute_c5(best_h)
    except Exception:
        parent_n = 384
        best_n = parent_n
        best_h = np.zeros(parent_n)
        best_h[:parent_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Phase 1: Round to binary and efficient 1-swap
    h_bin = round_to_binary(best_h, best_n)
    h_ls, c5_ls = efficient_1swap(h_bin, best_n, max_rounds=20)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
        best_n = best_n

    # Phase 2: Targeted reduction on hot shifts
    if remaining() > 60:
        h_t, c5_t = targeted_reduction(h_ls, best_n, max_rounds=30)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 3: Simulated annealing (1-swap)
    if remaining() > 80:
        n_iter = min(4000, int(remaining() * 8))
        h_sa, c5_sa = simulated_annealing(h_t if c5_t < best_c5 else h_ls,
                                          best_n, 0.4, 0.005, n_iter, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Multi-swap SA (2-swap)
    if remaining() > 80:
        n_iter = min(2500, int(remaining() * 5))
        h_ms, c5_ms = multi_swap_sa(best_h, best_n, 0.6, 0.01, n_iter, rng, swap_size=2)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()
            best_n = best_n

    # Phase 5: Multi-resolution exploration
    parent_bin = round_to_binary(best_h, best_n)
    for n_t in [512, 640, 768, 896, 576]:
        if remaining() < 80:
            break
        if n_t % 2 != 0:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_interp = project_feasible(h_interp, n_t)
        h_bin_t = round_to_binary(h_interp, n_t)
        c5_t = compute_c5(h_bin_t)
        
        # 1-swap at this resolution
        if remaining() > 40:
            h_ls_t, c5_ls_t = efficient_1swap(h_bin_t, n_t, max_rounds=15)
            if c5_ls_t < c5_t:
                c5_t = c5_ls_t
                h_bin_t = h_ls_t.copy()
        
        # SA at this resolution
        if remaining() > 30 and c5_t < best_c5 + 0.01:
            n_iter_t = min(1500, int(remaining() * 4))
            h_sa_t, c5_sa_t = simulated_annealing(h_bin_t, n_t, 0.3, 0.005, n_iter_t, rng)
            if c5_sa_t < c5_t:
                c5_t = c5_sa_t
                h_bin_t = h_sa_t.copy()
        
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_bin_t.copy()
            best_n = n_t

    # Phase 6: Continuous SLSQP refinement
    n = best_n
    h = best_h.copy()
    for beta in [3000, 10000, 30000, 80000]:
        if remaining() < 15:
            break
        h_opt = slsqp_run(h, n, beta, maxiter=min(300, int(remaining() // 10)))
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
        if remaining() < 15:
            break

    # Phase 7: Re-round and refine
    if remaining() > 30:
        h_cr = round_to_binary(best_h, best_n)
        c5_cr = compute_c5(h_cr)
        h_r, c5_r = efficient_1swap(h_cr, best_n, max_rounds=20)
        if c5_r < c5_cr:
            c5_cr = c5_r
            h_cr = h_r
        if c5_cr < best_c5:
            best_c5 = c5_cr
            best_h = h_cr.copy()
            best_n = best_n

    # Phase 8: More SA from best binary
    if remaining() > 40:
        h_cur = round_to_binary(best_h, best_n) if (best_h > 0.5).sum() != best_n // 2 else best_h.copy()
        n_iter = min(3000, int(remaining() * 6))
        h_sa2, c5_sa2 = simulated_annealing(h_cur, best_n, 0.25, 0.003, n_iter, rng)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            best_n = best_n

    # Phase 9: Final SLSQP + targeted
    if remaining() > 20:
        h_fin = slsqp_run(best_h, best_n, 50000, maxiter=300)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 10: Final binary polish
    if remaining() > 15:
        h_fb = round_to_binary(best_h, best_n)
        c5_fb = compute_c5(h_fb)
        h_fp, c5_fp = efficient_1swap(h_fb, best_n, max_rounds=25)
        if c5_fp < c5_fb:
            c5_fb = c5_fp
            h_fb = h_fp
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()
            best_n = best_n

    # Phase 11: Final targeted reduction
    if remaining() > 15:
        h_td, c5_td = targeted_reduction(best_h, best_n, max_rounds=20)
        if c5_td < best_c5:
            best_c5 = c5_td
            best_h = h_td.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
