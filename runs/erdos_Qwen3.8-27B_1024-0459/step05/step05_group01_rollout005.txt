                    h[i2] = 0.0

                    h[j2] = 1.0
                else:
                    h[best_i] = 0.0
                    h[best_j] = 1.0
            else:
                h[best_i] = 0.0
                h[best_j] = 1.0
            best_corr += best_delta
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            improved = True
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                # Random kick
                for _ in range(rng.randint(3, 10)):
                    i = ones[rng.randint(n_ones)]
                    j = zeros[rng.randint(n_zeros)]
                    h[i] = 0.0
                    h[j] = 1.0
                best_corr = compute_max_corr(h)
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0

    return h, best_corr * dx


def binary_sa(h, n, time_limit=120, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    dx = 2.0 / n
    best_corr = compute_max_corr(h)
    best_h = h.copy()
    start = time.time()
    total_steps = max(500, int(time_limit * 3000))
    T_start = 2.0
    T_end = 0.01

    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()

    for step in range(total_steps):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        frac = step / total_steps
        T = T_start * (T_end / T_start) ** frac

        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones < 1 or n_zeros < 1:
            break

        # Decide move type
        r = rng.random()
        if r < 0.7:
            # Single swap
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            h[i] = 0.0
            h[j] = 1.0
        elif r < 0.9:
            # 2-swap
            if n_ones >= 2 and n_zeros >= 2:
                i1 = ones[rng.randint(n_ones)]
                i2 = ones[rng.randint(n_ones)]
                j1 = zeros[rng.randint(n_zeros)]
                j2 = zeros[rng.randint(n_zeros)]
                if i1 != i2 and j1 != j2:
                    h[i1] = 0.0
                    h[i2] = 0.0
                    h[j1] = 1.0
                    h[j2] = 1.0
                else:
                    i = ones[rng.randint(n_ones)]
                    j = zeros[rng.randint(n_zeros)]
                    h[i] = 0.0
                    h[j] = 1.0

        new_corr = compute_max_corr(h)
        delta = new_corr - best_corr
        if delta < 0:
            best_corr = new_corr
            best_h = h.copy()
        else:
            if delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-10)):
                pass  # Accept
            else:
                # Revert
                ones2 = np.where(h == 1.0)[0]
                zeros2 = np.where(h == 0.0)[0]
                # Just recompute from best and redo
                h = best_h.copy()
                ones = np.where(h == 1.0)[0].copy()
                zeros = np.where(h == 0.0)[0].copy()
                continue

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

    return best_h, best_corr * dx


def smooth_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def smooth_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(smooth_obj, h0, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < smooth_obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def projected_gradient(h, n, beta, max_iters=5000, lr=0.01):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        g = smooth_grad(h)
        v = 0.9 * v + lr * g
        h_new = h - v
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 1000 == 0:
            lr = lr * 0.8
    return best_h, best_c5


def top_k_equalize(h, n, top_k=5, max_iters=300):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= wk_val if False else w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.005, 0.01, 0.02, 0.05, 0.002, 0.08, 0.15]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load initial
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(384)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive binary search at parent resolution
    h_bin = (best_h > 0.5).astype(float)
    n_half = best_n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(best_n)
        h_bin[sort_idx[:n_half]] = 1.0

    # Multiple starting points for binary search
    starts = [h_bin.copy()]
    starts.append(theta_morse(best_n))
    starts.append(random_binary(best_n, rng))
    # Also try shifted and patterned
    h_pat = np.zeros(best_n)
    for i in range(best_n):
        h_pat[i] = 1.0 if (i % 2 == 0) else 0.0
    if int(np.sum(h_pat)) != n_half:
        h_pat[0] = 1.0 if n_half > int(np.sum(h_pat)) else 0.0
    starts.append(project_feasible(h_pat, best_n))

    t_start = time.time()
    for si, h_s in enumerate(starts):
        if remaining_time() < 30:
            break
        alloc = max(30, int(remaining_time() * (0.35 if si == 0 else 0.15)))
        h_b, c5_b = binary_sa(h_s, best_n, time_limit=alloc, rng=rng)
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_bin = h_b.copy()

        # Refine with hill climbing
        if remaining_time() > 15:
            h_b2, c5_b2 = binary_hill_climb(h_b, best_n, time_limit=min(60, int(remaining_time() * 0.4)), rng=rng)
            if c5_b2 < best_c5 - 1e-14:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                h_bin = h_b2.copy()

    # Phase 2: Continuous refinement of best binary
    h_cont = h_bin.copy()
    for beta in [100, 500, 2000, 8000, 30000, 100000, 300000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h_cont, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cont = h_new.copy()

    # Phase 3: Top-k equalization
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h_cont, best_n, top_k=6, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h_cont = h_eq.copy()

    # Phase 4: More SLSQP polish
    for beta in [50000, 200000, 500000, 1000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h_cont, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cont = h_new.copy()

    # Phase 5: Multi-resolution search
    n_trials = []
    for nn in [320, 352, 384, 416, 448, 480, 512, 544, 576, 608, 640, 768]:
        if nn != best_n and 160 <= nn <= 800 and nn % 2 == 0:
            n_trials.append(nn)
    n_trials.sort(key=lambda x: abs(x - best_n))

    for n_t in n_trials[:6]:
        if remaining_time() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)

        # Binary search at this resolution
        h_bin_t = (h_feas > 0.5).astype(float)
        nh = n_t // 2
        no = int(np.sum(h_bin_t))
        if no != nh:
            si2 = np.argsort(-h_bin_t)
            h_bin_t = np.zeros(n_t)
            h_bin_t[si2[:nh]] = 1.0

        alloc = max(20, int(remaining_time() * 0.25))
        h_bt, c5_bt = binary_sa(h_bin_t, n_t, time_limit=alloc, rng=rng)
        h_bt, c5_bt2 = binary_hill_climb(h_bt, n_t, time_limit=min(30, int(remaining_time() * 0.2)), rng=rng)
        c5_bt = min(c5_bt, c5_bt2)
        if c5_bt < best_c5 - 1e-14:
            best_c5 = c5_bt
            best_h = h_bt.copy()
            best_n = n_t

        # Continuous refinement
        if remaining_time() > 15:
            for beta in [1000, 10000, 50000, 200000]:
                if remaining_time() < 8:
                    break
                h_opt = slsqp_opt(h_bt, n_t, beta, maxiter=300)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5 - 1e-14:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_t
                    h_bt = h_opt.copy()

    # Phase 6: Further binary search from best continuous (rounded)
    if remaining_time() > 30:
        h_bin_final = (best_h > 0.5).astype(float)
        nh2 = best_n // 2
        no2 = int(np.sum(h_bin_final))
        if no2 != nh2:
            si3 = np.argsort(-h_bin_final)
            h_bin_final = np.zeros(best_n)
            h_bin_final[si3[:nh2]] = 1.0

        # SA from this point
        alloc = max(30, int(remaining_time() * 0.3))
        h_f, c5_f = binary_sa(h_bin_final, best_n, time_limit=alloc, rng=rng)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_f.copy()

        # Hill climb
        if remaining_time() > 15:
            h_f2, c5_f2 = binary_hill_climb(h_f, best_n, time_limit=min(45, int(remaining_time() * 0.3)), rng=rng)
            if c5_f2 < best_c5 - 1e-14:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    # Phase 7: Final continuous polish
    h_final = best_h.copy()
    for beta in [100000, 500000, 2000000]:
        if remaining_time() < 8:
            break
        h_new = slsqp_opt(h_final, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_final = h_new.copy()

    if remaining_time() > 8:
        h_eq2, c5_eq2 = top_k_equalize(h_final, best_n, top_k=8, max_iters=300)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final SLSQP
    if remaining_time() > 8:
        h_pol = slsqp_opt(best_h, best_n, 1000000, maxiter=500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5 - 1e-14:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
