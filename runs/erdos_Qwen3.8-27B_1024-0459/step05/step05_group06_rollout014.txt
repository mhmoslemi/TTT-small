```python
import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_OPT = 0
_BETA_OPT = 1000.0
_PENALTY_OPT = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_value_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for t in range(-(n-1), n):
        w = weights[n-1+t]
        if w < 1e-14:
            continue
        if t >= 0:
            npairs = n - t
            if npairs <= 0:
                continue
            if t > 0:
                grad[t:] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[t:t+npairs]
            else:
                grad[:npairs] += w * (1.0 - h[:npairs]) - w * h[:npairs]
        else:
            aoff = -t
            npairs = n - aoff
            if npairs <= 0:
                continue
            grad[:npairs] += w * (1.0 - h[aoff:aoff+npairs])
            grad[aoff:] -= w * h[:npairs]
    grad *= dx
    return val, grad

def _obj(x):
    val, _ = smooth_max_value_grad(x, _N_OPT, _BETA_OPT)
    penalty = _PENALTY_OPT * (np.sum(x) - _N_OPT / 2.0) ** 2
    return val + penalty

def _grad(x):
    val, g = smooth_max_value_grad(x, _N_OPT, _BETA_OPT)
    g = g + 2.0 * _PENALTY_OPT * (np.sum(x) - _N_OPT / 2.0) * np.ones(_N_OPT)
    return g

def lbfgsb_optimize(h, n, beta, penalty=1000.0, maxiter=3000):
    global _N_OPT, _BETA_OPT, _PENALTY_OPT
    _N_OPT = n
    _BETA_OPT = beta
    _PENALTY_OPT = penalty
    bounds = [(0.0, 1.0)] * n
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12, 'iprint': 0, 'maxfun': 5000})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    threshold = 0.5
    h_bin = (h >= threshold).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Need to flip some 1s to 0s
        ones_idx = np.where(h_bin == 1)[0]
        # Flip the ones with lowest original h value
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - target):
            h_bin[ones_idx[order[i]]] = 0
    elif n_ones < target:
        # Need to flip some 0s to 1s
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(target - n_ones):
            h_bin[zeros_idx[order[i]]] = 1
    return h_bin

def binary_local_search(h_bin, n, max_iters=50000):
    """Local search on binary solutions: swap a 1 and a 0."""
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best).astype(float)
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        improved = False
        # Try swapping random pairs
        for attempt in range(50):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            # Compute new corr
            # Flipping h[i]: 1->0, h[j]: 0->1
            # Effect on corr:
            # For each lag t, the change in corr[n-1+t]:
            # Originally: corr includes h[i]*(1-h[i+t]) and h[j]*(1-h[j+t]) etc.
            # After flip: h[i]=0, h[j]=1
            
            # Simpler: just compute new corr
            best[i] = 0
            best[j] = 1
            new_corr = compute_corr(best)
            new_c5 = np.max(new_corr) * (2.0/n)
            
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_corr = new_corr
                improved = True
                # Update ones/zeros
                ones = np.where(best == 1)[0]
                zeros = np.where(best == 0)[0]
                break
            else:
                best[i] = 1
                best[j] = 0
        if not improved:
            # Maybe try 3-opt or just break
            break
    return best, best_c5
```

I need a more efficient approach for the binary problem. Let me think about incremental updates.

When I flip two positions (i: 1→0, j: 0→1), the change in corr at lag t:
- corr[n-1+t] = Σ_i h[i+t]*(1-h[i])
- The change comes from terms involving positions i and j

For position p that was flipped (h[p] changed from old to new):
- In corr[n-1+t], h[p] appears:
  - As h[i+t] when i+t=p, i.e., i=p-t (if 0≤p-t<n): contribution changes by (new-old)*(1-h[p-t])
  - As (1-h[i]) when i=p (if 0≤p<n): contribution changes by h[p+t]*(-(new-old)) = (old-new)*h[p+t]

So Δcorr[n-1+t] = (new_h[p]-old_h[p]) * [(1-h[p-t]) * 1[0≤p-t<n] - h[p+t] * 1[0≤p+t<n]]

For a swap (i: 1→0, j: 0→1):
Δcorr[n-1+t] = (-1) * [(1-h[i-t])*1[0≤i-t<n] - h[i+t]*1[0≤i+t<n]] + (1) * [(1-h[j-t])*1[0≤j-t<n] - h[j+t]*1[0≤j+t<n]]

This is O(n) to compute for all t (just two vector operations). So each swap evaluation is O(n) = O(640) for n=640. Much better!

```python
        t_range_1 = np.arange(max(0, p-n+1), min(p, n-1) + 1)  # t values where 0≤p-t<n
        # Actually: 0≤p-t<n → p-n < t ≤ p, so t in [p-n+1, p] intersect [-(n-1), n-1]
        t1_lo = max(p-n+1, -(n-1))
        t1_hi = min(p, n-1)
        if t1_lo <= t1_hi:
            t_arr = np.arange(t1_lo, t1_hi+1)
            eff[n-1+t_arr] += delta_val * (1.0 - h_curr[p - t_arr])
        
        # -h[p+t] for t where 0≤p+t<n
        # 0≤p+t<n → -p ≤ t < n-p, so t in [-p, n-1-p] intersect [-(n-1), n-1]
        t2_lo = max(-p, -(n-1))
        t2_hi = min(n-1-p, n-1)
        if t2_lo <= t2_hi:
            t_arr2 = np.arange(t2_lo, t2_hi+1)
            eff[n-1+t_arr2] += delta_val * (-h_curr[p + t_arr2])
        
        return eff
    
    result = flip_effect(i, -1.0, h) + flip_effect(j, 1.0, h)
    return result
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_OPT = 0
_BETA_OPT = 1000.0
_PEN_OPT = 500.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:
            h[free] += diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_value_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for t in range(-(n-1), n):
        w = weights[n-1+t]
        if w < 1e-14:
            continue
        if t >= 0:
            npairs = n - t
            if npairs <= 0:
                continue
            if t > 0:
                grad[t:t+npairs] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[t:t+npairs]
            else:
                grad[:npairs] += w * (1.0 - 2.0 * h[:npairs])
        else:
            aoff = -t
            npairs = n - aoff
            if npairs <= 0:
                continue
            grad[:npairs] += w * (1.0 - h[aoff:aoff+npairs])
            grad[aoff:aoff+npairs] -= w * h[:npairs]
    grad *= dx
    return val, grad

def _obj_fn(x):
    val, _ = smooth_max_value_grad(x, _N_OPT, _BETA_OPT)
    s = np.sum(x) - _N_OPT / 2.0
    return val + _PEN_OPT * s * s

def _grad_fn(x):
    val, g = smooth_max_value_grad(x, _N_OPT, _BETA_OPT)
    s = np.sum(x) - _N_OPT / 2.0
    g = g + 2.0 * _PEN_OPT * s
    return g

def lbfgsb_optimize(h, n, beta, penalty=500.0, maxiter=2000, time_limit=120.0):
    global _N_OPT, _BETA_OPT, _PEN_OPT
    _N_OPT = n
    _BETA_OPT = beta
    _PEN_OPT = penalty
    bounds = [(0.0, 1.0)] * n
    c5_before = compute_c5(h)
    t_start = time.time()
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': 1e-14, 'iprint': 0,
                               'maxfun': 8000, 'factr': 1e10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def flip_effect_single(h, n, p, delta_val):
    eff = np.zeros(2 * n - 1)
    t1_lo = max(p - n + 1, -(n - 1))
    t1_hi = min(p, n - 1)
    if t1_lo <= t1_hi:
        t_arr = np.arange(t1_lo, t1_hi + 1)
        eff[n - 1 + t_arr] += delta_val * (1.0 - h[p - t_arr])
    t2_lo = max(-p, -(n - 1))
    t2_hi = min(n - 1 - p, n - 1)
    if t2_lo <= t2_hi:
        t_arr2 = np.arange(t2_lo, t2_hi + 1)
        eff[n - 1 + t_arr2] += delta_val * (-h[p + t_arr2])
    return eff

def binarize_h(h, n):
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - target):
            h_bin[ones_idx[order[i]]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(target - n_ones):
            h_bin[zeros_idx[order[i]]] = 1.0
    return h_bin

def binary_hill_climb(h_bin, n, max_swaps=200000, rng_seed=42):
    rng = np.random.RandomState(rng_seed)
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    corr = compute_corr(best).astype(float)
    max_corr = np.max(corr)
    ones_list = np.where(best == 1)[0]
    zeros_list = np.where(best == 0)[0]
    no_improve = 0
    for it in range(max_swaps):
        if remaining_time() < 10:
            break
        if no_improve > 5000:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:
            continue
        delta = flip_effect_single(best, n, i, -1.0) + flip_effect_single(best, n, j, 1.0)
        new_corr = corr + delta
        new_max = np.max(new_corr)
        if new_max < max_corr - 1e-12:
            best[i] = 0.0
            best[j] = 1.0
            corr = new_corr
            max_corr = new_max
            best_c5 = max_corr * (2.0 / n)
            no_improve = 0
            mask_i = ones_list == i
            if np.sum(mask_i) == 1:
                idx_i = np.where(mask_i)[0][0]
                ones_list = np.delete(ones_list, idx_i)
            mask_j = zeros_list == j
            if np.sum(mask_j) == 1:
                idx_j = np.where(mask_j)[0][0]
                zeros_list = np.delete(zeros_list, idx_j)
            ones_list = np.append(ones_list, j)
            zeros_list = np.append(zeros_list, i)
            ones_list = np.sort(ones_list)
            zeros_list = np.sort(zeros_list)
        else:
            no_improve += 1
    return best, best_c5

def coord_descent_fast(h, n, n_sweeps=30, time_limit=120.0):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best).astype(float)
    t_start = time.time()
    for sweep in range(n_sweeps):
        if remaining_time() < (time_limit - (time.time() - t_start)) + 5 or remaining_time() < 5:
            break
        improved_any = False
        max_corr = np.max(best_corr)
        for i in range(n):
            if remaining_time() < 5:
                break
            t0 = best[i]
            if t0 <= 1e-12 or t0 >= 1.0 - 1e-12:
                cands = [0.0, 1.0] if t0 <= 0.5 else [0.0, 1.0]
            else:
                cands = [0.0, 1.0, max(0.0, t0 - 0.3), min(1.0, t0 + 0.3),
                         max(0.0, t0 - 0.1), min(1.0, t0 + 0.1), 1.0 - t0]
            delta_pos = flip_effect_single(best, n, i, -t0)
            base_corr = best_corr - delta_pos
            best_t = t0
            best_f = max_corr
            for t in cands:
                if abs(t - t0) < 1e-12:
                    continue
                new_corr = base_corr + t * (flip_effect_single(best, n, i, 1.0))
                f = np.max(new_corr)
                if f < best_f - 1e-13:
                    best_f = f
                    best_t = t
            if best_t != t0:
                lo = max(0.0, best_t - 0.2)
                hi = min(1.0, best_t + 0.2)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base_corr + c1 * flip_effect_single(best, n, i, 1.0))
                f2 = np.max(base_corr + c2 * flip_effect_single(best, n, i, 1.0))
                for _ in range(40):
                    if hi - lo < 1e-12:
                        break
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base_corr + c1 * flip_effect_single(best, n, i, 1.0))
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base_corr + c2 * flip_effect_single(best, n, i, 1.0))
                t_opt = (lo + hi) / 2.0
                f_opt = np.max(base_corr + t_opt * flip_effect_single(best, n, i, 1.0))
                if f_opt < best_f:
                    best_f = f_opt
                    best_t = t_opt
            if best_f < max_corr - 1e-12:
                best[i] = best_t
                best_corr = compute_corr(best)
                max_corr = np.max(best_corr)
                best_c5 = max_corr * (2.0 / n)
                improved_any = True
        if not improved_any:
            break
    best = project_feasible(best, n)
    best_c5 = compute_c5(best)
    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def slsqp_polish(h, n, beta, maxiter=300):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n

    def _con_fun(x):
        return np.sum(x) - n / 2.0

    def _con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(lambda x: smooth_max_value_grad(x, n, beta)[0],
                      h, jac=lambda x: smooth_max_value_grad(x, n, beta)[1],
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()
```
