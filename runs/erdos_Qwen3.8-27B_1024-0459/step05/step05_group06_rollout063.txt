            top_z = np.argpartition(z_score, min(15, nz))[:min(15, nz)]
            top_o = np.argpartition(-o_score, min(15, no))[:min(15, no)]
            
            # Generate candidates
            cand_i = zeros[top_z]
            cand_j = ones[top_o]
            
            # Also try: for the worst-contributing positions
            # Positions i where h[i]=1 and h[i+d]=0 (contribute to A(d))
            offset_d = hd - n_half
            if offset_d > 0:
                contrib_pairs = [(i, i + offset_d) for i in range(n - offset_d) if h[i] == 1 and h[i + offset_d] == 0]
                if len(contrib_pairs) > 2:
                    # Try swapping first of a contributing pair with a random 0 elsewhere
                    import random
                    rng_local = np.random.RandomState(it)
                    extra_i = cand_i.copy()
                    extra_j = cand_j.copy()
                    for _ in range(5):
                        if len(contrib_pairs) > 0:
                            pi, pj = contrib_pairs[rng_local.randint(len(contrib_pairs))]
                            # Flip pi: 1->0, need a 0 to flip to 1
                            z_pool = zeros[np.isin(zeros, [pi], invert=True)]
                            if len(z_pool) > 0:
                                z_idx = rng_local.randint(len(z_pool))
                                extra_i = np.append(extra_i, z_pool[z_idx])
                                extra_j = np.append(extra_j, pi)
            
            cand_i = np.unique(cand_i)
            cand_j = np.unique(cand_j)
            
            nc = min(len(cand_i), 30)
            no2 = min(len(cand_j), 30)
            cand_i = cand_i[:nc]
            cand_j = cand_j[:no2]
            
            if len(cand_i) == 0 or len(cand_j) == 0:
                continue
            
            # Evaluate all candidate swaps
            ci = np.array(cand_i)
            cj = np.array(cand_j)
            # new_corr = corr + effects[ci] - effects[cj]
            # Shape: (nc, no2, ncorr) - too big. Do it in batches.
            batch = 20
            for bi in range(0, len(ci), batch):
                ci_b = ci[bi:bi+batch]
                ni_b = ci_b[:, None, :]  # (lb, 1, ncorr)
                oj_b = cj[None, :, :]    # (1, nb, ncorr)
                new_corrs = corr[None, None, :] + effects[ci_b][:, None, :] - effects[cj][None, :, :]
                maxes = np.max(new_corrs, axis=2)  # (lb, nb)
                for ii in range(maxes.shape[0]):
                    for jj in range(maxes.shape[1]):
                        if maxes[ii, jj] < best_new_max - 1e-14:
                            best_new_max = maxes[ii, jj]
                            best_swap = (ci_b[ii], cj[jj])
                            best_delta = max_corr - maxes[ii, jj]
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 1.0
            h[j] = 0.0
            corr = compute_corr(h)
            effects = compute_flip_effects(h, n)
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            new_c5 = np.max(corr) * (2.0 / n)
            if new_c5 < best_c5:
                best_c5 = new_c5
                stagnation = 0
            else:
                stagnation += 1
                # Revert
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                effects = compute_flip_effects(h, n)
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
        else:
            stagnation += 1
        
        it += 1
        if stagnation > max_stagnation:
            break
    
    return h, compute_c5(h)

def local_search_binary_fast(h, n, max_iter=3000, time_limit=300.0):
    """Faster local search focusing on reducing specific offsets."""
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = np.max(corr) * (2.0 / n)
    ncorr = 2 * n - 1
    n_half = n - 1
    
    it = 0
    stagnation = 0
    max_stagnation = 80
    
    while it < max_iter and remaining_time() > 5:
        max_corr = np.max(corr)
        
        # Get top hot offsets
        top_k = [np.argmax(corr)]
        corr_copy = corr.copy()
        for _ in range(4):
            k = np.argmax(corr_copy)
            if corr[k] < max_corr - 1.0:
                break
            if k not in top_k:
                top_k.append(k)
            corr_copy[k] = -1e10
        
        best_swap = None
        best_new_max = max_corr + 0.5
        
        for hd in top_k:
            d = hd - n_half
            if d == 0:
                continue
            
            # Find positions that contribute to this offset
            if d > 0:
                # A(d) = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
                # Contributors: h[i]=1, h[i+d]=0
                contrib_1 = [i for i in range(n - d) if h[i] == 1 and h[i + d] == 0]
                # We want to reduce these: flip i (1->0) or flip i+d (0->1)
                # Option A: flip i from 1->0, need to flip some j from 0->1 (j != i)
                # Option B: flip i+d from 0->1, need to flip some j from 1->0 (j != i+d)
                
                candidates = []
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                
                # Option A: for each contributing i, swap with a 0-position far from hot spots
                for i in contrib_1[:10]:
                    # Try swapping h[i]=1 with h[j]=0 for j in zeros
                    # Effect on corr[d]: reduces by 1 (the contributing term)
                    # But might increase other offsets
                    # Just try a few random zeros
                    for j in zeros[:5]:
                        if j == i or (d > 0 and j == i + d) or (-d > 0 and j == i - d):
                            continue
                        candidates.append((j, i))  # j: 0->1, i: 1->0
                    # Also try structured: find j that minimizes increase elsewhere
                    if len(zeros) > 0:
                        mid_j = zeros[len(zeros)//2]
                        if mid_j != i and abs(mid_j - i) > d:
                            candidates.append((mid_j, i))
                
                # Option B: for each contributing pair, flip i+d (0->1)
                for i in contrib_1[:10]:
                    j2 = i + d
                    if h[j2] == 0:
                        for j in ones[:5]:
                            if j == j2 or j == i:
                                continue
                            candidates.append((j2, j))
                        if len(ones) > 0:
                            mid_j = ones[len(ones)//2]
                            if mid_j != j2 and mid_j != i:
                                candidates.append((j2, mid_j))
                
            else:
                # Negative offset: d < 0, let dd = -d
                dd = -d
                # A(-dd) = sum_{i=0}^{n-1-dd} h[i+dd]*(1-h[i])
                # Contributors: h[i+dd]=1, h[i]=0
                contrib_1 = [i + dd for i in range(n - dd) if h[i + dd] == 1 and h[i] == 0]
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
                
                candidates = []
                for j in contrib_1[:10]:
                    # j is 1, flip to 0; need i (0->1) where h[i]=0 and i+dd was the pair
                    for zi in zeros[:5]:
                        if zi == j or zi == j - dd:
                            continue
                        candidates.append((zi, j))
                    if len(zeros) > 0:
                        mid_z = zeros[len(zeros)//2]
                        if mid_z != j and abs(mid_z - j) > dd:
                            candidates.append((mid_z, j))
                    
                    i2 = j - dd
                    if h[i2] == 0:
                        for oj in ones[:5]:
                            if oj == i2 or oj == j:
                                continue
                            candidates.append((i2, oj))
                        if len(ones) > 0:
                            mid_o = ones[len(ones)//2]
                            if mid_o != i2 and mid_o != j:
                                candidates.append((i2, mid_o))
            
            # Evaluate candidates
            if not candidates:
                continue
            # Remove duplicates
            candidates = list(set(candidates))[:200]
            
            for (zi, oj) in candidates:
                if h[zi] != 0 or h[oj] != 1:
                    continue
                # Compute new corr quickly: just update the changed entries
                # h[zi]: 0->1, h[oj]: 1->0
                new_corr = corr.copy()
                # Effect of h[zi] 0->1:
                for dd2 in range(1, n):
                    # Positive offset dd2
                    if zi + dd2 < n:
                        new_corr[n_half + dd2] += (1.0 - h[zi + dd2])
                    if zi - dd2 >= 0:
                        new_corr[n_half + dd2] -= h[zi - dd2]
                    # Negative offset
                    if zi - dd2 >= 0:
                        new_corr[n_half - dd2] += (1.0 - h[zi - dd2])
                    if zi + dd2 < n:
                        new_corr[n_half - dd2] -= h[zi + dd2]
                # Effect of h[oj] 1->0:
                for dd2 in range(1, n):
                    if oj + dd2 < n:
                        new_corr[n_half + dd2] -= (1.0 - h[oj + dd2])
                    if oj - dd2 >= 0:
                        new_corr[n_half + dd2] += h[oj - dd2]
                    if oj - dd2 >= 0:
                        new_corr[n_half - dd2] -= (1.0 - h[oj - dd2])
                    if oj + dd2 < n:
                        new_corr[n_half - dd2] += h[oj + dd2]
                
                new_max = np.max(new_corr)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (zi, oj, new_corr)
        
        if best_swap is not None:
            zi, oj, new_corr = best_swap
            h[zi] = 1.0
            h[oj] = 0.0
            corr = new_corr
            new_c5 = np.max(corr) * (2.0 / n)
            if new_c5 < best_c5:
                best_c5 = new_c5
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1
        
        it += 1
        if stagnation > max_stagnation:
            break
    
    return h, compute_c5(h)

        max_offsets = np.where(corr == max_corr)[0]
        
        improved = False

            if d == 0:
                continue
            
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break

            best_max = max_corr
            
            # Try all pairs involving the hot offset
            # For offset d > 0: want to reduce corr[n_half + d]
            # Flip 0->1 at i reduces corr[n_half+d] if h[i+d]=1 (term was 0*(1-1)=0, becomes 1*(1-1)=0... no change)
            # Actually: flipping i from 0 to 1 at positive offset d:
            #   if i < n-d: corr[n_half+d] += (1-h[i+d])
            #   if i >= d: corr[n_half+d] -= h[i-d]

            #   if j < n-d: corr[n_half+d] -= (1-h[j+d])
