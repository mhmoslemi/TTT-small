Anyway, let me focus on the practical optimization. The key is to find a good binary pattern and then refine it continuously.

Let me write a more focused program:

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TOPK = 5
_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

# ============ L-BFGS-B with log-sum-exp smoothing ============

def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty

def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g

def lbfgs_pass(h, n, beta, top_k=5, maxiter=600):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 8})
        return proj(res.x, n)
    except Exception:
        return h

# ============ Binary local search with diverse moves ============

def binary_local_search_v2(h_init, n, max_iters=10000, top_k=4):
    """Enhanced binary local search with multiple move types."""
    h_bin = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h_bin == 1)[0]
            to_remove = set(ones[np.argsort(np.random.RandomState(0).rand(len(ones)))[:n_ones - n_half]])
            for i in to_remove:
                h_bin[i] = 0
        else:
            zeros = np.where(h_bin == 0)[0]
            to_add = set(zeros[np.argsort(np.random.RandomState(0).rand(len(zeros)))[:n_half - n_ones]])
            for i in to_add:
                h_bin[i] = 1
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5
    rng = np.random.RandomState(42)
    
    # Compute top-k gradients for scoring
    for it in range(max_iters):
        if remaining() < 3:
            break
        
        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]
        
        # Gradient for scoring swaps
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)
        
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        best_delta = -1e-10
        best_move = None
        
        # Type 1: Single swap (1→0 at i, 0→1 at j)
        # Score: want g[i] very negative, g[j] very positive
        n_try = min(15, len(ones))
        n_try_z = min(15, len(zeros))
        top_ones = ones[np.argsort(-g[ones])[:n_try]]  # most negative g
        top_zeros = zeros[np.argsort(-g[zeros])[:n_try_z]]  # most positive g
        
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                nc = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = ('swap', i, j)
        
        # Type 2: 2-swap (two 1s→0, two 0s→1)
        if remaining() > 10:
            n2 = min(6, len(ones))
            n2z = min(6, len(zeros))
            top_ones2 = ones[np.argsort(-g[ones])[:n2]]
            top_zeros2 = zeros[np.argsort(-g[zeros])[:n2z]]
            for i1 in top_ones2:
                for i2 in top_ones2:
                    if i1 >= i2:
                        continue
                    for j1 in top_zeros2:
                        for j2 in top_zeros2:
                            if j1 >= j2:
                                continue
                            if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                                continue
                            h_bin[i1] = 0.0
                            h_bin[i2] = 0.0
                            h_bin[j1] = 1.0
                            h_bin[j2] = 1.0
                            nc = compute_c5(h_bin)
                            h_bin[i1] = 1.0
                            h_bin[i2] = 1.0
                            h_bin[j1] = 0.0
                            h_bin[j2] = 0.0
                            delta = nc - cur_c5
                            if delta < best_delta:
                                best_delta = delta
                                best_move = ('2swap', i1, i2, j1, j2)
        
        # Type 3: Random swap for exploration
        if best_move is None:
            for _ in range(4):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                nc = compute_c5(h_bin)
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = ('swap', i, j)
        
        if best_move is None:
            break
        
        # Apply move
        if best_move[0] == 'swap':
            _, i, j = best_move
            h_bin[i] = 0.0
            h_bin[j] = 1.0
        elif best_move[0] == '2swap':
            _, i1, i2, j1, j2 = best_move
            h_bin[i1] = 0.0
            h_bin[i2] = 0.0
            h_bin[j1] = 1.0
            h_bin[j2] = 1.0
        
        cur_c5 = compute_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()
    
    return best_h, best_c5

# ============ Top-correlation equalization ============

def top_equalize_v2(h, n, top_k=5, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_vals = corr[idx]
        c_max = c_vals[0]
        c_min = c_vals[-1]
        if c_max - c_min < 1e-10:
            break
        
        # Direction to equalize: reduce max, increase min
        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            break
        d /= dn
        
        improved = False
        for alpha in [0.05, 0.03, 0.01, 0.08, 0.02, 0.005, 0.1, 0.015]:
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        
        if not improved:
            # Try negative direction
            for alpha in [0.03, 0.01, 0.05, 0.02]:
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        
        if not improved:
            # Try pairs of top correlations
            if top_k >= 2:
                for a_idx in range(min(3, top_k)):
                    for b_idx in range(a_idx + 1, min(3, top_k)):
                        g_a = corr_grad_single(h, idx[a_idx], n)
                        g_b = corr_grad_single(h, idx[b_idx], n)
                        d2 = 0.5 * (g_a + g_b)
                        d2 -= np.mean(d2)
                        dn2 = np.linalg.norm(d2)
                        if dn2 < 1e-13:
                            continue
                        d2 /= dn2
                        for alpha in [0.03, 0.01, 0.05]:
                            h_try = proj(h - alpha * d2, n)
                            c5_try = compute_c5(h_try)
                            if c5_try < best_c5 - 1e-14:
                                h = h_try
                                best_c5 = c5_try
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
        
        if not improved:
            break
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
