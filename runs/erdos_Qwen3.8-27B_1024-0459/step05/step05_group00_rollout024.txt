def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):

        diff = target - np.sum(h)
        if abs(diff) < 1e-15:

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n

def smooth_max_obj_grad(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -80.0, 80.0)
    ex = np.exp(shifted)
    s = np.sum(ex)
    obj = (m + np.log(s) / beta) * (2.0 / n)
    w = ex / s
    g = np.zeros(n)
    active = w > 1e-10

            np_ = n - ao

        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
