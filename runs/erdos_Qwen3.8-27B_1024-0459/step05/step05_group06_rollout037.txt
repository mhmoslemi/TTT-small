    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            t0 = best[i]
            s_all = np.arange(-(n - 1), n)
            ip = i + s_all
            im = i - s_all
            valid_p = (ip >= 0) & (ip < n)
            valid_m = (im >= 0) & (im < n)

            a = np.zeros(ncorr)
            if np.any(valid_p):
                a[valid_p] += (1.0 - best[ip[valid_p]])
            if np.any(valid_m):
                a[valid_m] += (-best[im[valid_m]])

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            fc1 = coord_de_objective(best_corr, a, c1, t0, n)
            fc2 = coord_de_objective(best_corr, a, c2, t0, n)
            for _ in range(40):
                if hi - lo < 1e-14:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = coord_de_objective(best_corr, a, c1, t0, n)
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    fc2 = coord_de_objective(best_corr, a, c2, t0, n)
            t_opt = (lo + hi) / 2.0
            f_opt = coord_de_objective(best_corr, a, t_opt, t0, n)

            if f_opt * dx < best_c5 - 1e-16:
                best[i] = t_opt
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True

        if not improved:
            break

    return best, best_c5

def sa_block_move(current, n, rng, T, ncorr_hint):
    block_sizes = [3, 5, 8, 12, 20]
    B = block_sizes[rng.randint(len(block_sizes))]
    start = rng.randint(0, max(1, n - B + 1))
    noise_scale = max(T * 3.0, 0.02)
    trial = current.copy()
    trial[start:start + B] += rng.normal(0, noise_scale, B)
    trial = project_feasible(trial, n)
    return trial

def simulated_annealing(h, n, deadline, rng, time_fraction=0.3):
    best = h.copy()
    best_c5 = compute_c5(best)
    current = h.copy()
    current_c5 = best_c5

    T0 = 0.015
    T_min = 0.00005
    alpha = 0.9997
    T = T0
    sa_start = time.time()
    sa_duration = (deadline - sa_start) * time_fraction

    no_improve = 0
    while time.time() - sa_start < sa_duration:
        trial = sa_block_move(current, n, rng, T)
        trial_c5 = compute_c5(trial)
        delta = trial_c5 - current_c5

        if delta < 0 or rng.uniform() < np.exp(-delta / max(T, 1e-12)):
            current = trial
            current_c5 = trial_c5
            no_improve = 0
            if current_c5 < best_c5:
                best = current.copy()
                best_c5 = current_c5
        else:
            no_improve += 1

        T = max(T * alpha, T_min)
        if no_improve > 8000:
            T = T0 * 0.4
            no_improve = 0

    return best, best_c5

def iterated_local_search(h, n, n_runs=15, rng=None, deadline=None):
    best = h.copy()
    best_c5 = compute_c5(best)

    for run_i in range(n_runs):
        if deadline is not None and time.time() > deadline:
            break

        pert = best.copy()
        n_blocks = rng.randint(1, 4)
        for _ in range(n_blocks):
            B = rng.randint(5, 35)
            start = rng.randint(0, max(1, n - B + 1))
            pert[start:start + B] = 1.0 - pert[start:start + B]
        noise_scale = 0.03 + 0.04 * rng.random()
        pert += rng.normal(0, noise_scale, n)
        pert = project_feasible(pert, n)

        opt_h, opt_c5 = coord_descent(pert, n, n_sweeps=25, rng=rng, deadline=deadline)
        if opt_c5 < best_c5:
            best = opt_h.copy()
            best_c5 = opt_c5

    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def active_set_polish(h, n, deadline, rng):
    """Focus optimization on reducing the active (maximizing) shifts."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n

    for iteration in range(30):
        if deadline is not None and time.time() > deadline:
            break

        max_val = np.max(best_corr)
        threshold = max_val - 0.3
        active = np.where(best_corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(best_corr)])

        # Do coordinate descent but only check against active set for speed
        order = rng.permutation(n)
        improved = False
        for i in order:
            t0 = best[i]
            s_all = np.arange(-(n - 1), n)
            ip = i + s_all
            im = i - s_all
            valid_p = (ip >= 0) & (ip < n)
            valid_m = (im >= 0) & (im < n)

            a = np.zeros(len(s_all))
            if np.any(valid_p):
                a[valid_p] += (1.0 - best[ip[valid_p]])
            if np.any(valid_m):
                a[valid_m] += (-best[im[valid_m]])

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)

            def eval_t(t):
                vals = best_corr + (t - t0) * a
                vals[n - 1] = best_corr[n - 1] + (t - t0) - (t * t - t0 * t0)
                return np.max(vals)

            fc1 = eval_t(c1)
            fc2 = eval_t(c2)
            for _ in range(30):
                if hi - lo < 1e-13:
                    break
                if fc1 < fc2:
                    hi = c2
                    c2 = c1
                    fc2 = fc1
                    c1 = hi - gr * (hi - lo)
                    fc1 = eval_t(c1)
                else:
                    lo = c1
                    c1 = c2
                    fc1 = fc2
                    c2 = lo + gr * (hi - lo)
                    fc2 = eval_t(c2)
            t_opt = (lo + hi) / 2.0
            f_opt = eval_t(t_opt)

            if f_opt * dx < best_c5 - 1e-16:
                best[i] = t_opt
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True

        if not improved:
            break

    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15
    rng = np.random.RandomState(seed)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = project_feasible(h0, n0)
    except Exception:
        n0 = 640
        h0 = np.zeros(n0)
        idx = rng.choice(n0, n0 // 2, replace=False)
        h0[idx] = 1.0
        h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Phase 1: Initial coordinate descent
    h1, c5_1 = coord_descent(best_h, best_n, n_sweeps=40, rng=rng, deadline=deadline)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Iterated local search (main global exploration)
    if time.time() < deadline - 120:
        h2, c5_2 = iterated_local_search(best_h, best_n, n_runs=20, rng=rng,
                                         deadline=min(deadline - 120, start_time + 500))
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        h2b, c5_2b = coord_descent(h2, best_n, n_sweeps=30, rng=rng, deadline=deadline)
        if c5_2b < best_c5:
            best_c5 = c5_2b
            best_h = h2b.copy()

    # Phase 3: Simulated annealing
    if time.time() < deadline - 100:
        h3, c5_3 = simulated_annealing(best_h, best_n, deadline=deadline, rng=rng, time_fraction=0.35)
        h3 = project_feasible(h3, best_n)
        c5_3 = compute_c5(h3)
        h3b, c5_3b = coord_descent(h3, best_n, n_sweeps=35, rng=rng, deadline=deadline)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3b.copy()

    # Phase 4: More ILS with different perturbation strengths
    if time.time() < deadline - 80:
        h4 = best_h.copy()
        for il in range(10):
            if time.time() > deadline - 80:
                break
            pert = h4.copy()
            # Larger, more structured perturbation
            n_seg = rng.randint(2, 5)
            seg_len = n_best_n = best_n // n_seg
            for s in range(n_seg):
                st = s * seg_len
                en = min(st + seg_len, best_n)
                if rng.random() < 0.5:
                    pert[st:en] = 1.0 - pert[st:en]
                else:
                    pert[st:en] += rng.normal(0, 0.1, en - st)
            pert = project_feasible(pert, best_n)
            opt_h, opt_c5 = coord_descent(pert, best_n, n_sweeps=20, rng=rng, deadline=deadline)
            if opt_c5 < best_c5:
                best_c5 = opt_c5
                best_h = opt_h.copy()
                h4 = opt_h.copy()

    # Phase 5: Try higher resolution
    for mult in [1.5, 2.0]:
        if time.time() > deadline - 120:
            break
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if rn > 1600 or rn < best_n:
            continue
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        h_r, c5_r = coord_descent(h_r, rn, n_sweeps=25, rng=rng, deadline=deadline)
        if time.time() < deadline - 40:
            # Small SA at higher res
            h_r2, c5_r2 = simulated_annealing(h_r, rn, deadline=deadline, rng=rng, time_fraction=0.15)
            c5_r2 = compute_c5(h_r2)
            h_r2, c5_r2 = coord_descent(h_r2, rn, n_sweeps=20, rng=rng, deadline=deadline)
            if c5_r2 < c5_r:
                h_r = h_r2
                c5_r = c5_r2

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 6: Final intensive coordinate descent
    if time.time() < deadline - 10:
        h_fin, c5_fin = coord_descent(best_h, best_n, n_sweeps=60, rng=rng, deadline=deadline)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 7: Very fine polish - just a few more sweeps
    if time.time() < deadline - 3:
        h_fin2, c5_fin2 = coord_descent(best_h, best_n, n_sweeps=20, rng=rng, deadline=deadline)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
