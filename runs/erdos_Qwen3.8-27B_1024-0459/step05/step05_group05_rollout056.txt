def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad(h, k_off, n):
    g = np.zeros(n)
    if k_off >= 0:
        p = n - k_off
        g[k_off:k_off + p] += (1.0 - h[:p])
        g[:p] -= h[k_off:k_off + p]
    else:
        a = -k_off
        p = n - a
        g[:p] += (1.0 - h[a:a + p])
        g[a:a + p] -= h[:p]
    return g


def slsqp_smooth(h, n, beta, maxiter=500):
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h, n)

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        s = corr - m
        c = np.clip(beta * s, -50.0, 50.0)
        e = np.exp(c)
        return (m + np.log(np.sum(e)) / beta) * dx

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        s = corr - m
        c = np.clip(beta * s, -50.0, 50.0)
        e = np.exp(c)
        w = (e / np.sum(e)) * dx
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                p = n - off
                g[off:off + p] += wk * (1.0 - x[:p])
                g[:p] -= wk * x[off:off + p]
            else:
                a = -off
                p = n - a
                g[:p] += wk * (1.0 - x[a:a + p])
                g[a:a + p] -= wk * x[:p]
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def thue_morse_vec(n, rng):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            rem = ones_idx[perm[:n_ones - n_half]]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            add = zeros_idx[perm[:n_half - n_ones]]
            h[add] = 1
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

        g /= len(top_idx)
        g -= np.mean(g)

        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            step = min(step * 1.1, 0.3)
        else:
            step *= 0.85
            if step < 0.001:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse_vec(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP progressive at parent resolution
    for beta in [100, 500, 2000, 8000, 20000, 50000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: L-BFGS-B with augmented Lagrangian
    if remaining() > 15:
        h_lb, c5_lb = lbfgs_auglag(h, n, maxiter=500)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()
            h = h_lb.copy()

    # Phase 3: Equalization
    for tk in [3, 5, 8]:
        if remaining() < 10:
            break
        h_eq, c5_eq = equalize(h, n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Systematic binary local search
    if remaining() > 30:
        h_bin = binarize(h)
        h_b, c5_b = systematic_binary(h_bin, n, rng, time_limit=min(100, int(remaining() * 0.25)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
            h = h_b.copy()
            # Refine continuously from binary
            h_cont = slsqp_smooth(h_b, n, 10000, 400)
            c5_cont = compute_c5(h_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()
                h = h_cont.copy()
            h_eq2, c5_eq2 = equalize(h, n, top_k=5, max_iters=300)
            if c5_eq2 < best_c5:
                best_c5 = c5_eq2
                best_h = h_eq2.copy()
                h = h_eq2.copy()
        else:
            h = best_h.copy()

    # Phase 5: Subgradient descent
    if remaining() > 15:
        h_sg, c5_sg = subgradient_descent(h, n, n_iter=200)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 6: Diverse starts
    if remaining() > 80:
        # Thue-Morse
        h_tm = thue_morse_vec(n, rng)
        h_tm = slsqp_smooth(h_tm, n, 3000, 400)
        h_tm, c5_tm = equalize(h_tm, n, top_k=5, max_iters=300)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            best_n = n

        # Perturbed parent
        for pert_scale in [0.05, 0.1, 0.2]:
            if remaining() < 20:
                break
            h_p = best_h + rng.randn(n) * pert_scale
            h_p = project_feasible(h_p, n)
            h_p = slsqp_smooth(h_p, n, 5000, 400)
            h_p, c5_p = equalize(h_p, n, top_k=5, max_iters=200)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
                best_n = n

        # Fourier-based init: sum of a few cosines clipped
        if remaining() > 40:
            for trial in range(3):
                h_f = np.zeros(n)
                for freq in range(1, 6):
                    amp = rng.uniform(0.1, 0.4)
                    phase = rng.uniform(0, 2 * np.pi)
                    h_f += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
                h_f = 0.5 + h_f
                h_f = np.clip(h_f, 0, 1)
                h_f = project_feasible(h_f, n)
                h_f = slsqp_smooth(h_f, n, 3000, 300)
                h_f, c5_f = equalize(h_f, n, top_k=5, max_iters=200)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_f.copy()
                    best_n = n

    # Phase 7: Resolution sweep
    if remaining() > 60:
        n_candidates = []
        for nt in [256, 320, 384, 448, 512, 576, 640, 768]:
            if nt != best_n and 128 <= nt <= 768 and nt % 2 == 0:
                n_candidates.append(nt)
        n_candidates.sort(key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:6]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)

            h_t = slsqp_smooth(h_t, n_t, 3000, 300)
            h_t = slsqp_smooth(h_t, n_t, 10000, 300)
            h_t, c5_t = equalize(h_t, n_t, top_k=5, max_iters=200)

            if remaining() > 15 and c5_t < best_c5 + 0.002:
                h_tb = binarize(h_t)
                h_tb, c5_tb = systematic_binary(h_tb, n_t, rng, time_limit=25)
                if c5_tb < c5_t:
                    c5_t = c5_tb
                    h_t = h_tb.copy()
                    h_t = slsqp_smooth(h_t, n_t, 10000, 300)
                    h_t, c5_t = equalize(h_t, n_t, top_k=5, max_iters=200)

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 8: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    for beta in [5000, 20000, 80000, 200000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    for tk in [3, 5, 7, 10]:
        if remaining() < 8:
            break
        h_eq, c5_eq = equalize(h, n, top_k=tk, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final L-BFGS-B polish
    if remaining() > 15:
        h_lb, c5_lb = lbfgs_auglag(h, n, maxiter=400)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    # Final binary search
    if remaining() > 20:
        h_bin = binarize(best_h)
        h_b, c5_b = systematic_binary(h_bin, n, rng, time_limit=min(50, int(remaining() * 0.4)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = slsqp_smooth(h_b, n, 50000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Final subgradient polish
    if remaining() > 10:
        h_sg, c5_sg = subgradient_descent(best_h, best_n, n_iter=150)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
