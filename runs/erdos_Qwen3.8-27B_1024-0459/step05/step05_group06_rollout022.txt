```python
def compute_c_vector(i, best, n):
    ncorr = 2 * n - 1
    c = np.zeros(ncorr)
    
    # Positive offsets: o = 0, 1, ..., n-1 -> k = n-1, n, ..., 2n-2
    # First factor: (1-h[i+o]) if i+o < n
    # Second factor: -h[i-o] if i >= o
    o_pos = np.arange(n)  # 0 to n-1
    k_pos = n - 1 + o_pos  # n-1 to 2n-2
    
    # First factor
    valid_f = i + o_pos < n  # i+o < n
    first_f = np.where(valid_f, 1.0 - best[i + o_pos[valid_f]], 0.0)
    # Actually need to be careful with indexing
    idx_f = i + o_pos
    first = np.zeros(n)
    mask_f = idx_f < n
    first[mask_f] = 1.0 - best[idx_f[mask_f]]
    
    # Second factor
    valid_s = o_pos <= i  # i >= o, so i-o >= 0
    idx_s = i - o_pos[valid_s]
    second = np.zeros(n)
    second[valid_s] = -best[idx_s]
    
    c[k_pos] = first + second
    
    # Negative offsets: a = 1, 2, ..., n-1 -> k = n-2, n-3, ..., 0
    a_neg = np.arange(1, n)  # 1 to n-1
    k_neg = n - 1 - a_neg  # n-2 to 0
    
    # First factor: (1-h[i-a]) if i >= a
    idx_f2 = i - a_neg
    first2 = np.zeros(n-1)
    mask_f2 = idx_f2 >= 0
    first2[mask_f2] = 1.0 - best[idx_f2[mask_f2]]
    
    # Second factor: -h[i+a] if i+a < n
    idx_s2 = i + a_neg
    second2 = np.zeros(n-1)
    mask_s2 = idx_s2 < n
    second2[mask_s2] = -best[idx_s2[mask_s2]]
    
    c[k_neg] = first2 + second2
    
    return c
```

```python
from scipy.optimize import minimize
import time

_N_OPT = 0
_BETA_OPT = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13 and n > 0:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

    return smooth_value_grad(x, _N_OPT, _BETA_OPT)[0]

def _grad_fn(x):
    return smooth_value_grad(x, _N_OPT, _BETA_OPT)[1]

    return np.sum(x) - _N_OPT / 2.0

def slsqp_optimize(h, n, beta, maxiter=400):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-13:

    except Exception:

def compute_c_vec(i, h, n):
    """Compute coefficient vector c[k] = d(corr[k])/d(h[i])"""
    ncorr = 2 * n - 1
    c = np.zeros(ncorr)
    
    # For offset o >= 0: k = n-1+o
    # corr[k] = sum_{j=0}^{n-1-o} h[j]*(1-h[j+o])
    # h[i] appears as:
    #   first: (1-h[i+o]) if i+o < n (i.e., i < n-o)
    #   second: -h[i-o] if i >= o (so i-o >= 0) and i-o <= n-1-o (i <= n-1, always true)
    
    o = np.arange(n)  # 0 to n-1
    k_idx = n - 1 + o  # n-1 to 2n-2
    
    # First factor: (1-h[i+o]) for i+o < n
    idx1 = i + o
    m1 = idx1 < n
    first = np.zeros(n)
    first[m1] = 1.0 - h[idx1[m1]]
    
    # Second factor: -h[i-o] for i >= o
    m2 = (o <= i)
    idx2 = i - o[m2]
    second = np.zeros(n)
    second[m2] = -h[idx2]
    
    c[k_idx] = first + second
    
    # For offset o < 0, let a = -o (1 to n-1): k = n-1-a
    # corr[k] = sum_{j=a}^{n-1} h[j]*(1-h[j-a])
    # h[i] appears as:
    #   first: (1-h[i-a]) if i >= a (so i-a >= 0) and i-a < n (always)
    #   second: -h[i+a] if i+a < n
    
    a = np.arange(1, n)  # 1 to n-1
    k_idx2 = n - 1 - a  # n-2 to 0
    
    # First factor: (1-h[i-a]) for i >= a
    idx1b = i - a
    m1b = idx1b >= 0
    firstb = np.zeros(n - 1)
    firstb[m1b] = 1.0 - h[idx1b[m1b]]
    
    # Second factor: -h[i+a] for i+a < n
    idx2b = i + a
    m2b = idx2b < n
    secondb = np.zeros(n - 1)
    secondb[m2b] = -h[idx2b[m2b]]
    
    c[k_idx2] = firstb + secondb
    
    return c

def exact_1d_min(base, c):
    """Minimize max_k (base[k] + c[k]*t) for t in [0,1]. Returns (t*, f*)."""
    # Check if c is all zeros
    if np.allclose(c, 0.0):
        return 0.0, np.max(base)
    
    # Evaluate at endpoints
    f0 = np.max(base + c * 0.0)  # = np.max(base)
    f1 = np.max(base + c * 1.0)
    
    # If c is all positive, optimum is at t=0
    if np.min(c) > 1e-14:
        return 0.0, f0
    # If c is all negative, optimum is at t=1
    if np.max(c) < -1e-14:
        return 1.0, f1
    
    # Binary search on subgradient
    t_lo = 0.0
    t_hi = 1.0
    t = 0.5
    
    for _ in range(50):
        if t_hi - t_lo < 1e-14:
            break
        vals = base + c * t
        m = np.max(vals)
        tol = 1e-9 * max(1.0, abs(m))
        active_mask = vals >= m - tol
        active_c = c[active_mask]
        
        if len(active_c) == 0:
            # Shouldn't happen, but fallback
            break
        
        if np.min(active_c) > 1e-12:
            t_hi = t
        elif np.max(active_c) < -1e-12:
            t_lo = t
        else:
            # 0 in subgradient, optimal
            break
        t = (t_lo + t_hi) / 2.0
    
    f_t = np.max(base + c * t)
    best_t = t
    best_f = f_t
    
    if f0 < best_f - 1e-15:
        best_t = 0.0
        best_f = f0
    if f1 < best_f - 1e-15:
        best_t = 1.0
        best_f = f1
    
    return best_t, best_f

def coord_descent_exact(h, n, n_sweeps=30):
    """Coordinate descent with exact 1D minimization."""
    best = h.copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_corr = np.array(compute_corr(best), dtype=float)
    best_c5 = np.max(best_corr) * dx
    
    for sweep in range(n_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        
        # Alternate direction each sweep
        order = np.arange(n) if sweep % 2 == 0 else np.arange(n - 1, -1, -1)
        
        for i in order:
            if remaining_time() < 0.5:
                break
            
            t0 = best[i]
            c = compute_c_vec(i, best, n)
            
            # Skip if c is all zeros
            if np.max(np.abs(c)) < 1e-15:
                continue
            
            base = best_corr - c * t0
            t_opt, f_opt = exact_1d_min(base, c)
            
            # Update if it doesn't increase max
            if f_opt <= np.max(best_corr) + 1e-13 and abs(t_opt - t0) > 1e-13:
                best[i] = t_opt
                best_corr = base + c * t_opt
                if np.max(best_corr) * dx < best_c5 - 1e-15:
                    best_c5 = np.max(best_corr) * dx
                    improved = True
                elif abs(t_opt - t0) > 1e-10:
                    improved = True  # Allow neutral moves
        
        # Project to feasibility

            best = project_feasible(best, n)

            best_c5 = np.max(best_corr) * dx
        
        if not improved:
            # Try one more sweep in case numerical issues prevented last update
            if sweep > 0 and not improved:
                break
    
    return best, best_c5
```

```python
import numpy as np

_N_OPT = 0
_BETA_OPT = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13 and n > 0:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

    return smooth_value_grad(x, _N_OPT, _BETA_OPT)[0]

def _grad_fn(x):
    return smooth_value_grad(x, _N_OPT, _BETA_OPT)[1]

def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

    return np.ones(_N_OPT)

def slsqp_optimize(h, n, beta, maxiter=400):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)

def compute_c_vec(i, h, n):
    ncorr = 2 * n - 1
    c = np.zeros(ncorr)
    o = np.arange(n)
    k_idx = n - 1 + o
    idx1 = i + o
    m1 = idx1 < n
    first = np.zeros(n)
    first[m1] = 1.0 - h[idx1[m1]]
    m2 = (o <= i)
    idx2 = i - o[m2]
    second = np.zeros(n)
    second[m2] = -h[idx2]
    c[k_idx] = first + second
    a = np.arange(1, n)
    k_idx2 = n - 1 - a
    idx1b = i - a
    m1b = idx1b >= 0
    firstb = np.zeros(n - 1)
    firstb[m1b] = 1.0 - h[idx1b[m1b]]
    idx2b = i + a
    m2b = idx2b < n
    secondb = np.zeros(n - 1)
    secondb[m2b] = -h[idx2b[m2b]]
    c[k_idx2] = firstb + secondb
    return c

def exact_1d_min(base, c):
    if np.allclose(c, 0.0, atol=1e-15):
        return 0.0, np.max(base)
    f0 = np.max(base)
    f1 = np.max(base + c)
    if np.min(c) > 1e-14:
        return 0.0, f0
    if np.max(c) < -1e-14:
        return 1.0, f1
    t_lo = 0.0
    t_hi = 1.0
    t = 0.5

            break

        m = np.max(vals)

        if len(active_c) == 0:

        if np.min(active_c) > 1e-12:

        elif np.max(active_c) < -1e-12:
            t_lo = t
        else:

        t = (t_lo + t_hi) / 2.0
    f_t = np.max(base + c * t)
    best_t = t
    best_f = f_t
    if f0 < best_f - 1e-15:

        best_f = f0
    if f1 < best_f - 1e-15:
        best_t = 1.0
        best_f = f1
    return best_t, best_f

            best_h = h_new.copy()

    # Phase 4: More coordinate descent
    if remaining_time() > 40:
        n_sweeps = max(20, int(remaining_time() // 10))
        n_sweeps = min(n_sweeps, 200)
        h_new, c5_new = coord_descent_exact(h, n, n_sweeps=n_sweeps)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Random perturbations to escape local optima
    if remaining_time() > 60:
        n_perturb = max(5, n // 40)
        for pert in range(8):
            if remaining_time() < 30:
                break
            hp, c5p = try_perturb_and_optimize(best_h, best_n, rng, n_perturb, 0.1)
            if c5p < best_c5:
                best_c5 = c5p
                best_h = hp.copy()
                n_perturb = max(3, n_perturb // 2)

    # Phase 6: Higher resolution
    for rn in [960, 1280]:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        h_r = slsqp_optimize(h_r, rn, 5000, 200)
        if remaining_time() > 40:
            n_sweeps = max(10, int(remaining_time() // 15))
            n_sweeps = min(n_sweeps, 80)
            h_r, c5_r = coord_descent_exact(h_r, rn, n_sweeps=n_sweeps)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 7: Final intensive coordinate descent on best
    n = best_n
    h = best_h.copy()
    if remaining_time() > 20:
        n_sweeps = max(10, int(remaining_time() // 8))
        n_sweeps = min(n_sweeps, 150)
        h_new, c5_new = coord_descent_exact(h, n, n_sweeps=n_sweeps)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 8: Final SLSQP
    if remaining_time() > 10:
        h_new = slsqp_optimize(best_h, best_n, 200000, 150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: One last coord descent
    if remaining_time() > 8:
        h_new, c5_new = coord_descent_exact(best_h, best_n, n_sweeps=20)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)
        final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
```
