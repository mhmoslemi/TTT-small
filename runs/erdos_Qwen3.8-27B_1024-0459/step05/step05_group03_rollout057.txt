    idx = np.argsort(-h)

    h_bin[idx[:n//2]] = 1.0

    h = h_init.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_ones = len(ones)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    T = t_start
    cool_factor = (t_end / t_start) ** (1.0 / n_iters)
    
    for it in range(n_iters):
        if remaining() < 2:
            break
        
        # Pick random swap
        ia = rng.randint(n_ones)
        ib = rng.randint(n - n_ones)
        pa = ones[ia]
        pb = zeros[ib]
        
        # Perform swap
        h[pa] = 0.0
        h[pb] = 1.0
        ones[ia] = pb
        zeros[ib] = pa
        
        # Evaluate

        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):

                best_c5 = new_c5
                best_h = h.copy()

            # Revert
            h[pa] = 1.0
            h[pb] = 0.0
            ones[ia] = pa
            zeros[ib] = pb

            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)

            nc = compute_c5(h)

            h[pa] = 1.0
            h[pb] = 0.0
            if delta > best_delta:
                best_delta = delta
                best_pa, best_pb = pa, pb
        if best_pa < 0 or best_delta < 1e-15:
            break
        h[best_pa] = 0.0
        h[best_pb] = 1.0
        idx_pa = np.where(ones == best_pa)[0][0]
        idx_pb = np.where(zeros == best_pb)[0][0]
        ones[idx_pa] = best_pb
        zeros[idx_pb] = best_pa
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5


def ils_binary(h_init, n, rng, n_rounds=40, perturb_k=4):
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_ones = n // 2
    for _ in range(n_rounds):
        if remaining() < 15.0:
            break
        h_ls, c5_ls = greedy_ls(h_cur, n, rng, max_rounds=25, n_trials=3000)
        h_cur = h_ls
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_cur.copy()
        ones = np.where(h_cur == 1)[0].copy()
        zeros = np.where(h_cur == 0)[0].copy()
        for _ in range(perturb_k):
            ia = rng.randint(n_ones)
            ib = rng.randint(n_ones)
            h_cur[ones[ia]] = 0.0
            h_cur[zeros[ib]] = 1.0
            ones[ia] = zeros[ib]
            zeros[ib] = ones[ia]
    return best_h, best_c5


def smooth_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -45.0, 45.0)
    ex = np.exp(sh)
    dx = 2.0 / len(x)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_grad(x, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -45.0, 45.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            g[0:n - s] += wk * (1.0 - x[s:n])
            g[s:n] -= wk * x[0:n - s]
        else:
            a = -s
            g[a:n] += wk * (1.0 - x[0:n - a])
            g[0:n - a] -= wk * x[a:n]
    return g


def slsqp_optimize(h, n, beta, maxiter=300):
    h0 = project_feasible(h.copy(), n)

    def obj(x):
        return smooth_obj(x, beta)

    def grad(x):
        return smooth_grad(x, beta)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_h(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_n = parent_n
        best_c5 = compute_c5(best_h)
    except Exception:
        parent_n = 384
        best_n = parent_n
        best_h = np.zeros(parent_n)
        best_h[:parent_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    n0 = best_n

    # Phase 1: Binary SA from parent at native resolution
    if remaining() > 60:
        h_bin = to_binary(best_h, n0)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        # SA with restarts
        for restart in range(4):
            if remaining() < 30:
                break
            t_start = 0.004 if restart == 0 else 0.002
            t_end = 3e-6
            n_iters = min(800000, int(remaining() // 0.06))
            h_sa, c5_sa = sa_binary(h_bin, n0, t_start, t_end, n_iters, rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h_bin = h_sa.copy()
            # Perturb for next restart
            h_bin2 = h_sa.copy()
            ones = np.where(h_bin2 == 1)[0]
            zeros = np.where(h_bin2 == 0)[0]
            for _ in range(3 + restart):
                ia = rng.randint(len(ones))
                ib = rng.randint(len(zeros))
                h_bin2[ones[ia]] = 0.0
                h_bin2[zeros[ib]] = 1.0
            h_bin = h_bin2

    # Phase 2: ILS on best binary
    if remaining() > 40:
        h_bin = to_binary(best_h, best_n)
        h_ils, c5_ils = ils_binary(h_bin, best_n, rng, n_rounds=35, perturb_k=4)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()

    # Phase 3: Greedy local search intensive
    if remaining() > 30:
        h_bin = to_binary(best_h, best_n)
        h_gs, c5_gs = greedy_ls(h_bin, best_n, rng, max_rounds=60, n_trials=5000)
        if c5_gs < best_c5:
            best_c5 = c5_gs
            best_h = h_gs.copy()

    # Phase 4: SLSQP continuous polish
    if remaining() > 30:
        h_pol = to_binary(best_h, best_n).astype(float)
        for beta in [2000, 10000, 50000, 200000]:
            if remaining() < 15:
                break
            h_new = slsqp_optimize(h_pol, best_n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_pol = h_new.copy()

    # Phase 5: Multi-resolution exploration
    if remaining() > 60:
        for n_t in [512, 448, 640]:
            if remaining() < 30:
                break
            if n_t % 2 != 0:
                n_t += 1
            h_int = interp_h(best_h, best_n, n_t)
            h_int = project_feasible(h_int, n_t)
            h_bin_t = to_binary(h_int, n_t)

            # Quick SA
            t_s = 0.003
            t_e = 5e-6
            n_it = min(300000, int(remaining() // 0.08))
            h_sa_t, c5_sa_t = sa_binary(h_bin_t, n_t, t_s, t_e, n_it, rng)
            if c5_sa_t > best_c5:
                h_best_t = h_bin_t
                c5_best_t = c5_sa_t
            else:
                h_best_t = h_sa_t
                c5_best_t = compute_c5(h_bin_t)
                if c5_best_t > c5_sa_t:
                    h_best_t = h_sa_t
                    c5_best_t = c5_sa_t

            # SLSQP at this resolution
            if remaining() > 15:
                h_pol_t = slsqp_optimize(h_best_t, n_t, 10000, maxiter=200)
                c5_pol_t = compute_c5(h_pol_t)
                if c5_pol_t < c5_best_t:
                    c5_best_t = c5_pol_t
                    h_best_t = h_pol_t

            if c5_best_t < best_c5:
                best_c5 = c5_best_t
                best_h = h_best_t.copy()
                best_n = n_t

    # Phase 6: Final ILS + SLSQP at best resolution
    if remaining() > 25:
        h_bin_f = to_binary(best_h, best_n)
        h_ils_f, c5_ils_f = ils_binary(h_bin_f, best_n, rng, n_rounds=20, perturb_k=3)
        if c5_ils_f < best_c5:
            best_c5 = c5_ils_f
            best_h = h_ils_f.copy()
        h_pol_f = slsqp_optimize(to_binary(best_h, best_n), best_n, 100000, maxiter=300)
        c5_pol_f = compute_c5(h_pol_f)
        if c5_pol_f < best_c5:
            best_c5 = c5_pol_f
            best_h = h_pol_f.copy()

    # Phase 7: One more SA round on the polished solution (binary)
    if remaining() > 20:
        h_bin7 = to_binary(best_h, best_n)
        h_sa7, c5_sa7 = sa_binary(h_bin7, best_n, 0.001, 1e-7, 200000, rng)
        if c5_sa7 < best_c5:
            best_c5 = c5_sa7
            best_h = h_sa7.copy()
        h_pol7 = slsqp_optimize(best_h, best_n, 200000, maxiter=200)
        c5_pol7 = compute_c5(h_pol7)
        if c5_pol7 < best_c5:
            best_c5 = c5_pol7
            best_h = h_pol7.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
