    n_ones = int(h.sum())

        h[to_rm] = 0

        if time.time() - start >= time_limit or remaining() < 15:
            break
        
        pert_size = 3 + int(3 * np.exp(-iters_since_improve / 500.0))
        pert_size = max(2, min(pert_size, 12))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(pert_size):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0, 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        corr = compute_corr(h)
        cur_max = float(corr.max())
    
    return best_h, best_max * dx

def sa_binary(h_bin, n, time_limit, rng):
    """Simulated Annealing binary search."""
    h = h_bin.astype(float).copy()
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, target - n_ones, replace=False)] = 1
    
    corr = compute_corr(h)
    cur_max = float(corr.max())
    dx = 2.0 / n
    best_h = h.copy()
    best_max = cur_max
    
    start = time.time()
    T_init = 0.5
    T_min = 0.005
    
    while time.time() - start < time_limit and remaining() > 15:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / time_limit)
        T = T_init * (T_min / T_init) ** frac
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Generate candidates
        candidates = []
        top_k = np.argsort(-corr)[:8]
        for s in top_k:
            off = s - (n - 1)
            if off >= 0:
                contrib = np.where((h[off:] == 1) & (h[:n-off] == 0))[0]
                if len(contrib) > 0:
                    picks = rng.choice(contrib, min(5, len(contrib)), replace=False)
                    for ci in picks:
                        candidates.append((ci + off, ci))
            else:
                ao = -off
                contrib = np.where((h[:n-ao] == 1) & (h[ao:] == 0))[0]
                if len(contrib) > 0:
                    picks = rng.choice(contrib, min(5, len(contrib)), replace=False)
                    for ci in picks:
                        candidates.append((ci, ci + ao))
        for _ in range(20):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        # Find best move (allow worsening with SA probability)
        best_move = None
        best_score = cur_max
        
        seen = set()
        for (i, j) in candidates:
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            
            h[i], h[j] = 0, 1
            new_corr = compute_corr(h)
            new_max = float(new_corr.max())
            delta = new_max - cur_max
            
            if delta < 0:
                score = new_max
            else:
                if rng.rand() < np.exp(-delta / max(T, 1e-10)):
                    score = new_max
                else:
                    h[i], h[j] = 1, 0
                    continue
            
            h[i], h[j] = 1, 0
            if score < best_score:
                best_score = score
                best_move = (i, j, new_corr.copy())
        
        if best_move is not None:
            i, j, new_corr = best_move
            h[i], h[j] = 0, 1
            corr = new_corr
            cur_max = best_score
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    
    return best_h, best_max * dx

def interpolate_h(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def continuous_refine(h, n, beta, maxiter):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x)
        m = corr.max()
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(ex.sum()) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = corr.max()
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / ex.sum()
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao+np_])
                g[ao:ao+np_] -= wk * x[:np_]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-12:
            break
        
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off+np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao+np_])
                g_avg[ao:ao+np_] -= h[:np_]
        
        bot_idx = np.argsort(corr)[:3]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off+np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao+np_])
                g_bot[ao:ao+np_] -= h[:np_]
        
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.03, 0.06, 0.015, 0.04]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(h.sum())
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones-target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target-n_ones]] = 1
    return h

def greedy_construct(n, rng):
    h = np.zeros(n)
    n_half = n // 2
    positions = list(range(n))
    rng.shuffle(positions)
    for _ in range(n_half):
        best_pos = -1
        best_max = float('inf')
        for p in positions:
            if h[p] > 0:
                continue
            h[p] = 1
            corr = compute_corr(h)
            mx = float(corr.max())
            h[p] = 0
            if mx < best_max:
                best_max = mx
                best_pos = p
        if best_pos >= 0:
            h[best_pos] = 1
        else:
            for p in positions:
                if h[p] == 0:
                    h[p] = 1
                    break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse_init(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: ILS binary search at parent resolution (main workhorse - 55% budget)
    n = best_n
    h_bin = make_binary(best_h, n, rng)
    
    ils_time = min(700, int(remaining() * 0.55))
    if ils_time > 30:
        h_b1, c5_b1 = ils_binary(h_bin, n, ils_time, rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n
    
    # Phase 2: SA from different starting points (15% budget)
    sa_time_each = min(80, int(remaining() * 0.07))
    if sa_time_each > 10:
        starts = [
            thue_morse_init(n),
            greedy_construct(n, np.random.RandomState(seed + 200)),
            make_binary(thue_morse_init(n) + 0.3 * rng.rand(n), n, rng),
        ]
        for sp in starts:
            if remaining() < 20:
                break
            h_sa, c5_sa = sa_binary(sp, n, sa_time_each, rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n
        
        # Also run SA from current best binary
        h_b_bin = make_binary(best_h, n, rng)
        h_sa2, c5_sa2 = sa_binary(h_b_bin, n, sa_time_each, rng)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            best_n = n
    
    # Phase 3: Try other resolutions (15% budget)
    n = best_n
    resolutions = []
    for dn in [-16, -8, 8, 16, 32, -32, 48]:
        rn = n + dn
        if 256 <= rn <= 640 and rn % 2 == 0 and rn != n:
            resolutions.append(rn)
    # Add some "nice" resolutions
    for rn in [320, 360, 400, 448, 480, 512, 560]:
        if rn != n and 256 <= rn <= 640 and rn % 2 == 0:
            if rn not in resolutions:
                resolutions.append(rn)
    
    resolutions.sort(key=lambda x: abs(x - n))
    res_time_each = min(50, int(remaining() * 0.04))
    
    for n_t in resolutions[:5]:
        if remaining() < 30 or res_time_each < 10:
            break
        h_interp = interpolate_h(best_h, best_n, n_t)
        h_bin_t = make_binary(h_interp, n_t, rng)
        h_r, c5_r = ils_binary(h_bin_t, n_t, res_time_each, rng)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_t
    
    # Phase 4: Continuous refinement (10% budget)
    n = best_n
    h = best_h.copy()
    for beta in [200, 1000, 5000, 20000]:
        if remaining() < 10:
            break
        h_new = continuous_refine(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Top-shifts equalization
    if remaining() > 15:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 6: More ILS from best continuous (binarized)
    if remaining() > 30:
        h_bin2 = make_binary(best_h, n, rng)
        ils_time2 = min(100, int(remaining() * 0.3))
        if ils_time2 > 15:
            h_b2, c5_b2 = ils_binary(h_bin2, n, ils_time2, rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
    
    # Phase 7: Final polish - continuous
    n = best_n
    h = best_h.copy()
    for beta in [10000, 30000, 80000]:
        if remaining() < 8:
            break
        h_new = continuous_refine(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 8: Final equalization
    if remaining() > 8:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=8, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Ensure feasibility and final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
