import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def precompute_FS(h, n):
    L = np.arange(-(n - 1), n)
    F = np.zeros((n, 2 * n - 1))
    S = np.zeros((n, 2 * n - 1))
    p = np.arange(n)
    pf = p[:, None] + L[None, :]
    sf = p[:, None] - L[None, :]
    vf = (pf >= 0) & (pf < n)
    vs = (sf >= 0) & (sf < n)
    idx_f = np.where(vf, pf, 0)
    idx_s = np.where(vs, sf, 0)
    F = np.where(vf, 1.0 - h[idx_f], 0.0)
    S = np.where(vs, h[idx_s], 0.0)
    return F, S, L


def update_FS(F, S, h, n, pos, L):
    p = pos
    pf = p + L
    sf = p - L
    vf = (pf >= 0) & (pf < n)
    vs = (sf >= 0) & (sf < n)
    F[p] = 0.0
    S[p] = 0.0
    idx_f = pf[vf]
    idx_s = sf[vs]
    F[p, vf] = 1.0 - h[idx_f]
    S[p, vs] = h[idx_s]

            new_C = C[None, None, :] + delta
            new_max = new_max_vals = new_C.max(axis=2)

            for i in range(len(ones)):

def binary_sa(h, n, T0=0.03, T1=0.0005, max_iters=3000, rng=None):

        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    if abs(h.sum() - n / 2) > 1e-10:
        h = round_to_binary(h, n)
    F, S, L = precompute_FS(h, n)
    C = compute_corr(h)
    cur_c5 = float(np.max(C) * 2.0 / n)
    best_c5 = cur_c5
    best_h = h.copy()

    for it in range(max_iters):
        if remaining() < 15:
            break
        T = T0 * (T1 / T0) ** (it / max_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        if a == b:
            continue
        delta_C = -F[a] + S[a] + F[b] - S[b]
        new_max = float(np.max(C + delta_C))

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / n
    w = w * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def slsqp_run(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def nesterov_refine(h, n, beta, max_iters=2000):
    h = project_feasible(h, n).copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0

    for it in range(max_iters):
        if remaining() < 5:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        y_new = h + alpha * (h - y)
        g = smooth_grad(y_new, beta)
        lr = 2.0 / (beta * (it + 10))
        h_new = project_feasible(y_new - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new
    return best_h, best_c5

                                max_iters=5000, rng=rng)
        c5_sa = compute_c5(h_sa)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: More greedy rounds from SA result
    if remaining() > 60:
        h_g2, c5_g2 = binary_greedy_vectorized(h_sa, best_n, max_rounds=300, tabu_len=50)
        c5_g2 = compute_c5(h_g2)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    # Phase 4: Continuous refinement (SLSQP + Nesterov)
    if remaining() > 60:
        h_cont = best_h.copy()
        for beta in [300, 1000, 3000, 10000, 30000, 80000]:
            if remaining() < 15:
                break
            h_cont = slsqp_run(h_cont, best_n, beta, maxiter=400)
            c5_c = compute_c5(h_cont)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_cont.copy()

    if remaining() > 40:
        h_nes, c5_nes = nesterov_refine(best_h, best_n, beta=10000, max_iters=1500)
        if c5_nes < best_c5:
            best_c5 = c5_nes
            best_h = h_nes.copy()

    # Phase 5: Quadratic residue initialization (n=396, p=397)
    if remaining() > 80:
        h_qr = quadratic_residue_init(397)
        n_qr = len(h_qr)
        c5_qr = compute_c5(h_qr)
        if c5_qr < best_c5 + 0.05:  # worth trying
            h_qr, c5_qr2 = binary_greedy_vectorized(h_qr, n_qr, max_rounds=300, tabu_len=40)
            c5_qr2 = compute_c5(h_qr)
            if c5_qr2 < best_c5:
                best_c5 = c5_qr2
                best_h = h_qr.copy()
                best_n = n_qr
            if remaining() > 30:
                h_qr_c = slsqp_run(h_qr, n_qr, 10000, maxiter=300)
                c5_qr3 = compute_c5(h_qr_c)
                if c5_qr3 < best_c5:
                    best_c5 = c5_qr3
                    best_h = h_qr_c.copy()
                    best_n = n_qr

    # Phase 6: Try n=382 (p=383)
    if remaining() > 60:
        h_qr2 = quadratic_residue_init(383)
        n_qr2 = len(h_qr2)
        c5_qr2 = compute_c5(h_qr2)
        if c5_qr2 < best_c5 + 0.05:
            h_qr2, c5_qr2b = binary_greedy_vectorized(h_qr2, n_qr2, max_rounds=250, tabu_len=40)
            c5_qr2b = compute_c5(h_qr2)
            if c5_qr2b < best_c5:
                best_c5 = c5_qr2b
                best_h = h_qr2.copy()
                best_n = n_qr2
            if remaining() > 20:
                h_qr2c = slsqp_run(h_qr2, n_qr2, 10000, maxiter=250)
                c5_qr2c = compute_c5(h_qr2c)
                if c5_qr2c < best_c5:
                    best_c5 = c5_qr2c
                    best_h = h_qr2c.copy()
                    best_n = n_qr2

    # Phase 7: Multi-resolution at original best_n
    if remaining() > 60:
        for scale in [1.25, 0.8]:
            nn = int(best_n * scale)
            if nn < 128 or nn > 800 or nn % 2 != 0:
                continue
            h_interp = interp_to(best_h, best_n, nn)
            h_interp = project_feasible(h_interp, nn)
            h_bin_i = round_to_binary(h_interp, nn)
            h_bin_i, c5_i = binary_greedy_vectorized(h_bin_i, nn, max_rounds=200, tabu_len=30)
            c5_i = compute_c5(h_bin_i)
            if c5_i < best_c5:
                best_c5 = c5_i
                best_h = h_bin_i.copy()
                best_n = nn
            if remaining() > 20 and c5_i < best_c5 + 0.01:
                h_ic = slsqp_run(h_bin_i, nn, 10000, maxiter=300)
                c5_ic = compute_c5(h_ic)
                if c5_ic < best_c5:
                    best_c5 = c5_ic
                    best_h = h_ic.copy()
                    best_n = nn

    # Phase 8: Iterative binary-continuous refinement
    if remaining() > 80:
        h_iter, c5_iter = binary_iterative_refine(best_h, best_n, max_rounds=15)
        if c5_iter < best_c5:
            best_c5 = c5_iter
            best_h = h_iter.copy()

    # Phase 9: Final intensive SLSQP
    if remaining() > 30:
        for beta in [50000, 150000, 400000]:
            if remaining() < 10:
                break
            h_f = slsqp_run(best_h, best_n, beta, maxiter=500)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 10: Final binary tweak
    if remaining() > 20:
        h_bf = round_to_binary(best_h, best_n)
        h_bf, c5_bf = binary_greedy_vectorized(h_bf, best_n, max_rounds=200, tabu_len=60)
        c5_bf = compute_c5(h_bf)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        if remaining() > 15:
            h_pol = slsqp_run(h_bf, best_n, 100000, maxiter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    else:
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
