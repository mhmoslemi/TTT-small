    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def random_01_struct(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return project_feasible(h, n)

def block_struct(n, n_blocks, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    block_len = n // (2 * n_blocks)
    gap_len = (n - n_blocks * block_len) // n_blocks
    pos = 0
    rng.shuffle_list = None
    offsets = sorted(rng.choice(n, n_blocks, replace=False))
    used = set()
    h = np.zeros(n)
    for i in range(n_blocks):
        start = offsets[i] % n
        end = start + block_len
        for j in range(start, end):
            h[j % n] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = random_01_struct(best_n, 0)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP at moderate beta
    for beta in [200, 800, 4000]:
        if remaining_time() < 30:
            break
        h, c5_new = slsqp_optimize(h, n, beta, 250)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: Coordinate descent
    if remaining_time() > 60:
        h_cd, c5_cd = coord_descent(h, n, n_sweeps=100)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 3: Subgradient descent (broad search)
    if remaining_time() > 120:
        h_sg, c5_sg = subgrad_descent(h, n, n_iter=2000, n_active=6)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: More SLSQP at higher beta
    for beta in [10000, 50000, 200000]:
        if remaining_time() < 25:
            break
        h, c5_new = slsqp_optimize(h, n, beta, 300)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 5: Coordinate descent again
    if remaining_time() > 50:
        h_cd, c5_cd = coord_descent(h, n, n_sweeps=80)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()

    # Phase 6: Random restarts with different structures
    n_restarts = 5
    for r in range(n_restarts):
        if remaining_time() < 80:
            break
        if r == 0:
            h_r = h + rng.uniform(-0.1, 0.1, n)
            h_r = project_feasible(h_r, n)
        elif r == 1:
            h_r = random_01_struct(n, seed + 100 + r)
        elif r == 2:
            # Smoothed version of random
            h_r = random_01_struct(n, seed + 200 + r)
            kernel = np.ones(5) / 5.0
            h_r = np.convolve(h_r, kernel, mode='same')
            h_r = project_feasible(h_r, n)
        elif r == 3:
            # Flip half the bits of current best
            h_r = h.copy()
            flip = rng.choice(n, n // 4, replace=False)
            h_r[flip] = 1.0 - h_r[flip]
            h_r = project_feasible(h_r, n)
        else:
            # Block structure
            nb = rng.randint(3, 10)
            h_r = block_struct(n, nb, seed + 300 + r)

        h_r, c5_r = slsqp_optimize(h_r, n, 5000, 150)
        if remaining_time() > 30:
            h_r, c5_r = coord_descent(h_r, n, n_sweeps=40)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            h = h_r.copy()

    # Phase 7: Try higher resolution
    for mult in [1.5, 2.0]:
        if remaining_time() < 80:
            break
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if rn > 2048 or rn < 128:
            continue
        h_hi = interp_to(best_h, best_n, rn)
        h_hi = project_feasible(h_hi, rn)
        for beta in [5000, 30000]:
            if remaining_time() < 30:
                break
            h_hi, _ = slsqp_optimize(h_hi, rn, beta, 200)
        if remaining_time() > 20:
            h_hi, c5_hi = coord_descent(h_hi, rn, n_sweeps=30)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = rn

    # Phase 8: Intensive subgradient on best
    if remaining_time() > 100:
        h_sg, c5_sg = subgrad_descent(best_h, best_n, n_iter=3000, n_active=10)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 40:
        h, c5_new = slsqp_optimize(best_h, best_n, 500000, 200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 20:
        h_cd, c5_cd = coord_descent(best_h, best_n, n_sweeps=50)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 11: Ultra-fine coordinate descent (only interior points)
    if remaining_time() > 15:
        h_fine = best_h.copy()
        c5_fine = best_c5
        for i in range(len(h_fine)):
            if remaining_time() < 2:
                break
            ti = h_fine[i]
            if 0.01 < ti < 0.99:
                for t in [0.0, 1.0, max(0.0, ti - 0.05), min(1.0, ti + 0.05)]:
                    if abs(t - ti) < 1e-8:
                        continue
                    h_fine[i] = t
                    f = compute_c5(h_fine)
                    if f < c5_fine - 1e-15:
                        c5_fine = f
                    else:
                        h_fine[i] = ti
                        break
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        cnt = np.sum(free)
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, final_c5, best_n)
