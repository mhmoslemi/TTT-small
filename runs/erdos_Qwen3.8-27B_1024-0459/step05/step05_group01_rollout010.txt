def slsqp_opt(h, n, beta, maxiter=500):

            return project_feasible(res.x, n)
    except Exception:

    return h0


_CUR_N = 0
_SM_BETA = 1000.0


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g

def top_k_equalize(h, n, top_k=5, max_iters=300):

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:

    return np.interp(xn, xp, src)

def hashes_mix(s):
    """Simple hash for seeding rngs from string names."""
    v = 0
    for c in s:

    return v
