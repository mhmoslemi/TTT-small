$$C_5 = \max_k \int h(x)(1 - h(x+k)) dx$$

The discretization uses n_points samples over [0,2], with dx = 2/n_points.
The objective is computed as: max(np.correlate(h, 1-h, mode="full")) * dx

5. **Key mathematical insight**: For the binary case, if h has n/2 ones among n positions, then:
   - corr[k] = sum of h[i]*h[i+k-(n-1)] over valid i (for lag k-(n-1))
   - Actually, corr[k] from np.correlate(h, 1-h, mode='full') gives:
     - For k >= n-1: sum_{i=0}^{n-1-(k-(n-1))} h[i] * (1-h[i+k-(n-1)])
     - For k < n-1: sum_{i=k-(n-1)+1}^{n-1} h[i] * (1-h[i+k-(n-1)])
   
   Wait, let me think more carefully. np.correlate(a, b, mode='full')[k] = sum_i a[i] * b[i + k - (len(b)-1)]
   
   Actually, np.correlate(a, v, mode='full')[k] = sum_n a[n+k] * conj(v[n]) for valid n
   
   Let me just use the formula directly. With a=h, b=1-h:
   corr[k] = sum over valid i of h[i] * (1-h[i + (k-(n-1))])
   
   For k = n-1: corr[n-1] = sum_i h[i] * (1-h[i]) = sum_i h[i] - sum_i h[i]^2 = n/2 (for binary with n/2 ones)
   
   Wait, that's interesting. For binary h with exactly n/2 ones:
   corr[n-1] = sum_i h[i](1-h[i]) = n/2 (since h[i](1-h[i]) = 0 for binary)
   
   Hmm wait: if h[i] is 0 or 1, then h[i](1-h[i]) = 0 always. So corr[n-1] = 0!
   
   Actually let me re-check. For binary h:
   - h[i] * (1-h[i]) = 0 for all i
   - So the autocorrelation at lag 0 of h*(1-h) is 0
   
   But np.correlate(h, 1-h, mode='full') is NOT the autocorrelation of h*(1-h). It's the cross-correlation of h and (1-h).
   
   corr[k] = sum_i h[i] * (1-h[i + k - (n-1)])
   
   For k = n-1: corr[n-1] = sum_i h[i] * (1-h[i]) = 0 (for binary)
   
   For k = n-1+j (j > 0): corr[n-1+j] = sum_{i=0}^{n-1-j} h[i] * (1-h[i+j])
   This is: (number of i where h[i]=1 and h[i+j]=0) for i in valid range
   
   For k = n-1-j (j > 0): corr[n-1-j] = sum_{i=0}^{j-1+...} ... 
   
   Actually for the cross-correlation:
   corr[k] = sum_i h[i] * (1-h[i + k - (n-1)])
   
   Let m = k - (n-1). Then:
   corr[n-1+m] = sum_i h[i] * (1-h[i+m]) for valid i
   
   For m = 0: sum h[i]*(1-h[i]) = 0 for binary
   For m > 0: sum_{i=0}^{n-1-m} h[i]*(1-h[i+m]) = #{i: h[i]=1, h[i+m]=0} in range
   For m < 0: sum_{i=-m}^{n-1} h[i]*(1-h[i+m]) = #{i: h[i]=1, h[i+m]=0} in range
   
   So for binary h, the max of corr[k] * dx where k != n-1 gives the overlap.
   
   The maximum possible value: if h is all 1s in first half, 0s in second half:
   For m=1: sum h[i]*(1-h[i+1]) - at most 1 (the boundary)
   For m=n/2: sum_{i=0}^{n/2-1} h[i]*(1-h[i+n/2]) = n/2 (all 1s correlated with 0s)
   So corr[n-1+n/2] = n/2, giving C5 = (n/2) * 2/n = 1. That's the worst case.
   
   For a random binary h: E[h[i]*h[i+m]] ≈ 1/4, so E[corr[m]] ≈ (n-m)/4, giving C5 ≈ (n/4)*2/n ≈ 1/2.
   
   The optimal should be around 1/3 based on theory.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def make_swap_delta(h, i, j, n):
    """Compute correlation change for swap h[i]:1->0, h[j]:0->1. O(n)."""
    r = 2 * n - 1
    dr = np.zeros(r)
    # Effect of h[i] -> h[i]-1 (delta=-1)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    # Effect of h[j] -> h[j]+1 (delta=+1)
    lo3 = max(j, 0)
    hi3 = min(j + n - 1, r - 1)
    if lo3 <= hi3:
        ks = np.arange(lo3, hi3 + 1)
        dr[lo3:hi3+1] += (1.0 - h[j + n - 1 - ks])
    lo4 = max(n - 1 - j, 0)
    hi4 = min(2 * (n - 1) - j, r - 1)
    if lo4 <= hi4:
        ks = np.arange(lo4, hi4 + 1)
        dr[lo4:hi4+1] -= h[j + ks - (n - 1)]
    dr[n - 1] -= 2.0
    return dr

def multi_point_swap(h, n, rng, num_pairs=2):
    """Try swapping multiple pairs simultaneously."""
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    
    # Select random pairs
    idx_ones = rng.choice(len(ones), size=min(num_pairs, len(ones)), replace=False)
    idx_zeros = rng.choice(len(zeros), size=min(num_pairs, len(zeros)), replace=False)
    
    h_new = h.copy()
    for idx in idx_ones:
        h_new[ones[idx]] = 0
    for idx in idx_zeros:
        h_new[zeros[idx]] = 1
    
    new_corr = compute_corr(h_new)
    new_max = float(np.max(new_corr))
    return new_max

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            rng = np.random.RandomState(0)
            rng.shuffle(ones)
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            rng = np.random.RandomState(0)
            rng.shuffle(zeros)
            h[zeros[:target - n_ones]] = 1
    return h

def periodic_pattern(n, period, offset=0):
    """Create a periodic binary pattern."""
    h = np.zeros(n)
    for i in range(n):
        if (i + offset) % period < period // 2:
            h[i] = 1.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            np.random.RandomState(1).shuffle(ones)
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            np.random.RandomState(1).shuffle(zeros)
            h[zeros[:target - n_ones]] = 1
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _TGT,
             'jac': lambda x: np.ones(_N)}]
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / _BETA) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k=5, max_iters=200):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]
        
        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= d
        
        improved = False
        for step in [0.01, 0.02, 0.05, 0.01, 0.03, 0.08, 0.005, 0.04, 0.06]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

                    idx = np.where(h_start == 0)[0]
                    h_start[idx[:n_half - n_o]] = 1
        
        t_s = min(100, int(remaining() * 0.12))
        if t_s < 20:
            t_s = 20
        h_bs, c5_bs = sa_binary_local_search(h_start, n, time_limit=t_s, rng=np.random.RandomState(seed + 1000 + s_idx))
        if c5_bs < best_c5:
            print(f"Phase 2 start {s_idx}: C5={c5_bs:.6f}")
            best_c5 = c5_bs
            best_h = h_bs.copy()
            best_n = n
    
    # Phase 3: Try other resolutions
    resolutions = sorted(set([
        n, n + 32, n + 64, n + 128, n - 32, n - 64,
        256, 320, 448, 512, 640, 768
    ]))
    resolutions = [r for r in resolutions if 128 <= r <= 896 and r % 2 == 0 and r != n]
    resolutions.sort(key=lambda x: abs(x - n))

        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:

                idx = np.where(h_bin_t == 1)[0]
                np.random.RandomState(seed + n_t).shuffle(idx)
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                np.random.RandomState(seed + n_t).shuffle(idx)
                h_bin_t[idx[:nt_half - nt_ones]] = 1

        if t_bi < 20:
            t_bi = 20
        h_b3, c5_b3 = sa_binary_local_search(h_bin_t, n_t, time_limit=t_bi, rng=np.random.RandomState(seed + 2000 + n_t))
        if c5_b3 < best_c5:
            print(f"Phase 3 (n={n_t}): C5={c5_b3:.6f}")
            best_c5 = c5_b3
            best_h = h_b3.copy()
            best_n = n_t
    
    # Phase 4: Continuous refinement on best
    n = best_n
    h = best_h.copy()

        if remaining() < 15:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            print(f"Phase 4 (beta={beta}): C5={c5_new:.6f}")
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining() > 20:

        if c5_eq < best_c5:

            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
