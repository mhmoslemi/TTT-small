            if best_pair is not None:

                h_bin[i] = 0.0
                h_bin[j] = 1.0
                best_c5 = compute_c5(h_bin)
                best_h = h_bin.copy()
                improved = True
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            best_c5 = compute_c5(h_bin)
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def von_neumann(n):
    h = np.zeros(n)
    for i in range(n):
        v = i + 1
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def random_balanced(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def coordinate_refine(h, n, max_sweeps=20):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        imp = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0]
            if old > 0.01 and old < 0.99:
                cands.extend([0.25, 0.5, 0.75])
            best_cand = old
            best_v = best_c5
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                cv = compute_c5(h)
                if cv < best_v - 1e-13:
                    best_v = cv
                    best_cand = c
            h[i] = best_cand
            h = project_feasible(h, n)
            if best_v < best_c5:
                best_c5 = best_v
                imp = True
        if not imp:
            break
    return h, best_c5


def active_set_cycle(h, n, k_start=5, k_max=20, n_cycles=10, beta_start=500, beta_end=50000):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for cyc in range(n_cycles):
        if remaining_time() < 10:
            break
        corr = compute_corr(h)
        k = min(k_start + cyc * 2, k_max, 2 * n - 1)
        active_idx = np.argsort(-corr)[:k]
        beta = int(beta_start * (beta_end / beta_start) ** (cyc / max(1, n_cycles - 1)))
        beta = max(100, min(beta, 200000))
        h_new = run_lbfgs(h, n, beta, active_idx, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            h = h_new
        else:
            corr2 = compute_corr(h)
            active_idx2 = np.argsort(-corr2)[:k]
            h_eq, c5_eq = top_k_equip(h, n, [k], max_iters=100)
            if c5_eq < best_c5 - 1e-14:
                best_c5 = c5_eq
                h = h_eq
    return h, best_c5


def multi_res_opt(h_best, n_best, c5_best, target_ns, time_per_n):
    global _G_N, _G_BETA, _G_ACTIVE, _G_PEN
    for n_t in target_ns:
        if remaining_time() < 30:
            break
        if n_t == n_best:
            continue
        h_interp = interp_to(h_best, n_best, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        h_work = h_feas.copy()
        # Active set optimization
        h_work, c5_act = active_set_cycle(h_work, n_t, k_start=5, k_max=15,
                                          n_cycles=6, beta_start=300, beta_end=30000)
        if c5_act < c5_best - 1e-14:
            c5_best = c5_act
            h_best = h_work.copy()
            n_best = n_t
        # Equip
        h_eq, c5_eq = top_k_equip(h_best, n_best, [5, 10], max_iters=200)
        if c5_eq < c5_best - 1e-14:
            c5_best = c5_eq
            h_best = h_eq.copy()
        # Final SLSQP polish at this resolution
        if remaining_time() > 15:
            h_sl = run_slq(h_best, n_best, 50000, maxiter=300)
            c5_sl = compute_c5(h_sl)
            if c5_sl < c5_best - 1e-14:
                c5_best = c5_sl
                h_best = h_sl.copy()
    return h_best, n_best, c5_best


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _G_N, _G_BETA, _G_ACTIVE, _G_PEN
    _T0 = time.time()
    _BUDGET = budget_s
    _G_ACTIVE = None
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP warm-up at parent resolution
    h = best_h.copy()
    for beta in [200, 1000, 5000, 20000]:
        if remaining_time() < 15:
            break
        h_new = run_slq(h, best_n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Active-set L-BFGS-B optimization (key difference from previous)
    if remaining_time() > 20:
        h_act, c5_act = active_set_cycle(h, best_n, k_start=4, k_max=16,
                                         n_cycles=12, beta_start=200, beta_end=80000)
        if c5_act < best_c5 - 1e-14:
            best_c5 = c5_act
            best_h = h_act.copy()
            h = h_act.copy()

    # Phase 3: Top-k equalization
    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equip(h, best_n, [4, 8, 12], max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Multi-resolution exploration
    res_candidates = []
    for d in [-4, -2, 2, 4, 8, 16, 32, 64]:
        nn = best_n + d
        if 160 <= nn <= 900 and nn % 2 == 0:
            res_candidates.append(nn)
    for mult in [0.85, 1.1, 1.25, 1.5, 1.75, 2.0]:
        nn = int(best_n * mult)
        if 160 <= nn <= 900 and nn % 2 == 0:
            res_candidates.append(nn)
    # Also try power-of-2-ish resolutions
    for pn in [256, 320, 384, 512, 640, 768]:
        if 160 <= pn <= 900 and pn % 2 == 0:
            res_candidates.append(pn)
    res_candidates = sorted(set(res_candidates))
    res_candidates.sort(key=lambda x: abs(x - best_n))
    res_candidates = [x for x in res_candidates if x != best_n][:6]

    h_best2, n_best2, c5_best2 = multi_res_opt(best_h, best_n, best_c5,
                                               res_candidates, 120)
    if c5_best2 < best_c5 - 1e-14:
        best_c5 = c5_best2
        best_h = h_best2.copy()
        best_n = n_best2
        h = h_best2.copy()

    # Phase 5: Binary local search with pair swaps
    if remaining_time() > 25:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:n_half]] = 1.0
        sa_time = min(100, max(10, int(remaining_time() * 0.25)))
        h_b, c5_b = binary_pair_swap(h_bin, best_n, time_limit=sa_time, rng=rng)
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Refine binary result continuously
        h_b_c = h_b.copy()
        h_b_c, c5_b_c = active_set_cycle(h_b_c, best_n, k_start=5, k_max=12,
                                         n_cycles=5, beta_start=500, beta_end=50000)
        if c5_b_c < best_c5 - 1e-14:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()

    # Phase 6: Alternative starting points
    if remaining_time() > 40:
        alts = []
        alts.append(theta_morse(best_n))
        alts.append(von_neumann(best_n))
        alts.append(random_balanced(best_n, rng))
        # Shifted version
        alts.append(project_feasible(np.roll(best_h, 2), best_n))
        # Inverted
        alts.append(project_feasible(1.0 - best_h, best_n))

        for h_alt in alts:
            if remaining_time() < 25:
                break
            h_opt, c5_opt = active_set_cycle(h_alt, best_n, k_start=5, k_max=10,
                                             n_cycles=5, beta_start=500, beta_end=30000)
            h_opt, c5_opt = top_k_equip(h_opt, best_n, [5, 8], max_iters=150)
            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h = h_opt.copy()

    # Phase 7: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    if remaining_time() > 20:
        h_act, c5_act = active_set_cycle(h, n, k_start=3, k_max=10,
                                         n_cycles=8, beta_start=100, beta_end=100000)
        if c5_act < best_c5 - 1e-14:
            best_c5 = c5_act
            best_h = h_act.copy()
            h = h_act.copy()

    if remaining_time() > 12:
        h_eq, c5_eq = top_k_equip(h, n, [3, 5, 8], max_iters=300)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining_time() > 10:
        h_sl = run_slq(h, n, 100000, maxiter=400)
        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5 - 1e-14:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    if remaining_time() > 8:
        h_eq2, c5_eq2 = top_k_equip(best_h, n, [4, 7], max_iters=200)
        if c5_eq2 < best_c5 - 1e-14:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 8: Coordinate refinement
    if remaining_time() > 8:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=15)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Phase 9: One more L-BFGS-B pass with very high beta
    if remaining_time() > 10:
        corr_f = compute_corr(best_h)
        active_f = np.argsort(-corr_f)[:6]
        h_final = run_lbfgs(best_h, n, 100000, active_f, maxiter=300)
        c5_f = compute_c5(h_final)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
