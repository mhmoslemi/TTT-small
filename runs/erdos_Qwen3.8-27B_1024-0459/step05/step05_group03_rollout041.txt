```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum h = n/2} using bisection on Lagrange multiplier."""
    target = n / 2.0
    # Bisection on lambda: h[i] = clip(y[i] - lambda, 0, 1)
    lo = h.min() - target - 2.0
    hi = h.max() - target + 2.0
    # Actually, let's think about this more carefully.
    # We want sum clip(h[i] - lambda, 0, 1) = n/2
    # As lambda → -∞, sum → n
    # As lambda → +∞, sum → 0
    # So we need lambda such that sum clip(h[i] - lambda, 0, 1) = n/2
    
    def sum_at_lambda(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    
    # Find range
    if sum_at_lambda(lo) < target:
        lo -= 1.0
    while sum_at_lambda(lo) < target and lo > -100:
        lo -= 1.0
    if sum_at_lambda(hi) > target:
        hi += 1.0
    while sum_at_lambda(hi) > target and hi < 100:
        hi += 1.0
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = sum_at_lambda(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-15:
            break
    
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g

def smooth_obj_and_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    s = np.sum(ex)
    w = ex / s
    dx = 2.0 / len(h)
    obj = (m + np.log(s) / beta) * dx
    grad = corr_grad_weighted(h, w * dx)
    return obj, grad

def lbfgs_smooth(h, n, beta, maxiter=2000):
    """L-BFGS-B with projection for smooth objective."""
    h_cur = project_feasible(h.copy(), n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    def _obj(x):
        hp = project_feasible(x, n)
        return smooth_obj_and_grad(hp, beta)[0]
    
    def _grad(x):
        hp = project_feasible(x, n)
        return smooth_obj_and_grad(hp, beta)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj, h_cur, jac=_grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12, 'iprint': 0})
        h_opt = project_feasible(res.x, n)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    except Exception:
        pass
    
    return best_h, best_c5
```

```python
    h = project_feasible(h.copy(), n)
    y = h.copy()
    v = np.zeros(n)
```

Wait, the Lipschitz constant for the smooth objective is important for convergence. The smooth objective is:
f(h) = (max corr + log(sum exp(beta*(corr - max corr)))/beta) * dx

The Hessian of f is bounded by beta * (max eigenvalue of the correlation Jacobian squared) * dx.

For the correlation, the "Jacobian" is a matrix of shift operators, and its norm is bounded by about 2 (since each h[i] appears in at most 2 correlation terms with coefficient at most 1).

So L ≈ beta * 4 * dx = beta * 8/n.

For n=384, beta=10000: L ≈ 10000 * 8/384 ≈ 208.
Step size: lr = 1/L ≈ 0.005.

With Nesterov, the convergence rate is O(L/T²), so to get to accuracy ε, we need T ≈ sqrt(L/ε) iterations.

For ε = 0.001 (in terms of the objective), T ≈ sqrt(208/0.001) ≈ 456 iterations.

That's feasible within our time budget.

But the issue is that the smooth approximation with large beta is very close to the non-smooth max, and the gradient is dominated by the single worst lag. This makes the optimization slow because it only addresses one lag at a time.

```python
    row = np.zeros(2*n-1)
    # Part 1: (1-h[m+k-(n-1)]) for k where m+k-(n-1) in [0, n-1]
    # k in [n-1-m, 2n-2-m]

    if lo2 <= hi2:
```

        idx = np.arange(m - lo2 + (n - 1), m - hi2 + (n - 1) + 1)
        row[lo2:hi2 + 1] -= h[idx][::-1]
    return row
