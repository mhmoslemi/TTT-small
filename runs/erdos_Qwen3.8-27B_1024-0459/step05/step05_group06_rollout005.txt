                if t < 0.0 or t > 1.0:
                    continue
                f = np.max(base + a * t + b * t * t) * dx
                if f < best_f:
                    best_f = f
                    best_t = t

            margin = min(0.4, abs(best_t - t0) + 0.1)
            lo = max(0.0, best_t - margin)
            hi = min(1.0, best_t + margin)
            if hi - lo > 1e-12:
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                for _ in range(45):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                t_mid = (lo + hi) / 2.0
                f_mid = np.max(base + a * t_mid + b * t_mid * t_mid) * dx
                if f_mid < best_f:
                    best_f = f_mid
                    best_t = t_mid

            if abs(best_t - t0) > 1e-13:
                delta = best_t - t0
                best_i = best[i]
                best[i] = best_t
                best_corr += delta * a + (best_t * best_t - best_i * best_i) * b
                new_c5 = np.max(best_corr) * dx
                if new_c5 < best_c5 + 1e-14:
                    best_c5 = new_c5
                    improved_any = True
                else:
                    best[i] = best_i
                    best_corr -= delta * a + (best_t * best_t - best_i * best_i) * b

        if sweep % 20 == 19:
            best_corr = compute_corr(best)
            best_c5 = np.max(best_corr) * dx

        if not improved_any:
            break

    return best, best_c5

def pair_exchange(h, n, n_rounds=15):
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best = h.copy()
    best_corr = compute_corr(best)
    best_c5 = np.max(best_corr) * dx

    for rnd in range(n_rounds):
        if remaining_time() < 5:
            break
        orig_c5 = best_c5
        improved = False
        ones = np.where(best > 0.6)[0]
        zeros = np.where(best < 0.4)[0]
        if len(ones) == 0 or len(zeros) == 0:
            interior = np.where((best > 0.1) & (best < 0.9))[0]
            if len(interior) >= 2:
                ones = interior[:len(interior)//2]
                zeros = interior[len(interior)//2:]
            else:
                break

        for i_idx in range(len(ones)):
            if remaining_time() < 3:
                break
            ii = ones[i_idx]
            for j_idx in range(len(zeros)):
                jj = zeros[j_idx]
                if ii == jj:
                    continue
                delta = best[ii] - best[jj]
                if abs(delta) < 0.05:
                    continue
                best[ii] -= delta * 0.5
                best[jj] += delta * 0.5
                best[ii] = np.clip(best[ii], 0.0, 1.0)
                best[jj] = np.clip(best[jj], 0.0, 1.0)
                best_corr2 = compute_corr(best)
                c5_2 = np.max(best_corr2) * dx
                if c5_2 < best_c5 - 1e-10:
                    best_c5 = c5_2
                    best_corr = best_corr2
                    improved = True
                else:
                    best[ii] += delta * 0.5
                    best[jj] -= delta * 0.5
                    best_corr = best_corr2

        best = project_feasible(best, n)
        best_corr = compute_corr(best)
        best_c5 = np.max(best_corr) * dx
        if not improved:
            break

    return best, best_c5

def smooth_max_obj(x, n, beta, top_k=3):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[::-1]
    top = corr[idx[:top_k]]
    shifted = np.minimum(beta * (top - top[0]), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    smax = top[0] + np.log(wsum) / beta
    return smax * dx

def smooth_max_grad(x, n, beta, top_k=3):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = np.argsort(corr)[::-1]
    top_idx = idx[:top_k]
    top_vals = corr[top_idx]
    shifted = np.minimum(beta * (top_vals - top_vals[0]), 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        if wi < 1e-15:
            continue
        k = int(ki)
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += wi * (1.0 - x[:npairs])
                grad[:npairs] -= wi * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += wi * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= wi * x[:npairs]
    grad *= dx
    return grad

def slsqp_optimize(h, n, beta=50000, top_k=5, maxiter=300):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj(x, n, beta, top_k)

    def jac(x):
        return smooth_max_grad(x, n, beta, top_k)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    c5_before = compute_c5(h)
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def nesterov_subgrad(h, n, n_iter=500, top_k=5):
    dx = 2.0 / n
    x = h.copy()
    v = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(best)
    base_lr = 0.08

    for it in range(n_iter):
        if remaining_time() < 5:
            break
        lr = base_lr / (1.0 + it * 0.01)
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for ki in idx:
            k = int(ki)
            offset = k - (n - 1)
            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    g[offset:offset + npairs] += (1.0 - x[:npairs])
                    g[:npairs] -= x[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    g[:npairs] += (1.0 - x[aoff:aoff + npairs])
                    g[aoff:aoff + npairs] -= x[:npairs]
        g /= top_k
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        g /= gnorm
        v = 0.9 * v - lr * g
        x_new = x + v
        x_new = project_feasible(x_new, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = x_new.copy()
        x = x_new

    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    seq = np.array([0], dtype=float)
    while len(seq) < n:
        seq = np.concatenate([seq, 1.0 - seq])
    return seq[:n]

def initial_h_values_check():
    try:
        return np.array(initial_h_values, dtype=float)
    except (NameError, Exception):
        return None

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = initial_h_values_check()
    if parent is not None:
        parent = project_feasible(parent, len(parent))
        best_h = parent.copy()
        best_c5 = compute_c5(parent)
        best_n = len(parent)
    else:
        best_n = 640
        h0 = np.zeros(best_n)
        idx = rng.choice(best_n, best_n // 2, replace=False)
        h0[idx] = 1.0
        best_h = project_feasible(h0, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Polished coordinate descent at parent resolution
    corr = compute_corr(h)
    h, c5 = coord_descent(h, n, corr, n_sweeps=40)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Pair exchange for structure
    if remaining_time() > 100:
        h2, c5_2 = pair_exchange(h, n, n_rounds=8)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            h = h2.copy()
        # Quick coord descent after pair exchange
        corr2 = compute_corr(h)
        h3, c5_3 = coord_descent(h, n, corr2, n_sweeps=20)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
            h = h3.copy()

    # Phase 3: SLSQP polish at current resolution
    if remaining_time() > 80:
        h4 = slsqp_optimize(h, n, beta=50000, top_k=3, maxiter=250)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
            h = h4.copy()

    # Phase 4: Upsample to higher resolution
    target_n = max(1280, min(2048, best_n * 2))
    if target_n % 2 != 0:
        target_n += 1
    if remaining_time() > 200:
        h_hi = interp_to(best_h, best_n, target_n)
        h_hi = project_feasible(h_hi, target_n)
        c5_hi = compute_c5(h_hi)

        corr_hi = compute_corr(h_hi)
        h_hi, c5_hi = coord_descent(h_hi, target_n, corr_hi, n_sweeps=50)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = target_n

        if remaining_time() > 80:
            h_hi = slsqp_optimize(h_hi, target_n, beta=30000, top_k=4, maxiter=200)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = target_n

        if remaining_time() > 60:
            corr_hi = compute_corr(h_hi)
            h_hi, c5_hi = coord_descent(h_hi, target_n, corr_hi, n_sweeps=30)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = target_n

    # Phase 5: Try even higher resolution
    if remaining_time() > 150:
        target_n2 = min(3072, target_n * 2)
        if target_n2 % 2 != 0:
            target_n2 += 1
        if target_n2 > target_n:
            h_hi2 = interp_to(best_h, best_n, target_n2)
            h_hi2 = project_feasible(h_hi2, target_n2)
            c5_hi2 = compute_c5(h_hi2)
            corr_hi2 = compute_corr(h_hi2)
            h_hi2, c5_hi2 = coord_descent(h_hi2, target_n2, corr_hi2, n_sweeps=30)
            if c5_hi2 < best_c5:
                best_c5 = c5_hi2
                best_h = h_hi2.copy()
                best_n = target_n2
            if remaining_time() > 40:
                h_hi2 = slsqp_optimize(h_hi2, target_n2, beta=20000, top_k=3, maxiter=150)
                c5_hi2 = compute_c5(h_hi2)
                if c5_hi2 < best_c5:
                    best_c5 = c5_hi2
                    best_h = h_hi2.copy()
                    best_n = target_n2

    # Phase 6: Random restarts with different structures
    n_work = best_n
    if remaining_time() > 120:
        # Thue-Morse based
        tm_n = 1 << (n_work.bit_length() - 1)
        if tm_n > 128 and tm_n <= 4096:
            tm = thue_morse(tm_n)
            tm_c5 = compute_c5(tm)
            if tm_c5 < 0.45:
                corr_tm = compute_corr(tm)
                tm, tm_c5 = coord_descent(tm, tm_n, corr_tm, n_sweeps=30)
                if tm_c5 < best_c5:
                    best_c5 = tm_c5
                    best_h = tm.copy()
                    best_n = tm_n

        # Random structured starts
        for rst in range(8):
            if remaining_time() < 30:
                break
            h_r = np.zeros(n_work)
            pattern = rng.choice(2, size=n_work)
            h_r[pattern == 1] = 1.0
            # Ensure exactly n/2 ones
            n_ones = int(np.sum(h_r))
            if n_ones > n_work // 2:
                excess = rng.choice(np.where(h_r > 0.5)[0], n_ones - n_work // 2, replace=False)
                h_r[excess] = 0.0
            elif n_ones < n_work // 2:
                deficit = rng.choice(np.where(h_r < 0.5)[0], n_work // 2 - n_ones, replace=False)
                h_r[deficit] = 1.0
            h_r = project_feasible(h_r, n_work)
            c5_r = compute_c5(h_r)
            if c5_r < 0.42:
                corr_r = compute_corr(h_r)
                h_r, c5_r = coord_descent(h_r, n_work, corr_r, n_sweeps=20)
                h_r = slsqp_optimize(h_r, n_work, beta=30000, top_k=3, maxiter=150)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()

    # Phase 7: Nesterov subgradient at best resolution
    if remaining_time() > 40:
        h_ng, c5_ng = nesterov_subgrad(best_h, best_n, n_iter=300, top_k=5)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()

    # Phase 8: Final intensive coordinate descent
    if remaining_time() > 20:
        corr_final = compute_corr(best_h)
        h_final, c5_final = coord_descent(best_h, best_n, corr_final, n_sweeps=50)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 10:
        h_polish = slsqp_optimize(best_h, best_n, beta=200000, top_k=2, maxiter=200)
        c5_polish = compute_c5(h_polish)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish.copy()

    # Phase 10: One last coord descent
    if remaining_time() > 5:
        corr_last = compute_corr(best_h)
        h_last, c5_last = coord_descent(best_h, best_n, corr_last, n_sweeps=15)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    target_sum = best_n / 2.0
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
