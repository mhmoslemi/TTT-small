def active_set_step(h, n, k_active=5, step_size=0.01):
    """Top-k active set subgradient step"""
    corr = compute_corr(h)
    max_c = np.max(corr)
    # Find active set: correlations within 50% of max
    threshold = max_c - 0.5 * max_c  # or use a fixed offset
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > k_active:
        # Keep the top k
        sorted_idx = np.argsort(corr[active])[::-1]
        active = active[sorted_idx[:k_active]]
    
    # Compute gradient as sum of gradients of active correlations
    grad = np.zeros(n)
    for k in active:
        offset = int(k) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

def topk_gradient(h, n, k=5):
    """Compute gradient of the average of the top-k correlations."""
    corr = compute_corr(h)
