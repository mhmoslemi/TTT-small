    wsum = np.sum(exp_vals)
    val = (m + np.log(wsum) / beta) * (2.0 / n)
    return val

def smooth_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    ev = np.exp(shifted)
    ws = np.sum(ev)
    return (m + np.log(ws) / beta) * (2.0 / n)

def smooth_obj_grad(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    ev = np.exp(shifted)
    ws = np.sum(ev)
    weights = ev / ws
    dx = 2.0 / n
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return grad

def slsqp_opt(h, n, beta, maxiter=400):
    def obj(x):
        return smooth_obj(x, n, beta)
    def objg(x):
        return smooth_obj_grad(x, n, beta)
    def eqf(x):
        return np.sum(x) - n / 2.0
    def eqj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, h)

            j //= 2
        h[i] = 1.0 if v > 0.5 else 0.0

            break

            idx = np.arange(start, start + block_size)
            old_block = best[idx].copy()
            f_old = np.max(best_corr)

            # Compute combined A matrix for all coordinates in block
            A_mats = []
            for i in idx:
                i_plus = i + offsets
                i_minus = i - offsets
                vp = (i_plus >= 0) & (i_plus < n)
                vm = (i_minus >= 0) & (i_minus < n)
                A_i = np.zeros(ncorr)

                if np.any(vp):
                    A_i[vp] -= best[i_plus[vp]]
                A_i[n - 1] = 0.0
                A_mats.append(A_i)

            base = best_corr.copy()
            for i, A_i in zip(idx, A_mats):
                base -= A_i * old_block[i - start]
            base[n - 1] = best_corr[n - 1]
            for j, i in enumerate(idx):
                base[n - 1] -= old_block[j] * (1.0 - old_block[j])

            best_block = old_block.copy()
            best_f = f_old

            # Try flipping each element
            for j in range(block_size):
                trial = old_block.copy()
                trial[j] = 1.0 - old_block[j] if abs(old_block[j] - 0.5) > 0.1 else 0.5
                new_corr = base.copy()
                for k2, i in enumerate(idx):
                    new_corr += A_mats[k2] * (trial[k2] - old_block[k2])
                new_corr[n - 1] = base[n - 1]
                for k2 in range(block_size):
                    new_corr[n - 1] += trial[k2] * (1.0 - trial[k2])

    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Coordinate descent on parent at original resolution
    h = best_h.copy()
    h, c5 = coord_descent(h, n, max_sweeps=200)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP at original resolution
    h = best_h.copy()
    for beta in [500, 3000, 15000]:
        if remaining_time() < 40:
            break
        h_new = slsqp_opt(h, n, beta, 350)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 3: More coordinate descent
    if remaining_time() > 60:
        h = best_h.copy()
        h, c5 = coord_descent(h, n, max_sweeps=150)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Local perturbations + CD
    if remaining_time() > 120:
        for p in range(15):
            if remaining_time() < 25:
                break
            h_new, c5_new = local_perturb_cd(best_h, n, rng, n_pert=15 + p * 3, n_sweeps=40)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Interpolate to n=1024 and optimize
    if remaining_time() > 200:
        target_n = 1024
        h_high = interp_to(best_h, best_n, target_n)
        h_high = project_feasible(h_high, target_n)

        h_new, c5_new = coord_descent(h_high, target_n, max_sweeps=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = target_n

        if remaining_time() > 120:
            h_new, c5_new = coord_descent(best_h, target_n, max_sweeps=150)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = target_n

        if remaining_time() > 80:
            h_slq = slsqp_opt(best_h, target_n, 2000, 300)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()
                best_n = target_n
                if remaining_time() > 40:
                    h_new, c5_new = coord_descent(best_h, target_n, max_sweeps=100)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = target_n

    # Phase 6: Try Thue-Morse at n=1024 as independent start
    if remaining_time() > 200:
        n_tm = 1024
        h_tm = thue_morse_init(n_tm)
        h_new, c5_new = coord_descent(h_tm, n_tm, max_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_tm
        elif remaining_time() > 100:
            # Combine: use TM where it's better, parent where parent is better
            h_parent_at_nm = interp_to(best_h, best_n, n_tm)
            h_parent_at_nm = project_feasible(h_parent_at_nm, n_tm)
            corr_tm = compute_corr(h_new)
            corr_par = compute_corr(h_parent_at_nm)
            # Where TM has lower local correlation contribution, use TM
            # Simple: just do CD from the better of the two
            if c5_new < compute_c5(h_parent_at_nm):
                pass  # already using h_new
            else:
                h_combined = h_parent_at_nm
                h_new2, c5_new2 = coord_descent(h_combined, n_tm, max_sweeps=150)
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
                    best_h = h_new2.copy()
                    best_n = n_tm

    # Phase 7: Try n=1280
    if remaining_time() > 150:
        target_n2 = 1280
        h_high2 = interp_to(best_h, best_n, target_n2)
        h_high2 = project_feasible(h_high2, target_n2)
        h_new, c5_new = coord_descent(h_high2, target_n2, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = target_n2
        if remaining_time() > 60:
            h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = target_n2

    # Phase 8: Final local perturbations at best resolution
    if remaining_time() > 60:
        n_cur = best_n
        for p in range(10):
            if remaining_time() < 15:
                break
            h_new, c5_new = local_perturb_cd(best_h, n_cur, rng, n_pert=12, n_sweeps=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 9: Block coordinate descent
    if remaining_time() > 40:
        h_new, c5_new = block_coord_descent(best_h, best_n, block_size=4, max_sweeps=30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 25:
        n_cur = best_n
        h_slq = slsqp_opt(best_h, n_cur, 50000, 300)
        c5_slq = compute_c5(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()

    # Phase 11: Final CD
    if remaining_time() > 10:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
