                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        # Strategy 2: Random swaps for exploration
        if best_swap is None or stale > 50:
            for _ in range(10):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = float(np.max(compute_corr(h_bin))) * dx
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is None:
            # Try block moves: shift a run of consecutive 1s
            found_block = False
            for _ in range(5):
                i0 = int(ones[rng.randint(len(ones))])
                block = [i0]
                # Extend block
                while len(block) < 8:
                    if block[-1] + 1 < n and h_bin[block[-1] + 1] == 1:
                        block.append(block[-1] + 1)
                    elif block[0] > 0 and h_bin[block[0] - 1] == 1:
                        block.insert(0, block[0] - 1)
                    else:
                        break
                if len(block) < 2:
                    continue
                # Try shifting block right by 1
                target = block[-1] + 1
                if target < n and h_bin[target] == 0:
                    h_bin[block[0]] = 0
                    h_bin[target] = 1
                    nc = float(np.max(compute_corr(h_bin))) * dx
                    delta = nc - cur_c5
                    if delta < best_delta:
                        h_bin = h_bin.copy()
                        best_delta = delta
                        best_swap = ('block', block, target)
                        found_block = True
                    else:
                        h_bin[block[0]] = 1
                        h_bin[target] = 0
                # Try shifting block left by 1
                target_l = block[0] - 1
                if target_l >= 0 and h_bin[target_l] == 0:
                    h_bin[block[-1]] = 0
                    h_bin[target_l] = 1
                    nc = float(np.max(compute_corr(h_bin))) * dx
                    delta = nc - cur_c5
                    if delta < best_delta:
                        h_bin = h_bin.copy()
                        best_delta = delta
                        best_swap = ('block', block, target_l)
                        found_block = True
                    else:
                        h_bin[block[-1]] = 1
                        h_bin[target_l] = 0
            if not found_block and best_swap is None:
                break
            if best_swap is not None and best_swap[0] == 'block':
                _, blk, tgt = best_swap
                # Already applied above
                if h_bin.sum() != n_half:
                    best_h = h_bin.copy()
                    best_c5 = cur_c5
                cur_c5 = float(np.max(compute_corr(h_bin))) * nc
                continue

        if best_swap is None:
            break

        if isinstance(best_swap, tuple) and len(best_swap) == 2:
            i, j = best_swap
            h_bin[i] = 0
            h_bin[j] = 1
            cur_c5 = float(np.max(compute_corr(h_bin))) * dx
        else:
            cur_c5 = float(np.max(compute_corr(h_bin))) * dx

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()
            stale = 0

    return best_h, best_c5


def nesterov_gradient(h, n, top_k=5, max_iters=6000, step_init=0.025):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init
    stale = 0

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 1500:
                break

        idx = np.argsort(-corr)[:top_k]
        c_top = corr[idx]
        m = c_top[0]
        sh = 50.0 * (c_top - m)
        sh = np.clip(sh, -50.0, 0.0)
        w = np.exp(sh)
        w /= np.sum(w)

        g = np.zeros(n)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)
        g *= dx
        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.015, 0.12)
            else:
                step *= 0.9
        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.7

        if it % 800 == 0 and it > 0:
            step *= 0.9
            momentum = min(momentum + 0.003, 0.94)

    return best_h, best_c5


def top_equalize(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    stale = 0

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5_cur = float(np.max(corr)) * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            stale = 0
        else:
            stale += 1
            if stale > 100:
                break

        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        if len(idx) > 1:
            c_min = corr[idx[-1]]
        else:
            c_min = c_max
        if c_max - c_min < 1e-10:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_max
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn

        improved = False
        for scale in [3, 1, 2, 0.5, 5, 1.5, 0.1, 4, 0.2, 0.8]:
            alpha = 0.015 * scale
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for scale in [2, 1, 3, 0.5, 4]:
                alpha = 0.01 * scale
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def refine_binary(h_bin, n, budget_check=True):
    """Refine a binary solution: continuous opt then binary search again."""
    h_ref = h_bin.astype(float).copy()
    for beta in [2000, 8000, 20000]:
        if budget_check and remaining() < 5:
            break
        h_ref = lbfgs_pass(h_ref, n, beta, top_k=5, maxiter=400)
    c5_ref = compute_c5(h_ref)

    # Binary polish
    h_b2 = (h_ref > 0.5).astype(float)
    h_bl, c5_bl = binary_local_search(h_b2, n, max_iters=2000, top_k=4, patience=150)
    c5_bl_ref = compute_c5(h_bl)

    # Final continuous from binary
    h_final = h_bl.astype(float).copy()
    if remaining() > 8:
        for beta in [5000, 15000, 40000]:
            if remaining() < 5:
                break
            h_final = lbfgs_pass(h_final, n, beta, top_k=4, maxiter=300)
    c5_final = compute_c5(h_final)

    # Pick best
    best_c = min(c5_ref, c5_bl_ref, c5_final)
    if c5_final <= best_c:
        return h_final, c5_final
    elif c5_bl_ref <= best_c:
        return h_bl, c5_bl_ref
    else:
        return h_ref, c5_ref


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick L-BFGS at parent resolution
    for beta in [200, 1000, 5000]:
        if remaining() < 10:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Nesterov gradient
    if remaining() > 30:
        h_pg, c5_pg = nesterov_gradient(h, n, top_k=5, max_iters=5000, step_init=0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 3: Top equalize
    if remaining() > 15:
        h_eq, c5_eq = top_equalize(h, n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Binary local search (main workhorse)
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_bl, c5_bl = binary_local_search(h_bin, best_n, max_iters=8000, top_k=4, patience=300)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

        # Refine continuously
        h_ref, c5_ref = refine_binary(h_bl, best_n)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 5: More L-BFGS with high beta
    for beta in [10000, 30000, 60000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(best_h, best_n, beta, top_k=4, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Binary search again after continuous refinement
    if remaining() > 30:
        h_bin2 = (best_h > 0.5).astype(float)
        h_bl2, c5_bl2 = binary_local_search(h_bin2, best_n, max_iters=6000, top_k=5, patience=250)
        if c5_bl2 < best_c5:
            best_c5 = c5_bl2
            best_h = h_bl2.copy()
        h_ref2, c5_ref2 = refine_binary(h_bl2, best_n)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2.copy()

    # Phase 7: Try nearby resolutions
    n_candidates = []
    for d in [-4, -2, 2, 4, 6, 8, -6, 10, -8, 12]:
        nn = best_n + d
        if 128 <= nn <= 768 and nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))
    n_candidates = [nn for nn in n_candidates if nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:6]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # Quick continuous opt
        for beta in [1000, 5000, 15000]:
            if remaining() < 10:
                break
            h_feas = lbfgs_pass(h_feas, n_t, beta, top_k=5, maxiter=300)

        # Binary search at this resolution
        h_b3 = (h_feas > 0.5).astype(float)
        h_bl3, c5_bl3 = binary_local_search(h_b3, n_t, max_iters=3000, top_k=4, patience=200)
        h_ref3, c5_ref3 = refine_binary(h_bl3, n_t)

        c5_cand = min(compute_c5(h_feas), c5_bl3, c5_ref3)
        if c5_ref3 <= c5_cand:
            h_cand = h_ref3
        elif c5_bl3 <= c5_cand:
            h_cand = h_bl3
        else:
            h_cand = h_feas

        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    if remaining() > 20:
        h_pg3, c5_pg3 = nesterov_gradient(h, n, top_k=4, max_iters=4000, step_init=0.015)
        if c5_pg3 < best_c5:
            best_c5 = c5_pg3
            best_h = h_pg3.copy()
            h = h_pg3.copy()

    if remaining() > 15:
        h_eq3, c5_eq3 = top_equalize(h, n, top_k=6, max_iters=250)
        if c5_eq3 < best_c5:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()
            h = h_eq3.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_local_search(h_bin_f, best_n, max_iters=5000, top_k=5, patience=300)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        h_rf, c5_rf = refine_binary(h_bf, best_n)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()

    # Final L-BFGS polish
    if remaining() > 10:
        for beta in [20000, 50000]:
            if remaining() < 5:
                break
            h_new = lbfgs_pass(best_h, best_n, beta, top_k=4, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        # Re-optimize if projection changed things
        best_h = proj(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
