import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def to_binary(h, n):
    """Round h to binary with exactly n/2 ones."""
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def sa_binary(h, n, max_iters, T_start, T_end, rng):
    """Simulated annealing on binary pattern."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        # Pick random swap
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        
        # Perform swap
        h[a] = 0.0
        h[b] = 1.0
        ones[ia] = b
        zeros[ib] = a
        
        c5_new = compute_c5(h)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        elif c5_new > best_c5:
            delta = c5_new - best_c5
            if rng.random() > np.exp(-delta / (T + 1e-10)):
                # Reject: undo
                h[a] = 1.0
                h[b] = 0.0
                ones[ia] = a
                zeros[ib] = b
    return best_h, best_c5

def local_search_binary(h, n, max_rounds=50):
    """Try all swaps, pick the best improvement each round."""
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for rnd in range(max_rounds):
        if remaining() < 10:
            break
        best_improve = -1
        best_pair = None
        corr = np.correlate(h, 1-h, mode='full')
        c_max = np.max(corr)
        
        # Try each 1→0 position
        for ia in range(len(ones)):
            a = ones[ia]
            h[a] = 0.0
            # Try each 0→1 position
            for ib in range(len(zeros)):
                b = zeros[ib]
                h[b] = 1.0
                c5_new = np.max(np.correlate(h, 1-h, mode='full'))
                if c5_new < c_max + 1e-15:
                    imp = c_max - c5_new
                    if imp > best_improve:
                        best_improve = imp
                        best_pair = (ia, ib)
                h[b] = 0.0
            h[a] = 1.0
        
        if best_pair is None:
            break
        ia, ib = best_pair
        a = ones[ia]
        b = zeros[ib]
        h[a] = 0.0
        h[b] = 1.0
        ones = np.concatenate([ones[:ia], ones[ia+1:], [b]])
        zeros = np.concatenate([zeros[:ib], zeros[ib+1:], [a]])
        best_c5 = compute_c5(h)
    
    return h, best_c5

Actually, let me think about this more carefully. For a swap (a: 1→0, b: 0→1), the change in corr at lag k is:

For positive lag k (1 to n-1), corr_pos[k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]):
- Position a contributes: if a ≤ n-1-k, old = 1*(1-h[a+k]), new = 0. Δ -= (1-h[a+k])
- Position a as "target": if a-k ≥ 0, old = h[a-k]*(1-1) = 0, new = h[a-k]*(1-0) = h[a-k]. Δ += h[a-k]
- Position b contributes: if b ≤ n-1-k, old = 0, new = 1*(1-h[b+k]). Δ += (1-h[b+k])
- Position b as "target": if b-k ≥ 0, old = h[b-k]*(1-0) = h[b-k], new = h[b-k]*(1-1) = 0. Δ -= h[b-k]

So Δ_corr_pos[k] = -(1-h[a+k]) + h[a-k] + (1-h[b+k]) - h[b-k]
(terms included only when indices are valid)

For negative lag -k (stored at index 2n-2-k in the full correlation):
This is corr_neg[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]):
- Position a as "source" (h[i+k] = h[a], so i = a-k): if a ≥ k, old = 1*(1-h[a-k]), new = 0. Δ -= (1-h[a-k])
- Position a as "target" (h[i] = h[a], so i = a): if a ≤ n-1-k, old = h[a+k]*(1-1) = 0, new = h[a+k]*(1-0) = h[a+k]. Δ += h[a+k]
- Position b as "source" (i = b-k): if b ≥ k, old = 0, new = 1*(1-h[b-k]). Δ += (1-h[b-k])
- Position b as "target" (i = b): if b ≤ n-1-k, old = h[b+k]*(1-0) = h[b+k], new = 0. Δ -= h[b+k]

So Δ_corr_neg[k] = -(1-h[a-k]) + h[a+k] + (1-h[b-k]) - h[b+k]

Total delta at index in full correlation:
- Index n-1+k (positive lag k): Δ = -(1-h[a+k]) + h[a-k] + (1-h[b+k]) - h[b-k]
- Index n-1-k (negative lag -k, i.e., lag = -k): Δ = -(1-h[a-k]) + h[a+k] + (1-h[b-k]) - h[b+k]

(Where I use h[a+k] only if a+k < n, h[a-k] only if a-k ≥ 0, etc.)

This is O(n) per swap to compute the full delta vector. Then I can find the new max in O(n) as well.
