import time

_BETA_G = 1000.0

def compute_corr(h):

def corr_grad_vec(h, k):
    n = len(h)
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        m = n - off
        g[off:] += (1.0 - h[:m])
        g[:m] -= h[off:]
    else:
        a = -off
        m = n - a
        g[:m] += (1.0 - h[a:])
        g[a:] -= h[:m]
    return g


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _sm_fun(x):

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA_G
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_jac(x):
    n = _N_G
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA_G
    z = np.clip(beta * (corr - m), -50.0, 50.0)

    S = np.sum(ez)

    _BETA_G = beta

                       bounds=bounds, constraints=cons,

        else:

        h_final, c5_f = multi_beta_slsqp(h_final, n_final, [500000, 2000000, 10000000], maxiter_each=1000)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_final.copy()
            h_final = h_final.copy()

    if remaining_time() > 15:
        h_final, c5_f = active_equip(h_final, n_final, top_k=3, max_iters=500)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_final.copy()
        h_final, c5_f = active_equip(h_final, n_final, top_k=5, maxiter=500)
        if c5_f < best_c5 - 1e-14:
            best_c5 = c5_f
            best_h = h_final.copy()

    # Phase 10: Coordinate descent
    if remaining_time() > 15:
        h_cd, c5_cd = coord_descent(best_h, n_final, max_sweeps=25)
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 11: Final SLSQP
    if remaining_time() > 10:
        h_last = slsqp_opt(best_h, n_final, 10000000, maxiter=1000, ftol=1e-18)
        c5_last = compute_c5(h_last)
        if c5_last < best_c5 - 1e-14:
            best_c5 = c5_last
            best_h = h_last.copy()

    # Phase 12: Final active-set polish
    if remaining_time() > 8:
        h_eq, c5_eq = active_equip(best_h, n_final, top_k=4, max_iters=300)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n_final)
