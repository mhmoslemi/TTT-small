            h_trial = slsqp_optimize(h_trial, n_struct, 5000, maxiter=200)
            c5_trial = compute_c5(h_trial)
            
            if remaining() > 10:
                h_trial, c5_trial = corr_aware_hill_climb(h_trial, n_struct, max_iters=500)
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                best_n = n_struct
    
    # === Phase 4: Intensive optimization at best solution ===
    n = best_n
    h = best_h.copy()
    
    # Very high beta SLSQP
    for beta in [50000, 200000, 1000000, 5000000]:
        if remaining() < 8:
            break
        maxit = min(500, int(remaining() // 8))
        if maxit < 15:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=maxit)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    # Projected gradient steps
    if remaining() > 20:
        for _ in range(100):
            if remaining() < 5:
                break
            h = proj_grad_step(h, n, 50000, max_line=20)
            c5_chk = compute_c5(h)
            if c5_chk < best_c5:
                best_c5 = c5_chk
                best_h = h.copy()
    
    # Correlation-aware hill climb (intensive)
    if remaining() > 20:
        h_cc, c5_cc = corr_aware_hill_climb(best_h, best_n, max_iters=3000)
        if c5_cc < best_c5:
            best_c5 = c5_cc
            best_h = h_cc.copy()
    
    # === Phase 5: Binary refinement ===
    if remaining() > 20:
        h_bin = round_to_binary(best_h, best_n)
        h_bin, c5_bin = binary_ls(h_bin, best_n, max_rounds=40)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        
        # Multi-swap
        if remaining() > 15:
            h_ms, c5_ms = binary_ls_multi_swap(h_bin, best_n, max_rounds=15, swap_count=2)
            if c5_ms < best_c5:
                best_c5 = c5_ms
                best_h = h_ms.copy()
        
        # SLSQP from best binary
        if remaining() > 15:
            h_from_bin = slsqp_optimize(h_bin, best_n, 100000, maxiter=400)
            c5_fb = compute_c5(h_from_bin)
            if c5_fb < best_c5:
                best_c5 = c5_fb
                best_h = h_from_bin.copy()
    
    # === Phase 6: Final micro-optimization ===
    if remaining() > 10:
        # Try very small perturbations around best solution
        for _ in range(50):
            if remaining() < 3:
                break
            pert_scale = 0.005 / (1 + _ * 0.1)
            pert = rng.randn(best_n) * pert_scale
            pert -= np.mean(pert)
            h_try = project_feasible(best_h + pert, best_n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # === Final validation ===
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    best_c5 = min(best_c5, final_c5)
    
    return (best_h, best_c5, best_n)
