Alternative: For each a, compute base_a = corr + down[a] (vector of length 2n-1). Then find the b that minimizes max(base_a + up[b]). 

For a fixed a, I want to minimize max_k (base_a[k] + up[b, k]) over b in zeros.
This is equivalent to: for each b, compute max_k (base_a[k] + up[b, k]), then pick the b with the smallest such max.

I can compute this for all b simultaneously:
- up is a matrix of shape (n, 2n-1), where up[b, k] is the change in corr[k] from flipping position b from 0 to 1

def compute_down_up(h, n):
    """Compute the effect of each position flipping on all correlations."""
    ncorr = 2 * n - 1
    # down[p, k]: change in corr[k] if h[p] goes 1->0
    # up[p, k]: change in corr[k] if h[p] goes 0->1 (= -down[p,k] when h[p]=0)
    
    # For np.correlate(h, 1-h, 'full'):
    # corr[k] where k = 0..2n-2
    # offset = k - (n-1)
    # If offset >= 0: corr = sum_{i=0}^{n-1-offset} h[i+offset]*(1-h[i])
    # If offset < 0: corr = sum_{i=0}^{n-1+offset} h[i]*(1-h[i-offset]) (offset negative, so -offset positive)
    #   Actually: corr = sum_{i=0}^{n-1+|offset|} h[i]*(1-h[i+|offset|])
    
    # For position p going 1->0:
    # In the "h" part: h[p] appears when i+offset=p, i.e., i=p-offset
    #   valid: 0<=i<n and 0<=p<n (always true for p)
    #   so need 0<=p-offset<n
    # In the "(1-h)" part: (1-h[p]) appears when i=p
    #   need 0<=p+offset<n
    
    down = np.zeros((n, ncorr))
    offsets = np.arange(ncorr) - (n - 1)  # (ncorr,)
    
    for p in range(n):
        # h[p] * (1-h[p-offset]) term (p is in the "h" position)
        valid_h = (p - offsets >= 0) & (p - offsets < n)
        # When valid, the term is 1*(1-h[p-offset]), after flip: 0
        # change: -(1-h[p-offset])
        i_val = np.clip(p - offsets, 0, n-1)
        down[p, valid_h] = -(1.0 - h[i_val[valid_h]])
        
        # h[p+offset] * (1-h[p]) term (p is in the "(1-h)" position)
        valid_1mh = (p + offsets >= 0) & (p + offsets < n)
        # When valid, the term is h[p+offset]*0 = 0, after flip: h[p+offset]*1
        # change: +h[p+offset]
        i_val2 = np.clip(p + offsets, 0, n-1)
        down[p, valid_1mh] += h[i_val2[valid_1mh]]
    
    up = -down  # up[p,k] = -down[p,k] (flipping 0->1 is opposite of 1->0)
    # But wait: up[p,k] is the change when h[p] goes 0->1.
    # If h[p]=0, going to 1: 
    #   In "h" position: was 0*(1-h[p-offset])=0, now 1*(1-h[p-offset]). Change: +(1-h[p-offset])
    #   In "(1-h)" position: was h[p+offset]*(1-0)=h[p+offset], now h[p+offset]*0. Change: -h[p+offset]
    # So up[p,k] = (1-h[p-offset]) for valid_h - h[p+offset] for valid_1mh
    # Which is indeed -down[p,k]
    
    return down, up
