                nc, new_corr_arr = swap_eval_fast(int(p), int(q), corr)
                if nc < cur_max:
                    delta = nc - cur_max
                    if delta < best_delta:
                        best_delta = delta
                        best_p = int(p)
                        best_q = int(q)
                        best_new_corr = new_corr_arr
                        accept_swap = True
        
        # 2-2 swaps: random pairs
        if not accept_swap and remaining() > 10:
            n_2sw = 15
            for _ in range(n_2sw):
                p1 = int(ones_list[rng.randint(len(ones_list))])
                p2 = int(ones_list[rng.randint(len(ones_list))])
                while p2 == p1 and len(ones_list) > 1:
                    p2 = int(ones_list[rng.randint(len(ones_list))])
                q1 = int(zeros_list[rng.randint(len(zeros_list))])
                q2 = int(zeros_list[rng.randint(len(zeros_list))])
                while q2 == q1 and len(zeros_list) > 1:
                    q2 = int(zeros_list[rng.randint(len(zeros_list))])
                while q1 == p1 or q1 == p2 or q2 == p1 or q2 == p2:
                    q1 = int(zeros_list[rng.randint(len(zeros_list))])
                    q2 = int(zeros_list[rng.randint(len(zeros_list))])
                    while q2 == q1:
                        q2 = int(zeros_list[rng.randint(len(zeros_list))])
                
                h[p1] = 0; h[p2] = 0; h[q1] = 1; h[q2] = 1
                nc = compute_c5(h)
                h[p1] = 1; h[p2] = 1; h[q1] = 0; h[q2] = 0
                delta2 = nc - cur_c5
                if delta2 < best_delta:
                    best_delta = delta2
                    best_p = p1  # encode 2-2 swap
                    best_q = p2
                    best_new_corr = None
                    # Store the 2-2 swap
                    best_p = (-p1, -p2, -q1, -q2)
                    accept_swap = True
        
        # Also try random 1-1 swaps for exploration
        if not accept_swap and it % 20 == 0:
            for _ in range(8):
                p = int(ones_list[rng.randint(len(ones_list))])
                q = int(zeros_list[rng.randint(len(zeros_list))])
                nc, new_corr_arr = swap_eval_fast(p, q, corr)
                if nc < cur_max:
                    delta = nc - cur_max
                    if delta < best_delta:
                        best_delta = delta
                        best_p = p
                        best_q = q
                        best_new_corr = new_corr_arr
                        accept_swap = True
        
        # Simulated annealing: accept worse swaps occasionally
        if not accept_swap:
            for _ in range(3):
                p = int(ones_list[rng.randint(len(ones_list))])
                q = int(zeros_list[rng.randint(len(zeros_list))])
                nc, new_corr_arr = swap_eval_fast(p, q, corr)
                delta = nc - cur_max
                if delta < 0 or rng.rand() < np.exp(-abs(delta) / (temperature * max(dx, 1e-6))):
                    if delta < best_delta:
                        best_delta = delta
                        best_p = p
                        best_q = q
                        best_new_corr = new_corr_arr
                        accept_swap = True
        
        if accept_swap and best_p is not None:
            if isinstance(best_p, tuple):
                # 2-2 swap
                pa, pb, qa, qb = best_p
                h[pa] = 0; h[pb] = 0; h[qa] = 1; h[qb] = 1
            else:
                h[best_p] = 0
                h[best_q] = 1
            temperature *= cool_rate
            if temperature < min_temp:
                temperature = min_temp
        elif it % 100 == 0 and it > 0:
            # Restart exploration: random perturbation
            if remaining() < 10:
                break
            n_pert = rng.randint(3, 10)
            for _ in range(n_pert):
                p = int(ones_list[rng.randint(len(ones_list))])
                q = int(zeros_list[rng.randint(len(zeros_list))])
                h[p] = 0; h[q] = 1
            temperature = max(temperature * 5.0, 0.005)
        
    return best_h, best_c5


def fourier_init(n, n_modes=3, seed=0):
    """Create a diverse initialization using Fourier modes."""
    rng = np.random.RandomState(seed)
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = np.zeros(n)
    for fm in range(n_modes):
        freq = rng.uniform(0.5, 8.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        h += amp * np.sin(2 * np.pi * freq * x + phase)
    h += 0.5
    h = proj(h, n)
    # Round toward binary
    thresh = 0.5 + rng.uniform(-0.1, 0.1)
    h_bin = (h > thresh).astype(float)
    # Fix sum
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        idx_ones = np.where(h_bin == 1)[0]
        # Remove from positions where h is closest to threshold
        h_vals_ones = h[idx_ones]
        order = np.argsort(h_vals_ones)[:n_ones - target]
        for idx in order:
            h_bin[idx_ones[idx]] = 0
    elif n_ones < target:
        idx_zeros = np.where(h_bin == 0)[0]
        h_vals_zeros = h[idx_zeros]
        order = np.argsort(-h_vals_zeros)[:target - n_ones]
        for idx in order:
            h_bin[idx_zeros[idx]] = 1
    return h_bin


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK
    _T0 = time.time()
    _BUDGET = budget_s
    _TOPK = 5
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = fourier_init(parent_n, n_modes=4, seed=1)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick L-BFGS refinement at parent resolution
    for beta in [500, 2000, 8000, 20000]:
        if remaining() < 15:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Nesterov gradient
    if remaining() > 30:
        h_pg, c5_pg = nesterov_hc(h, n, top_k=5, max_iters=3000, step_init=0.04)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 3: Binary local search (main workhorse)
    h_bin = (best_h > 0.5).astype(float)
    # Ensure correct count
    n_ones = int(np.sum(h_bin))
    target = best_n // 2
    if n_ones != target:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(best_n)
        h_bin[sort_idx[:target]] = 1.0

    h_bs, c5_bs = binary_search(h_bin, best_n, max_iters=15000, rng=rng)
    if c5_bs < best_c5:
        best_c5 = c5_bs
        best_h = h_bs.copy()

    # Phase 4: Continuous refinement from best binary
    h_ref = best_h.copy()
    for beta in [5000, 15000, 40000]:
        if remaining() < 10:
            break
        h_ref = lbfgs_pass(h_ref, best_n, beta, top_k=4, maxiter=300)
    c5_ref = compute_c5(h_ref)
    if c5_ref < best_c5:
        best_c5 = c5_ref
        best_h = h_ref.copy()

    # Phase 5: Try alternate resolution n=480
    n_alt = 480
    if remaining() > 60:
        h_alt = interp_to(best_h, best_n, n_alt)
        h_alt = proj(h_alt, n_alt)
        # Quick L-BFGS
        for beta in [1000, 5000, 15000]:
            if remaining() < 10:
                break
            h_alt = lbfgs_pass(h_alt, n_alt, beta, top_k=5, maxiter=300)
        # Nesterov
        if remaining() > 20:
            h_alt, c5_alt_pg = nesterov_hc(h_alt, n_alt, top_k=5, max_iters=2000, step_init=0.04)
        # Binary search
        h_alt_bin = (h_alt > 0.5).astype(float)
        n_ones_alt = int(np.sum(h_alt_bin))
        if n_ones_alt != n_alt // 2:
            sort_idx = np.argsort(-h_alt_bin)
            h_alt_bin = np.zeros(n_alt)
            h_alt_bin[sort_idx[:n_alt // 2]] = 1.0
        if remaining() > 30:
            h_alt_bin, c5_alt_bs = binary_search(h_alt_bin, n_alt, max_iters=10000, rng=rng)
            if c5_alt_bs < best_c5:
                best_c5 = c5_alt_bs
                best_h = h_alt_bin.copy()
                best_n = n_alt
            # Continuous refinement
            if remaining() > 15:
                h_alt_ref = h_alt_bin.copy()
                h_alt_ref = lbfgs_pass(h_alt_ref, n_alt, 20000, top_k=4, maxiter=250)
                c5_alt_ref = compute_c5(h_alt_ref)
                if c5_alt_ref < best_c5:
                    best_c5 = c5_alt_ref
                    best_h = h_alt_ref.copy()
                    best_n = n_alt

    # Phase 6: Try n=512
    n_alt2 = 512

        h_alt2 = proj(h_alt2, n_alt2)
        for beta in [1000, 5000]:
            if remaining() < 10:
                break
            h_alt2 = lbfgs_pass(h_alt2, n_alt2, beta, top_k=5, maxiter=250)
        if remaining() > 30:
            h_alt2, c5_alt2_pg = nesterov_hc(h_alt2, n_alt2, top_k=5, max_iters=1500, step_init=0.04)
        h_alt2_bin = (h_alt2 > 0.5).astype(float)
        n_ones_a2 = int(np.sum(h_alt2_bin))
        if n_ones_a2 != n_alt2 // 2:
            sort_idx = np.argsort(-h_alt2_bin)
            h_alt2_bin = np.zeros(n_alt2)
            h_alt2_bin[sort_idx[:n_alt2 // 2]] = 1.0
        if remaining() > 30:
            h_alt2_bin, c5_alt2_bs = binary_search(h_alt2_bin, n_alt2, max_iters=8000, rng=rng)
            if c5_alt2_bs < best_c5:
                best_c5 = c5_alt2_bs
                best_h = h_alt2_bin.copy()
                best_n = n_alt2
            if remaining() > 15:
                h_alt2_ref = h_alt2_bin.copy()
                h_alt2_ref = lbfgs_pass(h_alt2_ref, n_alt2, 20000, top_k=4, maxiter=200)
                c5_alt2_ref = compute_c5(h_alt2_ref)

                    best_h = h_alt2_ref.copy()
                    best_n = n_alt2

    # Phase 7: Multi-start with Fourier init at best_n
    if remaining() > 60:
        for fs in range(3):
            h_f = fourier_init(best_n, n_modes=4, seed=100 + fs * 7)
            c5_f = compute_c5(h_f)
            # Quick optimization
            if remaining() < 20:
                break
            h_f = lbfgs_pass(h_f, best_n, 5000, top_k=5, maxiter=200)
            h_f, c5_f = nesterov_hc(h_f, best_n, top_k=5, max_iters=1000, step_init=0.03)
            h_f_bin = (h_f > 0.5).astype(float)
            n_ones_f = int(np.sum(h_f_bin))
            if n_ones_f != best_n // 2:
                sort_idx = np.argsort(-h_f_bin)
                h_f_bin = np.zeros(best_n)
                h_f_bin[sort_idx[:best_n // 2]] = 1.0
            if remaining() > 20:
                h_f_bin, c5_f_bin = binary_search(h_f_bin, best_n, max_iters=5000, rng=rng)
                if c5_f_bin < best_c5:
                    best_c5 = c5_f_bin
                    best_h = h_f_bin.copy()
                if remaining() > 10:
                    h_f_ref = h_f_bin.copy()
                    h_f_ref = lbfgs_pass(h_f_ref, best_n, 20000, top_k=4, maxiter=200)
                    c5_f_ref = compute_c5(h_f_ref)
                    if c5_f_ref < best_c5:

                        best_h = h_f_ref.copy()

    # Phase 8: Intensive binary search on best solution
    if remaining() > 30:
        h_bin_final = (best_h > 0.5).astype(float)
        n_ones_fin = int(np.sum(h_bin_final))

            h_bin_final = np.zeros(best_n)

            best_h = h_bin_final.copy()

    # Phase 9: Final continuous polish
    if remaining() > 15:
