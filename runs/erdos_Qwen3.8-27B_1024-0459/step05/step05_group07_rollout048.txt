def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h))) * (2.0 / n)

def project_feasible(h, n):
    h = np.asarray(h, float).copy()
    np.clip(h, 0, 1, out=h)
    target = 0.5 * n
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        m = (h > 1e-12) & (h < 1 - 1e-12)
        c = m.sum()
        if c: h[m] += d/c
        else:
            i = np.argmin(h) if d > 0 else np.argmax(h)
            h[i] += d
        np.clip(h, 0, 1, out=h)
    return h

def tleft():
    return _BUDGET_S - (time.time() - _START_TIME)

def coord_descent(h, n, max_sweeps=300, tol=1e-12):
    best = h.astype(float).copy()
    corr = compute_corr(best).copy()
    dx = 2.0 / n
    nc = 2*n - 1
    k0 = n - 1
    for sweep in range(max_sweeps):
        if tleft() < 5: break
        imp = False
        for i in range(n):
            old = best[i]
            offs = np.arange(nc) - k0
            ip = i + offs
            im = i - offs
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            A = np.zeros(nc)
            if vp.any(): A[vp] += 1.0 - best[ip[vp]]
            if vm.any(): A[vm] -= best[im[vm]]
            A[k0] = 0.0
            base = corr - A * old
            bk0 = corr[k0] - old*(1.0-old)
            f_old = max(bk0 + old - old*old, np.max(base))
            bt = old; bf = f_old
            for tc in (0.0, 1.0, 0.5):
                fc = max(bk0 + tc - tc*tc, np.max(base + A*tc))
                if fc < bf - tol: bt = tc; bf = fc
            if bt != old:

                else: lo, hi = max(0.0, bt-0.3), old
                if hi-lo < 1e-10: hi = min(1.0, lo+0.1)
                gr = 0.6180339887
                c1 = hi - gr*(hi-lo); c2 = lo + gr*(hi-lo)
                def ft(t): return max(bk0+t-t*t, np.max(base+A*t))
                f1, f2 = ft(c1), ft(c2)
                for _ in range(50):

                best[i] = bt; imp = True
        if not imp: break
