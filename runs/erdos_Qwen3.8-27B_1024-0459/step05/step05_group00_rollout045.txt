    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new <= c5_old + 1e-12:
                return h_new
    except Exception:
        pass
    return h


def nesterov_subgrad(h, n, time_limit, K=20, beta=80):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    h_prev = h.copy()
    start = time.time()
    n_iter = max(int(time_limit * 40), 50)

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        # Nesterov momentum
        if it == 0:
            y = h_curr.copy()
        else:
            gamma = (it - 1.0 + np.sqrt(1.0 + 4.0 * (it - 1.0) ** 2)) / (2.0 * it)
            y = h_curr + gamma * (h_curr - h_prev)
            y = np.clip(y, 0.0, 1.0)

        g_y, c_max = smooth_max_topk_grad(y, n, K, beta)
        gy_n = np.linalg.norm(g_y)
        if gy_n < 1e-15:
            break

        step_size = 0.15 / gy_n
        step_size = min(step_size, 0.4) * (1.0 + 0.5 / (1.0 + it * 0.01))

        h_new = project_feasible(y - step_size * g_y, n)
        c5_new = compute_c5(h_new)

        h_prev = h_curr.copy()
        h_curr = h_new

        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_curr.copy()

    return h_best, c5_best


def fine_gradient_descent(h, n, time_limit, K=10, beta=150):
    h_curr = h.copy()
    h_best = h.copy()
    c5_best = compute_c5(h)
    start = time.time()
    n_iter = max(int(time_limit * 80), 200)

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 3:
            break

        g, c_max = smooth_max_topk_grad(h_curr, n, K, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break

        base_step = 0.02 / gn
        improved = False
        for factor in [1.0, 0.5, 0.25, 2.0, 0.1, 4.0]:
            step = min(base_step * factor, 0.2)
            h_try = project_feasible(h_curr - step * g, n)
            c_try = compute_c5(h_try)
            if c_try < c5_best - 1e-14:
                c5_best = c_try
                h_best = h_try.copy()
                h_curr = h_try.copy()
                improved = True
                break
        if not improved and it > n_iter // 2:
            # Try accepting any feasible move to escape
            step = base_step * 0.5
            h_try = project_feasible(h_curr - step * g, n)
            h_curr = h_try

    return h_best, c5_best


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def perturb_h(base_h, n, rng, sigma=0.02):
    h_pert = base_h + rng.randn(n) * sigma
    return project_feasible(h_pert, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    # Get initial solution
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP sweep at parent resolution with various K and beta
    for K in [10, 15, 20, 30]:
        for beta in [100, 300, 1000, 3000, 8000, 20000]:
            if remaining() < 8:
                break
            h_new = slsqp_optimize(h, n, K, beta, maxiter=350)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 2: Nesterov subgradient for landscape exploration
    if remaining() > 40:
        sg_time = min(80.0, remaining() * 0.15)
        h_sg, c5_sg = nesterov_subgrad(h, n, sg_time, K=20, beta=80)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: SLSQP refinement after subgradient
    for K in [8, 12, 20, 30]:
        for beta in [2000, 8000, 30000]:
            if remaining() < 8:
                break
            h_new = slsqp_optimize(h, n, K, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Multi-resolution exploration
    res_candidates = []
    for ratio in [1.25, 1.5, 1.75, 2.0, 2.5]:
        r = int(round(best_n * ratio / 64) * 64)
        if 256 <= r <= 2048 and r != best_n and r % 2 == 0:
            res_candidates.append(r)
    res_candidates = sorted(set(res_candidates), key=lambda x: abs(x - best_n))

    for n_t in res_candidates[:4]:
        if remaining() < 80:
            break

        h_t = project_feasible(interp_to(best_h, best_n, n_t), n_t)
        c5_t = compute_c5(h_t)

        # SLSQP at new resolution
        for K in [15, 25, 40]:
            for beta in [500, 2000, 10000]:
                if remaining() < 10:
                    break
                h_t = slsqp_optimize(h_t, n_t, K, beta, maxiter=300)
                c5_t = compute_c5(h_t)

        # Nesterov subgradient
        if remaining() > 30:
            h_t, c5_t = nesterov_subgrad(h_t, n_t, min(40.0, remaining() * 0.25), K=30, beta=80)

        # Final SLSQP
        for beta in [20000, 50000]:
            if remaining() < 8:
                break
            h_t = slsqp_optimize(h_t, n_t, 20, beta, maxiter=400)
            c5_t = compute_c5(h_t)

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 5: Perturbation-based multi-start at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 60:
        for pert_i in range(4):
            if remaining() < 30:
                break
            h_pert = perturb_h(h, n, rng, sigma=0.015 + 0.01 * pert_i)
            c5_pert = compute_c5(h_pert)
            if c5_pert > best_c5 + 0.01:
                continue  # Too far from optimum
            # Quick SLSQP from perturbation
            for beta in [2000, 10000]:
                if remaining() < 10:
                    break
                h_pert = slsqp_optimize(h_pert, n, 15, beta, maxiter=250)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert.copy()

    # Phase 6: Intensive final SLSQP at best resolution
    n = best_n
    h = best_h.copy()
    for K in [6, 8, 10, 15, 20]:
        for beta in [5000, 20000, 80000]:
            if remaining() < 5:
                break
            h_new = slsqp_optimize(h, n, K, beta, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 7: Fine gradient descent (very small steps)
    if remaining() > 15:
        fg_time = min(30.0, remaining() * 0.3)
        h_fg, c5_fg = fine_gradient_descent(best_h, best_n, fg_time, K=8, beta=200)
        if c5_fg < best_c5:
            best_c5 = c5_fg
            best_h = h_fg.copy()

    # Phase 8: Final SLSQP polish
    for beta in [50000, 100000, 200000]:
        if remaining() < 5:
            break
        h_new = slsqp_optimize(best_h, best_n, 10, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
