def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        for j in range(n):
            if h[j] > 1e-14 and h[j] < 1.0 - 1e-14:
                h[j] = np.clip(h[j] + diff, 0.0, 1.0)
                break
    return h

def fast_coord_descent(h, n, max_sweeps=100):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 8:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_p = (i_plus >= 0) & (i_plus < n)
            valid_m = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_m):
                A[valid_m] += (1.0 - best[i_minus[valid_m]])
            if np.any(valid_p):
                A[valid_p] -= best[i_plus[valid_p]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            best_f = f_old

                    best_t = t

def _g_constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0


def _g_constraint_sum_jac(x):
    return np.ones(_N_WORK)


def _smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n

    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):

            break

        # Find active set (correlations within 5% of max)
        threshold = max_val - 0.05 * max_val
        active = np.where(best_corr >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(best_corr)])
        if len(active) > 12:
            active = active[np.argsort(best_corr[active])[::-1][:12]]

        # Compute subgradient
        grad = np.zeros(n)
        for k_idx in active:

            if offset >= 0:
                npairs = n - offset
                if npairs > 0:
                    grad[offset:offset + npairs] += (1.0 - best[:npairs])
                    grad[:npairs] -= best[offset:offset + npairs]
            else:
                aoff = -offset
                npairs = n - aoff
                if npairs > 0:
                    grad[:npairs] += (1.0 - best[aoff:aoff + npairs])
                    grad[aoff:aoff + npairs] -= best[:npairs]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Adaptive step size
        step = 0.015 / (1.0 + 0.005 * it)
        h_new = best - step * grad / gnorm
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

            best_c5 = c5_new

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        h_f, c5_f = fast_coord_descent(best_h.copy(), best_n, max_sweeps=60)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 10: One more SLSQP if time
    if remaining_time() > 10:
        h_new = slsqp_refine(best_h.copy(), best_n, 50000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final: coordinate descent + projection
    if remaining_time() > 5:
        h_f, c5_f = fast_coord_descent(best_h.copy(), best_n, max_sweeps=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
