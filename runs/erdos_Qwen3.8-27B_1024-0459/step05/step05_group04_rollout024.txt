def projected_gradient_step(h, n, beta, top_k, step_size):
    """One step of projected gradient with smooth max approximation."""
    corr = compute_corr(h)
    m = np.max(corr)
    
    # Smooth max via top-k weighting
    idx = np.argsort(-corr)[:top_k]
    weights = np.zeros(len(corr))
    for k_idx in idx:
        weights[k_idx] = 1.0 / top_k
    
    # Also add a small log-sum-exp component for global smoothness
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    smax_weights = ex / np.sum(ex)
    
    # Combined weights
    w = 0.5 * weights + 0.5 * smax_weights
    w = w / np.sum(w)
    
    # Compute weighted gradient
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    
    # Take step and project
    h_new = project_feasible(h - step_size * g, n)
    return h_new

            stall = 0
            # Increase step slightly
            step = min(step * 1.1, 0.3)
            # Increase beta and top_k over time
            if time.time() - start > time_limit * 0.3:
                beta = max(beta * 1.2, 500)
                top_k = min(top_k + 1, 15)
            if time.time() - start > time_limit * 0.6:
                beta = max(beta * 1.3, 2000)
                top_k = min(top_k + 1, 20)
        else:
            stall += 1
            step *= 0.7
            if stall > 30:
                # Try a random perturbation
                pert = rng.randn(n) * 0.05
                h_pert = project_feasible(h + pert, n)
                c5_pert = compute_c5(h_pert)
                if c5_pert > best_c5 * 1.01:
                    # Too bad, just reset step
                    step = 0.05
                else:
                    h = h_pert
                stall = 0
    
    return best_h, best_c5

def top_shifts_equalize(h, n, top_k=8, max_iters=300):
    h = h.copy()
    best_c5 = compute_c5(h)
    step_size = 0.02
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]
        
        # Also pull up bottom shifts
        bot_idx = np.argsort(corr)[:3]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [step_size, step_size*0.5, step_size*2, step_size*0.3, step_size*0.7, step_size*3]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                step_size = min(step * 1.05, 0.1)
                improved = True
                break
        
        if not improved:
            step_size *= 0.6
            if step_size < 0.001:
                break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    
    # Phase 1: Binary local search at parent resolution
    h_bin = np.zeros(n)
    sort_idx = np.argsort(-best_h)
    h_bin[sort_idx[:n//2]] = 1.0
    
    t1 = min(250, int(remaining() * 0.25))
    if t1 > 20:
        h_b1, c5_b1 = binary_local_search(h_bin, n, time_limit=t1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n
    
    # Phase 2: Try different starting points at same resolution
    for trial in range(3):
        if remaining() < 40:
            break
        if trial == 0:
            sp = thue_morse(n)
        elif trial == 1:
            sp = (rng.rand(n) > 0.5).astype(float)
            n_o = int(np.sum(sp))
            target = n // 2
            if n_o != target:
                if n_o > target:
                    idx = np.where(sp == 1)[0]
                    sp[idx[:n_o - target]] = 0
                else:
                    idx = np.where(sp == 0)[0]
                    sp[idx[:target - n_o]] = 1
        else:
            # Mutate the best binary
            sp = (best_h > 0.5).astype(float).copy()
            for _ in range(10):
                i = rng.randint(n)
                if sp[i] == 1:
                    sp[i] = 0
            n_o = int(np.sum(sp))
            target = n // 2
            if n_o != target:
                if n_o > target:
                    idx = np.where(sp == 1)[0]
                    sp[idx[:n_o - target]] = 0
                else:
                    idx = np.where(sp == 0)[0]
                    sp[idx[:target - n_o]] = 1
        
        t_sub = min(80, int(remaining() * 0.1))
        if t_sub > 15:
            h_b2, c5_b2 = binary_local_search(sp, n, time_limit=t_sub, rng=rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n
    
    # Phase 3: Try higher resolutions
    higher_ns = [n + 32, n + 64, n + 128, 512, 640, 768]
    higher_ns = sorted(set([x for x in higher_ns if x > n and x <= 1024 and x % 2 == 0]))
    
    for n_t in higher_ns[:4]:
        if remaining() < 60:
            break
        
        # Interpolate current best to new resolution
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.argsort(-h_bin_t)
                h_bin_t[idx[nt_ones - nt_half:]] = 0
            else:
                idx = np.argsort(h_bin_t)
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        
        t_bi = min(100, int(remaining() * 0.1))
        if t_bi > 15:
            h_b3, c5_b3 = binary_local_search(h_bin_t, n_t, time_limit=t_bi, rng=rng)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t
    
    # Also try lower resolution (sometimes coarser is better for binary)
    lower_ns = [n - 32, n - 64, n - 128]
    lower_ns = [x for x in lower_ns if x >= 128 and x % 2 == 0]
    for n_t in lower_ns[:2]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.argsort(-h_bin_t)
                h_bin_t[idx[nt_ones - nt_half:]] = 0
            else:
                idx = np.argsort(h_bin_t)
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        
        t_bi = min(60, int(remaining() * 0.06))
        if t_bi > 10:
            h_b3, c5_b3 = binary_local_search(h_bin_t, n_t, time_limit=t_bi, rng=rng)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t
    
    # Phase 4: Continuous optimization at best resolution
    n = best_n
    h = best_h.copy()
    t_cont = min(200, int(remaining() * 0.2))
    if t_cont > 20:
        h_c, c5_c = continuous_optimize(h, n, time_limit=t_cont, rng=rng)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
    
    # Phase 5: Top-shifts equalization
    if remaining() > 30:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=8, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 6: More continuous optimization
    if remaining() > 30:
        t_c2 = min(150, int(remaining() * 0.15))
        if t_c2 > 15:
            h_c2, c5_c2 = continuous_optimize(h, n, time_limit=t_c2, rng=rng)
            if c5_c2 < best_c5:
                best_c5 = c5_c2
                best_h = h_c2.copy()
    
    # Phase 7: Try even finer resolution if time allows
    if remaining() > 100:
        n_fine = min(best_n + 128, 1024)
        if n_fine % 2 != 0:
            n_fine += 1
        h_fine = interp_to(best_h, best_n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        
        t_cf = min(80, int(remaining() * 0.1))
        if t_cf > 15:
            h_fine, c5_fine = continuous_optimize(h_fine, n_fine, time_limit=t_cf, rng=rng)
            if remaining() > 20:
                h_fine, c5_fine2 = top_shifts_equalize(h_fine, n_fine, top_k=6, max_iters=200)
                c5_fine = min(c5_fine, c5_fine2)
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine.copy()
                best_n = n_fine
    
    # Phase 8: Final polish
    n = best_n
    h = best_h.copy()
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    if remaining() > 30:
        t_final = min(60, int(remaining() * 0.2))
        if t_final > 10:
            h_f, c5_f = continuous_optimize(best_h, best_n, time_limit=t_final, rng=rng)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
