    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0)

    for _ in range(500):

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13 and n > 0:
        free = (h > 0) & (h < 1)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def coord_descent(h, n, n_sweeps=200):
    """Fast coordinate descent with 1D exact minimization."""
    best = h.copy()
    best_corr = compute_corr(best)
    best_c5 = np.max(best_corr) * (2.0 / n)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    
    # Pre-compute valid masks for each i (too much memory for large n, compute on the fly)

        if remaining_time() < 5:
            break
        improved = 0
        for i in range(n):
            if remaining_time() < 1:
                break
            v0 = best[i]
            if v0 < 1e-15 or v0 > 1.0 - 1e-15:
                # Allow move from boundary
                pass
            
            # g[k] = coefficient of h[i]=v in corr[k]
            # h-term: v*(1-h[i-t]) where t=shifts[k], valid if 0≤i-t<n
            # 1-h term: h[i+t]*(1-v) where valid if 0≤i+t<n → contributes -h[i+t]*v
            
            sh = shifts  # shape (2n-1,)
            idx_h = i - sh  # for h-term: need 0 <= idx_h < n
            idx_1h = i + sh  # for 1-h term: need 0 <= idx_1h < n
