        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 200.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 400.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=5, maxiter=800):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxiter * 12,
                                'maxls': 50})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def binary_block_search(h_bin, n, max_iters=3000, top_k=4):
    """Binary local search with block moves, multi-swaps, and gradient direction."""
    h_bin = h_bin.astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5
    patience = 0
    max_patience = 80

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h_bin)
        cur_c5 = float(np.max(corr)) * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                # Random restart from perturbation
                for _ in range(10):
                    ones = np.where(h_bin == 1)[0]
                    zeros = np.where(h_bin == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
                    h_bin[i] = 0
                    h_bin[j] = 1
                    nc = compute_c5(h_bin)
                    if nc < cur_c5:
                        cur_c5 = nc
                        break
                    else:
                        h_bin[i] = 1
                        h_bin[j] = 0
                patience = 0

        idx_top = np.argsort(-corr)[:top_k]

        # Compute weighted gradient
        c_top = corr[idx_top]
        w_max = c_top[0]
        w_min = c_top[-1]
        w_range = max(w_max - w_min, 1e-10)
        weights = ((c_top - w_min) / w_range) ** 2 + 0.1
        weights /= weights.sum()

        g = np.zeros(n)
        for j, ki in enumerate(idx_top):
            g += weights[j] * corr_grad_single(h_bin, ki, n)

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_move = None
        best_move_type = None

        # Type 1: Gradient-directed single swap
        n_try = min(10, len(ones), len(zeros))
        score_remove = g[ones]
        score_add = -g[zeros]
        top_o = np.argsort(-score_remove)[:n_try]
        top_z = np.argsort(-score_add)[:n_try]

        for ii in range(n_try):
            i = ones[top_o[ii]]
            for jj in range(n_try):
                j = zeros[top_z[jj]]
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j)
                    best_move_type = 'swap'

        # Type 2: Block shift (shift contiguous block of 1s)
        # Find blocks of 1s
        blocks = []
        in_block = False
        bs = 0
        for i in range(n):
            if h_bin[i] == 1 and not in_block:
                bs = i
                in_block = True
            elif h_bin[i] == 0 and in_block:
                blocks.append((bs, i))
                in_block = False
        if in_block:
            blocks.append((bs, n))

        for (bstart, bend) in blocks:
            blen = bend - bstart
            if blen < 2 or blen > 10:
                continue
            for shift in [-3, -2, -1, 1, 2, 3]:
                ns = bstart + shift
                ne = bend + shift
                if ns < 0 or ne > n:
                    continue
                # Check new position is all zeros
                ok = True
                for p in range(ns, ne):
                    if h_bin[p] == 1:
                        ok = False
                        break
                if not ok:
                    continue
                # Perform move
                h_bin[bstart:bend] = 0
                h_bin[ns:ne] = 1
                nc = compute_c5(h_bin)
                h_bin[ns:ne] = 0
                h_bin[bstart:bend] = 1
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = (bstart, bend, shift)
                    best_move_type = 'block'

        # Type 3: Double swap (two simultaneous swaps)
        if best_delta > -1e-12 and len(ones) > 2 and len(zeros) > 2:
            n_try2 = min(5, len(ones), len(zeros))
            top_o2 = np.argsort(-score_remove)[:n_try2]
            top_z2 = np.argsort(-score_add)[:n_try2]
            for ii1 in range(n_try2):
                i1 = ones[top_o2[ii1]]
                for ii2 in range(ii1 + 1, n_try2):
                    i2 = ones[top_o2[ii2]]
                    for jj1 in range(n_try2):
                        j1 = zeros[top_z2[jj1]]
                        for jj2 in range(jj1 + 1, n_try2):
                            j2 = zeros[top_z2[jj2]]
                            if len(set([i1, i2, j1, j2])) < 4:
                                continue
                            h_bin[i1] = 0
                            h_bin[i2] = 0
                            h_bin[j1] = 1
                            h_bin[j2] = 1
                            nc = compute_c5(h_bin)
                            h_bin[j1] = 0
                            h_bin[j2] = 0
                            h_bin[i1] = 1
                            h_bin[i2] = 1
                            delta = nc - cur_c5
                            if delta < best_delta:
                                best_delta = delta
                                best_move = (i1, i2, j1, j2)
                                best_move_type = 'dswap'

        if best_move is not None and best_delta < -1e-14:
            if best_move_type == 'swap':
                i, j = best_move
                h_bin[i] = 0
                h_bin[j] = 1
            elif best_move_type == 'block':
                bstart, bend, shift = best_move
                h_bin[bstart:bend] = 0
                h_bin[bstart + shift:bend + shift] = 1
            elif best_move_type == 'dswap':
                i1, i2, j1, j2 = best_move
                h_bin[i1] = 0
                h_bin[i2] = 0
                h_bin[j1] = 1
                h_bin[j2] = 1
            cur_c5 = compute_c5(h_bin)
        else:
            # No improving move found; try random to escape
            improved = False
            for _ in range(8):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                if nc < cur_c5:
                    h_bin[i] = 0
                    h_bin[j] = 1
                    cur_c5 = nc
                    improved = True
                    break
            if not improved:
                # Accept a slightly worse move (simulated annealing step)
                if np.random.random() < 0.3:
                    i = ones[np.random.randint(len(ones))]
                    j = zeros[np.random.randint(len(zeros))]
                    h_bin[i] = 0
                    h_bin[j] = 1
                    cur_c5 = compute_c5(h_bin)

    return best_h, best_c5


def nesterov_gradient(h, n, top_k=5, max_iters=8000, step_init=0.04):
    """Projected gradient with Nesterov-like momentum."""
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        c_top = corr[idx]
        w_max = c_top[0]
        w_min = c_top[-1]
        w_range = max(w_max - w_min, 1e-10)
        weights = ((c_top - w_min) / w_range) ** 2 + 0.05
        weights /= weights.sum()

        g = np.zeros(n)
        for j, ki in enumerate(idx):
            g += weights[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-10)] = 0.0
        g[(h <= 1e-10)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.03, 0.2)
            else:
                step *= 0.9
        else:
            h = best_h.copy()
            v *= 0.4
            step *= 0.65

        if it % 800 == 0 and it > 0:
            step *= 0.9
            momentum = min(momentum + 0.008, 0.95)

    return best_h, best_c5


def top_equalize(h, n, top_k=5, max_iters=400):
    """Directly equalize top correlations."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-9:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_max
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn

        improved = False
        for scale in [0.03, 0.015, 0.05, 0.01, 0.02, 0.04, 0.008, 0.025]:
            h_try = proj(h + scale * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for scale in [0.02, 0.01, 0.03]:
                h_try = proj(h - scale * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Aggressive L-BFGS-B at parent resolution
    beta_schedule = [50, 150, 400, 1000, 3000, 8000, 20000, 50000]
    for beta in beta_schedule:
        if remaining() < 10:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=700)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Nesterov gradient
    if remaining() > 30:
        h_pg, c5_pg = nesterov_gradient(h, n, top_k=5, max_iters=10000, step_init=0.035)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 3: Top equalization
    if remaining() > 15:
        h_eq, c5_eq = top_equalize(h, n, top_k=6, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: More L-BFGS-B with high beta
    for beta in [30000, 80000, 150000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=4, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Binary local search with block moves (KEY IMPROVEMENT)
    if remaining() > 40:
        h_bin_init = (best_h > 0.5).astype(float)
        h_bl, c5_bl = binary_block_search(h_bin_init, best_n, max_iters=5000, top_k=4)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

        # Refine binary solution continuously
        h_ref = h_bl.copy()
        for beta in [5000, 20000, 50000]:
            if remaining() < 5:
                break
            h_ref = lbfgs_pass(h_ref, best_n, beta, top_k=4, maxiter=400)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 6: Higher resolution attempts
    n_candidates = sorted(set([
        best_n + 2, best_n - 2,
        int(best_n * 1.33), int(best_n * 1.5),
        int(best_n * 1.2), best_n + 8,
    ]))
    n_candidates = [nn for nn in n_candidates if 128 <= nn <= 1024 and nn % 2 == 0]
    n_candidates = [nn for nn in n_candidates if nn != best_n]
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # Quick L-BFGS-B at new resolution
        for beta in [300, 1500, 5000, 15000]:
            if remaining() < 10:
                break
            h_feas = lbfgs_pass(h_feas, n_t, beta, top_k=5, maxiter=500)

        c5_feas = compute_c5(h_feas)

        # Binary search at this resolution
        h_b2 = (h_feas > 0.5).astype(float)
        h_bl2, c5_bl2 = binary_block_search(h_b2, n_t, max_iters=3000, top_k=4)

        # Refine
        h_ref2 = h_bl2.copy()
        for beta in [5000, 20000]:
            if remaining() < 5:
                break
            h_ref2 = lbfgs_pass(h_ref2, n_t, beta, top_k=4, maxiter=300)
        c5_ref2 = compute_c5(h_ref2)

        c5_candidate = min(c5_feas, c5_bl2, c5_ref2)
        if c5_ref2 <= c5_candidate:
            h_candidate = h_ref2
        elif c5_bl2 <= c5_candidate:
            h_candidate = h_bl2
        else:
            h_candidate = h_feas

        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            best_n = n_t

    # Phase 7: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    if remaining() > 20:
        h_pg2, c5_pg2 = nesterov_gradient(h, n, top_k=4, max_iters=6000, step_init=0.02)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()
            h = h_pg2.copy()

    for beta in [50000, 100000, 200000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=3, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final top equalize
    if remaining() > 10:
        h_eq2, c5_eq2 = top_equalize(h, n, top_k=8, max_iters=400)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final binary polish with block moves
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_block_search(h_bin_f, best_n, max_iters=3000, top_k=5)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        # Refine
        h_rf = (best_h > 0.5).astype(float) if c5_bf < best_c5 else best_h.copy()
        for beta in [30000, 100000]:
            if remaining() < 5:
                break
            h_rf = lbfgs_pass(h_rf, best_n, beta, top_k=4, maxiter=300)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
