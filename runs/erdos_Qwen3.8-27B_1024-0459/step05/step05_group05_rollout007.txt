    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50, 50)

    # For o >= 0 (k = o + n - 1):
    #   grad[o:] += w[k] * (1-h[:n-o])
    #   grad[:n-o] -= w[k] * h[o:]
    # For o < 0 (a = -o, k = n-1-a):
    #   grad[:n-a] += w[k] * (1-h[a:])
    #   grad[a:] -= w[k] * h[:n-a]

        wk = w[k] * dx
        if wk < 1e-15:

            continue
        L = n - o
        grad[:L] += wk * (1.0 - h[o:])
        grad[o:] -= wk * h[:L]
    
    return obj, grad

import numpy as np
import time

_T0 = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def thue_morse(n, rng):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones > n_half:
        ones_idx = np.where(h == 1)[0]
        perm = rng.permutation(len(ones_idx))
        h[ones_idx[perm[:n_ones - n_half]]] = 0
    elif n_ones < n_half:
        zeros_idx = np.where(h == 0)[0]
        perm = rng.permutation(len(zeros_idx))
        h[zeros_idx[perm[:n_half - n_ones]]] = 1
    return h

def smooth_grad_descent(h, n, betas, max_iters_total=200000):
    """Projected gradient with adaptive beta."""
    h = project_feasible(h, n).copy()
    best_h = h.copy()

    step = 0.05
    momentum_v = np.zeros(n)
    
    for beta in betas:
        if remaining() < 5:
            break
        # Anneal: reduce step as beta increases
        step = max(0.01, 0.3 / (1 + beta/50))
        
        for it in range(max_iters_total):
            if remaining() < 3:
                break
            
            # Compute smooth objective and gradient
            corr = compute_corr(h)
            m = np.max(corr)
            shifted = np.clip(beta * (corr - m), -50, 50)
            exp_v = np.exp(shifted)
            sum_exp = np.sum(exp_v)
            w = exp_v / sum_exp
            dx = 2.0 / n
            
            grad = np.zeros(n)
            for o in range(0, n):
                k = o + n - 1
                wk = w[k] * dx
                if wk < 1e-14:
                    continue
                L = n - o
                grad[o:] += wk * (1.0 - h[:L])
                grad[:L] -= wk * h[o:]
            
            for o in range(1, n):
                k = n - 1 - o
                wk = w[k] * dx
                if wk < 1e-14:
                    continue
                L = n - o
                grad[:L] += wk * (1.0 - h[o:])
                grad[o:] -= wk * h[:L]
            
            # Center gradient (remove mean for sum constraint)
            grad -= np.mean(grad)
            
            # Nesterov momentum
            momentum_v = 0.85 * momentum_v + grad
            h_new = project_feasible(h - step * momentum_v, n)
            
            c5_new = compute_c5(h_new)

                best_h = h_new.copy()
                h = h_new.copy()

                step = max(step * 0.95, 0.005)
            
            # Periodically check actual max
            if it % 200 == 0:
