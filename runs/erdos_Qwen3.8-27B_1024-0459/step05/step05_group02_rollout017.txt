        448, 480, 512, 576, 640, 768,
    ]))
    res_list = [r for r in res_list if 192 <= r <= 1024 and r % 2 == 0 and r != best_n]
    res_list = sorted(set(res_list), key=lambda x: abs(x - best_n))

    for n_t in res_list[:8]:
        if remaining() < 35:
            break
        t_alloc = min(remaining() * 0.15, 120.0)
        h_t = interp_to(best_h, best_n, n_t)
        h_t, c5_t = optimize_at_resolution(h_t, n_t, t_alloc)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 3: Final intensive refinement at best resolution
    if remaining() > 25:
        n_f = best_n
        h_f = best_h.copy()

        for beta in [15000, 50000, 150000]:
            if remaining() < 8:
                break
            h_f = lbfgs_pass(h_f, n_f, beta, 6, 300)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

        h_bf = to_binary(best_h, n_f)
        for tk in [4, 5, 6]:
            if remaining() < 10:
                break
            h_bf, c5_bf = binary_search(h_bf, n_f, max_iters=2000, top_k=tk, n_cand=40, do_2swap=True)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()

        hr = best_h.copy()
        for beta in [30000, 100000]:
            if remaining() < 5:
                break
            hr = lbfgs_pass(hr, n_f, beta, 5, 200)
        c5_hr = compute_c5(hr)
        if c5_hr < best_c5:
            best_c5 = c5_hr
            best_h = hr.copy()

    # Final feasibility check
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev < best_c5:
            best_c5 = ev
    except Exception:
        pass

    return (best_h, best_c5, best_n)
