        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g, h) + 1e-12)

    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta, h + delta])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])

    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), c_max_val


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    n = best_n
    h = best_h.copy()

    # Phase 1: Subgradient descent at parent resolution
    if remaining() > 30:
        h_sg, c5_sg = top_k_subgrad_descend(h, n, min(60.0, remaining() * 0.2), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 2: L-BFGS-B with smooth max at parent resolution
    for beta in [200, 500, 1000, 3000]:
        if remaining() < 10:
            break
        h_lb, c5_lb, _ = lbfgsb_optimize(h, n, beta, maxiter=300, time_limit=min(120.0, remaining() * 0.3))
        c5_check = compute_c5(h_lb)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h_lb.copy()
            h = h_lb.copy()

    # Phase 3: SLSQP polish
    for beta in [5000, 10000, 20000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Cutting plane iterations
    if remaining() > 30:
        for it in range(50):
            if remaining() < 5:
                break
            h_lp, t_lp = cutting_plane_step(h, n, K=20, delta=0.04)
            if t_lp < compute_c5(h) - 1e-10:
                best_val = compute_c5(h)
                for alpha in [1.0, 0.5, 0.3, 0.7]:
                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        h = h_try
                if compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
            elif it > 25:
                break

    # SLSQP after cutting plane
    for beta in [15000, 30000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Try higher resolutions
    for n_t in [1024, 2048]:
        if remaining() < 60:
            break
        if n_t == best_n:
            continue
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        h_cur = h_feas.copy()
        c5_cur = c5_int

        # Subgradient at new resolution
        if remaining() > 30:
            h_cur, c5_cur = top_k_subgrad_descend(h_cur, n_t, min(40.0, remaining() * 0.25), rng)

        # L-BFGS-B at new resolution
        for beta in [300, 1000, 3000, 8000]:
            if remaining() < 15:
                break
            h_lb, c5_lb, _ = lbfgsb_optimize(h_cur, n_t, beta, maxiter=200, time_limit=min(100.0, remaining() * 0.3))
            c5_check = compute_c5(h_lb)
            if c5_check < c5_cur:
                c5_cur = c5_check
                h_cur = h_lb.copy()

        # SLSQP polish
        for beta in [5000, 15000]:
            if remaining() < 10:
                break
            h_cur = slsqp_optimize(h_cur, n_t, beta, maxiter=300)
            c5_cur = compute_c5(h_cur)

        # Cutting plane
        if remaining() > 20:
            for it in range(40):
                if remaining() < 5:
                    break
                h_lp, t_lp = cutting_plane_step(h_cur, n_t, K=20, delta=0.03)
                if t_lp < compute_c5(h_cur) - 1e-10:
                    best_val = compute_c5(h_cur)
                    for alpha in [1.0, 0.5, 0.3]:
                        h_try = project_feasible(h_cur + alpha * (h_lp - h_cur), n_t)
                        c_try = compute_c5(h_try)
                        if c_try < best_val:
                            best_val = c_try
                            h_cur = h_try
                    c5_cur = compute_c5(h_cur)
                elif it > 20:
                    break

        # Final L-BFGS-B
        if remaining() > 15:
            h_lb, c5_lb, _ = lbfgsb_optimize(h_cur, n_t, 10000, maxiter=200, time_limit=min(80.0, remaining() * 0.4))
            c5_lb = compute_c5(h_lb)
            if c5_lb < c5_cur:
                c5_cur = c5_lb
                h_cur = h_lb.copy()

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_t

    # Phase 6: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    for cycle in range(3):
        if remaining() < 25:
            break
        # Subgradient
        if remaining() > 15:
            h_sg, c5_sg = top_k_subgrad_descend(h, n, min(30.0, remaining() * 0.2), rng)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()
        # L-BFGS-B
        for beta in [5000, 15000]:
            if remaining() < 10:
                break
            h_lb, c5_lb, _ = lbfgsb_optimize(h, n, beta, maxiter=300, time_limit=min(80.0, remaining() * 0.25))
            c5_lb = compute_c5(h_lb)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb.copy()
                h = h_lb.copy()
        # SLSQP
        if remaining() > 10:
            h_new = slsqp_optimize(h, n, 20000, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 7: Random perturbations + re-optimization
    n = best_n
    h = best_h.copy()
    n_restarts = 5
    if remaining() > n_restarts * 30:
        for r in range(n_restarts):
            if remaining() < 30:
                break
            noise_scale = 0.05 / (1 + r)
            h_pert = h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, n)
            # Quick L-BFGS-B
            h_lb, c5_lb, _ = lbfgsb_optimize(h_pert, n, 2000, maxiter=200, time_limit=20.0)
            c5_lb = compute_c5(h_lb)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb.copy()
                h = h_lb.copy()
            # SLSQP polish on best
            if remaining() > 10:
                h_new = slsqp_optimize(h, n, 10000, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()

    # Phase 8: Final polish
    n = best_n
    h = best_h.copy()
    for beta in [30000, 50000, 100000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Very fine subgradient
    if remaining() > 10:
        h_sg, c5_sg = top_k_subgrad_descend(h, n, min(20.0, remaining() * 0.5), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final L-BFGS-B
    if remaining() > 15:
        h_lb, c5_lb, _ = lbfgsb_optimize(best_h, best_n, 50000, maxiter=300, time_limit=min(100.0, remaining() * 0.5))
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    # Ensure final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
