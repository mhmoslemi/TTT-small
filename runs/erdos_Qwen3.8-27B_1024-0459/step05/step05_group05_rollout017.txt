import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

    sort_idx = np.argsort(-h)

Actually, I can be smarter. For a point swap (i: 1→0, j: 0→1), the change in corr[off] is:

For shift off >= 0: corr[off] = Σ_{n=0}^{n-1-off} h[n+off]*(1-h[n])
- Effect of h[i]: if i is in [0, n-1-off], the term (1-h[i]) changes from 0 to 1, contribution += h[i+off]

  (These could overlap if off=0, but we're off>0 for the interesting case)

For shift off < 0 (let ao = -off > 0): corr[-ao] = Σ_{n=ao}^{n-1} h[n]*(1-h[n-ao])
- Effect of h[i]: if i is in [ao, n-1], h[i] changes 1→0, contribution -= (1-h[i-ao])
  if i is in [0, n-1-ao], (1-h[i]) changes 0→1, contribution += h[i+ao]

This is O(1) per shift, so we can compute the full δcorr vector in O(n) time!

def swap_delta(h, i, j, n):
    """Compute the change in corr vector when h[i]: 1->0 and h[j]: 0->1."""
    delta = np.zeros(2*n-1)
    # i was 1, becomes 0
    # j was 0, becomes 1
    
    # For shift off >= 0: corr[off] = sum_{m=0}^{n-1-off} h[m+off]*(1-h[m])
    # Effect of changing h[i] (1->0):
    #   In the h[m+off] slot: m+off = i => m = i-off, valid if 0 <= i-off <= n-1-off
    #     term was h[i]*(1-h[i-off]) = (1-h[i-off]), becomes 0, delta = -(1-h[i-off])
    #   In the (1-h[m]) slot: m = i, valid if 0 <= i <= n-1-off
    #     term was h[i+off]*(1-h[i]) = 0 (since h[i]=1), becomes h[i+off]*(1-0) = h[i+off]
    #     delta = +h[i+off]
    
    # Effect of changing h[j] (0->1):
    #   In the h[m+off] slot: m+off = j => m = j-off, valid if 0 <= j-off <= n-1-off
    #     term was h[j]*(1-h[j-off]) = 0, becomes (1-h[j-off]), delta = +(1-h[j-off])
    #   In the (1-h[m]) slot: m = j, valid if 0 <= j <= n-1-off
    #     term was h[j+off]*(1-h[j]) = h[j+off], becomes 0, delta = -h[j+off]
    
    # Similarly for shift off < 0 (a = -off > 0): corr index = n-1-a
    # corr[-a] = sum_{m=a}^{n-1} h[m]*(1-h[m-a])
    # Effect of h[i] (1->0):
    #   In h[m]: m = i, valid if a <= i <= n-1
    #     term was h[i]*(1-h[i-a]) = (1-h[i-a]), becomes 0, delta = -(1-h[i-a])
    #   In (1-h[m-a]): m-a = i => m = i+a, valid if a <= i+a <= n-1 => 0 <= i <= n-1-a
    #     term was h[i+a]*(1-h[i]) = 0, becomes h[i+a], delta = +h[i+a]
    
    # Effect of h[j] (0->1):
    #   In h[m]: m = j, valid if a <= j <= n-1
    #     term was h[j]*(1-h[j-a]) = 0, becomes (1-h[j-a]), delta = +(1-h[j-a])
    #   In (1-h[m-a]): m-a = j => m = j+a, valid if a <= j+a <= n-1 => 0 <= j <= n-1-a
    #     term was h[j+a]*(1-h[j]) = h[j+a], becomes 0, delta = -h[j+a]
    
    # Let me index the corr array as corr[k] where k goes from 0 to 2n-2
    # k = 0 corresponds to shift -(n-1), k = n-1 corresponds to shift 0, k = 2n-2 corresponds to shift n-1
    # So shift off >= 0: k = n-1+off
    #    shift off < 0 (a = -off): k = n-1-off = n-1+a
    
    # For shift off >= 0 (k = n-1+off):
    for off in range(1, n):  # off = shift, 1 to n-1
        k = n - 1 + off
        # h[i] effect (1->0):
        #   m = i-off in range [0, n-1-off]: i-off >= 0 => i >= off, i-off <= n-1-off => i <= n-1
        if off <= i <= n-1:

        #   m = i in range [0, n-1-off]: i <= n-1-off
        if 0 <= i <= n-1-off:
            delta[k] += h[i+off]
        # h[j] effect (0->1):
        if off <= j <= n-1:
            delta[k] += (1 - h[j-off])
        if 0 <= j <= n-1-off:
            delta[k] -= h[j+off]
    
    # For shift a > 0 (k = n-1-a):
    for a in range(1, n):
        k = n - 1 - a
        # h[i] effect (1->0):
        #   m = i in range [a, n-1]: a <= i
        if a <= i:

        #   m = i+a in range [a, n-1]: i+a <= n-1 => i <= n-1-a
        if i <= n-1-a:
            delta[k] += h[i+a]
        # h[j] effect (0->1):
        if a <= j:
            delta[k] += (1 - h[j-a])
        if j <= n-1-a:
            delta[k] -= h[j+a]
    
    return delta

    #   off ranges from 1 to i (and also off=0 but that's trivial for binary)
    #   delta at k=n-1+off: -(1-h[i-off])

    #   delta at k=n-1-a: -(1-h[i-a])

    #   delta at k=n-1-a: +h[i+a]

    if n-1-i > 0:

        delta[n-1+off_range] += h[i + off_range]
    # Neg shifts: a in [1, i]: delta[n-1-a] -= (1-h[i-a])
    if i > 0:
        a_range = np.arange(1, i+1)
        delta[n-1-a_range] -= (1 - h[i - a_range])
    # Neg shifts: a in [1, n-1-i]: delta[n-1-a] += h[i+a]
    if n-1-i > 0:

        delta[n-1-a_range] += h[i + a_range]
    
    # h[j]: 0->1
    # Pos shifts: off in [1, j]: delta[n-1+off] += (1-h[j-off])
    if j > 0:
        off_range = np.arange(1, j+1)
        delta[n-1+off_range] += (1 - h[j - off_range])
    # Pos shifts: off in [1, n-1-j]: delta[n-1+off] -= h[j+off]
    if n-1-j > 0:
        off_range = np.arange(1, n-j)
        delta[n-1+off_range] -= h[j + off_range]

def swap_delta(h, i, j, n):
    """Compute corr change vector for swap: h[i] 1->0, h[j] 0->1. O(n) vectorized."""
    length = 2 * n - 1
    delta = np.zeros(length)
    off = np.arange(1, n, dtype=np.int64)

    # Positive shifts: index n-1+off
    m = off <= i
    if np.any(m):
        r = off[m]
        delta[n - 1 + r] -= (1.0 - h[i - r])
    m = off <= (n - 1 - i)
    if np.any(m):
        r = off[m]
        delta[n - 1 + r] += h[i + r]

    m = off <= j
    if np.any(m):
        r = off[m]
        delta[n - 1 + r] += (1.0 - h[j - r])

    best_max = cur_max

    zeros = np.where(h == 0)[0].copy()
    hi = h[i] if False else 0  # dummy
    while time.time() - start < time_limit and remaining() > 5:

    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_improve = 0
    max_no = 50
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    while time.time() - start < time_limit and remaining() > 5:
        if no_improve >= max_no:
            k = int(rng.randint(2, 6))
            for _ in range(k):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
            no_improve = 0
            continue
        improved = False
        for _ in range(n_tries_per_iter):
            if time.time() - start > time_limit or remaining() < 5:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            d = swap_delta(h, i, j, n)
            nm = float(np.max(corr + d))
            if nm < cur_max - 1e-15:
                h[i] = 0.0
                h[j] = 1.0
                corr += d
                cur_max = nm
                pi = int(np.where(ones == i)[0][0])
                pj = int(np.where(zeros == j)[0][0])
                ones[pi] = j
                zeros[pj] = i
                improved = True
                break
        if improved:
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1
    return best_h, best_max * dx


def slsqp_smooth(h, n, beta, maxiter=400):
    """Log-sum-exp smoothed max via SLSQP."""
    N = n
    TGT = n / 2.0
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = compute_corr(x)
        m = float(np.max(corr))
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / N)

    def grad(x):
        corr = compute_corr(x)
        m = float(np.max(corr))
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(N)
        wc = w * (2.0 / N)
        for k in range(2 * N - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (N - 1)
            if off >= 0:
                np_ = N - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = N - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def cfun(x):
        return np.sum(x) - TGT

    def cjac(x):
        return np.ones(N)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def top_shifts_focus(h, n, top_k=5, max_iters=300):
    """Directional subgradient focusing on top-k worst shifts."""
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-9:
            break
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(idx)
        g -= np.mean(g)
        dn = np.linalg.norm(g)
        if dn < 1e-14:
            break
        d = -g / dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h_bin_parent = binarize(best_h, n)
    h_tm = thue_morse(n, rng)

    # Phase 1: SA from parent binary
    if remaining() > 120:
        tl = min(250.0, remaining() * 0.25)
        h_s1, c5_s1 = sa_binary(h_bin_parent, n, rng, time_limit=tl, t_init=3.0, t_floor=0.05, cool=0.9997)
        if c5_s1 < best_c5:
            best_c5 = c5_s1
            best_h = h_s1.copy()
            best_n = n

    # Phase 2: SA from Thue-Morse
    if remaining() > 120:
        tl = min(250.0, remaining() * 0.25)
        h_s2, c5_s2 = sa_binary(h_tm, n, rng, time_limit=tl, t_init=2.5, t_floor=0.04, cool=0.9997)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()
            best_n = n

    # Phase 3: Hill climb from current best
    if remaining() > 60:
        h_bc = binarize(best_h, best_n)
        tl = min(120.0, remaining() * 0.3)
        h_hc, c5_hc = hill_climb_binary(h_bc, best_n, rng, time_limit=tl, n_tries_per_iter=40)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc.copy()
            best_n = best_n

    # Phase 4: SA from best with lower temperature (fine-tuning)
    if remaining() > 80:
        h_bc = binarize(best_h, best_n)
        tl = min(200.0, remaining() * 0.3)
        h_s4, c5_s4 = sa_binary(h_bc, best_n, rng, time_limit=tl, t_init=1.0, t_floor=0.02, cool=0.9998)
        if c5_s4 < best_c5:
            best_c5 = c5_s4
            best_h = h_s4.copy()
            best_n = best_n

    # Phase 5: Continuous refinement via SLSQP
    n = best_n
    if remaining() > 60:
        h_cont = binarize(best_h, n)
        for beta in [1000, 5000, 20000, 80000]:
            if remaining() < 8:
                break
            h_cont = slsqp_smooth(h_cont, n, beta, maxiter=300)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
            best_n = n
        h_fs, c5_fs = top_shifts_focus(h_cont, n, top_k=5, max_iters=200)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs.copy()
            best_n = n

    # Phase 6: Multi-resolution sweep
    if remaining() > 80:
        n_base = best_n
        n_cands = []
        for dt in [-8, -4, -2, 2, 4, 8, 12, 16]:
            nt = n_base + dt
            if 128 <= nt <= 700 and nt % 2 == 0 and nt != n_base:
                n_ones = nt // 2
                if n_ones * 2 == nt:
                    n_cands.append(nt)
        n_cands.sort(key=lambda x: abs(x - n_base))
        for n_t in n_cands[:6]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            h_tb = binarize(h_t, n_t)
            tl = min(40.0, remaining() * 0.4)
            h_sr, c5_sr = sa_binary(h_tb, n_t, rng, time_limit=tl, t_init=2.0, t_floor=0.04, cool=0.9997)
            if c5_sr < best_c5:
                best_c5 = c5_sr
                best_h = h_sr.copy()
                best_n = n_t

    # Phase 7: Final intensive hill climb
    if remaining() > 30:
        n = best_n
        h_bc = binarize(best_h, n)
        tl = min(60.0, remaining() * 0.5)
        h_fin, c5_fin = hill_climb_binary(h_bc, n, rng, time_limit=tl, n_tries_per_iter=60)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n

    # Phase 8: Slight continuous polish
    if remaining() > 15:
        n = best_n
        h_pol = project_feasible(best_h, n)
        h_pol = slsqp_smooth(h_pol, n, 50000, maxiter=200)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n
        h_tsf, c5_tsf = top_shifts_focus(h_pol, n, top_k=6, max_iters=150)
        if c5_tsf < best_c5:
            best_c5 = c5_tsf
            best_h = h_tsf.copy()
            best_n = n

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
