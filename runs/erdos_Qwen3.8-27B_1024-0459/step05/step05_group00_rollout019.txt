One more thought: the "flow" algorithm. Here's how it works:
- Compute gradient g = ∂F/∂h (where F is the smooth max)
- The index j* = argmax(g) means F increases most rapidly when h[j*] increases → we want to DECREASE h[j*]
- The index i* = argmin(g) means F increases least rapidly (or decreases) when h[i*] increases → we want to INCREASE h[i*]
- Move δ from h[j*] to h[i*]: h[i*] += δ, h[j*] -= δ
- This maintains the sum constraint and (if δ is small) the box constraints
- Choose δ by 1D line search

This is essentially the steepest descent direction restricted to the constraint manifold, and it's much more efficient than the full gradient step because it only changes 2 coordinates at a time.

        if gn < 1e-14:
            break
        base_step = 0.04 * n / gn
        step = base_step / (1.0 + 0.15 * it)
        step = min(step, 0.2)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
    return best_h, best_c5

def slsqp_smooth(h, n, beta, maxiter=500):
    def obj_func(x):
        return smooth_max_obj(x, n, beta)
    def grad_func(x):
        return smooth_max_grad(x, n, beta)
    def cons_eq(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-9:
                return h_new
    except Exception:
        pass
    return h

def trust_constr_smooth(h, n, beta, maxiter=300):
    def obj_func(x):
        return smooth_max_obj(x, n, beta)
    def grad_func(x):
        return smooth_max_grad(x, n, beta)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    constraints = [None]
    from scipy.optimize import LinearConstraint
    A = np.ones((1, n))
    lc = LinearConstraint(A, n/2.0, n/2.0)
    try:
        res = minimize(obj_func, h, jac=grad_func, method='trust-constr',
                       bounds=bounds, constraints=lc,
                       options={'maxiter': maxiter, 'gtol': 1e-16, 'xtol': 1e-16})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-9:
                return h_new
    except Exception:
        pass
    return h

def flow_local_search(h, n, max_iter=2000, time_limit=60.0):
    global _T0, _BUDGET
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        beta = 10000.0
        g = smooth_max_grad(h, n, beta)
        # Find max and min gradient indices
        i_max = np.argmax(g)  # want to decrease h[i_max]
        i_min = np.argmin(g)  # want to increase h[i_min]
        if i_max == i_min:
            break
        # Try moving delta from i_max to i_min
        delta_max = min(h[i_max], 1.0 - h[i_min])
        for delta in [0.01, 0.005, 0.002, 0.001, 0.02, 0.0005, 0.05]:
            if delta > delta_max:
                continue
            h_try = h.copy()
            h_try[i_min] += delta
            h_try[i_max] -= delta
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c_try
                break
        else:
            # Try second-best pair
            order_max = np.argsort(-g)
            order_min = np.argsort(g)
            found = False
            for a in order_max[:5]:
                for b in order_min[:5]:
                    if a == b:
                        continue
                    dm = min(h[a], 1.0 - h[b])
                    for delta in [0.005, 0.002, 0.001]:
                        if delta > dm:
                            continue
                        h_try = h.copy()
                        h_try[b] += delta
                        h_try[a] -= delta
                        c_try = compute_c5(h_try)
                        if c_try < best_c5 - 1e-15:
                            h = h_try
                            best_c5 = c_try
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if not found:
                break
    return h, best_c5

def cutting_plane_lp(h, n, K=25, delta_trust=0.05):
    corr = compute_corr(h)
    c_max = np.max(corr)
    idx = np.argsort(-corr)[:K]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        fk = corr[k]
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-fk + np.dot(g, h))
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    I = np.eye(n)
    A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                         np.hstack([-I, np.zeros((n, 1))])])
    b_trust = np.concatenate([h + delta_trust, h + delta_trust])
    A_ub = np.vstack([A_ub, A_trust])
    b_ub = np.concatenate([b_ub, b_trust])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), c_max

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 512
        parent_h = np.zeros(n0)
        parent_h[:n0 // 2] = 1.0
    n = len(parent_h)
    best_h = parent_h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n

    # Phase 1: Quick subgradient descent
    if remaining() > 30:
        h_sg, c5_sg = subgradient_descent(best_h, n, min(30.0, remaining() * 0.1), rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 2: SLSQP with increasing beta
    for beta in [500, 2000, 8000, 30000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(best_h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Flow local search (fundamentally different from gradient descent)
    if remaining() > 30:
        h_flow, c5_flow = flow_local_search(best_h, n, max_iter=3000, time_limit=min(50.0, remaining() * 0.3))
        if c5_flow < best_c5:
            best_c5 = c5_flow
            best_h = h_flow.copy()

    # Phase 4: More SLSQP passes
    for beta in [50000, 100000, 200000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(best_h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Cutting plane
    if remaining() > 40:
        for it in range(200):
            if remaining() < 5:
                break
            h_lp, t_lp = cutting_plane_lp(best_h, n, K=30, delta_trust=0.04)
            cur_c5 = compute_c5(best_h)
            if t_lp < cur_c5 - 1e-10:
                best_step = 0
                best_val = cur_c5
                for alpha in [1.0, 0.7, 0.5, 0.3]:
                    h_try = project_feasible(best_h + alpha * (h_lp - best_h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        best_step = alpha
                if best_step > 0:
                    best_h = project_feasible(best_h + best_step * (h_lp - best_h), n)
                    c5_check = compute_c5(best_h)
                    if c5_check < best_c5:
                        best_c5 = c5_check
            elif it > 80:
                break

    # Phase 6: SLSQP + Flow alternating
    for cycle in range(3):
        if remaining() < 15:
            break
        if remaining() > 10:
            h_new = slsqp_smooth(best_h, n, 80000, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        if remaining() > 15:
            h_flow, c5_flow = flow_local_search(best_h, n, max_iter=2000, time_limit=min(20.0, remaining() * 0.3))
            if c5_flow < best_c5:
                best_c5 = c5_flow
                best_h = h_flow.copy()

    # Phase 7: Try higher resolution
    for n_t in [768, 1024]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        h_cur = h_feas.copy()
        c5_cur = c5_int
        # Subgradient
        if remaining() > 20:
            h_sg, c5_sg = subgradient_descent(h_cur, n_t, min(15.0, remaining() * 0.2), rng)
            if c5_sg < c5_cur:
                h_cur = h_sg.copy()
                c5_cur = c5_sg
        # SLSQP
        for beta in [3000, 15000, 60000]:
            if remaining() < 10:
                break
            h_new = slsqp_smooth(h_cur, n_t, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_cur = h_new.copy()
        # Flow
        if remaining() > 15:
            h_flow, c5_flow = flow_local_search(h_cur, n_t, max_iter=1500, time_limit=min(15.0, remaining() * 0.3))
            if c5_flow < c5_cur:
                c5_cur = c5_flow
                h_cur = h_flow.copy()
        # Cutting plane
        if remaining() > 20:
            for it in range(60):
                if remaining() < 5:
                    break
                h_lp, t_lp = cutting_plane_lp(h_cur, n_t, K=20, delta_trust=0.03)
                if t_lp < compute_c5(h_cur) - 1e-10:
                    best_val = compute_c5(h_cur)
                    for alpha in [1.0, 0.5, 0.3]:
                        h_try = project_feasible(h_cur + alpha * (h_lp - h_cur), n_t)
                        c_try = compute_c5(h_try)
                        if c_try < best_val:
                            best_val = c_try
                            h_cur = h_try
                    c5_cur = compute_c5(h_cur)
                elif it > 30:
                    break
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_t

    # Phase 8: Final intensive polish
    n = best_n
    h = best_h.copy()
    
    # Very high beta SLSQP
    for beta in [100000, 300000, 500000]:
        if remaining() < 5:
            break
        h = slsqp_smooth(h, n, beta, maxiter=500)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()

    # Final flow
    if remaining() > 10:
        h_flow, c5_flow = flow_local_search(best_h, n, max_iter=5000, time_limit=min(30.0, remaining() * 0.4))
        if c5_flow < best_c5:
            best_c5 = c5_flow
            best_h = h_flow.copy()

    # Final micro-steps: pairwise coordinate optimization
    if remaining() > 10:
        h = best_h.copy()
        for sweep in range(20):
            if remaining() < 3:
                break
            improved = False
            for i in range(n):
                if remaining() < 2:
                    break
                gi = smooth_max_grad(h, n, 50000.0)[i]
                if gi < -1e-8:
                    # Want to increase h[i]
                    for di in [0.001, 0.005, 0.002, 0.01, 0.0005]:
                        if h[i] + di > 1.0:
                            continue
                        h_try = h.copy()
                        h_try[i] += di
                        # Redistribute from a random other coordinate
                        j = (i + 1) % n
                        if h_try[j] >= di:
                            h_try[j] -= di
                            c_try = compute_c5(h_try)
                            if c_try < best_c5:
                                h = h_try
                                best_c5 = c_try
                                best_h = h.copy()
                                improved = True
                                break
                elif gi > 1e-8:
                    # Want to decrease h[i]
                    for di in [0.001, 0.005, 0.002, 0.01, 0.0005]:
                        if h[i] - di < 0.0:
                            continue
                        h_try = h.copy()
                        h_try[i] -= di
                        j = (i + 1) % n
                        if h_try[j] + di <= 1.0:
                            h_try[j] += di
                            c_try = compute_c5(h_try)
                            if c_try < best_c5:
                                h = h_try
                                best_c5 = c_try
                                best_h = h.copy()
                                improved = True
                                break
            if not improved and sweep > 2:
                break

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
