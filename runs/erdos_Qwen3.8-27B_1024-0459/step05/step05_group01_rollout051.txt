def project_feasible(h, n):

def corr_gradient(k, h, n, sign=1.0):
    """Gradient of correlation at lag k. k is index in full correlation output (0 to 2n-2)."""
    g = np.zeros(n)
    off = k - (n - 1)  # actual shift
    if off >= 0:
        np_ = n - off
        g[off:] += sign * (1.0 - h[:np_])
        g[:np_] -= sign * h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += sign * (1.0 - h[a:])
        g[a:] -= sign * h[:np_]
    return g

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

    for n_t in sorted(n_targets, key=lambda x: abs(x - best_n)):
        if remaining_time() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        t_alloc = min(60, int(remaining_time() * 0.12))
        h_opt, c5_opt = subgradient_nesterov(h_feas, n_t, top_k=6,
                                             max_iters=10000, time_limit=t_alloc)
        h_opt, c5_opt = top_k_descent(h_opt, n_t, top_k=5,
                                      max_iters=500, time_limit=30.0)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

    # Phase 7: Binary simulated annealing + continuous refinement
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(100, int(remaining_time() * 0.2))
        h_b, c5_b = binary_anim(h_bin, best_n, sa_time, rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

        if remaining_time() > 30:
            t_alloc = min(80, int(remaining_time() * 0.25))
            h_r, c5_r = subgradient_nesterov(h_b, best_n, top_k=8,
                                             max_iters=15000, time_limit=t_alloc)
            h_r, c5_r = top_k_descent(h_r, best_n, top_k=6,
                                      max_iters=800, time_limit=40.0)
            if c5_r < best_c5 - 1e-14:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 8: Alternative starting points
    if remaining_time() > 50:
        starts = [thue_morse(best_n), random_binary(best_n, rng)]
        h_shifted = np.roll(best_h, 2)
        starts.append(project_feasible(h_shifted, best_n))
        for h_s in starts:
            if remaining_time() < 25:
                break
            t_alloc = min(50, int(remaining_time() * 0.2))
            h_a, c5_a = subgradient_nesterov(h_s, best_n, top_k=6,
                                             max_iters=10000, time_limit=t_alloc)
            h_a, c5_a = top_k_descent(h_a, best_n, top_k=5,
                                      max_iters=400, time_limit=25.0)
            if c5_a < best_c5 - 1e-14:
                best_c5 = c5_a
                best_h = h_a.copy()

    # Phase 9: Coordinate descent fine-tuning
    if remaining_time() > 20:
        h_cd, c5_cd = coord_descent(best_h, best_n, max_sweeps=10,
                                    time_limit=min(60, int(remaining_time() * 0.2)))
        if c5_cd < best_c5 - 1e-14:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 10: Final SLSQP polish
    if remaining_time() > 20:
        h_final = slsqp_opt(best_h, best_n, 1000000, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 11: Final top-k equalization
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_descent(best_h, best_n, top_k=8,
                                    max_iters=1500, time_limit=30.0)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility and validate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None:
            if eval_c5 < best_c5:
                best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
