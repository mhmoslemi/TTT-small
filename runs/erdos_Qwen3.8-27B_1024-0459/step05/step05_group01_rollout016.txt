import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 1000.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 10.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(150):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    h = np.array(h, dtype=float).flatten()
    n_half = n // 2
    idx = np.argsort(-h)[:n_half]
    s = np.zeros(n)
    s[idx] = 1.0
    return s


def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return to_binary(h, n)


def periodic_pattern(n, period_len):
    s = np.zeros(n)
    per = np.zeros(period_len)
    per[:period_len // 2] = 1.0
    for i in range(n):
        s[i] = per[i % period_len]
    return to_binary(s, n)


def random_balanced(n, rng):

    h[rng.choice(n, n // 2, replace=False)] = 1.0

def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 60.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g

    return np.sum(x) - _CUR_N / 2.0

    return np.ones(_CUR_N)


def slsqp_opt(h, n, beta, maxiter=500):

    _CUR_N = n
    _SM_BETA = beta

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and res.fun < _sm_obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

                g_dir[a:] -= w_k * h[:np_]

                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def coordinate_refine(h, n, max_sweeps=40):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0, old]
            if 0.0 < old < 1.0:
                cands.extend([old * 0.5, old + (1.0 - old) * 0.5,
                              max(0.0, old - 0.2), min(1.0, old + 0.2)])
            cands = [max(0.0, min(1.0, c)) for c in cands]
            cands = list(set(cands))
            cur_c5 = compute_c5(h)
            best_val = old
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < cur_c5 - 1e-13:
                    cur_c5 = c5_v
                    best_val = c
                    improved = True
            h[i] = best_val
            h = project_feasible(h, n)
        if not improved:
            break
        if best_c5 > compute_c5(h):
            best_c5 = compute_c5(h)
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    if _BUDGET > budget_s:
        _BUDGET = budget_s
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = np.zeros(384)
        parent_h[:192] = 1.0
    parent_n = len(parent_h)
    parent_h = project_feasible(parent_h, parent_n)

    best_h = parent_h.copy()
    best_n = parent_n
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP on parent
    h = best_h.copy()
    for beta in [500, 2000, 10000, 50000]:
        if remaining_time() < 15:
            break
        h = slsqp_opt(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()

    h, c5_eq = top_k_equalize(h, best_n, top_k=5, max_iters=300)
    if c5_eq < best_c5 - 1e-14:
        best_c5 = c5_eq
        best_h = h.copy()

    # Phase 2: Binary NEH search (main workhorse)
    # Start from best continuous solution rounded to binary
    s0 = to_binary(best_h, best_n)
    neh_time = max(30, int(remaining_time() * 0.35))
    s_best, c5_best = binary_neh(s0, best_n, neh_time, rng, max_restarts=6, sa_rounds=400)
    if c5_best < best_c5 - 1e-14:
        best_c5 = c5_best
        best_h = project_feasible(s_best, best_n)

    # Phase 2b: NEH from Thue-Morse
    if remaining_time() > 60:
        tm = thue_morse(best_n)
        tm_time = max(20, int(remaining_time() * 0.2))
        s_tm, c5_tm = binary_neh(tm, best_n, tm_time, rng, max_restarts=4, sa_rounds=300)
        if c5_tm < best_c5 - 1e-14:
            best_c5 = c5_tm
            best_h = project_feasible(s_tm, best_n)

    # Phase 2c: NEH from random
    if remaining_time() > 40:
        rb = random_balanced(best_n, rng)
        rb_time = max(15, int(remaining_time() * 0.1))
        s_rb, c5_rb = binary_neh(rb, best_n, rb_time, rng, max_restarts=3, sa_rounds=200)
        if c5_rb < best_c5 - 1e-14:
            best_c5 = c5_rb
            best_h = project_feasible(s_rb, best_n)

    # Phase 3: Continuous refinement of best binary
    h = best_h.copy()
    for beta in [5000, 20000, 100000, 500000, 1000000]:
        if remaining_time() < 10:
            break
        h = slsqp_opt(h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()

    # Top-k equalization with larger k
    for tk in [5, 8, 12, 16]:
        if remaining_time() < 8:
            break
        h, c5_eq = top_k_equalize(best_h, best_n, top_k=tk, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h.copy()

    # Phase 4: Try different resolutions
    n_opts = [best_n - 4, best_n, best_n + 4, best_n + 8, best_n + 16,
              best_n + 32, best_n + 64]
    n_opts = [max(128, min(1024, x)) for x in n_opts]
    n_opts = [x if x % 2 == 0 else x + 1 for x in n_opts]
    n_opts = sorted(set(n_opts), key=lambda x: abs(x - best_n))

    for n_t in n_opts[:5]:
        if remaining_time() < 40:
            break
        if n_t == best_n:
            continue
        h_int = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_int, n_t)
        c5_int = compute_c5(h_feas)

        # Quick NEH
        s_t = to_binary(h_feas, n_t)
        neh_t = max(15, int(remaining_time() * 0.12))
        s_t, c5_t = binary_neh(s_t, n_t, neh_t, rng, max_restarts=2, sa_rounds=200)

        # SLSQP refine
        h_t = project_feasible(s_t, n_t)
        for beta in [5000, 50000, 300000]:
            if remaining_time() < 8:
                break
            h_t = slsqp_opt(h_t, n_t, beta, maxiter=400)
        c5_t = compute_c5(h_t)
        h_t, c5_t = top_k_equalize(h_t, n_t, top_k=6, max_iters=300)

        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 5: Final intensive polish
    h = best_h.copy()
    n = best_n
    for beta in [200000, 1000000, 3000000]:
        if remaining_time() < 8:
            break
        h = slsqp_opt(h, n, beta, maxiter=800)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()

    if remaining_time() > 12:
        h, c5_eq = top_k_equalize(best_h, n, top_k=10, max_iters=600)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h.copy()

    if remaining_time() > 15:
        h, c5_cr = coordinate_refine(best_h, n, max_sweeps=30)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h.copy()

    # Final SLSQP polish
    if remaining_time() > 8:
        h_fin = slsqp_opt(best_h, n, 5000000, maxiter=300)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
