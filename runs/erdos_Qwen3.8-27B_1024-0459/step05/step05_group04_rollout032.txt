```python
def subgradient_step(h, n, top_k=10):
    """Find a direction that reduces the top-k correlations."""
    corr = compute_corr(h)
    # Get the top-k shifts
    top_shifts = np.argsort(-corr)[:top_k]
    
    # Compute gradients for each top shift
    # G is (top_k x n)
    G = np.zeros((top_k, n))
    for idx, k in enumerate(top_shifts):
        G[idx] = corr_gradient(h, k, n)
    
    # Solve: max_g min_{i} G[i]·g s.t. ||g||_2 ≤ 1, and h + eps*g is feasible (for small eps)
    # This is: max_g t s.t. G[i]·g ≥ t for all i, ||g||_2 ≤ 1
    # The constraint ||g||_2 ≤ 1 makes this non-standard for LP.
    # Alternative: just use the average gradient
    g = -np.mean(G, axis=0)
    # Project to feasible direction: sum(g) = 0 (to preserve integral)
    g -= np.mean(g)
    # Also need to respect box constraints (approximately):
    # If h[i] is near 0, don't push g[i] negative; if near 1, don't push g[i] positive
    mask_low = h < 0.05
    mask_high = h > 0.95
    g[mask_low & (g < 0)] = 0
    g[mask_high & (g > 0)] = 0
    g -= np.mean(g)  # re-center
    
    return g
```
