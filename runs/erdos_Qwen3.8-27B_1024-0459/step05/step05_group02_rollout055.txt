import numpy as np
from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def _sm_fun(x):

    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty

    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]

    for beta in [200, 800, 3000, 10000, 30000]:
        if remaining() < 12:
            break
        h1 = lbfgs_pass(h1, n1, beta, top_k=5, maxiter=700)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n1

    if remaining() > 40:
        h_ng, c5_ng = nesterov_opt(h1, n1, top_k=5, max_iters=10000, step_init=0.03)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()
            best_n = n1

    # Binary search at 768
    if remaining() > 80:
        h_bin = (best_h > 0.5).astype(float)
        h_b1, c5_b1 = binary_search_ann(h_bin, n1, max_iters=20000, top_k=5)

        h_ref1 = h_b1.copy()
        for beta in [5000, 20000, 50000]:
            if remaining() < 8:
                break
            h_ref1 = lbfgs_pass(h_ref1, n1, beta, top_k=4, maxiter=400)
        c5_ref1 = compute_c5(h_ref1)

        if c5_ref1 < best_c5:
            best_c5 = c5_ref1
            best_h = h_ref1.copy()
            best_n = n1
        elif c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n1

    # Top equalize at 768
    if remaining() > 20:
        h_eq, c5_eq = top_equalize(best_h, best_n, top_k=6, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 2: Try n=1024
    n2 = 1024
    if remaining() > 150:
        h2 = interp_to(best_h, best_n, n2)
        h2 = proj(h2, n2)

        for beta in [500, 2000, 8000, 25000]:
            if remaining() < 12:
                break
            h2 = lbfgs_pass(h2, n2, beta, top_k=5, maxiter=600)

        if remaining() > 40:
            h_ng2, c5_ng2 = nesterov_opt(h2, n2, top_k=5, max_iters=8000, step_init=0.03)
            if c5_ng2 < compute_c5(h2):
                h2 = h_ng2

        if remaining() > 60:
            h_bin2 = (h2 > 0.5).astype(float)
            h_b2, c5_b2 = binary_search_ann(h_bin2, n2, max_iters=15000, top_k=5)

            h_ref2 = h_b2.copy()
            for beta in [5000, 20000]:
                if remaining() < 8:
                    break
                h_ref2 = lbfgs_pass(h_ref2, n2, beta, top_k=4, maxiter=350)
            c5_ref2 = compute_c5(h_ref2)

            c5_2 = min(compute_c5(h2), c5_b2, c5_ref2)
            if c5_ref2 <= c5_2:
                h_cand = h_ref2
            elif c5_b2 <= c5_2:
                h_cand = h_b2
            else:
                h_cand = h2

            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_cand.copy()
                best_n = n2

    # Phase 3: Try n=512
    n3 = 512
    if remaining() > 100:
        h3 = interp_to(best_h, best_n, n3)
        h3 = proj(h3, n3)

        for beta in [500, 3000, 10000]:
            if remaining() < 10:
                break
            h3 = lbfgs_pass(h3, n3, beta, top_k=5, maxiter=500)

        if remaining() > 50:
            h_bin3 = (h3 > 0.5).astype(float)
            h_b3, c5_b3 = binary_search_ann(h_bin3, n3, max_iters=15000, top_k=4)

            h_ref3 = h_b3.copy()
            for beta in [5000, 15000]:
                if remaining() < 8:
                    break
                h_ref3 = lbfgs_pass(h_ref3, n3, beta, top_k=4, maxiter=300)
            c5_ref3 = compute_c5(h_ref3)

            c5_3 = min(compute_c5(h3), c5_b3, c5_ref3)
            if c5_ref3 <= c5_3:
                h_cand = h_ref3
            elif c5_b3 <= c5_3:
                h_cand = h_b3
            else:
                h_cand = h3

            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h_cand.copy()
                best_n = n3

    # Phase 4: Fourier-based restart at 768
    if remaining() > 120:
        for fs in [1, 7, 42]:
            if remaining() < 40:
                break
            h_f = fourier_init(768, n_modes=3, seed=fs)
            h_f = proj(h_f, 768)

            for beta in [1000, 5000, 20000]:
                if remaining() < 8:
                    break
                h_f = lbfgs_pass(h_f, 768, beta, top_k=5, maxiter=400)

            if remaining() > 30:
                h_bin_f = (h_f > 0.5).astype(float)
                h_bf, c5_bf = binary_search_ann(h_bin_f, 768, max_iters=8000, top_k=5)
                h_rf = h_bf.copy()
                if remaining() > 10:
                    h_rf = lbfgs_pass(h_rf, 768, 10000, top_k=4, maxiter=300)
                c5_rf = compute_c5(h_rf)

                c5_f = min(compute_c5(h_f), c5_bf, c5_rf)
                if c5_rf <= c5_f:
                    h_cand = h_rf
                elif c5_bf <= c5_f:
                    h_cand = h_bf
                else:
                    h_cand = h_f

                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_cand.copy()
                    best_n = 768

    # Phase 5: Final intensive refinement
    n = best_n
    h = best_h.copy()

    if remaining() > 30:
        h_pg, c5_pg = nesterov_opt(h, n, top_k=4, max_iters=8000, step_init=0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    if remaining() > 15:
        h_eq2, c5_eq2 = top_equalize(h, n, top_k=6, max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    # Final binary polish
    if remaining() > 50:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_search_ann(h_bin_f, best_n, max_iters=15000, top_k=4)
        h_rf = h_bf.copy()
        for beta in [10000, 30000, 80000]:
            if remaining() < 5:
                break
            h_rf = lbfgs_pass(h_rf, best_n, beta, top_k=3, maxiter=300)
        c5_rf = compute_c5(h_rf)

        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()
        elif c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
