    change = np.zeros(2*n-1)

        contrib = 0.0

    return change

def slsqp_refine(h, n, beta=50000, maxiter=300):
    """Light SLSQP refinement with soft-min objective."""
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        return (m + np.log(np.sum(ez)) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        g = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:np_]
        return g
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = best_nc
            c5 = float(np.max(corr) * 2.0 / n)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i != j:
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
        if stall > max_stall // 2:
            trial_n = max(100, trial_n // 2)

    return best_h, best_c5


def binary_simulated_annealing(h, n, time_limit, rng):
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * 2.0 / n)
    best_h = h.copy()
    start = time.time()
    T_start = 0.3
    T_end = 0.001

    while time.time() - start < time_limit and remaining_time() > 5:
        frac = (time.time() - start) / time_limit
        T = T_start * (T_end / T_start) ** frac
        T *= max(0.5, 1.0 + 0.15 * rng.randn())
        T = max(T, 0.0005)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        current_max = float(np.max(corr))

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        nc = compute_corr(h)
        nm = float(np.max(nc))
        h[i] = 1.0
        h[j] = 0.0

        delta = nm - current_max
        if delta < -1e-13 or rng.random() < np.exp(-delta / T):
            h[i] = 0.0
            h[j] = 1.0
            corr = nc
            c5 = nm * 2.0 / n
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5


def binary_2swap(h, n, time_limit, rng, n_trials=200):
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * 2.0 / n)
    best_h = h.copy()
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        current_max = float(np.max(corr))
        best_delta = 0.0
        best_mv = None
        best_nc = None

        for _ in range(n_trials):
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            i1, i2 = ones[i1], ones[i2]
            j1, j2 = zeros[j1], zeros[j2]
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            nc = compute_corr(h)
            nm = float(np.max(nc))
            d = nm - current_max
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
            if d < best_delta - 1e-13:
                best_delta = d
                best_mv = (i1, i2, j1, j2)
                best_nc = nc

        if best_mv is not None:
            i1, i2, j1, j2 = best_mv
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            corr = best_nc
            c5 = float(np.max(corr) * 2.0 / n)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            h[ones[i1]] = 0.0
            h[ones[i2]] = 0.0
            h[zeros[j1]] = 1.0
            h[zeros[j2]] = 1.0
            corr = compute_corr(h)

    return best_h, best_c5


def targeted_binary_search(h, n, time_limit, rng, n_trials=300):
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * 2.0 / n)
    best_h = h.copy()
    start = time.time()
    stall = 0

    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        current_max = float(np.max(corr))
        top_idx = np.argsort(-corr)[:8]
        best_delta = 0.0
        best_i = -1
        best_j = -1
        best_nc = None

        nt = min(n_trials, len(ones) * len(zeros))
        for _ in range(nt):
            target_lag = top_idx[rng.randint(len(top_idx))]
            ell = target_lag - (n - 1)

            if ell >= 0:
                valid_is = np.arange(0, n - ell)
            else:
                valid_is = np.arange(-ell, n)

            good_ones = []
            good_zeros = []
            for i in valid_is:
                if ell >= 0:
                    partner = i + ell
                else:
                    partner = i - ell
                if h[i] == 1.0 and h[partner] == 0.0:
                    good_ones.append(i)
                elif h[i] == 0.0 and h[partner] == 1.0:
                    good_zeros.append(i)

            if len(good_ones) > 0:
                i = good_ones[rng.randint(len(good_ones))]
                j = zeros[rng.randint(len(zeros))]
            elif len(good_zeros) > 0:
                j = good_zeros[rng.randint(len(good_zeros))]
                i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            nc = compute_corr(h)
            nm = float(np.max(nc))
            d = nm - current_max
            h[i] = 1.0
            h[j] = 0.0
            if d < best_delta - 1e-13:
                best_delta = d
                best_i = i
                best_j = j
                best_nc = nc

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = best_nc
            c5 = float(np.max(corr) * 2.0 / n)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i != j:
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)

    return best_h, best_c5


def _slsqp_obj(x):
    n = _CUR_N
    corr = compute_corr(x)
    m = np.max(corr)
    z = np.clip(_SM_BETA * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / _SM_BETA) * (2.0 / n)


def _slsqp_grad(x):
    n = _CUR_N
    corr = compute_corr(x)
    m = np.max(corr)
    z = np.clip(_SM_BETA * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    dx = 2.0 / n
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _slsqp_eq(x):
    return np.sum(x) - _CUR_N / 2.0


def _slsqp_eqjac(x):
    return np.ones(_CUR_N)


def slsqp_refine(h, n, beta=50000, maxiter=300):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eqjac}]
    try:
        res = minimize(_slsqp_obj, h0, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < _slsqp_obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return make_binary(h, n)


def periodic_pattern(n, period_len):
    h = np.zeros(n)
    for i in range(n):
        pos = i % period_len
        if pos < period_len // 2:
            h[i] = 1.0
    return make_binary(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = float(budget_s)
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(384)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive binary search at parent resolution
    h_bin = make_binary(best_h, best_n)

    t1 = min(250, remaining_time() * 0.25)
    h1, c5_1 = binary_best_improvement(h_bin, best_n, t1, rng, n_trials=600)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = project_feasible(h1, best_n)

    if remaining_time() > 50:
        t2 = min(150, remaining_time() * 0.18)
        h2, c5_2 = binary_simulated_annealing(h1, best_n, t2, rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = project_feasible(h2, best_n)

    if remaining_time() > 50:
        t3 = min(150, remaining_time() * 0.18)
        h3, c5_3 = binary_2swap(h1, best_n, t3, rng, n_trials=250)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = project_feasible(h3, best_n)

    if remaining_time() > 40:
        t4 = min(120, remaining_time() * 0.15)
        h4, c5_4 = targeted_binary_search(best_h, best_n, t4, rng, n_trials=500)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = project_feasible(h4, best_n)

    # Phase 2: Multi-resolution exploration
    n_list = []
    for d in [-64, -48, -32, -16, 0, 16, 32, 48, 64, 96, 128]:
        nn = best_n + d
        if 128 <= nn <= 900 and nn % 2 == 0:
            n_list.append(nn)
    for mult in [0.75, 0.85, 1.15, 1.3, 1.5, 1.75, 2.0]:
        nn = int(round(best_n * mult))
        if 128 <= nn <= 900 and nn % 2 == 0:
            n_list.append(nn)
    n_list = sorted(set(n_list))
    n_list.sort(key=lambda x: abs(x - best_n))

    for n_t in n_list[:7]:
        if remaining_time() < 80:
            break
        h_t = interp_to(best_h, best_n, n_t)
        h_t_bin = make_binary(h_t, n_t)

        t5 = min(120, remaining_time() * 0.3)
        h_t1, c5_t1 = binary_best_improvement(h_t_bin, n_t, t5, rng, n_trials=400)

        c5_t = c5_t1
        h_t_best = h_t1
        if remaining_time() > 40:
            t6 = min(60, remaining_time() * 0.2)
            h_t2, c5_t2 = binary_2swap(h_t1, n_t, t6, rng, n_trials=150)
            if c5_t2 < c5_t:
                c5_t = c5_t2
                h_t_best = h_t2
        if remaining_time() > 30:
            t7 = min(40, remaining_time() * 0.2)
            h_t3, c5_t3 = targeted_binary_search(h_t_best, n_t, t7, rng, n_trials=300)
            if c5_t3 < c5_t:
                c5_t = c5_t3
                h_t_best = h_t3

        # Light SLSQP polish at this resolution
        if remaining_time() > 20:
            h_ts = slsqp_refine(h_t_best, n_t, beta=30000, maxiter=200)
            c5_ts = compute_c5(h_ts)
            if c5_ts < c5_t:
                c5_t = c5_ts
                h_t_best = h_ts

        if c5_t < best_c5 - 1e-14:
            best_c5 = c5_t
            best_h = h_t_best.copy()
            best_n = n_t

    # Phase 3: Alternative starting points
    alt_ns = [best_n, best_n + 32 if best_n + 32 <= 900 and (best_n+32)%2==0 else best_n - 32 if best_n - 32 >= 128 else best_n]
    for n_alt in set(alt_ns):
        if remaining_time() < 80:
            break
        # Thue-Morse start
        h_tm = theta_morse(n_alt)
        t8 = min(80, remaining_time() * 0.25)
        h_tm1, c5_tm = binary_best_improvement(h_tm, n_alt, t8, rng, n_trials=400)
        if remaining_time() > 30:
            h_tm2, c5_tm2 = binary_2swap(h_tm1, n_alt, min(40, remaining_time()*0.2), rng)
            if c5_tm2 < c5_tm:
                c5_tm = c5_tm2
                h_tm1 = h_tm2
        if c5_tm < best_c5 - 1e-14:
            h_tmf = slsqp_refine(h_tm1, n_alt, beta=50000, maxiter=200)
            c5_tmf = compute_c5(h_tmf)
            if c5_tmf < c5_tm:
                c5_tm = c5_tmf
                h_tm1 = h_tmf
            best_c5 = c5_tm
            best_h = h_tm1.copy()
            best_n = n_alt

        if remaining_time() < 60:
            break
        # Periodic pattern
        for per in [8, 12, 16, 24, 32]:
            if remaining_time() < 40:
                break
            h_pp = periodic_pattern(n_alt, per)
            t9 = min(50, remaining_time() * 0.2)
            h_pp1, c5_pp = binary_best_improvement(h_pp, n_alt, t9, rng, n_trials=350)
            if c5_pp < best_c5 - 1e-14:
                h_ppf = slsqp_refine(h_pp1, n_alt, beta=50000, maxiter=200)
                c5_ppf = compute_c5(h_ppf)
                if c5_ppf < c5_pp:
                    c5_pp = c5_ppf
                    h_pp1 = h_ppf
                best_c5 = c5_pp
                best_h = h_pp1.copy()
                best_n = n_alt

    # Phase 4: Final intensive refinement at best resolution
    if remaining_time() > 30:
        h_f_bin = make_binary(best_h, best_n)
        t10 = min(200, remaining_time() * 0.35)
        h_f1, c5_f1 = binary_best_improvement(h_f_bin, best_n, t10, rng, n_trials=700)
        if c5_f1 < best_c5:
            best_c5 = c5_f1
            best_h = project_feasible(h_f1, best_n)

    if remaining_time() > 30:
        h_f_bin2 = make_binary(best_h, best_n)
        t11 = min(100, remaining_time() * 0.25)
        h_f2, c5_f2 = targeted_binary_search(h_f_bin2, best_n, t11, rng, n_trials=500)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = project_feasible(h_f2, best_n)

    if remaining_time() > 20:
        h_f_bin3 = make_binary(best_h, best_n)
        t12 = min(60, remaining_time() * 0.2)
        h_f3, c5_f3 = binary_2swap(h_f_bin3, best_n, t12, rng, n_trials=200)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = project_feasible(h_f3, best_n)

    # Phase 5: SLSQP final polish
    if remaining_time() > 12:
        h_pol = slsqp_refine(best_h, best_n, beta=100000, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    if remaining_time() > 8:
        h_pol2 = slsqp_refine(best_h, best_n, beta=300000, maxiter=300)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2

    # Final feasibility and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
