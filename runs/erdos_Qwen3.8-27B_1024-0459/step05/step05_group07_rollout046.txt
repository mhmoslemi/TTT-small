import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14 and n > 0:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            idx = int(np.argmin(np.abs(h - 0.5)))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def pm_value(x, n, p):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    return np.mean(corr**p) ** (1.0/p)

def pm_grad(x, n, p):
    """Gradient of power-mean objective."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    v = corr * dx  # v_k = corr_k * dx
    ncorr = 2 * n - 1
    S = np.mean(v**p)
    pm = S ** (1.0/p)
    
    # d(pm)/d(v_k) = (1/p) * pm^(1-p) * p * v_k^(p-1) / ncorr = pm^(1-p) * v_k^(p-1) / ncorr
    weights = pm**(1-p) * np.power(v, p-1) / ncorr
    # d(v_k)/d(x_j) = dx * d(corr_k)/d(x_j)
    # So d(pm)/d(x_j) = sum_k weights[k] * dx * d(corr_k)/d(x_j)
    
    grad = np.zeros(n)
    # Vectorized: for each shift k, the gradient contribution is:
    # weights[k] * dx * [(1-x[j+k]) - x[j-k]] (valid indices only)
    
    for k in range(ncorr):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        offset = k - (n-1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * x[:npairs]
    return pm, grad
