                   he[n + best_b + k_vals] - he[n + best_b - k_vals])
    h[best_a] = 0.0
    h[best_b] = 1.0
    corr = corr + delta_final
    c_max = float(corr.max())
    return h, corr, c_max, True


def binary_local_search(h, n, k_vals, max_passes, time_func, min_time):
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    c_max = float(corr.max())
    for _ in range(max_passes):
        if time_func() < min_time:
            break
        h, corr, c_max, improved = binary_one_pass(h, corr, c_max, n, k_vals)
        if not improved:
            break
    return h, corr, c_max


def binary_ils(h_init, n, time_func, min_time, max_rounds=300):
    k_vals = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c_max = float(np.correlate(h, 1 - h, mode='full').max())
    rng = np.random.RandomState(12345)
    for rnd in range(max_rounds):
        if time_func() < min_time + 20:
            break
        h, corr, c_max = binary_local_search(h, n, k_vals, max_passes=20000,
                                             time_func=time_func, min_time=min_time)
        if c_max < best_c_max:
            best_c_max = c_max
            best_h = h.copy()
        if time_func() < min_time + 20:
            break
        n_pert = rng.randint(2, 8)
        h_cert = h.copy()
        cmax_cert = c_max
        for _ in range(n_pert):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
        corr_new = np.correlate(h, 1 - h, mode='full').astype(np.float64)
        cmax_new = float(corr_new.max())
        if cmax_new > best_c_max + 20:
            h = h_cert.copy()
            c_max = cmax_cert
    return best_h, best_c_max * dx


def sa_binary(h_init, n, time_func, min_time, T_start=4.0, T_end=0.05, total_steps=3000000):
    k_vals = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = h_init.copy()
    he = np.zeros(4 * n, dtype=np.float64)
    he[n:n + n] = h
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    c_max = float(corr.max())
    best_h = h.copy()
    best_c_max = c_max
    rng = np.random.RandomState(67890)
    step = 0
    ls_counter = 0
    while step < total_steps and time_func() > min_time:
        frac = min(1.0, step / total_steps)
        T = T_start * (T_end / T_start) ** frac
        a_idx = rng.randint(n // 2)
        b_idx = rng.randint(n // 2)
        ones = np.where(he[n:n + n] == 1)[0]
        zeros = np.where(he[n:n + n] == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[a_idx % len(ones)]
        b = zeros[b_idx % len(zeros)]
        delta = (he[n + a + k_vals] + he[n + a - k_vals] -
                 he[n + b + k_vals] - he[n + b - k_vals])
        new_c_max = float((corr + delta).max())
        delta_max = new_c_max - c_max
        if delta_max <= 0 or rng.random() < np.exp(-delta_max / max(T, 1e-10)):
            h[a] = 0.0
            h[b] = 1.0
            he[n + a] = 0.0
            he[n + b] = 1.0
            corr = corr + delta
            c_max = new_c_max
            if c_max < best_c_max:
                best_c_max = c_max
                best_h = h.copy()
        step += 1
        ls_counter += 1
        if ls_counter >= 50000:
            ls_counter = 0
            if time_func() > min_time + 30:
                h, corr, c_max = binary_local_search(h, n, k_vals, max_passes=200,
                                                     time_func=time_func, min_time=min_time)
                he = np.zeros(4 * n, dtype=np.float64)
                he[n:n + n] = h
                if c_max < best_c_max:
                    best_c_max = c_max
                    best_h = h.copy()
    return best_h, best_c_max * dx


def continuous_refine(h_init, n, time_func, min_time):
    k_vals = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    he = np.zeros(4 * n, dtype=np.float64)
    he[n:n + n] = h
    corr = np.correlate(h, 1 - h, mode='full').astype(np.float64)
    c_max = float(corr.max())
    momentum = np.zeros(n)
    mu = 0.7
    step_size = 0.3
    for it in range(50000):
        if time_func() < min_time:
            break
        top_k = 5
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w = (c_top - c_top.min() + 1e-10) ** 2
        w /= w.sum()
        g = np.zeros(n)
        for idx, wk in zip(top_idx, w):
            k_shift = int(idx) - (n - 1)
            if k_shift >= 0:
                np_ = n - k_shift
                g[k_shift:k_shift + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[k_shift:k_shift + np_]
            else:
                a_off = -k_shift
                np_ = n - a_off
                g[:np_] += wk * (1.0 - h[a_off:a_off + np_])
                g[a_off:a_off + np_] -= wk * h[:np_]
        g *= dx
        g -= np.mean(g)
        momentum = mu * momentum - step_size * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        step_size *= 0.9997
        if it % 1000 == 0:
            mu = min(0.92, mu + 0.02)
    return best_h, best_c5


def slsqp_smooth(h_init, n, beta, maxiter, time_func, min_time):
    h0 = project_feasible(h_init, n)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        def _obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(c)
            sh = np.clip(beta * (c - m), -50.0, 50.0)
            ex = np.exp(sh)
            return (m + np.log(np.sum(ex)) / beta) * dx

        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        if res.x is not None and res.fun < _obj(h0):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_n = parent_n
        best_c5 = compute_c5(best_h)
    except Exception:
        parent_n = 384
        best_n = parent_n
        rng0 = np.random.RandomState(seed)
        best_h = rng0.rand(parent_n)
        best_h = project_feasible(best_h, parent_n)
        best_c5 = compute_c5(best_h)

    n_base = best_n

    # Phase 1: Binary ILS at base resolution
    h_bin = round_to_binary(best_h, n_base)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    if remaining() > 60:
        h_b, c5_b = binary_ils(h_bin, n_base, remaining, 30, max_rounds=500)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 2: SA at base resolution
    if remaining() > 100:
        h_sa, c5_sa = sa_binary(h_bin, n_base, remaining, 60)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        # Local search from SA result
        if remaining() > 60:
            k_vals = np.arange(-(n_base - 1), n_base)
            h_ls, corr_ls, cmax_ls = binary_local_search(h_sa, n_base, k_vals,
                                                         max_passes=50000,
                                                         time_func=remaining, min_time=30)
            c5_ls = cmax_ls * (2.0 / n_base)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()

    # Phase 3: Try higher resolution (2x)
    n_2x = n_base * 2
    if n_2x <= 1024 and remaining() > 120:
        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up, n_2x)
        h_up_bin = round_to_binary(h_up, n_2x)
        c5_up = compute_c5(h_up_bin)
        if remaining() > 80:
            h_up_b, c5_up_b = binary_ils(h_up_bin, n_2x, remaining, 40, max_rounds=200)
            if c5_up_b < c5_up:
                c5_up = c5_up_b
                h_up = h_up_b.copy()
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_2x

    # Phase 4: Try 3x resolution
    n_3x = n_base * 3
    if n_3x % 2 == 0 and n_3x <= 1536 and remaining() > 100:
        h_interp = interp_to(best_h, best_n, n_3x)
        h_3x_bin = round_to_binary(h_interp, n_3x)
        if remaining() > 60:
            h_3x_b, c5_3x = binary_ils(h_3x_bin, n_3x, remaining, 40, max_rounds=150)
            if c5_3x < best_c5:
                best_c5 = c5_3x
                best_h = h_3x_b.copy()
                best_n = n_3x

    # Phase 5: Continuous refinement at best resolution
    if remaining() > 30:
        h_cont, c5_cont = continuous_refine(best_h, best_n, remaining, 15)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 6: SLSQP from best binary
    if remaining() > 40:
        h_bin_now = round_to_binary(best_h, best_n)
        for beta in [5000, 20000]:
            if remaining() < 20:
                break
            h_slq = slsqp_smooth(h_bin_now, best_n, beta, maxiter=300,
                                 time_func=remaining, min_time=10)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()
                h_bin_now = h_slq.copy()

    # Phase 7: Final binary polish at best resolution
    if remaining() > 40:
        h_bin_final = round_to_binary(best_h, best_n)
        k_vals_f = np.arange(-(best_n - 1), best_n)
        h_f, corr_f, cmax_f = binary_local_search(h_bin_final, best_n, k_vals_f,
                                                  max_passes=100000,
                                                  time_func=remaining, min_time=15)
        c5_f = cmax_f * (2.0 / best_n)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        # Then continuous refine from there
        if remaining() > 20:
            h_cf, c5_cf = continuous_refine(h_f, best_n, remaining, 10)
            if c5_cf < best_c5:
                best_c5 = c5_cf
                best_h = h_cf.copy()

    # Phase 8: Multi-start binary ILS with random initializations
    if remaining() > 60:
        for trial in range(3):
            if remaining() < 30:
                break
            rng_t = np.random.RandomState(seed + 100 * trial)
            h_rand = rng_t.rand(best_n)
            h_rand_bin = round_to_binary(h_rand, best_n)
            c5_r = compute_c5(h_rand_bin)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_rand_bin.copy()
            if remaining() > 30:
                h_r_b, c5_r_b = binary_ils(h_rand_bin, best_n, remaining, 20, max_rounds=100)
                if c5_r_b < best_c5:
                    best_c5 = c5_r_b
                    best_h = h_r_b.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
