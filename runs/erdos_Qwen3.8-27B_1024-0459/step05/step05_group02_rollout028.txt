        g -= np.mean(g)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        score_one = -g[ones]
        score_zero = g[zeros]

        n_try_o = min(10, len(ones))
        n_try_z = min(10, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try_o]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        best_delta = 0.0
        best_swap = None
        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                nc = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            cur_c5 += best_delta
        else:
            # Try random pairs
            for _ in range(4):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                nc = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
            if best_swap is not None:
                i, j = best_swap
                h[i] = 0.0
                h[j] = 1.0
                cur_c5 += best_delta
            else:
                stuck_count += 10

    return best_h, best_c5


def subbinary_refine(h, n, max_iters=2000, top_k=4, rng=None):
    """Refine near-binary solution by allowing small deviations from 0/1."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        cur_c5 = float(np.max(corr)) * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h, ki, n)
        g -= np.mean(g)

        # Find positions near 0 or 1 where gradient suggests moving inward
        # Positions near 1: if g[i] < 0, decreasing h[i] helps
        # Positions near 0: if g[i] > 0, increasing h[i] helps
        near_one = np.where((h > 0.05) & (h < 0.95))[0]
        if len(near_one) == 0:
            # All are very close to 0 or 1, try to create some intermediate values
            at_one = np.where(h > 0.5)[0]
            at_zero = np.where(h < 0.5)[0]
            # Try to create a new intermediate value at the most beneficial position
            if len(at_one) > 0 and len(at_zero) > 0:
                cand_decrease = at_one[np.argsort(-g[at_one])[:5]]
                cand_increase = at_zero[np.argsort(-g[at_zero])[:5]]
                for i in cand_decrease:
                    h_try = h.copy()
                    h_try[i] = 0.8
                    h_try = proj(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        h = h_try
                        best_c5 = c5_try
                        break
                for j in cand_increase:
                    h_try = h.copy()
                    h_try[j] = 0.2
                    h_try = proj(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        h = h_try
                        best_c5 = c5_try
                        break

            continue

        # For positions with intermediate values, try small adjustments
        improved = False
        for i in near_one:
            if g[i] < -0.01:
                h_try = h.copy()
                h_try[i] = max(0.0, h[i] - 0.05)
                h_try = proj(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            elif g[i] > 0.01:
                h_try = h.copy()
                h_try[i] = min(1.0, h[i] + 0.05)
                h_try = proj(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break

        if not improved:
            # Try L-BFGS step from current point
            h_new = lbfgs_pass(h, n, beta=5000, top_k=4, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
            else:
                break

    return best_h, best_c5


def top_equalize(h, n, top_k=5, max_iters=500):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-13:
            break
        d /= dn

        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.04, 0.001, 0.03, 0.015]:
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for alpha in [0.01, 0.02, 0.005]:
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def full_pipeline_at_resolution(h_start, n, rng, time_budget_frac=1.0):
    """Run a full optimization pipeline at a given resolution."""
    t_start = time.time()
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)

    def time_ok():
        return (time.time() - t_start) < time_budget_frac * _BUDGET - 10 and remaining() > 15

    # Step 1: Quick L-BFGS
    for beta in [1000, 5000, 20000]:
        if not time_ok():
            break
        h_new = lbfgs_pass(best_h, n, beta, top_k=5, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Step 2: Binary threshold and SA search
    h_bin = (best_h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        sort_idx = np.argsort(h_bin)
        for idx in sort_idx:
            if h_bin[idx] > 0.5:
                h_bin[idx] = 0.0
                n_ones -= 1
                if n_ones == n // 2:
                    break
    elif n_ones < n // 2:
        sort_idx = np.argsort(-h_bin)
        for idx in sort_idx:
            if h_bin[idx] < 0.5:
                h_bin[idx] = 1.0
                n_ones += 1
                if n_ones == n // 2:
                    break

    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # SA search
    if time_ok():
        h_sa, c5_sa = binary_sa_search(h_bin, n, max_iters=4000, top_k=3, T_init=0.004, T_min=0.0001, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_bin = h_sa.copy()

    # Greedy search
    if time_ok():
        h_g, c5_g = binary_greedy_search(h_bin, n, max_iters=3000, top_k=3, rng=rng)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            h_bin = h_g.copy()

    # Step 3: Sub-binary refinement
    if time_ok():
        h_sub, c5_sub = subbinary_refine(best_h, n, max_iters=1500, top_k=4, rng=rng)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Step 4: Top equalization
    if time_ok():
        h_eq, c5_eq = top_equalize(best_h, n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Step 5: Final L-BFGS
    for beta in [10000, 30000, 80000]:
        if not time_ok():
            break
        h_new = lbfgs_pass(best_h, n, beta, top_k=4, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Step 6: Final SA from refined solution
    if time_ok():
        h_bin2 = (best_h > 0.5).astype(float)
        n_ones2 = int(np.sum(h_bin2))
        if n_ones2 != n // 2:
            sort_idx = np.argsort(-best_h)
            h_bin2 = np.zeros(n)
            h_bin2[sort_idx[:n // 2]] = 1.0
        h_sa2, c5_sa2 = binary_sa_search(h_bin2, n, max_iters=2000, top_k=4, T_init=0.002, T_min=0.00005, rng=rng)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        # Refine
        if time_ok():
            h_r, c5_r = subbinary_refine(best_h, n, max_iters=1000, top_k=4, rng=rng)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Full pipeline at parent resolution
    h1, c51 = full_pipeline_at_resolution(best_h, best_n, rng, time_budget_frac=0.40)
    if c51 < best_c5:
        best_c5 = c51
        best_h = h1.copy()

    # Phase 2: Try higher resolutions
    n_candidates = [best_n + 2, best_n + 4, best_n + 8]
    # Also try some structured resolutions
    for nn in [512, 640, 768]:
        if nn not in n_candidates and abs(nn - best_n) <= 200:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))
    n_candidates = [nn for nn in n_candidates if nn >= 256 and nn <= 1024 and nn % 2 == 0 and nn != best_n]

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)
        # Also try a binary start at this resolution
        h_bin_start = (h_feas > 0.5).astype(float)
        n_ones_t = int(np.sum(h_bin_start))
        if n_ones_t > n_t // 2:
            sort_idx = np.argsort(h_bin_start)
            for idx in sort_idx:
                if h_bin_start[idx] > 0.5:
                    h_bin_start[idx] = 0.0
                    n_ones_t -= 1
                    if n_ones_t == n_t // 2:
                        break
        elif n_ones_t < n_t // 2:
            sort_idx = np.argsort(-h_bin_start)
            for idx in sort_idx:
                if h_bin_start[idx] < 0.5:
                    h_bin_start[idx] = 1.0
                    n_ones_t += 1
                    if n_ones_t == n_t // 2:
                        break

        # Use the better start
        c5_binary = compute_c5(h_bin_start)
        c5_interp = compute_c5(h_feas)
        if c5_binary < c5_interp:
            h_start = h_bin_start
        else:
            h_start = h_feas

        h_t, c5_t = full_pipeline_at_resolution(h_start, n_t, rng, time_budget_frac=0.15)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 3: Intensive local search at best resolution
    n = best_n
    h = best_h.copy()

    # Multiple SA restarts with different temperatures
    h_bin_cur = (h > 0.5).astype(float)
    n_ones_cur = int(np.sum(h_bin_cur))
    if n_ones_cur != n // 2:
        sort_idx = np.argsort(-h)
        h_bin_cur = np.zeros(n)
        h_bin_cur[sort_idx[:n // 2]] = 1.0

    for T_init in [0.006, 0.003, 0.001, 0.008]:
        if remaining() < 20:
            break
        h_sa, c5_sa = binary_sa_search(h_bin_cur, n, max_iters=3000, top_k=4, T_init=T_init, T_min=0.00005, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_bin_cur = h_sa.copy()
            h = h_sa.copy()

        # Sub-binary refine after SA
        if remaining() > 15:
            h_sub, c5_sub = subbinary_refine(h, n, max_iters=800, top_k=4, rng=rng)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
                h = h_sub.copy()

    # Phase 4: Final intensive L-BFGS with very high beta
    for beta in [20000, 50000, 100000, 200000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(best_h, best_n, beta, top_k=3, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Final top equalize and polish
    if remaining() > 10:
        h_eq, c5_eq = top_equalize(best_h, best_n, top_k=6, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 6: One more round of binary search + refinement
    if remaining() > 25:
        h_bin_final = (best_h > 0.5).astype(float)
        n_ones_f = int(np.sum(h_bin_final))
        if n_ones_f != best_n // 2:
            sort_idx = np.argsort(-best_h)
            h_bin_final = np.zeros(best_n)
            h_bin_final[sort_idx[:best_n // 2]] = 1.0

        h_gf, c5_gf = binary_greedy_search(h_bin_final, best_n, max_iters=5000, top_k=4, rng=rng)
        if c5_gf < best_c5:
            best_c5 = c5_gf
            best_h = h_gf.copy()

        if remaining() > 10:
            h_sf, c5_sf = subbinary_refine(best_h, best_n, max_iters=1000, top_k=5, rng=rng)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
