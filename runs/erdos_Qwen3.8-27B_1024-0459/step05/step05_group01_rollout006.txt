                np_ = n - off
                g[off:] += (1.0 - h[:np_])
                g[:np_] -= h[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += (1.0 - h[a:])
                g[a:] -= h[:np_]
        g /= len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        d = -g / gn
        # Backtracking line search
        improved = False
        for trial in range(15):
            step_try = step * (0.5 ** trial)
            ht = project_feasible(h + step_try * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = ht.copy()
                improved = True
                break
            elif cn < compute_c5(h) - 1e-14:
                h = ht
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < 1e-6:
                break
        else:
            step = min(step * 1.1, 2.0)
    return best_h, best_c5


def top_k_equalize(h, n, top_k=6, max_iters=800):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        active_range = c_max - c_min_top
        for pos, k in enumerate(idx_top):
            w_k = (c_max - corr[k]) / (active_range * top_k) + 1.0 / top_k
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        for step in [0.003, 0.005, 0.01, 0.02, 0.04, 0.001, 0.008, 0.015, 0.06, 0.1]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def binary_simulated_annealing(h, n, time_limit=180, rng=None):
    """Simulated annealing for binary solutions with structured moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    start = time.time()
    T = 0.15
    T_min = 0.001
    cooling = 0.995
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5

    while time.time() - start < time_limit and remaining_time() > 5:
        if T < T_min:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Choose move type
        move_type = rng.random()
        if move_type < 0.5:
            # Single pair swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        elif move_type < 0.75:
            # Structured swap: swap at offset m
            m = rng.randint(1, n // 2)
            i = ones[rng.randint(len(ones))]
            j = (i + m) % n
            if h[j] == 0.0:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                delta = new_c5 - cur_c5
                if delta < 0 or rng.random() < np.exp(-delta / T):
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        else:
            # Block swap
            block_len = rng.randint(1, max(2, n // 64))
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            # Swap h[i] with h[j]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0

        T *= cooling

    # Local search polish
    for _ in range(200):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_trials = min(500, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h.copy()
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return best_h, best_c5


def multi_pair_local_search(h, n, time_limit=120, rng=None):
    """Try multiple simultaneous pair swaps for faster convergence."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0

    while time.time() - start < time_limit and remaining_time() > 5:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        # Greedy: find best single swap
        best_delta = 1e-10
        best_pair = None
        n_trials = min(400, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            delta = c_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)
            h[i] = 1.0
            h[j] = 0.0

        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
            improved = True
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
            # Random kick
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()

    return best_h, best_c5


def coordinate_refine(h, n, max_sweeps=20):
    """Refine by trying to optimize each coordinate."""
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved_sweep = False
        corr_cache = compute_corr(h)
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0]
            if 0.1 < old < 0.9:
                cands.extend([0.1, 0.25, 0.5, 0.75, 0.9])
            cands.append(min(1.0, max(0.0, old + 0.1)))
            cands.append(min(1.0, max(0.0, old - 0.1)))

            best_cand_c5 = compute_c5(h)
            best_cand_val = old
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-12:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
            if best_cand_c5 < best_c5:
                best_c5 = best_cand_c5
        if not improved_sweep:
            break
    return h, best_c5


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def random_balanced(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def periodic_pattern(n, period, offset=0):
    """Create a periodic 0-1 pattern with given period and offset."""
    h = np.zeros(n)
    pat = np.zeros(period)
    n_ones_pat = period // 2
    for i in range(n_ones_pat):
        pat[(i * 2) % period] = 1.0
    h[:] = pat[:n % period] if n % period > 0 else pat
    # Repeat
    for i in range(n):
        h[i] = pat[(i + offset) % period]
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_n = 384
        best_h = np.zeros(best_n)
        best_h[::2] = 1.0

    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with progressive soft-max at parent resolution
    h = best_h.copy()
    for beta in [200, 1000, 5000, 20000, 100000, 500000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization
    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=6, max_iters=800)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: Subgradient method
    if remaining_time() > 30:
        h_sg, c5_sg = subgrad_opt(h, best_n, max_iters=3000, time_limit=80)
        if c5_sg < best_c5 - 1e-14:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 4: More SLSQP after equalization
    for beta in [200000, 500000, 2000000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        if remaining_time() > 10:
            h_eq2, c5_eq2 = top_k_equalize(h, best_n, top_k=8, max_iters=400)
            if c5_eq2 < best_c5 - 1e-14:
                best_c5 = c5_eq2
                best_h = h_eq2.copy()
                h = h_eq2.copy()

    # Phase 5: Binary simulated annealing
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        if int(np.sum(h_bin)) != best_n // 2:
            sort_idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[sort_idx[:best_n // 2]] = 1.0

        sa_time = min(150, max(20, int(remaining_time() * 0.25)))
        h_b, c5_b = binary_simulated_annealing(h_bin, best_n, time_limit=sa_time, rng=rng)

        # Refine binary result continuously
        h_b_c = project_feasible(h_b, best_n)
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 10:
                break
            h_b_c = slsqp_opt(h_b_c, best_n, beta, maxiter=400)
        c5_b_c = compute_c5(h_b_c)
        h_b_c, c5_b_c = top_k_equalize(h_b_c, best_n, top_k=6, max_iters=300)

        if c5_b_c < best_c5 - 1e-14:
            best_c5 = c5_b_c
            best_h = h_b_c.copy()
            h = h_b_c.copy()
        if c5_b < best_c5 - 1e-14:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 6: Multi-resolution search
    if remaining_time() > 80:
        n_candidates = [512, 640, 768]
        for n_t in n_candidates:
            if remaining_time() < 40:
                break
            h_interp = interp_to(best_h, best_n, n_t)
            h_feas = project_feasible(h_interp, n_t)
            c5_int = compute_c5(h_feas)

            h_opt = h_feas
            for beta in [1000, 10000, 100000]:
                if remaining_time() < 15:
                    break
                h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=300)
            c5_opt = compute_c5(h_opt)
            h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=5, max_iters=200)

            if c5_opt < best_c5 - 1e-14:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t
                h = h_opt.copy()

    # Phase 7: Alternative starting points
    alts = []
    alts.append(theta_morse(best_n))
    alts.append(random_balanced(best_n, rng))
    alts.append(periodic_pattern(best_n, 8, 0))
    alts.append(periodic_pattern(best_n, 12, 1))

    for h_alt in alts:
        if remaining_time() < 40:
            break
        h_opt = h_alt
        for beta in [5000, 50000, 500000]:
            if remaining_time() < 10:
                break
            h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=300)
        c5_opt = compute_c5(h_opt)
        h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=6, max_iters=300)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 8: Multi-pair local search on best binary
    if remaining_time() > 40:
        h_bin2 = (best_h > 0.5).astype(float)
        if int(np.sum(h_bin2)) != best_n // 2:
            sort_idx = np.argsort(-h_bin2)
            h_bin2 = np.zeros(best_n)
            h_bin2[sort_idx[:best_n // 2]] = 1.0
        h_mp, c5_mp = multi_pair_local_search(h_bin2, best_n, time_limit=min(80, int(remaining_time() * 0.3)), rng=rng)
        if c5_mp < best_c5 - 1e-14:
            best_c5 = c5_mp
            best_h = h_mp.copy()
        # Continuous refinement of binary result
        h_mp_c = project_feasible(h_mp, best_n)
        for beta in [10000, 100000]:
            if remaining_time() < 10:
                break
            h_mp_c = slsqp_opt(h_mp_c, best_n, beta, maxiter=300)
        h_mp_c, c5_mp_c = top_k_equalize(h_mp_c, best_n, top_k=5, max_iters=200)
        if c5_mp_c < best_c5 - 1e-14:
            best_c5 = c5_mp_c
            best_h = h_mp_c.copy()

    # Phase 9: Final intensive refinement
    n = best_n
    h = best_h.copy()
    for beta in [500000, 2000000, 5000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=10, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 10: Coordinate refinement
    if remaining_time() > 15:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=15)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Final polish
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 5000000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with evaluator
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
