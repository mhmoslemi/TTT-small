import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def compute_C_vector(h, i, n):
    """Compute the C vector for coordinate i."""
    ncorr = 2 * n - 1
    C = np.zeros(ncorr)
    # A part: contribution from h[i] as the first factor
    # h[i] appears in corr[m] when j+offset_m = i, i.e., j = i - offset_m
    # offset_m = m - (n-1), so j = i - m + n - 1
    # Valid when 0 <= j < n, i.e., 0 <= i - m + n - 1 < n
    # => n-1 <= m <= n-1+i (from i-m+n-1 >= 0 => m <= i+n-1)
    # => i <= m (from i-m+n-1 < n => m > i-1 => m >= i)
    a_lo = max(0, i)
    a_hi = min(2*n-2, i + n - 1)
    if a_hi >= a_lo:
        m_vals = np.arange(a_lo, a_hi + 1)
        j_vals = i - m_vals + n - 1
        valid = (j_vals >= 0) & (j_vals < n)
        C[m_vals[valid]] += (1.0 - h[j_vals[valid]])
    
    # B part: contribution from h[i] as the second factor (in 1-h[j])
    # h[i] appears in (1-h[j]) when j = i, so corr[m] includes h[j+offset_m]*(1-h[i])
    # i.e., h[i + offset_m] * (1-h[i]) where offset_m = m-(n-1)
    # So the term is h[i + m - n + 1] * (1-h[i])
    # When h[i] changes by delta, this term changes by -delta * h[i+m-n+1]
    # Valid when 0 <= i + m - n + 1 < n
    # => n-1-i <= m <= 2n-2-i
    b_lo = max(0, n - 1 - i)
    b_hi = min(2*n-2, 2*n - 2 - i)
    if b_hi >= b_lo:
        m_vals = np.arange(b_lo, b_hi + 1)
        j_vals = i + m_vals - n + 1
        valid = (j_vals >= 0) & (j_vals < n)
        C[m_vals[valid]] -= h[j_vals[valid]]
    
    return C

def golden_section_1d(corr, C, t0, n, max_iter=60):
    """Minimize g(t) = max(corr + (t-t0)*C) over t in [0,1]."""
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    # Quick check at endpoints and current
    f_current = np.max(corr)  # g(t0)
    f_0 = np.max(corr - t0 * C)  # g(0)
    f_1 = np.max(corr + (1.0 - t0) * C)  # g(1)
    
    if f_0 <= f_current and f_0 <= f_1:
        return 0.0
    if f_1 <= f_current and f_1 <= f_0:
        return 1.0
    
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = np.max(corr + (c1 - t0) * C)
    f2 = np.max(corr + (c2 - t0) * C)
    
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = np.max(corr + (c1 - t0) * C)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = np.max(corr + (c2 - t0) * C)
    
    return (lo + hi) / 2.0

                elif new_max < best_c5 / (2.0/n):
                    best_c5 = new_max * (2.0 / n)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
