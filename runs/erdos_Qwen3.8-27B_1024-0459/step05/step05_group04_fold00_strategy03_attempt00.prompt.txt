def make_swap_delta(h, i, j, n):
    """Compute correlation change for swap h[i]:1->0, h[j]:0->1. O(n)."""
    r = 2 * n - 1

    # First factor: -(1-h[i+n-1-k]) for k in [i, i+n-1]
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    # Second factor: +h[i+k-(n-1)] for k in [n-1-i, 2n-2-i]

def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _TGT,
             'jac': lambda x: np.ones(_N)}]
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / _BETA) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0

    if n_ones != target:
        if n_ones > target:

            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def greedy_construct(n, rng):
    """Greedy: place 1s to minimize max correlation so far."""
    h = np.zeros(n)
    n_half = n // 2
    used = set()
    positions = list(range(n))
    rng.shuffle(positions)
    
    for _ in range(n_half):
        best_pos = -1
        best_max = float('inf')
        for p in positions:
            if p in used:
                continue
            h[p] = 1
            corr = compute_corr(h)
            mx = float(np.max(corr))
            h[p] = 0
            if mx < best_max:
                best_max = mx
                best_pos = p
        if best_pos >= 0:
            h[best_pos] = 1
            used.add(best_pos)
        else:
            for p in positions:
                if p not in used:
                    h[p] = 1
                    used.add(p)
                    break
    return h
