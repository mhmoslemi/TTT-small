```python
def swap_delta(h, p, q, n):
    """Compute change in full correlation when h[p] 1->0, h[q] 0->1."""
    offset = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # Term 1: -(1-h[p+offset]) where valid if 0 <= p+offset < n
    t1 = np.zeros(2*n-1)
    valid1 = (p + offset >= 0) & (p + offset < n)
    t1[valid1] = -(1.0 - h[p + offset[valid1]])
    
    # Term 2: h[p-offset] where valid if 0 <= p-offset < n
    t2 = np.zeros(2*n-1)
    valid2 = (p - offset >= 0) & (p - offset < n)
    t2[valid2] = h[p - offset[valid2]]
    
    # Term 3: +(1-h[q+offset]) where valid if 0 <= q+offset < n
    t3 = np.zeros(2*n-1)
    valid3 = (q + offset >= 0) & (q + offset < n)
    t3[valid3] = 1.0 - h[q + offset[valid3]]
    
    # Term 4: -h[q-offset] where valid if 0 <= q-offset < n
    t4 = np.zeros(2*n-1)
    valid4 = (q - offset >= 0) & (q - offset < n)
    t4[valid4] = -h[q - offset[valid4]]
    
    return t1 + t2 + t3 + t4
```

```python
def binary_full_search(h_bin, n, max_iters=500, top_lags=None):

    dx = 2.0 / n

    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
            
        # Precompute change vectors
        offset = np.arange(2*n-1) - (n-1)
        
        # Change when removing a 1 at position p
        # out_change[p] = delta_full when h[p] goes 1->0
        # in_change[q] = delta_full when h[q] goes 0->1
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # For efficiency, compute in batches
        best_delta_max = 0  # want to minimize max, so want most negative change in max
        cur_max_val = float(np.max(full_corr))
        
        best_p = None
        best_q = None
        best_new_max = cur_max_val
        
        # Vectorized: for each p in ones, compute out_change
        # out_change[p, k] for k = 0..2n-2
        # = -(1-h[p+offset[k]]) [if valid] + h[p-offset[k]] [if valid]
        
        # This is tricky to fully vectorize for all p at once due to different valid ranges
        # But we can loop over p (192 iterations) and vectorize over k (767) and q (192)
        
        for p in ones:
            # Compute out_change for this p (vector of length 2n-1)
            oc = np.zeros(2*n-1)
            v1 = (p + offset >= 0) & (p + offset < n)
            oc[v1] -= (1.0 - h[p + offset[v1]])
            v2 = (p - offset >= 0) & (p - offset < n)
            oc[v2] += h[p - offset[v2]]
            
            # For each q in zeros, compute in_change and total
            # in_change[q] = (1-h[q+offset]) [if valid] - h[q-offset] [if valid]
            ic = np.zeros((n_zeros, 2*n-1))
            for qi, q in enumerate(zeros):
                v3 = (q + offset >= 0) & (q + offset < n)
                ic[qi, v3] += (1.0 - h[q + offset[v3]])
                v4 = (q - offset >= 0) & (q - offset < n)
                ic[qi, v4] -= h[q - offset[v4]]
            
            # new_corr[q_idx] = full_corr + oc + ic[q_idx]
            # new_max[q_idx] = max(new_corr[q_idx])
            new_max = np.max(full_corr[None, :] + oc[None, :] + ic, axis=1)
            
            qmin_idx = np.argmin(new_max)
            if new_max[qmin_idx] < best_new_max:
                best_new_max = new_max[qmin_idx]
                best_p = p
                best_q = zeros[qmin_idx]
        
        if best_p is None or best_new_max >= cur_max_val - 1e-12:
            break  # No improvement
        
        # Apply swap
        h[best_p] = 0
        h[best_q] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        full_corr = compute_corr_full(h)
        cur_max_val = float(np.max(full_corr))
        c5 = cur_max_val * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):

        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)

            h[mask] += diff / cnt
        else:

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr_full(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty

def _sm_grad(x):

    m = c[0]

def binary_full_local_search(h_init, n, max_iters=800, restart_every=100):
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    full_corr = compute_corr_full(h)
    best_h = h.copy()
    best_c5 = float(np.max(full_corr)) * dx
    cur_max = float(np.max(full_corr))

    offset = np.arange(2 * n - 1) - (n - 1)

    for it in range(max_iters):
        if remaining() < 5.0:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break

        # Precompute in_changes for all zeros: (n_zeros, 2n-1)
        ic = np.zeros((n_zeros, 2 * n - 1), dtype=np.float64)
        for qi, q in enumerate(zeros):
            v3 = (q + offset >= 0) & (q + offset < n)
            ic[qi, v3] += (1.0 - h[q + offset[v3]])
            v4 = (q - offset >= 0) & (q - offset < n)
            ic[qi, v4] -= h[q - offset[v4]]

        best_new_max = cur_max
        best_p = -1
        best_q_idx = -1

        for p in ones:
            oc = np.zeros(2 * n - 1, dtype=np.float64)
            v1 = (p + offset >= 0) & (p + offset < n)
            oc[v1] -= (1.0 - h[p + offset[v1]])
            v2 = (p - offset >= 0) & (p - offset < n)
            oc[v2] += h[p - offset[v2]]

            combined = full_corr[None, :] + oc[None, :] + ic
            new_maxes = np.max(combined, axis=1)
            qm = np.argmin(new_maxes)
            if new_maxes[qm] < best_new_max:
                best_new_max = new_maxes[qm]
                best_p = p
                best_q_idx = qm

        if best_p < 0 or best_new_max >= cur_max - 1e-12:
            # No improving single swap found. Try restart.
            if it % restart_every == restart_every - 1 and restart_every > 0:
                # Perturb: flip 3 random positions
                rng_r = np.random.RandomState(it)
                flip_ones = ones[rng_r.choice(n_ones, size=min(3, n_ones), replace=False)]
                flip_zeros = zeros[rng_r.choice(n_zeros, size=min(3, n_zeros), replace=False)]
                for idx in flip_ones[:min(3, len(flip_zeros))]:
                    h[idx] = 0.0
                for idx in flip_zeros[:min(3, len(flip_ones))]:
                    h[idx] = 1.0
                # Rebalance
                n_ones2 = int(np.sum(h))
                if n_ones2 > n_half:

                    extra = extra[rng_r.choice(len(extra), size=n_ones2 - n_half, replace=False)]
                    h[extra] = 0.0
                elif n_ones2 < n_half:
                    extra = np.where(h == 0)[0]
                    extra = extra[rng_r.choice(len(extra), size=n_half - n_ones2, replace=False)]
                    h[extra] = 1.0

                cur_max = float(np.max(full_corr))
            else:
                break

        # Apply best swap
        best_q = zeros[best_q_idx]
        h[best_p] = 0.0
        h[best_q] = 1.0

def projected_gradient(h, n, top_k=5, max_iters=5000, step_init=0.025):
    h = proj(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 4.0:
            break
        corr = compute_corr_full(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        c_top = corr[idx]
        weights = c_top - c_top[-1] + 0.05
        weights = np.maximum(weights, 0.01)
        weights /= np.sum(weights)

        g = np.zeros(n)
        for j, ki in enumerate(idx):
            g += weights[j] * corr_grad_single(h, ki, n)
        g -= np.mean(g)
        g[h >= 1.0 - 1e-9] = 0.0
        g[h <= 1e-9] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.03, 0.12)
        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.7

        if it % 500 == 0 and it > 0:
            step *= 0.9
            momentum = min(momentum + 0.003, 0.94)

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):

def binary_block_search(h_init, n, max_iters=200):
    """Try block swaps (swap 2 adjacent pairs) to escape single-swap local opt."""
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(12345)

    for it in range(max_iters):
        if remaining() < 5.0:
            break
        cur_c5 = compute_c5(h)

        # Try random block swaps of size 2
        improved = False
        for _ in range(30):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            c5_try = compute_c5(h)
            if c5_try < cur_c5 - 1e-12:
                cur_c5 = c5_try
                improved = True
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h.copy()
                break
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
        if not improved:
            # Try single flips (accept worse moves as in simulated annealing)
            found_any = False
            for _ in range(10):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c5_try = compute_c5(h)
                delta = c5_try - cur_c5
                # Accept with probability based on "temperature"
                if delta < 0 or rng.random() < np.exp(-delta / 0.001):
                    cur_c5 = c5_try
                    found_any = True
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if not found_any:
                break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick continuous optimization to refine initial pattern
    for beta in [200, 800, 3000]:
        if remaining() < 10:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=4, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Aggressive binary full local search (main improvement)
    h_bin = (best_h > 0.5).astype(float)
    h_bl, c5_bl = binary_full_local_search(h_bin, best_n, max_iters=600)
    if c5_bl < best_c5:
        best_c5 = c5_bl
        best_h = h_bl.copy()
        h = h_bl.copy()

    # Phase 3: Continuous refinement of the binary pattern
    # Start slightly off binary to allow L-BFGS-B to move
    h_cont = best_h.copy().astype(float)
    h_cont[h_cont > 0.5] = 0.5 + 0.3 * (h_cont[h_cont > 0.5] - 0.5) * 2
    h_cont[h_cont <= 0.5] = 0.5 - 0.3 * (0.5 - h_cont[h_cont <= 0.5]) * 2
    h_cont = np.clip(h_cont, 0.05, 0.95)
    h_cont = proj(h_cont, best_n)

    for beta in [3000, 10000, 30000]:
        if remaining() < 8:
            break
        h_ref = lbfgs_pass(h_cont, best_n, beta, top_k=3, maxiter=300)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h_cont = h_ref.copy()

    # Also try from the pure binary
    for beta in [5000, 15000]:
        if remaining() < 8:
            break
        h_ref2 = lbfgs_pass(h_bl.copy(), best_n, beta, top_k=3, maxiter=300)
        c5_ref2 = compute_c5(h_ref2)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2.copy()

    # Phase 4: Projected gradient for fine-tuning
    if remaining() > 20:
        h_pg, c5_pg = projected_gradient(h, best_n, top_k=4, max_iters=4000, step_init=0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 5: Top equalize
    if remaining() > 10:
        h_eq, c5_eq = top_equalize(h, best_n, top_k=4, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: Block search to escape local optima, then re-run full search
    if remaining() > 30:
        h_blk, c5_blk = binary_block_search(best_h, best_n, max_iters=150)
        if c5_blk < best_c5:
            best_c5 = c5_blk
            best_h = h_blk.copy()
        # Re-run full local search from the (possibly improved) binary
        h_blk2, c5_blk2 = binary_full_local_search(best_h, best_n, max_iters=300)
        if c5_blk2 < best_c5:
            best_c5 = c5_blk2
            best_h = h_blk2.copy()
        # Refine
        h_ref3 = lbfgs_pass(best_h, best_n, 10000, top_k=3, maxiter=250)
        c5_ref3 = compute_c5(h_ref3)
        if c5_ref3 < best_c5:
            best_c5 = c5_ref3
            best_h = h_ref3.copy()

    # Phase 7: Try different resolutions
    n_candidates = []
    for nn in [best_n - 4, best_n - 2, best_n + 2, best_n + 4, best_n + 8,
               best_n + 16, int(best_n * 1.1), int(best_n * 1.2),
               int(best_n * 1.3), int(best_n * 1.5)]:
        nn = nn // 2 * 2  # make even
        if 128 <= nn <= 768 and nn != best_n:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # Quick L-BFGS at new resolution
        for beta in [1000, 5000]:
            if remaining() < 15:
                break
            h_feas = lbfgs_pass(h_feas, n_t, beta, top_k=4, maxiter=300)

        # Binary full search at new resolution
        h_bin_t = (h_feas > 0.5).astype(float)
        h_bl_t, c5_bl_t = binary_full_local_search(h_bin_t, n_t, max_iters=300)

        # Refine
        h_ref_t = lbfgs_pass(h_bl_t, n_t, 8000, top_k=3, maxiter=250)
        c5_ref_t = compute_c5(h_ref_t)

        c5_cand = min(c5_bl_t, c5_ref_t, compute_c5(h_feas))
        if c5_ref_t <= c5_bl_t:
            h_cand = h_ref_t
        else:
            h_cand = h_bl_t

        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_t

    # Phase 8: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    # SLSQP for precise constraint
    for beta in [20000, 50000]:
        if remaining() < 8:
            break
        h_new = slsqp_pass(h, n, beta, top_k=3, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_full_local_search(h_bin_f, best_n, max_iters=400)
        h_rf = lbfgs_pass(h_bf, best_n, 12000, top_k=3, maxiter=250)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()
        elif c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Final top equalize
    if remaining() > 6:
        h_eq2, c5_eq2 = top_equalize(best_h, best_n, top_k=5, max_iters=150)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_h = proj(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
```
