def structured_init(n, rng):

    h = np.zeros(n)

    # Find next prime

        is_p = True
        for i in range(2, int(p ** 0.5) + 1):
            if p % i == 0:
                is_p = False
                break
        if is_p:
            break
        p += 1

    for x in range(p):
        qr.add((x * x) % p)
    for i in range(n):

    return project_feasible(h, n)

        h, c5 = top_k_equalize(h, n, top_k=8, max_iters=500)
    return h, c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N, _SM_BETA, _PENALTY
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution
    h, c5 = multi_phase_optimize(best_h.copy(), best_n, time_limit=300)
    if c5 < best_c5 - 1e-14:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Try alternative starts at parent resolution
    alt_starts = [
        theta_morse(best_n),
        random_balanced(best_n, rng),
        structured_init(best_n, rng),
    ]
    for h_alt in alt_starts:
        if remaining_time() < 40:
            break
        h_opt, c5_opt = multi_phase_optimize(h_alt, best_n, time_limit=120)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Multi-resolution search (higher resolutions have more DOF)
    n_list = sorted(set([best_n]))
    for mult in [1.2, 1.35, 1.5, 1.75, 2.0, 2.5, 3.0]:
        nn = int(best_n * mult)
        if nn % 2 != 0:
            nn += 1
        if 200 <= nn <= 1200:
            n_list.append(nn)

    # Also try lower resolutions (sometimes fewer variables help)
    for mult in [0.75, 0.6, 0.5]:
        nn = int(best_n * mult)
        if nn % 2 != 0:
            nn += 1
        if 120 <= nn < best_n:
            n_list.append(nn)

    n_list.sort(key=lambda x: abs(x - best_n))

    for n_t in n_list:
        if remaining_time() < 80:
            break
        if n_t == best_n:
            continue
        # Interpolate current best to new resolution
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        # Also generate fresh starts at this resolution
        h_fresh = random_balanced(n_t, rng)
        h_tm = theta_morse(n_t)

        candidates = [h_feas, h_fresh, h_tm]
        best_local_c5 = c5_int
        best_local_h = h_feas.copy()

        for h_cand in candidates:
            if remaining_time() < 50:
                break
            h_opt, c5_opt = multi_phase_optimize(h_cand, n_t, time_limit=150)
            if c5_opt < best_local_c5:
                best_local_c5 = c5_opt
                best_local_h = h_opt.copy()

        # Binary local search at this resolution
        if remaining_time() > 50 and n_t <= 512:
            h_bin = (best_local_h > 0.5).astype(float)
            h_bl, c5_bl = binary_local_search(h_bin, n_t, time_limit=40, rng=rng)
            if c5_bl < best_local_c5:
                best_local_c5 = c5_bl
                best_local_h = h_bl.copy()
            # Refine
            if c5_bl < best_local_c5 or compute_c5(h_bl) < best_local_c5:
                h_ref, c5_ref = multi_phase_optimize(h_bl, n_t, time_limit=60)
                if c5_ref < best_local_c5:
                    best_local_c5 = c5_ref
                    best_local_h = h_ref.copy()

        if best_local_c5 < best_c5 - 1e-14:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()
            best_n = n_t

    # Phase 4: Binary SA at best resolution
    if remaining_time() > 60 and best_n <= 512:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(80, int(remaining_time() * 0.3))
        h_sa, c5_sa = binary_sa(h_bin, best_n, time_limit=sa_time, rng=rng)
        # Refine
        h_sa_ref, c5_sa_ref = multi_phase_optimize(h_sa, best_n, time_limit=80)
        if c5_sa_ref < best_c5 - 1e-14:
            best_c5 = c5_sa_ref
            best_h = h_sa_ref.copy()
        if c5_sa < best_c5 - 1e-14:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 5: Intensive final refinement at best resolution
    n = best_n
    h = best_h.copy()
    for beta in [500000, 2000000, 5000000]:
        if remaining_time() < 10:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=6, max_iters=1000)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining_time() > 8:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=10, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final SLSQP polish
    if remaining_time() > 8:
        h_final = slsqp_opt(best_h, n, 3000000, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
