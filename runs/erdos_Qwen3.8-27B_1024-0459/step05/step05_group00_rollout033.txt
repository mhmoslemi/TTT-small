```python
def smooth_max_and_grad(h, beta, min_weight=1e-10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    sm = (m + np.log(np.sum(exp_s)) / beta) * dx
    
    # Only compute gradient for lags with significant weight
    active = w > min_weight
    g = np.zeros(n)
    for k in np.where(active)[0]:
        wk = w[k] * dx
        lag = k - (n - 1)
        if lag >= 0:
            npairs = n - lag
            g[lag:] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[lag:]
        else:
            m2 = -lag
            npairs = n - m2
            g[:npairs] += wk * (1.0 - h[m2:])
            g[m2:] -= wk * h[:npairs]
    return sm, g
```

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def topk_smooth_max_and_grad(h, beta, K=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(-corr)[:K]
    top_corr = corr[top_idx]
    m = top_corr.max()
    shifted = beta * (top_corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    sm = (m + np.log(np.sum(exp_s)) / beta) * dx
    g = np.zeros(n)
    for i in range(K):
        wk = w[i] * dx
        if wk < 1e-15:
            continue
        k = top_idx[i]
        lag = k - (n - 1)
        if lag >= 0:
            npairs = n - lag
            g[lag:] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[lag:]
        else:
            m2 = -lag
            npairs = n - m2
            g[:npairs] += wk * (1.0 - h[m2:])
            g[m2:] -= wk * h[:npairs]
    return sm, g


def full_smooth_max_and_grad(h, beta, min_weight=1e-10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    sm = (m + np.log(np.sum(exp_s)) / beta) * dx
    g = np.zeros(n)
    active = w > min_weight
    active_idx = np.where(active)[0]
    for k in active_idx:
        wk = w[k] * dx
        lag = k - (n - 1)
        if lag >= 0:
            npairs = n - lag
            g[lag:] += wk * (1.0 - h[:npairs])
            g[:npairs] -= wk * h[lag:]
        else:
            m2 = -lag
            npairs = n - m2
            g[:npairs] += wk * (1.0 - h[m2:])
            g[m2:] -= wk * h[:npairs]
    return sm, g


def adam_optimize(h, n, beta_start, beta_end, n_iter, lr, time_limit, K=20, use_full=False):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    patience = 0
    patience_limit = 80

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        t += 1
        progress = min(it / max(n_iter - 1, 1), 1.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        lr_cur = lr / (1.0 + 0.3 * progress)

        if use_full:
            obj, g = full_smooth_max_and_grad(h, beta, min_weight=1e-12)
        else:
            obj, g = topk_smooth_max_and_grad(h, beta, K)

        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g ** 2
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)

        h_new = h - lr_cur * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1

        h = h_new
        if patience > patience_limit:
            break

    return best_h, best_c5


def al_lbfgs(h, n, beta, K=20, max_outer=30, max_inner=100):
    lam = 0.0
    mu = 1.0
    h_cur = h.copy()
    base_c5 = compute_c5(h)

    for outer in range(max_outer):
        rem = _BUDGET - (time.time() - _T0)
        if rem < 15:
            break

        my_mu = mu
        my_lam = lam

        def obj(x):
            val, g = topk_smooth_max_and_grad(x, beta, K)
            s = np.sum(x) - n / 2.0
            return val + 0.5 * my_mu * s * s + my_lam * s

        def grad_fn(x):
            val, g = topk_smooth_max_and_grad(x, beta, K)
            s = np.sum(x) - n / 2.0
            return g + (my_mu * s + my_lam) * np.ones(n)

        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(fun=obj, x0=h_cur, jac=grad_fn, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_inner, 'ftol': 1e-14, 'gtol': 1e-6, 'maxfun': max_inner * 3})
            h_cur = np.clip(res.x, 0.0, 1.0)
        except Exception:
            break

        h_cur = project_feasible(h_cur, n)
        c5_cur = compute_c5(h_cur)
        if c5_cur < base_c5 - 1e-12:
            base_c5 = c5_cur

        s = np.sum(h_cur) - n / 2.0
        lam = lam + mu * s
        if abs(s) > 1e-5:
            mu = min(mu * 2.0, 1e6)
        elif abs(s) < 1e-12:
            break

    h_final = project_feasible(h_cur, n)
    c5_final = compute_c5(h_final)
    if c5_final <= compute_c5(h) + 1e-10:
        return h_final
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coordinated_local_search(h, n, n_iter=200, group_size=8, time_limit=30.0):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        # Pick a random group of coordinates to perturb
        idx = np.random.choice(n, size=min(group_size, n), replace=False)
        offsets = np.random.uniform(-0.05, 0.05, size=len(idx))

        for scale in [1.0, 0.5, 0.2, 0.1]:
            h_try = h.copy()
            h_try[idx] += offsets * scale
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                break

    return best_h, best_c5


def równoważenie_top(h, n, K=10, max_iters=100, time_limit=30.0):
    """Try to equalize the top K correlation values."""
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break

        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:K]
        top_vals = corr[top_idx]
        c_max = top_vals.max()
        c_min = top_vals.min()

        if c_max - c_min < 1e-10:
            break

        # Gradient of (max - min) of top K to push them together
        # and reduce the max
        g = np.zeros(n)
        for k in top_idx:
            wk = 1.0 / K
            lag = k - (n - 1)
            if lag >= 0:
                npairs = n - lag
                g[lag:] += wk * (1.0 - h[:npairs])
                g[:npairs] -= wk * h[lag:]
            else:
                m2 = -lag
                npairs = n - m2
                g[:npairs] += wk * (1.0 - h[m2:])
                g[m2:] -= wk * h[:npairs]

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        for step in [0.05, 0.03, 0.01, 0.08, 0.02, 0.005]:
            h_try = project_feasible(h - step * g / gn, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                break

    return best_h, best_c5
