    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 100.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 200.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=6, maxiter=500):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(sm_fun, h, jac=sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': maxiter * 12,
                                'maxls': 50})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def binary_search_intensive(h_init, n, max_iters=10000, top_k=6,
                            cand_size=40, rng=None):
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h = np.zeros(n)
    h[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = cur_max * dx
    stall = 0

    for it in range(max_iters):
        if remaining() < 2:
            break

        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        cur_c5 = cur_max * dx

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > max_iters // 2:
            break

        idx_top = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h, ki, n)
        g -= np.mean(g)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        score_one = -g[ones]
        score_zero = g[zeros]

        n_try_o = min(cand_size, len(ones))
        n_try_z = min(cand_size, len(zeros))
        best_o = ones[np.argsort(-score_one)[:n_try_o]]
        best_z = zeros[np.argsort(-score_zero)[:n_try_z]]

        best_delta = 0.0
        best_swap = None

        for i in best_o:
            for j in best_z:
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                nc = float(np.max(compute_corr(h)))
                h[i] = 1.0
                h[j] = 0.0
                delta = nc - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)

        for _ in range(8):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            nc = float(np.max(compute_corr(h)))
            h[i] = 1.0
            h[j] = 0.0
            delta = nc - cur_max
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j)

        if best_swap is None or best_delta >= 0:
            if it % 500 == 0 and it > 500:
                for _ in range(4):
                    ii = int(ones[rng.randint(len(ones))])
                    jj = int(zeros[rng.randint(len(zeros))])
                    if ii != jj:
                        h[ii] = 0.0
                        h[jj] = 1.0
                nc = float(np.max(compute_corr(h))) * dx
                if nc > best_c5 + 0.002:
                    h = best_h.copy()
                    corr = compute_corr(h)
                    cur_max = float(np.max(corr))
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
            else:
                break
        else:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0

    return best_h, best_c5


def nesterov_gradient(h, n, top_k=6, max_iters=6000, step_init=0.015):
    h = proj(h, n).copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        for ki in idx:
            g += corr_grad_single(h, ki, n)
        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.001:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.08)
            else:
                step *= 0.92
        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.75

        if it % 600 == 0 and it > 0:
            step *= 0.92
            momentum = min(momentum + 0.003, 0.95)

    return best_h, best_c5


def top_equalize(h, n, top_k=6, max_iters=400):
    h = proj(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn

        improved = False
        for alpha in [0.005, 0.02, 0.01, 0.04, 0.015, 0.03, 0.008]:
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            for alpha in [0.01, 0.02, 0.03]:
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_quasiperiodic(n, freq, phase, rng):
    r = np.arange(n)
    h = (np.sin(2.0 * np.pi * freq * r / n + phase) > 0).astype(float)
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        excess = n_ones - n // 2
        one_idx = np.where(h == 1)[0]
        to_flip = rng.choice(one_idx, excess, replace=False)
        h[to_flip] = 0
    elif n_ones < n // 2:
        deficit = n // 2 - n_ones
        zero_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zero_idx, deficit, replace=False)
        h[to_flip] = 1
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = proj(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        best_h = make_random_binary(parent_n, rng)
        best_c5 = compute_c5(best_h)
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1a: Intensive binary search from parent
    if remaining() > 60:
        h_bin = (h > 0.5).astype(float)
        h_b, c5_b = binary_search_intensive(h_bin, n, max_iters=5000, top_k=6,
                                            cand_size=40, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()

    # Phase 1b: Continuous refinement
    for beta in [1000, 4000, 12000, 30000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=6, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 1c: Nesterov gradient
    if remaining() > 20:
        h_pg, c5_pg = nesterov_gradient(h, n, top_k=6, max_iters=5000, step_init=0.012)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 1d: Top equalize
    if remaining() > 10:
        h_eq, c5_eq = top_equalize(h, n, top_k=6, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 1e: Binary from continuous + refine
    if remaining() > 40:
        h_bin2 = (h > 0.5).astype(float)
        h_b2, c5_b2 = binary_search_intensive(h_bin2, n, max_iters=5000, top_k=6,
                                              cand_size=45, rng=rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            h = h_b2.copy()
        h_ref = lbfgs_pass(h_b2, n, 15000, top_k=6, maxiter=400)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h = h_ref.copy()

    # Phase 2: Try different resolutions
    resolutions = sorted(set([
        best_n + 2, best_n + 6, best_n + 12,
        int(best_n * 1.25), int(best_n * 1.5), int(best_n * 1.75),
        512, 640, 768, 1024, 1280
    ]))
    resolutions = [r for r in resolutions if 300 <= r <= 1280 and r % 2 == 0 and r != best_n]
    resolutions.sort(key=lambda x: abs(x - best_n))

    for n_t in resolutions[:5]:
        if remaining() < 50:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = proj(h_interp, n_t)

        # Binary search at new resolution
        h_bin_t = (h_feas > 0.5).astype(float)
        h_bt, c5_bt = binary_search_intensive(h_bin_t, n_t, max_iters=3000,
                                              top_k=6, cand_size=35, rng=rng)
        if c5_bt < best_c5:
            best_c5 = c5_bt
            best_h = h_bt.copy()
            best_n = n_t
            h = h_bt.copy()

        # Continuous at new resolution
        if remaining() > 20:
            h_ct = lbfgs_pass(h_bt, n_t, 8000, top_k=6, maxiter=400)
            c5_ct = compute_c5(h_ct)
            if c5_ct < best_c5:
                best_c5 = c5_ct
                best_h = h_ct.copy()
                best_n = n_t
                h = h_ct.copy()

    # Phase 3: Multi-start from diverse initializations at best_n
    n = best_n
    h = best_h.copy()

    diverse_inits = []
    # Random binary
    diverse_inits.append(make_random_binary(n, rng))
    # Quasiperiodic
    diverse_inits.append(make_quasiperiodic(n, 7, 0.3, rng))
    diverse_inits.append(make_quasiperiodic(n, 13, 1.7, rng))
    diverse_inits.append(make_quasiperiodic(n, 3, 0.8, rng))
    # Perturbed parent
    h_pert = h.copy()
    for _ in range(n // 32):
        ones_p = np.where(h_pert == h_pert.max())[0]
        zeros_p = np.where(h_pert == h_pert.min())[0]
        if len(ones_p) > 0 and len(zeros_p) > 0:
            ii = int(ones_p[rng.randint(len(ones_p))])
            jj = int(zeros_p[rng.randint(len(zeros_p))])
            if h_pert[ii] > 0.5:
                h_pert[ii] -= 1.0
                h_pert[jj] += 1.0
    h_pert = np.clip(h_pert, 0, 1)
    diverse_inits.append(h_pert)

    for init in diverse_inits:
        if remaining() < 80:
            break
        h_t = proj(init, n)
        h_bt, c5_bt = binary_search_intensive(h_t, n, max_iters=3000, top_k=6,
                                              cand_size=35, rng=rng)
        if c5_bt < best_c5:
            best_c5 = c5_bt
            best_h = h_bt.copy()
            h = h_bt.copy()
        if remaining() > 20:
            h_ct = lbfgs_pass(h_bt, n, 10000, top_k=6, maxiter=300)
            c5_ct = compute_c5(h_ct)
            if c5_ct < best_c5:
                best_c5 = c5_ct
                best_h = h_ct.copy()
                h = h_ct.copy()

    # Phase 4: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # More binary search
    if remaining() > 40:
        h_bin_f = (h > 0.5).astype(float)
        h_bf, c5_bf = binary_search_intensive(h_bin_f, n, max_iters=6000, top_k=7,
                                              cand_size=50, rng=rng)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
            h = h_bf.copy()

    # Continuous refinement with high beta
    for beta in [20000, 40000, 80000]:
        if remaining() < 6:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Nesterov final
    if remaining() > 15:
        h_nf, c5_nf = nesterov_gradient(h, n, top_k=7, max_iters=4000, step_init=0.01)
        if c5_nf < best_c5:
            best_c5 = c5_nf
            best_h = h_nf.copy()
            h = h_nf.copy()

    # Top equalize final
    if remaining() > 8:
        h_ef, c5_ef = top_equalize(h, n, top_k=8, max_iters=300)
        if c5_ef < best_c5:
            best_c5 = c5_ef
            best_h = h_ef.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_p = (best_h > 0.5).astype(float)
        h_bp, c5_bp = binary_search_intensive(h_bin_p, n, max_iters=4000, top_k=7,
                                              cand_size=50, rng=rng)
        if c5_bp < best_c5:
            best_c5 = c5_bp
            best_h = h_bp.copy()
        h_rp = lbfgs_pass(h_bp, n, 25000, top_k=5, maxiter=300)
        c5_rp = compute_c5(h_rp)
        if c5_rp < best_c5:
            best_c5 = c5_rp
            best_h = h_rp.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
