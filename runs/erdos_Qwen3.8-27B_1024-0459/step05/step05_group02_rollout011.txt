def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

            c5_t = min(c5_t, c5_bt, c5_rt)
            if c5_rt <= c5_t:
                h_t = h_rt
            elif c5_bt <= c5_t:
                h_t = h_bt

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 6: Random restarts at best resolution
    n = best_n
    for restart in range(3):
        if remaining() < 40:
            break
        h_r = random_binary_init(n, rng)
        h_r = lse_optimize(h_r, n, 200, maxiter=300)
        h_r = lse_optimize(h_r, n, 5000, maxiter=300)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Fourier-based initialization
    if remaining() > 30:
        h_f = fourier_init(n, rng, amplitude=0.25)
        h_f = lse_optimize(h_f, n, 500, maxiter=300)
        h_f = lse_optimize(h_f, n, 10000, maxiter=300)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()
    if remaining() > 20:
        h_pg2, c5_pg2 = topk_smooth(h, n, beta=100, top_k=8)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()
            h = h_pg2.copy()

    for beta in [50000, 100000, 200000]:
        if remaining() < 10:
            break
        h_new = lse_optimize(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Final binary polish
    if remaining() > 25:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_search_2flip(h_bin_f, best_n, max_iters=4000, top_k=5)
        h_rf = lse_optimize(h_bf, best_n, 50000, maxiter=400)
        c5_rf = compute_c5(h_rf)
        if min(c5_bf, c5_rf) < best_c5:
            if c5_bf <= c5_rf:
                best_c5 = c5_bf
                best_h = h_bf
            else:
                best_c5 = c5_rf
                best_h = h_rf

    # One more LSE pass
    if remaining() > 10:
        h_final = lse_optimize(best_h, best_n, 100000, maxiter=200)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5
        best_h = proj(best_h, best_n)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
