    fa = np.fft.fft(a, fsize)

def slsqp_opt(h, n, beta, maxiter=500):
    global _CUR_N
    _CUR_N = n
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(lambda x: _smooth_obj(x, beta), h0,
                       jac=lambda x: _smooth_grad(x, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success and res.fun < _smooth_obj(h0, beta) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:

def projected_gradient_step(h, n, beta, step_size, momentum=0.9, velocity=None):
    """One step of momentum projected gradient descent on smoothed objective."""
    global _CUR_N
    _CUR_N = n
    grad = _smooth_grad(h, beta)
    if velocity is None:
        velocity = np.zeros(n)
    velocity = momentum * velocity - step_size * grad
    h_new = h + velocity
    h_new = project_feasible(h_new, n)
    return h_new, velocity

def momentum_opt(h, n, beta, max_iters=2000, init_step=0.01):
    """Optimize with momentum projected gradient."""
    h = project_feasible(h, n)
    velocity = np.zeros(n)
    step = init_step

def top_k_equalize(h, n, top_k=5, max_iters=500, min_gap=1e-10):
    """Equalize the top-k worst correlations."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    momentum = 0.7
    velocity = np.zeros(n)
    step = 0.01
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        gap = c_max - c_min_top
        if gap < min_gap:
            break
        
        # Compute weighted subgradient
        weights = np.zeros(len(idx_top))
        weights[0] = 1.0
        for pos in range(1, len(idx_top)):
            weight = -1.0 / (len(idx_top) - 1)
            weights[pos] = weight
        
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = weights[pos] * dx
            if abs(w_k) < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        
        # Momentum step
        velocity = momentum * velocity + d
        vn = np.linalg.norm(velocity)
        if vn > 0:
            d = velocity / vn
        
        improved = False
        for s in [step, step*2, step*0.5, step*4, step*0.25]:
            ht = project_feasible(h + s * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                step = min(s * 1.1, 0.1)
                break
        
        if not improved:
            step *= 0.7
            if step < 0.001:
                break
    
    return h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

def binary_sa(h, n, time_limit=60, rng=None):
    """Simulated annealing on binary solutions."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    
    dx = 2.0 / n
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Initial temperature based on typical delta
    # Typical change in C5 from a random swap is about 2*dx = 4/n
    T = 2.0 / n * 2.0  # Start a bit above typical delta
    T_min = T * 0.01
    cooling = 0.9995
    
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Propose a swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        delta = c_new - best_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h[i] = 0.0
            h[j] = 1.0
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        T = max(T_min, T * cooling)
        
        if no_improve > 5000 and T <= T_min * 1.01:
            break
    
    return best_h, best_c5

    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try a batch of random swaps, pick the best
        n_trials = min(200, len(ones) * len(zeros))
        best_delta = 1e-10
        best_pair = None
        
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            delta = c_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                # Random kick: do 5 random swaps
                for _ in range(5):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = h[j], h[i]
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                best_c5 = compute_c5(h)
                no_improve = 0
    
    return best_h, best_c5

def smooth_obj_factory(n, beta):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        return (m + np.log(np.sum(ez)) / beta) * dx
    return obj


def smooth_grad_factory(n, beta):
    dx = 2.0 / n
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:np_]
        return g
    return grad


def slsqp_smooth(h, n, beta, maxiter=500):

    obj_fn = smooth_obj_factory(n, beta)
    grad_fn = smooth_grad_factory(n, beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',

        if len(ones) == 0 or len(zeros) == 0:
            break

        n_trials = min(300, len(ones) * len(zeros))
        best_delta = 1e-10
        best_pair = None

        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            delta = c_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)

        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 15:
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = h[j], h[i]
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = theta_morse(384)
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLP optimization on parent (main workhorse)
    h = best_h.copy()
    h_slp, c5_slp = slp_optimize(h, best_n, max_iters=3000, alpha_init=0.5, top_frac=0.02)
    if c5_slp < best_c5 - 1e-15:
        best_c5 = c5_slp
        best_h = h_slp.copy()

    # Phase 2: Smooth SLSQP refinement
    h = best_h.copy()
    for beta in [100, 500, 2000, 10000]:
        if remaining_time() < 25:
            break
        h_new = slsqp_smooth(h, best_n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Top-k equalization
    if remaining_time() > 15:
        h_eq, c5_eq = top_k_equalize(h, best_n, top_k=5, max_iters=600)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Second SLP pass
    if remaining_time() > 30:
        h_slp2, c5_slp2 = slp_optimize(h, best_n, max_iters=2000, alpha_init=0.3, top_frac=0.01)
        if c5_slp2 < best_c5 - 1e-15:
            best_c5 = c5_slp2
            best_h = h_slp2.copy()

    # Phase 5: Binary SA
    if remaining_time() > 80:
        h_bin = (best_h > 0.5).astype(float)
        sa_time = min(int(remaining_time() * 0.2), 150)
        h_sa, c5_sa = binary_sa(h_bin, best_n, time_limit=sa_time, rng=rng)

        # Refine binary continuously
        h_ref = h_sa.copy()
        if remaining_time() > 20:
            h_slp3, c5_slp3 = slp_optimize(h_ref, best_n, max_iters=1500, alpha_init=0.3, top_frac=0.02)
            if c5_slp3 < best_c5 - 1e-15:
                best_c5 = c5_slp3
                best_h = h_slp3.copy()

        for beta in [5000, 50000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_smooth(h_ref, best_n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_ref = h_new.copy()

        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Greedy binary
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        g_time = min(int(remaining_time() * 0.25), 120)
        h_g, c5_g = binary_greedy(h_bin, best_n, time_limit=g_time, rng=rng)

        h_ref2 = h_g.copy()
        if remaining_time() > 20:
            h_slp4, c5_slp4 = slp_optimize(h_ref2, best_n, max_iters=1000, alpha_init=0.3, top_frac=0.02)
            if c5_slp4 < best_c5 - 1e-15:
                best_c5 = c5_slp4
                best_h = h_slp4.copy()

        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 7: Try higher resolution
    if remaining_time() > 120:
        n_high = best_n * 2
        if n_high % 2 != 0:
            n_high += 1
        h_high = interp_to(best_h, best_n, n_high)
        h_high = project_feasible(h_high, n_high)

        h_slp5, c5_slp5 = slp_optimize(h_high, n_high, max_iters=1500, alpha_init=0.4, top_frac=0.02)
        if c5_slp5 < best_c5 - 1e-15:
            best_c5 = c5_slp5
            best_h = h_slp5.copy()
            best_n = n_high

        if remaining_time() > 30:
            h_hi_eq, c5_hi_eq = top_k_equalize(h_slp5, n_high, top_k=5, max_iters=300)
            if c5_hi_eq < best_c5 - 1e-15:
                best_c5 = c5_hi_eq
                best_h = h_hi_eq.copy()

        if remaining_time() > 30:
            h_slp6, c5_slp6 = slp_optimize(best_h, best_n, max_iters=1000, alpha_init=0.3, top_frac=0.01)
            if c5_slp6 < best_c5 - 1e-15:
                best_c5 = c5_slp6
                best_h = h_slp6.copy()

    # Phase 8: Different initialization (Thue-Morse)
    if remaining_time() > 60:
        h_tm = theta_morse(best_n)
        h_tm_slp, c5_tm = slp_optimize(h_tm, best_n, max_iters=1500, alpha_init=0.4, top_frac=0.02)
        if c5_tm < best_c5 - 1e-15:
            best_c5 = c5_tm
            best_h = h_tm_slp.copy()

    # Phase 9: Final intensive polish
    n = best_n
    h = best_h.copy()

    for top_f in [0.01, 0.005, 0.002]:
        if remaining_time() < 10:
            break
        h_final, c5_final = slp_optimize(h, n, max_iters=1000, alpha_init=0.3, top_frac=top_f)
        if c5_final < best_c5 - 1e-15:
            best_c5 = c5_final
            best_h = h_final.copy()
            h = h_final.copy()

    if remaining_time() > 10:
        h_eq2, c5_eq2 = top_k_equalize(h, n, top_k=8, max_iters=300)
        if c5_eq2 < best_c5 - 1e-15:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Final SLSQP
    if remaining_time() > 10:
        h_f = slsqp_smooth(best_h, n, 200000, maxiter=500)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5 - 1e-15:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
