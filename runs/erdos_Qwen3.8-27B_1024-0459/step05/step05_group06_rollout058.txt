import numpy as np
import time
from scipy.optimize import minimize

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def solve_lp_cvxpy(n, solver_list=['CBC', 'GLOP', 'GLPK', 'ECOS']):
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    dx = 2.0 / n
    nshifts = 2 * n - 1
    
    # Build y variables and constraints
    y_vars = []  # list of (k, i, j, y_var)
    constraints = [
        t >= 0,
        h >= 0,
        h <= 1,
        cp.sum(h) == n / 2.0
    ]
    
    for k in range(nshifts):
        m = k - (n - 1)
        if m >= 0:
            i_range = range(0, n - m)
        else:
            i_range = range(-m, n)
        
        shift_ys = []
        for i in i_range:
            j = i + m
            y = cp.Variable()
            y_vars.append((k, i, j, y))
            constraints.append(y <= h[i])
            constraints.append(y <= 1 - h[j])
            constraints.append(y >= h[i] - h[j])
            constraints.append(y >= 0)
            shift_ys.append(y)
        
        if shift_ys:
            constraints.append(cp.sum(shift_ys) * dx <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    for solver in solver_list:
        try:

            if prob.status in ['optimal', 'optimal_inaccurate'] and h.value is not None:

        except Exception:

    nshifts = 2 * n - 1
    dx = 2.0 / n

            var_idx += 1

    coo_col = []
    coo_val = []

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        wsum = 1e-300
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad
