# Globals
_T0 = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmax(h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def smooth_max_fn(h, beta, topk):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:topk]
    c = corr[idx]
    m = c[0]
    sh = beta * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / beta) * dx
    penalty = 50.0 * (np.sum(h) - n / 2.0) ** 2
    return val + penalty

def smooth_max_grad_fn(h, beta, topk):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:topk]
    c = corr[idx]
    m = c[0]
    sh = beta * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(h, ki, n)
    g *= dx
    g += 100.0 * (np.sum(h) - n / 2.0)
    return g

def lbfgs(h, beta, top_k, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return smooth_max_fn(x, beta, top_k)
    
    def grad(x):
        return smooth_max_grad_fn(x, beta, top_k)
    
    h = proj(h, n)
    try:
        res = minimize(fun, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-19,
                                'gtol': 1e-12, 'maxls': 50,
                                'maxfun': maxiter * 20})
        return proj(res.x, n)
    except Exception:
        return h

    best_h = h.copy()

    v = np.zeros(n)
    momentum = 0.8

        w = np.exp(sh)
        w /= np.sum(w)

        g[h >= 1.0 - 1e-10] = 0.0
        g[h <= 1e-10] = 0.0

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmax(h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def smooth_max_fn(h, beta, topk):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:topk]
    c = corr[idx]
    m = c[0]
    sh = beta * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / beta) * dx
    penalty = 80.0 * (np.sum(h) - n / 2.0) ** 2
    return val + penalty


def smooth_max_grad_fn(h, beta, topk):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:topk]
    c = corr[idx]
    m = c[0]
    sh = beta * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(h, ki, n)
    g *= dx
    g += 160.0 * (np.sum(h) - n / 2.0)
    return g


def lbfgs(h, beta, top_k, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return smooth_max_fn(x, beta, top_k)

    def grad(x):
        return smooth_max_grad_fn(x, beta, top_k)

    h = proj(h, n)
    try:
        res = minimize(fun, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-19,
                                'gtol': 1e-12, 'maxls': 50,
                                'maxfun': maxiter * 20})
        return proj(res.x, n)
    except Exception:
        return h


def nesterov(h, n, top_k, max_iters, step_init):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.8
    step = step_init

    for it in range(max_iters):
        if remaining() < 5:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        c_top = corr[idx]
        max_c = c_top[0]
        sh = c_top - max_c
        sh = np.clip(sh, -30, 30)
        w = np.exp(sh)
        w /= np.sum(w)

        g = np.zeros(n)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.sum(g) / n
        g[h >= 1.0 - 1e-10] = 0.0
        g[h <= 1e-10] = 0.0

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        momentum = min(momentum + 0.0005, 0.95)
        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < c5 + 1e-12:
            h = h_new
            step = min(step * 1.005, 0.08)
        else:
            h = best_h.copy()
            v *= 0.3
            step *= 0.65

        if it % 1000 == 0 and it > 0:
            step *= 0.9

    return best_h, best_c5


def binary_sa(h, n, max_iters, top_k, t_init, t_final, rng_seed=42):
    rng = np.random.RandomState(rng_seed)
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if abs(int(np.sum(h_bin)) - n_half) > 0:
        sort_idx = np.argsort(-h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_c5 = best_c5

    total = max(max_iters, 1)
    stall = 0

    for it in range(max_iters):
        if remaining() < 5:
            break

        frac = float(it) / total
        T = t_init * (t_final / t_init) ** frac

        corr = compute_corr(h_bin)
        idx_top = np.argsort(-corr)[:top_k]

        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        score_one = -g[ones]
        score_zero = g[zeros]

        best_delta = 0
        best_move = None

        n_try = min(10, len(ones))
        top_ones = ones[np.argsort(-score_one)[:n_try]]
        n_try_z = min(10, len(zeros))

        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_move = (i, j)

        # 2-swap with small search
        if remaining() > 10:
            n_try2 = min(3, len(ones))
            top_ones2 = ones[np.argsort(-score_one)[:n_try2]]
            n_try_z2 = min(3, len(zeros))

                            j1, j2 = top_zeros2[c], top_zeros2[d]
                            h_bin[i1] = 0
                            h_bin[i2] = 0
                            h_bin[j1] = 1
                            h_bin[j2] = 1
                            nc = compute_c5(h_bin)
                            h_bin[i1] = 1
                            h_bin[i2] = 1
                            h_bin[j1] = 0
                            h_bin[j2] = 0
                            delta = nc - cur_c5
                            if delta < best_delta:
                                best_delta = delta
                                best_move = (i1, i2, j1, j2)

        if best_move is not None:
            if len(best_move) == 2:
                i, j = best_move
                h_bin[i] = 0
                h_bin[j] = 1
            else:
                i1, i2, j1, j2 = best_move
                h_bin[i1] = 0
                h_bin[i2] = 0
                h_bin[j1] = 1
                h_bin[j2] = 1

            stall = 0
        else:

            accepted = False

                else:
                    h_bin[i] = 1
                    h_bin[j] = 0

            else:

                    break

        if cur_c5 < best_c5 - 1e-15:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def top_equalize(h, n, top_k, max_iters):
    h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        if len(idx) < 2:
            break
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn

        improved = False
        for alpha in [0.04, 0.02, 0.01, 0.03, 0.005, 0.015]:
            h_try = proj(h + alpha * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
                break

        if not improved:
            for alpha in [0.02, 0.01, 0.03]:
                h_try = proj(h - alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def coordinate_descent(h, n, n_passes=3):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    for p in range(n_passes):
        if remaining() < 10:
            break
        improved_any = False
        for i in range(n):
            if remaining() < 3:
                break
            old_val = h[i]
            for trial in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(trial - old_val) < 1e-10:
                    continue
                h[i] = trial
                h_proj = proj(h, n)
                c5_try = compute_c5(h_proj)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    h = h_proj.copy()
                    improved_any = True
                    break
            h[i] = old_val
        if not improved_any:
            break

    return h, best_c5
