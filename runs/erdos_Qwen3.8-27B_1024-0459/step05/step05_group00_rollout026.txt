def binary_local_search(h, n, max_rounds=50):
    """
    For near-binary h, try flipping pairs to reduce max correlation.
    """
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        
        # Identify positions with h > 0.5 and h < 0.5
        pos = np.where(h > 0.5)[0]
        neg = np.where(h < 0.5)[0]
        
        if len(pos) == 0 or len(neg) == 0:
            break
        
        # For efficiency, only consider a subset
        # Sort by gradient direction: move mass from high grad to low grad
        g, _ = _get_top_gradient(h, n, K=10)
        
        best_improvement = 0
        best_i = -1
        best_j = -1
        
        # Limit search to top candidates
        pos_score = g[pos]  # higher = more beneficial to decrease
        neg_score = g[neg]  # lower = more beneficial to increase
        
        # Take top 20 from each
        pos_top = pos[np.argsort(-pos_score)][:min(20, len(pos))]
        neg_top = neg[np.argsort(neg_score)][:min(20, len(neg))]
        
        for i in pos_top:
            for j in neg_top:
                # Swap: h[i] -= delta, h[j] += delta
                # Determine delta based on current values
                delta = min(h[i], 1.0 - h[j])
                if delta < 0.05:
                    continue
                
                h_try = h.copy()
                h_try[i] -= delta
                h_try[j] += delta
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                
                improvement = best_c5 - c_try
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
                    best_h_try = h_try.copy()
        
        if best_i >= 0:
            h = best_h_try
            best_h = h.copy()
            best_c5 = best_c5 - best_improvement
        else:
            break
    
    return best_h, best_c5

def _get_top_gradient(h, n, K=10):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(50.0 * (c_top - c_top.max()))
    weights /= weights.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        g += weights[i] * grad_corr_k(h, k, n)
    return g, np.max(corr)

def cutting_plane(h, n, K=20, delta=0.05, max_iters=100):
    """Cutting plane LP with trust region"""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        idx = np.argsort(-corr)[:K]
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub_list = []
        b_ub_list = []
        
        for k in idx:
            g = grad_corr_k(h, k, n)
            fk = corr[k]
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_ub_list.append(row)
            b_ub_list.append(-fk + np.dot(g, h))
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Trust region
        I = np.eye(n)
        A_trust = np.vstack([np.hstack([I, np.zeros((n, 1))]),
                             np.hstack([-I, np.zeros((n, 1))])])
        b_trust = np.concatenate([h + delta, h + delta])
        
        A_ub = np.vstack([A_ub, A_trust])
        b_ub = np.concatenate([b_ub, b_trust])
        
        bounds = [(0, 1)] * n + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                t_new = res.x[-1]
                if t_new < compute_c5(h) - 1e-10:
                    # Line search
                    best_val = compute_c5(h)
                    for alpha in [1.0, 0.7, 0.5, 0.3]:
                        h_try = project_feasible(h + alpha * (h_new - h), n)
                        c_try = compute_c5(h_try)
                        if c_try < best_val:
                            best_val = c_try
                            h = h_try
                    if compute_c5(h) < best_c5:
                        best_c5 = compute_c5(h)
                        best_h = h.copy()
                elif it > 50:
                    break
        except Exception:
            break
    
    return best_h, best_c5
