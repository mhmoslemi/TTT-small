import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TOPK = 5
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        m = n - offset
        g[offset:offset + m] += (1.0 - h[:m])
        g[:m] -= h[offset:offset + m]
    else:
        ao = -offset
        m = n - ao
        g[:m] += (1.0 - h[ao:ao + m])
        g[ao:ao + m] -= h[:m]
    return g


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 200.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad(x, ki, n)
    g *= dx
    g += 400.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_smooth(h, n, beta, top_k=5, maxiter=500):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxiter * 15})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h


def topk_qp_equalize(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_top = corr[idx]
        c_max = c_top[0]
        c_min_top = c_top[-1]

        if c_max - c_min_top < 1e-9:
            # Already equalized; try to reduce all
            G = np.stack([corr_grad(h, ki, n) for ki in idx])
            g_mean = np.mean(G, axis=0)
            g_mean -= np.mean(g_mean)
            gn = np.linalg.norm(g_mean)
            if gn < 1e-13:
                break
            g_mean /= gn
        else:
            G = np.stack([corr_grad(h, ki, n) for ki in idx])
            c_avg = np.mean(c_top)
            target_shift = -(c_top - c_avg)
            GGt = G @ G.T
            try:
                w = G.T @ np.linalg.solve(GGt + 1e-10 * np.eye(top_k), target_shift)
            except Exception:
                w = G.T @ np.linalg.pinv(GGt) @ target_shift
            w -= np.mean(w)
            g_mean = w
            gn = np.linalg.norm(g_mean)
            if gn < 1e-13:
                break
            g_mean /= gn

        improved = False
        for alpha in [0.02, 0.01, 0.005, 0.002, 0.001, 0.03, 0.04, 0.0005]:
            h_try = proj(h + alpha * g_mean, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
            h_try = proj(h - alpha * g_mean, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                h = h_try
                best_c5 = c5_try
                improved = True
                break
        if not improved:
            break

    return h, best_c5


def binary_simulated_annealing(h_bin, n, rng, max_iters=100000):
    h_bin = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    best_h = h_bin.copy()
    best_corr = compute_corr(h_bin).copy()
    best_c5 = float(np.max(best_corr)) * dx
    cur_c5 = best_c5

    T_start = 0.15
    T_end = 0.001
    n_iters = max(1, max_iters)

    for it in range(n_iters):
        if remaining() < 3:
            break

        T = T_start * (T_end / T_start) ** (it / n_iters)

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Choose move type: 70% single swap, 20% double swap, 10% block shift
        move_type = rng.random()
        if move_type < 0.7:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if i == j:
                continue
            h_bin[i] = 0
            h_bin[j] = 1
            corr = compute_corr(h_bin)
            new_c5 = float(np.max(corr)) * dx
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / (T * dx)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_bin.copy()
            else:
                h_bin[i] = 1
                h_bin[j] = 0
        elif move_type < 0.9:
            # Double swap: swap two 1s with two 0s
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1 = int(ones[rng.randint(len(ones))])
            i2 = int(ones[rng.randint(len(ones))])
            j1 = int(zeros[rng.randint(len(zeros))])
            j2 = int(zeros[rng.randint(len(zeros))])
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            h_bin[i1] = 0
            h_bin[i2] = 0
            h_bin[j1] = 1
            h_bin[j2] = 1
            corr = compute_corr(h_bin)
            new_c5 = float(np.max(corr)) * dx
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / (T * dx * 2)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_bin.copy()
            else:
                h_bin[i1] = 1
                h_bin[i2] = 1
                h_bin[j1] = 0
                h_bin[j2] = 0
        else:
            # Block shift: move a block of consecutive 1s
            block_len = rng.randint(1, 4)
            if len(ones) < block_len:
                continue
            # Find a block of consecutive 1s
            segs = []
            start = ones[0]
            prev = ones[0]
            for idx in ones[1:]:
                if idx == prev + 1:
                    prev = idx
                else:
                    if prev - start + 1 >= block_len:
                        segs.append(start)
                    start = idx
                    prev = idx
            if prev - start + 1 >= block_len:
                segs.append(start)
            if not segs:
                continue
            seg_start = segs[rng.randint(len(segs))]
            shift = rng.randint(-3, 4)
            if shift == 0:
                continue
            new_start = seg_start + shift
            if new_start < 0 or new_start + block_len > n:
                continue
            # Check zeros at target
            target_positions = list(range(new_start, new_start + block_len))
            source_positions = list(range(seg_start, seg_start + block_len))
            if any(h_bin[p] == 1 for p in target_positions if p not in source_positions):
                continue
            h_bin[seg_start:seg_start + block_len] = 0
            h_bin[new_start:new_start + block_len] = 1
            corr = compute_corr(h_bin)
            new_c5 = float(np.max(corr)) * dx
            delta = new_c5 - cur_c5
            if delta < 0 or rng.random() < np.exp(-delta / (T * dx)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_bin.copy()
            else:
                h_bin[seg_start:seg_start + block_len] = 1
                h_bin[new_start:new_start + block_len] = 0

    return best_h, best_c5


def binary_local_search(h_bin, n, max_iters=10000, top_k=3, rng=None):

    h_bin = (h_bin > 0.5).astype(float).copy()

    cur_c5 = best_c5
    corr = compute_corr(h_bin)

        corr = compute_corr(h_bin)

        cur_c5 = float(np.max(corr)) * dx

        # Compute gradient direction
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad(h_bin, ki, n)
        g -= np.mean(g)

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_move = None

        # Evaluate top candidates for swap
        score_one = -g[ones]
        score_zero = g[zeros]
        n_try_o = min(20, len(ones))
        n_try_z = min(20, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try_o]]

        for i in top_ones:

                h_bin[i] = 0
                h_bin[j] = 1
                nc = float(np.max(compute_corr(h_bin))) * dx

        # Try double swaps

        top_ones2 = ones[np.argsort(-score_one)[:n_try_o2]]

        top_zeros2 = zeros[np.argsort(-score_zero)[:n_try_z2]]

        for i1 in top_ones2:
            for i2 in top_ones2[i1 > top_ones2[0]:] if i1 != top_ones2[0] else []:
                pass  # complex, skip for now

        # Random exploration if stuck
        if best_move is None:
            stuck_count += 1
            for _ in range(10):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0

                nc = float(np.max(compute_corr(h_bin))) * dx

                delta = nc - cur_c5

                    best_move = (i, j, 1)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def projected_gradient(h, n, top_k=5, max_iters=20000, step_init=0.02):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - c_top[-1] + 0.5)
        w = np.maximum(w, 0.1)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-8)] = 0.0
        g[(h <= 1e-8)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.03, 0.12)
        else:
            h = best_h.copy()
            v *= 0.4
            step *= 0.65

        if it % 2000 == 0 and it > 0:
            step *= 0.8
            momentum = min(momentum + 0.01, 0.93)

    return best_h, best_c5

    # Final projected gradient
    if remaining() > 20:
        h_pg2, c5_pg2 = projected_gradient(best_h, best_n, top_k=4, max_iters=8000, step_init=0.01)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()

    # Final QP
    if remaining() > 10:
        h_eq3, c5_eq3 = topk_qp_equalize(best_h, best_n, top_k=5, max_iters=150)
        if c5_eq3 < best_c5:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
