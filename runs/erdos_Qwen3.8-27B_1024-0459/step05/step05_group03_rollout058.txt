def swap_delta(h, a, b, n):
    """Compute change in all correlations when swapping h[a]:1->0, h[b]:0->1."""
    delta = np.zeros(2 * n - 1)
    off_shift = n - 1
    for d in range(-(n - 1), n):
        dd = 0.0
        if 0 <= a + d < n:
            dd += -(1.0 - h[a + d])
        if 0 <= a - d < n:
            dd += h[a - d]
        if 0 <= b + d < n:
            dd += (1.0 - h[b + d])
        if 0 <= b - d < n:
            dd += -h[b - d]
        delta[d + off_shift] = dd
    return delta

def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with smooth max approximation."""
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        corr = compute_corr(x)
        m = float(np.max(corr))
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        dx = 2.0 / len(x)
        return (m + np.log(float(np.sum(ex))) / beta) * dx
    
    def _grad(x):
        corr = compute_corr(x)
        m = float(np.max(corr))
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / float(np.sum(ex))
        dx = 2.0 / len(x)
        w = w * dx
        # Gradient of weighted correlation
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-18:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                a2 = -off
                np_ = n - a2
                g[:np_] += wk * (1.0 - x[a2:a2 + np_])
                g[a2:a2 + np_] -= wk * x[:np_]
        return g
    
    def _cons(x):
        return float(np.sum(x)) - n / 2.0
    
    def _cons_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def structured_pattern(n, type_str='balanced', rng=None):
    """Generate structured binary patterns."""
    h = np.zeros(n)
    half = n // 2
    if type_str == 'block':
        h[:half] = 1.0
    elif type_str == 'alternating':
        h[::2] = 0.0
        h[1::2] = 0.0
        for i in range(n):
            if i < half:
                h[i] = 1.0
            else:
                h[i] = 0.0
    elif type_str == 'quasi_periodic':
        period = max(4, n // 48)
        pattern = [1] * (period // 2) + [0] * (period - period // 2)
        for i in range(n):
            h[i] = pattern[i % period]
        # Fix count
        ones_count = np.sum(h)
        if ones_count > half:
            excess = int(ones_count - half)
            idx = np.where(h == 1)[0]
            rng.shuffle(idx)
            h[idx[:excess]] = 0.0
        elif ones_count < half:
            deficit = int(half - ones_count)
            idx = np.where(h == 0)[0]
            rng.shuffle(idx)
            h[idx[:deficit]] = 1.0
    elif type_str == 'goldens':
        # Based on golden ratio irrational rotation
        phi = (1.0 + 5.0 ** 0.5) / 2.0
        for i in range(n):
            h[i] = 1.0 if np.floor((i + 1) / phi) - np.floor(i / phi) == 1 else 0.0
        # Fix count
        ones_count = np.sum(h)
        if ones_count > half:
            excess = int(ones_count - half)
            idx = np.where(h == 1)[0]
            rng.shuffle(idx)
            h[idx[:excess]] = 0.0
        elif ones_count < half:
            deficit = int(half - ones_count)
            idx = np.where(h == 0)[0]
            rng.shuffle(idx)
            h[idx[:deficit]] = 1.0
    elif type_str == 'random':
        positions = np.arange(n)
        rng.shuffle(positions)
        h[positions[:half]] = 1.0
    return h

        h_interp = project_feasible(h_interp, n5)
        h_opt, c5_opt = optimize_at_resolution(h_interp, n5, alloc_per_res // 2, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n5
    
    # Multi-start with structured patterns at parent resolution
    if remaining() > 60:
        for pat_type in ['quasi_periodic', 'goldens', 'block']:
            if remaining() < 30:
                break
            h_pat = structured_pattern(parent_n, pat_type, rng)
            h_opt, c5_opt = optimize_at_resolution(h_pat, parent_n, 30, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = parent_n
    
    # Final intensive polish at best resolution
    if remaining() > 30:
        n = best_n
        h = best_h.copy()
        # Binary search at best
        h_bin = round_to_binary(h, n)
        h_b, c5_b = binary_greedy_improve(h_bin, n, max_rounds=200, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()
        
        # Anneal
        if remaining() > 20:
            h_a, c5_a = binary_anneal(h, n, max_iters=5000, T0=1.5, T1=0.005, rng=rng)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()
                h = h_a.copy()
        
        # SLSQP
        if remaining() > 15:
            for beta in [50000, 200000]:
                if remaining() < 8:
                    break
                h_s = slsqp_smooth(h, n, beta, maxiter=200)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                    h = h_s.copy()
        
        # Subgradient
        if remaining() > 10:
            h_sg, c5_sg = subgrad_equalize(h, n, max_iters=500, top_k=12, rng=rng)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
        
        # Final binary
        if remaining() > 8:
            h_bin_f = round_to_binary(best_h, best_n)
            h_bf, c5_bf = binary_greedy_improve(h_bin_f, best_n, max_rounds=100, rng=rng)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
