    # d/d(h[j]) = (1-h[j+d]) if j <= n-1-d [from h[i-d] term where i=j+d, i-d=j]
    #           - h[j-d] if j >= d [from (1-h[i]) term where i=j]
    
    dm = w * dx  # weights times dx
    
    for d in range(1, n):
        # Positive offset d: c[n-1+d]
        wd = dm[n-1+d]
        if wd > 1e-16:
            np_ = n - d
            # g[j] += wd * (1-h[j-d]) for j = d, ..., n-1 (from h[j] in h[i+d]*(1-h[i]) where i=j-d)
            g[d:] += wd * (1.0 - h[:np_])
            # g[j] += wd * (-h[j+d]) for j = 0, ..., n-1-d (from (1-h[j]) in h[j+d]*(1-h[j]))
            g[:np_] -= wd * h[d:]
        
        # Negative offset d: c[n-1-d]
        wd = dm[n-1-d]
        if wd > 1e-16:
            np_ = n - d
            # c[n-1-d] = sum_{i=d}^{n-1} h[i-d]*(1-h[i])
            # Let j = i-d, i = j+d, j = 0,...,n-1-d
            # c[n-1-d] = sum_{j=0}^{n-1-d} h[j]*(1-h[j+d])
            # g[j] += wd*(1-h[j+d]) for j=0,...,n-1-d
            g[:np_] += wd * (1.0 - h[d:])
            # g[j] += wd*(-h[j-d]) for j=d,...,n-1
            g[d:] -= wd * h[:np_]
    
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def corr_gradient_weighted(h, weights):
    """Compute sum_k weights[k] * d(corr[k])/dh."""
    n = len(h)
    g = np.zeros(n)
    for d in range(1, n):
        # Positive offset d: corr[n-1+d] involves h[i] and h[i+d] for i=0..n-1-d
        wd = weights[n-1+d]
        if wd > 1e-16:
            np_ = n - d
            g[:np_] -= wd * h[d:]         # -h[i+d] w.r.t. h[i]
            g[d:] += wd * (1.0 - h[:np_]) # (1-h[i]) w.r.t. h[i+d]
        
        # Negative offset d: corr[n-1-d] involves h[i] and h[i+d] for i=0..n-1-d
        wd = weights[n-1-d]
        if wd > 1e-16:
            np_ = n - d
            g[:np_] += wd * (1.0 - h[d:]) # (1-h[i+d]) w.r.t. h[i]
            g[d:] -= wd * h[:np_]         # -h[i] w.r.t. h[i+d]
    return g

def smooth_obj_grad(h, beta, lam=0.0):
    """Objective and gradient for smooth max + penalty."""
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -100, 100)
    ex = np.exp(shifted)
    se = np.log(ex.sum()) / beta
    obj = (m + se) * dx
    
    # Gradient of smooth max
    w = ex / ex.sum()  # softmax weights
    w *= dx
    g = corr_gradient_weighted(h, w)
    
    # Add penalty gradient
    if lam > 0:
        s = np.sum(h) - n / 2.0
        obj += lam * s * s
        g += 2.0 * lam * s
    
    return obj, g

def lfbgbs_optimize(h, n, beta, lam, maxiter=500, ftol=1e-14):
    """L-BFGS-B optimization with smooth max + penalty."""
    h0 = project_feasible(h, n)
    
    def obj(x):
        f, _ = smooth_obj_grad(x, beta, lam)
        return f
    
    def grad(x):
        _, g = smooth_obj_grad(x, beta, lam)
        return g
    
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxfun': maxiter * 10})
        if res.fun < obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def lfbgbs_projected(h, n, beta, lam, maxiter=300, proj_every=5):
    """L-BFGS-B with periodic projection of sum constraint."""
    h_cur = project_feasible(h, n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    # Actually, let's just use L-BFGS-B and project at the end
    # The penalty should keep us feasible

    c5_opt = compute_c5(h_opt)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
    
    return best_h, best_c5

def peak_shave(h, n, top_k=10, max_iters=500, step0=0.1):
    """Targeted peak shaving: reduce the worst lags."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step0
    momentum = np.zeros(n)
    mu = 0.7
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        # Weight top-k lags
        top_idx = np.argsort(-corr)[:top_k]
        c_top = corr[top_idx]
        w_rel = c_top - c_top.min() + 1e-10
        w_rel = w_rel ** 3  # Sharper weighting
        w_rel /= w_rel.sum()
        
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_rel
        weights *= dx
        
        g = corr_gradient_weighted(h, weights)
        g -= np.mean(g)  # Preserve sum
        
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.95
            if step < 1e-6:
                break
        
        if it % 100 == 0:
            mu = min(0.9, mu + 0.02)
    
    return best_h, best_c5
