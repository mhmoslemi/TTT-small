def solve_1d_subproblem(p, q, lam):
    """Solve min_{t in [0,1]} max_d (p_d * t + q_d) + lam * t."""
    n = len(p)
    
    def f_val(t):
        vals = p * t + q + lam * t
        return np.max(vals)
    
    def f_subgrad(t):
        vals = p * t + q + lam * t
        mx = np.max(vals)
        active = np.abs(vals - mx) < 1e-12
        return np.min(p[active] + lam), np.max(p[active] + lam)
    
    lo, hi = 0.0, 1.0
    f_lo = f_val(lo)
    f_hi = f_val(hi)
    
    if f_lo <= f_hi + 1e-15:
        sg_lo, _ = f_subgrad(lo)
        if sg_lo > -1e-12:
            return lo
    if f_hi <= f_lo + 1e-15:
        _, sg_hi = f_subgrad(hi)
        if sg_hi < 1e-12:
            return hi

        if sg_min > 1e-10:
            hi = mid
        elif sg_max < -1e-10:
            lo = mid
        else:

def coordinate_descent(h, n, max_sweeps=200, rng=None, verbose=False):
    """Binary coordinate descent with exact 1D subproblem solving."""
    h = np.array(h, dtype=float).copy()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lam = 0.0
    
    if rng is None:
        rng = np.random.RandomState(0)
    
    for sweep in range(max_sweeps):
        rem = _BUDGET - (time.time() - _T0) - 3.0
        if rem < 2:
            break
        
        corr = compute_all_corrs(h)
        order = rng.permutation(n)
        
        for i in order:
            p = np.zeros(2 * n - 1)
            L = i
            R = n - 1 - i
            
            # First factor: index m in [i, i+L], p[m] = 1 - h[i+L-m]
            m_start = i
            m_end = i + L
            idx = np.arange(m_start, m_end + 1)
            src = np.arange(L, -1, -1)
            p[idx] += 1.0 - h[src]
            
            # Second factor: index m in [R, R+L], p[m] = -h[m - R]
            m2_start = R
            m2_end = R + L
            idx2 = np.arange(m2_start, m2_end + 1)
            p[idx2] += -h[np.arange(0, L + 1)]
            
            q = corr - p * h[i]
            t_new = solve_1d_subproblem(p, q, lam)
            delta = t_new - h[i]
            
            if abs(delta) > 1e-15:
                h[i] = t_new
                # Update corr incrementally
                # First factor regions
                corr[idx] += delta * (1.0 - h[src])
                corr[idx2] -= delta * h[np.arange(0, L + 1)]
                # Adjust for the quadratic term at k=0 (index n-1)
                # c_0 term: h[i]*(1-h[i]) -> linear approx was (1-2h[i])*delta
                # Actual change: (h[i]+d)(1-h[i]-d) - h[i](1-h[i]) = d(1-2h[i]-d)
                # We already added delta*(1-h[i_old]) - delta*h[i_old] = delta*(1-2*h[i_old])
                # Need to subtract delta^2
                # But we already updated h[i] = t_new, so h[i] is now t_new
                # The linear parts we added: delta * (1 - h[i_old]) for first factor at k=0
                # and -delta * h[i_old] for second factor at k=0
                # Actual change in c_0 from i: (t_new)(1-t_new) - h[i_old](1-h[i_old])
                # = t_new - t_new^2 - h[i_old] + h[i_old]^2
                # = delta - (t_new^2 - h[i_old]^2)
                # = delta - delta*(t_new + h[i_old])
                # = delta*(1 - t_new - h[i_old])
                # What we added: delta*(1-h[i_old]) + (-delta)*h[i_old] = delta*(1-2*h[i_old])
                # But h[i_old] was the OLD value. Since we updated h[i] before computing,
                # let me redo:
                # We computed p using h BEFORE the update. So p at index n-1 includes
                # (1-h[i_old]) from first factor and -h[i_old] from second factor.
                # So q[n-1] = corr[n-1] - p[n-1]*h[i_old]
                # After update: new c_0 = p[n-1]*t_new + q[n-1]
                # = p[n-1]*t_new + corr[n-1] - p[n-1]*h[i_old]
                # = corr[n-1] + p[n-1]*(t_new - h[i_old])
                # = corr[n-1] + p[n-1]*delta
                # But p[n-1] = (1-h[i_old]) + (-h[i_old]) = 1 - 2*h[i_old]
                # So new c_0 = corr[n-1] + delta*(1-2*h[i_old])
                # Actual new c_0 = corr[n-1] - h[i_old]*(1-h[i_old]) + t_new*(1-t_new)
                # = corr[n-1] + t_new - t_new^2 - h[i_old] + h[i_old]^2
                # = corr[n-1] + delta - (t_new^2 - h[i_old]^2)
                # = corr[n-1] + delta - delta*(t_new + h[i_old])
                # = corr[n-1] + delta*(1 - t_new - h[i_old])
                # Difference: delta*(1-2*h[i_old]) - delta*(1-t_new-h[i_old]) = delta*(t_new - h[i_old]) = delta^2
                # So we need to subtract delta^2 from corr[n-1]
                corr[n - 1] -= delta * delta
                
                if h[i] < 0:
                    h[i] = 0
                elif h[i] > 1:
                    h[i] = 1
        
        # Update lambda to maintain sum constraint
        s = np.sum(h)
        target = n / 2.0
        lam_update = 0.1 * (s - target)
        lam = 0.9 * lam + 0.1 * lam_update + 0.01 * (s - target)
        # Simpler: just project and reset lam
        h = project_feasible(h, n)
        lam = 0.0
        
        c5_curr = compute_c5(h)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
    
    return best_h, best_c5


def smooth_max_slsqp(h, n, beta, maxiter=500):
    dx = 2.0 / n
    
    def sm_obj(x):
        corr = compute_all_corrs(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx
    
    def sm_grad(x):
        corr = compute_all_corrs(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def sc(x):
        return np.sum(x) - n / 2.0
    
    def scj(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-12:
                return h_new
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def random_binary_h(n, rng):
    idx = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def perturb_h(h, n, sigma, rng):
    h_new = h + sigma * rng.randn(n)
    return project_feasible(h_new, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = random_binary_h(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)
    
    # Phase 1: Coordinate descent at parent resolution
    if remaining() > 30:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_sweeps=300, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = best_n
    
    # Phase 2: SLSQP polish
    for beta in [200, 500, 1000, 3000]:
        if remaining() < 10:
            break
        h_new = smooth_max_slsqp(best_h, best_n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 3: Coordinate descent again after SLSQP
    if remaining() > 30:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_sweeps=300, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 4: Try higher resolution
    for target_n in [768, 1024, 1536, 2048]:
        if remaining() < 60:
            break
        h_interp = interp_to(best_h, best_n, target_n)
        h_feas = project_feasible(h_interp, target_n)
        c5_int = compute_c5(h_feas)
        
        # Coordinate descent at new resolution
        h_cd, c5_cd = coordinate_descent(h_feas, target_n, max_sweeps=150, rng=rng)
        if c5_cd < c5_int:
            c5_int = c5_cd
            h_feas = h_cd.copy()
        
        # SLSQP at new resolution
        for beta in [500, 2000, 8000]:
            if remaining() < 15:
                break
            h_feas = smooth_max_slsqp(h_feas, target_n, beta, maxiter=300)
            c5_int = compute_c5(h_feas)
        
        # More coordinate descent
        if remaining() > 20:
            h_cd, c5_cd = coordinate_descent(h_feas, target_n, max_sweeps=200, rng=rng)
            if c5_cd < c5_int:
                c5_int = c5_cd
                h_feas = h_cd.copy()
        
        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_feas.copy()
            best_n = target_n
    
    # Phase 5: Multi-start exploration at best resolution
    n = best_n
    n_starts = 8
    for s in range(n_starts):
        if remaining() < 40:
            break
        
        # Different initializations
        if s == 0:
            # Perturb best solution
            h_try = perturb_h(best_h, n, 0.05, rng)
        elif s == 1:
            # Perturb more
            h_try = perturb_h(best_h, n, 0.1, rng)
        elif s == 2:
            # Random binary
            h_try = random_binary_h(n, rng)
        elif s == 3:
            # Structured: alternating blocks
            block = max(1, n // 32)
            h_try = np.zeros(n)
            for b in range(0, n, 2 * block):
                h_try[b:b+block] = 1.0
            h_try = project_feasible(h_try, n)
        elif s == 4:
            # Random with smoothing
            h_try = np.zeros(n)
            h_try[::2] = 1.0
            h_try = project_feasible(h_try, n)
            h_try = perturb_h(h_try, n, 0.15, rng)
        elif s == 5:
            # Cosine-like
            x = np.linspace(0, 2, n)
            h_try = 0.5 + 0.3 * np.cos(np.pi * x + 0.5)
            h_try = project_feasible(np.clip(h_try, 0, 1), n)
        elif s == 6:
            # Two-frequency
            x = np.linspace(0, 2, n)
            h_try = 0.5 + 0.25 * np.cos(3 * np.pi * x) + 0.15 * np.cos(7 * np.pi * x + 1.0)
            h_try = project_feasible(np.clip(h_try, 0, 1), n)
        else:
            # Random smooth
            freqs = rng.randint(1, 10, size=4)
            phases = rng.uniform(0, 2 * np.pi, size=4)
            amps = rng.uniform(0.1, 0.4, size=4)
            x = np.linspace(0, 2, n)
            h_try = 0.5 * np.ones(n)
            for fq, ph, am in zip(freqs, phases, amps):
                h_try += am * np.cos(fq * np.pi * x + ph)
            h_try = project_feasible(np.clip(h_try, 0, 1), n)
        
        c5_try = compute_c5(h_try)
        
        # Coordinate descent from this start
        h_cd, c5_cd = coordinate_descent(h_try, n, max_sweeps=150, rng=rng)
        if c5_cd < c5_try:
            c5_try = c5_cd
            h_try = h_cd.copy()
        
        # SLSQP polish
        if remaining() > 10 and c5_try < best_c5 + 0.05:
            for beta in [500, 2000]:
                if remaining() < 8:
                    break
                h_try = smooth_max_slsqp(h_try, n, beta, maxiter=200)
            c5_try = compute_c5(h_try)
        
        # Final CD
        if remaining() > 15 and c5_try < best_c5 + 0.02:
            h_cd, c5_cd = coordinate_descent(h_try, n, max_sweeps=100, rng=rng)
            if c5_cd < c5_try:
                c5_try = c5_cd
                h_try = h_cd copy if False else h_cd.copy()
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 6: Intensive refinement at best solution
    n = best_n
    h = best_h.copy()
    
    for cycle in range(5):
        if remaining() < 15:
            break
        
        # CD
        h_cd, c5_cd = coordinate_descent(h, n, max_sweeps=200, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h = h_cd.copy()
        
        # SLSQP
        if remaining() > 10:
            for beta in [3000, 10000, 30000]:
                if remaining() < 8:
                    break
                h = smooth_max_slsqp(h, n, beta, maxiter=400)
            c5_tmp = compute_c5(h)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()
    
    # Phase 7: Try even higher resolution with best solution
    for target_n in [3072, 4096]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, target_n)
        h_feas = project_feasible(h_interp, target_n)
        c5_int = compute_c5(h_feas)
        
        h_cd, c5_cd = coordinate_descent(h_feas, target_n, max_sweeps=100, rng=rng)
        if c5_cd < c5_int:
            c5_int = c5_cd
            h_feas = h_cd.copy()
        
        if remaining() > 15:
            h_feas = smooth_max_slsqp(h_feas, target_n, 5000, maxiter=200)
            c5_int = compute_c5(h_feas)
        
        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_feas.copy()
            best_n = target_n
    
    # Final: ensure feasibility and compute
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
