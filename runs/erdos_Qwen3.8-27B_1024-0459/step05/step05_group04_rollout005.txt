            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def random_binary(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            h[idx[:n_ones - target]] = 0
        else:
            idx = np.where(h == 0)[0]
            h[idx[:target - n_ones]] = 1
    return h

def continuous_optimize(h, n, time_limit):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    betas = [50, 200, 500, 2000, 5000, 10000, 20000, 50000]
    for beta in betas:
        if remaining() < 10:
            break
        t_alloc = min(40, remaining() * 0.4)
        if t_alloc < 5:
            break
        h1, c5_1 = nesterov_projected(h, n, beta, t_alloc, lr_init=0.05, mu=0.8)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            h = h1.copy()
        if remaining() < 10:
            break
        t_alloc2 = min(30, remaining() * 0.3)
        if t_alloc2 < 5:
            break
        h2, c5_2 = projected_gradient(h, n, beta, t_alloc2, lr_init=0.05)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            h = h2.copy()
        if remaining() > 10:
            h3, c5_3 = top_shifts_refine(h, n, min(15, remaining() * 0.2), top_k=6)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                h = h3.copy()
    return best_h, best_c5

def slsqp_step(h, n, beta, maxiter=300):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons_phi = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}]
    try:
        res = minimize(obj_wrapped, h0, jac=grad_wrapped, method='SLSQP',
                       bounds=bounds, constraints=cons_phi, args=(n, beta),
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None and np.all(res.x >= -0.1) and np.all(res.x <= 1.1):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def cons_fun(x, n):
    return np.sum(x) - n / 2.0

def cons_jac(x, n):
    return np.ones(n)

def obj_wrapped(x, n, beta):
    return obj_soft_max(x, n, beta)

def grad_wrapped(x, n, beta):
    return grad_soft_max(x, n, beta)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5

    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h_bin = np.zeros(n)
    sort_idx = np.argsort(-best_h)
    h_bin[sort_idx[:n // 2]] = 1.0

    # Phase 1: Binary SA at parent resolution
    t_bi = min(250, int(remaining() * 0.3))
    if t_bi > 20:
        h_b1, c5_b1 = binary_sa(h_bin, n, t_bi, rng, T_init=5.0, T_min=0.3)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n

    # Phase 2: Alternative binary starts
    starts = [thue_morse(n), random_binary(n, np.random.RandomState(seed + 7)),
              random_binary(n, np.random.RandomState(seed + 13))]
    for si, sp in enumerate(starts):
        if remaining() < 30:
            break
        t_s = min(100, int(remaining() * 0.12))
        if t_s < 10:
            break
        h_b2, c5_b2 = binary_sa(sp, n, t_s, np.random.RandomState(seed + 100 + si))
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = n

    # Phase 3: Continuous optimization at parent resolution
    if remaining() > 30:
        h_cont = best_h.copy()
        h_cont = project_feasible(h_cont, n)
        t_cont = min(200, int(remaining() * 0.3))
        if t_cont > 10:
            h_c1, c5_c1 = continuous_optimize(h_cont, n, t_cont)
            if c5_c1 < best_c5:
                best_c5 = c5_c1
                best_h = h_c1.copy()
                best_n = n

    # Phase 4: SLSQP polish
    if remaining() > 20:
        for beta in [5000, 20000, 50000]:
            if remaining() < 10:
                break
            h_new = slsqp_step(best_h, best_n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 5: Try higher resolutions
    for n_t in [n + 32, n + 64, n + 128]:
        if remaining() < 50:
            break
        if n_t > 1200:
            break
        if n_t % 2 != 0:
            n_t += 1
        h_interp = interp_to(best_h, best_n, n_t)
        h_interp = project_feasible(h_interp, n_t)
        t_high = min(100, int(remaining() * 0.2))
        if t_high < 10:
            break
        h_fh, c5_fh = continuous_optimize(h_interp, n_t, t_high)
        if c5_fh < best_c5:
            best_c5 = c5_fh
            best_h = h_fh.copy()
            best_n = n_t

    # Phase 6: Binary search at best resolution (refined)
    if remaining() > 40:
        n_final = best_n
        h_bin_f = (best_h > 0.5).astype(float)
        n_half = n_final // 2
        n_ones = int(h_bin_f.sum())
        if n_ones != n_half:
            if n_ones > n_half:
                idx = np.where(h_bin_f == 1)[0]
                h_bin_f[idx[:n_ones - n_half]] = 0
            else:
                idx = np.where(h_bin_f == 0)[0]
                h_bin_f[idx[:n_half - n_ones]] = 1
        t_fin = min(120, int(remaining() * 0.4))
        if t_fin > 10:
            h_bf, c5_bf = binary_sa(h_bin_f, n_final, t_fin, np.random.RandomState(seed + 999), T_init=3.0, T_min=0.3)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()
                best_n = n_final

    # Phase 7: Final continuous polish
    if remaining() > 15:
        n_final = best_n
        h_final = project_feasible(best_h, n_final)
        for beta in [10000, 30000, 80000]:
            if remaining() < 8:
                break
            h_new = projected_gradient(h_final, n_final, beta, min(15, remaining() * 0.3), lr_init=0.03)
            c5_new = compute_c5(h_new[0])
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new[0]
                h_final = h_new[0]
        if remaining() > 10:
            h_eq, c5_eq = top_shifts_refine(best_h, best_n, min(20, remaining() * 0.4), top_k=8)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    return (best_h, best_c5, best_n)
