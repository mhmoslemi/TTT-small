```python
def reweighting_opt(h, n, iters=100, p=10):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    w = np.ones(2*n-1)
    for it in range(iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        # Update weights
        w = np.maximum(0, corr - m + 1.0) ** p  # shift so all positive
        w = w / np.sum(w)
        # Compute weighted gradient
        g = np.zeros(n)
        for k in range(2*n-1):
            if w[k] < 1e-12:
                continue
            off = k - (n-1)
            g += w[k] * corr_gradient_at(h, off, n)
        g *= dx
        # Project to feasible
        g -= np.mean(g)
        # Line search
        best_step = 0.01
        best_c5 = compute_c5(h)
        for step in [0.05, 0.02, 0.01, 0.005, 0.1, 0.001]:
            h_new = project_feasible(h - step * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_step = step
                h = h_new
        if best_step == 0.01:
            # No improvement, try smaller steps
```

```python
                d_i[k_start2:k_end2+1] += h[i + np.arange(k_start2, k_end2 + 1)]
            
            # For each j in zeros, compute new max
            # new_c[k] = corr[k] + d_i[k] + d_j[k]
            base = corr + d_i
            
            for j in zeros:
                d_j = np.zeros(2*n - 1)
                # First term: 1-h[j-k] for valid k
                k_start = max(0, j - n + 1)
                k_end = min(j, 2*n - 2)
                if k_start <= k_end:
                    m_vals = j - np.arange(k_start, k_end + 1)
                    d_j[k_start:k_end+1] += 1.0 - h[m_vals]
                
                # Second term: -h[j+k] for valid k
                k_start2 = 0
                k_end2 = min(n - 1 - j, 2*n - 2)
                if k_start2 <= k_end2:
                    d_j[k_start2:k_end2+1] -= h[j + np.arange(k_start2, k_end2 + 1)]
                
                new_max = float(np.max(base + d_j))
                delta = new_max - cur_max
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            cur_max += best_delta
            corr = compute_corr(h)  # Recompute to avoid accumulation errors
            cur_max = float(np.max(corr))
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            break
    
    return best_h, best_max * dx
```
