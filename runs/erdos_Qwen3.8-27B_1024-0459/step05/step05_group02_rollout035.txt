def corr_grad_single(h, k, n):
    """Gradient of corr[k] with respect to h."""

    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / beta) * dx
    penalty = 100.0 * (np.sum(x) - tgt) ** 2
    return val + penalty


def smooth_grad_topk(x, n, beta, top_k, tgt):
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:top_k]
    c = corr[idx]
    m = c[0]
    sh = beta * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 200.0 * (np.sum(x) - tgt)
    return g


def smooth_grad_all(x, n, beta, tgt):
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    significant = w > 1e-8
    for j in np.where(significant)[0]:
        g += w[j] * corr_grad_single(x, j, n)
    g *= dx
    g += 200.0 * (np.sum(x) - tgt)
    return g


def lbfgs_opt(h, n, beta, top_k, maxiter, use_all_grad=False):
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    tgt = n / 2.0

    def obj(x):
        return smooth_obj_full(x, n, beta, tgt)

    def grad(x):
        if use_all_grad and n <= 512:
            return smooth_grad_all(x, n, beta, tgt)
        else:
            return smooth_grad_topk(x, n, beta, top_k, tgt)

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 12,
                                'maxls': 30})
        return proj(res.x, n)
    except Exception:
        return h


def equalize_corrs(h, n, top_k=6, max_iters=400):
    h = h.copy()
    best_c5 = compute_c5(h)
    step_sizes = [0.04, 0.025, 0.015, 0.008, 0.004, 0.002]

    for it in range(max_iters):
        if remaining() < 2:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_vals = corr[idx]
        if c_vals[0] - c_vals[-1] < 1e-9:
            break

        improved = False
        # Primary direction: max - min among top_k
        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn > 1e-13:
            d /= dn
            for alpha in step_sizes:
                h_try = proj(h + alpha * d, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
            if not improved:
                for alpha in step_sizes[:4]:
                    h_try = proj(h - alpha * d, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-15:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
                        break

        if not improved:
            # Try pairwise equalization
            for a in range(min(top_k - 1, 4)):
                for b in range(a + 1, min(top_k, 5)):
                    d2 = corr_grad_single(h, idx[b], n) - corr_grad_single(h, idx[a], n)
                    d2 -= np.mean(d2)
                    dn2 = np.linalg.norm(d2)
                    if dn2 < 1e-13:
                        continue
                    d2 /= dn2
                    for alpha in step_sizes[:4]:
                        h_try = proj(h + alpha * d2, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-15:
                            h = h_try
                            best_c5 = c5_try
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            break

    return h, best_c5


def binary_hill_climb(h, n, max_iters, top_k=5):
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if int(np.sum(h_bin)) != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    cur_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    stall_count = 0

    for it in range(max_iters):
        if remaining() < 2:
            break

        corr = compute_corr(h_bin)
        corr_s = corr * dx
        cur_c5 = float(np.max(corr_s))
        idx_top = np.argsort(-corr_s)[:top_k]

        g = np.zeros(n)
        w = corr_s[idx_top] - corr_s[idx_top].min() + 0.01
        w /= w.sum()
        for j, ki in enumerate(idx_top):
            g += w[j] * corr_grad_single(h_bin, ki, n)
        # Add anti-variance term
        small_idx = np.argsort(corr_s)[:3]
        for ki in small_idx:
            g -= 0.05 * corr_grad_single(h_bin, ki, n)
        g -= np.mean(g)

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        score_one = -g[ones]
        score_zero = g[zeros]

        n_try = min(20, len(ones))
        n_try_z = min(20, len(zeros))
        top_ones = ones[np.argsort(-score_one)[:n_try]]
        top_zeros = zeros[np.argsort(-score_zero)[:n_try_z]]

        best_delta = 0
        best_swap = None
        best_new_c5 = cur_c5

        for i in top_ones:
            for j in top_zeros:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = float(np.max(compute_corr(h_bin) * dx))
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_swap = (i, j)
                    best_new_c5 = nc

        if best_swap is None:
            stall_count += 1
            found = False
            n_rand = min(20, 5 + stall_count * 3)
            for _ in range(n_rand):
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = float(np.max(compute_corr(h_bin) * dx))
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                    best_new_c5 = nc
                    found = True

            if not found:
                if stall_count > 15:
                    # Multi-swap perturbation
                    n_pert = np.random.randint(2, 5)
                    for _ in range(n_pert):
                        i = ones[np.random.randint(len(ones))]
                        j = zeros[np.random.randint(len(zeros))]
                        if i != j:
                            h_bin[i] = 0
                            h_bin[j] = 1
                    cur_c5 = float(np.max(compute_corr(h_bin) * dx))
                    ones = np.where(h_bin == 1)[0]
                    zeros = np.where(h_bin == 0)[0]
                    stall_count = 0
                    continue
                else:
                    break
        else:
            stall_count = 0

        i, j = best_swap
        h_bin[i] = 0
        h_bin[j] = 1
        cur_c5 = best_new_c5
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def binary_sa(h, n, max_iters, t_init=0.003, t_min=0.00005, cool=0.9997):
    h_bin = (h > 0.5).astype(float).copy()
    n_half = n // 2
    if int(np.sum(h_bin)) != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    cur_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    cur_temp = t_init

    for it in range(max_iters):
        if remaining() < 2:
            break
        cur_temp = max(t_min, cur_temp * cool)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]

        h_bin[i] = 0
        h_bin[j] = 1
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5

        if delta < 0 or np.random.random() < np.exp(-delta / cur_temp):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Smooth optimization with increasing beta
    for beta in [100, 300, 1000, 3000, 8000, 20000, 50000]:
        if remaining() < 12:
            break
        use_all = (beta >= 10000 and n <= 512)
        h = lbfgs_opt(h, n, beta, top_k=6, maxiter=400, use_all_grad=use_all)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Equalize correlations
    if remaining() > 15:
        h, c5_eq = equalize_corrs(h, n, top_k=7, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h.copy()

    # Phase 3: Binary hill climbing
    if remaining() > 25:
        h_bin = (best_h > 0.5).astype(float)
        h_bg, c5_bg = binary_hill_climb(h_bin, n, max_iters=4000, top_k=5)
        if c5_bg < best_c5:
            best_c5 = c5_bg
        best_h_for_binary = h_bg if c5_bg < best_c5 else h_bin

    # Phase 4: Simulated annealing
    if remaining() > 25:
        h_sa, c5_sa = binary_sa(best_h_for_binary, n, max_iters=8000, t_init=0.002, t_min=0.00003, cool=0.99985)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 5: Refine best binary continuously
    if remaining() > 20:
        h_ref = best_h.copy() if (best_h > 0.5).sum() != (best_h < 0.5).sum() else best_h.copy()
        for beta in [8000, 25000, 60000]:
            if remaining() < 8:
                break
            h_ref = lbfgs_opt(h_ref, n, beta, top_k=5, maxiter=300, use_all_grad=True)
            c5 = compute_c5(h_ref)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()

    # Phase 6: Equalize again
    if remaining() > 12:
        h_eq2, c5_eq2 = equalize_corrs(best_h, n, top_k=8, max_iters=200)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 7: Try nearby resolutions
    n_candidates = sorted(set([
        n + 16, n + 32, n + 48, n + 64,
        int(n * 1.5), int(n * 1.75), int(n * 2.0),
        512, 768,
    ]))
    n_candidates = [nn for nn in n_candidates if 200 <= nn <= 1100 and nn % 2 == 0]
    n_candidates = [nn for nn in n_candidates if nn != n]
    n_candidates.sort(key=lambda x: abs(x - n))

    for n_t in n_candidates[:3]:
        if remaining() < 80:
            break
        h_interp = interp_to(best_h, n, n_t)
        h_feas = proj(h_interp, n_t)

        for beta in [2000, 10000, 30000]:
            if remaining() < 25:
                break
            h_feas = lbfgs_opt(h_feas, n_t, beta, top_k=5, maxiter=300, use_all_grad=(n_t <= 512))

        h_feas, c5_feas = equalize_corrs(h_feas, n_t, top_k=6, max_iters=150)

        h_b2 = (h_feas > 0.5).astype(float)
        h_bg2, c5_bg2 = binary_hill_climb(h_b2, n_t, max_iters=2000, top_k=5)

        c5_cand = min(c5_feas, c5_bg2)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_feas.copy() if c5_feas <= c5_bg2 else h_bg2.copy()
            best_n = n_t
            n = n_t

    # Phase 8: Final intensive binary search
    if remaining() > 30 and best_n == n:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_hill_climb(h_bin_f, n, max_iters=3000, top_k=6)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        # SA on best
        if remaining() > 20:
            h_sa2, c5_sa2 = binary_sa(h_bf, n, max_iters=5000, t_init=0.001, t_min=0.00002, cool=0.9999)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()

    # Phase 9: Final continuous polish
    if remaining() > 10:
        for beta in [50000, 100000]:
            if remaining() < 6:
                break
            hpf = lbfgs_opt(best_h, best_n, beta, top_k=5, maxiter=200, use_all_grad=True)
            c5pf = compute_c5(hpf)
            if c5pf < best_c5:
                best_c5 = c5pf
                best_h = hpf.copy()

    final_c5 = compute_c5(proj(best_h, best_n))
    best_h = proj(best_h, best_n)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
