```python
def penalty_lbfgsb(h, n, beta, lambda_pen, maxiter=200):
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        sm = m + np.log(np.sum(ex)) / beta
        penalty = lambda_pen * (np.sum(x) - n/2)**2
        return sm * (2.0/n) + penalty
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        dx = 2.0/n
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k] * dx
            if wk < 1e-15: continue
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao+np_])
                g[ao:ao+np_] -= wk * x[:np_]
        g += 2 * lambda_pen * (np.sum(x) - n/2)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    return project_feasible(res.x, n)
```
