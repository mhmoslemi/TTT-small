def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    g = np.zeros(n)

        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def smooth_max_obj_grad(h, n, beta):
    """Compute smooth max objective and gradient."""
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -50.0, 50.0)
    ex = np.exp(cl)
    obj = (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    w = ex / np.sum(ex)
    g = np.zeros(n)
    kc = w * (2.0 / n)
    for k in range(2 * n - 1):
        wk = kc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return obj, g

def cut_plane_lp(h, n, K=30, delta=0.05):
