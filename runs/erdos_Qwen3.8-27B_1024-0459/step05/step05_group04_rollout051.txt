        inner_iter = 0
        max_inner = 200
        while inner_iter < max_inner:
            if time.time() - start > total_time * 0.9:
                break
            h_new = h - step_size * g
            h_new = project_feasible(h_new, n)
            
            obj_new = softmax_objective(h_new, n, beta)
            if obj_new < obj:
                h = h_new
                obj = obj_new
                inner_iter += 1
                # Update gradient
                g = softmax_gradient(h, n, beta)
                g -= np.mean(g)
            else:
                step_size *= 0.5
                if step_size < 1e-10:
                    break
        
        # Check actual C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def top_shifts_equalize_custom(h, n, top_k=8, max_iters=1000, time_limit=100):
    """More aggressive equalization of top shifts."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-12:
            # All top shifts are equal, check if we can push them all down
            idx_all = np.argsort(-corr)[:top_k + 2]
            c_max = corr[idx_all[0]]
            c_min_all = corr[idx_all[-1]]
            if c_max - c_min_all < 1e-10:
                break
        
        # Compute combined gradient for top shifts (to push them down)
        # and for next shifts (to not push them up too much)
        g_top = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_top[:np_] += (1.0 - h[off:off+np_])
                g_top[off:] += -h[:np_]
            else:
                ao = -off
                np_ = n - ao
                g_top[ao:] += (1.0 - h[:np_])
                g_top[:np_] += -h[ao:]
        
        # Gradient to push up the shifts just below top
        idx_below = np.argsort(-corr)[top_k:top_k+4]
        g_below = np.zeros(n)
        for k in idx_below:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_below[:np_] += (1.0 - h[off:off+np_])
                g_below[off:] += -h[:np_]
            else:
                ao = -off
                np_ = n - ao
                g_below[ao:] += (1.0 - h[:np_])
                g_below[:np_] += -h[ao:]
        
        # Direction: push top down, allow below to go up
        d = -g_top + 0.3 * g_below
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        # Make sum-neutral
        d -= np.mean(d)
        
        # Line search
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.001, 0.1, 0.15, 0.0005]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.001, 0.002, 0.0005, 0.003, 0.0008]:
                ht = project_feasible(h + step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-15:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
