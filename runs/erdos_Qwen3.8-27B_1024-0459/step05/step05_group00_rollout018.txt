def smooth_max_grad(x, beta, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum

    g = np.zeros(n)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for k in range(nfull):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            g[offset:offset + npairs] += wk * (1.0 - x[:npairs])
            g[:npairs] -= wk * x[offset:offset + npairs]
        else:
            a = -offset
            npairs = n - a
            g[:npairs] += wk * (1.0 - x[a:a + npairs])
            g[a:a + npairs] -= wk * x[:npairs]
    return g


def slsqp_optimize(h, n, beta, maxiter=500):
    def obj_func(x):
        return smooth_max_value(x, beta, n)

    def grad_func(x):
        return smooth_max_grad(x, beta, n)

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h.copy()


def precompute_dcorr(h, n):
    nfull = 2 * n - 1
    k_range = np.arange(nfull)
    dcorr = np.zeros((n, nfull))
    for i in range(n):
        j1 = i + k_range - (n - 1)
        j2 = i - k_range + (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        dcorr[i, v1] += (1.0 - h[j1[v1]])
        dcorr[i, v2] -= h[j2[v2]]
    return dcorr


def local_pair_search(h, n, time_limit=60, max_passes=20):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    for _pass in range(max_passes):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 15:
            break

        corr = compute_corr(h)
        c_max_val = float(np.max(corr))

        ones_mask = h >= 0.5
        zeros_mask = h <= 0.5
        ones_count = int(np.count_nonzero(ones_mask))
        zeros_count = int(np.count_nonzero(zeros_mask))

        if ones_count < 2 or zeros_count < 2:
            break

        dcorr = precompute_dcorr(h, n)
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]

        improved = False
        nfull = 2 * n - 1

        for i in ones_idx:
            if _BUDGET - (time.time() - _T0) < 15:
                break
            base = corr - dcorr[i]
            base_max = np.max(base)
            if base_max >= c_max_val - 1e-13:
                continue

            new_maxs = np.max(base[None, :] + dcorr[zeros_idx], axis=1)
            best_pos = int(np.argmin(new_maxs))
            best_new_max = float(new_maxs[best_pos])

            if best_new_max < c_max_val - 1e-12:
                j = zeros_idx[best_pos]
                h[i] = 0.0
                h[j] = 1.0
                h = project_feasible(h, n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    improved = True
                    break

        if not improved:
            break

    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    x_new = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(x_new, xp, src_h)


def subgradient_step(h, n, K=20, beta_w=50.0, step_size=0.05):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta_w * (c_top - c_top.max()))
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= w_sum

    g = np.zeros(n)
    nfull = 2 * n - 1
    for wt, k in zip(weights, idx):
        if wt < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            np_ = n - offset
            g[offset:offset + np_] += wt * (1.0 - h[:np_])
            g[:np_] -= wt * h[offset:offset + np_]
        else:
            a = -offset
            np_ = n - a
            g[:np_] += wt * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wt * h[:np_]
    return g


def subgradient_descent(h, n, time_limit=60):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    n_iter = max(20, int(time_limit * 40))

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break

        progress = it / max(n_iter, 1)
        K = 30 if progress < 0.3 else (20 if progress < 0.7 else 12)
        bw = 30.0 if progress < 0.3 else (60.0 if progress < 0.7 else 100.0)

        g = subgradient_step(h, n, K=K, beta_w=bw)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        base_step = 0.04 / (1.0 + 0.05 * it)
        step = min(base_step, 0.2) * n / (gn + 1e-14)

        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h - step * factor * g, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    # Phase 0: Initialize from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        best_n = len(parent_h)
        best_h = project_feasible(parent_h, best_n)
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = project_feasible(best_h, best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with increasing beta at parent resolution
    n = best_n
    h = best_h.copy()
    for beta in [200, 800, 3000, 10000, 30000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Subgradient descent at parent resolution
    if remaining() > 15:
        h_sg, c5_sg = subgradient_descent(h.copy(), n, time_limit=min(40.0, remaining() * 0.2))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 3: Local pair search at parent resolution
    if remaining() > 10:
        h_lp, c5_lp = local_pair_search(h.copy(), n, time_limit=min(50.0, remaining() * 0.25))
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()

    # SLSQP polish after pair search
    for beta in [15000, 50000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(h.copy(), n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Try higher resolutions
    res_list = []
    for mult in [1.5, 2.0, 2.5, 3.0]:
        r = int(round(best_n * mult / 64) * 64)
        if 64 <= r <= 3072 and r % 2 == 0 and r != best_n:
            res_list.append(r)
    res_list = sorted(set(res_list))

    for n_t in res_list:
        if remaining() < 60:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)

        # SLSQP at new resolution
        for beta in [500, 3000, 10000]:
            if remaining() < 15:
                break
            h_cur = slsqp_optimize(h_feas.copy(), n_t, beta, maxiter=400)
            c5_cur = compute_c5(h_cur)
            if c5_cur < c5_int:
                c5_int = c5_cur
                h_feas = h_cur.copy()

        # Subgradient at new resolution
        if remaining() > 15:
            h_sg2, c5_sg2 = subgradient_descent(h_feas.copy(), n_t, time_limit=min(20.0, remaining() * 0.2))
            if c5_sg2 < c5_int:
                c5_int = c5_sg2
                h_feas = h_sg2.copy()

        # Local pair search at new resolution (only if n_t not too large)
        if n_t <= 1024 and remaining() > 15:
            h_lp2, c5_lp2 = local_pair_search(h_feas.copy(), n_t, time_limit=min(30.0, remaining() * 0.2))
            if c5_lp2 < c5_int:
                c5_int = c5_lp2
                h_feas = h_lp2.copy()

        # Final SLSQP
        if remaining() > 10:
            h_final = slsqp_optimize(h_feas.copy(), n_t, 20000, maxiter=500)
            c5_fin = compute_c5(h_final)
            if c5_fin < c5_int:
                c5_int = c5_fin
                h_feas = h_final.copy()

        if c5_int < best_c5 - 1e-12:
            best_c5 = c5_int
            best_h = h_feas.copy()
            best_n = n_t
            n = n_t
            h = h_feas.copy()

    # Phase 5: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    for cycle in range(3):
        if remaining() < 20:
            break

        # SLSQP
        for beta in [20000, 60000, 150000]:
            if remaining() < 8:
                break
            h_new = slsqp_optimize(h.copy(), n, beta, maxiter=500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        # Local pair search
        if n <= 1024 and remaining() > 15:
            h_lp, c5_lp = local_pair_search(h.copy(), n, time_limit=min(30.0, remaining() * 0.25))
            if c5_lp < best_c5 - 1e-12:
                best_c5 = c5_lp
                best_h = h_lp.copy()
                h = h_lp.copy()

        # Subgradient
        if remaining() > 10:
            h_sg, c5_sg = subgradient_descent(h.copy(), n, time_limit=min(15.0, remaining() * 0.2))
            if c5_sg < best_c5 - 1e-12:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    # Phase 6: Final polish
    for beta in [50000, 100000, 200000]:
        if remaining() < 5:
            break
        h_new = slsqp_optimize(best_h.copy(), best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final local pair search if resolution allows
    if best_n <= 1024 and remaining() > 10:
        h_lp, c5_lp = local_pair_search(best_h.copy(), best_n, time_limit=min(20.0, remaining() * 0.5))
        if c5_lp < best_c5 - 1e-13:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Ensure final feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
