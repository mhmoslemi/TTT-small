```python
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty

def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 100.0 * (np.sum(x) - _TGT)
    return g

def lbfgs_pass(h, n, beta, top_k=5, maxiter=600):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 10})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h

def compute_F_matrix(h, n):
    """Precompute F(x, d) = (1-h[x+d])*[valid] - h[x-d]*[valid] for all x, d."""
    # d ranges from -(n-1) to (n-1), total 2n-1 values
    # F has shape (n, 2n-1)
    F = np.zeros((n, 2 * n - 1))
    for d in range(-(n-1), n):
        col = d + (n - 1)  # column index
        # term1: (1-h[x+d]) for valid x+d
        # x+d in [0, n-1] => x in [-d, n-1-d] => x in [max(0,-d), min(n-1, n-1-d)]
        if d >= 0:
            # x+d valid when x in [0, n-1-d]
            for x in range(n - d):
                F[x, col] += (1.0 - h[x + d])
            # x-d valid when x in [d, n-1]
            for x in range(d, n):
                F[x, col] -= h[x - d]
        else:
            # d < 0, let ao = -d
            ao = -d
            # x+d = x-ao valid when x in [ao, n-1]
            for x in range(ao, n):
                F[x, col] += (1.0 - h[x - ao])
            # x-d = x+ao valid when x in [0, n-1-ao]
            for x in range(n - ao):
                F[x, col] -= h[x + ao]
    return F
```

```python
                F[:n-d, col] += (1.0 - h[d:n])
            # term2: positions x where x-d >= 0, i.e., x >= d

            F[ao:, col] += (1.0 - h[:n-ao])
            # term2: positions x where x+ao < n, i.e., x < n-ao
            F[:n-ao, col] -= h[ao:]
```

```python
def best_swap_local_search(h_bin, n, max_iters=5000):
    """Find best individual swap using precomputed F matrix."""
    h = h_bin.copy()
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr)) * dx
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        
        # Recompute F (since h changed)
        F = compute_F_matrix(h, n)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_score = np.inf
        best_swap = None
        
        for i in ones:
            # For this i, find best j
            # new_corr[k] = corr[k] + F[j, k] - F[i, k] for all k
            # score[j] = max(new_corr)
            diff = F[zeros, :] - F[i, :]  # (n_zeros, 2n-1)
            new_corrs = corr[None, :] + diff
            scores = new_corrs.max(axis=1)
            j_local = zeros[scores.argmin()]
            if scores.min() < best_score:
                best_score = scores.min()
                best_swap = (i, j_local)
        
        if best_swap is None:
            break
        
        cur_c5 = float(np.max(corr)) * dx
        if best_score * dx >= cur_c5 - 1e-12:
            # No improvement possible
            break
        
        i, j = best_swap
        h[i] = 0
        h[j] = 1
        corr = compute_corr(h)
        cur_c5 = float(np.max(corr)) * dx
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
def sa_local_search(h_bin, n, max_iters=2000, T_init=0.5, T_final=0.01):
    """Simulated annealing on binary sequence."""
    h = h_bin.copy()
    dx = 2.0 / n
    best_h = h.copy()
    corr = compute_corr(h)
    cur_c5 = float(np.max(corr)) * dx
    best_c5 = cur_c5
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        
        T = T_init * (T_final / T_init) ** (it / max_iters)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_corr = compute_corr(h)
        new_c5 = float(np.max(new_corr)) * dx
        
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            corr = new_corr
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5
```

```python
    """For all pairs (i,j) with h[i]=1, h[j]=0, compute the max of new_corr.
    Returns a (n_ones, n_zeros) array of new max correlations."""
    corr = compute_corr(h)

    zeros = np.where(h == 0)[0]
    n_ones = len(ones)

            F[ao:, col] += h[:n-ao]
            # x-d = x+ao valid: x <= n-1-ao
            F[:n-ao, col] -= (1.0 - h[ao:])
    
    # Approximate scores: new_corr[m] ≈ corr[m] + F[i, m] - F[j, m]
    # (using the formula Δcorr = A(i,d) - A(j,d) where A = F as defined above)
    
    # Wait, I defined F[x,d] = h[x+d] - (1-h[x-d]).
    # And Δcorr[m] = A(i,d) - A(j,d) where A(x,d) = h[x+d] - (1-h[x-d]).
    # So new_corr[m] ≈ corr[m] + F[i,m] - F[j,m]
    
    scores = np.full((n_ones, n_zeros), np.inf)
    for ii in range(n_ones):
        i = ones[ii]
        # For all j: new_corr ≈ corr + F[i,:] - F[j,:]
        # score[j] = max(new_corr)
        diff = F[zeros, :] - F[i, :]  # (n_zeros, 2n-1)
        new_corrs = corr[None, :] + diff
        scores[ii, :] = new_corrs.max(axis=1)
    
    return scores, ones, zeros
```

This is O(n_ones * n_zeros * n) = O(n³/4) ≈ 14M for n=384. In numpy, the inner loop (for each i, computing the max over a (192, 767) array) should take about 1-2ms. With 192 iterations, total is about 0.2-0.4s per full pass.
