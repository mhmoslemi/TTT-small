import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TOPK = 10
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 100.0 * (np.sum(x) - _TGT) ** 2
    return val + penalty


def _sm_grad(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]
    sh = _BETA * (c - m)
    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j, ki in enumerate(idx):
        g += w[j] * corr_grad_single(x, ki, n)
    g *= dx
    g += 200.0 * (np.sum(x) - _TGT)
    return g


def lbfgs_pass(h, n, beta, top_k=10, maxiter=800):

    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = top_k
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 10})
        h_new = proj(res.x, n)
        return h_new
    except Exception:
        return h

def top_equalize(h, n, top_k=8, max_iters=500):

        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-9:
            break

        g_max = corr_grad_single(h, idx[0], n)
        g_min = corr_grad_single(h, idx[-1], n)
        d = g_min - g_max
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_max
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn

        improved = False
        for scale in [4, 2, 1, 0.5, 0.25, 6, 0.1, 8, 0.05, 3]:
            alpha = 0.01 * scale

            if c5_try < best_c5 - 1e-15:
                h = h_try
                best_c5 = c5_try
                improved = True

                if c5_try < best_c5 - 1e-15:

                    best_c5 = c5_try
                    improved = True

def binary_sah(h_init, n, max_iters=5000, top_k=5, T_init=0.05, T_min=0.001):
    """Simulated annealing on binary solutions with multi-swap moves."""
    h_bin = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_bin)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    dx = 2.0 / n
    cur_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = cur_c5
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

    for it in range(max_iters):
        if remaining() < 3:
            break
        T = T_init * (T_min / T_init) ** (it / max_iters)
        
        # Choose move type
        if np.random.random() < 0.7:
            # Single swap
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c5 = new_c5
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        else:
            # Double swap (2 ones -> 2 zeros)
            if len(ones) < 2 or len(zeros) < 2:
                break
            i1, i2 = np.random.choice(len(ones), 2, replace=False)
            j1, j2 = np.random.choice(len(zeros), 2, replace=False)
            oi = h_bin.copy()
            h_bin[i1] = 0.0
            h_bin[i2] = 0.0
            h_bin[j1] = 1.0
            h_bin[j2] = 1.0
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):

                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
            else:
                h_bin[:] = oi[:]

    return best_h, best_c5


def binary_local_search(h_init, n, max_iters=3000, top_k=4):

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def fourier_init(n, rng):
    """Generate a random initialization with controlled frequency content."""
    x = (np.arange(n) + 0.5) * 2.0 / n
    h = 0.5 * np.ones(n)
    # Add a few low-frequency components
    for _ in range(rng.randint(2, 5)):
        f = rng.uniform(1.0, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        h += amp * np.cos(2 * np.pi * f * x + phase)
    h = np.clip(h, 0, 1)
    h = proj(h, n)
    return h

def iterated_binary_search(h_init, n, n_rounds=5, max_iters_per_round=2000, top_k=4):
    """Iterated local search: optimize, perturb, re-optimize."""
    best_h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(best_h))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        best_h = np.zeros(n)
        best_h[sort_idx[:n_half]] = 1.0

    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()

    for rnd in range(n_rounds):
        if remaining() < 5:
            break
        # Local search
        cur_h, cur_c5 = binary_local_search(cur_h, n, max_iters=max_iters_per_round, top_k=top_k)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = cur_h.copy()

        if remaining() < 5:
            break

        # Perturb: flip 3-8 random bits
        n_flip = np.random.randint(3, 9)
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if len(ones) < n_flip or len(zeros) < n_flip:
            break
        flip_ones = np.random.choice(len(ones), min(n_flip, len(ones)), replace=False)
        flip_zeros = np.random.choice(len(zeros), min(n_flip, len(zeros)), replace=False)
        nf = min(len(flip_ones), len(flip_zeros))
        for idx_i in flip_ones[:nf]:
            cur_h[ones[idx_i]] = 0.0
        for idx_j in flip_zeros[:nf]:
            cur_h[zeros[idx_j]] = 1.0

    return best_h, best_c5


def refine_continuous(h_bin, n, max_beta=50000):
    """Refine a binary solution with continuous optimization."""
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for beta in [2000, 8000, 20000, max_beta]:
        if remaining() < 6:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=10, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = best_h.copy()
    # Top equalize
    if remaining() > 5:
        h_eq, c5_eq = top_equalize(best_h, n, top_k=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Interpolate to higher resolution and optimize
    n_target = max(parent_n, 512)
    n_target = n_target // 2 * 2  # ensure even
    if n_target != parent_n:
        h_work = interp_to(best_h, parent_n, n_target)
        h_work = proj(h_work, n_target)
    else:
        h_work = best_h.copy()
        n_target = parent_n
    n = n_target

    for beta in [50, 200, 800, 3000, 10000, 30000]:
        if remaining() < 12:
            break
        h_new = lbfgs_pass(h_work, n, beta, top_k=12, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h_work = h_new.copy()

    # Phase 2: Nesterov gradient
    if remaining() > 80:
        h_pg, c5_pg = nesterov_grad(h_work, n, top_k=12, max_iters=12000, step_init=0.02)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n
            h_work = h_pg.copy()

    # Phase 3: Top equalization
    if remaining() > 30:
        h_eq, c5_eq = top_equalize(h_work, n, top_k=12, max_iters=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            best_n = n
            h_work = h_eq.copy()

    # Phase 4: Aggressive binary optimization
    if remaining() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_bl, c5_bl = iterated_binary_search(h_bin, n, n_rounds=6, max_iters_per_round=1800, top_k=5)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            best_n = n

        # Continuous refinement of best binary
        if remaining() > 30:
            h_ref, c5_ref = refine_continuous(h_bl, n, max_beta=40000)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n

    # Phase 5: Simulated annealing
    if remaining() > 80:
        h_bin_sa = (best_h > 0.5).astype(float)
        h_sa, c5_sa = binary_sah(h_bin_sa, n, max_iters=6000, T_init=0.03, T_min=0.0002)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n
        # Refine
        if remaining() > 25:
            h_ref_sa, c5_ref_sa = refine_continuous(h_sa, n, max_beta=30000)
            if c5_ref_sa < best_c5:
                best_c5 = c5_ref_sa
                best_h = h_ref_sa.copy()
                best_n = n

    # Phase 6: Try even higher resolution
    n_high2 = max(n + 128, int(n * 1.25) // 2 * 2)
    if n_high2 <= 900 and remaining() > 150:
        h_hi = interp_to(best_h, best_n, n_high2)
        h_hi = proj(h_hi, n_high2)
        for beta in [500, 3000, 15000]:
            if remaining() < 12:
                break
            h_hi = lbfgs_pass(h_hi, n_high2, beta, top_k=12, maxiter=400)
        c5_hi = compute_c5(h_hi)
        if remaining() > 40:
            h_hi, c5_hi_eq = top_equalize(h_hi, n_high2, top_k=10, max_iters=300)
            c5_hi = min(c5_hi, c5_hi_eq)
        if remaining() > 60:
            h_bin_hi = (h_hi > 0.5).astype(float)
            h_bl_hi, c5_bl_hi = iterated_binary_search(h_bin_hi, n_high2, n_rounds=4,
                                                       max_iters_per_round=1200, top_k=4)
            c5_hi = min(c5_hi, c5_bl_hi)
            if c5_bl_hi < c5_hi:
                h_hi = h_bl_hi
        if remaining() > 25:
            h_ref_hi, c5_ref_hi = refine_continuous(h_hi, n_high2, max_beta=25000)
            c5_hi = min(c5_hi, c5_ref_hi)
            if c5_ref_hi < c5_hi:
                h_hi = h_ref_hi
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_high2

    # Phase 7: More L-BFGS at higher beta
    n = best_n
    h = best_h.copy()
    for beta in [20000, 50000, 100000]:
        if remaining() < 10:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=15, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: Final top equalize with larger k
    if remaining() > 15:
        h_eq2, c5_eq2 = top_equalize(h, n, top_k=15, max_iters=400)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Phase 9: Final binary polish
    if remaining() > 40:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = iterated_binary_search(h_bin_f, best_n, n_rounds=4,
                                             max_iters_per_round=1500, top_k=5)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        if remaining() > 20:
            h_rf, c5_rf = refine_continuous(h_bf, best_n, max_beta=50000)
            if c5_rf < best_c5:
                best_c5 = c5_rf
                best_h = h_rf.copy()

    # Phase 10: Small perturbation restarts
    for _ in range(3):
        if remaining() < 30:
            break
        h_pert = best_h.copy()
        mask = (h_pert > 0.1) & (h_pert < 0.9)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_pert[mask] += rng.uniform(-0.08, 0.08, size=cnt)
        h_pert = proj(h_pert, best_n)
        h_pert = lbfgs_pass(h_pert, best_n, 10000, top_k=12, maxiter=300)
        c5_pert = compute_c5(h_pert)
        if remaining() > 8:
            h_pert, c5_pert = top_equalize(h_pert, best_n, top_k=10, max_iters=200)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()

    # Ensure feasibility and final evaluation
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
