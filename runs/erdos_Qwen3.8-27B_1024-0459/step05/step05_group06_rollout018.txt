def compute_c5(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            min_val = np.min(h)
            mask2 = h == min_val
            h[mask2] += diff / np.sum(mask2)
        else:
            max_val = np.max(h)
            mask2 = h == max_val
            h[mask2] += diff / np.sum(mask2)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(free)
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def slsqp_optimize(h, n, beta, maxiter=400):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})

    beta_coeff = np.zeros(ncorr)
    gamma_coeff = np.zeros(ncorr)
    gamma_coeff[n - 1] = -1.0

    if max_off_src > 0:

            beta_coeff[n - 1 + off] += (1.0 - best[i + off])
    # h[i] as target: valid if i-off >= 0, i.e., off <= i
    max_off_tgt = i
    if max_off_tgt > 0:
        for off in range(1, max_off_tgt + 1):
            beta_coeff[n - 1 + off] += (-best[i - off])
    # off < 0: aoff = -off, k = n-1-aoff
    # h[i] as source: valid if i-aoff >= 0, i.e., aoff <= i

    _RNG = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution from parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: SLSQP polish with moderate beta
    for beta in [2000, 10000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_optimize(h, n, beta, 350)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-16:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: Coordinate descent at current resolution
    if remaining_time() > 60:
        h_new, c5_new = coord_descent(h, n, n_sweeps=40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Thresholding + binary local search
    if remaining_time() > 80:
        h_bin = (h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Local search on binary
        h_bin2, c5_bin2 = local_search_binary(h_bin, n, n_iters=5000)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Phase 4: Un-threshold and coordinate descent
    if remaining_time() > 60:
        h_unth = best_h.copy()
        # Add small noise to break 0-1 structure
        frac_mask = (h_unth > 0.01) & (h_unth < 0.99)
        if np.sum(frac_mask) < 10:
            h_unth = best_h + _RNG.uniform(-0.05, 0.05, n) * 0.1
            h_unth = project_feasible(h_unth, n)
        h_new, c5_new = coord_descent(h_unth, n, n_sweeps=35)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: SLSQP with high beta for precision
    if remaining_time() > 30:
        h_new = slsqp_optimize(best_h, n, 50000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: More coordinate descent
    if remaining_time() > 40:
        h_new, c5_new = coord_descent(best_h, n, n_sweeps=30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 7: Try higher resolution
    higher_res = [int(best_n * 1.5), int(best_n * 2.0)]
    # Ensure even
    for idx_hr in range(len(higher_res)):
        if higher_res[idx_hr] % 2 != 0:
            higher_res[idx_hr] += 1
    higher_res = [rn for rn in higher_res if 64 <= rn <= 2048]
    higher_res = sorted(set(higher_res))

    for rn in higher_res:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        # SLSQP at higher res
        if remaining_time() > 40:
            h_r = slsqp_optimize(h_r, rn, 5000, 250)

        # Coordinate descent at higher res (fewer sweeps due to cost)
        if remaining_time() > 40:
            h_r, c5_r = coord_descent(h_r, rn, n_sweeps=n_sweeps_for_res(rn))

        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 8: Binary local search at best resolution
    if remaining_time() > 60:
        h_b = (best_h > 0.5).astype(float)
        h_b = project_feasible(h_b, best_n)
        h_b2, c5_b2 = local_search_binary(h_b, best_n, n_iters=30000)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
        # Also try coordinate descent from the binary solution
        if c5_b2 < best_c5 + 0.001 and remaining_time() > 30:
            h_cd, c5_cd = coord_descent(h_b2, best_n, n_sweeps=25)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 9: Random perturbations
    n_perturb = max(10, best_n // 20)
    for pert in range(20):
        if remaining_time() < 30:
            break
        h_pert = best_h.copy()
        idx_p = _RNG.choice(best_n, n_pert, replace=False)
        h_pert[idx_p] += _RNG.uniform(-0.2, 0.2, n_pert)
        h_pert = project_feasible(h_pert, best_n)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5 + 0.002:
            h_new, c5_new = coord_descent(h_pert, best_n, n_sweeps=20)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 10: Final polish
    if remaining_time() > 15:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=20)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 10:
        h_new = slsqp_optimize(best_h, best_n, 200000, 150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining_time() > 5:
        h_new, c5_new = coord_descent(best_h, best_n, n_sweeps=10)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final validation and constraint enforcement
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        cnt = np.sum(free)
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)
    # One more projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)

def n_sweeps_for_res(n):
    if n <= 640:
        return 30
    elif n <= 1000:
        return 20
    elif n <= 1400:
        return 15
    else:
        return 10
