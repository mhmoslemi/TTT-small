        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

    def obj_fn(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val
    
    def grad_fn(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)

            if wk < 1e-15:

                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g


def topk_gradient(h, n, K=25, beta=50.0):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    w = np.exp(beta * (c_top - c_top.max()))
    s = w.sum()
    if s < 1e-30:
        w = np.ones(K) / K
    else:
        w /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if w[i] > 1e-14:
            g += w[i] * grad_corr_k(h, k, n)
    return g


def adam_optimizer(h, n, time_limit, rng):
    lr = 0.015
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    start = time.time()
    no_improve = 0
    best_lr = lr

    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        t += 1

        progress = min(t / 5000.0, 1.0)
        K = int(35 - 20 * progress)
        K = max(K, 8)
        beta_sm = 20.0 + 80.0 * progress

        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        w = np.exp(beta_sm * (corr[idx] - c_max))
        w /= w.sum()

        g = np.zeros(n)
        for i, k in enumerate(idx):
            if w[i] > 1e-14:
                g += w[i] * grad_corr_k(h, k, n)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * g * g
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_t = best_lr * (1.0 + 0.3 / (1.0 + t / 200.0))
        h_new = project_feasible(h - lr_t * m_hat / (np.sqrt(v_hat) + eps), n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-16:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

            no_improve += 1

                    lr = best_lr

                        best_h = h_try.copy()
                        best_c5 = c5_try

            best_lr *= 0.98

    return best_h, best_c5


def smooth_max_slsqp(h, n, beta, maxiter=300):
    dx_val = 2.0 / n
    h0 = h.copy()

    def obj_fn(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx_val

    def grad_fn(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx_val
        active = np.where(wc > 1e-12)[0]
        for k in active:
            wk = wc[k]
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def sc_fn(x):
        return np.sum(x) - n / 2.0

    def sc_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc_fn, 'jac': sc_jac}]

    try:
        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c_check = compute_c5(h_new)
            if c_check <= compute_c5(h0) + 1e-10:
                return h_new
    except Exception:
        pass

    return h0


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def pair_swap_optimize(h, n, time_limit, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    deltas = [0.001, 0.002, 0.005, 0.01, 0.003, 0.008, 0.015, 0.02]

    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = deltas[rng.randint(len(deltas))]
        s = 1.0 if rng.rand() < 0.5 else -1.0

        if 0.0 <= h[i] + s * d <= 1.0 and 0.0 <= h[j] - s * d <= 1.0:
            h_try = h.copy()
            h_try[i] += s * d
            h_try[j] -= s * d
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_h = h_try.copy()
                best_c5 = c5_try
                h = h_try

    return best_h, best_c5


def coord_descent(h, n, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    h = h.copy()
    candidates = [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]

    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        improved = False
        for i in range(n):
            old = h[i]
            for d in candidates:
                new_val = old + d
                if new_val < 0.0 or new_val > 1.0:
                    continue
                h[i] = new_val
                h_feas = project_feasible(h, n)
                c5_try = compute_c5(h_feas)
                if c5_try < best_c5 - 1e-16:
                    best_c5 = c5_try
                    best_h = h_feas.copy()
                    h = h_feas.copy()
                    improved = True
                    break
                else:
                    h[i] = old
        if not improved:
            break

    return best_h, best_c5


def running_subgradient(h, n, time_limit, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    start = time.time()
    step_schedule = 0.08

    while time.time() - start < time_limit:
        if _BUDGET - (time.time() - _T0) < 5:
            break

        corr = compute_corr(h)
        k_max = int(np.argmax(corr))
        g = grad_corr_k(h, k_max, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        direction = -g / gn
        improved = False
        for step in [step_schedule, step_schedule * 0.5, step_schedule * 2,
                     step_schedule * 0.25, step_schedule * 0.1, step_schedule * 4]:
            h_try = project_feasible(h + step * direction, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-16:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try.copy()
                improved = True
                break

        if not improved:
            step_schedule *= 0.95
            if step_schedule < 0.001:
                break

    return best_h, best_c5
