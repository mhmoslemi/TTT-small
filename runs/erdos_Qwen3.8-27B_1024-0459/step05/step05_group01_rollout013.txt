        v = mu * v - step * g
        h_new = project_feasible(h + v, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            no_improve = 0
            step = min(step * 1.1, 0.2)
        else:
            no_improve += 1
            step *= 0.8
            v *= 0.5
            if no_improve > 50:
                # Reset and try smaller step
                v = np.zeros(n)
                step = 0.001
                no_improve = 0
                if step < 1e-7:
                    break
    
    return best_h, best_c5

def binary_full_sweep(h, n, max_rounds=100, time_limit=60.0):
    """Exhaustive best-swap binary optimization"""
    h = h.copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    start = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_c5_try = best_c5
        best_swap = None
        
        # Sample if too many pairs
        n_pairs = len(ones) * len(zeros)
        if n_pairs > 40000:
            # Random sampling
            rng = np.random.RandomState(rnd)
            n_samples = 40000
            for _ in range(n_samples):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_corr = compute_corr(h)
                new_max = np.max(new_corr) * dx
                h[i] = 1.0
                h[j] = 0.0
                if new_max < best_c5_try - 1e-15:
                    best_c5_try = new_max
                    best_swap = (i, j)
        else:
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    new_corr = compute_corr(h)
                    new_max = np.max(new_corr) * dx
                    h[i] = 1.0
                    h[j] = 0.0
                    if new_max < best_c5_try - 1e-15:
                        best_c5_try = new_max
                        best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corr = compute_corr(h)
        best_c5 = np.max(corr) * dx
    
    return h, best_c5

def slsqp_smooth_max(h, n, beta, maxiter=500):
    """SLSQP with smooth max approximation"""
    h0 = project_feasible(h, n)
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        return (m + np.log(np.sum(ez)) / beta) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        z = np.clip(beta * (corr - m), -50.0, 50.0)
        ez = np.exp(z)
        S = np.sum(ez)
        w = ez / S
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:])
                g[a:] -= wk * x[:np_]
        return g
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.success or res.fun < obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0
