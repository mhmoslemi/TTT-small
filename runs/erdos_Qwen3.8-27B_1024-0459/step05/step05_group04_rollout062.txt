    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)
    hi3 = min(j + n - 1, r - 1)
    if lo3 <= hi3:
        ks = np.arange(lo3, hi3 + 1)
        dr[lo3:hi3+1] += (1.0 - h[j + n - 1 - ks])
    lo4 = max(n - 1 - j, 0)
    hi4 = min(2 * (n - 1) - j, r - 1)
    if lo4 <= hi4:
        ks = np.arange(lo4, hi4 + 1)
        dr[lo4:hi4+1] -= h[j + ks - (n - 1)]
    dr[n - 1] -= 2.0
    return dr

def block_move_delta(h, start, end, new_start, n):
    """Compute correlation change for moving block [start:end] to new_start."""
    n_hook = end - start
    r = 2 * n - 1
    dr = np.zeros(r)
    old_block = h[start:end].copy()
    # Subtract contribution of old block positions
    for pos in range(start, end):
        lo = max(pos, 0)
        hi = min(pos + n - 1, r - 1)
        if lo <= hi:
            ks = np.arange(lo, hi + 1)
            dr[lo:hi+1] -= (1.0 - h[pos + n - 1 - ks]) * old_block[pos - start]
        lo2 = max(n - 1 - pos, 0)
        hi2 = min(2*(n-1) - pos, r - 1)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2 + 1)
            dr[lo2:hi2+1] -= h[pos + ks - (n - 1)] * old_block[pos - start]
    # Add contribution of new block positions
    for idx in range(n_hook):
        pos = new_start + idx
        if pos < 0 or pos >= n:
            break
        lo = max(pos, 0)
        hi = min(pos + n - 1, r - 1)
        if lo <= hi:
            ks = np.arange(lo, hi + 1)
            dr[lo:hi+1] += (1.0 - h[pos + n - 1 - ks]) * old_block[idx]
        lo2 = max(n - 1 - pos, 0)
        hi2 = min(2*(n-1) - pos, r - 1)
        if lo2 <= hi2:
            ks = np.arange(lo2, hi2 + 1)
            dr[lo2:hi2+1] += h[pos + ks - (n - 1)] * old_block[idx]
    return dr

def binary_sa(h_init, n, time_limit=200, rng=None):
    """Simulated annealing on binary with swap and block moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h_init > 0.5).copy().astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h_init)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    start_t = time.time()
    T0 = 0.5
    Tend = 0.01
    n_steps = 20000
    cool_rate = (Tend / T0) ** (1.0 / max(1, n_steps))
    T = T0
    
    stall = 0
    for step in range(n_steps):
        if time.time() - start_t > time_limit or remaining() < 15:
            break
        T *= cool_rate
        if T < 1e-8:
            T = 1e-8
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose move type
        move_type = rng.random()
        best_move = None
        best_new_max = cur_max + 1e-30
        found_improve = False
        
        if move_type < 0.7:
            # Single swap: pick from high-correlation area + random
            top_k = np.argsort(-corr)[:6]
            candidates = set()
            for s in top_k:
                off = s - (n - 1)
                if off >= 0:
                    np_ = n - off
                    contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                    for ci in contrib[:8]:
                        candidates.add((int(ci + off), int(ci)))
                else:
                    ao = -off
                    np_ = n - ao
                    contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                    for ci in contrib[:8]:
                        candidates.add((int(ci), int(ci + ao)))
            # Random candidates
            for _ in range(20):
                candidates.add((int(ones[rng.randint(len(ones))]), int(zeros[rng.randint(len(zeros))])))
            
            for (i, j) in candidates:
                if h[i] != 1 or h[j] != 0:
                    continue
                dr = make_swap_delta(h, i, j, n)
                new_corr = corr + dr
                new_max = float(np.max(new_corr))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_move = ('swap', i, j, dr)
                    if new_max < cur_max:
                        found_improve = True
        else:
            # Block move: move a short block of 1s
            block_len = rng.randint(1, min(4, len(ones)//3 + 1))
            if len(ones) > block_len:
                # Pick a random contiguous block of 1s
                candidates_bs = []
                for _ in range(10):
                    ri = rng.randint(0, len(ones))
                    start_pos = int(ones[ri])
                    # Find contiguous ones from start_pos
                    end_pos = start_pos + 1
                    while end_pos < n and h[end_pos] == 1 and end_pos - start_pos < block_len:
                        end_pos += 1
                    blen = end_pos - start_pos
                    if blen >= 1 and blen <= block_len:
                        new_start = rng.randint(0, n - blen + 1)
                        if new_start != start_pos:
                            # Check no overlap
                            if new_start + blen <= start_pos or new_start >= end_pos:
                                candidates_bs.append((start_pos, end_pos, new_start))
                
                for (bs, be, ns) in candidates_bs[:5]:
                    dr = block_move_delta(h, bs, be, ns, n)
                    new_corr = corr + dr
                    new_max = float(np.max(new_corr))
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_move = ('block', bs, be, ns, dr)
                        if new_max < cur_max:
                            found_improve = True
        
        if best_move is not None:
            mtype = best_move[0]
            if mtype == 'swap':
                _, i, j, dr = best_move
                delta = best_new_max - cur_max
                if delta < 0 or rng.random() < np.exp(-delta / (T * dx + 1e-15)):
                    h[i] = 0
                    h[j] = 1
                    corr = corr + dr
                    cur_max = best_new_max
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            elif mtype == 'block':
                _, bs, be, ns, dr = best_move
                delta = best_new_max - cur_max
                if delta < 0 or rng.random() < np.exp(-delta / (T * dx + 1e-15)):
                    old_block = h[bs:be].copy()
                    h[bs:be] = 0
                    h[ns:ns+(be-bs)] = old_block
                    corr = corr + dr
                    cur_max = best_new_max
                    if cur_max < best_max:
                        best_max = cur_max
                        best_h = h.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
        else:
            stall += 1
        
        if stall > 500:
            # Perturbation: accept 5 random worsening moves
            for _ in range(5):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    dr = make_swap_delta(h, i, j, n)
                    h[i] = 0
                    h[j] = 1
                    corr = corr + dr
            cur_max = float(np.max(corr))
            stall = 0
    
    return best_h, best_max * dx

def projected_gradient_step(h, n, beta, max_iter=2000, base_lr=0.05):
    """Custom projected gradient descent with softmax-approximated max."""
    dx = 2.0 / n
    n_full = 2 * n - 1
    
    def soften(corr):
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -80.0, 80.0)
        sw = np.exp(sh)
        sw /= np.sum(sw)
        return m + np.log(np.sum(np.exp(sh))) / beta, sw
    
    def softmax_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        obj, _ = soften(c)
        return obj * dx
    
    def softmax_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = np.clip(beta * (c - m), -80.0, 80.0)
        sw = np.exp(sh)
        sw /= np.sum(sw)
        sw *= dx
        g = np.zeros(n)
        for k in range(n_full):
            wk = sw[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    h_curr = project_feasible(h, n)
    best_x = h_curr.copy()
    best_val = softmax_obj(h_curr)
    vel = np.zeros(n)  # momentum
    
    for it in range(max_iter):
        if remaining() < 20:
            break
        lr = base_lr / (1.0 + it * 0.002)
        g = softmax_grad(h_curr)
        vel = 0.9 * vel - lr * g
        h_new = h_curr + vel
        h_new = project_feasible(h_new, n)
        
        new_val = softmax_obj(h_new)
        if new_val < best_val + 1e-14:
            h_curr = h_new
            if new_val < best_val:
                best_val = new_val
                best_x = h_new.copy()
        else:
            # Line search: backtrack
            for alpha in [0.5, 0.25, 0.1, 0.05]:
                h_try = project_feasible(h_curr + alpha * vel, n)
                v_try = softmax_obj(h_try)
                if v_try < best_val + 1e-14:
                    h_curr = h_try
                    if v_try < best_val:
                        best_val = v_try
                        best_x = h_try.copy()
                    break
            else:
                vel *= 0.5
    
    return best_x

def equalize_top_shifts(h, n, top_k=8, max_iters=500):
    """Iteratively equalize the top-k correlations."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if remaining() < 10:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        
        n_full = 2 * n - 1
        # Weighted gradient: push top-k down, pull bottom up
        g_top = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_top[off:off + np_] += (1.0 - h[:np_])
                g_top[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_top[:np_] += (1.0 - h[ao:ao + np_])
                g_top[ao:ao + np_] -= h[:np_]
        
        bot_idx = np.argsort(corr)[:min(4, n_full)]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        
        d = g_bot - g_top
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.005, 0.01, 0.02, 0.03, 0.05, 0.008, 0.015, 0.04, 0.08, 0.012]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    
    # Phase 1: Binary SA at parent resolution (exploitation)
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    
    t1 = min(200, int(remaining() * 0.25))
    if t1 > 30:
        h_b1, c5_b1 = binary_sa(h_bin, n, time_limit=t1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
    
    # Phase 2: Try alternative starting points at parent resolution
    alt_starts = [thue_morse(n)]
    # Random with spectral shaping
    for _ in range(2):
        freqs = np.fft.fftfreq(n)
        noise = np.zeros(n)
        for f in range(1, n // 2 + 1):
            amp = rng.random() * (1.0 / (1.0 + 0.1 * f))
            noise += amp * np.cos(2 * np.pi * f * np.arange(n) / n + rng.random() * 2 * np.pi)
        alt_starts.append((noise > np.median(noise)).astype(float))
    
    for sp in alt_starts:
        if remaining() < 40:
            break
        sp_ones = int(np.sum(sp))
        if sp_ones != n_half:
            sort_idx = np.argsort(-sp)
            sp = np.zeros(n)
            sp[sort_idx[:n_half]] = 1.0
        sp_c5 = compute_c5(sp)
        if sp_c5 < best_c5 + 0.02:
            t_alt = min(100, int(remaining() * 0.1))
            if t_alt > 20:
                h_b2, c5_b2 = binary_sa(sp, n, time_limit=t_alt, rng=rng)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_b2.copy()
    
    # Phase 3: Binary SA at other resolutions
    res_candidates = []
    for mult in [0.75, 1.0, 1.25, 1.5]:
        nr = int(n * mult)
        nr = nr if nr % 2 == 0 else nr + 1
        if 192 <= nr <= 640 and nr != n:
            res_candidates.append(nr)
    for delta in [-32, 16, 32, 64]:
        nr = n + delta
        if nr % 2 == 0 and 192 <= nr <= 640:
            res_candidates.append(nr)
    res_candidates = sorted(set(res_candidates))[:4]
    
    for n_t in res_candidates:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        
        t_bi = min(80, int(remaining() * 0.07))
        if t_bi > 15:
            h_b3, c5_b3 = binary_sa(h_bin_t, n_t, time_limit=t_bi, rng=rng)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t
    
    # Phase 4: Continuous optimization - projected gradient (main exploitation)
    n = best_n
    h = project_feasible(best_h, n)
    
    for beta in [200, 500, 1000, 2000, 5000, 10000, 20000, 40000, 80000]:
        if remaining() < 25:
            break
        t_pg = min(80, int(remaining() * 0.08))
        max_it = max(500, t_pg * 15)
        h_new = projected_gradient_step(h, n, beta, max_iter=max_it, base_lr=0.03)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Equalization
    if remaining() > 30:
        h_eq, c5_eq = equalize_top_shifts(h, n, top_k=6, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 6: More continuous refinement with very high beta
    for beta in [50000, 100000, 200000]:
        if remaining() < 15:
            break
        h_new = projected_gradient_step(h, n, beta, max_iter=1000, base_lr=0.01)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 7: Try finer resolution continuous
    if remaining() > 80:
        n_fine = min(n + 64, 512)
        if n_fine > n and n_fine % 2 == 0:
            h_fine = interp_to(best_h, n, n_fine)
            h_fine = project_feasible(h_fine, n_fine)
            for beta in [2000, 5000, 10000, 30000]:
                if remaining() < 20:
                    break
                h_fine = projected_gradient_step(h_fine, n_fine, beta, max_iter=800, base_lr=0.02)
            c5_fine = compute_c5(h_fine)
            if remaining() > 20:
                h_fine, c5_fine2 = equalize_top_shifts(h_fine, n_fine, top_k=6, max_iters=200)
                c5_fine = min(c5_fine, c5_fine2)
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine.copy()
                best_n = n_fine
    
    # Phase 8: Binary->Continuous refinement cycle
    if remaining() > 40:
        n = best_n
        # Round to binary, do quick SA, then continuous
        h_bin8 = (best_h > 0.5).astype(float)
        n_half = n // 2
        n_ones = int(np.sum(h_bin8))
        if n_ones != n_half:
            sort_idx = np.argsort(-best_h)
            h_bin8 = np.zeros(n)
            h_bin8[sort_idx[:n_half]] = 1.0
        h_b8, c5_b8 = binary_sa(h_bin8, n, time_limit=min(50, int(remaining() * 0.3)), rng=rng)
        h8 = project_feasible(h_b8, n)
        for beta in [5000, 20000, 50000]:
            if remaining() < 15:
                break
            h8 = projected_gradient_step(h8, n, beta, max_iter=500, base_lr=0.02)
        c5_8 = compute_c5(h8)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()
    
    # Phase 9: Final equalization and polish
    if remaining() > 15:
        n = best_n
        h = best_h.copy()
        h_eq, c5_eq = equalize_top_shifts(h, n, top_k=8, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
        h_new = projected_gradient_step(best_h, best_n, 100000, max_iter=500, base_lr=0.005)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
