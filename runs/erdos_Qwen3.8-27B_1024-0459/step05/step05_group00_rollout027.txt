def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Distribute among entries with most room
        if diff > 0:
            idx = np.argsort(1.0 - h)[:int(abs(diff)) + 1]
        else:
            idx = np.argsort(h)[:int(abs(diff)) + 1]
        for i in idx:
            adj = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += adj
            diff -= adj
    return h

        g[offset:] += (1.0 - h[:npairs])

Let me also add one more technique: **coordinate descent with exact line search**.

For each coordinate i:
- Fix all other coordinates
- The objective as a function of h[i] is: max_k f_k(h[i]) where each f_k is a quadratic in h[i]
- The maximum of quadratics is a piecewise quadratic function
- We can find the minimum by checking the critical points

This is exact for the 1D subproblem and might give better results than gradient-based methods.

For a fixed h[i-1], h[i+1], ..., the function f_k as a function of h[i] = x:
f_k(x) = a_k * x² + b_k * x + c_k (quadratic in x)

The maximum of these quadratics over x ∈ [0,1] (with the sum constraint absorbing the change) is a piecewise quadratic function. Its minimum can be found by:
1. Finding all pairwise intersection points of the quadratics
2. Evaluating the max at each intersection point
3. Also checking the boundaries x=0 and x=1

This is O(K²) per coordinate, and we have n coordinates, so O(n*K²) per full sweep. For n=512, K=20, that's about 200K operations per sweep. Very fast!

But the sum constraint complicates things: if we change h[i], we need to change another coordinate to maintain the sum. So it's really a 2D subproblem (change h[i] and h[j] simultaneously).

- h[i] += δ, h[j] -= δ (maintains sum)
- For a fixed pair (i,j), the objective is max_k f_k(δ) where each f_k is quadratic in δ
- Find the optimal δ by checking critical points

This is O(K²) per pair, and we try n² pairs (or a subset), giving O(n²K²) per sweep. For n=512, K=20, that's 512²*400 ≈ 100M. That might be too slow per sweep.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-15:
                break
            adj = np.sign(diff) * min(abs(diff), (1.0 - h[i]) if diff > 0 else h[i])
            if abs(adj) > 1e-16:
                h[i] += adj
                diff -= adj
    return h


def grad_corr_k(h, k, n):

        g[:npairs] -= h[offset:]
    else:

        candidate_starts = []
        
        # Random starts
        for _ in range(3):
            h_rand = random_feasible(n, rng)
            candidate_starts.append(h_rand)
        
        # Block structures
        for _ in range(3):
            h_blk = block_structure(n, rng)
            candidate_starts.append(h_blk)
        
        # Perturbed parent
        for _ in range(2):
            h_pert = h + rng.normal(0, 0.05, n)
            h_pert = project_feasible(h_pert, n)
            candidate_starts.append(h_pert)
        
        for h_start in candidate_starts:
            if remaining() < 30:
                break
            
            # Quick subgradient
            h_c, c5_c = subgradient_descent(h_start, n, min(10.0, remaining() * 0.2), rng)
            
            # Quick SLSQP
            if remaining() > 15:
                h_c = smooth_max_slsqp(h_c, n, 3000, maxiter=300)
                c5_c = compute_c5(h_c)
            
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                h = h_c.copy()
    
    # Phase 5: Multi-resolution search
    res_list = []
    for target in [384, 640, 768, 1024, 1280]:
        res_list.append(target)
    res_list = [r for r in res_list if r != best_n and 256 <= r <= 1536 and r % 2 == 0]
    res_list = sorted(set(res_list), key=lambda x: abs(x - best_n))
    
    for n_t in res_list[:4]:
        if remaining() < 60:
            break
        
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        
        # Subgradient at new resolution
        if remaining() > 30:
            h_cur, c5_cur = subgradient_descent(h_feas, n_t, min(25.0, remaining() * 0.3), rng)
            if c5_cur > c5_int:
                h_cur = h_feas.copy()
                c5_cur = c5_int
        else:
            h_cur = h_feas.copy()
            c5_cur = c5_int
        
        # SLSQP at new resolution
        for beta in [1000, 5000, 20000, 50000]:
            if remaining() < 15:
                break
            h_cur = smooth_max_slsqp(h_cur, n_t, beta, maxiter=500)
            c5_cur = compute_c5(h_cur)
        
        # Brief SA at new resolution
        if remaining() > 30 and c5_cur < best_c5 + 0.001:
            h_cur, c5_cur = simulated_annealing(h_cur, n_t, min(30.0, remaining() * 0.3), rng)
            if remaining() > 10:
                h_cur = smooth_max_slsqp(h_cur, n_t, 30000, maxiter=400)
                c5_cur = compute_c5(h_cur)
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n_t
            h = h_cur.copy()
            n = n_t
    
    # Phase 6: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    cycles = 3
    for cycle in range(cycles):
        if remaining() < 30:
            break
        
        # Subgradient
        if remaining() > 15:
            h, c5_tmp = subgradient_descent(h, n, min(20.0, remaining() * 0.2), rng)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()
        
        # SLSQP with high beta
        for beta in [10000, 30000, 80000, 200000]:
            if remaining() < 8:
                break
            h = smooth_max_slsqp(h, n, beta, maxiter=800)
            c5_tmp = compute_c5(h)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h.copy()
        
        # Brief SA for escape
        if remaining() > 20:
            h_sa, c5_sa = simulated_annealing(h, n, min(15.0, remaining() * 0.2), rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h = h_sa.copy()
                # Re-optimize
                if remaining() > 10:
                    h = smooth_max_slsqp(h, n, 50000, maxiter=600)
                    c5_chk = compute_c5(h)
                    if c5_chk < best_c5:
                        best_c5 = c5_chk
                        best_h = h.copy()
    
    # Phase 7: Final precision optimization
    h = best_h.copy()
    for beta in [50000, 100000, 200000, 500000]:
        if remaining() < 5:
            break
        h = smooth_max_slsqp(h, best_n, beta, maxiter=1000)
        c5_tmp = compute_c5(h)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h.copy()
    
    # Phase 8: Final micro-optimization with top-K gradient
    if remaining() > 5:
        h = best_h.copy()
        n = best_n
        for it in range(100):
            if remaining() < 3:
                break
            corr = compute_corr(h)
            top_k = np.argsort(-corr)[:12]
            c_top = corr[top_k]
            beta_w = 200.0
            w = np.exp(beta_w * (c_top - c_top.max()))
            w /= w.sum()
            
            g = np.zeros(n)
            for i, k in enumerate(top_k):
                g += w[i] * grad_corr_k(h, k, n)
            
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            
            for step in [0.005, 0.002, 0.01, 0.003, 0.008, 0.001]:
                h_try = project_feasible(h - step * g / gn, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
    
    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
