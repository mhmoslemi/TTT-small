import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def thue_morse_init(n):
    """Thue-Morse sequence: h[i] = 1 if popcount(i) is odd, 0 otherwise."""
    h = np.zeros(n)
    for i in range(n):
        # Count 1-bits in i
        v = i
        cnt = 0
        while v:
            cnt += (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 0 else 0.0
    # Ensure exactly n/2 ones (Thue-Morse has this property for n = 2^k)
    return project_feasible(h, n)

def interp_to_n(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    fp = src
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, fp)

def coord_descent(h, n, max_sweeps=300):
    """Fast coordinate descent for the Erdős problem."""
    best = h.astype(float).copy()
    corr = compute_corr(best)
    mid = n - 1
    ncorr = 2 * n - 1
    dx = 2.0 / n
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    
    # Precompute for each i the valid indices
    # For coordinate i, with offset k (where k = index - mid):
    # i_plus = i + k, valid when 0 <= i+k < n (contributes (1-h[i+k]))
    # i_minus = i - k, valid when 0 <= i-k < n (contributes -h[i-k])
    
    for sweep in range(max_sweeps):
        if remaining_time() < 3.0:
            break
        improved = False
        for i in range(n):
            if remaining_time() < 1.0:
                break
            old = best[i]
            
            # Compute A: A[k] = (1-h[i+k]) for valid i+k, minus h[i-k] for valid i-k
            # Using offset = k - mid, so k = offset + mid
            # i + offset in [0,n): contributes (1 - h[i+offset])
            # i - offset in [0,n): contributes -h[i-offset]
            
            i_plus = i + offsets  # array of length ncorr
            i_minus = i - offsets
            vp = (i_plus >= 0) & (i_plus < n)
            vm = (i_minus >= 0) & (i_minus < n)
            
            A = np.zeros(ncorr)
            if np.any(vm):
                A[vm] += (1.0 - best[i_minus[vm]])
            if np.any(vp):
                A[vp] -= best[i_plus[vp]]
            
            # For mid (offset 0): A[mid] = (1-best[i]) - best[i] = 1 - 2*old
            # But we handle the quadratic separately
            # Linear part for mid: coefficient is (1-2*old), quadratic part: -delta^2
            # Parameterization: new_corr[mid] = corr[mid] + (new-old)(1-2*old) - (new-old)^2
            # = corr[mid] - old + new + (1-2old)(new-old) - (new-old)^2 ... let me just use:
            # new_corr[mid] = best_new*(1-best_new) + (corr[mid] - old*(1-old))
            # So: new_corr[mid] = base_mid + quad_part where base_mid = corr[mid] - old*(1-old)
            # and quad_part = best_new*(1-best_new)
            
            # For k != mid: new_corr[k] = corr[k] + A[k]*(new-old) = (corr[k] - A[k]*old) + A[k]*new
            base = corr - A * old
            base[mid] = corr[mid] - old * (1.0 - old)  # quadratic will be added separately
            # A[mid] = 1 - 2*old (linear coefficient if we parameterize as base + A*new + quad(new))
            # but we'll handle mid separately in the evaluation
            
            # Objective: min new max( max_{k!=mid} [base[k] + A[k]*new], base[mid] + new*(1-new) )
            # where new in [0,1]
            
            # Quick check: evaluate at 0, 1, old
            val_old = np.max(corr)
            
            # At new=0:
            aff0 = base.copy()
            aff0[mid] += 0.0  # 0*(1-0) = 0
            v0 = np.max(aff0)
            
            # At new=1:
            aff1 = base + A
            aff1[mid] += 1.0 * (1.0 - 1.0)  # = 0
            v1 = np.max(aff1)
            
            best_new = old
            best_val = val_old
            if v0 < best_val - 1e-15:
                best_val = v0
                best_new = 0.0
            if v1 < best_val - 1e-15:
                best_val = v1
                best_new = 1.0
            
            # Try the "top-2 crossing" point
            # Evaluate at best_new to find which constraints are active
            aff_bn = base + A * best_new
            aff_bn[mid] = aff_bn[mid] + best_new * (1.0 - best_new)
            top2 = np.argsort(aff_bn)[::-1][:2]
            i1, i2 = int(top2[0]), int(top2[1])
            
            if i1 != mid and i2 != mid:
                if A[i1] != A[i2]:
                    x_star = (base[i2] - base[i1]) / (A[i1] - A[i2])
                    if 0.0 <= x_star <= 1.0:
                        aff_x = base + A * x_star
                        aff_x[mid] += x_star * (1.0 - x_star)
                        vx = np.max(aff_x)
                        if vx < best_val - 1e-15:
                            best_val = vx
                            best_new = x_star
            elif i1 == mid and i2 != mid:
                # crossing of quad and affine: base[mid] + x(1-x) = base[i2] + A[i2]*x
                # base[mid] + x - x² = base[i2] + A[i2]*x
                # x² + (A[i2]-1)x + (base[i2]-base[mid]) = 0
                # x = (1-A[i2] ± sqrt((1-A[i2])² - 4*(base[i2]-base[mid]))) / 2
                a_coef = 1.0
                b_coef = A[i2] - 1.0
                c_coef = base[i2] - base[mid]
                disc = b_coef*b_coef - 4*a_coef*c_coef
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    for x_cand in [(-b_coef + sqrt_disc) / 2.0, (-b_coef - sqrt_disc) / 2.0]:
                        if 0.0 <= x_cand <= 1.0:
                            aff_x = base + A * x_cand
                            aff_x[mid] += x_cand * (1.0 - x_cand)
                            vx = np.max(aff_x)
                            if vx < best_val - 1e-15:
                                best_val = vx
                                best_new = x_cand
            
            # Golden section refinement
            lo = max(0.0, best_new - 0.25)
            hi = min(1.0, best_new + 0.25)
            if hi - lo > 1e-10:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                
                aff_c1 = base + A * c1
                aff_c1[mid] += c1 * (1.0 - c1)
                f1 = np.max(aff_c1)
                aff_c2 = base + A * c2
                aff_c2[mid] += c2 * (1.0 - c2)
                f2 = np.max(aff_c2)
                
                for _ in range(35):
                    if f1 < f2:
                        hi = c2; c2 = c1; f2 = f1
                        c1 = hi - gr*(hi - lo)
                        aff_c1 = base + A * c1
                        aff_c1[mid] += c1 * (1.0 - c1)
                        f1 = np.max(aff_c1)
                    else:
                        lo = c1; c1 = c2; f1 = f2
                        c2 = lo + gr*(hi - lo)
                        aff_c2 = base + A * c2
                        aff_c2[mid] += c2 * (1.0 - c2)
                        f2 = np.max(aff_c2)
                
                x_gold = (lo + hi) / 2.0
                aff_g = base + A * x_gold
                aff_g[mid] += x_gold * (1.0 - x_gold)
                fg = np.max(aff_g)
                if fg < best_val - 1e-15:
                    best_val = fg
                    best_new = x_gold
            
            if best_new != old:
                delta = best_new - old
                corr += A * delta
                corr[mid] += best_new * (1.0 - best_new) - old * (1.0 - old)
                improved = True
        
        if not improved:
            break
    
    return best, np.max(corr) * dx

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    val = (m + np.log(wsum) / beta) * dx
    return val

def smooth_max_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    mid = n - 1
    offsets = np.arange(-(n - 1), n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - mid
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return grad

def slsqp_polish(h, n, beta=5000, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    sum_target = n / 2.0
    
    def obj(x):
        return smooth_max_obj(x, n, beta)
    
    def obj_grad(x):
        return smooth_max_grad(x, n, beta)
    
    def cons_fun(x):
        return np.sum(x) - sum_target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            return trial
    except Exception:
        pass
    return h.copy()

Let me also think about the problem from a different angle. The objective is:
C₅ = max_k corr[k] * dx

where corr[k] = sum_i h[i] * (1 - h[i+k]) (for valid indices).

If h is binary (0 or 1), then corr[k] = number of positions where h[i]=1 and h[i+k]=0.

For a binary sequence of length n with n/2 ones, the total number of (i, k) pairs where h[i]=1 and h[i+k]=0 is:
sum_k corr[k] = sum_i h[i] * (number of k where h[i+k]=0)
= sum_i h[i] * (n - 1 - sum_k h[i+k] for k≠0)
= (n/2) * (n-1) - sum_i h[i] * sum_{k≠0} h[i+k]
= (n/2)(n-1) - sum_{i,j: i≠j} h[i]h[j]
= (n/2)(n-1) - (n/2)(n/2 - 1)
= (n/2)(n-1) - (n/2)((n-2)/2)
= (n/2)[(n-1) - (n-2)/2]
= (n/2)[(2n-2-n+2)/2]
= (n/2)(n/2)
= n²/4

So sum_k corr[k] = n²/4 for any binary sequence with n/2 ones.

There are 2n-1 shifts (from -(n-1) to n-1). The average corr[k] = n²/(4(2n-1)) ≈ n/8 for large n.

But the max is what we care about. For a "random" binary sequence, corr[k] ≈ (n-|k|)/4 with some fluctuation of order sqrt((n-|k|)/16).

The Thue-Morse sequence has the property that its autocorrelation is bounded by O(sqrt(n)), which is much better than the random case.

For the Thue-Morse sequence of length n=2^m:
- corr[0] = 0 (since h is binary)
- For k≠0: |corr[k]| ≤ C * sqrt(n) for some constant C

So max corr[k] ≈ C*sqrt(n), and C₅ ≈ C*sqrt(n) * 2/n = 2C/sqrt(n).

For n=640: C₅ ≈ 2C/25.3 ≈ 0.079C

def compute_c5(h):

    dx = 2.0 / n

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):

        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 0 else 0.0
    return project_feasible(h, n)

def smooth_max_obj(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = np.minimum(beta * (corr - m), 0.0)
    ev = np.exp(shifted)
    ws = float(np.sum(ev))
    return (m + np.log(ws) / beta) * dx

def smooth_max_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    shifted = np.minimum(beta * (corr - m), 0.0)
    ev = np.exp(shifted)
    ws = float(np.sum(ev))
    weights = ev / ws
    grad = np.zeros(n)
    mid = n - 1
    offsets = np.arange(-(n - 1), n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - mid
        if offset >= 0:
            np_ = n - offset
            if np_ > 0:
                grad[offset:offset + np_] += w * (1.0 - x[:np_])
                grad[:np_] -= w * x[offset:offset + np_]
        else:
            aoff = -offset
            np_ = n - aoff
            if np_ > 0:
                grad[:np_] += w * (1.0 - x[aoff:aoff + np_])
                grad[aoff:aoff + np_] -= w * x[:np_]
    grad *= dx
    return grad

def slsqp_polish(h, n, beta=5000, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    sum_target = n / 2.0

    def obj(x):
        return smooth_max_obj(x, n, beta)

    def obj_grad(x):
        return smooth_max_grad(x, n, beta)

    def cons_fun(x):
        return np.sum(x) - sum_target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            if compute_c5(trial) <= compute_c5(h) + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def block_flip(h, n, rng, n_flips=6):

    idx = rng.choice(n, min(n_flips, n), replace=False)
