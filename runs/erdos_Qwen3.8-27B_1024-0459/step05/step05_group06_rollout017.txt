def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = np.where((h > 0.01) & (h < 0.99))[0]
        if len(free) > 0:
            h[free[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def smooth_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_s = np.max(shifted)
    exp_vals = np.exp(shifted - max_s)
    wsum = exp_vals.sum()
    weights = exp_vals / wsum
    val = (m + (max_s + np.log(wsum)) / beta) * dx
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            grad[:np_] += w * (1.0 - h[off:off + np_])
            grad[off:] += -w * h[:np_]
        else:
            aoff = -off
            np_ = n - aoff
            grad[aoff:] += w * (1.0 - h[:np_])
            grad[:np_] += -w * h[aoff:]
    grad *= dx
    return val, grad

def slsqp_step(h, n, beta, maxiter=300):
    def obj(x):
        return smooth_obj_grad(x, n, beta)[0]
    def grad(x):
        return smooth_obj_grad(x, n, beta)[1]
    bounds = [(0.0, 1.0)] * n
    target_sum = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def adam_optimize(h, n, beta, lr=0.005, n_steps=5000):
    m矩 = np.zeros(n)
    v矩 = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    b1, b2, eps = 0.85, 0.999, 1e-8
    cur_lr = lr
    stall = 0
    for t in range(1, n_steps + 1):
        if remaining_time() < 5:
            break
        if t % 500 == 0:
            cur_lr *= 0.7
        val, grad = smooth_obj_grad(h, n, beta)
        m矩 = b1 * m矩 + (1 - b1) * grad
        v矩 = b2 * v矩 + (1 - b2) * (grad * grad)
        m_hat = m矩 / (1.0 - b1 ** t)
        v_hat = v矩 / (1.0 - b2 ** t)
        h_new = h - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if stall > 2000:
            break
    return best_h, best_c5

def coord_descent(h, n, n_sweeps=40):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    for sweep in range(n_sweeps):
        if remaining_time() < 8:
            break
        improved = False
        for i in range(n):
            if remaining_time() < 2:
                break
            t0 = best[i]
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n - 1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] = 1.0
                    b[k] = -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += 1.0 - best[i + off]
                    if 0 <= i - off < n:
                        a[k] += -best[i - off]
            base = best_corr - a * t0 - b * t0 * t0
            grid = np.linspace(0.0, 1.0, 25)
            cands = np.unique(np.concatenate([[0.0, 1.0, t0, 1.0 - t0], grid]))
            vals = np.array([np.max(base + a * c + b * c * c) * dx for c in cands])
            bi = np.argmin(vals)
            best_t = cands[bi]
            best_f = vals[bi]
            if best_f < best_c5 - 1e-13 and abs(best_t - t0) > 1e-10:
                lo = max(0.0, best_t - 0.08)
                hi = min(1.0, best_t + 0.08)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                for _ in range(45):
                    if hi - lo < 1e-13:
                        break
                    if f1 < f2:
                        hi, c2, f2 = c2, c1, f1
                        c1 = hi - gr * (hi - lo)
                        f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    else:
                        lo, c1, f1 = c1, c2, f2
                        c2 = lo + gr * (hi - lo)
                        f2 = np.max(base + a * c2 + h * c2 * c2) if False else np.max(base + a * c2 + b * c2 * c2) * dx
                t_opt = (lo + hi) / 2.0
                f_opt = np.max(base + a * t_opt + b * t_opt * t_opt) * dx
                if f_opt < best_f:
                    best_t, best_f = t_opt, f_opt
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved = True
        if not improved:
            break
    return best, best_c5

def simulate_anneal(h, n, n_iter=5000, T0=0.01, Tmin=1e-6):
    best = h.copy()
    best_c5 = compute_c5(best)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    rng = np.random.RandomState(12345)
    for it in range(n_iter):
        if remaining_time() < 5:
            break
        T = T0 * (Tmin / T0) ** (it / max(n_iter, 1))
        n_pert = rng.randint(2, max(3, n // 40) + 1)
        idx = rng.choice(n, n_pert, replace=False)
        cur_p = cur.copy()
        cur_p[idx] += rng.uniform(-0.05, 0.05, n_pert)
        cur_p = project_feasible(cur_p, n)
        c5_p = compute_c5(cur_p)
        delta = c5_p - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
            cur = cur_p
            cur_c5 = c5_p
            if c5_p < best_c5:
                best_c5 = c5_p
                best = cur_p.copy()
    return best, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = rng.choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Adam-based smooth max optimization
    for beta in [500, 2000, 10000]:
        if remaining_time() < 40:
            break
        h_new, c5_new = adam_optimize(h, n, beta, lr=0.003, n_steps=3000)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 2: SLSQP refinement
    for beta in [1000, 10000, 80000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_step(h, n, beta, 350)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-15:
            h = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 3: Intensive coordinate descent
    if remaining_time() > 60:
        h, c5_new = coord_descent(h, n, n_sweeps=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 4: More SLSQP at high precision
    for beta in [100000, 500000]:
        if remaining_time() < 20:
            break
        h_new = slsqp_step(best_h, best_n, beta, 250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Coordinate descent again
    if remaining_time() > 40:
        h, c5_new = coord_descent(best_h, best_n, n_sweeps=40)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    # Phase 6: Try different resolutions
    res_list = []
    for mult in [0.5, 1.5, 2.0]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if 64 <= rn <= 3000:
            res_list.append(rn)
    res_list = sorted(set(res_list))
    for rn in res_list:
        if remaining_time() < 70:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)
        if c5_r <= best_c5 + 0.005:
            for beta in [2000, 20000, 150000]:
                if remaining_time() < 20:
                    break
                h_r = slsqp_step(h_r, rn, beta, 200)
            if remaining_time() > 25:
                h_r, c5_r = coord_descent(h_r, rn, n_sweeps=25)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                best_n = rn

    # Phase 7: Simulated annealing on best
    if remaining_time() > 120:
        h_sa, c5_sa = simulate_anneal(best_h, best_n, n_iter=3000, T0=0.008, Tmin=1e-7)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        else:
            # Still polish the SA result and compare
            if c5_sa < best_c5 + 0.003:
                h_sa = slsqp_step(h_sa, best_n, 50000, 200)
                h_sa, c5_sa = coord_descent(h_sa, best_n, n_sweeps=20)
                c5_sa = compute_c5(h_sa)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()

    # Phase 8: Random perturbations + local opt
    n_use = best_n
    n_pert = max(4, n_use // 60)
    for pert in range(40):
        if remaining_time() < 30:
            break
        h_p = best_h.copy()
        idx = rng.choice(n_use, n_pert, replace=False)
        h_p[idx] += rng.uniform(-0.12, 0.12, n_pert)
        h_p = project_feasible(h_p, n_use)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5 + 0.003:
            h_p, _ = adam_optimize(h_p, n_use, 5000, lr=0.002, n_steps=1500)
            h_p = slsqp_step(h_p, n_use, 30000, 150)
            h_p, _ = coord_descent(h_p, n_use, n_sweeps=15)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 9: Final polish cycle
    if remaining_time() > 30:
        best_h, c5_new = coord_descent(best_h, best_n, n_sweeps=40)
        if c5_new < best_c5:
            best_c5 = c5_new
    if remaining_time() > 15:
        best_h = slsqp_step(best_h, best_n, 1000000, 150)
        c5_new = compute_c5(best_h)
        if c5_new < best_c5:
            best_c5 = c5_new
    if remaining_time() > 10:
        best_h, c5_new = coord_descent(best_h, best_n, n_sweeps=25)
        if c5_new < best_c5:
            best_c5 = c5_new

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure exact sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            best_h[free] += diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
