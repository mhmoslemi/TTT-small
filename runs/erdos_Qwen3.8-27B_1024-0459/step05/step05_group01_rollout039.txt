def top_k_descent(h, n, top_k=5, max_iters=400):
    """Gradient descent focused on top-k active lags."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break

        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (top_k - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]

        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn

        improved = False
        for step in [0.005, 0.01, 0.02, 0.003, 0.008, 0.05, 0.015, 0.04]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break

    return h, best_c5

    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)

            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Top-k descent
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_descent(best_h, best_n, top_k=5, max_iters=500)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 3: Higher resolution attempt
    n_hi_candidates = []
    for mult in [1.5, 2.0, 2.5, 3.0]:
        nh = int(best_n * mult)
        if nh % 2 != 0:
            nh += 1
        if 200 <= nh <= 1200:
            n_hi_candidates.append(nh)
    n_hi_candidates = sorted(set(n_hi_candidates))

    for n_hi in n_hi_candidates[:2]:
        if remaining_time() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_hi)
        h_feas = project_feasible(h_interp, n_hi)
        h_bin_hi = (h_feas > 0.5).astype(float)

        sa_budget2 = min(200, max(15, int(remaining_time() * 0.25)))
        h_sa_hi, c5_sa_hi = binary_sa_multi_restart(h_bin_hi, n_hi, sa_budget2, rng,
                                                     n_restarts=2)
        h_cont_hi = h_sa_hi
        for beta in [1000, 10000, 50000]:
            if remaining_time() < 10:
                break
            h_cont_hi = slsqp_refine(h_cont_hi, n_hi, beta, maxiter=400)

        c5_hi = compute_c5(h_cont_hi)
        if c5_hi < best_c5 - 1e-14:
            best_c5 = c5_hi
            best_h = h_cont_hi
            best_n = n_hi

    # Phase 4: Alternative starting points at best resolution
    alt_starts = [theta_morse(best_n)]
    # Random balanced
    h_rand = np.zeros(best_n)
    h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
    alt_starts.append(h_rand)
    # Shifted parent
    h_shift = np.roll(best_h, best_n // 3)
    alt_starts.append(project_feasible(h_shift, best_n))

    for h_alt in alt_starts:
        if remaining_time() < 40:
            break
        h_bin_alt = (h_alt > 0.5).astype(float)
        sa_budget3 = min(120, max(10, int(remaining_time() * 0.15)))
        h_sa_alt, c5_sa_alt = binary_sa(h_bin_alt, best_n, sa_budget3, rng,
                                         T_start=2.0, T_end=0.003, cool_per_iter=0.9997)
        h_cont_alt = h_sa_alt
        for beta in [5000, 50000]:
            if remaining_time() < 8:
                break
            h_cont_alt = slsqp_refine(h_cont_alt, best_n, beta, maxiter=400)
        c5_alt = compute_c5(h_cont_alt)
        if c5_alt < best_c5 - 1e-14:
            best_c5 = c5_alt
            best_h = h_cont_alt.copy()

    # Phase 5: Final intensive SA at best resolution
    if remaining_time() > 30:
        h_bin_final = (best_h > 0.5).astype(float)
        sa_budget4 = min(200, max(15, int(remaining_time() * 0.4)))
        h_final, c5_final = binary_sa(h_bin_final, best_n, sa_budget4, rng,
                                       T_start=2.5, T_end=0.001, cool_per_iter=0.9996)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

        # Continuous polish on the new best
        for beta in [10000, 50000, 200000]:
            if remaining_time() < 8:
                break
            h_polish = slsqp_refine(best_h, best_n, beta, maxiter=500)
            c5_polish = compute_c5(h_polish)
            if c5_polish < best_c5 - 1e-14:
                best_c5 = c5_polish
                best_h = h_polish.copy()

    # Phase 6: Final top-k
    if remaining_time() > 8:
        h_eq, c5_eq = top_k_descent(best_h, best_n, top_k=6, max_iters=400)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Ensure feasibility and final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
