def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Find index with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_f_k(h, k, n):
    """Gradient of f_k w.r.t. h. f_k = sum_i h[i]*(1-h[i-(k-n+1)])"""
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:off+np_] += (1.0 - h[:np_])
        g[:np_] -= h[off:off+np_]
    else:
        ao = -off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao+np_])
        g[ao:ao+np_] -= h[:np_]
    return g

def subgradient_step(h, n, top_k=10, max_iters=500, time_limit=60):
    """Subgradient method with exact line search on piecewise linear max."""
    start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c_max = np.max(corr)
        cur_c5 = c_max * (2.0/n)
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
        
        # Weights: softmax over top K shifts
        idx = np.argsort(-corr)[:top_k]
        sh = (corr[idx] - c_max) / 0.005  # temperature
        w = np.exp(sh)
        w /= np.sum(w)
        
        # Compute subgradient
        g = np.zeros(n)
        for i, k in enumerate(idx):
            g += w[i] * grad_f_k(h, k, n)
        
        # Also add negative gradient from bottom to pull them up
        bot_idx = np.argsort(corr)[:min(3, top_k)]
        for k in bot_idx:
            g -= 0.1 * grad_f_k(h, k, n)
        
        # Line search: minimize max_k(f[k] + alpha * g[k]) * (2/n)
        # where f = corr
        # equivalent to: minimize max_k(corr[k] + alpha * G[k])
        # where G[k] = sum_i w_i * ∂f_k/∂... 
        # Actually G is the gradient of the weighted sum, not the max
        
        # For exact line search on max:
        # direction d_k = sum_i w_i * (∂f_k/∂h * 1) ... this is getting complicated
        # Simpler: just do a gradient step with projected line search
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        
        best_alpha = 0.0
        best_new_max = c_max

                best_alpha = alpha

        h = project_feasible(h + best_alpha * g, n)
    
    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return h, best_c5

                    # Actually at k=n-1: f = sum h[m](1-h[m])
                    # h[i]: was 1*(1-1)=0, now 0*(1-0)=0: no change
                    # h[j]: was 0*(1-0)=0, now 1*(1-1)=0: no change
                    # So correction is 0 at k=n-1 for binary
                    dk = 0.0
                new_corr[k] += dk

            new_max = float(np.max(new_corr))
            delta = new_max - cur_max
            if delta < 0:
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_move = (i, j, new_corr)
            else:
                # SA: accept with probability exp(-delta/temp)
                if temp > 0.001 and rng.rand() < np.exp(-delta / max(temp, 1e-10)):
                    if new_max < best_new_max + 0.5:
                        best_new_max = new_max
                        best_move = (i, j, new_corr)

        if best_move is not None:
            i, j, new_corr = best_move
            h[i] = 0
            h[j] = 1
            corr = new_corr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                for _ in range(5):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        h[i] = 0
                        h[j] = 1
                        ones = np.where(h == 1)[0]
                        zeros = np.where(h == 0)[0]
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
                temp *= 1.5

    return best_h, best_max * dx

def gradient_descent_exact(h, n, time_limit=60, top_k=6):
    start_t = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(300):
        if time.time() - start_t > time_limit or remaining() < 5:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        cur_c5 = c_max * (2.0 / n)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        sh = (corr[idx] - c_max) / 0.003
        w = np.exp(sh)
        w /= np.sum(w)

        g = np.zeros(n)
        for i, k in enumerate(idx):
            g += w[i] * grad_f_k(h, k, n)

        bot_idx = np.argsort(corr)[:3]
        for k in bot_idx:
            g -= 0.15 * grad_f_k(h, k, n)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        improved = False
        for step in [0.003, 0.007, 0.015, 0.03, 0.06, 0.002, 0.01, 0.02, 0.05, 0.1]:
            ht = project_feasible(h + step * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            for step in [0.001, 0.002, 0.005, 0.01, 0.0005]:
                ht = project_feasible(h - step * g, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        for i in range(parent_n):
            parent_h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n

    # Phase 1: Binary local search with SA
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    if int(np.sum(h_bin)) != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    t1 = min(250, int(remaining() * 0.2))
    if t1 > 30:
        h_b1, c5_b1 = binary_sa_search(h_bin, n, time_limit=t1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n

    # Phase 2: Multi-start binary search
    starts = []
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i ^ seed).count('1') % 2 == 0 else 0.0
    if int(np.sum(tm)) != n_half:
        if int(np.sum(tm)) > n_half:
            tm[rng.choice(np.where(tm==1)[0], int(np.sum(tm))-n_half, replace=False)] = 0
        else:
            tm[rng.choice(np.where(tm==0)[0], n_half-int(np.sum(tm)), replace=False)] = 1
    starts.append(tm)
    rp = (rng.rand(n) > 0.5).astype(float)
    if int(np.sum(rp)) != n_half:
        if int(np.sum(rp)) > n_half:
            rp[rng.choice(np.where(rp==1)[0], int(np.sum(rp))-n_half, replace=False)] = 0
        else:
            rp[rng.choice(np.where(rp==0)[0], n_half-int(np.sum(rp)), replace=False)] = 1
    starts.append(rp)

    for sp in starts:
        if remaining() < 30:
            break
        t_sp = min(100, int(remaining() * 0.1))
        if t_sp < 20:
            break
        h_bs, c5_bs = binary_sa_search(sp, n, time_limit=t_sp, rng=rng)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()
            best_n = n

    # Phase 3: Continuous optimization from best
    n_cur = best_n
    h_cur = best_h.copy()

    for beta in [200, 1000, 5000, 20000, 80000]:
        if remaining() < 20:
            break
        h_new = slsqp_optimize(h_cur, n_cur, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new.copy()
            best_n = n_cur

    # Phase 4: Top-shifts equalization
    if remaining() > 30:
        h_eq, c5_eq = top_shifts_equalize(h_cur, n_cur, top_k=8, max_iters=500, time_limit=min(80, int(remaining()*0.06)))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h_cur = h_eq.copy()
            best_n = n_cur

    # Phase 5: Gradient descent with exact steps
    if remaining() > 30:
        h_gd, c5_gd = gradient_descent_exact(h_cur, n_cur, time_limit=min(80, int(remaining()*0.06)), top_k=8)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()
            h_cur = h_gd.copy()
            best_n = n_cur

    # Phase 6: Try nearby resolutions
    n_base = best_n
    res_try = sorted(set([n_base - 16, n_base - 8, n_base - 4, n_base + 4, n_base + 8, n_base + 16,
                          n_base + 32, n_base - 32]))
    res_try = [r for r in res_try if 192 <= r <= 768 and r % 2 == 0 and r != n_base]
    res_try.sort(key=lambda x: abs(x - n_base))

    for n_t in res_try[:4]:
        if remaining() < 50:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_interp = project_feasible(h_interp, n_t)

        # Quick SLSQP
        for beta in [2000, 20000]:
            if remaining() < 15:
                break
            h_t = slsqp_optimize(h_interp, n_t, beta, maxiter=300)
            h_interp = h_t

        # Equalize
        if remaining() > 20:
            h_t, c5_t = top_shifts_equalize(h_interp, n_t, top_k=6, max_iters=200, time_limit=min(30, int(remaining()*0.04)))
            h_interp = h_t

        c5_t = compute_c5(h_interp)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_interp.copy()
            best_n = n_t

    # Phase 7: Binary refinement on best continuous solution
    n_cur = best_n
    h_bin2 = (best_h > 0.5).astype(float)
    n_half2 = n_cur // 2
    if int(np.sum(h_bin2)) != n_half2:
        sort_idx = np.argsort(-best_h)
        h_bin2 = np.zeros(n_cur)
        h_bin2[sort_idx[:n_half2]] = 1.0

    if remaining() > 40:
        h_b3, c5_b3 = binary_sa_search(h_bin2, n_cur, time_limit=min(100, int(remaining()*0.08)), rng=rng)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()
            best_n = n_cur

    # Phase 8: Final continuous polish
    n_cur = best_n
    h_cur = best_h.copy()
    for beta in [10000, 50000, 200000]:
        if remaining() < 15:
            break
        h_new = slsqp_optimize(h_cur, n_cur, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new.copy()
            best_n = n_cur

    if remaining() > 20:
        h_eq2, c5_eq2 = top_shifts_equalize(h_cur, n_cur, top_k=10, max_iters=400, time_limit=min(60, int(remaining()*0.06)))
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            best_n = n_cur

    # Phase 9: Try one more even/odd resolution
    if remaining() > 40:
        for n_t in [best_n + 2, best_n - 2, best_n + 6, best_n - 6]:
            if n_t < 192 or n_t > 1024 or n_t % 2 != 0:
                continue
            if remaining() < 20:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            if remaining() > 15:
                h_t = slsqp_optimize(h_t, n_t, 20000, maxiter=300)
            if remaining() > 10:
                h_t, c5_t = top_shifts_equalize(h_t, n_t, top_k=8, max_iters=200, time_limit=min(25, int(remaining()*0.03)))
            else:
                c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Final: ensure feasibility and compute exact value
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
