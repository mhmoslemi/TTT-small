def solve_1d_minimax(overlaps, gj, gk, t_low, t_high):
    """
    Find t in [t_low, t_high] that minimizes max_s (overlaps[s] + t * (gj[s] - gk[s]))
    """
    n_s = len(overlaps)
    diff = gj - gk  # shape (n_s,)
    
    # Check if the function is constant
    if np.all(np.abs(diff) < 1e-15):
        return 0.0
    
    # Evaluate at boundaries and 0
    candidates = [t_low, t_high, 0.0]
    candidates = [max(t_low, min(t_high, c)) for c in candidates]
    
    # Find the minimum by "walking" from 0
    t = 0.0
    for _ in range(n_s + 1):
        val = overlaps + t * diff
        s_star = np.argmax(val)
        slope = diff[s_star]
        if abs(slope) < 1e-15:
            break
        if slope > 0:
            # Moving in positive t increases the max, so optimum is to the left
            # Find the t where another line becomes >= the current max
            current_val = val[s_star]
            # Find breakpoints: t such that overlaps[s] + t*diff[s] = current_val
            # t = (current_val - overlaps[s]) / diff[s] for diff[s] != 0
            valid = (np.abs(diff) > 1e-15) & (np.arange(n_s) != s_star)
            if not np.any(valid):
                # Only one non-constant line, go to boundary
                t = t_low
                break
            t_bps = (current_val - overlaps[valid]) / diff[valid]
            # We want the largest t_bp < t (since we're going left)
            t_bps_left = t_bps[t_bps < t - 1e-15]
            if len(t_bps_left) > 0:
                t = max(t_low, t_bps_left.max())
            else:
                t = t_low
        else:
            # slope < 0, optimum is to the right
            current_val = val[s_star]
            valid = (np.abs(diff) > 1e-15) & (np.arange(n_s) != s_star)
            if not np.any(valid):
                t = t_high
                break
            t_bps = (current_val - overlaps[valid]) / diff[valid]
            t_bps_right = t_bps[t_bps > t + 1e-15]
            if len(t_bps_right) > 0:
                t = min(t_high, t_bps_right.min())
            else:
                t = t_high
    
    # Evaluate at t, t_low, t_high and pick the best
    for tt in [t, t_low, t_high, 0.0]:
        tt = max(t_low, min(t_high, tt))
