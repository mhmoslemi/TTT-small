import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 100.0


def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_gradient_single(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g


def _sm_obj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -100.0, 100.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx


def _sm_grad(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -100.0, 100.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - x[a:])
            g[a:] -= wk * x[:np_]
    return g


def _eq_fun(x):
    return np.sum(x) - _CUR_N / 2.0


def _eq_jac(x):
    return np.ones(_CUR_N)


def slsqp_smooth(h, n, beta, maxiter=500):
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_sm_obj, h0, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < _sm_obj(h0) - 1e-12:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def active_set_optimize(h, n, maxiter=1000, threshold_frac=0.97):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    for it in range(maxiter):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        threshold = c_max * threshold_frac
        active_idx = np.where(corr >= threshold)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        g = np.zeros(n)
        for k in active_idx:
            g += -corr_gradient_single(h, k, n)
        g /= len(active_idx)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        improved = False
        steps = [0.02, 0.01, 0.005, 0.04, 0.002, 0.08, 0.001, 0.15, 0.3]
        for step in steps:
            h_new = project_feasible(h + step * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def pgd_topk(h, n, top_k, lr_init=0.005, maxiter=2000):
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(maxiter):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx = np.argpartition(corr, -top_k)[-top_k:]
        g = np.zeros(n)
        w = 1.0 / top_k
        for k in idx:
            g += w * corr_gradient_single(h, k, n)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_new = project_feasible(h - lr * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 300 == 299:
            lr *= 0.85
    return best_h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def binary_swap_search(h, n, time_limit=40, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    while time.time() - start < time_limit and remaining_time() > 5:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = 300
        best_delta = 1e-10
        best_pair = None
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            delta = c_new - best_c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (int(i), int(j))
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = compute_c5(h)
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                break
    return best_h, best_c5


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)
