    beta = _SM_BETA
    z = np.clip(beta * (corr - m), -60.0, 0.0)
    ez = np.exp(z)
    return (m + np.log(np.sum(ez)) / beta) * dx

        idx1 = np.arange(k_lag, n)
        g[idx1] += (1.0 - h[idx1 - k_lag])
        idx2 = np.arange(0, n - k_lag)
        g[idx2] -= h[idx2 + k_lag]
    else:
        m = -k_lag
        idx1 = np.arange(0, n - m)
        g[idx1] += (1.0 - h[idx1 + m])
        idx2 = np.arange(m, n)
        g[idx2] -= h[idx2 - m]
    return g


def subgradient_polish(h, n, n_steps=100, time_limit=60):
    """Subgradient method on the non-smooth objective with projection."""
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 1.0 / np.sqrt(n)
    for step in range(n_steps):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        k_lag = max_idx - (n - 1)
        g = corr_grad_single(h, k_lag, n)
        g *= dx
        # Check convergence
        if np.linalg.norm(g) < 1e-12:
            break
        g /= np.linalg.norm(g)
        # Project gradient to feasible direction (orthogonal to constraint)
        g_proj = g - np.mean(g)
        # Line search
        for t in [alpha, alpha/2, alpha/4, alpha/8, alpha/16]:
            h_try = project_feasible(h - t * g_proj, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                break
        alpha *= 0.995
    return best_h, best_c5


def top_k_refine(h, n, top_k=4, max_iters=800):
    """Refine by equalizing top-k correlations and minimizing the max."""
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        if it % 500 == 0:
            pass  # debug
        # Direction to equalize
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            k_lag = k - (n - 1)
            gk = corr_grad_single(h, k_lag, n)
            if pos == 0:
                g_dir += gk
            else:
                g_dir -= gk
        # Also add a component to reduce the max
        g_max = np.zeros(n)
        for k in idx_top:
            k_lag = k - (n - 1)
            g_max += corr_grad_single(h, k_lag, n)
        g_max /= top_k
        # Combined direction: equalize + reduce
        d = g_dir * 0.5 + g_max
        d -= np.mean(d)  # project to feasible subspace
        gn = np.linalg.norm(d)
        if gn < 1e-14:
            # Try pure equalization
            d = g_dir
            d -= np.mean(d)
            gn = np.linalg.norm(d)
            if gn < 1e-14:
                break
            d /= gn
        else:
            d /= gn
        improved = False
        for step in [0.001, 0.005, 0.01, 0.02, 0.003, 0.008, 0.002, 0.05, 0.0005]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            # Try opposite direction
            for step in [0.005, 0.01, 0.002, 0.008]:
                ht = project_feasible(h - step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)


def random_balanced(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Load parent
    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive SLSQP
    h = best_h.copy()
    for beta in [200, 500, 1500, 5000, 15000, 50000, 150000, 500000, 1500000]:
        if remaining_time() < 15:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-k equalization (more aggressive)
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_refine(h, best_n, top_k=4, max_iters=1000)
        if c5_eq < best_c5 - 1e-15:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 3: SLSQP with very high beta
    for beta in [500000, 1500000, 5000000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 4: Subgradient polish
    if remaining_time() > 20:
        h_sg, c5_sg = subgradient_polish(h, best_n, n_steps=200,
                                         time_limit=min(80, int(remaining_time() * 0.15)))
        if c5_sg < best_c5 - 1e-15:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # Phase 5: Top-k refine again
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_refine(h, best_n, top_k=5, max_iters=800)
