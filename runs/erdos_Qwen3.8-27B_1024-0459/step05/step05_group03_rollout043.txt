        # For lag k > 0 (full index = k + n - 1):
        # Delta = h[p+k]*[p+k < n] - (1-h[p-k])*[p >= k]
        for k in range(1, n):
            # k > 0
            delta_k = 0.0
            if p + k < n:
                delta_k += h[p + k]
            if p >= k:
                delta_k -= (1.0 - h[p - k])
            effects[p, k + n - 1] = delta_k
            # k < 0: full index = -k + n - 1
            delta_negk = 0.0
            if p >= k:
                delta_negk += h[p - k]
            if p + k < n:
                delta_negk -= (1.0 - h[p + k])
            effects[p, -k + n - 1] = delta_negk
    return effects


def compute_effects_fast(h, n):
    """Faster vectorized computation of flip effects."""
    n_corr = 2 * n - 1
    effects = np.zeros((n, n_corr), dtype=np.float64)
    lags = np.arange(n)  # 0, 1, ..., n-1
    for p in range(n):
        if h[p] < 0.5:
            continue
        # For k = 1 to n-1
        ks = np.arange(1, n)
        # Positive lags: full index = k + n - 1
        idx_pos = ks + n - 1
        # h[p+k] where p+k < n
        pk = p + ks
        valid_pk = pk < n
        term1 = np.zeros(n - 1)
        term1[valid_pk] = h[pk[valid_pk]]
        # (1-h[p-k]) where p >= k, i.e., k <= p
        valid_pk2 = ks <= p
        term2 = np.zeros(n - 1)
        pk2 = p - ks
        term2[valid_pk2] = 1.0 - h[pk2[valid_pk2]]
        effects[p, idx_pos] = term1 - term2
        # Negative lags: full index = -k + n - 1
        idx_neg = -ks + n - 1
        # h[p-k] where p >= k
        term3 = np.zeros(n - 1)
        term3[valid_pk2] = h[pk2[valid_pk2]]
        # (1-h[p+k]) where p+k < n
        term4 = np.zeros(n - 1)
        term4[valid_pk] = 1.0 - h[pk[valid_pk]]
        effects[p, idx_neg] = term3 - term4
    return effects


def binary_local_search(h_bin, n, max_rounds=500, time_limit=None):
    """Intensive binary local search with vectorized candidate evaluation."""
    h = (h_bin > 0.5).astype(float).copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_c5 = current_max * dx
    
    start_time = time.time()
    
    for rnd in range(max_rounds):
        if remaining() < 5:
            break
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
            
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Compute effects for all one-positions
        effects = compute_effects_fast(h, n)
        
        # Vectorized: for all (pa, pb) pairs, compute approximate new max
        # new_corr_approx = corr + effects[pa] - effects[pb]
        # (since 0->1 flip effect is negative of 1->0 flip effect)
        
        # To save memory, process in batches if needed
        batch_size = max(1, min(n_ones, 64))
        best_improve = 1e-10
        best_pa, best_pb = -1, -1
        best_new_corr = None
        
        for batch_start in range(0, n_ones, batch_size):
            batch_end = min(batch_start + batch_size, n_ones)
            pa_batch = ones[batch_start:batch_end]
            
            # candidates shape: (batch, n_zeros, 2*n-1)
            base = corr[None, None, :] + effects[pa_batch][:, None, :]
            candidates = base - effects[zeros][None, :, :]
            maxes = np.max(candidates, axis=2)  # (batch, n_zeros)
            
            flat_min = np.min(maxes)
            if flat_min < current_max - best_improve:
                best_improve = current_max - flat_min
                idx = np.argmin(maxes)
                bi, bj = divmod(idx, n_zeros)
                best_pa = pa_batch[bi]
                best_pb = zeros[bj]
        
        if best_pa < 0:
            break  # No improvement found
        
        # Apply swap and verify
        h[best_pa] = 0.0
        h[best_pb] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        
        if new_max < current_max - 1e-12:
            current_max = new_max
            corr = new_corr
            if current_max * dx < best_c5:
                best_c5 = current_max * dx
                best_h = h.copy()
        else:
            # Revert
            h[best_pa] = 1.0
            h[best_pb] = 0.0
            # Try a few more candidates
            # Re-sort and try next best
            effects2 = compute_effects_fast(h, n)
            tried = set()
            tried.add((best_pa, best_pb))
            found = False
            for batch_start in range(0, n_ones, batch_size):
                if found:
                    break
                if remaining() < 5:
                    break
                batch_end = min(batch_start + batch_size, n_ones)
                pa_batch = ones[batch_start:batch_end]
                base = corr[None, None, :] + effects2[pa_batch][:, None, :]
                candidates = base - effects2[zeros][None, :, :]
                maxes = np.max(candidates, axis=2)
                
                # Get top few candidates
                flat_maxes = maxes.ravel()
                flat_order = np.argsort(flat_maxes)
                for fi in flat_order[:20]:
                    bi, bj = divmod(fi, n_zeros)
                    pa_c = pa_batch[bi]
                    pb_c = zeros[bj]
                    if (pa_c, pb_c) in tried:
                        continue
                    tried.add((pa_c, pb_c))
                    h[pa_c] = 0.0
                    h[pb_c] = 1.0
                    ec = compute_corr(h)
                    em = float(np.max(ec))
                    h[pa_c] = 1.0
                    h[pb_c] = 0.0
                    if em < current_max - 1e-12:
                        h[pa_c] = 0.0
                        h[pb_c] = 1.0
                        current_max = em
                        corr = ec
                        if current_max * dx < best_c5:
                            best_c5 = current_max * dx
                            best_h = h.copy()
                        found = True
                        break
            if not found:
                break  # No improvement at all, stop
    
    return best_h, best_c5


def nesterov_binary(h, n, max_iters=3000):
    """Nesterov-style search specifically for binary solutions using swaps."""
    h = (h > 0.5).astype(float).copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    
    # Keep score history for tabu
    tabu = {}
    tabu_tenure = 50
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute effects
        effects = compute_effects_fast(h, n)
        
        # Find best swap
        # Process in batches
        batch_size = 64
        best_max = np.inf
        best_pa, best_pb = -1, -1
        
        for bs in range(0, len(ones), batch_size):
            be = min(bs + batch_size, len(ones))
            pa_b = ones[bs:be]
            base = corr[None, None, :] + effects[pa_b][:, None, :]
            cands = base - effects[zeros][None, :, :]
            maxes = np.max(cands, axis=2)
            
            # Apply tabu: mark tabu moves as very bad
            for (tpa, tpb), expiry in list(tabu.items()):
                if it > expiry:
                    del tabu[(tpa, tpb)]
                    continue
                # Find position in maxes
                for i, pa_val in enumerate(pa_b):
                    if pa_val == tpa:
                        for j, pb_val in enumerate(zeros):
                            if pb_val == tpb:
                                maxes[i, j] = np.inf
            
            m = np.min(maxes)
            if m < best_max:
                best_max = m
                idx = np.argmin(maxes)
                bi, bj = divmod(idx, len(zeros))
                best_pa = pa_b[bi]
                best_pb = zeros[bj]
        
        if best_pa < 0:
            break
        
        # Accept even if not improving (simulated annealing style)
        if best_max < current_max := float(np.max(corr)):
            h[best_pa] = 0.0
            h[best_pb] = 1.0
            corr = compute_corr(h)
            tabu[(best_pb, best_pa)] = it + tabu_tenure
            if np.max(corr) * dx < best_c5:
                best_c5 = np.max(corr) * dx
                best_h = h.copy()
        else:
            # Occasionally accept non-improving moves (annealing)
            if it % 100 == 50 and np.random.random() < 0.1:
                h[best_pa] = 0.0
                h[best_pb] = 1.0
                corr = compute_corr(h)
                tabu[(best_pb, best_pa)] = it + tabu_tenure
            else:
                break
    
    return best_h, best_c5


def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    w = w * dx
    return corr_grad_weighted(h, w)


def slsqp_refine(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def thue_morse(n):
    """Generate first n bits of Thue-Morse sequence."""
    tm = np.zeros(n, dtype=float)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ ((i & 1))
    # Make it have exactly n/2 ones
    if n % 2 == 0:
        target = n // 2
        count = int(np.sum(tm))
        if count != target:
            # Adjust
            idx = np.where(tm == 1)[0]
            idx0 = np.where(tm == 0)[0]
            if count > target:
                for i in range(count - target):
                    tm[idx[i]] = 0
            else:
                for i in range(target - count):
                    tm[idx0[i]] = 1
    return tm


def greedy_balanced(n):
    """Greedy construction: place ones to minimize max correlation so far."""
    h = np.zeros(n)
    corr = np.zeros(2 * n - 1)
    ones_needed = n // 2
    
    for _ in range(ones_needed):
        best_pos = 0
        best_max = np.inf
        # Try a subset of positions for speed
        candidates = np.arange(n)
        # Sample if too many
        if len(candidates) > 200:
            rng = np.random.RandomState(42)
            candidates = rng.choice(n, 200, replace=False)
        for p in candidates:
            h[p] = 1.0
            c = compute_corr(h)
            m = np.max(c)
            if m < best_max:
                best_max = m
                best_pos = p
            h[p] = 0.0
        h[best_pos] = 1.0
    return h
