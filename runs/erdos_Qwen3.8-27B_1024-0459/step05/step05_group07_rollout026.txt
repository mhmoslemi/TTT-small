        im = i - d_range

        jm = j - d_range

        valid_ip = (ip >= 0) & (ip < n)
        valid_im = (im >= 0) & (im < n)
        valid_jp = (jp >= 0) & (jp < n)
        valid_jm = (jm >= 0) & (jm < n)
        
        delta[valid_ip] += h[ip[valid_ip]]
        delta[valid_im] += h[im[valid_im]]
        delta[valid_jp] -= h[jp[valid_jp]]
        delta[valid_jm] -= h[jm[valid_jm]]
        
        new_corr = best_corr + delta
        new_max = np.max(new_corr)
        
        if use_sa:

                zeros = np.where(h == 0)[0]

Actually, I just realized something important. The `np.correlate(h, 1-h, mode='full')` for a binary h where h has value 1 at some positions and 0 elsewhere:

corr[k] = ∑_i h[i] * (1-h[i+k-n+1])  ... no wait.

Let me be precise. `np.correlate(a, v, mode='full')` computes:
c[k] = ∑_i a[i+k] * conj(v[i]) for k = -(len(v)-1), ..., len(a)-1

Actually, from numpy docs: `c_k = \sum_n \bar{a}_{n+k} v_n`

So `np.correlate(h, 1-h, mode='full')`:
c[k] = ∑_n h[n+k] * (1-h[n]) for valid n

where k ranges from -(n-1) to n-1.

For k ≥ 0: c[k] = ∑_{n=0}^{n-1-k} h[n+k] * (1-h[n])
For k < 0: c[k] = ∑_{n=-k}^{n-1} h[n+k] * (1-h[n])

In both cases, c[k] counts (weighted) the number of positions where h is 1 at the "shifted" position and 0 at the "original" position.

For binary h: c[k] = number of n where h[n+k]=1 and h[n]=0 (for the valid range).

This is the number of 0→1 transitions when shifting by k. Or equivalently, the cross-correlation of h with (1-h).

OK so my analysis was correct. For binary h with n/2 ones, the expected value of c[k] for large |k| is approximately (n-|k|)/4.

Now, the maximum possible value of c[k] is min(n-|k|, n/2) (can't have more 1s than exist, and can't have more positions than the valid range).

We want max_k c[k] * dx ≤ 0.38, so max_k c[k] ≤ 0.38 * n/2 = 0.19 * n = 121.6 for n=640.

So we need c[k] ≤ 121 for all k. The "random" expected value is about 160 for small k, so we need a significant reduction.

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)

            h[mask] += diff / cnt
        else:

def fast_coord_descent_cont(h, n, max_sweeps=100):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))

            t_best = old
            f_best = f_old

            for t_cand in [0.0, 0.5, 1.0]:
                mid = base + A * t_cand
                mid[n - 1] = base[n - 1] + t_cand - t_cand * t_cand
                f_cand = np.max(mid)
                if f_cand < f_best - 1e-14:
                    t_best = t_cand
                    f_best = f_cand

            if t_best != old:
                if t_best > old:
                    lo = old
                    hi = min(1.0, t_best + 0.1)
                else:
                    lo = max(0.0, t_best - 0.1)
                    hi = old
                if hi - lo < 1e-10:
                    hi = lo + 1e-8

                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                def _eval_t(t):
                    mid = base + A * t
                    mid[n - 1] = base[n - 1] + t - t * t
                    return np.max(mid)

                f1 = _eval_t(c1)
                f2 = _eval_t(c2)
                for _ in range(30):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _eval_t(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = _eval_t(c2)
                t_best = (lo + hi) / 2.0
                f_best = _eval_t(t_best)

            if f_best < f_old - 1e-13:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                improved = True

        if not improved:
            break

    return best, np.max(best_corr) * dx


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]


def _g_smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]


def _g_constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0

def slsqp_refine(h, n, beta, maxiter=300):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_constraint_sum, 'jac': _g_constraint_sum_jac}]
    try:
        res = minimize(_g_smooth_obj, h, jac=_g_smooth_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-25, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)

def active_set_gradient_step(h, n, n_active=20, step_size=0.05, beta=500):
    dx = 2.0 / n
    corr = compute_corr(h)
    ncorr = len(corr)
    top_k = np.argsort(corr)[::-1][:n_active]
    corr_sub = corr[top_k]
    m = np.max(corr_sub)
    shifted = np.minimum(beta * (corr_sub - m), 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)

    grad = np.zeros(n)
    for k_idx in top_k:
        w = weights[np.where(top_k == k_idx)[0][0]]
        if w < 1e-14:
            continue
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * h[:npairs]
    grad *= dx
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h
    h_new = h - step_size * grad
    return project_feasible(h_new, n)


def multi_scale_attack(h, n, rng):
    """Try to improve by identifying worst lags and directly reducing them."""
    best = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(best)
    corr = compute_corr(best)

    for iteration in range(50):
        if remaining_time() < 10:
            break
        # Find the worst lag
        worst_k = int(np.argmax(corr))
        offset = worst_k - (n - 1)

        # Try to reduce this specific lag by adjusting h
        if offset >= 0:
            npairs = n - offset
            pairs = [(i, i + offset) for i in range(npairs)]
        else:
            aoff = -offset
            npairs = n - aoff
            pairs = [(i, i - aoff) for i in range(aoff, n)]

        # For each pair (i, j) contributing to this lag, try to reduce h[i]*(1-h[j])
        # This means either decrease h[i] or increase h[j]
        # But we must maintain the sum constraint
        # Strategy: find the pair with maximum contribution and try to reduce it
        contributions = []
        for (i, j) in pairs:
            contributions.append((best[i] * (1 - best[j]), i, j))
        contributions.sort(reverse=True)

        improved = False
        for contrib, i, j in contributions[:5]:
            if contrib < 1e-6:
                break
            # Try to reduce: decrease h[i] or increase h[j]
            # Decrease h[i] by delta, increase some other position to maintain sum
            # Simple approach: move a bit of mass from i to j
            delta = min(contrib * 0.5, best[i] * 0.3, (1.0 - best[j]) * 0.3)
            if delta < 1e-6:
                continue
            trial = best.copy()
            trial[i] -= delta
            trial[j] += delta
            trial = np.clip(trial, 0, 1)
            trial = project_feasible(trial, n)
            c5_trial = compute_c5(trial)
            if c5_trial < best_c5 - 1e-10:
                best = trial
                best_c5 = c5_trial
                corr = compute_corr(best)
                improved = True
                break

        if not improved:
            break

    return best, best_c5

    if remaining_time() > 60:
        h_sls = slsqp_refine(best_h, n, beta=3000, maxiter=400)
        c5_sls = compute_c5(h_sls)
        if c5_sls < best_c5:
            best_c5 = c5_sls
            best_h = h_sls.copy()

    # Phase 5: More binary SA with different parameters
    if remaining_time() > 300:
        h_bin3 = (best_h > 0.5).astype(float)
        n_ones3 = int(np.sum(h_bin3))
        if n_ones3 > n // 2:
            ones_idx = np.where(h_bin3 == 1)[0]
            h_bin3[ones_idx[:n_ones3 - n // 2]] = 0
        elif n_ones3 < n // 2:
            zeros_idx = np.where(h_bin3 == 0)[0]
            h_bin3[zeros_idx[:n // 2 - n_ones3]] = 1

        h_sa3, c5_sa3 = binary_sa_search(h_bin3, n, rng, max_iters=400000, T0=1.2, T1=0.005, patience=12000)
        if c5_sa3 < best_c5:
            best_c5 = c5_sa3
            best_h = h_sa3.copy()

    # Phase 6: Active set gradient + coordinate descent
    if remaining_time() > 120:
        h_act = best_h.copy()
        for it in range(200):
            if remaining_time() < 10:
                break
            h_act = active_set_gradient_step(h_act, n, n_active=30, step_size=0.03, beta=200)
            c5_act = compute_c5(h_act)
            if c5_act < best_c5 - 1e-10:
                best_c5 = c5_act
                best_h = h_act.copy()
            elif it > 50 and it % 20 == 0:
                h_cd2, c5_cd2 = fast_coord_descent_cont(h_act, n, max_sweeps=30)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()
                    h_act = h_cd2.copy()

    # Phase 7: Higher resolution
    if remaining_time() > 200:
        high_n = 2 * n
        if high_n <= 2048:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            h_high_cd, c5_high = fast_coord_descent_cont(h_high, high_n, max_sweeps=50)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high_cd.copy()
                best_n = high_n

            if remaining_time() > 60:
                h_high_sls = slsqp_refine(best_h, best_n, beta=5000, maxiter=200)
                c5_high_sls = compute_c5(h_high_sls)
                if c5_high_sls < best_c5:
                    best_c5 = c5_high_sls
                    best_h = h_high_sls.copy()
                    best_n = high_n

    # Phase 8: Multi-scale attack
    if remaining_time() > 60:
        h_ms, c5_ms = multi_scale_attack(best_h, len(best_h), rng)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 20:
        n_cur = len(best_h)
        h_final, c5_final = fast_coord_descent_cont(best_h, n_cur, max_sweeps=60)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 10: Final SLSQP
    if remaining_time() > 15:
        n_cur = len(best_h)
        h_fsls = slsqp_refine(best_h, n_cur, beta=20000, maxiter=200)
        c5_fsls = compute_c5(h_fsls)
        if c5_fsls < best_c5:
            best_c5 = c5_fsls
            best_h = h_fsls.copy()

    # Final projection
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
