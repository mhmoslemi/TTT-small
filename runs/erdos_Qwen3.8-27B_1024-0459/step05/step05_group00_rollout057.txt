Actually, I realize the key issue: $C_k(h) = \sum h_i(1-h_j)$ over appropriate indices is a **quadratic** function of $h$. Specifically, it's bilinear (linear in each argument when the other is fixed). So the problem $\min_h \max_k C_k(h)$ is a min-max of bilinear functions, which is nonconvex.

2. Evaluate actual $\max_k C_k(h_{LP})$

def grad_corr_k(h, k, n):

    g = np.zeros(n)

    else:

The correlation is:
$c_k = \sum_{max(0,k-n+1)}^{min(n-1, k+n-1)} h_i \cdot h_{i-n+1+k}$ (full mode)

Wait, let me think about this more carefully. `np.correlate(a, v, mode='full')` computes:
$c_k = \sum_n a_{n+k} \overline{v_n}$ for $k = -(len(v)-1), ..., len(a)-1$

In our case, `a = h`, `v = 1-h`, so:
$c_k = \sum_n h_{n+k}(1-h_n)$

For the gradient $\partial c_k / \partial h_i$:
- $h_i$ appears as $h_{n+k}$ when $n = i-k$: contributes $(1-h_{i-k})$ if $0 \leq i-k < n$
- $h_i$ appears as $h_n$ when $n = i$: contributes $-h_{i+k}$ if $0 \leq i+k < n$

So:
$\frac{\partial c_k}{\partial h_i} = (1-h_{i-k}) \cdot \mathbf{1}[0 \leq i-k < n] - h_{i+k} \cdot \mathbf{1}[0 \leq i+k < n]$

In the code's indexing, `offset = k - (n-1)` maps the correlation index to the actual lag. Let me verify: for `np.correlate(h, 1-h, mode='full')`, the output has length $2n-1$, indexed from 0 to $2n-2$. The lag at index $k$ is $k - (n-1)$. So for index $k$, the actual shift is $d = k - (n-1)$.

Then $c_k = \sum_{j: 0 \leq j < n, 0 \leq j+d < n} h_{j+d}(1-h_j)$

$\partial c_k / \partial h_i = (1-h_{i-d}) \cdot \mathbf{1}[0 \leq i-d < n] - h_{i+d} \cdot \mathbf{1}[0 \leq i+d < n]$

        g[:npair] -= h[offset:offset + npair]
    else:
        a = -offset
        npair = n - a
        g[:npair] += (1.0 - h[a:a + npair])
        g[a:a + npair] -= h[:npair]

        c_top = corr[idx]
        weights = np.exp(beta_g * (c_top - c_top.max()))
        s = weights.sum()
        if s < 1e-30:
            weights = np.ones(K) / K
        else:
            weights /= s
        g = np.zeros(n)
        for i, k in enumerate(idx):
            if weights[i] > 1e-12:
                g += weights[i] * grad_corr_k(h, k, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g_norm = g / gn
        base_step = 0.04 / (1.0 + 0.05 * it)
        base_step = min(base_step, 0.15)
        improved = False
        for step in [base_step, base_step * 0.5, base_step * 2, base_step * 0.25, base_step * 0.1]:
            ht = project_feasible(h - step * g_norm, n)
            cn = compute_c5(ht)
            if cn < best_c5:
                best_c5 = cn
                best_h = ht.copy()
                h = ht.copy()
                improved = True
                break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    best_h = project_feasible(parent_h, parent_n)
    best_c5 = compute_c5(best_h)
    best_n = parent_n

    # Phase 1: SLSQP at parent resolution - increasing beta
    for beta in [2000, 8000, 25000, 60000, 150000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(best_h, best_n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Equalize top shifts
    if remaining() > 20:
        h_eq, c5_eq = equalize_top(best_h, best_n, K=6, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 3: Cutting plane at parent resolution
    if remaining() > 40:
        for it in range(150):
            if remaining() < 5:
                break
            h_lp, t_lp = cutting_plane_lp(best_h, best_n, K=30, delta=0.12)
            if t_lp < best_c5 - 1e-10:
                improved = False
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.8, 0.2]:
                    h_try = project_feasible(best_h + alpha * (h_lp - best_h), best_n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c5:
                        best_c5 = c_try
                        best_h = h_try.copy()
                        improved = True
                        break
                if not improved:
                    break
            elif it > 80:
                break

    # Phase 4: SLSQP after cutting plane
    for beta in [100000, 300000, 800000]:
        if remaining() < 10:
            break
        h_new = slsqp_optimize(best_h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Top-K gradient descent
    if remaining() > 20:
        h_tk, c5_tk = topk_gradient_descent(best_h, best_n, K=8, beta=80.0, max_iters=200)
        if c5_tk < best_c5:
            best_c5 = c5_tk
            best_h = h_tk.copy()

    # Phase 6: Try higher resolution (2x)
    n_hi = best_n * 2
    if remaining() > 250 and n_hi <= 2048:
        h_hi = interp_to(best_h, best_n, n_hi)
        h_hi = project_feasible(h_hi, n_hi)
        c5_hi = compute_c5(h_hi)

        for beta in [5000, 20000, 80000, 200000]:
            if remaining() < 30:
                break
            h_hi = slsqp_optimize(h_hi, n_hi, beta, maxiter=500)
            c5_hi = compute_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n_hi

        if remaining() > 30:
            h_eq, c5_eq = equalize_top(best_h, best_n, K=6, max_iters=200)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()

        if remaining() > 40:
            for it in range(60):
                if remaining() < 5:
                    break
                h_lp, t_lp = cutting_plane_lp(best_h, best_n, K=25, delta=0.08)
                if t_lp < best_c5 - 1e-10:
                    improved = False
                    for alpha in [1.0, 0.5, 0.3, 0.7]:
                        h_try = project_feasible(best_h + alpha * (h_lp - best_h), best_n)
                        c_try = compute_c5(h_try)
                        if c_try < best_c5:
                            best_c5 = c_try
                            best_h = h_try.copy()
                            improved = True
                            break
                    if not improved:
                        break
                elif it > 30:
                    break

        for beta in [300000, 600000, 1200000]:
            if remaining() < 15:
                break
            h_new = slsqp_optimize(best_h, best_n, beta, maxiter=600)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        if remaining() > 15:
            h_tk, c5_tk = topk_gradient_descent(best_h, best_n, K=6, beta=100.0, max_iters=150)
            if c5_tk < best_c5:
                best_c5 = c5_tk
                best_h = h_tk.copy()

    # Phase 7: Multi-restart with perturbations at best resolution
    if remaining() > 80:
        for restart in range(4):
            if remaining() < 40:
                break
            noise_scale = 0.03 * (restart + 1) / 4.0
            noise = rng.randn(best_n) * noise_scale
            h_pert = project_feasible(best_h + noise, best_n)
            h_cur = h_pert.copy()
            c5_cur = compute_c5(h_cur)
            if c5_cur > best_c5 * 1.05:
                continue
            for beta in [8000, 50000, 200000]:
                if remaining() < 15:
                    break
                h_cur = slsqp_optimize(h_cur, best_n, beta, maxiter=400)
                c5_cur = compute_c5(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()

    # Phase 8: Final intensive SLSQP
    for beta in [500000, 1500000, 4000000]:
        if remaining() < 8:
            break
        h_new = slsqp_optimize(best_h, best_n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Final equalize + gradient descent
    if remaining() > 15:
        h_eq, c5_eq = equalize_top(best_h, best_n, K=8, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining() > 10:
        h_tk, c5_tk = topk_gradient_descent(best_h, best_n, K=5, beta=150.0, max_iters=100)
        if c5_tk < best_c5:
            best_c5 = c5_tk
            best_h = h_tk.copy()

    # Phase 10: Final SLSQP polish
    if remaining() > 8:
        h_new = slsqp_optimize(best_h, best_n, 5000000, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
