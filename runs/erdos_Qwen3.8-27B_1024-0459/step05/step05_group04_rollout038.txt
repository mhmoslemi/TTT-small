```python
def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with soft-max smoothing."""
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return m * (2.0 / n)
        return (m + np.log(s) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
```

```python
def coordinate_step(h, i, n):
    """Optimize h[i] while keeping other coordinates fixed."""
    # f_k(h) = sum_t h[t]*(1-h[t+n-1-k]) for valid t
    # When we change h[i], the affected terms are:
    #   If i is the first factor (t=i): f_k changes by (1-h[i+n-1-k]) * delta for k where valid
    #   If i is the second factor (t+n-1-k=i): f_k changes by -h[t] * delta for k where valid
    
    # Let's compute the current f_k for all k, then figure out how they change with h[i]
    corr = compute_corr(h)
    
    # For a unit increase in h[i] (delta=+1):
    # First factor: k in [i, i+n-1], change = (1-h[i+n-1-k])
    # Second factor: k in [n-1-i, 2n-2-i], change = -h[k-(n-1-i)]
    
    # So df_k/dh[i] = (1-h[i+n-1-k]) for k in [i, i+n-1]
    #                + (-h[k-(n-1-i)]) for k in [n-1-i, 2n-2-i]
    # (with appropriate bounds)
    
    # The new corr would be: corr + delta * df/dh[i]
    # We want to choose delta in [-(h[i]), 1-h[i]] to minimize max(corr + delta * df)
    
    df = np.zeros(2*n-1)
    # First factor
    lo = i
    hi = min(i + n - 1, 2*n-2)
    if lo <= hi:
        ks = np.arange(lo, hi+1)
        df[lo:hi+1] += (1.0 - h[i + n - 1 - ks])
    # Second factor  
    lo2 = max(n-1-i, 0)
    hi2 = min(2*n-2-i, 2*n-2)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2+1)
        t_vals = ks - (n-1-i)
        df[lo2:hi2+1] += -h[t_vals]
    
    # Now minimize max(corr + delta * df) over delta in [-h[i], 1-h[i]]
    # This is a 1D convex piecewise-linear problem
    # The optimal delta is where the "slope" changes sign
    
    # values(delta) = corr + delta * df
    # We want min_delta max_k values(delta)_k
    
    # The breakpoints are where corr[k] + delta*df[k] = corr[k'] + delta*df[k']
    # i.e., delta = (corr[k'] - corr[k]) / (df[k] - df[k']) for df[k] != df[k']
    
    # More efficient: sort by df values, then the max is determined by the "upper envelope"
    # Actually, for a fixed set of linear functions, the min of the max can be found
    # by binary search on the answer.
    
    # Binary search: find the smallest M such that there exists delta with
    # corr[k] + delta * df[k] <= M for all k
    # For each k: delta <= (M - corr[k]) / df[k] if df[k] > 0
    #             delta >= (M - corr[k]) / df[k] if df[k] < 0
    #             no constraint if df[k] == 0 (need corr[k] <= M)
    
    lo_delta = -h[i]
    hi_delta = 1.0 - h[i]
    
    # First, check if the current value is already optimal
    # (i.e., the derivative of max at current point is 0 or the point is at a boundary)
    
    # For efficiency, let's use a simpler approach: ternary search or golden section
    # Since the function is convex in delta, we can use binary search on the derivative
    
    # Actually, the cleanest approach: the optimal delta minimizes
    # g(delta) = max_k (corr[k] + delta * df[k])
    # This is a convex piecewise-linear function. The minimum is at a point where
    # the subgradient contains 0, i.e., where the "kinks" change from negative to positive slope.
    
    # The slope of g at delta is the max of df[k] over k in the argmax set.
    # So we want the largest delta where min df[k] in argmax <= 0 and the smallest delta where max df[k] in argmax >= 0.
    
    # Simple approach: sample delta values and find the minimum
    n_samples = 100
    deltas = np.linspace(lo_delta, hi_delta, n_samples)
    best_d = 0.0
    best_val = float('inf')
    for d in deltas:
        val = np.max(corr + d * df)
        if val < best_val:
            best_val = val
            best_d = d
    h[i] += best_d
    return h[i]
```

```python
def make_swap_delta(h, i, j, n):
```

```python
def slsqp_smooth(h, n, beta, maxiter=200):
    h = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return m * (2.0 / n)
        return (m + np.log(s) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
```
