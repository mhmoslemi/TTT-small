import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def to_binary(h, n):
    n_half = n // 2
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h_bin == 1)[0]
            conf = h[ones]
            order = np.argsort(conf)[:n_ones - n_half]
            h_bin[ones[order]] = 0
        else:
            zeros = np.where(h_bin == 0)[0]
            conf = h[zeros]
            order = np.argsort(-conf)[:n_half - n_ones]
            h_bin[zeros[order]] = 1
    return h_bin

def swap_new_max(h, i, j, n, corr):
    """Compute new max correlation after swap i(1->0), j(0->1). Fast O(n) vectorized."""
    nm1 = n - 1
    new_corr = corr.copy()
    
    # Position i (1->0):
    d_pos_i = n - 1 - i
    d_neg_i = i
    if d_pos_i > 0:
        d = np.arange(1, d_pos_i + 1)
        new_corr[nm1 + d] += h[i + d] - 1.0
        new_corr[nm1 - d] += h[i + d]
    if d_neg_i > 0:
        d = np.arange(1, d_neg_i + 1)
        new_corr[nm1 + d] += h[i - d]
        new_corr[nm1 - d] += h[i - d] - 1.0
    
    # Position j (0->1):
    d_pos_j = n - 1 - j
    d_neg_j = j
    if d_pos_j > 0:
        d = np.arange(1, d_pos_j + 1)
        new_corr[nm1 + d] += 1.0 - h[j + d]
        new_corr[nm1 - d] -= h[j + d]
    if d_neg_j > 0:
        d = np.arange(1, d_neg_j + 1)
        new_corr[nm1 + d] -= h[j - d]
        new_corr[nm1 - d] += 1.0 - h[j - d]
    
    return float(np.max(new_corr)), new_corr

def binary_local_search_fast(h_bin, n, time_limit=100, rng=None):
    """Fast binary local search with vectorized swap evaluation."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h_bin.copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    stall = 0
    max_stall = 300
    
    start = time.time()
    while time.time() - start < time_limit and remaining() > 15:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Generate candidates from top shifts
        top_k = np.argsort(-corr)[:20]
        candidates = []
        seen = set()
        
        for s in top_k[:12]:
            off = int(s) - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                    for ci in contrib[:20]:
                        key = (int(ci) + off, int(ci))
                        if key not in seen:
                            seen.add(key)
                            candidates.append(key)
            else:
                ao = -off
                np_ = n - ao
                if np_ > 0:
                    contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                    for ci in contrib[:20]:
                        key = (int(ci), int(ci) + ao)
                        if key not in seen:
                            seen.add(key)
                            candidates.append(key)
        
        # Add random candidates
        n_rand = 50
        for _ in range(n_rand):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            key = (i, j)
            if key not in seen:
                seen.add(key)
                candidates.append(key)

        best_move = None
        best_new_max = cur_max + 1e-15

        top_shift_set = set(top_k[:12].tolist())

                best_new_max = new_max

            h[j] = 1

def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with smooth max approximation."""
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj(h) - 1e-12:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def top_shifts_equalize(h, n, top_k=5, max_iters=200):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        
        g_avg = np.zeros(n)
        for k in idx:
            off = int(k) - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]
        
        bot_idx = np.argsort(corr)[:min(3, len(corr))]
        g_bot = np.zeros(n)
        for k in bot_idx:
            off = int(k) - (n - 1)
            if off >= 0:
                np_ = n - off
                g_bot[off:off + np_] += (1.0 - h[:np_])
                g_bot[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_bot[:np_] += (1.0 - h[ao:ao + np_])
                g_bot[ao:ao + np_] -= h[:np_]
        
        d = g_bot - g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= d
        
        improved = False
        for step in [0.01, 0.03, 0.05, 0.02, 0.08, 0.005, 0.1, 0.015, 0.04, 0.06]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5
