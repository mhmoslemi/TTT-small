    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[:np_] += wk * (1.0 - h[off:off + np_])
            g[off:off + np_] -= wk * h[:np_]
        else:
            a = -off
            np_ = n - a
            g[a:a + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[a:a + np_]
    return g


def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    return corr_grad_weighted(h, w * dx)


def slsqp_run(h, n, beta, maxiter=300):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def binary_local_search_v2(h, n, max_rounds=80):
    """Fast binary local search: O(n^3) per round, vectorized over q for each p."""
    h = (h > 0.5).astype(float).copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = float(np.max(c) * dx)
    best_h = h.copy()
    c_max = float(np.max(c))

    for rnd in range(max_rounds):
        if remaining() < 5:
            break

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break

        # Precompute add_effects for all zeros positions
        add_effects = np.zeros((n_zeros, 2 * n - 1), dtype=np.float64)
        for iq in range(n_zeros):
            q = zeros[iq]
            q_lo = n - 1 - q
            q_hi = n - 1 + q
            q_pos = np.arange(0, 2 * q + 1)
            add_effects[iq, q_lo:q_hi + 1] = 1.0 - 2.0 * h[q_pos]

        best_improve = 0.0
        best_p = -1
        best_iq = -1

        for ia in range(n_ones):
            p = ones[ia]
            k_lo = n - 1 - p
            k_hi = n - 1 + p
            pos = np.arange(0, 2 * p + 1)
            delta_remove = np.zeros(2 * n - 1)
            delta_remove[k_lo:k_hi + 1] = 2.0 * h[pos] - 1.0

            base = c + delta_remove
            # new_c[iq, :] = base + add_effects[iq, :]
            new_maxes = np.max(base[None, :] + add_effects, axis=1)
            improves = c_max - new_maxes
            idx = int(np.argmax(improves))
            if improves[idx] > best_improve:
                best_improve = improves[idx]
                best_p = p
                best_iq = idx

        if best_p < 0:
            break

        zp = zeros[best_iq]
        h[best_p] = 0.0
        h[zp] = 1.0
        c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        new_c_max = float(np.max(c))
        c5_new = new_c_max * dx
        c_max = new_c_max

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    return best_h, best_c5


def simulated_annealing_binary(h, n, max_iters=8000, seed=123):
    h = (h > 0.5).astype(float).copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c5 = float(np.max(c) * dx)
    best_c5 = cur_c5
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    T_start = 0.3
    T_end = 0.005

    for it in range(max_iters):
        if remaining() < 5:
            break
        T = T_start * (T_end / T_start) ** (float(it) / float(max_iters))

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        if p == q:
            continue

        h[p] = 0.0
        h[q] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr) * dx)

        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            c = new_corr.astype(np.float64)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            delta = new_c5 - cur_c5
            if delta < T * (-np.log(max(rng.uniform(), 1e-300))):
                cur_c5 = new_c5
                c = new_corr.astype(np.float64)
            else:
                h[q] = 0.0
                h[p] = 1.0

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def thue_morse_init(n):
    """Generate Thue-Morse binary sequence of length n with exactly n//2 ones."""
    tm = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        tm[i] = c & 1
    # Adjust to have exactly n//2 ones
    n_ones = int(np.sum(tm))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(tm == 1)[0]
        rng = np.random.RandomState(7)
        to_flip = rng.choice(one_idx, n_ones - target, replace=False)
        tm[to_flip] = 0
    elif n_ones < target:
        zero_idx = np.where(tm == 0)[0]
        rng = np.random.RandomState(7)
        to_flip = rng.choice(zero_idx, target - n_ones, replace=False)
        tm[to_flip] = 1
    return tm


def projected_gradient_smooth(h, n, beta, max_iters=3000):
    """Nesterov-accelerated projected gradient on smooth max."""
    h = project_feasible(h, n)
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0

    for it in range(max_iters):
        if remaining() < 5:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        g = smooth_grad(h, beta)
        lr = 8.0 / (beta * (it + 50) ** 0.7)
        y_new = y + alpha * (y - h) - lr * g
        h_new = project_feasible(y_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        best_h = parent_h.copy()
        best_n = parent_n
        best_c5 = compute_c5(best_h)
    except Exception:
        parent_n = 384
        best_n = parent_n
        best_h = thue_morse_init(parent_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Up-resolve to n=768 and optimize
    n_main = 768
    h_main = interp_to(best_h, best_n, n_main)
    h_main = project_feasible(h_main, n_main)
    c5_main = compute_c5(h_main)
    if c5_main < best_c5:
        best_c5 = c5_main
        best_h = h_main.copy()
        best_n = n_main

    # SLSQP continuous optimization at n=768
    for beta in [200, 1000, 5000, 20000]:
        if remaining() < 20:
            break
        h_opt = slsqp_run(h_main, n_main, beta, maxiter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_main
            h_main = h_opt.copy()

    # Phase 2: Binary local search at n=768
    if remaining() > 30:
        h_bin = round_to_binary(h_main, n_main)
        h_b, c5_b = binary_local_search_v2(h_bin, n_main, max_rounds=100)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n_main
            h_main = h_b.copy()

    # Phase 3: SLSQP polish after binary LS
    if remaining() > 20:
        for beta in [5000, 30000, 100000]:
            if remaining() < 15:
                break
            h_pol = slsqp_run(h_main, n_main, beta, maxiter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_main
                h_main = h_pol.copy()

    # Phase 4: Binary LS again after continuous polish
    if remaining() > 30:
        h_bin2 = round_to_binary(h_main, n_main)
        h_b2, c5_b2 = binary_local_search_v2(h_bin2, n_main, max_rounds=80)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = n_main
            h_main = h_b2.copy()

    # Phase 5: Simulated annealing
    if remaining() > 40:
        h_sa, c5_sa = simulated_annealing_binary(h_main, n_main, max_iters=6000, seed=99)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_main
            h_main = h_sa.copy()
        # Another round of binary LS after SA
        if remaining() > 20:
            h_b3, c5_b3 = binary_local_search_v2(h_main, n_main, max_rounds=60)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_main
                h_main = h_b3.copy()

    # Phase 6: Try Thue-Morse initialization at n=768
    if remaining() > 60:
        h_tm = thue_morse_init(n_main)
        h_tm_b, c5_tm = binary_local_search_v2(h_tm, n_main, max_rounds=100)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm_b.copy()
            best_n = n_main
            h_main = h_tm_b.copy()
        # SLSQP from Thue-Morse result
        if c5_tm < best_c5 + 0.005 and remaining() > 30:
            for beta in [5000, 30000]:
                if remaining() < 15:
                    break
                h_tm_opt = slsqp_run(h_main, n_main, beta, maxiter=300)
                c5_tm_opt = compute_c5(h_tm_opt)
                if c5_tm_opt < best_c5:
                    best_c5 = c5_tm_opt
                    best_h = h_tm_opt.copy()
                    best_n = n_main
                    h_main = h_tm_opt.copy()

    # Phase 7: Try n=960
    if remaining() > 80:
        n_alt = 960
        h_alt = interp_to(best_h, best_n, n_alt)
        h_alt = project_feasible(h_alt, n_alt)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5 + 0.002:
            h_alt = slsqp_run(h_alt, n_alt, 5000, maxiter=200)
            c5_alt = compute_c5(h_alt)
            h_alt_bin = round_to_binary(h_alt, n_alt)
            h_alt_b, c5_alt_b = binary_local_search_v2(h_alt_bin, n_alt, max_rounds=60)
            if c5_alt_b < c5_alt:
                c5_alt = c5_alt_b
                h_alt = h_alt_b.copy()
            if remaining() > 20:
                h_alt = slsqp_run(h_alt, n_alt, 20000, maxiter=200)
                c5_alt = compute_c5(h_alt)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                best_n = n_alt
                h_main = h_alt.copy()

    # Phase 8: Try n=640
    if remaining() > 60:
        n_alt2 = 640
        h_alt2 = interp_to(best_h, best_n, n_alt2)
        h_alt2 = project_feasible(h_alt2, n_alt2)
        h_alt2_bin = round_to_binary(h_alt2, n_alt2)
        h_alt2_b, c5_alt2_b = binary_local_search_v2(h_alt2_bin, n_alt2, max_rounds=80)
        c5_alt2 = c5_alt2_b
        if remaining() > 20:
            h_alt2_opt = slsqp_run(h_alt2_b, n_alt2, 10000, maxiter=200)
            c5_alt2_opt = compute_c5(h_alt2_opt)
            if c5_alt2_opt < c5_alt2:
                c5_alt2 = c5_alt2_opt
                h_alt2_b = h_alt2_opt.copy()
        if c5_alt2 < best_c5:
            best_c5 = c5_alt2
            best_h = h_alt2_b.copy()
            best_n = n_alt2
            h_main = h_alt2_b.copy()

    # Phase 9: Final intensive polish at best resolution
    n_final = best_n
    h_final = best_h.copy()

    for beta in [10000, 50000, 150000, 400000]:
        if remaining() < 10:
            break
        h_p = slsqp_run(h_final, n_final, beta, maxiter=400)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = n_final
            h_final = h_p.copy()

    # Final binary LS
    if remaining() > 15:
        h_fb = round_to_binary(h_final, best_n)
        h_fb, c5_fb = binary_local_search_v2(h_fb, best_n, max_rounds=60)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()
            best_n = best_n
            h_final = h_fb.copy()

    # Final SLSQP
    if remaining() > 10:
        h_fp = slsqp_run(h_final, best_n, 100000, maxiter=300)
        c5_fp = compute_c5(h_fp)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    # Ensure exact integral
    total = np.sum(best_h)
    target = best_n / 2.0
    if abs(total - target) > 1e-12:
        best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
