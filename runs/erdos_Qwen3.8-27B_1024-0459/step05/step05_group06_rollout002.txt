    wsum = np.sum(exp_vals)

    weights = exp_vals / wsum

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def smooth_max_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - h[:npairs])
                grad[:npairs] -= w * h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * h[:npairs]
    grad *= dx
    return val, grad, corr, m, weights

def projected_gradient_method(h, n, beta, n_iter, momentum=0.85):
    best = h.copy()
    best_c5 = compute_c5(best)
    velocity = np.zeros(n)
    step = 0.01
    c5_prev = best_c5
    
    for it in range(n_iter):
        if remaining_time() < 10:
            break
        val, grad, corr, m, weights = smooth_max_and_grad(best, n, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        velocity = momentum * velocity - step * grad
        trial = best + velocity
        trial = project_feasible(trial, n)
        c5_trial = compute_c5(trial)
        
        if c5_trial > best_c5 + 0.001:
            step *= 0.5
            velocity = momentum * velocity
            trial = best + velocity
            trial = project_feasible(trial, n)
            c5_trial = compute_c5(trial)
        
        if c5_trial < best_c5 - 1e-15:
            best = trial
            best_c5 = c5_trial
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.8
            velocity = np.zeros(n)
        
        if it % 500 == 0 and it > 0:
            step = 0.01
    
    return best, best_c5

def coord_descent(h, n, n_sweeps, verbose=False):
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for sweep in range(n_sweeps):

            break
        improved_any = False
        
        for i in range(n):
            if remaining_time() < 1:
                break
            t0 = best[i]
            
            a = np.zeros(ncorr)
            b = np.zeros(ncorr)
            for off in range(-(n - 1), n):
                k = n - 1 + off
                if off == 0:
                    a[k] += 1.0
                    b[k] += -1.0
                else:
                    if 0 <= i + off < n:
                        a[k] += (1.0 - best[i + off])
                    if 0 <= i - off < n:
                        a[k] += (-best[i - off])
            
            base = best_corr - a * t0 - b * t0 * t0
            
            lo, hi = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = np.max(base + a * c1 + b * c1 * c1)
            f2 = np.max(base + a * c2 + b * c2 * c2)
            
            for _ in range(50):
                if hi - lo < 1e-10:
                    break
                if f1 < f2:
                    hi = c2; c2 = c1; f2 = f1
                    c1 = hi - gr * (hi - lo)
                    f1 = np.max(base + a * c1 + b * c1 * c1)
                else:
                    lo = c1; c1 = c2; f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = np.max(base + a * c2 + b * c2 * c2)
            
            t_opt = (lo + hi) / 2.0
            f_opt = np.max(base + a * t_opt + b * t_opt * t_opt)
            
            if f_opt * dx < best_c5 - 1e-13:
                best[i] = t_opt
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved_any = True

def slsqp_optimize(h, n, beta, maxiter=300):
    dx = 2.0 / n
    
    def obj_fn(x):
        return smooth_max_and_grad(x, n, beta)[0]
    
    def grad_fn(x):
        return smooth_max_and_grad(x, n, beta)[1]
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    c5_before = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def block_slsqp(h, n, block_size=32, beta=5000, maxiter=100):

    best = h.copy()
    best_c5 = compute_c5(best)
    dx = 2.0 / n

        if remaining_time() < 10:
            break

        bsize = end - start
        x0 = best[start:end].copy()
        
        def obj_block(xb):
            x_full = best.copy()
            x_full[start:end] = xb
            return smooth_max_and_grad(x_full, n, beta)[0]
        
        def grad_block(xb):
            x_full = best.copy()
            x_full[start:end] = xb
            _, g = smooth_max_and_grad(x_full, n, beta)
            return g[start:end]
        
        bounds = [(0.0, 1.0)] * bsize
        max_iter = min(maxiter, 50)
        
        try:
            res = minimize(obj_block, x0, jac=grad_block, method='SLSQP',
                         bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):

                best = project_feasible(best, n)
                c5_new = compute_c5(best)
                if c5_new > best_c5 + 1e-10:
                    best[start:end] = x0
                    best = project_feasible(best, n)
                else:
                    best_c5 = compute_c5(best)
        except Exception:
            pass
