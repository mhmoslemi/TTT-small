```python
def mw_optimize(h, n, max_iters=1000, time_limit=300):
    """Multiplicative weights / game-theoretic approach."""
    dx = 2.0 / n
    h = project_feasible(h, n)
    start = time.time()
    
    # Initialize weights uniformly
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    
    for it in range(max_iters):
        if time.time() - start > time_limit or remaining_time() < 10:
            break
        
        # Compute weighted objective gradient
        corr = compute_corr(h)
        
        # Update weights (multiplicative weights)
        c_max = np.max(corr)
        c_min = np.min(corr)
        if c_max - c_min < 1e-12:
            break
        # Exponential weights
        eta = 0.5 / (c_max - c_min)
        w *= np.exp(eta * corr)
        w /= np.sum(w)
        
        # Compute gradient of weighted objective
        grad = np.zeros(n)
        for k in range(n_lags):
            if w[k] < 1e-10:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                grad[off:] += w[k] * (1.0 - h[:np_])
                grad[:np_] -= w[k] * h[off:]
            else:
                a = -off
                np_ = n - a
                grad[:np_] += w[k] * (1.0 - h[a:])
                grad[a:] -= w[k] * h[:np_]
        
        # Gradient descent step with projection
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        d = -grad / gn
        
        # Line search
        best_h = h
        best_c5 = compute_c5(h)
        for step in [0.01, 0.005, 0.02, 0.001, 0.05, 0.002]:
            h_try = project_feasible(h + step * d, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        
        h = best_h
    
    return h, compute_c5(h)
```

```python
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_CUR_N = 0
_SM_BETA = 100.0

def remaining_time():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).flatten(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_single(h, n, k):
    """Gradient of corr[k] with respect to h."""
    g = np.zeros(n)
    off = k - (n - 1)
    if off >= 0:
        np_ = n - off
        g[off:] += (1.0 - h[:np_])
        g[:np_] -= h[off:]
    else:
        a = -off
        np_ = n - a
        g[:np_] += (1.0 - h[a:])
        g[a:] -= h[:np_]
    return g

def smooth_max_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    z = np.clip(beta * (corr - m), -50.0, 50.0)
    ez = np.exp(z)
    S = np.sum(ez)
    w = ez / S
    sm = (m + np.log(S) / beta) * dx
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        g += wk * corr_grad_single(h, n, k)
    g *= dx
    return sm, g

def lbfgs_opt(h, n, beta, maxiter=500, mu=1000.0):
    """L-BFGS-B with penalty for equality constraint."""
    h0 = project_feasible(h, n).copy()
    target = n / 2.0
    
    def obj(x):
        sm, _ = smooth_max_and_grad(x, n, beta)
        pen = mu * (np.sum(x) - target) ** 2
        return sm + pen
    
    def grad(x):
        sm, g = smooth_max_and_grad(x, n, beta)
        pen_g = 2 * mu * (np.sum(x) - target) * np.ones(n)
        return g + pen_g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10,
                                           'ftol': 1e-14, 'maxfun': maxiter * 3})
    result = project_feasible(res.x, n)
    return result

def directed_descent(h, n, top_k=5, max_iters=300, time_limit=60):
    """Directly reduce the top-k lags."""
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        
        # Compute combined gradient for top-k lags
        g_dir = np.zeros(n)
        for k in idx_top:
            g_dir += corr_grad_single(h, n, k)
        
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = -g_dir / gn
        
        # Line search
        improved = False
        for step in [0.01, 0.005, 0.02, 0.001, 0.05, 0.002, 0.08, 0.03]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        
        if not improved:
            # Try smaller steps
            for step in [0.0005, 0.001, 0.002]:
                ht = project_feasible(h + step * d, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn

                    break
            if not improved:

    return h, best_c5

def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xn = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xn, xp, src)

def slsqp_opt(h, n, beta, maxiter=300):
    """SLSQP with smooth max and equality constraint."""
    global _CUR_N, _SM_BETA
    _CUR_N = n
    _SM_BETA = beta
    h0 = project_feasible(h, n)
    
    def obj(x):
        sm, _ = smooth_max_and_grad(x, n, beta)
        return sm
    
    def grad(x):
        _, g = smooth_max_and_grad(x, n, beta)
        return g
    
    def eq_fun(x):
        return np.sum(x) - n / 2.0
    
    def eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.fun < 1e6:  # Check it didn't fail
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def theta_morse(n):
    h = np.zeros(n)

        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
```

```python
def lbfgs_auglag(h, n, beta, maxiter=300, mu=100.0, n_outer=3):
    """L-BFGS-B with augmented Lagrangian for the equality constraint."""
    h0 = project_feasible(h, n).copy()
    target = n / 2.0
    lam = 0.0
    
    for outer in range(n_outer):
        def obj(x):
            sm, _ = smooth_max_and_grad(x, n, beta)
            r = np.sum(x) - target
            return sm + lam * r + mu / 2 * r * r
        
        def grad(x):
            sm, g = smooth_max_and_grad(x, n, beta)
            r = np.sum(x) - target
            g = g + (lam + mu * r) * np.ones(n)
            return g
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter // n_outer,
                                               'gtol': 1e-8, 'ftol': 1e-14})
        h0 = res.x
        r = np.sum(h0) - target
        lam += mu * r
        if abs(r) < 1e-8:
            break
    
    return project_feasible(h0, n)
```
