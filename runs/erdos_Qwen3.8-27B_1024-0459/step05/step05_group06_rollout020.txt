        base_diff = best_corr - 0.0  # we'll parameterize as delta

        # h[i] -> t_i + delta, h[j] -> t_j - delta
        # corr_new[k] = best_corr[k] + delta * (alpha_i[k] - alpha_j[k])
        # Constraint: 0 <= t_i + delta <= 1, 0 <= t_j - delta <= 1
        # So delta in [max(-t_i, t_j-1), min(1-t_i, t_j)]

        d_lo = max(-t_i, t_j - 1.0)
        d_hi = min(1.0 - t_i, t_j)
        if d_hi - d_lo < 1e-12:
            continue

        # Minimize max_k (best_corr[k] + delta * alpha_diff[k]) over delta in [d_lo, d_hi]
        # Ternary search
        lo, hi = d_lo, d_hi
        for _ in range(45):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(best_corr + m1 * alpha_diff)
            f2 = np.max(best_corr + m2 * alpha_diff)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        d_opt = (lo + hi) / 2.0
        new_c5 = np.max(best_corr + d_opt * alpha_diff) * dx

        if new_c5 < best_c5 - 1e-14:
            best[i] = t_i + d_opt
            best[j] = t_j - d_opt
            best = np.clip(best, 0.0, 1.0)
            best_corr = compute_corr(best)
            best_c5 = compute_c5(best)

    best = project_feasible(best, n)
    best_c5 = compute_c5(best)
    return best, best_c5

def smooth_value_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _obj_fn(x):
    return smooth_value_grad(x, _N_OPT, _BETA_OPT)[0]

def _grad_fn(x):
    return smooth_value_grad(x, _N_OPT, _BETA_OPT)[1]

def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _con_jac(x):
    return np.ones(_N_OPT)

def slsqp_optimize(h, n, beta, maxiter=300):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-13:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def simulated_annealing_coord(h, n, n_iters=500, seed=42):
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    T_init = 0.005
    T_end = 0.0001
    n_accepted = 0

    for it in range(n_iters):
        if remaining_time() < 3:
            break
        T = T_init * (T_end / T_init) ** (it / max(1, n_iters - 1))
        i = rng.randint(n)
        t_old = best[i]
        alpha = compute_alpha_for_coord(i, best, n)
        base = best_corr - t_old * alpha

        # Sample a random candidate
        t_new = np.clip(t_old + rng.normal(0, 0.1 + 0.3 * T), 0.0, 1.0)
        if abs(t_new - t_old) < 1e-8:
            continue

        new_c5 = np.max(base + t_new * alpha) * dx
        delta = new_c5 - best_c5

        if delta < 0 or rng.uniform() < np.exp(-delta / (T * 0.01)):
            best[i] = t_new
            best_corr = base + t_new * alpha
            if new_c5 < best_c5:
                best_c5 = new_c5
                n_accepted += 1

    best = project_feasible(best, n)
    best_c5 = compute_c5(best)
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_tmp = np.zeros(best_n)
        idx = np.random.RandomState(0).choice(best_n, best_n // 2, replace=False)
        h_tmp[idx] = 1.0
        best_h = project_feasible(h_tmp, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Intensive fast coordinate descent at parent resolution
    n1 = best_n
    h1 = best_h.copy()
    h1, c5_1 = fast_coord_descent(h1, n1, n_sweeps=200, seed=seed)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Pair swap descent
    if remaining_time() > 100:
        h2 = best_h.copy()
        h2, c5_2 = pair_swap_descent(h2, best_n, n_pairs=500, seed=seed + 1)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: SLSQP polish
    if remaining_time() > 80:
        h3 = slsqp_optimize(best_h, best_n, 5000, 300)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        # Fast coord after SLSQP
        h3, c5_3b = fast_coord_descent(h3, best_n, n_sweeps=50, seed=seed + 2)
        if c5_3b < best_c5:
            best_c5 = c5_3b
            best_h = h3.copy()

    # Phase 4: Simulated annealing_coord
    if remaining_time() > 100:
        h4 = simulated_annealing_coord(best_h, best_n, n_iters=800, seed=seed + 3)
        h4, c5_4 = fast_coord_descent(h4, best_n, n_sweeps=50, seed=seed + 4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Upscale to higher resolution
    target_resolutions = []
    for mult in [1.5, 2.0, 2.5]:
        rn = int(best_n * mult)
        if rn % 2 != 0:
            rn += 1
        if best_n < rn <= 3200:
            target_resolutions.append(rn)
    # Add some absolute values
    for rn in [1280, 1920, 2560]:
        if rn > best_n and rn <= 3200 and rn % 2 == 0:
            if rn not in target_resolutions:
                target_resolutions.append(rn)
    target_resolutions = sorted(set(target_resolutions))

    for rn in target_resolutions:
        if remaining_time() < 80:
            break
        h_r = interp_to(best_h, best_n, rn)
        h_r = project_feasible(h_r, rn)
        c5_r = compute_c5(h_r)

        # Coordinate descent at higher resolution
        h_r, c5_r = fast_coord_descent(h_r, rn, n_sweeps=80, seed=seed + 10)

        # Pair swaps
        if remaining_time() > 40:
            h_r, c5_r = pair_swap_descent(h_r, rn, n_pairs=300, seed=seed + 20)

        # SLSQP
        if remaining_time() > 40:
            h_r = slsqp_optimize(h_r, rn, 3000, 200)
            c5_r = compute_c5(h_r)
            h_r, c5_r = fast_coord_descent(h_r, rn, n_sweeps=50, seed=seed + 30)

        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = rn

    # Phase 6: Random restarts around best solution
    n = best_n
    for restart in range(15):
        if remaining_time() < 30:
            break
        h_p = best_h.copy()
        # Perturb a random subset
        n_pert = max(10, n // 16)
        idx = rng.choice(n, n_pert, replace=False)
        h_p[idx] += rng.uniform(-0.2, 0.2, n_pert)
        h_p = project_feasible(h_p, n)
        
        c5_p = compute_c5(h_p)
        # Only pursue if reasonably close
        if c5_p < best_c5 + 0.01:
            h_p, c5_p = fast_coord_descent(h_p, n, n_sweeps=40, seed=seed + 100 + restart)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 7: 0-1 discretization attempt
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, best_n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Also try: keep fractional values but push towards 0/1
        h_pushed = best_h.copy()
        h_pushed = np.where(h_pushed > 0.7, 1.0, h_pushed)
        h_pushed = np.where(h_pushed < 0.3, 0.0, h_pushed)
        h_pushed = project_feasible(h_pushed, best_n)
        c5_pushed = compute_c5(h_pushed)
        if c5_pushed < best_c5:
            best_c5 = c5_pushed
            best_h = h_pushed.copy()

    # Phase 8: More aggressive coordinate descent with different orderings
    if remaining_time() > 40:
        h8 = best_h.copy()
        h8, c5_8 = fast_coord_descent(h8, best_n, n_sweeps=100, seed=seed + 500)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Phase 9: Final pair swaps + coord descent
    if remaining_time() > 25:
        h9 = best_h.copy()
        h9, c5_9 = pair_swap_descent(h9, best_n, n_pairs=300, seed=seed + 600)
        h9, c5_9 = fast_coord_descent(h9, best_n, n_sweeps=50, seed=seed + 700)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    # Phase 10: Final SLSQP with very high beta
    if remaining_time() > 20:
        h10 = slsqp_optimize(best_h, best_n, 100000, 200)
        c5_10 = compute_c5(h10)
        h10, c5_10 = fast_coord_descent(h10, best_n, n_sweeps=30, seed=seed + 800)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    # Phase 11: Very fine coordinate descent at the end
    if remaining_time() > 10:
        h11, c5_11 = fast_coord_descent(best_h, best_n, n_sweeps=60, seed=seed + 900)
        if c5_11 < best_c5:
            best_c5 = c5_11
            best_h = h11.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1.0 - 1e-14)
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
