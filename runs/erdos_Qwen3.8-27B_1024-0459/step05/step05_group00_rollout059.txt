import numpy as np
from scipy.optimize import minimize, linprog
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * 2.0 / n


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_k(h, k, n):
    g = np.zeros(n)
    d = k - (n - 1)
    if d >= 0:
        m = n - d
        g[d:d + m] += 1.0 - h[:m]
        g[:m] -= h[d:d + m]
    else:
        a = -d
        m = n - a
        g[:m] += 1.0 - h[a:a + m]
        g[a:a + m] -= h[:m]
    return g


def slsqp_solver(h, n, K, beta, iters=500):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]

    def f(x):
        v = np.empty(K)
        for i, k in enumerate(idx):
            d = k - (n - 1)
            if d >= 0:
                v[i] = np.dot(x[d:], 1.0 - x[:n - d])
            else:
                a = -d
                v[i] = np.dot(x[:n - a], 1.0 - x[a:])
        m = v.max()
        s = np.exp(np.clip(beta * (v - m), -80.0, 80.0))
        return (m + np.log(s.sum()) / beta) * (2.0 / n)

    def df(x):
        v = np.empty(K)
        for i, k in enumerate(idx):
            d = k - (n - 1)
            if d >= 0:
                v[i] = np.dot(x[d:], 1.0 - x[:n - d])
            else:
                a = -d
                v[i] = np.dot(x[:n - a], 1.0 - x[a:])
        m = v.max()
        s = np.exp(np.clip(beta * (v - m), -80.0, 80.0))
        w = s / s.sum()
        g = np.zeros(n)
        for i in range(K):
            g += w[i] * grad_k(x, idx[i], n)
        return g * (2.0 / n)

    try:
        r = minimize(f, h, jac=df, method='SLSQP',
                     bounds=[(0, 1)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': iters, 'ftol': 1e-18, 'iprint': 0})
        if r.x is not None:
            return project_feasible(r.x, n)
    except Exception:
        pass
    return h


def lp_direction(h, n, K=15):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = []
    b_ub = []
    for k in idx:
        g = grad_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub.append(row)
        b_ub.append(float(np.dot(g, h) - corr[k]))

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]

    try:
        r = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                    A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if r.success:
            return r.x[:n]
    except Exception:
        pass
    try:
        r = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                    A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='revised simplex')
        if r.success:
            return r.x[:n]
    except Exception:
        pass
    return None


def interp_up(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src_h)


def subgradient_step(h, n, K, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    c_max = c_top.max()
    cl = np.clip(beta * (c_top - c_max), -60.0, 60.0)
    w = np.exp(cl)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(K):
        if w[i] > 1e-14:
            g += w[i] * grad_k(h, idx[i], n)
    return g
