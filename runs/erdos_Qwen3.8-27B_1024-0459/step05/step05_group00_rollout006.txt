def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_k(h, k, n):
    """Gradient of corr[k] with respect to h."""
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:n] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:n]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:n])
        g[a:n] -= h[:npairs]
    return g

def corr_k(h, k, n):
    """Compute correlation at shift k."""
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        return np.dot(h[:npairs], 1.0 - h[offset:offset+npairs])
    else:
        a = -offset
        npairs = n - a
        return np.dot(h[a:a+npairs], 1.0 - h[:npairs])

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_ACTIVE_K = []
_N_ACT = 0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k, n):
    g = np.zeros(n)
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs > 0:
            g[offset:offset + npairs] += (1.0 - h[:npairs])
            g[:npairs] -= h[offset:offset + npairs]
    else:
        a = -offset
        npairs = n - a
        if npairs > 0:
            g[:npairs] += (1.0 - h[a:a + npairs])
            g[a:a + npairs] -= h[:npairs]
    return g


def corr_k(h, k, n):
    offset = k - (n - 1)
    if offset >= 0:
        npairs = n - offset
        if npairs <= 0:
            return 0.0
        return float(np.dot(h[:npairs], 1.0 - h[offset:offset + npairs]))
    else:
        a = -offset
        npairs = n - a
        if npairs <= 0:
            return 0.0
        return float(np.dot(h[a:a + npairs], 1.0 - h[:npairs]))


def _obj_t(x):
    return x[-1]


def _obj_t_grad(x):
    g = np.zeros(len(x))
    g[-1] = 1.0
    return g


def _con_fun(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    vals = np.zeros(_N_ACT)
    for j in range(_N_ACT):
        k = _ACTIVE_K[j]
        vals[j] = t - corr_k(h, k, n)
    return vals


def _con_jac(x):
    h = x[:-1]
    n = len(h)
    J = np.zeros((_N_ACT, n + 1))
    for j in range(_N_ACT):
        k = _ACTIVE_K[j]
        gk = grad_corr_k(h, k, n)
        J[j, :n] = -gk
        J[j, -1] = 1.0
    return J


def _eq_fun(x):
    n = len(x) - 1
    return np.sum(x[:n]) - n / 2.0


def _eq_jac(x):
    n = len(x) - 1
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g

    h = project_feasible(h, n).copy()

        _N_ACT = len(active_set)

        x0 = np.append(h.copy(), c_max + 1e-6)
        bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
        cons = [{'type': 'ineq', 'fun': _con_fun, 'jac': _con_jac},
                {'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        try:
            res = minimize(_obj_t, x0, jac=_obj_t_grad, method='SLSQP',

def smooth_max_slsqp(h, n, beta, maxiter=400):
    dx = 2.0 / n

    def sm_obj(x):
        c = compute_corr(x)
        m = np.max(c)
        sh = beta * (c - m)
        sh = np.clip(sh, -50.0, 50.0)
        ex = np.exp(sh)
        s = np.sum(ex)
        if s < 1e-300:
            return m * dx
        return (m + np.log(s) / beta) * dx

    def sm_grad(x):
        c = compute_corr(x)
        m = np.max(c)
        sh = beta * (c - m)
        sh = np.clip(sh, -50.0, 50.0)
        ex = np.exp(sh)
        s = np.sum(ex)
        if s < 1e-300:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        else:
            w = ex / s
        g = np.zeros(n)
        for k in range(len(c)):
            wk = w[k]
            if wk < 1e-14:
                continue
            gk = grad_corr_k(x, k, n)
            g += wk * gk
        return g * dx

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    h0 = h.copy()
    try:
        res = minimize(sm_obj, h0, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h0) + 1e-10:
                return h_new
    except Exception:
        pass
    return h
