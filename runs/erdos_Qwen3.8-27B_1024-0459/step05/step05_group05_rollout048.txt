def projected_grad_descent(h, n, beta, n_iter=500, lr_init=0.015, patience=40):
    """Projected gradient descent with LSE smoothing."""
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0

    for it in range(n_iter):
        if remaining() < 5:
            break
        if no_improve > patience:
            break

        _, g = lse_obj_grad(h, n, beta)
        g -= np.mean(g)

        h_new = project_feasible(h - lr * g, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            no_improve = 0
            lr = min(lr * 1.03, 0.04)
        else:
            no_improve += 1
            lr *= 0.93
            if lr < 1e-8:
                break

    return best_h, best_c5


def lp_subgradient_step(h, n, top_k=5):
    """LP-based step: compute subgradient from top correlations, take LP step."""
    corr = compute_corr(h)
    top_idx = np.argsort(-corr)[:top_k]
    m = corr[top_idx[0]]
    # Weight by proximity to max
    weights = np.zeros(2 * n - 1)
    for idx in top_idx:
        weights[idx] = np.exp(-(corr[idx] - m) * 50.0)
    weights /= np.sum(weights)

    w = weights * (2.0 / n)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            if np_ > 0:
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            if np_ > 0:
                g[:np_] += wk * (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= wk * h[:np_]

    # LP step: solve min sum(g_j * h_j) s.t. 0<=h_j<=1, sum(h)=n/2
    # Optimal: h_j=1 where g_j smallest, h_j=0 where g_j largest
    order = np.argsort(g)
    h_lp = np.zeros(n)
    rem = n / 2.0
    for i in order:
        if rem >= 1.0 - 1e-12:
            h_lp[i] = 1.0
            rem -= 1.0
        elif rem > 1e-12:
            h_lp[i] = rem
            rem = 0.0
            break

    # Blend: take a fraction of the LP step
    alpha = 0.3
    h_new = project_feasible((1.0 - alpha) * h + alpha * h_lp, n)
    return h_new


def binary_sa(h_bin, n, rng, time_limit=60, top_k=5):
    """Simulated annealing on binary vectors."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - n_half]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:n_half - n_ones]]] = 1

    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    T = 0.5
    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > 3000:
            break

        corr = compute_corr(h)
        max_val = np.max(corr)
        top_idx = np.argsort(-corr)[:top_k]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_swap = None

        # Random swaps focused on hot regions
        n_try = 10
        for _ in range(n_try):
            # Pick from hot region
            k_hot = top_idx[rng.randint(len(top_idx))]
            off = k_hot - (n - 1)
            if off >= 0:
                np_ = n - off
                faults_10 = np.where((h[:np_] == 1) & (h[off:off + np_] == 0))[0]
                faults_01 = np.where((h[:np_] == 0) & (h[off:off + np_] == 1))[0]
            else:
                ao = -off
                np_ = n - ao
                faults_10 = np.where((h[ao:ao + np_] == 1) & (h[:np_] == 0))[0]
                faults_01 = np.where((h[ao:ao + np_] == 0) & (h[:np_] == 1))[0]

            if len(faults_10) > 0:
                i = faults_10[rng.randint(len(faults_10))]
                if off >= 0:
                    i_abs = i
                else:
                    i_abs = i + ao
            else:
                i_abs = ones[rng.randint(len(ones))]

            j = zeros[rng.randint(len(zeros))]
            if j == i_abs:
                continue

            h[i_abs] = 0
            h[j] = 1
            nc = np.max(compute_corr(h))
            delta = nc - max_val
            h[i_abs] = 1
            h[j] = 0

            if delta < best_delta - 1e-14 or (delta < 1e-10 and rng.rand() < np.exp(-delta / T)):
                best_delta = delta
                best_swap = (i_abs, j)

        # 2-swap: swap two 1s with two 0s
        if len(ones) >= 2 and len(zeros) >= 2:
            for _ in range(4):
                i1, i2 = ones[rng.randint(len(ones))], ones[rng.randint(len(ones))]
                j1, j2 = zeros[rng.randint(len(zeros))], zeros[rng.randint(len(zeros))]
                if len(set([i1, i2, j1, j2])) < 4:
                    continue
                h[i1] = 0; h[i2] = 0
                h[j1] = 1; h[j2] = 1
                nc = np.max(compute_corr(h))
                delta = nc - max_val
                h[i1] = 1; h[i2] = 1
                h[j1] = 0; h[j2] = 0

                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_swap = (i1, i2, j1, j2)

        if best_swap is not None:
            if len(best_swap) == 4:
                i1, i2, j1, j2 = best_swap
                h[i1] = 0; h[i2] = 0
                h[j1] = 1; h[j2] = 1
            else:
                i, j = best_swap
                h[i] = 0; h[j] = 1
            no_improve = 0
            cur_max = float(np.max(compute_corr(h)))
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

        T *= 0.9997
        if T < 0.01:
            T = 0.01

    return best_h, best_max * dx


def slsqp_phase(h, n, beta, maxiter=300):
    """SLSQP with LSE smoothing."""
    dx = 2.0 / n

    def obj_fn(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50, 50)
        ev = np.exp(shifted)
        return (m + np.log(np.sum(ev)) / beta) * dx

    def grad_fn(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50, 50)
        ev = np.exp(shifted)
        w = ev / np.sum(ev) * dx
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-18:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    g[off:off + np_] += wk * (1.0 - x[:np_])
                    g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ > 0:
                    g[:np_] += wk * (1.0 - x[ao:ao + np_])
                    g[ao:ao + np_] -= wk * x[:np_]
        return g

    def constr(x):
        return np.sum(x) - n / 2.0

    def constr_jac(x):
        return np.ones(n)

    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr, 'jac': constr_jac}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binary_local_search_intensive(h_bin, n, rng, time_limit=60, top_k=5):
    """More intensive binary local search."""
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - n_half]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:n_half - n_ones]]] = 1

    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit and remaining() > 5:
        no_improve += 1
        if no_improve > 8000:
            break

        corr = compute_corr(h)
        max_val = np.max(corr)
        top_idx = np.argsort(-corr)[:top_k]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_swap = None
        n_candidates = 16

        for _ in range(n_candidates):
            k_sel = top_idx[rng.randint(len(top_idx))]
            off = k_sel - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    faults = np.where((h[:np_] == 1) & (h[off:off + np_] == 0))[0]
                else:
                    faults = np.array([], dtype=int)
            else:
                ao = -off
                np_ = n - ao
                if np_ > 0:
                    faults = np.where((h[ao:ao + np_] == 1) & (h[:np_] == 0))[0]
                else:
                    faults = np.array([], dtype=int)

            if len(faults) > 0:
                fi = faults[rng.randint(len(faults))]
                if off >= 0:
                    i = fi
                else:
                    i = fi + ao
            else:
                i = ones[rng.randint(len(ones))]

            j = zeros[rng.randint(len(zeros))]
            if j == i:
                continue

            h[i] = 0
            h[j] = 1
            nc = np.max(compute_corr(h))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0

            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            cur_max += best_delta
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

    return best_h, best_max * dx


def top_shifts_gradient(h, n, top_k=5, n_iter=300, lr_init=0.02):
    """Focus gradient on top-k correlations to equalize them."""
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0

    for it in range(n_iter):
        if remaining() < 5:
            break
        if no_improve > 30:
            break

        corr = compute_corr(h)
        top_idx = np.argsort(-corr)[:top_k]
        m = corr[top_idx[0]]

        # Equal-weight gradient of top-k correlations
        g = np.zeros(n)
        for k_idx in top_idx:
            off = k_idx - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ > 0:
                    g[off:off + np_] += (1.0 - h[:np_])
                    g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ > 0:
                    g[:np_] += (1.0 - h[ao:ao + np_])
                    g[ao:ao + np_] -= h[:np_]
        g /= top_k
        g *= dx
        g -= np.mean(g)

        h_new = project_feasible(h - lr * g, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            lr *= 0.92
            if lr < 1e-7:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP to get into good neighborhood
    for beta in [100, 500, 2000, 8000, 30000]:
        if remaining() < 10:
            break
        h_new = slsqp_phase(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Projected gradient descent with increasing beta (main workhorse)
    for beta in [500, 2000, 8000, 30000, 100000]:
        if remaining() < 10:
            break
        h_new, c5_new = projected_grad_descent(h, n, beta, n_iter=400, lr_init=0.012, patience=30)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Top-shifts equalization
    for tk in [3, 5, 8, 10, 15]:
        if remaining() < 10:
            break
        h_eq, c5_eq = top_shifts_gradient(h, n, top_k=tk, n_iter=400, lr_init=0.015)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: LP subgradient steps
    if remaining() > 20:
        h_lp = h.copy()
        best_c5_lp = compute_c5(h_lp)
        for it in range(200):
            if remaining() < 5:
                break
            h_new_lp = lp_subgradient_step(h_lp, n, top_k=5)
            c5_lp = compute_c5(h_new_lp)
            if c5_lp < best_c5_lp - 1e-15:
                best_c5_lp = c5_lp
                h_lp = h_new_lp.copy()
            else:
                h_lp = project_feasible(0.8 * h_lp + 0.2 * h_new_lp, n)
                c5_lp = compute_c5(h_lp)
                if c5_lp < best_c5_lp:
                    best_c5_lp = c5_lp
        if best_c5_lp < best_c5:
            best_c5 = best_c5_lp
            best_h = h_lp.copy()
            h = h_lp.copy()

    # Phase 5: Binary SA + continuous refinement
    if remaining() > 40:
        h_bin = (h > 0.5).astype(float)
        n_avail = int(np.sum(h_bin))
        target_n = n // 2
        if n_avail != target_n:
            if n_avail > target_n:
                idx = np.argsort(-h)[:n_avail - target_n]
                h_bin[idx] = 0
            else:
                idx = np.argsort(h)[:target_n - n_avail]
                h_bin[idx] = 1

        sa_time = min(60, max(15, int(remaining() * 0.25)))
        h_b, c5_b = binary_sa(h_bin, n, rng, time_limit=sa_time, top_k=5)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

        # Continuous refinement from binary
        h_cont = slsqp_phase(h_b, n, 5000, maxiter=300)
        h_cont, c5_cont = projected_grad_descent(h_cont, n, 20000, n_iter=300, lr_init=0.01)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
            h = h_cont.copy()

    # Phase 6: More intensive projected gradient at high beta
    for beta in [50000, 200000]:
        if remaining() < 15:
            break
        h_new, c5_new = projected_grad_descent(h, n, beta, n_iter=600, lr_init=0.008, patience=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Multi-resolution exploration
    if remaining() > 60:
        for delta in [-2, 2, -4, 4, -6, 6]:
            if remaining() < 20:
                break
            n_t = best_n + delta
            if n_t < 64 or n_t > 600 or n_t % 2 != 0:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t
            h_t = slsqp_phase(h_t, n_t, 5000, maxiter=200)
            h_t, c5_t = projected_grad_descent(h_t, n_t, 20000, n_iter=200, lr_init=0.01)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t
                h = h_t.copy()

    # Phase 8: Diverse starts
    if remaining() > 50:
        # Thue-Morse
        h_tm = np.zeros(best_n)
        for i in range(best_n):
            h_tm[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
        n_ones = int(np.sum(h_tm))
        n_half = best_n // 2
        if n_ones != n_half:
            if n_ones > n_half:
                oi = np.where(h_tm == 1)[0]
                h_tm[oi[:n_ones - n_half]] = 0
            else:
                zi = np.where(h_tm == 0)[0]
                h_tm[zi[:n_half - n_ones]] = 1
        h_tm = slsqp_phase(h_tm, best_n, 3000, maxiter=200)
        h_tm, c5_tm = projected_grad_descent(h_tm, best_n, 10000, n_iter=300)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            h = h_tm.copy()

        # Random start
        if remaining() > 30:
            h_r = rng.uniform(0, 1, best_n)
            h_r = project_feasible(h_r, best_n)
            h_r = slsqp_phase(h_r, best_n, 2000, maxiter=200)
            h_r, c5_r = projected_grad_descent(h_r, best_n, 8000, n_iter=300)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
                h = h_r.copy()

    # Phase 9: Final intensive refinement
    n = best_n
    h = best_h.copy()

    for beta in [10000, 50000, 200000, 500000]:
        if remaining() < 8:
            break
        h_new, c5_new = projected_grad_descent(h, n, beta, n_iter=500, lr_init=0.005, patience=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    for tk in [3, 5, 8]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_gradient(h, n, top_k=tk, n_iter=300, lr_init=0.008)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 10: Final binary search + refine
    if remaining() > 25:
        h_bin = (h > 0.5).astype(float)
        h_b, c5_b = binary_local_search_intensive(h_bin, n, rng, time_limit=min(40, int(remaining() * 0.5)), top_k=5)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Quick continuous from binary
        if remaining() > 10:
            h_c = slsqp_phase(h_b, n, 10000, maxiter=200)
            h_c, c5_c = projected_grad_descent(h_c, n, 50000, n_iter=200, lr_init=0.005)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Final LP polish
    if remaining() > 8:
        h_lp = best_h.copy()
        for it in range(50):
            if remaining() < 3:
                break
            h_new_lp = lp_subgradient_step(h_lp, n, top_k=3)
            c5_lp = compute_c5(h_new_lp)
            if c5_lp < compute_c5(h_lp):
                h_lp = h_new_lp.copy()
        c5_final_lp = compute_c5(h_lp)
        if c5_final_lp < best_c5:
            best_c5 = c5_final_lp
            best_h = h_lp.copy()

    # Ensure feasibility and final validation
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
