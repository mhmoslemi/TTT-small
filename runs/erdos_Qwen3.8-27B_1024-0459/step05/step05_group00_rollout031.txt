            # d/dh[j] = (1-h[j+d]) for j in [0, n-1-d] (j as first factor)
            #          - h[j-d] for j in [d, n-1] (j as second factor)
            g[:n-d] += (1.0 - h[d:n])
            g[d:] -= h[:n-d]
    else:
        s = -d
        if s < n:
            # corr = sum_{i=s}^{n-1} h[i]*(1-h[i-s])
            # d/dh[j] = (1-h[j-s]) for j in [s, n-1] (j as first factor)
            #          - h[j+s] for j in [0, n-1-s] (j as second factor)
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    return g

def smooth_max_slsqp(h, n, beta, maxiter=300):
    def objective(x):
        corr = get_corr(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(ex.sum()) / beta) * (2.0 / n)
    
    def gradient(x):
        corr = get_corr(x)
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / ex.sum()
        wc = w * (2.0 / n)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if wc[k] < 1e-12:
                continue
            d = k - (n - 1)
            g += wc[k] * grad_single_lag(x, d, n)
        return g
    
    def constraint_eq(x):
        return np.sum(x) - n / 2.0
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_jac}]
    
    try:
        res = minimize(objective, h, jac=gradient, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def subgradient_step(h, n, K=20, beta=50.0):
    corr = get_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    w = np.exp(beta * (c_top - c_top.max()))
    w /= w.sum()
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if w[i] > 1e-12:
            d = k - (n - 1)
            g += w[i] * grad_single_lag(h, d, n)
    return g, corr, idx

def coordinate_descent(h, n, K=15, max_passes=10000, time_limit=60.0):
    """Coordinate descent: for each coordinate j, find optimal δ with partner l."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    # Precompute active lags
    corr = get_corr(h)
    active_idx = np.argsort(-corr)[:K]
    
    passes = 0
    for p in range(max_passes):
        if time.time() - start_time > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 10:
            break
        
        improved = False
        # Update active set every 10 passes
        if p % 10 == 0:
            corr = get_corr(h)
            active_idx = np.argsort(-corr)[:K]
            active_d = active_idx - (n - 1)
            
        # Cycle through coordinates
        for j in range(n):
            if _BUDGET - (time.time() - _T0) < 10:
                return h, best_c5
            
            # Choose partner l
            # Strategy: l = (j + n//3) % n or l = (j - n//3) % n
            # Or random
            l = (j + n // 3) % n
            if l == j:
                l = (j + 1) % n
            
            s = h[j] + h[l]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            
            # Compute A_d = ∂corr_d/∂h[j] - ∂corr_d/∂h[l] for each active lag
            A = np.zeros(K)
            for a, k in enumerate(active_idx):
                d = k - (n - 1)
                if d == 0:
                    # corr_0 = sum_i h[i](1-h[i]) = n/2 - sum h[i]^2
                    # ∂/∂h[j] = 1 - 2*h[j]
                    A[a] = (1 - 2*h[j]) - (1 - 2*h[l])
                else:
                    g_j = 0.0
                    g_l = 0.0
                    if d >= 0:
                        if j + d < n:
                            g_j += (1.0 - h[j + d])
                        if j - d >= 0:
                            g_j -= h[j - d]
                        if l + d < n:
                            g_l += (1.0 - h[l + d])
                        if l - d >= 0:
                            g_l -= h[l - d]
                    else:
                        s2 = -d
                        if j >= s2:
                            g_j += (1.0 - h[j - s2])
                        if j + s2 < n:
                            g_j -= h[j + s2]
                        if l >= s2:
                            g_l += (1.0 - h[l - s2])
                        if l + s2 < n:
                            g_l -= h[l + s2]
                    A[a] = g_j - g_l
            
            # Solve: minimize max_a (corr[active_idx[a]] + A[a] * δ)
            # subject to lo - h[j] <= δ <= hi - h[j]
            delta_min = lo - h[j]
            delta_max = hi - h[j]
            
            # The optimal δ minimizes the maximum of K affine functions of δ
            # Strategy: try all pairs and boundaries
            best_delta = 0.0
            best_val = np.max(corr[active_idx] + A * 0.0)  # current max
            
            # Check boundaries
            for delta_try in [delta_min, delta_max]:
                val = np.max(corr[active_idx] + A * delta_try)
                if val < best_val:
                    best_val = val
                    best_delta = delta_try
            
            # Check intersections
            for i in range(K):
                for j2 in range(i + 1, K):
                    if abs(A[i] - A[j2]) < 1e-15:
                        continue
                    delta_star = (corr[active_idx[j2]] - corr[active_idx[i]]) / (A[i] - A[j2])
                    if delta_min - 1e-12 <= delta_star <= delta_max + 1e-12:
                        delta_star = np.clip(delta_star, delta_min, delta_max)
                        val = np.max(corr[active_idx] + A * delta_star)
                        if val < best_val:
                            best_val = val
                            best_delta = delta_star
            
            # Apply update
            if abs(best_delta) > 1e-12:
                h[j] += best_delta
                h[l] -= best_delta
                # Quick feasibility check
                if h[j] < -1e-10 or h[j] > 1 + 1e-10 or h[l] < -1e-10 or h[l] > 1 + 1e-10:
                    h[j] = np.clip(h[j], 0, 1)
                    h[l] = s - h[j]
                    h[l] = np.clip(h[l], 0, 1)
                improved = True
        
        passes += 1
        
        # Periodic evaluation
        if p % 50 == 0 or p == max_passes - 1:
            h = project_feasible(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
            # Reset active set
            corr = get_corr(h)
            active_idx = np.argsort(-corr)[:K]
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

Wait, I have a bug in the coordinate descent. The `active_d` variable is computed but not used. Also, the O(K^2) inner loop for intersections might be slow in Python. Let me optimize it.

For K=15, K^2/2 = 112.5 pairs. For n=512 coordinates, that's 512 * 112.5 ≈ 57K pair checks per pass. Each check involves computing max of K values, so total is 57K * 15 ≈ 856K operations per pass. In Python, this might be ~10-50ms per pass. So in 120 seconds, we get ~2400-12000 passes. That's reasonable.

But the A[d] computation is also O(K) per coordinate, so that's another 512 * 15 = 7.7K operations per pass (negligible).

The bottleneck is the O(K^2) intersection check. For K=15, that's 105 pairs. Let me see if I can vectorize it.

Actually, let me reconsider the approach. Instead of the O(K^2) exact 1D LP solve, I can use a simpler heuristic:
- Try a few candidate δ values (0, delta_min, delta_max, and a few in between)
- Pick the best one

This is O(K) per coordinate instead of O(K^2). Less optimal but much faster in Python.

Or better: I can vectorize the intersection computation. For all pairs (i, j), compute delta_star = (c_j - c_i) / (A_i - A_j), then filter valid ones and evaluate. This can be done with numpy.

Let me rewrite the coordinate descent with vectorization:
