def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            break
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:

            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -40.0, 40.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -40.0, 40.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g

                       bounds=bounds, constraints=cons,

def subgradient_max(h, n, n_iter=200):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.03
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 0.8
        top_idx = np.where(corr >= threshold)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]

        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(top_idx)
        g -= np.mean(g)

        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.92
            if step < 0.001:
                break
    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):

            best_c5 = c5_b
            best_h = h_b.copy()

        # 2-point swap search from best binary
        if remaining() > 60:
            h_b2, c5_b2 = binary_search_2pt(best_h, best_n, rng,
                                            min(300, int(remaining() * 0.3)))
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()

    # Phase 3: Resolution sweep with binary search
    if remaining() > 80:
        n_base = best_n
        n_candidates = [256, 320, 384, 448, 512, 576, 640, 704, 768, 832]
        n_candidates = [nt for nt in n_candidates if 128 <= nt <= 800 and nt % 2 == 0]
        # Sort by closeness to base
        n_candidates.sort(key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:7]:
            if remaining() < 25:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            h_t_bin = binarize(h_t, n_t)

            # Binary search at this resolution
            t_alloc = min(100, int(remaining() * 0.2))
            h_tb, c5_tb = binary_local_search_intensive(h_t_bin, n_t, rng, t_alloc)
            if c5_tb < best_c5:
                best_c5 = c5_tb
                best_h = h_tb.copy()
                best_n = n_t

            # Also try Thue-Morse at this resolution
            if remaining() > 20:
                h_tm = thue_morse(n_t, rng)
                h_tm_b, c5_tm = binary_local_search_intensive(h_tm, n_t, rng,
                                                              min(50, int(remaining() * 0.15)))
                if c5_tm < best_c5:
                    best_c5 = c5_tm
                    best_h = h_tm_b.copy()
                    best_n = n_t

    # Phase 4: Diverse binary starts at best resolution
    if remaining() > 40:
        n = best_n
        # Inverse Thue-Morse
        h_tm = thue_morse(n, rng)
        h_tm_inv = 1.0 - h_tm
        h_tm_inv = project_feasible(h_tm_inv, n)
        h_bi, c5_bi = binary_local_search_intensive(h_tm_inv, n, rng,
                                                    min(60, int(remaining() * 0.2)))
        if c5_bi < best_c5:
            best_c5 = c5_bi
            best_h = h_bi.copy()

        # Random binary starts
        for rs in range(3):
            if remaining() < 15:
                break
            h_r = (rng.rand(n) < 0.5).astype(float)
            h_r = project_feasible(h_r, n)
            h_rb, c5_rb = binary_local_search_intensive(h_r, n, rng,
                                                        min(40, int(remaining() * 0.2)))
            if c5_rb < best_c5:
                best_c5 = c5_rb
                best_h = h_rb.copy()

    # Phase 5: Continuous refinement from best binary
    if remaining() > 20:
        n = best_n
        h_c = project_feasible(best_h, n)
        for beta in [2000, 8000, 25000]:
            if remaining() < 8:
                break
            h_new = slsqp_smooth(h_c, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_c = h_new.copy()
            else:
                break

    # Phase 6: Subgradient polish
    if remaining() > 15:
        h_sg, c5_sg = subgradient_max(best_h, best_n, n_iter=150)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 7: Final intensive binary search
    if remaining() > 30:
        h_bin_final = binarize(best_h, best_n)
        t_final = min(200, int(remaining() * 0.5))
        h_fb, c5_fb = binary_local_search_intensive(h_bin_final, best_n, rng, t_final)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()

        # 2-point swap from this
        if remaining() > 20:
            h_fb2, c5_fb2 = binary_search_2pt(best_h, best_n, rng,
                                              min(100, int(remaining() * 0.4)))
            if c5_fb2 < best_c5:
                best_c5 = c5_fb2
                best_h = h_fb2.copy()

    # Phase 8: Try a few more resolutions with focused search
    if remaining() > 40:
        n_best = best_n
        for delta in [-32, -16, -8, 8, 16, 32, 64]:
            if remaining() < 15:
                break
            n_t = n_best + delta
            if n_t < 128 or n_t > 800 or n_t % 2 != 0:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            h_t_bin = binarize(h_t, n_t)
            h_tb, c5_tb = binary_local_search_intensive(h_t_bin, n_t, rng,
                                                        min(60, int(remaining() * 0.25)))
            if c5_tb < best_c5:
                best_c5 = c5_tb
                best_h = h_tb.copy()
                best_n = n_t

    # Final continuous refinement
    if remaining() > 10:
        h_final_c = slsqp_smooth(best_h, best_n, 50000, maxiter=300)
        c5_final_c = compute_c5(h_final_c)
        if c5_final_c < best_c5:
            best_c5 = c5_final_c
            best_h = h_final_c.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
