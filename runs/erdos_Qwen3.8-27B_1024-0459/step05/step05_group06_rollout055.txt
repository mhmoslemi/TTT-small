    wsum = np.sum(exp_vals)
    weights_top = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    # Build full weight vector
    weights = np.zeros(ncorr)
    weights[top_idx] = weights_top
    
    # Compute gradient
    grad = np.zeros(n)
    active_idx = top_idx[weights[top_idx] > 1e-12]
    for k_idx in active_idx:
        w = weights[k_idx]
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _obj_fn(x):
    return topk_smooth_grad(x, _N_OPT, _BETA_OPT, _TOPK_OPT)[0]

def _grad_fn(x):
    return topk_smooth_grad(x, _N_OPT, _BETA_OPT, _TOPK_OPT)[1]

def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _con_jac(x):
    return np.ones(_N_OPT)

def slsqp_optimize(h, n, beta, maxiter=400, topk=0):
    global _N_OPT, _BETA_OPT, _TOPK_OPT
    _N_OPT = n
    _BETA_OPT = beta
    _TOPK_OPT = topk
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-13:
                return trial
    except Exception:
        pass
    return h.copy()

def coord_descent_fast(h, n, n_sweeps=30):
    """Faster coordinate descent using vectorized operations."""
    best = h.copy()
    best_c5 = compute_c5(best)
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        improved_any = False
        # Focus on interior points first, then all
        order = np.zeros(n, dtype=int)
        interior = (best > 1e-10) & (best < 1.0 - 1e-10)
        order[interior] = 0
        order[~interior] = 1
        nporder = np.argsort(order, kind='stable')
        
        for i in nporder:
            if remaining_time() < 2:
                break
            t0 = best[i]
            # Precompute linear and quadratic coefficients for corr changes
            # corr[k] = sum_i' h[i'](1-h[i'+k']) 
            # When we change h[i] from t0 to t, the change is:
            # For offset 0: h[i](1-h[i]) -> t(1-t), change = t(1-t) - t0(1-t0)
            # For offset > 0 where i < n-offset: h[i](1-h[i+offset]) -> t(1-h[i+offset])
            #   change = (t - t0)(1-h[i+offset])
            # For offset > 0 where i-offset >= 0: h[i-offset](1-h[i]) -> h[i-offset](1-t)
            #   change = (t0 - t)h[i-offset]
            # For offset < 0, symmetric
            
            a = np.zeros(ncorr)  # linear coefficient in t
            b = np.zeros(ncorr)  # quadratic coefficient in t (only for offset 0)
            
            # offset 0 (k = n-1)
            a[n-1] = 1.0  # coefficient of t
            b[n-1] = -1.0  # coefficient of t^2
            
            # For positive offsets
            for off in range(1, n):
                k_pos = n - 1 + off  # k index for positive offset
                k_neg = n - 1 - off  # k index for negative offset
                if i + off < n:
                    # h[i] appears as first factor: h[i]*(1-h[i+off])
                    a[k_pos] += (1.0 - best[i + off])
                if i - off >= 0:
                    # h[i] appears as second factor (1-h[i]): h[i-off]*(1-h[i])
                    a[k_neg] += (-best[i - off])
            
            base = best_corr - a * t0 - b * t0 * t0
            
            # Evaluate at candidate points
            cands = np.array([0.0, 1.0, t0, 1.0 - t0,
                           max(0.0, t0 - 0.15), min(1.0, t0 + 0.15),
                           max(0.0, t0 * 0.5), min(1.0, t0 * 1.5)])
            cands = cands[np.abs(cands - t0) > 1e-12]
            
            best_t = t0
            best_f = np.max(base) * dx
            
            for t in cands:
                f = np.max(base + a * t + b * t * t) * dx
                if f < best_f - 1e-15:
                    best_f = f
                    best_t = t

            # Golden section if improvement found
            if best_t != t0:
                margin = 0.25
                lo = max(0.0, best_t - margin)
                hi = min(1.0, best_t + margin)
                if hi - lo > 1e-12:
                    gr = (np.sqrt(5.0) - 1.0) / 2.0
                    c1 = hi - gr * (hi - lo)
                    c2 = lo + gr * (hi - lo)
                    f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                    f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                    for _ in range(70):
                        if hi - lo < 1e-14:
                            break
                        if f1 < f2:
                            hi = c2; c2 = c1; f2 = f1
                            c1 = hi - gr * (hi - lo)
                            f1 = np.max(base + a * c1 + b * c1 * c1) * dx
                        else:
                            lo = c1; c1 = c2; f1 = f2
                            c2 = lo + gr * (hi - lo)
                            f2 = np.max(base + a * c2 + b * c2 * c2) * dx
                    t_opt = (lo + hi) / 2.0
                    f_opt = np.max(base + a * t_opt + b * t_opt * t_opt) * dx
                    if f_opt < best_f:
                        best_f = f_opt
                        best_t = t_opt

            if best_f < best_c5 - 1e-14:
                best[i] = best_t
                best_corr = compute_corr(best)
                best_c5 = compute_c5(best)
                improved_any = True

        if not improved_any:
            break

    return best, best_c5

def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

def remaining_time():

def topk_smooth_grad(x, n, beta, topk):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    ncorr = len(corr)
    if topk > 0 and topk < ncorr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(ncorr)
    ctop = corr[top_idx]
    m = np.max(ctop)
    shifted = np.minimum(beta * (ctop - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights_top = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    weights = np.zeros(ncorr)
    weights[top_idx] = weights_top
    grad = np.zeros(n)
    active = top_idx[weights_top > 1e-12]
    for k_idx in active:
        w = weights[k_idx]
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _grad_fn(x):

def slsqp_optimize(h, n, beta, maxiter=400, topk=0):

    _BETA_OPT = beta
    _TOPK_OPT = topk
