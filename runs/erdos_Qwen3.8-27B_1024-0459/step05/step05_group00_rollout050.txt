        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50, 50)

def coord_optimize(h, j, n, K=20):
    """Optimize h[j] by finding min of max_k (a_k*t + b_k) for t in [0,1]"""
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    
    # For each active shift, compute a_k and b_k
    t_old = h[j]
    a_vals = []
    b_vals = []
    for k in idx:
        off = k - (n - 1)
        # corr_k = sum_i h[i]*(1-h[i+off]) (approximately)
        # d(corr_k)/d(h[j]) = a_k
        # corr_k = a_k * h[j] + b_k
        g = grad_corr_offset(h, off, n)
        a_k = g[j]
        b_k = corr[k] - a_k * t_old
        a_vals.append(a_k)
        b_vals.append(b_k)
    
    a_vals = np.array(a_vals)
    b_vals = np.array(b_vals)
    
    # Find breakpoints: a_k1 * t + b_k1 = a_k2 * t + b_k2
    # t = (b_k2 - b_k1) / (a_k1 - a_k2)
    candidates = [0.0, 1.0]
    for i in range(len(a_vals)):
        for jj in range(i+1, len(a_vals)):
            da = a_vals[i] - a_vals[jj]
            if abs(da) > 1e-14:
                t_bp = (b_vals[jj] - b_vals[i]) / da
                if 0.0 <= t_bp <= 1.0:
                    candidates.append(t_bp)
    
    # Evaluate f(t) at all candidates
    best_t = t_old
    best_val = np.inf
    for t in candidates:
        f = np.max(a_vals * t + b_vals)
        if f < best_val:
            best_val = f
            best_t = t

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_offset(h, off, n):
    g = np.zeros(n)
    if off >= 0:
        npairs = n - off
        if npairs <= 0:
            return g
        g[off:] += (1.0 - h[:npairs])
        g[:npairs] -= h[off:]
    else:
        a = -off
        npairs = n - a
        if npairs <= 0:
            return g
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def top_k_gradient(h, n, K=20, beta=50.0):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]
    c_top = corr[idx]
    weights = np.exp(beta * (c_top - c_top.max()))
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(K) / K
    else:
        weights /= s
    g = np.zeros(n)
    for i, k in enumerate(idx):
        if weights[i] > 1e-12:
            g += weights[i] * grad_corr_offset(h, k - (n - 1), n)
    return g, np.max(corr)


def smooth_max_obj_and_grad(h, n, beta):
    corr = compute_corr(h)
    dx = 2.0 / n
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(shifted)
    sm_val = (m + np.log(np.sum(ex)) / beta) * dx
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-15:
            continue
        off = k - (n - 1)
        g += w[k] * grad_corr_offset(h, off, n)
    g *= dx
    return sm_val, g

def coord_descent_sweep(h, n, K=30, max_sweeps=50, time_limit=None):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - start > time_limit:
            break
        rem = _BUDGET - (time.time() - _T0)
        if rem < 5:
            break
        corr = compute_corr(h)
        idx_active = np.argsort(-corr)[:K]
        for j in range(n):
            if rem < 3:
                break
            t_old = h[j]
            a_vals = np.zeros(K)
            b_vals = np.zeros(K)
            for ii, k in enumerate(idx_active):
                off = k - (n - 1)
                g = grad_corr_offset(h, off, n)
                a_vals[ii] = g[j]
                b_vals[ii] = corr[k] - a_vals[ii] * t_old
            candidates = [0.0, 1.0, t_old]
            for ii in range(K):
                for jj in range(ii + 1, K):
                    da = a_vals[ii] - a_vals[jj]
                    if abs(da) > 1e-14:
                        t_bp = (b_vals[jj] - b_vals[ii]) / da
                        if -0.01 <= t_bp <= 1.01:
                            t_bp = max(0.0, min(1.0, t_bp))
                            candidates.append(t_bp)
            best_t = t_old

            best_c5 = c5_cur
        elif sweep > 10:
            change = best_c5 - c5_cur
            if change < -1e-12 and sweep > 5:
                break
    return h, best_c5


def lbfgs_smooth_max(h, n, beta, maxiter=300):
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(shifted)
        sm = (m + np.log(np.sum(ex)) / beta) * dx
        penalty = 500.0 * (np.sum(x) - target) ** 2
        return sm + penalty

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(shifted)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-15:
                continue
            off = k - (n - 1)
            g += w[k] * grad_corr_offset(x, off, n)
        g *= dx
        g += 1000.0 * (np.sum(x) - target)
        return g

    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h.copy()


def slsqp_smooth_max(h, n, beta, maxiter=300):
    dx = 2.0 / n

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(shifted)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(shifted)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-15:
                continue
            off = k - (n - 1)
            g += w[k] * grad_corr_offset(x, off, n)
        g *= dx
        return g

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return h.copy()


def equalize_top(h, n, K=8, max_iters=200):

        corr = compute_corr(h)
        idx = np.argsort(-corr)[:K]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-10:
            break
        g_avg = np.zeros(n)
        for k in idx:
            g_avg += grad_corr_offset(h, k - (n - 1), n)
        g_avg /= K
        g_dir = -g_avg
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        g_dir /= gn
        improved = False
        for step in [0.02, 0.04, 0.01, 0.03, 0.06, 0.005, 0.015, 0.05]:
            ht = project_feasible(h + step * g_dir, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-15:
                h = ht
                best_c5 = cn
                improved = True
                break

def random_perturb(h, n, n_pert=20, sigma=0.05, rng=None):

def interp_to(src_h, src_n, dst_n):

    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
