def project_feasible(h, n):

def make_swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)

def compute_corr(h):

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def make_swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)

def soft_max_obj_grad(h, n, beta, mu):
    """Compute soft-max objective and gradient with sum penalty."""
    corr = compute_corr(h)
    c_max = np.max(corr)
    sh = corr - c_max
    cl = np.clip(beta * sh, -500.0, 50.0)
    ex = np.exp(cl)
    s_ex = np.sum(ex)
    w = ex / s_ex
    obj = (c_max + np.log(s_ex) / beta) * (2.0 / n) + 0.5 * mu * (np.sum(h) - n/2.0)**2

    # Gradient
    g = np.zeros(n)
    wc = w * (2.0 / n)
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    g += mu * (np.sum(h) - n/2.0)
    return obj, g

def projected_grad(h, n, time_limit=200, beta=10000):

    best_h = h.copy()

        g_sum = np.sum(g)
        g_proj = g - g_sum / n

        # Backtracking line search
        improved = False
        for _ in range(30):

            c5_try = compute_c5(h_try)
            if c5_try < c5_curr - 1e-15:
                h = h_try
                best_h = h_try.copy()
                best_c5 = c5_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            alpha *= 0.3
            if alpha < 1e-10:
                break
            alpha = max(alpha, 1e-8)
        else:
            alpha = min(alpha * 1.1, 2.0 / (n * beta))

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0

    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

    sort_idx = np.argsort(-h)
