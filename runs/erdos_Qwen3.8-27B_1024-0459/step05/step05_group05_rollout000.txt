def thue_morse_seq(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rng.shuffle(ones_idx)
            h[ones_idx[:n_ones - n_half]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            rng.shuffle(zeros_idx)
            h[zeros_idx[:n_half - n_ones]] = 1
    return h

def corr_grad_at(h, offset, n):

def binary_sa_search(h_bin, n, rng, time_limit):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rng.shuffle(ones_idx)
            h[ones_idx[:n_ones - n_half]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            rng.shuffle(zeros_idx)
            h[zeros_idx[:n_half - n_ones]] = 1

    dx_v = 2.0 / n
    corr = compute_corr(h)
    cur_max = int(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_improve = 0
    T = 1.0
    T_min = 0.01

    while time.time() - start < time_limit and remaining() > 3:
        no_improve += 1
        if no_improve > 8000:
            break
        T = max(T_min, T * 0.99997)

        corr = compute_corr(h)
        max_val = int(np.max(corr))
        top_idx = np.argsort(-corr)[:5]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_move = None
        n_try = 20

        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = int(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta:
                best_delta = delta
                best_move = ((i, j), delta)

        # Targeted: fix worst correlation
        for _ in range(10):
            k_hot = top_idx[rng.randint(len(top_idx))]
            off = k_hot - (n - 1)
            if off >= 0:
                m = n - off
                faults = np.where((h[:m] == 1) & (h[off:] == 0))[0]
            else:
                ao = -off
                m = n - ao
                faults = np.where((h[ao:] == 1) & (h[:m] == 0))[0]
            if len(faults) == 0:
                continue
            fi = faults[rng.randint(len(faults))]
            if off >= 0:
                i = fi
            else:
                i = fi + ao
            if i < 0 or i >= n or h[i] != 1:
                continue
            j = zeros[rng.randint(len(zeros))]
            if j == i:
                continue
            h[i] = 0
            h[j] = 1
            nc = int(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta:
                best_delta = delta
                best_move = ((i, j), delta)

        # Pair swap for diversity
        if len(ones) >= 2 and len(zeros) >= 2:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 != i2 and j1 != j2 and i1 != j1 and i1 != j2 and i2 != j1 and i2 != j2:
                h[i1] = 0
                h[i2] = 0
                h[j1] = 1
                h[j2] = 1
                nc = int(np.max(compute_corr(h)))
                delta = nc - max_val
                h[i1] = 1
                h[i2] = 1
                h[j1] = 0
                h[j2] = 0
                if delta < best_delta:
                    best_delta = delta
                    best_move = (('pair', i1, i2, j1, j2), delta)

        if best_move is not None:
            accept = best_delta < 0
            if not accept and T > T_min:
                if rng.random() < np.exp(-best_delta / T):
                    accept = True
            if accept:
                mv = best_move[0]
                if mv == 'pair':
                    _, i1, i2, j1, j2 = mv
                    h[i1] = 0
                    h[i2] = 0
                    h[j1] = 1
                    h[j2] = 1
                else:
                    i, j = mv
                    h[i] = 0
                    h[j] = 1
                cur_max += best_move[1]
                no_improve = 0
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
        elif no_improve > 300:
            # Random perturbation to escape
            n_flip = rng.randint(2, 5)
            ones_now = np.where(h == 1)[0]
            zeros_now = np.where(h == 0)[0]
            for _ in range(n_flip):
                if len(ones_now) == 0 or len(zeros_now) == 0:
                    break
                i = ones_now[rng.randint(len(ones_now))]
                j = zeros_now[rng.randint(len(zeros_now))]
                h[i] = 0
                h[j] = 1
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
            T = min(1.0, T * 2.0)

    return best_h, best_max * dx_v


def corr_equalize(h, n, n_iter=500):
    h = project_feasible(h, n).copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.03

    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        top_k = 5
        idx = np.argsort(-corr)[:top_k]
        # Only consider correlations close to max
        thresh = corr[idx[-1]] - 2
        active = np.where(corr >= thresh)[0]

        g = np.zeros(n)
        for k in active:
            off = k - (n - 1)
            g += corr_grad_at(h, off, n)
        g /= len(active)
        g -= np.mean(g)

        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-15:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
            step = min(0.1, step * 1.05)
        else:
            step *= 0.85
            if step < 0.001:
                break

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse_seq(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP refinement (10% time)
    phase1_time = 0.10 * budget_s
    for beta in [200, 1000, 5000, 20000]:
        if remaining() < phase1_time * 0.2 + 3:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Intensive binary SA search (30% time)
    phase2_time = 0.30 * budget_s
    if remaining() > phase2_time * 0.3:
        h_bin = binarize(h, n)
        t2 = min(phase2_time, int(remaining() * 0.35))
        h_b, c5_b = binary_sa_search(h_bin, n, rng, time_limit=t2)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h = h_b.copy()

    # Phase 3: Correlation equalization (15% time)
    if remaining() > 30:
        h_eq, c5_eq = corr_equalize(h, n, n_iter=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 4: Try Thue-Morse and other structured starts (15% time)
    if remaining() > 40:
        # Thue-Morse at n
        h_tm = thue_morse_seq(n, rng)
        h_tm = slsqp_optimize(h_tm, n, 5000, maxiter=300)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            h = h_tm.copy()

        # Binary SA from Thue-Morse
        h_tm_bin = binarize(h_tm, n)
        t4 = min(30, int(remaining() * 0.15))
        h_tmb, c5_tmb = binary_sa_search(h_tm_bin, n, rng, time_limit=t4)
        if c5_tmb < best_c5:
            best_c5 = c5_tmb
            best_h = h_tmb.copy()
            h = h_tmb.copy()

        # Random start
        h_r = rng.uniform(0, 1, n)
        h_r = project_feasible(h_r, n)
        h_r = slsqp_optimize(h_r, n, 5000, maxiter=300)
        h_rb = binarize(h_r, n)
        t5 = min(20, int(remaining() * 0.1))
        h_rb2, c5_rb = binary_sa_search(h_rb, n, rng, time_limit=t5)
        if c5_rb < best_c5:
            best_c5 = c5_rb
            best_h = h_rb2.copy()
            h = h_rb2.copy()

    # Phase 5: Multi-resolution exploration (20% time)
    if remaining() > 60:
        n_candidates = []
        for nt in range(max(320, best_n - 20), min(440, best_n + 21), 2):
            if nt % 2 == 0 and nt >= 256:
                n_candidates.append(nt)

        for n_t in n_candidates:
            if remaining() < 15:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

            # Quick SA
            h_t_bin = binarize(h_t, n_t)
            tt = min(15, int(remaining() * 0.08))
            if tt > 5:
                h_tb, c5_tb = binary_sa_search(h_t_bin, n_t, rng, time_limit=tt)
                if c5_tb < best_c5:
                    best_c5 = c5_tb
                    best_h = h_tb.copy()
                    best_n = n_t

            # Quick SLSQP
            if remaining() > 10:
                h_t = slsqp_optimize(h_t, n_t, 10000, maxiter=200)
                c5_t = compute_c5(h_t)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    best_n = n_t

    # Phase 6: Final intensive refinement at best resolution (remaining)
    n = best_n
    h = best_h.copy()

    # SLSQP with increasing beta
    for beta in [5000, 20000, 50000, 100000]:
        if remaining() < 5:
            break
        h_new = slsqp_optimize(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Correlation equalization
    if remaining() > 15:
        h_eq, c5_eq = corr_equalize(h, n, n_iter=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary SA
    if remaining() > 15:
        h_bin = binarize(h, n)
        tf = min(30, int(remaining() * 0.3))
        if tf > 5:
            h_b, c5_b = binary_sa_search(h_bin, n, rng, time_limit=tf)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                # Continuous refinement from binary
                if remaining() > 5:
                    h_c = slsqp_optimize(h_b, n, 20000, maxiter=300)
                    c5_c = compute_c5(h_c)
                    if c5_c < best_c5:
                        best_c5 = c5_c
                        best_h = h_c.copy()

    # Final micro-optimization: tiny perturbations
    if remaining() > 10:
        for _ in range(20):
            if remaining() < 3:
                break
            h_p = best_h + rng.randn(n) * 0.01
            h_p = project_feasible(h_p, n)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

        # Targeted: reduce the single worst correlation
        h_f = best_h.copy()
        corr_f = compute_corr(h_f)
        worst_k = int(np.argmax(corr_f))
        off_f = worst_k - (n - 1)
        g_f = corr_grad_at(h_f, off_f, n)
        g_f -= np.mean(g_f)
        for step in [0.01, 0.02, 0.005, 0.03, 0.015]:
            h_try = project_feasible(h_f - step * g_f, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_f = h_try.copy()
                break

    # Ensure feasibility and validate
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
