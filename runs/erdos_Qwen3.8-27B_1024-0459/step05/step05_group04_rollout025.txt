                        dr = np.zeros(len(corr))
                        for ii in range(i, i2+1):
                            dr += make_swap_delta_single(h, ii, -1, n)
                        for jj in range(j, j2+1):
                            dr += make_swap_delta_single(h, jj, +1, n)
                        new_corr = corr + dr
                        new_max = float(np.max(new_corr))
                        if new_max < cur_max or (new_max - cur_max) < T and rng.rand() < np.exp(-(new_max - cur_max) / T):
                            h[i:i2+1] = 0
                            h[j:j2+1] = 1
                            corr = new_corr
                            cur_max = new_max
                            stall = 0
                            if cur_max < best_max:
                                best_max = cur_max
                                best_h = h.copy()
                        break
        
        # Single swaps
        best_move = None
        best_new_max = cur_max
        seen = set()
        
        for (i, j) in candidates:
            if i >= n or j >= n or i < 0 or j < 0:
                continue
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            
            dr = make_swap_delta(h, i, j, n)
            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr)
        
        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall > max_stall:
                # Perturbation: accept k random worsening moves
                k = rng.randint(2, 5)
                for _ in range(k):
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    if h[i] == 1 and h[j] == 0:
                        dr = make_swap_delta(h, i, j, n)
                        h[i] = 0
                        h[j] = 1
                        corr = corr + dr
                cur_max = float(np.max(corr))
                stall = 0
        
        T = max(T * cool_rate, T_min)
    
    # Final greedy polish
    stall2 = 0
    while stall2 < 200 and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        top_k = np.argsort(-corr)[:10]
        candidates = []
        for s in top_k[:8]:
            off = s - (n - 1)
            if off >= 0:
                np_ = n - off
                contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for ci in contrib[:25]:
                    candidates.append((ci + off, ci))
            else:
                ao = -off
                np_ = n - ao
                contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for ci in contrib[:25]:
                    candidates.append((ci, ci + ao))
        for _ in range(40):
            candidates.append((ones[rng.randint(len(ones))], zeros[rng.randint(len(zeros))]))
        
        best_move = None
        best_new_max = cur_max
        seen = set()
        for (i, j) in candidates:
            if i >= n or j >= n or i < 0 or j < 0:
                continue
            if h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            dr = make_swap_delta(h, i, j, n)
            new_corr = corr + dr
            new_max = float(np.max(new_corr))
            if new_max < best_new_max:
                best_new_max = new_max
                best_move = (i, j, dr)
        
        if best_move is not None:
            i, j, dr = best_move
            h[i] = 0
            h[j] = 1
            corr = corr + dr
            cur_max = best_new_max
            stall2 = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            break
    
    return best_h, best_max * dx

def slsqp_smooth(h, n, beta, maxiter=300):
    h = project_feasible(h, n)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def projected_gradient_refine(h, n, max_iters=2000, top_k=6):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        
        # Subtract mean to maintain sum constraint
        g -= np.mean(g)
        
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        
        improved = False
        for step in [0.05, 0.02, 0.01, 0.03, 0.005, 0.08, 0.015, 0.04, 0.001]:
            ht = h + step * g
            ht = project_feasible(ht, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        
        if not improved:
            # Try random perturbations
            found = False
            for _ in range(20):
                mask = rng_rand_mask(n)
                pert = (np.random.RandomState(it + 7).rand(n) - 0.5) * 0.1 * mask
                ht = project_feasible(h + pert, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-14:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    found = True
                    improved = True
                    break
            if not found:
                if it > 100:
                    break
    
    return best_h, best_c5

def rng_rand_mask(n):
    return np.ones(n)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            rng = np.random.RandomState(0)
            rm = rng.choice(ones, n_ones - target, replace=False)
            h[rm] = 0
        else:
            zeros = np.where(h == 0)[0]
            rng = np.random.RandomState(0)
            add = rng.choice(zeros, target - n_ones, replace=False)
            h[add] = 1
    return h

def zadoff_chu_binary(n, rng):
    # Create a Zadoff-Chu-like sequence and threshold
    N = n
    k = rng.randint(1, N)
    zc = np.exp(-1j * np.pi * k * np.arange(N) * (np.arange(N) + 1) / N)
    h = (np.real(zc) > 0).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, n_ones - target, replace=False)] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, target - n_ones, replace=False)] = 1
    return h

def random_structured(n, rng):
    # Create pattern with some structure
    h = (rng.rand(n) > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, n_ones - target, replace=False)] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, target - n_ones, replace=False)] = 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    parent_n = 384
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    # Phase 1: Binary SA at parent resolution
    n = best_n
    h_bin = np.zeros(n)
    sort_idx = np.argsort(-best_h)
    h_bin[sort_idx[:n // 2]] = 1.0
    
    t1 = min(200, int(remaining() * 0.25))
    if t1 > 30:
        h_b1, c5_b1 = sa_binary_search(h_bin, n, time_limit=t1, rng=np.random.RandomState(seed))
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n
    
    # Phase 2: Multiple diverse binary starts at parent resolution
    starts = [
        thue_morse(n),
        zadoff_chu_binary(n, np.random.RandomState(seed + 1)),
        random_structured(n, np.random.RandomState(seed + 2)),
        random_structured(n, np.random.RandomState(seed + 3)),
    ]
    
    # Also try from best binary found
    starts.append(h_b1 if c5_b1 < best_c5 else h_bin)
    
    for si, sp in enumerate(starts):
        if remaining() < 60:
            break
        sp_c5 = compute_c5(sp)
        t2 = min(100, int(remaining() * 0.12))
        if t2 > 20:
            h_b2, c5_b2 = sa_binary_search(sp, n, time_limit=t2, rng=np.random.RandomState(seed + 100 + si))
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n
    
    # Phase 3: Try other resolutions
    test_ns = sorted(set([n, 256, 320, 512, 640]))
    test_ns = [x for x in test_ns if 128 <= x <= 768 and x % 2 == 0]
    
    for n_t in test_ns:
        if remaining() < 80:
            break
        if n_t == n:
            continue
        
        # Interpolate and threshold
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        
        t3 = min(80, int(remaining() * 0.1))
        if t3 > 20:
            h_b3, c5_b3 = sa_binary_search(h_bin_t, n_t, time_limit=t3, rng=np.random.RandomState(seed + 200))
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t
    
    # Phase 4: Continuous refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    for beta in [500, 2000, 10000, 50000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Projected gradient refinement
    if remaining() > 20:
        h_pg, c5_pg = projected_gradient_refine(h, n, max_iters=3000, top_k=5)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()
    
    # Phase 6: More SLSQP with higher beta
    for beta in [20000, 100000, 200000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 7: Try finer resolution
    if remaining() > 60:
        n_fine = min(n + 64, 768)
        if n_fine != n:
            h_fine = interp_to(best_h, n, n_fine)
            h_fine = project_feasible(h_fine, n_fine)
            for beta in [5000, 20000, 100000]:
                if remaining() < 15:
                    break
                h_fine = slsqp_smooth(h_fine, n_fine, beta, maxiter=300)
            c5_fine = compute_c5(h_fine)
            if remaining() > 20:
                h_fine, c5_fine2 = projected_gradient_refine(h_fine, n_fine, max_iters=2000, top_k=6)
                c5_fine = min(c5_fine, c5_fine2)
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine.copy()
                best_n = n_fine
    
    # Phase 8: Final polish
    n = best_n
    h = best_h.copy()
    for beta in [50000, 200000, 500000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    if remaining() > 15:
        h_pg2, c5_pg2 = projected_gradient_refine(h, n, max_iters=3000, top_k=8)
        if c5_pg2 < best_c5:
            best_c5 = c5_pg2
            best_h = h_pg2.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
