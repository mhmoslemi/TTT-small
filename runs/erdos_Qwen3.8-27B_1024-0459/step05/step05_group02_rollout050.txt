def top_equalize(h, n, top_k=5, max_iters=400):

def interp_to(src_h, src_n, dst_n):

        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def perturb_reoptimize(h, n, num_flip, beta=5000):
    """Perturb by flipping a few values, then re-optimize."""
    h_p = h.copy()
    idx = np.random.choice(n, num_flip, replace=False)
    for i in idx:
        h_p[i] = 1.0 - h_p[i]
    h_p = proj(h_p, n)
    h_p = lbfgs_pass_softmax(h_p, n, beta, top_k=5, maxiter=300)
    return h_p
