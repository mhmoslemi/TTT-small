def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

def make_swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)
    hi3 = min(j + n - 1, r - 1)
    if lo3 <= hi3:

                g[ao:ao + np_] -= wk * x[:np_]

        {
            'type': 'eq',
            'fun': obj_constraint,
            'jac': obj_jac_constraint,
            'args': ()
        }
    ]

        res = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',

def subgrad_equalize(h, n, max_iters=800, top_k=8):

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    # Create binary from parent
    h_bin = np.zeros(n)
    sort_idx = np.argsort(-best_h)
    n_half = n // 2
    h_bin[sort_idx[:n_half]] = 1.0
    
    # Phase 1: ILS on best binary (most important phase)
    t1 = min(500, int(remaining() * 0.35))
    if t1 > 30:
        h_b1, c5_b1 = binary_il_search(h_bin, n, t1, rng, n_restarts=12)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n
    
    # Also run standard binary search from the ILS result
    t1b = min(120, int(remaining() * 0.1))
    if t1b > 15:
        h_b1b, c5_b1b = binary_ils(h_bin, n, t1b, rng)
        if c5_b1b < best_c5:
            best_c5 = c5_b1b
            best_h = h_b1b.copy()
            best_n = n
    
    # Phase 2: Try different binary starting points
    starts = []
    # Thue-Morse
    tm = thue_morse(n)
    starts.append(tm)
    # Shifted Thue-Morse
    tm_shift = np.roll(thue_morse(n), n // 3)
    n_o = int(np.sum(tm_shift))
    if n_o != n_half:
        ones = np.where(tm_shift == 1)[0]
        tm_shift[ones[:abs(n_o - n_half)]] = 1.0 if n_o < n_half else 0.0
    starts.append(tm_shift)
    # Random
    rand_h = (rng.rand(n) > 0.5).astype(float)
    n_o = int(np.sum(rand_h))
    if n_o != n_half:
        if n_o > n_half:
            idx = np.where(rand_h == 1)[0]
            rand_h[idx[:n_o - n_half]] = 0
        else:
            idx = np.where(rand_h == 0)[0]
            rand_h[idx[:n_half - n_o]] = 1
    starts.append(rand_h)
    
    for sp in starts:
        if remaining() < 30:
            break
        sp_c5 = compute_c5(sp)
        # Quick local search to see if promising
        t_quick = min(60, int(remaining() * 0.08))
        if t_quick > 10:
            h_s, c5_s = binary_ils(sp, n, t_quick, np.random.RandomState(seed + 7))
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n
                # Deep dive
                t_deep = min(100, int(remaining() * 0.1))
                if t_deep > 15:
                    h_d, c5_d = binary_il_search(h_s, n, t_deep, rng, n_restarts=5)
                    if c5_d < best_c5:
                        best_c5 = c5_d
                        best_h = h_d.copy()
                        best_n = n
    
    # Phase 3: Multi-resolution binary search
    res_list = sorted(set([
        n, n - 12, n - 6, n + 6, n + 12, n + 24,
        n - 24, n + 36, int(n * 0.8), int(n * 1.2),
    ]))
    res_list = [r for r in res_list if 200 <= r <= 600 and r % 2 == 0 and r != n]
    res_list.sort(key=lambda x: abs(x - n))
    
    for n_t in res_list[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1
        
        t_bi = min(60, int(remaining() * 0.07))
        if t_bi > 10:
            h_b3, c5_b3 = binary_il_search(h_bin_t, n_t, t_bi, rng, n_restarts=4)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t
    
    # Phase 4: Continuous refinement - SLSQP with increasing beta
    n = best_n
    h = best_h.copy()
    
    beta_schedule = [50, 100, 300, 1000, 3000, 10000, 30000]
    for beta in beta_schedule:
        if remaining() < 12:
            break
        maxiter = 300 if beta < 1000 else 200
        t_check = time.time()
        h_new = slsqp_smooth(h, n, beta, maxiter=maxiter)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 5: Subgradient equalization
    if remaining() > 20:
        h_eq, c5_eq = subgrad_equalize(h, n, max_iters=600, top_k=6)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
    
    # Phase 6: More SLSQP at high beta
    for beta in [50000, 100000, 200000]:
        if remaining() < 10:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    # Phase 7: Subgradient again
    if remaining() > 15:
        h_eq2, c5_eq2 = subgrad_equalize(h, n, max_iters=400, top_k=4)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()
    
    # Phase 8: Try finer resolution
    if remaining() > 80:
        n_fine = best_n + 64
        h_fine = interp_to(best_h, best_n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)
        for beta in [2000, 10000, 50000]:
            if remaining() < 15:
                break
            h_fine = slsqp_smooth(h_fine, n_fine, beta, maxiter=200)
        c5_fine = compute_c5(h_fine)
        if remaining() > 20:
            h_fine, c5_fine2 = subgrad_equalize(h_fine, n_fine, max_iters=300, top_k=6)
            c5_fine = min(c5_fine, c5_fine2)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine
    
    # Phase 9: Final ILS on the best binary at best_n
    if remaining() > 60:
        n = best_n
        h_bin_final = np.zeros(n)
        sort_idx = np.argsort(-best_h)
        n_half = n // 2
        h_bin_final[sort_idx[:n_half]] = 1.0
        t_final = min(120, int(remaining() * 0.2))
        if t_final > 15:
            h_ff, c5_ff = binary_il_search(h_bin_final, n, t_final, rng, n_restarts=6)
            if c5_ff < best_c5:
                best_c5 = c5_ff
                best_h = h_ff.copy()
                best_n = n
            # Then do continuous on the improved binary
            if remaining() > 30:
                h_c = best_h.copy()
                for beta in [5000, 30000, 100000]:
                    if remaining() < 10:
                        break
                    h_c = slsqp_smooth(h_c, best_n, beta, maxiter=150)
                c5_c = compute_c5(h_c)
                if c5_c < best_c5:
                    best_c5 = c5_c
                    best_h = h_c.copy()
    
    # Phase 10: Final polish
    n = best_n
    h = best_h.copy()
    for beta in [50000, 150000, 500000]:
        if remaining() < 6:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
    
    if remaining() > 8:
        h_eq3, c5_eq3 = subgrad_equalize(h, n, max_iters=300, top_k=5)
        if c5_eq3 < best_c5:
            best_c5 = c5_eq3
            best_h = h_eq3.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
