def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def thue_morse_init(n):

        for p in range(12):
            if remaining_time() < 40:
                break
            h_pert = best_h.copy()
            noise_scale = 0.2 / (1.0 + p * 0.5)
            frac = 0.2 + 0.1 * (p % 3)
            mask = rng.random(n) < frac
            h_pert[mask] += rng.randn(np.sum(mask)) * noise_scale
            h_pert = project_feasible(h_pert, n)

            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=60)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

            if remaining_time() > 50:
                h_slq = slsqp_refine(h_new, n, 10000, 300)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    h_new, c5_new = fast_coord_descent(h_slq, n, max_sweeps=40)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()

    # Phase 5: Thue-Morse based start
    if remaining_time() > 100:
        h_tm = thue_morse_init(n)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 + 0.08:
            h_new, c5_new = fast_coord_descent(h_tm, n, max_sweeps=150)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 40:
                    h_slq = slsqp_refine(h_new, n, 20000, 400)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        h_new, c5_new = fast_coord_descent(h_new, n, max_sweeps=50)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()

    # Phase 6: Higher resolution
    if remaining_time() > 150:
        high_n = 2 * n
        if high_n <= 2048:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            c5_high = compute_c5(h_high)

            h_new, c5_new = fast_coord_descent(h_high, high_n, max_sweeps=80)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = high_n

            if remaining_time() > 60:
                h_slq = slsqp_refine(h_new, high_n, 5000, 300)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    best_n = high_n
                    h_new, c5_new = fast_coord_descent(h_new, high_n, max_sweeps=50)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = high_n

    # Phase 7: Subgradient + CD alternating
    if remaining_time() > 80:
        n_cur = best_n
        h = best_h.copy()
        for it in range(800):
            if remaining_time() < 5:
                break
            step = 0.03 / (1.0 + 0.02 * it)
            h_sg = active_set_subgrad(h, n_cur, step, n_active=10)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5 - 1e-14:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()
            elif it % 40 == 20 and remaining_time() > 15:
                h_new, c5_new = fast_coord_descent(h, n_cur, max_sweeps=15)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    if it > 100:
                        break

    # Phase 8: Final SLSQP with very high beta
    if remaining_time() > 20:
        n_cur = best_n
        for beta in [100000, 500000]:
            if remaining_time() < 10:
                break
            h_new = slsqp_refine(best_h, n_cur, beta, min(500, int(remaining_time() * 0.7)))
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 10: One more SLSQP + CD
    if remaining_time() > 10:
        n_cur = best_n
        h_new = slsqp_refine(best_h, n_cur, 300000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if remaining_time() > 5:
            h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
