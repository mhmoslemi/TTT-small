def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            adj = diff / cnt
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            if diff > 0:
                idx = int(np.argmin(h))
                h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            else:

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def fast_coord_descent(h, n, max_sweeps=200, tol=1e-14):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        max_improvement = 0.0
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            def _eval_t(t):
                mid = base + A * t
                mid[n - 1] = base[n - 1] + t - t * t
                return np.max(mid)

            f_old = _eval_t(old)

            # Check endpoints and midpoint
            f_0 = _eval_t(0.0)
            f_1 = _eval_t(1.0)
            f_half = _eval_t(0.5)

            t_best = old
            f_best = f_old
            if f_0 < f_best:
                t_best, f_best = 0.0, f_0
            if f_1 < f_best:
                t_best, f_best = 1.0, f_1
            if f_half < f_best:
                t_best, f_best = 0.5, f_half

            # Golden section refinement around best
            if t_best < 0.999 * f_old or f_best < f_old - 1e-13:
                if t_best < old:
                    lo, hi = 0.0, min(1.0, old)
                else:
                    lo, hi = old, 1.0
                if hi - lo < 1e-12:
                    lo, hi = max(0.0, old - 0.1), min(1.0, old + 0.1)
                    if hi - lo < 1e-12:
                        lo, hi = 0.0, 1.0

                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = _eval_t(c1)
                f2 = _eval_t(c2)
                for _ in range(50):

                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _eval_t(c1)

                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return project_feasible(h, n)


def random_binary_init(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h


def active_set_gradient(h, n, n_active=10):
    """Compute gradient toward reducing the top active lags."""
    corr = compute_corr(h)
    max_corr = np.max(corr)
    threshold = max_corr - 0.3
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        idx = active[np.argsort(corr[active])[::-1][:n_active]]
        active = idx

    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    return grad

    """Perturb a base solution and run coordinate descent."""
    h_pert = h_base.copy()
    # Structured perturbation: flip a random subset of near-binary values
    mask = rng.random(n) < alpha_perturb
    h_pert[mask] = np.clip(h_pert[mask] + rng.randn(np.sum(mask)) * noise_scale, 0.0, 1.0)
    h_pert = project_feasible(h_pert, n)

    h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=max_sweeps)
    return h_new, c5_new


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent at parent resolution
    h = best_h.copy()
    h, c5 = fast_coord_descent(h, best_n, max_sweeps=80)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: SLSQP at parent resolution
    if remaining_time() > 100:
        for beta in [1000, 5000, 20000]:
            if remaining_time() < 30:
                break
            h_new = slsqp_refine(best_h, best_n, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        # Final coord descent
        h, c5 = fast_coord_descent(best_h, best_n, max_sweeps=50)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: Go to higher resolution n=1280
    if remaining_time() > 200:
        n2 = 1280
        h2 = interp_to(best_h, best_n, n2)
        h2 = project_feasible(h2, n2)
        c5_2 = compute_c5(h2)

        # Coordinate descent at n2
        h2, c5_2 = fast_coord_descent(h2, n2, max_sweeps=100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

        # SLSQP at n2
        if remaining_time() > 80:
            for beta in [2000, 10000, 50000]:
                if remaining_time() < 30:
                    break
                h_new = slsqp_refine(best_h, best_n, beta, 400)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n2
            # Coord descent polish
            h, c5 = fast_coord_descent(best_h, best_n, max_sweeps=50)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n2

    # Phase 4: Try Thue-Morse at high resolution
    if remaining_time() > 200:
        n3 = 1280
        h_tm = thue_morse_init(n3)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5 + 0.02:
            h_tm, c5_tm = fast_coord_descent(h_tm, n3, max_sweeps=100)
            if c5_tm < best_c5:
                best_c5 = c5_tm
                best_h = h_tm.copy()
                best_n = n3
            if remaining_time() > 60:
                h_new = slsqp_refine(best_h, best_n, 10000, 300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n3
                h, c5 = fast_coord_descent(best_h, best_n, max_sweeps=40)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n = n3

    # Phase 5: Multiple random restarts
    if remaining_time() > 100:
        n_cur = best_n
        n_restarts = min(6, int(remaining_time() / 40))
        for r in range(n_restarts):
            if remaining_time() < 30:
                break
            noise = 0.2 / (1 + r * 0.5)
            alpha = 0.25
            h_new, c5_new = perturb_and_optimize(best_h, n_cur, rng, noise,
                                                 max_sweeps=60, alpha_perturb=alpha)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_cur
            # Quick SLSQP on improved
            if remaining_time() > 50 and c5_new < best_c5 + 0.001:
                h_slq = slsqp_refine(h_new, n_cur, 5000, 200)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    best_n = n_cur

    # Phase 6: Try n=2560 if time allows
    if remaining_time() > 300:
        n4 = 2560
        h4 = interp_to(best_h, best_n, n4)
        h4 = project_feasible(h4, n4)
        h4, c5_4 = fast_coord_descent(h4, n4, max_sweeps=60)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
            best_n = n4
        if remaining_time() > 80:
            h_new = slsqp_refine(best_h, best_n, 5000, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n4
            h, c5 = fast_coord_descent(best_h, best_n, max_sweeps=40)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n4

    # Phase 7: Subgradient + coordinate descent alternating
    if remaining_time() > 60:
        n_cur = best_n
        h = best_h.copy()
        for it in range(2000):
            if remaining_time() < 5:
                break
            step = 0.03 / (1.0 + it * 0.005)
            grad = active_set_gradient(h, n_cur, n_active=8)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            h_new = h - step * grad
            h_new = project_feasible(h_new, n_cur)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
            elif it % 20 == 10 and remaining_time() > 20:
                h_cd, c5_cd = fast_coord_descent(h, n_cur, max_sweeps=15)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    h = h_cd.copy()
                else:
                    break

    # Phase 8: Final SLSQP polish with very high beta
    if remaining_time() > 15:
        n_cur = best_n
        for beta in [20000, 100000, 500000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(best_h, n_cur, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        h, c5 = fast_coord_descent(best_h, n_cur, max_sweeps=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 9: Final coordinate descent
    if remaining_time() > 8:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
