import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)

            break

        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)

def smooth_corr_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0

        c5 = compute_c5(h)
        if c5 < best_c5:

            best_h = h.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def smooth_corr_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def smooth_corr_grad(h, beta):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / n
    w = w * dx
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def precompute_effect(h, n):
    """Precompute effect_1to0[p, k] for all positions p and shifts k."""
    nk = 2 * n - 1
    effect = np.zeros((n, nk), dtype=np.float64)
    idx = np.arange(nk)
    for p in range(n):
        # first_loss: (1 - h[p+k-(n-1)]) where 0 <= p+k-(n-1) < n
        j = p + idx - (n - 1)
        valid1 = (j >= 0) & (j < n)
        effect[p] -= np.where(valid1, 1.0 - h[j[valid1]], 0.0)
        # second_gain: h[p-k+(n-1)] where 0 <= p-k+(n-1) < n
        j2 = p - idx + (n - 1)
        valid2 = (j2 >= 0) & (j2 < n)
        effect[p] += np.where(valid2, h[j2[valid2]], 0.0)
    return effect


def fast_binary_ls(h_bin, n, max_rounds=100, sa_temp=0.0, patience=10):
    """Vectorized binary local search with single swaps."""
    h = h_bin.copy().astype(np.float64)
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * (2.0 / n))
    best_h = h.copy()
    dx = 2.0 / n
    no_improve = 0
    T = sa_temp

    for rnd in range(max_rounds):
        if remaining() < 5:
            break
        effect = precompute_effect((h > 0.5).astype(np.float64), n)
        ones_mask = h > 0.5
        zeros_mask = ~ones_mask
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        c_max = np.max(corr)
        best_new_max = c_max - 1e-15
        best_pair = None
        
        for ia in ones_idx:
            delta_a = effect[ia]
            # For all b in zeros_idx: new_corr = corr + delta_a - effect[b]
            # We want to minimize max(new_corr)
            candidates = corr[None, :] + delta_a[None, :] - effect[zeros_idx, :]
            new_maxes = np.max(candidates, axis=1)
            best_local = np.argmin(new_maxes)
            if new_maxes[best_local] < best_new_max:
                best_new_max = new_maxes[best_local]
                best_pair = (ia, zeros_idx[best_local])

            h[pa] = 0.0
            h[pb] = 1.0

            if c5_new <= best_c5 + T:
                best_c5 = min(best_c5, c5_new)
                best_h = h.copy()
                no_improve = 0
            else:
                h[pa] = 1.0
                h[pb] = 0.0
                no_improve += 1
                T *= 0.9
        else:
            no_improve += 1
        
        if no_improve > patience:
            break
    
    return best_h, best_c5


def block_move_ls(h_bin, n, max_rounds=30, block_sizes=(2, 3, 4)):
    """Try moving contiguous blocks of 1s to different positions."""
    h = h_bin.copy().astype(np.float64)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if remaining() < 5:
            break
        best_imp = 1e-15
        best_move = None
        best_new_c5 = best_c5
        
        for bs in block_sizes:
            # Find all contiguous blocks of 1s of length >= bs
            for start in range(0, n - bs + 1):
                if any(h[start:start+bs] < 0.5):
                    continue
                # Try moving this block to various positions
                for dest in range(0, n - bs + 1):
                    if dest == start:
                        continue
                    # Check dest doesn't overlap with source
                    if dest < start + bs and dest + bs > start:
                        continue
                    h2 = h.copy()
                    h2[start:start+bs] = 0.0
                    h2[dest:dest+bs] = 1.0
                    c52 = compute_c5(h2)
                    if c52 < best_new_c5 - 1e-15:
                        best_new_c5 = c52
                        best_move = (start, dest, bs)
                        best_imp = best_c5 - c52
        
        if best_move is not None:
            s, d, bs = best_move
            h[s:s+bs] = 0.0
            h[d:d+bs] = 1.0

def nesterov_opt(h, n, beta, max_iters, lr_init=0.02, lr_decay=0.01):
    h = project_feasible(h, n)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        alpha = (t_val - 1.0) / t_new
        y = y + alpha * (y - h)
        g = smooth_corr_grad(h, beta)
        lr = lr_init / (1.0 + lr_decay * it)
        h_new = project_feasible(y - lr * g, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        y = h
        h = h_new
        t_val = t_new
    
    return best_h, best_c5


def slsqp_opt(h, n, beta, maxiter=300):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj_fun(x):
        return smooth_corr_obj(x, beta)
    
    def grad_fun(x):
        return smooth_corr_grad(x, beta)
    
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj_fun(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

    sorted_idx = np.argsort(-h)

def structured_binary(n, rng, pattern_type=0):
    h = np.zeros(n)
    half = n // 2
    if pattern_type == 0:
        # Golden ratio based
        golden = 0.618033988749895
        positions = sorted((int(i * golden * n) % n) for i in range(half))
        h[np.array(positions[:half])] = 1.0
        # Fix count
        cnt = np.count_nonzero(h)
        if cnt > half:
            h[np.where(h == 1)[0][half:]] = 0.0
        elif cnt < half:
            zeros = np.where(h == 0)[0]
            h[zeros[:half - cnt]] = 1.0
    elif pattern_type == 1:
        # Thue-Morse like
        h = np.zeros(n)
        for i in range(n):
            h[i] = np.binary_repr(i).count('1') % 2
        # Ensure exactly half ones
        sorted_idx = np.argsort(-h)
        h2 = np.zeros(n)
        h2[sorted_idx[:half]] = 1.0
        h = h2
    elif pattern_type == 2:
        # Random with local structure
        h = random_binary(n, rng)
        # Smooth slightly
        for _ in range(3):
            h2 = h.copy()
            for i in range(1, n - 1):
                if h[i-1] + h[i+1] > h[i] * 1.5:
                    h2[i] = 1.0
                elif h[i-1] + h[i+1] < h[i] * 0.5:
                    h2[i] = 0.0
            cnt = np.count_nonzero(h2)
            if cnt == half:
                h = h2
    return h


def coordinate_descent(h, n, max_rounds=50):

                best_h = h_ref.copy()

    # Phase 6: Final intensive polish
    n = best_n
    h = best_h.copy()
    for beta in [10000, 50000, 150000]:
        if remaining() < 10:
            break
        h_sl = slsqp_opt(h, n, beta, maxiter=min(400, int(remaining() // 8)))
        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            h = h_sl.copy()
        else:
            h = h_sl  # still use as starting point for next

    # Phase 7: Final Nesterov with very high beta
    if remaining() > 15:
        h_fin, c5_fin = nesterov_opt(best_h, n, 50000, max_iters=300, lr_init=0.005, lr_decay=0.02)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 8: Final binary tweak
    if remaining() > 10:
        h_bf = round_to_binary(best_h, n)
        h_bf, c5_bf = fast_binary_ls(h_bf, n, max_rounds=20, sa_temp=0.0, patience=5)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()
        # Quick SLSQP from binary
        if remaining() > 8:
            h_pol = slsqp_opt(h_bf, n, 50000, maxiter=200)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5
    else:
        best_h = project_feasible(best_h, best_n)
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
