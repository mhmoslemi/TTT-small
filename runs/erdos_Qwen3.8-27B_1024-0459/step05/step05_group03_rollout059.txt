    """SLSQP optimization with smooth max objective."""
    h0 = project_feasible(np.array(h, dtype=float), n)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex) * dx
        # Gradient of sum_k w[k] * corr[k] w.r.t. x
        # corr[k] = sum_i x[i+k-(n-1)] * (1-x[i]) for valid i
        g = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-14:
                continue
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off+np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:a+np_])
                g[a:a+np_] -= wk * x[:np_]
        return g
    
    def cons(x):
        return np.sum(x) - n/2.0
    
    def cons_jac(x):
        return np.ones(n)

                       bounds=bounds,

                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})

            h[q1] = 1.0
            h[q2] = 1.0
            nc = np.max(compute_corr_full(h))
            h[p1] = 1.0
            h[p2] = 1.0
            h[q1] = 0.0
            h[q2] = 0.0
            imp = c_max - nc
            if imp > best_improve:
                best_improve = imp
                best_move = (int(p1), int(p2), int(q1), int(q2))
                best_type = 2

        if best_move is None:
            break

        if best_type == 1:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
        elif best_type == 2:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 0.0
            h[best_move[2]] = 1.0
            h[best_move[3]] = 1.0

        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            break

    return best_h, best_c5


def slsqp_polish(h, n, beta, maxiter=300):
    h0 = project_feasible(np.array(h, dtype=float), n)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex) * dx
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += wk * (1.0 - x[a:a + np_])
                g[a:a + np_] -= wk * x[:np_]
        return g

    def _cons(x):
        return np.sum(x) - n / 2.0

    def _cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-14:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0

    best_h = parent_h.copy()
    best_c5 = compute_c5(parent_h)
    best_n = parent_n

    # Phase 1: Binary local search at parent resolution
    h_bin = round_to_binary(parent_h, parent_n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Greedy exhaustive LS
    h_g, c5_g = greed_binary_ls(h_bin, parent_n, rng, max_rounds=80)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
        h_bin = h_g.copy()

    # SA from the best binary
    if remaining() > 60:
        h_sa, c5_sa = sa_binary(h_bin, parent_n, rng,
                                n_iters=min(150000, int(remaining() // 3)),
                                t_start=1.0, t_end=0.01)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_bin = h_sa.copy()

    # SA with perturbations (multi-restart)
    for restart in range(6):
        if remaining() < 30:
            break
        h_pert = h_bin.copy()
        n_perturb = rng.randint(10, 40)
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_perturb):
            pa = ones[rng.randint(len(ones))]
            pb = zeros[rng.randint(len(zeros))]
            h_pert[pa], h_pert[pb] = h_pert[pb], h_pert[pa]

        h_sa_r, c5_sa_r = sa_binary(h_pert, parent_n, rng,
                                    n_iters=min(80000, int(remaining() // 8)),
                                    t_start=0.6, t_end=0.008)
        if c5_sa_r < best_c5:
            best_c5 = c5_sa_r
            best_h = h_sa_r.copy()
            h_bin = h_sa_r.copy()
        # Also try greedy from this SA result
        if c5_sa_r < best_c5 + 0.001 and remaining() > 20:
            h_g2, c5_g2 = greed_binary_ls(h_sa_r, parent_n, rng, max_rounds=20)
            if c5_g2 < best_c5:
                best_c5 = c5_g2
                best_h = h_g2.copy()
                h_bin = h_g2.copy()

    # Top-k reduction LS
    if remaining() > 30:
        for tk in [3, 5, 8]:
            if remaining() < 15:
                break
            h_tk, c5_tk = topk_reduction_ls(h_bin, parent_n, rng,
                                            max_rounds=100, top_k=tk)
            if c5_tk < best_c5:
                best_c5 = c5_tk
                best_h = h_tk.copy()
                h_bin = h_tk.copy()

    # 2-swap local search
    if remaining() > 30:
        h_2s, c5_2s = two_swap_ls(h_bin, parent_n, rng, max_rounds=200)
        if c5_2s < best_c5:
            best_c5 = c5_2s
            best_h = h_2s.copy()
            h_bin = h_2s.copy()

    # Phase 2: Try other resolutions
    other_ns = [512, 448, 640]
    for n_try in other_ns:
        if remaining() < 80:
            break
        h_interp = interp_to(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_bin_t = round_to_binary(h_interp, n_try)
        c5_t = compute_c5(h_bin_t)

        # Greedy LS
        h_g_t, c5_g_t = greed_binary_ls(h_bin_t, n_try, rng, max_rounds=30)
        if c5_g_t < c5_t:
            c5_t = c5_g_t
            h_bin_t = h_g_t.copy()

        # SA
        if remaining() > 40:
            h_sa_t, c5_sa_t = sa_binary(h_bin_t, n_try, rng,
                                        n_iters=min(60000, int(remaining() // 5)),
                                        t_start=0.6, t_end=0.01)
            if c5_sa_t < c5_t:
                c5_t = c5_sa_t
                h_bin_t = h_sa_t.copy()

        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_bin_t.copy()
            best_n = n_try

    # Phase 3: Try random binary initializations at parent_n
    if remaining() > 60:
        for ri in range(4):
            if remaining() < 20:
                break
            h_rand = random_binary(parent_n, rng)
            c5_r = compute_c5(h_rand)
            if c5_r < best_c5 + 0.02:  # Only if reasonably close
                h_gr, c5_gr = greed_binary_ls(h_rand, parent_n, rng, max_rounds=30)
                if c5_gr < c5_r:
                    c5_r = c5_gr
                    h_rand = h_gr.copy()
                if remaining() > 20:
                    h_sr, c5_sr = sa_binary(h_rand, parent_n, rng,
                                            n_iters=min(40000, int(remaining() // 4)),
                                            t_start=1.5, t_end=0.02)
                    if c5_sr < c5_r:
                        c5_r = c5_sr
                        h_rand = h_sr.copy()
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_rand.copy()
                    best_n = parent_n

    # Phase 4: Continuous polish (only accept if strictly better)
    if remaining() > 60:
        h_cont = best_h.copy()
        for beta in [1000, 10000, 100000]:
            if remaining() < 20:
                break
            h_opt = slsqp_polish(h_cont, best_n, beta, maxiter=400)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_cont = h_opt.copy()
            else:
                h_cont = h_opt.copy()

    # Phase 5: Final polish rounds
    if remaining() > 30:
        # More SA at the best
        h_final_sa, c5_final_sa = sa_binary(best_h, best_n, rng,
                                            n_iters=min(100000, int(remaining() // 4)),
                                            t_start=0.3, t_end=0.002)
        if c5_final_sa < best_c5:
            best_c5 = c5_final_sa
            best_h = h_final_sa.copy()

    # Final projection and validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    # Validate with evaluate_erdos_solution
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
