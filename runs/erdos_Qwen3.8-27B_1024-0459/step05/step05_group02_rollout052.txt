def proj(h, n):

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        if np_ <= 0:
            return g
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao

def _sm_fun(x):
    n = _N
    dx = 2.0 / n
    corr = compute_corr(x)
    idx = np.argsort(-corr)[:_TOPK]
    c = corr[idx]
    m = c[0]

    sh = np.clip(sh, -60.0, 60.0)
    ex = np.exp(sh)
    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 200.0 * (np.sum(x) - _TGT) ** 2

def _sm_grad(x):

    dx = 2.0 / n

def fista_pass(h, n, max_iter=5000, topk=5, beta_start=100, beta_end=50000):
    global _N, _BETA, _TOPK, _TGT
    n = int(n)
    _N = n
    _TOPK = topk
    _TGT = n / 2.0
    h = proj(h, n)
    h_curr = h.copy()
    h_prev = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    for it in range(max_iter):
        if remaining() < 3:
            break
        # Adaptive beta
        frac = it / max(max_iter - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        _BETA = beta

        corr = compute_corr(h_curr)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_curr.copy()

        # Gradient of smooth surrogate
        idx = np.argsort(-corr)[:topk]
        c_top = corr[idx]
        m = c_top[0]
        sh = beta * (c_top - m)
        sh = np.clip(sh, -60.0, 60.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h_curr, ki, n)
        g *= dx

        # FISTA step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        g_curr = (1.0 - t / t_new) * h_curr
        # Backtracking line search
        step = 1.0 / (beta * dx + 1.0)
        for _ in range(20):
            y = proj(g_curr - step * g, n)
            f_new = _sm_fun(y)
            f_old = _sm_fun(g_curr)
            # Second-order test
            diff = y - g_curr
            if np.dot(diff, (f_new - f_old - np.dot(g, diff)) / (0.5 * step * np.dot(diff, diff) + 1e-30)) <= 0.9:
                break
            step *= 0.5

        h_prev = h_curr.copy()
        h_curr = proj(g_curr - step * g, n)
        t = t_new

    return best_h, best_c5


def topk_equalize(h, n, topk=5, max_iter=2000):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx_all = np.argsort(-corr)[:topk]
        c_vals = corr[idx_all]
        c_max = c_vals[0]
        c_min = c_vals[-1]

        if c_max - c_min < 1e-10:
            break

        # Direction: increase min, decrease max
        g_dir = np.zeros(n)
        for ki in idx_all:
            gv = corr_grad_single(h, ki, n)
            if ki == idx_all[0]:
                g_dir -= gv
            elif ki == idx_all[-1]:
                g_dir += gv
        g_dir -= np.mean(g_dir)
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            # Only reduce max
            g_dir = -corr_grad_single(h, idx_all[0], n)
            g_dir -= np.mean(g_dir)
            gn = np.linalg.norm(g_dir)
            if gn < 1e-14:
                break
        g_dir /= gn

        improved = False
        for alpha in [0.05, 0.03, 0.01, 0.08, 0.02, 0.005, 0.1, 0.015]:
            h_try = proj(h + alpha * g_dir, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                h = h_try

                best_h = h.copy()
                improved = True
                break
        if not improved:

    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def lbfgs_pass(h, n, beta, topk=5, maxiter=400):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = topk
    h = proj(h, n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-10, 'maxfun': maxiter * 10})
        return proj(res.x, n)
    except Exception:
        return h


def thue_morse(n, period_n):
    """Generate a Thue-Morse-like pattern of length n based on pattern of length period_n."""
    tm = np.zeros(period_n)
    for i in range(period_n):
        tm[i] = 1.0 if (bin(i).count('1') % 2 == 0) else 0.0
    # Tile to length n
    result = np.tile(tm, n // period_n + 1)[:n]
    return result


def boxcar_pattern(n, blocks):
    """Generate periodic block pattern: k ones followed by k zeros."""
    h = np.zeros(n)
    period = blocks * 2
    for start in range(0, n, period):
        end = min(start + blocks, n)
        h[start:end] = 1.0
    return h


def optimize_one(h0, n, time_alloc):
    """Run a full optimization pipeline on a single starting point with given time budget."""
    t_start = time.time()
    h = proj(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # FISTA with adaptive beta
    elapsed = time.time() - t_start
    if elapsed < time_alloc * 0.35:
        h_f, c5_f = fista_pass(h, n, max_iter=3000, topk=5, beta_start=50, beta_end=30000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        elapsed = time.time() - t_start

    # L-BFGS with increasing beta
    if time.time() - t_start < time_alloc * 0.6:
        for beta in [2000, 10000, 30000]:
            if time.time() - t_start > time_alloc * 0.6:
                break
            if remaining() < 5:
                break
            h_new = lbfgs_pass(best_h, n, beta, topk=5, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Top-K equalization
    if time.time() - t_start < time_alloc * 0.8:
        if remaining() < 5:
            return best_h, best_c5
        h_eq, c5_eq = topk_equalize(best_h, n, topk=5, max_iter=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Binary 2-flip search
    if time.time() - t_start < time_alloc and remaining() > 5:
        h_bin_init = (best_h > 0.5).astype(float)
        h_b, c5_b = binary_2flip_search(h_bin_init, n, max_iters=2000, topk=4)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Final L-BFGS polish
    if remaining() > 5 and time.time() - t_start < time_alloc:
        h_new = lbfgs_pass(best_h, n, 50000, topk=4, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Generate diverse initializations
    n_base = best_n
    initializations = []

    # 1. Parent
    initializations.append(('parent', parent_h.copy(), parent_n))

    # 2. Thue-Morse at n_base
    tm = thue_morse(n_base, n_base // 2)
    tm = proj(tm, n_base)
    initializations.append(('thue_morse', tm, n_base))

    # 3. Random binary at n_base
    rb = np.zeros(n_base)
    idx = rng.permutation(n_base)
    rb[idx[:n_base // 2]] = 1.0
    initializations.append(('random', rb, n_base))

    # 4. Boxcar pattern
    bc = boxcar_pattern(n_base, n_base // 8)
    bc = proj(bc, n_base)
    initializations.append(('boxcar', bc, n_base))

    # 5. Interpolated parent at higher resolution
    for n_t in [n_base + 48, n_base + 96]:
        if n_t % 2 == 0 and n_t <= 600:
            hp = interp_to(parent_h, parent_n, n_t)
            hp = proj(hp, n_t)
            initializations.append(('interp_' + str(n_t), hp, n_t))

    # 6. Fourier-based initialization
    for freq in [3, 5]:
        x = np.linspace(0, 2, n_base, endpoint=False)
        ifreq = 0.3 * np.cos(2 * np.pi * freq * x) + 0.2 * np.sin(4 * np.pi * freq * x)
        fh = 0.5 + ifreq
        fh = np.clip(fh, 0.05, 0.95)
        fh = proj(fh, n_base)
        initializations.append(('fourier_' + str(freq), fh, n_base))

    # Optimize each initialization sequentially
    n_init = len(initializations)
    time_per_init = max(60, min(200, (budget_s - 60) / n_init))

    for name, h0, n0 in initializations:
        if remaining() < 30:
            break
        alloc = min(time_per_init, remaining() - 10)
        if alloc < 30:
            break
        h_opt, c5_opt = optimize_one(h0, n0, alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n0

    # Phase 2: Intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    if remaining() > 60:
        # FISTA again with higher precision
        alloc2 = min(180, remaining() * 0.3)
        h_f2, c5_f2 = fista_pass(h, n, max_iter=4000, topk=6, beta_start=200, beta_end=80000)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()
            h = h_f2.copy()

    if remaining() > 40:
        # More L-BFGS at very high beta
        for beta in [50000, 100000, 200000]:
            if remaining() < 15:
                break
            h_new = lbfgs_pass(h, n, beta, topk=4, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    if remaining() > 30:
        # Top-K equalization (more aggressive)
        h_eq, c5_eq = topk_equalize(h, n, topk=6, max_iter=1000)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining() > 60:
        # Binary 2-flip search (extended)
        h_bin = (best_h > 0.5).astype(float)
        h_b2, c5_b2 = binary_2flip_search(h_bin, n, max_iters=5000, topk=5)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            h = h_b2.copy()

    # Phase 3: Try a couple more resolutions
    if remaining() > 90:
        for n_t in [n - 32, n + 32, n + 64]:
            if n_t < 128 or n_t > 700 or n_t % 2 != 0 or n_t == n:
                continue
            if remaining() < 40:
                break
            h_interp = interp_to(best_h, best_n, n_t)
            h_interp = proj(h_interp, n_t)
            h_opt, c5_opt = optimize_one(h_interp, n_t, min(80, remaining() - 20))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_t

    # Phase 4: Final micro-refinement
    n = best_n
    h = best_h.copy()

    if remaining() > 20:
        # Very high beta L-BFGS
        h_new = lbfgs_pass(h, n, 500000, topk=3, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if remaining() > 15:
        # Final topk equalize
        h_eq, c5_eq = topk_equalize(best_h, n, topk=8, max_iter=500)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    if remaining() > 30:
        # Final binary polish
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_2flip_search(h_bin_f, n, max_iters=3000, topk=6)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-9:
        best_c5 = final_c5
    else:
        # Projecting may have slightly changed things; use the projected value
        best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
