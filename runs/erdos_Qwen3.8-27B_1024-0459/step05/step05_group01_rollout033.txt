            h_bin = np.zeros(n_t)
            h_bin[idx[:n_half]] = 1.0

        # SA phase
        sa_time = min(120, max(8, int(remaining_time() * 0.2)))
        h_sa, c5_sa = binary_sa(h_bin, n_t, time_limit=sa_time, rng=rng)

        # Greedy refinement
        gr_time = min(60, max(5, int(remaining_time() * 0.1)))
        h_gr, c5_gr = binary_greedy_swap(h_sa, n_t, time_limit=gr_time, rng=rng)

        best_local_c5 = min(c5_sa, c5_gr)
        best_local_h = h_sa if c5_sa <= c5_gr else h_gr

        # Continuous refinement
        h_ref = best_local_h.copy()
        for beta in [1000, 10000, 100000]:
            if remaining_time() < 10:
                break
            h_ref = slsqp_refine(h_ref, n_t, beta, maxiter=400)
        c5_ref = compute_c5(h_ref)

        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n_t

        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()
            best_n = n_t

    # Phase 3: Second round - more SA on the current best
    if remaining_time() > 30:
        h_bin = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones != n_half:
            idx = np.argsort(-h_bin)
            h_bin = np.zeros(best_n)
            h_bin[idx[:n_half]] = 1.0

        sa_time = min(250, max(20, int(remaining_time() * 0.35)))
        h_sa2, c5_sa2 = binary_sa(h_bin, best_n, time_limit=sa_time,
                                  rng=rng, t_start=0.02, t_end=0.00005)

        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

        # Greedy after SA
        gr_time = min(80, max(5, int(remaining_time() * 0.1)))
        h_gr2, c5_gr2 = binary_greedy_swap(best_h, best_n, time_limit=gr_time, rng=rng)
        if c5_gr2 < best_c5:
            best_c5 = c5_gr2
            best_h = h_gr2.copy()

        # Continuous refinement
        for beta in [5000, 50000, 200000, 500000]:
            if remaining_time() < 10:
                break
            h_ref = slsqp_refine(best_h, best_n, beta, maxiter=600)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Phase 4: Top-k equalization
    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(best_h, best_n, top_k=5, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Phase 5: Try a few more resolutions if time permits
    if remaining_time() > 40:
        extra_ns = [int(best_n * 1.5), int(best_n * 0.75), best_n + 64, best_n - 64]
        for n_e in extra_ns:
            if remaining_time() < 20:
                break
            if n_e % 2 != 0 or n_e < 128 or n_e > 1500:
                continue
            h_e = interp_to(best_h, best_n, n_e)
            h_e = project_feasible(h_e, n_e)
            c5_e = compute_c5(h_e)

            h_bin_e = (h_e > 0.5).astype(float)
            nh = n_e // 2
            no = int(np.sum(h_bin_e))
            if no != nh:
                idx = np.argsort(-h_bin_e)
                h_bin_e = np.zeros(n_e)
                h_bin_e[idx[:nh]] = 1.0

            sa_t = min(40, max(5, int(remaining_time() * 0.15)))
            h_sa_e, c5_sa_e = binary_sa(h_bin_e, n_e, time_limit=sa_t, rng=rng)

            for beta in [5000, 50000]:
                if remaining_time() < 8:
                    break
                h_sa_e = slsqp_refine(h_sa_e, n_e, beta, maxiter=300)
            c5_e_ref = compute_c5(h_sa_e)

            if c5_e_ref < best_c5:
                best_c5 = c5_e_ref
                best_h = h_sa_e.copy()
                best_n = n_e

    # Phase 6: Final polish
    if remaining_time() > 12:
        h_final = slsqp_refine(best_h, best_n, 1000000, maxiter=1000)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if remaining_time() > 8:
        h_eq2, c5_eq2 = top_k_equalize(best_h, best_n, top_k=6, max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
