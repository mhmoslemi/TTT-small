        #   - h[p-k] if 0<=p-k<n else 0          [p-k as i, h[p] was the (1-h[i+k]) factor]

        #   rem[p, n-1]: p as i: -(1-h[p]) = -0 = 0 (since h[p]=1). p-k=p: h[p]=1. 
        #   Wait, for k=0, the term i=p is h[p](1-h[p]) = 0. And i=p-k = p is the same term!
        #   So for k=0, the two contributions are the same term. We'd double count.
        #   But since corr at k=0 is always 0 for binary, and changing binary never makes it non-zero,
        #   we can just skip k=0.

        #   p as i: old = 0, new = 1*(1-h[p+k]). Δ = 1-h[p+k] (if 0<=p+k<n)
        #   p-k as i: old = h[p-k]*(1-0) = h[p-k], new = h[p-k]*(1-1) = 0. Δ = -h[p-k] (if 0<=p-k<n)

        # rem: (h[p+k]-1) + h[p-k] = -(1-h[p+k]) + h[p-k]
        # add: (1-h[p+k]) + (-h[p-k]) = (1-h[p+k]) - h[p-k]
        # So add = -rem. Yes!

        # Δcorr[j] = rem[a,j] + add[b,j] = rem[a,j] - rem[b,j]

        # then for each pair (a,b), the change is rem[a,:] - rem[b,:].

                k = j - (n - 1)
                if k == 0:
                    continue
                val = 0.0
                if 0 <= p + k < n:
                    val += h[p+k] - 1.0
                if 0 <= p - k < n:
                    val += h[p - k]
                rem[p, j] = val

        # new_corr = corr + rem[a,:] - rem[b,:]
        # new_max = max(new_corr)
        # We want to minimize new_max.
        
        # Vectorized: for each a, compute rem[a,:] - rem[b,:] for all b in zeros
        # Then take max along the j axis.
        
        # delta = rem[a][:,None] - rem[zeros][None,:]  shape (n2, n_zeros)
        # new_corr_max = np.max(corr[:,None] + delta, axis=0)  shape (n_zeros,)
        
        best_improve = 1e-15
        best_pair = None
        
        for ia in range(len(ones)):
            a = ones[ia]
            delta = rem[a][:, None] - rem[zeros][None, :]  # (n2, n_zeros)
            new_max = np.max(corr[:, None] + delta, axis=0)  # (n_zeros,)
            improve = c_max - new_max  # (n_zeros,)
            if improve.max() > best_improve:
                best_improve = improve.max()
                best_pair = (a, zeros[improve.argmax()])
            # Simulated annealing: check worst swap with probability
            if T > 1e-6:
                # Accept a random swap with prob exp(-Δ/T)
                rand_j = rng.randint(len(zeros))
                rand_improve = improve[rand_j]
                if rand_improve < 0:

                    if rng.random() < min(1.0, prob):
                        if best_improve <= 0 or rand_improve > best_improve:
                            # Actually for SA, we accept even if worse
                            best_improve = max(best_improve, rand_improve)
                            best_pair = (a, zeros[rand_j])
                    # Hmm, this logic is getting complicated. Let me simplify.
        
        if best_pair is None:
            # Try random swap for SA
            if T > 1e-6 and rng.random() < 0.3:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]
                if a != b:
                    best_pair = (a, b)
            if best_pair is None:
                T *= cooling
                continue
        
        a, b = best_pair
        h[a] = 0.0
        h[b] = 1.0
        corr = compute_corr(h)
        c5_new = float(np.max(corr) * dx)
        
        # Update rem table (it depends on h!)
        # Actually, rem depends on h because it uses h[p+k] and h[p-k].
        # After the swap, we need to update rem.
        # rem[p,j] only depends on h at positions p+k and p-k for each j.
        # Changing h[a] and h[b] affects rem at positions where p+k={a,b} or p-k={a,b}.
        # This is complex. Let's just recompute rem every few rounds.
        
        # For now, recompute rem every round (cost: O(n²) = 147k for n=384, fast)
        for p in range(n):
            for j in range(n2):
                k = j - (n - 1)
                if k == 0:
                    rem[p, j] = 0.0
                    continue

                if 0 <= p + k < n:
                    val += h[p+k] - 1.0
                if 0 <= p - k < n:
                    val += h[p-k]
                rem[p, j] = val

            best_c5 = c5_new
            best_h = h.copy()
        
        T *= cooling
    
    return best_h, best_c5

def compute_rem_table(h, n):
    n2 = 2 * n - 1
    rem = np.zeros((n, n2))
    # For position p and lag k: rem[p, (n-1)+k] = (h[p+k]-1 if 0<=p+k<n else 0) + (h[p-k] if 0<=p-k<n else 0)
    # Vectorize over p for fixed k, or use broadcasting.
    for k in range(-(n-1), n):
        if k == 0:
            continue
        j = (n-1) + k
        # p+k valid: 0 <= p+k < n => -k <= p < n-k
        pk_valid = (np.arange(n) + k >= 0) & (np.arange(n) + k < n)
        pk_val = np.where(pk_valid, h[np.arange(n) + k], 0.0) - 1.0  # h[p+k]-1
        pk_val = np.where(pk_valid, pk_val, 0.0)
        
        pk_minus_valid = (np.arange(n) - k >= 0) & (np.arange(n) - k < n)
        pk_minus_val = np.where(pk_minus_valid, h[np.arange(n) - k], 0.0)
        
        rem[:, j] = pk_val + pk_minus_val
    return rem

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def compute_rem_table(h, n):
    n2 = 2 * n - 1
    rem = np.zeros((n, n2))
    idx = np.arange(n)
    for k in range(-(n - 1), n):
        if k == 0:
            continue
        j = (n - 1) + k
        pk = idx + k
        valid_pk = (pk >= 0) & (pk < n)
        val1 = np.zeros(n)
        if valid_pk.any():
            val1[valid_pk] = h[pk[valid_pk]] - 1.0
        pk2 = idx - k
        valid_pk2 = (pk2 >= 0) & (pk2 < n)
        val2 = np.zeros(n)
        if valid_pk2.any():
            val2[valid_pk2] = h[pk2[valid_pk2]]
        rem[:, j] = val1 + val2
    return rem


def fast_binary_ls(h_init, n, max_rounds=2000, T0=0.5, cooling=0.999, seed=42):
    rng = np.random.RandomState(seed)
    h = (h_init > 0.5).astype(float).copy()
    if np.sum(h) != n / 2:
        h = round_to_binary(h, n)
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    T = T0
    n2 = 2 * n - 1
    rem = compute_rem_table(h, n)

    for rnd in range(max_rounds):
        if remaining() < 10:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        c_max = np.max(corr)
        best_improve = 1e-15
        best_pair = None

        for ia in range(len(ones)):
            a = ones[ia]
            delta = rem[a][:, None] - rem[zeros][None, :]
            new_max = np.maximum(corr[:, None] + delta, axis=0)
            improve = c_max - new_max
            best_idx = np.argmax(improve)
            if improve[best_idx] > best_improve:
                best_improve = improve[best_idx]
                best_pair = (a, zeros[best_idx])

            if T > 1e-4 and rnd < 500:
                rand_j = rng.randint(len(zeros))
                if improve[rand_j] < 0:
                    prob = min(1.0, np.exp(improve[rand_j] / (T * dx)))
                    if rng.random() < prob:
                        if best_improve < -1e10:
                            best_pair = (a, zeros[rand_j])
                            best_improve = improve[rand_j]

        if best_pair is None:
            if T > 1e-4 and rng.random() < 0.5:
                a = ones[rng.randint(len(ones))]
                b = zeros[rng.randint(len(zeros))]

                    best_pair = (a, b)

                continue

        if a == b:

def slsqp_polish(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(sh)
        w = ex / np.sum(ex) * dx
        g = np.zeros(n)
        n2 = 2 * n - 1
        for k in range(n2):
            if w[k] < 1e-16:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += w[k] * (1.0 - x[:np_])
                g[:np_] -= w[k] * x[off:off + np_]
            else:
                a = -off
                np_ = n - a
                g[:np_] += w[k] * (1.0 - x[a:a + np_])
                g[a:a + np_] -= w[k] * x[:np_]
        return g

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def make_periodic_pattern(period, ones_in_period, n, offset=0):
    h = np.zeros(n)
    pat = np.zeros(period)
    pat[:ones_in_period] = 1.0
    for i in range(n):
        h[i] = pat[(i + offset) % period]
    total = h.sum()
    target = n / 2.0
    if total != target:
        h = round_to_binary(h, n)
    return h


def make_thue_morse(n):
    tm = np.zeros(n)
    if n > 0:
        tm[0] = 1.0
        for i in range(1, n):
            c = bin(i).count('1')
            tm[i] = 1.0 if c % 2 == 0 else 0.0
    return round_to_binary(tm, n)


def make_modular_pattern(n, p, threshold):
    h = np.zeros(n)
    x = np.arange(n) / n * 2.0
    h = (x % p / p < threshold).astype(float)

            best_h = h_fb3.copy()
            best_n = n

    # Phase 5: SLSQP from best binary at multiple betas
    n = best_n
    h_cur = round_to_binary(best_h, n)
    for beta in [5000, 20000, 80000, 200000]:
        if remaining() < 15:
            break
        h_pol = slsqp_polish(h_cur, n, beta, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            h_cur = h_pol.copy()

    # Phase 6: Final binary refinement
    if remaining() > 30:
        h_bin_final = round_to_binary(best_h, best_n)
        h_fb4, c5_fb4 = fast_binary_ls(h_bin_final, best_n, max_rounds=3000, T0=0.05, cooling=0.9998, seed=seed + 999)
        if c5_fb4 < best_c5:
            best_c5 = c5_fb4
            best_h = h_fb4.copy()

    # Phase 7: One more SLSQP pass
    if remaining() > 15:
        h_final_pol = slsqp_polish(best_h, best_n, 100000, maxiter=400)
        c5_final = compute_c5(h_final_pol)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_pol.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-12:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
