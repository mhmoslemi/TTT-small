def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def _smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]


def _smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]


def _constraint_sum(x):
    return np.sum(x) - _N_WORK / 2.0


def _constraint_sum_jac(x):
    return np.ones(_N_WORK)


def slsqp_refine(h, n, beta, maxiter):

    _N_WORK = n
    _BETA_WORK = beta
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(_smooth_obj, h.copy(), jac=_smooth_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-30, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):

            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)

    except Exception:

    return h.copy()


def interp_to(src_h, src_n, dst_n):

def fast_coord_descent(h, n, max_sweeps=200, anneal_T=0.0, anneal_rate=0.98):
    """Fast coordinate descent with optional simulated annealing."""
    best = h.copy()
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    T = anneal_T

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)
            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0
            base = best_corr - A * old
            quad_base = best_corr[n - 1] - old * (1.0 - old)

            def _eval_f(t):
                mid = base + A * t
                mid[n - 1] = quad_base + t - t * t
                return np.max(mid)

            f_old = _eval_f(old)

            # Quick candidates
            t_best = old
            f_best = f_old
            for t_c in [0.0, 1.0, 0.5]:
                f_c = _eval_f(t_c)
                if f_c < f_best - 1e-14:
                    f_best = f_c
                    t_best = t_c

            # Golden section if we found improvement
            if t_best != old:
                if t_best > old:
                    lo, hi = old, min(1.0, t_best + 0.3)
                else:
                    lo, hi = max(0.0, t_best - 0.3), old
                if hi - lo < 1e-12:
                    hi = min(1.0, lo + 1e-8)
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = _eval_f(c1)
                f2 = _eval_f(c2)
                for _ in range(50):
                    if f1 < f2:
                        hi, c2, f2 = c2, c1, f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _eval_f(c1)
                    else:
                        lo, c1, f1 = c1, c2, f2
                        c2 = lo + gr * (hi - lo)
                        f2 = _eval_f(c2)
                t_best = (lo + hi) / 2.0
                f_best = _eval_f(t_best)

            # Annealing: accept worsening moves with probability
            accept = (f_best < f_old - 1e-14)
            if not accept and T > 1e-10 and f_best > f_old:
                import math
                delta = f_best - f_old
                if delta / T < 50.0:
                    if np.random.random() < math.exp(-delta / T):
                        accept = True

            if accept:
                if f_best < best_c5 / dx - 1e-14:
                    pass  # will update best_c5 below
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True
        T *= anneal_rate
        if not improved:
            break
    return best, best_c5


def pair_coord_descent(h, n, max_sweeps=30):
    """Optimize pairs of coordinates simultaneously via 2D search."""
    best = h.copy()
    best_corr = compute_corr(best)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(0, n - 1, 2):
            j = i + 1
            old_i, old_j = best[i], best[j]

            i_plus = i + offsets
            i_minus = i - offsets
            vp = (i_plus >= 0) & (i_plus < n)
            vm = (i_minus >= 0) & (i_minus < n)
            A_i = np.zeros(ncorr)
            if np.any(vm):
                A_i[vm] += (1.0 - best[i_minus[vm]])
            if np.any(vp):
                A_i[vp] -= best[i_plus[vp]]
            A_i[n - 1] = 0.0

            j_plus = j + offsets
            j_minus = j - offsets
            vp_j = (j_plus >= 0) & (j_plus < n)
            vm_j = (j_minus >= 0) & (j_minus < n)
            A_j = np.zeros(ncorr)
            if np.any(vm_j):
                A_j[vm_j] += (1.0 - best[j_minus[vm_j]])
            if np.any(vp_j):
                A_j[vp_j] -= best[j_plus[vp_j]]
            A_j[n - 1] = 0.0

            f_old = np.max(best_corr)
            best_ti, best_tj = old_i, old_j
            best_f = f_old

            # Coarse grid
            for ti in [0.0, old_i, 1.0]:
                for tj in [0.0, old_j, 1.0]:
                    nc = best_corr + A_i * (ti - old_i) + A_j * (tj - old_j)
                    nc[n - 1] += ti * (1 - ti) - old_i * (1 - old_i)
                    nc[n - 1] += tj * (1 - tj) - old_j * (1 - old_j)
                    f_new = np.max(nc)
                    if f_new < best_f - 1e-14:
                        best_f = f_new
                        best_ti, best_tj = ti, tj

            # Refine
            if best_ti != old_i or best_tj != old_j:
                best_ti_c, best_tj_c = best_ti, best_tj
                step_i, step_j = 0.1, 0.1
                for _ in range(40):
                    found = False
                    for di in [step_i, -step_i, 0.0]:
                        for dj in [step_j, -step_j, 0.0]:
                            ti = best_ti_c + di
                            tj = best_tj_c + dj
                            if 0.0 <= ti <= 1.0 and 0.0 <= tj <= 1.0:
                                nc = best_corr + A_i * (ti - old_i) + A_j * (tj - old_j)
                                nc[n - 1] += ti * (1 - ti) - old_i * (1 - old_i)
                                nc[n - 1] += tj * (1 - tj) - old_j * (1 - old_j)
                                f_new = np.max(nc)
                                if f_new < best_f - 1e-15:
                                    best_f = f_new
                                    best_ti, best_tj = ti, tj
                                    found = True
                    if not found:
                        step_i *= 0.5
                        step_j *= 0.5
                        if step_i < 1e-6:
                            break

            if best_ti != old_i or best_tj != old_j:
                best[i] = best_ti
                best[j] = best_tj
                best_corr += A_i * (best_ti - old_i) + A_j * (best_tj - old_j)
                best_corr[n - 1] += best_ti * (1 - best_ti) - old_i * (1 - old_i)
                best_corr[n - 1] += best_tj * (1 - best_tj) - old_j * (1 - old_j)
                best_c5 = np.max(best_corr) * dx
                improved = True
        if not improved:
            break
    return best, best_c5


def binary_push(h, n, frac=0.0):
    """Push values near 0 or 1 to be exactly 0 or 1."""
    hp = h.copy()
    mask_low = hp < frac + 0.01
    mask_high = hp > 1.0 - frac - 0.01
    hp[mask_low] = 0.0
    hp[mask_high] = 1.0
    return project_feasible(hp, n)

        parent_n = len(parent_h)

        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        h_init = np.zeros(best_n)
        ones = rng.choice(best_n, best_n // 2, replace=False)
        h_init[ones] = 1.0
        best_h = h_init
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Intensive coordinate descent with mild annealing
    h = best_h.copy()
    h, c5 = fast_coord_descent(h, n, max_sweeps=200, anneal_T=0.003, anneal_rate=0.975)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    # Phase 2: Pair coordinate descent
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = pair_coord_descent(h, n, max_sweeps=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 3: SLSQP with increasing beta
    if remaining_time() > 60:
        h = best_h.copy()
        for beta in [300, 1500, 8000, 40000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_refine(h, n, beta, 500)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-15:
                h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    # Phase 4: Coordinate descent after SLSQP
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Multi-resolution - downsample to n/2, optimize, upsample
    if remaining_time() > 150:
        n_low = n // 2
        h_low = interp_to(best_h, n, n_low)
        h_low = project_feasible(h_low, n_low)
        h_low, c5_low = fast_coord_descent(h_low, n_low, max_sweeps=150, anneal_T=0.002, anneal_rate=0.97)
        if remaining_time() > 60:
            h_low = slsqp_refine(h_low, n_low, 5000, 400)
            c5_low = compute_c5(h_low)
        # Upsample and refine
        h_high = interp_to(h_low, n_low, n)
        h_high = project_feasible(h_high, n)
        h_high, c5_high = fast_coord_descent(h_high, n, max_sweeps=80)
        if remaining_time() > 20:
            h_high = slsqp_refine(h_high, n, 10000, 300)
            c5_high = compute_c5(h_high)
        h_high, c5_high = fast_coord_descent(h_high, n, max_sweeps=50)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high.copy()

    # Phase 6: Diverse restarts with different perturbations
    if remaining_time() > 100:
        for p in range(12):
            if remaining_time() < 25:
                break
            h_pert = best_h.copy()
            noise_scale = 0.2 / (1.0 + p * 0.4)
            frac_perturb = 0.25 / (1.0 + p * 0.2)
            mask = rng.random(n) < frac_perturb
            h_pert[mask] += rng.randn(int(np.sum(mask))) * noise_scale
            h_pert = project_feasible(h_pert, n)

            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=80, anneal_T=0.001, anneal_rate=0.97)
            if remaining_time() > 15:
                h_new = slsqp_refine(h_new, n, 5000, 300)
                c5_new = compute_c5(h_new)
            h_new, c5_new2 = fast_coord_descent(h_new, n, max_sweeps=40)
            c5_new = min(c5_new, c5_new2)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 7: Try higher resolution
    if remaining_time() > 120:
        n_high = n + 160  # 800
        if n_high > 0:
            h_high = interp_to(best_h, n, n_high)
            h_high = project_feasible(h_high, n_high)
            h_high, c5_high = fast_coord_descent(h_high, n_high, max_sweeps=60, anneal_T=0.002, anneal_rate=0.98)
            if remaining_time() > 40:
                h_high = slsqp_refine(h_high, n_high, 8000, 300)
                c5_high = compute_c5(h_high)
            h_high, c5_high2 = fast_coord_descent(h_high, n_high, max_sweeps=40)
            c5_high = min(c5_high, c5_high2)
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high.copy()
                best_n = n_high

    # Phase 8: Binary push + re-optimize (find structure)
    if remaining_time() > 60:
        for threshold in [0.05, 0.1, 0.15]:
            if remaining_time() < 15:
                break
            h_bin = binary_push(best_h, n, frac=threshold)
            h_bin, c5_bin = fast_coord_descent(h_bin, n, max_sweeps=80)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                if remaining_time() > 20:
                    h_bin = slsqp_refine(h_bin, n, 10000, 300)
                    c5_bin = compute_c5(h_bin)
                    if c5_bin < best_c5:
                        best_c5 = c5_bin
                        best_h = h_bin.copy()

    # Phase 9: Try n=512 and n=768
    for n_try in [n - 128, n + 128]:
        if remaining_time() < 40 or n_try < 256:
            continue
        h_try = interp_to(best_h, best_n, n_try)
        h_try = project_feasible(h_try, n_try)
        h_try, c5_try = fast_coord_descent(h_try, n_try, max_sweeps=60, anneal_T=0.002, anneal_rate=0.98)
        if remaining_time() > 20:
            h_try = slsqp_refine(h_try, n_try, 5000, 300)
            c5_try = compute_c5(h_try)
        h_try, c5_try2 = fast_coord_descent(h_try, n_try, max_sweeps=40)
        c5_try = min(c5_try, c5_try2)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try

    # Phase 10: Final intensive polishing
    n_cur = best_n
    h = best_h.copy()
    h, c5 = fast_coord_descent(h, n_cur, max_sweeps=150, anneal_T=0.001, anneal_rate=0.99)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

    if remaining_time() > 10:
        for beta in [50000, 200000]:
            if remaining_time() < 5:
                break
            h_new = slsqp_refine(best_h, best_n, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final coordinate descent
    if remaining_time() > 5:
        h_new, c5_new = fast_coord_descent(best_h, best_n, max_sweeps=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
