def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -60.0, 60.0)
    ex = np.exp(sh)
    s = np.sum(ex)
    dx = 2.0 / len(h)
    return (m + np.log(s) / beta) * dx


def smooth_max_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -60.0, 60.0)
    ex = np.exp(sh)
    s = np.sum(ex)
    w = ex / s
    nx = len(h)
    g = np.zeros(nx)
    for k in range(2 * nx - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        off = k - (nx - 1)
        if off >= 0:
            np_ = nx - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:

            np_ = nx - a

    return g * (2.0 / nx)

        return smooth_max_grad(x, beta)

        return np.sum(x) - n / 2.0

        return np.ones(n)

def periodic_init(n, period, rng):

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
