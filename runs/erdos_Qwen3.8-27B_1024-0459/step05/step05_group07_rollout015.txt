def project_feasible(h, n):

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = len(corr)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset

                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def thue_morse_binary(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h


def interp_to(src_h, src_n, dst_n):

    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

            A_j[n - 1] = 0.0

            base = best_corr - A_i * old_i - A_j * old_j
            base[n - 1] = best_corr[n - 1] - old_i * (1 - old_i) - old_j * (1 - old_j)

            f_old = np.max(best_corr)
            best_ti, best_tj = old_i, old_j
            best_f = f_old

            # Coarse 2D grid
            for ti in [0.0, 0.33, 0.5, 0.67, 1.0]:
                for tj in [0.0, 0.33, 0.5, 0.67, 1.0]:
                    vals = base + A_i * ti + A_j * tj
                    vals[n - 1] = base[n - 1] + ti * (1 - ti) + tj * (1 - tj)
                    f = np.max(vals)
                    if f < best_f - 1e-15:
                        best_f = f
                        best_ti, best_tj = ti, tj

            if best_ti != old_i or best_tj != old_j:
                # 2D Nelder-Mead style refinement
                ti, tj = best_ti, best_tj
                for _ in range(20):
                    improved_step = False
                    for dti in [-0.1, 0.1, -0.03, 0.03]:
                        for dtj in [-0.1, 0.1, -0.03, 0.03]:
                            tti = ti + dti
                            ttj = tj + dtj
                            if 0 <= tti <= 1 and 0 <= ttj <= 1:
                                vals = base + A_i * tti + A_j * ttj
                                vals[n - 1] = base[n - 1] + tti*(1-tti) + ttj*(1-ttj)
                                f = np.max(vals)
                                if f < best_f - 1e-14:
                                    best_f = f
                                    best_ti, best_tj = tti, ttj
                                    ti, tj = tti, ttj
                                    improved_step = True
                    if not improved_step:
                        break

            if best_ti != old_i or best_tj != old_j:
                best[i] = best_ti
                best[j] = best_tj
                best_corr += A_i * (best_ti - old_i) + A_j * (best_tj - old_j)
                best_corr[n - 1] += best_ti*(1-best_ti) + best_tj*(1-best_tj) - old_i*(1-old_i) - old_j*(1-old_j)
                improved = True

        if not improved:
            break

    return best, np.max(best_corr) * dx


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_binary(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n

    # Phase 1: Coordinate descent from parent
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = coord_descent(h, n, max_sweeps=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Binary SA (Thue-Morse init at n=640)
    if remaining_time() > 120:
        h_tm = thue_morse_binary(n)
        h_tm_b = (h_tm > 0.5).astype(float)
        # Ensure n/2 ones
        n_ones = int(np.sum(h_tm_b))
        if n_ones > n // 2:
            ones_idx = np.where(h_tm_b == 1)[0]
            h_tm_b[ones_idx[n // 2:]] = 0
        elif n_ones < n // 2:
            zeros_idx = np.where(h_tm_b == 0)[0]
            h_tm_b[zeros_idx[:n // 2 - n_ones]] = 1

        h_bin, c5_bin = binary_sa(h_tm_b, n, rng, max_iters=2000000)
        # Evaluate binary solution
        c5_bin_actual = compute_c5(h_bin)
        if c5_bin_actual < best_c5:
            best_c5 = c5_bin_actual
            best_h = h_bin.copy()

        # Phase 2b: Relax best binary and coordinate descent
        if remaining_time() > 60:
            h_relax = binary_to_continuous(h_bin, n, rng, fraction_relax=0.05)
            h_relax, c5_relax = coord_descent(h_relax, n, max_sweeps=150)
            if c5_relax < best_c5:
                best_c5 = c5_relax
                best_h = h_relax.copy()

    # Phase 3: SLSQP with increasing beta on best continuous
    if remaining_time() > 40:
        h = best_h.copy()
        for beta in [1000, 5000, 20000, 80000]:
            if remaining_time() < 20:
                break
            h_new = slsqp_refine(h, n, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-14:
                h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    # Phase 4: More coordinate descent sweeps
    if remaining_time() > 40:
        h = best_h.copy()
        h, c5 = coord_descent(h, n, max_sweeps=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Pair coordinate descent
    if remaining_time() > 60:
        h = best_h.copy()
        h, c5 = pair_coord_descent(h, n, max_sweeps=30)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Follow up with single coord descent
        if remaining_time() > 30:
            h, c5 = coord_descent(h, n, max_sweeps=100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 6: Random perturbations + coordinate descent
    if remaining_time() > 80:
        for p in range(6):
            if remaining_time() < 30:
                break
            h_pert = best_h.copy()
            noise_scale = 0.1 / (1.0 + p * 0.5)
            mask = rng.random(n) < 0.2
            if np.sum(mask) > 0:
                h_pert[mask] += rng.randn(int(np.sum(mask))) * noise_scale
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = coord_descent(h_pert, n, max_sweeps=80)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 20:
                    h_slq = slsqp_refine(h_new, n, 10000, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()

    # Phase 7: Try different resolution n=512
    if remaining_time() > 120:
        n2 = 512
        h2 = project_feasible(thue_morse_binary(n2), n2)
        # Binary SA at n=512
        h2_b = (h2 > 0.5).astype(float)
        h2_sa, c5_2sa = binary_sa(h2_b, n2, rng, max_iters=1500000)
        c5_2sa_actual = compute_c5(h2_sa)
        if c5_2sa_actual < best_c5:
            best_c5 = c5_2sa_actual
            best_h = h2_sa.copy()
            best_n = n2
        # Continuous from 512
        if remaining_time() > 60:
            h2_relax = binary_to_continuous(h2_sa, n2, rng, fraction_relax=0.05)
            h2_relax, c5_2r = coord_descent(h2_relax, n2, max_sweeps=100)
            if c5_2r < best_c5:
                best_c5 = c5_2r
                best_h = h2_relax.copy()
                best_n = n2

    # Phase 8: Interpolate best to higher resolution and optimize
    if remaining_time() > 100 and best_n < 1280:
        n3 = best_n * 2
        if n3 <= 1280:
            h3 = interp_to(best_h, best_n, n3)
            h3 = project_feasible(h3, n3)
            h3, c5_3 = coord_descent(h3, n3, max_sweeps=100)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n3
            if remaining_time() > 30:
                h3_slq = slsqp_refine(h3, n3, 5000, 300)
                c5_3slq = compute_c5(h3_slq)
                if c5_3slq < best_c5:
                    best_c5 = c5_3slq
                    best_h = h3_slq.copy()
                    best_n = n3

    # Phase 9: Final SLSQP polish with very high beta
    if remaining_time() > 15:
        n_cur = best_n
        h = best_h.copy()
        for beta in [50000, 200000, 500000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_refine(h, n_cur, beta, 300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 15:
        h_new, c5_new = coord_descent(best_h, best_n, max_sweeps=150)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final SLSQP
    if remaining_time() > 10:
        h_new = slsqp_refine(best_h, best_n, 100000, 200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, n_final)
