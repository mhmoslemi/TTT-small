    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2*(n-1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = j
    hi3 = min(j + n - 1, r - 1)
    if lo3 <= hi3:
        ks = np.arange(lo3, hi3 + 1)
        dr[lo3:hi3+1] += (1.0 - h[j + n - 1 - ks])
    lo4 = max(n-1-j, 0)
    hi4 = min(2*(n-1)-j, r-1)
    if lo4 <= hi4:
        ks = np.arange(lo4, hi4+1)
        dr[lo4:hi4+1] -= h[j + ks - (n-1)]
    dr[n-1] -= 2.0
    return dr

def binary_local(h_init, n, time_limit, rng, aggressive=False):
    h = (h_init > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(h.sum())
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h==1)[0]
            h[rng.choice(ones, n_ones-n_half, replace=False)] = 0
        else:
            zeros = np.where(h==0)[0]
            h[rng.choice(zeros, n_half-n_ones, replace=False)] = 1
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    start = time.time()
    stall = 0
    T = 0.3
    T_min = 0.02
    cool = 0.9997
    max_stall = 150 if not aggressive else 80

    while time.time() - start < time_limit and remaining() > 8:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        candidates = []
        n_top = min(12, len(corr))
        top_idx = np.argpartition(-corr, n_top)[:n_top]
        for s in top_idx:
            off = s - (n-1)
            if off >= 0:
                np_ = n - off
                contrib = np.where((h[off:off+np_] == 1) & (h[:np_] == 0))[0]
                for ci in contrib[:25]:
                    candidates.append((int(ci+off), int(ci)))
            else:
                ao = -off
                np_ = n - ao
                contrib = np.where((h[:np_] == 1) & (h[ao:ao+np_] == 0))[0]
                for ci in contrib[:25]:
                    candidates.append((int(ci), int(ci+ao)))
        nr = 40
        ri = rng.randint(len(ones), nr)
        rj = rng.randint(len(zeros), nr)
        for a in range(nr):
            candidates.append((int(ones[ri[a]]), int(zeros[rj[a]])))
        best_new_max = cur_max + 1e-15
        best_i = -1
        best_j = -1
        best_dr = None
        seen = set()
        for (i, j) in candidates:
            if i == j or h[i] != 1 or h[j] != 0:
                continue
            key = (i, j)
            if key in seen:
                continue
            seen.add(key)
            dr = swap_delta(h, i, j, n)
            nm = corr + dr
            nmx = float(np.max(nm))
            if nmx < best_new_max:
                best_new_max = nmx
                best_i = i
                best_j = j
                best_dr = dr
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = corr + best_dr
            cur_max = best_new_max
            stall = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            stall += 1
            if stall <= 3 or (stall <= max_stall and rng.rand() < 0.3):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                if h[i]==1 and h[j]==0:
                    dr = swap_delta(h, i, j, n)
                    nm = corr + dr
                    nmx = float(np.max(nm))
                    delta = nmx - cur_max
                    if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
                        h[i] = 0
                        h[j] = 1
                        corr = nm
                        cur_max = nmx
                        if cur_max < best_max:
                            best_max = cur_max
                            best_h = h.copy()
                        stall = 0
            elif stall > max_stall:
                n_pert = 3 if not aggressive else 6
                for _ in range(n_pert):
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    if h[i]==1 and h[j]==0:
                        dr = swap_delta(h, i, j, n)
                        h[i] = 0
                        h[j] = 1
                        corr = corr + dr
                cur_max = float(np.max(corr))
                stall = 0
                ones = np.where(h==1)[0]
                zeros = np.where(h==0)[0]
        T = max(T_min, T * cool)
    return best_h, best_max * dx

def slsqp_smooth(h, n, beta, maxiter=300):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _TGT,
             'jac': lambda x: np.ones(_N)}]
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / _BETA) * (2.0 / n)
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(_BETA * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2*n-1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao+np_])
                g[ao:ao+np_] -= wk * x[:np_]
        return g
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 384
        h = np.zeros(best_n)
        for i in range(best_n):
            if bin(i).count('1') % 2 == 0:
                h[i] = 1.0
        n_ones = int(h.sum())
        tgt = best_n // 2
        if n_ones != tgt:
            if n_ones > tgt:
                idx = np.where(h==1)[0]
                h[idx[:n_ones-tgt]] = 0
            else:
                idx = np.where(h==0)[0]
                h[idx[:tgt-n_ones]] = 1
        best_h = h
        best_c5 = compute_c5(h)

    n = best_n

    # Phase 1: Main binary local search
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    if int(h_bin.sum()) != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0
    t1 = min(450, int(remaining() * 0.35))
    if t1 > 30:
        h_b1, c5_b1 = binary_local(h_bin, n, t1, rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()

    # Phase 2: Second binary pass (refine further)
    if remaining() > 80:
        t2 = min(250, int(remaining() * 0.2))
        h_b2, c5_b2 = binary_local(best_h, n, t2, rng, aggressive=True)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 3: Try nearby resolutions
    resolutions = []
    for delta in [-16, -8, 8, 16, 24, 32, -32]:
        nt = best_n + delta
        if 128 <= nt <= 768 and nt % 2 == 0:
            resolutions.append(nt)
    resolutions = list(set(resolutions))
    resolutions.sort(key=lambda x: abs(x - best_n))

    for n_t in resolutions[:4]:
        if remaining() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(h_bin_t.sum())
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t==1)[0]
                h_bin_t[idx[:nt_ones-nt_half]] = 0
            else:
                idx = np.where(h_bin_t==0)[0]
                h_bin_t[idx[:nt_half-nt_ones]] = 1
        t3 = min(70, int(remaining() * 0.06))
        if t3 > 10:
            h_b3, c5_b3 = binary_local(h_bin_t, n_t, t3, rng)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
                best_n = n_t

    # Phase 4: Continuous refinement
    n = best_n
    h = best_h.copy()
    for beta in [100, 500, 2000, 10000]:
        if remaining() < 15:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 5: Final binary polish from continuous solution
    if remaining() > 50:
        h_bin5 = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        if int(h_bin5.sum()) != n_half:
            sort_idx = np.argsort(-best_h)
            h_bin5 = np.zeros(best_n)
            h_bin5[sort_idx[:n_half]] = 1.0
        t5 = min(100, int(remaining() * 0.08))
        h_b5, c5_b5 = binary_local(h_bin5, best_n, t5, rng)
        if c5_b5 < best_c5:
            best_c5 = c5_b5
            best_h = h_b5.copy()

    # Phase 6: Final continuous polish
    if remaining() > 15:
        h = best_h.copy()
        for beta in [20000, 50000]:
            if remaining() < 10:
                break
            h_new = slsqp_smooth(h, best_n, beta, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        best_c5 = min(best_c5, eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
