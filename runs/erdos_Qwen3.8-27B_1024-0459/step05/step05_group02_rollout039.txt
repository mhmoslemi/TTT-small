def compute_corr(h):

def corr_grad_single(h, k, n):
    offset = k - (n - 1)

        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def compute_c5(h):

    return np.correlate(h, 1.0 - h, mode='full')

def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

        nt = min(n_try, len(ones), len(zeros))
        # Most negative gradient for ones (we want to remove them)
        one_scores = -g[ones]  # higher = more negative g = better to remove
        zero_scores = g[zeros]  # higher = more positive g = better to add
        top_ones = ones[np.argsort(-one_scores)[:nt]]
        top_zeros = zeros[np.argsort(-zero_scores)[:nt]]

        for i in top_ones:
            for j in top_zeros:
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                h_bin[i] = 1
                h_bin[j] = 0
                delta = nc - cur_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_move = (i, j, True)  # True = single swap

        # 2-swaps if no single swap found
        if use_2swap and best_move is None:
            nt2 = min(4, len(ones), len(zeros))
            top_ones2 = ones[np.argsort(-one_scores)[:nt2]]
            top_zeros2 = zeros[np.argsort(-zero_scores)[:nt2]]
            for a in range(nt2):
                for b in range(a+1, nt2):
                    for c in range(nt2):
                        for d in range(c+1, nt2):
                            i1, i2 = top_ones2[a], top_ones2[b]
                            j1, j2 = top_zeros2[c], top_zeros2[d]
                            if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                                continue
                            h_bin[i1] = 0
                            h_bin[i2] = 0
                            h_bin[j1] = 1
                            h_bin[j2] = 1
                            nc = compute_c5(h_bin)
                            h_bin[i1] = 1
                            h_bin[i2] = 1
                            h_bin[j1] = 0
                            h_bin[j2] = 0
                            delta = nc - cur_c5
                            if delta < best_delta - 1e-15:
                                best_delta = delta
                                best_move = (i1, i2, j1, j2, False)  # False = double swap

        if best_move is not None:
            if best_move[2] if len(best_move) == 3 else (not best_move[4]):
                # single swap
                i, j = best_move[0], best_move[1]
                h_bin[i] = 0
                h_bin[j] = 1
            else:
                i1, i2, j1, j2 = best_move[0], best_move[1], best_move[2], best_move[3]
                h_bin[i1] = 0
                h_bin[i2] = 0
                h_bin[j1] = 1
                h_bin[j2] = 1
            cur_c5 = compute_c5(h_bin)
            no_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_bin.copy()
        else:
            no_improve += 1
            # Random restart attempt
            found = False
            for _ in range(10):
                i = int(ones[np.random.randint(len(ones))])
                j = int(zeros[np.random.randint(len(zeros))])
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                nc = compute_c5(h_bin)
                delta = nc - cur_c5
                # Accept with probability based on improvement
                if delta < 0 or (no_improve > 50 and np.random.random() < 0.1):
                    if delta < 0:
                        cur_c5 = nc
                        if cur_c5 < best_c5:
                            best_c5 = cur_c5
                            best_h = h_bin.copy()
                        no_improve = 0
                        found = True
                        break
                    else:
                        h_bin[i] = 1
                        h_bin[j] = 0
            if not found and no_improve > 200:
                break

    return best_h, best_c5

def simulated_annealing(h_init, n, max_iters=10000, T_init=0.02, T_min=0.001):
    """Simulated annealing for binary sequences."""
    h = (h_init > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        sort_idx = np.argsort(-h)
        h = np.zeros(n)
        h[sort_idx[:n_half]] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T = T_init
    cool = (T_min / T_init) ** (1.0 / max(max_iters, 1))

    for it in range(max_iters):
        if remaining() < 5:
            break
        T *= cool

        # Choose a guided random swap
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:3]
        g = np.zeros(n)
        for ki in idx_top:
            g += corr_grad_single(h, ki, n)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Weighted random selection based on gradient
        weights_ones = np.maximum(-g[ones], 0.01)
        weights_zeros = np.maximum(g[zeros], 0.01)
        weights_ones /= weights_ones.sum()
        weights_zeros /= weights_zeros.sum()

        i = int(ones[np.random.choice(len(ones), p=weights_ones)])
        j = int(zeros[np.random.choice(len(zeros), p=weights_zeros)])
        if i == j:
            continue

        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5

        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0

    return best_h, best_c5
