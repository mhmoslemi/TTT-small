                        val -= best[i + m]
                A[k] = val
            
            # But wait, at offset=0 (k = n-1):
            # corr[n-1] = sum_j h[j]*(1-h[j])
            # h[i] contributes: (1-h[i]) - h[i] = 1 - 2*h[i]
            # But this is linear in h[i], not quadratic!
            # Wait: h[i]*(1-h[i]) = h[i] - h[i]^2
            # d/dh[i]: 1 - 2*h[i]
            # So the contribution to corr[n-1] from changing h[i] by dt is (1-2*old)*dt + (-1)*dt^2
            # Hmm, this is quadratic in the new value t:
            # New corr[n-1] = old_corr[n-1] - old*(1-old) + t*(1-t)
            # = old_corr[n-1] + t - t^2 - old + old^2
            # = (old_corr[n-1] - old + old^2) + t - t^2
            
            # For other k (offset ≠ 0), the change is linear in t.
            # So A[n-1] should be 0 (we handle the quadratic separately)
            
            A[n - 1] = 0.0
            
            # New corr at value t:
            # For k ≠ n-1: corr_new[k] = corr_old[k] + A[k] * (t - old)
            # For k = n-1: corr_new[n-1] = corr_old[n-1] - old*(1-old) + t*(1-t)
            #               = (corr_old[n-1] - old + old^2) + t - t^2
            
            base = best_corr.copy() - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)
            
            # Objective: f(t) = max(max_{k≠n-1}(base[k] + A[k]*t), base[n-1] + t - t^2)
            # This is convex in t on [0,1]
            
            def f_of_t(t):
                val_linear = np.max(base + A * t)  # includes k=n-1 but A[n-1]=0, base[n-1] is the constant
                val_quad = base[n - 1] + t - t * t
                return max(val_linear, val_quad)
            
            # Evaluate at current
            f_old = f_of_t(old)
            
            # Try some candidates
            best_t = old
            best_f = f_old
            
            # Golden section search on [0, 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo = 0.0
            hi = 1.0
            
            # First check boundaries and midpoint
            f_lo = f_of_t(0.0)
            f_hi = f_of_t(1.0)
            f_mid = f_of_t(0.5)
            
            if f_lo < best_f - 1e-14:
                best_t = 0.0
                best_f = f_lo
            if f_hi < best_f - 1e-14:
                best_t = 1.0
                best_f = f_hi
            if f_mid < best_f - 1e-14:
                best_t = 0.5
                best_f = f_mid
            
            # Narrow the search interval
            if best_t > 0.1 and best_t < 0.9:
                lo = max(0.0, best_t - 0.3)
                hi = min(1.0, best_t + 0.3)
            else:
                if best_t == 0:
                    hi = 0.4
                elif best_t == 1:
                    lo = 0.6
                else:
                    if best_t < 0.5:
                        hi = 0.8
                    else:
                        lo = 0.2
            
            if hi - lo > 1e-10:
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                f1 = f_of_t(c1)
                f2 = f_of_t(c2)
                for _ in range(50):
                    if hi - lo < 1e-12:
                        break
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = f_of_t(c1)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = f_of_t(c2)
                
                t_star = (lo + hi) / 2.0
                f_star = f_of_t(t_star)
                if f_star < best_f - 1e-14:
                    best_t = t_star
                    best_f = f_star
            
            if best_f < f_old - tol:
                delta = best_t - old
                best[i] = best_t
                best_corr += A * delta
                best_corr[n - 1] = base[n - 1] + best_t - best_t * best_t
                improved = True
        
        if not improved:
            break
    
    return best, np.max(best_corr) * dx

    ncorr = 2 * n - 1
    A = np.zeros(ncorr)
    
    # Positive offsets: k = n-1+offset, offset = 0, ..., n-1
    # A[n-1+offset] += (1-h[i-offset]) if offset <= i
    # A[n-1+offset] += -h[i+offset] if offset <= n-1-i

from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_WORK = 0
_BETA_WORK = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        order = np.argsort(np.abs(h - 0.5))
        remaining = diff
        for idx in order:
            if abs(remaining) < 1e-14:
                break
            if remaining > 0:
                adj = min(remaining, 1.0 - h[idx])
            else:
                adj = max(remaining, -h[idx])
            h[idx] += adj
            remaining -= adj
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 2.0)

def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _g_smooth_obj(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[0]

def _g_smooth_obj_grad(x):
    return smooth_obj_and_grad(x, _N_WORK, _BETA_WORK)[1]

def _g_const_sum(x):
    return np.sum(x) - _N_WORK / 2.0

def _g_const_sum_jac(x):
    return np.ones(_N_WORK)

def slsqp_refine(h, n, beta, maxiter):
    global _N_WORK, _BETA_WORK
    _N_WORK = n
    _BETA_WORK = float(beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_const_sum, 'jac': _g_const_sum_jac}]
    try:
        res = minimize(_g_smooth_obj, h, jac=_g_smooth_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-25, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def compute_A(best, i, n):
    """Compute the linear coefficient vector A for coordinate i."""
    ncorr = 2 * n - 1
    A = np.zeros(ncorr)
    
    # Positive offsets: k = (n-1) + offset, offset = 0, ..., n-1
    # A[n-1+off] += (1 - best[i-off]) for off <= i
    # A[n-1+off] += -best[i+off] for off <= n-1-i
    
    mf = min(i, n - 1)
    if mf >= 0:
        offs = np.arange(mf + 1)
        A[n - 1 + offs] += (1.0 - best[i - offs])
    
    ms = min(n - 1 - i, n - 1)
    if ms >= 0:
        offs = np.arange(ms + 1)
        A[n - 1 + offs] -= best[i + offs]
    
    # Negative offsets: k = (n-1) - m, m = 1, ..., n-1
    # A[n-1-m] += (1 - best[i-m]) for m <= i
    # A[n-1-m] += -best[i+m] for m <= n-1-i
    
    mnf = min(i, n - 1)
    if mnf >= 1:
        ms_arr = np.arange(1, mnf + 1)
        A[n - 1 - ms_arr] += (1.0 - best[i - ms_arr])
    
    mns = min(n - 1 - i, n - 1)
    if mns >= 1:
        ms_arr = np.arange(1, mns + 1)
        A[n - 1 - ms_arr] -= best[i + ms_arr]
    
    A[n - 1] = 0.0
    return A
