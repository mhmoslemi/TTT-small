def compute_grad_matrix(h, n):
    """Compute G[i,k] = ∂corr[k]/∂h[i] for all i,k."""
    G = np.zeros((n, 2*n-1))
    for offset in range(-(n-1), n):
        k = n - 1 + offset
        if offset >= 0:
            # (1-h[i-offset]) for i in [offset, n-1]
            idx = np.arange(offset, n)
            G[idx, k] += (1.0 - h[idx - offset])
            # -h[i+offset] for i in [0, n-1-offset]
            idx2 = np.arange(0, n-offset)
            G[idx2, k] -= h[idx2 + offset]
        else:
            a = -offset
            # (1-h[i+a]) for i in [0, n-1-a]
            idx = np.arange(0, n-a)
            G[idx, k] += (1.0 - h[idx + a])
            # -h[i-a] for i in [a, n-1]
            idx2 = np.arange(a, n)
            G[idx2, k] -= h[idx2 - a]
    return G

        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12 and n > 0:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def to_binary_feasible(h, n, rng):
    """Round h to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    # Normalize: find threshold
    h_sorted = np.sort(h)[::-1]
    # We need exactly n//2 ones
    n_ones = n // 2
    # Take top n_ones entries and set to 1
    h_bin = np.zeros(n)
    if n_ones > 0:
        top_idx = np.argsort(h)[::-1][:n_ones]
        h_bin[top_idx] = 1.0
    # Verify sum
    diff = n/2.0 - np.sum(h_bin)
    if abs(diff) > 1e-10:
        # Shouldn't happen with even n, but handle gracefully
        h_bin = project_feasible(h_bin, n)
    return h_bin

def compute_grad_at_lag(h, offset, n):
    """Compute g_i = d(corr[n-1+offset])/d(h[i]) for given offset."""
    g = np.zeros(n)
    if offset >= 0:
        idx = np.arange(offset, n)
        g[idx] += (1.0 - h[idx - offset])
        idx2 = np.arange(0, n - offset)
        g[idx2] -= h[idx2 + offset]
    else:
        a = -offset
        idx = np.arange(0, n - a)
        g[idx] += (1.0 - h[idx + a])
        idx2 = np.arange(a, n)
        g[idx2] -= h[idx2 - a]
    return g

def fast_swap_descent(h, n, max_iters=20000, multi_k=1):
    """Fast swap-based coordinate descent for nearly-binary h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    h = h.copy()
    
    ones_mask = h > 0.5
    zeros_mask = h < 0.5
    ones_set = set(np.where(ones_mask)[0].tolist())
    zeros_set = set(np.where(zeros_mask)[0].tolist())
    
    if len(ones_set) == 0 or len(zeros_set) == 0:
        return h, best_c5
    
    stagnation = 0
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        
        # Find top multi_k lags
        top_lags = np.argsort(corr)[::-1][:multi_k]
        
        best_gain = 0
        best_i = -1
        best_j = -1
        
        for k_star in top_lags:
            offset = int(k_star) - (n - 1)
            g = compute_grad_at_lag(h, offset, n)
            
            # Find i in ones with max g[i], j in zeros with min g[j]
            if len(ones_set) == 0 or len(zeros_set) == 0:
                break
            ones_list = list(ones_set)
            zeros_list = list(zeros_set)
            g_ones = g[ones_list]
            g_zeros = g[zeros_list]
            
            i_cand = ones_list[int(np.argmax(g_ones))]
            j_cand = zeros_list[int(np.argmin(g_zeros))]
            
            gain = g[i_cand] - g[j_cand]
            if gain > best_gain:
                best_gain = gain
                best_i = i_cand
                best_j = j_cand
        
        if best_gain <= 1e-10:
            stagnation += 1
            if stagnation > 50:
                break
            continue
        
        if best_i < 0 or best_j < 0:
            break
        
        # Perform swap
        h[best_i] = 0.0
        h[best_j] = 1.0
        ones_set.remove(best_i)
        zeros_set.remove(best_j)
        ones_set.add(best_j)
        zeros_set.add(best_i)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            stagnation = 0
        else:
            # Revert
            h[best_i] = 1.0
            h[best_j] = 0.0
            ones_set.remove(best_j)
            zeros_set.remove(best_i)
            ones_set.add(best_i)
            zeros_set.add(best_j)
            corr = np.correlate(h, 1.0 - h, mode='full')
            stagnation += 1
            if stagnation > 200:
                break
    
    return h, best_c5

def random_swap_perturb(h, n, n_swaps, rng):
    """Randomly perform n_swaps random swaps, return modified h."""
    h_mod = h.copy()
    ones_list = np.where(h_mod > 0.5)[0].tolist()
    zeros_list = np.where(h_mod < 0.5)[0].tolist()
    
    for _ in range(n_swaps):
        if not ones_list or not zeros_list:
            break
        i = int(rng.randint(len(ones_list)))
        j = int(rng.randint(len(zeros_list)))
        h_mod[ones_list[i]] = 0.0
        h_mod[zeros_list[j]] = 1.0
        ones_list[i] = zeros_list.pop(j)
        zeros_list[j] = 1
        # Actually let's just redo:
        # This is getting messy, let me simplify
    
    # Simpler approach
    h_mod = h.copy()
    for _ in range(n_swaps):
        ones_mask = h_mod > 0.5
        zeros_mask = h_mod < 0.5

def smooth_max_grad(h, n, beta):
    """Compute smooth max of correlations and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    
    if beta < 1:
        beta = 1.0
    
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    
    val = (m + np.log(wsum) / beta) * dx
    
    # Gradient: sum_k weights[k] * dx * d(corr[k])/d(h)
    grad = np.zeros(n)
    ncorr = 2 * n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-12:
            continue
        offset = k - (n - 1)
        g_k = compute_grad_at_lag(h, offset, n)
        grad += w * g_k
    
    grad *= dx
    return val, grad

def gradient_descent_smooth(h, n, beta, n_steps, lr_init=0.01):
    """Gradient descent with projection for smooth max."""
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        if remaining_time() < 3:
            break
        val, grad = smooth_max_grad(h, n, beta)
        
        # Project gradient onto {sum(h) = n/2} direction
        # Subtract mean to make zero-sum
        grad_centered = grad - np.mean(grad)
        
        lr = lr_init / (1.0 + step * 0.01)
        h_new = h - lr * grad_centered
        
        # Project onto [0,1] and fix sum
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new <= best_c5 + 1e-12:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
        else:
            # Try smaller step
            lr2 = lr * 0.1
            h_new2 = h - lr2 * grad_centered
            h_new2 = project_feasible(h_new2, n)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 <= best_c5 + 1e-12:
                h = h_new2
                if c5_new2 < best_c5:
                    best_c5 = c5_new2
    
    return h, best_c5

def equalize_top(h, n, n_top=5, max_iters=500):
    """Try to equalize top correlations while reducing max."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    
    ones_set = set(np.where(h > 0.5)[0].tolist())
    zeros_set = set(np.where(h < 0.5)[0].tolist())
    
    if not ones_set or not zeros_set:
        return h, best_c5
    
    for it in range(max_iters):
        if remaining_time() < 5:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        current_max = np.max(corr)
        
        if current_max * dx >= best_c5 + 1e-10:
            break
        
        # Find top lags
        top_lags = np.argsort(corr)[::-1][:n_top]
        top_vals = np.array([corr[k] for k in top_lags])
        spread = np.max(top_vals) - np.min(top_vals)
        
        # Find swap that reduces the most active lag
        k_star = int(top_lags[0])
        offset = k_star - (n - 1)
        g = compute_grad_at_lag(h, offset, n)
        
        ones_list = list(ones_set)
        zeros_list = list(zeros_set)
        g_ones = g[ones_list]
        g_zeros = g[zeros_list]
        
        i_star = ones_list[int(np.argmax(g_ones))]
        j_star = zeros_list[int(np.argmin(g_zeros))]
        
        gain = g[i_star] - g[j_star]
        if gain < 1e-8:
            break
        
        # Try the swap
        h[i_star] = 0.0
        h[j_star] = 1.0
        ones_set.remove(i_star)
        zeros_set.remove(j_star)
        ones_set.add(j_star)
        zeros_set.add(i_star)
        
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = np.max(new_corr) * dx
        
        if new_max < best_c5:
            best_c5 = new_max
        else:
            # Revert
            h[i_star] = 1.0
            h[j_star] = 0.0
            ones_set.remove(j_star)
            zeros_set.remove(i_star)
            ones_set.add(i_star)
            zeros_set.add(j_star)
            if gain < 0.5:
                break
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_periodic_initial(n, period_pattern):
    """Create a periodic binary pattern."""
    h = np.zeros(n)
    p = len(period_pattern)
    for i in range(n):
        h[i] = period_pattern[i % p]
    # Adjust to get exactly n/2 ones
    n_ones_target = n // 2
    n_ones_current = int(np.sum(h > 0.5))
    if n_ones_current != n_ones_target:
        # Adjust: flip some entries
        if n_ones_current > n_ones_target:
            excess = n_ones_current - n_ones_target
            one_idx = np.where(h > 0.5)[0]
            to_flip = one_idx[:excess]
            h[to_flip] = 0.0
        else:
            deficit = n_ones_target - n_ones_current
            zero_idx = np.where(h < 0.5)[0]
            to_flip = zero_idx[:deficit]
            h[to_flip] = 1.0
    return h
