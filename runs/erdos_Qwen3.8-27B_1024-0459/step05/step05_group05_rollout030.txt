def project_onto_simplex(h, target_sum):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target_sum}"""
    # This is equivalent to finding x that minimizes ||x - h||^2
    # subject to 0 <= x_i <= 1 and sum(x_i) = target_sum
    
    # By KKT, x_i = clip(h_i - lambda) for some lambda
    # We need sum(clip(h_i - lambda)) = target_sum
    
    # Bisection on lambda
    lo, hi = min(h) - target_sum, max(h)  # generous bounds
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

def lbfgs_smooth(h0, n, base_beta, penalty, maxiter=1000):
    """L-BFGS-B with penalty for sum constraint."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(base_beta * sh, -50, 50)
        ex = np.exp(cl)
        sm = m + np.log(np.sum(ex)) / base_beta
        pen = penalty * (np.sum(x) - target)**2
        return (sm + pen) * dx
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(base_beta * sh, -50, 50)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx
        for k in range(2*n-1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n-1)
            if off >= 0:
                np_ = n - off
                g[off:off+np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off+np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao+np_])
                g[ao:ao+np_] -= wk * x[:np_]
        # Penalty gradient
        g += 2 * penalty * (np.sum(x) - target)
        return g
    
    bounds = [(0, 1)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
    return res.x
```

    """Project onto {x: 0<=x_i<=1, sum(x)=n/2} via bisection on clip(h-lambda)."""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - target - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _corr_gradient(h, n):
    """Compute gradient of all correlations with respect to h."""
    g = np.zeros(n)
    for k in range(2 * n - 1):
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += (1.0 - h[:np_])
            g[:np_] -= h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= h[:np_]
    return g


def _smoothed_max_obj(x, n, beta, penalty):
    dx = 2.0 / n
    target = n / 2.0
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    sm = m + np.log(np.sum(ex)) / beta
    pen = penalty * (np.sum(x) - target) ** 2
    return (sm + pen) * dx


def _smoothed_max_grad(x, n, beta, penalty):
    dx = 2.0 / n
    target = n / 2.0
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -80.0, 80.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    g += 2.0 * penalty * (np.sum(x) - target)
    return g


def lbfgs_optimize(h0, n, beta, penalty, maxiter=2000):
    """L-BFGS-B optimization with smoothed max and penalty for sum constraint."""
    h0 = np.clip(h0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _smoothed_max_obj(x, n, beta, penalty)

    def grad(x):
        return _smoothed_max_grad(x, n, beta, penalty)

    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-18,
                                'maxfun': maxiter * 3})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_exact(res.x, n)
    except Exception:
        pass
    return project_exact(h0, n)


def slsqp_smooth(h, n, beta, penalty, maxiter=300):
    """SLSQP with equality constraint for the sum."""
    h = project_exact(h, n)
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -80.0, 80.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -80.0, 80.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * dx
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def constr_fun(x):
        return np.sum(x) - target

    def constr_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_exact(res.x, n)
    except Exception:
        pass
    return h


def subgradient_descent(h, n, n_iter=500):
    """Projected subgradient descent on the max correlation."""
    dx = 2.0 / n
    h = project_exact(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.03
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 0.5
        top_idx = np.where(corr >= threshol')[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]

        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(top_idx)
        g -= np.mean(g)

        h_new = project_exact(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.95
            if step < 0.001:
                break
    return best_h, best_c5


def binary_local_search(h_bin, n, rng, time_limit=100):
    """Gradient-guided binary local search with 2-swap moves."""
    h = (h_bin > 0.5).astype(float).copy()
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - target]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:target - n_ones]]] = 1

    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_improve = 0
    max_no_improve = 8000

    while time.time() - start < time_limit and remaining() > 3:
        no_improve += 1
        if no_improve > max_no_improve:
            break

        corr = compute_corr(h)
        max_val = float(np.max(corr))
        top_k = min(8, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_k]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0.0
        best_swap = None

        # Strategy 1: Focus on hot offsets
        for _ in range(40):
            k_hot = top_idx[rng.randint(len(top_idx))]
            off = k_hot - (n - 1)
            if off >= 0:
                valid = ones[(ones + off) < n]
                valid = valid[h[valid + off] == 0]
            else:
                ao = -off
                valid = ones[ones >= ao]
                valid = valid[h[valid - ao] == 0]
            if len(valid) == 0:
                valid = ones
            i = valid[rng.randint(len(valid))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        # Strategy 2: Random swaps
        for _ in range(20):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            nc = float(np.max(compute_corr(h)))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            cur_max += best_delta
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            # Try 2-swap moves to escape local minima
            if no_improve > 300 and rng.randint(3) == 0:
                found = False
                for _ in range(8):
                    if len(ones) < 2 or len(zeros) < 2:
                        break
                    p = rng.permutation(len(ones))[:2]
                    q = rng.permutation(len(zeros))[:2]
                    i1, i2 = ones[p[0]], ones[p[1]]
                    j1, j2 = zeros[q[0]], zeros[q[1]]
                    if len(set([i1, i2, j1, j2])) < 4:
                        continue
                    h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                    nc = float(np.max(compute_corr(h)))
                    h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
                    if nc < max_val - 1e-14:
                        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                        cur_max = nc
                        no_improve = 0
                        if cur_max < best_max:
                            best_max = cur_max
                            best_h = h.copy()
                        found = True
                        break
            if no_improve > 500:
                # Larger perturbation
                n_flip = rng.randint(2, 5)
                for _ in range(n_flip):
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    i = ones_now[rng.randint(len(ones_now))]
                    j = zeros_now[rng.randint(len(zeros_now))]
                    h[i] = 0; h[j] = 1
                cur_max = float(np.max(compute_corr(h)))

    return best_h, best_max * dx


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def intensive_refinement(h, n, time_limit=200):
    """Multi-technique intensive refinement at a fixed resolution."""
    h = project_exact(h, n)
    best_c5 = compute_c5(h)
    start = time.time()

    # L-BFGS-B with increasing beta
    for beta in [100, 500, 2000, 8000, 30000, 100000]:
        if remaining() < 5 or time.time() - start > time_limit:
            break
        penalty = max(1.0, 0.01 * beta)
        h_new = lbfgs_optimize(h, n, beta, penalty, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new.copy()

    # SLSQP polish
    for beta in [10000, 50000, 200000]:
        if remaining() < 5 or time.time() - start > time_limit:
            break
        h_new = slsqp_smooth(h, n, beta, 1.0, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new.copy()

    # Subgradient descent
    if remaining() > 10 and time.time() - start < time_limit - 5:
        h_sg, c5_sg = subgradient_descent(h, n, n_iter=200)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            h = h_sg.copy()

    # Another L-BFGS-B round
    if remaining() > 5 and time.time() - start < time_limit - 5:
        for beta in [50000, 200000]:
            if remaining() < 3:
                break
            h_new = lbfgs_optimize(h, n, beta, 50.0, maxiter=1000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h = h_new.copy()

    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_exact(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: L-BFGS-B at parent resolution (budget ~150s)
    phase1_end = _T0 + 150
    for beta in [50, 200, 1000, 5000, 20000, 80000, 300000]:
        if time.time() > phase1_end or remaining() < 8:
            break
        penalty = max(1.0, 0.005 * beta)
        h_new = lbfgs_optimize(h, n, beta, penalty, maxiter=2000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: SLSQP refinement (budget ~80s)
    phase2_end = min(_T0 + 230, _T0 + _BUDGET * 0.35)
    for beta in [5000, 30000, 150000]:
        if time.time() > phase2_end or remaining() < 6:
            break
        h_new = slsqp_smooth(h, n, beta, 1.0, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 3: Resolution sweep (budget ~250s)
    phase3_end = min(_T0 + 480, _T0 + _BUDGET * 0.65)
    n_candidates = []
    for delta in [-8, -6, -4, -3, -2, -1, 1, 2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64]:
        nt = best_n + delta
        if 192 <= nt <= 800 and nt % 2 == 0:
            n_candidates.append(nt)
    for nt in [256, 288, 320, 352, 384, 416, 448, 480, 512, 544, 576, 640]:
        if 192 <= nt <= 800 and nt % 2 == 0 and nt not in n_candidates:
            n_candidates.append(nt)
    n_candidates.sort(key=lambda x: abs(x - best_n))

    for n_t in n_candidates[:10]:
        if time.time() > phase3_end or remaining() < 15:
            break
        h_t = interp_to(best_h, best_n, n_t)
        h_t = project_exact(h_t, n_t)
        c5_t0 = compute_c5(h_t)
        if c5_t0 < best_c5:
            best_c5 = c5_t0
            best_h = h_t.copy()
            best_n = n_t

        # Quick L-BFGS-B at this resolution
        for beta in [1000, 10000, 100000]:
            if time.time() > phase3_end or remaining() < 5:
                break
            penalty = max(1.0, 0.005 * beta)
            h_t2 = lbfgs_optimize(h_t, n_t, beta, penalty, maxiter=1000)
            c5_t2 = compute_c5(h_t2)
            if c5_t2 < c5_t0:
                c5_t0 = c5_t2
                h_t = h_t2.copy()
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

    # Phase 4: Intensive refinement at best resolution (budget ~250s)
    n = best_n
    h = best_h.copy()
    phase4_end = min(_T0 + 730, _T0 + _BUDGET * 0.88)
    h, best_c5 = intensive_refinement(h, n, time_limit=min(250, max(20, int(phase4_end - time.time()))))
    best_h = h.copy()
    best_n = n

    # Phase 5: Binary local search (budget ~150s)
    if remaining() > 30:
        h_bin = binarize(best_h, best_n)
        bl_time = min(150, int(remaining() * 0.4))
        h_b, c5_b = binary_local_search(h_bin, best_n, rng, time_limit=bl_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            # Continuous refinement from binary
            h_c = lbfgs_optimize(h_b, best_n, 50000, 50.0, maxiter=2000)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
        # Second binary search from slightly perturbed start
        if remaining() > 20:
            h_bin2 = binarize(best_h + rng.randn(best_n) * 0.05, best_n)
            h_b2, c5_b2 = binary_local_search(h_bin2, best_n, rng, time_limit=min(60, int(remaining() * 0.3)))
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()

    # Phase 6: Multi-start explorations (budget ~100s)
    if remaining() > 40:
        # Thue-Morse start
        h_tm = thue_morse(best_n, rng)
        h_tm = lbfgs_optimize(h_tm, best_n, 10000, 10.0, maxiter=2000)
        c5_tm = compute_c5(h_tm)
        h_tm = slsqp_smooth(h_tm, best_n, 50000, 1.0, maxiter=300)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()

        # Inverse Thue-Morse
        if remaining() > 20:
            h_tm2 = 1.0 - thue_morse(best_n, rng)
            h_tm2 = project_exact(h_tm2, best_n)
            h_tm2 = lbfgs_optimize(h_tm2, best_n, 10000, 10.0, maxiter=1500)
            c5_tm2 = compute_c5(h_tm2)
            if c5_tm2 < best_c5:
                best_c5 = c5_tm2
                best_h = h_tm2.copy()

        # Random starts
        for rs in range(3):
            if remaining() < 15:
                break
            h_r = rng.uniform(0.2, 0.8, best_n)
            h_r = project_exact(h_r, best_n)
            h_r = lbfgs_optimize(h_r, best_n, 5000, 5.0, maxiter=1500)
            h_r = lbfgs_optimize(h_r, best_n, 50000, 50.0, maxiter=1500)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

        # Perturbed best
        if remaining() > 15:
            h_p = best_h + rng.randn(best_n) * 0.08
            h_p = project_exact(h_p, best_n)
            h_p = lbfgs_optimize(h_p, best_n, 30000, 30.0, maxiter=2000)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 7: Final intensive polish (remaining - 10s safety)
    n = best_n
    h = best_h.copy()
    final_time = max(10, int(remaining() - 10))

    for beta in [50000, 200000, 500000, 1000000]:
        if remaining() < 4:
            break
        h_new = lbfgs_optimize(h, n, beta, max(10.0, 0.001 * beta), maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # SLSQP final
    if remaining() > 8:
        h_new = slsqp_smooth(h, n, 500000, 10.0, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Subgradient final
    if remaining() > 10:
        h_sg, c5_sg = subgradient_descent(h, n, n_iter=int(final_time * 2))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # One more L-BFGS-B after subgradient
    if remaining() > 5:
        h_new = lbfgs_optimize(best_h, best_n, 1000000, 100.0, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final binary search if time allows
    if remaining() > 15:
        h_bin = binarize(best_h, best_n)
        h_b, c5_b = binary_local_search(h_bin, best_n, rng, time_limit=min(40, int(remaining() - 5)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = lbfgs_optimize(h_b, best_n, 200000, 100.0, maxiter=1000)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Ensure feasibility
    final_h = project_exact(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate with evaluate_erdos_solution if available
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
