                lo = max(0.0, lo)
                hi = min(1.0, hi)

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def pair_swap(h, n, rng, n_pairs=200):
    """Try swapping/complementing pairs of values"""
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode='full')
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    for _ in range(n_pairs):
        i, j = rng.choice(n, 2, replace=False)
        # Complement both: h[i] -> 1-h[i], h[j] -> 1-h[j]
        if abs(best[i] - best[j]) < 1e-12:
            continue
        best[i], best[j] = 1.0 - best[i], 1.0 - best[j]
        c5_new = compute_c5(best)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            corr = np.correlate(best, 1.0 - best, mode='full')
        else:
            best[i], best[j] = 1.0 - best[i], 1.0 - best[j]  # revert
    
    return best, best_c5
