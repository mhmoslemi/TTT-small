            best_c5 = c1
            best_h = h1.copy()
            best_n = best_n

    # Phase 2: Try higher resolutions
    res_list = []
    for mult in [1.5, 2.0, 2.5, 3.0]:
        rn = int(round(best_n * mult / 64) * 64)
        if 320 <= rn <= 3072 and rn % 2 == 0 and rn != best_n:
            res_list.append(rn)
    res_list = sorted(set(res_list))

    for rn in res_list[:4]:
        if rem() < 40:
            break
        h_interp = interp_to(best_h, best_n, rn)
        h_feas = project_feasible(h_interp, rn)
        c_int = compute_c5(h_feas)
        t_alloc = min(200.0, rem() * 0.2)
        h_new, c_new = optimize_at_resolution(h_feas, rn, t_alloc)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            best_n = rn

    # Phase 3: Exact subgradient refinement at best resolution
    if rem() > 20:
        t_sg = min(150.0, rem() * 0.15)
        h_sg, c_sg = subgradient_exact(best_h, t_sg)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg.copy()

    # Phase 4: Top-K gradient descent
    if rem() > 15:
        h_gd, c_gd = topk_grad_descent(best_h, best_n, min(100.0, rem() * 0.1), K=25, beta_w=80.0)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd.copy()

    # Phase 5: SLSQP final polish with very high beta
    if rem() > 10:
        for beta in [500000, 1000000, 2000000]:
            if rem() < 8:
                break
            h_new = slsqp_smooth(best_h, beta, maxiter=200)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()

    # Phase 6: Try binary local search as alternative
    if rem() > 30:
        t_bs = min(80.0, rem() * 0.1)
        h_bin, c_bin = binary_local_search(best_h, best_n, t_bs, max_rounds=300)
        if c_bin < best_c5:
            # Use binary as starting point for continuous optimization
            h_ref = interp_to(h_bin, best_n, best_n)
            h_ref = project_feasible(h_ref, best_n)
            if rem() > 20:
                h_pol, c_pol = optimize_at_resolution(h_ref, best_n, min(40.0, rem() * 0.2))
                if c_pol < best_c5:
                    best_c5 = c_pol
                    best_h = h_pol.copy()
            else:
                best_c5 = c_bin
                best_h = h_bin.copy()

    # Phase 7: Final trust-constr attempt at best resolution
    if rem() > 15:
        h_tc = tc_smooth(best_h, 200000, maxiter=100)
        c_tc = compute_c5(h_tc)
        if c_tc < best_c5:
            best_c5 = c_tc
            best_h = h_tc.copy()

    # Phase 8: Very fine top-K descent
    if rem() > 10:
        h_fine, c_fine = topk_grad_descent(best_h, best_n, min(50.0, rem() * 0.5), K=10, beta_w=200.0)
        if c_fine < best_c5:
            best_c5 = c_fine
            best_h = h_fine.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
