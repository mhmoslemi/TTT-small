    - h[p] appears in (1-h[i+shift]) when i+shift=p, i=p-shift: contributes -h[p-shift] if 0≤p-shift<n
    
    So: ∂C[m]/∂h[p] = (1-h[p+shift])*[0≤p+shift<n] - h[p-shift]*[0≤p-shift<n]
    where shift = m - n + 1.

    ΔC[m] = δ * ((1-h[p+shift_m])*[0≤p+shift_m<n] - h[p-shift_m]*[0≤p-shift_m<n])
    
    where shift_m = m - n + 1.

    shifts = arange(2n-1) - n + 1  (length 2n-1)
    valid1 = (p + shifts >= 0) & (p + shifts < n)
    valid2 = (p - shifts >= 0) & (p - shifts < n)
    delta_C = δ * ((1-h)[valid indices for p+shifts] - h[valid indices for p-shifts])

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -60.0, 60.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx

def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g

def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -60.0, 60.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    w = w * dx
    return corr_grad_weighted(h, w)

def slsqp_run(h, n, beta, maxiter=500):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        return smooth_obj(x, beta)
    def _grad(x):
        return smooth_grad(x, beta)
    def _sc(x):
        return np.sum(x) - n / 2.0
    def _scj(x):
        return np.ones(n)
    
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.success or res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def binary_local_search(h_bin, n, max_rounds=30, verbose=False):
    """Efficient binary local search using influence vectors."""
    h = (h_bin > 0.5).astype(float).copy()
    dx = 2.0 / n
    C = compute_corr(h) * dx
    best_c5 = float(np.max(C))
    best_h = h.copy()
    shifts = np.arange(2 * n - 1) - (n - 1)  # shift values
    
    for rnd in range(max_rounds):
        if remaining() < 15:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        c_max = np.max(C)
        best_improve = 1e-15
        best_pair = None
        
        # For each position, precompute its influence vector
        # inf_p[m] = (1-h[p+shift_m])*[valid] - h[p-shift_m])*[valid]
        # This is O(n) per position, O(n²) total
        
        # Actually, let's compute the influence matrix all at once
        # inf[p, m] where m indexes the 2n-1 correlation values
        p_idx = np.arange(n)[:, None]  # n x 1
        s_idx = shifts[None, :]  # 1 x (2n-1)
        p_plus_s = p_idx + s_idx  # n x (2n-1)
        p_minus_s = p_idx - s_idx  # n x (2n-1)
        
        valid_plus = (p_plus_s >= 0) & (p_plus_s < n)  # n x (2n-1)
        valid_minus = (p_minus_s >= 0) & (p_minus_s < n)  # n x (2n-1)
        
        # (1-h[p+shift]) for valid positions
        one_minus_h = 1.0 - h
        # Clip indices for fancy indexing
        clip_plus = np.clip(p_plus_s, 0, n - 1)
        clip_minus = np.clip(p_minus_s, 0, n - 1)
        
        term_plus = np.where(valid_plus, one_minus_h[clip_plus], 0.0)  # n x (2n-1)
        term_minus = np.where(valid_minus, h[clip_minus], 0.0)  # n x (2n-1)
        
        inf_matrix = term_plus - term_minus  # n x (2n-1)
        
        # Now for a swap (p: 1→0, q: 0→1):
        # ΔC = -inf_matrix[p] + inf_matrix[q]
        # new_C = C + ΔC
        # new_max = max(new_C)
        
        # To find the best swap, we need to minimize max(C - inf[p] + inf[q])
        # over p in ones, q in zeros
        
        # This is expensive to do exhaustively. Let's use a heuristic:
        # Focus on positions that affect the current maximum shifts.
        
        top_k_shifts = np.argsort(-C)[:min(20, 2*n-1)]  # top 20 shifts by value
        
        # For each one position, compute how much it "contributes" to the top shifts
        # For each zero position, compute how much it "reduces" the top shifts
        
        one_contrib = np.zeros((len(ones), len(top_k_shifts)))
        zero_contrib = np.zeros((len(zeros), len(top_k_shifts)))
        
        for ia, p in enumerate(ones):
            one_contrib[ia] = inf_matrix[p, top_k_shifts]
        for ib, q in enumerate(zeros):
            zero_contrib[ib] = inf_matrix[q, top_k_shifts]
            
        # For swap (p, q): new_C at top shifts = C[top] - one_contrib[ia] + zero_contrib[ib]
        # We want this to be small (below c_max)
        
        # Heuristic: find p with high contribution to current max, q with low (or negative)
        # at the same shifts
        
        # Score for p: weighted sum of contributions to top shifts
        top_weights = C[top_k_shifts] / (C[top_k_shifts].sum() + 1e-30)
        p_scores = one_contrib @ top_weights  # higher = more "responsible" for max
        q_scores = zero_contrib @ top_weights  # lower = less "additive" to max
        
        # Try the top candidates
        n_try_p = min(30, len(ones))
        n_try_q = min(30, len(zeros))
        
        top_p = np.argsort(-p_scores)[:n_try_p]
        top_q = np.argsort(q_scores)[:n_try_q]
        
        for ia_idx in range(len(top_p)):
            p = ones[top_p[ia_idx]]
            for ib_idx in range(len(top_q)):
                q = zeros[top_q[ib_idx]]
                # Compute new max
                new_C = C - inf_matrix[p] + inf_matrix[q]
                nm = np.max(new_C)
                imp = c_max - nm
                if imp > best_improve:
                    best_improve = imp
                    best_pair = (p, q)
                    
                    if best_improve > 0.5 * dx:  # big improvement, stop early
                        break
            if best_improve > 0.5 * dx:
                break
        
        if best_pair is None:
            break
        
        p, q = best_pair
        h[p] = 0.0
        h[q] = 1.0
        C = compute_corr(h) * dx
        c5_new = np.max(C)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        if verbose:
            print(f"Round {rnd}: C5={c5_new:.8f}, best={best_c5:.8f}")
    
    return best_h, best_c5
