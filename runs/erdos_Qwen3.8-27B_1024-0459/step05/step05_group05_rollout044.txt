def corr_gradient_at(h, k_off, n):
    g = np.zeros(n)
    if k_off >= 0:
        np_ = n - k_off
        g[k_off:k_off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def smooth_max_obj_grad(h, beta):
    n = len(h)

Actually, for the full correlation of h with (1-h):
corr[k] = sum_i h[i] * (1-h[i+k-(n-1)]) for valid i

This is equivalent to the cross-correlation. In the frequency domain, cross-correlation is the product of one DFT with the conjugate of the other.

Let me think about whether FFT-based correlation would help... For n=384, direct correlation is about 384*767 ≈ 294K operations. FFT-based would be O(n log n) ≈ 384*9 ≈ 3.4K operations. That's a significant speedup!

But np.correlate already uses an efficient algorithm, and for n=384 it's fast enough. The bottleneck is the number of evaluations, not the cost per evaluation.

1. Efficient Adam with curvature-aware learning rate
2. Multi-beta curriculum
3. Improved binary search with targeted swaps
4. Multi-resolution sweep
5. Diverse initializations

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def corr_grad_single(h, k_off, n):
    g = np.zeros(n)
    if k_off >= 0:
        np_ = n - k_off
        g[k_off:k_off + np_] += (1.0 - h[:np_])
        g[:np_] -= h[k_off:k_off + np_]
    else:
        ao = -k_off
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(beta * sh, -60.0, 60.0)
    ex = np.exp(cl)
    s = np.sum(ex)
    val = (m + np.log(s) / beta) * dx
    w = ex / s
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - h[ao:ao + np_])
            g[ao:ao + np_] -= wk * h[:np_]
    return val, g


def adam_optimize(h_init, n, beta, n_iter, base_lr=0.005):
    h = project_feasible(h_init, n).copy()
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t = 0
    b1, b2 = 0.85, 0.999
    eps = 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = base_lr
    stall = 0
    best_stall = best_c5

    for it in range(n_iter):
        if time.time() - _T0 > _BUDGET - 8:
            break
        t += 1
        val, g = smooth_obj_grad(h, beta)
        g -= np.mean(g)
        g = np.where(h > 0.98, -np.abs(g) * 0.5, g)
        g = np.where(h < 0.02, np.abs(g) * 0.5, g)

        m_acc = b1 * m_acc + (1 - b1) * g
        v_acc = b2 * v_acc + (1 - b2) * (g * g)
        m_hat = m_acc / (1.0 - b1 ** t)
        v_hat = v_acc / (1.0 - b2 ** t)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1

        h = h_new

        if stall % 300 == 0:
            if best_c5 > best_stall + 1e-14:
                lr *= 0.85
            best_stall = best_c5
            if lr < 1e-7:
                break

    return best_h, best_c5


def binary_ls(h_bin, n, rng, time_limit=60):
    h = (h_bin > 0.5).astype(float).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            perm = rng.permutation(len(ones_idx))
            h[ones_idx[perm[:n_ones - n_half]]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            perm = rng.permutation(len(zeros_idx))
            h[zeros_idx[perm[:n_half - n_ones]]] = 1

    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    start = time.time()
    no_imp = 0

    while time.time() - start < time_limit and remaining() > 5:
        no_imp += 1
        if no_imp > 10000:
            break

        corr = compute_corr(h)
        max_val = np.max(corr)
        top_k = min(6, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_k]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_delta = 0
        best_swap = None
        n_try = 10

        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            nc = np.max(compute_corr(h))
            delta = nc - max_val
            h[i] = 1
            h[j] = 0
            if delta < best_delta - 1e-14:
                best_delta = delta
                best_swap = (i, j)

        for kk in range(min(4, len(top_idx))):
            k_hot = top_idx[kk]
            off = k_hot - (n - 1)
            if off >= 0:
                fault_mask_a = (h[:n - off] == 1) & (h[off:] == 0)
                fault_idx = np.where(fault_mask_a)[0]
            else:
                ao = -off
                fault_mask_a = (h[ao:] == 1) & (h[:n - ao] == 0)
                fault_idx = np.where(fault_mask_a)[0]

            for fi in fault_idx[:2]:
                if off >= 0:
                    i_sw = fi
                else:
                    i_sw = fi + ao
                if i_sw < 0 or i_sw >= n or h[i_sw] != 1:
                    continue
                j = zeros[rng.randint(len(zeros))]
                if j == i_sw:
                    continue
                h[i_sw] = 0
                h[j] = 1
                nc = np.max(compute_corr(h))
                delta = nc - max_val
                h[i_sw] = 1
                h[j] = 0
                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_swap = (i_sw, j)

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            no_imp = 0
            cur_max = float(np.max(compute_corr(h)))
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            if no_imp > 400:
                n_flip = rng.randint(2, 5)
                ones_now = np.where(h == 1)[0]
                zeros_now = np.where(h == 0)[0]
                for _ in range(n_flip):
                    if len(ones_now) == 0 or len(zeros_now) == 0:
                        break
                    ii = ones_now[rng.randint(len(ones_now))]
                    jj = zeros_now[rng.randint(len(zeros_now))]
                    h[ii] = 0
                    h[jj] = 1
                    ones_now = np.where(h == 1)[0]
                    zeros_now = np.where(h == 0)[0]
                cur_max = float(np.max(compute_corr(h)))

    return best_h, best_max * dx


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def binarize(h, n):
    h = np.array(h, dtype=float)
    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sort_idx[:n_half]] = 1.0
    return h_bin


def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = ones_idx[rng.permutation(len(ones_idx))[:n_ones - n_half]]
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = zeros_idx[rng.permutation(len(zeros_idx))[:n_half - n_ones]]
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def slsqp_refine(h, n, beta, maxiter=200):
    dx = 2.0 / n
    best_c5 = compute_c5(h)

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / beta) * dx

    def jac(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        w = ex / s
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k] * dx
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5_res = compute_c5(h_res)
        if c5_res < best_c5:
            return h_res, c5_res
    except Exception:
        pass
    return h, best_c5
