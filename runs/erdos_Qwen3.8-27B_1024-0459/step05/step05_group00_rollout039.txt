            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            for factor in [0.5, 0.25, 0.1]:
                h_try = project_feasible(h + factor * v, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    break
    return best_h, best_c5


def cutting_plane_lp(h, n, K, delta):
    corr = compute_corr(h)
    idx = np.argsort(-corr)[:K]

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub_list = []
    b_ub_list = []
    for k in idx:
        g = grad_corr_k(h, k, n)
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(corr[k] - np.dot(g, h))

    A_trust = np.vstack([
        np.hstack([np.eye(n), np.zeros((n, 1))]),
        np.hstack([-np.eye(n), np.zeros((n, 1))])
    ])
    b_trust = np.concatenate([h + delta, -h + delta])

    A_ub = np.vstack([A_ub_list, A_trust])
    b_ub = np.concatenate([b_ub_list, b_trust])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return h.copy(), float('inf')


def basin_hopping(h, n, time_limit, rng, best_c5, best_h):
    global _T0, _BUDGET
    h = h.copy().astype(float)
    current_c5 = compute_c5(h)

    start = time.time()
    perturbation_scale = 0.03

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 30:
            break

        n_pert = rng.randint(1, 6)
        indices = rng.choice(n, n_pert, replace=False)
        h_pert = h.copy()
        h_pert[indices] += rng.uniform(-perturbation_scale, perturbation_scale, n_pert)
        h_pert = project_feasible(h_pert, n)

        h_opt = slsqp_smooth_max(h_pert, n, beta=5000, maxiter=300)
        c5_opt = compute_c5(h_opt)

        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()
            current_c5 = c5_opt
            perturbation_scale = max(0.005, perturbation_scale * 0.85)
        else:
            perturbation_scale = min(0.05, perturbation_scale * 1.05)

    return best_h, best_c5


def fine_coordinate_search(h, n, time_limit, rng):
    global _T0, _BUDGET
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        rem = _BUDGET - (time.time() - _T0)
        if rem < 10:
            break
        improved = False
        for _ in range(min(200, int((time_limit - (time.time() - start)) * 5))):
            if time.time() - start >= time_limit:
                break
            rem = _BUDGET - (time.time() - _T0)
            if rem < 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(-0.02, 0.02)
            h_try = h.copy()
            h_try[i] += delta
            h_try[j] -= delta
            h_try = np.clip(h_try, 0.0, 1.0)
            if not (0 <= h_try[i] <= 1 and 0 <= h_try[j] <= 1):
                continue
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                h = h_try.copy()
                improved = True
        if not improved:
            break
    return h, best_c5

        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 512
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    h = best_h.copy()

    # Phase 1: Subgradient with momentum (60s)
    if remaining() > 30:
        t1 = min(60.0, remaining() * 0.1)
        h, c5 = subgradient_momentum(h, n, t1, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: SLSQP with increasing beta
    for beta in [500, 2000, 8000, 20000, 50000]:
        if remaining() < 15:
            break

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:

        h = best_h.copy()

                    h_try = project_feasible(h + alpha * (h_lp - h), n)
                    c_try = compute_c5(h_try)
                    if c_try < best_val:
                        best_val = c_try
                        h = h_try
                if compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
            else:

        if remaining() > 15:
            h = slsqp_smooth_max(h, n, beta=30000, maxiter=500)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        h = best_h.copy()
