def corr_grad_single(h, k, n):
    """Gradient of correlation at index k with respect to h."""
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g

def _sm_fun(x):

    dx = 2.0 / n

    val = (m + np.log(np.sum(ex)) / _BETA) * dx
    penalty = 50.0 * (np.sum(x) - _TGT) ** 2

def nesterov_grad(h, n, top_k=5, max_iters=5000, step_init=0.03, rng=None):

    v = np.zeros(n)
    momentum = 0.9
    step = step_init
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx if (dx := 2.0/n) else 0
        # Actually let me not use walrus, let me be explicit
        dx_val = 2.0 / n
        c5 = float(np.max(corr)) * dx_val
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        idx = np.argsort(-corr)[:top_k]
        c_top = corr[idx]
        m = c_top[0]
        # Soft weights
        beta_w = 50.0
        w = np.exp(beta_w * (c_top - m))
        w /= np.sum(w)
        
        g = np.zeros(n)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)
        g -= np.mean(g)
        g[(h >= 1.0 - 1e-10)] = 0.0
        g[(h <= 1e-10)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-13:

        g /= gn
        
        v = momentum * v - step * g

            h = best_h.copy()
            v *= 0.5
            step *= 0.7
        
        if it % 2000 == 0 and it > 0:
            step *= 0.9
            momentum = min(momentum + 0.005, 0.97)
    
    return best_h, best_c5

        if c5_c768 <= c5_b768:
            h_768 = h_c768
        else:
            h_768 = h_b768

        if c5_768 < best_c5:
            best_c5 = c5_768
            best_h = h_768.copy()
            best_n = n768

    # Phase 4: Even higher resolution - n=1024
    n1024 = 1024
    if remaining() > 150:
        h_interp1k = interp_to(best_h, best_n, n1024)
        h_feas1k = proj(h_interp1k, n1024)

        h_b1k = (h_feas1k > 0.5).astype(float)
        h_b1k, c5_b1k = binary_local_search(h_b1k, n1024, max_iters=6000, top_k=5, rng=rng)

        # Jitter at 1024
        if remaining() > 30:
            jitters1k = jitter_binary(h_b1k, n1024, n_jitters=8, rng=rng)
            for h_j, c5_j in jitters1k:
                if c5_j < c5_b1k:
                    c5_b1k = c5_j
                    h_b1k = h_j.copy()
            if remaining() > 20:
                h_b1kj, c5_b1kj = binary_local_search(h_b1k, n1024, max_iters=2500, top_k=5, rng=rng)
                if c5_b1kj < c5_b1k:
                    c5_b1k = c5_b1kj
                    h_b1k = h_b1kj.copy()

        # Continuous at 1024
        h_c1k = h_b1k.copy()
        for beta in [1000, 5000, 20000]:
            if remaining() < 10:
                break
            h_c1k = lbfgs_pass(h_c1k, n1024, beta, top_k=5, maxiter=350)
        c5_c1k = compute_c5(h_c1k)

        if remaining() > 10:
            h_c1k, c5_c1k = top_equalize(h_c1k, n1024, top_k=5, max_iters=250)

        if remaining() > 15:
            h_c1k, c5_c1k = nesterov_grad(h_c1k, n1024, top_k=4, max_iters=2500, step_init=0.02, rng=rng)

        c5_1k = min(c5_b1k, c5_c1k)
        if c5_c1k <= c5_b1k:
            h_1k = h_c1k
        else:
            h_1k = h_b1k

        if c5_1k < best_c5:
            best_c5 = c5_1k
            best_h = h_1k.copy()
            best_n = n1024

    # Phase 5: Final intensive refinement at best resolution
    n = best_n
    h = best_h.copy()

    # Aggressive L-BFGS with increasing beta
    for beta in [5000, 20000, 60000, 150000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=4, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Nesterov with small step
    if remaining() > 20:
        h_ng2, c5_ng2 = nesterov_grad(h, n, top_k=4, max_iters=4000, step_init=0.015, rng=rng)
        if c5_ng2 < best_c5:
            best_c5 = c5_ng2
            best_h = h_ng2.copy()
            h = h_ng2.copy()

    # Binary polish + continuous refine
    if remaining() > 30:
        h_bin_f = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_local_search(h_bin_f, n, max_iters=4000, top_k=4, rng=rng)
        h_rf = h_bf.copy()
        for beta in [5000, 20000, 80000]:
            if remaining() < 5:
                break
            h_rf = lbfgs_pass(h_rf, n, beta, top_k=4, maxiter=350)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()
        elif c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Final top equalize
    if remaining() > 10:
        h_eq2, c5_eq2 = top_equalize(best_h, best_n, top_k=7, max_iters=300)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()

    # One more nesterov pass
    if remaining() > 15:
        h_ng3, c5_ng3 = nesterov_grad(best_h, best_n, top_k=5, max_iters=3000, step_init=0.01, rng=rng)
        if c5_ng3 < best_c5:
            best_c5 = c5_ng3
            best_h = h_ng3.copy()

    # Ensure feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
