def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:

    return val, grad

def block_coord_descent(h, n, block_size=2, max_sweeps=50, rng=None):

            break

            idxs = list(range(start, start + block_size))
            old_vals = [best[i] for i in idxs]

            # Compute combined A matrix
            A_combined = np.zeros((block_size, ncorr))
            for bi, i in enumerate(idxs):
                i_plus = i + offsets
                i_minus = i - offsets
                valid_plus = (i_plus >= 0) & (i_plus < n)
                valid_minus = (i_minus >= 0) & (i_minus < n)
                A_combined[bi][valid_minus] += (1.0 - best[i_minus[valid_minus]])
                A_combined[bi][valid_plus] -= best[i_plus[valid_plus]]
                A_combined[bi][mid] = 0.0

            # Fix the quadratic term for offset 0

            # Grid search: try a few values for each coordinate
            # For block_size=2, try a 5x5 grid

            # Add current values
            for v in old_vals:

                    grid_vals.append(v)
            grid_vals = sorted(set(grid_vals))

            best_f = np.max(best_corr)
            best_combo = old_vals[:]

            for v1 in grid_vals:

                    for v2 in grid_vals:
                        new_base = base.copy()
                        new_base += A_combined[0] * (v1 - old_vals[0]) + A_combined[1] * (v2 - old_vals[1])

    return best, best_c5

            mask = rng.random(n) < 0.25
            n_mask = np.sum(mask)
            if n_mask > 0:
                h_pert[mask] += rng.randn(n_mask) * noise_scale
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=60, use_random_order=True)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                # Quick SLSQP
                if remaining_time() > 20:
                    h_slq = slsqp_refine(h_new, n, 10000, 200)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        h_cd, c5_cd = fast_coord_descent(h_slq, n, max_sweeps=30, use_random_order=True)
                        if c5_cd < best_c5:
                            best_c5 = c5_cd
                            best_h = h_cd.copy()

    # Phase 6: Try higher resolution n=1280
    if remaining_time() > 120:
        high_n = 1280
        h_high = interp_to(best_h, best_n, high_n)
        h_high = project_feasible(h_high, high_n)
        c5_high = compute_c5(h_high)

        h_new, c5_new = fast_coord_descent(h_high, high_n, max_sweeps=80, use_random_order=True)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = high_n

        if remaining_time() > 60:
            h_slq = slsqp_refine(best_h, high_n, 5000, 300)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()
                best_n = high_n
                h_cd, c5_cd = fast_coord_descent(h_slq, high_n, max_sweeps=40, use_random_order=True)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = high_n

    # Phase 7: Try even higher resolution n=1920
    if remaining_time() > 100:
        high_n2 = 1920
        h_high2 = interp_to(best_h, best_n, high_n2)
        h_high2 = project_feasible(h_high2, high_n2)
        h_new, c5_new = fast_coord_descent(h_high2, high_n2, max_sweeps=50, use_random_order=True)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = high_n2
            if remaining_time() > 30:
                h_slq = slsqp_refine(h_new, high_n2, 3000, 200)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    best_n = high_n2

    # Phase 8: Block coordinate descent at best resolution
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = block_coord_descent(h, best_n, block_size=2, max_sweeps=20, rng=rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Follow up
        if remaining_time() > 15:
            h2, c52 = fast_coord_descent(best_h, best_n, max_sweeps=30, use_random_order=True)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2.copy()

    # Phase 9: More perturbations at best resolution
    if remaining_time() > 60:
        for p in range(15):
            if remaining_time() < 15:
                break
            h_pert = best_h.copy()
            noise_scale = 0.08 / (1 + p * 0.15)
            mask = rng.random(best_n) < 0.2
            n_mask = np.sum(mask)
            if n_mask > 0:
                h_pert[mask] += rng.randn(n_mask) * noise_scale
            h_pert = project_feasible(h_pert, best_n)
            h_new, c5_new = fast_coord_descent(h_pert, best_n, max_sweeps=50, use_random_order=True)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 20:
                    h_slq = slsqp_refine(h_new, best_n, 20000, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()

    # Phase 10: Final intensive optimization
    if remaining_time() > 20:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, best_n, max_sweeps=60, use_random_order=True)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 11: Final SLSQP
    if remaining_time() > 10:
        h_new = slsqp_refine(best_h, best_n, 50000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 12: Final coordinate descent
    if remaining_time() > 8:
        h_new, c5_new = fast_coord_descent(best_h, best_n, max_sweeps=40, use_random_order=True)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
