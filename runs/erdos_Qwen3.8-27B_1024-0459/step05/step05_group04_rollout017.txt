def make_swap_delta(h, i, j, n):

def subgradient_descent(h, n, max_iters=5000, top_k=3, lr_init=0.1, lr_decay=0.999):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    dx = 2.0 / n
    start = time.time()

            break
        
        g, corr = corr_gradient(h, n, top_k)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new

def softmax_slsqp(h, n, beta, maxiter=300):
    """SLSQP with softmax smoothing of the max."""
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return m * (2.0 / n)
        return (m + np.log(s) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if np_ <= 0:
                    continue
                if off == 0:
                    g[:] += wk * (1.0 - 2.0 * x)
                else:
                    g[:np_] += wk * (1.0 - x[off:])
                    g[off:] -= wk * x[:np_]
            else:
                ao = -off
                np_ = n - ao
                if np_ <= 0:
                    continue
                if ao == 0:
                    g[:] += wk * (1.0 - 2.0 * x)
                else:
                    g[ao:] += wk * (1.0 - x[:np_])
                    g[:np_] -= wk * x[:np_]
        return g
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    h0 = h.copy()
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.success or (hasattr(res, 'fun') and res.fun < obj(h0)):
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h0)
            if c5_new < c5_old:
                return h_new, c5_new
    except Exception:
        pass
    
    c5_old = compute_c5(h0)
    return h0, c5_old

def top_shifts_equalize(h, n, top_k=5, max_iters=300):
    """Directly reduce the top-k shifts by moving mass away from them."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        
        # Compute average gradient of top-k
        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                if off == 0:
                    g_avg[:] += (1.0 - 2.0 * h)
                else:
                    g_avg[:np_] += (1.0 - h[off:])
                    g_avg[off:] -= h[:np_]
            else:
                ao = -off
                np_ = n - ao
                if ao == 0:
                    g_avg[:] += (1.0 - 2.0 * h)
                else:
                    g_avg[ao:] += (1.0 - h[:np_])
                    g_avg[:np_] -= h[:np_]
        g_avg /= top_k
        
        # Direction: negative gradient (to reduce correlation)
        d = -g_avg
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in [0.05, 0.02, 0.01, 0.03, 0.08, 0.005, 0.015, 0.1]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def compute_corr(h):

def project_feasible(h, n):

def make_swap_delta(h, i, j, n):
    r = 2 * n - 1
    dr = np.zeros(r)
    lo = max(i, 0)
    hi = min(i + n - 1, r - 1)
    if lo <= hi:
        ks = np.arange(lo, hi + 1)
        dr[lo:hi+1] += -(1.0 - h[i + n - 1 - ks])
    lo2 = max(n - 1 - i, 0)
    hi2 = min(2 * (n - 1) - i, r - 1)
    if lo2 <= hi2:
        ks = np.arange(lo2, hi2 + 1)
        dr[lo2:hi2+1] += h[i + ks - (n - 1)]
    lo3 = max(j, 0)

def corr_gradient(h, n, idx):
    """Compute gradient of average of correlations at given indices."""
    g = np.zeros(n)
    for k in idx:
        s = k - (n - 1)
        for j in range(n):
            val = 1.0
            js = j + s
            if 0 <= js < n:
                val -= h[js]
            jm = j - s
            if 0 <= jm < n:
                val -= h[jm]
            g[j] += val
    g /= len(idx)
    return g

def corr_gradient_fast(h, n, idx):
    """Faster gradient using vectorized operations."""
    g = np.zeros(n)
    for k in idx:
        s = k - (n - 1)
        if s == 0:
            g[:] += 1.0 - 2.0 * h
        elif s > 0:
            np_ = n - s
            g[:np_] += 1.0 - h[s:]
            g[s:] -= h[:np_]
        else:
            ao = -s
            np_ = n - ao
            g[ao:] += 1.0 - h[:np_]
            g[:np_] -= h[:np_]
    g /= len(idx)
    return g

def softmax_slsqp(h, n, beta, maxiter=300):
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return m * (2.0 / n)
        return (m + np.log(s) / beta) * (2.0 / n)

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            if s == 0:
                g[:] += wk * (1.0 - 2.0 * x)
            elif s > 0:
                np_ = n - s
                if np_ > 0:
                    g[:np_] += wk * (1.0 - x[s:])
                    g[s:] -= wk * x[:np_]
            else:
                ao = -s
                np_ = n - ao
                if np_ > 0:
                    g[ao:] += wk * (1.0 - x[:np_])
                    g[:np_] -= wk * x[:np_]
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    h0 = h.copy()
    obj0 = obj(h0)
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if hasattr(res, 'fun') and res.fun < obj0:
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h0)
            if c5_new < c5_old:
                return h_new, c5_new
    except Exception:
        pass

    c5_old = compute_c5(h0)
    return h0, c5_old

def subgradient_descent(h, n, max_iters=5000, top_k=5, lr_init=0.05, lr_decay=0.9995):

    best_h = h.copy()

        if lr < 1e-7:
            break

            step = lr * scale

def top_shifts_equalize(h, n, top_k=8, max_iters=300):

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary local search at parent resolution
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    t1 = min(250, int(remaining() * 0.22))
    if t1 > 30:
        h_b1, c5_b1 = binary_local_search(h_bin, n, time_limit=t1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()
            best_n = n

    # Phase 2: Alternative binary starts
    for start_name in ['thue_morse', 'random_antiperiodic']:
        if remaining() < 40:
            break
        if start_name == 'thue_morse':
            sp = thue_morse(n)
        else:
            # Anti-periodic: h[i] = 1 - h[i + n/2]
            half = n // 2
            sp = np.zeros(n)
            sp[:half] = 1.0
            sp[half:] = 0.0
            # Add some randomness
            for _ in range(n // 8):
                i = rng.randint(half)
                sp[i] = 1.0 - sp[i]
                sp[i + half] = 1.0 - sp[i + half]
            n_o = int(np.sum(sp))
            if n_o != n_half:
                if n_o > n_half:
                    idx = np.where(sp == 1)[0]
                    sp[idx[:n_o - n_half]] = 0
                else:
                    idx = np.where(sp == 0)[0]
                    sp[idx[:n_half - n_o]] = 1

        sp_c5 = compute_c5(sp)
        if sp_c5 < best_c5 + 0.02:
            t_bi = min(100, int(remaining() * 0.09))
            if t_bi > 15:
                h_b2, c5_b2 = binary_local_search(sp, n, time_limit=t_bi, rng=rng)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_b2.copy()
                    best_n = n

    # Phase 3: Continuous refinement at parent resolution
    h = best_h.copy()
    n_cur = best_n

    # Subgradient descent (fast, helps escape SLSQP local minima)
    if remaining() > 30:
        h_sg, c5_sg = subgradient_descent(h, n_cur, max_iters=4000, top_k=5, lr_init=0.04, lr_decay=0.9997)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            h = h_sg.copy()

    # SLSQP with increasing beta
    for beta in [50, 200, 1000, 5000, 20000, 80000]:
        if remaining() < 12:
            break
        h_new, c5_new = softmax_slsqp(h, n_cur, beta, maxiter=250)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Top-shifts equalization
    if remaining() > 15:
        h_eq, c5_eq = top_shifts_equalize(h, n_cur, top_k=10, max_iters=250)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Another subgradient pass
    if remaining() > 20:
        h_sg2, c5_sg2 = subgradient_descent(h, n_cur, max_iters=3000, top_k=8, lr_init=0.02, lr_decay=0.9998)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            h = h_sg2.copy()

    # Phase 4: Try higher resolutions
    higher_resolutions = sorted(set([n_cur + 32, n_cur + 64, n_cur + 128]))
    higher_resolutions = [r for r in higher_resolutions if r <= 1024 and r % 2 == 0]

    for n_t in higher_resolutions[:3]:
        if remaining() < 50:
            break

        h_interp = interp_to(best_h, best_n, n_t)
        h_interp = project_feasible(h_interp, n_t)

        # Binary at higher resolution
        h_bin_t = (h_interp > 0.5).astype(float)
        nt_half = n_t // 2
        nt_ones = int(np.sum(h_bin_t))
        if nt_ones != nt_half:
            if nt_ones > nt_half:
                idx = np.where(h_bin_t == 1)[0]
                h_bin_t[idx[:nt_ones - nt_half]] = 0
            else:
                idx = np.where(h_bin_t == 0)[0]
                h_bin_t[idx[:nt_half - nt_ones]] = 1

        best_at_n_t = h_interp.copy()
        best_c5_t = compute_c5(h_interp)

        t_bi = min(50, int(remaining() * 0.05))
        if t_bi > 10:
            h_b3, c5_b3 = binary_local_search(h_bin_t, n_t, time_limit=t_bi, rng=rng)
            if c5_b3 < best_c5_t:
                best_c5_t = c5_b3
                best_at_n_t = h_b3.copy()

        # Continuous refinement at higher resolution
        h_c = best_at_n_t.copy()
        h_c = project_feasible(h_c, n_t)

        for beta in [200, 2000, 20000]:
            if remaining() < 12:
                break
            h_new, c5_new = softmax_slsqp(h_c, n_t, beta, maxiter=200)
            if c5_new < best_c5_t:
                best_c5_t = c5_new
                h_c = h_new.copy()

        if remaining() > 15:
            h_eq_t, c5_eq_t = top_shifts_equalize(h_c, n_t, top_k=8, max_iters=150)
            if c5_eq_t < best_c5_t:
                best_c5_t = c5_eq_t
                h_c = h_eq_t.copy()

        c5_t_final = compute_c5(h_c)
        if c5_t_final < best_c5_t:
            best_c5_t = c5_t_final

        if best_c5_t < best_c5:
            best_c5 = best_c5_t
            best_h = h_c.copy()
            best_n = n_t

    # Phase 5: Final intensive polishing at best resolution
    n = best_n
    h = best_h.copy()

    # Multiple SLSQP passes with very high beta
    for beta in [50000, 150000, 300000, 500000]:
        if remaining() < 8:
            break
        h_new, c5_new = softmax_slsqp(h, n, beta, maxiter=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Top-shifts with many shifts
    if remaining() > 12:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=15, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()

    # Final subgradient
    if remaining() > 12:
        h_sg, c5_sg = subgradient_descent(h, n, max_iters=3000, top_k=15, lr_init=0.005, lr_decay=0.9999)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
