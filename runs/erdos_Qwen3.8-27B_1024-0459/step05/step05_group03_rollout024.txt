import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def smooth_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(ex)) / beta) * dx


def corr_grad_weighted(h, weights):
    n = len(h)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-16:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - h[:np_])
            g[:np_] -= wk * h[off:off + np_]
        else:
            a = -off
            np_ = n - a
            g[:np_] += wk * (1.0 - h[a:a + np_])
            g[a:a + np_] -= wk * h[:np_]
    return g


def smooth_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -50.0, 50.0)
    ex = np.exp(sh)
    w = ex / np.sum(ex)
    dx = 2.0 / len(h)
    w = w * dx
    return corr_grad_weighted(h, w)


def slsqp_run(h, n, beta, maxiter=400):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return smooth_obj(x, beta)

    def _grad(x):
        return smooth_grad(x, beta)

    def _sc(x):
        return np.sum(x) - n / 2.0

    def _scj(x):
        return np.ones(n)

    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sc, 'jac': _scj}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < _obj(h0) - 1e-15:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0


def greedy_binary_ls(h, n, max_rounds=25):
    h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if remaining() < 5:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr_cur = compute_corr(h)
        c_max = np.max(corr_cur)
        best_improve = 1e-15
        best_pair = None
        for ia in range(len(ones)):
            pa = ones[ia]
            h[pa] = 0.0
            for ib in range(len(zeros)):
                pb = zeros[ib]
                h[pb] = 1.0
                nc = np.max(compute_corr(h))
                h[pb] = 0.0
                imp = c_max - nc
                if imp > best_improve:
                    best_improve = imp
                    best_pair = (pa, pb)
            h[pa] = 1.0
        if best_pair is None:
            break
        pa, pb = best_pair
        h[pa] = 0.0
        h[pb] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
    return h, best_c5

        rng = np.random.RandomState(42)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(total_rounds):
        if remaining() < 10:
            break
        # Greedy to convergence
        h, c5 = greedy_binary_ls(h, n, max_rounds=12)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Perturb: adaptive number of swaps
        if it < total_rounds // 3:
            pert_size = rng.randint(3, 8)
        elif it < 2 * total_rounds // 3:
            pert_size = rng.randint(2, 5)
        else:
            pert_size = rng.randint(1, 3)

        h_pert = h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]

                i = rng.randint(len(ones))
                j = rng.randint(len(zeros))
                h_pert[ones[i]] = 0.0
                h_pert[zeros[j]] = 1.0
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
        h = h_pert

    return best_h, best_c5


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def multi_topk_subgrad(h, n, top_k_list=None, max_iters=500):
    """Subgradient method focusing on top-k active lags with decreasing k."""
    if top_k_list is None:
        top_k_list = [3, 5, 8, 12, 20]
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    mu = 0.8
    step = 0.03

    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        tk = top_k_list[min(it // (max_iters // len(top_k_list)), len(top_k_list) - 1)]
        top_idx = np.argsort(-corr)[:tk]
        c_top = corr[top_idx]
        w_top = c_top - c_top.min() + 1e-10
        w_top = w_top ** 3
        w_top /= w_top.sum()
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = w_top
        weights *= dx
        g = corr_grad_weighted(h, weights)
        g -= np.mean(g)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        step *= 0.9997
        if it % 200 == 0:
            mu = min(0.93, mu + 0.01)
    return best_h, best_c5
