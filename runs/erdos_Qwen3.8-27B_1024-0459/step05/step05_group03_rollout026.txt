_T0 = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(120):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)

    elif n_ones < target:

def sa_binary(h_bin, n, t_start, t_end, max_iters, seed=42):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    h = h_bin.astype(float).copy()
    corr = compute_corr(h)
    cur_e = float(np.max(corr))
    best_e = cur_e
    best_h = h.copy()
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    for it in range(max_iters):
        if remaining() < 3:
            break
        frac = it / max_iters
        t = t_start * (t_end / t_start) ** frac

        ii = rng.randint(len(ones_list))
        jj = rng.randint(len(zeros_list))
        i = ones_list[ii]
        j = zeros_list[jj]

        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_e = float(np.max(new_corr))

            ones_list[ii] = j
            zeros_list[jj] = i
            if cur_e < best_e:
                best_e = cur_e
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_e * (2.0 / n)

        for ia in range(len(ones)):

            best_max = cur_max
            best_h = h.copy()
