        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(shifted)
        w = ex / np.sum(ex)
        g = np.zeros(n_local)
        for k_idx in range(2 * n_local - 1):
            wk = w[k_idx]

                cnt = n_local - a

                g[a:a+cnt] -= wk * x[:cnt]

        return np.sum(x) - n / 2.0

        return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_rng = np.random.RandomState(42)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 3.0)


def smooth_max_val(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -50.0, 50.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def slsqp_opt(h, n, beta, maxiter=300):
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_val(x, beta) * (2.0 / n)

    def grad(x):
        nl = len(x)
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -50.0, 50.0)
        ex = np.exp(shifted)
        w = ex / np.sum(ex)
        g = np.zeros(nl)
        for k_idx in range(2 * nl - 1):
            wk = w[k_idx]
            if wk < 1e-14:
                continue
            off = k_idx - (nl - 1)
            if off >= 0:
                cnt = nl - off
                g[off:off + cnt] += wk * (1.0 - x[:cnt])
                g[:cnt] -= wk * x[off:off + cnt]
            else:
                a = -off
                cnt = nl - a
                g[:cnt] += wk * (1.0 - x[a:a + cnt])
                g[a:a + cnt] -= wk * x[:cnt]
        return g * (2.0 / nl)

    def eq(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq, 'jac': eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h0) - 1e-15:
            return h_new
    except Exception:
        pass
    return h0


def sa_binary(h_bin, n, max_iters=60000, T0=0.004, cooling=0.99997, rng=None):
    if rng is None:
        rng = _rng
    h = h_bin.astype(float).copy()
    m = n // 2
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = T0
    reheat_interval = 15000
    reheat_temp = 0.002

    for it in range(max_iters):
        if remaining() < 20:
            break
        if it > 0 and it % reheat_interval == 0 and T < reheat_temp:
            T = reheat_temp

        ia = rng.randint(m)
        ib = rng.randint(m)
        pa = ones[ia]
        pb = zeros[ib]

        h[pa] = 0.0
        h[pb] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones[ia] = pb
            zeros[ib] = pa
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[pa] = 1.0
            h[pb] = 0.0

        T *= cooling

    return best_h, best_c5


def greedy_binary(h_bin, n, max_rounds=15, rng=None):
    if rng is None:
        rng = _rng
    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if remaining() < 25:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        corr_cur = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr_cur) * dx

        best_improve = 1e-12
        best_pair = None
        total_pairs = len(ones) * len(zeros)

        if total_pairs <= 80000:
            for ia in range(len(ones)):
                pa = ones[ia]
                h[pa] = 0.0
                for ib in range(len(zeros)):
                    pb = zeros[ib]
                    h[pb] = 1.0
                    nc = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
                    h[pb] = 0.0
                    imp = c_max - nc
                    if imp > best_improve:
                        best_improve = imp
                        best_pair = (pa, pb)
                h[pa] = 1.0
        else:
            for _ in range(80000):
                pa = ones[rng.randint(len(ones))]
                pb = zeros[rng.randint(len(zeros))]
                h[pa] = 0.0
                h[pb] = 1.0
                nc = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
                h[pb] = 0.0
                h[pa] = 1.0
                imp = c_max - nc
                if imp > best_improve:
                    best_improve = imp
                    best_pair = (pa, pb)

        if best_pair is None:
            break
        pa, pb = best_pair
        h[pa] = 0.0
        h[pb] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def round_to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def perturb_binary(h_bin, n, n_swaps, rng=None):
    if rng is None:
        rng = _rng
    h = h_bin.astype(float).copy()
    for _ in range(n_swaps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        pa = ones[rng.randint(len(ones))]
        pb = zeros[rng.randint(len(zeros))]
        h[pa] = 0.0
        h[pb] = 1.0
    return h


def optimize_at_resolution(h_start, n, time_limit, rng=None):
    if rng is None:
        rng = _rng
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    # SLSQP progressive
    h = project_feasible(h_start, n)
    for beta in [50, 200, 1000, 5000, 20000, 80000, 200000]:
        if remaining() < time_limit:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=min(300, max(50, int(remaining() / 8))))
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Binary + SA + Greedy
    h_bin = round_to_binary(h, n)
    for trial in range(3):
        if remaining() < time_limit + 20:
            break
        if trial == 0:
            h_start_sa = h_bin.copy()
        elif trial == 1:
            h_start_sa = perturb_binary(h_bin, n, 8, rng)
        else:
            h_start_sa = perturb_binary(h_bin, n, 20, rng)

        h_sa, c5_sa = sa_binary(h_start_sa, n, max_iters=50000,
                                T0=0.003 + 0.002 * trial, cooling=0.99996, rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_bin = h_sa.copy()

        if remaining() > time_limit + 15:
            h_g, c5_g = greedy_binary(h_bin, n, max_rounds=12, rng=rng)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
                h_bin = h_g.copy()

    # SLSQP from best binary
    if remaining() > time_limit + 15:
        h = best_h.copy()
        for beta in [20000, 100000, 300000]:
            if remaining() < time_limit + 10:
                break
            h_new = slsqp_opt(h, n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    return best_h, best_c5
