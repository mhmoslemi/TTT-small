def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def round_to_binary(h, n):
    """Round h to binary, preserving exactly n/2 ones."""
    h = np.array(h, dtype=float)
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    target_ones = n // 2
    if len(ones) > target_ones:
        # Remove excess ones (those closest to 0.5)
        excess = ones[np.argsort(h[ones])]
        h[excess[:len(ones) - target_ones]] = 0
    elif len(ones) < target_ones:
        need = target_ones - len(ones)
        add = zeros[np.argsort(-h[zeros])]
        h[add[:need]] = 1
    h[h >= 0.5] = 1.0
    h[h < 0.5] = 0.0
    return h


def swap_delta(h, a, b, n):
    """Compute delta in correlations when swapping h[a]=1 to 0 and h[b]=0 to 1."""
    delta = np.zeros(n)
    # Term 1: -(1-h[a+k]) for k=0..n-1-a
    la = n - a
    if la > 0:
        delta[:la] -= (1.0 - h[a:a + la])
    # Term 2: +h[a-k] for k=0..a
    lb = a + 1
    if lb > 0:
        delta[:lb] += h[a::-1][:lb]
    # Term 3: +(1-h[b+k]) for k=0..n-1-b
    lc = n - b
    if lc > 0:
        delta[:lc] += (1.0 - h[b:b + lc])
    # Term 4: -h[b-k] for k=0..b
    ld = b + 1
    if ld > 0:
        delta[:ld] -= h[b::-1][:ld]
    return delta


def binary_best_improvement(h, n, max_passes=200, time_limit=None):
    """Best-improvement local search on binary solutions."""
    global _T0, _BUDGET
    h = np.array(h, dtype=float)
    h = round_to_binary(h, n)
    corr = compute_corr(h)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    start = time.time()
    
    for pas in range(max_passes):
        if time_limit and time.time() - start > time_limit:
            break
        if _BUDGET - (time.time() - _T0) < 5:
            break
        
        best_improvement = 0
        best_a, best_b = -1, -1
        best_max_after = np.max(corr[:(2*n-1)-(n-1):1]) if False else None
        
        # We need to track the full correlation (2n-1 values)
        # But for the objective, we only care about max(corr) * (2/n)
        # corr from np.correlate has 2n-1 values
        # The interior (shift 0) is at index n-1
        # We care about the max over all 2n-1 values
        
        full_corr = compute_corr(h)
        current_max = np.max(full_corr)
        
        improved = False
        for a in ones:
            for b in zeros:
                if a == b:
                    continue
                d = swap_delta(h, a, b, n)
                # d has length n (shifts 0 to n-1)
                # full_corr has length 2n-1, indexed from -(n-1) to n-1
                # full_corr[n-1+k] corresponds to shift +k
                # full_corr[n-1-k] corresponds to shift -k
                # d[k] for k=0..n-1 corresponds to shift +k
                # We also need the effect on negative shifts
                # By symmetry of the swap, delta for shift -k is the same as for shift +k
                # Actually no, let's just compute the full thing
                
                # Apply to positive shifts (indices n-1 to 2n-2 in full_corr)
                new_full = full_corr.copy()
                new_full[n-1:] += d  # shifts 0, 1, ..., n-1
                # For negative shifts (indices 0 to n-2), shift -k:
                # The delta for shift -k is the same computation with k replaced
                # Actually, delta for shift -k when swapping a→0, b→1:
                # corr[-k] = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
                # This is the same as corr[k] with h reversed... 
                # Let me just compute it directly for negative shifts too
                
                # Actually, let's use the fact that:
                # corr_full[j] for j=0..2n-2 corresponds to correlation at "offset" j-(n-1)
                # For offset = k (positive): sum_{i} h[i]*(1-h[i+k]) for valid i
                # For offset = -k: sum_{i} h[i]*(1-h[i-k]) for valid i
                # 
                # For the swap, the delta at offset -k is:
                # Same formula but with i and i+k swapped... 
                # Actually by the definition of correlate, the negative shifts are
                # computed as: for offset = -k, sum_i h[i+k]*(1-h[i]) for valid i
                # = sum_j h[j]*(1-h[j-k]) for j = k..n-1
                # This is NOT the same as the positive shift delta.
                
                # Let me just compute the negative shift delta directly.
                # For offset = -k (k > 0): corr = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
                # Swap effect:
                # h[a] 1→0: affects terms where i=a (if a>=k, a<n): change -(1-h[a-k])
                #                    and where i-k=a i.e. i=a+k (if a+k<n): change +h[a+k]
                # h[b] 0→1: affects terms where i=b (if b>=k, b<n): change +(1-h[b-k])
                #                    and where i-k=b i.e. i=b+k (if b+k<n): change -h[b+k]
                
                neg_delta = np.zeros(n)  # neg_delta[k] for k=0..n-1, neg_delta[0]=0
                for k in range(1, n):
                    val = 0.0
                    # h[a] 1→0
                    # i=a valid if a>=k: change -(1-h[a-k])
                    if a >= k:
                        val -= (1.0 - h[a - k])
                    # i-k=a, i=a+k valid if a+k<n: change +h[a+k]
                    if a + k < n:
                        val += h[a + k]
                    # h[b] 0→1
                    # i=b valid if b>=k: change +(1-h[b-k])
                    if b >= k:
                        val += (1.0 - h[b - k])
                    # i-k=b, i=b+k valid if b+k<n: change -h[b+k]
                    if b + k < n:
                        val -= h[b + k]
                    neg_delta[k] = val
                
                new_full[:n-1] += neg_delta[::-1]  # indices 0..n-2 are shifts -(n-1)..-1
                # new_full[0] is shift -(n-1), new_full[n-2] is shift -1
                # neg_delta[k] is for shift -k, so new_full[n-1-k] += neg_delta[k]
                # i.e., new_full[0:n-1] += neg_delta[n-1:0:-1]
                
                new_max = np.max(new_full)
                if new_max < current_max - 1e-12:
                    if best_a == -1 or new_max < best_max_after:
                        best_max_after = new_max
                        best_a, best_b = a, b
                    improved = True
        
        if not improved or best_a == -1:
            break
        
        # Apply the best swap
        h[best_a] = 0
        h[best_b] = 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5


def binary_sa(h, n, time_limit=60.0):
    """Simulated annealing on binary solutions."""
    global _T0, _BUDGET
    rng = np.random.RandomState(42)
    h = round_to_binary(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    full_corr = compute_corr(h)
    current_max = np.max(full_corr)
    current_c5 = current_max * 2.0 / n
    
    T = 0.5
    T_min = 0.001
    cooling = 0.995
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    start = time.time()
    
    while time.time() - start < time_limit and T > T_min:
        if _BUDGET - (time.time() - _T0) < 5:
            break
        
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        if a == b:
            continue
        
        d = swap_delta(h, a, b, n)
        new_full = full_corr.copy()
        new_full[n-1:] += d
        
        # Negative shifts
        neg_delta = np.zeros(n)
        for k in range(1, n):
            val = 0.0
            if a >= k:
                val -= (1.0 - h[a - k])
            if a + k < n:
                val += h[a + k]
            if b >= k:
                val += (1.0 - h[b - k])
            if b + k < n:
                val -= h[b + k]
            neg_delta[k] = val
        new_full[:n-1] += neg_delta[n-1:0:-1]
        
        new_max = np.max(new_full)
        delta_c5 = (new_max - current_max) * 2.0 / n
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            h[a] = 0
            h[b] = 1
            full_corr = new_full
            current_max = new_max
            current_c5 = new_max * 2.0 / n
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        T *= cooling
    
    return best_h, best_c5


def multi_start_binary(h_init, n, n_starts=5, time_per_start=30.0, rng=None):
    """Try multiple initializations and local searches."""
    global _T0, _BUDGET
    if rng is None:
        rng = np.random.RandomState(42)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    initializations = []
    # Start 1: parent rounded
    initializations.append(round_to_binary(h_init, n))
    # Start 2: random
    for _ in range(min(n_starts - 1, 4)):
        arr = np.zeros(n)
        idx = rng.choice(n, size=n//2, replace=False)
        arr[idx] = 1.0
        initializations.append(arr)
    
    for h_start in initializations:
        if _BUDGET - (time.time() - _T0) < 10:
            break
        h_bi, c5_bi = binary_best_improvement(h_start, n, max_passes=50, time_limit=min(15.0, time_per_start))
        if c5_bi < best_c5:
            best_c5 = c5_bi
            best_h = h_bi.copy()
        # SA from the best-improvement result
        if _BUDGET - (time.time() - _T0) < 10:
            break
        h_sa, c5_sa = binary_sa(h_bi, n, time_limit=min(20.0, time_per_start))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    return best_h, best_c5


def continuous_slsqp(h, n, beta=5000, maxiter=400):
    """SLSQP optimization with smooth max approximation."""
    def sm_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = ex.sum()
        if s < 1e-30:
            return m * 2.0 / n
        return (m + np.log(s) / beta) * 2.0 / n

    def sm_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = ex.sum()
        if s < 1e-30:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w = ex / s
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def sc(x):
        return np.sum(x) - n / 2.0

    def scj(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sc, 'jac': scj}]
    try:
        res = minimize(sm_obj, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            return h_new
    except Exception:
        pass
    return project_feasible(h, n)


def interp_to(src_h, src_n, dst_n):
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 512
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _T0) - 5.0)

    n = parent_n
    h = parent_h.copy()

    # Phase 1: Binary local search at parent resolution
    if remaining() > 30:
        t_alloc = min(120.0, remaining() * 0.25)
        h_bin, c5_bin = multi_start_binary(h, n, n_starts=4, time_per_start=min(30.0, t_alloc / 4), rng=rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n

    # Additional best-improvement passes on the best binary solution
    if remaining() > 30:
        h_bi, c5_bi = binary_best_improvement(best_h, best_n, max_passes=100, time_limit=min(60.0, remaining() * 0.3))
        if c5_bi < best_c5:
            best_c5 = c5_bi
            best_h = h_bi.copy()
            best_n = best_n

    # Phase 2: SA refinement
    if remaining() > 30:
        h_sa, c5_sa = binary_sa(best_h, best_n, time_limit=min(60.0, remaining() * 0.3))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = best_n

    # Phase 3: Continuous SLSQP polish (starting from best binary)
    if remaining() > 20:
        h_cont = best_h.copy()
        for beta in [3000, 8000, 20000]:
            if remaining() < 10:
                break
            h_cont = continuous_slsqp(h_cont, best_n, beta=beta, maxiter=300)
            c5_cont = compute_c5(h_cont)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont.copy()

    # Phase 4: Try different resolutions
    res_list = []
    base_n = best_n
    for mult in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
        rn = int(round(base_n * mult / 16) * 16)
        if 128 <= rn <= 1024 and rn % 2 == 0 and rn != base_n:
            res_list.append(rn)
    res_list = sorted(set(res_list))

    for n_t in res_list:
        if remaining() < 40:
            break
        
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_int = compute_c5(h_feas)
        
        # Binary local search at this resolution
        t_budget = min(60.0, remaining() * 0.4)
        h_bin_t, c5_bin_t = multi_start_binary(h_feas, n_t, n_starts=3, time_per_start=min(20.0, t_budget / 3), rng=rng)
        c5_t = min(c5_bin_t, c5_int)
        h_t = h_bin_t if c5_bin_t < c5_int else h_feas
        
        # More best-improvement
        if remaining() > 20:
            h_t2, c5_t2 = binary_best_improvement(h_t, n_t, max_passes=50, time_limit=min(30.0, remaining() * 0.3))
            if c5_t2 < c5_t:
                c5_t = c5_t2
                h_t = h_t2
        
        # SA
        if remaining() > 15:
            h_t3, c5_t3 = binary_sa(h_t, n_t, time_limit=min(20.0, remaining() * 0.2))
            if c5_t3 < c5_t:
                c5_t = c5_t3
                h_t = h_t3
        
        # SLSQP
        if remaining() > 10:
            h_t4 = continuous_slsqp(h_t, n_t, beta=5000, maxiter=200)
            c5_t4 = compute_c5(h_t4)
            if c5_t4 < c5_t:
                c5_t = c5_t4
                h_t = h_t4
        
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n_t

    # Phase 5: Intensive refinement at best resolution
    n = best_n
    h = best_h.copy()
    
    for cycle in range(3):
        if remaining() < 20:
            break
        
        # Binary local search
        h_bi, c5_bi = binary_best_improvement(h, n, max_passes=80, time_limit=min(40.0, remaining() * 0.3))
        if c5_bi < best_c5:
            best_c5 = c5_bi
            best_h = h_bi.copy()
            h = h_bi.copy()
            n = best_n
        
        # SA
        if remaining() > 20:
            h_sa, c5_sa = binary_sa(h, n, time_limit=min(30.0, remaining() * 0.3))
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h = h_sa.copy()
        
        # SLSQP
        if remaining() > 10:
            for beta in [10000, 30000]:
                if remaining() < 8:
                    break
                h = continuous_slsqp(h, n, beta=beta, maxiter=300)
                c5_chk = compute_c5(h)
                if c5_chk < best_c5:
                    best_c5 = c5_chk
                    best_h = h.copy()

    # Phase 6: Final fine-tuning
    # Try individual coordinate adjustments
    if remaining() > 5:
        h = best_h.copy()
        c5_cur = compute_c5(h)
        for it in range(200):
            if remaining() < 3:
                break
            improved = False
            idx = rng.randint(n)
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h_try = h.copy()
                # Adjust h[idx] and compensate elsewhere
                h_try[idx] = np.clip(h_try[idx] + delta, 0.0, 1.0)
                h_try = project_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-12:
                    h = h_try
                    c5_cur = c5_try
                    improved = True
                    break
            if not improved:
                break
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Final SLSQP polish
    if remaining() > 5:
        h_final = continuous_slsqp(best_h, best_n, beta=50000, maxiter=500)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
