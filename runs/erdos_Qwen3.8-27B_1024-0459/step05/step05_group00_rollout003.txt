import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_corr_k(h, k_idx, n):
    g = np.zeros(n)
    offset = k_idx - (n - 1)
    if offset >= 0:
        npairs = n - offset
        g[offset:] += (1.0 - h[:npairs])
        g[:npairs] -= h[offset:]
    else:
        a = -offset
        npairs = n - a
        g[:npairs] += (1.0 - h[a:])
        g[a:] -= h[:npairs]
    return g


def smooth_max_evaluate(x, n, beta):
    corr = compute_corr(x)
    m = np.max(corr)
    scaled = beta * (corr - m)
    scaled = np.clip(scaled, -400.0, 400.0)
    ex = np.exp(scaled)
    s = ex.sum()
    dx = 2.0 / n
    val = (m + np.log(s) / beta) * dx
    w = ex / s
    g = np.zeros(n)
    wc = w * dx
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-14:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return val, g


def slsqp_optimize(h, n, beta, maxiter=1000):
    def obj(x):
        v, _ = smooth_max_evaluate(x, n, beta)
        return v

    def grad(x):
        _, g = smooth_max_evaluate(x, n, beta)
        return g

    def eq_fun(x):
        return np.sum(x) - n / 2.0

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]

    best_x = h.copy()
    best_f = compute_c5(h)
    try:
        res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            f_new = compute_c5(h_new)
            if f_new < best_f:
                best_x = h_new
                best_f = f_new
    except Exception:
        pass
    return best_x
