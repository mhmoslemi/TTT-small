import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TOPK = 5
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def proj(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Distribute remainder among values closest to 0.5
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def corr_grad_single(h, k, n):
    offset = k - (n - 1)
    g = np.zeros(n)
    if offset >= 0:
        np_ = n - offset
        g[offset:offset + np_] += (1.0 - h[:np_])
        g[:np_] -= h[offset:offset + np_]
    else:
        ao = -offset
        np_ = n - ao
        g[:np_] += (1.0 - h[ao:ao + np_])
        g[ao:ao + np_] -= h[:np_]
    return g


def _sm_fun(x):

def projected_gradient_nesterov(h, n, top_k=10, max_iters=10000, step_init=0.02):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    momentum = 0.85
    step = step_init

    for it in range(max_iters):
        if remaining() < 3:
            break

        corr = compute_corr(h)
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        idx = np.argsort(-corr)[:top_k]
        g = np.zeros(n)
        c_top = corr[idx]
        w = (c_top - np.mean(c_top)) + 0.05
        w = np.maximum(w, 0.005)
        w /= np.sum(w)
        for j, ki in enumerate(idx):
            g += w[j] * corr_grad_single(h, ki, n)

        g -= np.mean(g)
        g[(h >= 1.0 - 1e-9)] = 0.0
        g[(h <= 1e-9)] = 0.0
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        v = momentum * v - step * g
        h_new = proj(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 + 0.002:
            h = h_new
            if c5_new <= c5:
                step = min(step * 1.01, 0.12)
        else:
            h = best_h.copy()
            v *= 0.5
            step *= 0.75

        if it % 500 == 0 and it > 0:
            step *= 0.9
            momentum = min(momentum + 0.003, 0.93)

    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def fourier_init(n, seed, n_modes=4, max_freq=5):
    """Generate initialization with controlled frequency content."""
    rng = np.random.RandomState(seed)
    h = np.ones(n) * 0.5
    for m in range(n_modes):
        freq = rng.randint(1, max_freq + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        h += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    h = np.clip(h, 0, 1)
    h = proj(h, n)
    return h

            sz = zeros[np.argsort(-g[zeros])[:np_z]]
            for a in range(np_o):
                for b in range(a + 1, np_o):
                    for c in range(np_z):
                        for d in range(c + 1, np_z):
                            i1, i2 = so[a], so[b]
                            j1, j2 = sz[c], sz[d]
                            if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                                continue
                            h_bin[i1] = 0.0
                            h_bin[i2] = 0.0
                            h_bin[j1] = 1.0
                            h_bin[j2] = 1.0
                            nc = compute_c5(h_bin)
                            h_bin[i1] = 1.0
                            h_bin[i2] = 1.0
                            h_bin[j1] = 0.0
                            h_bin[j2] = 0.0
                            delta = nc - cur_c5
                            if delta < best_delta:
                                best_delta = delta
                                best_move = (i1, i2, j1, j2)
                                move_type = 1

        if best_move is None or best_delta >= 0:
            found = False
            for _ in range(10):
                if pair_flips and len(ones) > 1 and len(zeros) > 1 and rng.random() < 0.35:
                    i1 = int(ones[rng.randint(len(ones))])
                    i2 = int(ones[rng.randint(len(ones))])
                    j1 = int(zeros[rng.randint(len(zeros))])
                    j2 = int(zeros[rng.randint(len(zeros))])
                    if len(set([i1, i2, j1, j2])) < 4:
                        continue
                    h_bin[i1] = 0.0; h_bin[i2] = 0.0; h_bin[j1] = 1.0; h_bin[j2] = 1.0
                    nc = compute_c5(h_bin)
                    h_bin[i1] = 1.0; h_bin[i2] = 1.0; h_bin[j1] = 0.0; h_bin[j2] = 0.0
                    delta = nc - cur_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_move = (i1, i2, j1, j2)
                        move_type = 1
                        found = True
                else:
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    if i == j:
                        continue
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    nc = compute_c5(h_bin)
                    h_bin[i] = 1.0
                    h_bin[j] = 0.0
                    delta = nc - cur_c5
                    if delta < best_delta:
                        best_delta = delta
                        best_move = (i, j)
                        move_type = 0
                        found = True
            if not found:
                break

        if move_type == 0:
            i, j = best_move
            h_bin[i] = 0.0
            h_bin[j] = 1.0
        else:
            i1, i2, j1, j2 = best_move
            h_bin[i1] = 0.0; h_bin[i2] = 0.0; h_bin[j1] = 1.0; h_bin[j2] = 1.0

        cur_c5 = compute_c5(h_bin)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_bin.copy()

    return best_h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def slsqp_pass(h, n, beta, maxiter=150):
    global _N, _BETA, _TOPK, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    _TOPK = 6
    h = proj(h, n)

    def _sc(x):
        return np.sum(x) - _TGT

    def _scj(x):
        return np.ones(_N)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_fun, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return proj(res.x, n)
    except Exception:
        return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TOPK, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Quick refinement at parent resolution
    n = best_n
    h = best_h.copy()
    for beta in [200, 800, 3000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=10, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Upsample to n=768 and optimize
    n_hi = 768
    if remaining() > 120:
        h_hi = interp_to(best_h, best_n, n_hi)
        h_hi = proj(h_hi, n_hi)
        for beta in [500, 2000, 8000, 25000]:
            if remaining() < 10:
                break
            h_hi = lbfgs_pass(h_hi, n_hi, beta, top_k=12, maxiter=500)
        h_hi, c5_hi = top_equalize(h_hi, n_hi, top_k=10, max_iters=250)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_hi
            h = h_hi.copy()

    # Phase 3: Binary local search at current best
    if remaining() > 100:
        n_cur = best_n
        h_bin_init = (best_h > 0.5).astype(float)
        h_bl, c5_bl = binary_local_search(h_bin_init, n_cur, max_iters=8000, top_k=5, pair_flips=True)
        c5_bl = compute_c5(h_bl)
        h_ref = h_bl.copy()
        for beta in [8000, 30000, 80000]:
            if remaining() < 8:
                break
            h_ref = lbfgs_pass(h_ref, n_cur, beta, top_k=6, maxiter=350)
        c5_ref = compute_c5(h_ref)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h = h_ref.copy()
        elif c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            h = h_bl.copy()

    # Phase 4: Try n=1024
    if remaining() > 150:
        n_hh = 1024
        h_hh = interp_to(best_h, best_n, n_hh)
        h_hh = proj(h_hh, n_hh)
        for beta in [1000, 5000, 20000]:
            if remaining() < 10:
                break
            h_hh = lbfgs_pass(h_hh, n_hh, beta, top_k=12, maxiter=450)
        h_hh, c5_hh = top_equalize(h_hh, n_hh, top_k=10, max_iters=200)
        c5_hh = compute_c5(h_hh)
        h_bin2 = (h_hh > 0.5).astype(float)
        h_bl2, c5_bl2 = binary_local_search(h_bin2, n_hh, max_iters=5000, top_k=5, pair_flips=True)
        h_ref2 = h_bl2.copy()
        for beta in [10000, 50000]:
            if remaining() < 8:
                break
            h_ref2 = lbfgs_pass(h_ref2, n_hh, beta, top_k=8, maxiter=300)
        c5_ref2 = compute_c5(h_ref2)
        c5_cand = min(c5_hh, c5_bl2, c5_ref2)
        if c5_ref2 <= c5_cand:
            h_cand = h_ref2
        elif c5_bl2 <= c5_cand:
            h_cand = h_bl2
        else:
            h_cand = h_hh
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
            best_n = n_hh
            h = h_cand.copy()

    # Phase 5: Nesterov at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 40:
        h_pg, c5_pg = nesterov_descend(h, n, top_k=12, max_iters=8000, step_init=0.012)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 6: High-beta L-BFGS sweep
    for beta in [15000, 60000, 200000]:
        if remaining() < 8:
            break
        h_new = lbfgs_pass(h, n, beta, top_k=5, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Final equalize
    if remaining() > 12:
        h_eq, c5_eq = top_equalize(h, n, top_k=10, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 8: Final binary polish
    if remaining() > 30:
        h_bf = (best_h > 0.5).astype(float)
        h_bf, c5_bf = binary_local_search(h_bf, n, max_iters=4000, top_k=4, pair_flips=True)
        h_rf = h_bf.copy()
        for beta in [30000, 100000]:
            if remaining() < 5:
                break
            h_rf = lbfgs_pass(h_rf, n, beta, top_k=4, maxiter=250)
        c5_rf = compute_c5(h_rf)
        if c5_rf < best_c5:
            best_c5 = c5_rf
            best_h = h_rf.copy()
        elif c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Phase 9: SLSQP final
    if remaining() > 10:
        h_sl = slsqp_pass(best_h, n, 80000, maxiter=100)
        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Final feasibility
    best_h = proj(best_h, best_n)
    final_c5 = compute_c5(best_h)
    best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
