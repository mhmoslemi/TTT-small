def slsqp_minimax(h, n, beta, maxiter=500):
    """SLSQP with soft-max approximation."""
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    state = {'n': n, 'beta': beta, 'target': target}
    
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(state['beta'] * sh, -50.0, 50.0)
        ex = np.exp(cl)
        return (m + np.log(np.sum(ex)) / state['beta']) * (2.0 / state['n'])
    
    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(state['beta'] * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(state['n'])
        wc = w * (2.0 / state['n'])
        for k in range(2 * state['n'] - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (state['n'] - 1)
            if off >= 0:
                np_ = state['n'] - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = state['n'] - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def _cons_fun(x):
        return np.sum(x) - state['target']
    
    def _cons_jac(x):
        return np.ones(state['n'])
    
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < _obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            # Remove from positions that contribute most to max corr
            corr = compute_corr(h)
            max_c = corr.max()
            ones = np.where(h == 1)[0]
            # Sort by contribution (in simple shifts)
            contrib = np.zeros(n)
            for k in range(2*n-1):
                off = k - (n-1)
                if off >= 0:
                    np_ = n - off
                    contrib[off:off+np_] += h[:np_]
                else:
                    ao = -off
                    np_ = n - ao
                    contrib[:np_] += h[ao:ao+np_]
            sorted_ones = ones[np.argsort(-contrib[ones])]
            h[sorted_ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

        return (m + np.log(np.sum(ex)) / beta) * (2.0 / n)

    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def _cons_fun(x):
        return np.sum(x) - target

    def _cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.fun < _obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def active_set_optimize(h, n, time_limit=60):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    while time.time() - start < time_limit and remaining() > 10:
        corr = compute_corr(h)
        max_val = np.max(corr)
        thresh = max_val - 0.03 * max_val
        active = np.where(corr >= thresh)[0]

        g = np.zeros(n)
        for k in active:
            g += corr_gradient_one(h, k, n)
        g /= max(len(active), 1)

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn

        improved = False
        for step in [0.003, 0.008, 0.015, 0.03, 0.005, 0.06, 0.01, 0.02, 0.04, 0.002]:
            h_new = project_feasible(h - step * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                improved = True
                break
        if not improved:
            break

    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 0 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            h[ones[:n_ones - target]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target - n_ones]] = 1
    return h

def continuous_sa(h, n, time_limit=120, rng=None):
    """Simulated annealing on continuous variables."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()
    T = 0.05
    T_min = 0.001
    cooling = 0.995

    while time.time() - start < time_limit and remaining() > 10 and T > T_min:
        # Random perturbation
        n_pert = rng.randint(3, 15)
        idx = rng.choice(n, n_pert, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.uniform(-0.1, 0.1, n_pert)
        h_new = project_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5

        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

        T *= cooling

    return best_h, best_c5

def multi_start_continuous(h_init, n, rng, time_limit=200):
    """Try multiple perturbations and optimize each."""
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    start = time.time()

    for attempt in range(20):
        if time.time() - start > time_limit or remaining() < 15:
            break

        if attempt == 0:
            h_try = h_init.copy()
        else:
            h_try = h_init.copy()
            idx = rng.choice(n, rng.randint(5, 20), replace=False)
            h_try[idx] += rng.uniform(-0.15, 0.15, len(idx))
            h_try = project_feasible(h_try, n)

        # Quick active set
        h_try, c5_try = active_set_optimize(h_try, n, time_limit=8)
        # SLSQP polish
        for beta in [2000, 20000]:
            if remaining() < 10:
                break
            h_try = slsqp_minimax(h_try, n, beta, maxiter=150)
        c5_try = compute_c5(h_try)

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary local search at parent resolution (quick)
    n = best_n
    h_bin = (best_h > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        sort_idx = np.argsort(-best_h)
        h_bin = np.zeros(n)
        h_bin[sort_idx[:n_half]] = 1.0

    t1 = min(150, int(remaining() * 0.18))
    if t1 > 20:
        h_b1, c5_b1 = binary_local_search(h_bin, n, time_limit=t1, rng=rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = h_b1.copy()

    # Phase 2: Thue-Morse binary search (different structure)
    if remaining() > 60:
        h_tm = thue_morse(n)
        tm_c5 = compute_c5(h_tm)
        if tm_c5 < best_c5 + 0.01:
            t2 = min(80, int(remaining() * 0.1))
            if t2 > 10:
                h_b2, c5_b2 = binary_local_search(h_tm, n, time_limit=t2, rng=rng)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_b2.copy()

    # Phase 3: Continuous optimization at parent resolution
    h = best_h.copy()
    if remaining() > 80:
        h_eq, c5_eq = active_set_optimize(h, n, time_limit=min(60, int(remaining() * 0.1)))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    if remaining() > 50:
        for beta in [1000, 5000, 20000]:
            if remaining() < 12:
                break
            h_new = slsqp_minimax(h, n, beta, maxiter=250)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 4: Going to higher resolution n=512
    if remaining() > 120:
        n_fine = 512
        h_fine = interp_to(best_h, best_n, n_fine)
        h_fine = project_feasible(h_fine, n_fine)

        # Binary at fine resolution
        h_bin_f = (h_fine > 0.5).astype(float)
        nf_half = n_fine // 2
        nf_ones = int(np.sum(h_bin_f))
        if nf_ones != nf_half:
            if nf_ones > nf_half:
                idx = np.where(h_bin_f == 1)[0]
                h_bin_f[idx[:nf_ones - nf_half]] = 0
            else:
                idx = np.where(h_bin_f == 0)[0]
                h_bin_f[idx[:nf_half - nf_ones]] = 1

        t3 = min(80, int(remaining() * 0.12))
        if t3 > 10:
            h_bf, c5_bf = binary_local_search(h_bin_f, n_fine, time_limit=t3, rng=rng)
            c5_bf_eval = compute_c5(h_bf)

            # Compare: keep best at this resolution
            h_fine_candidate = 0.4 * h_fine + 0.6 * h_bf
            h_fine_candidate = project_feasible(h_fine_candidate, n_fine)
            c5_cand = compute_c5(h_fine_candidate)

            if c5_cand < compute_c5(h_fine):
                h_fine = h_fine_candidate

        # Continuous optimization at fine resolution
        if remaining() > 60:
            h_fine, c5_fine = active_set_optimize(h_fine, n_fine, time_limit=min(40, int(remaining() * 0.12)))
            for beta in [3000, 15000, 60000]:
                if remaining() < 12:
                    break
                h_fine = slsqp_minimax(h_fine, n_fine, beta, maxiter=200)
            c5_fine = compute_c5(h_fine)

            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine.copy()
                best_n = n_fine

    # Phase 5: Try n=384 with more aggressive continuous search
    if remaining() > 100:
        n2 = 384
        h_384 = interp_to(best_h, best_n, n_384) if best_n != 384 else best_h.copy()
        h_384 = project_feasible(h_384, n_384)
        h_384, c5_384 = multi_start_continuous(h_384, n_384, rng, time_limit=min(120, int(remaining() * 0.2)))
        if c5_384 < best_c5:
            best_c5 = c5_384
            best_h = h_384.copy()
            best_n = n_384

    # Phase 6: Try n=768
    if remaining() > 100:
        n_big = 768
        h_big = interp_to(best_h, best_n, n_big)
        h_big = project_feasible(h_big, n_big)

        h_bin_big = (h_big > 0.5).astype(float)
        nb_half = n_big // 2
        nb_ones = int(np.sum(h_bin_big))
        if nb_ones != nb_half:
            if nb_ones > nb_half:
                idx = np.where(h_bin_big == 1)[0]
                h_bin_big[idx[:nb_ones - nb_half]] = 0
            else:
                idx = np.where(h_bin_big == 0)[0]
                h_bin_big[idx[:nb_half - nb_ones]] = 1

        t4 = min(60, int(remaining() * 0.1))
        if t4 > 10:
            h_bb, c5_bb = binary_local_search(h_bin_big, n_big, time_limit=t4, rng=rng)
            h_big_candidate = 0.3 * h_big + 0.7 * h_bb
            h_big_candidate = project_feasible(h_big_candidate, n_big)
            if compute_c5(h_big_candidate) < compute_c5(h_big):
                h_big = h_big_candidate

        if remaining() > 50:
            h_big, c5_big = active_set_optimize(h_big, n_big, time_limit=min(40, int(remaining() * 0.12)))
            for beta in [5000, 30000]:
                if remaining() < 10:
                    break
                h_big = slsqp_minimax(h_big, n_big, beta, maxiter=150)
            c5_big = compute_c5(h_big)

            if c5_big < best_c5:
                best_c5 = c5_big
                best_h = h_big.copy()
                best_n = n_big

    # Phase 7: Final polish at best resolution
    n = best_n
    h = best_h.copy()
    if remaining() > 30:
        h_eq, c5_eq = active_set_optimize(h, n, time_limit=min(40, int(remaining() * 0.25)))
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    for beta in [10000, 50000, 150000]:
        if remaining() < 8:
            break
        h_new = slsqp_minimax(h, n, beta, maxiter=250)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 8: One more round of multi-start
    if remaining() > 40:
        h_ms, c5_ms = multi_start_continuous(h, n, rng, time_limit=min(60, int(remaining() * 0.5)))
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Final: ensure feasibility and evaluate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
