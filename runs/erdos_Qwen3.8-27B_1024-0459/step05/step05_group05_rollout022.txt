def solve_ilp(n, time_limit=120):
    """Solve the ILP for minimum max correlation."""
    dx = 2.0 / n
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # Precompute offsets
    # For d > 0: c[d] = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
    #            c[-d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
    
    # Create z variables for bilinear terms
    # z_pos[d][i] = h[i+d] * (1-h[i]) for d>0, i=0..n-1-d
    # z_neg[d][i] = h[i] * (1-h[i+d]) for d>0, i=0..n-1-d
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2, t >= 0]
    
    # This might be too large... let's see
    # For n=128: sum_{d=1}^{127} (128-d) = 127*128/2 = 8128
    # Total z vars: 2*8128 = 16256
    
    # Actually, let's be smarter. Instead of explicit z variables,
    # for binary h, h[i+d]*(1-h[i]) ≤ h[i+d] and ≤ 1-h[i]
    # And h[i+d]*(1-h[i]) ≥ h[i+d] - h[i]
    # Sum of h[i+d]*(1-h[i]) over i = sum of z_pos[d][i]
    
    # Let's just use the constraints directly:
    # sum_i h[i+d]*(1-h[i]) ≤ t/dx
    # For binary h, h[i+d]*(1-h[i]) ≤ h[i+d] (trivial)
    # and h[i+d]*(1-h[i]) ≤ 1-h[i] (trivial)
    # But we need the SUM, not individual terms.
    
    # Actually, for binary h, h[i+d]*(1-h[i]) is already 0 or 1.
    # The SUM sum_i h[i+d]*(1-h[i]) is just the count of i where h[i+d]=1 and h[i]=0.
    # This IS linear in h! Because for binary h, h[i+d]*(1-h[i]) = h[i+d] - h[i+d]*h[i]
    # But h[i+d]*h[i] is NOT linear...
    
    # Hmm, so it's still bilinear. We need z variables.
    
    # Let me restructure: for each d>0, create z arrays
    # This will be a large model for n=128 (16256 z vars)
    # Let me try it with CBC
    
    try:
        # Build model incrementally
        z_vars = {}  # (type, d, i) -> variable
        z_constraints = []
        
        sum_pos = {}  # d -> sum of z_pos[d][i]
        sum_neg = {}  # d -> sum of z_neg[d][i]
        
        total_z = 0
        for d in range(1, n):
            num = n - d
            if d > 0:
                zp = cp.Variable(num, boolean=True)
                zn = cp.Variable(num, boolean=True)
                z_vars[('pos', d)] = zp
                z_vars[('neg', d)] = zn
                
                # Constraints for z_pos[d][i] = h[i+d] * (1-h[i])
                # z ≤ h[i+d]
                # z ≤ 1-h[i]
                # z ≥ h[i+d] - h[i]
                # z ≥ 0 (implicit for boolean ≥ 0)
                
                z_constraints.append(zp <= h[d:d+num])
                z_constraints.append(zp <= 1 - h[0:num])
                z_constraints.append(zp >= h[d:d+num] - h[0:num])
                
                # Constraints for z_neg[d][i] = h[i] * (1-h[i+d])
                zn_constraints = []
                z_constraints.append(zn <= h[0:num])
                z_constraints.append(zn <= 1 - h[d:d+num])
                z_constraints.append(zn >= h[0:num] - h[d:d+num])
                
                sum_pos[d] = cp.sum(zp)
                sum_neg[d] = cp.sum(zn)
                total_z += 2 * num
        
        # Objective constraints
        t_bound = t / dx
        for d in range(1, n):
            constraints.append(sum_pos[d] <= t_bound)
            constraints.append(sum_neg[d] <= t_bound)
        
        # Note: c[0] = 0 for binary, so no constraint needed
        
        prob = cp.Problem(cp.Minimize(t), constraints + z_constraints)
        
        # Solve
        prob.solve(solver=cp.CBC, time_limit=time_limit)
        
        if h.value is not None and t.value is not None:
            h_sol = np.round(h.value).astype(float)
            h_sol = np.clip(h_sol, 0, 1)
            # Fix sum
            n_ones = int(np.sum(h_sol))
            n_half = n // 2
            if n_ones > n_half:
                ones_idx = np.where(h_sol == 1)[0]
                h_sol[ones_idx[:n_ones-n_half]] = 0
            elif n_ones < n_half:
                zeros_idx = np.where(h_sol == 0)[0]
                h_sol[zeros_idx[:n_half-n_ones]] = 1
            
            c5 = compute_c5(h_sol)
            return h_sol, c5, t.value
    except Exception as e:
        print(f"ILP failed for n={n}: {e}")
    
    return None, None, None

                        j = fi
                    if 0 <= i < n and 0 <= j < n and h[i] != h[j]:
                        h[i], h[j] = h[j], h[i]
                        nc = compute_c5(h)
                        if nc < best_c5 - 1e-14:
                            best_c5 = nc
                            best_h = h.copy()
                            cur_c5 = nc
                            improved = True
                            break
                        else:
                            h[i], h[j] = h[j], h[i]
                        break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5


def simulated_annealing(h_bin, n, rng, time_limit=150):
    """Simulated annealing with diverse block moves."""
    h = ensure_n_half(h_bin, n, rng)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()

    T_init = 0.08
    T_final = 0.0005

    while time.time() - start < time_limit and remaining() > 5:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / time_limit)
        T = T_init * (T_final / T_init) ** progress

        # Choose move
        L = rng.choice([1, 1, 2, 2, 3, 4, 5, 6, 8, 12, 16])
        L = min(L, n // 4)
        max_pos = n - L
        if max_pos <= 0:
            break
        p1 = rng.randint(max_pos + 1)
        p2 = rng.randint(max_pos + 1)
        if abs(p1 - p2) < L:
            continue

        h[p1:p1 + L], h[p2:p2 + L] = h[p2:p2 + L].copy(), h[p1:p1 + L].copy()
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[p1:p1 + L], h[p2:p2 + L] = h[p2:p2 + L].copy(), h[p1:p1 + L].copy()

    return best_h, best_c5


def top_shifts_equalize(h, n, top_k=5, max_iters=300):
    """Equalize top correlations using gradient direction."""
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-9:
            break
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            g += corr_gradient_at(h, off, n)
        g /= top_k
        # Subtract gradient of minimum for balanced direction
        g_min = np.zeros(n)
        bot_idx = np.argsort(corr)[:top_k]
        for k in bot_idx:
            off = k - (n - 1)
            g_min += corr_gradient_at(h, off, n)
        g_min /= top_k
        d = g - g_min
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.5, 0.2]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g


def _sc(x):
    return np.sum(x) - _TGT


def _scj(x):
    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=400):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:
        res = minimize(_sm_obj, h, jac=_sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < _sm_obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def iterted_local_search(h_bin, n, rng, time_limit=120):
    """Iterated local search: local search + perturbation + local search."""
    h = ensure_n_half(h_bin, n, rng)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    n_iters = 0

    while time.time() - start < time_limit and remaining() > 5:
        n_iters += 1
        # Local search phase
        h_ls, c5_ls = block_local_search(h, n, rng, time_limit=min(20, max(5, (time_limit - (time.time() - start)) * 0.6)))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        h = h_ls.copy()

        # Perturbation
        pert_size = rng.randint(3, min(10, n // 10))
        for _ in range(pert_size):
            L = rng.randint(1, min(8, n // 8))
            p1 = rng.randint(n - L + 1)
            p2 = rng.randint(n - L + 1)
            if abs(p1 - p2) >= L:
                h[p1:p1 + L], h[p2:p2 + L] = h[p2:p2 + L].copy(), h[p1:p1 + L].copy()

        if time.time() - start > time_limit * 0.8:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Continuous refinement on parent (SLSQP with progressive beta)
    for beta in [200, 1000, 5000, 20000, 80000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Top-shifts equalization
    if remaining() > 15:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=5, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()
        h_eq2, c5_eq2 = top_shifts_equalize(h, n, top_k=8, max_iters=150)
        if c5_eq2 < best_c5:
            best_c5 = c5_eq2
            best_h = h_eq2.copy()
            h = h_eq2.copy()

    # Phase 3: Intensive binary local search from best continuous
    if remaining() > 40:
        h_bin = binarize(h, n)
        ls_time = min(150, max(20, int(remaining() * 0.2)))
        h_b, c5_b = block_local_search(h_bin, n, rng, time_limit=ls_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            # Continuous refinement from binary
            h_c = slsqp_smooth(h_b, n, 5000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                h = h_c.copy()
            else:
                h = h_b.copy()
        else:
            h = best_h.copy()

    # Phase 4: Simulated annealing
    if remaining() > 60:
        h_bin = binarize(h, n)
        sa_time = min(200, max(30, int(remaining() * 0.25)))
        h_sa, c5_sa = simulated_annealing(h_bin, n, rng, time_limit=sa_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            h_c = slsqp_smooth(h_sa, n, 10000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
                h = h_c.copy()
            else:
                h = h_sa.copy()
        else:
            h = best_h.copy()

    # Phase 5: Iterated local search for deeper exploration
    if remaining() > 60:
        h_bin = binarize(h, n)
        ils_time = min(180, max(20, int(remaining() * 0.2)))
        h_ils, c5_ils = iterted_local_search(h_bin, n, rng, time_limit=ils_time)
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()
            h = h_ils.copy()

    # Phase 6: Multi-start from structured sequences
    if remaining() > 80:
        # Thue-Morse
        h_tm = thue_morse(n, rng)
        h_tm_b, c5_tm = block_local_search(h_tm, n, rng, time_limit=min(60, int(remaining() * 0.2)))
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm_b.copy()
            best_n = n

        # Inverse Thue-Morse
        h_tm2 = 1.0 - h_tm
        h_tm2 = ensure_n_half(h_tm2, n, rng)
        h_tm2_b, c5_tm2 = block_local_search(h_tm2, n, rng, time_limit=min(40, int(remaining() * 0.15)))
        if c5_tm2 < best_c5:
            best_c5 = c5_tm2
            best_h = h_tm2_b.copy()
            best_n = n

        # Random structured start
        for rs in range(3):
            if remaining() < 20:
                break
            h_r = rng.uniform(0, 1, n)
            h_r = project_feasible(h_r, n)
            h_r_bin = binarize(h_r, n)
            h_r_b, c5_r = block_local_search(h_r_bin, n, rng, time_limit=min(40, int(remaining() * 0.15)))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r_b.copy()
                best_n = n

        # Perturbed parent
        h_p = best_h + rng.randn(n) * 0.08
        h_p = project_feasible(h_p, n)
        h_p_bin = binarize(h_p, n)
        h_p_b, c5_p = block_local_search(h_p_bin, n, rng, time_limit=min(50, int(remaining() * 0.2)))
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p_b.copy()
            best_n = n

    # Phase 7: Resolution sweep
    if remaining() > 80:
        n_candidates = []
        for delta in [-16, -12, -8, -6, -4, -2, 2, 4, 6, 8, 12, 16, 24, 32, 48, 64]:
            nt = best_n + delta
            if 128 <= nt <= 800 and nt % 2 == 0:
                n_candidates.append(nt)
        for nt in [128, 192, 256, 320, 384, 448, 512, 576, 640]:
            if nt % 2 == 0 and nt not in n_candidates:
                n_candidates.append(nt)
        n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - best_n))

        for n_t in n_candidates[:10]:
            if remaining() < 25:
                break
            if n_t <= 0 or n_t % 2 != 0:
                continue
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

            # SLSQP on this resolution
            for beta in [2000, 10000]:
                if remaining() < 15:
                    break
                h_t = slsqp_smooth(h_t, n_t, beta, 250)
            c5_t = compute_c5(h_t)

            # Binary local search
            if remaining() > 20:
                h_t_bin = binarize(h_t, n_t)
                h_t_b, c5_t_b = block_local_search(h_t_bin, n_t, rng, time_limit=min(30, int(remaining() * 0.3)))
                if c5_t_b < c5_t:
                    c5_t = c5_t_b
                    h_t = h_t_b.copy()

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n_t

    # Phase 8: Final intensive refinement
    n = best_n
    h = best_h.copy()

    # SLSQP polish
    for beta in [5000, 20000, 60000, 150000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Top-shifts with various k
    for tk in [3, 5, 8, 12]:
        if remaining() < 8:
            break
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=tk, max_iters=200)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Final binary local search (thorough)
    if remaining() > 30:
        h_bin = binarize(h, n)
        h_b, c5_b = block_local_search(h_bin, n, rng, time_limit=min(60, int(remaining() * 0.4)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            h_c = slsqp_smooth(h_b, n, 20000, 300)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Final SA polish
    if remaining() > 20:
        h_bin = binarize(best_h, best_n)
        h_sa, c5_sa = simulated_annealing(h_bin, best_n, rng, time_limit=min(30, int(remaining() * 0.5)))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, best_n)
