            # Term 2: h[j] * (1-h[j-k+offset]) changes when j-k+offset = i, i.e., k = j-i+offset
            #   For fixed i, k = j-i+offset where j is any index
            #   But we're changing h[i], so the (1-h) term changes when j-k+offset = i
            #   The change is: h[j] * (-dt) where the condition is j-k+offset = i
            #   For corr[k], this contributes -dt * h[j] where j = i+k-offset = i+k-n+1
            #   Valid when 0 ≤ i+k-n+1 < n, i.e., k ∈ [n-1-i, n-1-i+n-1] = [n-1-i, 2n-2-i]
            #   And k ∈ [0, 2n-2]
            #   So k ∈ [max(0, n-1-i), min(2n-2, 2n-2-i)] = [n-1-i, 2n-2-i] (since i < n-1)
            
            # Combined: dcorr[k]/dh[i] for k in [0, 2n-2]:
            #   If k >= i and k <= 2n-2: (1 - h[i - k + n - 1])  [let idx1 = i-k+n-1, valid when 0<=idx1<n]
            #   If k >= n-1-i and k <= 2n-2-i: -h[i+k-n+1]  [let idx2 = i+k-n+1, valid when 0<=idx2<n]
            
            # Let me just compute this numerically for correctness:
            dc = np.zeros(ncorr)
            # Term 1: h[i] * (1 - h[i-k+n-1])
            k_min1 = max(0, i)  # i-k+n-1 >= 0 => k <= i+n-1, and k>=0
            k_max1 = min(2*n-2, i+n-1)  # i-k+n-1 <= n-1 => k >= i
            # Wait: i-k+n-1 >= 0 => k <= i+n-1
            # i-k+n-1 <= n-1 => k >= i
            # So k in [i, min(2n-2, i+n-1)]
            for k in range(max(0, i), min(2*n-1, i+n)):
                idx = i - k + n - 1
                if 0 <= idx < n:
                    dc[k] += (1.0 - h[idx])
            
            # Term 2: -h[i+k-n+1] when j=i+k-n+1 and we're in the (1-h[j-k+offset]) slot
            # Actually: corr[k] = sum_j h[j]*(1-h[j-k+n-1])
            # If j-k+n-1 = i, then j = i+k-n+1
            # Contribution: h[i+k-n+1] * (-1) to the (1-h) part... 
            # Wait: the term is h[j]*(1-h[j-k+n-1]). If j-k+n-1 = i, then:
            # h[i+k-n+1]*(1-h[i]) - but we're changing h[i], so:
            # d/dh[i] [h[j]*(1-h[j-k+n-1])] = h[j]*(-1) when j-k+n-1=i
            # = -h[i+k-n+1]
            # Valid when 0 <= i+k-n+1 < n, i.e., n-1-i <= k <= 2n-2-i
            for k in range(max(0, n-1-i), min(2*n-1, 2*n-1-i)):
                idx = i + k - n + 1
                if 0 <= idx < n:
                    dc[k] -= h[idx]
            
            # Now corr_new[k] = corr[k] + dc[k] * dt
            # We want to minimize max(corr[k] + dc[k]*dt) over dt in [max(0, h[i]-t0), min(1, h[i]+...)]
            # Actually h_new[i] = h[i] + dt, and we need h_new[i] in [0,1]
            # So dt in [-h[i], 1-h[i]]
            
            # The function f(dt) = max_k (corr[k] + dc[k]*dt) is convex in dt
            # Find its minimum in [-h[i], 1-h[i]]
            
            # Use golden section or ternary search
            lo = -h[i]
            hi = 1.0 - h[i]
            
            # Evaluate at a few points to get a rough minimum
            n_eval = 20
            candidates = np.linspace(lo, hi, n_eval)
            f_vals = np.array([np.max(corr + dc * dt) for dt in candidates])
            best_idx = np.argmin(f_vals)
            
            # Refine around best candidate
            if best_idx > 0 and best_idx < n_eval - 1:
                rlo = candidates[max(0, best_idx-1)]
                rhi = candidates[min(n_eval-1, best_idx+1)]
            else:
                rlo, rhi = lo, hi
            
            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c1 = rhi - gr * (rhi - rlo)
            c2 = rlo + gr * (rhi - rlo)
            f1 = np.max(corr + dc * c1)
            f2 = np.max(corr + dc * c2)
            for _ in range(40):
                if rhi - rlo < 1e-14:
                    break
                if f1 < f2:
                    rhi = c2
                    c2 = c1
                    f2 = f1
                    c1 = rhi - gr * (rhi - rlo)
                    f1 = np.max(corr + dc * c1)
                else:
                    rlo = c1
                    c1 = c2
                    f1 = f2
                    c2 = rlo + gr * (rhi - rlo)
                    f2 = np.max(corr + dc * c2)
            
            dt_opt = (rlo + rhi) / 2.0
            f_opt = np.max(corr + dc * dt_opt)
            
            # Check if this improves over current max
            curr_max = np.max(corr)
            if f_opt < curr_max - 1e-12:
                h[i] += dt_opt
                h[i] = np.clip(h[i], 0, 1)
                corr = compute_corr(h)  # Recompute for accuracy
                improved = True
        
        if not improved and sweep > 0:
            break
    
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0
_N_OPT = 0
_BETA_OPT = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        elif diff > 0:
            idx = np.argmin(h)
            h[idx] = np.min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = np.max(0.0, h[idx] + diff)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13 and n > 0:
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)

def smooth_val_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad

def _obj_fn(x):
    return smooth_val_grad(x, _N_OPT, _BETA_OPT)[0]

def _grad_fn(x):
    return smooth_val_grad(x, _N_OPT, _BETA_OPT)[1]

def _con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _con_jac(x):
    return np.ones(_N_OPT)

def slsqp_opt(h, n, beta, maxiter=400):
    global _N_OPT, _BETA_OPT
    _N_OPT = n
    _BETA_OPT = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    c5_before = compute_c5(h)
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            if c5_trial <= c5_before + 1e-13:
                return trial
    except Exception:
        pass
    return h.copy()

def fast_coord_descent(h, n, n_sweeps=100):
    h = h.copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = compute_c5(h)
    k_range = np.arange(ncorr)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if remaining_time() < 5:

        corr = compute_corr(h)
        curr_max = np.max(corr)
        improved = False
        n_interior = 0

        for i in range(n):
            if h[i] <= 1e-12 or h[i] >= 1.0 - 1e-12:
                continue
            n_interior += 1

            # Compute dc: derivative of corr w.r.t. h[i]
            dc = np.zeros(ncorr)
            # Term 1: h[i]*(1-h[i-k+n-1]) -> derivative w.r.t. h[i] is (1-h[i-k+n-1])
            mask1 = (k_range >= i) & (k_range <= i + n - 1)
            if np.any(mask1):
                idx1 = i - k_range[mask1] + n - 1
                dc[mask1] += (1.0 - h[idx1])
            # Term 2: h[j]*(1-h[j-k+n-1]) where j-k+n-1=i -> j=i+k-n+1
            # derivative w.r.t. h[i] of (1-h[i]) term: -h[j]
            mask2 = (k_range >= n - 1 - i) & (k_range <= 2 * n - 2 - i)
            if np.any(mask2):
                idx2 = i + k_range[mask2] - n + 1
                dc[mask2] -= h[idx2]

            # Minimize f(dt) = max(corr + dc*dt) for dt in [-h[i], 1-h[i]]
            lo_dt = -h[i]
            hi_dt = 1.0 - h[i]

            # Coarse grid to find approximate minimum
            n_coarse = 16
            cands = np.linspace(lo_dt, hi_dt, n_coarse)
            f_coarse = np.array([np.max(corr + dc * dt) for dt in cands])
            bi = np.argmin(f_coarse)

            if bi > 0 and bi < n_coarse - 1:
                rlo = cands[bi - 1]
                rhi = cands[bi + 1]
            else:
                rlo = lo_dt
                rhi = hi_dt

            # Golden section refinement
            c1 = rhi - gr * (rhi - rlo)
            c2 = rlo + gr * (rhi - rlo)
            f1 = np.max(corr + dc * c1)
            f2 = np.max(corr + dc * c2)

                    break
                if f1 < f2:
                    rhi = c2
                    c2 = c1
                    f2 = f1
                    c1 = rhi - gr * (rhi - rlo)

                else:
                    rlo = c1
                    c1 = c2
                    f1 = f2
                    c2 = rlo + gr * (rhi - rlo)
                    f2 = np.max(corr + dc * c2)

            dt_opt = (rlo + rhi) / 2.0
            # Also check endpoints

            f_hi = np.max(corr + dc * hi_dt)
            f_opt = np.max(corr + dc * dt_opt)

                f_opt = f_lo
            if f_hi < f_opt:

                f_opt = f_hi

            if f_opt < curr_max - 1e-13:
                h[i] = h[i] + dt_opt
                h[i] = np.clip(h[i], 0.0, 1.0)
                corr = compute_corr(h)
                curr_max = np.max(corr)
                improved = True

        if not improved:
            break

    best_c5 = compute_c5(h)
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def block_sweep(h, n, block_size=2, n_sweeps=30):
    h = h.copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    k_range = np.arange(ncorr)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if remaining_time() < 5:
            break
        corr = compute_corr(h)
        curr_max = np.max(corr)
        improved = False

        for start in range(0, n - block_size + 1, block_size):
            idxs = np.arange(start, start + block_size)
            vals = h[idxs].copy()
            if np.all(vals <= 1e-12) or np.all(vals >= 1.0 - 1e-12):
                continue

            # Compute Jacobian: dcorr/d(h[idxs])
            J = np.zeros((ncorr, block_size))
            for di, i in enumerate(idxs):
                mask1 = (k_range >= i) & (k_range <= i + n - 1)
                if np.any(mask1):
                    idx1 = i - k_range[mask1] + n - 1
                    J[mask1, di] += (1.0 - h[idx1])
                mask2 = (k_range >= n - 1 - i) & (k_range <= 2 * n - 2 - i)
                if np.any(mask2):
                    idx2 = i + k_range[mask2] - n + 1
                    J[mask2, di] -= h[idx2]

            # For block_size=2: minimize max(corr + J1*t1 + J2*t2)
            # Use grid search over a small region
            if block_size == 1:
                continue

            # Simple approach: try a small grid of perturbations
            search_range = 0.3
            n_grid = 8
            best_f = curr_max
            best_delta = np.zeros(block_size)
            for d1_offset in np.linspace(-search_range, search_range, n_grid):
                if block_size == 2:
                    for d2_offset in np.linspace(-search_range, search_range, n_grid):
                        delta = np.array([d1_offset, d2_offset])
                        new_vals = vals + delta
                        if np.any(new_vals < -0.01) or np.any(new_vals > 1.01):
                            continue
                        new_corr = corr + J @ delta
                        f = np.max(new_corr)
                        if f < best_f - 1e-12:
                            best_f = f
                            best_delta = delta.copy()
                elif block_size == 3:
                    for d2_offset in np.linspace(-search_range, search_range, 5):
                        for d3_offset in np.linspace(-search_range, search_range, 5):
                            delta = np.array([d1_offset, d2_offset, d3_offset])
                            new_vals = vals + delta
                            if np.any(new_vals < -0.01) or np.any(new_vals > 1.01):
                                continue
                            new_corr = corr + J @ delta
                            f = np.max(new_corr)
                            if f < best_f - 1e-12:
                                best_f = f
                                best_delta = delta.copy()
            if np.any(best_delta != 0):
                h[idxs] = np.clip(h[idxs] + best_delta, 0, 1)
                corr = compute_corr(h)
                curr_max = np.max(corr)
                improved = True

        if not improved:
            break

    return h, compute_c5(h)

                                                                int(np.sum(nudge_mask)))
            h_pert = project_feasible(h_pert, n)
            h_pert, c5_pert = fast_coord_descent(h_pert, n, n_sweeps=30)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    # Phase 8: Binarization + re-optimization
    if remaining_time() > 25:
        h_bin = best_h.copy()
        h_bin[h_bin < 0.15] = 0.0
        h_bin[h_bin > 0.85] = 1.0
        h_bin = project_feasible(h_bin, best_n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5 + 0.01:
            h_bin, c5_bin2 = fast_coord_descent(h_bin, best_n, n_sweeps=50)
            c5_bin2 = compute_c5(h_bin)
            if c5_bin2 < best_c5:
                best_c5 = c5_bin2
                best_h = h_bin.copy()

    # Phase 9: One more SLSQP at very high beta
    if remaining_time() > 12:
        h_last = slsqp_opt(best_h, best_n, 1000000, 200)
        c5_last = compute_c5(h_last)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    # Phase 10: Final coordinate descent
    if remaining_time() > 8:
        h_last, c5_last = fast_coord_descent(best_h, best_n, n_sweeps=30)
        c5_last = compute_c5(h_last)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    # Final validation and projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    # Ensure exact sum constraint
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        free = (best_h > 1e-14) & (best_h < 1.0 - 1e-14)
        if np.sum(free) > 0:
            best_h[free] += diff / np.sum(free)
            best_h = np.clip(best_h, 0.0, 1.0)
            final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
