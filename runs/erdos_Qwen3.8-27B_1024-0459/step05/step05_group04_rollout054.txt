def make_swap_delta(h, i, j, n):
    """Compute correlation change for swap h[i]:1->0, h[j]:0->1."""
    r = 2 * n - 1
    dr = np.zeros(r)
    # Effect of h[i] changing by -1
    # corr[k] = sum_i h[i] * (1 - h[i + k - (n-1)]) (with appropriate bounds)
    # Actually, np.correlate(h, 1-h, 'full') computes:
    # corr[k] = sum_j h[j] * (1-h)[j + k - (n-1)] for valid j
    # Let's just compute the effect directly.
    
    # When h[i] changes by delta, corr[k] changes by:
    # delta * (1-h)[i + k - (n-1)] for the first factor being h[i]
    # - delta * h[i + k - (n-1)] for the second factor... wait.

    # j + k - (n-1) >= 0 => j >= (n-1) - k
    # j + k - (n-1) < n => j < n - k + (n-1) = 2n - 1 - k
    
    # So corr[k] = sum_j h[j] * (1-h[j+k-(n-1)]) over j in [max(0, n-1-k), min(n-1, 2n-2-k)]
    
    # Effect of h[i] -> h[i] - 1 (delta = -1):
    # First, h[i] appears as the first factor: contribution to corr[k] is -(1-h[i+k-(n-1)])
    #   valid when 0 <= i+k-(n-1) < n, i.e., (n-1)-i <= k <= (n-1)+i
    # Second, h[i] appears as the second factor (in 1-h): contribution is -1*( change in (1-h) )
    #   (1-h[i]) -> (1-h[i]+1) = 2-h[i], so delta in (1-h) = +1
    #   Wait, h[i] -> h[i]-1, so (1-h[i]) -> 1-(h[i]-1) = 2-h[i], delta = +1
    #   This affects corr[k] where j + k - (n-1) = i, i.e., k = i - j + (n-1)

    #   Actually corr[k] = sum_j h[j]*(1-h[j+k-(n-1)])
    #   If h[i] changes by -1, then (1-h[i]) changes by +1
    #   This affects corr[k] for all j where j+k-(n-1) = i, i.e., k = i-(n-1)+j
    #   For such k, corr[k] increases by h[j] for each valid j
    #   j ranges: 0 <= j < n and 0 <= k < 2n-1
    #   k = i - (n-1) + j, so j = k - i + (n-1)
    #   Valid: 0 <= j < n => i-(n-1) <= k <= i-(n-1)+n-1 = i
    #   Also need 0 <= k < 2n-1
    #   So for k in [max(0, i-n+1), min(i, 2n-2)]: corr[k] += h[k-i+(n-1)]
    
    # Similarly for h[j] -> h[j] + 1 (delta = +1):
    # First factor: h[j] increases by 1, so corr[k] += (1-h[j+k-(n-1)])
    #   valid when 0 <= j+k-(n-1) < n, i.e., (n-1)-j <= k <= (n-1)+j
    # Second factor: (1-h[j]) decreases by 1, so corr[k] -= h[l] where l+k-(n-1)=j
    #   k = j-(n-1)+l, valid l: 0<=l<n
    #   k in [max(0, j-n+1), min(j, 2n-2)]: corr[k] -= h[k-j+(n-1)]

    # First: for k where (n-1)-i <= k <= (n-1)+i: corr[k] -= (1 - h[k - (n-1) + i])
    #   but we need 0 <= k-(n-1)+i < n, which is equivalent to the k range above

    k_lo2 = max(0, i - n + 1)

    k_lo3 = max(0, n - 1 - j)
    k_hi3 = min(2 * n - 2, n - 1 + j)
    if k_lo3 <= k_hi3:

def subgradient_equalize(h, n, max_iters=500, top_k=8):
    """Subgradient method to equalize top correlations."""
    h = np.array(h, dtype=float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        
        if c_max - c_min_top < 1e-9:
            break
        
        # Compute subgradient: average gradient of top-k correlations
        g = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        
        g /= top_k  # average
        
        # Line search
        improved = False
        for step in [0.005, 0.01, 0.02, 0.03, 0.05, 0.002, 0.008, 0.04, 0.06, 0.015]:
            ht = project_feasible(h - step * g, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-12:
                h = ht
                best_c5 = cn
                best_h = h.copy()
                improved = True
                break
        if not improved:
            # Try smaller steps
            for step in [0.001, 0.0005, 0.002]:
                ht = project_feasible(h - step * g, n)
                cn = compute_c5(ht)
                if cn < best_c5 - 1e-12:
                    h = ht
                    best_c5 = cn
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)
