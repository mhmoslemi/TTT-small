def _smooth_obj(x):

def subgradient_step(h, n, step_size, n_active=10):

    max_corr = np.max(corr)
    threshold = max_corr - 0.3
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        idx = active[np.argsort(corr[active])[::-1][:n_active]]
        active = idx
    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h
    grad /= gnorm
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

                if b2 < b1:
                    b1, b2 = b2, b1
                h_pert[b1:b1+blen], h_pert[b2:b2+blen] = h_pert[b2:b2+blen].copy(), h_pert[b1:b1+blen].copy()

            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=120)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                # Polish
                if remaining_time() > 25:
                    h_slq = slsqp_refine(h_new, n, 5000, 200)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                    h2, c5_2 = fast_coord_descent(h_new, n, max_sweeps=60)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()

    # Phase 5: Structured initializations
    if remaining_time() > 60:
        structured_inits = [
            thue_morse_init(n),
            random_binary_init(n, rng),
            random_binary_init(n, np.random.RandomState(seed + 1)),
        ]
        for SI in structured_inits:
            if remaining_time() < 30:
                break
            c5_si = compute_c5(SI)
            if c5_si < best_c5 + 0.02:
                h_new, c5_new = fast_coord_descent(SI, n, max_sweeps=100)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    if remaining_time() > 20:
                        h_slq = slsqp_refine(h_new, n, 10000, 300)
                        c5_slq = compute_c5(h_slq)
                        if c5_slq < best_c5:
                            best_c5 = c5_slq
                            best_h = h_slq.copy()
                            h2, c5_2 = fast_coord_descent(h_new, n, max_sweeps=60)
                            if c5_2 < best_c5:
                                best_c5 = c5_2
                                best_h = h2.copy()

    # Phase 6: Try different resolutions
    if remaining_time() > 100:
        for try_n in [512, 768]:
            if remaining_time() < 50:
                break
            h_try = interp_to(best_h, best_n, try_n)
            h_try = project_feasible(h_try, try_n)
            h_new, c5_new = fast_coord_descent(h_try, try_n, max_sweeps=80)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = try_n
                if remaining_time() > 30:
                    h_slq = slsqp_refine(h_new, try_n, 3000, 200)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()

    # Phase 7: Subgradient + coordinate descent
    if remaining_time() > 50:
        h = best_h.copy()
        for it in range(300):
            if remaining_time() < 8:
                break
            step = 0.015 / (1.0 + it * 0.008)
            h_sg = subgradient_step(h, best_n, step, n_active=8)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5 - 1e-15:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()
            elif it % 80 == 40:
                h_new, c5_new = fast_coord_descent(h, best_n, max_sweeps=30)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                else:
                    break

    # Phase 8: Final thorough coordinate descent
    if remaining_time() > 15:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, best_n, max_sweeps=200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 9: Final SLSQP polish
    if remaining_time() > 12:
        h_new = slsqp_refine(best_h, best_n, 80000, 300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h2, c5_2 = fast_coord_descent(h_new, best_n, max_sweeps=80)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 10: One last coordinate descent
    if remaining_time() > 8:
        h2, c5_2 = fast_coord_descent(best_h, best_n, max_sweeps=150)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
