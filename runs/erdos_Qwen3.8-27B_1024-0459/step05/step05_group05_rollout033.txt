For a swap (i: 1→0, j: 0→1), the affected offsets are those d where i+d is valid OR j+d is valid OR i = j+d (i.e., d = i-j) OR j = i+d (d = j-i).

For offset d:
- corr[d] = sum_{i: 0≤i<n, 0≤i+d<n} h[i]*(1-h[i+d])

After swap:
- h_new[i] = 0 (was 1)
- h_new[j] = 1 (was 0)
- Everything else unchanged.

Change in corr[d]:
- Terms involving h[i] as the "first" factor: h[i]*(1-h[i+d]) was 1*(1-h[i+d]) = 1-h[i+d], now 0. Change: -(1-h[i+d]) = h[i+d]-1.
  Valid when 0 ≤ i < n and 0 ≤ i+d < n.
  
- Terms involving h[i] as the "second" factor: h[i-d]*(1-h[i]) was h[i-d]*(1-1) = 0, now h[i-d]*(1-0) = h[i-d]. Change: +h[i-d].
  Valid when 0 ≤ i-d < n and 0 ≤ i < n, i.e., 0 ≤ i-d < n.
  
- Terms involving h[j] as the "first" factor: h[j]*(1-h[j+d]) was 0*(...) = 0, now 1*(1-h[j+d]) = 1-h[j+d]. Change: +(1-h[j+d]).
  Valid when 0 ≤ j < n and 0 ≤ j+d < n.
  
- Terms involving h[j] as the "second" factor: h[j-d]*(1-h[j]) was h[j-d]*(1-0) = h[j-d], now h[j-d]*(1-1) = 0. Change: -h[j-d].
  Valid when 0 ≤ j-d < n and 0 ≤ j < n.

So for each offset d, the total change is at most 4 terms (some may not be valid).

Now, the key insight: for a given swap (i,j), which offsets d are affected?
- d such that 0 ≤ i+d < n: d in [-i, n-1-i]
- d such that 0 ≤ i-d < n: d in [i-(n-1), i]
- d such that 0 ≤ j+d < n: d in [-j, n-1-j]
- d such that 0 ≤ j-d < n: d in [j-(n-1), j]

The union of these ranges is [-max(i,j), n-1-min(i,j)] approximately. So it's O(n) offsets affected.

But I can compute the delta vector in O(n) time:

```python
def compute_swap_delta(h, i, j, n):
    """Returns delta[d + n - 1] = change in correlation at offset d."""
    delta = np.zeros(2*n-1)
    # Offset d corresponds to index d + (n-1) in the correlation array
    
    # h[i]: 1→0
    # As first factor: affects offsets d where 0 ≤ i+d < n, i.e., d in [-i, n-1-i]
    for d in range(-i, n-i):
        if 0 <= i+d < n:
            delta[d + n - 1] += (h[i+d] - 1)  # change is -(1-h[i+d]) = h[i+d]-1
    
    # As second factor: affects offsets d where 0 ≤ i-d < n, i.e., d in [i-(n-1), i]
    for d in range(i-(n-1), i+1):
        if 0 <= i-d < n:
            delta[d + n - 1] += h[i-d]  # change is +h[i-d]
    
    # h[j]: 0→1
    # As first factor: affects offsets d where 0 ≤ j+d < n
    for d in range(-j, n-j):
        if 0 <= j+d < n:
            delta[d + n - 1] += (1 - h[j+d])
    
    # As second factor: affects offsets d where 0 ≤ j-d < n
    for d in range(j-(n-1), j+1):
        if 0 <= j-d < n:
            delta[d + n - 1] += -h[j-d]
    
    return delta
```

import numpy as np
from scipy.optimize import minimize
import time

_T0 = 0.0
_BUDGET = 1300.0
_N = 0
_BETA = 1.0
_TGT = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining():
    return max(0.0, _BUDGET - (time.time() - _T0) - 2.0)


def _sm_obj(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    return (m + np.log(np.sum(ex)) / _BETA) * dx_v


def _sm_grad(x):
    n = _N
    dx_v = 2.0 / n
    corr = compute_corr(x)
    m = np.max(corr)
    sh = corr - m
    cl = np.clip(_BETA * sh, -50.0, 50.0)
    ex = np.exp(cl)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    wc = w * dx_v
    for k in range(2 * n - 1):
        wk = wc[k]
        if wk < 1e-15:
            continue
        off = k - (n - 1)
        if off >= 0:
            np_ = n - off
            g[off:off + np_] += wk * (1.0 - x[:np_])
            g[:np_] -= wk * x[off:off + np_]
        else:
            ao = -off
            np_ = n - ao
            g[:np_] += wk * (1.0 - x[ao:ao + np_])
            g[ao:ao + np_] -= wk * x[:np_]
    return g


def _sc(x):
    return np.sum(x) - _TGT

    return np.ones(_N)


def slsqp_smooth(h, n, beta, maxiter=500):
    global _N, _BETA, _TGT
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj}]
    try:

                       bounds=bounds, constraints=cons,

        if res.success or res.fun < _sm_obj(h):

def binarize(h, n):

    n_half = n // 2
    sort_idx = np.argsort(-h)
    h_bin = np.zeros(n)

def thue_morse(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            if rng is not None:
                rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            else:
                rem = ones_idx[:n_ones - n_half]
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            if rng is not None:
                add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            else:
                add = zeros_idx[:n_half - n_ones]
            h[add] = 1
    return h


def compute_swap_delta(h, i, j, n):
    """Compute change in correlation vector when h[i]:1->0, h[j]:0->1. O(n) vectorized."""
    delta = np.zeros(2 * n - 1)
    off = n - 1
    h_rev = h[::-1]
    
    # h[i]: 1->0, as first factor in h[i+d]*(1-h[i]): delta[d+off] += h[i+d] - 1
    # d in [-i, n-1-i], h[i+d] for these d is h[0:n-i]
    a = off - i
    b = off + n - i
    delta[a:b] += h[:n - i] - 1.0
    
    # h[i]: 1->0, as second factor: h[i-d]*(1-h[i]) changes from 0 to h[i-d]
    # d in [i-n+1, i], h[i-d] is h_rev
    a2 = i
    b2 = i + n
    delta[a2:b2] += h_rev
    
    # h[j]: 0->1, as first factor: h[j]*(1-h[j+d]) changes from 0 to 1-h[j+d]
    # d in [-j, n-1-j], h[j+d] for these d is h[0:n-j]
    a3 = off - j
    b3 = off + n - j
    delta[a3:b3] += 1.0 - h[:n - j]
    
    # h[j]: 0->1, as second factor: h[j-d]*(1-h[j]) changes from h[j-d] to 0
    # d in [j-n+1, j], h[j-d] is h_rev
    a4 = j
    b4 = j + n
    delta[a4:b4] += -h_rev
    
    return delta


def efficient_binary_search(h_init, n, rng, time_limit):
    """Efficient binary local search with O(n) swap evaluation and simulated annealing."""
    h = h_init.copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rem = rng.choice(ones_idx, n_ones - n_half, replace=False)
            h[rem] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            add = rng.choice(zeros_idx, n_half - n_ones, replace=False)
            h[add] = 1
    
    dx = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    
    start = time.time()
    temp = 0.5
    no_improve = 0
    max_no_improve = 50000
    iteration = 0
    
    while time.time() - start < time_limit and remaining() > 5:
        iteration += 1
        no_improve += 1
        if no_improve > max_no_improve:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Generate candidates
        candidates = []
        
        # Random swaps
        n_rand = 30
        for _ in range(n_rand):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i != j:
                candidates.append((i, j))
        
        # Targeted swaps: focus on worst lags
        top_k = min(8, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_k]
        n_target = 30
        for _ in range(n_target):
            lag_idx = top_idx[rng.randint(len(top_idx))]
            d = lag_idx - (n - 1)
            if d == 0:
                continue
            if d > 0:
                # corr at offset d: sum_i h[i+d]*(1-h[i]) for i in [0, n-1-d]
                # "Fault": h[i+d]=1, h[i]=0 → swapping i (0->1) helps
                overlap = n - d
                faults = np.where((h[d:] == 1) & (h[:overlap] == 0))[0]
                if len(faults) > 0:
                    fi = faults[rng.randint(len(faults))]
                    # i = fi (the 0 position), j = a 1 position to swap with
                    j = ones[rng.randint(len(ones))]
                    if j != fi:
                        candidates.append((j, fi))
                # Also: positions where h[i+d]=0, h[i]=1 → "anti-faults"
                # Swapping i+ d (0->1) with some 1 reduces this
                antifaults = np.where((h[d:] == 0) & (h[:overlap] == 1))[0]
                if len(antifaults) > 0:
                    fi = antifaults[rng.randint(len(antifaults))]
                    # h[fi+d] = 0, h[fi] = 1
                    # If we set h[fi+d]=1 and h[fi]=0, we reduce corr[d] by 1
                    i_swap = fi + d
                    j_swap = fi
                    if 0 <= i_swap < n and h[i_swap] == 0 and h[j_swap] == 1 and i_swap != j_swap:
                        candidates.append((j_swap, i_swap))
            else:
                # d < 0: offset is -|d|, corr = sum_i h[i+d]*(1-h[i]) = sum_i h[i-|d|]*(1-h[i])
                ao = -d
                overlap = n - ao
                # h[i-ao]*(1-h[i]), i in [ao, n-1]
                # Fault: h[i-ao]=1, h[i]=0
                faults = np.where((h[:overlap] == 1) & (h[ao:] == 0))[0]
                if len(faults) > 0:
                    fi = faults[rng.randint(len(faults))]
                    j = ones[rng.randint(len(ones))]
                    if j != fi + ao:
                        pass  # This is more complex, skip for now
                    i_swap = fi
                    # h[fi]=1, h[fi+ao]=0 → swap: set h[fi]→0, h[fi+ao]→1
                    if 0 <= fi < n and 0 <= fi + ao < n and h[fi] == 1 and h[fi + ao] == 0:
                        candidates.append((fi, fi + ao))
        
        # Evaluate candidates with early termination
        best_candidate_max = cur_max
        best_candidate = None
        best_delta = None
        
        for (i, j) in candidates:
            if i < 0 or i >= n or j < 0 or j >= n:
                continue
            if i == j:
                continue
            if h[i] != 1 or h[j] != 0:
                continue
            
            # Quick filter: check if this swap reduces the current max
            # The current max is at position argmax(corr)
            # Check if delta at that position is negative
            delta = compute_swap_delta(h, i, j, n)
            
            # Quick check: max position
            max_pos = np.argmax(corr)
            if corr[max_pos] + delta[max_pos] >= best_candidate_max - 1e-15:
                # This swap won't beat the current best at the max position
                # Still might be better if max moves... but likely not
                # Do full check only if it looks promising
                if delta[max_pos] > -0.5:
                    continue
            
            new_max = float(np.max(corr + delta))
            if new_max < best_candidate_max - 1e-15:
                best_candidate_max = new_max
                best_candidate = (i, j)
                best_delta = delta.copy()
        
        # Accept with simulated annealing
        if best_candidate is not None:
            i, j = best_candidate
            h[i] = 0
            h[j] = 1
            corr = corr + best_delta
            cur_max = best_candidate_max
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            # No improving swap found
            # Occasionally accept a random swap (annealing)
            if temp > 0.01 and rng.random() < 0.1:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                if i != j:
                    delta = compute_swap_delta(h, i, j, n)
                    new_max = float(np.max(corr + delta))
                    delta_e = (new_max - cur_max) * n / 8.0  # normalize
                    if rng.random() < np.exp(-delta_e / max(temp, 1e-6)):
                        h[i] = 0
                        h[j] = 1
                        corr = corr + delta
                        cur_max = new_max
        
        # Cool down
        if no_improve % 500 == 0:
            temp *= 0.95
        if temp < 0.001:
            temp = 0.001
            
        # Periodic restart from best
        if no_improve % 10000 == 0:
            h = best_h.copy()
            corr = compute_corr(h)
            cur_max = float(np.max(corr))
    
    return best_h, best_max * dx


def top_shifts_focus(h, n, top_k=5, max_iters=300):
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-9:
            break
        k_top = idx[0]
        k_bot = idx[-1]
        off_t = k_top - (n - 1)
        off_b = k_bot - (n - 1)
        
        g_t = np.zeros(n)
        if off_t >= 0:
            np_ = n - off_t
            g_t[off_t:off_t + np_] += (1.0 - h[:np_])
            g_t[:np_] -= h[off_t:off_t + np_]
        else:
            ao = -off_t
            np_ = n - ao
            g_t[:np_] += (1.0 - h[ao:ao + np_])
            g_t[ao:ao + np_] -= h[:np_]
        
        g_b = np.zeros(n)
        if off_b >= 0:
            np_ = n - off_b
            g_b[off_b:off_b + np_] += (1.0 - h[:np_])
            g_b[:np_] -= h[off_b:off_b + np_]
        else:
            ao = -off_b
            np_ = n - ao
            g_b[:np_] += (1.0 - h[ao:ao + np_])
            g_b[ao:ao + np_] -= h[:np_]
        
        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn
        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015, 0.5, 0.2]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def subgradient_max(h, n, n_iter=200):
    h = project_feasible(h, n).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.05
    for it in range(n_iter):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        threshold = m - 0.5
        top_mask = corr >= threshold
        top_idx = np.where(top_mask)[0]
        if len(top_idx) == 0:
            top_idx = [np.argmax(corr)]
        
        g = np.zeros(n)
        for k in top_idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]
        g /= len(top_idx)
        
        g -= np.mean(g)
        g = np.where(h > 0.99, -np.abs(g), g)
        g = np.where(h < 0.01, np.abs(g), g)
        
        h_new = project_feasible(h - step * g, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            step *= 0.9
            if step < 0.001:
                break
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    
    n = best_n
    
    # Phase 1: Quick SLSQP to get a good continuous starting point
    if remaining() > 100:
        for beta in [200, 1000, 5000]:
            if remaining() < 10:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 2: Main binary search (most of the time)
    if remaining() > 60:
        h_bin = binarize(best_h, n)
        sa_time = min(int(remaining() * 0.5), 600)
        h_b, c5_b = efficient_binary_search(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n
    
    # Phase 3: Top-shifts equalization on best
    if remaining() > 20:
        h_eq, c5_eq = top_shifts_focus(best_h, best_n, top_k=5, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
    
    # Phase 4: More binary search from the improved point
    if remaining() > 40:
        h_bin2 = binarize(best_h, best_n)
        sa_time2 = min(int(remaining() * 0.3), 300)
        h_b2, c5_b2 = efficient_binary_search(h_bin2, best_n, rng, time_limit=sa_time2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()
            best_n = best_n
    
    # Phase 5: Continuous refinement
    if remaining() > 20:
        for beta in [5000, 15000, 40000]:
            if remaining() < 8:
                break
            h_c = slsqp_smooth(best_h, best_n, beta, maxiter=400)
            c5_c = compute_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
    
    # Phase 6: Multi-resolution exploration
    if remaining() > 60:
        n_res = [best_n - 16, best_n - 8, best_n + 8, best_n + 16, 
                 best_n - 32, best_n + 32, 256, 320, 512, 640]
        n_res = [x for x in n_res if 128 <= x <= 700 and x % 2 == 0]
        n_res = list(set(n_res))
        n_res.sort(key=lambda x: abs(x - best_n))
        
        for n_t in n_res[:6]:
            if remaining() < 20:
                break
            
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            h_t_bin = binarize(h_t, n_t)
            c5_t0 = compute_c5(h_t_bin)
            
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t_bin.copy()
                best_n = n_t
            
            # Quick binary search at this resolution
            if remaining() > 25:
                h_tb, c5_tb = efficient_binary_search(h_t_bin, n_t, rng, time_limit=min(30, int(remaining() * 0.3)))
                if c5_tb < best_c5:
                    best_c5 = c5_tb
                    best_h = h_tb.copy()
                    best_n = n_t
    
    # Phase 7: Diverse starts
    if remaining() > 50:
        # Thue-Morse
        h_tm = thue_morse(best_n, rng)
        c5_tm = compute_c5(h_tm)
        if remaining() > 30:
            h_tm2, c5_tm2 = efficient_binary_search(h_tm, best_n, rng, time_limit=min(30, int(remaining() * 0.3)))
            if c5_tm2 < best_c5:
                best_c5 = c5_tm2
                best_h = h_tm2.copy()
        
        # Inverse Thue-Morse
        if remaining() > 30:
            h_tm3 = thue_morse(best_n, rng)
            h_tm3 = 1.0 - h_tm3
            h_tm3 = project_feasible(h_tm3, best_n)
            h_tm3_bin = binarize(h_tm3, best_n)
            h_tm4, c5_tm4 = efficient_binary_search(h_tm3_bin, best_n, rng, time_limit=min(25, int(remaining() * 0.25)))
            if c5_tm4 < best_c5:
                best_c5 = c5_tm4
                best_h = h_tm4.copy()
        
        # Random start
        if remaining() > 30:
            h_r = rng.uniform(0, 1, best_n)
            h_r = project_feasible(h_r, best_n)
            h_r_bin = binarize(h_r, best_n)
            h_r2, c5_r2 = efficient_binary_search(h_r_bin, best_n, rng, time_limit=min(25, int(remaining() * 0.25)))
            if c5_r2 < best_c5:
                best_c5 = c5_r2
                best_h = h_r2.copy()
    
    # Phase 8: Final intensive binary search
    if remaining() > 30:
        h_bin_final = binarize(best_h, best_n)
        final_time = min(int(remaining() * 0.6), 120)
        h_f, c5_f = efficient_binary_search(h_bin_final, best_n, rng, time_limit=final_time)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            best_n = best_n
    
    # Phase 9: Final continuous polish
    if remaining() > 15:
        h_polish = slsqp_smooth(best_h, best_n, 20000, maxiter=300)
        c5_polish = compute_c5(h_polish)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish.copy()
        
        h_eq_final, c5_eq_final = top_shifts_focus(best_h, best_n, top_k=3, max_iters=200)
        if c5_eq_final < best_c5:
            best_c5 = c5_eq_final
            best_h = h_eq_final.copy()
    
    # Phase 10: Subgradient final
    if remaining() > 10:
        h_sg, c5_sg = subgradient_max(best_h, best_n, n_iter=80)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Ensure feasibility
    final_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h
    
    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
