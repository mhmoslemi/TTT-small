    # Precompute
    ones_list = np.where(h_bin == 1)[0].copy()
    zeros_list = np.where(h_bin == 0)[0].copy()
    
    rng = np.random.RandomState(12345)
    n_total = max_iters
    
    for it in range(n_total):
        if remaining() < 3:
            break
        
        T = t_start * (t_end / t_start) ** (it / max(1, n_total - 1))
        
        # Choose move type: 1-swap (80%) or 2-swap (20%)
        move_type = rng.random() < 0.7
        
        if move_type:
            # 1-swap: flip one 1 and one 0
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = int(ones_list[rng.randint(len(ones_list))])
            j = int(zeros_list[rng.randint(len(zeros_list))])
            if i == j:
                continue
            
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            
            if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
                cur_c5 = new_c5
                # Update lists
                ones_list = np.where(h_bin == 1)[0]
                zeros_list = np.where(h_bin == 0)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        else:
            # 2-swap: flip two 1s and two 0s
            if len(ones_list) < 2 or len(zeros_list) < 2:
                break
            i1 = int(ones_list[rng.randint(len(ones_list))])
            i2 = int(ones_list[rng.randint(len(ones_list))])
            j1 = int(zeros_list[rng.randint(len(zeros_list))])
            j2 = int(zeros_list[rng.randint(len(zeros_list))])
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            
            h_bin[i1] = 0.0
            h_bin[i2] = 0.0
            h_bin[j1] = 1.0
            h_bin[j2] = 1.0
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            
            if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
                cur_c5 = new_c5
                ones_list = np.where(h_bin == 1)[0]
                zeros_list = np.where(h_bin == 0)[0]
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h_bin.copy()
            else:
                h_bin[i1] = 1.0
                h_bin[i2] = 1.0
                h_bin[j1] = 0.0
                h_bin[j2] = 0.0
    
    return best_h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def multiple_start_opt(h, n, num_starts=4, time_per=30):
    """Try multiple perturbed starts."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(99988)
    
    for s in range(num_starts):
        if remaining() < time_per:
            break
        # Create perturbation
        pert = rng.randn(n) * 0.05
        h_try = proj(h + pert, n)
        c5_try = compute_c5(h_try)
        
        # Quick L-BFGS
        for beta in [500, 2000, 8000]:
            if remaining() < 5:
                break
            h_try = lbfgs_pass(h_try, n, beta, top_k=5, maxiter=200)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    return best_h, best_c5
