import numpy as np

import time

_START_TIME = 0.0
_BUDGET_S = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)


def thue_morse_seq(n):
    """Generate Thue-Morse binary sequence of length n."""
    h = np.zeros(n)
    for i in range(n):
        v = i
        parity = 0
        while v:
            parity ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if parity == 1 else 0.0
    return h


def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    # Threshold at 0.5, then fix count
    b = (h > 0.5).astype(float)
    total = np.sum(b)
    target = n / 2.0
    if total > target:
        # Need to flip some 1s to 0s
        ones = np.where(b == 1)[0]
        # Flip the ones with lowest h values
        order = np.argsort(h[ones])
        for idx in order[:int(total - target)]:
            b[ones[idx]] = 0.0
    elif total < target:
        # Need to flip some 0s to 1s
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])  # Flip zeros with highest h values
        for idx in order[:int(target - total)]:
            b[zeros[idx]] = 1.0
    return b


def binary_swap_improve(h, n, rng, n_trials=50000):
    """Improve binary solution via random swaps."""
    b = make_binary(h, n)
    corr = compute_corr(b).astype(float)
    best_max = np.max(corr)
    dx = 2.0 / n
    
    ones = np.where(b == 1)[0].copy()
    zeros = np.where(b == 0)[0].copy()
    
    offsets = np.arange(-(n-1), n)
    
    best_b = b.copy()
    best_corr = corr.copy()
    
    for t in range(n_trials):
        if remaining_time() < 5:
            break
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Compute delta corr
        # When h[i] goes 1->0 and h[j] goes 0->1:
        # delta_corr[k] = h[i-k] - h[i+k] + (1-h[j+k]) - h[j-k]
        # where out-of-bounds index gives 0
        
        i_plus = i + offsets
        i_minus = i - offsets
        j_plus = j + offsets
        j_minus = j - offsets
        
        valid_ip = (i_plus >= 0) & (i_plus < n)
        valid_im = (i_minus >= 0) & (i_minus < n)
        valid_jp = (j_plus >= 0) & (j_plus < n)
        valid_jm = (j_minus >= 0) & (j_minus < n)
        
        delta = np.zeros(2*n-1)
        # -(1-h[i+k]) for valid i+k
        if np.any(valid_ip):
            delta[valid_ip] -= (1.0 - b[i_plus[valid_ip]])
        # +h[i-k] for valid i-k
        if np.any(valid_im):
            delta[valid_im] += b[i_minus[valid_im]]
        # +(1-h[j+k]) for valid j+k
        if np.any(valid_jp):
            delta[valid_jp] += (1.0 - b[j_plus[valid_jp]])
        # -h[j-k] for valid j-k
        if np.any(valid_jm):
            delta[valid_jm] -= b[j_minus[valid_jm]]
        
        new_corr = corr + delta
        new_max = np.max(new_corr)
        
        if new_max < best_max - 1e-15:
            # Accept
            b[i] = 0.0
            b[j] = 1.0
            corr = new_corr
            best_max = new_max
            best_b = b.copy()
            best_corr = new_corr
            # Update ones and zeros lists
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
    
    return best_b, best_max * dx


def fast_coord_descent(h, n, max_sweeps=200):
    """Vectorized coordinate descent with exact per-coordinate optimization."""
    best = h.copy()
    best_corr = compute_corr(best).astype(float)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    
    offsets = np.arange(-(n - 1), n, dtype=np.int64)
    
    for sweep in range(max_sweeps):
        if remaining_time() < 5:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)
            
            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0
            
            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)
            
            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))
            
            t_candidate = 0.5
            f_cand = max(base[n - 1] + t_candidate - t_candidate * t_candidate,
                         np.max(base + A * t_candidate))
            t_best = old
            f_best = f_old
            if f_cand < f_best - 1e-14:
                t_best = t_candidate
                f_best = f_cand
            
            f_0 = max(base[n - 1], np.max(base))
            if f_0 < f_best - 1e-14:
                t_best = 0.0
                f_best = f_0
            f_1 = max(base[n - 1], np.max(base + A))
            if f_1 < f_best - 1e-14:
                t_best = 1.0
                f_best = f_1
            
            # Golden section
            if t_best != old:
                if t_best > old:
                    lo = old
                    hi = min(1.0, t_best + 0.2)
                else:
                    lo = max(0.0, t_best - 0.2)
                    hi = old
                if hi - lo < 1e-10:
                    hi = lo + 1e-6
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                
                def _eval_t(t_val, base_arr, A_arr):
                    mid = base_arr + A_arr * t_val
                    mid[n - 1] = base_arr[n - 1] + t_val - t_val * t_val
                    return np.max(mid)
                
                f1 = _eval_t(c1, base, A)
                f2 = _eval_t(c2, base, A)
                for _ in range(35):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _eval_t(c1, base, A)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        f2 = _eval_t(c2, base, A)
                t_best = (lo + hi) / 2.0
                f_best = _eval_t(t_best, base, A)
            
            if f_best < f_old - 1e-13:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True
        
        if not improved:
            break
    
    return best, best_c5


def block_coord_descent(h, n, block_size=4, max_sweeps=50):

    best = h.copy()

                i_plus = i + offsets
                i_minus = i - offsets
                valid_plus = (i_plus >= 0) & (i_plus < n)
                valid_minus = (i_minus >= 0) & (i_minus < n)
                
                A = np.zeros(2*n-1)
                if np.any(valid_minus):
                    A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
                if np.any(valid_plus):

                A[n-1] = 0.0
                
                # Find best value for this coordinate
                base = best_corr - A * old
                base[n-1] = best_corr[n-1] - old * (1.0 - old)
                
                # Try a few candidates
                cands = [0.0, 1.0, 0.25, 0.5, 0.75, old]
                best_t = old
                best_f = max(base[n-1] + old - old*old, np.max(best_corr))
                for t in cands:

                    if f < best_f - 1e-14:

                        best_t = t
                
                if best_t != old:
                    best[i] = best_t
                    best_corr += A * (best_t - old)
                    best_corr[n-1] += best_t*(1-best_t) - old*(1-old)

        new_c5 = np.max(best_corr) * dx
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
        elif not improved:
            break
    
    return best, best_c5


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    
    grad = np.zeros(n)
    ncorr = 2*n - 1
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset+npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset+npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff+npairs])
                grad[aoff:aoff+npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def slsqp_refine(h, n, beta, maxiter):
    _n_work = n
    _beta_work = beta
    
    def _obj(x):
        return smooth_obj_and_grad(x, _n_work, _beta_work)[0]
    
    def _obj_grad(x):
        return smooth_obj_and_grad(x, _n_work, _beta_work)[1]
    
    def _cons_fun(x):
        return np.sum(x) - _n_work / 2.0
    
    def _cons_jac(x):
        return np.ones(_n_work)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def lp_jump(h, n, rng):
    """LP-based jump: linearize the objective and solve the resulting LP."""
    import cvxpy as cp
    
    corr = compute_corr(h)
    max_corr = np.max(corr)
    # Active set: top correlations
    threshold = max_corr - 0.02 * max_corr
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > 50:
        active = np.sort(np.argsort(corr)[-50:])
    
    # Linearize: f_k(h) ≈ f_k(h0) + grad_k . (h - h0)
    # f_k(h) = sum_i h[i]*(1-h[i+k]) for valid i
    # grad_k[j] = (1-h[j+k]) - h[j-k] for valid indices
    # (h[x] = 0 for x outside [0,n-1])
    
    dx = 2.0 / n
    ncorr = 2 * n - 1
    
    # Compute linearized constraints
    # For each active k: f_k(h0) + sum_j grad_k[j]*(h[j]-h0[j]) <= t
    # Which is: sum_j grad_k[j]*h[j] <= t - f_k(h0) + sum_j grad_k[j]*h0[j]
    # = t - (f_k(h0) - sum_j grad_k[j]*h0[j])
    # Note: f_k(h0) - sum_j grad_k[j]*h0[j] is the "constant" part
    
    # Actually, let's just build the LP directly
    var_h = cp.Variable(n)
    var_t = cp.Variable()
    
    constraints = [
        var_h >= 0,
        var_h <= 1,
        cp.sum(var_h) == n / 2.0
    ]
    
    offsets = np.arange(-(n-1), n)
    
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        # Compute gradient of f_k at h0
        grad_k = np.zeros(n)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                # f_k = sum_{i=0}^{n-1-offset} h[i]*(1-h[i+offset])
                # df_k/dh[j] = (1-h[j+offset]) for j < n-offset
                #            - h[j-offset] for j >= offset
                #            (1-h[j+offset]) - h[j-offset] for offset <= j < n-offset
                for j in range(npairs):
                    grad_k[j] += (1.0 - h[j + offset])
                for j in range(offset, n):
                    grad_k[j] -= h[j - offset]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                for j in range(npairs):
                    grad_k[j] += (1.0 - h[j + aoff])
                for j in range(aoff, n):
                    grad_k[j] -= h[j - aoff]
        
        # Compute f_k(h0)
        f_k0 = corr[k_idx]
        
        # Constant: f_k0 - grad_k . h0
        const = f_k0 - np.dot(grad_k, h)
        
        # Constraint: grad_k . h + const <= t
        constraints.append(grad_k @ var_h + const <= var_t)
    
    objective = cp.Minimize(var_t * dx)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)
        if var_h.value is not None:
            new_h = np.array(var_h.value).flatten()
            new_h = project_feasible(new_h, n)
            new_c5 = compute_c5(new_h)
            orig_c5 = compute_c5(h)
            if new_c5 < orig_c5:
                return new_h, new_c5
    except Exception:
        pass
    
    return h.copy(), compute_c5(h)


def interp_to(src_h, src_n, dst_n):

**Better strategy**: Instead of binary swaps, use **"delta" moves**: pick two random positions i and j, and transfer a small amount ε from h[i] to h[j] (or vice versa). This preserves the sum constraint and allows continuous exploration.

**Delta move algorithm:**
1. Pick i, j randomly (i ≠ j)
2. If h[i] > ε and h[j] < 1-ε: transfer ε from i to j
3. Check if C₅ improved
4. If yes, accept; if no, reject (or accept with probability exp(-Δ/T) for simulated annealing)

This is essentially a form of MCMC / simulated annealing on the continuous space.

For the efficiency, we can use the incremental correlation update:
When h[i] changes by -ε and h[j] changes by +ε:
delta_corr[k] = A[k]*(-ε) + B[k]*ε

where A[k] is the gradient of corr[k] with respect to h[i] (i.e., the coefficient of h[i] in corr[k] when h[i] is the variable), and B[k] is the gradient with respect to h[j].

For corr[k] = sum_m h[m]*(1-h[m+k]):
∂corr[k]/∂h[i] = (1-h[i+k]) - h[i-k] (with boundary handling)

So:
delta_corr[k] = -ε * [(1-h[i+k]) - h[i-k]] + ε * [(1-h[j+k]) - h[j-k]]
= ε * [(h[j-k] - h[j+k]) - (h[i-k] - h[i+k])]
= ε * [(h[j-k] - h[i-k]) - (h[j+k] - h[i+k])]

This is O(1) per k, so O(n) per move. Very fast!

def delta_moves(h, n, rng, n_moves=100000, eps_base=0.05, cooling=0.999):

        j = rng.randint(n)
        if i == j:

        # Try both
        for sign in [1, -1]:

                if best[i] < eps or best[j] > 1 - eps:

                ii, jj = i, j

            jj_plus = jj + offsets

                best[jj] += dj

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def thue_morse_seq(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        parity = 0
        while v:
            parity ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if parity == 1 else 0.0
    return h

def smooth_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    vals = np.exp(shifted)
    wsum = np.sum(vals)
    return (m + np.log(wsum) / beta) * (2.0 / n)


def smooth_max_grad(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    vals = np.exp(shifted)
    wsum = np.sum(vals)
    weights = vals / wsum
    dx = 2.0 / n
    ncorr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(ncorr):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return grad


def slsqp_refine(h, n, beta, maxiter):
    n_w = n
    beta_w = beta

    def _obj(x):
        return smooth_max_obj(x, n_w, beta_w)

    def _obj_grad(x):
        return smooth_max_grad(x, n_w, beta_w)

    def _cons_fun(x):
        return np.sum(x) - n_w / 2.0

    def _cons_jac(x):
        return np.ones(n_w)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()

            if c5_tmp < best_c5:

                best_h = h_tmp.copy()

    # Phase 5: More delta moves with different parameters
    if remaining_time() > 60:
        h_tmp = best_h.copy()
        h_tmp, c5_tmp = delta_moves(h_tmp, n, rng, n_moves=400000, eps_base=0.02)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_tmp.copy()
        h_ref = best_h.copy()
        h_ref, c5_ref = fast_coord_descent(h_ref, n, max_sweeps=80)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    # Phase 6: Perturbation-based multi-start
    if remaining_time() > 80:
        for p in range(6):
            if remaining_time() < 30:
                break
            scale = 0.08 + 0.04 * p
            h_new, c5_new = perturb_and_refine(best_h, n, rng, scale=scale, coord_sweeps=50)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            # Quick SLSQP on improved
            if c5_new < best_c5 + 0.001 and remaining_time() > 20:
                h_slq = slsqp_refine(h_new, n, 5000, 200)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()

    # Phase 7: Try different n values
    if remaining_time() > 120:
        for trial_n in [512, 768, 1024]:
            if remaining_time() < 50:
                break
            try:
                h_tok = interp_to(best_h, best_n, trial_n)
                h_tok = project_feasible(h_tok, trial_n)
                h_tok, c5_tok = fast_coord_descent(h_tok, trial_n, max_sweeps=60)
                h_bin_tok = make_binary(h_tok, trial_n)
                h_bin_tok, c5_bin_tok = binary_swap_improve(h_bin_tok, trial_n, rng, n_trials=80000)
                if c5_bin_tok < c5_tok:
                    h_tok = h_bin_tok
                    c5_tok = c5_bin_tok
                h_tok, c5_tok = fast_coord_descent(h_tok, trial_n, max_sweeps=40)
                if c5_tok < best_c5:
                    best_c5 = c5_tok
                    best_h = h_tok.copy()
                    best_n = trial_n
            except Exception:
                continue

    # Phase 8: Final delta moves
    if remaining_time() > 30:
        h_tmp = best_h.copy()
        h_tmp, c5_tmp = delta_moves(h_tmp, best_n, rng, n_moves=200000, eps_base=0.015)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_tmp.copy()

    # Phase 9: Final coordinate descent + SLSQP
    if remaining_time() > 15:
        h_ref, c5_ref = fast_coord_descent(best_h, best_n, max_sweeps=100)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()

    if remaining_time() > 10:
        h_slq = slsqp_refine(best_h, best_n, 50000, 200)
        c5_slq = compute_c5(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()

    # Final projection
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
