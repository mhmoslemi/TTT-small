def slsqp_smooth(h, n, beta, maxiter=300):
    """SLSQP with log-sum-exp smooth max."""
    h0 = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    
    # Precompute the index structure for correlation
    # corr[k] = sum over valid (i,j) pairs where j-i = k-(n-1)
    # For the gradient, we need for each k:
    #   d(corr[k])/d(h[m]) for all m
    
    def obj(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        return (m + np.log(s) / beta) * (2.0 / n)
    
    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -50.0, 50.0)
        ex = np.exp(cl)
        w = ex / np.sum(ex)
        g = np.zeros(n)
        wc = w * (2.0 / n)
        r = 2 * n - 1
        for k in range(r):
            wk = wc[k]
            if wk < 1e-15:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g
    
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < obj(h0):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

    """Project v onto {0<=x<=1, sum(x)=target}."""
    x = v.copy()
    for _ in range(100):
        x = np.clip(x, 0.0, 1.0)
        diff = target - np.sum(x)
        if abs(diff) < 1e-14:

                g[:np_] += w * (1.0 - h[ao:ao+np_])
                g[ao:ao+np_] -= w * h[:np_]
