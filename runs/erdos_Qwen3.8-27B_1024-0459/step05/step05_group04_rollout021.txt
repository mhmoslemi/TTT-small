def to_binary(h, n):
    """Convert to binary with exactly n/2 ones, preferring values closer to 1."""
    n_half = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n_half]] = 1.0
    return h_bin

def sa_binary(h_bin, n, time_limit, rng, T0=3.0, Tmin=0.005, alpha=0.9997):
    """Simulated annealing on binary representation."""
    h = h_bin.copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        h = to_binary(h, n)
    
    corr = compute_corr(h)
    cur_e = float(np.max(corr))
    best_h = h.copy()
    best_e = cur_e
    dx = 2.0 / n
    
    start = time.time()
    T = T0
    steps = 0
    no_improve = 0
    
    while time.time() - start < time_limit and remaining() > 5:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_corr = compute_corr(h)
        new_e = float(np.max(new_corr))
        
        delta = new_e - cur_e
        if delta <= 0 or (T > Tmin and rng.random() < math.exp(-delta / T)):
            cur_e = new_e
            corr = new_corr
            if cur_e < best_e:
                best_e = cur_e
                best_h = h.copy()
                no_improve = 0
            steps += 1
        else:
            h[i] = 1
            h[j] = 0
        
        T *= alpha
        if T < Tmin:
            T = Tmin
        
        # Reheat
        if steps > 0 and steps % 50000 == 0 and no_improve > 20000:
            T = T0 * 0.5
            no_improve = 0
    
    return best_h, best_e * dx

def targeted_local_search(h_bin, n, time_limit, rng):

    h = h_bin.copy()

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()

            off = k - (n - 1)
