                    h[b] = 1.0
                    corr = new_corr
                    # Update lists
                    ones_list = np.array([x for x in ones_list if x != a] + [b])
                    ones_list = np.unique(ones_list)
                    zeros_list = np.array([x for x in zeros_list if x != b] + [a])
                    zeros_list = np.unique(zeros_list)
                    current_max = new_max
                    if new_max < best_max:
                        best_max = new_max
                        best_h = h.copy()
                    chosen = True
                    break
        if not chosen:
            pass

    return best_h, best_max * dx


def fast_coord_descent(h, n, max_sweeps=100):
    best = h.copy()
    best_corr = compute_corr(best).astype(float).copy()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    best_c5 = np.max(best_corr) * dx
    offsets = np.arange(-(n - 1), n, dtype=np.int64)

    for sweep in range(max_sweeps):
        if remaining_time() < 8.0:
            break
        improved = False
        for i in range(n):
            old = best[i]
            i_plus = i + offsets
            i_minus = i - offsets
            valid_plus = (i_plus >= 0) & (i_plus < n)
            valid_minus = (i_minus >= 0) & (i_minus < n)

            A = np.zeros(ncorr)
            if np.any(valid_minus):
                A[valid_minus] += (1.0 - best[i_minus[valid_minus]])
            if np.any(valid_plus):
                A[valid_plus] -= best[i_plus[valid_plus]]
            A[n - 1] = 0.0

            base = best_corr - A * old
            base[n - 1] = best_corr[n - 1] - old * (1.0 - old)

            f_old = max(base[n - 1] + old - old * old, np.max(best_corr))

            # Evaluate at a few candidates
            candidates = [0.0, 1.0, 0.5, old]
            # Add points near where linear part peaks
            A_max_idx = np.argmax(base + A * 0.5)
            t_linear = 0.5
            for _ in range(15):
                val_at_t = base + A * t_linear
                val_at_t[n - 1] = base[n - 1] + t_linear - t_linear * t_linear
                peak_idx = np.argmax(val_at_t)
                if peak_idx == n - 1:
                    # Quadratic dominates, optimum at 0.5
                    t_new = 0.5
                else:
                    Ak = A[peak_idx]
                    if abs(Ak) < 1e-15:
                        t_new = t_linear
                    elif Ak > 0:
                        t_new = max(0.0, min(1.0, -base[peak_idx] / Ak))
                    else:
                        # Linear is decreasing, try boundary
                        t_new = 0.0 if base[peak_idx] + A[peak_idx] > base[n - 1] else 1.0
                        t_new = max(0.0, min(1.0, t_new))
                if abs(t_new - t_linear) < 1e-10:
                    break
                t_linear = 0.5 * (t_linear + t_new)
            candidates.append(t_linear)

            t_best = old
            f_best = f_old
            for tc in candidates:
                tc = max(0.0, min(1.0, tc))
                mid = base + A * tc
                mid[n - 1] = base[n - 1] + tc - tc * tc
                f_c = np.max(mid)
                if f_c < f_best - 1e-14:
                    f_best = f_c
                    t_best = tc

            # Golden section refinement if improvement found
            if t_best != old and abs(t_best - old) > 1e-6:
                if t_best > old:
                    lo, hi = max(0.0, old - 0.05), min(1.0, t_best + 0.05)
                else:
                    lo, hi = max(0.0, t_best - 0.05), min(1.0, old + 0.05)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                def _ev(t):
                    mid2 = base + A * t
                    mid2[n - 1] = base[n - 1] + t - t * t
                    return np.max(mid2)

                f1, f2 = _ev(c1), _ev(c2)
                for _ in range(35):
                    if f1 < f2:
                        hi, c2, f2 = c2, c1, f1
                        c1 = hi - gr * (hi - lo)
                        f1 = _ev(c1)
                    else:
                        lo, c1, f1 = c1, c2, f2
                        c2 = lo + gr * (hi - lo)
                        f2 = _ev(c2)
                t_best = (lo + hi) / 2.0
                f_best = _ev(t_best)

            if f_best < f_old - 1e-13:
                best[i] = t_best
                best_corr += A * (t_best - old)
                best_corr[n - 1] += t_best * (1.0 - t_best) - old * (1.0 - old)
                best_c5 = np.max(best_corr) * dx
                improved = True
        if not improved:
            break
    return best, best_c5


def smooth_obj_and_grad(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.minimum(beta * (corr - m), 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    val = (m + np.log(wsum) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += w * (1.0 - x[:npairs])
                grad[:npairs] -= w * x[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += w * (1.0 - x[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= w * x[:npairs]
    grad *= dx
    return val, grad


def slsqp_refine(h, n, beta, maxiter):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_obj_and_grad(x, n, beta)[0]

    def obj_grad(x):
        return smooth_obj_and_grad(x, n, beta)[1]

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            trial = project_feasible(res.x, n)
            c5_trial = compute_c5(trial)
            c5_orig = compute_c5(h)
            if c5_trial <= c5_orig + 1e-12:
                return trial
    except Exception:
        pass
    return h.copy()


def projected_gradient_step(h, n, step_size, n_active=10):
    corr = compute_corr(h)
    max_corr = np.max(corr)
    threshold = max_corr - 0.5
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) > n_active:
        active = active[np.argsort(corr[active])[::-1][:n_active]]

    grad = np.zeros(n)
    for k_idx in active:
        offset = int(k_idx) - (n - 1)
        if offset >= 0:
            npairs = n - offset
            if npairs > 0:
                grad[offset:offset + npairs] += (1.0 - h[:npairs])
                grad[:npairs] -= h[offset:offset + npairs]
        else:
            aoff = -offset
            npairs = n - aoff
            if npairs > 0:
                grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
                grad[aoff:aoff + npairs] -= h[:npairs]

    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-16:
        return h
    h_new = h - step_size * grad / gnorm
    return project_feasible(h_new, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _rng
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    _rng = np.random.RandomState(seed)
    rng = _rng

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        best_n = 640
        best_h = thue_morse_init(best_n)
        best_c5 = compute_c5(best_h)

    n = best_n
    print(f"Initial: n={n}, C5={best_c5:.6f}", flush=True)

    # Phase 1: Coordinate descent from parent
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        print(f"After coord descent: C5={best_c5:.6f}", flush=True)

    # Phase 2: Binary local search (key new approach)
    if remaining_time() > 60:
        h_bin = (best_h > 0.5).astype(float)
        # Ensure exactly n/2 ones
        n_ones = int(np.sum(h_bin))
        if n_ones > n // 2:
            ones_idx = np.where(h_bin == 1.0)[0]
            _rng.shuffle(ones_idx)
            h_bin[ones_idx[:n_ones - n // 2]] = 0.0
        elif n_ones < n // 2:
            zeros_idx = np.where(h_bin == 0.0)[0]
            _rng.shuffle(zeros_idx)
            h_bin[zeros_idx[:n // 2 - n_ones]] = 1.0

        h_bin, c5_bin = binary_local_search(h_bin, n, max_iters=8000, rng=rng,
                                            t_start=0.08, t_end=0.0005, n_focus=8)
        # Now do continuous refinement from binary solution
        h_cont = project_feasible(h_bin, n)
        h_cont, c5_cont = fast_coord_descent(h_cont, n, max_sweeps=80)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
        print(f"After binary search + refine: C5={best_c5:.6f}", flush=True)

    # Phase 3: SLSQP refinement
    if remaining_time() > 40:
        h = best_h.copy()
        for beta in [2000, 10000, 50000]:
            if remaining_time() < 15:
                break
            h_new = slsqp_refine(h, n, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-15:
                h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        print(f"After SLSQP: C5={best_c5:.6f}", flush=True)

    # Phase 4: More coordinate descent
    if remaining_time() > 30:
        h = best_h.copy()
        h, c5 = fast_coord_descent(h, n, max_sweeps=100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 5: Multiple perturbations
    if remaining_time() > 60:
        for p in range(12):
            if remaining_time() < 30:
                break
            h_pert = best_h.copy()
            noise_scale = 0.10 / (1 + p * 0.3)
            mask = rng.random(n) < 0.25
            h_pert[mask] += rng.randn(np.sum(mask)) * noise_scale
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = fast_coord_descent(h_pert, n, max_sweeps=60)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if remaining_time() > 40:
                    h_slq = slsqp_refine(h_new, n, 10000, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        h_new, c5_new = fast_coord_descent(h_new, n, max_sweeps=40)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
        print(f"After perturbations: C5={best_c5:.6f}", flush=True)

    # Phase 6: Second round of binary local search with different params
    if remaining_time() > 50:
        h_bin2 = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin2))
        if n_ones > n // 2:
            ones_idx = np.where(h_bin2 == 1.0)[0]
            _rng.shuffle(ones_idx)
            h_bin2[ones_idx[:n_ones - n // 2]] = 0.0
        elif n_ones < n // 2:
            zeros_idx = np.where(h_bin2 == 0.0)[0]
            _rng.shuffle(zeros_idx)
            h_bin2[zeros_idx[:n // 2 - n_ones]] = 1.0

        h_bin2, c5_bin2 = binary_local_search(h_bin2, n, max_iters=10000, rng=rng,
                                              t_start=0.04, t_end=0.0002, n_focus=5)
        h_cont2 = project_feasible(h_bin2, n)
        h_cont2, c5_cont2 = fast_coord_descent(h_cont2, n, max_sweeps=80)
        if c5_cont2 < best_c5:
            best_c5 = c5_cont2
            best_h = h_cont2.copy()
        print(f"After 2nd binary search: C5={best_c5:.6f}", flush=True)

    # Phase 7: Try higher resolution
    if remaining_time() > 100:
        high_n = min(2 * n, 1600)
        if high_n > n:
            h_high = interp_to(best_h, n, high_n)
            h_high = project_feasible(h_high, high_n)
            c5_high = compute_c5(h_high)
            if c5_high < best_c5 + 0.01:
                h_new, c5_new = fast_coord_descent(h_high, high_n, max_sweeps=50)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = high_n
                if remaining_time() > 50:
                    h_slq = slsqp_refine(h_new, high_n, 5000, 300)
                    c5_slq = compute_c5(h_slq)
                    if c5_slq < best_c5:
                        best_c5 = c5_slq
                        best_h = h_slq.copy()
                        best_n = high_n
                        h_new, c5_new = fast_coord_descent(h_new, high_n, max_sweeps=30)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            best_n = high_n
        print(f"After high res: C5={best_c5:.6f}, n={best_n}", flush=True)

    # Phase 8: Thue-Morse based exploration
    if remaining_time() > 50:
        h_tm = thue_morse_init(n)
        h_tm, c5_tm = fast_coord_descent(h_tm, n, max_sweeps=80)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            if remaining_time() > 30:
                h_slq = slsqp_refine(h_tm, n, 20000, 300)
                c5_slq = compute_c5(h_slq)
                if c5_slq < best_c5:
                    best_c5 = c5_slq
                    best_h = h_slq.copy()
                    h_tm, c5_tm = fast_coord_descent(h_tm, n, max_sweeps=40)
                    if c5_tm < best_c5:
                        best_c5 = c5_tm
                        best_h = h_tm.copy()

    # Phase 9: Projected gradient + coordinate descent alternating
    if remaining_time() > 40:
        n_cur = best_n
        h = best_h.copy()
        for it in range(300):
            if remaining_time() < 5:
                break
            step = 0.015 / (1.0 + it * 0.02)
            h_sg = projected_gradient_step(h, n_cur, step, n_active=8)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5 - 1e-15:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h = h_sg.copy()
            elif it % 30 == 15 and remaining_time() > 20:
                h_new, c5_new = fast_coord_descent(h, n_cur, max_sweeps=15)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                elif c5_new < c5_sg:
                    h = h_new.copy()
                else:
                    break

    # Phase 10: Final SLSQP polish
    if remaining_time() > 15:
        n_cur = best_n
        h = best_h.copy()
        for beta in [20000, 100000]:
            if remaining_time() < 8:
                break
            h_new = slsqp_refine(h, n_cur, beta, 400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

    # Phase 11: Final coordinate descent
    if remaining_time() > 10:
        n_cur = best_n
        h_new, c5_new = fast_coord_descent(best_h, n_cur, max_sweeps=60)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Final projection and evaluation
    n_final = best_n
    best_h = project_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print(f"FINAL: C5={final_c5:.8f}, n={n_final}", flush=True)
    return (best_h, final_c5, n_final)
