        # Double swap (2-for-2) for escaping
        if best_i < 0 and stagnation > 50:
            if len(ones_indices) >= 2 and len(zeros_indices) >= 2:
                for _ in range(10):
                    i1 = int(ones_indices[rng.randint(len(ones_indices))])
                    i2_val = int(ones_indices[rng.randint(len(ones_indices))])
                    j1 = int(zeros_indices[rng.randint(len(zeros_indices))])
                    j2 = int(zeros_indices[rng.randint(len(zeros_indices))])
                    if i1 == i2_val or j1 == j2 or i1 == j1 or i1 == j2 or i2_val == j1 or i2_val == j2:
                        continue
                    h[i1] = 0
                    h[i2_val] = 0
                    h[j1] = 1
                    h[j2] = 1
                    nm = float(np.max(compute_corr(h)))
                    h[i1] = 1
                    h[i2_val] = 1
                    h[j1] = 0
                    h[j2] = 0
                    delta = nm - cur_max
                    if delta < best_delta - 1e-12:
                        best_delta = delta
                        best_i, best_j = i1, j1
                        best_j2 = j2

        if best_i >= 0:
            h[best_i] = 0
            if 'best_j2' in dir() and best_i < 0:
                pass
            # Handle double swap case
            if best_delta < -1.5:
                # Might be double swap - just apply single for safety
                pass
            h[best_j] = 1
            stagnation = 0
            nm = float(np.max(compute_corr(h)))
            if nm < best_max:
                best_max = nm
                best_h = h.copy()
        else:
            stagnation += 1
            if stagnation > 80:
                n_flips = min(1 + stagnation // 100, 6)
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                for _ in range(n_flips):
                    if len(oi) == 0 or len(zi) == 0:
                        break
                    fi = int(oi[rng.randint(len(oi))])
                    fj = int(zi[rng.randint(len(zi))])
                    h[fi] = 0
                    h[fj] = 1
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
                best_max = max(best_max, float(np.max(compute_corr(h))))
                if float(np.max(compute_corr(h))) < best_max:
                    best_max = float(np.max(compute_corr(h)))
                    best_h = h.copy()
                stagnation = 0

    return best_h, best_max * dx


def binary_search_v3(h_bin, n, rng, time_limit=120):
    """More systematic binary search with simulated annealing."""
    h = (h_bin > 0.5).astype(np.float64).copy()
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        if n_ones > n_half:
            idx1 = np.where(h == 1)[0]
            rng.shuffle(idx1)
            h[idx1[:n_ones - n_half]] = 0
        else:
            idx0 = np.where(h == 0)[0]
            rng.shuffle(idx0)
            h[idx0[:n_half - n_ones]] = 1

    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n

    start = time.time()
    temp = 2.0
    min_temp = 0.3
    cool_rate = 0.995
    no_improve = 0

    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h)
        cur_max = float(np.max(corr))

        # Reset temperature if we found new best
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        top_idx = np.argpartition(-corr, min(6, 2*n-2))[:min(6, 2*n-1)]
        top_idx = top_idx[np.argsort(-corr[top_idx])]

        ones_indices = np.where(h == 1)[0]
        zeros_indices = np.where(h == 0)[0]
        if len(ones_indices) == 0 or len(zeros_indices) == 0:
            break

        # Generate candidates
        candidates = []

        # Fault-based candidates at hot shifts
        for k_idx in top_idx[:4]:
            m = int(k_idx) - (n - 1)
            if m == 0:
                continue
            abs_m = abs(m)
            if abs_m >= n:
                continue
            if m > 0:
                faults = np.where((h[:n-m] == 1) & (h[m:] == 0))[0]
                if len(faults) > 15:
                    faults = rng.choice(faults, 15, replace=False)
                for t in faults:
                    candidates.append((int(t), int(t + m)))
            else:
                ao = abs_m
                faults = np.where((h[:n-ao] == 0) & (h[ao:] == 1))[0]
                if len(faults) > 15:
                    faults = rng.choice(faults, 15, replace=False)
                for t in faults:
                    candidates.append((int(t + ao), int(t)))

        # Random candidates
        for _ in range(25):
            i = int(ones_indices[rng.randint(len(ones_indices))])
            j = int(zeros_indices[rng.randint(len(zeros_indices))])
            if i != j:
                candidates.append((i, j))

        if not candidates:
            break

        # Evaluate candidates
        best_cand_delta = float('inf')
        best_cand = None

        rng.shuffle(candidates)
        for (i, j) in candidates[:60]:
            if h[i] != 1 or h[j] != 0:
                continue
            h[i] = 0
            h[j] = 1
            nm = float(np.max(compute_corr(h)))
            h[i] = 1
            h[j] = 0
            delta = nm - cur_max
            if delta < best_cand_delta:
                best_cand_delta = delta
                best_cand = (i, j)

        if best_cand is None:
            temp *= cool_rate
            continue

        # Accept if improving or with SA probability
        i, j = best_cand
        accept = best_cand_delta < 0
        if not accept and temp > min_temp:
            # SA: accept with probability exp(-delta/temp)
            prob = np.exp(-best_cand_delta / temp)
            accept = (rng.random() < prob)

        if accept:
            h[i] = 0
            h[j] = 1
            temp *= cool_rate
        else:
            temp *= cool_rate

        if no_improve > 200:
            temp = max(temp, 1.5)
            no_improve = 0

    return best_h, best_max * dx


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def top_shifts_focus(h, n, top_k=5, max_iters=200):
    h = h.copy()
    dx_v = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 3:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min_top = corr[idx[-1]]
        if c_max - c_min_top < 1e-9:
            break
        k_top = idx[0]
        k_bot = idx[-1]
        off_t = k_top - (n - 1)
        off_b = k_bot - (n - 1)

        g_t = np.zeros(n)
        g_b = np.zeros(n)
        for g, off in [(g_t, off_t), (g_b, off_b)]:
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += (1.0 - h[:np_])
                g[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= h[:np_]

        d = g_b - g_t
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            d = -g_t
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                break
        d /= dn

        improved = False
        for step in [0.02, 0.05, 0.01, 0.08, 0.005, 0.03, 0.1, 0.015]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def thue_morse_init(n, rng=None):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1') % 2
        h[i] = 1.0 if c == 1 else 0.0
    n_ones = int(np.sum(h))
    n_half = n // 2
    if n_ones != n_half:
        if n_ones > n_half:
            ones_idx = np.where(h == 1)[0]
            rng.shuffle(ones_idx)
            h[ones_idx[:n_ones - n_half]] = 0
        else:
            zeros_idx = np.where(h == 0)[0]
            rng.shuffle(zeros_idx)
            h[zeros_idx[:n_half - n_ones]] = 1
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _N, _BETA, _TGT
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse_init(parent_n, rng)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    n = best_n
    h = best_h.copy()

    # Phase 1: Quick SLSQP to improve continuous solution
    for beta in [2000, 8000, 30000]:
        if remaining() < 5:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=300)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Binarize and intensive binary search (MAIN PHASE)
    if remaining() > 100:
        h_bin = binarize(h, n)
        c5_bin_init = compute_c5(h_bin)
        if c5_bin_init < best_c5:
            best_c5 = c5_bin_init
            best_h = h_bin.copy()

        # Binary search v2 (targeted)
        sa_time = min(90, max(20, int(remaining() * 0.35)))
        h_b, c5_b = binary_search_v2(h_bin, n, rng, time_limit=sa_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

        # Binary search v3 (simulated annealing)
        if remaining() > 30:
            sa_time2 = min(60, max(10, int(remaining() * 0.2)))
            h_b3, c5_b3 = binary_search_v3(h_bin, n, rng, time_limit=sa_time2)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()

        # More binary search from best
        if remaining() > 30:
            h_bin2 = binarize(best_h, n)
            sa_time3 = min(50, max(10, int(remaining() * 0.2)))
            h_b4, c5_b4 = binary_search_v2(h_bin2, n, rng, time_limit=sa_time3)
            if c5_b4 < best_c5:
                best_c5 = c5_b4
                best_h = h_b4.copy()

    # Phase 3: Try other resolutions
    if remaining() > 60:
        n_base = best_n
        n_candidates = [n_base - 4, n_base - 2, n_base + 2, n_base + 4,
                        n_base + 6, n_base + 8, n_base + 12, n_base + 16]
        for nt in [256, 320, 384, 448, 512]:
            if nt not in n_candidates and abs(nt - n_base) <= 128:
                n_candidates.append(nt)

        n_candidates = [x for x in n_candidates if 128 <= x <= 640 and x % 2 == 0]
        n_candidates.sort(key=lambda x: abs(x - n_base))

        for n_t in n_candidates[:6]:
            if remaining() < 15:
                break
            h_t = interp_to(best_h, best_n, n_t)
            h_t = project_feasible(h_t, n_t)
            c5_t0 = compute_c5(h_t)
            if c5_t0 < best_c5:
                best_c5 = c5_t0
                best_h = h_t.copy()
                best_n = n_t

            # Binary at this resolution
            h_t_bin = binarize(h_t, n_t)
            c5_tb = compute_c5(h_t_bin)
            if c5_tb < best_c5:
                best_c5 = c5_tb
                best_h = h_t_bin.copy()
                best_n = n_t

            if remaining() > 15:
                tl = min(25, max(5, int(remaining() * 0.15)))
                h_tb, c5_tb2 = binary_search_v2(h_t_bin, n_t, rng, time_limit=tl)
                if c5_tb2 < best_c5:
                    best_c5 = c5_tb2
                    best_h = h_tb.copy()
                    best_n = n_t

            # Quick continuous refinement
            if remaining() > 10:
                h_t = slsqp_smooth(h_t, n_t, 5000, 200)
                c5_tc = compute_c5(h_t)
                if c5_tc < best_c5:
                    best_c5 = c5_tc
                    best_h = h_t.copy()
                    best_n = n_t

    # Phase 4: Diverse starts at best resolution
    if remaining() > 40:
        n = best_n

        # Thue-Morse start
        h_tm = thue_morse_init(n, rng)
        h_tm_bin = binarize(h_tm, n)
        c5_tm_b = compute_c5(h_tm_bin)
        if c5_tm_b < best_c5:
            best_c5 = c5_tm_b
            best_h = h_tm_bin.copy()

        if remaining() > 15:
            tl = min(20, max(5, int(remaining() * 0.2)))
            h_tm_b, c5_tm = binary_search_v2(h_tm_bin, n, rng, time_limit=tl)
            if c5_tm < best_c5:
                best_c5 = c5_tm
                best_h = h_tm_b.copy()

        # Continuous Thue-Morse
        if remaining() > 15:
            h_tm_c = slsqp_smooth(h_tm, n, 5000, 200)
            c5_tmc = compute_c5(h_tm_c)
            if c5_tmc < best_c5:
                best_c5 = c5_tmc
                best_h = h_tm_c.copy()

        # Random starts
        for rs in range(3):
            if remaining() < 10:
                break
            h_r = rng.uniform(0, 1, n)
            h_r = project_feasible(h_r, n)
            h_r_bin = binarize(h_r, n)
            c5_rb = compute_c5(h_r_bin)
            if c5_rb < best_c5:
                best_c5 = c5_rb
                best_h = h_r_bin.copy()
            if remaining() > 10:
                tl = min(15, max(5, int(remaining() * 0.15)))
                h_rb, c5_r = binary_search_v2(h_r_bin, n, rng, time_limit=tl)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_rb.copy()

    # Phase 5: Final refinement at best resolution
    n = best_n
    h = best_h.copy()

    # Try top_shifts on continuous version
    if remaining() > 15:
        h_cont = project_feasible(h, n)
        for tk in [3, 5, 8]:
            if remaining() < 5:
                break
            h_eq, c5_eq = top_shifts_focus(h_cont, n, top_k=tk, max_iters=150)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()
                h = h_eq.copy()

    # One more round of binary search on the best
    if remaining() > 20:
        h_bin_final = binarize(best_h, n)
        tl = min(40, max(5, int(remaining() * 0.3)))
        h_bf, c5_bf = binary_search_v2(h_bin_final, n, rng, time_limit=tl)
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

        # SA variant
        if remaining() > 10:
            tl2 = min(20, max(5, int(remaining() * 0.2)))
            h_bf3, c5_bf3 = binary_search_v3(h_bin_final, n, rng, time_limit=tl2)
            if c5_bf3 < best_c5:
                best_c5 = c5_bf3
                best_h = h_bf3.copy()

    # Final SLSQP polish from best binary
    if remaining() > 10:
        h_polish = slsqp_smooth(best_h, n, 20000, 300)
        c5_polish = compute_c5(h_polish)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish.copy()

        # Top-shifts equalization
        if remaining() > 5:
            h_eq, c5_eq = top_shifts_focus(h_polish, n, top_k=5, max_iters=100)
            if c5_eq < best_c5:
                best_c5 = c5_eq
                best_h = h_eq.copy()

    # Ensure feasibility
    final_n = best_n
    final_h = project_feasible(best_h, final_n)
    final_c5 = compute_c5(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = final_h

    # Validate
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and float(eval_c5) < best_c5:
            best_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, best_c5, final_n)
