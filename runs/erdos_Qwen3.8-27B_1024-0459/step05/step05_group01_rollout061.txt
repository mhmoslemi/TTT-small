def exhaustive_pair_search(h, n, rng=None):
    """Try all possible pair swaps and keep the best."""
    dx = 2.0 / n
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, compute_c5(h)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for i in ones:
        for j in zeros:
            h[i] = 0.0
            h[j] = 1.0
            c = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
            h[i] = 1.0
            h[j] = 0.0
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
                h = best_h.copy()
    return best_h, best_c5

            break
        ones = list(np.where(h_bin == 1.0)[0])
        zeros = list(np.where(h_bin == 0.0)[0])
        if not ones or not zeros:
            break
        best_delta = 1e-13
        best_pair = None
        new_c5 = best_c5
        for i in ones:
            for j in zeros:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c = np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full')) * dx
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                if c < best_c5 - best_delta:
                    best_delta = best_c5 - c
                    best_pair = (i, j)
                    new_c5 = c
        if best_pair is None:
            break
        i, j = best_pair
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        best_c5 = new_c5
        best_h = h_bin.copy()
    return best_h, best_c5


def two_swap_search(h_bin, n, n_trials=3000, time_limit=30):
    dx = 2.0 / n
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    rng = np.random.RandomState(123)
    for _ in range(n_trials):
        if time.time() - start > time_limit or remaining_time() < 5:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1, i2 = rng.choice(len(ones), 2, replace=False)
        j1, j2 = rng.choice(len(zeros), 2, replace=False)
        a1, a2 = ones[i1], ones[i2]
        b1, b2 = zeros[j1], zeros[j2]
        h_bin[a1] = 0.0
        h_bin[a2] = 0.0
        h_bin[b1] = 1.0
        h_bin[b2] = 1.0
        c = np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full')) * dx
        h_bin[a1] = 1.0
        h_bin[a2] = 1.0
        h_bin[b1] = 0.0
        h_bin[b2] = 0.0
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = h_bin.copy()
            h_bin = best_h.copy()
    return best_h, best_c5


def top_k_equalize(h, n, top_k=6, max_iters=800):
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        corr = compute_corr(h)
        idx_top = np.argsort(-corr)[:top_k]
        c_max = corr[idx_top[0]]
        c_min_top = corr[idx_top[-1]]
        if c_max - c_min_top < 1e-10:
            break
        g_dir = np.zeros(n)
        for pos, k in enumerate(idx_top):
            w_k = 1.0 if pos == 0 else (-1.0 / (len(idx_top) - 1))
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_dir[off:] += w_k * (1.0 - h[:np_])
                g_dir[:np_] -= w_k * h[off:]
            else:
                a = -off
                np_ = n - a
                g_dir[:np_] += w_k * (1.0 - h[a:])
                g_dir[a:] -= w_k * h[:np_]
        gn = np.linalg.norm(g_dir)
        if gn < 1e-14:
            break
        d = g_dir / gn
        improved = False
        steps = [0.003, 0.005, 0.01, 0.02, 0.008, 0.015, 0.03, 0.05, 0.1]
        for step in steps:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-13:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def coordinate_refine(h, n, max_sweeps=40):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved_sweep = False
        for i in range(n):
            old = h[i]
            cands = [0.0, 1.0]
            if 0.0 < old < 1.0:
                cands.extend([old * 0.5, old + (1.0 - old) * 0.5,
                              max(0.0, old - 0.1), min(1.0, old + 0.1),
                              max(0.0, old - 0.05), min(1.0, old + 0.05)])
            cands = sorted(set(c for c in cands if 0.0 <= c <= 1.0))
            best_cand_c5 = best_c5
            best_cand_val = old
            for c in cands:
                h[i] = c
                h = project_feasible(h, n)
                c5_v = compute_c5(h)
                if c5_v < best_cand_c5 - 1e-12:
                    best_cand_c5 = c5_v
                    best_cand_val = c
                    improved_sweep = True
            h[i] = best_cand_val
            h = project_feasible(h, n)
            if best_cand_c5 < best_c5:
                best_c5 = best_cand_c5
        if not improved_sweep:
            break
    return h, best_c5


def theta_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt += v & 1
            v >>= 1
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    n_half = n // 2
    n_ones = int(np.sum(h))
    if n_ones != n_half:
        idx = np.argsort(-h)
        h = np.zeros(n)
        h[idx[:n_half]] = 1.0
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET, _CUR_N
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float)
    except Exception:
        best_h = np.zeros(384)
        best_h[:192] = 1.0
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: Progressive SLSQP on parent
    h = best_h.copy()
    for beta in [200, 1000, 10000, 100000, 500000]:
        if remaining_time() < 30:
            break
        h_new = slsqp_opt(h, best_n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 2: Exhaustive binary search at parent n
    if remaining_time() > 60:
        t_lim = min(90, int(remaining_time() * 0.25))
        h_bin, c5_bin = exhaustive_binary_search(h, best_n, max_rounds=60, time_limit=t_lim)
        # Refine the binary result with SLSQP
        h_ref = h_bin.copy()
        for beta in [5000, 50000, 300000]:
            if remaining_time() < 15:
                break
            h_ref = slsqp_opt(h_ref, best_n, beta, maxiter=600)
        c5_ref = compute_c5(h_ref)
        h_ref, c5_ref = top_k_equalize(h_ref, best_n, top_k=6, max_iters=400)
        if c5_ref < best_c5 - 1e-14:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            h = h_ref.copy()
        # Also check raw binary
        if c5_bin < best_c5 - 1e-14:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: Two-swap search to escape local minima
    if remaining_time() > 40:
        h_bin2 = (best_h > 0.5).astype(float)
        n_half = best_n // 2
        n_ones = int(np.sum(h_bin2))
        if n_ones != n_half:
            idx = np.argsort(-h_bin2)
            h_bin2 = np.zeros(best_n)
            h_bin2[idx[:n_half]] = 1.0
        h2s, c5_2s = two_swap_search(h_bin2, best_n, n_trials=5000, time_limit=30)
        if c5_2s < compute_c5((best_h > 0.5).astype(float)) - 1e-14:
            h2s_r = slsqp_opt(h2s, best_n, 50000, maxiter=600)
            c5_2s_r = compute_c5(h2s_r)
            h2s_r, c5_2s_r = top_k_equalize(h2s_r, best_n, top_k=6, max_iters=300)
            if c5_2s_r < best_c5 - 1e-14:
                best_c5 = c5_2s_r
                best_h = h2s_r.copy()
                h = h2s_r.copy()

    # Phase 4: Multi-resolution search
    n_list = []
    for nn in [best_n - 2, best_n, best_n + 2, best_n + 16, best_n + 48,
               best_n + 64, best_n + 96, best_n + 128, best_n + 192]:
        if 128 <= nn <= 1024 and nn % 2 == 0:
            n_list.append(nn)
    for mult in [0.75, 0.875, 1.25, 1.5, 2.0]:
        nn = int(round(best_n * mult / 2.0)) * 2
        if 128 <= nn <= 1024:
            n_list.append(nn)
    n_list = sorted(set(n_list), key=lambda x: -abs(x - best_n))
    n_list = sorted(n_list, key=lambda x: abs(x - best_n))

    for n_t in n_list[:6]:
        if remaining_time() < 40:
            break
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        h_opt = h_feas
        for beta in [1000, 10000, 100000]:
            if remaining_time() < 15:
                break
            h_opt = slsqp_opt(h_opt, n_t, beta, maxiter=500)
        c5_opt = compute_c5(h_opt)
        # Binary search at this resolution
        if remaining_time() > 60:
            h_b_t, c5_b_t = exhaustive_binary_search(h_opt, n_t, max_rounds=30, time_limit=50)
            h_b_t = slsqp_opt(h_b_t, n_t, 50000, maxiter=500)
            c5_b_t = compute_c5(h_b_t)
            if c5_b_t < c5_opt:
                h_opt = h_b_t
                c5_opt = c5_b_t
        h_opt, c5_opt = top_k_equalize(h_opt, n_t, top_k=6, max_iters=300)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t
            h = h_opt.copy()

    # Phase 5: Alternative starting points
    alt_starts = [theta_morse(best_n)]
    h_rand = np.zeros(best_n)
    h_rand[rng.choice(best_n, best_n // 2, replace=False)] = 1.0
    alt_starts.append(h_rand)

    for h_alt in alt_starts:
        if remaining_time() < 40:
            break
        h_opt = h_alt
        for beta in [1000, 10000, 100000]:
            if remaining_time() < 15:
                break
            h_opt = slsqp_opt(h_opt, best_n, beta, maxiter=500)
        c5_opt = compute_c5(h_opt)
        if remaining_time() > 50:
            h_b_alt, c5_b_alt = exhaustive_binary_search(h_opt, best_n, max_rounds=20, time_limit=40)
            h_b_alt = slsqp_opt(h_b_alt, best_n, 50000, maxiter=500)
            c5_b_alt = compute_c5(h_b_alt)
            if c5_b_alt < c5_opt:
                h_opt = h_b_alt
                c5_opt = c5_b_alt
        h_opt, c5_opt = top_k_equalize(h_opt, best_n, top_k=6, max_iters=400)
        if c5_opt < best_c5 - 1e-14:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h = h_opt.copy()

    # Phase 6: Final intensive SLSQP + equalization
    n = best_n
    h = best_h.copy()
    for beta in [200000, 800000, 2000000]:
        if remaining_time() < 12:
            break
        h_new = slsqp_opt(h, n, beta, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    if remaining_time() > 10:
        h_eq, c5_eq = top_k_equalize(h, n, top_k=8, max_iters=600)
        if c5_eq < best_c5 - 1e-14:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 7: Coordinate refinement
    if remaining_time() > 15:
        h_cr, c5_cr = coordinate_refine(best_h, n, max_sweeps=30)
        if c5_cr < best_c5 - 1e-14:
            best_c5 = c5_cr
            best_h = h_cr.copy()
            h = h_cr.copy()

    # Phase 8: Final SLSQP polish
    if remaining_time() > 10:
        h_final = slsqp_opt(best_h, n, 500000, maxiter=800)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5 - 1e-14:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, n)
