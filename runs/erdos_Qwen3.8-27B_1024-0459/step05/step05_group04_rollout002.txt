def projected_gradient_descent(h, n, time_limit=120, rng=None):
    """Projected gradient with Barzilai-Borwein step size for continuous optimization."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = project_feasible(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    start = time.time()
    gamma = 0.5  # smoothing parameter for soft-max
    prev_grad = None
    prev_x = None
    bb_alpha = 0.1

    while time.time() - start < time_limit and remaining() > 5:
        corr = compute_corr(h)
        m = np.max(corr)
        # Soft gradient: weighted average of gradients of top correlations
        exp_vals = np.exp(-(corr - m) * gamma)
        w = exp_vals / np.sum(exp_vals)
        w = w * dx

        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-18:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - h[:np_])
                g[:np_] -= wk * h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - h[ao:ao + np_])
                g[ao:ao + np_] -= wk * h[:np_]

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_x is not None:
            s = h - prev_x
            y = g - prev_grad
            sys2 = np.dot(s, s)
            sy = np.dot(s, y)
            yy2 = np.dot(y, y)
            if sys2 > 1e-16 and sy > 1e-18:
                alpha_bb = sys2 / sy
                beta_bb = sy / yy2 if yy2 > 1e-16 else 0.1
                alpha = bb_alpha * alpha_bb / (1.0 + bb_alpha)
            else:
                alpha = 0.01
        else:
            alpha = 0.01
        alpha = min(alpha, 0.5)

        idx = 0
        while alpha > 1e-6 and idx < 30:
            idx += 1
            h_new = project_feasible(h - alpha * g, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 or c5_new < compute_c5(h):
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                prev_grad = prev_x if prev_x is not None else None
                prev_x = h
                h = h_new
                break
            alpha *= 0.5
        else:
            if idx >= 30:
                # No improvement, try random perturbation
                if rng.randint(20) == 0:
                    h = project_feasible(h + 0.01 * rng.randn(n) * np.gradient(h) * 0 + rng.randn(n) * 0.05, n)
                break

        prev_grad = g
        prev_x = h

        # Increase gamma occasionally for sharper focus
        gamma *= 1.02

    return best_h, best_c5

        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            s = 1e-300
        return (m + np.log(s) / beta) * (2.0 / n)

    def grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        sh = corr - m
        cl = np.clip(beta * sh, -60.0, 60.0)
        ex = np.exp(cl)
        s = np.sum(ex)
        if s < 1e-300:
            return np.zeros(n)
        wv = ex / s
        wc = wv * (2.0 / n)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = wc[k]
            if wk < 1e-18:
                continue
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g[off:off + np_] += wk * (1.0 - x[:np_])
                g[:np_] -= wk * x[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g[:np_] += wk * (1.0 - x[ao:ao + np_])
                g[ao:ao + np_] -= wk * x[:np_]
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return ones_vec

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        if res.success or res.fun < obj(h):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def top_shifts_equalize(h, n, top_k=6, max_iters=400):
    h = project_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if remaining() < 5:
            break
        corr = compute_corr(h)
        idx = np.argsort(-corr)[:top_k]
        c_max = corr[idx[0]]
        c_min = corr[idx[-1]]
        if c_max - c_min < 1e-12:
            break

        g_avg = np.zeros(n)
        for k in idx:
            off = k - (n - 1)
            if off >= 0:
                np_ = n - off
                g_avg[off:off + np_] += (1.0 - h[:np_])
                g_avg[:np_] -= h[off:off + np_]
            else:
                ao = -off
                np_ = n - ao
                g_avg[:np_] += (1.0 - h[ao:ao + np_])
                g_avg[ao:ao + np_] -= h[:np_]

        d = -g_avg / (np.linalg.norm(g_avg) + 1e-15)
        improved = False
        for step in [0.005, 0.01, 0.02, 0.03, 0.05, 0.015, 0.04, 0.008, 0.025, 0.06]:
            ht = project_feasible(h + step * d, n)
            cn = compute_c5(ht)
            if cn < best_c5 - 1e-14:
                h = ht
                best_c5 = cn
                improved = True
                break
        if not improved:
            break
    return h, best_c5

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def to_binary(h, n, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    thresh = 0.5 + 0.01 * (rng.rand(n) - 0.5)
    h_bin = (h > thresh).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones != n_half:
        if n_ones > n_half:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(-h[ones])
            h_bin[ones[order[:n_ones - n_half]]] = 0
        else:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            h_bin[zeros[order[:n_half - n_ones]]] = 1
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _BUDGET
    _T0 = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float).copy()
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 384
        parent_h = thue_morse(parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Binary local search at parent resolution
    n = best_n
    h_bin = to_binary(best_h, n, rng)
    t1_limit = min(250, int(remaining() * 0.22))
    if t1_limit > 20:
        h_b, c5_b = binary_local_search(h_bin, n, time_limit=t1_limit, rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n

    # Phase 2: Try alternative starting points at same resolution
    alt_starts = []
    tm = thue_morse(n)
    alt_starts.append(('tm', tm))
    # Random
    rand_h = (rng.rand(n) > 0.5).astype(float)
    n_half = n // 2
    n_ones = int(np.sum(rand_h))
    if n_ones != n_half:
        if n_ones > n_half:
            idx = np.where(rand_h == 1)[0]
            rand_h[idx[:n_ones - n_half]] = 0
        else:
            idx = np.where(rand_h == 0)[0]
            rand_h[idx[:n_half - n_ones]] = 1
    alt_starts.append(('rand', rand_h))

    for name, sp in alt_starts:
        if remaining() < 40:
            break
        t2 = min(100, int(remaining() * 0.08))
        if t2 > 15:
            h_b2, c5_b2 = binary_local_search(sp, n, time_limit=t2, rng=rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n

    # Phase 3: Try higher resolutions
    resolutions_to_try = []
    for factor in [1.25, 1.5, 2.0, 2.5]:
        nr = int(best_n * factor)
        nr = (nr // 2) * 2  # make even
        if 128 <= nr <= 2048:
            resolutions_to_try.append(nr)
    # Also add some absolute values
    for nr in [512, 640, 768, 1024]:
        if nr != best_n and 128 <= nr <= 2048:
            resolutions_to_try.append(nr)
    resolutions_to_try = sorted(set(resolutions_to_try))
    # Sort by closeness to best_n
    resolutions_to_try.sort(key=lambda x: abs(x - best_n))

    for n_t in resolutions_to_try[:5]:
        if remaining() < 50:
            break
        # Interpolate and binarize
        h_interp = interp_to(best_h, best_n, n_t)
        h_bin_t = to_binary(h_interp, n_t, rng)

        t3 = min(100, int(remaining() * 0.06))
        if t3 < 15:
            continue
        h_b3, c5_b3 = binary_local_search(h_bin_t, n_t, time_limit=t3, rng=rng)
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()
            best_n = n_t

    # Phase 4: Continuous refinement (projected gradient)
    n = best_n
    h = best_h.copy()
    t4_limit = min(200, int(remaining() * 0.18))
    if t4_limit > 20:
        h_pg, c5_pg = projected_gradient_descent(h, n, time_limit=t4_limit, rng=rng)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h = h_pg.copy()

    # Phase 5: Top-shifts equalization
    if remaining() > 30:
        h_eq, c5_eq = top_shifts_equalize(h, n, top_k=8, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Phase 6: SLSQP refinement with increasing beta
    for beta in [500, 2000, 8000, 20000, 50000]:
        if remaining() < 15:
            break
        slsqp_time = remaining()
        if slsqp_time < 12:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=600)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Phase 7: Second round of binary search at potentially new best
    if remaining() > 60:
        h_bin2 = to_binary(best_h, best_n, rng)
        t7 = min(80, int(remaining() * 0.06))
        if t7 > 20:
            h_b7, c5_b7 = binary_local_search(h_bin2, best_n, time_limit=t7, rng=np.random.RandomState(seed+200))
            if c5_b7 < best_c5:
                best_c5 = c5_b7
                best_h = h_b7.copy()

        # Quick continuous polish after binary
        if remaining() > 40:
            h2 = best_h.copy()
            h_pg2, c5_pg2 = projected_gradient_descent(h2, best_n, time_limit=min(60, int(remaining()*0.05)), rng=rng)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2.copy()

    # Phase 8: Final equalization and SLSQP
    if remaining() > 20:
        h3, c5_3 = top_shifts_equalize(best_h, best_n, top_k=10, max_iters=300)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    if remaining() > 15:
        h_new = slsqp_smooth(best_h, best_n, 50000, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 9: Very high resolution attempt if time permits
    if remaining() > 120:
        n_fine = best_n + 256
        n_fine = (n_fine // 2) * 2
        if n_fine <= 2500:
            h_fine = interp_to(best_h, best_n, n_fine)
            h_fine = project_feasible(h_fine, n_fine)
            t9 = min(90, int(remaining() * 0.07))
            if t9 > 20:
                h_fine_pg, c5_fine = projected_gradient_descent(h_fine, n_fine, time_limit=t9, rng=rng)
                if remaining() > 30:
                    h_fine_eq, c5_fine2 = top_shifts_equalize(h_fine_pg, n_fine, top_k=8, max_iters=200)
                    if c5_fine2 < c5_fine:
                        c5_fine = c5_fine2
                        h_fine_pg = h_fine_eq
                if c5_fine < best_c5:
                    best_c5 = c5_fine
                    best_h = h_fine_pg.copy()
                    best_n = n_fine

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
