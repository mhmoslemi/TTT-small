    t0 = time.time()
    
    best_h = None
    best_score = float('inf')
    best_n = 0
    
    # Evaluate initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        h_init = np.clip(h_init, 0.0, 1.0)
        score = compute_c5(h_init)
        if score < best_score:
            best_h = h_init.copy()
            best_score = score
            best_n = len(h_init)
    
    # Try binary version of initial
    if best_h is not None and best_n % 2 == 0:
        h_bin = _binary_round(best_h, best_n)
        s_bin = compute_c5(h_bin)
        if s_bin < best_score:
            best_score = s_bin
            best_h = h_bin.copy()
    
    # Optimization loop over resolutions
    for n in [45, 90, 180, 360]:
        tl = budget_s - (time.time() - t0)
        if tl < 60:
            break
        
        # Create initial guess
        if best_h is not None and best_n == n:
            x0 = best_h.copy()
        elif best_h is not None:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            x0 = np.interp(x_new, x_old, best_h)
        else:
            x0 = np.full(n, 0.5)
        
        # Repeated L-BFGS-B with different parameters
        for T, lam in [(30, 50), (50, 100), (100, 200)]:
            tl = budget_s - (time.time() - t0)
            if tl < 30:
                break
            
            tl_sub = budget_s - (time.time() - t0)
            maxiter = min(300, int(tl_sub * 10))
            x_opt = _run_lbfgsb(x0, n, T, lam, maxiter=maxiter)
            score = compute_c5(x_opt)
            if score < best_score:
                best_h = x_opt.copy()
                best_score = score
                best_n = n
                x0 = x_opt.copy()
        
        # Binary refinement
        if n % 2 == 0:
            h_bin = _binary_round(x0, n)
            s_bin = compute_c5(h_bin)
            if s_bin < best_score:
                best_score = s_bin
                best_h = h_bin.copy()
                best_n = n
            
            # Local search on binary
            tl = budget_s - (time.time() - t0)
            if tl > 60:
                h_ls, s_ls = _local_search_binary(h_bin, max_iters=min(500, int(tl * 5)))
                if s_ls < best_score:
                    best_score = s_ls
                    best_h = h_ls.copy()
                    best_n = n
