def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    """Project h to satisfy sum(h) = n/2 and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0, 1)
    # Iterative projection
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Adjust non-saturated entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def soft_max_value(values, beta):
    """Stable soft max."""
    m = np.max(values)
    if beta < 1e-10:

    return m + np.log(np.sum(np.exp(beta * (values - m)))) / beta

    # corr[k] = sum_{i in valid range} h[i+shift] * (1-h[i])
    # d/dh[j] = sum_{i: i+shift=j} (1-h[i]) - sum_{i: i=j} h[i+shift]
    #         = (1-h[j-shift]) * [valid] - h[j+shift] * [valid]

    for j in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            w = weights[k] * dx
            if w < 1e-14:
                continue
            shift = k - (n - 1)
            i_min = max(0, -shift)
            i_max = min(n - 1, n - 1 - shift)
            
            # h[j] as h[i+shift]: i = j - shift
            i_val = j - shift
            if i_min <= i_val <= i_max:
                g += w * (1.0 - h[i_val])
            
            # h[j] as h[i]: i = j
            if i_min <= j <= i_max:
                js = j + shift
                if 0 <= js <= n - 1:
                    g += w * (-h[js])
        
        grad[j] = g
        # Add penalty gradient
        target = n / 2.0
        grad[j] += 2.0 * penalty * (np.sum(h) - target)
    
    return grad
