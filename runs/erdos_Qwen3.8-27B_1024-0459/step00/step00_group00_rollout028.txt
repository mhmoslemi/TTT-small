  ```python
  import numpy as np
  from scipy.optimize import minimize, fmin_l_bfgs_b
  import time

  def compute_c5(h):
      n = len(h)
      dx = 2.0 / n
      c = np.correlate(h, 1.0 - h, mode='full')
      return float(np.max(c) * dx)

  def smooth_max_obj(h, p):
      c = np.correlate(h, 1.0 - h, mode='full')
      c_safe = np.maximum(c, 1e-12)
      return float(np.sum(np.power(c_safe, p)) ** (1.0 / p))

  def smooth_max_grad(h, p):
      n = len(h)
      c = np.correlate(h, 1.0 - h, mode='full')
      c_safe = np.maximum(c, 1e-12)
      sp = np.sum(np.power(c_safe, p))
      factor = sp ** (1.0 / p - 1.0)
      grad = np.zeros(n)
      for j in range(n):
          s = 0.0
          for p_idx in range(2 * n - 1):
              k = p_idx - (n - 1)
              val = 0.0
              if 0 <= j - k < n:
                  val += (1.0 - h[j - k])
              if 0 <= j + k < n:
                  val -= h[j + k]
              s += c_safe[p_idx] ** (p - 1) * val
          grad[j] = factor * s
      return grad

  def project_to_constraints(h, n):
      """Project h onto {0≤h≤1, sum(h)=n/2} using water-filling."""
      h = h.copy()
      for _ in range(100):
          h = np.clip(h, 0, 1)
          excess = h.sum() - n / 2.0
          if abs(excess) < 1e-12:
              break
          if excess > 0:
              mask = h > 1e-8
              if mask.sum() == 0:
                  break
              h[mask] -= excess / mask.sum()
          else:
              mask = h < 1 - 1e-8
              if mask.sum() == 0:
                  break
              h[mask] -= excess / mask.sum()
      h = np.clip(h, 0, 1)
      return h

  def run(seed=42, budget_s=1300, **kwargs):
      start_time = time.time()
      np.random.seed(seed)

      # Get initial solution
      if 'initial_h_values' in globals() and initial_h_values is not None:
          parent_h = np.asarray(initial_h_values, dtype=float)
          n_parent = len(parent_h)

      def time_left():

      # Optimization at various resolutions
      for target_n in [n_parent, 90, 180, 360]:
          if time_left() < 90:

              h0 = project_to_constraints(h0, target_n)

          n = target_n

                   'jac': lambda x: np.ones(n)}]

                  break

                      x0=h_start,

                      constraints=cons,
                      options={'maxiter': 300, 'ftol': 1e-14}

                  h_res = res.x
                  h_res = project_to_constraints(h_res, n)
                  c5_val = compute_c5(h_res)

                  if c5_val < best_c5:
                      best_c5 = c5_val
                      best_h = h_res.copy()
                      h0 = h_res.copy()

              except Exception:
                  pass

      # Final polish with exact max (subgradient)
      if time_left() > 30:
          n = len(best_h)
          h_curr = best_h.copy()
          dx = 2.0 / n
          for step in range(500):

              max_idx = np.argmax(c)
              # Subgradient with respect to h:

              sg = np.zeros(n)

                      sg[j] += (1.0 - h_curr[j - k])
                  if 0 <= j + k < n:

              lr = 0.01 / n
              h_curr = h_curr - lr * sg
              h_curr = project_to_constraints(h_curr, n)

          c5_val = compute_c5(h_curr)
          if c5_val < best_c5:
              best_c5 = c5_val
              best_h = h_curr

      return (best_h, best_c5, len(best_h))
  ```
