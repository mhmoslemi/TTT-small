import numpy as np
from scipy.optimize import minimize
import time

def project(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def eval_c5(h, n):
    """Compute C5 = max_k correlate(h, 1-h)[k] * dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    """Smooth approximation of max using log-sum-exp."""
    M = np.max(x)
    s = x - M
    s = np.clip(s, -100.0, 0.0)
    e = np.exp(beta * s)
    return float(M + np.log(np.sum(e)) / beta)

def make_objective(n_pts, beta, pen):
    """Create the objective function for optimization."""
    ts = n_pts / 2.0
    dx = 2.0 / n_pts
    def f(h):
        hc = np.clip(h, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode='full') * dx
        return smooth_max(corr, beta) + pen * (hc.sum() - ts) ** 2
    return f

def opt_lbfgsb(h_start, n_pts, beta, pen, maxiter):
    """Run L-BFGS-B optimization from h_start."""
    f = make_objective(n_pts, beta, pen)
    bounds = [(0.0, 1.0)] * n_pts
    try:
        res = minimize(f, h_start.copy(), method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project(res.x, n_pts)
    except Exception:
        return project(h_start, n_pts)

def coordinate_descent(h, n, n_sweeps=10, beta=500.0):
    """Simple coordinate descent: optimize each h[i] in 1D."""
    dx = 2.0 / n
    ts = n / 2.0
    h = project(h, n)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            # Try a small grid of new values
            candidates = np.linspace(0.0, 1.0, 21)
            best_val = old_val
            best_obj = np.inf
            for cv in candidates:
                test_h = h.copy()
                test_h[i] = cv
                # Adjust sum constraint: shift all others
                diff = ts - test_h.sum()
                others = np.ones(n)
                others[i] = 0
                n_others = n - 1
                if n_others > 0:
                    shift = diff / n_others
                    adj = test_h + shift * others
                    adj = np.clip(adj, 0.0, 1.0)
                    adj[i] = cv
                    # Re-project
                    for _ in range(50):
                        d2 = ts - adj.sum()
                        if abs(d2) < 1e-14:
                            break
                        adj = adj + (d2 / n_others) * others
                        adj = np.clip(adj, 0.0, 1.0)
                        adj[i] = cv
                    if abs(adj.sum() - ts) > 0.5:
                        continue
                    corr = np.correlate(adj, 1.0 - adj, mode='full') * dx
                    obj = smooth_max(corr, beta)
                    if obj < best_obj:
                        best_obj = obj
                        best_val = cv
            if best_val != old_val:
                h[i] = best_val
                improved = True
        if not improved:
            break
    return project(h, n)
