    """Compute all shift overlap values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx, dx

    vals, dx = compute_all_shifts(h)
    return np.max(vals)

def project_constraint(h, n):
    """Project h to satisfy sum(h) = n/2 and 0<=h<=1."""
    target = n / 2.0
    current_sum = np.sum(h)
    if np.abs(current_sum - target) < 1e-10:
        return h
    # Adjust proportionally
    # First try uniform shift
    delta = (target - current_sum) / n
    h2 = h + delta
    if np.all(h2 >= 0) and np.all(h2 <= 1):
        return h2
    # If not, use iterative clipping
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        mask_pos = h > 1.0
        mask_neg = h < 0.0
        if np.any(mask_pos):
            overflow = np.sum(h[mask_pos] - 1.0)
            h[mask_pos] = 1.0
            diff -= overflow
        if np.any(mask_neg):
            underflow = np.sum(h[mask_neg])
            h[mask_neg] = 0.0
            diff += underflow
        free = (h > 0) & (h < 1)
        if np.any(free):
            h[free] += diff / np.sum(free)
    h = np.clip(h, 0, 1)
    return h

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions at top level

def compute_c5vals(h):
    """Compute all shift overlap values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(h):
    """Compute C5 bound."""
    return np.max(compute_c5vals(h))

def project_sum(h, target_sum):
    """Project h to satisfy sum=h_target with 0<=h<=1."""
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            break
        h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, k):
    """Top-k mean as smooth max approximation."""
    k = min(k, len(vals))
    return np.mean(np.sort(vals)[::-1][:k])

def obj_gradient(h, n, top_k=5, penalty=500.0):
    """Compute penalized objective and its gradient."""
    target = n / 2.0
    dx = 2.0 / n
    vals = compute_c5vals(h)
    obj_smooth = smooth_max(vals, top_k)
    cv = np.sum(h) - target
    obj = obj_smooth + penalty * cv * cv
    
    # Gradient of smooth max
    top_idx = np.argsort(vals)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        m = idx - (n - 1)
        if m >= 0:
            nm = n - m
            if nm > 0:
                grad[:nm] += (1.0 - h[m:])
            if m < n:
                grad[m:] -= h[:nm]
        else:
            a = -m
            na = n - a
            grad[a:] += (1.0 - h[:na])
            if na > 0:
                grad[:na] -= h[a:]
    grad *= (dx / top_k)
    
    # Gradient of penalty
    grad += 2.0 * penalty * cv
    
    return obj, grad

    t0 = time.time()
    target = n / 2.0
    h0 = project_sum(h0, target)
    
    def fun(x):
        if time.time() - t0 > time_limit:
            raise ValueError("timeout")
        return obj_gradient(x, n, top_k, penalty)
    
    try:
        res = minimize(fun, h0, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': 1000, 'maxfun': 50000,
                               'ftol': 1e-12, 'gtol': 1e-9, 'eps': 1e-7})
    except (ValueError, RuntimeError):
        res = None
    
    if res is not None and res.x is not None:
        h_new = project_sum(res.x, target)
    else:
        h_new = h0
    
    return h_new, compute_c5(h_new)

    h_dst = np.interp(x_dst, x_src, h_src)

def coordinate_descent(h, n, time_limit=100, steps=5000):
    """Simple coordinate descent for refinement."""
    t0 = time.time()
    h = h.copy()
    target = n / 2.0
    h = project_sum(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for step in range(steps):
        if time.time() - t0 > time_limit:
            break
        i = step % n
        old_val = h[i]
        # Try to find best value for h[i] given others fixed
        # The objective as a function of h[i] is piecewise quadratic
        # Simple approach: try several candidate values
        candidates = [old_val, 0.0, 1.0, old_val*0.5, old_val*1.5,

        best_local = old_val
        best_local_c5 = best_c5  # approximate: current c5 is upper bound

            h[i] = cv
            # Re-project to maintain sum
            # Actually, changing one element changes sum by (cv - old_val)
            # For coordinate descent, let's just adjust and check
            new_c5 = compute_c5(h)
            if new_c5 < best_local_c5:
                best_local_c5 = new_c5
                best_local = cv
        
        h[i] = best_local
        # Re-project
        h = project_sum(h, target)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(h):
    return np.max(compute_c5vals(h))

def project_sum(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        count = np.sum(mask)
        if count == 0:
            break
        h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def smooth_max_topk(vals, k):
    k = min(k, len(vals))
    return np.mean(np.sort(vals)[::-1][:k])

def obj_gradient(h, n, top_k=5, penalty=2000.0):
    target = n / 2.0
    dx = 2.0 / n
    vals = compute_c5vals(h)
    obj_s = smooth_max_topk(vals, top_k)
    cv = np.sum(h) - target
    obj = obj_s + penalty * cv * cv
    top_idx = np.argsort(vals)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        m = idx - (n - 1)
        if m >= 0:
            nm = n - m
            if nm > 0:
                grad[:nm] += (1.0 - h[m:])
            if m < n:
                grad[m:] -= h[:nm]
        else:
            a = -m
            na = n - a
            grad[a:] += (1.0 - h[:na])
            if na > 0:
                grad[:na] -= h[a:]
    grad *= (dx / top_k)
    grad += 2.0 * penalty * cv
    return obj, grad

def run_lbgfsb(h0, n, time_limit=250, top_k=5, penalty=2000.0):
    t0 = time.time()
    target = n / 2.0
    h0 = project_sum(h0, target)
    timeout_flag = [False]
    
    def fun(x):
        if time.time() - t0 > time_limit:
            timeout_flag[0] = True
            raise ValueError("timeout")
        return obj_gradient(x, n, top_k, penalty)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    try:
        res = minimize(fun, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 2000, 'maxfun': 100000,
                                'ftol': 1e-14, 'gtol': 1e-10})
        if res.x is not None:
            h_new = project_sum(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    except (ValueError, RuntimeError, Exception):
        pass
    
    return best_h, best_c5

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    x_src = (np.arange(n_src) + 0.5) * dx_src
    dx_dst = 2.0 / n_dst
    x_dst = (np.arange(n_dst) + 0.5) * dx_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    return np.clip(h_dst, 0.0, 1.0)

def refinesolve(h, n, time_limit=100):
    t0 = time.time()
    target = n / 2.0
    h = project_sum(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.02
    n_steps = 3000
    n_smooth = 10
    for step in range(n_steps):

            break
        for _ in range(n_smooth):
            vals = compute_c5vals(h)

                else:
                    a = -m
                    na = n - a
                    grad[a:] += (1.0 - h[:na])

                        grad[:na] -= h[a:]
            grad *= (2.0 / n) / 5

            h_new = h - lr * grad
            h_new = project_sum(h_new, target)
