    # Phase 4: Random restarts with local optimization
    n_restarts = 0
    while rem() > 50 and n_restarts < 30:
        pert_scale = 0.03 + 0.04 * (n_restarts / 30.0)
        h_r = best_h + rng.normal(0, pert_scale, n)
        h_r = project_feasible(h_r, n)
        h_r = do_slsqp(h_r, n, 0.01, 150, t0, budget_s)
        c_r = compute_c5(h_r)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()
        n_restarts += 1

    # Phase 5: Try n=48 (even number for cleaner structure)
    if rem() > 120:
        n2 = 48
        # Upsample best_h to n2
        indices_old = np.linspace(0, n - 1, n)
        indices_new = np.linspace(0, n - 1, n2)
        h2 = np.interp(indices_new, indices_old, best_h)
        h2 = project_feasible(h2, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            # Optimize at n2
            h2 = do_slsqp(h2, n2, 0.02, 200, t0, budget_s)
            h2 = do_coord_descent(h2, n2, 30, t0, budget_s)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
                n = n2

    # Phase 6: Try bipolar random initializations
    if rem() > 80:
        for _ in range(5):
            if rem() < 40:
                break
            h_b = make_bipolar(n, rng)
            h_b = do_slsqp(h_b, n, 0.04, 200, t0, budget_s)
            c_b = compute_c5(h_b)
            if c_b < best_c5:
                best_c5 = c_b
                best_h = h_b.copy()
                h_b2, c_b2 = do_subgrad_loop(h_b.copy(), n, 1000, 0.03, 0.997, t0, budget_s)
                if c_b2 < best_c5:
                    best_c5 = c_b2
                    best_h = h_b2.copy()

    # Phase 7: Final coordinate descent
    if rem() > 15:
        best_h, best_c5 = do_coord_descent(best_h, n, 50, t0, budget_s)

    # Phase 8: One more subgradient pass
    if rem() > 20:
        h_sg2, c_sg2 = do_subgrad_loop(best_h.copy(), n, 2000, 0.01, 0.998, t0, budget_s)
        if c_sg2 < best_c5:
            best_c5 = c_sg2
            best_h = h_sg2.copy()

    # Final polish
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
