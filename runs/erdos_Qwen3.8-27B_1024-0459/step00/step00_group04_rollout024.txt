import time

def compute_c5_exact(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_smoothed(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    if temp < 1e-10:
        return max_c
    return max_c + temp * np.log(np.sum(np.exp((corr - max_c) / temp)))

def objective(h, n, dx, temp):
    return compute_c5_smoothed(h, dx, temp)

    # Numerical gradient

        h_plus[i] += eps
