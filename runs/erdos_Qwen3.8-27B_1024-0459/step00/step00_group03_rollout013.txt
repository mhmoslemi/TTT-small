import numpy as np
from scipy import optimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_max_lag(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    return max_idx - (n_points - 1), np.max(corr)

def grad_for_lag(h, lag, n):
    """Compute df_lag/dh where f_lag is the correlation at the given lag."""
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1 - 2 * h
    elif lag > 0:
        grad[lag:] += (1 - h[:n - lag])
        grad[:n - lag] += -h[lag:]
    else:
        m = -lag
        grad[:n - m] += (1 - h[m:])
        grad[m:] += -h[:n - m]
    return grad

def project_onto_feasible(h, n):
    """Project h onto {h: sum(h) = n/2, 0 <= h <= 1}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nfree = np.sum(free)
        if nfree == 0:
            # All at bounds; nudge slightly
            h = np.clip(h + diff / n, 0, 1)
            continue
        h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new, n_old):
    """Upsample h from length n_old to n_new."""
    if n_new == n_old:
        return h.copy()
    # Linear interpolation
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h, period=2)  # hmm, periodic?
    # Actually, let's just use regular interpolation (not periodic)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def downsample(h, n_new, n_old):
    """Downsample h from length n_old to n_new."""
    if n_new == n_old:
        return h.copy()
    # Average blocks
    block_size = n_old / n_new
    h_new = np.zeros(n_new)
    for i in range(n_new):
        start = int(i * block_size)
        end = int((i + 1) * block_size)
        h_new[i] = np.mean(h[start:end])
    h_new = np.clip(h_new, 0, 1)
    return h_new

def optimize_continuous(h, n, time_budget):
    """Projected gradient descent with adaptive step size."""
    start = time.time()
    dx = 2.0 / n
    h = project_onto_feasible(h.copy(), n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    alpha = 0.1
    no_improve_count = 0
    max_no_improve = 200
    stagnation_threshold = 1e-8
    
    while time.time() - start < time_budget:
        # Compute gradient at the max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        grad = grad_for_lag(h, lag, n) * dx
        
        # Gradient step
        h_new = h - alpha * grad
        h_new = project_onto_feasible(h_new, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve_count = 0
            alpha = min(alpha * 1.05, 1.0)
        else:
            h = h_new  # still move, even if not improvement globally
            if c5_new > best_c5 + stagnation_threshold:
                alpha *= 0.9
                no_improve_count += 1
            else:
                no_improve_count += 1
        
        if no_improve_count > max_no_improve:
            # Try a different approach: reset alpha and add noise
            alpha = 0.05
            noise = np.random.randn(n) * 0.01
            h = project_onto_feasible(h + noise, n)
            no_improve_count = 0
    
    return best_h, best_c5

def optimize_binary(h_init, n, time_budget):
    """Local search on binary sequences."""
    start = time.time()
    # Convert to binary
    h = (h_init >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    while np.sum(h) > n/2:
        # Find a 1 to flip to 0 that reduces objective
        ones = np.where(h == 1)[0]
        best_flip = None
        best_real = np.inf
        for i in ones:
            h[i] = 0
            real = compute_c5(h, n)
            if real < best_real:
                best_real = real
                best_flip = i
            h[i] = 1
        if best_flip is not None:
            h[best_flip] = 0
        else:
            h[ones[0]] = 0
    
    while np.sum(h) < n/2:
        zeros = np.where(h == 0)[0]
        best_flip = None
        best_real = np.inf
        for i in zeros:
            h[i] = 1
            real = compute_c5(h, n)
            if real < best_real:
                best_real = real
                best_flip = i
            h[i] = 0
        if best_flip is not None:
            h[best_flip] = 1
        else:
            h[zeros[0]] = 1
    
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    # Local search: flip pairs
    improvement = True
    while improvement and time.time() - start < time_budget:
        improvement = False
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        
        for i in ones:
            if time.time() - start > time_budget:
                break
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    improvement = True
                    # Restart from new state
                    ones = list(np.where(h == 1)[0])
                    zeros = list(np.where(h == 0)[0])
                    break
                else:
                    h[i] = 1
                    h[j] = 0
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
    
    return best_h, best_c5

import numpy as np
from scipy import optimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = free.sum()
        if nf == 0:
            h = np.clip(h + diff / n, 0, 1)
            continue
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag, n):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def optimize_pg(h_init, n, time_budget, rng):
    """Projected gradient descent."""
    start = time.time()
    h = project_feasible(h_init, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    alpha = 1.0
    stagnant = 0
    
    while time.time() - start < time_budget:
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        g = grad_lag(h, lag, n) * dx
        
        # Backtracking line search
        for _ in range(30):
            h_try = project_feasible(h - alpha * g, n)
            c5_try = compute_c5(h_try, n)
            c5_cur = compute_c5(h, n)
            if c5_try < c5_cur - 1e-12 * max(1, c5_cur):
                h = h_try
                break
            alpha *= 0.5
        
        c5_now = compute_c5(h, n)
        if c5_now < best_c5 - 1e-10:
            best_c5 = c5_now
            best_h = h.copy()
            stagnant = 0
            alpha = min(alpha * 1.1, 10.0)
        else:
            stagnant += 1
            alpha *= 0.7
        
        if stagnant > 100:
            # Restart with noise
            h = project_feasible(h + rng.randn(n) * 0.02, n)
            stagnant = 0
            alpha = 1.0
        if alpha < 1e-8:
            break
    
    return best_h, best_c5

def optimize_binary_local(h_init, n, time_budget, rng):
    """Local search on binary sequences."""
    start = time.time()
    h = (h_init >= 0.5).astype(float)
    
    # Balance
    while h.sum() > n / 2:
        ones = np.where(h == 1)[0]
        best_i, best_val = None, np.inf
        for i in ones:
            h[i] = 0
            v = compute_c5(h, n)
            if v < best_val:
                best_val = v
                best_i = i
            h[i] = 1
        if best_i is not None:
            h[best_i] = 0
        else:
            h[ones[rng.integers(len(ones))]] = 0
    
    while h.sum() < n / 2:
        zeros = np.where(h == 0)[0]
        best_j, best_val = None, np.inf
        for j in zeros:
            h[j] = 1
            v = compute_c5(h, n)
            if v < best_val:
                best_val = v
                best_j = j
            h[j] = 0
        if best_j is not None:
            h[best_j] = 1
        else:
            h[zeros[rng.integers(len(zeros))]] = 1
    
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    # Hill climbing: flip pairs
    no_improve = 0
    while time.time() - start < time_budget and no_improve < 50:
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        improved = False
        
        # Random order for diversity
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in ones:
            if time.time() - start > time_budget:
                break
            for j in zeros:
                h[i] = 0
                h[j] = 1
                v = compute_c5(h, n)
                if v < best_c5 - 1e-12:
                    best_c5 = v
                    best_h = h.copy()
                    improved = True
                    ones = list(np.where(h == 1)[0])
                    zeros = list(np.where(h == 0)[0])
                    rng.shuffle(ones)
                    rng.shuffle(zeros)
                    break
                else:
                    h[i] = 1
                    h[j] = 0
        
        if not improved:
            no_improve += 1
            if no_improve < 5:
                # Random perturbation
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                    h[i] = 0
                    h[j] = 1
                    best_h = h.copy()
                    best_c5 = compute_c5(h, n)
                    no_improve = 0
                else:
                    break
    
    return best_h, best_c5
