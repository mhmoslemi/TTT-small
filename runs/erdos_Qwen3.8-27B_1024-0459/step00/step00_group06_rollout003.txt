def project_sum(x, target):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    target = float(target)
    n = x.size
    if n == 0: return x
    if target <= 0.0: return np.zeros(n)
    if target >= n: return np.ones(n)
    cur = float(x.sum())
    if abs(cur - target) <= 1e-12:
        return x
    # quick guess tau = (cur-target)/n
    tau = (cur - target) / n
    if -1.5 < tau < 1.5:
        y = np.clip(x - tau, 0.0, 1.0)
        if abs(float(y.sum()) - target) <= 1e-10:
            return y
    lo = -1.0; hi = 1.0
    # function sum(clip(x - mid)) decreasing in mid
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = 0.5*(lo+hi)
    return np.clip(x - mid, 0.0, 1.0)

def project_sum(x, target):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    cur = float(x.sum())
    if abs(cur - target) <= 1e-12:
        return x

    tau_guess = (cur - target) / float(n)
    if -1.5 < tau_guess < 1.5:
        y = np.clip(x - tau_guess, 0.0, 1.0)
        if abs(float(y.sum()) - target) <= 1e-10:
            return y

    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    return np.clip(x - mid, 0.0, 1.0)


def resample_h(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    n_old = h_old.size
    n_new = int(n_new)
    if n_old == n_new:
        return h_old.copy()
    if n_old == 0:
        return np.zeros(n_new, dtype=float)
    if n_old == 1:
        return np.full(n_new, float(h_old[0]), dtype=float)
    old_x = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / float(n_old))
    new_x = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    return np.interp(new_x, old_x, h_old, left=float(h_old[0]), right=float(h_old[-1]))


def c5_value(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0 or not np.all(np.isfinite(h)):
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    if m < 0.0 and m > -1e-12:
        m = 0.0
    return m * dx


def compute_O_values(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    acorr = np.correlate(h, h, mode="full")
    pos = np.empty(n, dtype=float)
    neg = np.empty(n, dtype=float)

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]

    ab0 = float(acorr[n - 1])
    v0 = dx * (total - ab0)
    if v0 < 0.0 and v0 > -1e-12:
        v0 = 0.0
    pos[0] = v0
    neg[0] = v0

    for e in range(1, n):
        ab = float(acorr[n - 1 + e])
        sa = float(prefix[n - e])
        sb = float(total - prefix[e])
        vp = dx * (sa - ab)
        vn = dx * (sb - ab)
        if vp < 0.0 and vp > -1e-12:
            vp = 0.0
        if vn < 0.0 and vn > -1e-12:
            vn = 0.0
        pos[e] = vp
        neg[e] = vn
    return pos, neg


def soft_value(h, alpha):
    h = np.asarray(h, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    n = h.size
    if n == 0:
        return 0.0
    pos, neg = compute_O_values(h)
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = pos[0]
    vals[1:n] = pos[1:]
    vals[n:] = neg[1:]
    if alpha <= 0.0:
        return float(np.mean(vals))
    m = float(vals.max())
    ex = np.exp(alpha * (vals - m))
    s = float(ex.sum())
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + np.log(s) / alpha)


def soft_gradient(h, alpha):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0 or not np.all(np.isfinite(h)):
        return g

    pos, neg = compute_O_values(h)
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = pos[0]
    vals[1:n] = pos[1:]
    vals[n:] = neg[1:]

    if alpha <= 0.0:
        w = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=float)
    else:
        m = float(vals.max())
        ex = np.exp(alpha * (vals - m))
        s = float(ex.sum())
        if s <= 0.0 or not np.isfinite(s):
            w = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=float)
        else:
            w = ex / s

    dx = 2.0 / float(n)
    g += (w[0] * dx) * (1.0 - 2.0 * h)

    wp = w[1:n]
    wn = w[n:2 * n - 1]
    for e in range(1, n):
        a = h[:-e]
        b = h[e:]

        cp = wp[e - 1] * dx
        if cp != 0.0:
            g[:-e] += cp * (1.0 - b)
            g[e:] -= cp * a

        cm = wn[e - 1] * dx
        if cm != 0.0:
            g[e:] += cm * (1.0 - a)
            g[:-e] -= cm * b
    return g


def soft_f_and_g(h, alpha):

    n = h.size
    if not np.all(np.isfinite(h)):
        return 1e100, np.zeros(n, dtype=float)
    pos, neg = compute_O_values(h)
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = pos[0]
    vals[1:n] = pos[1:]
    vals[n:] = neg[1:]

    if alpha <= 0.0:
        return float(np.mean(vals)), np.zeros(n, dtype=float)

    m = float(vals.max())
    ex = np.exp(alpha * (vals - m))
    s = float(ex.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.full(2 * n - 1, 1.0 / float(2 * n - 1), dtype=float)
        f = m
    else:
        w = ex / s
        f = m + np.log(s) / alpha

    g = (w[0] * dx) * (1.0 - 2.0 * h)

    wp = w[1:n]
    wn = w[n:2 * n - 1]
    for e in range(1, n):
        a = h[:-e]
        b = h[e:]

        cm = wn[e - 1] * dx
        if cm != 0.0:

def lbfgs_objective(x):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    if not np.all(np.isfinite(h)):
        return 1e100
    f = soft_value(h, _SOFT_ALPHA)
    err = float(np.sum(h) - _PEN_TARGET)
    return float(f + 0.5 * _PENALTY_MU * err * err)


def lbfgs_gradient(x):
    h = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    if not np.all(np.isfinite(h)):
        return np.zeros_like(h)
    g = soft_gradient(h, _SOFT_ALPHA)
    err = float(np.sum(h) - _PEN_TARGET)
    g += _PENALTY_MU * err
    return g


def random_start(n, kind, rng, parent):

    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))

    if kind == 0 and parent is not None:
        return resample_h(parent, n)

    if kind == 1:
        h = np.full(n, 0.5, dtype=float)
        kmax = max(1, min(8, n // 4))
        for m in range(1, kmax + 1):
            if rng.random() < 0.75:
                amp = rng.uniform(0.05, 0.45) / np.sqrt(float(m))
                phase = rng.uniform(0.0, 2.0 * np.pi)
                h += amp * np.cos(float(m) * np.pi * x + phase)
        return h

    if kind == 2:
        low = 3
        high = max(4, n // 6 + 1)
        nb = int(rng.integers(low, high))
        block = rng.random(nb)
        xb = (np.arange(nb, dtype=float) + 0.5) * (2.0 / float(nb))
        h = np.interp(x, xb, block, left=float(block[0]), right=float(block[-1]))
        return h

    if kind == 3:
        amp = rng.uniform(0.18, 0.46)
        f = int(rng.integers(1, max(2, 7)))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h = 0.5 + amp * np.sin(float(f) * np.pi * x + phase)
        h = h + rng.normal(0.0, 0.08, n)
        return h

    if kind == 4:
        h = np.zeros(n, dtype=float)
        k = int(round(0.5 * float(n)))
        k = min(max(k, 0), n)
        if k > 0:
            idx = rng.choice(n, size=k, replace=False)
            h[idx] = 1.0
        wlen = min(5, max(1, n))
        if wlen > 1:
            w = np.ones(wlen, dtype=float)
            w /= float(w.size)
            h = np.convolve(h, w, mode="same")
        return h

    h = 0.5 + 0.32 * np.sin(np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
    return h

def projected_gradient_polish(h, n, limit, alpha, max_iter):

    step = 0.12
    for _ in range(max_iter):
        if time.perf_counter() > limit - 1.0:
            break
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-11 or not np.isfinite(gnorm):
            break
        direction = -g / gnorm

        accepted = False

            continue

            best_h = h.copy()

    c = c5_value(h)

        best_c = c5_value(best_h)

    global_cands = []
    uid = 0
    gen_deadline = start + min(60.0, max(5.0, total * 0.055))

    for n in n_list:
        if time.perf_counter() > gen_deadline:
            break

        local = []

        if parent is not None:
            hp = resample_h(parent, n)
            c, hproj = add_local_candidate(local, hp, n, 6)
            if c is not None and np.isfinite(c) and c < best_c:
                best_c = float(c)
                best_h = hproj.copy()
                best_n = int(n)
            hm = 0.5 * (hp + hp[::-1])
            c, hproj = add_local_candidate(local, hm, n, 6)
            if c is not None and np.isfinite(c) and c < best_c:
                best_c = float(c)
                best_h = hproj.copy()
                best_n = int(n)

        if best_h is not None:
            hb = resample_h(best_h, n) if int(best_h.size) != int(n) else best_h.copy()
            c, hproj = add_local_candidate(local, hb, n, 6)
            if c is not None and np.isfinite(c) and c < best_c:
                best_c = float(c)
                best_h = hproj.copy()
                best_n = int(n)

        if n <= 128:
            nrand = 50
        elif n <= 200:
            nrand = 30
        elif n <= 288:
            nrand = 16
        else:
            nrand = 10

        scale = max(0.2, min(1.0, total / 800.0))
        nrand = max(4, int(nrand * scale))

        for _ in range(nrand):
            if time.perf_counter() > gen_deadline:
                break
            kind = int(rng.integers(0, 5))
            h = random_start(n, kind, rng, parent)
            c, hproj = add_local_candidate(local, h, n, 6)
            if c is not None and np.isfinite(c) and c < best_c:
                best_c = float(c)
                best_h = hproj.copy()
                best_n = int(n)

        if best_h is not None:
            hb = resample_h(best_h, n) if int(best_h.size) != int(n) else best_h.copy()
            c, hproj = add_local_candidate(local, hb, n, 6)
            if c is not None and np.isfinite(c) and c < best_c:
                best_c = float(c)
                best_h = hproj.copy()
                best_n = int(n)

        for cc, hh in local:
            global_cands.append((float(cc), uid, int(n), hh))
            uid += 1

    if best_h is None:
        best_n = 96
        best_h = project_sum(np.full(best_n, 0.5, dtype=float), 0.5 * float(best_n))
        best_c = c5_value(best_h)

    global_cands.sort()

    selected = []
    selected_uids = []
    selected_ns = []
    max_selected = 14

    for cand in global_cands:
        if len(selected) >= 6:
            break
        cval, cand_uid, cn, ch = cand
        if cand_uid in selected_uids:
            continue
        selected.append(cand)
        selected_uids.append(cand_uid)

    for cand in global_cands:
        if len(selected) >= max_selected:
            break
        cval, cand_uid, cn, ch = cand
        if cand_uid in selected_uids:
            continue
        already = False
        for ns in selected_ns:
            if ns == cn:
                already = already or False
                already = True
                break
        if already:
            continue
        selected.append(cand)
        selected_uids.append(cand_uid)
        selected_ns.append(cn)

    reserve = 90.0 if int(best_n) <= 220 else 45.0

    rank = 0
    for cand in selected:
        rem = deadline - time.perf_counter()
        if rem < reserve + 25.0:
            break

        c0, cand_uid, cn, ch = cand
        rank += 1

        if rank == 1:
            base = 300.0
        elif rank == 2:
            base = 220.0
        elif rank == 3:
            base = 170.0
        elif rank == 4:
            base = 145.0
        else:
            base = 120.0

        if int(cn) > 288:
            base *= 0.65
        elif int(cn) > 200:
            base *= 0.8

        alloc = min(base, max(18.0, rem - reserve - 10.0))
        if alloc < 18.0:
            break

        limit = time.perf_counter() + alloc
        if limit > deadline - 10.0:
            limit = deadline - 10.0

        if alloc > 210.0:
            max_stages = 5
        elif alloc > 130.0:
            max_stages = 4
        elif alloc > 75.0:
            max_stages = 3
        else:
            max_stages = 2

        try:
            bh, bc, _last = optimize_candidate(ch, cn, limit, rng, max_stages)
        except Exception:
            bh = project_sum(ch, 0.5 * float(cn))
            bc = c5_value(bh)

        if np.isfinite(bc) and bc < best_c - 1e-14:
            best_c = float(bc)
            best_h = bh.copy()
            best_n = int(cn)

    attempt = 0
    while attempt < 4:
        rem = deadline - time.perf_counter()
        if rem < reserve + 80.0:
            break
        attempt += 1

        n_use = int(best_n)
        if n_use <= 0:
            n_use = 96

        kind = int(rng.integers(0, 5))
        h = random_start(n_use, kind, rng, parent)
        c, hproj = add_local_candidate([], h, n_use, 1)
        if c is None or not np.isfinite(c):
            continue

        alloc = min(260.0, max(30.0, rem - reserve - 20.0))
        limit = time.perf_counter() + alloc
        if limit > deadline - 12.0:
            limit = deadline - 12.0

        if alloc > 200.0:
            max_stages = 5
        elif alloc > 120.0:
            max_stages = 4
        else:
            max_stages = 3

        try:
            bh, bc, _last = optimize_candidate(hproj, n_use, limit, rng, max_stages)
            if np.isfinite(bc) and bc < best_c - 1e-14:
                best_c = float(bc)
                best_h = bh.copy()
                best_n = int(n_use)
        except Exception:
            pass

    rem = deadline - time.perf_counter()
    for alpha in (100.0, 300.0, 800.0):
        rem = deadline - time.perf_counter()
        if rem < 18.0:
            break
        try:
            _final, ph, pc = projected_gradient_polish(
                best_h, best_n, deadline - 1.0, alpha, 22
            )
            if np.isfinite(pc) and pc < best_c - 1e-14:
                best_c = float(pc)
                best_h = ph.copy()
        except Exception:
            break

    best_h = project_sum(best_h, 0.5 * float(best_n))
    best_c = c5_value(best_h)
    if not np.isfinite(best_c):
        best_n = 96
        best_h = project_sum(np.full(best_n, 0.5, dtype=float), 0.5 * float(best_n))
        best_c = c5_value(best_h)

    try:
        ev = float(evaluate_erdos_solution(best_h))
        if np.isfinite(ev):
            best_c = min(best_c, ev)
    except Exception:
        pass

    return best_h.tolist(), float(best_c), int(best_n)
