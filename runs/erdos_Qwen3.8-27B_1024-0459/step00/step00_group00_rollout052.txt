def obj_func(h, n, pen_w, beta):
    h_c = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h_c, 1.0 - h_c, mode='full')
    if beta < 1e6:
        max_c = np.max(corr)
        shifted = beta * (corr - max_c)
        log_sum = max_c + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
        c5_smooth = log_sum * dx
    else:
        c5_smooth = np.max(corr) * dx
    s = float(np.sum(h_c)) - n / 2.0
    return c5_smooth + pen_w * s * s

def optimize_at_n(n, h_starts, pen_w, beta, time_limit):
    best_h = None
    best_score = 1e10
    bounds = [(0.0, 1.0)] * n
    start_t = time.time()
    for h0 in h_starts:
        if time.time() - start_t > time_limit:
            break
        h0 = project_feasible(h0.copy(), n)
        res = minimize(obj_func, h0, args=(n, pen_w, beta),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8})
        if res.x is not None:
            h_sol = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            score = compute_c5(h_sol)
            if score < best_score:
                best_score = score
                best_h = h_sol.copy()
    return best_h, best_score

def projected_grad_step(h, n, step_size, beta):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    if beta < 1e6:
        max_c = np.max(corr)
        weights = np.exp(beta * (corr - max_c))
        weights /= np.sum(weights)
    else:
        weights = np.zeros(2 * n - 1)
        max_val = np.max(corr)
        mask = np.isclose(corr, max_val, atol=1e-10)
        weights[mask] = 1.0 / np.sum(mask)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if weights[idx] < 1e-12:
            continue
        lag = idx - (n - 1)
        w = weights[idx]
        if lag >= 0:
            for j in range(n - lag):
                grad[j] += w * (1.0 - h[j + lag])
                if j >= lag:
                    grad[j] -= w * h[j - lag]
        else:
            m = -lag
            for j in range(m, n):
                grad[j] += w * (1.0 - h[j - m])
            for j in range(n - m):
                grad[j] -= w * h[j + m]
    grad *= dx
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

def refine_projected(h, n, n_iters, step_size, beta):
    h = h.copy()
    for _ in range(n_iters):
        h = projected_grad_step(h, n, step_size, beta)
    return h

def make_starts(n, parent, rng):
    starts = []
    if parent is not None:
        h0 = interp_periodic(parent, n)
        h0 = project_feasible(h0, n)
        starts.append(h0)
    if n % 2 == 0:
        h0 = np.full(n, 0.5)
        starts.append(h0)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h_sin = 0.5 + 0.35 * np.sin(np.pi * x)
    starts.append(project_feasible(h_sin, n))
    h_cos = 0.5 + 0.35 * np.cos(np.pi * x)
    starts.append(project_feasible(h_cos, n))
    h0 = np.zeros(n)
    w = max(1, n // 4)
    c = n // 3
    h0[max(0, c - w // 2):min(n, c + w // 2 + 1)] = 1.0
    starts.append(project_feasible(h0, n))
    h_rand = np.clip(rng.randn(n) * 0.2 + 0.5, 0.0, 1.0)
    starts.append(project_feasible(h_rand, n))
    if parent is not None:
        h_pert = interp_periodic(parent, n).copy()
        h_pert += rng.randn(n) * 0.05
        starts.append(project_feasible(h_pert, n))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    try:
        parent = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError, TypeError):
        parent = None
    best_h = None
    best_score = 1e10
    best_n = 45
    if parent is not None:
        best_h = parent.copy()
        best_score = compute_c5(best_h)
        best_n = len(best_h)
    resolutions = [45, 90, 180, 360]
    penalty_seq = [1e4, 1e5, 1e6]
    beta_seq = [50.0, 100.0, 1e8]
    for n in resolutions:
        if time.time() - t_start > budget_s - 90:
            break
        allocated = min(300, max(30, int((budget_s - 90) * (1.0 / max(1, len(resolutions) - resolutions.index(n)))))
        starts = make_starts(n, parent, rng)
        for pen_w, beta in zip(penalty_seq, beta_seq):
            if time.time() - t_start > budget_s - 60:
                break
            h_opt, score = optimize_at_n(n, starts, pen_w, beta, time_limit=allocated)
            if h_opt is not None and score < best_score:
                best_score = score
                best_h = h_opt.copy()
                best_n = n
                starts = [h_opt] + starts
    if best_h is not None:
        n_best = len(best_h)
        for beta in [50.0, 100.0]:
            h_ref = refine_projected(best_h, n_best, 200, 0.01, beta)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
    if best_h is None:
        best_h = np.full(45, 0.5)
        best_score = compute_c5(best_h)
        best_n = 45
    return (best_h, best_score, len(best_h))
