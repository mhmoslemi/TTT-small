import time
import math
import numpy as np

_CACHE = {}


def _next_pow2(v):
    p = 1
    while p < v:
        p <<= 1
    return p


def get_fft_size(n):
    return _next_pow2(max(1, 2 * int(n)))


def get_lag_arrays(n, m):
    key = (n, m)
    if key in _CACHE:
        return _CACHE[key]
    if n > 1:
        d = np.arange(1, n, dtype=np.intp)
        md = m - d
    else:
        d = np.empty(0, dtype=np.intp)
        md = np.empty(0, dtype=np.intp)
    _CACHE[key] = (d, md)
    return d, md


def get_q_parts(h, n, m=None):
    if m is None:
        m = get_fft_size(n)
    H = np.fft.rfft(h, n=m)
    ac = np.fft.irfft(H * np.conj(H), n=m)[:n]

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)

    q0 = float(pref[n] - ac[0])
    if n > 1:
        pos = pref[n - 1:0:-1] - ac[1:]
        neg = (pref[n] - pref[1:]) - ac[1:]
        return H, q0, pos, neg
    return H, q0, None, None


def compute_qmax(h, n, m=None):
    _, q0, pos, neg = get_q_parts(h, n, m)
    best = q0
    if pos is not None:
        mp = float(np.max(pos))
        mn = float(np.max(neg))
        if mp > best:
            best = mp
        if mn > best:
            best = mn
    return float(best)


def eval_f_only(h, n, tau, m):
    _, q0, pos, neg = get_q_parts(h, n, m)
    mval = q0
    if pos is not None:
        if float(np.max(pos)) > mval:
            mval = float(np.max(pos))
        if float(np.max(neg)) > mval:
            mval = float(np.max(neg))

        pos = pos - mval
        neg = neg - mval
        s_pos = float(np.exp(pos, out=pos).sum())
        s_neg = float(np.exp(neg, out=neg).sum())
        e0 = math.exp(q0 - mval)
        Z = e0 + s_pos + s_neg
    else:
        Z = 1.0

    f = mval + tau * math.log(Z)
    return f, float(mval)


def eval_f_grad(h, n, tau, m, d, md):
    H, q0, pos, neg = get_q_parts(h, n, m)

    mval = q0
    if pos is not None:
        if float(np.max(pos)) > mval:
            mval = float(np.max(pos))
        if float(np.max(neg)) > mval:
            mval = float(np.max(neg))

    e0 = math.exp(q0 - mval)
    P = np.zeros(n, dtype=float)
    N = np.zeros(n, dtype=float)
    M = np.zeros(n, dtype=float)

    if pos is not None:
        pos = pos - mval
        neg = neg - mval
        s_pos = float(np.exp(pos, out=pos).sum())
        s_neg = float(np.exp(neg, out=neg).sum())
        Z = e0 + s_pos + s_neg
        p0 = e0 / Z
        P[1:] = pos / Z
        N[1:] = neg / Z
        M[1:] = P[1:] + N[1:]
    else:
        Z = 1.0
        p0 = 1.0

    f = mval + tau * math.log(Z)

    cumP = np.cumsum(P)
    idxR = n - 1 - np.arange(n)
    R = cumP[idxR]

    cumN = np.cumsum(N)
    L = cumN

    first = p0 + R + L

    if n > 1:
        k = np.zeros(m, dtype=float)
        k[d] = M[1:]
        k[md] = M[1:]
        product = np.fft.irfft(H * np.fft.rfft(k, n=m), n=m)[:n]
    else:
        product = np.zeros(1, dtype=float)

    grad = first - product - 2.0 * p0 * h
    return f, float(mval), grad


def fix_sum(x, S):
    x = np.array(x, dtype=float, copy=True)
    S = float(S)

    for _ in range(4):
        res = S - float(x.sum())
        if abs(res) < 1e-13:
            break

        if res > 0.0:
            room = 1.0 - x
            tot = float(room.sum())
            if tot <= 0.0:
                break
            if tot <= res * (1.0 + 1e-12):
                x = np.minimum(1.0, x + res / tot)
            else:
                x = x + res * (room / tot)
        else:
            mass = x
            tot = float(mass.sum())
            if tot <= 0.0:
                break
            if tot <= (-res) * (1.0 + 1e-12):
                x = np.maximum(0.0, x + res / tot)
            else:
                x = x + res * (mass / tot)

        x = np.clip(x, 0.0, 1.0)

    res = S - float(x.sum())
    if abs(res) > 1e-13:
        if res > 0.0:
            idx = int(np.argmax(1.0 - x))
            x[idx] = float(np.clip(x[idx] + res, 0.0, 1.0))
        else:
            idx = int(np.argmin(x))
            x[idx] = float(np.clip(x[idx] + res, 0.0, 1.0))
    return x


def project_box_sum(a, S):
    a = np.asarray(a, dtype=float)
    n = int(a.size)
    S = float(S)
    if n == 0:
        return a

    a0 = np.clip(a, 0.0, 1.0)
    s0 = float(a0.sum())

    if abs(s0 - S) <= 1e-12:
        return fix_sum(a0, S)

    eps = 1e-12

    if s0 > S:
        c = np.sort(a0)[::-1]
        pref = np.concatenate(([0.0], np.cumsum(c)))
        k = np.arange(1, n + 1)
        theta = (pref[k] - S) / k

        low = np.empty(n, dtype=float)
        if n > 1:
            low[:-1] = c[1:]
            low[-1] = 0.0
        else:
            low[0] = 0.0
        high = c

        valid = (theta >= low - eps) & (theta <= high + eps) & (theta >= -eps)
        idxs = np.nonzero(valid)[0]
        if idxs.size > 0:
            theta = float(theta[idxs[0]])
            x = np.clip(a0 - theta, 0.0, 1.0)
        else:
            lo = 0.0
            hi = float(max(0.0, c[0]))
            for _ in range(60):
                mid = 0.5 * (lo + hi)
                val = float(np.sum(np.clip(a0 - mid, 0.0, None)))
                if val > S:
                    lo = mid
                else:
                    hi = mid
            theta = 0.5 * (lo + hi)
            x = np.clip(a0 - theta, 0.0, 1.0)
    else:
        b = np.sort(a0)
        pref = np.concatenate(([0.0], np.cumsum(b)))
        k = np.arange(1, n + 1)
        u = (pref[k] + float(n) - S) / k

        low = b
        high = np.empty(n, dtype=float)
        if n > 1:
            high[:-1] = b[1:]
            high[-1] = np.inf
        else:
            high[0] = np.inf

        valid = (u >= low - eps) & (u <= high + eps) & (u >= -eps) & (u <= 1.0 + eps)
        idxs = np.nonzero(valid)[0]
        if idxs.size > 0:
            u = float(u[idxs[0]])
            phi = 1.0 - u
            x = np.clip(a0 + phi, 0.0, 1.0)
        else:
            lo = 0.0
            hi = 1.0
            for _ in range(60):
                mid = 0.5 * (lo + hi)
                val = float(np.sum(np.clip(a0 + mid, 0.0, 1.0)))
                if val < S:
                    lo = mid
                else:
                    hi = mid
            phi = 0.5 * (lo + hi)
            x = np.clip(a0 + phi, 0.0, 1.0)

    return fix_sum(x, S)


def linear_min_unit_sum(g, target_count):
    g = np.asarray(g, dtype=float)
    n = int(g.size)
    target_count = int(round(float(target_count)))

    if target_count <= 0:
        return np.zeros(n, dtype=float)
    if target_count >= n:
        return np.ones(n, dtype=float)

    idx = np.argpartition(g, target_count - 1)[:target_count]
    y = np.zeros(n, dtype=float)
    y[idx] = 1.0
    return y


def upsample(h_old, n_new):
    h_old = np.asarray(h_old, dtype=float)
    old_n = int(h_old.size)
    n_new = int(n_new)

    if old_n == 0 or n_new == old_n:
        h = np.asarray(h_old, dtype=float)
        if h.size != n_new:
            return h
        return fix_sum(np.clip(h, 0.0, 1.0), 0.5 * n_new)

    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h = np.interp(x_new, x_old, h_old)
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, 0.5 * n_new)
    return h


def make_random_h(n, rng):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n, dtype=float)

    for mode in range(1, 7):
        amp = rng.uniform(0.05, 0.28) / math.sqrt(float(mode))
        phase = rng.uniform(0.0, 2.0 * math.pi)
        h += amp * np.sin(math.pi * float(mode) * x + phase)

    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, 0.5 * n)
    return h


def make_smooth_binary_h(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    count = n // 2
    if count > 0:
        idx = rng.choice(n, size=count, replace=False)
        h[idx] = 1.0

    ks = min(7, n)
    if ks % 2 == 0:
        ks -= 1
    if ks < 1:
        ks = 1
    kernel = np.ones(ks, dtype=float) / float(ks)
    h = np.convolve(h, kernel, mode="same")[:n]

    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, 0.5 * n)
    return h


def compute_exact_c5(h, n=None):
    h = np.asarray(h, dtype=float)
    if n is None:
        n = int(h.size)
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return max(0.0, float(np.max(c) * dx))


def optimize_stage(h, n, stage_deadline, rng):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if n % 2 == 1:
        h = upsample(h, n + 1)
        n = int(h.size)

    S = 0.5 * float(n)
    count = int(round(S))
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, S)

    m = get_fft_size(n)
    d, md = get_lag_arrays(n, m)
    invdx = 0.5 / float(n)

    qmax0 = compute_qmax(h, n, m)
    best_h = h.copy()
    best_c5 = qmax0 * invdx

    tau = max(1.0, qmax0 / 20.0)
    tau_min = max(0.05, qmax0 / 512.0)
    taus = []
    t = tau
    while t > tau_min:
        taus.append(float(t))
        t *= 0.5
    taus.append(float(tau_min))
    if len(taus) > 12:
        taus = taus[:12]

    last_lam = 0.5
    perturb_budget = 8

    for tau in taus:
        if time.time() >= stage_deadline - 0.5:
            break

        no_accept = 0
        no_improve = 0
        iter_count = 0

        while time.time() < stage_deadline - 0.5:
            f, maxq, grad = eval_f_grad(h, n, tau, m, d, md)
            curr_c5 = maxq * invdx

            if curr_c5 < best_c5 - 1e-8:
                best_c5 = curr_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 350 and tau > taus[-1] * 1.01:
                break

            y = linear_min_unit_sum(grad, count)
            dvec = y - h
            nrm2 = float(np.dot(dvec, dvec))

            if nrm2 < 1e-20:
                if perturb_budget > 0:
                    g2 = grad + rng.standard_normal(n) * 1e-6
                    y = linear_min_unit_sum(g2, count)
                    dvec = y - h
                    nrm2 = float(np.dot(dvec, dvec))
                if nrm2 < 1e-20:
                    break

            dotg = float(np.dot(grad, dvec))

            if dotg >= -1e-13:
                if perturb_budget > 0 and count > 0:
                    idx = rng.choice(n, size=count, replace=False)
                    yr = np.zeros(n, dtype=float)
                    yr[idx] = 1.0
                    dvec = yr - h
                    nrm2 = float(np.dot(dvec, dvec))
                    dotg = float(np.dot(grad, dvec))
                if dotg >= -1e-13 or nrm2 < 1e-20:
                    break

            cands = []
            if last_lam > 0.0:
                cands.append(last_lam)
            cands.extend([1.0, 0.55, 0.25, 0.08, 0.02, 0.005])

            seen = set()
            uniq = []
            for lam in cands:
                if lam <= 0.0 or lam > 1.0:
                    continue
                key = round(float(lam), 6)
                if key not in seen:
                    seen.add(key)
                    uniq.append(float(lam))

            accepted = False
            for lam in uniq:
                if nrm2 * lam * lam < 1e-26:
                    continue

                ht = h + lam * dvec
                if ht.max() > 1.0 + 1e-9 or ht.min() < -1e-9:
                    ht = np.clip(ht, 0.0, 1.0)

                ft, maxq_t = eval_f_only(ht, n, tau, m)
                test_c5 = maxq_t * invdx

                if dotg < 0.0:
                    armijo = f + 1e-4 * dotg * lam
                else:
                    armijo = f - 1e-12

                if ft <= armijo or test_c5 < curr_c5 - 1e-8:
                    h = ht
                    if abs(h.sum() - S) > 1e-7:
                        h = fix_sum(h, S)

                    if test_c5 < best_c5 - 1e-8:
                        best_c5 = test_c5
                        best_h = h.copy()
                        no_improve = 0

                    accepted = True
                    last_lam = lam
                    break

            if not accepted:
                no_accept += 1
                if no_accept >= 3:
                    if perturb_budget > 0:
                        idx = rng.choice(n, size=count, replace=False)
                        yr = np.zeros(n, dtype=float)
                        yr[idx] = 1.0
                        h = 0.88 * h + 0.12 * yr
                        h = project_box_sum(h, S)
                        perturb_budget -= 1
                    no_accept = 0
                    if perturb_budget < 0:

            else:

                if last_lam < 1.0:

    c_h = compute_exact_c5(h, n)
    if c_h < best_c5:
        best_c5 = c_h
        best_h = h.copy()

    c_best = compute_exact_c5(best_h, n)

            h_rand = make_smooth_binary_h(n_final, rng)

        rem = deadline - time.time()
        stage_deadline = min(deadline - 2.0, time.time() + max(25.0, 0.45 * rem))
        if stage_deadline > time.time() + 10.0:
            h_st, c_st = optimize_stage(h_rand, n_final, stage_deadline, rng)
            c_st = compute_exact_c5(h_st, int(h_st.size))
            if c_st < best_c5:
                best_h = h_st
                best_c5 = c_st

    best_h = np.asarray(best_h, dtype=float)
    n_best = int(best_h.size)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, 0.5 * n_best)
    c5_bound = compute_exact_c5(best_h, n_best)

    return best_h, float(c5_bound), n_best
