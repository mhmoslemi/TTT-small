    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = tau * (corr - m)

    return float((m + np.log(np.sum(np.exp(s))) / tau) * dx)

def make_feasible(h, n):
    """Project h to be feasible: sum(h) = n/2, 0 <= h <= 1."""
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            # Need to increase some h values
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    # Final exact adjustment
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:

        for i in range(n):
            if 0 < h[i] < 1:
                h[i] += diff
                h[i] = np.clip(h[i], 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, tau=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    if m < 1e-10:
        return 0.0
    shifted = corr - m
    shifted = np.clip(shifted / m, -5.0, 0.0)
    return float((m + np.log(np.sum(np.exp(tau * shifted))) / tau) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h
