            for _ in range(rng.integers(3, 8)):

                b = zeros[rng.integers(len(zeros))]
                h[a], h[b] = h[b], h[a]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            current_c5 = compute_c5(h)
            stall = 0
    
    return best_h, best_c5

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    global_start = time.time()
    deadline = global_start + budget_s * 0.92
    rng = np.random.default_rng(seed)
    
    # Get initial values
    h_init = None
    n_init = 70
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
    
    if h_init is None:
        h_init = np.zeros(n_init)
        h_init[::2] = 1.0
        h_init = project_feasible(h_init, n_init)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n_init
    
    # Phase 1: Binary local search at initial resolution
    t1_deadline = global_start + budget_s * 0.20
    h_bin, c5_bin = binary_local_search(h_init, n_init, rng, t1_deadline)
    if c5_bin < best_c5:
        best_h = h_bin
        best_c5 = c5_bin
        best_n = n_init
    # Also try from binarized best
    # Phase 2: Continuous local search at initial resolution
    t2_deadline = global_start + budget_s * 0.45
    h_cont, c5_cont = local_search_cont(h_init, n_init, rng, t2_deadline)
    if c5_cont < best_c5:
        best_h = h_cont
        best_c5 = c5_cont
        best_n = n_init
    
    # Phase 3: SA at initial resolution
    if time.time() < deadline:
        t3_deadline = min(global_start + budget_s * 0.55, deadline)
        h_sa, c5_sa = simulated_annealing(best_h, best_n, rng, t3_deadline)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa
            best_n = n_init
    
    # Phase 4: Try higher resolution
    if time.time() < deadline:
        n2 = best_n * 2
        h2 = upsample(best_h, best_n, n2)
        t4_deadline = min(global_start + budget_s * 0.72, deadline)
        h2, c5_2 = local_search_cont(h2, n2, rng, t4_deadline)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
            best_n = n2
        # Binary at higher res
        if time.time() < deadline:
            t4b_deadline = min(global_start + budget_s * 0.82, deadline)
            h2b, c5_2b = binary_local_search(h2, n2, rng, t4b_deadline)
            if c5_2b < best_c5:
                best_h = h2b
                best_c5 = c5_2b
                best_n = n2
    
    # Phase 5: Final refinement at best resolution
    if time.time() < deadline:
        t5_deadline = deadline
        h_ref, c5_ref = local_search_cont(best_h, best_n, rng, t5_deadline)
        if c5_ref < best_c5:
            best_h = h_ref
            best_c5 = c5_ref
            best_n = best_n
    
    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
