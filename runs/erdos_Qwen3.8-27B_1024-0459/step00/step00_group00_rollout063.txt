import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def enforce_constraints(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_jac_single(h, n, j):
    k = j - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        idx1 = np.arange(k, n)
        grad[idx1] += (1.0 - h[idx1 - k])
        idx2 = np.arange(0, n - k)
        grad[idx2] -= h[idx2 + k]
    else:
        kp = -k
        idx1 = np.arange(0, n - kp)
        grad[idx1] += (1.0 - h[idx1 + kp])
        idx2 = np.arange(kp, n)
        grad[idx2] -= h[idx2 - kp]
    return grad

def smooth_obj_grad(h, n, alpha, pen):
    vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = vals.max()
    shifted = vals - m
    exp_vals = np.exp(shifted)
    logsumexp = m + np.log(exp_vals.sum()) / alpha
    weights = exp_vals / exp_vals.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    thresh = 1e-10
    active = np.where(weights > thresh)[0]
    for j in active:
        grad += weights[j] * corr_jac_single(h, n, j)
    grad *= dx
    s = h.sum() - n / 2.0
    obj = logsumexp + pen * s * s
    grad += 2.0 * pen * s * np.ones(n)
    return obj, grad

def slsqp_fun(h, n, alpha, pen):
    return smooth_obj_grad(h, n, alpha, pen)

    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return h

def make_initial_two_mode(n, a1=0.2, b1=0.15, a2=0.1, b2=0.08):
    x = np.linspace(0.0, 2.0, n, endpoint=False)

    h = np.clip(h, 0.0, 1.0)
    return h

def make_initial_binary_spread(n):
    h = np.zeros(n)
    half = n // 2
    indices = np.sort(np.random.choice(n, half, replace=False))
    h[indices] = 1.0
    return h

def resample_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    old_x = np.linspace(0.0, 2.0, n_old, endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h_old)

def binary_local_search(h, n, iterations, rng):
    half = n // 2
    h = np.zeros(n)
    indices = np.sort(rng.choice(n, half, replace=False))
    h[indices] = 1.0
    h = enforce_constraints(h, n)
    best_c5 = compute_c5(h, n)
    for it in range(iterations):
        if rng.random() < 0.3:
            pos_ones = np.where(h >= 0.5)[0]
            pos_zeros = np.where(h < 0.5)[0]
            if len(pos_ones) == 0 or len(pos_zeros) == 0:
                continue
            i = pos_ones[rng.integers(len(pos_ones))]
            j = pos_zeros[rng.integers(len(pos_zeros))]
            h_new = h.copy()
            h_new[i] = 0.0
            h_new[j] = 1.0
            h_new = enforce_constraints(h_new, n)
        else:
            i = rng.integers(n)
            delta = rng.uniform(-0.15, 0.15)
            h_new = h.copy()
            h_new[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h_new = enforce_constraints(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-12:
            h = h_new
            best_c5 = c5_new
    return h, best_c5
