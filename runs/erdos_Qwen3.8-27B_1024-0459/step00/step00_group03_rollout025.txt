from scipy.optimize import minimize
import time

def project(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    h = np.clip(h, 0, 1)
    for _ in range(100):
        deficit = target_sum - np.sum(h)
        if abs(deficit) < 1e-15:
            break
        if deficit > 0:
            # Can only increase elements that are < 1
            mask = h < 1 - 1e-15
            if not np.any(mask):
                break
            h[mask] += deficit / np.sum(mask)
            h = np.clip(h, 0, 1)
        else:
            # Can only decrease elements that are > 0
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += deficit / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def compute_correlation(h, dx):

    if abs(diff) > 1e-14:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(mask):

            result = np.clip(result, 0.0, 1.0)
    return result

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(c)
    exp_c = np.exp(beta * (c - max_c))
    sum_exp = np.sum(exp_c)
    smooth_max = max_c + np.log(sum_exp) / beta
    penalty = gamma * (np.sum(h) - target_sum) ** 2
    return smooth_max + penalty, c, sum_exp

def compute_grad(h, beta, gamma, target_sum, dx):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(c)
    exp_c = np.exp(beta * (c - max_c))
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        j_start = max(0, s)
        j_end = min(n - 1, s + n - 1)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end + 1)
            grad[idx] += wm * (1.0 - h[idx - s])
        j_start2 = max(0, -s)
        j_end2 = min(n - 1, n - 1 - s)
        if j_start2 <= j_end2:
            idx2 = np.arange(j_start2, j_end2 + 1)
            grad[idx2] += wm * (-h[idx2 + s])
    
    penalty_grad = 2.0 * gamma * (np.sum(h) - target_sum)
    grad += penalty_grad * np.ones(n)
    return grad

def project_exact(v, target_sum):
    """Exact Euclidean projection onto {0<=x<=1, sum(x)=target_sum}."""
    n = len(v)
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.clip(v - tau, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-14:
