    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exps = np.exp(shifted)
    s = np.sum(exps)
    val = (max_c + np.log(s) / beta) * dx
    return val, exps / s

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-10:

        k = idx - (n - 1)

            nk = n - k
            if nk > 0:
                grad[k:] += w * (1.0 - h[:nk])
