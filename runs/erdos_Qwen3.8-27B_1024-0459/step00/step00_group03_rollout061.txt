    return np.max(corr) * dx, corr

def project_to_constraint(h, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Final adjustment for numerical precision
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-15:
        h_proj += diff / N
        h_proj = np.clip(h_proj, 0.0, 1.0)

def log_sum_exp_obj(h, dx, beta):
    half = 1.0 - h
    corr = np.correlate(h, half, mode='full') * dx
    # Shift to avoid overflow
    cmax = np.max(corr)

    return cmax + np.log(np.sum(np.exp(beta * (corr - cmax)))) / beta

    n = len(h)

        f1 = log_sum_exp_obj(h, dx, beta)

        grad[i] = (f1 - f0) / eps
    # Add constraint penalty gradient
    grad += 2 * 1e8 * (np.sum(h) - n/2.0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_and_corr(h, dx):
    half = 1.0 - h
    corr = np.correlate(h, half, mode='full')
    return np.max(corr) * dx, corr

def project_to_constraint(h, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    h_proj += diff / N
    h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def lse_objective(h, dx, beta):
    half = 1.0 - h
    corr = np.correlate(h, half, mode='full') * dx
    cmax = np.max(corr)
    if cmax < 1e-15:
        return 0.0
    return cmax + np.log(np.sum(np.exp(beta * (corr - cmax)))) / beta

def lse_gradient(h, dx, beta):
    N = len(h)
    half = 1.0 - h
    corr = np.correlate(h, half, mode='full') * dx
    cmax = np.max(corr)
    if cmax < 1e-15:
        return np.zeros(N)
    logits = beta * (corr - cmax)
    expl = np.exp(logits)
    Z = np.sum(expl)
    p = expl / Z  # softmax weights over lags
    
    grad = np.zeros(N)
    for k in range(len(corr)):
        lag = k - (N - 1)
        weight = dx * p[k]
        for m in range(N):
            idx1 = m - lag
            idx2 = m + lag
            if 0 <= idx1 < N:
                grad[m] += weight * (1.0 - h[idx1])
            if 0 <= idx2 < N:
                grad[m] -= weight * h[idx2]
    return grad

def lse_gradient_fast(h, dx, beta):
    N = len(h)
    half = 1.0 - h
    corr = np.correlate(h, half, mode='full') * dx
    cmax = np.max(corr)
    if cmax < 1e-15:
        return np.zeros(N)
    logits = beta * (corr - cmax)
    expl = np.exp(logits)
    Z = np.sum(expl)
    p = expl / Z
    
    grad = np.zeros(N)
    for k in range(len(p)):
        if p[k] < 1e-12:
            continue
        lag = k - (N - 1)
        w = dx * p[k]
        if lag >= 0:
            idx1 = np.arange(lag, N)
            idx2 = np.arange(0, N - lag)
            grad[idx1] += w * (1.0 - h[:N-lag])
            if N - lag > 0:
                grad[:N-lag] -= w * h[lag:]
        else:
            al = -lag
            idx1 = np.arange(0, N - al)
            idx2 = np.arange(al, N)
            grad[:N-al] += w * (1.0 - h[al:])
            grad[al:] -= w * h[:N-al]
    return grad

def constraint_fun(h, target):
    return np.sum(h) - target

def constraint_jac(h, N):
    return np.ones(N)
