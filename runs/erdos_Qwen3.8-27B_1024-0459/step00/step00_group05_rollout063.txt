Let me think about this differently. The problem is equivalent to:
C5 = max_k ∫₀² h(x)(1-h(x-k)) dx (over all valid k)

For k near 0: ∫ h(x)(1-h(x)) dx = 1 - ∫h² dx (since ∫h = 1)
To minimize this, maximize ∫h², so make h spiky.

For k near 1: ∫ h(x)(1-h(x-1)) dx. If h is "shift-invariant" in some sense, this could be large.

For k near 2: only a small overlap region, so the integral is small.

The key tension: making h spiky reduces the k=0 term but might increase intermediate lags.

Actually, for the discretized problem, let me think about what structure minimizes the max correlation.

The correlation at lag k involves the "cross-correlation" between h and 1-h with shift k. If h is a "Bernoulli" sequence with p=1/2 (random), the expected correlation at lag k is approximately (n-k)/4, and C5 ≈ (n/4) * 2/n = 1/2.

To do better than 1/2, we need to make the correlations at ALL lags smaller than n/4.

One approach: make h "anti-correlated" with its own shifts. Like a Welch sequence or a Barker sequence. But those are for 0/1 sequences and the correlation is different.

Actually, for a 0/1 sequence with exactly n/2 ones:
- Correlation at lag 0: always 0 (since h*(1-h) = 0 for 0/1 values)
- Correlation at lag k: number of positions where h[i+k]=1 and h[i]=0 (or vice versa, depending on direction)

Wait, for a 0/1 sequence, h[i+k]*(1-h[i]) is 1 when h[i+k]=1 and h[i]=0. So the correlation at lag k counts the number of "0→1" transitions at distance k.

For a random 0/1 sequence with n/2 ones, the expected count at each lag k is (n-|k|)/4.

To minimize the maximum, we want a sequence where the "0→1" count is as uniform as possible across all lags. This is related to "balanced" sequences or "low autocorrelation" sequences.

But we're not restricted to 0/1 values! h can take any value in [0,1]. This gives us more flexibility.

For the gradient-based approach:

- The subgradient at h is: (dx) * grad_k*(h) where k* is the argmax

For the projected gradient:

- Project to {0 ≤ h ≤ 1, sum(h) = n/2}

The projection is: minimize ||h_new - h||² subject to 0 ≤ h ≤ 1, sum(h) = n/2.
This can be done by:
1. h_clipped = clip(h_new, 0, 1)
2. If sum(h_clipped) > n/2: reduce some values (those closest to 1)

3. More precisely, find λ such that sum(clip(h_new + λ, 0, 1)) = n/2

Actually, the correct projection onto {0≤h≤1, sum(h)=n/2} is:
h_proj[i] = clip(h_new[i] + λ, 0, 1) where λ is chosen so that sum(h_proj) = n/2.

This is found by bisection on λ.

import numpy as np
import time

def project_to_feasible(h, n_target):
    """Project h onto {0<=h<=1, sum(h)=n_target}"""
    # Bisection on lambda
    lo = -np.max(h) - 1
    hi = 1 - np.min(h)
    for _ in range(200):
        mid = (lo + hi) / 2
        h_try = np.clip(h + mid, 0, 1)
        s = np.sum(h_try)
        if s > n_target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo+hi)/2, 0, 1)

def compute_C5_values(h):
    """Return all correlation values (without dx scaling)"""
    return np.correlate(h, 1.0 - h, mode="full")

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_C5_values(h)) * dx

def grad_of_correlation_at_lag(h, lag):
    """Gradient of correlation at a specific lag"""
    n = len(h)
    # correlation at lag k: sum_i h[i+k]*(1-h[i]) for valid i
    # d/dh[j] = sum over i where j=i+k: (1-h[i]) + sum over i where j=i: -h[i+k]
    grad = np.zeros(n)
    if lag >= 0:
        # i ranges from 0 to n-1-lag
        for i in range(n - lag):
            grad[i + lag] += (1 - h[i])
            grad[i] += -h[i + lag]
    else:
        # lag < 0, let offset = -lag
        # i ranges from -lag to n-1
        offset = -lag
        for i in range(offset, n):
            grad[i - offset] += (1 - h[i])
            grad[i] += -h[i - offset]
    return grad
```

def smooth_max_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")  # length 2n-1
    # corr[k] corresponds to lag = k-(n-1)
    logits = tau * corr * dx
    logits -= np.max(logits)  # for numerical stability
    p = np.exp(logits)
    p /= np.sum(p)
    
    # Now compute gradient
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        lag = k_idx - (n-1)
        if p[k_idx] < 1e-15:
            continue
        if lag >= 0:
            m = n - lag
            grad[lag:] += p[k_idx] * (1 - h[:m])
            grad[:m] += -p[k_idx] * h[lag:]
        else:
            offset = -lag
            m = n - offset
            grad[:m] += p[k_idx] * (1 - h[offset:])
            grad[offset:] += -p[k_idx] * h[:m]
    return grad
```

import numpy as np
import time

def project_to_feasible(h, n_target=None):
    n = len(h)
    if n_target is None:
        n_target = n / 2.0
    # Find lambda such that sum(clip(h + lambda, 0, 1)) = n_target
    # sum(clip(h+lambda, 0, 1)) is monotonically increasing in lambda
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_try)
        if s > n_target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = n_target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_single_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[lag:] += (1.0 - h[:m])
        grad[:m] += -h[lag:]
    else:
        offset = -lag
        m = n - offset
        grad[:m] += (1.0 - h[offset:])
        grad[offset:] += -h[:m]
    return grad

def subgradient_of_max_c5(h, smooth_tau=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlation(h)
    scaled = corr * dx
    if smooth_tau > 0:
        # Softmax weighting
        logits = smooth_tau * scaled
        logits -= np.max(logits)
        weights = np.exp(logits)
        weights /= np.sum(weights)
    else:
        max_val = np.max(scaled)
        active = np.where(np.abs(scaled - max_val) < 1e-10)[0]
        weights = np.zeros(len(corr))
        weights[active] = 1.0 / len(active)
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-15:
            continue
        lag = k_idx - (n - 1)
        g = grad_single_lag(h, lag)
        grad += weights[k_idx] * g
    return grad, float(np.max(scaled))

def optimize_subgradient(h_init, max_iter=5000, tau_start=1.0, tau_end=50.0, time_limit=None, t_start=None):
    n = len(h_init)
    h = h_init.astype(float).copy()
    n_target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alpha = 0.01  # initial step size
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        grad, c5_val = subgradient_of_max_c5(h, smooth_tau=tau_start + (tau_end - tau_start) * it / max_iter)
        
        momentum = beta * momentum + (1 - beta) * grad
        h_trial = h - alpha * momentum
        h_trial = project_to_feasible(h_trial, n_target)
        
        # Backtracking
        a = alpha
        c5_trial = compute_c5(h_trial)
        for _ in range(20):
            if c5_trial < c5_val * (1 - 1e-5) or c5_trial <= best_c5:
                break
            a *= 0.5
            h_trial = h - a * momentum
            h_trial = project_to_feasible(h_trial, n_target)
            c5_trial = compute_c5(h_trial)
        
        h = h_trial
        c5_val = c5_trial
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            alpha *= 1.01  # slightly increase step size on improvement
        else:
            alpha *= 0.95
        
        alpha = np.clip(alpha, 1e-4, 1.0)
    
    return best_h, best_c5

def interpolate_to_resolution(h_old, n_new):
    """Interpolate h from old resolution to new resolution"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    # Re-project
    h_new = project_to_feasible(h_new, n_new / 2.0)
    return h_new

def random_initial(n, seed=42):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0, 1, n)
    h = project_to_feasible(h, n / 2.0)
    return h

import numpy as np
import time
from scipy.optimize import minimize


def project_to_feasible(h, n_target=None):
    n = len(h)
    if n_target is None:
        n_target = n / 2.0
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(120):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_try)
        if s > n_target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = n_target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode="full")


def grad_single_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:] += (1.0 - h[:m])
            grad[:m] += -h[lag:]
    else:
        offset = -lag
        m = n - offset
        if m > 0:
            grad[:m] += (1.0 - h[offset:])
            grad[offset:] += -h[:m]
    return grad


def smooth_c5_value(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    shifted = tau * (scaled - m)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return m + log_sum / tau


def smooth_c5_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    logits = tau * (scaled - m)
    p = np.exp(logits)
    p_sum = np.sum(p)
    if p_sum < 1e-300:
        return np.zeros(n)
    p /= p_sum
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = p[k_idx]
        if w < 1e-15:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            mm = n - lag
            if mm > 0:
                grad[lag:] += w * (1.0 - h[:mm])
                grad[:mm] += -w * h[lag:]
        else:
            offset = -lag
            mm = n - offset
            if mm > 0:
                grad[:mm] += w * (1.0 - h[offset:])
                grad[offset:] += -w * h[:mm]
    return dx * grad


def lbgfs_optimize(h0, tau, n_target, max_iter=500, lam=50.0):
    n = len(h0)
    h0 = project_to_feasible(h0, n_target).copy()

    def obj(h):
        val = smooth_c5_value(h, tau)
        penalty = lam * (np.sum(h) - n_target) ** 2
        return val + penalty

    def grad(h):
        g = smooth_c5_gradient(h, tau)
        penalty_g = 2.0 * lam * (np.sum(h) - n_target)
        return g + penalty_g

    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
    h_res = project_to_feasible(res.x, n_target)
    return h_res


def subgradient_optimize(h_init, max_iter=5000, tau_schedule=None, time_limit=None, t_start=None):
    n = len(h_init)
    h = h_init.astype(float).copy()
    n_target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.05
    momentum_vec = np.zeros(n)
    beta_mom = 0.8
    no_improve_count = 0

    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:

        logits -= np.max(logits)
        p = np.exp(logits)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.zeros(len(corr))
            p[np.argmax(corr)] = 1.0
        else:
            p /= p_sum

        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            if p[k_idx] < 1e-14:

            lag = k_idx - (n - 1)

            grad += p[k_idx] * g

        h_trial = h - alpha * momentum_vec
        h_trial = project_to_feasible(h_trial, n_target)

def interpolate_to_resolution(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_to_feasible(h_new, n_new / 2.0)
    return h_new

def two_step_pattern(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    block = max(2, n // 10)
    filled = 0
    i = 0

        end = min(i + length, n)
        h[i:end] = 1.0
        filled += end - i
        i = end + rng.randint(1, max(2, block // 2))
    h = project_to_feasible(h, n / 2.0)
    return h
