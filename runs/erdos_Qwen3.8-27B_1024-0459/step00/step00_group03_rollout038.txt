import numpy as np
import time
from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_g_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    """Project to {x: 0<=x[i]<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    
    # If already close, just clip
    if abs(h.sum() - target) < 1e-10 and h.min() >= -1e-12 and h.max() <= 1 + 1e-12:
        return np.clip(h, 0, 1)
    
    # Bisection on lambda: find lambda such that sum(clip(h-lambda, 0, 1)) = target
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(h - (lo + hi) / 2.0, 0, 1)
    # Fine correction
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        interior = (result > 1e-9) & (result < 1.0 - 1e-9)
        if interior.sum() > 0:
            result[interior] += diff / interior.sum()
            result = np.clip(result, 0, 1)
    return result

def upsample(h, factor):
    """Upsample h by integer factor"""
    n_old = len(h)
    n_new = n_old * factor
    h_new = np.repeat(h, factor)
    # Sum is automatically n_new/2 since sum(h_new) = factor * sum(h) = factor * n_old/2 = n_new/2
    return h_new

def subgradient_descent(h, n_iters, alpha_init=0.1, decay=0.999, active_tol=0.005):
    """Run subgradient descent"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    alpha = alpha_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        g = get_g_values(h)
        c5 = g.max()
        
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
        
        # Active set
        active = np.where(g >= c5 - active_tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(g)])
        
        # Compute subgradient
        grad = np.zeros(n)
        for idx in active:
            d = idx - (n - 1)
            # Term 1: (1 - h[i+d]) for valid i+d
            lo1 = max(0, -d)
            hi1 = min(n - 1, n - 1 - d)
            if lo1 <= hi1:
                idxs = np.arange(lo1, hi1 + 1)
                grad[idxs] += (1.0 - h[idxs + d])
            # Term 2: -h[i-d] for valid i-d
            lo2 = max(0, d)
            hi2 = min(n - 1, n - 1 + d)
            if lo2 <= hi2:
                idxs = np.arange(lo2, hi2 + 1)
                grad[idxs] -= h[idxs - d]
        
        grad *= dx
        grad /= len(active)
        
        # Update
        h = h - alpha * grad
        h = project_feasible(h, n)
        
        alpha *= decay
    
    return best_h, best_c5

def paired_swap_local_search(h, max_sweeps=5, tol=1e-8):
    """Local search using paired swaps"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        g_current = get_g_values(h)
        
        for i in range(n):
            for j in range(i + 1, n):
                # Feasible range for delta
                delta_min = max(-h[i], h[j] - 1.0)
                delta_max = min(1.0 - h[i], h[j])
                
                if delta_max - delta_min < 1e-10:
                    continue
                
                # Compute c_d for all d
                # c_d = (1-h[i+d])_v - h[i-d]_v - (1-h[j+d])_v + h[j-d]_v
                # where _v means "value if index valid, else 0"
                
                # Vectorized computation
                d_range = np.arange(-(n-1), n)  # all possible lags
                
                c = np.zeros(2 * n - 1)
                
                # (1-h[i+d]) for valid i+d
                valid1 = (i + d_range >= 0) & (i + d_range < n)
                c[valid1] += (1.0 - h[i + d_range[valid1]])
                
                # -h[i-d] for valid i-d
                valid2 = (i - d_range >= 0) & (i - d_range < n)
                c[valid2] -= h[i - d_range[valid2]]
                
                # -(1-h[j+d]) for valid j+d
                valid3 = (j + d_range >= 0) & (j + d_range < n)
                c[valid3] -= (1.0 - h[j + d_range[valid3]])
                
                # +h[j-d] for valid j-d
                valid4 = (j - d_range >= 0) & (j - d_range < n)
                c[valid4] += h[j - d_range[valid4]]
                
                # g_d(delta) = g_d(0) + dx * c_d * delta
                # C5(delta) = max_d (g_d(0) + dx * c_d * delta)
                
                # Minimize over delta in [delta_min, delta_max]
                # Use bisection/ternary search or scipy
                
                # Quick check: evaluate at endpoints and midpoint
                vals = []
                for test_d in [delta_min, delta_max, (delta_min + delta_max) / 2]:
                    vals.append(np.max(g_current + dx * c * test_d))
                
                best_test_d = [delta_min, delta_max, (delta_min + delta_max) / 2][np.argmin(vals)]
                
                # Refine with minimize_scalar
                def obj(delta):
                    return float(np.max(g_current + dx * c * delta))
                
                res = minimize_scalar(obj, bounds=(delta_min, delta_max), method='bounded', 
                                     options={'xatol': 1e-8})
                
                if res.fun < best_c5 - tol:
                    h[i] += res.x
                    h[j] -= res.x
                    h = np.clip(h, 0, 1)
                    best_c5 = res.fun
                    improved = True
                    g_current = get_g_values(h)
        
        if not improved:
            break
    
    return h, best_c5

import numpy as np
import time
from scipy.optimize import minimize_scalar

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_g_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    if abs(h.sum() - target) < 1e-10 and h.min() >= -1e-12 and h.max() <= 1.0 + 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        interior = (result > 1e-9) & (result < 1.0 - 1e-9)
        if interior.sum() > 0:
            result[interior] += diff / interior.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def upsample(h, factor):
    return np.repeat(h, factor)

def subgradient_descent(h, n_iters, alpha_init, decay, active_tol, time_limit):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    alpha = alpha_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        g = get_g_values(h)
        c5 = float(g.max())
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        threshold = c5 - active_tol
        active = np.where(g >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(g)])
        grad = np.zeros(n)
        for idx in active:
            d = idx - (n - 1)
            lo1 = max(0, -d)
            hi1 = min(n - 1, n - 1 - d)
            if lo1 <= hi1:
                idxs = np.arange(lo1, hi1 + 1)
                grad[idxs] += (1.0 - h[idxs + d])
            lo2 = max(0, d)
            hi2 = min(n - 1, n - 1 + d)
            if lo2 <= hi2:
                idxs = np.arange(lo2, hi2 + 1)
                grad[idxs] -= h[idxs - d]
        grad *= dx
        grad /= len(active)
        h = h - alpha * grad
        h = project_feasible(h, n)
        alpha *= decay
    return best_h, best_c5

def find_optimal_delta(g_current, c, dx, delta_min, delta_max):
    best_val = np.inf
    best_delta = 0.0
    for _ in range(40):
        mid = (delta_min + delta_max) / 2.0
        f_mid = float(np.max(g_current + dx * c * mid))
        f_left = float(np.max(g_current + dx * c * delta_min))
        f_right = float(np.max(g_current + dx * c * delta_max))
        if f_mid < best_val:
            best_val = f_mid
            best_delta = mid
        if delta_max - delta_min < 1e-10:
            break
        f_left_third = float(np.max(g_current + dx * c * (delta_min + (delta_max - delta_min) / 3.0)))
        f_right_third = float(np.max(g_current + dx * c * (delta_max - (delta_max - delta_min) / 3.0))))
        if f_left_third < f_right_third:
            delta_max = delta_max - (delta_max - delta_min) / 3.0
        else:
            delta_min = delta_min + (delta_max - delta_min) / 3.0
    return best_delta, best_val

def paired_swap_local_search(h, max_sweeps, tol, time_limit, random_pairs_fraction=1.0, seed=42):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        improved = False
        g_current = get_g_values(h)
        c5_current = float(g_current.max())
        
        if random_pairs_fraction < 1.0:
            all_pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            n_pairs = int(len(all_pairs) * random_pairs_fraction)
            pairs = rng.choice(len(all_pairs), size=n_pairs, replace=False)
            pairs = [all_pairs[k] for k in pairs]
        else:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        
        for i, j in pairs:
            if time.time() - start > time_limit:
                break
            delta_min = max(-h[i], h[j] - 1.0)
            delta_max = min(1.0 - h[i], h[j])
            if delta_max - delta_min < 1e-10:
                continue
            
            d_range = np.arange(-(n - 1), n)
            c = np.zeros(2 * n - 1)
            valid1 = (i + d_range >= 0) & (i + d_range < n)
            c[valid1] += (1.0 - h[i + d_range[valid1]])
            valid2 = (i - d_range >= 0) & (i - d_range < n)
            c[valid2] -= h[i - d_range[valid2]]
            valid3 = (j + d_range >= 0) & (j + d_range < n)
            c[valid3] -= (1.0 - h[j + d_range[valid3]])
            valid4 = (j - d_range >= 0) & (j - d_range < n)
            c[valid4] += h[j - d_range[valid4]]
            
            best_delta, best_val = find_optimal_delta(g_current, c, dx, delta_min, delta_max)
            
            if best_val < best_c5 - tol:
                h[i] += best_delta
                h[j] -= best_delta
                h = np.clip(h, 0.0, 1.0)
                best_c5 = best_val
                improved = True
                g_current = get_g_values(h)
        
        if not improved:
            break
    return h, best_c5

def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = n // 2
    indices = rng.choice(n, size=ones, replace=False)
    h[indices] = 1.0
    return h

def sinusoid_init(n, freq=1, amp=0.4):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)
