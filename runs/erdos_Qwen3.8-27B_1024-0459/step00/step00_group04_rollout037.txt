```python
    """Smooth approximation of max."""
    max_corr = np.max(corr)
    exp_corr = np.exp(T * (corr - max_corr))  # shifted for numerical stability
    return max_corr + np.log(np.sum(exp_corr)) / T

def smooth_max_grad(corr, T=10):

    max_corr = np.max(corr)
    exp_corr = np.exp(T * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr

def compute_single_grad(h, k_star):
    """Compute gradient of the k_star-th correlation w.r.t. h."""
    n = len(h)
    s = k_star - (n - 1)
    grad = np.zeros(n)
    
    if s >= 0:
        upper = n - 1 - s
        # (1-h[i+s]) for i in [0, upper]
        grad[0:upper+1] += (1 - h[s:n])
        # -h[i+s] for i in [0, upper]  (from the -h[n]*h[n+s] term, n=i)
        grad[0:upper+1] -= h[s:n]
        # -h[i-s] for i in [s, n-1] (from the -h[n]*h[n+s] term, n=i-s)
        grad[s:n] -= h[0:upper+1]
    else:

        upper = n - 1 - t
        # (1-h[i-t]) for i in [t, n-1]
        grad[t:n] += (1 - h[0:upper+1])
        # -h[i-t] for i in [t, n-1]
        grad[t:n] -= h[0:upper+1]
        # -h[i+t] for i in [0, upper]
        grad[0:upper+1] -= h[t:n]
    
    return grad

    """Project onto {0<=h<=1, sum(h)=target_sum}."""
    lo = -1.0
    hi = 1.0

        if s > target_sum:
            lo = mid
        else:
            hi = mid

    return np.clip(h - tau, 0, 1)

def subgradient_optimize(h0, n_iters=50000, verbose=False):

    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Nesterov momentum
    v = np.zeros(n)
    mu = 0.9  # momentum parameter
    
    # Adaptive step size
    alpha = 0.1 / n  # initial step
    min_alpha = 1e-10
    max_alpha = 1.0 / n
    
    no_improve = 0

        y = h + mu * v
        corr = compute_correlations(y)
        k_star = np.argmax(corr)
        c5_val = corr[k_star]
        
        grad = compute_single_grad(y, k_star)

            best_h = h_new.copy()
            no_improve = 0
        else:

            alpha = min(alpha * 1.05, max_alpha)
        else:

        h = h_new
        
        if verbose and it % 5000 == 0:
            print(f"Iter {it}: C5={compute_c5(h):.6f}, best={best_c5:.6f}, alpha={alpha:.8f}")

            v = np.zeros(n)
            alpha = 0.1 / n
            no_improve = 0
            if mu <= 0.501:
```

One more thought: I should also try a "smoothing" approach where I minimize a differentiable approximation of the max. This can be more effective than the subgradient method because it uses information from all near-max lags, not just the argmax.

The smoothed max can be computed as:
F_T(h) = (1/T) * log(sum_k exp(T * corr_k))

where corr_k = correlation at lag k * dx.

The gradient of F_T is:
∇F_T = (1/T) * sum_k exp(T*corr_k) / sum_j exp(T*corr_j) * ∇corr_k

This is a weighted average of the individual gradients, with weights proportional to exp(T*corr_k). For large T, this concentrates on the max; for small T, it's more spread out.

The advantage of using this is that it's a smooth function, so we can use more sophisticated optimizers.

However, computing all 2N-1 gradients is O(N²) per iteration (same as computing all correlations), so it's not more expensive than the subgradient method (which also computes all correlations).

Let me implement both and use the smoothed version as the primary method.

```python
def smooth_max_and_grad(h, T=50):
    """Compute smoothed max of correlations and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    
    # Compute weights for smooth max
    max_corr = np.max(corr)
    shifted = T * (corr - max_corr)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    smooth_val = max_corr + np.log(np.sum(exp_shifted)) / T
    
    # Compute gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    # Instead of computing all 2N-1 gradients individually,
    # we can use the fact that the gradient is a weighted sum
    # and find an efficient formula
    
    # Actually, let's just compute it directly for the top few lags
    # Or compute all of them (O(N^2) total)
    
    for k in range(2 * n - 1):
        if weights[k] > 1e-10:
            grad += weights[k] * compute_single_grad(h, k)
    
    return smooth_val, grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_single_grad(h, k_star):
    n = len(h)
    m = k_star - (n - 1)
    grad = np.zeros(n)
    if m >= 0:
        grad[m:n] += 1.0
        grad[m:n] -= h[0:n - m]
        grad[0:n - m] -= h[m:n]
    else:
        t = -m
        grad[0:n - t] += 1.0
        grad[0:n - t] -= h[t:n]
        grad[t:n] -= h[0:n - t]
    return grad

def project(h, target_sum):
    lo = -1.5
    hi = 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def upscale_to(h, target_n):
    cur_n = len(h)
    if target_n == cur_n:
        return h.copy()
    x_old = np.arange(cur_n)
    x_new = np.arange(target_n) * (cur_n / target_n)
    h_new = np.interp(x_new, x_old, h, period=cur_n)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project(h_new, target_n / 2.0)
    return h_new

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_corr = np.max(corr)
    shifted = T * (corr - max_corr)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    smooth_val = max_corr + np.log(total) / T
    grad = np.zeros(n)
    top_indices = np.argsort(weights)[-40:]
    for k in top_indices:
        w = weights[k]
        if w > 1e-10:
            grad += w * compute_single_grad(h, k)
    return smooth_val, grad, corr, max_corr

def optimize_smoothed(h0, time_limit, seed=42, verbose=False):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_smooth = best_c5

    v = np.zeros(n)
    mu = 0.9
    alpha = 2.0 / n
    min_alpha = 1e-14
    max_alpha = 50.0 / n

    T = 20.0
    T_min = 80.0

    no_improve = 0
    patience = 30000
    it = 0
    start = time.time()

    while time.time() - start < time_limit:
        it += 1

        y = h + mu * v
        smooth_val, grad, corr, max_corr = smooth_max_grad(y, T)

        v = mu * v - alpha * grad
        h_new = h + v
        h_new = project(h_new, target_sum)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_smooth = smooth_val
            no_improve = 0
        else:
            no_improve += 1

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            direction_improvement = np.dot(grad, h_new - h)
            if direction_improvement < 0:
                alpha = min(alpha * 1.08, max_alpha)
            else:
                alpha = max(alpha * 0.7, min_alpha)

        h = h_new

        # Anneal temperature
        progress = min(it / 100000.0, 1.0)
        T = min(T + 0.01, T_min)

        if no_improve > patience:
            # Restart
            mu = max(mu * 0.9, 0.5)
            v[:] = 0.0
            alpha = 2.0 / n
            no_improve = 0
            T = max(T - 20.0, 10.0)
            # Small random perturbation
            h = h + rng.randn(n) * 0.02
            h = project(h, target_sum)
            if mu <= 0.501:
                break

        if verbose and it % 20000 == 0:
            elapsed = time.time() - start
            print(f"    Iter {it}: C5={best_c5:.6f}, T={T:.1f}, alpha={alpha:.8f}, elapsed={elapsed:.1f}s")

    return best_h, best_c5
```
