        steps = [0.05, 0.02, 0.01, 0.005, 0.002]
        h_cd, c5_cd = _coord_descent(best_h, best_n, steps, sweeps=4)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Random restarts
    n_restarts = max(3, int(rem() / 40))
    for i in range(n_restarts):
        if rem() < 25:
            break
        noise_scale = rng.uniform(0.03, 0.15)
        h_pert = _project(best_h + rng.normal(0, noise_scale, best_n), best_n)
        h_new, c5_new = _opt_lbgfs(h_pert, best_n, 100, 200, 5000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Coord descent on the new candidate
        if rem() < 15:
            break
        h_cd, c5_cd = _coord_descent(h_new, best_n, [0.02, 0.01, 0.005], sweeps=2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Final high-beta polish
    if rem() > 20:
        for beta in [200, 500, 1000]:
            if rem() < 10:
                break
            h_new, c5_new = _opt_lbgfs(best_h, best_n, beta, 200, 3000)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        h_cd, c5_cd = _coord_descent(best_h, best_n, [0.01, 0.005, 0.002, 0.001], sweeps=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Try binary projection
    if rem() > 15:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = _project(h_bin, best_n)
        c5_bin = _eval_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h_cd, c5_cd = _coord_descent(best_h, best_n, [0.5, 0.25, 0.1], sweeps=3)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final: ensure valid
    best_h = _project(best_h, best_n)
    final_c5 = _eval_c5(best_h)

    try:
        v = evaluate_erdos_solution(best_h)
        if v is not None:
            final_c5 = float(v)
    except (NameError, Exception):
        pass

    return (best_h, final_c5, best_n)
