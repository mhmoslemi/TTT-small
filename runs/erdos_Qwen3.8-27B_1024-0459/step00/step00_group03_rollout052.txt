        best_h = project_h(rng.random(n), n)
        best_c5 = compute_c5(best_h, n)
        best_n = n

    # Final verification
    final_c5 = compute_c5(best_h, best_n)
    if abs(best_h.sum() - best_n / 2.0) > 1e-6:
        best_h = project_h(best_h, best_n)
        final_c5 = compute_c5(best_h, best_n)

    return best_h, final_c5, best_n
