            grad[idx2] -= h[idx2 + p]
        
        c_val = corr[k_star] * (2.0 / n)
        return grad, c_val, s
    
    def subgrad_topk(h, n, k=3):
        """Compute weighted subgradient using top-k shifts."""
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        top_idx = np.argsort(corr)[::-1][:k]
        weights = np.exp(corr[top_idx] - corr[top_idx[0]])
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, ki in zip(weights, top_idx):
            s = ki - n + 1
            g = np.zeros(n)
            if s >= 0:
                idx1 = np.arange(n - s)
                g[idx1] += (1.0 - h[idx1 + s])
                if s > 0:
                    idx2 = np.arange(s, n)
                    g[idx2] -= h[idx2 - s]
            else:
                p = -s
                idx1 = np.arange(p, n)
                g[idx1] += (1.0 - h[idx1 - p])
                idx2 = np.arange(0, n - p)
                g[idx2] -= h[idx2 + p]
            grad += w * g
        
        c_val = corr[top_idx[0]] * dx
        return grad, c_val
    
    def optimize_binary_swaps(h, n, max_iters=50000):
        """Local search: swap 1s and 0s to reduce max correlation."""
        h = h.copy()
        n_ones = int(np.sum(h > 0.5))
        c5 = compute_c5(h, n)
        one_positions = list(np.where(h > 0.5)[0])
        zero_positions = list(np.where(h <= 0.5)[0])
        no_improve = 0
        temp = 1.0
        
        for it in range(max_iters):
            if elapsed() > budget_s - 10:
                break
            if no_improve > 5000:
                break
            
            # Pick random 1 and 0 to swap
            i = one_positions[np.random.randint(len(one_positions))]
            j = zero_positions[np.random.randint(len(zero_positions))]
            if i == j:
                continue
            
            # Try swap
            h[i], h[j] = 0.0, 1.0
            c5_new = compute_c5(h, n)
            
            if c5_new < c5 or (temp > 0.01 and np.random.rand() < np.exp(-(c5_new - c5) / temp)):
                c5 = c5_new
                one_positions.remove(i)
                one_positions.append(j)
                zero_positions.remove(j)
                zero_positions.append(i)
                no_improve = 0
            else:
                h[i], h[j] = 1.0, 0.0
                no_improve += 1
            
            temp = max(0.01, temp * 0.999)
        
        return h, c5
    
    def optimize_gradient(h, n, max_iter=5000, base_lr=0.5):
        """Projected subgradient descent."""
        h = project_feasible(h.copy(), n)
        best_local = h.copy()
        best_local_c5 = compute_c5(h, n)
        patience = 0
        
        for it in range(max_iter):
            if elapsed() > budget_s - 5:
                break
            
            grad, c_val = subgrad_topk(h, n, k=5)
            lr = base_lr / (1.0 + 0.05 * it)
            
            h_new = project_feasible(h - lr * grad, n)
            c5_new = compute_c5(h_new, n)
            
            if c5_new < best_local_c5 - 1e-10:
                best_local = h_new.copy()
                best_local_c5 = c5_new
                patience = 0
            else:
                patience += 1
                if patience > 500:
                    break
            
            h = h_new
        
        return best_local, best_local_c5
    
    def make_thue_morse(n):
        tm = np.array([0])
        while len(tm) < n:
            tm = np.concatenate([tm, 1 - tm])
        tm = tm[:n].copy()
        # Adjust to exactly n//2 ones
        n_ones = int(np.sum(tm))
        target = n // 2
        if n_ones > target:
            idx = np.where(tm == 1)[0]
            np.random.shuffle(idx)
            tm[idx[:n_ones - target]] = 0
        elif n_ones < target:
            idx = np.where(tm == 0)[0]
            np.random.shuffle(idx)
            tm[idx[:target - n_ones]] = 1
        return tm.astype(float)
    
    def make_dyadic(n):
        """Dyadic/radio-structured sequence."""
        h = np.zeros(n)
        # Place 1s at positions based on dyadic intervals
        # This creates a well-spread pattern
        for level in range(int(np.log2(n)) + 1):
            step = n // (2 ** level)
            if step == 0:
                break
            # Mark every other block
            for i in range(0, n, 2 * step):
                h[i:i+step] = 1 - h[i:i+step]
        # Adjust count
        n_ones = int(np.round(np.sum(h)))
        target = n // 2
        if n_ones != target:
            if n_ones > target:
                idx = np.where(h == 1)[0]
                np.random.shuffle(idx)
                h[idx[:n_ones - target]] = 0
            else:
                idx = np.where(h == 0)[0]
                np.random.shuffle(idx)
                h[idx[:target - n_ones]] = 1
        return h
