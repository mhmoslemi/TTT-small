    center = n - 1
    max_val = corr.max()
    active = np.where(corr >= max_val * 0.95)[0]
    if len(active) > top_k:
        active = active[np.argsort(-corr[active])[:top_k]]
    grad = np.zeros(n)
    for idx in active:
        L = idx - center
        if L >= 0:
            if L == 0:
                grad += (1.0 - 2.0 * h)
            else:
                grad[L:] += (1.0 - h[:n - L])
                grad[:n - L] -= h[L:]
        else:
            K = -L
            grad[:n - K] += (1.0 - h[K:])
            grad[K:] -= h[:n - K]
    grad /= max(len(active), 1)
    return grad - np.mean(grad)

def _sa_run(h_init, n, n_iters, rng, T0, T1, start_t, budget, target):
    h = h_init.copy()
    c5 = _eval_c5(h)
    best_c5 = c5
    best_h = h.copy()
    stagnant = 0
    max_stagnant = 5000

    for it in range(n_iters):
        if time.time() - start_t > budget - 2:
            break
        if best_c5 < target:
            break

        T = T0 * (T1 / T0) ** (it / max(n_iters - 1, 1))
        h_save = h.copy()

        r = rng.random()
        if r < 0.3:
            a, b = rng.choice(n, 2, replace=False)
            d = rng.uniform(0.1, 0.9) * min(h[a], 1.0 - h[b])
            if d < 1e-6:
                h = h_save
                continue
            h[a] -= d
            h[b] += d
        elif r < 0.55:
            a, b = rng.choice(n, 2, replace=False)
            d = rng.uniform(0.1, 0.9) * min(h[b], 1.0 - h[a])
            if d < 1e-6:
                h = h_save
                continue
            h[a] += d
            h[b] -= d
        elif r < 0.7:
            grad = _subgradient_direction(h, n, 3)
            scale = rng.uniform(0.05, 0.3)
            h = h - scale * grad
            h = np.clip(h, 0.0, 1.0)
        elif r < 0.85:
            k = rng.randint(3, min(25, n))
            idxs = rng.choice(n, k, replace=False)
            h[idxs] += rng.uniform(-0.3, 0.3, k)
            h = np.clip(h, 0.0, 1.0)
        else:
            a, b = rng.choice(n, 2, replace=False)
            h[a] = 1.0 if rng.random() < 0.5 else 0.0
            h[b] = 0.0 if rng.random() < 0.5 else 1.0

        h = _make_valid(h, n)
        c5_new = _eval_c5(h)

        if c5_new <= c5:
            c5 = c5_new
            stagnant = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            stagnant += 1
            delta = c5_new - c5
            if T > 1e-15:
                accept_prob = math.exp(-min(delta / T, 50.0))
                if rng.random() < accept_prob:
                    c5 = c5_new
                else:
                    h = h_save
            else:
                h = h_save

        if stagnant > max_stagnant:
            h = h_save
            idxs = rng.choice(n, min(n, max(10, n // 5)), replace=False)
            h[idxs] += rng.uniform(-0.4, 0.4, len(idxs))
            h = _make_valid(h, n)
            c5 = _eval_c5(h)
            stagnant = 0

    return best_c5, best_h

def _greedy_descent(h_init, n, max_iters, rng, start_t, budget, target):
    h = h_init.copy()
    c5 = _eval_c5(h)
    best_c5 = c5
    best_h = h.copy()

    for it in range(max_iters):
        if time.time() - start_t > budget - 2:
            break
        if best_c5 < target:
            break

        grad = _subgradient_direction(h, n, 5)
        for lr in [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]:
            h_try = _make_valid(h - lr * grad, n)
            c5_try = _eval_c5(h_try)
            if c5_try < c5:
                h = h_try
                c5 = c5_try
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                break
            else:
                h = h

    return best_c5, best_h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    current_target = 0.38085

    try:
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    except NameError:
        n = 140
        h = np.zeros(n)
        for i in range(n // 2):
            h[2 * i] = 1.0

    h = _make_valid(h, n)
    c5 = _eval_c5(h)
    best_c5 = c5
    best_h = h.copy()
    best_n = n

    # Phase 1: Greedy + SA at starting resolution
    c5_g, h_g = _greedy_descent(h, n, 200, rng, t0, budget_s, best_c5)
    if c5_g < best_c5:
        best_c5 = c5_g
        best_h = h_g.copy()
        best_n = n

    c5_s, h_s = _sa_run(h_g, n, 15000, rng, 0.05, 0.0005, t0, budget_s, best_c5)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()
        best_n = n

    # Try binary projection
    h_b = _binary_project(best_h, best_n)
    c5_b = _eval_c5(h_b)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b
        best_n = best_n

    # Phase 2: Multi-resolution
    for factor in [2, 3, 4, 6, 8]:
        elapsed = time.time() - t0
        if elapsed > budget_s - 50:
            break
        n_new = best_n * factor
        if n_new > 6000 or n_new <= best_n:
            continue

        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _make_valid(h_up, n_new)

        c5_up = _eval_c5(h_up)
        target = min(best_c5, c5_up)

        # Greedy first
        remaining = budget_s - (time.time() - t0) - 20
        g_iters = min(100, int(remaining * 5))
        c5_g2, h_g2 = _greedy_descent(h_up, n_new, max(50, g_iters), rng, t0, budget_s, target)
        if c5_g2 < target:
            target = c5_g2

        # SA
        sa_iters = max(5000, int((budget_s - (time.time() - t0) - 30) * 25))
        c5_s2, h_s2 = _sa_run(h_g2, n_new, sa_iters, rng, 0.03, 0.0003, t0, budget_s, target)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2.copy()
            best_n = n_new

        # Binary at new res
        h_b2 = _binary_project(best_h, best_n)
        c5_b2 = _eval_c5(h_b2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2
            best_n = best_n

    # Phase 3: Random restarts at best resolution
    attempt = 0
    while time.time() - t0 < budget_s - 15:
        attempt += 1
        sigma = 0.1 + 0.1 * (attempt % 4)
        h_r = best_h + rng.normal(0, sigma, best_n)
        h_r = _make_valid(h_r, best_n)

        iters = max(3000, int((budget_s - (time.time() - t0) - 15) * 15))
        c5_r, h_r = _sa_run(h_r, best_n, iters, rng, 0.04, 0.0005, t0, budget_s, best_c5)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

        h_br = _binary_project(h_r, best_n)
        c5_br = _eval_c5(h_br)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = h_br

        # Greedy polish
        c5_gp, h_gp = _greedy_descent(best_h, best_n, 100, rng, t0, budget_s, best_c5)
        if c5_gp < best_c5:
            best_c5 = c5_gp
            best_h = h_gp.copy()

    # Phase 4: Try a couple of fresh random starts
    while time.time() - t0 < budget_s - 10:
        n_rand = best_n
        h_rand = rng.uniform(0, 1, n_rand)
        h_rand = _make_valid(h_rand, n_rand)
        c5_r, h_r = _sa_run(h_rand, n_rand, 4000, rng, 0.06, 0.001, t0, budget_s, best_c5)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_rand

    # Final validation and return
    best_h = _make_valid(best_h, best_n)
    s = np.sum(best_h)
    target_sum = best_n / 2.0
    if abs(s - target_sum) > 1e-8:
        best_h = _make_valid(best_h, best_n)
    final_c5 = _eval_c5(best_h)

    return best_h, final_c5, best_n
