def project_box_sum(x, S):
    x = np.asarray(x, dtype=float)
    n = x.size
    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if abs(s - S) <= 1e-12:
        return x
    lo = -1.0
    hi = 1.0
    # ensure f(lo)>0 and f(hi)<0; if not, extend
    for _ in range(20):
        mid = 0.5*(lo+hi)
        y = np.clip(x - mid, 0.0, 1.0)
        ys = float(y.sum())
        if ys > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - hi, 0.0, 1.0)
    # correct residual
    res = S - float(y.sum())
    if abs(res) > 1e-8:
        if res > 0:
            idx = np.argmax(1.0 - y)
        else:
            idx = np.argmax(y)
        room = (1.0 - y[idx]) if res > 0 else y[idx]
        adj = res if abs(res)<=room else (room if room>=0 else -room)
        y[idx] += adj
    return np.clip(y,0.0,1.0)

        else:
            hi = mid

    y = np.clip(x - hi, 0.0, 1.0)
    for _ in range(10):
        res = float(S) - float(y.sum())
        if abs(res) < 1e-14:
            break
        if res > 0.0:
            room = 1.0 - y
            idx = int(np.argmax(room))
            adj = min(res, float(room[idx]))
        else:
            idx = int(np.argmax(y))
            adj = max(res, -float(y[idx]))
        if adj == 0.0:
            break
        y[idx] += adj
    return np.clip(y, 0.0, 1.0)


def enforce_exact_sum(h):
    h = np.asarray(h, dtype=float).ravel()
    S = 0.5 * float(h.size)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        res = S - float(h.sum())
        if abs(res) < 1e-14:
            break
        if res > 0.0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            adj = min(res, float(room[idx]))
        else:
            idx = int(np.argmax(h))
            adj = max(res, -float(h[idx]))
        if adj == 0.0:
            break
        h[idx] += adj
    return np.clip(h, 0.0, 1.0)


def validate_candidate(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return h, float("inf")
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, 0.5 * float(h.size))
    h = enforce_exact_sum(h)
    c = manual_c5(h)
    return h, c


def recompute_state(h):
    h = np.asarray(h, dtype=float).ravel()
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, 0.5 * float(h.size))
    h = enforce_exact_sum(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    C = float(np.max(c) * (2.0 / float(h.size)))
    return h, c, C


def start_smooth(n, seed):
    if n <= 0:
        return np.array([], dtype=float)
    if n == 1:
        return np.full(1, 0.5, dtype=float)
    rng = np.random.default_rng(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 * np.ones(n, dtype=float)
    amps = (0.35, 0.18, 0.08)
    for k in range(3):
        amp = amps[k] * float(rng.uniform(0.5, 1.2))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        h = h + amp * np.sin(2.0 * np.pi * (k + 1) * x + phase)
    h += float(rng.uniform(-0.05, 0.05))
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, 0.5 * float(n))


def start_runs(n, seed):
    if n <= 0:
        return np.array([], dtype=float)
    rng = np.random.default_rng(seed)
    h = np.zeros(n, dtype=float)
    pos = 0
    bit = int(rng.integers(0, 2))
    while pos < n:
        run = int(rng.integers(2, 6))
        if run < 1:
            run = 1
        end = min(n, pos + run)
        if bit:
            h[pos:end] = 1.0
        pos = end
        bit = 1 - bit
    return project_box_sum(h, 0.5 * float(n))


def start_random_binary(n, seed):
    if n <= 0:
        return np.array([], dtype=float)
    rng = np.random.default_rng(seed)
    h = np.zeros(n, dtype=float)
    S = int(round(0.5 * float(n)))
    if 0 < S < n:
        idxs = rng.choice(n, S, replace=False)
        h[idxs] = 1.0
    elif S >= n:
        h[:] = 1.0
    return project_box_sum(h, 0.5 * float(n))


def parent_to_n(parent, n, mode):
    p = np.asarray(parent, dtype=float).ravel()
    N = p.size
    if n <= 0:
        return np.array([], dtype=float)
    if N == 0:
        return np.full(n, 0.5, dtype=float)
    if N == 1:
        h = np.full(n, float(p[0]), dtype=float)
    elif n == N:
        h = p.copy()
    elif mode == "linear":
        xp = np.linspace(0.0, float(N - 1), n)
        h = np.interp(xp, np.arange(N, dtype=float), p)
    else:
        src = np.floor(np.arange(n) * float(N) / float(n)).astype(np.int64)
        src = np.clip(src, 0, N - 1)
        h = p[src]
    h = np.clip(h, 0.0, 1.0)
    h = project_box_sum(h, 0.5 * float(n))
    return h


def active_gradient(h, c, M=6):
    n = h.size
    L = c.size
    if n <= 0 or L <= 0:
        return np.zeros(n, dtype=float)
    center = (L - 1) // 2
    k = int(M)
    if k < 1:
        k = 1
    if k > L:
        k = L
    if k == 1:
        top = np.array([int(np.argmax(c))], dtype=int)
    else:
        top = np.argpartition(-c, k - 1)[:k]
    g = np.zeros(n, dtype=float)
    w = 1.0 / float(k)
    for m in top:
        mi = int(m)
        kk = mi - center
        if kk == 0:
            g += w * (1.0 - 2.0 * h)
        else:
            d = abs(int(kk))
            if kk > 0:
                g[d:] += w
            else:
                g[:-d] += w
            g[d:] -= w * h[:-d]
            g[:-d] -= w * h[d:]
    return g


def coordinate_search(h0, c0, C0, time_end, seed):
    n = int(h0.size)
    if n <= 1 or time.time() >= time_end:
        return h0.copy(), float(C0)

    S = 0.5 * float(n)
    dx = 2.0 / float(n)
    L = 2 * n - 1
    center = n - 1

    h = np.asarray(h0, dtype=float).copy()
    c = np.asarray(c0, dtype=float).copy()
    C = float(C0)

    best_h = h.copy()
    best_c = C
    oh_rev = (1.0 - h)[::-1].copy()
    rng = np.random.default_rng(seed)
    lin = np.zeros(L, dtype=float)
    c_test = np.empty(L, dtype=float)

    no_improve = 0
    perturb_fail = 0

    while time.time() < time_end:
        c = np.correlate(h, 1.0 - h, mode="full")
        C = float(np.max(c) * dx)
        if C < best_c:
            best_c = C
            best_h = h.copy()
        oh_rev = (1.0 - h)[::-1].copy()

        if time.time() >= time_end:
            break

        order = [int(v) for v in rng.permutation(n)]
        improved = False
        pair_ct = 0

        for ii in order:
            i = ii
            if time.time() >= time_end:
                break
            hi = h[i]
            gain_room = 1.0 - hi
            if gain_room <= 0.0:
                continue
            for jj in order:
                j = jj
                if i == j:
                    continue
                hj = h[j]
                maxd = min(gain_room, hj)
                maxd -= 1e-12
                if maxd <= 0.0:
                    continue

                a = i
                b = j

                lin.fill(0.0)
                lin[a:a + n] += oh_rev
                lin[center - a:center - a + n] -= h
                lin[b:b + n] -= oh_rev
                lin[center - b:center - b + n] += h

                d = b - a
                idx1 = center + d
                idx2 = center - d

                best_delta = 0.0
                bestC = C
                for frac in COORD_FRACTIONS:
                    delta = frac * maxd
                    if delta <= 0.0:
                        continue
                    c_test[:] = c
                    c_test += delta * lin
                    if d:
                        delta2 = delta * delta
                        c_test[idx1] += delta2
                        c_test[idx2] += delta2
                    val = float(np.max(c_test) * dx)
                    if val < bestC:
                        bestC = val
                        best_delta = delta

                if best_delta > 0.0 and bestC < C - 1e-12:
                    c += best_delta * lin
                    if d:
                        delta2 = best_delta * best_delta
                        c[idx1] += delta2
                        c[idx2] += delta2

                    h[a] += best_delta
                    h[b] -= best_delta
                    oh_rev[center - a] -= best_delta
                    oh_rev[center - b] += best_delta

                    if h[a] > 1.0:
                        h[a] = 1.0
                        oh_rev[center - a] = 0.0
                    elif h[a] < 0.0:
                        h[a] = 0.0
                        oh_rev[center - a] = 1.0
                    if h[b] > 1.0:
                        h[b] = 1.0
                        oh_rev[center - b] = 0.0
                    elif h[b] < 0.0:
                        h[b] = 0.0
                        oh_rev[center - b] = 1.0

                    C = bestC
                    improved = True
                    if C < best_c:
                        best_c = C
                        best_h = h.copy()

                pair_ct += 1
                if (pair_ct & 8191) == 0 and time.time() >= time_end:
                    break

        if improved:
            no_improve = 0
            perturb_fail = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                h = h + 0.02 * rng.normal(size=n)
                h = project_box_sum(h, S)
                c2 = np.correlate(h, 1.0 - h, mode="full")
                C2 = float(np.max(c2) * dx)
                if C2 > best_c + 0.05:
                    h = best_h.copy()
                    c = np.correlate(h, 1.0 - h, mode="full")
                    C = float(np.max(c) * dx)
                else:
                    c = c2
                    C = C2
                no_improve = 0
                perturb_fail += 1
                if perturb_fail >= 5:
                    break

    hrec, crec, Creccalc = recompute_state(best_h)
    if Creccalc < best_c:
        best_h = hrec
        best_c = Creccalc
    return best_h, best_c


def stochastic_search(h0, c0, C0, time_end, seed):
    n = int(h0.size)
    if n <= 1 or time.time() >= time_end:
        return h0.copy(), float(C0)

    S = 0.5 * float(n)
    dx = 2.0 / float(n)
    L = 2 * n - 1
    center = n - 1

    h = np.asarray(h0, dtype=float).copy()
    c = np.asarray(c0, dtype=float).copy()
    C = float(C0)

    best_h = h.copy()
    best_c = C
    oh_rev = (1.0 - h)[::-1].copy()

    rng = np.random.default_rng(seed)
    lin = np.zeros(L, dtype=float)
    c_test = np.empty(L, dtype=float)

    T = max(1e-5, 0.5 * dx)
    T0 = T
    low = None
    high = None

    it = 0
    no_improve = 0
    recompute_ct = 0

    while time.time() < time_end:
        if it % 2000 == 0:
            g = active_gradient(h, c, 6)
            q_lo = float(np.quantile(g, 0.25))
            q_hi = float(np.quantile(g, 0.75))
            low = np.nonzero(g <= q_lo)[0]
            high = np.nonzero(g >= q_hi)[0]
            if low.size < 4:
                take = max(2, min(8, n))
                low = np.argsort(g)[:take]
            if high.size < 4:
                take = max(2, min(8, n))
                high = np.argsort(g)[-take:]

        if low is not None and low.size > 0 and high.size > 0 and rng.random() < 0.75:
            a = int(low[int(rng.integers(low.size))])
            b = int(high[int(rng.integers(high.size))])
        else:
            a = int(rng.integers(n))
            b = int(rng.integers(n))

        if a == b:
            b = int(rng.integers(n))
            if a == b:
                b = (b + 1) % n

        m1 = min(1.0 - h[a], h[b])
        m2 = min(1.0 - h[b], h[a])
        if m2 > m1:
            a, b = b, a
            maxd = m2
        else:
            maxd = m1

        maxd -= 1e-12
        if maxd <= 0.0:
            it += 1
            continue

        lin.fill(0.0)
        lin[a:a + n] += oh_rev
        lin[center - a:center - a + n] -= h
        lin[b:b + n] -= oh_rev
        lin[center - b:center - b + n] += h

        d = b - a
        idx1 = center + d
        idx2 = center - d

        best_delta = 0.0
        bestC = C
        for frac in STOCH_FRACTIONS:
            delta = frac * maxd
            if delta <= 0.0:
                continue
            c_test[:] = c
            c_test += delta * lin
            if d:
                delta2 = delta * delta
                c_test[idx1] += delta2
                c_test[idx2] += delta2
            val = float(np.max(c_test) * dx)
            if val < bestC:
                bestC = val
                best_delta = delta

        accept = best_delta > 0.0 and bestC < C - 1e-12
        if (not accept) and best_delta > 0.0 and (bestC - C) < 3.0 * dx and T > 1e-6:
            if math.exp((C - bestC) / T) > rng.random():
                accept = True

        if accept:
            c += best_delta * lin
            if d:
                delta2 = best_delta * best_delta
                c[idx1] += delta2
                c[idx2] += delta2

            h[a] += best_delta
            h[b] -= best_delta
            oh_rev[center - a] -= best_delta
            oh_rev[center - b] += best_delta

            if h[a] > 1.0:
                h[a] = 1.0
                oh_rev[center - a] = 0.0
            elif h[a] < 0.0:
                h[a] = 0.0
                oh_rev[center - a] = 1.0
            if h[b] > 1.0:
                h[b] = 1.0
                oh_rev[center - b] = 0.0
            elif h[b] < 0.0:
                h[b] = 0.0
                oh_rev[center - b] = 1.0

            C = bestC
            if C < best_c:
                best_c = C
                best_h = h.copy()
            no_improve = 0
            T = max(1e-6, T * 0.999)
        else:
            no_improve += 1
            if no_improve % 5000 == 0:
                T = max(1e-6, T * 0.9)
            if no_improve > 50000:
                h = h + 0.03 * rng.normal(size=n)
                h = project_box_sum(h, S)
                c2 = np.correlate(h, 1.0 - h, mode="full")
                C2 = float(np.max(c2) * dx)
                if C2 > best_c + 0.05:
                    h = best_h.copy()
                    c = np.correlate(h, 1.0 - h, mode="full")
                    C = float(np.max(c) * dx)
                else:
                    c = c2
                    C = C2
                no_improve = 0
                T = T0

        recompute_ct += 1
        if recompute_ct % 5000 == 0:
            c = np.correlate(h, 1.0 - h, mode="full")
            C = float(np.max(c) * dx)
            oh_rev = (1.0 - h)[::-1].copy()
            if C < best_c:
                best_c = C
                best_h = h.copy()

        it += 1
        if (it & 4095) == 0 and time.time() >= time_end:
            break

    hrec, crec, Creccalc = recompute_state(best_h)
    if Creccalc < best_c:
        best_h = hrec
        best_c = Creccalc
    return best_h, best_c


def local_search(h0, time_end, seed):
    h, c, C = recompute_state(h0)
    best_h = h.copy()
    best_c = C
    n = int(h.size)

    if time.time() >= time_end:
        return best_h, best_c

    remain = max(0.0, time_end - time.time())
    if n <= FULL_COORD_THRESHOLD and remain > 4.0:
        coord_end = time.time() + 0.85 * remain
        bh, bc = coordinate_search(h, c, C, coord_end, seed)
        if bc < best_c:
            best_c = bc
            best_h = bh.copy()

    if time.time() < time_end:
        h2, c2, C2 = recompute_state(best_h)
        bh, bc = stochastic_search(h2, c2, C2, time_end, seed + 101)
        if bc < best_c:
            best_c = bc
            best_h = bh.copy()

    vh, vc = validate_candidate(best_h)
    if vc <= best_c + 1e-12:
        return vh, vc
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    usable = max(1.0, float(budget_s) - 1.0)
    reserve = max(1.0, min(60.0, 0.08 * usable))
    optim_end = t0 + max(1.0, usable - reserve)
    end_all = t0 + usable

    start_records = []
    idx = 0
    parent_work = None

    parent = globals().get("initial_h_values")
    if parent is not None:
        pv = np.asarray(parent, dtype=float).ravel()
        if pv.size > 0:
            hp, hc = validate_candidate(pv)
            if math.isfinite(hc):
                start_records.append((float(hc), idx, int(hp.size), hp))
                idx += 1
            parent_work = pv

    Np = int(parent_work.size) if parent_work is not None else 80
    base = Np
    if base < 32:
        base = 32
    if base % 2:
        base += 1

    n_set = set()
    n_list_initial = (base, base * 2, base * 3, base * 4, base * 5, base * 6)
    for n in n_list_initial:
        if 32 <= n <= 512 and n % 2 == 0:
            n_set.add(int(n))
    for n in (64, 80, 96, 128, 160, 240, 320, 480):
        if 32 <= n <= 512 and n % 2 == 0:
            n_set.add(int(n))
    n_list = sorted(n_set)

    for n in n_list:
        if idx > 80:
            break

        if parent_work is not None:
            hrep = parent_to_n(parent_work, n, "repeat")
            hv, cv = validate_candidate(hrep)
            if math.isfinite(cv):
                start_records.append((float(cv), idx, n, hv))
                idx += 1

            if n != Np:
                hlin = parent_to_n(parent_work, n, "linear")
                hv, cv = validate_candidate(hlin)
                if math.isfinite(cv):
                    start_records.append((float(cv), idx, n, hv))
                    idx += 1

        if n >= 16:
            hr = start_random_binary(n, int(seed) + 1000 * idx)
            hv, cv = validate_candidate(hr)
            if math.isfinite(cv):
                start_records.append((float(cv), idx, n, hv))
                idx += 1

            hs = start_smooth(n, int(seed) + 2000 * idx)
            hv, cv = validate_candidate(hs)
            if math.isfinite(cv):
                start_records.append((float(cv), idx, n, hv))
                idx += 1

            hu = start_runs(n, int(seed) + 3000 * idx)
            hv, cv = validate_candidate(hu)
            if math.isfinite(cv):
                start_records.append((float(cv), idx, n, hv))
                idx += 1

    if not start_records:
        h0 = start_smooth(80, int(seed))
        hv, cv = validate_candidate(h0)
        start_records.append((float(cv), 0, 80, hv))

    start_records.sort()
    active = start_records[:10]

    best_h = None
    best_c = float("inf")
    best_n = 0
    for rec in active:
        if rec[0] < best_c:
            best_c = float(rec[0])
            best_h = rec[3]
            best_n = int(rec[2])

    chunk_list = (60.0, 30.0, 20.0, 10.0, 5.0)
    cycle = 0
    while cycle < len(chunk_list) and time.time() < optim_end:
        base_chunk = chunk_list[cycle]
        for ai, rec in enumerate(active):
            if time.time() >= optim_end:
                break
            c0 = float(rec[0])
            if c0 > best_c + 0.08 and cycle > 0:
                continue
            h0 = rec[3]
            rem = optim_end - time.time()
            chunk = min(base_chunk, rem)
            if h0.size > 400:
                chunk = min(chunk, 25.0)
            if chunk < 3.0:
                continue
            seed_use = int(seed) + 1000 * cycle + 17 * ai
            hopt, copt = local_search(h0, time.time() + chunk, seed_use)
            hval, cval = validate_candidate(hopt)
            if math.isfinite(cval) and cval < best_c:
                best_c = float(cval)
                best_h = hval
                best_n = int(hval.size)
        cycle += 1

    if best_h is None:
        rec = active[0]
        best_h = rec[3]
        best_c = float(rec[0])
        best_n = int(rec[2])

    best_h, best_c_final = validate_candidate(best_h)
    if best_c_final < best_c:
        best_c = best_c_final
    else:
        best_c = best_c_final

    if time.time() < end_all - 1.0:
        pol_h, pol_c = local_search(best_h, end_all, int(seed) + 777001)
        pol_h, pol_c = validate_candidate(pol_h)
        if math.isfinite(pol_c) and pol_c < best_c:
            best_c = float(pol_c)
            best_h = pol_h

    best_h, best_c = validate_candidate(best_h)
    best_n = int(best_h.size)
    if not math.isfinite(best_c):
        best_c = 0.38085
    return best_h, float(best_c), best_n
