_deadline = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_fun(x):
    return compute_c5(x)

def con_fun(x):
    return x.sum() - _N_CUR / 2.0

def con_jac(x):
    return np.ones(_N_CUR)

def set_current_n(n):
    global _N_CUR
    _N_CUR = n

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):

    np.random.seed(seed)

        best_h = h_init.copy()

def corr_jac_row(m, h, n):
    u = 1.0 - h
    d = m - n + 1
    row = np.zeros(n)
    if d >= 0:
        # u[j-d] valid for j >= d
        row[d:] = u[:n-d]
        # h[j+d] valid for j < n-d
        row[:n-d] -= (1.0 - u)[d:]  # h[j+d] = h[j+d], for j in 0..n-d-1
        # Wait, h[j+d] for j in range(n-d) is h[d:n]
        row[:n-d] -= h[d:]

        # d < 0, let d' = -d
        dprime = -d
        # u[j-d] = u[j+d'] valid for j < n-d'
        row[:n-dprime] = u[dprime:]
        # h[j+d] = h[j-d'] valid for j >= d'
        row[dprime:] -= h[:n-dprime]
    return row
