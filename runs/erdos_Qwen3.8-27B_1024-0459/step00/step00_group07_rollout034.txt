```python
import numpy as np
import time

def fix_sum(h, target_sum):
    """Fix the sum of h to be exactly target_sum while keeping 0<=h<=1."""
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            amt = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + amt, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            amt = diff / np.sum(mask)  # amt is negative
            h[mask] = np.maximum(h[mask] + amt, 0.0)
    return h

def eval_c5(h, dx):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
   corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def coordinate_descent(h, n, dx, time_limit, rng):
    """Optimize h using coordinate descent with sum constraint."""
    desired_sum = n / 2.0
    c_full = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(c_full) * dx
    t_start = time.time()
    
    tau = np.arange(-(n-1), n)
    
    passes = 0
    while time.time() - t_start < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            old_v = h[j]
            if old_v <= 1e-15 or old_v >= 1.0 - 1e-15:
                continue
            
            # Compute D_tau
            idx1 = j + tau
            idx2 = j - tau
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            D = np.zeros(2*n-1)
            D[valid1] = 1.0 - h[idx1[valid1].astype(int)]
            D[valid2] -= h[idx2[valid2].astype(int)]
            
            # B = c_full - old_v * D
            B = c_full - old_v * D
            
            # Find v in [0,1] minimizing max(B + v*D)
            # Ternary search
            lo, hi = 0.0, 1.0
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(B + m1 * D)
                f2 = np.max(B + m2 * D)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            v_new = (lo + hi) / 2.0
            
            # Also check endpoints
            f_0 = np.max(B)  # v=0
            f_1 = np.max(B + D)  # v=1
            f_mid = np.max(B + v_new * D)
            
            min_f = min(f_0, f_1, f_mid)
            if min_f == f_0:
                v_new = 0.0
            elif min_f == f_1:
                v_new = 1.0
            
            if abs(v_new - old_v) > 1e-10:
                h[j] = v_new
                c_full = B + v_new * D
                improved = True
        
        # Fix sum
        h = fix_sum(h, desired_sum)
        c_full = np.correlate(h, 1.0 - h, mode='full')
        
        current_c5 = np.max(c_full) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = initial_h_values.copy()
    best_n = n_init

        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 30  # 30s buffer
        if remaining < 30:
            break
        
        dx = 2.0 / n
        desired_sum = n / 2.0
        
        # Scale parent
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, n_init), initial_h_values)
        h = h * (desired_sum / np.sum(h))
        h = np.clip(h, 0, 1)
        h = fix_sum(h, desired_sum)
        
        # Coordinate descent
        alloc = remaining * 0.4
        h, c5 = coordinate_descent(h.copy(), n, dx, alloc, rng)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        
        # Random perturbation + re-optimize

                best_h = h2.copy()
                best_n = n
    
    # Final check
    return best_h, best_c5, best_n
```
