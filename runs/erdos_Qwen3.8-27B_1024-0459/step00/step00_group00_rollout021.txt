    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = (1.0 - h)
            mask = room > 1e-15
            if not mask.any():
                break
            h[mask] += room[mask] * (diff / room[mask].sum())
        else:
            avail = h.copy()
            mask = avail > 1e-15
            if not mask.any():
                break
            h[mask] -= avail[mask] * (-diff / avail[mask].sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_old, n_new):
    idx = (np.arange(n_new) * n_old) // n_new
    idx = np.clip(idx, 0, n_old - 1)
    return h[idx].copy()

def make_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    h_bin = np.sort(h)[::-1]  # largest first
    target = n // 2
    if n % 2 == 0:
        h_bin[:target] = 1.0
        h_bin[target:] = 0.0
    else:
        h_bin[:target] = 1.0
        h_bin[target] = 1.0  # or 0.5
        h_bin[target+1:] = 0.0
        # One 0.5 for the fractional part
    # Actually for odd n, sum = n/2 which is x.5
    # So we need target integers summing to n/2
    # E.g., n=45: need sum=22.5. 22 ones + 1 half + 22 zeros.
    if n % 2 == 1:
        half = (n - 1) // 2  # 22 for n=45
        h_bin[:half] = 1.0
        h_bin[half] = 0.5
        h_bin[half+1:] = 0.0
    return h_bin

def slsqp_optimize(h, n, maxiter=500):
    h_start = repair(h.copy(), n)
    res = minimize(
        compute_c5, h_start,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    if res.x is not None and np.all(np.isfinite(res.x)):
        h_opt = repair(res.x, n)
        return h_opt, compute_c5(h_opt)
    return h_start, compute_c5(h_start)

def coord_desc_simple(h, n, max_sweeps=30):
    """Simple coordinate descent trying a few values per coordinate."""
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            old = h[j]
            best_obj = compute_c5(h)
            best_val = old
            for val in [0.0, 1.0, old * 0.5 if old > 0 else 0.5, 
                        old * 1.5 if old < 1 else 0.5, 0.5]:
                val = np.clip(val, 0, 1)
                if abs(val - old) < 1e-10:
                    continue
                h[j] = val
                h = repair(h, n)
                obj = compute_c5(h)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = val
                else:
                    h[j] = best_val
                    h = repair(h, n)
            h[j] = best_val
            h = repair(h, n)
            if best_val != old:
                improved = True
        if not improved:
            break
    return h, compute_c5(h)

            h[mask] += room[mask] * (diff / total)
