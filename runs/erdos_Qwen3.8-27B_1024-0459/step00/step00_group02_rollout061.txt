    n = len(h)
    grad = np.zeros(n)
    # First term: (1-h[m-d]) for valid m-d
    # m ranges over [0,n-1], m-d ranges over [0,n-1]
    # If d >= 0: m-d valid for m in [d, n-1], value is (1-h[m-d])
    # If d < 0: m-d valid for m in [0, n-1+d], value is (1-h[m-d]) = (1-h[m+|d|])
    
    if d >= 0:

        grad[dd:] -= h[:n-dd]  # m from dd to n-1, m+d=m-dd from 0 to n-1-dd
    
    return grad

import numpy as np
import time

def grad_for_lag(h, d):
    """Compute gradient of corr at lag d with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        if d < n:
            grad[d:] += (1.0 - h[:n - d])
            grad[:n - d] -= h[d:]
    else:
        dd = -d
        if dd < n:
            grad[:n - dd] += (1.0 - h[dd:])
            grad[dd:] -= h[:n - dd]
    return grad

def soft_max_weighted_grad(h, n, eps):
    """Compute the soft-max weighted gradient of the max correlation."""
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')  # shape (2n-1,)
    corr_vals = corr * dx
    
    # Soft-max weights
    max_val = np.max(corr_vals)
    shifted = corr_vals - max_val
    weights = np.exp(shifted / eps)
    total_w = np.sum(weights)
    weights /= total_w
    
    # Compute weighted gradient
    grad = np.zeros(n)
    n_minus_1 = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        d = k - n_minus_1
        grad += w * grad_for_lag(h, d)
    
    return grad * dx, max_val

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    
    current_sum = np.sum(h)
    if abs(current_sum - target) < 1e-12:
        return h
    
    # Binary search for uniform shift
    lo, hi = -1.0, 1.0
    # Ensure bracketing
    while np.sum(np.clip(h + lo, 0, 1)) > target:
        lo *= 2
    while np.sum(np.clip(h + hi, 0, 1)) < target:
        hi *= 2
    
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    c = 0.5 * (lo + hi)
    h_proj = np.clip(h + c, 0.0, 1.0)
    
    # Fine adjustment
    residual = target - np.sum(h_proj)
    if abs(residual) > 1e-14:
        h_proj += residual / n
        h_proj = np.clip(h_proj, 0.0, 1.0)
    
    return h_proj

def evaluate_c5(h, n):

    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    return np.max(corr) * dx

def upsample_arr(arr, new_n):
    """Upsample arr to new_n points using linear interpolation."""
    old_n = len(arr)
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

        inits.append(np.clip(0.5 + 0.35 * np.sin(2 * np.pi * 2 * xs), 0, 1))
        inits.append(np.clip(0.5 + 0.35 * np.cos(2 * np.pi * 3 * xs) + 0.1 * np.sin(2 * np.pi * 5 * xs), 0, 1))

        per_init = time_budget / len(inits)

        for ii, hinit in enumerate(inits):
            rem = budget_s - (time.time() - t0)
            if rem < 30:
                break
            tl = min(per_init * 1.3, rem * 0.95)
            ho, co = _optimize(hinit, n, tl, t0, rng)
            if co < bc:
                bh = ho.copy()
                bc = co
                bn = n

        rem = budget_s - (time.time() - t0)
        if rem > 80 and bn == n:
            tl = min(rem * 0.3, 180)
            ho, co = _local_refine(bh, n, tl, t0, rng)
            if co < bc:
                bh = ho.copy()
                bc = co

        # Second optimization pass on best found so far at this resolution
        rem = budget_s - (time.time() - t0)
        if rem > 60 and bn == n:
            tl = min(rem * 0.4, 200)
            ho, co = _optimize(bh, n, tl, t0, rng)
            if co < bc:
                bh = ho.copy()
                bc = co

    # Final local refinement
    rem = budget_s - (time.time() - t0)
    if rem > 30 and bh is not None:
        tl = min(rem * 0.5, 150)
        ho, co = _local_refine(bh, bn, tl, t0, rng)
        if co < bc:
            bh = ho.copy()
            bc = co

    # Verification
    if bh is not None:
        v = _score(bh, bn)
        if v > 1.0:
            if hp is not None:
                bh = hp.copy()
                bc = _score(hp, len(hp))
                bn = len(hp)
            else:
                n_fb = 45
                bh = np.ones(n_fb)
                bh[n_fb // 2:] = 0.0
                bc = _score(bh, n_fb)
                bn = n_fb

    return bh, bc, bn
