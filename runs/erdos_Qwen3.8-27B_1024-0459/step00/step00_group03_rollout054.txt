import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    return np.max(corr) * dx

def compute_c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    c5 = np.max(corr) * dx
    
    # Find all argmax positions for subgradient
    max_val = np.max(corr)
    argmax_positions = np.where(corr >= max_val - 1e-10)[0]
    
    grad = np.zeros(n)
    for j in argmax_positions:
        k = j - (n - 1)
        for m in range(n):
            g = 0.0
            idx1 = m - k
            if 0 <= idx1 < n:
                g += 1.0 - h[idx1]
            idx2 = m + k
            if 0 <= idx2 < n:
                g -= h[idx2]
            grad[m] += g
    grad *= dx / len(argmax_positions)
    
    return c5, grad

def project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        mid = (lo + hi) / 2
        h_try = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_try)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

    # Try to use initial_h_values
    try:
        h_parent = np.array(initial_h_values, dtype=np.float64)
        n_parent = len(h_parent)
    except (NameError, UnboundLocalError, TypeError):
        h_parent = None
        n_parent = 0

    if h_parent is not None:
        h_parent = project_to_feasible(h_parent, n_parent)
        c5_p = compute_c5(h_parent)
        best_c5 = c5_p
        best_h = h_parent.copy()
        best_n = n_parent

        # Optimize parent
        t_alloc = deadline(0.20) - elapsed()
        if t_alloc > 5:
            h_opt, c5_opt = pgd_coarse_to_fine(h_parent, n_parent, time_limit=t_alloc)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_parent

    # Try multiple resolutions
    resolutions = [80, 100, 120, 140, 160, 200, 250, 300]
    for n in resolutions:
        if elapsed() > deadline(0.82):
            break
        t_remaining = deadline(0.82) - elapsed()
        if t_remaining < 20:
            break

        # Start from upsampled best
        if best_h is not None and len(best_h) < n:
            h_start = upsample_h(best_h, n)
        elif best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_start = project_to_feasible(np.interp(x_new, x_old, best_h), n)
        else:
            h_start = random_smooth_init(n, seed + n)

        t_alloc = t_remaining * 0.35
        h_opt, c5_opt = pgd_coarse_to_fine(h_start, n, time_limit=t_alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

        # Random binary start
        if elapsed() < deadline(0.82) and deadline(0.82) - elapsed() > 15:
            h_rb = random_binary_init(n, seed + n * 7 + 3)
            t_alloc2 = deadline(0.82) - elapsed()
            if t_alloc2 > 10:
                h_opt2, c5_opt2 = pgd(h_rb, n, lr=0.03, n_iters=3000,
                                      time_limit=t_alloc2 * 0.5)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
                    best_n = n

        # Random smooth start
        if elapsed() < deadline(0.82) and deadline(0.82) - elapsed() > 10:
            h_rs = random_smooth_init(n, seed + n * 11 + 5)
            t_alloc3 = deadline(0.82) - elapsed()
            if t_alloc3 > 8:
                h_opt3, c5_opt3 = pgd(h_rs, n, lr=0.02, n_iters=2000,
                                      time_limit=t_alloc3 * 0.6)
                if c5_opt3 < best_c5:
                    best_c5 = c5_opt3
                    best_h = h_opt3.copy()
                    best_n = n

    # Final refinement phase
    if best_h is not None and elapsed() < deadline(0.95):
        n_best = len(best_h)
        t_rem = deadline(0.95) - elapsed()

        # Fine PGD
        if t_rem > 10:
            h_fin, c5_fin = pgd(best_h, n_best, lr=0.005, n_iters=10000,
                                time_limit=t_rem * 0.4)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

        # Annealing
        if elapsed() < deadline(0.97):
            t_rem2 = deadline(0.97) - elapsed()
            if t_rem2 > 15:
                h_ann, c5_ann = anneal_and_reoptimize(best_h, n_best,
                                                      seed + 999, t_rem2 * 0.7)
                if c5_ann < best_c5:
                    best_c5 = c5_ann
                    best_h = h_ann.copy()

        # Binary push and re-optimize
        if elapsed() < deadline(0.99):
            t_rem3 = deadline(0.99) - elapsed()
            if t_rem3 > 10:
                h_bin = (best_h > 0.45).astype(np.float64)
                h_bin = project_to_feasible(h_bin, n_best)
                h_bin_opt, c5_bin = pgd(h_bin, n_best, lr=0.02, n_iters=3000,
                                        time_limit=t_rem3 * 0.6)
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin_opt.copy()

    # Fallback
    if best_h is None:
        n = 70
        rng = np.random.RandomState(seed)
        best_h = project_to_feasible(rng.uniform(0, 1, n), n)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final verification
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
