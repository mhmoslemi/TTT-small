import numpy as np
from scipy import optimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_sum(v, target_sum):
    """Project v onto {x: sum(x)=target_sum, 0<=x<=1} via bisection on Lagrange multiplier."""
    if target_sum <= 0:
        return np.zeros_like(v)
    if target_sum >= len(v):
        return np.ones_like(v)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if abs(s - target_sum) < 1e-14:
            return p
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5*(lo+hi), 0.0, 1.0)

def _objective_and_grad(x, dx):
    """Compute C5(x) and subgradient."""
    h = x
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    obj = corr[max_idx] * dx
    s = max_idx - (N - 1)
    grad = np.zeros(N)
    if s >= 0:
        for n in range(N - s):
            i = n + s
            grad[i] += (1.0 - h[n])
            grad[n] -= h[i]
    else:
        t = -s
        for n in range(t, N):
            i = n - t
            grad[n] += (1.0 - h[i])
            grad[i] -= h[n]
    grad *= dx
    return obj, grad

def _grad_descent(h, dx, target_sum, max_iters=5000, lr=0.01, momentum=0.9):
    """Projected gradient descent with momentum."""
    N = len(h)
    x = h.copy()
    v = np.zeros(N)
    best_x = h.copy()
    best_f = _compute_c5(h, dx)
    
    for it in range(max_iters):
        _, grad = _objective_and_grad(x, dx)
        # Update velocity
        v = momentum * v - lr * grad
        # Gradient step
        x_new = x + v
        # Project
        x_new = _project_sum(x_new, target_sum)
        x_new = np.clip(x_new, 0.0, 1.0)
        # Ensure sum constraint
        x_new = _project_sum(x_new, target_sum)
        x = x_new
        
        f = _compute_c5(x, dx)
        if f < best_f:
            best_f = f
            best_x = x.copy()
    
    return best_x, best_f

def _scipy_optimize(h, dx, target_sum, timeout=60):
    """Use scipy SLSQP to optimize."""
    N = len(h)
    start = time.time()
    
    def fun(x):
        return _compute_c5(x, dx)
    
    def jac(x):
        _, g = _objective_and_grad(x, dx)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(len(x))}]
    bounds = [(0.0, 1.0)] * N
    
    result = optimize.minimize(fun, h, jac=jac, method='SLSQP',
                                bounds=bounds, constraints=constraints,
                                options={'maxiter': 500, 'ftol': 1e-10, 'iprint': 0})
    
    if result.x is not None:
        # Ensure constraints
        result.x = np.clip(result.x, 0.0, 1.0)
        result.x = _project_sum(result.x, target_sum)
        if _compute_c5(result.x, dx) < result.fun:
            # Prefer the projected version if it's actually better
            pass
        return result.x, _compute_c5(result.x, dx)
    return h, _compute_c5(h, dx)

def _interpolate(h, n_new):
    """Interpolate h from its resolution to n_new points."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_interp = np.interp(x_new, x_old, h)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return _project_sum(h_interp, n_new / 2)

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.default_rng(seed)

import numpy as np
from scipy import optimize
import time


def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _project_sum(v, target_sum):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    if target_sum >= n:
        return np.ones(n)
    lo, hi = -5.0, 5.0
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if abs(s - target_sum) < 1e-13:
            return p
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def _obj_and_grad(x, dx):
    h = x
    N = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_idx = int(np.argmax(corr))
    obj = corr[max_idx] * dx
    s = max_idx - (N - 1)
    grad = np.zeros(N)
    if s >= 0:
        grad[s:] += (1.0 - h[:N - s])
        grad[:N - s] -= h[s:]
    else:
        t = -s
        grad[:N - t] += (1.0 - h[t:])
        grad[t:] -= h[:N - t]
    grad *= dx
    return obj, grad


def _constraint_sum(x, target_sum):
    return np.sum(x) - target_sum


def _constraint_sum_jac(x):
    return np.ones(len(x))


def _scipy_slsqp(h0, dx, target_sum, maxiter=300):
    N = len(h0)
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': lambda x: _constraint_sum(x, target_sum),
             'jac': lambda x: _constraint_sum_jac(x)}]

    def fun(x):
        return _compute_c5(x, dx)

    def jac(x):
        _, g = _obj_and_grad(x, dx)
        return g

    res = optimize.minimize(fun, h0, method='SLSQP', jac=jac,
                            bounds=bounds, constraints=cons,
                            options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
    if res.success or res.fun < fun(h0):
        x = np.clip(res.x, 0.0, 1.0)
        x = _project_sum(x, target_sum)
        return x, _compute_c5(x, dx)
    return h0, _compute_c5(h0, dx)


def _grad_descent(h0, dx, target_sum, max_iters=5000, lr0=0.005, momentum=0.85):
    N = len(h0)
    x = h0.copy()
    v = np.zeros(N)
    best_x = h0.copy()
    best_f = _compute_c5(x, dx)
    stall = 0

    for it in range(max_iters):
        _, grad = _obj_and_grad(x, dx)
        lr = lr0 / (1.0 + 0.01 * it)
        v = momentum * v - lr * grad
        x_new = x + v
        x_new = _project_sum(x_new, target_sum)
        x = x_new
        f = _compute_c5(x, dx)
        if f < best_f - 1e-12:
            best_f = f
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
    return best_x, best_f


def _interpolate(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_interp = np.interp(x_new, x_old, h)
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return _project_sum(h_interp, n_new / 2.0)


def _smooth_max_obj(x, dx, alpha=50.0):
    """Smooth approximation of max via log-sum-exp for use with BFGS."""
    h = x
    N = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    shifted = corr * dx
    m = np.max(shifted)
    return m + np.log(np.sum(np.exp(alpha * (shifted - m)))) / alpha


def _run(sc=42, budget_s=1300, **kw):
    start_time = time.time()
    rng = np.random.default_rng(sc)

    def elapsed():
        return time.time() - start_time

    def has_time(remaining=30.0):
        return elapsed() < budget_s - remaining

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(160, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h_init = np.clip(h_init, 0.0, 1.0)
    h_init = _project_sum(h_init, target_sum)

    best_h = h_init.copy()
    best_c5 = _compute_c5(best_h, dx)
    best_n = n

    # Phase 1: SLSQP at initial resolution
    if has_time(60):
        h1, c5_1 = _scipy_slsqp(best_h, dx, target_sum, maxiter=400)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Gradient descent refinement
    if has_time(60):
        h2, c5_2 = _grad_descent(best_h, dx, target_sum, max_iters=4000, lr0=0.004, momentum=0.85)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Try higher resolutions
    for mult in [2, 4, 6]:
        n_new = n * mult
        if n_new > 1200:
            break
        if not has_time(120):
            break
        dx_new = 2.0 / n_new
        ts_new = n_new / 2.0
        h_new = _interpolate(best_h, n_new)
        c5_new = _compute_c5(h_new, dx_new)

        # SLSQP at new resolution
        if has_time(60):
            h_new, c5_new = _scipy_slsqp(h_new, dx_new, ts_new, maxiter=200)

        # Gradient descent
        if has_time(60):
            h_new, c5_new = _grad_descent(h_new, dx_new, ts_new, max_iters=3000, lr0=0.003, momentum=0.85)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_new
            n = best_n
            dx = 2.0 / n
            target_sum = n / 2.0

    # Phase 4: More SLSQP on best
    dx_best = 2.0 / best_n
    ts_best = best_n / 2.0
    if has_time(60):
        h4, c5_4 = _scipy_slsqp(best_h, dx_best, ts_best, maxiter=500)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Random restarts with perturbation
    if has_time(90):
        n_tries = 0
        while has_time(20) and n_tries < 8:
            n_tries += 1
            scale = 0.05 + 0.03 * n_tries
            h_pert = best_h + rng.normal(0, scale, best_n)
            h_pert = _project_sum(h_pert, ts_best)
            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = _project_sum(h_pert, ts_best)

            h_r, c5_r = _scipy_slsqp(h_pert, dx_best, ts_best, maxiter=150)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Final gradient descent polish
    if has_time(30):
        h6, c5_6 = _grad_descent(best_h, dx_best, ts_best, max_iters=3000, lr0=0.002, momentum=0.8)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / len(best_h)
    best_h = _project_sum(best_h, len(best_h) / 2.0)
    best_c5 = _compute_c5(best_h, dx_final)

    return best_h, best_c5, len(best_h)


def run(seed=42, budget_s=1300, **kwargs):
    return _run(seed=seed, budget_s=budget_s, **kwargs)
