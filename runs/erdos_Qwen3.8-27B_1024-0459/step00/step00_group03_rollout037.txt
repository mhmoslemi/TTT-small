def project_sum(h, target):

    lo, hi = -2.0, 2.0

def top_k_grad(h, n, k=3, temp=20.0):

    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    cv = corr * dx

def random_init(n, seed, style="smooth"):
