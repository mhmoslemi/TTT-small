    shifted = np.where(shifted < -50, -50, shifted)
    return m + np.log(np.sum(np.exp(shifted))) / temp

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum} using water-filling."""
    h = np.clip(h, 0, 1).copy()
    # If sum is already close, return
    if abs(np.sum(h) - target_sum) < 1e-10:
        return h
    # Bisection on tau
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def objective_smoothed(h, n, temp, penalty_w):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    obj = smooth_max(vals, temp)
    s = np.sum(h) - n / 2.0
    obj += penalty_w * s * s
    return obj

def gradient_smoothed(h, n, temp, penalty_w):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = temp * (vals - m)
    shifted = np.where(shifted < -50, -50, shifted)
    w = temp * np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient: d/dh[j] of sum_k w_k * dx * corr[k]
    # corr[k] = sum_i h[i]*(1-h[i+k-n+1]) ... alignment matters
    # Using the formula: d corr[k]/dh[j] = dx*(1-2*h[k-j+n-1]) approximately
    # More precisely, use: grad[j] = sum_k w[k] * dx * (1 - 2*h[j + (k - n + 1)])
    # where the index j+(k-n+1) must be in [0, n-1]
    
    # Let's use the shift interpretation:
    # In np.correlate(h, 1-h, 'full'), index k corresponds to shift k-(n-1)
    # Let shift s = k - (n-1), so s ranges from -(n-1) to (n-1)
    # corr[k] = sum_{i: valid} h[i] * (1 - h[i+s])
    # d/dh[j]: (1-h[j+s]) if 0<=j+s<n, minus h[j-s] if 0<=j-s<n
    # = (1-h[j+s])*1[valid] - h[j-s]*1[valid]
    
    # Efficient computation via convolution:
    # grad[j] = sum_s w[s+n-1] * dx * [(1-h[j+s]) - h[j-s]]
    # = dx * sum_s w[s+n-1] * 1 - dx * sum_s w[s+n-1]*h[j+s] - dx * sum_s w[s+n-1]*h[j-s]
    # Let W = sum w = 1 (since w sums to 1)
    # term1 = dx * 1
    # term2 = dx * convolve(w, h)[j + n - 1]  (linear conv, appropriate index)
    # term3 = dx * convolve(w_rev, h)[j]  (where w_rev reverses w)
    
    # Actually let's be more careful. Use np.convolve for linear convolution.
    # conv_w_h = np.convolve(w, h, mode='full') has length 2n-1
    # conv_w_rev_h = np.convolve(w[::-1], h, mode='full') has length 2n-1
    
    conv_wh = np.convolve(w, h, mode='full')  # length 2n-1
    w_rev = w[::-1].copy()
    conv_wrev_h = np.convolve(w_rev, h, mode='full')  # length 2n-1
    
    # For index j in [0, n-1]:
    # (w * h)[j] in the sense of sum_s w[s]*h[j+s] corresponds to conv_wh[j + n - 1]
    # sum_s w[s]*h[j-s] = sum_s w_rev[s]*h[j+s] ... this is getting confusing
    
    # Let me just do it directly for clarity (n is at most ~360, so O(n^2) is fine)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            s = k - (n - 1)  # shift
            # Contribution from h[j] as the first factor: (1 - h[j+s]) if j+s in [0,n-1]
            js = j + s
            if 0 <= js < n:
                g += (1.0 - h[js])
            # Contribution from h[j] as the second factor: -h[j-s] if j-s in [0,n-1]
            js2 = j - s
            if 0 <= js2 < n:
                g -= h[js2]
        grad[j] = w[k] * 0  # placeholder
        # Actually we need to weight by w[k]
        grad[j] = 0.0
    
    # Redo properly
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            s = k - (n - 1)
            js = j + s
            contrib = 0.0
            if 0 <= js < n:
                contrib += (1.0 - h[js])
            js2 = j - s
            if 0 <= js2 < n:
                contrib -= h[js2]
            g += w[k] * contrib
        grad[j] = dx * g
    
    # Add penalty gradient
    s_sum = np.sum(h) - n / 2.0
    grad += 2.0 * penalty_w * s_sum * np.ones(n)
    
    return grad

def gradient_smoothed_fast(h, n, temp, penalty_w):
    """Faster gradient using the structure of the problem."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = temp * (vals - m)
    shifted = np.where(shifted < -50, -50, shifted)
    w = temp * np.exp(shifted) / np.sum(np.exp(shifted))
    
    # The gradient at position j is:
    # dx * sum_k w[k] * [(1-h[j+(k-n+1)])*1[0<=j+(k-n+1)<n] - h[j-(k-n+1)]*1[0<=j-(k-n+1)<n]]
    
    # Let u[k] = w[k] for k in [0, 2n-2]
    # First term: sum_k u[k] * (1-h[j+(k-n+1)]) where j+(k-n+1) in [0,n-1]
    #   = sum_{s: 0<=j+s<n} u[s+n-1] * (1-h[j+s])
    #   = sum_s' u[j+s'-n+1]... 
    
    # Let's use convolution:
    # A[j] = sum_{valid s} w[s+n-1] * (1-h[j+s])
    # B[j] = sum_{valid s} w[s+n-1] * h[j-s]
    # grad[j] = dx * (A[j] - B[j])
    
    # A[j] = sum_s u[s] * (1-h[j+s]) where u[s] = w[s+n-1] for s in [-(n-1), n-1]
    # This is a convolution of u with (1-h), evaluated at appropriate indices.
    
    # Let's just compute efficiently:
    # u has 2n-1 elements: u[k] = w[k] for k = 0,...,2n-2
    # A[j] = sum_{k: 0<=j+(k-n+1)<n} u[k] * (1-h[j+(k-n+1)])
    # Let s = k-(n-1), so k = s+n-1, s ranges from -(n-1) to (n-1)
    # A[j] = sum_{s: 0<=j+s<n} u[s+n-1] * (1-h[j+s])
    # = sum_{j+s in [0,n-1]} u[s+n-1] * (1-h[j+s])
    
    # This is (1-h) convolved with u, at index j (with u centered)
    # Specifically: A = convolve(1-h, u_centered) where u_centered is u padded
    
    u = np.zeros(2 * n - 1)
    u[:] = w[:]
    
    # A[j] = sum_s u[s+n-1] * (1-h[j+s]) for valid j+s
    # = (u * (1-h))[j + n - 1] in full convolution? Let me check.
    # np.convolve(a, b, 'full')[k] = sum_i a[i]*b[k-i]
    # So np.convolve(1-h, u, 'full')[j+n-1] = sum_i (1-h)[i]*u[j+n-1-i]
    # Let s = j+n-1-i, i = j+n-1-s
    # = sum_s (1-h[j+n-1-s])*u[s] for valid s
    # Hmm, not quite what I want.
    
    # Let me try: np.convolve(u, 1-h, 'full')[j + n - 1]
    # = sum_i u[i]*(1-h)[j+n-1-i]
    # Let s = n-1-(j+n-1-i) = i-j... no.
    # = sum_i u[i]*(1-h[j+n-1-i])
    # Let t = n-1-i, i = n-1-t
    # = sum_t u[n-1-t]*(1-h[j+t]) for valid t
    # = sum_t u_rev[t]*(1-h[j+t]) where u_rev = u[::-1]
    
    # I want: sum_s u[s+n-1]*(1-h[j+s]) for 0<=j+s<n
    # This is: (u_shifted * (1-h))[j] where u_shifted[s] = u[s+n-1]
    # In np.convolve(a,b,'full')[j] = sum_i a[i]*b[j-i]
    # If a = u (length 2n-1), b = 1-h (length n):
    # np.convolve(u, 1-h, 'full')[j + n - 1] = sum_i u[i]*(1-h)[j+n-1-i]
    # Let s = j+n-1-i: i = j+n-1-s
    # = sum_s u[j+n-1-s]*(1-h)[s] ... not right either
    
    # I'll just do it with a loop for correctness since n <= 360
    # O(n * (2n-1)) = O(n^2) which is fine for n=360 (about 260K ops)
    
    grad = np.zeros(n)
    # Precompute: for each shift s = -(n-1), ..., (n-1), the weight is w[s+n-1]
    for j in range(n):
        acc = 0.0
        for k in range(2 * n - 1):
            s = k - (n - 1)
            # First part: h[j] as first factor in h[i]*(1-h[i+s]), j=takes role of i
            # (1-h[j+s]) if 0<=j+s<n
            idx1 = j + s
            if 0 <= idx1 < n:
                acc += w[k] * (1.0 - h[idx1])
            # Second part: h[j] as second factor, h[j-s]*(1-h[j-s+s])... 
            # Actually: in corr[k] = sum_i h[i]*(1-h[i+s]), if j = i+s then h[j] appears
            # d/dh[j] of h[i+s] = 1 when i+s=j, i.e., i=j-s
            # term: h[j-s]*(1-h[j-s+s]) ... no
            # d/dh[j] of (1-h[i+s]) = -1 when i+s=j, i.e., i=j-s
            # So: -h[j-s] if 0<=j-s<n
            idx2 = j - s
            if 0 <= idx2 < n:
                acc -= w[k] * h[idx2]
        grad[j] = dx * acc
    
    s_sum = np.sum(h) - n / 2.0
    grad += 2.0 * penalty_w * s_sum
    
    return grad

def optimize_h(h_init, n, temp_start, temp_end, penalty_w, max_iter, time_limit, seed=42):
    """Optimize h using projected gradient with backtracking."""
    rng = np.random.RandomState(seed)
    h = np.array(h_init, dtype=float).copy()
    h = project_feasible(h, n / 2.0)
    best_h = h.copy()
    best_obj = compute_c5(h)
    t0 = time.time()
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        # Anneal temperature
        frac = it / max(max_iter - 1, 1)
        temp = temp_start * (temp_end / temp_start) ** frac
        temp = max(temp, 0.01)
        
        obj = objective_smoothed(h, n, temp, penalty_w)
        grad = gradient_smoothed_fast(h, n, temp, penalty_w)
        
        # Backtracking line search
        step = 1.0
        for _ in range(30):
            new_h = project_feasible(h - step * grad, n / 2.0)
            new_obj = objective_smoothed(new_h, n, temp, penalty_w)
            if new_obj < obj - 1e-4 * step * np.dot(grad, grad):
                break
            step *= 0.5
        else:
            new_h = h.copy()
            step = 0.0
        
        h = new_h.copy()
        
        # Check actual C5
        c5 = compute_c5(h)
        if c5 < best_obj:
            best_obj = c5
            best_h = h.copy()
        
        # Small random perturbation to escape plateaus
        if it > 0 and it % 50 == 0:
            pert = rng.uniform(-0.01, 0.01, n)
            h = project_feasible(h + pert, n / 2.0)
    
    return best_h, best_obj

def coordinate_descent(h, n, max_rounds=200, time_limit=60.0, seed=0):
    """Coordinate descent to polish the solution."""
    rng = np.random.RandomState(seed)
    h = np.array(h, dtype=float).copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in rng.permutation(n):
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            # Try moving h[j] in small steps
            # The effect on corr: changing h[j] by delta changes
            # corr[k] by delta*(1-h[j+s_k]) - h[j-s_k]*delta for relevant k
            # For simplicity, try several values
            for delta in [0.05, -0.05, 0.02, -0.02, 0.1, -0.1, 0.01, -0.01]:
                new_val = old + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[j] = new_val
                # Renormalize approximately (adjust one other element)
                diff = np.sum(h) - n / 2.0
                if abs(diff) < 1e-10:
                    c5 = compute_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        improved = True
                    else:
                        h[j] = old
                else:
                    h[j] = old
            # Renormalize
            h = project_feasible(h, n / 2.0)
        # Final projection
        h = project_feasible(h, n / 2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            # Try again with smaller steps
            if rnd > 50:
                break
    
    return h, best_c5

def interpolate_h(h_old, n_old, n_new):
    """Interpolate h from n_old to n_new points."""
    if n_new == n_old:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        h0 = np.full(45, 0.5)
        for i in range(len(h0) // 2):
            h0[i] = 1.0
    n0 = len(h0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at original resolution
    if time_left() > 60:
        # Gradient-based optimization at n0
        t_budget = min(time_left() * 0.2, 120)
        h_opt, c5_opt = optimize_h(
            h0, n0, temp_start=20.0, temp_end=0.1,
            penalty_w=1000.0, max_iter=2000, time_limit=t_budget, seed=seed
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        # Coordinate descent polish
        h_cd, c5_cd = coordinate_descent(best_h, n0, max_rounds=100, time_limit=min(time_left()*0.1, 30), seed=seed+1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    # Phase 2: Try higher resolutions
    resolutions = []
    for mult in [2, 3, 4, 6, 8]:
        n_try = n0 * mult
        if n_try not in resolutions:
            resolutions.append(n_try)
    # Also add some specific values
    for nv in [60, 90, 120, 180, 240, 360]:
        if nv not in resolutions and nv > n0:
            resolutions.append(nv)
    resolutions = sorted(resolutions)
    
    for n in resolutions:
        if time_left() < 30:
            break
        if n > 720:  # Cap resolution
            continue
        
        # Interpolate from best
        h_interp = interpolate_h(best_h, len(best_h), n)
        h_interp = project_feasible(h_interp, n / 2.0)
        
        # Also try from original h0
        h_interp2 = interpolate_h(h0, n0, n)
        h_interp2 = project_feasible(h_interp2, n / 2.0)
        
        # Optimize at this resolution
        t_budget = min(time_left() * 0.4, 180)
        max_iter = min(3000, max(500, int(t_budget * 2)))
        
        h_opt1, c5_opt1 = optimize_h(
            h_interp, n, temp_start=15.0, temp_end=0.1,
            penalty_w=500.0, max_iter=max_iter, time_limit=t_budget*0.5, seed=seed+n
        )
        
        h_opt2, c5_opt2 = optimize_h(
            h_interp2, n, temp_start=15.0, temp_end=0.1,
            penalty_w=500.0, max_iter=max_iter, time_limit=t_budget*0.5, seed=seed+n+100
        )
        
        # Pick better of the two
        if c5_opt1 <= c5_opt2:
            h_curr, c5_curr = h_opt1, c5_opt1
        else:
            h_curr, c5_curr = h_opt2, c5_opt2
        
        # Coordinate descent
        if time_left() > 20:
            h_cd, c5_cd = coordinate_descent(h_curr, n, max_rounds=80, time_limit=min(time_left()*0.15, 30), seed=seed+n+200)
            if c5_cd < c5_curr:
                h_curr, c5_curr = h_cd, c5_cd
        
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr
            # Update parent for next resolution
            h0 = best_h.copy()
            n0 = n
    
    # Phase 3: Final polish at the best resolution
    if time_left() > 10:
        n_best = len(best_h)
        t_polish = min(time_left() * 0.5, 60)
        # Gradient step with low temperature
        h_pol, c5_pol = optimize_h(
            best_h, n_best, temp_start=5.0, temp_end=0.01,
            penalty_w=2000.0, max_iter=2000, time_limit=t_polish*0.5, seed=seed+999
        )
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
        
        # Coordinate descent
        h_cd, c5_cd = coordinate_descent(best_h, n_best, max_rounds=150, time_limit=t_polish*0.5, seed=seed+888)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    # Phase 4: Random restarts if time remains
    if time_left() > 30:
        n_best = len(best_h)
        for restart in range(5):
            if time_left() < 15:
                break
            # Perturb best solution
            rng = np.random.RandomState(seed + 500 + restart)
            h_pert = best_h + rng.normal(0, 0.05, n_best)
            h_pert = project_feasible(h_pert, n_best / 2.0)
            
            t_b = min(time_left() * 0.3, 60)
            h_r, c5_r = optimize_h(
                h_pert, n_best, temp_start=10.0, temp_end=0.1,
                penalty_w=1000.0, max_iter=1500, time_limit=t_b, seed=seed+500+restart
            )
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
    
    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
