```
def balance_binary(h):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n == 0: return h
    target = n // 2
    s = int(np.rint(float(h.sum())))
    diff = s - target
    if diff == 0:
       # round near binary
       if np.all(np.abs(h - np.round(h)) < 1e-12):
          h = np.round(h)
       return h
    need_type = 1.0 if diff > 0 else 0.0
    flips = abs(diff)
    positions = []
    for a,b in ((1,2),(1,4),(3,4),(5,8),(3,8),(1,8),(7,8),(1,10),(9,10)):
        positions.append(int((a*n)//b))
    selected=[]
    for p in positions:
       if 0 <= p < n and abs(h[p]-need_type) < 0.5 and p not in selected:
          selected.append(p)
          if len(selected) == flips: break
    allp = np.where(np.abs(h - need_type) < 0.5)[0]
    if allp.size:
       distance = np.abs(allp - (n // 2))
       order = np.argsort(distance)
       for idx in order:
          p = int(allp[idx])
          if p not in selected:
             selected.append(p)
          if len(selected) == flips: break
    if len(selected) < flips:
       # last resort: mass adjust
       return make_feasible(h)
    h[selected[:flips]] = 1.0 - h[selected[:flips]]
    h = np.where(h > 0.5, 1.0, 0.0)
    return h
```

```
import numpy as np
import time

_TAPS_CACHE = {}

def mask_sort_key(x):
    return (x.bit_count(), x)

def get_parent_h():
    try:
        h = np.asarray(initial_h_values, dtype=float)
    except Exception:
        return None
    if h.size == 0 or not np.all(np.isfinite(h)):
        return None
    return h

def make_feasible(h):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(50):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / float(mask.sum())
        else:
            h += diff / float(n)
        h = np.clip(h, 0.0, 1.0)
    return h

def objective_fft(h, dx=None):
    arr = np.asarray(h, dtype=np.float64)
    n = arr.size
    if n == 0:
        return float('inf')
    if dx is None:
        dx = 2.0 / float(n)
    m = 2*n - 1
    size = 1
    while size < m:
        size <<= 1
    fa = np.fft.rfft(arr, size)
    fb = np.fft.rfft(1.0 - arr, size)
    c = np.fft.irfft(fa * fb, size)
    if c.size > m:
        c = c[:m]
    if c.size == 0:
        return float('inf')
    # numerical clip to nonnegative for safety
    mn = float(np.min(c))
    if mn < 0.0 and mn > -1e-8:
        c = np.clip(c, 0.0, None)
    return float(np.max(c) * dx)

def objective_exact(h):
    arr = np.asarray(h, dtype=np.float64)
    n = arr.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    if n > 20000:
        return objective_fft(arr, dx)
    c = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(c) * dx)

def evaluate_exact(h):
    hf = make_feasible(h)
    c = objective_exact(hf)
    return hf, float(c)

def period_of_lfsr(m, feedback_mask):
    init = (1 << m) - 1
    state = init
    L = (1 << m) - 1
    top_bit = 1 << (m - 1)
    for step in range(1, L + 1):
        fb = (state & feedback_mask).bit_count() & 1
        state = (state >> 1) | (fb * top_bit)
        if state == init:
            return step
    return L + 1

def find_primitive_taps(m):
    global _TAPS_CACHE
    if m in _TAPS_CACHE:
        return _TAPS_CACHE[m]
    L = (1 << m) - 1
    max_extra = 1 << (m - 1)
    options = list(range(1, max_extra))
    options.sort(key=mask_sort_key)

def generate_msequence_bits(m, taps, length):
    fm = 0
    for d in taps:
        fm |= 1 << d
    state = (1 << m) - 1
    top_bit = 1 << (m - 1)
    out = np.empty(length, dtype=np.uint8)
    for i in range(length):
        out[i] = state & 1
        fb = (state & fm).bit_count() & 1
        state = (state >> 1) | (fb * top_bit)
    return out

def balance_binary(h):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n == 0:
        return h
    target = n // 2
    s = int(np.rint(float(h.sum())))
    diff = s - target
    if diff == 0:
        if np.all(np.abs(h - np.round(h)) < 1e-12):
            h = np.round(h)
        return h
    need_type = 1.0 if diff > 0 else 0.0
    flips = abs(diff)
    positions = []
    for a, b in ((1,2),(1,4),(3,4),(5,8),(3,8),(1,8),(7,8),(1,10),(9,10)):
        positions.append(int((a*n)//b))
    selected = []
    for p in positions:
        if 0 <= p < n and abs(h[p] - need_type) < 0.5 and p not in selected:
            selected.append(p)
            if len(selected) == flips:
                break
    allp = np.where(np.abs(h - need_type) < 0.5)[0]
    if allp.size:
        center = n // 2
        order = np.argsort(np.abs(allp - center))
        for idx in order:
            p = int(allp[idx])
            if p not in selected:
                selected.append(p)
            if len(selected) == flips:
                break
    if len(selected) < flips:
        return make_feasible(h)
    h[selected[:flips]] = 1.0 - h[selected[:flips]]
    h = np.where(h > 0.5, 1.0, 0.0)
    return h

def build_msequence_candidates(m, max_phases=4):
    taps = find_primitive_taps(m)
    if taps is None:
        return []
    L = (1 << m) - 1
    bits = generate_msequence_bits(m, taps, L)
    out = []
    phases = min(int(max_phases), L)
    for phase in range(phases):
        pp = int(phase)
        period = np.concatenate((bits[pp:], bits[:pp])).copy()
        period = np.where(period > 0, 1.0, 0.0)
        # repeated twice, then balanced
        rep = np.tile(period, 2).astype(np.float64)
        rep = balance_binary(rep)
        out.append(rep)
        # anti-periodic and exactly balanced
        anti = np.concatenate((period, 1.0 - period)).astype(np.float64)
        anti = np.where(anti > 0.5, 1.0, 0.0)
        out.append(anti)
    return out

def random_balanced_binary(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(int(n), dtype=np.float64)
    h[:int(n)//2] = 1.0
    rng.shuffle(h)
    return h
```

```python
import numpy as np
import time

_TAPS_CACHE = {}


def mask_sort_key(x):
    return (x.bit_count(), x)


def get_parent_h():
    try:
        h = np.asarray(initial_h_values, dtype=float)
    except Exception:
        return None
    if h.size == 0 or not np.all(np.isfinite(h)):
        return None
    return h


def make_feasible(h):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n == 0:
        return h

    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(50):
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break

        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / float(mask.sum())
        else:
            h += diff / float(n)
        h = np.clip(h, 0.0, 1.0)

    return h


def objective_fft(h, dx=None):
    arr = np.asarray(h, dtype=np.float64)
    n = arr.size
    if n == 0:
        return float("inf")

    if dx is None:
        dx = 2.0 / float(n)

    m = 2 * n - 1
    size = 1
    while size < m:
        size <<= 1

    # np.correlate(a, b) = np.convolve(a, b[::-1])
    b = (1.0 - arr)[::-1]
    fa = np.fft.rfft(arr, size)
    fb = np.fft.rfft(b, size)
    c = np.fft.irfft(fa * fb, size)

    if c.size > m:
        c = c[:m]
    if c.size == 0:
        return float("inf")

    # Remove tiny negative numerical noise.
    if float(np.min(c)) < 0.0:
        c = np.clip(c, 0.0, None)

    return float(np.max(c) * dx)


def objective_exact(h):
    arr = np.asarray(h, dtype=np.float64)
    n = arr.size
    if n == 0:
        return float("inf")

    dx = 2.0 / float(n)

    if n > 20000:
        return objective_fft(arr, dx)

    c = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(c) * dx)


def evaluate_exact(h):
    hf = make_feasible(h)
    c = objective_exact(hf)
    return hf, float(c)


def pad_to_next_power_of_two(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)

    if n <= 1:
        return np.array([0.0, 1.0], dtype=float)

    new_n = 1 << (n - 1).bit_length()
    if new_n == n:
        return h.copy()

    target = 0.5 * float(new_n)
    need = target - float(h.sum())
    cap = new_n - n

    pad = np.full(cap, 0.5, dtype=float)
    diff = need - 0.5 * float(cap)

    i = 0
    while abs(diff) > 1e-12 and i < cap:
        if diff > 0.0:
            add = min(0.5, diff)
            pad[i] += add
            diff -= add
        else:
            sub = min(0.5, -diff)
            pad[i] -= sub
            diff += sub
        i += 1

    if abs(diff) > 1e-12:
        pad += diff / float(cap)
        pad = np.clip(pad, 0.0, 1.0)

    return np.concatenate((h, pad))


def period_of_lfsr(m, feedback_mask):
    init = (1 << m) - 1
    state = init
    L = (1 << m) - 1
    top_bit = 1 << (m - 1)

    for step in range(1, L + 1):
        fb = (state & feedback_mask).bit_count() & 1
        state = (state >> 1) | (fb * top_bit)
        if state == init:
            return step

    return L + 1


def find_primitive_taps(m, deadline=None):
    global _TAPS_CACHE

    if m in _TAPS_CACHE:
        return _TAPS_CACHE[m]

    if deadline is not None and time.monotonic() >= deadline:
        return None

    L = (1 << m) - 1
    max_extra = 1 << (m - 1)
    options = list(range(1, max_extra))
    options.sort(key=mask_sort_key)

    found = None
    timed_out = False

    for extra in options:
        if deadline is not None and time.monotonic() >= deadline:
            timed_out = True
            break

        fm = 1 | (extra << 1)
        if period_of_lfsr(m, fm) == L:
            found = [d for d in range(m) if (fm >> d) & 1]
            break

    if not timed_out:
        _TAPS_CACHE[m] = found

    return found


def generate_msequence_bits(m, taps, length):
    fm = 0
    for d in taps:
        fm |= 1 << d

    state = (1 << m) - 1
    top_bit = 1 << (m - 1)
    out = np.empty(length, dtype=np.uint8)

    for i in range(length):
        out[i] = state & 1
        fb = (state & fm).bit_count() & 1
        state = (state >> 1) | (fb * top_bit)

    return out


def balance_binary(h):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n == 0:
        return h

    target = n // 2
    s = int(np.rint(float(h.sum())))
    diff = s - target

    if diff == 0:
        if np.all(np.abs(h - np.round(h)) < 1e-12):
            h = np.round(h)
        return h

    need_type = 1.0 if diff > 0 else 0.0
    flips = abs(diff)

    positions = []
    for a, b in ((1, 2), (1, 4), (3, 4), (5, 8), (3, 8), (1, 8), (7, 8), (1, 10), (9, 10)):
        positions.append(int((a * n) // b))

    selected = []
    for p in positions:
        if 0 <= p < n and abs(h[p] - need_type) < 0.5 and p not in selected:
            selected.append(p)
            if len(selected) == flips:
                break

    allp = np.where(np.abs(h - need_type) < 0.5)[0]
    if allp.size:
        center = n // 2
        order = np.argsort(np.abs(allp - center))
        for idx in order:
            p = int(allp[idx])
            if p not in selected:
                selected.append(p)
            if len(selected) == flips:
                break

    if len(selected) < flips:
        return make_feasible(h)

    h[selected[:flips]] = 1.0 - h[selected[:flips]]
    h = np.where(h > 0.5, 1.0, 0.0)
    return h


def build_msequence_candidates(m, max_phases=4):
    taps = find_primitive_taps(m)
    if taps is None:
        return []

    L = (1 << m) - 1
    bits = generate_msequence_bits(m, taps, L)
    out = []
    phases = min(int(max_phases), L)

    for phase in range(phases):
        pp = int(phase)
        period = np.concatenate((bits[pp:], bits[:pp])).copy()
        period = np.where(period > 0, 1.0, 0.0)

        # Two identical periods, then exact balance correction.
        rep = np.tile(period, 2).astype(np.float64)
        rep = balance_binary(rep)
        out.append(pad_to_next_power_of_two(rep))

        # Antiperiodic two-block sequence: exactly balanced without correction.
        anti = np.concatenate((period, 1.0 - period)).astype(np.float64)
        anti = np.where(anti > 0.5, 1.0, 0.0)
        out.append(pad_to_next_power_of_two(anti))

    return out


def random_balanced_binary(n, seed):
    rng = np.random.default_rng(int(seed))
    n = int(n)
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h


def random_candidate_search(n, trials, seed, deadline):
    n = int(n)
    best_h = None
    best_c = float("inf")

    for t in range(int(trials)):
        if time.monotonic() >= deadline:
            break

        hh = random_balanced_binary(n, int(seed) + 9973 + t)
        c = objective_fft(hh, 2.0 / float(n))
        if np.isfinite(c) and c < best_c:
            best_c = c
            best_h = hh
            if c < 0.2502:
                break

    if best_h is None:
        return None, float("inf")

    hh, cc = evaluate_exact(best_h)
    return hh, cc


def merge_best(new_h, new_c, best_h, best_c, best_n):
    if new_h is None or not np.isfinite(new_c):
        return best_h, best_c, best_n

    new_h = np.asarray(new_h, dtype=float)
    if new_h.size == 0:
        return best_h, best_c, best_n

    if new_c < best_c - 1e-15:
        return new_h, float(new_c), int(new_h.size)

    return best_h, best_c, best_n
```
