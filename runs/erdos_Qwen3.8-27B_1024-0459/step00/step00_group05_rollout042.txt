import numpy as np
import time
from scipy.optimize import minimize
import scipy

def compute_c5(h, n_points):
    """Compute C5 for a given h vector."""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def project_h(h, n_points):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        h = h / s * target
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment: if still not exact, adjust individual elements
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Need to add diff to some elements
            order = np.argsort(-h)  # elements with most room (closest to 1 first? No, closest to 0)
            order = np.argsort(h)  # elements closest to 0 first
            for idx in order:
                room = 1.0 - h[idx]
                add = min(room, diff)
                h[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(-h)  # elements closest to 1 first
            for idx in order:
                room = h[idx]
                sub = min(room, -diff)
                h[idx] -= sub
                diff += sub
                if abs(diff) <= 1e-12:
                    break
    return h

def smooth_max(h, n_points, epsilon=0.01):
    """Smooth approximation of max of correlation values."""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    # Log-sum-exp for smooth max
    m = np.max(conv)
    if m > 10 * epsilon:
        return m + epsilon * (np.log(np.sum(np.exp((conv - m) / epsilon))) )
    else:
        return m + epsilon * (np.log(np.sum(np.exp(conv / epsilon))) - np.log(len(conv)))

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new points."""
    # Linear interpolation

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_h(h_new, n_new)

def project_h(h, n_points):

def corr_grad_single(h, k, n):
    """Gradient of c_k = sum_i h[i]*（1-h[i+k]) with respect to h."""
    g = np.zeros(n)

            g[i + k] += (1.0 - h[i])
            g[i] -= h[i + k]

def smooth_max_grad(h, n, epsilon):

    dx = 2.0 / n
