def make_perturbation(n, seed, bandwidth=8):

def smooth_gradient(h, f, tau, n):

       f2=np.zeros(L); k=min(f.size,L); f2[:k]=f[:k]; f=f2

def quantize_exact(h, n):
    n=int(n)
    if n <=0: return np.zeros(0)
    h=np.asarray(h,float)
    if h.size != n: h=upsample_h(h,n)
    Q = 1 << _QUANT_BITS
    if n % 2 == 0:
        targetI = (n // 2) * Q
    else:
        targetI = (n * Q) // 2
    targetI = int(targetI)
    h = np.clip(h, 0.0, 1.0)
    q = np.rint(h * float(Q)).astype(np.int64)
    q = np.clip(q, 0, Q)
    s = int(np.sum(q))
    diff = targetI - s
    if diff > 0:
        cap = Q - q
        order = np.argsort(-cap)
        for idx in order:
            if diff <= 0: break
            add = int(diff)
            c = int(cap[idx])
            if add > c: add = c
            if add > 0:
                q[idx] += add
                diff -= add
    elif diff < 0:
        excess = int(-diff)
        order = np.argsort(-q)
        for idx in order:
            if excess <= 0: break
            rem = int(q[idx])
            if rem > 0:
                sub = excess if excess < rem else rem
                q[idx] -= sub
                excess -= sub
    return q.astype(np.float64) / float(Q)

def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return x

    target = float(target)
    y = np.clip(x, 0.0, 1.0)
    s = float(np.sum(y))

    if abs(s - target) <= 1.0e-10:
        return y

    err = target - s
    z = np.clip(y + err / float(x.size), 0.0, 1.0)
    if abs(float(np.sum(z)) - target) <= 1.0e-8:
        return z

    lo = -1.0
    hi = 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        z = np.clip(y + mid, 0.0, 1.0)
        zs = float(np.sum(z))
        if zs < target:
            lo = mid
        else:
            hi = mid

    return np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)


def upsample_h(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)

    m = int(h.size)
    if m == n:
        return h.copy()
    if m == 0:
        return np.full(n, 0.5, dtype=float)

    x0 = (np.arange(m, dtype=float) + 0.5) * (2.0 / float(m))
    x1 = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    out = np.interp(x1, x0, h, left=float(h[0]), right=float(h[-1]))
    return np.clip(out, 0.0, 1.0)


def exact_evaluate(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if n <= 0 or h.size != n:
        return _INF
    if not np.all(np.isfinite(h)):
        return _INF

    c = np.correlate(h, 1.0 - h, mode="full")
    if c.size == 0:
        return _INF
    return float(np.max(c)) * (2.0 / float(n))


def fast_c_values(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)

    L = 2 * n - 1
    use_fft = _HAS_FFT and n > 96
    if use_fft:
        c = signal.fftconvolve(h, (1.0 - h)[::-1], mode="full")
    else:
        c = np.correlate(h, 1.0 - h, mode="full")

    c = np.real(np.asarray(c, dtype=float))
    if c.size != L:
        tmp = np.zeros(L, dtype=float)
        k = min(c.size, L)
        if k > 0:
            tmp[:k] = c[:k]
        c = tmp

    c = np.maximum(c, 0.0)
    return c * (2.0 / float(n))


def smooth_gradient(h, f, tau, n):
    h = np.asarray(h, dtype=float)
    f = np.asarray(f, dtype=float)
    n = int(n)

    L = 2 * n - 1
    if f.size != L:
        tmp = np.zeros(L, dtype=float)
        k = min(f.size, L)
        if k > 0:
            tmp[:k] = f[:k]
        f = tmp

    if f.size == 0:
        return None, np.zeros(n, dtype=float), 0.0

    m = float(np.max(f))
    if tau <= 0.0:
        tau = 1.0e-12

    low = m - 60.0 * tau
    ex = np.zeros_like(f, dtype=float)
    mask = f >= low
    ex[mask] = np.exp((f[mask] - m) / tau)
    z = float(np.sum(ex))

    if not np.isfinite(z) or z <= 0.0:
        P = np.full(f.size, 1.0 / float(f.size), dtype=float)
        smooth_val = m
    else:
        P = ex / z
        smooth_val = m + tau * math.log(z)

    use_fft = _HAS_FFT and n > 96
    if use_fft:
        A = signal.fftconvolve(1.0 - h, P, mode="full")
        B = signal.fftconvolve(h, P[::-1], mode="full")
    else:
        A = np.convolve(1.0 - h, P, mode="full")
        B = np.convolve(h, P[::-1], mode="full")

    A = np.real(np.asarray(A, dtype=float))
    B = np.real(np.asarray(B, dtype=float))

    start = n - 1
    stop = 2 * n - 1
    ga = A[start:stop]
    gb = B[start:stop]

    if ga.size != n:
        tmp = np.zeros(n, dtype=float)
        k = min(ga.size, n)
        if k > 0:
            tmp[:k] = ga[:k]
        ga = tmp
    else:
        ga = ga[:n]

    if gb.size != n:
        tmp = np.zeros(n, dtype=float)
        k = min(gb.size, n)
        if k > 0:
            tmp[:k] = gb[:k]
        gb = tmp
    else:
        gb = gb[:n]

    grad = (2.0 / float(n)) * (ga - gb)
    return P, grad, smooth_val


def harmonic_h(n, mode, amp, phase=0.0):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)

    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    h = 0.5 + float(amp) * np.sin(math.pi * float(mode) * x + float(phase))
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, 0.5 * float(n))


def random_smooth_h(n, seed):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)

    rng = np.random.default_rng(seed)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)

    max_mode = max(2, min(16, n // 2))
    modes = []
    for _ in range(3):
        mode = int(rng.integers(1, max_mode + 1))
        trials = 0
        while mode in modes and trials < 12:
            mode = int(rng.integers(1, max_mode + 1))
            trials += 1
        modes.append(mode)

        amp = float(rng.uniform(0.12, 0.38))
        phase = float(rng.uniform(-math.pi, math.pi))
        h += amp * np.sin(math.pi * float(mode) * x + phase)

    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, 0.5 * float(n))


def choose_initial(n, seed, parent):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)

    if parent is not None:
        parent = np.asarray(parent, dtype=float)
        if parent.size > 0:
            h = upsample_h(parent, n)
            return project_box_sum(h, 0.5 * float(n))

    best_h = None
    best_c = _INF

    for mode in (1, 2, 3, 4, 5, 6):
        for amp in (0.20, 0.30, 0.38, 0.46):
            h = harmonic_h(n, mode, amp, 0.0)
            c = exact_evaluate(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    for i in range(8):
        h = random_smooth_h(n, seed + i * 131)
        c = exact_evaluate(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    if best_h is None:
        best_h = np.full(n, 0.5, dtype=float)

    return project_box_sum(best_h, 0.5 * float(n))


def make_perturbation(n, seed, bandwidth=8):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)

    rng = np.random.default_rng(seed)
    y = rng.normal(size=n)

    if n > 8:
        Y = np.fft.rfft(y)
        cutoff = max(2, min(Y.size, n // int(bandwidth) + 1))
        Y = Y.copy()
        if cutoff < Y.size:
            Y[cutoff:] = 0.0
        y = np.fft.irfft(Y, n=n)

    y = y - float(np.mean(y))
    sd = float(np.std(y))
    if sd > 1.0e-12:
        y = y / sd
    else:
        y = np.zeros(n, dtype=float)

    return y


def local_optimize(h_in, n, time_limit, seed=0, guide_c=None):
    n = int(n)
    time_limit = float(time_limit)
    if n <= 0 or time_limit <= 2.0:
        h = np.asarray(h_in, dtype=float)
        if h.size != n:
            h = upsample_h(h, n)
        h = project_box_sum(h, 0.5 * float(n))
        return h, exact_evaluate(h, n)

    t0 = time.time()
    target = 0.5 * float(n)

    h = np.asarray(h_in, dtype=float).copy()
    if h.size != n:
        h = upsample_h(h, n)
    h = project_box_sum(h, target)
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5, dtype=float)

    best_c = exact_evaluate(h, n)
    best_h = h.copy()

    guide = 0.45 if guide_c is None else float(guide_c)

    beta1 = 0.90
    beta2 = 0.999
    eps = 1.0e-8

    tau_start = 0.05
    tau_end = 0.0012
    lr_start = 0.10
    lr_end = 0.012

    m = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)

    it = 0
    no_improve = 0
    restarts = 0

    max_restarts = 6 if time_limit > 240.0 else 3
    exact_interval = 30 if n <= 600 else 60
    stall_limit = 900 if n > 200 else 600
    best_smooth = best_c + 0.10

    if float(np.std(h)) < 0.05:
        h = project_box_sum(h + 0.08 * make_perturbation(n, seed + 1, 6), target)

    while True:
        elapsed = time.time() - t0
        if elapsed >= time_limit - 1.0:
            break

        f = fast_c_values(h, n)
        if not np.all(np.isfinite(f)):
            h = best_h.copy()
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            continue

        mmax = float(np.max(f))
        if mmax <= 1.0e-14:
            c = exact_evaluate(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            break

        prog = elapsed / max(1.0e-9, time_limit)
        if prog > 1.0:
            prog = 1.0

        tau = tau_start * ((tau_end / tau_start) ** prog)
        if restarts > 0 and elapsed < 0.9 * time_limit:
            tau = max(tau, 0.012)
        if tau < 1.0e-8:
            tau = 1.0e-8

        lr = lr_start * ((lr_end / lr_start) ** prog)

        P, grad, smooth_val = smooth_gradient(h, f, tau, n)
        if P is None or not np.all(np.isfinite(grad)):
            grad = np.zeros(n, dtype=float)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        decay1 = 1.0 - beta1 ** it
        decay2 = 1.0 - beta2 ** it
        if decay1 <= 0.0:
            decay1 = 1.0
        if decay2 <= 0.0:
            decay2 = 1.0

        m_hat = m / decay1
        v_hat = v / decay2
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        max_step = float(np.max(np.abs(step))) if step.size else 0.0
        if max_step > 0.15:
            step = step * (0.15 / max_step)

        h_trial = h - step
        h_trial = project_box_sum(h_trial, target)

        if not np.all(np.isfinite(h_trial)):
            h = best_h.copy()
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            continue

        h = h_trial

        if it % exact_interval == 0:
            c = exact_evaluate(h, n)
            if c < best_c - 1.0e-12:
                best_c = c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += exact_interval
        else:
            no_improve += 1

        if smooth_val < best_smooth:
            best_smooth = smooth_val

        if no_improve > stall_limit:
            if restarts < max_restarts and time.time() - t0 < time_limit - 25.0:
                use_random = (best_c > guide + 0.01) and (restarts % 2 == 1)
                if use_random:
                    h = random_smooth_h(n, seed + 777 + restarts * 1000)
                else:
                    scale = 0.08 if restarts == 0 else max(0.02, 0.06 - 0.008 * restarts)
                    bw = max(4, n // 96)
                    h = best_h + scale * make_perturbation(n, seed + 555 + restarts * 1000, bw)

                h = project_box_sum(h, target)
                if not np.all(np.isfinite(h)):
                    h = best_h.copy()

                m.fill(0.0)
                v.fill(0.0)
                it = 0
                no_improve = 0
                restarts += 1
                best_smooth = best_c + 0.05
            else:
                if not np.allclose(h, best_h, atol=1.0e-7, rtol=0.0):
                    h = best_h.copy()
                    m.fill(0.0)
                    v.fill(0.0)
                    it = 0
                no_improve = int(0.5 * stall_limit)

    c = exact_evaluate(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    best_h = project_box_sum(best_h, target)
    c = exact_evaluate(best_h, n)
    return best_h, c


def quantize_exact(h, n):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)

    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = upsample_h(h, n)

    Q = 1 << _QUANT_BITS
    if n % 2 == 0:
        targetI = (n // 2) * Q
    else:
        targetI = (n * Q) // 2
    targetI = int(targetI)

    h = np.clip(h, 0.0, 1.0)
    q = np.rint(h * float(Q)).astype(np.int64)
    q = np.clip(q, 0, Q)

    s = int(np.sum(q))
    diff = targetI - s

    if diff > 0:
        cap = Q - q
        order = np.argsort(-cap)
        for idx in order:
            if diff <= 0:
                break
            add = int(diff)
            c = int(cap[idx])
            if add > c:
                add = c
            if add > 0:
                q[idx] += add
                diff -= add
    elif diff < 0:
        excess = int(-diff)
        order = np.argsort(-q)
        for idx in order:
            if excess <= 0:
                break
            rem = int(q[idx])
            if rem > 0:
                sub = excess if excess < rem else rem
                q[idx] -= sub
                excess -= sub

    return q.astype(np.float64) / float(Q)


def add_smoothing_candidates(cands, h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = upsample_h(h, n)

    cutoffs = (
        max(2, n // 32),
        max(2, n // 16),
        max(2, n // 8),
        max(2, n // 4),
    )

    for cutoff in cutoffs:
        Y = np.fft.rfft(h)
        Yc = Y.copy()
        if cutoff < Yc.size:
            Yc[cutoff:] = 0.0
        hs = np.real(np.fft.irfft(Yc, n=n))
        hs = project_box_sum(np.clip(hs, 0.0, 1.0), 0.5 * float(n))
        cs = exact_evaluate(hs, n)
        cands.append((hs.copy(), cs, n))


def choose_best_quantized(cands):
    best_c = _INF
    best_h = None
    best_n = 0

    for item in cands:
        h, c, n = item
        if h is None:
            continue
        n = int(n)
        if n <= 0:
            continue

        hq = quantize_exact(h, n)
        cq = exact_evaluate(hq, n)

        if cq < best_c:
            best_c = cq
            best_h = hq
            best_n = n

    return best_h, best_c, best_n


def run(seed=42, budget_s=1300, **kwargs):
    t_run = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if budget < 10.0:
        budget = 10.0

    final_reserve = min(30.0, max(5.0, 0.10 * budget))

    parent = globals().get("initial_h_values", None)

    cands = []
    best_h = None
    best_c = _INF
    best_n = 0
    p_n = 0

    if parent is not None:
        parent_arr = np.asarray(parent, dtype=float)
        p_n = int(parent_arr.size)
        if p_n > 0:
            h0 = project_box_sum(np.clip(parent_arr.copy(), 0.0, 1.0), 0.5 * float(p_n))
            c0 = exact_evaluate(h0, p_n)
            cands.append((h0.copy(), c0, p_n))
            if c0 < best_c:
                best_c = c0
                best_h = h0.copy()
                best_n = p_n

    if p_n > 0:
        base = p_n
        if base % 2 != 0:
            base += 1
    else:
        base = 140

    if p_n > 0:
        if budget < 300.0:
            resolutions = [base]
        elif budget < 800.0:
            resolutions = [base, 2 * base, 4 * base]
        else:
            resolutions = [base, 2 * base, 4 * base, 8 * base]
    else:
        if budget < 300.0:
            resolutions = [140]
        elif budget < 800.0:
            resolutions = [140, 280]
        else:
            resolutions = [140, 280, 560]

    max_n = 560 if _HAS_FFT else 280
    filtered = []
    for n in resolutions:
        n = int(n)
        if n < 40:
            n = 40
        if n % 2 != 0:
            n += 1
        if n > max_n:
            if filtered:
                break
            n = max_n
        if n not in filtered:
            filtered.append(n)
    resolutions = filtered

    Lres = len(resolutions)
    if Lres == 1:
        fracs = [1.0]
    elif Lres == 2:
        fracs = [0.15, 0.85]
    elif Lres == 3:
        fracs = [0.08, 0.30, 0.62]
    elif Lres == 4:
        fracs = [0.03, 0.12, 0.30, 0.55]
    else:
        fracs = [0.03, 0.10, 0.20, 0.30, 0.37]

    for idx in range(Lres):
        n = int(resolutions[idx])
        elapsed = time.time() - t_run
        remaining = budget - final_reserve - elapsed
        if remaining <= 15.0:
            break

        sum_rem = 0.0
        for j in range(idx, Lres):
            sum_rem += fracs[j]

        if sum_rem <= 0.0:
            frac = 1.0
        else:
            frac = fracs[idx] / sum_rem

        stage_time = remaining * frac
        if stage_time < 8.0:
            continue

        if best_h is None:
            h_initial = choose_initial(n, seed + idx * 199, parent if p_n > 0 else None)
        else:
            if n == best_n:
                h_initial = best_h.copy()
            else:
                h_initial = upsample_h(best_h, n)
                h_initial = project_box_sum(
                    h_initial + 0.01 * make_perturbation(n, seed + idx * 123, max(4, n // 128)),
                    0.5 * float(n),
                )

        h_opt, c_opt = local_optimize(
            h_initial,
            n,
            stage_time,
            seed + idx * 1000,
            guide_c=best_c if np.isfinite(best_c) else None,
        )

        if h_opt is not None and np.isfinite(c_opt):
            h_opt = project_box_sum(h_opt, 0.5 * float(n))
            c_opt = exact_evaluate(h_opt, n)

            if np.isfinite(c_opt):
                cands.append((h_opt.copy(), c_opt, n))
                if c_opt < best_c:
                    best_c = c_opt
                    best_h = h_opt.copy()
                    best_n = n
                    add_smoothing_candidates(cands, h_opt, n)
                    for item in cands:
                        hh, cc, nn = item
                        if np.isfinite(cc) and cc < best_c:
                            best_c = cc
                            best_h = hh.copy()
                            best_n = nn

    if not cands:
        n = p_n if p_n > 0 else 140
        h = choose_initial(n, seed, parent if p_n > 0 else None)
        c = exact_evaluate(h, n)
        cands.append((h, c, n))
        best_h = h
        best_c = c
        best_n = n

    best_quant_h, best_quant_c, best_quant_n = choose_best_quantized(cands)

    if best_quant_h is None:
        if best_h is not None and best_n > 0:
            best_quant_h = quantize_exact(best_h, best_n)
            best_quant_n = int(best_n)
        else:
            best_quant_n = 140
            h = choose_initial(best_quant_n, seed, None)
            best_quant_h = quantize_exact(h, best_quant_n)
        best_quant_c = exact_evaluate(best_quant_h, best_quant_n)

    return best_quant_h, float(best_quant_c), int(best_quant_n)
