    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode="full")

def project_onto_simplex(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x_i <= 1}"""
    h = np.array(h, dtype=float)
    n = len(h)
    h = np.clip(h, 0, 1)
    for _ in range(50):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:

        free = (h > 1e-10) & (h < 1 - 1e-10)
        if free.sum() == 0:
            # All at bounds, can't project exactly
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0, 1)
    return h

def project_simplex(h, target):

def gradient_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]

        g[k:] += (1.0 - h[:n - k])

        h_new, c5_new = try_slsqp(h0, n0, time.time() + 120)

            best_h, best_c5, best_n = h_new, c5_new, n0

    # Phase 2: Subgradient from best
    if time.time() < deadline - 60:
        h_new, c5_new = try_subgradient(best_h, best_n, time.time() + 300)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: SLSQP again from improved
    if time.time() < deadline - 60:
        h_new, c5_new = try_slsqp(best_h, best_n, time.time() + 120)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Subgradient again
    if time.time() < deadline - 30:
        h_new, c5_new = try_subgradient(best_h, best_n, time.time() + 200)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Try different n_points with various inits
    n_candidates = [50, 60, 70, 80, 100, 120, 150, 200]
    for n_try in n_candidates:
        if time.time() > deadline - 30:
            break
        if n_try == best_n:
            continue
        inits = [
            make_bump_init(n_try),
            make_sinusoidal_init(n_try, freq=1),
            make_sinusoidal_init(n_try, freq=2),
            make_random_init(n_try, rng),
        ]
        for hi in inits:
            if time.time() > deadline - 20:
                break
            h_new, c5_new = try_slsqp(hi, n_try, time.time() + 80)
            if h_new is not None and c5_new < best_c5:
                best_h, best_c5, best_n = h_new, c5_new, n_try
                # Follow up with subgradient
                if time.time() < deadline - 30:
                    h_sg, c5_sg = try_subgradient(best_h, best_n, time.time() + 100)
                    if c5_sg < best_c5:
                        best_h, best_c5 = h_sg, c5_new if c5_sg < c5_new else c5_new
                        best_c5 = c5_sg

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 10:
        h_new, c5_new = try_slsqp(best_h, best_n, time.time() + 60)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final subgradient polish
    if time.time() < deadline - 5:
        h_new, c5_new = try_subgradient(best_h, best_n, time.time() + 30, max_iters=5000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Ensure final constraint satisfaction
    best_h = project_simplex(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
