import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def c5_and_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)

    grad = np.zeros(n)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        # Term 1: (1-h[j-lag]) for j in [lag, n-1]
        grad[lag:] += (1.0 - h[:n - lag])
        # Term 2: -h[j+lag] for j in [0, n-1-lag]
        grad[:n - lag] -= h[lag:]
    else:
        a = -lag  # |lag|
        # Term 1: (1-h[j+|lag|]) for j in [0, n-1-a]
        grad[:n - a] += (1.0 - h[a:])
        # Term 2: -h[j-|lag|] for j in [a, n-1]
        grad[a:] -= h[:n - a]

    grad *= dx
    return c5, grad

def interpolate_to(h_old, n_new):

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
