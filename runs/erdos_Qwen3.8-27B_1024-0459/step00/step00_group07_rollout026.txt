def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_value(corr, alpha):
    c_max = np.max(corr)
    shifted = (corr - c_max) * alpha
    # log(sum(exp)) - log(len)
    return c_max + (np.log(np.sum(np.exp(shifted))) - np.log(len(corr))) / alpha

def gradient_of_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
