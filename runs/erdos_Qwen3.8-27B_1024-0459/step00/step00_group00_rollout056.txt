import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, full) * dx"""
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def gradient_c5(h, dx):
    """Compute subgradient of C5 w.r.t. h"""
    corr = np.correlate(h, 1-h, mode='full')
    max_idx = np.argmax(corr)
    n = len(h)
    s = max_idx - (n - 1)  # the shift achieving the max
    # c(s) = sum_j h[j+s] * (1-h[j]) for valid j
    grad = np.zeros(n)
    for j in range(n):
        if 0 <= j+s < n:
            # h[j+s] term: derivative w.r.t. h[j+s] is (1-h[j])
            grad[j+s] += (1-h[j]) * dx
            # (1-h[j]) term: derivative w.r.t. h[j] is -h[j+s]
            grad[j] += -h[j+s] * dx
    return grad

def project_onto_constraint(h, n):
    """Project h onto {sum(h) = n/2, 0 <= h <= 1}"""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    current_sum = np.sum(h)
    diff = target - current_sum
    if diff > 0:
        # Need to increase sum
        mask = h < 1
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0, 1)
    elif diff < 0:
        mask = h > 0
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0, 1)
    return h

import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def fix_sum(h, n):
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj_grad(h, dx, tau):
    n = len(h)
    corr_all = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr_all)
    shifted = tau * (corr_all - max_val)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    weights = exp_vals / wsum
    obj = max_val + np.log(wsum) / tau

    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        s = idx - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        for j in range(j_lo, j_hi):
            i = j + s
            grad[i] += w * (1.0 - h[j])
            grad[j] -= w * h[i]
    grad *= dx
    return obj, grad


def optimize_at_resolution(h_init, n, dx, seed, time_limit):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    best_local = h_init.copy()
    c5_best = compute_c5(h_init, dx)

    for tau in [3.0, 8.0, 20.0, 60.0, 150.0, 400.0, 1000.0]:
        if time.time() - t0 > time_limit:
            break
        h_cur = best_local.copy()
        n_steps = min(5000, max(500, int(time_limit * 3)))
        base_lr = 2.0 / (tau + 1.0)
        for step in range(n_steps):
            if time.time() - t0 > time_limit:
                break
            _, grad = smooth_max_obj_grad(h_cur, dx, tau)
            lr = base_lr / (1.0 + step * 0.0005)
            h_new = h_cur - lr * grad
            h_new = fix_sum(h_new, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5_best:
                c5_best = c5_new
                best_local = h_new.copy()
            if c5_new <= compute_c5(h_cur, dx) + 1e-10:
                h_cur = h_new
            else:
                lr *= 0.5
                h_new = h_cur - lr * grad
                h_new = fix_sum(h_new, n)
                if compute_c5(h_new, dx) < compute_c5(h_cur, dx):
                    h_cur = h_new
    return best_local, c5_best


def binary_local_search(h_start, n, dx, seed, time_limit):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    h = (h_start > 0.5).astype(float)
    h = fix_sum(h, n)
    c5_cur = compute_c5(h, dx)
    best = h.copy()
    best_c5 = c5_cur
    stall = 0
    max_stall = 200

    while time.time() - t0 < time_limit:
        improved = False
        for _ in range(500):
            if time.time() - t0 > time_limit:
                break
            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            hi = 1.0 - h[i]
            hj = 1.0 - h[j]
            if hi == hj:
                h[i] = hi
                h[j] = hj
                h = fix_sum(h, n)
                c5_new = compute_c5(h, dx)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    improved = True
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best = h.copy()
                else:
                    h[i] = 1.0 - hi
                    h[j] = 1.0 - hj
                    h = fix_sum(h, n)
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                h_best_now = best.copy()
                h_best_now[rng.randint(n)] += rng.uniform(-0.5, 0.5)
                h = (h_best_now > 0.5).astype(float)
                h = fix_sum(h, n)
                c5_cur = compute_c5(h, dx)
                stall = 0
    return best, best_c5


def make_pattern(n, pattern_type, rng):
    if pattern_type == 0:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif pattern_type == 1:
        h = np.zeros(n)
        period = 5
        for i in range(n):
            h[i] = 1.0 if (i % period) < 3 else 0.0
    elif pattern_type == 2:
        h = np.zeros(n)
        period = 7
        for i in range(n):
            h[i] = 1.0 if (i % period) < 4 else 0.0
    elif pattern_type == 3:
        h = np.zeros(n)
        period = 3
        for i in range(n):
            h[i] = 1.0 if (i % period) == 0 else 0.0
    elif pattern_type == 4:
        h = rng.rand(n)
    elif pattern_type == 5:
        h = np.zeros(n)
        period = 11
        for i in range(n):

        for i in range(n):
            h[i] = 1.0 if int(i * 2.0 / n * n) % 2 == 0 else 0.0
    else:
        h = np.ones(n) * 0.5
    return fix_sum(h, n)
