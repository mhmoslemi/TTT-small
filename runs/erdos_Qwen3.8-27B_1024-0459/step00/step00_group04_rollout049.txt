    """Compute C5 and its gradient"""
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    c5 = corr.max()
    
    # Gradient
    k_star = np.argmax(corr)
    grad = np.zeros(n)
    shift = k_star  # lag index 0 to 2n-2
    
    for j in range(n):
        # First term: (1-h[j+n-1-shift]) if valid
        idx1 = j + n - 1 - shift
        if 0 <= idx1 < n:
            grad[j] += (1.0 - h[idx1])
        # Second term: -h[j-n+1+shift] if valid
        idx2 = j - n + 1 + shift
        if 0 <= idx2 < n:
            grad[j] -= h[idx2]
    
    grad *= dx
    return c5, grad
