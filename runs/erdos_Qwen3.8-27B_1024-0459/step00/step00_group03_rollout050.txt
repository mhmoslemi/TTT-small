def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float((corr * dx).max())

def objective_fn(x, n_points):
    h = np.clip(x, 0.0, 1.0)
    return compute_c5(h, n_points)

def sum_eq(x, n_points):
    return float(x.sum() - n_points / 2.0)

def sum_eq_jac(x, n_points):
    return np.ones(n_points)

def project_feasible(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        if s <= 1e-15:
            h = np.full(n_points, target / n_points)
            break
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def run_slqip(h0, n, maxiter, ftol, time_fn, deadline):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    def obj_wrapper(x):
        if time_fn() > deadline:
            return compute_c5(np.clip(x, 0, 1), n) + 1.0
        return objective_fn(x, n)
    result = minimize(
        fun=obj_wrapper, x0=h0, args=(n,),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
    )
    return result.x

                return h

                c2 = 0.0  # const from i=p

                c4 = 0.0  # const from i=p-d
                if 0 <= p + d < n:

                if 0 <= p - d < n:
                    c3 = -h[p - d]

            # Minimize max_k(a[k]*t + b[k]) for t in [0,1]
