    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)

        return h

        return h

    h[mask] += adj

import numpy as np
import time

def compute_C5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
    cnt = np.sum(mask)
    if cnt == 0:
        mask = np.ones(n, dtype=bool)
        cnt = float(n)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def upsample_linear(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, n_new)
