    """One sweep of coordinate descent with Lagrange multiplier"""
    h_new = h.copy()
    for i in range(n):
        # For each k, compute the slope and intercept of c_k as a function of h[i]
        # c_k = ... + (1-h[i] terms involving h[i])
        # From our analysis:
        # For the j-th correlation component (j = 0..2n-2, k = j-(n-1)):
        # d(c_k)/d(h_i) = slope, and c_k = slope * h[i] + intercept (others fixed)
        
        # Actually, let me just compute c_k for t = 0 and t = 1 to get the line
        # c_k(t) = c_k(h with h[i]=t) = a_k * t + b_k
        # a_k = c_k(h with h[i]=1) - c_k(h with h[i]=0)
        # b_k = c_k(h with h[i]=0)
        
        h[i] = 0.0
        corr0 = corr_values(h_new, n)
        h[i] = 1.0
        corr1 = corr_values(h_new, n)
        a = corr1 - corr0  # slopes
        b = corr0  # intercepts
        
        # Minimize max_k (a_k * t + b_k) + lam * t
        # = max_k ((a_k + lam) * t + b_k)  ... wait, no
        # The Lagrangian adds lam * sum(h) = lam * (sum(h[j]) + h[i]) for j≠i
        # So the h[i]-dependent part is: max_k (a_k*t + b_k) + lam*t
        # This is NOT the same as max_k ((a_k + lam)*t + b_k)
        # Because the max and the +lam*t don't combine that way
        
        # So the objective is: f(t) = max_k (a_k*t + b_k) + lam*t
        # This is still convex PL in t
        
        # Find min over t in [0,1]
        best_t = 0.0
        best_val = np.inf
        for t in t_grid:
            val = np.max(a * t + b) + lam * t
            if val < best_val:
                best_val = val
                best_t = t
        h_new[i] = best_t
    
    return h_new

        mask2 = (idx >= 0) & (idx <= n - 1 - k)  # h[m+k] valid: 0 ≤ m ≤ n-1-k
        grad[mask1] += (1.0 - h[idx[mask1] - k])
        grad[mask2] -= h[idx[mask2] + k]
    else:
        kp = -k
        mask1 = (idx >= 0) & (idx <= n - 1 - kp)  # h[m+kp] valid: 0 ≤ m ≤ n-1-kp
        mask2 = (idx >= kp) & (idx <= n - 1)  # h[m-kp] valid: kp ≤ m ≤ n-1
        grad[mask1] += (1.0 - h[idx[mask1] + kp])
        grad[mask2] -= h[idx[mask2] - kp]
    return grad
