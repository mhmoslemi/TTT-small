    """Compute subgradient of max overlap at lag k_star.

        h_new = project_onto_constraint(h_new, target_sum)

        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        if new_c5 < current_c5 + 1e-12:
            h = h_new
            current_c5 = new_c5
            alpha *= decay
        else:
            alpha *= shrink
            alpha = max(alpha, 0.0005)

    return best_h, best_c5

def local_search_swap(h, max_passes=10000, time_limit=30.0, t0=None):
    n = len(h)
    if t0 is None:
        t0 = time.time()
    h = h.copy()
    current_c5 = evaluate_c5(h)
    improved = True
    passes = 0

    while improved and passes < max_passes:
        improved = False
        passes += 1
        if time.time() - t0 > time_limit:
            break
        for i in range(n):
            if improved:
                break
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-12:
                    continue
                h[i], h[j] = h[j], h[i]
                new_c5 = evaluate_c5(h)
                if new_c5 < current_c5 - 1e-15:
                    current_c5 = new_c5
                    improved = True
                    break
                h[i], h[j] = h[j], h[i]
    return h, current_c5

def make_random_binary(n, rng):
    target_sum = n / 2.0
    n_ones = int(target_sum)
    remainder = target_sum - n_ones
    h = np.zeros(n)
    positions = rng.permutation(n)
    h[positions[:n_ones]] = 1.0
    if remainder > 1e-10:
        h[positions[n_ones]] = remainder
    return h

def upsample_step(h, factor):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    best_h = None
    best_c5 = float('inf')

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = None

    if h_init is not None:
        n_init = len(h_init)
        h_init = project_onto_constraint(h_init, n_init / 2.0)
        c5_init = evaluate_c5(h_init)
        best_h = h_init.copy()
        best_c5 = c5_init

    # Phase 1: Subgradient descent from initial
    if h_init is not None and tl() > 40:
        h_sg, c5_sg = subgradient_descent(h_init, n_steps=4000, alpha0=0.4, decay=0.9995, shrink=0.93)
        if c5_sg < best_c5:
            best_h = h_sg.copy()
            best_c5 = c5_sg

    # Phase 2: Local search from best
    if tl() > 25:
        lt = min(45.0, tl() - 10)
        h_ls, c5_ls = local_search_swap(best_h, max_passes=20000, time_limit=lt, t0=t0)
        if c5_ls < best_c5:
            best_h = h_ls.copy()
            best_c5 = c5_ls

    # Phase 3: Random restarts at n=45
    n45 = 45
    if tl() > 30:
        n_restarts = int(min(25, tl() / 8))
        for r in range(n_restarts):
            if tl() < 12:
                break
            h_rand = make_random_binary(n45, rng)
            lt = min(5.0, tl() - 8)
            h_ls, c5 = local_search_swap(h_rand, max_passes=5000, time_limit=lt, t0=t0)
            if c5 < best_c5:
                best_h = h_ls.copy()
                best_c5 = c5

    # Phase 4: Try structured initializations
    if tl() > 30:
        n45 = 45
        # Alternating pattern
        h_alt = np.zeros(n45)
        for r in range(3):
            if tl() < 10:
                break
            h_alt = np.zeros(n45)
            for i in range(n45):
                if (i + r) % 2 == 0:
                    h_alt[i] = 1.0
            # Fix sum
            s = np.sum(h_alt)
            if s > 22.5:
                ones = [i for i in range(n45) if h_alt[i] == 1]
                for idx in ones:
                    h_alt[idx] = 0.0
                    if np.sum(h_alt) <= 22.5:
                        if np.sum(h_alt) < 22.5:
                            h_alt[idx] += 22.5 - np.sum(h_alt)
                        break
            elif s < 22.5:
                zeros = [i for i in range(n45) if h_alt[i] == 0]
                for idx in zeros:
                    h_alt[idx] = 1.0
                    if np.sum(h_alt) >= 22.5:
                        if np.sum(h_alt) > 22.5:
                            h_alt[idx] = 1.0 - (np.sum(h_alt) - 22.5)
                        break
            lt = min(5.0, tl() - 8)
            h_ls, c5 = local_search_swap(h_alt, max_passes=5000, time_limit=lt, t0=t0)
            if c5 < best_c5:
                best_h = h_ls.copy()
                best_c5 = c5

    # Phase 5: Higher resolution (n=90)
    if tl() > 80 and best_h is not None:
        n_high = 90
        factor = n_high // len(best_h) if len(best_h) > 0 else 1
        if len(best_h) == 45:
            h_high = np.repeat(best_h, 2)
        else:
            h_high = np.repeat(best_h, max(1, n_high // len(best_h)))
        h_high = h_high[:n_high]
        if len(h_high) < n_high:
            h_high = np.pad(h_high, (0, n_high - len(h_high)))
        h_high = project_onto_constraint(h_high, n_high / 2.0)

        if tl() > 50:
            h_sg, c5_sg = subgradient_descent(h_high, n_steps=2000, alpha0=0.2, decay=0.9995, shrink=0.93)
            if c5_sg < best_c5:
                best_h = h_sg.copy()
                best_c5 = c5_sg

        if tl() > 20:
            lt = min(40.0, tl() - 10)
            h_ls, c5 = local_search_swap(best_h, max_passes=10000, time_limit=lt, t0=t0)
            if c5 < best_c5:
                best_h = h_ls.copy()
                best_c5 = c5

    # Phase 6: More random restarts at n=90
    if tl() > 40:
        n90 = 90
        n_restarts = int(min(5, tl() / 15))
        for r in range(n_restarts):
            if tl() < 15:
                break
            h_rand = make_random_binary(n90, rng)
            lt = min(10.0, tl() - 10)
            h_ls, c5 = local_search_swap(h_rand, max_passes=5000, time_limit=lt, t0=t0)
            if c5 < best_c5:
                best_h = h_ls.copy()
                best_c5 = c5

    # Phase 7: Final polish on best solution
    if tl() > 15 and best_h is not None:
        lt = min(30.0, tl() - 3)
        h_ls, c5 = local_search_swap(best_h, max_passes=20000, time_limit=lt, t0=t0)
        if c5 < best_c5:
            best_h = h_ls.copy()
            best_c5 = c5

    # Final validation
    if best_h is None:
        best_h = make_random_binary(45, rng)
        best_c5 = evaluate_c5(best_h)

    # Ensure constraint is satisfied
    n = len(best_h)
    best_h = project_onto_constraint(best_h, n / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n
