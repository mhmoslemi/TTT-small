def compute_c5(h, n):

    conv = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):

        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-10:

        diff = target - s
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(free):
            break
        n_free = np.sum(free)
        h[free] += diff / n_free
    return h

def corr_lag(h, l, n):
    """Compute correlation at specific lag l (positive or negative)"""
    if l >= 0:
        m = min(n - l, n)
        return np.sum(h[l:l+m] * (1.0 - h[:m])) if l < n else 0.0
    else:
        l = -l
        m = n - l
        return np.sum(h[:m] * (1.0 - h[l:])) if l < n else 0.0

def gradient_c5(h, n, top_k=5):
    """Compute subgradient of C5 at h"""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    conv_dx = conv * dx
    # Find top lags
    top_indices = np.argsort(conv_dx)[-top_k:]
    weights = conv_dx[top_indices]
    weights = weights / np.sum(weights)
    
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        l = idx - (n - 1)  # actual lag in index units
        # Compute gradient at this lag
        for j in range(n):
            g = 0.0
            if l > 0:
                if j >= l:
                    g += (1.0 - h[j - l])
                if j + l < n:
                    g -= h[j + l]
            elif l < 0:
                ll = -l
                if j + ll < n:
                    g += (1.0 - h[j + ll])
                if j >= ll:
                    g -= h[j - ll]
            else:
                # l = 0: h[i](1-h[i]) = 0 always, gradient is 0
                pass
            grad[j] += w * g
    return grad * dx, conv_dx

def local_optimize(h, n, max_iter=5000, tol=1e-7):

    h = h.copy()
    h = project_feasible(h, n)
    best_c5 = compute_c5(h, n)[0]

def compute_c5_obj(h, n):

    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def project_feasible(h, n):

    target = n / 2.0

        if abs(s - target) < 1e-12:

        diff = target - s
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not np.any(free):

            if not np.any(free):
                break

        if nf == 0:

        h[free] += diff / nf

def grad_c5(h, n, n_active=5):

import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)

        if abs(diff) < 1e-12:

        nf = np.sum(free)
        if nf == 0:
            break
        h[free] += diff / nf
    return h

def grad_subgradient(h, n, n_active=3):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    k = min(n_active, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    w = conv[top_idx]
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n), conv

            grad[ll:] -= wi * h[:n-ll]

            grad += wi * (1.0 - 2.0 * h)

    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.05

import cvxpy as cp

def solve_milp(n, solver=cp.SCIP):
    dx = 2.0 / n
    # Variables
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    
    # w variables: for each lag k and each valid i
    # Lag k ranges from -(n-1) to (n-1)
    # For lag k, valid i are max(0,k) to min(n-1, n-1-k)
    w_vars = {}
    for k in range(-(n-1), n):
        i_start = max(0, k)
        i_end = min(n-1, n-1-k)
        n_i = i_end - i_start + 1
        if n_i > 0:
            w_vars[k] = cp.Variable(n_i)
    
    constraints = [cp.sum(h) == n/2]
    
    # Linking constraints
    for k, w in w_vars.items():
        i_start = max(0, k)
        i_end = min(n-1, n-1-k)
        n_i = i_end - i_start + 1
        for idx, i in enumerate(range(i_start, i_end+1)):
            j = i - k
            constraints.append(w[idx] <= h[i])
            constraints.append(w[idx] <= h[j])
            constraints.append(w[idx] >= h[i] + h[j] - 1)
            constraints.append(w[idx] >= 0)
        
        # Lag constraint: sum(w) >= n/2 - t/dx
        # Equivalently: t <= dx * (n/2 - sum(w))
        # Or: t + dx * sum(w) <= dx * n/2
        constraints.append(t + dx * cp.sum(w) <= dx * n / 2)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    if prob.status in ['optimal', 'optimal_inaccurate']:
        h_val = h.value
        t_val = t.value
        return h_val, t_val
    return None, None

import numpy as np
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf == 0:
            break
        h[free] += diff / nf
    return h

def grad_subgradient(h, n, n_active=3):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    k = min(n_active, len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    w = conv[top_idx].copy()
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n), conv
    w = w / w_sum
    grad = np.zeros(n)
    for ii in range(k):
        idx = top_idx[ii]
        l = idx - (n - 1)
        wi = w[ii]
        if l > 0:
            grad[l:] += wi * (1.0 - h[:n - l])
            grad[:n - l] -= wi * h[l:]
        elif l < 0:
            ll = -l
            grad[:n - ll] += wi * (1.0 - h[ll:])
            grad[ll:] -= wi * h[:n - ll]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad * dx, conv

def local_optimize(h, n, max_iter=5000, time_limit=30.0, n_active=3):
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.05
    momentum = 0.85
    v = np.zeros(n)
    start = time.time()
    stall = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        grad, _ = grad_subgradient(h, n, n_active=n_active)
        v = momentum * v - alpha * grad
        h_new = project_feasible(h + v, n)

        if c5_new < best_c5 - 1e-11:
            h = h_new

            stall = 0
            alpha = min(alpha * 1.02, 0.3)

            if alpha < 1e-5 or stall > 200:

    return best_h, best_c5

def upscale(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)

def try_milp(n, time_limit=300.0, initial_bound=None):
    try:
        import cvxpy as cp
    except ImportError:
        return None, None
    dx = 2.0 / n
    if n % 2 != 0:
        return None, None
    n_half = n // 2
    # Variables
    h_var = cp.Variable(n, boolean=True)
    t_var = cp.Variable()
    constraints = [cp.sum(h_var) == n_half, t_var >= 0]
    w_list = []
    lag_constraints = []
    for k in range(1, n):
        # Positive lag k: i from k to n-1, j = i-k from 0 to n-1-k
        n_i = n - k
        w = cp.Variable(n_i)
        w_list.append(w)
        for idx in range(n_i):
            i = k + idx
            j = idx
            constraints.append(w[idx] <= h_var[i])
            constraints.append(w[idx] <= h_var[j])
            constraints.append(w[idx] >= h_var[i] + h_var[j] - 1)
            constraints.append(w[idx] >= 0)
        # t + dx*sum(w) >= dx*n/2  => t >= dx*(n/2 - sum(w))
        lag_constraints.append(t_var >= dx * (n_half - cp.sum(w)))
    # Negative lags are symmetric for the max, so we only need positive
    constraints.extend(lag_constraints)
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    # Try solvers in order
    solvers_to_try = [cp.SCIP, cp.CBC, cp.GUROBI, cp.XPRESS]
    start = time.time()
    result_h = None
    result_t = None
    for solver in solvers_to_try:
        if time.time() - start > time_limit * 0.8:
            break
        try:
            prob.solve(solver=solver, time_limit=max(10, time_limit * 0.9))
            if prob.status in ['optimal', 'optimal_inaccurate']:
                result_t = float(t_var.value)
                h_val = h_var.value
                if h_val is not None:
                    result_h = np.round(h_val).astype(float)
                    break
        except Exception:
            continue
    if result_h is not None:
        result_h = np.clip(result_h, 0, 1)
        result_c5 = compute_c5(result_h, n)
        return result_h, result_c5
    return None, None
