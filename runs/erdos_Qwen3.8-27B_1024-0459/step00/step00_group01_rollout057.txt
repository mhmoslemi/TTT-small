import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    omp = 1 - h
    corr = np.correlate(h, omp, mode="full")
    return np.max(corr) * dx

def initial_binary(n, seed=42):
    rng = np.random.default_rng(seed)
    h = np.ones(n)
    positions = rng.choice(n, size=n//2, replace=False)
    h[positions] = 0
    return h

def local_search_binary(h, n, max_iters=10000, seed=42):
    """Flip bits to reduce max C5"""
    rng = np.random.default_rng(seed)
    h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    for it in range(max_iters):
        # Try flipping a random bit
        i = rng.integers(n)
        if h[i] == 1:
            # Flipping to 0 reduces sum, need to flip another to 1
            j = rng.integers(n)
            if h[j] == 0:
                h[i], h[j] = 0, 1
                new_c5 = compute_c5(h)
                if new_c5 < current_c5:
                    current_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                else:
                    h[i], h[j] = 1, 0
        # (For binary with fixed sum, we always swap a 1 and a 0)
    
    return h, best_c5

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    """Compute C5 for a given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(n, rng):
    """Create a random binary array with exactly n//2 ones."""
    h = np.zeros(n)
    positions = rng.choice(n, size=n//2, replace=False)
    h[positions] = 1.0
    return h

def sim_anneal_binary(h, n, rng, time_limit=30, T0=0.5, Tend=0.001):
    """Simulated annealing on binary sequences."""
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    start = time.time()
    total_iters = 0
    no_improve = 0
    max_no_improve = 50000
    
    while time.time() - start < time_limit:
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        
        # Pick random positions
        i = ones_pos[rng.integers(len(ones_pos))]
        j = zeros_pos[rng.integers(len(zeros_pos))]
        
        # Swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        
        # Temperature
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)
        T = T0 * (Tend / T0) ** frac
        
        # Metropolis acceptance
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            curr_c5 = new_c5
            # Update position tracking
            ones_pos = np.where(h == 1)[0]
            zeros_pos = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = 1.0, 0.0
        
        total_iters += 1
        if no_improve > max_no_improve:
            break
    
    return best_c5, best_h, total_iters

def relaxed_optimize(h0, n, time_limit=60):
    """Optimize using scipy SLSQP with equality constraint sum(h)=n/2."""
    target_sum = n / 2.0
    
    def objective(h):
        return compute_c5(h)
    
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    bounds = [(0, 1)] * n
    
    result = optimize.minimize(
        objective, h0, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-10}
    )
    
    c5 = compute_c5(result.x)
    # Ensure constraint satisfaction
    result.x = np.clip(result.x, 0, 1)
    diff = target_sum - np.sum(result.x)
    # Simple adjustment
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(result.x < 1)[0]
        else:
            idx = np.where(result.x > 0)[0]
        if len(idx) == 0:
            break
        i = idx[0]
        adj = np.clip(diff / len(idx), -result.x[i], 1 - result.x[i]) if diff > 0 else np.clip(diff / len(idx), -result.x[i], 1-result.x[i])
        # Simpler: just add diff/n to all, clip, and repeat
        result.x += diff / n
        result.x = np.clip(result.x, 0, 1)
        diff = target_sum - np.sum(result.x)
    c5 = compute_c5(result.x)
    return c5, result.x

A(-k) = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i]) for k > 0 (0→1 count)
