from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    exps = np.exp((scaled - m) / tau)
    sm = m + tau * np.log(np.sum(exps))
    s = np.sum(h) - n / 2.0
    return sm + penalty * s * s

def optimize_local(h0, max_iter=1000, tau_start=0.05, tau_end=0.001, penalty=100.0, time_limit=50):
    """Optimize h starting from h0 using L-BFGS-B with decreasing tau."""
    n = len(h0)
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    
    # Anneal tau
    tau = tau_start
    n_tau_steps = 5
    tau_schedule = np.geomspace(tau_start, tau_end, n_tau_steps)
    
    for tau in tau_schedule:
        if time.time() - t0 > time_limit:
            break
        result = minimize(
            smooth_max_obj, h, args=(tau, penalty),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': max_iter // n_tau_steps, 'ftol': 1e-10}
        )
        h = result.x
        # Project to feasible set (adjust sum)
        s = np.sum(h)
        target = n / 2.0
        if s > 0 and abs(s - target) > 1e-6:
            # Simple projection: scale and clip

import time

def evaluate_c5(h):
    """Compute C5 = max_k corr[k] * dx where corr = correlate(h, 1-h, 'full')."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_objective(h, tau, penalty):
    """Objective: smooth max of correlation*dx + constraint penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = corr * dx
    m = np.max(scaled)
    # Numerically stable log-sum-exp
    exps = np.exp((scaled - m) / tau)
    sm = m + tau * np.log(np.sum(exps))
    s = np.sum(h) - n / 2.0
    return sm + penalty * s * s

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2 approximately."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    target = n / 2.0
    if abs(s - target) < 1e-10:
        return h
    # Iterative water-filling style projection
    for _ in range(100):

        if abs(diff) < 1e-10:

    h_best = h.copy()
    stall = 0

    while time.time() - t0 < max_time:
        corr = get_corr_full(h)
        max_idx = int(np.argmax(corr))
        max_val = corr[max_idx]
        current_c5 = max_val * (2.0 / n)

        if current_c5 < best_c5 - 1e-15:
            best_c5 = current_c5
            h_best = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000:
                break

        lag = max_idx - (n - 1)

        if lag >= 0:
            if lag == 0:
                conflicts = np.array([], dtype=int)
            else:
                conflicts = np.where((h[:n - lag] == 1) & (h[lag:] == 0))[0]
        else:
            l = -lag
            conflicts = np.where((h[l:] == 1) & (h[:n - l] == 0))[0] + l

        improved = False

        if len(conflicts) > 0:
            for _ in range(n_tries):
                i = int(np.random.choice(conflicts))
                zeros_pos = np.where(h == 0)[0]
                if len(zeros_pos) == 0:
                    break
                j = int(np.random.choice(zeros_pos))
                h[i] = 0.0
                h[j] = 1.0
                new_max = np.max(get_corr_full(h)) * (2.0 / n)
                if new_max < current_c5:
                    best_c5 = new_max
                    h_best = h.copy()
                    stall = 0
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        else:
            for _ in range(n_tries):
                ones_pos = np.where(h == 1)[0]
                zeros_pos = np.where(h == 0)[0]
                if len(ones_pos) == 0 or len(zeros_pos) == 0:
                    break
                i = int(np.random.choice(ones_pos))
                j = int(np.random.choice(zeros_pos))
                h[i] = 0.0
                h[j] = 1.0
                new_max = np.max(get_corr_full(h)) * (2.0 / n)
                if new_max < current_c5:
                    best_c5 = new_max
                    h_best = h.copy()
                    stall = 0
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0

    return h_best


def lbfgs_optimize(h, n, tau_schedule, max_time=60.0):
    bounds = [(0.0, 1.0)] * n
    penalty = 500.0
    t0 = time.time()

    for tau in tau_schedule:
        elapsed = time.time() - t0
        if elapsed > max_time:
            break
        remaining_time = max_time - elapsed
        iters = max(100, min(800, int(remaining_time * 8)))
        res = minimize(
            smooth_obj, h, args=(tau, penalty),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': iters, 'ftol': 1e-12, 'gtol': 1e-7}
        )
        h = project_sum(res.x, n)

    return h


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_sum(h_new, n_new)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def time_ok(frac):
        return elapsed() < budget_s * frac

    # Initialize
    h = np.array(initial_h_values, dtype=np.float64).copy()
    n = len(h)
    c5 = evaluate_c5(h)
    h_best = h.copy()
    c5_best = c5
    n_best = n

    # Phase 1: L-BFGS-B at initial resolution
    if time_ok(0.15):
        h_proj = project_sum(h, n)
        h_opt = lbfgs_optimize(h_proj, n, tau_schedule=[0.1, 0.05, 0.02, 0.01, 0.005], max_time=20.0)
        c5_opt = evaluate_c5(h_opt)
        if c5_opt < c5_best:
            h_best = h_opt
            c5_best = c5_opt

    # Phase 2: Binary local search at initial resolution
    if time_ok(0.25):
        h_bin = binary_local_search(h_best, n, max_time=20.0)
        c5_bin = evaluate_c5(h_bin)
        if c5_bin < c5_best:
            h_best = h_bin
            c5_best = c5_bin

    # Phase 3: Multi-resolution optimization
    resolutions = [84, 168, 336, 672, 1344]
    for n_new in resolutions:
        if not time_ok(0.82):
            break

        # Interpolate
        h_interp = interpolate_h(h_best, n_best, n_new)

        # Continuous optimization
        if time_ok(0.75):
            max_t = min(40.0, budget_s * 0.75 - elapsed())
            if max_t > 5:
                h_opt = lbfgs_optimize(h_interp, n_new,
                                       tau_schedule=[0.05, 0.02, 0.01, 0.005, 0.002],
                                       max_time=max_t)
                c5_opt = evaluate_c5(h_opt)
                if c5_opt < c5_best:
                    h_best = h_opt
                    c5_best = c5_opt
                    n_best = n_new

        # Binary local search
        if time_ok(0.75):
            max_t = min(30.0, budget_s * 0.75 - elapsed())
            if max_t > 5:
                h_bin = binary_local_search(h_best, n_best, max_time=max_t)
                c5_bin = evaluate_c5(h_bin)
                if c5_bin < c5_best:
                    h_best = h_bin
                    c5_best = c5_bin
                    n_best = n_best

        # Go back to continuous from binary (binary is a valid continuous point)
        if time_ok(0.78) and n_best == n_new:
            max_t = min(20.0, budget_s * 0.78 - elapsed())
            if max_t > 5:
                h_cont = project_sum(h_best, n_best)
                h_opt = lbfgs_optimize(h_cont, n_best,
                                       tau_schedule=[0.01, 0.005, 0.002],
                                       max_time=max_t)
                c5_opt = evaluate_c5(h_opt)
                if c5_opt < c5_best:
                    h_best = h_opt
                    c5_best = c5_opt
                    n_best = n_best

    # Phase 4: Final high-resolution attempt
    if time_ok(0.88):
        n_final = 2016
        h_interp = interpolate_h(h_best, n_best, n_final)
        max_t = min(60.0, budget_s * 0.92 - elapsed())
        if max_t > 10:
            h_opt = lbfgs_optimize(h_interp, n_final,
                                   tau_schedule=[0.03, 0.01, 0.005, 0.002],
                                   max_time=max_t * 0.5)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < c5_best:
                h_best = h_opt
                c5_best = c5_opt
                n_best = n_final

            max_t2 = min(30.0, budget_s * 0.92 - elapsed())
            if max_t2 > 5:
                h_bin = binary_local_search(h_best, n_best, max_time=max_t2)
                c5_bin = evaluate_c5(h_bin)
                if c5_bin < c5_best:
                    h_best = h_bin
                    c5_best = c5_bin
                    n_best = n_best

    # Final projection and evaluation
    h_best = project_sum(h_best, n_best)
    c5_final = evaluate_c5(h_best)

    return h_best, c5_final, n_best
