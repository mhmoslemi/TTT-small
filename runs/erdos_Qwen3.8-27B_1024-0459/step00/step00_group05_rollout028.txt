```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def coord_step(h, j):
    """Find optimal value of h[j] given all other h fixed."""
    n = len(h)
    h_save = h[j]
    
    h[j] = 0
    corr0 = np.correlate(h, 1 - h, mode="full")
    
    h[j] = 1
    corr1 = np.correlate(h, 1 - h, mode="full")
    
    a = corr1 - corr0  # slopes (per unit change in h[j])
    b = corr0  # intercepts
    
    h[j] = h_save
    
    # Find t in [0,1] minimizing max_k (a[k]*t + b[k])
    # Evaluate at 0, 1, and intersection points
    f0 = np.max(b)
    f1 = np.max(a + b)
    best_t = 0 if f0 <= f1 else 1
    best_f = min(f0, f1)
    
    # Find intersection points
    for i in range(len(a)):
        for j2 in range(i+1, len(a)):
            if abs(a[i] - a[j2]) < 1e-12:
                continue
            t = (b[j2] - b[i]) / (a[i] - a[j2])
            if 0 <= t <= 1:
                f_t = np.max(a * t + b)
                if f_t < best_f - 1e-12:
                    best_f = f_t
                    best_t = t
    
    h[j] = best_t
    return best_f, best_t
```

- Find min of max(a*t + b) over t in [0,1]

For the last step, I can use the fact that the upper envelope of n lines has at most n vertices. I can compute the upper envelope in O(n log n) using a half-plane intersection algorithm, but that's complex. 

Simpler: sort lines by slope, then the upper envelope changes at O(n) points. Actually, let me just do the O(n^2) intersection check but vectorize it:

```python
a = corr1 - corr0  # shape (2n-1,)
b = corr0  # shape (2n-1,)

# For all pairs (i, j), compute intersection t_ij
# t_ij = (b_j - b_i) / (a_i - a_j)
# Only valid if a_i != a_j and 0 <= t_ij <= 1

# Vectorized:
da = a[:, None] - a[None, :]  # (m, m)
db = b[None, :] - b[:, None]  # (m, m)
with np.errstate(divide='ignore', invalid='ignore'):
    t = db / da  # (m, m)
t = np.where((da != 0) & (t >= 0) & (t <= 1), t, -1)

# For each valid t, compute f(t) = max_k (a_k * t + b_k)
valid = t[t >= 0]
if len(valid) > 0:
    # Evaluate at unique intersection points
    unique_t = np.unique(valid)
    # Also include 0 and 1
    unique_t = np.concatenate([unique_t, [0, 1]])
    # f at each t: max over k of a_k * t + b_k
    # Shape: (len(unique_t), m)
    vals = a[None, :] * unique_t[:, None] + b[None, :]
    f_vals = np.max(vals, axis=1)
    best_idx = np.argmin(f_vals)
    best_t = unique_t[best_idx]
```

```python
def project_to_constraint(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}."""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h.copy()
    
    # Bisection on lambda
    # We want sum(clip(h_i + lambda, 0, 1)) = target_sum
    # (note: lambda can be positive or negative)
    lo, hi = -2.0, 2.0
    for _ in range(60):

        trial = np.clip(h + mid, 0, 1)
        s_trial = np.sum(trial)
        if s_trial < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo+hi)/2, 0, 1)
    
    # Fix any tiny numerical errors

        free = (result > 1e-8) & (result < 1 - 1e-8)

            result = np.clip(result, 0, 1)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-13:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        trial = np.clip(h + mid, 0.0, 1.0)
        s_t = float(np.sum(trial))
        if s_t < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h + lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(result))
    if abs(diff) > 1e-14:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        nf = int(np.sum(free))

            result = np.clip(result, 0.0, 1.0)
        else:
            result += diff / len(result)
            result = np.clip(result, 0.0, 1.0)
    return result

def coordinate_descent(h, n_sweeps=50, time_budget=None, start_time=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    h = h.copy()
    target = n / 2.0

    for sweep in range(n_sweeps):
        if time_budget is not None and start_time is not None:
            if time.time() - start_time > time_budget:
                break
        order = rng.permutation(n)
        for j in order:
            h[j] = coord_optimal(h, j)
        h = project_to_constraint(h, target)

    return h

def upsample2(h):
    n = len(h)
    out = np.empty(2 * n)
    out[0::2] = h
    out[1::2] = h
    return out
```
