```python
def c5_smooth_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    one_m_h = 1.0 - h
    
    # Compute all g_k values using full correlation
    corr = np.correlate(h, one_m_h, mode='full')  # length 2n-1
    g = corr * dx
    
    # Soft max for smooth objective
    max_g = np.max(g)
    shifted = alpha * (g - max_g)
    exp_shifted = np.exp(shifted)
    Z = np.sum(exp_shifted)
    p = exp_shifted / Z  # Boltzmann weights, length 2n-1
    
    c5_smooth = max_g + np.log(Z) / alpha
    
    # Gradient computation
    # grad_h[j] = dx * sum_k p[k] * [(1-h[j+k]) - h[j-k]]
    # where k ranges from -(n-1) to (n-1), and p[k] = p[k+(n-1)]
    
    # Use the fact that:
    # sum_k p[k] * (1-h[j+k]) = correlate(1-h, p_reversed, 'same-like')
    # Let me just do it with explicit loops for clarity (n is small enough)
    
    grad_h = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k_idx in range(2*n-1):
            k = k_idx - (n - 1)
            pk = p[k_idx]
            if pk < 1e-15:
                continue
            if 0 <= j + k < n:
                s += pk * (1 - h[j + k])
            if 0 <= j - k < n:
                s -= pk * h[j - k]
        grad_h[j] = dx * s
    
    return c5_smooth, grad_h
```
