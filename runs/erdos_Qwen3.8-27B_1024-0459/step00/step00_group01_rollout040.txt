def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    c5 = corr[max_idx] * dx
    l = max_idx - (n - 1)
    
    grad = np.zeros(n)
    if l >= 0:
        # f = sum_{j=0}^{n-1-l} h[j](1-h[j+l])
        # d/dh[k] = (1-h[k+l]) for k+l <= n-1, minus h[k-l] for k >= l
        for k in range(n - l):
            grad[k] += (1.0 - h[k + l])
        for k in range(l, n):
            grad[k] -= h[k - l]
    else:
        m = -l
        for k in range(m, n):
            grad[k] += (1.0 - h[k - m])
        for k in range(n - m):
            grad[k] -= h[k + m]
    
    return c5, grad * dx

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        clipped = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(clipped)
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

    """Compute gradients of all correlation components. Returns (n, 2n-1) array."""
    n = len(h)

def optimize_slsqp(h_init, n, alpha_smooth=100.0, max_iter=500):
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        max_val = np.max(corr)
        return max_val + np.log(np.sum(np.exp(corr - max_val))) / alpha_smooth
    
    def gradient(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx

def resample_h(h_old, n_new):
    """Resample h from old resolution to new resolution."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new

def project_feasible(h, n):
