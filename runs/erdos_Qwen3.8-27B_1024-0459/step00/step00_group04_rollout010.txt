    if T < 1e-15:
        return c_max
    shifted = (c - c_max) / T
    shifted = np.clip(shifted, -700, 700)
    return c_max + T * np.log(np.sum(np.exp(shifted)))

    dx = 2.0 / n

    c_max = np.max(c)

        idx = np.argmax(c)

        shifted = (c - c_max) / T
        shifted = np.clip(shifted, -700, 700)
        weights = np.exp(shifted)

    grad = np.ones(n) * np.sum(weights)
    for m in range(len(c)):
        w = weights[m]
        if w < 1e-14:

        else:

                grad[o:] -= w * h[:n - o]
