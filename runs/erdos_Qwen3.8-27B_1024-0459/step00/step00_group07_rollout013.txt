import time

def compute_c5_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def get_c5(h):
    return np.max(compute_c5_values(h))

def make_binary(h, n_target):
    """Make a binary pattern with exactly n_target/2 ones"""
    h = np.round(h).astype(int)
    n_current = len(h)
    if n_current != n_target:
        # Upscale
        x_old = np.arange(n_current)
        x_new = np.arange(n_target) * n_current / n_target
        idx = np.floor(x_new).astype(int)
        h = h[idx]
    n_ones = np.sum(h)
    target = n_target // 2
    if n_ones > target:
        # Remove some ones
        idx = np.where(h == 1)[0]
        remove = np.random.choice(idx, n_ones - target, replace=False)
        h[remove] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        add = np.random.choice(idx, target - n_ones, replace=False)
        h[add] = 1
    return h

import numpy as np
import time

def get_c5(h):
    """Compute C5 for a given h array"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upscale(h, new_n):
    """Upscale h to new_n points using linear interpolation"""
    old_n = len(h)
    x_old = np.arange(old_n) / old_n * 2.0
    x_new = np.arange(new_n) / new_n * 2.0
    h_new = np.interp(x_new, x_old, h)
    # Snap to 0/1 and fix sum
    h_new = (h_new > 0.5).astype(np.int8)
    target = new_n // 2
    n_ones = h_new.sum()
    if n_ones > target:
        idx = np.where(h_new == 1)[0]
        # Remove random ones
        perm = np.random.permutation(idx)
        h_new[perm[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h_new == 0)[0]
        perm = np.random.permutation(idx)
        h_new[perm[:target - n_ones]] = 1
    return h_new.astype(np.float64)

            h[j] = 1 - orig
            c5 = get_c5(h.astype(np.float64))

            current_c5 = best_new_c5
            if current_c5 < best_c5:

                    if current_c5 < best_c5:

                stall = 0

def get_c5(h):
    """Compute C5 = max_k np.correlate(h, 1-h, mode='full')[k] * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, seed=42):
    """Ensure h is binary with sum = len(h)/2"""
    n = len(h)
    h_b = (h > 0.5).astype(np.int8)
    target = n // 2
    n_ones = h_b.sum()
    rng = np.random.RandomState(seed)
    if n_ones > target:
        idx = np.where(h_b == 1)[0]
        perm = rng.permutation(idx)
        h_b[perm[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h_b == 0)[0]
        perm = rng.permutation(idx)
        h_b[perm[:target - n_ones]] = 1
    return h_b

def upscale_binary(h, new_n, seed=42):
    """Upscale binary pattern to new_n, maintaining sum = new_n/2"""
    old_n = len(h)
    # For each new position, determine which old position it maps to
    # Use rounding for a "nearest" upscale
    scale = old_n / new_n
    new_positions = (np.arange(new_n) * scale + 0.5).astype(int)
    new_positions = np.clip(new_positions, 0, old_n - 1)
    h_new = h[new_positions]
    # Fix sum
    h_new = make_binary(h_new.astype(float), seed)
    return h_new
