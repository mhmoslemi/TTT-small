def compute_c5(h, n_points):
    """Compute C5 value for given h."""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy 0<=h<=1 and sum(h)=n_points/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    s = np.sum(h)
    if s < 1e-10:
        return np.full(n_points, target / n_points)
    # Scale
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Iterate to fix sum after clipping
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        # Adjust: add/subtract uniformly from non-clipped elements
        free = (h > 0) & (h < 1)
        diff = target - s
        if np.any(free):
            n_free = np.sum(free)
            adjustment = diff / n_free
            h[free] += adjustment
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    # Final fix
    s = np.sum(h)
    h += (target - s) / n_points
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def compute_grad(h, n_points):
    """Gradient of max_k corr[k] * dx with respect to h."""
    dx = 2.0 / n_points
    n = n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(conv)
    m = k_max - (n - 1)  # lag
    
    grad = np.zeros(n)
    if m >= 0:
        # c_m = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
        # d/dh[j]: 
        #   j as i+m: j-m = i, valid if 0<=j-m<=n-1-m → m<=j<=n-1
        #     contribution: (1-h[j-m])
        #   j as i: i=j, valid if 0<=j<=n-1-m
        #     contribution: -h[j+m]
        if m < n:
            grad[m:] += (1.0 - h[:n-m]) * dx
        if n - m > 0:
            grad[:n-m] += (-h[m:]) * dx
    else:
        p = -m
        # c_{-p} = sum_{i=p}^{n-1} h[i-p] * (1-h[i])
        # d/dh[j]:
        #   j as i-p: i=j+p, valid if p<=j+p<=n-1 → 0<=j<=n-1-p
        #     contribution: (1-h[j+p])
        #   j as i: i=j, valid if p<=j<=n-1
        #     contribution: -h[j-p]
        if n - p > 0:
            grad[:n-p] += (1.0 - h[p:]) * dx
        if p < n:
            grad[p:] += (-h[:n-p]) * dx
    
    return grad

def project_feasible(h, n_points):
    """Project to [0,1]^n ∩ {sum = n/2}."""
    n = n_points
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum
            free = h < 1.0 - 1e-10
            if not np.any(free):
                break
            add = diff / np.sum(free)
            h[free] = np.minimum(h[free] + add, 1.0)
        else:
            free = h > 1e-10
            if not np.any(free):
                break
            sub = (-diff) / np.sum(free)
            h[free] = np.maximum(h[free] - sub, 0.0)
    # Final correction
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Distribute remainder
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def objective_with_grad(h, n_points):
    """Objective and gradient for SLSQP."""
    c5 = compute_c5(h, n_points)
    g = compute_grad(h, n_points)
    return c5, g

def slsqp_optimize(h0, n_points, maxiter=200):
    """Optimize using SLSQP."""
    n = n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                    'jac': lambda x: np.ones(n)]}
    bounds = [(0, 1)] * n
    
    def obj(x):
        return compute_c5(x, n)
    
    def obj_grad(x):
        return compute_grad(x, n)
    
    result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
    return result.x, result.fun

def projected_gradient_descent(h0, n_points, n_iter=500, lr=0.01):
    """Projected gradient descent."""
    n = n_points
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    for it in range(n_iter):
        grad = compute_grad(h, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
        # Adapt learning rate
        if it % 100 == 0:
            lr *= 0.9
    
    return best_h, best_val

            best_h = h_new.copy()
        h = h_new
        if it % 50 == 49:
            lr *= 0.85
    return best_h, best_val


def binary_round(h, n_points):
    n = n_points
    target = n / 2.0
    h_sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:int(target)]] = 1.0
    return h_bin


def smooth_max_obj(h, n_points, temp=0.05):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = conv - np.max(conv)
    logsum = np.log(np.sum(np.exp(shifted / temp))) + np.max(conv) / temp
    return temp * logsum * n_points / n_points


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Initialize from parent
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if h0 is None or len(h0) == 0:
        n0 = 70
        h0 = np.full(n0, 0.5)
    else:
        n0 = len(h0)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_val = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: Polish initial with SLSQP (long)
    if time_left() > 60:
        try:
            h_try, v_try = slsqp_optimize(h0, n0, maxiter=1000, ftol=1e-12)
            h_try = project_feasible(h_try, n0)
            v_try = compute_c5(h_try, n0)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 2: Aggressive random restarts with SLSQP
    attempt = 0
    while time_left() > 300 and attempt < 500:
        strategy = attempt % 4
        if strategy == 0:
            h_rand = rng.uniform(0, 1, n0)
        elif strategy == 1:
            h_rand = rng.binomial(1, 0.5, n0).astype(float)
        elif strategy == 2:
            h_rand = np.mod(np.linspace(0, 1, n0) + rng.uniform(0, 0.3, 1), 1.0)
        else:
            h_rand = best_h + rng.normal(0, 0.3, n0)
            h_rand = np.clip(h_rand, 0, 1)
        h_rand = project_feasible(h_rand, n0)
        attempt += 1
        try:
            h_try, v_try = slsqp_optimize(h_rand, n0, maxiter=200, ftol=1e-10)
            h_try = project_feasible(h_try, n0)
            v_try = compute_c5(h_try, n0)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            try:
                h_try, v_try = projected_grad_descent(h_rand, n0, n_iter=100)
                if v_try < best_val:
                    best_h, best_val = h_try, v_try
            except Exception:
                pass

    # Phase 3: Perturb best and re-optimize (multiple scales)
    for i in range(40):
        if time_left() < 120:
            break
        for scale in [0.03, 0.05, 0.08, 0.12, 0.2]:
            if time_left() < 60:
                break
            if rng.random() < 0.3:
                h_pert = best_h + rng.normal(0, scale, n0)
            else:
                idx = rng.choice(n0, size=max(2, n0 // 5), replace=False)
                h_pert = best_h.copy()
                h_pert[idx] = np.clip(h_pert[idx] + rng.normal(0, scale * 2, len(idx)), 0, 1)
            h_pert = project_feasible(h_pert, n0)
            try:
                h_try, v_try = slsqp_optimize(h_pert, n0, maxiter=200, ftol=1e-10)
                h_try = project_feasible(h_try, n0)
                v_try = compute_c5(h_try, n0)
                if v_try < best_val:
                    best_h, best_val = h_try, v_try
            except Exception:
                pass

    # Phase 4: Try binary-rounded versions
    if time_left() > 80:
        h_bin = binary_round(best_h, best_n)
        # Add small perturbation to break symmetry
        h_bin = project_feasible(h_bin + rng.normal(0, 0.01, best_n), best_n)
        try:
            h_try, v_try = slsqp_optimize(h_bin, best_n, maxiter=300)
            h_try = project_feasible(h_try, best_n)
            v_try = compute_c5(h_try, best_n)
            if v_try < best_val:
                best_h, best_val = h_try, v_try
        except Exception:
            pass

    # Phase 5: Higher resolution
    if time_left() > 200:
        for n_fine in [140, 210, 280]:
            if time_left() < 100:
                break
            x_coarse = np.linspace(0, 2, best_n, endpoint=False)
            x_fine = np.linspace(0, 2, n_fine, endpoint=False)
            h_fine = np.interp(x_fine, x_coarse, best_h)
            h_fine = project_feasible(h_fine, n_fine)
            try:
                h_try, v_try = slsqp_optimize(h_fine, n_fine, maxiter=400, ftol=1e-10)
                h_try = project_feasible(h_try, n_fine)
                v_try = compute_c5(h_try, n_fine)
                if v_try < best_val:
                    best_h, best_val, best_n = h_try, v_try, n_fine
            except Exception:
                pass
            # Also try fresh random at higher res
            if time_left() > 80:
                try:
                    h_rand_fine = project_feasible(rng.uniform(0, 1, n_fine), n_fine)
                    h_try, v_try = slsqp_optimize(h_rand_fine, n_fine, maxiter=300)
                    h_try = project_feasible(h_try, n_fine)
                    v_try = compute_c5(h_try, n_fine)
                    if v_try < best_val:
                        best_h, best_val, best_n = h_try, v_try, n_fine
                except Exception:
                    pass

    # Phase 6: Final aggressive polish
    if time_left() > 30:
        for _ in range(5):
            if time_left() < 15:
                break
            try:
                h_try, v_try = slsqp_optimize(best_h, best_n, maxiter=500, ftol=1e-12)
                h_try = project_feasible(h_try, best_n)
                v_try = compute_c5(h_try, best_n)
                if v_try < best_val:
                    best_h, best_val = h_try, v_try
                else:
                    break
            except Exception:
                break

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_val = compute_c5(best_h, best_n)

    return (best_h, best_val, best_n)
