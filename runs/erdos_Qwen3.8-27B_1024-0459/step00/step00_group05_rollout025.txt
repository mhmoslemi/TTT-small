    grad *= dx
    return f, grad

            continue
        idx = np.arange(i_min, i_max)
        grad[idx] += w[k] * ((1.0 - h[idx + s]) - h[idx - s])
    grad *= dx
    return f, grad

def proj_grad_descent(h_init, n, t_list, max_iter_total, t_start, budget_s, reserve):
    """Projected gradient descent with increasing smoothing parameter."""
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h, n)
    lr = 0.05
    
    for t_val in t_list:
        lr = 0.05
        for it in range(max_iter_total):
            if time.time() - t_start > budget_s - reserve:
                return best_h, best_score
            h = project_h(h, n)
            f_raw = compute_c5(h, n)
            if f_raw < best_score:
                best_score = f_raw
                best_h = h.copy()
            _, g = smooth_obj_and_grad(h, n, t_val)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-8:
                break
            g = g / g_norm * min(g_norm, 1.0)
            # Backtracking
            alpha = lr
            improved = False
            for _ in range(25):
                h_try = project_h(h - alpha * g, n)
                f_try = compute_c5(h_try, n)
                if f_try < f_raw - 1e-12:
                    h = h_try
                    improved = True
                    if f_try < best_score:
                        best_score = f_try
                        best_h = h.copy()
                    break
                alpha *= 0.5
            if improved:
                lr = min(lr * 1.02, 0.5)
            else:
                lr *= 0.7
            if lr < 1e-7:
                break
    
    return best_h, best_score

def local_search(h_init, n, t_start, budget_s, reserve, rng, max_rounds=50):
    """Coordinate-wise local search."""
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h, n)
    
    for rnd in range(max_rounds):
        if time.time() - t_start > budget_s - reserve:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - t_start > budget_s - reserve:
                break
            old = h[i]
            try_vals = set()
            try_vals.add(0.0)
            try_vals.add(1.0)
            for d in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]:
                v = old + d
                if 0.0 <= v <= 1.0:
                    try_vals.add(round(v, 6))
            try_vals.add(round(rng.uniform(0, 1), 6))
            
            for cv in try_vals:
                if abs(cv - old) < 1e-9:
                    continue
                h_try = h.copy()
                h_try[i] = cv
                h_try = project_h(h_try, n)
                sc = compute_c5(h_try, n)
                if sc < best_score:
                    best_score = sc
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
        if not improved:
            break
    
    return best_h, best_score

def simulated_annealing(h_init, n, t_start, budget_s, reserve, rng, n_iter=8000):
    """Simulated annealing to escape local optima."""
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h, n)
    T = 0.04
    
    for it in range(n_iter):
        if time.time() - t_start > budget_s - reserve:
            break
        h_try = h.copy()
        k_pert = rng.randint(1, 5)
        idx = rng.choice(n, k_pert, replace=False)
        h_try[idx] += rng.normal(0, T * 1.5, k_pert)
        h_try = project_h(h_try, n)
        sc_try = compute_c5(h_try, n)
        sc_curr = compute_c5(h, n)
        delta = sc_try - sc_curr
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            if sc_try < best_score:
                best_score = sc_try
                best_h = h.copy()
        T *= 0.9995
        if T < 0.002:
            T = 0.03
    
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 70
        best_h = np.full(n, 0.5)
    
    best_h = project_h(best_h, n)
    best_score = compute_c5(best_h, n)
    
    # Phase 1: Multi-schedule projected gradient descent (~300s)
    t_list1 = [5, 20, 50, 100, 300, 500]
    h_cand, sc = proj_grad_descent(best_h, n, t_list1, max_iter_total=200,
                                    t_start=t_start, budget_s=budget_s, reserve=500)
    if sc < best_score:
        best_score = sc
        best_h = h_cand
    
    # Phase 2: Local search (~100s)
    h_cand, sc = local_search(best_h, n, t_start, budget_s, reserve=400, rng=rng, max_rounds=60)
    if sc < best_score:
        best_score = sc
        best_h = h_cand
    
    # Phase 3: Simulated annealing (~200s)
    h_cand, sc = simulated_annealing(best_h, n, t_start, budget_s, reserve=200, rng=rng, n_iter=10000)
    if sc < best_score:
        best_score = sc
        best_h = h_cand
    
    # Phase 4: More gradient descent with refined schedule
    t_list2 = [100, 300, 1000, 2000]
    h_cand, sc = proj_grad_descent(best_h, n, t_list2, max_iter_total=300,
                                    t_start=t_start, budget_s=budget_s, reserve=100)
    if sc < best_score:
        best_score = sc
        best_h = h_cand
    
    # Phase 5: Final local search polish
    h_cand, sc = local_search(best_h, n, t_start, budget_s, reserve=30, rng=rng, max_rounds=40)
    if sc < best_score:
        best_score = sc
        best_h = h_cand
    
    # Phase 6: Try higher resolution if time permits
    if budget_s - (time.time() - t_start) > 180:
        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False) + 1.0 / n2
        h_interp = np.interp(x_new, x_old, np.concatenate([best_h, best_h[:1]]))
        h_interp = project_h(h_interp, n2)
        sc_interp = compute_c5(h_interp, n2)
        
        h_hi, sc_hi = proj_grad_descent(h_interp, n2, [20, 50, 100, 300],
                                         max_iter_total=150,
                                         t_start=t_start, budget_s=budget_s, reserve=60)
        if sc_hi < best_score:
            best_score = sc_hi
            best_h = h_hi
            n = n2
        elif sc_interp < best_score:
            best_score = sc_interp
            best_h = h_interp
            n = n2
    
    # Final: one more local search pass
    if budget_s - (time.time() - t_start) > 20:
        h_cand, sc = local_search(best_h, n, t_start, budget_s, reserve=5, rng=rng, max_rounds=30)
        if sc < best_score:
            best_score = sc
            best_h = h_cand
    
    # Ensure validity
    best_h = project_h(best_h, n)
    best_score = compute_c5(best_h, n)
    
    return best_h, best_score, n
