            best_h = h_cd
            best_score = s_cd
            best_n = n_best
    
    # Final validation and return
    if best_h is None:
        # Ultimate fallback
        n = 100
        best_h = np.ones(n) * 0.5
        best_score = compute_c5_score(best_h)
        best_n = n
    
    # Ensure exact constraints
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, n_final)
    final_score = compute_c5_score(best_h)
    
    return (best_h, final_score, n_final)
