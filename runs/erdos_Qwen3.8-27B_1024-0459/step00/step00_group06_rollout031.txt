```python
    return np.correlate(h, 1-h, mode='full')

    corr = compute_corr(h)
    return np.max(corr) * dx

    scaled = alpha * corr * dx
    shifted = scaled - np.max(scaled)
    return np.max(scaled) + np.log(np.sum(np.exp(shifted))) / alpha

    corr = np.correlate(h, 1-h, mode='full')
    return smooth_max(corr, dx, alpha)

        s = hp.sum()
        if s > target_sum:
```

```python
def compute_gradient(h, w, dx):
    n = len(h)
    # w has length 2n-1
    # labels = arange(-(n-1), n)
    labels = np.arange(-(n-1), n)  # length 2n-1
    
    # For each j (0..n-1) and each label:
    # term1[j,k] = 1-h[j-labels[k]] if 0<=j-labels[k]<n else 0
    # term2[j,k] = h[j+labels[k]] if 0<=j+labels[k]<n else 0
    
    js = np.arange(n)[:, None]  # (n, 1)
    ls = labels[None, :]  # (1, 2n-1)
    
    idx1 = js - ls  # (n, 2n-1): j - label
    idx2 = js + ls  # (n, 2n-1): j + label
    
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    
    # term1: 1 - h[j-label]
    term1 = np.where(mask1, 1 - h[np.clip(idx1, 0, n-1)], 0.0)
    term2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    M = term1 - term2  # (n, 2n-1)
    grad = (M @ w) * dx  # (n,)
    return grad
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    lo = h.min() - target_sum
    hi = h.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_value(corr, dx, alpha):
    """Compute smooth max of corr*dx."""
    x = alpha * corr * dx
    x_shifted = x - np.max(x)
    return np.max(x) + np.log(np.sum(np.exp(x_shifted))) / alpha

def compute_smooth_grad(h, alpha):
    """Compute gradient of smooth max of correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    x = alpha * corr * dx
    x_shifted = x - np.max(x)
    w = np.exp(x_shifted) / np.sum(np.exp(x_shifted))
    
    # grad[j] = dx * sum_k w[k] * (1 - h[j-label_k] - h[j+label_k])
    # where label_k = k - (n-1)
    labels = np.arange(-(n - 1), n)  # (2n-1,)
    js = np.arange(n)[:, None]  # (n, 1)
    ls = labels[None, :]  # (1, 2n-1)
    
    idx1 = js - ls  # (n, 2n-1)
    idx2 = js + ls  # (n, 2n-1)
    
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    
    safe_idx1 = np.clip(idx1, 0, n - 1)
    safe_idx2 = np.clip(idx2, 0, n - 1)
    
    term1 = np.where(mask1, 1.0 - h[safe_idx1], 0.0)
    term2 = np.where(mask2, h[safe_idx2], 0.0)
    
    M = term1 - term2  # (n, 2n-1)
    grad = (M @ w) * dx
    return grad

def optimize_h(h, target_sum, time_limit, start_time):
    """Optimize h using projected gradient descent with adaptive learning rate."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    alphas = [10, 20, 50, 100, 200]
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        lr = 0.1
        for it in range(200):
            if time.time() - start_time > time_limit:
                break
        
            grad = compute_smooth_grad(h, alpha)
            
            # Try step
            h_new = h - lr * grad
            h_new = project_to_constraint(h_new, target_sum)
            
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                # Increase lr slightly
                lr = min(lr * 1.05, 1.0)
            else:
                lr *= 0.7
                if lr < 0.001:
                    break
        
        # Sharpen: push towards binary
        h = best_h.copy()
        # Optionally round
        rounded = np.where(h > 0.5, 1.0, 0.0)
        rounded = project_to_constraint(rounded, target_sum)
        c5_rounded = compute_c5(rounded)
        if c5_rounded < best_c5:
            best_c5 = c5_rounded
            best_h = rounded.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    # Get initial solution
    h0 = np.array(initial_h_values, dtype=float)
    best_h = h0.copy()

    # Optimize at original resolution

    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
    
    # Try other resolutions
    for n in [64, 128, 192, 256]:
        if time.time() - start_time > budget_s * 0.8:

        # Interpolate
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_constraint(h_interp, n / 2.0)
        
        time_limit = budget_s * 0.8
        h_opt, c5_opt = optimize_h(h_interp, n / 2.0, time_limit, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        n = len(best_h)
        rounded = np.where(best_h > 0.5, 1.0, 0.0)
        rounded = project_to_constraint(rounded, n / 2.0)
        c5_r = compute_c5(rounded)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = rounded.copy()
    
    return (best_h, best_c5, len(best_h))
```

```python
def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    n = len(h)
    lo = h.min() - target_sum
    hi = h.max()

def compute_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    x_shifted = x - np.max(x)

    labels = np.arange(-(n - 1), n)
    js = np.arange(n)[:, None]

    idx2 = js + ls

    mask2 = (idx2 >= 0) & (idx2 < n)

    s2 = np.clip(idx2, 0, n - 1)

    term2 = np.where(mask2, h[s2], 0.0)
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    n = len(h)
    lo = h.min() - target_sum
    hi = h.max()
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = alpha * corr * dx
    x_shifted = x - np.max(x)
    w = np.exp(x_shifted)
    w_sum = w.sum()
    w = w / w_sum
    labels = np.arange(-(n - 1), n)
    js = np.arange(n)[:, None]
    ls = labels[None, :]
    idx1 = js - ls
    idx2 = js + ls
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    s1 = np.clip(idx1, 0, n - 1)
    s2 = np.clip(idx2, 0, n - 1)
    term1 = np.where(mask1, 1.0 - h[s1], 0.0)
    term2 = np.where(mask2, h[s2], 0.0)
    M = term1 - term2
    return (M @ w) * dx

def optimize_continuous(h_init, target_sum, start_time, end_time):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alphas = [10.0, 30.0, 80.0, 200.0]
    for alpha in alphas:
        if time.time() >= end_time:
            break
        lr = 0.3
        stall = 0
        for it in range(500):
            if time.time() >= end_time:
                break
            grad = compute_gradient(h, alpha)
            h_new = h - lr * grad
            h_new = project_to_constraint(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.03, 1.5)
                stall = 0
            else:
                lr *= 0.75
                stall += 1
                if stall > 30 or lr < 0.002:
                    break
    return best_h, best_c5

def to_binary(h, target_sum):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    nb = int(target_sum)
    frac = target_sum - nb
    hb = np.zeros(n)
    for i in range(nb):
        hb[sorted_idx[i]] = 1.0
    if frac > 1e-9:
        hb[sorted_idx[nb]] = frac
    return hb

def binary_local_search(h_init, target_sum, start_time, end_time, max_passes=500):
    n = len(h_init)
    h = to_binary(h_init, target_sum)
    c5 = compute_c5(h)
    ones = set(int(x) for x in np.where(h >= 1.0 - 1e-9)[0])
    zeros = set(int(x) for x in np.where(h <= 1e-9)[0])
    for _ in range(max_passes):
        if time.time() >= end_time:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5
        best_swap = None
        ones_list = list(ones)
        zeros_list = list(zeros)
        for i in ones_list:
            h[i] = 0.0
            for j in zeros_list:
                h[j] = 1.0
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    best_swap = (i, j)
                h[j] = 0.0
            h[i] = 1.0
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            c5 = best_c5
            ones.discard(i)
            zeros.discard(j)
            ones.add(j)
            zeros.add(i)
        else:
            break
    return h, c5
```
