from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def smooth_max(c, beta):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def negative_smooth_max(h, n, beta, lam):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    val = -smooth_max(c, beta)
    s = np.sum(h) - n / 2.0
    val -= lam * s * s
    return val

def project_feasible(h, n):
    h = np.clip(h, 0, 1)
    s = np.sum(h) - n / 2.0
    if s > 0:
        # reduce where h is largest
        idx = np.argsort(h)[::-1]
        remaining = s
        for i in idx:
            red = min(h[i], remaining)
            h[i] -= red
            remaining -= red
            if remaining <= 0:
                break
    elif s < 0:
        # increase where h is smallest
        idx = np.argsort(h)
        remaining = -s
        for i in idx:
            inc = min(1 - h[i], remaining)
            h[i] += inc
            remaining -= inc
            if remaining <= 0:
                break
    return h

def binary_local_search(h_bin, n, max_iters=5000):
    """Local search for binary solution. Try flipping single bits."""
    h = h_bin.copy()
    current_c5 = compute_c5(h, n)
    
    for _ in range(max_iters):
        best_flip = None
        best_c5 = current_c5
        n_flip = np.sum(h == 1)
        n_zero = n - n_flip
        
        # Try flipping a 1 to 0
        if n_flip > 0:
            ones = np.where(h == 1)[0]
            for i in ones:
                h[i] = 0
                c5 = compute_c5(h, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_flip = (i, 0)
                h[i] = 1
        
        # Try flipping a 0 to 1
        if n_zero > 0:
            zeros = np.where(h == 0)[0]
            for i in zeros:

                c5 = compute_c5(h, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_flip = (i, 1)
                h[i] = 0
        
        if best_flip is None:

        h[best_flip[0]] = best_flip[1]
        current_c5 = best_c5

def smooth_obj(h, n, beta, lam):

    m = np.max(c)
    exp_c = np.exp(beta * (c - m))

    val = lse / beta
    s = np.sum(h) - n / 2.0
    val += lam * s * s
    return val


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = np.sum(h) - n / 2.0
        if abs(s) < 1e-10:
            break
        h = h - s / n
        h = np.clip(h, 0.0, 1.0)
    return h


def binary_swap_improve(h, n, max_swaps, deadline):
    c5 = compute_c5_max(h, n)
    for _ in range(max_swaps):

        c5_new = compute_c5_max(h, n)

            h[i] = 1.0

def interpolate_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)

            smooth_obj, h_start, args=args,
            method='L-BFGS-B',

            options={'maxiter': max_iter, 'maxfun': max_iter * n, 'ftol': 1e-12}

        h_opt = res.x

            best_local_c5 = c5_opt
            best_local = h_opt.copy()
        h_start = h_opt.copy()
    return best_local, best_local_c5
