import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx, c

def project(h, target_sum):
    """Project h onto {x: 0 ≤ x ≤ 1, sum(x) = target_sum}"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if diff > 0:
        # Need to increase sum: increase elements not at 1
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        inc = diff / mask.sum()
        h[mask] = np.minimum(h[mask] + inc, 1.0)
    elif diff < 0:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        dec = (-diff) / mask.sum()
        h[mask] = np.maximum(h[mask] - dec, 0.0)
    # Fine adjustment
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-10:
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff2 / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def gradient_max_corr(h, n):
    """Compute gradient of max correlation wrt h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)  # actual lag
    
    grad = np.zeros(n)
    if k >= 0:
        # c = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        # d/dc/dh[j] = (1-h[j-k]) for j in [k, n-1] - h[j+k] for j in [0, n-1-k]
        for j in range(k, n):
            grad[j] += (1 - h[j - k])
        for j in range(0, n - k):
            grad[j] -= h[j + k]
    else:
        m = -k
        # c = sum_{i=m}^{n-1} h[i-m] * (1-h[i])

        for j in range(m, n):
            grad[j] += (1 - h[j + m])
        for j in range(0, n - m):
            grad[j] -= h[j + m]
    
    return grad, corr[max_idx]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c) * dx, c

def project_constraint(h, target_sum):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        if not np.any(mask):
            return h
        h[mask] = np.minimum(h[mask] + diff / np.sum(mask), 1.0)
    else:
        mask = h > 0.0
        if not np.any(mask):
            return h
        h[mask] = np.maximum(h[mask] - (-diff) / np.sum(mask), 0.0)
    # Fine tune
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-10:
        mask = (h > 0) & (h < 1)
        if np.any(mask):
            h[mask] += diff2 / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def lag_gradient(h, k):
    """Compute gradient of correlation at lag k wrt h"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[k:] += (1 - h[:n-k])
            grad[:n-k] -= h[k:]
    else:
        m = -k
        if m < n:
            grad[:n-m] += (1 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

    for idx in top_indices:
        if idx == max_idx:
            continue
        k = idx - (n-1)
        grad += weight * lag_gradient(h, k)
    
    return grad

        if norm < 1e-10:

def optimize_sa(h, n, max_iters=50000, time_limit=100):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx, c

def project_constraint(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if diff > 1e-12:
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] = np.clip(h[mask] + diff / cnt, 0, 1)
    elif diff < -1e-12:
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:

                if h[i] < h[i + k]:
                    tmp = h[i].copy()
                    h[i] = h[i + k]
                    h[i + k] = tmp
                    h = project_constraint(h, target_sum)
                    c5_new = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
                    if c5_new < c5_curr:
                        c5_curr = c5_new
                        improved = True
                        if c5_curr < best_c5:
                            best_c5 = c5_curr
                            best_h = h.copy()
                    else:
                        h[i + k] = h[i].copy()
                        h[i] = tmp
                        h = project_constraint(h, target_sum)
        elif k < 0:
            m = -k
            for i in range(0, n - m, max(1, (n - m) // 80)):
                if h[i + m] < h[i]:
                    tmp = h[i].copy()
                    h[i] = h[i + m]
                    h[i + m] = tmp
                    h = project_constraint(h, target_sum)
                    c5_new = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
                    if c5_new < c5_curr:
                        c5_curr = c5_new
                        improved = True
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h.copy()
                    else:
                        h[i + m] = h[i].copy()
                        h[i] = tmp
                        h = project_constraint(h, target_sum)
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                h[i], h[j] = h[j], h[i]
                h = project_constraint(h, target_sum)
                c5_new = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
                if c5_new < c5_curr:
                    c5_curr = c5_new
                    improved = True
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
                    h = project_constraint(h, target_sum)
        if not improved:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and abs(h[i] - h[j]) > 0.01:
                h[i], h[j] = h[j], h[i]
                h = project_constraint(h, target_sum)
                c5_new = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
                if c5_new <= c5_curr:
                    c5_curr = c5_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[j], h[i]
                    h = project_constraint(h, target_sum)
        ticks += 1
        if ticks % 5000 == 0:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    else:
        n0 = 100
        h0 = np.full(n0, 0.5)

    c5_0, _ = compute_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_0

    def elapsed():
        return time.time() - t_start

    # Phase 1: gradient at initial resolution
    t1 = max(5, budget_s * 0.12)
    h1, c5_1 = optimize_gradient(h0, n0, t1, rng)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1

    if elapsed() > budget_s * 0.95:
        return (best_h, best_c5, len(best_h))

    # Phase 2: local search at initial resolution
    t_ls = max(5, budget_s * 0.08)
    h2, c5_2 = local_search(best_h, n0, rng, t_ls)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2

    if elapsed() > budget_s * 0.9:
        return (best_h, best_c5, len(best_h))

    # Phase 3: try multiple resolutions
    resolutions = [128, 160, 200, 256, 320]
    remaining = budget_s * 0.7 - elapsed()
    alloc_per = max(15, remaining / max(1, len(resolutions)))

    for n in resolutions:
        if elapsed() > budget_s * 0.7:
            break
        if n == n0:
            continue
        h_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_h)), best_h)
        h_interp = project_constraint(h_interp, n / 2.0)
        rem = min(alloc_per, budget_s * 0.7 - elapsed())
        if rem < 10:
            continue
        h_g, c5_g = optimize_gradient(h_interp, n, rem * 0.55, rng)
        h_l, c5_l = local_search(h_g, n, rng, min(rem * 0.3, 30))
        if c5_l < c5_g:
            h_g, c5_g = h_l, c5_l
        if c5_g < best_c5:
            best_c5, best_h = c5_g, h_g

    if elapsed() > budget_s * 0.9:
        final_c5, _ = compute_c5(best_h)
        best_c5 = min(best_c5, final_c5)
        return (best_h, best_c5, len(best_h))

    # Phase 4: further optimization at best resolution
    n_best = len(best_h)
    rem2 = max(5, budget_s * 0.88 - elapsed())
    h3, c5_3 = optimize_gradient(best_h, n_best, rem2 * 0.6, rng)
    h4, c5_4 = local_search(h3, n_best, rng, min(rem2 * 0.35, 40))
    if c5_4 < c5_3:
        h3, c5_3 = h4, c5_4
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3

    # Phase 5: random restarts at best resolution
    if elapsed() < budget_s * 0.92:
        n_r = len(best_h)
        for _ in range(3):
            if elapsed() > budget_s * 0.93:
                break
            h_rand = np.where(rng.random(n_r) < 0.5, 0.0, 1.0)
            h_rand = project_constraint(h_rand, n_r / 2.0)
            h_r, c5_r = optimize_gradient(h_rand, n_r, min(20, budget_s * 0.93 - elapsed()), rng)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r

    final_c5, _ = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, len(best_h))
