    """Compute C5 for a given h vector."""

    cor = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cor) * dx)

def _obj_and_grad(h, n):
    """Compute objective and gradient."""
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(cor))
    val = cor[max_idx] * dx
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] += -h[lag:]
    else:
        ll = -lag
        grad[:n-ll] += (1.0 - h[ll:])
        grad[ll:] += -h[:n-ll]
    return val, grad * dx

def _make_objective(n):
    """Create a smooth objective using log-sum-exp."""
    def smooth_obj(h):
        dx = 2.0 / n
        cor = np.correlate(h, 1.0 - h, mode="full") * dx
        # Soft max
        alpha = 50.0
        m = np.max(cor)
        return (m + np.log(np.sum(np.exp(cor - m))) / alpha)
    return smooth_obj
