    M=np.correlate(h, 1.0-h, mode='full')

    em=np.exp(m-mm)
    ss=em.sum()

def project_simplex(x, target_sum):
    x = np.asarray(x, dtype=float).copy()
    n = len(x)
    if n == 0:
        return x
    target_sum = float(target_sum)
    if target_sum <= 0.0:
        return np.zeros(n, dtype=float)
    if target_sum >= float(n):
        return np.ones(n, dtype=float)
    # bisection
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    z = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # exact adjustment
    for _ in range(5):
        diff = target_sum - float(z.sum())
        if abs(diff) < 1e-14 * max(1.0, abs(target_sum)):
            break
        if diff > 0:
            pos = np.where(z < 1.0 - 1e-12)[0]
            if pos.size == 0:

            for i in pos:
                add = min(inc, 1.0 - z[i])
                z[i] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
        else:
            pos = np.where(z > 1e-12)[0]
            if pos.size == 0:
                pos = np.arange(n)
            dec = -diff / pos.size
            for i in pos:
                sub = min(dec, z[i])
                z[i] -= sub
                diff += sub
                if abs(diff) < 1e-15:
                    break
        z = np.clip(z, 0.0, 1.0)

def project_simplex(x, target_sum):
    x = np.asarray(x, dtype=float).copy()
    n = len(x)
    if n == 0:
        return x

    target_sum = float(target_sum)
    if target_sum <= 0.0:
        return np.zeros(n, dtype=float)
    if target_sum >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0

    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    z = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(5):
        diff = target_sum - float(z.sum())
        if abs(diff) < 1e-14 * max(1.0, abs(target_sum)):
            break

        if diff > 0:
            pos = np.where(z < 1.0 - 1e-12)[0]
            if pos.size == 0:
                pos = np.arange(n)
            inc = diff / float(pos.size)
            for i in pos:
                add = min(inc, 1.0 - z[i])
                z[i] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break
        else:
            pos = np.where(z > 1e-12)[0]
            if pos.size == 0:
                pos = np.arange(n)
            dec = (-diff) / float(pos.size)
            for i in pos:
                sub = min(dec, z[i])
                z[i] -= sub
                diff += sub
                if abs(diff) < 1e-15:
                    break

        z = np.clip(z, 0.0, 1.0)

        order = np.argsort(np.minimum(z, 1.0 - z))[::-1]

                diff -= add
            else:

                z[i] -= sub
                diff += sub
            if abs(diff) < 1e-15:

    return z

def eval_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0

    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))


def smooth_obj_grad(h, beta):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / float(n)

    M = np.correlate(h, 1.0 - h, mode="full")
    C = M * dx

    m = beta * C
    mmax = float(np.max(m))
    em = np.exp(m - mmax)
    ss = float(np.sum(em))
    p = em / ss

    f = (mmax + float(np.log(ss))) / beta

    lin = np.convolve(p, np.ones(n, dtype=float), mode="valid")

    K = np.zeros(2 * n - 1, dtype=float)
    K[n - 1] = 2.0 * p[n - 1]
    if n > 1:
        w = p[n:2 * n - 1] + p[0:n - 1][::-1]
        K[n:2 * n - 1] = w
        K[0:n - 1] = w[::-1]

    quad_full = np.convolve(h, K, mode="full")
    quad = quad_full[n - 1:2 * n - 1]

    grad = dx * (lin - quad)
    return f, grad

def init_binary_M(bits):
    b = bits.astype(np.int64)
    M = np.correlate(b, 1 - b, mode="full").astype(np.int32, copy=False)
    M[len(bits) - 1] = 0
    return M


def apply_flip(bits, M, p, d):
    n = len(bits)
    d = int(d)
    M[p:n + p] += d * (1 - bits)[::-1]
    M[n - 1 - p:2 * n - 1 - p] -= d * bits
    M[n - 1] = 0


def apply_swap_inplace(bits, M, p, q):
    apply_flip(bits, M, p, -1)
    bits[p] = 0
    apply_flip(bits, M, q, 1)
    bits[q] = 1


def simulate_swap(bits, M, p, q):
    saved = M.copy()
    apply_flip(bits, saved, p, -1)
    bits[p] = 0
    apply_flip(bits, saved, q, 1)
    bits[q] = 1
    cnt = int(np.max(saved))
    bits[p] = 1
    bits[q] = 0
    return saved, cnt


def random_pair(bits, rng):
    n = len(bits)
    for _ in range(30):
        p = int(rng.integers(n))
        q = int(rng.integers(n))
        if bits[p] == 1 and bits[q] == 0 and p != q:
            return p, q

    ones = np.flatnonzero(bits)
    zeros = np.flatnonzero(1 - bits)
    if ones.size == 0 or zeros.size == 0:
        return 0, 1

    p = int(ones[int(rng.integers(ones.size))])
    q = int(zeros[int(rng.integers(zeros.size))])
    return p, q


def compute_flip_scores(bits, top_idx, weights):
    n = len(bits)
    D = np.zeros(n, dtype=float)
    one_minus = 1 - bits

    for t in range(len(top_idx)):
        idx = int(top_idx[t])
        w = float(weights[t])
        if w == 0.0:
            continue
        k = idx - (n - 1)
        if k == 0:
            continue

        s = max(0, k)
        e = min(n, n + k)
        if s < e:
            start_i = s - k
            ln = e - s
            D[s:e] += w * one_minus[start_i:start_i + ln]

        s = max(0, -k)
        e = min(n, n - k)
        if s < e:
            start_j = s + k
            ln = e - s
            D[s:e] -= w * bits[start_j:start_j + ln]

    return D

def make_bits_block(n, start_one=True):

def make_bits_random(n, rng):
    bits = np.zeros(n, dtype=np.int8)
    S = n // 2
    if S > 0:
        idx = rng.choice(n, size=S, replace=False)
        bits[idx] = 1
    return bits


def make_bits_sine(n, freq, rng):
    S = n // 2
    x = np.linspace(0.0, 2.0, n)
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    vals = 0.5 + 0.5 * np.sin(np.pi * float(freq) * x + phase)
    order = np.argsort(vals)
    bits = np.zeros(n, dtype=np.int8)
    bits[order[::-1][:S]] = 1
    return bits


def random_composition(S, R, rng):
    S = int(S)
    R = int(R)
    if S <= 0:
        return []
    if R <= 1:
        return [S]
    R = min(R, S)
    if R <= 1:
        return [S]
    if S - 1 < R - 1:
        return [1] * S

    cuts = np.sort(rng.choice(np.arange(1, S), size=R - 1, replace=False))
    parts = []
    prev = 0
    for c in cuts:
        c = int(c)
        parts.append(c - prev)
        prev = c
    parts.append(S - prev)
    return parts


def make_bits_runs(n, rng):
    S = n // 2
    if S <= 0:
        return np.zeros(n, dtype=np.int8)

    R = int(min(10, max(3, n // 6)))
    R = min(R, S)

    one_parts = random_composition(S, R, rng)
    zero_parts = random_composition(S, R, rng)

    if len(one_parts) != len(zero_parts):
        return make_bits_random(n, rng)

    start_one = rng.random() < 0.5
    vals = []
    if start_one:
        for a, b in zip(one_parts, zero_parts):
            vals.extend([1] * int(a))
            vals.extend([0] * int(b))
    else:
        for a, b in zip(zero_parts, one_parts):
            vals.extend([0] * int(a))
            vals.extend([1] * int(b))

    if len(vals) != n:
        return make_bits_random(n, rng)

    bits = np.array(vals, dtype=np.int8)
    if int(bits.sum()) != S:
        return make_bits_random(n, rng)

    return bits


def make_bits_from_parent(n, parent, rng):
    parent = np.asarray(parent, dtype=float).ravel()
    S = n // 2
    if S <= 0:
        return np.zeros(n, dtype=np.int8)
    if S >= n:
        return np.ones(n, dtype=np.int8)

    if parent.size == n:
        vals = parent
    else:
        xp = np.linspace(0.0, 2.0, parent.size)
        xn = np.linspace(0.0, 2.0, n)
        vals = np.interp(xn, xp, parent)

    jit = rng.random(n) * 1e-9
    order = np.argsort(vals + jit)
    bits = np.zeros(n, dtype=np.int8)
    bits[order[::-1][:S]] = 1
    return bits

                best_c = cr
                best_n = len(hr)

    final_reserve = min(180.0, max(60.0, 0.12 * total_budget))
    binary_deadline = deadline - final_reserve
    if binary_deadline < start_time + 20.0:
        binary_deadline = deadline - 10.0
        final_reserve = 10.0

    phases = [
        (44, 16),
        (46, 13),
        (48, 11),
        (50, 10),
        (54, 9),
        (60, 8),
        (70, 6),
        (80, 6),
        (100, 6),
        (120, 5),
    ]

    for phase_idx, (n, weight) in enumerate(phases):
        now = time.time()
        if now >= binary_deadline:
            break

        remaining = binary_deadline - now
        rem_w = 0
        for j in range(phase_idx, len(phases)):
            rem_w += phases[j][1]
        rem_w = max(1.0, float(rem_w))

        t_phase = remaining * float(weight) / rem_w
        if t_phase < 3.0:
            continue

        max_starts = 1
        if t_phase >= 15.0:
            max_starts = 2
        if t_phase >= 40.0:
            max_starts = 3
        if t_phase >= 80.0:
            max_starts = 4

        starts = get_starts(n, parent, rng, max_starts)
        if not starts:
            starts = [make_bits_block(n, True)]

        per_start = t_phase / float(len(starts))

        for bits0 in starts:
            t_start = min(per_start, binary_deadline - time.time())
            if t_start <= 1.0:
                break

            if n <= 80:
                bbits, bcnt = binary_local_search_steepest(n, bits0, t_start, rng)
            else:
                bbits, bcnt = binary_local_search_score(n, bits0, t_start, rng)

            if bcnt < 10**8:
                c5 = float(bcnt) * (2.0 / float(n))
                if c5 < best_c:
                    bh = make_feasible(bbits.astype(float, copy=True))
                    c5e = eval_c5(bh)
                    if c5e < best_c:
                        best_h = bh
                        best_c = c5e
                        best_n = n

    if best_h is not None:
        rem = deadline - time.time() - 5.0
        if rem > 15.0:
            t_ref = min(160.0, rem)
            hr = refine_continuous(best_h.copy(), t_ref, rng)
            hr = make_feasible(hr)
            cr = eval_c5(hr)
            if cr < best_c:
                best_h = hr
                best_c = cr
                best_n = len(hr)

    if best_h is None:
        n0 = 44
        x0 = np.linspace(0.0, 2.0, n0)
        h0 = 0.5 - 0.5 * np.cos(np.pi * x0)
        h0 = project_simplex(h0, 0.5 * float(n0))
        best_h = h0
        best_c = eval_c5(h0)
        best_n = n0
    else:
        best_h = make_feasible(best_h)
        best_c = eval_c5(best_h)
        best_n = len(best_h)

    return best_h, float(best_c), int(best_n)
