        print(f"Default: n={n0}, C5={best_c5:.8f}", flush=True)

    # Phase 1: SLSQP on initial resolution
    if time_left() > 80:
        try:
            res = slsqp_refine(best_h, len(best_h), maxiter=5000, ftol=1e-14)
            h_new = fix_constraint(res.x, len(best_h))
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            # Try snap
            h_snap = snap_refine(best_h, len(best_h), 0.01)
            c5_snap = compute_c5(h_snap)
            if c5_snap < best_c5:
                best_h = h_snap
                best_c5 = c5_snap
            print(f"Phase1 n={len(best_h)}: C5={best_c5:.8f}", flush=True)
        except Exception as e:
            print(f"Phase1 error: {e}", flush=True)

    # Phase 2: Higher resolutions
    n_cur = len(best_h)
    res_list = []
    for mult in [1.5, 2, 3]:
        nr = int(n_cur * mult)
        if nr > n_cur + 10 and nr <= 400:
            res_list.append(nr)
    # Also try specific values
    for nv in [140, 190, 250, 320, 400]:
        if nv > n_cur and nv not in res_list:
            res_list.append(nv)
    res_list = sorted(set(res_list))

    for n_new in res_list[:4]:
        if time_left() < 100:
            break
        h_interp = interpolate_to(best_h, n_new)
        h_start = fix_constraint(h_interp, n_new)
        try:
            res = slsqp_refine(h_start, n_new, maxiter=4000, ftol=1e-13)
            h_new = fix_constraint(res.x, n_new)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_snap = snap_refine(best_h if c5_new < best_c5 else h_new, n_new, 0.01)
            c5_snap = compute_c5(h_snap)
            if c5_snap < best_c5:
                best_h = h_snap
                best_c5 = c5_snap
            print(f"Phase2 n={n_new}: C5={compute_c5(fix_constraint(h_new, n_new)):.8f}, best={best_c5:.8f}", flush=True)
        except Exception as e:
            print(f"Phase2 n={n_new} error: {e}", flush=True)

    # Phase 3: Basin hopping
    n_bh = len(best_h)
    max_trials = min(40, int(max(0, time_left()) // 12))
    for i in range(max_trials):
        if time_left() < 25:
            break
        scale = 0.04 * (1.0 - i / max(max_trials, 1)) + 0.003
        h_pert = best_h + np.random.normal(0, scale, n_bh)
        h_pert = fix_constraint(h_pert, n_bh)
        try:
            res = slsqp_refine(h_pert, n_bh, maxiter=2000, ftol=1e-12)
            h_new = fix_constraint(res.x, n_bh)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                print(f"Basin {i}: C5={best_c5:.8f}", flush=True)
        except:
            pass

    # Phase 4: Final high-precision polish
    if time_left() > 20:
        try:
            res = slsqp_refine(best_h, len(best_h), maxiter=8000, ftol=1e-15)
            h_new = fix_constraint(res.x, len(best_h))
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            print(f"Phase4: C5={best_c5:.8f}", flush=True)
        except:
            pass

    # Final validation
    best_h = fix_constraint(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    print(f"FINAL: n={len(best_h)}, C5={final_c5:.8f}", flush=True)
    return (best_h, final_c5, len(best_h))
