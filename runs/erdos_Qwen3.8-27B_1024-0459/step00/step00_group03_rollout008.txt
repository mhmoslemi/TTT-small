import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def fix_sum(h):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(h)
    else:
        order = np.argsort(h)[::-1]
    for idx in order:
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, h[idx])
            h[idx] -= dec
            diff += dec
        if abs(diff) < 1e-14:
            break
    return h


def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = fix_sum(h_new)
    return h_new


def minimize_coord(h, i, n_grid=16, n_golden=10):
    n = len(h)
    grid = np.linspace(0.0, 1.0, n_grid)
    best_val = 1e9
    best_t = 0.5
    best_idx = 0
    for gi in range(n_grid):
        h[i] = grid[gi]
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_t = grid[gi]
            best_idx = gi
    a = grid[max(0, best_idx - 1)]
    b = grid[min(n_grid - 1, best_idx + 1)]
    if b - a < 1e-12:
        h[i] = best_t
        return
    inv_phi = (5.0 ** 0.5 - 1.0) / 2.0
    x1 = b - (b - a) * inv_phi
    x2 = a + (b - a) * inv_phi
    h[i] = x1
    f1 = compute_c5(h)
    h[i] = x2
    f2 = compute_c5(h)
    for _ in range(n_golden):
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = b - (b - a) * inv_phi
            h[i] = x1
            f1 = compute_c5(h)
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + (b - a) * inv_phi
            h[i] = x2
            f2 = compute_c5(h)
    h[i] = (a + b) * 0.5
    f_mid = compute_c5(h)
    h[i] = 0.0
    f_lo = compute_c5(h)
    h[i] = 1.0
    f_hi = compute_c5(h)
    if f_lo <= f_mid and f_lo <= f_hi:
        h[i] = 0.0
    elif f_hi <= f_mid and f_hi <= f_lo:
        h[i] = 1.0
    else:
        h[i] = (a + b) * 0.5


def coordinate_descent(h, max_passes=50, time_end=1e18):
    n = len(h)
    for p in range(max_passes):
        if time.time() > time_end - 30:
            break
        for i in range(n):
            if time.time() > time_end - 15:
                return h
            minimize_coord(h, i, n_grid=14, n_golden=8)
        h = fix_sum(h)
    return h


def perturb(h, n_pert, magnitude):
    n = len(h)
    h = h.copy()
    idx = np.random.choice(n, size=min(n_pert, n), replace=False)
    h[idx] += np.random.uniform(-magnitude, magnitude, size=len(idx))
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h)
    return h


def binary_round(h):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    target_ones = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target_ones:
        excess = n_ones - target_ones
        ones_idx = np.where(h_bin == 1.0)[0]
        for k in range(excess):
            h_bin[ones_idx[k]] = 0.0
    elif n_ones < target_ones:
        deficit = target_ones - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        for k in range(deficit):
            h_bin[zeros_idx[k]] = 1.0
    return h_bin
