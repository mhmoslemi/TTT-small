        best_n = n_init
    except (NameError, AttributeError, TypeError):
        pass

    if best_h is None:
        best_n = 45
        best_h = make_rand_binary(best_n, rng)
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at initial resolution
    n = best_n
    if time_left() > 30:
        res = do_slsqp(best_h, n, maxiter=800, ftol=1e-12)
        if res.x is not None:
            h_opt = fix_h(res.x, n)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n
        # Second pass with LSE
        if time_left() > 30:
            res2 = do_slsqp(best_h, n, maxiter=500, ftol=1e-12, use_lse=True)
            if res2.x is not None:
                h_opt2 = fix_h(res2.x, n)
                c5 = compute_c5(h_opt2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt2.copy()

    # Phase 2: Multiple resolutions with diverse starts
    resolutions = [45, 90, 64, 90, 128, 45, 90]
    min_time_per_n = 40

    for n in resolutions:
        if time_left() < min_time_per_n:
            break

        # Build list of starting points
        starts = []

        # Upscale best
        if best_h is not None:
            if len(best_h) == n:
                starts.append(best_h.copy())
                starts.append(fix_h(best_h + rng.randn(n) * 0.04, n))
            else:
                starts.append(upscale(best_h, n))

        # Random binary starts
        for _ in range(4):
            starts.append(make_rand_binary(n, rng))

        # Structured starts
        for freq in [1, 2, 3]:
            if n >= 2 * freq:
                starts.append(make_structured(n, freq, rng))

        # Binary with specific patterns
        if n >= 10:
            h_pat = np.zeros(n)
            h_pat[::2] = 1.0
            h_pat = fix_h(h_pat + rng.randn(n) * 0.08, n)
            starts.append(h_pat)

        for h0 in starts:
            if time_left() < 20:
                break

            # SLSQP nonsmooth
            res = do_slsqp(h0, n, maxiter=600, ftol=1e-11)
            h_cand = None
            if res.x is not None:
                h_cand = fix_h(res.x, n)
                c5 = compute_c5(h_cand)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cand.copy()
                    best_n = n

            # LSE smooth pass
            if h_cand is not None and time_left() > 25:
                res2 = do_slsqp(h_cand, n, maxiter=400, ftol=1e-11, use_lse=True)
                if res2.x is not None:
                    h_c2 = fix_h(res2.x, n)
                    c5 = compute_c5(h_c2)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_c2.copy()
                        best_n = n

    # Phase 3: L-BFGS-B exploration (fast for small n)
    for n in [45, 90]:
        if time_left() < 30:
            break
        for _ in range(3):
            if time_left() < 15:
                break
            if best_h is not None and len(best_h) == n:
                h0 = fix_h(best_h + rng.randn(n) * 0.03, n)
            else:
                h0 = make_rand_binary(n, rng)
            res = do_lbfgsb(h0, n, maxiter=2000, penalty=10000.0)
            if res.x is not None:
                h_c = fix_h(res.x, n)
                c5 = compute_c5(h_c)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_c.copy()
                    best_n = n

    # Phase 4: Final intensive polish at best resolution
    if best_h is not None and time_left() > 40:
        n = best_n
        for _ in range(5):
            if time_left() < 15:
                break
            h0 = fix_h(best_h + rng.randn(n) * 0.015, n)
            res = do_slsqp(h0, n, maxiter=1000, ftol=1e-13)
            if res.x is not None:
                h_c = fix_h(res.x, n)
                c5 = compute_c5(h_c)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_c.copy()
                    best_n = n

    # Final validation
    if best_h is None:
        best_n = 45
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    best_h = fix_h(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
