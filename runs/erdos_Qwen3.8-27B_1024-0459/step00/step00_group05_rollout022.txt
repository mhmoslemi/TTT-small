import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv * dx

def gradient_for_lag(h, lag_idx):
    n = len(h)
    ell = lag_idx - (n - 1)
    grad = np.zeros(n)
    if ell >= 0:
        grad[ell:n] += 1.0
        grad[ell:n] -= h[0:n - ell]
        grad[0:n - ell] -= h[ell:n]
    else:
        m = -ell
        grad[0:n - m] += 1.0
        grad[0:n - m] -= h[m:n]
        grad[m:n] -= h[0:n - m]
    return grad

def gradient_multi_lag(h, active_indices):
    n = len(h)
    grad = np.zeros(n)
    for idx in active_indices:
        grad += gradient_for_lag(h, idx)
    grad /= float(len(active_indices))
    return grad

def project_feasible(y, n_points):
    target = n_points / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(y - lam, 0.0, 1.0)
    # Fine correction for sum
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        n_free = int(np.sum((result > 1e-10) & (result < 1.0 - 1e-10)))
        if n_free > 0:
            result[(result > 1e-10) & (result < 1.0 - 1e-10)] += diff / n_free
            result = np.clip(result, 0.0, 1.0)
    return result

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

    t_start = time.time()

    c5_best = c5_cur
    h_best = h.copy()
    alpha = initial_alpha
    patience = 0
    max_patience = 80
    perturbation_scale = 0.04
    iter_count = 0

    while time.time() - t_start < time_limit:
        iter_count += 1
        lags = compute_all_lags(h)
        curr_max = float(lags.max())

        if not np.isfinite(curr_max):
            h = h_best.copy()
            continue

        if curr_max < c5_best - 1e-12:
            c5_best = curr_max
            h_best = h.copy()
            patience = 0
        else:
            patience += 1

        threshold = curr_max * 0.98
        active = np.where(lags >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(lags)])

        grad = gradient_multi_lag(h, active)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            h = h_best.copy()
            h += rng.normal(0, 0.02, n)
            h = project_feasible(h, n)
            alpha = initial_alpha
            patience = 0
            continue

        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        new_c5 = compute_c5(h_new)

        if not np.isfinite(new_c5):
            alpha *= 0.5
            continue

        if new_c5 < curr_max - 1e-14:
            h = h_new
            if new_c5 < c5_best:
                c5_best = new_c5
                h_best = h.copy()
            alpha = min(alpha * 1.03, 0.25)
        else:
            alpha *= 0.7
            if patience > max_patience:
                h = h_best.copy()
                noise = rng.normal(0, perturbation_scale, n)
                h += noise
                h = project_feasible(h, n)
                alpha = initial_alpha
                patience = 0
                max_patience = int(max_patience * 1.3)
                perturbation_scale *= 0.85
            elif patience > max_patience // 2:
                h = h_best.copy()
                h += rng.normal(0, perturbation_scale * 0.5, n)
                h = project_feasible(h, n)
                alpha = alpha * 0.5
                patience = 0

def fine_tune(h, n, time_limit, rng):
    """Very small step optimization for final polishing."""
    t_start = time.time()
    h = project_feasible(h.astype(np.float64).copy(), n)
    c5_best = compute_c5(h)
    h_best = h.copy()
    alpha = 0.01

    while time.time() - t_start < time_limit:
        lags = compute_all_lags(h)
        curr_max = float(lags.max())
        if curr_max < c5_best:
            c5_best = curr_max
            h_best = h.copy()

        threshold = curr_max * 0.995
        active = np.where(lags >= threshold)[0]
        grad = gradient_multi_lag(h, active)
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        new_c5 = compute_c5(h_new)
        if new_c5 < curr_max:
            h = h_new
            alpha = min(alpha * 1.02, 0.05)
        else:
            alpha *= 0.8
            if alpha < 0.001:
                h = h_best.copy()
                h += rng.normal(0, 0.005, n)
                h = project_feasible(h, n)
                alpha = 0.01

    return h_best, c5_best
