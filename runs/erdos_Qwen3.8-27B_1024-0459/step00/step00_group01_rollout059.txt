import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 objective value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h to satisfy sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    diff = target_sum - np.sum(h)
    h += diff / len(h)
    h = np.clip(h, 0, 1)
    # Iterative adjustment
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        # Add to elements that can absorb
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
        else:
            candidates = np.where(h > 0.0)[0]
        if len(candidates) == 0:
            break
        h[candidates] += diff / len(candidates)
        h = np.clip(h, 0, 1)
    return h

def logsumexp_objective(h, dx, T=50):
    """Smooth approximation of max using log-sum-exp."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Add small constant to ensure positivity
    corr = corr + 1e-6
    return np.log(np.sum(np.exp(T * corr)) / len(corr)) / T

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:top_k]

        m = j - (n - 1)

            grad[:n-mm] += (1.0 - h[mm:])
            grad[mm:] -= h[:n-mm]
