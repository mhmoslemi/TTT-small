import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return float(np.max(corr) * dx)

def upsample(h, new_n):
    """Upsample h from current resolution to new_n points using linear interpolation."""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0/old_n  # centers
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(new_x, old_x, np.append(h, h[0]))  # periodic
    # Fix sum
    target = new_n / 2
    delta = (target - np.sum(h_new)) / new_n
    h_new = np.clip(h_new + delta, 0, 1)
    # Fix sum again (clipping might have changed it)
    # Iteratively fix
    for _ in range(10):
        delta = (target - np.sum(h_new)) / new_n
        h_new = np.clip(h_new + delta, 0, 1)
        if abs(np.sum(h_new) - target) < 1e-10:
            break
    return h_new

def downsample(h, new_n):
    """Downsample h to new_n points."""
    old_n = len(h)
    # Average groups
    groups = np.array_split(np.arange(old_n), new_n)
    h_new = np.array([np.mean(h[g]) for g in groups])
    # Fix sum
    target = new_n / 2
    delta = (target - np.sum(h_new)) / new_n
    h_new = np.clip(h_new + delta, 0, 1)
    for _ in range(10):
        delta = (target - np.sum(h_new)) / new_n
        h_new = np.clip(h_new + delta, 0, 1)
    return h_new

def fix_sum(h, n):
    """Ensure sum(h) = n/2 with 0 <= h[i] <= 1."""
    target = n / 2
    for _ in range(20):
        delta = (target - np.sum(h)) / n
        h = np.clip(h + delta, 0, 1)
        if abs(np.sum(h) - target) < 1e-12:
            break
    return h

def sa_optimize(h, n, n_iterations, T_init=0.1, T_final=0.001, seed=42):
    """Simulated annealing optimization."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5
    current_h = h.copy()
    
    for it in range(n_iterations):
        # Temperature schedule
        progress = it / n_iterations
        T = T_init * (T_final / T_init) ** progress
        
        # Choose move size (decreasing over time)
        max_step = 0.1 * (1 - progress) + 0.01
        eps = rng.uniform(0.001, max_step)
        
        # Pick two indices
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        
        # Determine direction
        if current_h[i] >= current_h[j]:
            # Move from i to j
            max_eps = min(current_h[i], 1 - current_h[j])
        else:
            i, j = j, i
            max_eps = min(current_h[i], 1 - current_h[j])
        
        if max_eps < eps:
            eps = max_eps * 0.5 if max_eps > 1e-8 else 0
        if eps < 1e-10:
            continue
        
        current_h[i] -= eps
        current_h[j] += eps
        new_c5 = compute_c5(current_h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            current_h[i] += eps
            current_h[j] -= eps
    
    return best_h, best_c5

The "best" j to decrease: maximize g_j(k*) = (1-h[j+k*]) - h[j-k*], subject to h[j] > 0.

This is O(n) to compute! Then we just need to find δ_max which is O(n) (checking all k).

So each "greedy step" is O(n), and we can do millions of steps.

But there's a catch: after each step, the O(k) values change, and the bottleneck k* might change. Also, the greedy approach might get stuck in local optima.

A better approach: combine the greedy method with SA. Use the greedy direction to propose moves, but also allow random exploratory moves.

def greedy_sa(h, n, time_limit, seed=42):

    start = time.time()

    one_minus_h = 1 - current_h
    
    while time.time() - start < time_limit:
        progress = min(1.0, (time.time() - start) / time_limit)
        T = 0.05 * (0.001 / 0.05) ** progress
        
        # Compute correlation and find bottleneck
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        O = corr * (2.0 / n)  # This gives O(k) for each shift
        max_idx = np.argmax(O)
        max_val = O[max_idx]
        
        # The lag corresponding to max_idx in the full correlation
        # Full correlation of two length-n arrays has length 2n-1
        # Index max_idx corresponds to lag = max_idx - (n-1)
        k_star = max_idx - (n - 1)
        
        if rng.random() < 0.7:
            # Greedy move: reduce the bottleneck
            # g_i(k*) = (1-h[i+k*]) - h[i-k*]
            # We want to increase h where g is small, decrease where g is large
            g = np.zeros(n)
            for i in range(n):
                # (1-h[i+k*]) if i+k* is in [0,n), else 0
                ik = i + k_star
                if 0 <= ik < n:
                    g[i] += (1 - current_h[ik])
                # -h[i-k*] if i-k* is in [0,n), else 0
                ik2 = i - k_star
                if 0 <= ik2 < n:
                    g[i] -= current_h[ik2]
            
            # Find best i to increase (min g, h[i] < 1)
            candidates_inc = np.where(current_h < 1 - 1e-8)[0]
            candidates_dec = np.where(current_h > 1e-8)[0]
            
            if len(candidates_inc) == 0 or len(candidates_dec) == 0:
                continue
            
            best_i = candidates_inc[np.argmin(g[candidates_inc])]
            best_j = candidates_dec[np.argmax(g[candidates_dec])]
            
            if best_i == best_j:
                continue
            
            # Compute delta
            max_d = min(1 - current_h[best_i], current_h[best_j])
            
            # Check constraint: O(k) + delta*(g_i(k) - g_j(k)) <= max_val for all k
            # This is complex; for simplicity, use a small step
            step = min(max_d * 0.5, 0.01 * (1 - progress) + 0.001)
            
            current_h[best_i] += step
            current_h[best_j] -= step
        else:
            # Random exploratory move
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            if current_h[i] >= current_h[j]:
                step = min(current_h[i], 1 - current_h[j])
            else:
                i, j = j, i
                step = min(current_h[i], 1 - current_h[j])
            if step < 1e-10:
                continue
            step = step * rng.uniform(0.01, 0.5)
            current_h[i] -= step
            current_h[j] += step

        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # Undo the move
            # (Need to track what we did)
            current_h = np.clip(current_h, 0, 1)
            # Re-fix sum
            current_h = fix_sum(current_h, n)
    
    return best_h, best_c5

            h_opt, c5_opt = sa_search(h_hi, n_hi, t_alloc, seed=seed + 9000 + n_hi, T_init=0.015, T_min=0.0002)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_hi

    # Phase 4: Final polish
    if time_left() > 10 and best_h is not None:
        t_alloc = time_left() * 0.8
        h_opt, c5_opt = polish(best_h.copy(), best_n, t_alloc, seed=seed + 7777)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Validate and return
    if best_h is None:
        best_h = make_random_h(100, seed)
        best_c5 = compute_c5(best_h)
        best_n = 100

    best_h = fix_sum(best_h, best_n)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, best_n
