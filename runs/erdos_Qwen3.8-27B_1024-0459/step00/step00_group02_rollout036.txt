def compute_c5(h, dx):
    """Compute C5 = max_k dx * sum_i h[i]*(1-h[i+lag]) for all lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    lo = -float(np.max(h)) - 1.0
    hi = 1.0 - float(np.min(h)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h + lam, 0.0, 1.0)

def subgrad_for_lag(h, lag, dx):
    """Compute dx * d/dh [sum_i h[i]*(1-h[i+lag])] for a given lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    else:
        k = -lag
        if k < n:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
    return g * dx

def smoothed_grad(h, dx, alpha):
    """Compute smoothed gradient of max_k g_k(h) using softmax weighting."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx  # (2n-1,) array of g_k values
    
    c_shifted = c - float(np.max(c))
    exp_c = np.exp(alpha * c_shifted)
    p = exp_c / float(np.sum(exp_c))
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = p[idx]
        if wi < 1e-14:
            continue
        lag = idx - (n - 1)
        grad += wi * subgrad_for_lag(h, lag, dx)
    
    return alpha * grad

            x_new = np.linspace(0, 2, n2, endpoint=False)
            h2 = np.interp(x_new, x_old, best_h)
        else:
            h2 = np.ones(n2) * 0.5
        h2 = project_feasible(h2, n2 / 2.0)
        t2 = min(0.28 * budget_s, budget_s - (time.time() - t0))
        if t2 > 10:
            h2, c5_2 = run_optimization(h2, dx2, n2 / 2.0, t2, rng, alpha_init=35.0)
            h2, c5_2 = sharpen_and_check(h2, n2 / 2.0, dx2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2

    # Phase 3: Optimize at n=180
    if budget_s - (time.time() - t0) > 30:
        n3 = 180
        dx3 = 2.0 / n3
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n3, endpoint=False)
            h3 = np.interp(x_new, x_old, best_h)
        else:
            h3 = np.ones(n3) * 0.5
        h3 = project_feasible(h3, n3 / 2.0)
        t3 = min(0.25 * budget_s, budget_s - (time.time() - t0))
        if t3 > 10:
            h3, c5_3 = run_optimization(h3, dx3, n3 / 2.0, t3, rng, alpha_init=50.0)
            h3, c5_3 = sharpen_and_check(h3, n3 / 2.0, dx3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3

    # Phase 4: Random restarts at best resolution
    if budget_s - (time.time() - t0) > 20 and best_h is not None:
        n_r = len(best_h)
        dx_r = 2.0 / n_r
        target_r = n_r / 2.0
        remaining = budget_s - (time.time() - t0)
        n_restarts = min(6, int(remaining / 20))
        per_restart = min(remaining * 0.75 / max(n_restarts, 1), 80.0)

        for r in range(n_restarts):
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break

            init_type = r % 4
            if init_type == 0:
                h_r = rng.random(n_r) * 0.7 + 0.15
            elif init_type == 1:
                h_r = np.zeros(n_r)
                n_blocks = rng.randint(2, 7)
                for b in range(n_blocks):
                    start = rng.randint(0, max(1, n_r - 5))
                    length = rng.randint(5, max(6, n_r // 3))
                    h_r[start:min(start + length, n_r)] = 1.0
            elif init_type == 2:
                freq = rng.uniform(1.0, 5.0)
                phase = rng.uniform(0, 2 * np.pi)
                h_r = 0.5 + 0.35 * np.sin(2 * np.pi * freq * np.linspace(0, 2, n_r, endpoint=False) / 2.0 + phase)
            else:
                h_r = best_h + rng.normal(0, 0.1, n_r)

            h_r = project_feasible(h_r, target_r)
            t_r = min(per_restart, remaining - 10)
            if t_r < 5:
                break

            h_r, c5_r = run_optimization(h_r, dx_r, target_r, t_r, rng, alpha_init=35.0)
            h_r, c5_r = sharpen_and_check(h_r, target_r, dx_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r

    # Phase 5: Final refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 10 and best_h is not None:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        t_f = max(5, remaining - 5)
        h_f, c5_f = run_optimization(best_h, dx_f, n_f / 2.0, t_f, rng, alpha_init=60.0)
        h_f, c5_f = sharpen_and_check(h_f, n_f / 2.0, dx_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h, dx_final)

    return best_h, final_c5, n_final
