    """Compute C5 value for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n):
    """Repair h to satisfy constraints: 0 <= h <= 1, sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    # Adjust sum to n/2
    target = n / 2.0
    current_sum = np.sum(h)
    if current_sum == 0:
        h = np.full(n, 0.5)
        return h
    # Scale
    h = h * (target / current_sum)
    h = np.clip(h, 0, 1)
    # Fine-tune: distribute the remaining difference
    diff = target - np.sum(h)
    if diff > 0:
        # Need to add diff total
        # Add to elements that are < 1
        room = 1.0 - h
        # Distribute proportionally
        total_room = np.sum(room)
        if total_room > 0:
            h += room * (diff / total_room)
    elif diff < 0:
        # Need to subtract |diff| total
        room = h
        total_room = np.sum(room)
        if total_room > 0:
            h -= room * (-diff / total_room)

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    """Ensure 0 <= h[i] <= 1 and sum(h) = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if s == 0:
        return np.full(n, 0.5)
    # First pass: scale
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Second pass: fix remaining discrepancy
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Add to elements with room
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 0:
                h += room * (diff / total_room)
        else:
            # Remove from elements with surplus
            room = h
            total_room = np.sum(room)
            if total_room > 0:
                h -= room * (-diff / total_room)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix: adjust one element
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def gradient_single_shift(h, k):
    """Gradient of correlation at shift k w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    elif k < 0:
        kp = -k
        grad[:n-kp] += (1.0 - h[kp:])
        grad[kp:] -= h[:n-kp]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def upsample_linear(h, n_new):
    """Upsample h to n_new points using linear interpolation."""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return repair_h(h_new, n_new)

    sa_cool = 0.999
    
    n_iter = 0

        n_iter += 1
        progress = min(1.0, n_iter / 50000)
        
        # Compute correlations

        corr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        max_c = np.max(corr)
        
        # Find top-K active shifts
        K = min(10, len(corr))
        top_K = np.argsort(corr)[-K:]
        
        # Compute gradient (average of top-K)
        grad = np.zeros(n)
        for m in top_K:
            k = int(m - (n - 1))
            grad += gradient_single_shift(best_h, k)
        grad /= K
        grad *= dx
        
        # Learning rate
        lr = 0.05 * (1.0 - progress) + 0.005
        
        # Gradient step
        new_h = best_h - lr * grad
        new_h = repair_h(new_h, n)
        new_c5 = evaluate_c5(new_h)

            best_c5 = new_c5
            best_h = new_h
        else:

                if i != j:

                    delta = rng.uniform(-0.1, 0.1)
                    perturb_h[i] += delta
                    perturb_h[j] -= delta

                    perturb_c5 = evaluate_c5(perturb_h)
                    if perturb_c5 < best_c5:

                        best_h = perturb_h
                    elif rng.random() < np.exp(max(0, (best_c5 - perturb_c5) / sa_temp)):

            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)

def gradient_single_shift(h, k):
    n = len(h)

        grad[:n-kp] += (1.0 - h[kp:])
        grad[kp:] -= h[:n-kp]

        if n > 4000:
            break
        alloc2 = min(remaining() * 0.22, 180)
        if alloc2 < 15:
            break
        h_up = upsample_linear(h0, n)
        h_opt, c5_opt = optimize_inner(h_up, n, alloc2, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 3: Random binary restarts at promising resolutions
    if remaining() > 60:
        n_try = len(best_h)
        if n_try % 2 != 0:
            n_try = n_try + 1
        n_restarts = 3
        alloc3_each = min(remaining() * 0.1, 80)
        for _ in range(n_restarts):
            if remaining() < 30:
                break
            h_rand = random_binary_init(n_try, rng)
            h_opt, c5_opt = optimize_inner(h_rand, n_try, alloc3_each, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Phase 4: Final polish at best resolution
    if remaining() > 30:
        n_final = len(best_h)
        alloc4 = remaining() * 0.85
        h_opt, c5_opt = optimize_inner(best_h.copy(), n_final, alloc4, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Ensure validity
    best_h = repair_h(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    
    return best_h.tolist(), float(best_c5), len(best_h)
