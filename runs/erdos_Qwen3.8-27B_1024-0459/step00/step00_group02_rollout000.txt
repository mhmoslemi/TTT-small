def smooth_grad(h, beta):
    n=h.size
    dx=2.0/n
    y=1.0-h
    corr=np.correlate(h,y,mode='full')
    c=corr*dx
    m=c.max(initial=0.0) # np.max
    # weights stable
    z=beta*(c-m)
    e=np.exp(z)
    denom=e.sum()
    w=e/denom
    G=np.zeros(n)
    # skip weight tiny
    # Loop enumerate(w)
    for k, wk in enumerate(w):
       if wk < 1e-14: continue
       d=k-(n-1)
       if d >= 0:
          # g = 1 on [0:n-d] - h[d:n] - h[:n-d] shifted
          if n-d > 0:
             length = n-d
             G[:length] += wk
             G[:length] -= wk * h[d:d+length]
             G[d:d+length] -= wk * h[:length]
       else:
          a=-d
          length=n-a
          G[a:] += wk
          G[a:] -= wk * h[:length]
          G[:length] -= wk * h[a:a+length]
    return dx*G
```

import time

def c5_value(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 0.0
    if not np.all(np.isfinite(h)):
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return 0.0
    return float(np.max(corr) * dx)


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    n = int(x.size)
    if n == 0:
        return x.copy()
    z = np.clip(np.where(np.isfinite(x), x, 0.5), 0.0, 1.0).astype(float, copy=True)
    target = float(target)
    s = float(np.sum(z))
    if abs(s - target) <= 1e-12:
        return z

    if s > target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0

    for _ in range(50):
        mid = 0.5 * (lo + hi)
        sv = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if sv > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    y = np.clip(z - lam, 0.0, 1.0)

    res = target - float(np.sum(y))
    if res > 1e-13:
        order = np.argsort(y)
        left = res
        for idx in order:
            if left <= 1e-14:
                break
            add = min(left, 1.0 - float(y[idx]))
            if add > 0.0:
                y[idx] += add
                left -= add
    elif res < -1e-13:
        order = np.argsort(y)[::-1]
        left = -res
        for idx in order:
            if left <= 1e-14:
                break
            sub = min(left, float(y[idx]))
            if sub > 0.0:
                y[idx] -= sub
                left -= sub

    return y


def grad_lag(h, d):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g
    d = int(d)
    if d < -(n - 1) or d > (n - 1):
        return g

    if d >= 0:
        length = n - d
        if length <= 0:

        g[:length] += 1.0
        g[:length] -= h[d:d + length]
        g[d:d + length] -= h[:length]
    else:
        a = -d
        length = n - a
        if length <= 0:

        g[a:] += 1.0

def active_subgrad(h, corr, top_k=8):
    h = np.asarray(h, dtype=float)
    corr = np.asarray(corr, dtype=float)
    n = int(h.size)
    m = int(corr.size)
    if n == 0 or m == 0:
        return np.zeros(n, dtype=float)

    c = corr * (2.0 / float(n))
    if m <= top_k:
        idx = np.arange(m)
        vals = c[idx]
        w = np.ones(m, dtype=float)
    else:
        kk = int(max(1, top_k))
        idx = np.argpartition(c, -kk)[-kk:]
        vals = c[idx]
        vmax = float(np.max(vals))
        z = 20.0 * (vals - vmax)

        s = float(np.sum(w))

def smooth_c5(h, beta):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    m = float(np.max(c))
    beta = float(beta)
    if beta <= 0.0:
        return m
    z = beta * (c - m)
    s = float(np.sum(np.exp(z)))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return m + float(np.log(s)) / beta


def smooth_grad_c5(h, beta):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=float)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    m = float(np.max(c))
    beta = float(beta)

        return dx * active_subgrad(h, corr, top_k=12)
    z = beta * (c - m)
    w = np.exp(z)

        w = np.ones(c.size, dtype=float) / float(c.size)
    else:
        w = w / s

def upsample(h, n):
    h = np.asarray(h, dtype=float)
    m = int(h.size)
    n = int(n)
    if m == n:
        return h.copy()
    if m <= 0:
        return np.zeros(n, dtype=float)
    if n % m == 0:
        return np.repeat(h, n // m).astype(float, copy=True)

    old = np.linspace(0.0, 1.0, m, endpoint=False) + 0.5 / float(m)
    new = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / float(n)
    y = np.interp(new, old, h)
    return project_box_sum(y, 0.5 * float(n))


def random_binary(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    if n <= 0:
        return h
    k = n // 2
    if k > 0:

        h[idx] = 1.0
    if n % 2 == 1:
        zeros = np.where(h < 0.5)[0]
        if zeros.size > 0:
            j = int(zeros[int(rng.integers(zeros.size))])
            h[j] = 0.5
    return h


def random_smooth(n, rng):

    h = np.clip(rng.normal(0.5, 0.18, size=n), 0.0, 1.0)

def wave_start(n, rng):
    n = int(n)
    if n <= 1:
        return random_binary(n, rng)
    if n >= 10:
        f = int(rng.integers(2, 10))
    else:
        f = int(rng.integers(1, n + 1))
    phase = float(rng.uniform(0.0, 2.0 * np.pi))
    x = np.arange(n, dtype=float)
    s = np.sin(2.0 * np.pi * float(f) * x / float(n) + phase)
    h = (s >= 0.0).astype(float)
    return project_box_sum(h, 0.5 * float(n))


def build_lag_matrix(y):
    y = np.asarray(y, dtype=float)
    n = int(y.size)
    if n == 0:
        return np.zeros((0, 0), dtype=float)
    A = np.zeros((2 * n - 1, n), dtype=float)
    for m in range(2 * n - 1):
        d = (n - 1) - m
        if d >= 0:
            A[m, :n - d] = y[d:]
        else:
            a = -d
            A[m, a:] = y[:n - a]
    return A


def lp_best_response(y, target):
    y = np.asarray(y, dtype=float)
    n = int(y.size)
    if n == 0:
        return None
    target = float(target)
    A = build_lag_matrix(y)
    m = int(A.shape[0])

    c = np.zeros(n + 1, dtype=float)
    c[-1] = 1.0

    A_ub = np.zeros((m, n + 1), dtype=float)
    A_ub[:, :n] = A
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(m, dtype=float)

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([target], dtype=float)

    bounds = [(0.0, 1.0)] * n + [(0.0, float(n))]

    res = None
    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        try:
            res = optimize.linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
            )
        except Exception:
            res = None

    if res is not None and hasattr(res, "x") and res.x is not None:
        h = np.asarray(res.x[:n], dtype=float)
        if np.all(np.isfinite(h)):
            return project_box_sum(h, target)
    return None


def polish_lp(h0, max_iters=15, gamma=1.0, deadline=None):
    h0 = np.asarray(h0, dtype=float)
    n = int(h0.size)
    if n == 0:
        return h0.copy(), 0.0
    target = 0.5 * float(n)
    h = project_box_sum(h0, target)
    best_h = h.copy()
    best_c = c5_value(h)
    no_imp = 0

    for _ in range(int(max_iters)):
        if deadline is not None and time.time() > max(start + 1e-9, deadline - 1.0):
            break
        y = 1.0 - h
        hn = lp_best_response(y, target)
        if hn is None:
            break
        if gamma < 1.0:
            hn = project_box_sum((1.0 - float(gamma)) * h + float(gamma) * hn, target)
        c = c5_value(hn)
        if c < best_c - 1e-13:
            best_c = c
            best_h = hn.copy()
            no_imp = 0
        else:
            no_imp += 1
        h = hn.copy()
        if no_imp > 12:
            break

    best_c = min(best_c, c5_value(best_h))
    return best_h, best_c


def polish_slsqp(h0, rng, betas=(5.0, 20.0, 60.0), deadline=None,
                 attempts=2, maxit_scale=1.0):
    global _SL_TARGET, _SL_BETA
    h0 = np.asarray(h0, dtype=float)
    n = int(h0.size)
    if n == 0:
        return h0.copy(), 0.0
    target = 0.5 * float(n)
    h = project_box_sum(h0, target)
    best_h = h.copy()
    best_c = c5_value(h)
    _SL_TARGET = target

    bounds = [(0.0, 1.0)] * n
    cons = [
        {
            "type": "eq",
            "fun": _sl_sum,
            "jac": _sl_sum_jac,
        }
    ]

    base = int(9000.0 * float(maxit_scale) / float(n))
    base = max(18, min(220, base))

    current = h.copy()
    for beta in betas:
        if deadline is not None and time.time() > deadline - 0.75:
            break
        _SL_BETA = float(beta)
        for attempt in range(int(max(1, attempts))):
            if deadline is not None and time.time() > deadline - 0.5:
                break
            if attempt == 0:
                x0 = current.copy()
            else:
                noise = 0.04 * rng.standard_normal(n)
                x0 = project_box_sum(best_h + noise, target)

            res = None
            try:
                res = optimize.minimize(
                    _sl_obj,
                    x0,
                    method="SLSQP",
                    jac=_sl_grad,
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": base, "ftol": 1e-11},
                )
            except Exception:
                res = None

            if res is not None and hasattr(res, "x") and res.x is not None:
                cand = np.asarray(res.x, dtype=float)
            else:
                cand = x0

            cand = project_box_sum(cand, target)
            c = c5_value(cand)
            current = cand.copy()

            if c < best_c - 1e-13:
                best_c = c
                best_h = cand.copy()

    best_c = min(best_c, c5_value(best_h))
    return best_h, best_c


def pair_descent(h0, rng, max_evals=2000, deadline=None, top_k=6):
    h0 = np.asarray(h0, dtype=float)
    n = int(h0.size)
    if n <= 1:
        return h0.copy(), c5_value(h0)

    target = 0.5 * float(n)
    h = project_box_sum(h0, target)
    best_h = h.copy()
    best_c = c5_value(h)
    evals = 0
    max_evals = int(max(50, min(max_evals, 20000000.0 / float(n * n))))

    while evals < max_evals and (deadline is None or time.time() < deadline):
        y = 1.0 - h
        corr = np.correlate(h, y, mode="full")
        c = float(np.max(corr) * (2.0 / float(n)))
        if c < best_c:
            best_c = c
            best_h = h.copy()

        g = active_subgrad(h, corr, top_k=top_k)
        if not np.all(np.isfinite(g)) or float(np.max(np.abs(g))) < 1e-14:
            break

        pos = np.argsort(g)[-4:][::-1]
        neg = np.argsort(g)[:4]
        accepted = False
        cand_count = 0

        for aa in neg:
            if accepted:
                break
            for bb in pos:
                if accepted or cand_count >= 14:
                    break
                if aa == bb:
                    continue
                dmax = min(float(h[bb]), float(1.0 - h[aa]))
                if dmax <= 1e-12:
                    continue
                for frac in (0.5, 0.25, 0.1):
                    if evals >= max_evals or cand_count >= 14:
                        break
                    delta = frac * dmax
                    h2 = h.copy()
                    h2[aa] += delta
                    h2[bb] -= delta
                    if float(np.min(h2)) < -1e-12 or float(np.max(h2)) > 1.0 + 1e-12:
                        continue
                    if abs(float(np.sum(h2)) - target) > 1e-9:
                        h2 = project_box_sum(h2, target)
                    c2 = c5_value(h2)
                    evals += 1
                    cand_count += 1
                    if c2 < c - 1e-13:
                        h = h2
                        accepted = True
                        break
                if accepted:
                    break
            if accepted:
                break

        if not accepted:
            aa = -1
            bb = -1
            for _ in range(6):
                aa = int(rng.integers(n))
                bb = int(rng.integers(n))
                if aa != bb:
                    break
            if aa != bb:
                if float(g[aa]) < float(g[bb]):
                    inc, dec = aa, bb
                else:
                    inc, dec = bb, aa
                dmax = min(float(h[dec]), float(1.0 - h[inc]))
                if dmax > 1e-12:
                    delta = 0.05 * dmax
                    h2 = h.copy()
                    h2[inc] += delta
                    h2[dec] -= delta
                    if abs(float(np.sum(h2)) - target) > 1e-9:
                        h2 = project_box_sum(h2, target)
                    c2 = c5_value(h2)
                    evals += 1
                    if c2 < c - 1e-13:
                        h = h2

    best_c = min(best_c, c5_value(best_h))
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _SL_TARGET, _SL_BETA

    start = time.time()
    total = max(1.0, float(budget_s))
    deadline = start + total - 8.0
    if deadline <= start + 0.5:
        deadline = start + max(0.5, total)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")

    parent_h = None
    try:
        raw = initial_h_values
        arr = np.asarray(raw, dtype=float).reshape(-1)
        if arr.size > 0 and np.all(np.isfinite(arr)):
            parent_h = arr.copy()
    except Exception:
        parent_h = None

    if parent_h is None:
        parent_h = random_binary(46, rng)

    n0 = int(parent_h.size)
    if n0 <= 0:
        parent_h = random_binary(46, rng)
        n0 = int(parent_h.size)

    parent_h = project_box_sum(parent_h, 0.5 * float(n0))
    c0 = c5_value(parent_h)
    if np.isfinite(c0):
        best_h = parent_h.copy()
        best_c = c0

    max_n = 360
    ns = []
    for n in (n0, 2 * n0, 4 * n0):
        if 2 <= n <= max_n and n not in ns:
            ns.append(n)
    ns.sort()
    if len(ns) == 0:
        ns = [46]

    parent_deadline = start + max(1.0, 0.55 * (deadline - start))

    for n in ns:
        if time.time() > parent_deadline - 5.0:
            break

        h_start = upsample(parent_h, n)
        c_start = c5_value(h_start)
        if c_start < best_c:
            best_h = h_start.copy()
            best_c = c_start

        # A short discrete/coordinate polish before LP/SLSQP.
        h_pr, c_pr = pair_descent(h_start, rng, max_evals=400, deadline=parent_deadline)
        if c_pr < best_c:
            best_h = h_pr.copy()
            best_c = c_pr
        if np.isfinite(c_pr) and c_pr <= c_start + 1e-12:
            h_start = h_pr.copy()

        # LP best-response sequence.
        lp_h, lp_c = polish_lp(h_start, max_iters=12, gamma=1.0, deadline=parent_deadline)
        if lp_h is not None and lp_c < best_c:
            best_h = lp_h.copy()
            best_c = lp_c

        starts = [h_start]
        if lp_h is not None:
            if int(lp_h.size) == n and np.max(np.abs(lp_h - h_start)) > 1e-8:
                starts.append(lp_h)

        for hs in starts:
            if time.time() > parent_deadline - 10.0:
                break
            hh, cc = polish_slsqp(
                hs,
                rng,
                betas=(5.0, 20.0, 60.0),
                deadline=parent_deadline,
                attempts=2,
                maxit_scale=1.0,
            )
            if cc < best_c:
                best_h = hh.copy()
                best_c = cc

            if time.time() < parent_deadline - 5.0:
                pp, pc = pair_descent(hh, rng, max_evals=2500, deadline=parent_deadline)
                if pc < best_c:
                    best_h = pp.copy()
                    best_c = pc

        if time.time() < parent_deadline - 12.0:
            lp2_h, lp2_c = polish_lp(h_start, max_iters=8, gamma=0.75, deadline=parent_deadline)
            if lp2_h is not None and lp2_c < best_c:
                best_h = lp2_h.copy()
                best_c = lp2_c

    # Random restarts and local polishing, if time remains.
    random_deadline = deadline
    random_ns = []
    for n in ns:
        if 2 <= n <= max_n and n not in random_ns:
            random_ns.append(n)
    if len(random_ns) == 0:
        random_ns = [46]

    restart = 0
    while time.time() < random_deadline - 15.0:
        n = random_ns[restart % len(random_ns)]
        kind = restart % 3
        if kind == 0:
            hr = random_binary(n, rng)
        elif kind == 1:
            hr = wave_start(n, rng)
        else:
            hr = random_smooth(n, rng)

        hh, cc = polish_slsqp(
            hr,
            rng,
            betas=(10.0, 40.0),
            deadline=random_deadline,
            attempts=1,
            maxit_scale=0.45,
        )
        if cc < best_c:
            best_h = hh.copy()
            best_c = cc

        if time.time() < random_deadline - 8.0:
            pp, pc = pair_descent(hh, rng, max_evals=800, deadline=random_deadline)
            if pc < best_c:
                best_h = pp.copy()
                best_c = pc

        restart += 1
        if restart >= 80:
            break

    if best_h is None:
        best_h = parent_h.copy()
        best_c = c0

    best_h = project_box_sum(best_h, 0.5 * float(best_h.size))
    final_c = c5_value(best_h)
    if not np.isfinite(final_c):
        final_c = float("inf")

    if not np.isfinite(final_c) and best_c < final_c:
        final_c = best_c

    if time.time() > deadline + 5.0:
        pass

    return np.asarray(best_h, dtype=float), float(final_c), int(best_h.size)
