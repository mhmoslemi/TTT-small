def project_feasible(v, S):
    """Project v onto {h: 0<=h<=1, sum(h)=S} via bisection."""
    lo = -5.0

        s = np.sum(np.clip(v - mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)

    dx = 2.0 / n
