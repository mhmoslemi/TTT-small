    """Gradient of S(j) = sum_{valid i} h[i]*(1-h[i+j]) w.r.t. h"""

import numpy as np
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def grad_S_j(h, j, n):
    """Gradient of S(j) = sum_{valid i} h[i]*(1-h[i+j]) w.r.t. h"""
    g = np.zeros(n)
    m = np.arange(n)
    idx_a = m + j
    mask_a = (idx_a >= 0) & (idx_a < n)
    g[mask_a] += (1.0 - h[idx_a[mask_a]])
    idx_b = m - j
    mask_b = (idx_b >= 0) & (idx_b < n)
    g[mask_b] -= h[idx_b[mask_b]]
    return g


def optimize_grid(h, dx, target, n, rng, t0, budget_s, max_iter=50000):
    """Subgradient optimization on a fixed grid."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 0.08
    stall = 0
    iter_ct = 0

    for _ in range(max_iter):
        if budget_s - (time.time() - t0) < 5.0:
            break
        iter_ct += 1

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = float(np.max(corr)) * dx
        k_star = int(np.argmax(corr))
        j_star = k_star - (n - 1)

        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            scale = rng.choice([0.03, 0.05, 0.1, 0.15, 0.2])
            h = best_h + rng.normal(0, scale, n)
            h = project_feasible(h, target)
            stall = 0
            lr = 0.08
            continue

        g = grad_S_j(h, j_star, n)
        h_trial = h - lr * g
        h_trial = project_feasible(h_trial, target)
        c5_trial = compute_c5(h_trial, dx)

        if c5_trial < c5_cur:

                    h_alt = h - lr * g_alt
                    h_alt = project_feasible(h_alt, target)
                    c5_alt = compute_c5(h_alt, dx)
                    if c5_alt < best_c5:
                        h = h_alt
                        best_h = h_alt.copy()
                        best_c5 = c5_alt

                h = h_flip
                best_h = h_flip.copy()
                best_c5 = c5_flip
                stall = 0

    return best_h, best_c5
