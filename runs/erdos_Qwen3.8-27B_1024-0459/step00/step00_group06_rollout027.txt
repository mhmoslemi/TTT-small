    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _soft_max_obj(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = np.clip(alpha * (vals - m), -300, 300)
    return float(m + (1.0 / alpha) * np.log(np.sum(np.exp(shifted))))

def _eq_con(x, n):
    return np.sum(x) - n / 2.0

def _fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # Second pass for clipped elements
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-12:
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.sum(free) > 0:
