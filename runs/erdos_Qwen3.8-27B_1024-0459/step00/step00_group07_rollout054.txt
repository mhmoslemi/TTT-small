            h[j] = 0

import time

def _compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _compute_c5(h):
    return _compute_lags(h).max()

def _upsample_binary(h, new_n):
    """Upsample binary h to new_n using nearest neighbor, then fix count."""
    old_n = len(h)
    h_new = np.zeros(new_n)
    for i in range(new_n):
        old_i = min(int(i * old_n / new_n), old_n - 1)
        h_new[i] = h[old_i]
    # Fix count to new_n // 2
    target = new_n // 2
    current = int(h_new.sum())
    if current > target:
        ones = np.where(h_new == 1)[0]
        rng = np.random.RandomState(42)
        to_remove = rng.choice(len(ones), current - target, replace=False)
        h_new[ones[to_remove]] = 0
    elif current < target:
        zeros = np.where(h_new == 0)[0]
        rng = np.random.RandomState(42)
        to_add = rng.choice(len(zeros), target - current, replace=False)
        h_new[zeros[to_add]] = 1
    return h_new

def _local_search(h, seed=42, max_iter=200000, patience=5000, time_limit=60):
    """Local search by random bit-pair swaps for binary h."""
    rng = np.random.RandomState(seed)
    n = len(h)
    h = h.copy()
    c5 = _compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    start = time.time()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0
        h[j] = 1
        new_c5 = _compute_c5(h)
        if new_c5 < c5 - 1e-12:
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            no_improve = 0
        else:
            h[i] = 1
            h[j] = 0
            no_improve += 1
            if no_improve >= patience:
                break
    
    return best_h, best_c5

def _continuous_opt(h, seed=42, time_limit=30):
    """Gradient-based continuous optimization using log-sum-exp."""
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    
    best_c5 = _compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    # Log-sum-exp with increasing temperature
    T_values = [10, 20, 50, 100, 200, 500]
    lr = 0.1
    
    for T in T_values:
        if time.time() - start > time_limit:
            break
        for step in range(5000):
            if time.time() - start > time_limit:
                break
            
            lags = _compute_lags(h)
            # Log-sum-exp
            max_lag = lags.max()
            exp_lags = np.exp(T * (lags - max_lag))
            weights = exp_lags / exp_lags.sum()
            smooth_max = max_lag + np.log(exp_lags.sum()) / T
            
            # Gradient
            grad = np.zeros(n)
            for k in range(len(lags)):
                if weights[k] < 1e-10:
                    continue
                m = k - (n - 1)
                # ∂f_k/∂h[j] = dx * [(1-h[j+m]) * 1{j+m valid} - h[j-m] * 1{j-m valid}]
                if m >= 0:
                    # First term: j from 0 to n-1-m
                    if n - 1 - m >= 0:
                        j_idx = np.arange(0, n - m)
                        grad[j_idx] += weights[k] * dx * (1.0 - h[j_idx + m])
                    # Second term: j from m to n-1
                    if m <= n - 1:
                        j_idx = np.arange(m, n)
                        grad[j_idx] -= weights[k] * dx * h[j_idx - m]
                else:
                    # m < 0, let p = -m
                    p = -m
                    # First term: j from p to n-1
                    j_idx = np.arange(p, n)
                    grad[j_idx] += weights[k] * dx * (1.0 - h[j_idx + m])
                    # Second term: j from 0 to n-1-p
                    j_idx = np.arange(0, n - p)
                    grad[j_idx] -= weights[k] * dx * h[j_idx - m]
            
            # Gradient descent step
            h = h - lr * grad
            # Project to [0,1]
            h = np.clip(h, 0, 1)
            # Fix sum
            target = n / 2.0
            diff = target - h.sum()
            if abs(diff) > 1e-6:
                # Simple correction
                if diff > 0:
                    mask = h < 1.0
                    if mask.sum() > 0:
                        h[mask] += diff / mask.sum()
                        h = np.clip(h, 0, 1)
                else:
                    mask = h > 0.0
                    if mask.sum() > 0:
                        h[mask] += diff / mask.sum()
                        h = np.clip(h, 0, 1)
            
            c5 = _compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Adaptive learning rate
            if step % 500 == 0:
                lr *= 0.9
    
    return best_h, best_c5

def _corr_c5(h):

def _continuous_refine(h, seed, time_limit):
    """Gradient-based continuous refinement using log-sum-exp."""
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = _corr_c5(h)
    best_h = h.copy()
    t0 = time.time()
    lr = 0.05
    for T in [5, 10, 20, 50, 100]:
        if time.time() - t0 > time_limit:
            break
        for step in range(3000):
            if time.time() - t0 > time_limit:
                break
            lags = np.correlate(h, 1.0 - h, mode='full') * dx
            max_v = lags.max()
            shifted = T * (lags - max_v)
            exp_v = np.exp(shifted)
            w = exp_v / exp_v.sum()
            grad = np.zeros(n)
            active = np.where(w > 1e-8)[0]
            for k in active:
                m = k - (n - 1)
                if m >= 0:
                    idx1 = np.arange(0, n - m)
                    idx2 = np.arange(m, n)
                    grad[idx1] += w[k] * dx * (1.0 - h[idx1 + m])
                    grad[idx2] -= w[k] * dx * h[idx2 - m]
                else:
                    p = -m
                    idx1 = np.arange(p, n)
                    idx2 = np.arange(0, n - p)
                    grad[idx1] += w[k] * dx * (1.0 - h[idx1 + m])
                    grad[idx2] -= w[k] * dx * h[idx2 - m]
            h = h - lr * grad
            h = np.clip(h, 0.0, 1.0)
            target = n / 2.0
            diff = target - h.sum()
            if abs(diff) > 1e-8:
                if diff > 0:
                    mask = h < 1.0 - 1e-6
                    if mask.sum() > 0:
                        h[mask] += diff / mask.sum()
                else:
                    mask = h > 1e-6
                    if mask.sum() > 0:
                        h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
            c5 = _corr_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if step % 500 == 499:
                lr *= 0.85
