def gradient_c5(h, n, active_lags=None):

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    c5 = np.max(c)

            if 0 <= idx < len(c):

    hi1 = min(n, n + d)

    lo2 = max(0, -d)
    hi2 = min(n, n - d)
    if lo2 < hi2:

            grad += p[idx] * _grad_c_d(h, n, d)
