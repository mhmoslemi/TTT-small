from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def project_h(h, n):
    h = h.astype(float).copy()
    target = n / 2.0
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.sum(free)
        if nf == 0:
            break
        step = diff / nf
        if step > 0.5:
            step = 0.5
        elif step < -0.5:
            step = -0.5
        h[free] += step
    h = np.clip(h, 0.0, 1.0)
    return h

def make_obj(n, dx, beta):
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        return smooth_max(c, beta)
    return obj

    def cons_fun(h):
        return np.sum(h) - target
    def cons_jac(h):
        return np.ones(n)
