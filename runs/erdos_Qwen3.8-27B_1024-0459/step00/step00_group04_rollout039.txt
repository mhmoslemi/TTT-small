    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def make_binary(h, n):
    """Convert h to a binary vector with exactly n/2 ones."""
    # Threshold and adjust
    h_bin = (h >= 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    if n_ones > target_ones:
        # Remove some ones (those with lowest h values)
        one_indices = np.where(h_bin == 1)[0]
        # Sort by h value (ascending) and set the smallest to 0
        order = np.argsort(h[one_indices])
        for idx in order[:n_ones - target_ones]:
            h_bin[one_indices[idx]] = 0
    elif n_ones < target_ones:
        # Add some ones (those with highest h values among zeros)
        zero_indices = np.where(h_bin == 0)[0]
        order = np.argsort(h[zero_indices])[::-1]
        for idx in order[:target_ones - n_ones]:
            h_bin[zero_indices[idx]] = 1
    return h_bin

def binary_local_search(h, max_iters=10000):
    """Local search on binary vectors using swaps."""
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h)
    
    for iteration in range(max_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_improvement = 0
        best_swap = None
        
        # Try all swaps
        for i in ones:
            for j in zeros:
                h_swap = h.copy()
                h_swap[i] = 0
                h_swap[j] = 1
                c5 = compute_c5(h_swap)
                improvement = current_c5 - c5
                if improvement > best_improvement:

                    best_swap = (i, j)

            break

        h[best_swap[1]] = 1

import cvxpy as cp
import numpy as np

def solve_milp_erdos(n, initial_h=None, budget_s=100):
    """Solve the Erdős problem as a MILP."""
    dx = 2.0 / n
    N = n  # number of points
    
    # Variables
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # We need auxiliary variables for products h[i]*h[j]
    # The products that appear are h[a]*h[b] where a-b = lag
    # For all lags from -(N-1) to N-1, the pairs are (n+lag, n) for valid n
    # So the set of pairs is all (i, j) with i-j in [-(N-1), N-1], which is all pairs
    # But h[i]*h[j] = h[j]*h[i], so we only need i < j (and h[i] for i=j)
    
    # Let's use a dictionary to store the auxiliary variables
    z = {}
    for i in range(N):
        for j in range(i):  # j < i
            z[(i, j)] = cp.Variable()
    
    constraints = [cp.sum(h) == N // 2]
    
    # Constraints for auxiliary variables
    for (i, j), zij in z.items():
        constraints.append(zij <= h[i])
        constraints.append(zij <= h[j])
        constraints.append(zij >= h[i] + h[j] - 1)
        constraints.append(zij >= 0)
    
    # Correlation constraints
    # output[k] = sum_n h[n + k - (N-1)] * (1 - h[n]) for valid n
    # = sum_n h[n + k - (N-1)] - sum_n h[n + k - (N-1)] * h[n]
    
    for k in range(2 * N - 1):
        lag = k - (N - 1)
        # Valid n: 0 ≤ n ≤ N-1 and 0 ≤ n + lag ≤ N-1
        n_min = max(0, -lag)
        n_max = min(N - 1, N - 1 - lag)
        
        if n_min > n_max:
            continue
        
        # Linear part: sum_n h[n + lag]
        linear = 0
        for n in range(n_min, n_max + 1):
            linear += h[n + lag]
        
        # Product part: sum_n h[n + lag] * h[n]
        product = 0
        for n in range(n_min, n_max + 1):
            i = n + lag
            j = n
            if i == j:
                product += h[i]  # h[i]*h[i] = h[i] for binary

                product += z[(i, j)]
            else:
                product += z[(j, i)]
        
        constraints.append(dx * (linear - product) <= t)
    
    # Objective
    objective = cp.Minimize(t)
    
    prob = cp.Problem(objective, constraints)
    
    # Solve

    n = len(h)

    dx = 2.0 / n
    N = n
    
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    # Auxiliary variables for products h[i]*h[j] where i > j

        for j in range(i):
            z_vars[(i, j)] = cp.Variable()

        constraints.append(z <= h[i])
        constraints.append(z <= h[j])
        constraints.append(z >= h[i] + h[j] - 1)

    for k in range(2 * N - 1):

            j = n

            elif i > j:

            else:

        zeros = np.where(h == 0)[0]
        
        best_improvement = 1e-10  # Must be strictly improving
        best_i = -1
        best_j = -1
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0

                    best_i = i

                h[j] = 0.0

        h[best_j] = 1.0
        c5 -= best_improvement

def simulated_annealing(h, time_limit=120, seed=42):
    rng = np.random.default_rng(seed)
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur
    
    n = len(h_cur)
    T = 0.01
    T_min = 1e-6
    cooling = 0.9995
    
    start = time.time()
    while T > T_min and time.time() - start < time_limit:
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        c5_new = compute_c5(h_cur)
        delta = c5_new - c5_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_cur.copy()
        else:
            h_cur[j] = 0.0
            h_cur[i] = 1.0
        
        T *= cooling
    
    return h_best, c5_best

def solve_milp_erdos(n, time_limit=120):
    dx = 2.0 / n
    N = n
    
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    
    z_vars = {}
    for i in range(N):
        for j in range(i):
            z_vars[(i, j)] = cp.Variable()
    
    constraints = [cp.sum(h) == N // 2]
    
    for (i, j), z in z_vars.items():
        constraints.append(z <= h[i])
        constraints.append(z <= h[j])
        constraints.append(z >= h[i] + h[j] - 1)
    
    for k in range(2 * N - 1):
        lag = k - (N - 1)
        n_min = max(0, -lag)
        n_max = min(N - 1, N - 1 - lag)
        
        if n_min > n_max:
            continue
        
        expr = cp.Constant(0)
        for n_idx in range(n_min, n_max + 1):
            i = n_idx + lag
            j = n_idx
            expr += h[i]
            if i == j:
                expr -= h[i]
            elif i > j:
                expr -= z_vars[(i, j)]
            else:
                expr -= z_vars[(j, i)]
        
        constraints.append(dx * expr <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    solvers = [cp.CBC, cp.SCIP, cp.GLPK_MI]
    for solver in solvers:
        try:
            prob.solve(solver=solver, time_limit=time_limit)
            if h.value is not None:
                h_val = np.round(h.value).astype(float)
                if np.all((h_val >= 0) & (h_val <= 1)) and abs(np.sum(h_val) - N//2) < 1e-6:
                    c5 = compute_c5(h_val)
                    return h_val, c5
        except Exception:
            continue
    
    return None, None
