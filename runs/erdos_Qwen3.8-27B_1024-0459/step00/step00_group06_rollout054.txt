import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_onto_constraint(h, n):
    """Project h onto the feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need to increase sum
            mask = h < 1.0
        else:
            # Need to decrease sum
            mask = h > 0.0
        if np.any(mask):
            per = diff / np.sum(mask)
            h[mask] += per
            h = np.clip(h, 0, 1)
            # If still off, iterate
            for _ in range(10):
                diff = target - np.sum(h)
                if abs(diff) < 1e-10:
                    break
                if diff > 0:
                    mask = h < 1.0
                else:
                    mask = h > 0.0
                if not np.any(mask):
                    break
                per = diff / np.sum(mask)
                h[mask] += per
                h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)

        return budget_s - (time.time() - start_time)

            return np.sum(h) - target_sum
        
        cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones(n)}]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', bounds=bounds,

One more consideration: the gradient of the smooth max. For SLSQP, providing the Jacobian helps. The gradient of (1/β)log(Σexp(β*g_k)) with respect to h is:
(1/β) * Σ_k exp(β*g_k) * β * ∂g_k/∂h / Σ_k exp(β*g_k)
= Σ_k p_k * ∂g_k/∂h
where p_k = exp(β*g_k) / Σ_j exp(β*g_j) (softmax weights)

And g_k = corr_k * dx where corr_k = Σ_i h[i+k]*(1-h[i]) (with appropriate index ranges)

∂corr_k/∂h[j]:
- If j appears as h[i+k] (i.e., j = i+k for some valid i): ∂/∂h[j] of h[j]*(1-h[j-k]) = (1-h[j-k])
- If j appears as h[i] in the (1-h[i]) factor: ∂/∂h[j] of h[j+k]*(1-h[j]) = -h[j+k]
- For k=0: both apply, giving (1-h[j]) - h[j] = 1-2h[j]

import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta

    target = n / 2.0

    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        iterations = 0

            if diff > 0:

            else:

            per = diff / np.sum(mask)
            h[mask] += per
            h = np.clip(h, 0, 1)
            diff = target - np.sum(h)
            iterations += 1

_BETA = 5000.0

    return lse_max(h, _N_POINTS, _BETA)


def _constraint(h):
    return np.sum(h) - _TARGET_SUM


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def lse_max(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta


def check_and_fix(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)

        per = diff / cnt
        h[mask] += per
        np.clip(h, 0.0, 1.0, out=h)
