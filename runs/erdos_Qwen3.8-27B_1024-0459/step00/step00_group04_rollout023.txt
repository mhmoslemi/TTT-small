def compute_grad_for_lag(hh, n, lag):
    grad = np.zeros(n)
    # First term: (1-hh[i+lag]) for 0 <= i+lag < n
    # i+lag >= 0 → i >= -lag
    # i+lag < n → i < n-lag
    # i >= 0, i < n
    lo1 = max(0, -lag)
    hi1 = min(n, n - lag)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += (1.0 - hh[idx + lag])
    
    # Second term: -hh[i-lag] for 0 <= i-lag < n
    # i-lag >= 0 → i >= lag
    # i-lag < n → i < n+lag
    # i >= 0, i < n
    lo2 = max(0, lag)
    hi2 = min(n, n + lag)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] -= hh[idx - lag]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad_for_lag(hh, n, lag):
    grad = np.zeros(n)
    lo1 = max(0, -lag)
    hi1 = min(n, n - lag)
    if lo1 < hi1:
        idx = np.arange(lo1, hi1)
        grad[idx] += (1.0 - hh[idx + lag])
    lo2 = max(0, lag)
    hi2 = min(n, n + lag)
    if lo2 < hi2:
        idx = np.arange(lo2, hi2)
        grad[idx] -= hh[idx - lag]
    return grad

def obj_grad(h, n, dx, penalty):
    hh = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    corr = np.correlate(hh, 1.0 - hh, mode='full')
    max_val = np.max(corr)
    tol = 1e-10
    active = np.where(corr >= max_val - tol)[0]
    grad_total = np.zeros(n)
    for m in active:
        lag = int(m - (n - 1))
        grad_total += compute_grad_for_lag(hh, n, lag)
    grad_total /= len(active)
    grad_total *= dx
    s = np.sum(hh) - n / 2.0
    obj = max_val * dx + penalty * s * s
    grad_total += 2.0 * penalty * s
    return obj, grad_total

def project_to_constraint(h, n):
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 5e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def try_optimize(h_init, n, penalty, max_iter, ftol, gtol):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj_grad,
        h_init.copy(),
        args=(n, dx, penalty),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                 'maxfun': max_iter * 25, 'disp': False}
    )
    h_out = np.clip(res.x, 0.0, 1.0)
    h_out = project_to_constraint(h_out, n)
    return h_out

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    return np.clip(h_dst, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h_parent = np.full(42, 0.5)

    n_parent = len(h_parent)
    best_h = project_to_constraint(h_parent, n_parent)
    best_c5 = compute_c5(best_h)
    best_n = n_parent

    # Phase 1: Optimize at parent resolution
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.15:
        for pen in [1e4, 1e6]:
            h_opt = try_optimize(best_h, best_n, pen, 8000, 1e-16, 1e-10)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = best_n

    # Phase 2: Multi-resolution refinement
    resolutions = sorted(set([best_n]))
    for mult in [2, 4, 8]:
        nr = best_n * mult
        if nr <= 512:
            resolutions.append(nr)
    # Remove duplicates
    resolutions = sorted(set(resolutions))

    for n in resolutions:
        if n == best_n:
            continue
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.75:
            break

        h_curr = interpolate_h(best_h, n)
        h_curr = project_to_constraint(h_curr, n)

        # First pass with moderate penalty
        pen = 1e5
        max_it = int(min(6000, max(2000, (budget_s * 0.6 - elapsed) * 30)))
        h_opt = try_optimize(h_curr, n, pen, max_it, 1e-16, 1e-10)
        c5 = compute_c5(h_opt)
        improved = False
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n
            improved = True

        # Second pass with higher penalty for tighter constraint
        if improved or c5 < best_c5 + 0.001:
            elapsed = time.time() - start_time
            if elapsed < budget_s * 0.7:
                pen2 = 1e7
                max_it2 = int(min(4000, max(1000, (budget_s * 0.65 - elapsed) * 30)))
                h_opt2 = try_optimize(best_h if improved else h_opt, n, pen2, max_it2, 1e-16, 1e-11)
                c5_2 = compute_c5(h_opt2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h_opt2
                    best_n = n

    # Phase 3: Random restarts at best resolution
    elapsed = time.time() - start_time
    n_restart = best_n
    if n_restart < 84 and budget_s * 0.5 > elapsed:
        n_restart = 84
        h_restart_base = project_to_constraint(interpolate_h(best_h, n_restart), n_restart)
    else:
        h_restart_base = best_h.copy()
        n_restart = best_n

    for attempt in range(8):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.88:
            break

        scale = 0.03 * (1 + attempt * 0.5)
        h_pert = h_restart_base + rng.randn(n_restart) * scale
        h_pert = project_to_constraint(h_pert, n_restart)

        pen = 1e6
        max_it = int(min(5000, max(1500, (budget_s * 0.82 - elapsed) * 25)))
        try:
            h_opt = try_optimize(h_pert, n_restart, pen, max_it, 1e-16, 1e-11)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n = n_restart
        except Exception:
            pass

    # Phase 4: Try symmetric/structured patterns at high resolution
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.85:
        for n_try in [168, 336]:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.88:
                break
            # Try a block pattern: half 0s, half 1s arranged optimally
            h_block = np.zeros(n_try)
            h_block[::2] = 1.0
            h_block = project_to_constraint(h_block, n_try)
            pen = 1e6
            max_it = int(min(3000, max(1000, (budget_s * 0.85 - elapsed) * 20)))
            try:
                h_opt = try_optimize(h_block, n_try, pen, max_it, 1e-16, 1e-10)
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    best_n = n_try
            except Exception:
                pass

    # Final: ensure constraint is met and return
    best_h = project_to_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
