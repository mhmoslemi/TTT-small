import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_obj(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    # The full correlation has 2n-1 entries indexed 0 to 2n-2
    # lag = max_idx - (n-1)
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # result = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
        for i in range(n - lag):
            grad[i + lag] += (1.0 - h[i])
            grad[i] += (-h[i + lag])
    else:
        k = -lag
        # result = sum_{i=k}^{n-1} h[i] * (1-h[i-k])
        for i in range(k, n):
            grad[i] += (1.0 - h[i-k])
            grad[i-k] += (-h[i])
    grad *= dx
    return grad

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = None

    best_c5 = float('inf')
    best_n = 0

            break

            x_new = np.linspace(0, 2, n, endpoint=False) + 1.0/n
            h = np.interp(x_new, x_old, h_init, period=2.0)
        else:

            h = rng.randint(0, 2, size=n).astype(float)

        target_sum = n / 2.0

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_grad(h, n):
    """Compute C5 and a subgradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    # Find all max indices (for subgradient)
    max_val = corr.max()
    max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    grad = np.zeros(n)
    for max_idx in max_indices:
        lag = max_idx - (n - 1)
        if lag >= 0:
            # corr = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
            idx1 = np.arange(lag, n)
            idx2 = np.arange(0, n - lag)
            grad[idx1] += (1.0 - h[idx2])
            grad[idx2] -= h[idx1]
        else:
            k = -lag
            idx1 = np.arange(k, n)
            idx2 = np.arange(0, n - k)
            grad[idx1] += (1.0 - h[idx1 - k])  # h[i] * (1-h[i-k])

One more consideration: the gradient computation. For the full correlation of length 2n-1, computing all elements is O(n^2). Computing just the gradient at the max element is O(n). But for SLSQP, we need the full gradient evaluation (one f and one grad per iteration).

Actually, np.correlate(h, 1-h, mode='full') is O(n^2) for the full result. But we only need the max and the gradient at the max. Can we compute just the max more efficiently?

For the max, we need all corr values. For the gradient at the max, we just need the max index. So:
- Compute full corr: O(n^2) (or O(n log n) with FFT)
- Find max: O(n)
- Compute gradient at max: O(n)

Using FFT: corr can be computed via FFT in O(n log n). Let's use that for larger n.

Actually, np.correlate is already quite fast in numpy. For n=800, it's 800*1600 ≈ 1.3M multiply-adds, which is trivial. Let's not over-engineer.

def compute_c5_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = float(np.max(corr) * dx)
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        i = np.arange(0, n - lag)
        grad[i + lag] += (1.0 - h[i])
        grad[i] -= h[i + lag]

Let me also think about whether to use the "soft-max" of the top few correlations instead of just the max. This can help with convergence by smoothing the objective.

Actually, for SLSQP, the non-smoothness of max at points where multiple correlations are equal is usually handled fine. The main issue is that the gradient points in a specific direction (the direction that decreases the currently-maximizing correlation), which might not be the best direction overall.

A better approach: use the gradient of a "bottleneck" function:
f(h) = max_k g_k(h) + ε * sum_k max(g_k(h) - max_j g_j(h), 0)^2

This adds a smoothing term that pushes down all correlations that are near the max. But this complicates the gradient.

Simpler: just use SLSQP with the max and let it handle things. For the non-smooth points, SLSQP uses a numerical gradient step, which should be fine.

Alternatively, I can use the "sign" gradient: for each h, compute all g_k, find the top T (e.g., T=5) that are within ε of the max, and average their gradients. This gives a more "stable" gradient direction.
