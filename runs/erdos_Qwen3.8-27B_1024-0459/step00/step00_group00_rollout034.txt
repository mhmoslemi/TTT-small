    dx = 2.0 / n
    corr = np.correlate(h, 1-h, 'full') * dx

def optimize_coord(h, n, m):

    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    # Precompute the "base" correlation (excluding h[m] contributions)
    # and the coefficients a[i] for each correlation index i
    a = np.zeros(n_corr)
    b = np.zeros(n_corr)
    
    for i in range(n_corr):
        lag = i - (n - 1)
        j_lo = max(0, -lag)
        j_hi = min(n - 1, n - 1 - lag)

            # We'll handle this separately
            a[i] += 1.0  # coefficient of t
            # We need to track the -t² term
    
    # Evaluate on grid
    ts = np.linspace(0, 1, 201)
    best_t = 0.0
    best_val = np.inf
    
    for t in ts:

        # Handle quadratic: for the lag=0 term (i = n-1)
        # Actually, for i = n-1 (lag=0), the contribution is t - t² instead of a[i]*t
        # So: val[n-1] = b[n-1] + t - t² (instead of b[n-1] + a[n-1]*t)
        # a[n-1] was set to 1.0, so we need to subtract t²
        # Actually let me just compute directly
        worst = 0.0
        for i in range(n_corr):
            lag = i - (n - 1)
            if lag == 0:
                # This is the only case where quadratic appears

                # But wait, b[i] here already includes all non-m terms

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    h = h.copy()
    if diff > 0:
        order = np.argsort(-h)
        for idx in order:
            room = 1.0 - h[idx]
            adj = min(room, diff)
            h[idx] += adj
            diff -= adj
            if diff < 1e-12:
                break
    else:
        order = np.argsort(h)
        for idx in order:
            room = h[idx]
            adj = min(room, -diff)
            h[idx] -= adj
            diff += adj
            if diff > -1e-12:
                break
    return h

def optimize_coord_descent(h0, n, max_passes, time_limit, seed=42):

    ts_grid = np.linspace(0.0, 1.0, 101)
    
    for p in range(max_passes):
        if time.time() - t_start > time_limit:
            break
        order = rng.permutation(n)
        for step in range(n):
            if time.time() - t_start > time_limit:
                break
            m = order[step]
            
            a = np.zeros(n_corr)
            b = np.zeros(n_corr)
            has_quad = [False] * n_corr
            
            for i in range(n_corr):
                lag = i - (n - 1)
                j_lo = max(0, -lag)
                j_hi = min(n - 1, n - 1 - lag)
                bi = 0.0

                    elif j == m:

                    elif j2 == m:
                        bi += h[j]
                        ai += -h[j]
                    else:
                        bi += h[j] * (1.0 - h[j2])
                if quad:
                    ai += 1.0
                a[i] = ai
                b[i] = bi
                if quad:
                    has_quad[i] = True
            
            best_t = h[m]
            best_worst = np.inf
            for t in ts_grid:
                vals = b + a * t

                    if has_quad[qi]:
                        vals[qi] = b[qi] + t - t * t
                w = np.max(vals) * dx
                if w < best_worst:
                    best_worst = w
                    best_t = t
            
            h[m] = best_t
        
        h = project_sum(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def binary_local_search(h0, n, max_iters, time_limit, seed=0):

def soft_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c = corr / tau
    c_max = np.max(c)
    return float(tau * (c_max + np.log(np.sum(np.exp(c - c_max)))))

def soft_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    c = corr / tau
    c_max = np.max(c)
    exp_c = np.exp(c - c_max)
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        wi = w[i]
        if wi < 1e-10:
            continue
        lag = i - (n - 1)
        j_lo = max(0, -lag)
        j_hi = min(n - 1, n - 1 - lag)
        for j in range(j_lo, j_hi + 1):
            j2 = j + lag
            grad[j] += dx * wi * (1.0 - h[j2])
            if j2 != j:
                grad[j2] += -dx * wi * h[j]
            else:
                grad[j] += -dx * wi * h[j]
    return grad

def gradient_refine(h0, n, tau, time_limit):
    t_start = time.time()
    h = h0.copy()
    penalty = 200.0
    
    def fun(x):
        return soft_max_obj(x, tau) + penalty * (np.sum(x) - n / 2.0) ** 2
    
    def jac(x):
        g = soft_max_grad(x, tau)
        sv = np.sum(x) - n / 2.0
        g += 2.0 * penalty * sv / n
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 5000, 'gtol': 1e-9, 'maxfun': 50000})
        if time.time() - t_start < time_limit:
            h = np.clip(res.x, 0.0, 1.0)
    except:
        pass
    h = project_sum(h, n)
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    if n % 2 == 0:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    else:
        idx = rng.choice(n, (n - 1) // 2, replace=False)
        h[idx] = 1.0
        rem = np.setdiff1d(np.arange(n), idx)
        h[rem[rng.randint(len(rem))]] = 0.5
    return h

def expand_to_n(parent_h, new_n):

    h_new = project_sum(h_new, new_n)
    return h_new
