        h_minus[i] -= eps

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_objective(h, n_points, beta):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    if beta > 5000:
        return float(max_val)
    shifted = beta * (corr - max_val)
    # Clip for numerical stability
    shifted = np.maximum(shifted, -50)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return float(max_val + log_sum / beta)

def soft_max_grad(h, n_points, beta, eps=1e-6):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    if beta > 5000:
        # Hard max: subgradient from active set
        active = np.where(corr >= max_val - 1e-10)[0]
        grad = np.zeros(n_points)
        for k in active:
            # Analytical gradient of f_k
            m = k - (n_points - 1)
            if m >= 0:
                for i in range(n_points - m):
                    grad[i + m] += dx * (1.0 - h[i])
                    grad[i] -= dx * h[i + m]
            else:
                for i in range(-m, n_points):
                    grad[i] += dx * (1.0 - h[i + m])
                    grad[i + m] -= dx * h[i]
        grad /= len(active)
        return grad
    shifted = beta * (corr - max_val)
    shifted = np.maximum(shifted, -50)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    grad = np.zeros(n_points)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        m = k - (n_points - 1)
        if m >= 0:
            for i in range(n_points - m):
                grad[i + m] += weights[k] * dx * (1.0 - h[i])
                grad[i] -= weights[k] * dx * h[i + m]
        else:
            for i in range(-m, n_points):
                grad[i] += weights[k] * dx * (1.0 - h[i + m])
                grad[i + m] -= weights[k] * dx * h[i]
    return grad

def repair_h(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n_points, 0.5)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h0, n_points, beta_start, beta_end, max_iter=300, tol=1e-8):
    h_cur = h0.copy()
    for beta in [beta_start, beta_start*2, beta_start*4, beta_start*8,
                 beta_start*16, beta_start*32, beta_start*64]:
        beta = min(beta, beta_end)
        try:
            res = minimize(
                soft_max_objective,
                h_cur,
                args=(n_points, beta),
                jac=lambda x: soft_max_grad(x, n_points, beta),
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n_points,
                constraints=[{'type': 'eq',
                              'fun': lambda x: np.sum(x) - n_points/2.0,
                              'jac': lambda x: np.ones(n_points)}],
                options={'maxiter': max(30, max_iter//7), 'ftol': tol}
            )
            h_cur = res.x
            h_cur = repair_h(h_cur, n_points)
        except Exception:
            pass
    return h_cur

def top_k_objective(h, n_points, k):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(corr)[-k:]
    return float(np.mean(corr[idx]))

def try_initialization(n_points, rng, mode):
    if mode == 0:
        h = 0.5 * np.ones(n_points)
    elif mode == 1:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        if n_points % 2 != 0:
            h[n_points//2] = 0.5
    elif mode == 2:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = 0.5 + 0.25 * np.cos(2 * np.pi * x)
    elif mode == 3:
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = 0.5 + 0.3 * np.cos(4 * np.pi * x)
    elif mode == 4:
        # Thue-Morse like
        h = np.zeros(n_points)
        tm = 0
        for i in range(n_points):
            bits = bin(i).count('1')
            h[i] = 1.0 if bits % 2 == 0 else 0.0
    elif mode == 5:
        h = rng.uniform(0, 1, n_points)
    else:
        h = 0.5 * np.ones(n_points)
        h += rng.normal(0, 0.1, n_points)
    h = np.clip(h, 0, 1)
    h = repair_h(h, n_points)
    return h
