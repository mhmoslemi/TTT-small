import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_objective(h, n, t, lam):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    c = corr * t
    max_c = np.max(c)
    exp_c = np.exp(c - max_c)
    log_sum_exp = max_c + np.log(np.sum(exp_c))
    soft_max = log_sum_exp / t * dx
    sum_penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return soft_max + sum_penalty

def smooth_max_gradient(h, n, t, lam):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    c = corr * t
    max_c = np.max(c)
    exp_c = np.exp(c - max_c)
    weights = exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        w = wk * t * dx
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + s:hi1 + s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 - s:hi2 - s]
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    return grad

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_resolution(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    target = n_dst / 2.0
    h_dst = project_sum(h_dst, target)
    return h_dst

def get_obj_and_grad(n, t, lam):
    def obj(x):
        return smooth_max_objective(x, n, t, lam)
    def grad(x):
        return smooth_max_gradient(x, n, t, lam)
    return obj, grad
