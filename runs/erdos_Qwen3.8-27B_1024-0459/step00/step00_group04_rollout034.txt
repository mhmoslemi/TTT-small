import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    def S(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0))
    lo, hi = -3.0, 3.0
    while S(lo) > target:
        lo -= 1.0
    while S(hi) < target:
        hi += 1.0
    for _ in range(300):
        mid = (lo + hi) * 0.5
        if S(mid) < target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h_proj)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        if not np.any(mask):
            break
        h_proj[mask] += diff / np.sum(mask)
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def interp_resolution(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h, n_new)
    if n_old <= 1:
        return project_feasible(np.full(n_new, h[0]), n_new)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def optimize_lbfgsb(h, n, time_alloc):
    dx = 2.0 / n
    target = n / 2.0
    x0 = project_feasible(h, n)

    def obj(params):
        corr = np.correlate(params, 1.0 - params, mode='full')
        val = np.max(corr) * dx
        pen = 1e6 * (np.sum(params) - target) ** 2
        return val + pen

    bounds = [(0.0, 1.0)] * n
    maxiter = max(100, min(3000, int(time_alloc * 3)))

    c5_current = compute_c5(x0)
    best_x = x0.copy()
    best_val = c5_current

    t_start = time.time()
    try:
        res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxls': 40})
        h_candidate = project_feasible(res.x, n)
        c5_new = compute_c5(h_candidate)
        if c5_new < best_val:
            best_x = h_candidate
            best_val = c5_new
    except Exception:
        pass

    elapsed = time.time() - t_start
    if time_alloc - elapsed > 15:
        try:
            res2 = minimize(obj, best_x, method='L-BFGS-B', bounds=bounds,
                            options={'maxiter': maxiter, 'ftol': 1e-16,
                                     'gtol': 1e-12, 'maxls': 40})
            h_candidate2 = project_feasible(res2.x, n)
            c5_new2 = compute_c5(h_candidate2)
            if c5_new2 < best_val:
                best_x = h_candidate2
                best_val = c5_new2
        except Exception:
            pass

    return best_x

def active_set_descent(h, n, steps=200, lr=None):
    dx = 2.0 / n
    target = n / 2.0
    h = project_feasible(h, n)
    if lr is None:
        lr = 0.01
    for step in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        # Numerical gradient of c_{k_star} w.r.t. h
        eps = 1e-6
        grad = np.zeros(n)
        for i in range(n):
            h_p = h.copy()
            h_p[i] += eps
            c_p = np.correlate(h_p, 1.0 - h_p, mode='full')[k_star]
            h_m = h.copy()
            h_m[i] -= eps
            c_m = np.correlate(h_m, 1.0 - h_m, mode='full')[k_star]
            grad[i] = (c_p - c_m) / (2.0 * eps)
        h = h - lr * grad * dx
        h = project_feasible(h, n)
        lr *= 0.98
    return h
