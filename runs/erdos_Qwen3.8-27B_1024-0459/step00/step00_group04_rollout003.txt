class BudgetError(Exception):
    pass


# ---------------------------------------------------------------------------
# Correlation / objective helpers.

def _correlate_full(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    n = a.size
    if n <= 0:
        return np.array([], dtype=float)
    m = b.size
    if n <= 700 and m <= 700:
        return np.correlate(a, b, mode="full")
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b[::-1], size)
    out = np.fft.irfft(fa * fb, size)[:need]
    return out


def _convolve_full(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    na = a.size
    nb = b.size
    if na == 0 or nb == 0:
        return np.array([], dtype=float)
    if na * nb < 300000:
        return np.convolve(a, b, mode="full")
    need = na + nb - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:need]


def objective_values(h):
    h = np.asarray(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.0)
    n = h.size
    if n <= 1:
        return np.zeros(max(1, 2 * n - 1), dtype=float)
    dx = 2.0 / float(n)
    return dx * _correlate_full(h, 1.0 - h)


def objective_max(h):
    try:
        c = objective_values(h)
        if c.size == 0:
            return 0.0
        return float(np.max(c))
    except Exception:
        return float("inf")


# ---------------------------------------------------------------------------
# Feasibility helpers.
# ---------------------------------------------------------------------------

def project_box_sum(v, S):
    v = np.asarray(v, dtype=float)
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.5)
    S = float(S)
    if S <= 0.0:
        return np.zeros(n, dtype=float)
    if S >= float(n):
        return np.ones(n, dtype=float)

    vc = np.clip(v, 0.0, 1.0)
    s0 = float(vc.sum())
    if abs(s0 - S) <= 1e-12 * max(1, n):
        return vc

    lo = float(v.min()) - 1.0
    hi = float(v.max())
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        t = v - mid
        ss = float(np.sum(np.clip(t, 0.0, 1.0)))
        if ss > S:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    h = np.clip(v - lam, 0.0, 1.0)

    diff = S - float(h.sum())
    if abs(diff) > 1e-13:
        tol = 1e-12
        free = np.where((h > tol) & (h < 1.0 - tol))[0]
        if free.size > 0:
            h[free] += diff / float(free.size)
            h = np.clip(h, 0.0, 1.0)
        else:
            if diff > 0.0:
                idx = int(np.argmax(1.0 - h))
            else:
                idx = int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))

    # Final tiny correction.
    diff = S - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            slack = 1.0 - h
            if float(slack.max()) > 0.0:
                idx = int(np.argmax(slack))
                h[idx] = min(1.0, float(h[idx] + diff))
        else:
            if float(h.min()) > 0.0:
                idx = int(np.argmin(h))
                h[idx] = max(0.0, float(h[idx] + diff))
    return h


def make_feasible(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return h
    return project_box_sum(h, 0.5 * float(h.size))


def symmetrize_feasible(h):
    h = np.asarray(h, dtype=float)
    h = 0.5 * (h + h[::-1])
    return make_feasible(h)


def _upsample_h(h, n2):
    h = np.asarray(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.5)
    n1 = h.size
    n2 = int(n2)
    if n2 <= 1:
        return make_feasible(h[:1])
    if n2 == n1:
        return make_feasible(h)
    if n2 > 0 and n2 % n1 == 0:
        out = np.repeat(h, n2 // n1)
    else:
        x_old = np.linspace(0.0, 1.0, n1)
        x_new = np.linspace(0.0, 1.0, n2)
        out = np.interp(x_new, x_old, h)
    return make_feasible(out)


# ---------------------------------------------------------------------------
# Smoothed max objective and analytic gradients.
# ---------------------------------------------------------------------------

def _softmax_objective(c, tau):
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return 0.0, np.array([], dtype=float)
    cmax = float(np.max(c))
    m = c.size
    w = np.zeros(m, dtype=float)
    if not np.isfinite(cmax):
        w[int(np.argmax(c))] = 1.0
        return float("inf"), w

    if tau <= 1e-12:
        mask = c >= (cmax - 1e-10)
        if not np.any(mask):
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / float(np.sum(mask))
        return cmax, w

    z = (c - cmax) / tau
    z = np.where(z > 0.0, 0.0, z)
    z = np.where(z < -745.0, -745.0, z)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0.0:
        w[int(np.argmax(c))] = 1.0
        return cmax, w
    w = ez / s
    soft = cmax + tau * float(np.log(s))
    return soft, w


def _lag_gradient_one(h, s):
    n = h.size
    g = np.zeros(n, dtype=float)
    if s >= 0:
        ls = n - s
        if ls > 0:
            g[s:] += 1.0 - h[:ls]
            g[:ls] -= h[s:]
    else:
        l = -s
        ls = n - l
        if ls > 0:
            g[:ls] += 1.0 - h[l:]
            g[l:] -= h[:-l]
    return g


def _raw_weighted_gradient(h, W):
    h = np.asarray(h, dtype=float)
    W = np.asarray(W, dtype=float)
    n = h.size
    off = n - 1
    if W.size != 2 * n - 1:
        W2 = np.zeros(2 * n - 1, dtype=float)
        mn = min(W.size, W2.size)
        W2[:mn] = W[:mn]
        W = W2

    if n == 0:
        return np.array([], dtype=float)

    support = np.flatnonzero(W > 1e-14)
    k = support.size
    if k == 0:
        return np.zeros(n, dtype=float)

    # Sparse path: direct sum over active lags.
    if k <= max(16, n // 4):
        g = np.zeros(n, dtype=float)
        for idx in support:
            w = float(W[idx])
            s = int(idx) - off
            if s >= 0:
                ls = n - s
                if ls > 0:
                    g[s:] += w * (1.0 - h[:ls])
                    g[:ls] -= w * h[s:]
            else:
                l = -s
                ls = n - l
                if ls > 0:
                    g[:ls] += w * (1.0 - h[l:])
                    g[l:] -= w * h[:-l]
        return g

    # Dense path: convolutions.
    Wpos = W[off: off + n]
    Wneg = np.zeros(n, dtype=float)
    if n > 1:
        Wneg[1:] = W[0:n - 1][::-1]

    conv_pos = _convolve_full(h, Wpos)
    conv_posrev = _convolve_full(h, Wpos[::-1])
    conv_neg = _convolve_full(h, Wneg)
    conv_negrev = _convolve_full(h, Wneg[::-1])

    cum_pos = np.cumsum(Wpos)
    pref_neg = np.cumsum(Wneg)

    Apos = cum_pos - conv_pos[:n]
    Bpos = conv_posrev[off: off + n]
    Aneg = pref_neg[::-1] - conv_negrev[off: off + n]
    Bneg = conv_neg[:n]

    return Apos + Aneg - Bpos - Bneg


# ---------------------------------------------------------------------------
# SLSQP parameterization interface.
# ---------------------------------------------------------------------------

def _slsqp_h(x):
    x = np.asarray(x, dtype=float)
    x = np.where(np.isfinite(x), x, 0.5)
    xc = np.clip(x, 0.0, 1.0)
    if _PARAM_MODE == 1:
        return np.concatenate((xc, xc[::-1].copy()))
    return xc


def _slsqp_fun(x):
    try:
        h = _slsqp_h(x)
        c = objective_values(h)
        val, _ = _softmax_objective(c, _CUR_TAU)
        if np.isfinite(val):
            return float(val)
        return 1e10
    except Exception:
        return 1e10


def _slsqp_jac(x):
    try:
        h = _slsqp_h(x)
        c = objective_values(h)
        _, w = _softmax_objective(c, _CUR_TAU)
        raw = _raw_weighted_gradient(h, w)
        g = _CUR_DX * raw
        g = np.where(np.isfinite(g), g, 0.0)
        if _PARAM_MODE == 1:
            m = int(np.asarray(x).size)
            if g.size == 2 * m:
                return g[:m] + g[m:][::-1].copy()
        return g
    except Exception:
        return np.zeros_like(np.asarray(x, dtype=float))


def _eq_con_fun(x):
    try:
        return float(np.sum(x) - _CUR_TARGET)
    except Exception:
        return 0.0


def _eq_con_jac(x):
    try:
        return np.ones_like(x, dtype=float)
    except Exception:
        return np.ones(1, dtype=float)


def _update_best(h):
    global _BEST_H, _BEST_C, _BEST_N
    try:
        h = np.asarray(h, dtype=float)
        if h.size < 2:
            return
        h = make_feasible(h)
        c = objective_max(h)
        if np.isfinite(c) and c < _BEST_C:
            _BEST_H = h.copy()
            _BEST_C = float(c)
            _BEST_N = int(h.size)
    except Exception:
        pass


def make_h_from_x(x):
    try:
        h = _slsqp_h(x)
        if _CUR_N > 0 and h.size == _CUR_N:
            return project_box_sum(h, 0.5 * float(_CUR_N))
        return make_feasible(h)
    except Exception:
        return make_feasible(x)


def _slsqp_callback(xk):
    global _OPT_COUNT, _LAST_X
    try:
        arr = np.asarray(xk, dtype=float)
    except Exception:
        arr = None
    if arr is not None:
        _LAST_X = arr.copy()
    _OPT_COUNT = _OPT_COUNT + 1
    now = time.time()
    if now >= _OPT_LIMIT:
        if arr is not None:
            try:
                _update_best(make_h_from_x(arr))
            except Exception:
                pass
        raise BudgetError
    if _OPT_COUNT % 10 == 0 and arr is not None:
        try:
            _update_best(make_h_from_x(arr))
        except Exception:
            pass


# ---------------------------------------------------------------------------
# SLSQP refinement with time control.
# ---------------------------------------------------------------------------

def refine_slsqp(h0, time_limit, tau_seq, symmetric):
    global _PARAM_MODE, _CUR_N, _CUR_DX, _CUR_TARGET, _CUR_TAU
    global _LAST_X, _OPT_COUNT, _OPT_LIMIT, _OPT_START

    if h0 is None:
        return h0
    try:
        time_limit = max(0.0, float(time_limit))
    except Exception:
        time_limit = 0.0
    if time_limit < 2.0:
        return h0

    h0 = make_feasible(h0)
    n = h0.size
    if n < 4:
        return h0

    if symmetric and (n % 2 != 0):
        symmetric = False

    if symmetric:
        h0 = symmetrize_feasible(h0)
        m = n // 2
        target = 0.25 * float(n)
    else:
        m = n
        target = 0.5 * float(n)

    x0 = project_box_sum(h0[:m], target)

    _PARAM_MODE = 1 if symmetric else 0
    _CUR_N = n
    _CUR_DX = 2.0 / float(n)
    _CUR_TARGET = target

    taues = []
    try:
        for t in tau_seq:
            taues.append(float(t))
    except Exception:
        taues = [0.01]
    if len(taues) == 0:
        taues = [0.01]

    stage_start = time.time()
    budget = time_limit

    try:
        best_x = x0.copy()
        best_x_c = objective_max(make_h_from_x(best_x))
    except Exception:
        best_x_c = float("inf")

    for ti, tau in enumerate(taues):
        elapsed = time.time() - stage_start
        avail = budget - elapsed
        if avail < 2.0:
            break
        remaining = len(taues) - ti
        per_time = max(2.0, avail / float(remaining))

        _CUR_TAU = tau
        _LAST_X = None
        _OPT_COUNT = 0
        _OPT_START = time.time()
        _OPT_LIMIT = time.time() + per_time

        maxiter = int(max(20, min(120, 2 * max(1, m))))
        bounds = [(0.0, 1.0)] * int(m)
        constraints = [{"type": "eq", "fun": _eq_con_fun, "jac": _eq_con_jac}]

        x0 = project_box_sum(x0, target)

        try:
            res = opt.minimize(
                _slsqp_fun,
                x0,
                jac=_slsqp_jac,
                bounds=bounds,
                constraints=constraints,
                method="SLSQP",
                options={"maxiter": maxiter, "ftol": 1e-8, "disp": False},
                callback=_slsqp_callback,
            )
            if hasattr(res, "x") and res.x is not None:
                x0 = np.asarray(res.x, dtype=float)
            elif _LAST_X is not None:
                x0 = np.asarray(_LAST_X, dtype=float)
        except BudgetError:
            if _LAST_X is not None:
                x0 = np.asarray(_LAST_X, dtype=float)
        except Exception:
            pass

        try:
            h = make_h_from_x(x0)
            _update_best(h)
            c_cur = objective_max(h)
            if c_cur < best_x_c:
                best_x_c = c_cur
                best_x = x0.copy()
            elif c_cur > best_x_c + 0.003:
                x0 = best_x.copy()
                h = make_h_from_x(x0)
                _update_best(h)

            if symmetric:
                x0 = project_box_sum(h[:m], target)
            else:
                x0 = project_box_sum(h, target)
        except Exception:
            pass

    try:
        return make_h_from_x(x0)
    except Exception:
        return h0


# ---------------------------------------------------------------------------
# Simple projected steepest-descent polish on the true max overlap.
# ---------------------------------------------------------------------------

def active_polish(h, time_limit, delta=0.004):
    if h is None:
        return h
    try:
        time_limit = max(0.0, float(time_limit))
    except Exception:
        time_limit = 0.0
    if time_limit < 0.5:
        return h

    h = make_feasible(h)
    try:
        best_c = objective_max(h)
    except Exception:
        return h
    best_h = h.copy()

    start = time.time()
    steps = [
        0.10, 0.05, 0.025, 0.0125,
        0.00625, 0.003125, 0.0015625, 0.00078125
    ]

    for _ in range(300):
        if time.time() - start >= time_limit:
            break

        try:
            c = objective_values(h)
            fmax = float(np.max(c)) if c.size else 0.0
        except Exception:
            break

        if fmax < best_c:
            best_c = fmax
            best_h = h.copy()

        mask = c >= (fmax - max(1e-8, float(delta)))
        if not np.any(mask):
            mask = np.zeros(c.size, dtype=bool)
            mask[int(np.argmax(c))] = True
        W = np.zeros(c.size, dtype=float)
        W[mask] = 1.0 / float(np.sum(mask))

        raw = _raw_weighted_gradient(h, W)
        dx = 2.0 / float(h.size)
        grad = dx * raw
        grad = np.where(np.isfinite(grad), grad, 0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-12:
            break

        direction = -grad / gnorm
        improved = False
        for L in steps:
            try:
                y = make_feasible(h + L * direction)
                fy = objective_max(y)
                if fy < fmax - 1e-12:
                    h = y
                    fmax = fy
                    improved = True
                    break
            except Exception:
                pass
        if not improved:
            break

    return best_h

# Startups and perturbations.

def perturb_feasible(h, rng, scale):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n < 2:
        return make_feasible(h)
    x = np.linspace(0.0, 1.0, n)
    d = np.zeros(n, dtype=float)
    for f in (1, 2, 3, 5):
        amp = float(rng.normal(scale=0.5 * max(1e-6, scale)))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        d += amp * np.sin(2.0 * np.pi * float(f) * x + ph)
    d += 0.5 * scale * rng.normal(size=n)
    return make_feasible(h + d)


def make_starts(parent, n, rng, count):
    starts = []

    if parent is not None:
        p = np.asarray(parent, dtype=float)
        if p.size == n:
            starts.append(p.copy())
        elif p.size > 0:
            starts.append(_upsample_h(p, n))

    x = np.linspace(0.0, 1.0, n)

    starts.append(make_feasible(np.full(n, 0.5)))

    for mode in (1, 2, 3, 4, 6):
        h = 0.5 + 0.35 * np.sin(np.pi * float(mode) * x)
        starts.append(make_feasible(h))

    starts.append(make_feasible(np.sin(np.pi * x) ** 4))
    starts.append(make_feasible(1.0 - np.abs(2.0 * x - 1.0)))
    starts.append(make_feasible(0.5 + 0.5 * np.cos(2.0 * np.pi * x)))

    for _ in range(count):
        h = np.zeros(n, dtype=float)
        for f in (1, 2, 3, 5, 8):
            amp = float(rng.normal(scale=0.15))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(2.0 * np.pi * float(f) * x + ph)
        h = 0.5 + h
        starts.append(make_feasible(h))

    for _ in range(min(3, count)):
        h = np.zeros(n, dtype=float)
        k = max(1, n // 2)
        idx = rng.permutation(n)[:k]
        h[idx] = 1.0
        kernel = np.array([0.25, 0.5, 0.25], dtype=float)
        h = np.convolve(h, kernel, mode="same")
        starts.append(make_feasible(h))

    return starts


# ---------------------------------------------------------------------------
# Main entry point.
# ---------------------------------------------------------------------------

def run(seed=42, budget_s=1300, **kwargs):
    global _BEST_H, _BEST_C, _BEST_N

    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 1.0:
        budget = 1300.0
    deadline = t0 + budget - 8.0

    _BEST_H = None
    _BEST_C = float("inf")
    _BEST_N = 0

    try:
        rng = np.random.default_rng(seed)
    except Exception:
        rng = np.random.default_rng(42)

    # Use parent if provided.
    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is not None and parent.size >= 2 and bool(np.all(np.isfinite(parent))):
        if parent.size % 2 != 0:
            parent = _upsample_h(parent, parent.size + 1)
        parent = make_feasible(parent)
        _update_best(parent)
        n0 = int(parent.size)
        base = parent.copy()
    else:
        n0 = 64
        base = make_feasible(np.full(n0, 0.5))
        _update_best(base)

    # -----------------------------------------------------------------------
    # Refine at the parent resolution.
    # -----------------------------------------------------------------------

    rem = deadline - time.time()
    if rem > 30.0:
        refine_slsqp(
            base,
            min(240.0, 0.40 * rem),
            [0.02, 0.01, 0.005, 0.002, 0.001],
            False,
        )

    rem = deadline - time.time()
    if rem > 25.0:
        refine_slsqp(
            symmetrize_feasible(base),
            min(120.0, 0.25 * rem),
            [0.02, 0.01, 0.005],
            True,
        )

    rem = deadline - time.time()
    if rem > 60.0:
        starts = make_starts(parent, n0, rng, 8)
        for st in starts:
            rem = deadline - time.time()
            if rem < 25.0:
                break
            tlim = min(8.0, 0.20 * rem)
            if tlim < 2.5:
                break
            refine_slsqp(st, tlim, [0.02, 0.01, 0.005], False)

    if _BEST_H is not None:
        for _ in range(25):
            rem = deadline - time.time()
            if rem < 15.0:
                break
            tlim = min(4.0, rem / 12.0)
            if tlim < 2.0:
                break
            st = perturb_feasible(_BEST_H, rng, 0.05)
            refine_slsqp(st, tlim, [0.01, 0.005], False)

    # -----------------------------------------------------------------------
    # Multi-resolution refinement.
    # -----------------------------------------------------------------------

    if time.time() < deadline:
        max_n = 1008 if budget >= 1200.0 else 672
        if n0 > 200:
            max_n = max(max_n, n0 + 1)

        targets = []
        s = n0
        while s < max_n:
            s = s * 2
            if s <= max_n:
                targets.append(s)
            else:
                break
        if max_n not in targets and max_n > n0:
            targets.append(max_n)

        processed = set()
        for ns in targets:
            rem = deadline - time.time()
            if rem < 80.0:
                break
            if ns in processed:
                continue
            processed.add(ns)

            if _BEST_H is not None and ns <= _BEST_H.size:
                continue

            current = _BEST_H if _BEST_H is not None else base
            h_start = _upsample_h(current, ns)

            if ns <= 700:
                refine_slsqp(
                    h_start,
                    min(240.0, 0.40 * rem),
                    [0.02, 0.01, 0.005, 0.002],
                    False,
                )
            else:
                # Could not have been intended to fail.
                h_start2 = _upsample_h(_BEST_H if _BEST_H is not None else current, ns)
                refine_slsqp(
                    h_start2,
                    min(180.0, 0.40 * rem),
                    [0.02, 0.01, 0.005],
                    True,
                )
                rem = deadline - time.time()
                if rem > 120.0:
                    refine_slsqp(
                        _upsample_h(_BEST_H if _BEST_H is not None else current, ns),
                        min(120.0, 0.40 * rem),
                        [0.01, 0.005],
                        False,
                    )

            rem = deadline - time.time()
            if rem > 50.0 and _BEST_H is not None:
                for _ in range(3):
                    rem = deadline - time.time()
                    if rem < 20.0:
                        break
                    tlim = min(5.0, rem / 8.0)
                    if tlim < 2.0:
                        break
                    st = perturb_feasible(_BEST_H, rng, 0.03)
                    refine_slsqp(st, tlim, [0.01, 0.005], False)

    # -----------------------------------------------------------------------
    # Final light polish.
    # -----------------------------------------------------------------------

    if _BEST_H is not None:
        rem = deadline - time.time()
        if rem > 40.0:
            hpol = active_polish(_BEST_H, min(80.0, rem - 10.0), delta=0.003)
            _update_best(hpol)

    if _BEST_H is None:
        h_ret = make_feasible(np.full(64, 0.5))
    else:
        h_ret = make_feasible(_BEST_H)

    c_ret = objective_max(h_ret)
    if not np.isfinite(c_ret):
        c_ret = 0.5

    return h_ret, float(c_ret), int(h_ret.size)
