def project_to_feasible(h, n_points):
    target = n_points / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            total = np.sum(room)
            if total < 1e-15:
                break
            frac = min(1.0, diff / total)
            h[mask] += frac * room
        else:
            mask = h > 1e-10
            if not np.any(mask):
                break
            room = h[mask]
            total = np.sum(room)
            if total < 1e-15:
                break
            frac = min(1.0, -diff / total)
            h[mask] -= frac * room
    h = np.clip(h, 0.0, 1.0)
