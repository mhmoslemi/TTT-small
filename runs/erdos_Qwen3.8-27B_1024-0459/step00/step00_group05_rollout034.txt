So if j ∈ S(k): ∂c[k]/∂h[j] = (1 - h[k-j]) - h[k-j] = 1 - 2*h[k-j]
If j ∉ S(k): ∂c[k]/∂h[j] = 0

Great, so the gradient is simple:
grad[j] = 1 - 2*h[k-j] if 0 ≤ k-j ≤ n-1 and 0 ≤ j ≤ n-1, else 0

Since j is always in [0, n-1], the condition is just 0 ≤ k-j ≤ n-1, i.e., k-(n-1) ≤ j ≤ k.

So:
for j in range(max(0, k-(n-1)), min(n-1, k)+1):
    grad[j] = 1 - 2*h[k-j]

And grad[j] = 0 for all other j.

This is O(n) to compute, great!

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def full_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_constraints(h, n):
    """Project h onto {0 <= x_i <= 1, sum(x) = n/2} using bisection"""
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    lo = -max(1.0, float(np.max(h))) - 0.1
    hi = max(1.0, float(np.max(h))) + 0.1 - np.max(h)
    # Actually, the shift mu can be in [-1, 1] since h is in [0,1]
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    mu = (lo + hi) / 2.0
    return np.clip(h + mu, 0.0, 1.0)

def gradient_at_lag(h, k):
    """Gradient of c[k] = correlate(h, 1-h, 'full')[k] w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    j_lo = max(0, k - (n - 1))
    j_hi = min(n - 1, k)
    for j in range(j_lo, j_hi + 1):
        grad[j] = 1.0 - 2.0 * h[k - j]
    return grad

def subgradient_descent(h, n, n_iters, step_scale, time_limit):

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5_value(h)
    t0 = time.time()

        k_star = np.argmax(corr)
        c5 = corr[k_star] * dx
        if c5 < best_c5:

            best_h = h.copy()

        step = step_scale / (1.0 + 0.01 * t)
        h_new = h - step * dx * grad
        h = project_constraints(h_new, n)
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_constraints(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    lo = -1.0
    hi = 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def get_gradient(h, M):
    """Compute gradient of smooth approximation to max(c*k*dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = M * corr * dx
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    weights = exp_x / exp_x.sum()
    
    grad = np.zeros(n, dtype=np.float64)
    # For each lag k, add weights[k] * grad_k
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        # grad_k[j] = 1 - 2*h[k-j] for j in [j_lo, j_hi]
        indices = np.arange(j_lo, j_hi + 1)
        h_indices = k - indices
        grad[indices] += w * (1.0 - 2.0 * h[h_indices])
    return grad * dx

def smooth_obj(h, M):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    x = M * corr * dx
    x_max = x.max()
    return float((x_max + np.log(np.sum(np.exp(x - x_max)))) / M)

    h = h.copy()
    dx = 2.0 / n

    t0 = time.time()
    
    step = 0.5
    patience = 0
    for it in range(50000):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        val = corr[k_star] * dx
        if val < best_val:
            best_val = val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 500:
            step *= 0.8
            patience = 0
            if step < 1e-6:
                break

        j_lo = max(0, k_star - (n - 1))
        j_hi = min(n - 1, k_star)
        grad = np.zeros(n)
        indices = np.arange(j_lo, j_hi + 1)
        grad[indices] = 1.0 - 2.0 * h[k_star - indices]
        grad *= dx
        
        h = project_constraints(h - step * grad, n)
    
    return best_h, best_val

            if time.time() - t0 > tl:

            g = _sg(h, M)
            h = _proj(h - lr * g, n)
        v = _c5(h)
        if v < bv:
            bv = v
            best = h.copy()
    return best, bv


def _pol(h, n, tl):
    h = h.copy()
    dx = 2.0 / n
    best = h.copy()
    bv = _c5(h)
    t0 = time.time()
    lr = 0.35
    stall = 0
    nm1 = n - 1
    while time.time() - t0 < tl:
        corr = np.correlate(h, 1.0 - h, mode='full')
        ks = int(np.argmax(corr))
        v = corr[ks] * dx
        if v < bv - 1e-14:
            bv = v
            best = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 1500:
            lr *= 0.7
            stall = 0
            if lr < 1e-6:
                break
        jlo = max(0, ks - nm1)
        jhi = min(nm1, ks)
        idx = np.arange(jlo, jhi + 1)
        g = np.zeros(n)
        g[idx] = (1.0 - 2.0 * h[ks - idx]) * dx
        h = _proj(h - lr * g, n)
    return best, bv


def _topk_pol(h, n, tl, topk=5):
    h = h.copy()
    dx = 2.0 / n
    best = h.copy()
    bv = _c5(h)
    t0 = time.time()
    lr = 0.25
    stall = 0
    nm1 = n - 1
    while time.time() - t0 < tl:
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-topk:]
        v = corr.max() * dx
        if v < bv - 1e-14:
            bv = v
            best = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 1000:
            lr *= 0.7
            stall = 0
            if lr < 1e-6:
                break
        g = np.zeros(n)
        for ks in top_idx:
            jlo = max(0, ks - nm1)
            jhi = min(nm1, ks)
            idx = np.arange(jlo, jhi + 1)
            g[idx] += (1.0 - 2.0 * h[ks - idx])
        g *= dx / topk
        h = _proj(h - lr * g, n)
    return best, bv


def run(seed=42, budget_s=1300, **kwargs):
    tg = time.time()
    rng = np.random.default_rng(seed)

    h0 = np.asarray(initial_h_values, dtype=np.float64)
    n = len(h0)
    best_h = h0.copy()
    best_c5 = _c5(h0)

    # Phase 1: smooth gradient (30%)
    t1 = budget_s * 0.30
    h1, v1 = _smooth(h0, n, t1, 3.0, 50.0, 0.45)
    if v1 < best_c5:
        best_c5 = v1
        best_h = h1

    # Phase 2: non-smooth polish (15%)
    el = time.time() - tg
    t2 = min(budget_s * 0.15, budget_s - el - 20)
    if t2 > 5:
        h2, v2 = _pol(best_h, n, t2)
        if v2 < best_c5:
            best_c5 = v2
            best_h = h2

    # Phase 3: topk polish (10%)
    el = time.time() - tg
    t3 = min(budget_s * 0.10, budget_s - el - 20)
    if t3 > 5:
        h3, v3 = _topk_pol(best_h, n, t3, topk=4)
        if v3 < best_c5:
            best_c5 = v3
            best_h = h3

    # Phase 4: random restarts (25%)
    el = time.time() - tg
    t4 = min(budget_s * 0.25, budget_s - el - 60)
    if t4 > 30:
        nr = 8
        pr = t4 / nr
        for r in range(nr):
            if time.time() - tg > budget_s - 80:
                break
            ns = 0.08 + 0.06 * (r % 4)
            h_r = _proj(best_h + rng.standard_normal(n) * ns, n)
            h_r, v_r = _smooth(h_r, n, pr * 0.5, 4.0, 35.0, 0.35)
            h_r, v_r2 = _pol(h_r, n, pr * 0.3)
            v_rr = min(v_r, v_r2)
            h_cand = h_r if v_r <= v_r2 else best_h
            # Re-evaluate both candidates
            v_a = _c5(h_r)
            if v_a < best_c5:
                best_c5 = v_a
                best_h = h_r

    # Phase 5: higher resolution (remaining)
    el = time.time() - tg
    t5 = budget_s - el - 25
    if t5 > 80:
        n2 = n * 2
        fl = np.arange(n2) * n / n2
        ix = np.clip(np.floor(fl).astype(int), 0, n - 1)
        h5 = best_h[ix].copy()
        h5 = _proj(h5, n2)
        t5a = t5 * 0.50
        h5, v5a = _smooth(h5, n2, t5a, 3.0, 30.0, 0.30)
        t5b = t5 * 0.25
        h5, v5b = _pol(h5, n2, t5b)
        v5 = _c5(h5)
        if v5 < best_c5:
            best_c5 = v5
            best_h = h5
            n = n2

    # Phase 6: final polish at best resolution
    el = time.time() - tg
    t6 = min(budget_s * 0.08, budget_s - el - 5)
    if t6 > 5:
        nf = len(best_h)
        fin, vf = _topk_pol(best_h, nf, t6, topk=3)
        if vf < best_c5:
            best_c5 = vf
            best_h = fin
            n = nf

    # Final projection
    n_final = len(best_h)
    best_h = _proj(best_h, n_final)
    final_c5 = _c5(best_h)

    # Try to verify with evaluator
    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None:
            if isinstance(ev, (int, float, np.floating, np.integer)):
                final_c5 = float(ev)
            elif isinstance(ev, tuple) and len(ev) >= 1:
                final_c5 = float(ev[0])
    except Exception:
        pass

    return best_h, final_c5, n_final
