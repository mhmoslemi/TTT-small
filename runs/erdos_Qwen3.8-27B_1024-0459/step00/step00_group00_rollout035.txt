def smooth_c5(h, temp):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    s = (corr - m) / temp
    s = np.clip(s, -50.0, 50.0)
    e = np.exp(s)
    return m + temp * float(np.log(e.sum()))

def make_feasible(h, n):
    target = 0.5 * n
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room < 1e-14:
                break
            h += d * room / total_room
        else:
            room = h.copy()
            total_room = room.sum()
            if total_room < 1e-14:
                break
            h += d * room / total_room
        h = np.clip(h, 0.0, 1.0)
    return h
