from scipy import optimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, n):
    """Project h to satisfy: sum = n/2, 0 <= h <= 1"""
    h = np.clip(h, 0, 1)
    # Adjust sum
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-12:
        # Distribute adjustment
        diff = target_sum - current_sum
        if diff > 0:
            # Need to increase some values
            mask = h < 1.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask) * 0.999
        else:
            # Need to decrease some values
            mask = h > 0.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask) * 0.999
        h = np.clip(h, 0, 1)
        # Final adjustment
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Binary search for scaling
            # Or distribute to free variables
            mask = (h > 1e-6) & (h < 1 - 1e-6)
            if np.sum(mask) > 0:
                h[mask] += (target_sum - current_sum) / np.sum(mask)
                h = np.clip(h, 0, 1)
    return h

def upsamples_linear(h_old, n_new):
    """Upsample h_old to n_new points using linear interpolation"""
    n_old = len(h_old)
    # Sample positions in the old array
    positions_old = np.linspace(0, 1, n_old, endpoint=False)
    positions_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(positions_new, positions_old, np.concatenate([h_old, [h_old[0]]]))
    return project(h_new, n_new)

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

def upsample_linear(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    x_ext = np.append(x_old, 1.0)
    h_ext = np.append(h_old, h_old[0])
    h_new = np.interp(x_new, x_ext, h_ext)
    return project_feasible(h_new, n_new)

def compute_gradient(h, n, k_all, n_top):
    """Compute combined gradient from top n_top lags"""
    g = np.zeros(n)
    for k_idx in k_all[:n_top]:
        lag = k_idx - (n - 1)
        # Part 1: h[j] * (1-h[j+lag])
        j_lo = max(0, -lag)
        j_hi = min(n-1, n-1-lag)
        if j_lo <= j_hi:
            g[j_lo:j_hi+1] += (1.0 - h[j_lo+lag:j_hi+1+lag])
        # Part 2: h[j-lag] * (1-h[j])
        j_lo2 = max(0, lag)
        j_hi2 = min(n-1, n-1+lag)
        if j_lo2 <= j_hi2:
            g[j_lo2:j_hi2+1] -= h[j_lo2-lag:j_hi2+1-lag]
    g /= max(n_top, 1)
    g -= np.mean(g)
    return g
