def project_to_constraint(h, N):

        excess = s - N / 2.0

            h[i] -= reduce

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def lse_obj_and_grad(h, alpha, lam):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500, 500)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = cmax + np.log(sum_exp) / alpha
    cv = np.sum(h) - N / 2.0
    obj += lam * cv * cv
    
    weights = exp_s / sum_exp
    grad = np.zeros(N)
    
    for k in range(2 * N - 1):
        d = k - (N - 1)
        w = weights[k] * dx
        if d >= 0:
            idx1 = np.arange(d, N)
            grad[idx1] += w * (1.0 - h[idx1 - d])
            idx2 = np.arange(0, N - d)
            grad[idx2] += w * (-h[idx2 + d])
        else:
            s = -d
            idx1 = np.arange(0, N - s)
            grad[idx1] += w * (1.0 - h[idx1 + s])
            idx2 = np.arange(s, N)
            grad[idx2] += w * (-h[idx2 - s])
    
    grad += 2.0 * lam * cv
    return obj, grad

def project_h(h, N):
    h = np.clip(h, 0.0, 1.0)

        excess = s - target
        for i in order:
            red = min(h[i], excess)
            h[i] -= red
            excess -= red
            if excess <= 1e-12:
                break
    else:
        order = np.argsort(h)
        deficit = target - s
        for i in order:
            inc = min(1.0 - h[i], deficit)
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-12:

    best = h.copy()

        for i in range(N):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def lse_obj_and_grad(h, alpha, lam):
    N = len(h)
    dx = 2.0 / N
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -400, 400)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = cmax + np.log(sum_exp) / alpha
    cv = np.sum(h) - N / 2.0
    obj += lam * cv * cv
    weights = exp_s / sum_exp
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        d = k - (N - 1)
        w = weights[k] * dx
        if d >= 0:
            if d < N:
                idx1 = np.arange(d, N)
                grad[idx1] += w * (1.0 - h[idx1 - d])
                idx2 = np.arange(0, N - d)
                grad[idx2] += w * (-h[idx2 + d])
        else:
            s = -d
            if s < N:
                idx1 = np.arange(0, N - s)
                grad[idx1] += w * (1.0 - h[idx1 + s])
                idx2 = np.arange(s, N)
                grad[idx2] += w * (-h[idx2 - s])
    grad += 2.0 * lam * cv
    return obj, grad

def project_h(h, N):
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        order = np.argsort(h)[::-1]
        excess = s - target
        for i in order:
            red = min(h[i], excess)
            h[i] -= red
            excess -= red
            if excess <= 1e-14:
                break
    else:
        order = np.argsort(h)
        deficit = target - s
        for i in order:
            inc = min(1.0 - h[i], deficit)
            h[i] += inc
            deficit -= inc
            if deficit <= 1e-14:
                break
    return h

def make_bounds(N):
    return [(0.0, 1.0)] * N

def slsqp_constraint_val(x, N):
    return np.sum(x) - N / 2.0

def slsqp_constraint_grad(x, N):
    return np.ones(N)

def slsqp_jac(x, N):
    h = np.clip(x, 0, 1)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    kmax = np.argmax(corr)
    d = kmax - (N - 1)
    g = np.zeros(N)
    if d >= 0:
        if d < N:
            g[d:] += (1.0 - h[:N-d]) * dx / N
            g[:N-d] += (-h[d:]) * dx / N
    else:
        s = -d
        if s < N:
            g[:N-s] += (1.0 - h[s:]) * dx / N
            g[s:] += (-h[:N-s]) * dx / N
    return g

def optimize_lbfgs(h0, alpha, lam, max_iter=300):
    N = len(h0)
    bounds = make_bounds(N)

    def obj_fn(x):
        o, _ = lse_obj_and_grad(x, alpha, lam)
        return o

    def grad_fn(x):
        _, g = lse_obj_and_grad(x, alpha, lam)
        return g

    result = minimize(obj_fn, h0, method='L-BFGS-B', jac=grad_fn,
                      bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
    return result.x

def optimize_slsqp(h0, N, max_iter=100):
    bounds = make_bounds(N)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - N / 2.0, 'jac': lambda x: np.ones(N)}

    def obj_fn(x):
        return compute_c5(x)

    result = minimize(obj_fn, h0, method='SLSQP', bounds=bounds,
                      constraints=[cons],
                      options={'maxiter': max_iter, 'ftol': 1e-12})
    return result.x

def local_search_binary(h, N, max_iters=2000):
    """Iteratively flip bits to reduce C5 while maintaining N/2 ones"""
    best = h.copy()
    best_c5 = compute_c5(best)
    ones = np.where(best >= 0.5)[0]
    zeros = np.where(best < 0.5)[0]
    for it in range(max_iters):
        improved = False
        if len(ones) == 0 or len(zeros) == 0:
            break
        idx_o = np.random.choice(len(ones))
        idx_z = np.random.choice(len(zeros))
        o = ones[idx_o]
        z = zeros[idx_z]
        trial = best.copy()
        trial[o] = 0.0
        trial[z] = 1.0
        c5_trial = compute_c5(trial)
        if c5_trial < best_c5 - 1e-12:
            best = trial
            best_c5 = c5_trial
            ones[idx_o] = zeros[idx_z]
            zeros[idx_z] = o
            improved = True
    return best, best_c5
