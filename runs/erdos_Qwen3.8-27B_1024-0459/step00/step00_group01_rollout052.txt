def project_onto_simplex_box(y, target_sum):

    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)

    n = len(h)

import numpy as np
import time

def project_onto_simplex_box(y, target_sum):
    """Project y onto {h : 0 <= h[i] <= 1, sum(h) = target_sum}."""
    n = len(y)
    lo = float(y.min()) - target_sum / n - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = float(h.sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5_value(h):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_gradient_at_shift(h, d):
    """Gradient of C(d)*dx w.r.t. h where d is shift in grid units."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if d >= 0:
        js = np.arange(n - d)
        grad[js] += (1.0 - h[js + d])
        valid = js >= d
        grad[js[valid]] -= h[js[valid] - d]
    else:
        dp = -d
        js = np.arange(dp, n)
        grad[js] += (1.0 - h[js - dp])
        valid = js + dp < n
        grad[js[valid]] -= h[js[valid] + dp]
    return grad * dx

def compute_soft_max_gradient(h, eps, top_k):
    """Compute smoothed gradient using top-k softmax weighting of shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k = min(top_k, len(corr))
    top_idx = np.argsort(corr)[-k:]
    w = np.exp(corr[top_idx] / eps - np.max(corr[top_idx] / eps))
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(k):
        d_shift = int(top_idx[i]) - (n - 1)
        grad += w[i] * compute_gradient_at_shift(h, d_shift)
    return grad

def optimize_c5_smoothed(h_init, time_limit=60.0, noise_rate=0.02):
    """Smoothed subgradient descent to minimize C5."""
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy().astype(float)
    h = project_onto_simplex_box(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5_value(h)
    t0 = time.time()
    alpha = 0.4
    min_alpha = 0.003
    iteration = 0
    stall_count = 0
    while time.time() - t0 < time_limit:
        progress = min(1.0, iteration / 3000.0)
        eps = 0.06 * (1.0 - 0.9 * progress) + 0.003
        top_k = min(25, 2 * n - 1)
        g = compute_soft_max_gradient(h, eps, top_k)
        h_new = h - alpha * g
        if noise_rate > 0 and iteration % 50 == 0:
            h_new += noise_rate * (np.random.rand(n) - 0.5)
        h_new = project_onto_simplex_box(h_new, target_sum)
        h = h_new
        alpha = max(0.4 / (1.0 + 0.003 * iteration), min_alpha)
        c5 = compute_c5_value(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 500 and noise_rate < 0.1:
            noise_rate *= 2.0
            stall_count = 0
        iteration += 1
    return best_h, best_c5

def random_binary_string(n, seed):
    """Random binary string with exactly n//2 ones."""
    rng = np.random.RandomState(seed)
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return project_onto_simplex_box(h, n / 2.0)

def random_block_string(n, seed):
    """Random block-structured starting point."""
    rng = np.random.RandomState(seed)
    n_blocks = rng.randint(3, max(4, n // 15 + 1))
    boundaries = np.sort(rng.choice(np.arange(1, n), n_blocks - 1, replace=False))
    boundaries = np.concatenate([[0], boundaries, [n]])
    h = np.zeros(n)
    for i in range(n_blocks):
        if rng.random() > 0.4:
            h[boundaries[i]:boundaries[i+1]] = 1.0
    return project_onto_simplex_box(h, n / 2.0)

def golden_ratio_string(n, seed):
    """Starting point based on golden ratio fractional parts."""
    rng = np.random.RandomState(seed)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    offset = rng.random()
    centers = np.linspace(0, n, n, endpoint=False) + 0.5
    vals = np.abs(centers / n * phi + offset - np.round(centers / n * phi + offset))
    h = (vals < 0.5).astype(float)
    return project_onto_simplex_box(h, n / 2.0)

def interpolate_to_n(h_old, n_new):
    """Linearly interpolate and project to new resolution."""
    n_old = len(h_old)
    x_old = np.arange(n_old) * (2.0 / n_old) + (1.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new) + (1.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_onto_simplex_box(h_new, n_new / 2.0)

def try_binary_refinement(h, time_limit=30.0):
    """Try to improve by binarizing and local swaps."""
    n = len(h)
    target_sum = n / 2.0
    h_bin = (h >= 0.5).astype(float)
    if h_bin.sum() > target_sum:
        above = np.where(h_bin == 1.0)[0]
        _, order = np.sort(h[above]), np.argsort(h[above])
        n_remove = int(h_bin.sum()) - int(target_sum)
        h_bin[above[order[-n_remove:]]] = 0.0
    elif h_bin.sum() < target_sum:
        below = np.where(h_bin == 0.0)[0]
        _, order = np.sort(h[below]), np.argsort(h[below])
        n_add = int(target_sum) - int(h_bin.sum())
        h_bin[below[order[:n_add]]] = 1.0
    h_bin = project_onto_simplex_box(h_bin, target_sum)
    c5_bin = compute_c5_value(h_bin)
    c5_cont = compute_c5_value(h)
    return (h_bin, c5_bin) if c5_bin < c5_cont else (h, c5_cont)
