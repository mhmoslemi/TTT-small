def _slsqp_refine(h0, n, maxiter=100):
    """SLSQP continuous refinement."""
    dx = 2.0 / n
    target_sum = n / 2.0

    def objective(h):
        return _c5(h, dx)

    def objective_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        k_lag = k_max - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j - k_lag < n:
                grad[j] += (1.0 - h[j - k_lag])
            if 0 <= j + k_lag < n:
                grad[j] += (-h[j + k_lag])
        return grad * dx

    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
                   'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(objective, h0, jac=objective_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < _c5(h0, dx):
            h_res = np.clip(res.x, 0, 1)
            # Fix sum
            diff = target_sum - np.sum(h_res)
            h_res += diff / n
            h_res = np.clip(h_res, 0, 1)
            # Re-fix sum iteratively
            for _ in range(50):
                diff = target_sum - np.sum(h_res)
                if abs(diff) < 1e-12:

                mask = (h_res > 0.001) & (h_res < 0.999)

                    h_res[mask] += diff / np.sum(mask)
                else:
                    h_res += diff / n
                    h_res = np.clip(h_res, 0, 1)
            c5_res = _c5(h_res, dx)
