import time

def make_ramp(n):
    return np.linspace(1.0, 0.0, n)

def make_triangle(n):
    mid = (n-1)/2.0
    return np.maximum(0, 1 - np.abs(np.arange(n) - mid) / mid)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def satisfy_sum(h, n):
    """Adjust h to satisfy sum constraint while staying in [0,1]"""
    target = n / 2.0
    current = np.sum(h)
    if abs(current - target) < 1e-10:
        return h
    diff = target - current
    # Redistribute proportionally
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # If still off (due to clipping), iterate
    for _ in range(100):
        current = np.sum(h)
        diff = target - current
        if abs(diff) < 1e-12:
            break
        # Adjust non-saturated variables
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def satisfy_sum(h, n):

        if nmask == 0:

        h = h.copy()
        h[mask] += diff / nmask
        h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

        return j

        t = x[-1]

        return t - dx * corr

        {'type': 'ineq', 'fun': ineq_fun}
