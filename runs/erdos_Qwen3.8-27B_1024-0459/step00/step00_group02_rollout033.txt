    new_x = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

    n = len(h)
    dx = 2.0 / n
    if n == 1:
        return abs(h[0] * (1 - h[0])) * dx
    corr = np.correlate(h, 1 - h, mode='full')

    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Binary search for lambda
    lo, hi = -target, target
    for _ in range(100):
        mid = (lo + hi) / 2
        h_try = np.clip(h + mid, 0, 1)
        s_try = np.sum(h_try)
        if s_try > target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fine adjustment
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute evenly
        mask = (h_proj > 0) & (h_proj < 1)
        if np.any(mask):
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0, 1)

    n = len(h)

def upsample(h, new_n):

def project(h, n):

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -target - 1, target + 1

def smoothed_objective_and_grad(h, epsilon=0.01):
    """Compute smoothed max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = dx * corr  # g[k] = dx * corr[k]
    
    # Softmax with scaling
    g_max = g.max()
    exp_g = np.exp((g - g_max) / epsilon)
    Z = exp_g.sum()
    W = exp_g / Z  # softmax weights
    
    # Smoothed objective
    F = g_max + epsilon * np.log(Z)
    
    # Gradient: sum_k W[k] * grad_k
    grad = np.zeros(n)
    # For each shift, add its weighted gradient
    # To be efficient, only process shifts with significant weight
    threshold = 1.0 / (2 * n - 1) * 2  # 2/(2n-1)
    active = W > threshold
    
    for k_idx in np.where(active)[0]:
        w = W[k_idx]
        if w < 1e-6:
            continue
        s = k_idx - (n - 1)
        if s >= 0:
            grad[:n-s] -= w * dx * h[s:]
            grad[s:] += w * dx * (1.0 - h[:n-s])
        else:
            t = -s
            grad[t:] -= w * dx * h[:n-t]
            grad[:n-t] += w * dx * (1.0 - h[t:])
    
    return F, grad

import numpy as np
import time
from scipy.optimize import minimize

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * corr.max())

def project_constraint(h, n):
    target = 0.5 * n
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    lo = -target - 1.0
    hi = target + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        h_try = np.clip(h + mid, 0.0, 1.0)
        if h_try.sum() > target:
            hi = mid
        else:
            lo = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(h + lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = dx * float(corr.max())
    k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(corr)[-k:][::-1]
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] -= dx * h[s:]
            grad[s:] += dx * (1.0 - h[:n - s])
        else:
            t = -s
            grad[t:] -= dx * h[:n - t]
            grad[:n - t] += dx * (1.0 - h[t:])
    return grad / float(k), val

def smoothed_obj_and_grad(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = dx * corr
    g_max = float(g.max())
    exp_g = np.exp((g - g_max) / epsilon)
    Z = float(exp_g.sum())
    W = exp_g / Z
    F = g_max + epsilon * np.log(Z)
    grad = np.zeros(n)
    active_thresh = 1.0 / (2.0 * n)
    active = W > active_thresh
    active_idx = np.where(active)[0]
    for ki in active_idx:
        w = W[ki]
        if w < 1e-7:
            continue
        s = ki - (n - 1)
        if s >= 0:
            grad[:n - s] -= w * dx * h[s:]
            grad[s:] += w * dx * (1.0 - h[:n - s])
        else:
            t = -s
            grad[t:] -= w * dx * h[:n - t]
            grad[:n - t] += w * dx * (1.0 - h[t:])
    return F, grad

def optimize_subgradient(h_in, time_limit, rng, n, lr_init=0.01):
    h = h_in.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = lr_init
    momentum = np.zeros(n)
    beta = 0.85
    start = time.time()
    it = 0
    stalled = 0
    while time.time() - start < time_limit:
        grad, val = subgradient(h, top_k=5)
        it += 1
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1
        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h = project_constraint(h_new, n)
        if it % 300 == 0:
            if stalled > 200:
                lr *= 0.5
                if lr < 5e-5:
                    h = best_h + 0.02 * rng.standard_normal(n)
                    h = project_constraint(h, n)
                    lr = lr_init
                    momentum[:] = 0.0
                    stalled = 0
            else:
                lr = min(lr * 1.02, 0.06)
        if it % 2000 == 0:
            h = 0.6 * h + 0.4 * best_h
            h = project_constraint(h, n)
    final_val = compute_c5(h)
    if final_val < best_val:
        best_val = final_val
        best_h = h.copy()
    return best_h, best_val

def optimize_slsqp(h_in, time_limit, rng, n):
    target = 0.5 * n
    h0 = h_in.copy()
    epsilon = 0.015
    def obj(x):
        F, _ = smoothed_obj_and_grad(x, epsilon)
        return F
    def grad_obj(x):
        _, g = smoothed_obj_and_grad(x, epsilon)
        return g
    def eq_con(x):
        return x.sum() - target
    constraints = [{'type': 'eq', 'fun': eq_con}]
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    res = minimize(obj, h0, jac=grad_obj, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 500, 'ftol': 1e-8, 'disp': False})
    h_opt = project_constraint(res.x, n)
    val_opt = compute_c5(h_opt)
    elapsed = time.time() - t0
    return h_opt, val_opt, elapsed

def random_block_init(n, rng):
    h = np.zeros(n)
    pos = 0
    num_ones = 0
    while pos < n:
        blen = rng.randint(2, max(3, n // 8))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
            num_ones += blen
        pos += blen
    h = project_constraint(h, n)
    return h
