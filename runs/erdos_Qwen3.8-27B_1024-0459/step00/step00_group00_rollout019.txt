import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h, dx):
    """Compute C5 value for a given h."""
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    """Project h onto {0 <= h[i] <= 1, sum(h) = target_sum}."""
    # Find lambda such that sum(clamp(h[i] - lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        clamped = np.clip(h - mid, 0, 1)
        s = np.sum(clamped)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0, 1)

def subgradient_c5(h, dx):
    """Compute subgradient of C5 w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)
    # Find all k that achieve the maximum (within tolerance)
    active = np.abs(corr - max_val) < 1e-10

    n = len(h)
