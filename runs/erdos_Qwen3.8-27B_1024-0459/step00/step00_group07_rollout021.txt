import numpy as np
import scipy.optimize as so
import time
import math

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def objective_smooth(h, temperature=0.01):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Smooth max via log-sum-exp
    c_normalized = c / temperature
    c_max = np.max(c_normalized)
    if c_max > 500:
        return np.exp(c_max - 500) * temperature  # prevent overflow
    val = temperature * np.log(np.sum(np.exp(c_normalized - c_max)))
    return val

def objective_hard(h):
    return evaluate_c5(h)

def constraints_setup(n):
    def sum_constraint(h):
        return np.sum(h) - n / 2.0
    return [(sum_constraint, 'eq')]

def optimize_slsqp(h0, n, maxiter=200, ftol=1e-7):
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0, 'jac': lambda h: np.ones(n)}]
    
    def obj(h):
        return objective_smooth(h, temperature=0.005)
    
    res = so.minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
    return res.x, res.fun

def coordinate_descent(h, n, n_iter=10000):
    """Local search: try changing each h[i] to improve the objective."""
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    h_current = h.copy()
    
    for it in range(n_iter):
        improved = False
        for i in range(n):
            # Try small perturbations
            for delta in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:

                h_new[i] = np.clip(h_current[i] + delta, 0, 1)
                # Fix constraint: adjust another point

                if abs(diff) < 1e-10:

                        h_current = h_new
                        best_c5 = evaluate_c5(h_current)
                        improved = True

                else:

                    if diff > 0:
                        # Need to increase sum, find j with room
                        j = np.argmin(h_new)
                    else:
                        j = np.argmax(h_new)
                    h_new[j] = np.clip(h_new[j] + diff, 0, 1)
                    # Might not be exact, but approximate
                    if abs(np.sum(h_new) - n/2.0) < 1e-6:
                        c = evaluate_c5(h_new)
                        if c < best_c5:

                            improved = True
                            break

            h_sol = do_slsqp(h_up, n_t, T, maxiter=150, ftol=1e-8)
            c5 = evaluate_c5(h_sol)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sol.copy()
                h_up = best_h.copy()

    # Phase 5: Random restarts for diversity
    if time_ok(0.96):
        for trial in range(4):
            if not time_ok(0.96 - 0.01 * trial):
                break
            n_t = 160
            h_rand = np.random.uniform(0.0, 1.0, n_t)
            h_rand = project_to_constraint(h_rand)
            h_sol = do_slsqp(h_rand, n_t, 0.01, maxiter=150, ftol=1e-8)
            c5 = evaluate_c5(h_sol)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sol.copy()

    # Phase 6: Coordinate descent polish
    remaining = budget_s - (time.time() - t_start)
    if remaining > 25:
        polish_time = min(remaining - 8, 45)
        best_h = coordinate_refine(best_h, polish_time)
        best_c5 = evaluate_c5(best_h)

    # Final validation
    best_h = project_to_constraint(best_h)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
