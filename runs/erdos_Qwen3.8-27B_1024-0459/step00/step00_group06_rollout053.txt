def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    N = len(h)
    # Projection onto {x: 0<=x_i<=1, sum(x)=target_sum}
    # x_i = clip(h_i - mu, 0, 1)
    # Find mu such that sum(x) = target_sum
    
    def sum_at(mu):
        return np.sum(np.clip(h - mu, 0.0, 1.0))
    
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = sum_at(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    mu = (lo + hi) / 2.0
    return np.clip(h - mu, 0.0, 1.0)

def compute_gradient_at_max(h, N, dx):
    """Compute the subgradient of max_k corr_k at h"""
    corr = np.correlate(h, 1 - h, mode="full")
    max_val = np.max(corr)
    max_indices = np.where(np.isclose(corr, max_val, atol=1e-10))[0]
    
    # Average gradient over all max indices (subdifferential)
    grad = np.zeros(N)
    for k in max_indices:
        shift = k - (N - 1)
        # ∂result[k]/∂h[i] = (1-h[i-shift])*1[0≤i-shift<N] - h[i+shift]*1[0≤i+shift<N]
        # For i in 0..N-1:
        valid_m = (i - shift >= 0) & (i - shift < N)  # where m = i-shift is valid
        valid_p = (i + shift >= 0) & (i + shift < N)  # where i+shift is valid
        
        # Actually for each i in 0..N-1:
        # Term 1: (1-h[i-shift]) if 0 <= i-shift < N
        # Term 2: -h[i+shift] if 0 <= i+shift < N
        
        i_arr = np.arange(N)
        term1_idx = i_arr - shift
        term2_idx = i_arr + shift
        
        term1 = np.where((term1_idx >= 0) & (term1_idx < N), 1 - h[term1_idx], 0.0)
        term2 = np.where((term2_idx >= 0) & (term2_idx < N), -h[term2_idx], 0.0)
        
        grad += term1 + term2
    
    grad /= len(max_indices)
    return grad, max_val * dx

def optimize_h(h, N, dx, target_sum, max_time, seed):

    rng = np.random.RandomState(seed)
    start_time = time.time()

    h = project_onto_constraint(h, target_sum)
    
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full")

    # Adaptive step size

    lr_min = 1e-6
    lr_decay = 0.999
    
    for it in range(500000):
        if time.time() - start_time > max_time:
            break

        grad, current_c5 = compute_gradient_at_max(h, N, dx)

        h_new = h - lr * grad
        
        # Project
        h_new = project_onto_constraint(h_new, target_sum)
        
        # Evaluate
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        c5_new = np.max(corr_new) * dx

            best_c5 = c5_new
            best_h = h_new.copy()

        lr = max(lr * lr_decay, lr_min)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, corr

def project_onto_constraint(h, target_sum):
    N = len(h)
    def sum_at(mu):
        return np.sum(np.clip(h - mu, 0.0, 1.0))
    
    # Find mu such that sum(clip(h - mu, 0, 1)) = target_sum
    # sum decreases as mu increases
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if sum_at(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    
    mu = (lo + hi) * 0.5
    return np.clip(h - mu, 0.0, 1.0)

def gradient_of_corr(h, k, N):
    """Compute ∂corr[k]/∂h for a specific shift k"""
    shift = k - (N - 1)
    i_arr = np.arange(N)
    
    # term1: (1-h[i+shift]) if 0 <= i+shift < N
    idx1 = i_arr + shift
    term1 = np.where((idx1 >= 0) & (idx1 < N), 1.0 - h[idx1], 0.0)
    
    # term2: -h[i-shift] if 0 <= i-shift < N
    idx2 = i_arr - shift
    term2 = np.where((idx2 >= 0) & (idx2 < N), -h[idx2], 0.0)
    
    return term1 + term2

    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    tau = tau_init
    lr = 0.005
    no_improve_count = 0
    patience = 2000
    it = 0
    
    while time.time() - start_time < max_time:
