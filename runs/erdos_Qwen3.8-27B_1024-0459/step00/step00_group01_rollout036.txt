```python
    m = np.max(O)
```

```python
import numpy as np
import time

def compute_all_O(h):
    """Compute all overlap values O_k for the given h."""
    n = len(h)
    dx = 2.0 / n
    # Full correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(h):
    """Compute C5 = max_k O_k(h)."""
    return np.max(compute_all_O(h))

def project_to_feasible(x, n):
    """Project x onto {h: 0<=h_i<=1, sum(h_i)=n/2}."""
    target = n / 2.0
    # Bisection on lambda
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Correct for any numerical error
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def grad_O_j(j, h, n):
    """Compute gradient of O_j with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if j >= 0:
        # (1-h[m-j]) for j <= m < n
        if j < n:
            grad[j:] += (1.0 - h[:n-j])
        # -h[m+j] for 0 <= m <= n-1-j
        if j > 0:
            grad[:n-j] -= h[j:]
    else:
        t = -j
        # (1-h[m+t]) for 0 <= m <= n-1-t
        if t < n:
            grad[:n-t] += (1.0 - h[t:])
        # -h[m-t] for t <= m < n
        if t > 0:
            grad[t:] -= h[:n-t]
    return grad * dx

def soft_max_grad(h, alpha):
    """Compute gradient of soft-max of O_k values."""
    n = len(h)
    O = compute_all_O(h)
    # Softmax weights
    m = np.max(O)
    e = np.exp(alpha * (O - m))
    p = e / np.sum(e)
    
    # Only compute gradient for significant p_k
    grad = np.zeros(n)
    significant = np.where(p > 1e-8)[0]
    for k in significant:
        j = k - (n - 1)  # Convert correlation index to lag
        grad += p[k] * grad_O_j(j, h, n)
    return grad

def optimize_h(h0, max_iter=500, alpha_start=10.0, alpha_end=50.0, lr_start=1.0, lr_end=0.1):
    """Optimize h using soft-max gradient descent with projection."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        # Anneal alpha and learning rate
        frac = it / max_iter
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac
        
        # Compute gradient
        g = soft_max_grad(h, alpha)
        
        # Gradient step
        h_new = h - lr * g
        
        # Project onto feasible set
        h = project_to_feasible(h_new, n)
        
        # Update best
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
```
