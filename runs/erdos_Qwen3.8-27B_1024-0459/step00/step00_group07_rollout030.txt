- If j = i for some valid i: contribution is -h[i+k] = -h[j+k] (provided j+k is valid, i.e., 0≤j+k<n, and j is valid as i, i.e., 0≤j<n, and i=j gives i+k=j+k which must be valid)
- If j = i+k for some valid i: contribution is (1-h[i]) = (1-h[j-k]) (provided j-k is valid as i, i.e., 0≤j-k<n, and j is valid as i+k, i.e., 0≤j<n)

So:
d(full[p])/d(h[j]) = -h[j+k]·[0≤j<n AND 0≤j+k<n AND 0≤j<n (trivially true)] + (1-h[j-k])·[0≤j<n AND 0≤j-k<n AND 0≤j<n (trivially true)]

Simplifying (j is always in [0,n) since we're taking derivative w.r.t. h[j]):
= -h[j+k]·[0≤j+k<n] + (1-h[j-k])·[0≤j-k<n]

where k = p-(n-1).

Note: when k=0, both terms are -h[j] + (1-h[j]) = 1-2h[j], which makes sense since full[n-1] = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2, and d/dh[j] = 1 - 2h[j].

Great, so the gradient is:
grad[j] = dx * [ (1-h[j-k])·1{0≤j-k<n} - h[j+k]·1{0≤j+k<n} ]

for the active lag k.

    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    # Second pass for exact sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_active_lag(h, k):
    """Gradient of corr at lag k (k = p - (n-1) where p is the full index)"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # For h[j]: (1-h[j-k]) if 0<=j-k<n, minus h[j+k] if 0<=j+k<n
    if k >= 0:
        # j-k valid: j >= k, j < n
        # j+k valid: j+k < n, j >= 0 (always) → j < n-k
        valid1 = np.arange(k, n)  # j where j-k >= 0
        valid2 = np.arange(0, n-k)  # j where j+k < n
        grad[valid1] += (1.0 - h[valid1 - k])
        grad[valid2] -= h[valid2 + k]
    else:
        kk = -k
        # j-k = j+kk valid: j+kk < n → j < n-kk
        # j+k = j-kk valid: j-kk >= 0 → j >= kk
        valid1 = np.arange(0, n - kk)  # j where j+kk < n
        valid2 = np.arange(kk, n)  # j where j-kk >= 0
        grad[valid1] += (1.0 - h[valid1 + kk])
        grad[valid2] -= h[valid2 - kk]
    return grad * dx

    best_h = h.copy()

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    """Upscale h to n_new points via interpolation"""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)  # periodic interpolation
    target_sum = n_new / 2.0
    h_new = project_feasible(h_new, target_sum)
    return h_new

def gradient_corr(h, k):
    """
    Gradient of (corr at lag k) * dx w.r.t. h.
    k is the actual lag: k = p - (n-1) where p is the full output index.
    """
    n = len(h)
    dx = 2.0 / n

        # corr = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        # d/dh[j]: 
        #   j as i+k: (1-h[i]) = (1-h[j-k]), valid when k<=j<n
        #   j as i: -h[i+k] = -h[j+k], valid when 0<=j<n-k
        if k > 0:

        else:

            # d/dh[j] = (1-h[j]) - h[j] = 1-2h[j]
            grad[:] = 1.0 - 2.0 * h[:]
    else:
        kk = -k
        # corr = sum_{i=kk}^{n-1} h[i-kk]*(1-h[i])
        # Let j = i-kk: corr = sum_{j=0}^{n-1-kk} h[j]*(1-h[j+kk])

        #   j as first: (1-h[j+kk]), valid when j+kk < n → j < n-kk
        #   j as second: -h[j-kk], valid when j-kk >= 0 → j >= kk
        grad[:n-kk] += (1.0 - h[kk:])
        grad[kk:] -= h[:n-kk]
    
    return grad * dx

    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def upscale(h, n_new, rng=None):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    if rng is not None:
        h_new += 0.05 * rng.randn(n_new)
    target_sum = n_new / 2.0
    h_new = project_feasible(h_new, target_sum)
    return h_new

def gradient_corr_single(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h[:]
    elif k > 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        kk = -k
        grad[:n - kk] += (1.0 - h[kk:])
        grad[kk:] -= h[:n - kk]
    return grad * dx

def gradient_top_lags(h, top_lags, n_top=10):
    n = len(h)
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    sorted_idx = np.argsort(-corr_raw)[:top_lags]
    grad = np.zeros(n)
    for p in sorted_idx:
        k = p - (n - 1)
        grad += gradient_corr_single(h, k)
    return grad / len(sorted_idx)

def subgradient_method(h_init, budget_s=30, top_lags=10, seed=0):
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    lr = 0.08
    patience = 0
    max_patience = 150
    momentum = 0.0
    mu = 0.8
    
    while time.time() - t0 < budget_s:
        grad = gradient_top_lags(h, top_lags, n_top)
        momentum = mu * momentum - lr * grad
        h_new = h + momentum
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                lr *= 0.5
                patience = 0
                if lr < 1e-6:
                    break
                # Add perturbation to escape
                h_new = h + 0.02 * rng.randn(n)
                h_new = project_feasible(h_new, target_sum)
                h = h_new
                continue
        
        h = h_new
    
    return best_h, best_c5

def slsqp_optimize(h_init, budget_s=60, top_constraints=200):
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def obj_and_grad(x):
        h = x[:-1]
        t = x[-1]
        corr = np.correlate(h, 1.0 - h, mode='full')
        active = np.argsort(-corr)[:top_constraints]
        # objective: t + penalty for violated constraints
        pen = 0.0
        for k_idx in active:
            val = corr[k_idx] * dx - t
            if val > 0:
                pen += val * val
        obj = t + 100.0 * pen
        
        # gradient
        grad = np.zeros(n + 1)
        grad[-1] = 1.0 + 200.0 * sum(max(0, corr[k_idx]*dx - t) for k_idx in active)
        for k_idx in active:
            val = corr[k_idx] * dx - t
            if val > 0:
                k = k_idx - (n - 1)
                gk = gradient_corr_single(h, k)
                grad[:-1] += 200.0 * val * gk
        return obj, grad
    
    x0 = np.zeros(n + 1)
    x0[:-1] = h_init
    x0[-1] = compute_c5(h_init) + 0.01
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: x[:-1].sum() - target_sum,
         'jac': lambda x: np.concatenate([np.ones(n), [0.0]])}
    ]
    bounds = [(0, 1)] * n + [(0.01, 1.0)]
    
    t0 = time.time()
    try:
        res = minimize(obj_and_grad, x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'ftol': 1e-10})
        if time.time() - t0 < budget_s:
            h_res = project_feasible(res.x[:-1], target_sum)
            c5_res = compute_c5(h_res)
            return h_res, c5_res
    except:
        pass
    return h_init, compute_c5(h_init)

def make_binary(h, n):
    target = n / 2
    k = int(target)
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:k]] = 1.0
    return h_bin

def binary_2opt(h_bin, budget_s=60, seed=42):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    no_improve = 0
    max_no_improve = 300
    eval_count = 0
    
    while time.time() - t0 < budget_s and no_improve < max_no_improve:
        # Mix random and structured moves
        if eval_count % 5 == 0 and len(ones) > 2:
            # Structured: pick a 1, swap with a 0 at a specific distance
            i = ones[rng.randint(len(ones))]
            dist = rng.randint(1, max(2, n // 4))
            j = (i + dist) % n
            if h[j] == 1:
                j = (i - dist) % n
            if h[j] == 1:
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        if i == j or h[i] != 1 or h[j] != 0:
            eval_count += 1
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        eval_count += 1
        
        c5_new = compute_c5(h)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    
    return best_h, best_c5

def block_move_optimize(h_bin, budget_s=60, seed=7):
    """Try moving a block of consecutive 1s to a new position"""
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    no_improve = 0
    while time.time() - t0 < budget_s and no_improve < 200:
        # Find a block of 1s
        positions = np.where(h == 1)[0]
        if len(positions) < 2:
            break
        block_len = min(3, rng.randint(1, 4))
        start = positions[rng.randint(len(positions) - block_len)]
        # Find consecutive 1s starting at start
        block = [start]
        for offset in range(1, block_len):
            if start + offset < n and h[start + offset] == 1:
                block.append(start + offset)
            else:
                break
        if len(block) < 1:
            no_improve += 1
            continue
        if len(block) == 0:
            break
        
        # Choose new position (must be all 0s)
        for _ in range(10):
            new_start = rng.randint(0, n - len(block) + 1)
            positions_new = list(range(new_start, new_start + len(block)))
            if any(h[p] == 1 for p in positions_new):
                continue
            if set(positions_new) & set(block):
                continue
            # Execute move
            for p in block:
                h[p] = 0.0
            for p in positions_new:
                h[p] = 1.0
            
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
                break
            else:
                # Undo
                for p in block:
                    h[p] = 1.0
                for p in positions_new:
                    h[p] = 0.0
        else:
            no_improve += 1
    
    return best_h, best_c5

def random_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def structured_init(n, rng, style=0):
    """Create structured binary initializations"""
    h = np.zeros(n)
    half = n // 2
    if style == 0:
        # Thue-Morse like
        i = 0
        while i < n:
            h[i] = 1.0
            i += max(1, n // 20)
            if i < n and h[i] == 0:
                pass
            i += max(1, n // 20)
        h = project_feasible(h, half)
        return make_binary(h, n)
    elif style == 1:
        # Beatty sequence
        alpha = 0.618  # golden ratio conjugate
        positions = sorted(set(int(i / alpha) % n for i in range(half)))
        if len(positions) > half:
            positions = positions[:half]
        for p in positions:
            h[p % n] = 1.0
        return h
    elif style == 2:
        # Alternating blocks
        block_size = max(2, n // 20)
        i = 0
        val = 1.0
        while i < n:
            for j in range(i, min(i + block_size, n)):
                h[j] = val
            i += block_size
            val = 1.0 - val
        return make_binary(h, n)
    else:
        # Random with spacing
        idx = sorted(rng.choice(n, half, replace=False))
        for i in idx:
            h[i] = 1.0
        return h

def run(seed=42, budget_s=1300, **kwargs):
    t_global = time.time()
    rng = np.random.RandomState(seed)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except:
        h_parent = None
    
    if h_parent is None:
        n_p = 80
        h_parent = np.zeros(n_p)
        # Staircase initial
        for i in range(n_p):
            h_parent[i] = 1.0 if (i % 20) < 10 else 0.0
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    
    def elapsed():
        return time.time() - t_global
    
    def time_left():
        return budget_s - elapsed()
    
    # Phase 1: Upscale parent and optimize at n=300
    n1 = 300
    if time_left() > 200:
        h1 = upscale(h_parent, n1)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_h = h1.copy()
            best_c5 = c5_1
        
        # Subgradient
        if time_left() > 100:
            h1, c5_1 = subgradient_method(h1, budget_s=min(50, time_left() * 0.25), top_lags=10, seed=seed)
            if c5_1 < best_c5:
                best_h = h1.copy()
                best_c5 = c5_1
        
        # SLSQP
        if time_left() > 80:
            h1s, c5_1s = slsqp_optimize(h1, budget_s=min(60, time_left() * 0.3))
            if c5_1s < best_c5:
                best_h = h1s.copy()
                best_c5 = c5_1s
        
        # Binary round + 2opt
        h_bin1 = make_binary(best_h if len(best_h) == n1 else upscale(best_h, n1), n1)
        c5_bin1 = compute_c5(h_bin1)
        if c5_bin1 < best_c5:
            best_h = h_bin1.copy()
            best_c5 = c5_bin1
        
        if time_left() > 50:
            h_bin1, c5_bin1 = binary_2opt(h_bin1, budget_s=min(80, time_left() * 0.4), seed=seed+1)
            c5_bin1 = min(c5_bin1, compute_c5(h_bin1))
            if c5_bin1 < best_c5:
                best_h = h_bin1.copy()
                best_c5 = c5_bin1
        
        if time_left() > 40:
            h_blk, c5_blk = block_move_optimize(h_bin1, budget_s=min(40, time_left() * 0.3), seed=seed+2)
            c5_blk = min(c5_blk, compute_c5(h_blk))
            if c5_blk < best_c5:
                best_h = h_blk.copy()
                best_c5 = c5_blk
    
    # Phase 2: Higher resolution n=500
    n2 = 500
    if time_left() > 300:
        h2 = upscale(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2
        
        if time_left() > 200:
            h2, c5_2 = subgradient_method(h2, budget_s=min(80, time_left() * 0.3), top_lags=15, seed=seed+10)
            if c5_2 < best_c5:
                best_h = h2.copy()
                best_c5 = c5_2
        
        if time_left() > 150:
            h2s, c5_2s = slsqp_optimize(h2, budget_s=min(60, time_left() * 0.3))
            if c5_2s < best_c5:
                best_h = h2s.copy()
                best_c5 = c5_2s
        
        # Binary at n2
        h_bin2 = make_binary(best_h if len(best_h) == n2 else upscale(best_h, n2), n2)
        c5_bin2 = compute_c5(h_bin2)
        if c5_bin2 < best_c5:
            best_h = h_bin2.copy()
            best_c5 = c5_bin2
        
        if time_left() > 60:
            h_bin2, c5_bin2 = binary_2opt(h_bin2, budget_s=min(100, time_left() * 0.4), seed=seed+20)
            if c5_bin2 < best_c5:
                best_h = h_bin2.copy()
                best_c5 = c5_bin2
        
        if time_left() > 40:
            h_blk2, c5_blk2 = block_move_optimize(h_bin2, budget_s=min(50, time_left() * 0.3), seed=seed+30)
            if c5_blk2 < best_c5:
                best_h = h_blk2.copy()
                best_c5 = c5_blk2
    
    # Phase 3: Multiple random restarts at n=200
    n3 = 200
    if time_left() > 60:
        for attempt in range(20):
            if time_left() < 40:
                break
            # Different initializations
            style = attempt % 4
            if attempt < 4:
                h_r = structured_init(n3, rng, style)
            else:
                h_r = random_binary_init(n3, rng)
            
            c5_r = compute_c5(h_r)
            
            # Quick subgradient
            if time_left() > 30:
                h_nr, c5_nr = subgradient_method(h_r, budget_s=min(15, time_left() * 0.2), top_lags=8, seed=seed+100+attempt)
                c5_nr = compute_c5(h_nr)
                if c5_nr < c5_r:
                    h_r = h_nr
                    c5_r = c5_nr
            
            # Binary 2opt
            h_br = make_binary(h_r, n3)
            c5_br = compute_c5(h_br)
            if c5_br < c5_r:
                h_r = h_br
                c5_r = c5_br
            
            if time_left() > 25:
                h_br, c5_br = binary_2opt(h_r, budget_s=min(20, time_left() * 0.3), seed=seed+200+attempt)
                c5_br = compute_c5(h_br)
                if c5_br < c5_r:
                    h_r = h_br
                    c5_r = c5_br
            
            if c5_r < best_c5:
                best_h = h_r.copy()
                best_c5 = c5_r
    
    # Phase 4: If time remains, refine best solution at its resolution
    if time_left() > 60:
        n_best = len(best_h)
        # More 2-opt iterations
        if n_best % 2 == 0:
            h_ref, c5_ref = binary_2opt(best_h, budget_s=min(60, time_left() * 0.5), seed=seed+500)
            if c5_ref < best_c5:
                best_h = h_ref.copy()
                best_c5 = c5_ref
    
    # Final: ensure constraint satisfaction
    n_final = len(best_h)
    target = n_final / 2.0
    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, int(n_final)
