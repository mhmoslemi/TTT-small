def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def project_box_sum(y, S):
    y = np.asarray(y, dtype=float).ravel()
    n = int(y.size)
    if n == 0:
        return y
    S = float(S)
    if S <= 0.0:
        return np.zeros(n, dtype=float)
    if S >= float(n):
        return np.ones(n, dtype=float)

    y_max = float(y.max())

    hi = y_max

        np.clip(work, 0.0, 1.0, out=work)

def exact_objective_correlate(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = float(dx * float(corr.max()))
    if val < 0.0 and val > -1e-12:
        val = 0.0
    if val > 1.0 and val < 1.0 + 1e-12:
        val = 1.0
    return val


def estimate_objective(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return 0.0
    L = next_pow2(2 * n - 1)
    dx = 2.0 / float(n)

    H = np.fft.rfft(h, L)
    R = np.fft.irfft(H * np.conjugate(H), L)[:n]

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]

    A = prefix[n:0:-1]
    B = total - prefix[:n]

    SF = A - R
    max_s = float(SF.max())

        val = 1.0
    return float(val)


def eval_grad(h, T):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return 0.0, np.zeros(0, dtype=float)

    L = next_pow2(2 * n - 1)
    dx = 2.0 / float(n)
    T = max(float(T), 1e-9)

    H = np.fft.rfft(h, L)
    R = np.fft.irfft(H * np.conjugate(H), L)[:n]

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]

    A = prefix[n:0:-1]
    B = total - prefix[:n]

    SF = A - R
    qF = dx * SF

    qmax = float(qF.max())
    if n > 1:
        SG = B - R
        qG = dx * SG[1:]
        gm = float(qG.max())
        if gm > qmax:
            qmax = gm

    if qmax < 0.0 and qmax > -1e-12:
        qmax = 0.0

    eF = np.exp((qF - qmax) / T)
    if n > 1:
        eG = np.exp((qG - qmax) / T)
        wsum = float(eF.sum() + eG.sum())
        if wsum <= 0.0:
            wsum = 1.0
        alpha = eF / wsum
        beta = eG / wsum
    else:
        s = float(eF.sum())
        if s <= 0.0:
            s = 1.0
        alpha = eF / s
        beta = np.zeros(0, dtype=float)

    grad = np.zeros(n, dtype=float)

    if n == 1:
        grad[0] = float(alpha[0]) * (1.0 - 2.0 * h[0])
        return qmax, grad

    p0 = float(alpha[0])
    if p0 != 0.0:
        grad += p0 * (1.0 - 2.0 * h)

    arr = alpha[1:]
    prefix_arr = np.empty(n, dtype=float)
    prefix_arr[0] = 0.0
    prefix_arr[1:] = np.cumsum(arr)
    tailF = prefix_arr[n - 1 - np.arange(n)]

    alpha_pos = np.zeros(n, dtype=float)
    alpha_pos[1:] = arr

    beta_pos = np.zeros(n, dtype=float)
    if n > 1:
        beta_pos[1:] = beta

    prefixB = np.zeros(n, dtype=float)
    if n > 1:
        prefixB = np.cumsum(beta_pos)

    A_fft = np.fft.rfft(alpha_pos, L)
    B_fft = np.fft.rfft(beta_pos, L)

    convHA = np.fft.irfft(H * A_fft, L)[:n]
    convHB = np.fft.irfft(H * B_fft, L)[:n]
    corrB = np.fft.irfft(H * np.conjugate(B_fft), L)[:n]

    grad += tailF - convHA + prefixB - convHB - corrB

    return qmax, grad


def upsample(h, n):
    h = np.asarray(h, dtype=float).ravel()
    m = int(h.size)
    n = int(n)
    if n == m:
        return h.copy()
    if m > 0 and n % m == 0:
        return np.repeat(h, n // m)
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    if m == 1:
        return np.full(n, float(h[0]), dtype=float)
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])

        weights = np.array([0.10, 0.12, 0.14, 0.16, 0.18, 0.18, 0.12][:len(ns)], dtype=float)
    weights = weights / float(weights.sum())

    for i, n in enumerate(ns):
        elapsed = time.time() - t0
        if elapsed > budget_s - 35.0:
            break
        remaining = max(0.0, budget_s - elapsed)
        limit = remaining * float(weights[i])
        if limit < 20.0:
            continue

        h0 = upsample(best_h, n)
        h0 = project_box_sum(h0, 0.5 * n)

        if h0.size != n_parent:
            rng0 = np.random.default_rng(seed + 100000 + n)
            h0 = project_box_sum(h0 + rng0.normal(0.0, 0.01, n), 0.5 * n)

        deadline = t0 + limit
        best_h, best_obj = improve_one(h0, n, deadline, best_h, best_obj, seed + 200000 + n)

    elapsed = time.time() - t0
    remaining = max(0.0, budget_s - elapsed)

    if remaining > 90.0:
        ncur = int(best_h.size)
        limit1 = min(remaining * 0.55, 520.0)
        h0 = project_box_sum(best_h, 0.5 * ncur)
        deadline = t0 + limit1
        best_h, best_obj = improve_one(h0, ncur, deadline, best_h, best_obj, seed + 900000)

    elapsed = time.time() - t0
    remaining = max(0.0, budget_s - elapsed)

    if remaining > 70.0:
        ncur = int(best_h.size)
        n2 = ncur * 2
        if n2 <= 2048:
            h0 = upsample(best_h, n2)
            h0 = project_box_sum(h0, 0.5 * n2)
            rng1 = np.random.default_rng(seed + 9000000 + n2)
            h0 = project_box_sum(h0 + rng1.normal(0.0, 0.01, n2), 0.5 * n2)
            limit2 = min(remaining * 0.9, 520.0)
            deadline = t0 + limit2
            best_h, best_obj = improve_one(h0, n2, deadline, best_h, best_obj, seed + 9100000 + n2)

    best_h = project_box_sum(best_h, 0.5 * best_h.size)
    best_obj = exact_objective_correlate(best_h)

    return best_h.tolist(), float(best_obj), int(best_h.size)
