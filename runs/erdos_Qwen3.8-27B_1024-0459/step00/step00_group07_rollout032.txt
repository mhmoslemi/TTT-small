import numpy as np
from scipy.optimize import minimize, Bounds
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def grad_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    
    grad = np.zeros(n)
    if s >= 0:
        # c = sum_{i=0}^{n-1-s} h[i] * (1-h[i+s])
        valid_i = np.arange(n - s)
        grad[valid_i] += (1.0 - h[valid_i + s])
        grad[valid_i + s] -= h[valid_i]
    else:
        s2 = -s
        # c = sum_{i=0}^{n-1-s2} h[i+s2] * (1-h[i])
        valid_i = np.arange(n - s2)
        grad[valid_i + s2] += (1.0 - h[valid_i])
        grad[valid_i] -= h[valid_i + s2]
    
    grad *= dx
    return grad

def constraint_fun(h):
    return np.sum(h) - len(h) / 2.0

def make_initial(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def upscale_linear(init, n_target):
    n_init = len(init)
    x_src = np.linspace(0.0, 2.0 / n_init, n_init)
    x_tgt = np.linspace(0.0, 2.0 / n_target, n_target)
    h = np.interp(x_tgt, x_src, init)
    h = np.clip(h, 0.0, 1.0)
    # Adjust to satisfy sum constraint
    target = n_target / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute adjustment
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0, 1)
        # Second pass
        diff = target - np.sum(h)
        if diff > 0:
            mask = h < 1
            per = diff / max(mask.sum(), 1)
            h[mask] = np.minimum(h[mask] + per, 1.0)
        elif diff < 0:
            mask = h > 0
            per = diff / max(mask.sum(), 1)
            h[mask] = np.maximum(h[mask] + per, 0.0)
    return h

        return eval_c5(h)

        return grad_c5(h)

    bounds = Bounds(np.zeros(n), np.ones(n))

    best_val = obj(h0)
    
    result = minimize(
        obj, h0.copy(), jac=obj_grad,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,

    if result.fun < best_val:

        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0, 1)

                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h01[i] = 0
                h01[j] = 1
                c5_new = eval_c5(h01)
                if c5_new < c5_01 - 1e-10:
                    c5_01 = c5_new
                    improved = True
                    # Update lists
                    ones = np.where(h01 == 1)[0]
                    zeros = np.where(h01 == 0)[0]
                else:
                    h01[i] = 1
                    h01[j] = 0
            
            if c5_01 < best_c5:
                best_c5 = c5_01
                best_h = h01.copy()
    
    # Final evaluation
    best_c5 = eval_c5(best_h)
    # Ensure sum constraint
    target_sum = n / 2.0
    diff = target_sum - np.sum(best_h)
    if abs(diff) > 1e-10:
        mask = (best_h > 0) & (best_h < 1)
        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
            best_h = np.clip(best_h, 0, 1)
    
    return best_h, best_c5, n

Alternative: use the fact that for 0-1 valued h, the problem reduces to a combinatorial one. The key metric is:
For each shift s, count the number of positions where h[i+s]=1 and h[i]=0 (for s>0) or h[i]=1 and h[i+s]=0 (for s<0).

For 0-1 valued h with exactly n/2 ones:
C5 = (2/n) * max_s count_s

To minimize this, we want all counts to be as balanced as possible. The minimum possible max count is at least (n/2) * (n/2) / (n-1) (pigeonhole-like argument)... actually this isn't quite right.

For a random 0-1 string with n/2 ones, the expected count for shift s > 0 is:
E[count_s] = (n-s) * (1/2) * (1/2) = (n-s)/4 (approximately, ignoring the constraint of exactly n/2 ones)

The maximum over s of (n-s)/4 is (n-1)/4 (at s=1).
So C5 ≈ 2/n * (n-1)/4 ≈ 1/2 for large n.

For a 0-1 valued h, the correlate at shift s gives:

Minimizing max_s c_s is related to minimizing max_s d_s, but not identical.
