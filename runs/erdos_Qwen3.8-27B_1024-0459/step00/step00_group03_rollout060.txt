def make_binary(h, N):
    """Convert continuous h to binary with exactly N//2 ones."""
    h_bin = (h > 0.5).astype(float)
    ones = int(np.sum(h_bin))
    target = N // 2
    if ones > target:
        one_idx = np.where(h_bin == 1)[0]
        # Remove the ones with smallest h values
        order = np.argsort(h[one_idx])
        for k in range(ones - target):
            h_bin[one_idx[order[k]]] = 0
    elif ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for k in range(target - ones):
            h_bin[zero_idx[order[k]]] = 1
    return h_bin

import numpy as np
from scipy.optimize import minimize

# Module-level globals for optimizer callbacks
_g_N = 0
_g_dx = 0.0
_g_target = 0.0
_g_temp = 0.01
_g_penalty = 0.0


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, N):
    h = np.clip(h, 0.0, 1.0).copy()
    target = N / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 0.001) & (h < 0.999)
        n_mask = int(np.sum(mask))
        if n_mask == 0:
            if diff > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
        else:
            step = diff / n_mask
            h[mask] = np.clip(h[mask] + step, 0.0, 1.0)

            break
        idx = int(cands[0])

    sm = t * (m / t + np.log(np.sum(np.exp(exp_arg))))
