def make_block_start(n,rng):
    target=0.5*n
    h=np.zeros(n)

    if mode==0:
       # central one block
       length=target
       if length >=1:
          s=int(((n-length)//2))
          e=s+int(math.ceil(length))
          h[s:min(n,e)]=1.0

          l=min(a if one else b, n-pos)

def project_fixed_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    target = float(target)
    if target <= 0.0:
        return np.zeros_like(x)
    if target >= x.size:
        return np.ones_like(x)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        tau = 0.5*(lo+hi)
        y = np.clip(x - tau, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            lo = tau
        else:
            hi = tau
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    rem = target - float(np.sum(y))
    if rem > 1e-12:
        cap = 1.0 - y
        tot = float(np.sum(cap))
        if tot > 0.0:
            y += (rem / tot) * cap
    elif rem < -1e-12:
        tot = float(np.sum(y))
        if tot > 0.0:
            y += (rem / tot) * y
    y = np.clip(y, 0.0, 1.0)
    rem = target - float(np.sum(y))
    if abs(rem) > 1e-13:
        if rem > 0:
            idx = int(np.argmax(1.0 - y))
        else:
            idx = int(np.argmax(y))
        y[idx] = float(np.clip(y[idx] + rem, 0.0, 1.0))
    return y

import time
import math
import warnings
import numpy as np
from scipy.optimize import minimize

warnings.filterwarnings("ignore")

SL_STOP_DEADLINE = 0.0


def stop_callback(xk):
    return bool(time.perf_counter() > SL_STOP_DEADLINE)


def should_continue(deadline, margin=30.0):
    return bool(time.perf_counter() < float(deadline) - float(margin))


def project_fixed_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    if x.size == 0:
        return x
    target = float(target)
    if target <= 0.0:
        return np.zeros_like(x)
    if target >= float(x.size):
        return np.ones_like(x)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        y = np.clip(x - tau, 0.0, 1.0)
        s = float(np.sum(y))
        if s > target:
            lo = tau
        else:
            hi = tau

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    rem = target - float(np.sum(y))
    if rem > 1e-12:
        cap = 1.0 - y
        tot = float(np.sum(cap))
        if tot > 0.0:
            y += (rem / tot) * cap
    elif rem < -1e-12:
        tot = float(np.sum(y))
        if tot > 0.0:
            y += (rem / tot) * y

    y = np.clip(y, 0.0, 1.0)
    for _ in range(6):
        rem = target - float(np.sum(y))
        if abs(rem) <= 1e-13:
            break
        if rem > 0.0:
            idx = int(np.argmax(1.0 - y))
        else:
            idx = int(np.argmax(y))
        y[idx] = float(np.clip(y[idx] + rem, 0.0, 1.0))
    return y


def score_h(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c * dx))


def overlap_q_values(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=float)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def shifts_for_n(n):
    n = int(n)
    return np.arange(n - 1, -n, -1, dtype=int)


def add_grad_term(g, h, s, w):
    n = int(h.size)
    s = int(s)
    w = float(w)
    if w == 0.0:
        return
    if s > 0:
        g[: n - s] += w * (1.0 - h[s:])
        g[s:] -= w * h[: n - s]
    elif s < 0:
        t = -s
        g[t:] += w * (1.0 - h[: n - t])
        g[: n - t] -= w * h[t:]
    else:
        g += w * (1.0 - 2.0 * h)


def weighted_overlap_grad(h, weights):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    g = np.zeros(n, dtype=float)
    s_arr = shifts_for_n(n)
    for idx in range(weights.size):
        w = float(weights[idx])
        if abs(w) > 1e-15:
            add_grad_term(g, h, int(s_arr[idx]), w)
    return g


def weighted_overlap_grad_top(h, q, beta, topk):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    q = np.asarray(q, dtype=float)
    m = float(np.max(q))

    if beta <= 1e-6:
        idx = np.array([int(np.argmax(q))], dtype=int)
        w = np.array([1.0], dtype=float)
    else:
        if topk is not None and 0 < int(topk) < q.size:
            idx = np.argsort(q)[-int(topk):]
        else:
            idx = np.arange(q.size)
        z = float(beta) * (q[idx] - m)
        ez = np.exp(z)
        denom = float(np.sum(ez))
        if denom <= 0.0:
            w = np.zeros_like(ez)
            w[np.argmax(ez)] = 1.0
        else:
            w = ez / denom

    s_arr = shifts_for_n(n)[idx]
    g = np.zeros(n, dtype=float)
    for j in range(idx.size):
        add_grad_term(g, h, int(s_arr[j]), float(w[j]))
    return g


def smooth_max_obj(x, beta):
    q = overlap_q_values(x)
    m = float(np.max(q))
    beta = float(beta)
    if beta <= 1e-6:
        return m
    z = beta * (q - m)
    s = float(np.sum(np.exp(z)))
    if s <= 0.0:
        return m
    return m + math.log(s) / beta


def smooth_max_grad(x, beta):
    q = overlap_q_values(x)
    m = float(np.max(q))
    beta = float(beta)
    if beta <= 1e-6:
        w = np.zeros(q.size, dtype=float)
        w[int(np.argmax(q))] = 1.0
    else:
        z = beta * (q - m)
        ez = np.exp(z)
        s = float(np.sum(ez))
        if s <= 0.0:
            w = np.zeros(q.size, dtype=float)
            w[int(np.argmax(q))] = 1.0
        else:
            w = ez / s
    return weighted_overlap_grad(x, w)


def sum_eq_fun(x, target):
    return float(np.sum(x) - float(target))


def sum_eq_jac(x, target):
    return np.ones_like(x, dtype=float)


def lbfgs_penalty_obj(x, beta, target, mu):
    dev = float(np.sum(x) - float(target))
    return smooth_max_obj(x, beta) + 0.5 * float(mu) * dev * dev


def lbfgs_penalty_jac(x, beta, target, mu):
    grad = smooth_max_grad(x, beta)
    dev = float(np.sum(x) - float(target))
    if dev != 0.0:
        grad = grad + (float(mu) * dev) * np.ones_like(x)
    return grad


def lbfgs_penalty_once(h0, beta, target, maxiter, deadline):
    global SL_STOP_DEADLINE
    n = int(np.asarray(h0).size)
    h = project_fixed_sum(h0, target)
    SL_STOP_DEADLINE = float(deadline)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    try:
        res = minimize(
            lbfgs_penalty_obj,
            h,
            jac=lbfgs_penalty_jac,
            args=(float(beta), float(target), 100.0),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "gtol": 1e-5, "disp": 0},
            callback=stop_callback,
        )
        if res is not None and res.x is not None:
            h = res.x
    except Exception:
        pass
    return project_fixed_sum(h, target)


def run_slsqp_once(h0, beta, target, maxiter, deadline):
    global SL_STOP_DEADLINE
    n = int(np.asarray(h0).size)
    h = project_fixed_sum(h0, target)
    SL_STOP_DEADLINE = float(deadline)
    bounds = list(zip(np.zeros(n), np.ones(n)))
    constraints = [
        {
            "type": "eq",
            "fun": sum_eq_fun,
            "jac": sum_eq_jac,
            "args": (float(target),),
        }
    ]
    try:
        res = minimize(
            smooth_max_obj,
            h,
            args=(float(beta),),
            jac=smooth_max_grad,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": 1e-10, "disp": 0},
            callback=stop_callback,
        )
        if res is not None and res.x is not None:
            h = res.x
    except Exception:
        pass
    return project_fixed_sum(h, target)


def projected_polish(h0, max_iters, rng, deadline, beta=120.0, topk=64):
    h = project_fixed_sum(np.asarray(h0, dtype=float).copy(), 0.5 * len(h0))
    n = int(h.size)
    best_h = h.copy()
    best_score = score_h(h)

    step = 0.02
    beta_cur = float(beta)
    momentum = np.zeros(n, dtype=float)
    no_improve = 0

    for it in range(int(max_iters)):
        if time.perf_counter() > float(deadline):
            break

        q = overlap_q_values(h)
        cur = float(np.max(q))
        if cur < best_score:
            best_score = cur
            best_h = h.copy()

        g = weighted_overlap_grad_top(h, q, beta_cur, topk)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-14:
            h = project_fixed_sum(best_h + 0.02 * rng.standard_normal(n), 0.5 * n)
            momentum *= 0.0
            no_improve += 1
            if no_improve > 80:
                beta_cur = max(30.0, 0.6 * beta_cur)
                no_improve = 0
            continue

        momentum = 0.85 * momentum - step * g
        mnorm = float(np.linalg.norm(momentum))
        if mnorm > 0.25:
            momentum *= 0.25 / mnorm

        cand = project_fixed_sum(h + momentum, 0.5 * n)
        cs = score_h(cand)

        if cs <= cur + 1e-12:
            h = cand
            if cs < best_score:
                best_score = cs
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            step = min(0.05, step * 1.02)
        else:
            step *= 0.7
            momentum *= 0.5
            no_improve += 1
            if no_improve > 100 or step < 0.003:
                h = project_fixed_sum(best_h + 0.03 * rng.standard_normal(n), 0.5 * n)
                step = 0.02
                beta_cur = max(30.0, 0.6 * beta_cur)
                momentum *= 0.0
                no_improve = 0

        if it % 60 == 0 and it > 0:
            beta_cur = min(1200.0, beta_cur * 1.15)

    return best_h, best_score


def interpolate_resample(h, m):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    m = int(m)
    if n == 0:
        return np.zeros(m, dtype=float)
    if n == 1:
        return np.full(m, float(h[0]), dtype=float)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    y = np.linspace(0.0, 1.0, m, endpoint=False)
    return np.interp(y, x, h)


def upsample_h(h, m):
    m = int(m)
    v = interpolate_resample(h, m)
    return project_fixed_sum(np.clip(v, 0.0, 1.0), 0.5 * m)


def threshold_to_binary(h, target):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = float(target)
    order = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=float)
    m = int(math.floor(target + 1e-12))
    rem = target - float(m)

    if m > 0:
        b[order[:m]] = 1.0
    if rem > 1e-12 and m < n:
        b[order[m]] = min(1.0, rem)

    return project_fixed_sum(b, target)


def make_block_start(n, rng):
    n = int(n)
    target = 0.5 * n
    h = np.zeros(n, dtype=float)
    mode = int(rng.integers(0, 3))

    if mode == 0:
        length = target
        if length >= 1.0:
            s = int(((n - length) // 2))
            e = s + int(math.ceil(length))
            h[max(0, s) : min(n, e)] = 1.0
    elif mode == 1:
        length = 0.5 * target
        s1 = int(0.15 * n)
        e1 = s1 + int(math.ceil(length))
        h[max(0, s1) : min(n, e1)] = 1.0
        s2 = int(0.65 * n)
        e2 = s2 + int(math.ceil(length))
        h[max(0, s2) : min(n, e2)] = 1.0
    else:
        a = int(rng.integers(2, 8))
        b = int(rng.integers(2, 8))
        pos = 0
        one = bool(rng.random() < 0.5)
        while pos < n:
            length = min(a if one else b, n - pos)
            if one:
                h[pos : pos + length] = 1.0
            pos += length
            one = not one

    return project_fixed_sum(h, target)


def random_smooth_start(n, rng):
    n = int(n)
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    v = 0.5 + float(rng.normal(0.0, 0.08))
    for _ in range(4):
        f = int(rng.integers(1, 5))
        a = float(rng.uniform(0.05, 0.35))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        v = v + a * np.cos(2.0 * np.pi * f * x + ph)

        a2 = float(rng.uniform(0.01, 0.18))
        ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
        v = v + a2 * np.sin(2.0 * np.pi * f * x + ph2)

    v = 1.0 / (1.0 + np.exp(-4.0 * (v - 0.5)))
    return project_fixed_sum(v, 0.5 * n)


def random_binary_start(n, rng):
    n = int(n)
    target = 0.5 * n
    m = int(math.floor(target + 1e-12))
    rem = target - float(m)

    pos = np.linspace(0.0, 1.0, n, endpoint=False)
    center = (1.0 - 2.0 * np.abs(pos - 0.5)) ** 2
    w = (0.2 + 0.8 * center) * (0.4 + rng.random(n))

    order = np.argsort(w)
    b = np.zeros(n, dtype=float)
    if m > 0:
        b[order[-m:]] = 1.0
    if rem > 1e-12 and m < n:
        b[order[-(m + 1)]] = min(1.0, rem)

    return project_fixed_sum(b, target)


def choose_start(n, attempt, parent, best_h, rng):
    n = int(n)
    target = 0.5 * n
    attempt = int(attempt)

    if best_h is None:
        if attempt == 0:
            return make_block_start(n, rng)
        if attempt == 1:
            return random_binary_start(n, rng)
        return random_smooth_start(n, rng)

    if attempt == 0:
        if best_h.size == n:
            return project_fixed_sum(best_h, target)
        return upsample_h(best_h, n)

    if attempt == 1:
        lin = interpolate_resample(best_h, n)
        return threshold_to_binary(np.clip(lin, 0.0, 1.0), target)

    if attempt == 2:
        if parent is not None:
            p = np.asarray(parent, dtype=float).ravel()
            if p.size > 0:
                if p.size == n:
                    return project_fixed_sum(p, target)
                return upsample_h(p, n)
        return random_smooth_start(n, rng)

    if attempt == 3:
        return random_binary_start(n, rng)

    base = upsample_h(best_h, n)
    return project_fixed_sum(base + 0.05 * rng.standard_normal(n), target)


def rounding_variants(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    target = 0.5 * n
    out = []

    b = threshold_to_binary(h, target)
    out.append(b)

    r = np.where(h >= 0.5, 1.0, 0.0)
    r = project_fixed_sum(r, target)
    out.append(r)

    mid = 0.5 * (h + threshold_to_binary(h, target))
    mid = project_fixed_sum(mid, target)
    out.append(mid)

    return out


def optimize_sequence(h, target, n, deadline, rng):
    h = project_fixed_sum(np.asarray(h, dtype=float).copy(), target)
    n = int(n)
    target = float(target)

    best_h = h.copy()
    best_score = score_h(h)

    now = time.perf_counter()
    if n <= 64:
        lb_betas = (8.0, 30.0, 100.0)
        lb_base = 250
        beta_poly = 100.0
    elif n <= 256:
        lb_betas = (20.0, 80.0)
        lb_base = 120
        beta_poly = 150.0
    else:
        lb_betas = (40.0, 120.0)
        lb_base = 60
        beta_poly = 250.0

    for beta in lb_betas:
        if not should_continue(deadline, 25.0):
            break
        rem = max(0.0, float(deadline) - time.perf_counter())
        maxit = int(min(lb_base, max(20, rem / 0.10)))
        h2 = lbfgs_penalty_once(h, beta, target, maxit, deadline)
        s2 = score_h(h2)
        if s2 < best_score:
            best_h = h2
            best_score = s2
        h = h2

    if n <= 256:
        if n <= 96:
            sl_betas = (30.0, 100.0)
            sl_base = 80
        else:
            sl_betas = (50.0, 150.0)
            sl_base = 40
        for beta in sl_betas:
            if not should_continue(deadline, 45.0):
                break
            rem = max(0.0, float(deadline) - time.perf_counter())
            maxit = int(min(sl_base, max(10, rem / 0.25)))
            h2 = run_slsqp_once(h, beta, target, maxit, deadline)
            s2 = score_h(h2)
            if s2 < best_score:
                best_h = h2
                best_score = s2
            h = h2 if s2 <= best_score + 0.02 else best_h

    if should_continue(deadline, 45.0):
        start_poly = h if score_h(h) <= best_score + 0.01 else best_h
        h2, s2 = projected_polish(
            start_poly,
            100000,
            rng,
            float(deadline) - 10.0,
            beta=beta_poly,
            topk=64,
        )
        if s2 < best_score:
            best_h = h2
            best_score = s2

    if should_continue(deadline, 20.0):
        for var in rounding_variants(best_h):
            vv = project_fixed_sum(var, target)
            ss = score_h(vv)
            if ss < best_score:
                best_h = vv
                best_score = ss

    return best_h, best_score


def run_phase(n, phase_deadline, parent, best_h, best_score, rng):
    n = int(n)
    target = 0.5 * n
    phase_deadline = float(phase_deadline)

    remaining = phase_deadline - time.perf_counter()
    if remaining < 60.0:
        return best_h, best_score

    if n >= 360:
        max_attempts = 1
    elif remaining < 300.0:
        max_attempts = 1
    elif remaining < 800.0:
        max_attempts = 2
    else:
        max_attempts = 3

    for attempt in range(max_attempts):
        remaining = phase_deadline - time.perf_counter()
        if remaining < 90.0:
            break

        per_deadline = min(
            phase_deadline - 15.0,
            time.perf_counter() + max(60.0, remaining / float(max_attempts - attempt)),
        )

        st = choose_start(n, attempt, parent, best_h, rng)
        st = project_fixed_sum(st, target)
        ss = score_h(st)
        if ss < best_score:
            best_h = st
            best_score = ss

        h2, s2 = optimize_sequence(st, target, n, per_deadline, rng)
        if s2 < best_score:
            best_h = h2
            best_score = s2

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global SL_STOP_DEADLINE

    t0 = time.perf_counter()
    budget = float(budget_s)
    if budget <= 0.0:
        budget = 1.0
    deadline = t0 + budget
    SL_STOP_DEADLINE = deadline

    rng = np.random.default_rng(int(seed))

    parent = globals().get("initial_h_values", None)
    best_h = None
    best_score = float("inf")

    if parent is not None:
        p = np.asarray(parent, dtype=float).ravel()
        if p.size > 0:
            hp = project_fixed_sum(p, 0.5 * p.size)
            sp = score_h(hp)
            best_h = hp
            best_score = sp

    if best_h is None:
        best_h = make_block_start(45, rng)
        best_score = score_h(best_h)

    if budget < 300.0:
        phases = [(45, deadline - 2.0)]
        if budget > 150.0:
            phases.append((90, deadline - 2.0))
    else:
        phases = [
            (45, t0 + 0.12 * budget),
            (90, t0 + 0.40 * budget),
            (180, t0 + 0.78 * budget),
            (360, deadline - 2.0),
        ]

    for n, phd in phases:
        n = int(n)
        phd = float(phd)
        if phd < time.perf_counter() + 45.0:
            continue
        best_h, best_score = run_phase(n, phd, parent, best_h, best_score, rng)

    if best_h is not None:
        best_h = project_fixed_sum(best_h, 0.5 * len(best_h))
        best_score = score_h(best_h)

        for var in rounding_variants(best_h):
            vv = project_fixed_sum(var, 0.5 * len(var))
            ss = score_h(vv)
            if ss < best_score:
                best_h = vv
                best_score = ss

    return best_h, float(best_score), int(best_h.size)
