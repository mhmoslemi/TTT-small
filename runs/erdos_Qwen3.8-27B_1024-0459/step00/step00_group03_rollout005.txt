def safe_exp(x):
    if x < -745.0:
        return 0.0

def smooth_array(a, width=1):
    n = int(a.size)
    if width <= 0 or n <= 2:
        return a.copy()
    w = int(min(width, n // 2))
    if w <= 0:
        return a.copy()
    padded = np.pad(a, w, mode="edge")
    kernel = np.ones(2 * w + 1) / float(2 * w + 1)
    return np.convolve(padded, kernel, mode="valid")[:n]


def project_feasible(z, n):
    n = int(n)
    z = np.asarray(z, dtype=float).reshape(-1)
    if z.size != n:
        z = np.resize(z, n)
    z = np.nan_to_num(z, nan=0.5, posinf=1.0, neginf=0.0)

    target = 0.5 * n
    maxz = float(np.max(z))
    minz = float(np.min(z))
    lo = -maxz - 2.0
    hi = 1.0 - minz + 2.0

    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    return np.clip(z + lam, 0.0, 1.0)


def make_feasible(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size != n:
        h = np.resize(h, n)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)

    target = 0.5 * n
    s = float(np.sum(h))
    err = target - s

    if abs(err) <= 1e-10:
        if err != 0.0:
            free = (h > 1e-12) & (h < 1.0 - 1e-12)
            if np.any(free):
                cnt = int(np.sum(free))
                h[free] = np.clip(h[free] + err / float(cnt), 0.0, 1.0)

    h = project_feasible(h, n)

def resample_h(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    m = int(h.size)
    if m == n:
        return make_feasible(h.copy(), n)
    if m <= 0:

    x_old = (np.arange(m, dtype=float) + 0.5) * (2.0 / m)
    x_new = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)

    return make_feasible(y, n)


def evaluate_corr(h, n):
    n = int(n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_grad(h, n, beta, F_override=None):
    n = int(n)
    y = 2.0 * h - 1.0
    dx = 2.0 / n

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    if n > 0:
        np.cumsum(y, out=pref[1:])
    total = float(pref[n])

    pvals = np.empty(n, dtype=float)
    mvals = np.empty(n, dtype=float)

    for s in range(n):
        m = n - s
        if s == 0:
            R = float(np.dot(y, y))
            D = 0.0
        else:
            R = float(np.dot(y[:m], y[s:]))
            D = float(pref[m] - (total - pref[s]))
        base = 0.25 * dx * (float(m) - R)
        adj = 0.25 * dx * D
        pv = base + adj
        mv = base - adj
        if pv < 0.0 and pv > -1e-12:
            pv = 0.0
        if mv < 0.0 and mv > -1e-12:
            mv = 0.0
        pvals[s] = pv
        mvals[s] = mv

    if F_override is None:
        Fw = float(max(np.max(pvals), np.max(mvals)))
    else:
        Fw = float(F_override)

    maxq = float(max(np.max(pvals), np.max(mvals)))
    if maxq > Fw + 1e-10:
        Fw = maxq
    if Fw < 0.0:
        Fw = 0.0

    beta = float(beta)
    if beta < 1e-8:
        beta = 1.0

    g = np.zeros(n, dtype=float)

    vals = np.empty(2 * n - 1, dtype=float)
    if n > 0:
        pv = float(pvals[0])
        vals[0] = 1.0 if pv >= Fw else safe_exp(beta * (pv - Fw))
    for s in range(1, n):
        pv = float(pvals[s])
        mv = float(mvals[s])
        vals[2 * s - 1] = 1.0 if pv >= Fw else safe_exp(beta * (pv - Fw))
        vals[2 * s] = 1.0 if mv >= Fw else safe_exp(beta * (mv - Fw))

    wsum = float(np.sum(vals))
    if not math.isfinite(wsum) or wsum <= 0.0:
        idx = int(np.argmax(vals))
        vals = np.zeros_like(vals)
        vals[idx] = 1.0
        wsum = 1.0

    inv = 1.0 / wsum
    thresh = 1e-14

    w0 = float(vals[0]) * inv
    if w0 > thresh:
        g += w0 * (-dx * y)

    for s in range(1, n):
        pwt = float(vals[2 * s - 1]) * inv
        mwt = float(vals[2 * s]) * inv
        if (pwt + mwt) <= thresh:
            continue

        m = n - s
        Rg = np.zeros(n, dtype=float)

        Rg[s:] += y[:m]

            g += pwt * (negR + 0.5 * dx * Dg)
        if mwt > thresh:
            g += mwt * (negR - 0.5 * dx * Dg)

    return g

def random_feasible(n, rng):
    n = int(n)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    sig = np.zeros(n, dtype=float)
    for k in range(1, 7):
        ph = float(rng.normal(0.0, 1.0))
        amp = float(rng.normal(0.0, 1.0)) / float(k)
        sig += amp * np.cos(k * math.pi * x + ph)

        h = np.full(n, 0.5)
    else:
        h = 0.05 + 0.9 * (sig - mn) / span
    h = smooth_array(h, 1)
    return make_feasible(h, n)

                if math.isfinite(F0):
                    best_h = h0.copy()
                    best_F = F0
        except Exception:
            best_h = None

    if best_h is None:
        best_h = pattern_start(70)
        best_F = float(evaluate_corr(best_h, best_h.size))

    n0 = int(best_h.size)
    ns = []

    for cand in (
        n0,
        int(round(n0 * 1.25)),
        int(round(n0 * 1.6)),
        int(round(n0 * 2.0)),
        120,
        160,
        200,
        240,
        280,
    ):
        cand = int(cand)
        if 40 <= cand <= 300:
            if cand not in ns:
                ns.append(cand)

    ns.sort()
    if n0 in ns:
        ns.remove(n0)
        ns.insert(0, n0)

    for idx, n in enumerate(ns):
        rem_time = deadline - time.time()
        if rem_time <= 30.0:
            break

        if n == best_h.size:
            start = best_h.copy()
        else:
            start = resample_h(best_h, n)

        left_ns = max(1, len(ns) - idx)
        phase_budget = min(350.0, max(40.0, 0.85 * (deadline - time.time()) / float(left_ns)))

        limit1 = time.time() + max(18.0, 0.40 * phase_budget)
        limit1 = min(limit1, deadline - 25.0)
        if limit1 > time.time():
            it1 = int(max(80, min(1800, 90000.0 / max(2, n))))
            h1, F1 = local_optimize(start, n, 35.0, limit1, it1)
            if math.isfinite(F1) and F1 + 1e-12 < best_F:
                best_h = make_feasible(h1, n)
                best_F = F1

        if time.time() < deadline - 35.0:
            limit2 = time.time() + max(12.0, 0.60 * phase_budget)
            limit2 = min(limit2, deadline - 12.0)
            if limit2 > time.time():
                it2 = int(max(50, min(1000, 50000.0 / max(2, n))))
                base = best_h
                if n == base.size:
                    bstart = base.copy()
                else:
                    bstart = resample_h(base, n)
                h2, F2 = local_optimize(bstart, n, 180.0, limit2, it2)
                if math.isfinite(F2) and F2 + 1e-12 < best_F:
                    best_h = make_feasible(h2, n)
                    best_F = F2

    for extra_n in (n0, int(min(200, max(90, n0 + 30))), int(min(260, max(120, n0 + 60)))):
        rem_time = deadline - time.time()
        if rem_time <= 45.0:
            break
        extra_n = int(extra_n)
        if extra_n < 40 or extra_n > 300:
            continue

        if extra_n == best_h.size:
            noise = 0.025 * max(0.45, math.sqrt(70.0 / float(extra_n)))
            start = project_feasible(best_h + noise * rng.standard_normal(extra_n), extra_n)
        else:
            start = random_feasible(extra_n, rng)

        limit = time.time() + max(18.0, min(90.0, 0.45 * (deadline - time.time())))
        limit = min(limit, deadline - 15.0)
        if limit > time.time():
            it = int(max(50, min(600, 30000.0 / max(2, extra_n))))
            hr, Fr = local_optimize(start, extra_n, 80.0, limit, it)
            if math.isfinite(Fr) and Fr + 1e-12 < best_F:
                best_h = make_feasible(hr, extra_n)
                best_F = Fr

    if time.time() < deadline - 18.0:
        nfin = int(best_h.size)
        limit = deadline - 8.0
        it = int(max(60, min(1200, 50000.0 / max(2, nfin))))
        hfinal, Ffinal = local_optimize(best_h, nfin, 250.0, limit, it)
        if math.isfinite(Ffinal) and Ffinal + 1e-12 < best_F:
            best_h = make_feasible(hfinal, nfin)
            best_F = Ffinal

    final_n = int(best_h.size)
    best_h = make_feasible(best_h, final_n)
    final_F = float(evaluate_corr(best_h, final_n))
    if not math.isfinite(final_F):
        final_F = float(best_F) if math.isfinite(best_F) else 0.5

    return best_h, final_F, final_n
