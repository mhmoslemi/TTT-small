def corr_values(h, n_points):
    return np.correlate(h, 1.0 - h, mode="full")

def gradient_of_corr_shift(s, h, n_points):
    """Compute gradient of c[s] w.r.t. h, where s is the shift index.
    s ranges from -(n-1) to (n-1).
    Note: np.correlate index m corresponds to shift s = m - (n-1).

    for j in range(n):

        if 0 <= j + s < n:

        # Term 2: -h[j-s] if 0 <= j-s < n
        if 0 <= j - s < n:
            grad[j] -= h[j - s]
    return grad

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n_points):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n_points / 2.0
    s = np.sum(h)
    if s < 1e-15:
        h = np.zeros(n_points)
        n_full = int(target)
        h[:n_full] = 1.0
        frac = target - n_full
        if frac > 1e-12 and n_full < n_points:
            h[n_full] = frac
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_grad(h, n, beta):

    cv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(cv)
    if beta * abs(m) > 300:
        idx = int(np.argmax(cv))
        s = idx - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:n])
            grad[s:] -= h[:n - s]
        else:
            p = -s
            grad[p:] += (1.0 - h[:n - p])
            grad[:n - p] -= h[p:n]
        return m * dx, grad * dx
    shifted = beta * (cv - m)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    obj_val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    for idx in range(len(cv)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] += wi * (1.0 - h[s:n])
            grad[s:] -= wi * h[:n - s]
        else:
            p = -s
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] -= wi * h[p:n]
    grad *= dx
    return obj_val, grad


def slsqp_obj(h, n, beta):
    val, _ = smooth_max_and_grad(h, n, beta)
    return val


def slsqp_obj_grad(h, n, beta):
    _, g = smooth_max_and_grad(h, n, beta)
    return g


def sum_constraint(h, n):
    return np.sum(h) - n / 2.0


def sum_constraint_jac(h, n):
    return np.ones(len(h))


def slsqp_optimize(h0, n, beta, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    constraints = [
        {'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}
    ]
    result = minimize(
        slsqp_obj, h0.copy(), args=(n, beta),
        jac=slsqp_obj_grad, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': maxiter, 'ftol': ftol}
    )
    return result


def projected_subgradient(h0, n, max_iter, alpha_init, n_active):
    h = h0.copy()
    best = h.copy()
    best_c5 = compute_c5(h, n)
    alpha = alpha_init
    for it in range(max_iter):
        cv = np.correlate(h, 1.0 - h, mode="full")
        cur_max = np.max(cv) * (2.0 / n)
        if cur_max < best_c5 - 1e-12:
            best = h.copy()
            best_c5 = cur_max
        top_k = min(n_active, len(cv))
        top_idx = np.argsort(cv)[-top_k:]
        grad = np.zeros(n)
        for idx in top_idx:
            s = idx - (n - 1)
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:n])
                grad[s:] -= h[:n - s]
            else:
                p = -s
                grad[p:] += (1.0 - h[:n - p])
                grad[:n - p] -= h[p:n]
        grad /= top_k
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        cv_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        max_new = np.max(cv_new) * (2.0 / n)

        best = h
        best_c5 = c5_final
    return best, best_c5


def binary_local_search(h0, n, max_rounds):
    h = np.copy(h0)
    best_c5 = compute_c5(h, n)
    improved = True
    for round_num in range(max_rounds):
        improved = False
        for i in range(n):
            h_alt = h.copy()
            h_alt[i] = 1.0 - h_alt[i]
            h_alt = project_feasible(h_alt, n)
            c5_alt = compute_c5(h_alt, n)
            if c5_alt < best_c5 - 1e-12:
                h = h_alt
                best_c5 = c5_alt
                improved = True
        for i in range(n - 1):
            for j in range(i + 1, n):
                h_alt = h.copy()
                h_alt[i] = 1.0 - h_alt[i]

                h_alt = project_feasible(h_alt, n)

                if c5_alt < best_c5 - 1e-12:
                    h = h_alt
                    best_c5 = c5_alt
                    improved = True
        if not improved:
            break
    return h, best_c5


def upsample(h, factor):
    n_old = len(h)
    n_new = n_old * factor
    h_new = np.zeros(n_new)
    for i in range(n_old):
        for j in range(factor):
            h_new[i * factor + j] = h[i]
    return h_new
