import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, target_sum):
    """Project onto {0<=h[i]<=1} ∩ {sum(h)=target_sum} via bisection."""
    n = len(h)
    def f(lam):
        return float(np.clip(h - lam, 0.0, 1.0).sum()) - target_sum
    lo, hi = -1.0, 1.0
    if f(lo) < 0:
        lo = -2.0
        while f(lo) < 0:
            lo -= 1.0
    if f(hi) > 0:
        hi = 2.0
        while f(hi) > 0:
            hi += 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_c_k(h, k, dx):
    """Gradient of dx * c(k) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        nf = np.arange(n - k)
        grad[nf] += (1.0 - h[nf + k])
        ns = np.arange(k, n)
        grad[ns] += (-h[ns - k])
    else:
        kk = -k
        nf = np.arange(kk, n)
        grad[nf] += (1.0 - h[nf - kk])
        ns = np.arange(0, n - kk)
        grad[ns] += (-h[ns + kk])
    return grad * dx

def smooth_max_grad(h, n, dx, top_k=5, tau=0.3):
    """Compute gradient of smooth max using top-k shifts."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx
    # Get top_k indices
    top_indices = np.argsort(corr_scaled)[-top_k:][::-1]
    top_vals = corr_scaled[top_indices]
    # Softmax weights
    weights = np.exp((top_vals - top_vals[0]) / tau)
    weights /= weights.sum()
    grad = np.zeros(n)
    for idx, w in zip(top_indices, weights):
        k = idx - (n - 1)
        grad += w * grad_c_k(h, k, dx)
    return grad, corr_scaled
