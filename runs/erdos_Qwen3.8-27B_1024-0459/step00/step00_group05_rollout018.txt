                new_val = np.clip(new_val, 0.0, 1.0)
                if abs(new_val - old_val) < 1e-8:
                    continue
                # Adjust sum: change h[j] by (new_val - old_val), compensate elsewhere
                delta = new_val - old_val
                h[j] = new_val
                # Need to fix sum
                # Distribute -delta to other elements
                others = np.arange(n) != j
                cnt = others.sum()
                if cnt == 0:
                    h[j] = old_val
                    break
                h[others] -= delta / cnt
                h = np.clip(h, 0.0, 1.0)
                # Re-project
                h = project_h(h, n)
                if h[j] != new_val and abs(h[j] - new_val) > 0.05:
                    h = h.copy()
                    h[j] = new_val
                    h = project_h(h, n)
                c5_new = compute_c5(h, n)
                if c5_new < best_c5 - 1e-9:
                    best_c5 = c5_new
                    improved = True
                    old_val = new_val
                    break
                else:
                    h[j] = old_val
                    h = project_h(h, n)
            # Count one pass
        if not improved:
            break
    return h, best_c5

    """Add random noise and re-project."""
    noise = np.random.randn(n) * scale
    h_new = h + noise
    return project_h(h_new, n)

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx

def subgradient(h, n, top_k=5, temp=0.004):
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    max_c = c.max()
    idx = np.argsort(c)[-top_k:]
    w = np.exp((c[idx] - max_c) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, idx):
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += wi * (1.0 - h[s:])
            grad[s:] -= wi * h[:n - s]
        else:
            sa = -s
            grad[sa:] += wi * (1.0 - h[:n - sa])
            grad[:n - sa] -= wi * h[sa:]
    return grad * dx

def project_h(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14 and n > 0:
        j = int(np.argmin(np.abs(h - 0.5)))
        h[j] = float(np.clip(h[j] + diff, 0.0, 1.0))
    return h

    h = project_h(h, n)
    c5_cur = compute_c5(h, n)
    no_improve = 0
    for it in range(max_iter):
        improved = False
        for j in range(n):
            old_val = h[j]
            best_local = c5_cur
            best_local_h = None
            candidates = [0.0, 1.0, 0.5, old_val + 0.1, old_val - 0.1,
                          old_val + 0.05, old_val - 0.05, old_val + 0.02, old_val - 0.02]
            for nv in candidates:
                nv = float(np.clip(nv, 0.0, 1.0))
                if abs(nv - old_val) < 1e-6:
                    continue
                h[j] = nv
                h_new = project_h(h, n)
                c5_try = compute_c5(h_new, n)
                if c5_try < best_local:
                    best_local = c5_try
                    best_local_h = h_new.copy()
            if best_local_h is not None:
                h = best_local_h
                c5_cur = best_local
                improved = True
                no_improve = 0
            else:
                h[j] = old_val
                h = project_h(h, n)
                no_improve += 1
        if not improved or no_improve > 3 * n:
            break
    return h, c5_cur
