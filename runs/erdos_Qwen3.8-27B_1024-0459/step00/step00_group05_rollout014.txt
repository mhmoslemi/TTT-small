def smooth_max_objective(h, n_points, alpha=10.0, penalty=100.0):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    # Smooth max
    shifted = conv - np.max(conv)  # for numerical stability
    logsumexp = np.log(np.sum(np.exp(alpha * shifted))) / alpha + np.max(conv)
    # Penalty for sum constraint
    s = np.sum(h)
    target = n_points / 2.0
    penalty_term = penalty * (s - target) ** 2

def smooth_max_gradient(h, n_points, alpha=10.0, penalty=100.0):

        s = np.sum(x)
        if s < 1e-10:
            x = np.full(n_points, 0.5)
        else:
            x = x * (target_sum / s)
            x = np.clip(x, 0.0, 1.0)
        g, f = smooth_max_grad(x, n_points, alpha)
        # Add projection gradient for sum constraint
        s2 = np.sum(x)
        if s2 > 1e-10:
            g += 50.0 * (s2 - target_sum) / max(s2, 0.1) * np.ones(n_points)
        return f, g
    
    current_alpha = alpha_init
    h = project_feasible(h, n_points)
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        # Anneal alpha
        current_alpha = alpha_init + (alpha_final - alpha_init) * (it / max(max_iter - 1, 1))
        
        x = h.copy()
        res = minimize(
            lambda xx: obj_and_grad(xx, current_alpha)[0],
            x,
            jac=lambda xx: obj_and_grad(xx, current_alpha)[1],
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n_points,
            options={'maxiter': 30, 'gtol': 1e-4}
        )
        h_new = project_feasible(res.x, n_points)
        f_new = smooth_max_val(h_new, n_points, alpha_final)
        if f_new < best_local - 1e-10:
            best_local = f_new
            best_h = h_new.copy()
        elif f_new < best_local + 0.005:
            best_h = h_new.copy()
            best_local = f_new
    return best_h, best_local

def optimize_subgrad(h_start, n_points, time_limit, rng, current_best, lr_init=0.03):
    h = h_start.copy()
    h = project_feasible(h, n_points)
    c5_curr, _ = compute_c5(h, n_points)
    best_local = c5_curr
    best_h = h.copy()
    lr = lr_init
    patience = 0
    top_m = 5
    
    while time.time() < time_limit:
        c5_curr, conv = compute_c5(h, n_points)
        if c5_curr < best_local - 1e-10:
            best_local = c5_curr
            best_h = h.copy()
            patience = 0
            lr = min(lr * 1.01, 0.15)
        else:
            patience += 1
            lr = max(lr * 0.97, 0.002)
        
        if patience > 150:
            # Perturb
            scale = 0.03
            h = h.copy() + rng.uniform(-scale, scale, n_points)
            h = project_feasible(h, n_points)
            patience = 0
            lr = lr_init
            continue
        
        # Subgradient
        grad, conv, idxs = subgrad_max(h, n_points, top_m)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n_points)
        c5_new, _ = compute_c5(h_new, n_points)
        
        if c5_new < c5_curr:
            h = h_new
        else:
            # Try half step
            h_new2 = h - lr * 0.25 * grad
            h_new2 = project_feasible(h_new2, n_points)
            c5_new2, _ = compute_c5(h_new2, n_points)
            if c5_new2 < c5_curr:
                h = h_new2
            else:
                patience += 1
        
        if (int(patience) % 200) == 0 and patience > 0:
            pass
    
    return best_h, best_local

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except:
        h_init = None
    
    if h_init is not None:
        n_parent = len(h_init)
    else:
        n_parent = 70
        h_init = np.full(n_parent, 0.5)
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Phase 1: Optimize at parent resolution
    n1 = n_parent
    h1 = project_feasible(h_init.copy(), n1)
    c5_1, _ = compute_c5(h1, n1)
    best_c5 = c5_1
    best_h = h1.copy()
    best_n = n1
    print(f"Initial n={n1}, C5={c5_1:.6f}", flush=True)
    
    # Smooth max optimization (exploration phase)
    t_lim = t_start + min(200, budget_s * 0.15)
    h_smooth, c5_smooth = optimize_smooth(
        h1, n1, alpha_init=3.0, alpha_final=30.0,
        max_iter=50, time_limit=t_lim, rng=rng, current_best=best_c5
    )
    h_smooth = project_feasible(h_smooth, n1)
    c5_smooth_actual = compute_c5(h_smooth, n1)[0]
    print(f"After smooth opt: C5={c5_smooth_actual:.6f}", flush=True)
    if c5_smooth_actual < best_c5:
        best_c5 = c5_smooth_actual
        best_h = h_smooth.copy()
    
    # Subgradient refinement
    t_lim = t_start + min(500, budget_s * 0.4)
    h_sg, c5_sg = optimize_subgrad(h1, n1, t_lim, rng, best_c5, lr_init=0.03)
    h_sg = project_feasible(h_sg, n1)
    c5_sg = compute_c5(h_sg, n1)[0]
    print(f"After subgrad n={n1}: C5={c5_sg:.6f}", flush=True)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()
    
    # Also try from smooth solution
    if c5_smooth_actual < c5_sg:
        h_sg2, c5_sg2 = optimize_subgrad(h_smooth, n1, t_lim, rng, best_c5, lr_init=0.03)
        h_sg2 = project_feasible(h_sg2, n1)
        c5_sg2 = compute_c5(h_sg2, n1)[0]
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
            print(f"  Better from smooth: C5={c5_sg2:.6f}", flush=True)
    
    # Phase 2: Higher resolution
    elapsed = time.time() - t_start
    if budget_s - elapsed > 150:
        n2 = n1 * 2
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h, period=2.0)
        h2 = project_feasible(h2, n2)
        c5_2, _ = compute_c5(h2, n2)
        print(f"Interpolated n={n2}: C5={c5_2:.6f}", flush=True)
        
        t_lim = t_start + budget_s * 0.65
        h2_opt, c5_2_opt = optimize_subgrad(h2, n2, t_lim, rng, best_c5, lr_init=0.02)
        h2_opt = project_feasible(h2_opt, n2)
        c5_2_opt = compute_c5(h2_opt, n2)[0]
        print(f"Optimized n={n2}: C5={c5_2_opt:.6f}", flush=True)
        if c5_2_opt < best_c5:
            best_c5 = c5_2_opt
            best_h = h2_opt.copy()
            best_n = n2
    
    # Phase 3: Even higher resolution if time permits
    elapsed = time.time() - t_start
    if budget_s - elapsed > 120:
        n3 = best_n * 2
        if n3 <= 560:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n3, endpoint=False)
            h3 = np.interp(x_new, x_old, best_h, period=2.0)
            h3 = project_feasible(h3, n3)
            c5_3, _ = compute_c5(h3, n3)
            
            t_lim = t_start + budget_s * 0.88
            h3_opt, c5_3_opt = optimize_subgrad(h3, n3, t_lim, rng, best_c5, lr_init=0.015)
            h3_opt = project_feasible(h3_opt, n3)
            c5_3_opt = compute_c5(h3_opt, n3)[0]
            print(f"Optimized n={n3}: C5={c5_3_opt:.6f}", flush=True)
            if c5_3_opt < best_c5:
                best_c5 = c5_3_opt
                best_h = h3_opt.copy()
                best_n = n3
    
    # Final: try a few random restarts at best resolution if time remains
    elapsed = time.time() - t_start
    if budget_s - elapsed > 50:
        for restart in range(5):
            if time.time() - t_start > budget_s - 30:
                break
            perturbation = rng.uniform(-0.04, 0.04, best_n)
            h_r = project_feasible(best_h + perturbation, best_n)
            t_lim = t_start + budget_s - 30 - restart * 8
            if t_lim - time.time() < 5:
                break
            h_r_opt, c5_r = optimize_subgrad(h_r, best_n, t_lim, rng, best_c5, lr_init=0.02)
            h_r_opt = project_feasible(h_r_opt, best_n)
            c5_r = compute_c5(h_r_opt, best_n)[0]
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r_opt.copy()
                print(f"  Restart {restart}: C5={c5_r:.6f}", flush=True)
    
    # Ensure final solution is feasible
    best_h = project_feasible(best_h, best_n)
    final_c5, _ = compute_c5(best_h, best_n)
    
    # Verify
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10), "Feasibility violation"
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6, f"Sum constraint violation: {np.sum(best_h)} vs {best_n/2}"
    
    print(f"FINAL: n={best_n}, C5={final_c5:.6f}, time={time.time()-t_start:.1f}s", flush=True)
    
    return best_h, final_c5, best_n
