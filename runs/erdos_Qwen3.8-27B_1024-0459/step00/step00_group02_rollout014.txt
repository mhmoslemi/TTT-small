    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    corr = np.correlate(h_c, 1.0 - h_c, mode="full")

    dx = 2.0 / n
    h_c = np.clip(h, 0.0, 1.0)
