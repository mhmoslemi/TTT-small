        rng = np.random.RandomState(seed)
    else:
        rng = np.random
    h = rng.random(n)
    return project_feasible(h, n)

def binary_random(n, seed=None):
    if seed is not None:
        rng = np.random.RandomState(seed)
    else:
        rng = np.random
    h = np.zeros(n)
    if n % 2 == 0:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    else:
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
        remaining = np.setdiff1d(np.arange(n), idx)
        h[remaining[0]] = 0.5
    return h

def interpolate_h(h_src, n_target):
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_tgt = (np.arange(n_target) + 0.5) / n_target
    return np.interp(x_tgt, x_src, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    try:
        parent_h = initial_h_values
    except NameError:
        parent_h = None
    
    best_h = None
    best_c5 = float('inf')
    best_n = None
    
    def update_best(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n
    
    # Build list of n values to try
    n_candidates = set()
    if parent_h is not None:
        n_candidates.add(len(parent_h))
    for n in [44, 45, 46, 50, 56, 60, 64, 72, 80, 90, 100, 120, 128, 160, 200, 256]:
        n_candidates.add(n)
    n_list = sorted(n_candidates)
    
    # Phase 1: Subgradient on multiple n values
    # Allocate time proportionally (more time for smaller n initially)
    time_per_n = budget_s * 0.45 / len(n_list)
    
    for n in n_list:
        if time.time() - t0 > budget_s * 0.5:
            break
        
        # Create initial guess
        if parent_h is not None and len(parent_h) == n:
            h0 = parent_h.copy()
        elif parent_h is not None:
            h0 = interpolate_h(parent_h, n)
        elif n in (44, 45, 46):
            h0 = binary_random(n, seed=seed + n)
        else:
            h0 = random_feasible(n, seed=seed + n)
        
        h0 = project_feasible(h0, n)
        
        remaining = budget_s * 0.5 - (time.time() - t0)
        this_budget = max(5.0, min(time_per_n, remaining * 0.5))
        
        h_opt, c5 = run_subgradient(h0, n, time_limit=this_budget)
        update_best(h_opt, c5, n)
        
        # Also try smooth gradient descent from this result
        if time.time() - t0 < budget_s * 0.5:
            this_budget2 = max(3.0, min(15.0, budget_s * 0.5 - (time.time() - t0)))
            h_opt2, c5_2 = smooth_grad_descent(h_opt, n, time_limit=this_budget2)
            update_best(h_opt2, c5_2, n)
    
    # Phase 2: Local polishing on best
    if best_h is not None and time.time() - t0 < budget_s * 0.75:
        remain = budget_s * 0.75 - (time.time() - t0)
        h_pol, c5_pol = continuous_local_search(best_h, best_n, time_limit=remain * 0.6, seed=seed)
        update_best(h_pol, c5_pol, best_n)
    
    # Phase 3: Random restarts with subgradient
    if time.time() - t0 < budget_s * 0.92:
        n_try = best_n if best_n is not None else 45
        remain = budget_s * 0.92 - (time.time() - t0)
        num_restarts = max(1, int(remain / 15))
        for r in range(num_restarts):
            if time.time() - t0 > budget_s * 0.94:
                break
            h_rand = random_feasible(n_try, seed=seed + 1000 + r)
            h_opt, c5 = run_subgradient(h_rand, n_try, time_limit=12)
            update_best(h_opt, c5, n_try)
            
            # Also try binary random
            h_bin = binary_random(n_try, seed=seed + 2000 + r)
            h_opt2, c5_2 = run_subgradient(h_bin, n_try, time_limit=8)
            update_best(h_opt2, c5_2, n_try)
    
    # Phase 4: Final polish
    if best_h is not None and time.time() - t0 < budget_s * 0.99:
        remain = budget_s * 0.99 - (time.time() - t0)
        h_final, c5_final = continuous_local_search(best_h, best_n, time_limit=remain, seed=seed + 999)
        update_best(h_final, c5_final, best_n)
    
    if best_h is None:
        n = 45
        best_h = random_feasible(n, seed=seed)
        best_c5 = evaluate_c5(best_h)
        best_n = n
    
    return best_h, best_c5, best_n
