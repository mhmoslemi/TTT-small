    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Water-filling for sum constraint
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_of_max_corr(h):
    """Subgradient of max(corr) with respect to h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    j_star = np.argmax(corr)
    # j = j_star - (n-1) is the shift
    j = j_star - (n - 1)
    grad = np.zeros(n)
    for m in range(n):
        g = 0.0
        # First term: (1 - h[m+j]) if 0 <= m+j < n
        if 0 <= m + j < n:
            g += (1.0 - h[m + j])
        # Second term: -h[m-j] if 0 <= m-j < n
        if 0 <= m - j < n:
            g -= h[m - j]
        grad[m] = g
    return grad
