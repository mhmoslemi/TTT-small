            h_up_new = project_to_feasible(h_up_new, target_sum_up)
            c5_up_new = compute_c5(h_up_new, dx_up)

            if c5_up_new < compute_c5(h_up_best, dx_up) - 1e-12:
                h_up_best = h_up_new.copy()
                h_up = h_up_new
                pat_up = 0
            else:
                pat_up += 1
                if pat_up % 20 == 0:
                    step_up *= 0.8
                if pat_up > 150:
                    noise = np.random.randn(n_up) * 0.02
                    h_up = project_to_feasible(h_up_best + noise, target_sum_up)
                    pat_up = 0

            if step_up < 1e-8:
                break

        c5_up_final = compute_c5(h_up_best, dx_up)
        if c5_up_final < c5_best:
            h_best = h_up_best
            c5_best = c5_up_final
            best_n = n_up

    return h_best, c5_best, best_n
