def random_composition(total,k,rng):
    if total ==0:
        return np.zeros(k,dtype=np.int64)
    if k<=1:
        return np.array([total], dtype=np.int64)
    w=rng.random(k)
    total_w=w.sum()

       lens=np.ones(k); lens[:-1]=0; return lens
    w=w/total_w
    real=w*total

def eval_c(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


def project_box_sum(x, s_target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        ss = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if ss > s_target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(8):
        diff = s_target - float(np.sum(h))
        if abs(diff) < 1e-12:
            break
        interior = np.flatnonzero((h > 1e-11) & (h < 1.0 - 1e-11))
        if interior.size:
            idx = int(interior[0])
            h[idx] += diff
            h[idx] = min(1.0, max(0.0, h[idx]))
        else:
            if abs(h.sum()) > 1e-12:
                h = np.clip(h * (s_target / h.sum()), 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)


def count_max_binary(h):
    h = np.asarray(h, dtype=np.int8)
    hc = h.astype(np.int32)
    vc = (1 - h).astype(np.int32)
    c = np.correlate(hc, vc, mode="full")
    return int(c.max())


def fix_binary_sum(h, s_target):
    h = np.asarray(h, dtype=np.int8).copy()
    cur = int(h.sum())
    if cur > s_target:
        idx = np.flatnonzero(h == 1)[:cur - s_target]
        h[idx] = 0
    elif cur < s_target:
        idx = np.flatnonzero(h == 0)[:s_target - cur]
        h[idx] = 1
    return h


def resample_to_binary(values, n_points, rng):
    values = np.asarray(values, dtype=float)
    if values.size == n_points:
        vals = values
    else:
        oldx = np.linspace(0.0, 1.0, values.size)
        newx = np.linspace(0.0, 1.0, n_points)
        vals = np.interp(newx, oldx, values)
    s_target = n_points // 2
    order = np.argsort(-vals, kind="stable")
    h = np.zeros(n_points, dtype=np.int8)
    h[order[:s_target]] = 1
    return h


def random_balanced(n_points, rng):
    s_target = n_points // 2
    h = np.zeros(n_points, dtype=np.int8)
    if s_target > 0:
        idx = rng.choice(n_points, size=s_target, replace=False)
        h[idx] = 1
    return h


def random_composition(total, k, rng):
    total = int(total)
    k = int(k)
    if total == 0:
        return np.zeros(k, dtype=np.int64)
    if k <= 1:
        return np.array([total], dtype=np.int64)
    w = rng.random(k)
    tw = float(w.sum())
    if tw <= 0.0:
        out = np.zeros(k, dtype=np.int64)
        out[0] = total
        return out
    w = w / tw
    real = w * total
    lens = np.floor(real).astype(np.int64)
    rem = total - int(lens.sum())
    if rem > 0:
        frac = real - lens
        order = np.argsort(-frac, kind="stable")
        for t in range(rem):
            lens[int(order[t % k])] += 1
    return lens

def trend_start(n_points, rng):
    s_target = n_points // 2
    h = np.zeros(n_points, dtype=np.int8)
    h[:s_target] = 1
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)
    m = n_points // 4
    for _ in range(max(1, m)):
        i = int(ones[rng.integers(s_target)])
        j = int(zeros[rng.integers(s_target)])
        h[i] = 0
        h[j] = 1
    return h


def apply_delta(c, off, h, hm1, h1m, i, j):
    n = off + 1
    stop_i = i - 1
    if stop_i < 0:
        stop_i = None
    c[i + off : stop_i : -1] += hm1

    stop_j = j - 1
    if stop_j < 0:
        stop_j = None
    c[j + off : stop_j : -1] += h1m

    c[off + j - i] += 1
    c[off] -= 2

    c[off - i : off - i + n] += h
    c[off - j : off - j + n] -= h
    c[off + i - j] += 1


def undo_delta(c, off, h, hm1, h1m, i, j):
    n = off + 1
    stop_i = i - 1
    if stop_i < 0:
        stop_i = None
    c[i + off : stop_i : -1] -= hm1

    stop_j = j - 1
    if stop_j < 0:
        stop_j = None
    c[j + off : stop_j : -1] -= h1m

    c[off + j - i] -= 1
    c[off] += 2

    c[off - i : off - i + n] -= h
    c[off - j : off - j + n] += h
    c[off + i - j] -= 1


def commit_swap(h, hm1, h1m, ones, zeros, rank1, rank0, i, j):
    # i: 1 -> 0, j: 0 -> 1
    h[i] = 0
    h[j] = 1
    hm1[i] = -1
    hm1[j] = 0
    h1m[i] = 1
    h1m[j] = 0

    r1 = int(rank1[i])
    r0 = int(rank0[j])
    ones[r1] = j
    zeros[r0] = i
    rank1[j] = r1
    rank0[i] = r0
    rank1[i] = -1
    rank0[j] = -1


def active_offender_swap(h, c, max_count, off, rng):
    n = off + 1
    if max_count <= 0:
        return -1, -1
    active = np.where(c >= max_count)[0]
    if active.size == 0:
        return -1, -1
    for _ in range(6):
        idx = int(active[rng.integers(active.size)])
        k = idx - off
        if k > 0:
            d = k
            if d >= n:
                continue
            mask = (h[:n - d] == 0) & (h[d:] == 1)
        elif k < 0:
            d = -k
            if d <= 0 or d >= n:
                continue
            mask = (h[:-d] == 1) & (h[d:] == 0)
        else:
            continue
        pos = np.flatnonzero(mask)
        if pos.size:
            p = int(pos[rng.integers(pos.size)])
            if k > 0:
                return p + d, p
            else:
                return p, p + d
    return -1, -1


def init_binary_state(h_start):
    h_start = np.asarray(h_start, dtype=np.int8)
    n = int(h_start.size)
    s_target = n // 2
    h = fix_binary_sum(h_start.copy(), s_target)

    ones = np.flatnonzero(h == 1).astype(np.int32)
    zeros = np.flatnonzero(h == 0).astype(np.int32)

    rank1 = np.full(n, -1, dtype=np.int32)
    rank0 = np.full(n, -1, dtype=np.int32)
    rank1[ones] = np.arange(s_target, dtype=np.int32)
    rank0[zeros] = np.arange(s_target, dtype=np.int32)

    hm1 = (h - 1).astype(np.int8)
    h1m = (1 - h).astype(np.int8)

    hc = h.astype(np.int32)
    vc = (1 - h).astype(np.int32)
    c = np.correlate(hc, vc, mode="full").astype(np.int32)
    max_count = int(c.max())
    return h, ones, zeros, rank1, rank0, hm1, h1m, c, max_count


def local_search_binary(h_start, deadline, rng):
    n = int(np.asarray(h_start).size)
    off = n - 1
    s_target = n // 2

    h, ones, zeros, rank1, rank0, hm1, h1m, c, max_count = init_binary_state(h_start)
    best_m = max_count
    best_h = h.copy()

    if n <= 90:
        batch = 90
    elif n <= 150:
        batch = 55
    else:
        batch = 32

    max_stall = 12 * batch
    stall = 0
    perturb = 3

    while time.time() < deadline:
        best_new = max_count
        bi = -1
        bj = -1

        ai, aj = active_offender_swap(h, c, max_count, off, rng)
        if ai >= 0 and aj >= 0:
            apply_delta(c, off, h, hm1, h1m, ai, aj)
            nm = int(c.max())
            if nm < best_new:
                best_new = nm
                bi = ai
                bj = aj
            undo_delta(c, off, h, hm1, h1m, ai, aj)

        for _ in range(batch - 1):
            i = int(ones[rng.integers(s_target)])
            j = int(zeros[rng.integers(s_target)])
            apply_delta(c, off, h, hm1, h1m, i, j)
            nm = int(c.max())
            if nm < best_new:
                best_new = nm
                bi = i
                bj = j
            undo_delta(c, off, h, hm1, h1m, i, j)

        if bi >= 0 and best_new < max_count:
            apply_delta(c, off, h, hm1, h1m, bi, bj)
            commit_swap(h, hm1, h1m, ones, zeros, rank1, rank0, bi, bj)
            max_count = best_new
            stall = 0
            if max_count < best_m:
                best_m = max_count
                best_h = h.copy()
        else:
            stall += batch
            if stall >= max_stall:
                for _ in range(perturb):
                    i = int(ones[rng.integers(s_target)])
                    j = int(zeros[rng.integers(s_target)])
                    apply_delta(c, off, h, hm1, h1m, i, j)
                    commit_swap(h, hm1, h1m, ones, zeros, rank1, rank0, i, j)
                c = np.correlate(
                    h.astype(np.int32),

                max_count = int(c.max())

                if max_count > best_m + max(3, n // 40):
                    h, ones, zeros, rank1, rank0, hm1, h1m, c, max_count = init_binary_state(best_h)
                stall = 0
                perturb += 1
                if perturb > 14:
                    perturb = 14

    return best_h, best_m


def solve_cvxpy_fixed(n_points, target_t, time_limit):
    try:
        import cvxpy as cp
    except Exception:
        return None

    n_points = int(n_points)
    target_t = int(target_t)
    if n_points % 2 != 0 or n_points < 10:
        return None
    s_target = n_points // 2
    if target_t < 0:
        return None
    if target_t >= s_target:
        return random_balanced(n_points, np.random.default_rng(1234567))

    r = n_points - target_t - 1
    if r < 1:
        return None
    if r > n_points - 1:
        r = n_points - 1

    mz = r * n_points - r * (r + 1) // 2
    if mz <= 0:
        return random_balanced(n_points, np.random.default_rng(1234567))

    pair_z = {}
    idx = 0
    for i in range(n_points - 1):
        maxj = min(n_points, i + r + 1)
        for j in range(i + 1, maxj):
            pair_z[(i, j)] = idx
            idx += 1

    try:
        y = cp.Variable(n_points, boolean=True)
        z = cp.Variable(mz)
        cons = [cp.sum(y) == s_target]

        for (i, j), zi in pair_z.items():
            cons.append(z[zi] <= y[i])
            cons.append(z[zi] <= y[j])
            cons.append(z[zi] >= y[i] + y[j] - 1)

        for d in range(1, r + 1):
            pos_y = []
            pos_z = []
            neg_y = []
            for i in range(n_points - d):
                zi = pair_z[(i, i + d)]
                pos_y.append(y[i + d])
                pos_z.append(z[zi])
                neg_y.append(y[i])
            cons.append(cp.sum(pos_y) - cp.sum(pos_z) <= target_t)
            cons.append(cp.sum(neg_y) - cp.sum(pos_z) <= target_t)

        prob = cp.Problem(cp.Minimize(cp.sum(z)), cons)
        tl = max(1, int(time_limit * 0.92))
        try:
            prob.solve(solver=cp.CBC, max_time=tl)
        except Exception:
            prob.solve(solver=cp.CBC, solver_options={"timeLimit": tl})

        if y.value is None:
            return None
        yy = np.asarray(y.value).astype(float)
        out = (yy[:n_points] > 0.5).astype(np.int8)
        out = fix_binary_sum(out, s_target)
        if count_max_binary(out) <= target_t:

    except Exception:

    return None

        )
        beq = np.array([float(s_target)], dtype=float)

        bounds = Bounds(np.zeros(var_n, dtype=float), np.ones(var_n, dtype=float))
        integrality = np.zeros(var_n, dtype=int)
        integrality[:n_points] = 1
        obj = np.zeros(var_n, dtype=float)

        cons = LinearConstraint(a, -np.inf, b)
        try:
            res = milp(
                obj,
                constraints=cons,
                integrality=integrality,
                bounds=bounds,
                options={
                    "time_limit": max(1, int(time_limit * 0.92)),
                    "mip_gap": 0.0,
                    "presolve": True
                }
            )
        except Exception:
            return solve_cvxpy_fixed(n_points, target_t, time_limit)

        if res is None or res.x is None:
            return None

        yy = np.asarray(res.x)[:n_points]
        out = (yy > 0.5).astype(np.int8)
        out = fix_binary_sum(out, s_target)
        if count_max_binary(out) <= target_t:
            return out
    except Exception:
        return solve_cvxpy_fixed(n_points, target_t, time_limit)

    return None


def fractional_local(h_start, deadline, rng):
    h = np.asarray(h_start, dtype=float).copy()
    n = h.size
    if n == 0:
        return h, eval_c(h)
    s_target = float(n) / 2.0

    if not (np.all(h >= -1e-9) and np.all(h <= 1.0 + 1e-9)):
        h = project_box_sum(h, s_target)
    else:
        h = np.clip(h, 0.0, 1.0)
        if abs(float(h.sum()) - s_target) > 1e-7:
            h = project_box_sum(h, s_target)

    best_h = h.copy()
    best_c = eval_c(h)
    off = n - 1
    no_improve = 0

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_count = int(corr.max())
        if max_count <= 0:
            break
        active = np.where(corr >= max_count)[0]
        if active.size == 0:
            break
        idx = int(active[rng.integers(active.size)])
        k = idx - off

        cands = []
        if k > 0:
            d = k
            if d < n:
                w = (1.0 - h[:n - d]) * h[d:]
                if w.size and w.max() > 1e-8:
                    p = int(np.argmax(w))
                    i = p
                    j = p + d
                    for s in (0.10, 0.20, 0.30, 0.40, 0.50, 0.65, 0.80):
                        if h[i] + s <= 1.0 + 1e-12 and h[j] - s >= -1e-12:
                            cands.append((i, j, s))
        elif k < 0:
            d = -k
            if d < n:
                w = h[:-d] * (1.0 - h[d:])
                if w.size and w.max() > 1e-8:
                    p = int(np.argmax(w))
                    i = p
                    j = p + d
                    for s in (0.10, 0.20, 0.30, 0.40, 0.50, 0.65, 0.80):
                        if h[i] - s >= -1e-12 and h[j] + s <= 1.0 + 1e-12:
                            cands.append((i, j, s))

        for _ in range(2):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            s = 0.05 * (1.0 if rng.random() < 0.5 else -1.0)
            if -1e-12 <= h[i] + s <= 1.0 + 1e-12 and -1e-12 <= h[j] - s <= 1.0 + 1e-12:
                cands.append((i, j, s))

        best_candidate = None
        best_candidate_c = best_c
        for i, j, s in cands:
            oi = h[i]
            oj = h[j]
            h[i] = oi + s
            h[j] = oj - s
            cc = eval_c(h)
            h[i] = oi
            h[j] = oj
            if cc < best_candidate_c - 1e-12:
                best_candidate_c = cc
                best_candidate = (i, j, s)

        if best_candidate is not None:
            i, j, s = best_candidate
            h[i] += s
            h[j] -= s
            h = np.clip(h, 0.0, 1.0)
            if abs(float(h.sum()) - s_target) > 1e-8:
                h = project_box_sum(h, s_target)
            no_improve = 0
            if best_candidate_c < best_c:
                best_c = best_candidate_c
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 250:
                no_improve = 0
                for _ in range(4):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    s = 0.07 * (1.0 if rng.random() < 0.5 else -1.0)
                    if -1e-12 <= h[i] + s <= 1.0 + 1e-12 and -1e-12 <= h[j] - s <= 1.0 + 1e-12:
                        h[i] += s
                        h[j] -= s
                h = np.clip(h, 0.0, 1.0)
                if abs(float(h.sum()) - s_target) > 1e-8:
                    h = project_box_sum(h, s_target)

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0
    best_count_by_n = {}

    current_record = 0.38085
    parent = globals().get("initial_h_values", None)
    parent_proc = None

    if parent is not None:
        p_raw = np.asarray(parent, dtype=float)
        n_par = int(p_raw.size)
        s_par = float(n_par) / 2.0
        if (np.all(p_raw >= -1e-8) and np.all(p_raw <= 1.0 + 1e-8)
                and abs(float(p_raw.sum()) - s_par) <= 1e-6):
            p = np.clip(p_raw, 0.0, 1.0)
        else:
            p = project_box_sum(p_raw, s_par)
        parent_proc = p.copy()

        c_par = eval_c(p)
        best_c = c_par
        best_h = p.copy()
        best_n = n_par

        if n_par % 2 == 0 and n_par >= 20:
            pb = resample_to_binary(p, n_par, rng)
            cnt = count_max_binary(pb)
            cb = float(cnt) * (2.0 / n_par)
            if cb < current_record:
                best_count_by_n[n_par] = cnt
            if cb < best_c - 1e-12:
                best_c = cb
                best_h = pb.astype(float)
                best_n = n_par

    if best_h is None:
        n70 = 70
        h70 = random_balanced(n70, rng)
        cnt = count_max_binary(h70)
        best_c = float(cnt) * (2.0 / n70)
        best_h = h70.astype(float)
        best_n = n70
        best_count_by_n[n70] = cnt

    base_n = int(parent.size) if parent is not None else 70
    if base_n % 2 != 0 or base_n < 36:
        base_n = 70

    candidate_ns = [40, 48, 54, 60, 64, 70, 72, 80, 84, 90, 100, 108, 120, 140]
    for dn in (-10, -8, -6, -4, -2, 0, 2, 4, 6, 8, 10, 12):
        nn = base_n + dn
        if nn is not None and nn >= 36 and nn <= 200 and nn % 2 == 0:
            candidate_ns.append(nn)

    seen = set()
    total_ns = []
    for n in candidate_ns:
        if n not in seen and n >= 36 and n <= 220 and n % 2 == 0:
            seen.add(int(n))
            total_ns.append(int(n))

    order_items = []
    for n in total_ns:
        order_items.append((abs(n - base_n), n))
    order_items.sort()
    ordered_ns = [n for _, n in order_items]

    local_deadline = t0 + min(760.0, float(budget_s) * 0.58)
    if local_deadline > t0 + max(60.0, float(budget_s) - 380.0):
        local_deadline = t0 + max(60.0, float(budget_s) - 380.0)
    if local_deadline < t0 + 30.0:
        local_deadline = t0 + min(20.0, float(budget_s))

    for n in ordered_ns:
        if time.time() >= local_deadline - 5.0:
            break
        if n < 36 or n > 220:
            continue

        if n == base_n:
            n_allocate = 260.0
        elif n <= 90:
            n_allocate = 120.0
        elif n <= 140:
            n_allocate = 70.0
        else:
            n_allocate = 35.0

        rem = local_deadline - time.time()
        n_allocate = min(n_allocate, rem)
        if n_allocate < 15.0:
            continue

        starts = []
        if parent_proc is not None:
            starts.append(resample_to_binary(parent_proc, n, rng))
        starts.append(random_balanced(n, rng))
        starts.append(clump_start(n, rng))
        starts.append(trend_start(n, rng))
        if n > 130:
            starts = starts[:2]

        per = max(8.0, min(70.0, n_allocate / max(1, len(starts))))
        for st in starts:
            if time.time() >= local_deadline - 3.0:
                break
            dl = min(local_deadline, time.time() + per)
            if dl - time.time() < 5.0:
                break
            bh, bm = local_search_binary(st, dl, rng)
            bh = np.asarray(bh, dtype=np.int8)
            actual = count_max_binary(bh)
            cc = float(actual) * (2.0 / n)
            if n not in best_count_by_n or actual < best_count_by_n[n]:
                best_count_by_n[n] = actual
            if cc < best_c - 1e-12:
                best_c = cc
                best_h = bh.astype(float)
                best_n = n

    milp_deadline = min(t0 + float(budget_s) - 50.0, t0 + float(budget_s) * 0.92)
    milp_candidates = [base_n]
    for dn in (10, 20, -10, -20):
        nn = base_n + dn
        if nn >= 56 and nn <= 120 and nn % 2 == 0:
            milp_candidates.append(nn)
    for nn in (70, 80, 100, 60, 120):
        if 56 <= nn <= 120 and nn % 2 == 0:
            milp_candidates.append(nn)

    seen2 = set()
    milp_ns = []
    for n in milp_candidates:
        if n not in seen2:
            seen2.add(n)
            milp_ns.append(n)
    milp_ns.sort(reverse=True)
    if base_n in milp_ns:
        milp_ns.remove(base_n)
        milp_ns.insert(0, base_n)

    for n in milp_ns:
        if time.time() >= milp_deadline - 15.0:
            break
        if n < 56 or n > 120 or n % 2 != 0:
            continue

        lb = int(math.ceil(float(n * n) / (4.0 * (2 * n - 1))))
        target = int(math.floor(best_c * n / 2.0 - 1e-9))
        known = best_count_by_n.get(n, n // 2 + 1)
        if target >= known:
            target = known - 1
        if target < lb:
            continue

        tcur = target
        while tcur >= lb and time.time() < milp_deadline - 10.0:
            tl = min(220.0, milp_deadline - time.time())
            if tl < 25.0:
                break
            yh = solve_milp_fixed(n, tcur, tl)
            if yh is not None:
                yh = np.asarray(yh, dtype=np.int8)
                actual = count_max_binary(yh)
                cc = float(actual) * (2.0 / n)
                if n not in best_count_by_n or actual < best_count_by_n[n]:
                    best_count_by_n[n] = actual
                if cc < best_c - 1e-12:
                    best_c = cc
                    best_h = yh.astype(float)
                    best_n = n
                tcur -= 1
            else:
                break

    final_hard_deadline = t0 + float(budget_s) - 5.0
    if time.time() < final_hard_deadline - 20.0:
        frac_starts = []
        if parent_proc is not None and parent_proc.size <= 130:
            frac_starts.append(parent_proc)
        if best_h is not None and best_h.size <= 130:
            bh = np.asarray(best_h, dtype=float)
            if bh.size % 2 == 0:
                frac_starts.append(bh)

        seen3 = set()
        for st in frac_starts:
            if time.time() >= final_hard_deadline - 12.0:
                break
            key = (st.size, round(float(np.sum(st)), 6))
            if key in seen3:
                continue
            seen3.add(key)
            dl = min(final_hard_deadline, time.time() + 55.0)
            if dl - time.time() < 8.0:
                continue
            fb, fc = fractional_local(st, dl, rng)
            if fc < best_c - 1e-12:
                best_c = fc
                best_h = np.asarray(fb, dtype=float)
                best_n = int(best_h.size)

    if best_h is None:
        best_h = random_balanced(70, rng).astype(float)
        best_n = 70
        best_c = eval_c(best_h)

    best_h = np.asarray(best_h, dtype=float)
    n_out = int(best_h.size)
    s_out = float(n_out) / 2.0

    integer_like = (
        np.all((np.abs(best_h) < 1e-7) | (np.abs(best_h - 1.0) < 1e-7))
        and n_out % 2 == 0
    )
    if integer_like:
        snapped = (best_h > 0.5).astype(float)
        if abs(float(snapped.sum()) - s_out) <= 1e-6:
            best_h = snapped
            best_c = eval_c(best_h)
        else:
            best_h = project_box_sum(best_h, s_out)
            best_h = np.clip(best_h, 0.0, 1.0)
            best_c = eval_c(best_h)
    else:
        if np.all(best_h >= -1e-8) and np.all(best_h <= 1.0 + 1e-8) and abs(float(best_h.sum()) - s_out) <= 1e-6:
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h = project_box_sum(best_h, s_out)
            best_h = np.clip(best_h, 0.0, 1.0)
        best_c = eval_c(best_h)

    return best_h, float(best_c), n_out
