import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    # Water-filling: find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    lo = h.min() - target_sum / len(h)
    hi = h.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_grad(h, n):
    """Compute gradient of the max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    m_star = np.argmax(c)
    k = m_star - (n - 1)
    
    grad = np.zeros(n)
    # First part: (1-h[i-k]) for valid i
    i_start = max(0, k)
    i_end = min(n - 1, k + n - 1)
    if i_start <= i_end:
        j = np.arange(i_start, i_end + 1) - k
        grad[i_start:i_end+1] += (1 - h[j])
    
    # Second part: -h[i+k] for valid i
    i_start2 = max(0, -k)
    i_end2 = min(n - 1, n - 1 - k)
    if i_start2 <= i_end2:
        j2 = np.arange(i_start2, i_end2 + 1) + k
        grad[i_start2:i_end2+1] -= h[j2]
    
    return dx * grad

def compute_smooth_grad(h, n, eps):
    """Compute gradient of the smoothed max."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    c_shifted = c - np.max(c)
    p = np.exp(c_shifted / eps)
    p /= p.sum()
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-15:
            continue
        k = m - (n - 1)
        # First part
        i_start = max(0, k)
        i_end = min(n - 1, k + n - 1)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1) - k
            grad[i_start:i_end+1] += p[m] * (1 - h[j])
        # Second part
        i_start2 = max(0, -k)
        i_end2 = min(n - 1, n - 1 - k)
        if i_start2 <= i_end2:
            j2 = np.arange(i_start2, i_end2 + 1) + k
            grad[i_start2:i_end2+1] -= p[m] * h[j2]
    
    return dx / eps * grad

def optimize_projected(h0, n, max_iters=5000, time_limit=60):
    """Optimize using projected subgradient / gradient method."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    
    start = time.time()
    best_local_c5 = compute_c5(h)
    best_local_h = h.copy()
    
    # Phase 1: smooth max with decreasing epsilon
    epsils = [0.5, 0.1, 0.02, 0.005, 0.001, 0.0002]
    
    for eps in epsils:
        if time.time() - start > time_limit:
            break
        
        n_iter = max(50, int(3000 * eps / 0.5))
        lr = 0.01 * eps  # initial learning rate
        
        for it in range(n_iter):
            if time.time() - start > time_limit:
                break
            
            g = compute_smooth_grad(h, n, eps)
            
            # Step
            h_new = h - lr * g
            h_new = project_to_constraint(h_new, target_sum)
            
            # Check improvement
            c5_new = compute_c5(h_new)
            if c5_new < best_local_c5:
                best_local_c5 = c5_new
                best_local_h = h_new.copy()
            
            h = h_new
            
            # Decay learning rate
            lr *= 0.999
            if lr < 1e-8:
                lr = 1e-8
    
    return best_local_h, best_local_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_constraint(h, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    c_max = c.max()

                best_h = h_new.copy()
            
            h = h_new
            
            # Decay learning rate
            lr *= 0.999
            
            # Every 100 iters, evaluate and optionally restart
            if it % 100 == 0:
                pass  # no restart for now
    
    return best_h, best_c5
