def compute_C5(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    """Project to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h.copy()
    if s > target:
        excess = s - target
        # Reduce from largest values
        idx = np.argsort(-h)
        for i in idx:
            if excess <= 1e-12:
                break
            take = min(h[i], excess)
            h[i] -= take
            excess -= take
    else:
        deficit = target - s
        idx = np.argsort(h)
        for i in idx:
            if deficit <= 1e-12:
                break
            take = min(1.0 - h[i], deficit)
            h[i] += take
            deficit -= take
    return h

def make_binary(h, n):
    """Convert to a 0-1 vector with exactly n/2 ones."""
    h = np.clip(h, 0, 1)
    # Sort by value, take the top n/2 as 1s
    threshold = n / 2
    # Simple approach: sort and threshold
    sorted_idx = np.argsort(-h)
    result = np.zeros(n)
    # Put 1s at the top n/2 positions
    for i in range(n // 2):
        result[sorted_idx[i]] = 1.0
    return result

def local_search_swaps(h, n, dx, rng, time_limit, start_time, verbose=False):
    """Local search using swap moves on 0-1 vectors."""
    target = n / 2
    assert abs(np.sum(h) - target) < 1e-6, f"Sum is {np.sum(h)}, target is {target}"
    
    best_c5 = compute_C5(h, dx)
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    no_improve_count = 0
    max_no_improve = 50  # Restart after this many sweeps without improvement
    
    sweep = 0
    while time.time() - start_time < time_limit:
        sweep += 1
        improved = False
        
        # Shuffle orders
        ones_order = rng.permutation(len(ones))
        zeros_order = rng.permutation(len(zeros))
        
        for oi in ones_order:
            j = ones[oi]
            for zi in zeros_order:
                m = zeros[zi]
                # Try swap: h[j] 1→0, h[m] 0→1
                h[j] = 0.0
                h[m] = 1.0
                c5 = compute_C5(h, dx)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    break
                else:
                    h[j] = 1.0
                    h[m] = 0.0
            
            if time.time() - start_time > time_limit:
                break
        
        if not improved:
            no_improve_count += 1
            if no_improve_count >= max_no_improve:
                break
        else:
            no_improve_count = 0
    
    return h, best_c5

A "best improvement" strategy would evaluate all swaps and pick the best one, but that's O(n²) evaluations per sweep instead of O(n²) worst case. The first improvement is faster on average.

def kick(h, n, rng, strength=2):
    """Randomly perturb the solution."""
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for _ in range(strength):
        j = ones[rng.integers(len(ones))]
        m = zeros[rng.integers(len(zeros))]
        h[j], h[m] = h[m], h[j]
    return h
