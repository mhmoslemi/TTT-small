def active_gradient(h, f, top_k=8, tol=1e-9):
    n=len(h); m=len(f)
    current_max = f.max()
    # choose top k
    if top_k is None or top_k >= m: top_idx = np.argsort(f)[-max(1,min(m,top_k)):]
    else: top_idx = np.argpartition(f, -top_k)[-top_k:]
    # weights maybe uniform or softmax: w = 1/(len)
    G = np.zeros(n)
    for idx in top_idx:
        l = idx - (n-1)
        w=1.0/len(top_idx)

           L = n-l

           G[a:] += w*(1.0 - h[:n-a])

    return G

def random_binary(n, seed):
    rng = np.random.default_rng(seed)
    b = np.zeros(n, dtype=float)
    if n % 2 == 0:
       idx = rng.choice(n, n//2, replace=False)
       b[idx]=1.0
    else:

       rest = np.where(b==0)[0]
       if rest.size:
          b[rest[rng.integers(rest.size)]] = 0.5

def build_f(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.zeros(0, dtype=float)
    half = n - 1
    f = np.empty(2 * n - 1, dtype=float)
    q = 1.0 - h
    f[half] = float(np.dot(h, q))
    for l in range(1, n):
        r = n - l
        f[half + l] = float(np.dot(h[:r], q[l:]))
        f[half - l] = float(np.dot(h[l:], q[:r]))
    return f


def c5_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / float(n))


def active_gradient(h, f, top_k=12):
    n = h.size
    m = f.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=float)
    if top_k <= 0:
        top_k = 1
    top_k = int(min(top_k, m))
    if top_k >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(f, m - top_k)[m - top_k:]

    half = n - 1
    G = np.zeros(n, dtype=float)
    for ix in idx:
        l = int(ix) - half
        w = 1.0 / float(len(idx))
        if l >= 0:
            r = n - l
            if r <= 0:
                continue
            G[:r] += w * (1.0 - h[l:])
            G[l:] -= w * h[:r]
        else:
            a = -l
            r = n - a
            if r <= 0:
                continue
            G[a:] += w * (1.0 - h[:r])
            G[:r] -= w * h[a:]

    scale = float(np.max(np.abs(G)))
    if scale > 0.0:
        G /= scale
    return np.nan_to_num(G, nan=0.0, posinf=0.0, neginf=0.0)


def compute_pair_delta(h, p, q, delta, D):
    D.fill(0.0)
    if delta == 0.0 or p == q:
        return D

    n = h.size
    half = n - 1
    om = 1.0 - h
    rev = h[::-1]

    D[half - p:half - p + n] += delta * om
    D[p:p + n] -= delta * rev

    nd = -delta
    D[half - q:half - q + n] += nd * om
    D[q:q + n] -= nd * rev

    extra = delta * delta
    D[q - p + half] += extra
    D[p - q + half] += extra
    D[half] -= 2.0 * extra
    return D


def make_dyadic_sum_exact(h, n, p=20):

    target = int(round(0.5 * n * d))
    h = np.clip(h, 0.0, 1.0)
    k = np.rint(h * d).astype(np.int64)
    np.clip(k, 0, 1 << p, out=k)

    diff = target - int(k.sum())
    resid = h * d - k.astype(float)
    if diff > 0:
        order = np.argsort(-resid)
    else:
        order = np.argsort(resid)

                    diff -= 1
                    if diff == 0:
                        break
            else:
                if k[idx] > 0:
                    k[idx] -= 1
                    diff += 1
                    if diff == 0:
                        break

    return k.astype(float) / d

    best_h = h.copy()
    best_f = f.copy()

    rng = np.random.default_rng(seed)
    move_scale = 0.16
    temp = max(1e-3, 0.02 * max(1.0, curr))
    no_improve = 0
    total = 0
    G = active_gradient(h, f, 12)
    batch = 128

    while time.perf_counter() < deadline - 0.2:
        for _ in range(batch):
            if time.perf_counter() > deadline - 0.2:
                break

            total += 1
            if total % 8 == 0:
                G = active_gradient(h, f, 12)

            gmax = float(np.max(np.abs(G)))
            use_grad = gmax > 1e-12

            if use_grad:
                tsize = max(4, min(96, n // 6))
                if tsize < n:
                    top = np.argpartition(G, n - tsize)[n - tsize:]
                    bot = np.argpartition(G, tsize - 1)[:tsize]
                else:
                    top = np.arange(n)
                    bot = np.arange(n)

                donor = -1
                receiver = -1
                for __ in range(10):
                    d_candidate = int(top[int(rng.integers(top.size))])
                    r_candidate = int(bot[int(rng.integers(bot.size))])
                    if d_candidate != r_candidate:
                        donor = d_candidate
                        receiver = r_candidate
                        break
                if donor < 0:
                    donor = int(rng.integers(n))
                    receiver = (donor + 1) % n
            else:
                donor = int(rng.integers(n))
                receiver = int(rng.integers(n))
                if donor == receiver:
                    receiver = (receiver + 1) % n

            avail = min(float(h[donor]), float(1.0 - h[receiver]))
            if avail <= 1e-13:
                found = False
                for __ in range(10):
                    r2 = int(rng.integers(n))
                    if r2 != donor:
                        av2 = min(float(h[donor]), float(1.0 - h[r2]))
                        if av2 > 1e-13:
                            receiver = r2
                            avail = av2
                            found = True
                            break
                if not found:
                    donor = int(rng.integers(n))
                    receiver = int(rng.integers(n))
                    if donor == receiver:
                        receiver = (receiver + 1) % n
                    avail = min(float(h[donor]), float(1.0 - h[receiver]))
                    if avail <= 1e-13:
                        continue

            if use_grad:
                gd = float(G[donor] - G[receiver])
                if gd >= 0.0:
                    factor = min(1.0, gd)
                else:
                    if rng.random() < 0.65:
                        donor, receiver = receiver, donor
                        avail = min(float(h[donor]), float(1.0 - h[receiver]))
                        if avail <= 1e-13:
                            continue
                        gd = -gd
                        factor = min(1.0, gd)
                    else:
                        factor = 0.0
                amt = avail * move_scale * (0.1 + 1.9 * float(rng.random())) * (0.25 + 0.75 * factor)
            else:
                amt = avail * move_scale * (0.1 + 1.9 * float(rng.random()))

            if amt > avail:
                amt = avail
            if amt < 1e-7:
                if avail > 1e-7:
                    amt = min(avail, 1e-5)
                else:
                    continue

            compute_pair_delta(h, donor, receiver, -amt, D)
            f += D
            new_curr = float(f.max())
            diff = new_curr - curr

            if diff < -1e-12:
                accept = True
            elif diff < 1e-12:
                accept = (float(rng.random()) < 0.05)
            else:
                if diff < temp:
                    accept = (float(rng.random()) < math.exp(-diff / temp))
                else:
                    accept = False

            if accept:
                h[donor] -= amt
                h[receiver] += amt

                if h[donor] < -1e-9 or h[receiver] > 1.0 + 1e-9:
                    h = make_feasible(h)
                    f = build_f(h)
                    new_curr = float(f.max())
                else:
                    if h[donor] < 0.0:
                        h[donor] = 0.0
                    if h[receiver] > 1.0:
                        h[receiver] = 1.0

                curr = new_curr
                if curr < best - 1e-12:
                    best = curr
                    best_h = h.copy()
                    best_f = f.copy()
                    no_improve = 0
                    move_scale = min(0.5, move_scale * 1.08)
                else:
                    no_improve += 1
            else:
                f -= D
                no_improve += 1

        if no_improve > 180:
            move_scale = max(0.005, move_scale * 0.85)
            temp = max(1e-4, temp * 0.95)
            no_improve = 0

            if move_scale < 0.02:
                perturb = max(4, n // 16)
                for __ in range(perturb):
                    d_candidate = int(rng.integers(n))
                    r_candidate = int(rng.integers(n))
                    if d_candidate != r_candidate:
                        a_candidate = min(float(h[d_candidate]), float(1.0 - h[r_candidate]))
                        a_candidate *= float(rng.random())
                        if a_candidate > 0.0:
                            h[d_candidate] -= a_candidate
                            h[r_candidate] += a_candidate
                h = make_feasible(h)
                f = build_f(h)
                curr = float(f.max())
                if curr < best - 1e-12:
                    best = curr
                    best_h = h.copy()
                    best_f = f.copy()
                if curr > best + max(1.0, 0.5 * best):
                    h = best_h.copy()
                    f = best_f.copy()
                    curr = best
                move_scale = max(0.08, move_scale * 3.0)
                temp = max(1e-3, 0.02 * max(1.0, curr))

        if total % 5000 == 0:
            f_check = build_f(h)
            curr_check = float(f_check.max())
            if abs(curr_check - curr) > 1e-8 or curr_check < curr - 1e-12:
                f = f_check
                curr = curr_check
                if curr < best - 1e-12:
                    best = curr
                    best_h = h.copy()
                    best_f = f.copy()

        if time.perf_counter() > deadline - 0.2:
            break

    return best_h, best_f, float(best)


def candidate_by_kind(n, kind, seed):
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)
    if n % 2 != 0:
        n += 1

    if kind == "random":
        return random_binary(n, seed)
    if kind == "lowdisc":
        return low_disc_binary(n, seed)
    if kind == "sine1":
        return sine_candidate(n, seed, freq=1.0, phase=0.0)
    if kind == "sine2":
        return sine_candidate(n, seed, freq=2.0, phase=math.pi / 2.0)
    if kind == "block":
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h = 0.5 + 0.5 * np.sin(math.pi * 3.0 * x)
        return make_feasible(h)
    return linspace_candidate = None  # unused


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    total = max(10.0, float(budget_s))
    start = time.perf_counter()
    deadline = start + max(1.0, total - 5.0)

    best_h = None
    best_n = 0
    best_c = float("inf")

    parent = None
    try:
        parent = globals().get("initial_h_values", None)
    except Exception:
        parent = None

    if parent is not None:
        parent_arr = np.asarray(parent, dtype=float).ravel()
        if parent_arr.size > 0:
            clip_parent = np.clip(parent_arr, 0.0, 1.0)
            if abs(float(clip_parent.sum()) - 0.5 * parent_arr.size) < 1e-5:
                best_h, best_n, best_c = update_best(best_h, best_n, best_c, parent_arr, parent_arr.size)
            h_base = make_feasible(parent_arr)
            n_base = int(parent_arr.size)
        else:
            h_base = random_binary(80, seed)
            n_base = 80
    else:
        h_base = low_disc_binary(80, seed)
        n_base = 80

    if n_base < 2:
        n_base = 80
        h_base = random_binary(n_base, seed)

    h_base = make_feasible(h_base)
    best_h, best_n, best_c = update_best(best_h, best_n, best_c, h_base, n_base)

    base_n = int(best_n) if best_n > 0 else n_base
    base_n = max(8, base_n)
    if base_n % 2 != 0:
        base_n += 1

    target_list = []
    seen = set()

    def add_target(n_cand):
        n_cand = int(n_cand)
        if n_cand < 16:
            n_cand = 16
        if (n_cand % 2) != 0:
            n_cand += 1
        if n_cand <= 2048 and n_cand not in seen:
            seen.add(n_cand)
            target_list.append(n_cand)

    add_target(base_n)
    for mult in (2, 4, 8):
        add_target(base_n * mult)

    phase_weights = [3.0, 2.0, 1.5, 1.0, 0.7]

    remaining_candidates = len(target_list)
    for i, n_candidate in enumerate(target_list):
        now = time.perf_counter()
        rem = deadline - now - 2.0
        if rem < 10.0:
            break

        w = phase_weights[i] if i < len(phase_weights) else 0.7
        total_w = 0.0
        for j in range(i, remaining_candidates):
            total_w += phase_weights[j] if j < len(phase_weights) else 0.7

        alloc = rem * (w / max(1e-9, total_w))
        alloc = max(8.0, min(alloc, rem * 0.85))
        phase_deadline = now + alloc

        if n_candidate == base_n and len(target_list) > 0 and i == 0:
            h_candidate = h_base
        else:
            h_candidate = scale_vector_to_n(best_h if best_h is not None else h_base, n_candidate)

        f_candidate = build_f(h_candidate)
        h_out, f_out, _ = pair_local_search(h_candidate, f_candidate, phase_deadline, seed + 1000 * (i + 1) + n_candidate)
        best_h, best_n, best_c = update_best(best_h, best_n, best_c, h_out, n_candidate)

        remaining_candidates = len(target_list) - i - 1

    now = time.perf_counter()
    rem = deadline - now - 4.0
    small_specs = [
        (48, "random"),
        (64, "lowdisc"),
        (96, "sine1"),
        (128, "random"),
        (160, "sine2"),
    ]
    small_specs = [sp for sp in small_specs if sp[0] not in seen]
    if rem > 60.0 and small_specs:
        for i, (n_small, kind) in enumerate(small_specs):
            now = time.perf_counter()
            rem = deadline - now - 2.0
            if rem < 8.0:
                break
            w = max(0.4, 1.0 - 0.1 * i)
            total_w = sum(max(0.4, 1.0 - 0.1 * j) for j in range(i, len(small_specs)))
            alloc = rem * (w / max(1e-9, total_w))
            alloc = max(6.0, min(alloc, rem * 0.9))
            h_candidate = candidate_by_kind(n_small, kind, seed + 5000 + i + n_small)
            if h_candidate is None or h_candidate.size == 0:
                continue
            h_candidate = make_feasible(h_candidate)
            f_candidate = build_f(h_candidate)
            h_out, _, _ = pair_local_search(h_candidate, f_candidate, now + alloc, seed + 7000 + i + n_small)
            best_h, best_n, best_c = update_best(best_h, best_n, best_c, h_out, h_candidate.size)

    if best_h is None or best_n <= 0 or not np.isfinite(best_c):
        best_h = random_binary(80, seed)
        best_n = 80
        best_c = c5_exact(best_h)

    final_h = np.clip(np.asarray(best_h, dtype=float).ravel(), 0.0, 1.0)
    if abs(float(final_h.sum()) - 0.5 * final_h.size) > 1e-8:
        final_h = make_dyadic_sum_exact(final_h, final_h.size)

    final_c = c5_exact(final_h)
    if not np.isfinite(final_c):
        final_h = random_binary(80, seed)
        final_c = c5_exact(final_h)

    return final_h, float(final_c), int(best_n)
