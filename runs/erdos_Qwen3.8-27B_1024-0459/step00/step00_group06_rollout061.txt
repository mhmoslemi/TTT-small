    if tl() > 100:
        try:
            res = minimize(obj_slsqp, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n0,
                           constraints=[{'type': 'eq', 'fun': con_eq_fn, 'jac': con_eq_jac}],
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_try = make_feasible(res.x, n0)
            c = compute_c5(h_try)
            if c < best_c5:
                best_c5, best_h = c, h_try
        except Exception:
            pass

    # Phase 4: Multiple random restarts with NM
    n_restarts = 10
    for r in range(n_restarts):
        if tl() < 50:
            break
        scale = 0.03 + 0.07 * (n_restarts - r) / n_restarts
        noise = np.random.randn(n0) * scale
        h_start = make_feasible(best_h + noise, n0)
        try:
            res = minimize(obj_nm, h_start, method='Nelder-Mead',
                           options={'maxiter': 30000, 'xatol': 1e-11,
                                    'fatol': 1e-13, 'maxfev': 60000})
            h_try = make_feasible(res.x, n0)
            c = compute_c5(h_try)
            if c < best_c5:
                best_c5, best_h = c, h_try
                # SLSQP polish after improvement
                if tl() > 40:
                    try:
                        res2 = minimize(obj_slsqp, best_h, method='SLSQP',
                                       bounds=[(0.0, 1.0)] * n0,
                                       constraints=[{'type': 'eq', 'fun': con_eq_fn, 'jac': con_eq_jac}],
                                       options={'maxiter': 2000, 'ftol': 1e-14})
                        h_try2 = make_feasible(res2.x, n0)
                        c2 = compute_c5(h_try2)
                        if c2 < best_c5:
                            best_c5, best_h = c2, h_try2
                    except Exception:
                        pass
        except Exception:
            pass

    # Phase 5: Try different resolution (double)
    if tl() > 150:
        n2 = 2 * n0
        _n_global = n2
        x_c = np.linspace(0.0, 2.0, len(best_h))
        x_f = np.linspace(0.0, 2.0, n2)
        h2 = make_feasible(np.interp(x_f, x_c, best_h), n2)
        try:
            res = minimize(obj_nm, h2, method='Nelder-Mead',
                           options={'maxiter': 30000, 'xatol': 1e-11,
                                    'fatol': 1e-13, 'maxfev': 60000})
            h_try = make_feasible(res.x, n2)
            c = compute_c5(h_try)
            if c < best_c5:
                best_c5, best_h = c, h_try
        except Exception:
            pass
        try:
            res = minimize(obj_slsqp, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n2,
                           constraints=[{'type': 'eq', 'fun': con_eq_fn, 'jac': con_eq_jac}],
                           options={'maxiter': 3000, 'ftol': 1e-14})
            h_try = make_feasible(res.x, n2)
            c = compute_c5(h_try)
            if c < best_c5:
                best_c5, best_h = c, h_try
        except Exception:
            pass
        _n_global = n0

    # Phase 6: Final NM polish at best resolution
    if tl() > 80:
        _n_global = len(best_h)
        try:
            res = minimize(obj_nm, best_h, method='Nelder-Mead',
                           options={'maxiter': 50000, 'xatol': 1e-12,
                                    'fatol': 1e-15, 'maxfev': 100000})
            h_try = make_feasible(res.x, len(best_h))
            c = compute_c5(h_try)
            if c < best_c5:
                best_c5, best_h = c, h_try
        except Exception:
            pass

    # Final validation and return
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
