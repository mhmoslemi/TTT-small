def smooth_max(c, tau):
    """Log-sum-exp approximation of max."""
    c_scaled = tau * c
    c_max = np.max(c_scaled)
    return c_max / tau + np.log(np.sum(np.exp(c_scaled - c_max))) / tau

    corr = np.correlate(h, 1 - h, mode='full') * dx

def gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    # softmax weights
    c_scaled = tau * corr
    c_max = np.max(c_scaled)
    weights = np.exp(c_scaled - c_max)
    weights /= np.sum(weights)
    # Gradient of smooth_max = sum_k weights[k] * dc_k/dh
    # dc_k/dh[j] = dx * (1-h[j-k]) - dx * h[j+k] (with boundary handling)
    grad = np.zeros(n)

        actual_lag = k - (n-1)
        # dc_k/dh[j] = dx*(1-h[j-actual_lag]) - dx*h[j+actual_lag]
        # where out-of-range h is 0
        # For j in [0, n-1]:
        # First term: 1-h[j-actual_lag] if 0<=j-actual_lag<n else 1 (h=0 outside)
        # Second term: h[j+actual_lag] if 0<=j+actual_lag<n else 0
        j = np.arange(n)
        idx1 = j - actual_lag
        idx2 = j + actual_lag
        term1 = np.ones(n)  # 1-h[idx1], h=0 outside
        mask1 = (idx1 >= 0) & (idx1 < n)
        term1[mask1] = 1 - h[idx1[mask1]]
        term2 = np.zeros(n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        term2[mask2] = h[idx2[mask2]]
        grad += weights[k] * dx * (term1 - term2)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_trial) > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h_proj[mask] += diff / cnt
        else:
            mask = h_proj > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h_proj[mask] += diff / cnt
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def upscale_parent(h_parent, n):
    n_p = len(h_parent)
    x_parent = np.linspace(0.0, 2.0, n_p, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x_new, x_parent, h_parent)
    return h

def smooth_max_value(corr, tau):
    m = np.max(corr)
    shifted = tau * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / tau

def smooth_max_grad(corr, h, tau):
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    shifted = tau * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-300:
        idx_max = np.argmax(corr)
        weights = np.zeros_like(corr)
        weights[idx_max] = 1.0
    else:
        weights /= sw
    grad = np.zeros(n)
    threshold = 1e-8
    for idx in range(len(weights)):
        w = weights[idx]
        if w < threshold:
            continue
        lag = idx - (n - 1)
        j = np.arange(n)
        idx1 = j - lag
        t1 = np.ones(n)
        mask1 = (idx1 >= 0) & (idx1 < n)
        t1[mask1] = 1.0 - h[idx1[mask1]]
        idx2 = j + lag
        t2 = np.zeros(n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        t2[mask2] = h[idx2[mask2]]
        grad += w * dx * (t1 - t2)
    return grad

def optimize_resolution(h0, n, time_limit, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = eval_c5(h)

    t_start = time.time()

    tau_schedule = [30, 60, 120, 250, 500]
    penalty = 50.0

    for tau in tau_schedule:
        if time.time() - t_start > time_limit:
            break
        phase_time = time_limit / len(tau_schedule)

        def obj(h_arr):
            corr = np.correlate(h_arr, 1.0 - h_arr, mode='full') * dx
            sm = smooth_max_value(corr, tau)
            s_dev = np.sum(h_arr) - target_sum
            return sm + penalty * s_dev * s_dev

        def grad(h_arr):
            corr = np.correlate(h_arr, 1.0 - h_arr, mode='full') * dx
            g = smooth_max_grad(corr, h_arr, tau)
            s_dev = np.sum(h_arr) - target_sum
            g += 2.0 * penalty * s_dev
            return g

        bounds = [(0.0, 1.0)] * n
        try:
            result = minimize(obj, h, jac=grad, method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 800, 'ftol': 1e-14, 'gtol': 1e-10,
                                      'maxfun': 5000})
            h_new = result.x
            h_new = project_feasible(h_new, n)
            val_new = eval_c5(h_new)
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass

    return best_h, best_val

def local_refine(h, n, time_limit, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    best_val = eval_c5(h)
    t_start = time.time()
    step_size = 0.05
    no_improve_count = 0

    while time.time() - t_start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        g = np.zeros(n)
        j = np.arange(n)
        idx1 = j - lag
        t1 = np.ones(n)
        mask1 = (idx1 >= 0) & (idx1 < n)
        t1[mask1] = 1.0 - h[idx1[mask1]]
        idx2 = j + lag
        t2 = np.zeros(n)
        mask2 = (idx2 >= 0) & (idx2 < n)
        t2[mask2] = h[idx2[mask2]]
        g = dx * (t1 - t2)

        if np.all(np.abs(g) < 1e-10):
            break

        order = np.random.permutation(n)
        improved = False
        for j_idx in order:
            if time.time() - t_start > time_limit:
                break
            direction = -np.sign(g[j_idx])
            if direction == 0:
                direction = 1.0
            for delta in [step_size, step_size * 0.5, step_size * 0.1]:
                h_trial = h.copy()
                h_trial[j_idx] += direction * delta
                if h_trial[j_idx] < 0 or h_trial[j_idx] > 1:
                    continue
                h_trial = project_feasible(h_trial, n)
                c5_trial = eval_c5(h_trial)
                if c5_trial < best_val - 1e-12:
                    h = h_trial
                    best_val = c5_trial
                    improved = True
                    break
        if not improved:
            no_improve_count += 1
            if no_improve_count >= 5:
                break
            step_size *= 0.7
        else:
            no_improve_count = 0

    return h, best_val

def random_restart_optimize(n, time_limit, seed):
    np.random.seed(seed)
    h_rand = np.random.random(n)
    h_rand = project_feasible(h_rand, n)
    return optimize_resolution(h_rand, n, time_limit, seed)
