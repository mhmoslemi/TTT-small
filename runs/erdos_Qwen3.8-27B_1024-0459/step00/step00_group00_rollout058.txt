def c5_gradient(h, n):
    dx = 2.0 / n
    n2 = 2*n - 1
    # Compute all correlations
    corr = np.zeros(n2)
    for s in range(-(n-1), n):
        idx = s + n - 1  # index in corr array
        if s >= 0:
            # sum_i h[i]*(1-h[i+s]) for i=0..n-1-s
            i_range = np.arange(0, n-s)
            corr[idx] = np.sum(h[i_range] * (1 - h[i_range + s]))
        else:
            # sum_i h[i]*(1-h[i+s]) for i=-s..n-1
            i_range = np.arange(-s, n)
            corr[idx] = np.sum(h[i_range] * (1 - h[i_range + s]))
    
    s_max = np.argmax(corr) - (n-1)  # the shift that achieves max
    # Gradient of c_{s_max} w.r.t. h[m]
    grad = np.zeros(n)
    s = s_max
    for m in range(n):
        g = 0.0
        if 0 <= m + s < n:
            g += (1 - h[m+s])
        if 0 <= m - s < n:
            g -= h[m-s]
        grad[m] = g
    return corr * dx, grad * dx, s_max

def objective_with_grad(h, n):
    c5, grad, _ = c5_gradient(h, n)
    return c5, grad

```python
def c5_soft_and_grad(h, n, beta=100):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    # Softmax of corr (scaled by beta)
    corr_shifted = corr - np.max(corr)  # numerical stability
    weights = np.exp(beta * corr_shifted)
    weights /= np.sum(weights)
    c5_soft = dx * (np.max(corr) + np.log(np.sum(weights)) / beta)
    
    # Gradient: sum_k weights[k] * dc_k/dh
    grad = np.zeros(n)
    m = np.arange(n)
    for k in range(len(corr)):
        s = k - (n-1)
        if weights[k] < 1e-10:
            continue
        valid1 = (m + s >= 0) & (m + s < n)
        grad[valid1] += weights[k] * (1 - h[m[valid1] + s])
        valid2 = (m - s >= 0) & (m - s < n)
        grad[valid2] -= weights[k] * h[m[valid2] - s]
    grad *= dx
    
    return c5_soft, grad
```

import numpy as np
from scipy import optimize
import time

def evaluate(c_h, n):
    dx = 2.0 / n
    corr = np.correlate(c_h, 1.0 - c_h, mode='full')
    return float(np.max(corr) * dx)

def c5_soft_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    corr_max = np.max(corr)
    shifted = beta * (corr - corr_max)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    c5 = dx * (corr_max + np.log(sum_exp) / beta)
    
    grad = np.zeros(n)
    m = np.arange(n)
    for k in range(n_corr):
        wk = weights[k]
        if wk < 1e-6:
            continue
        s = k - (n - 1)
        mps = m + s
        mms = m - s
        v1 = (mps >= 0) & (mps < n)
        v2 = (mms >= 0) & (mms < n)
        if np.any(v1):
            grad[v1] += wk * (1.0 - h[mps[v1]])
        if np.any(v2):
            grad[v2] -= wk * h[mms[v2]]
    grad *= dx
    return c5, grad

def full_obj_grad(h, n, beta, penalty):
    c5, grad = c5_soft_obj_grad(h, n, beta)
    sv = np.sum(h) - n / 2.0
    c5 += penalty * sv * sv
    grad = grad + 2.0 * penalty * sv * np.ones(n)
    return c5, grad

def optimize_single(h0, n, beta, penalty, max_iter, time_limit):
    dx = 2.0 / n
    
    def func(x):
        val, _ = full_obj_grad(x, n, beta, penalty)
        return val
    
    def jac(x):
        _, g = full_obj_grad(x, n, beta, penalty)
        return g
    
    result = optimize.minimize(
        func, h0, jac=jac, method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': max_iter, 'ftol': 1e-10, 'gtol': 1e-8, 'maxls': 50}
    )
    h_opt = np.clip(result.x, 0.0, 1.0)
    # Fix sum constraint
    diff = n / 2.0 - np.sum(h_opt)
    if abs(diff) > 1e-6:
        # Distribute correction
        h_opt += diff / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
    
    c5 = evaluate(h_opt, n)
    return h_opt, c5

def random_feasible(n, rng):
    h = rng.random(n)
    # Scale to have sum = n/2
    target = n / 2.0
    h *= target / np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    # Fix sum
    diff = target - np.sum(h)
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        per = diff / len(idx) if len(idx) > 0 else 0
        for i in idx:
            h[i] = min(1.0, h[i] + per)
    elif diff < 0:
        idx = np.where(h > 0.0)[0]
        per = -diff / len(idx) if len(idx) > 0 else 0
        for i in idx:
            h[i] = max(0.0, h[i] - per)
    return h

def sinusoidal_init(n, freq, amp, phase, rng):
    x = np.arange(n) / n * 2.0
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    # Adjust for sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_parent, n_target):
    n_parent = len(h_parent)
    x_old = np.arange(n_parent)
    x_new = np.linspace(0, n_parent - 1, n_target)
    h_new = np.interp(x_new, x_old, h_parent)
    h_new = np.clip(h_new, 0.0, 1.0)
    # Fix sum
    target = n_target / 2.0
    diff = target - np.sum(h_new)
    h_new += diff / n_target
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c5 = np.inf
    best_n = 0
    
    parent_h = None

        parent_n = len(parent_h)
        parent_c5 = evaluate(parent_h, parent_n)
        best_h, best_c5, best_n = parent_h.copy(), parent_c5, parent_n
        print(f"Parent: n={parent_n}, C5={parent_c5:.6f}")
    
    resolutions = [45, 90, 180, 360, 720]
    
    for n in resolutions:

            break
        
        # Determine number of trials for this n
        max_time_per_n = min(200.0, (budget_s * 0.85 - (time.time() - start_time)) / max(1, len(resolutions) - (resolutions.index(n) if n in resolutions else 0)))
        n_trials = max(2, int(max_time_per_n / 30.0))
        
        for trial in range(n_trials):
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.9:
                break
            trial_time = min(60.0, (budget_s * 0.9 - elapsed) / max(1, n_trials - trial))
            
            # Choose initial guess
            if trial == 0 and parent_h is not None:
                h0 = interpolate_to_n(parent_h, n)
            elif trial == 1:
                h0 = sinusoidal_init(n, 1, 0.9, rng.random() * 2 * np.pi, rng)
            elif trial == 2:
                h0 = sinusoidal_init(n, 2, 0.9, rng.random() * 2 * np.pi, rng)
            elif trial == 3:
                h0 = sinusoidal_init(n, 3, 0.9, rng.random() * 2 * np.pi, rng)
            else:
                h0 = random_feasible(n, rng)
            
            # Optimize with increasing penalty
            beta = 50.0
            penalty = 100.0
            h_opt, c5_opt = optimize_single(h0, n, beta, penalty, max_iter=300, time_limit=trial_time)
            
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n
                print(f"n={n}, trial={trial}: C5={c5_opt:.6f} (new best)")
            else:
                print(f"n={n}, trial={trial}: C5={c5_opt:.6f}")
    
    # Final verification
    if best_h is not None:
        final_c5 = evaluate(best_h, best_n)
        best_c5 = final_c5
    
    return best_h, best_c5, best_n
