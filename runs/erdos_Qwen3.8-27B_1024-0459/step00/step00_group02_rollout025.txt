    # Binary search on lambda

    hi = h.max() + 1  # lambda large enough

        # Adjust to have exactly half ones

        zeros = np.where(seq == 0)[0]

            excess = len(ones) - half
            to_move = ones[:excess]
            to_fill = zeros[:excess]
            seq[to_move] = 0
            seq[to_fill] = 1
        elif len(ones) < half:
            deficit = half - len(ones)
            to_move = zeros[:deficit]
            to_fill = ones[:deficit]
            seq[to_move] = 1
            seq[to_fill] = 0
        h = seq.copy()
