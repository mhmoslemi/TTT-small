import numpy as np
import time

def eval_c5(h):
    """Compute C5 = max_k [correlate(h, 1-h, 'full')] * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    """Project h onto feasible set: 0<=h[i]<=1, sum(h)=n/2."""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h_src, n_target):
    """Upsample h to target resolution via linear interpolation."""
    n_src = len(h_src)
    if n_src == n_target:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_tgt, x_src, h_src)
    return make_feasible(h_new, n_target)

def random_01(n):
    """Random 0-1 initialization with exactly n//2 ones."""
    h = np.zeros(n)
    k = n // 2
    if n % 2 == 1:
        idx = np.random.choice(n, k, replace=False)
        h[idx] = 1.0
        rest = list(set(range(n)) - set(idx))
        h[np.random.choice(rest, 1)[0]] = 0.5
    else:
        idx = np.random.choice(n, k, replace=False)
        h[idx] = 1.0
    return h

def structured_init(n):
    """Multi-scale block structure initialization."""
    h = np.zeros(n)
    # Recursively assign blocks
    def fill(start, end, depth, val):
        length = end - start
        if depth <= 0 or length <= 2:
            h[start:end] = val
            return
        mid = start + length // 2
        fill(start, mid, depth - 1, 1.0 - val)
        fill(mid, end, depth - 1, val)
    fill(0, n, min(4, int(np.log2(n))), 0.0)
    return make_feasible(h, n)

def compute_active_gradient(h, n):
    """Compute gradient of the active (max) overlap constraint."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    lag = k_max - n + 1  # actual integer lag
    g = np.zeros(n)
    k = abs(lag)
    if k == 0:
        # f_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # df_0/dh[i] = 1 - 2*h[i]
        g = 1.0 - 2.0 * h
    else:
        # f_k(h) = sum_i h[i]*(1-h[i+k]) for i=0..n-1-k
        # df_k/dh[j] = (1-h[j+k]) for j=0..n-1-k
        #              + (-h[j-k]) for j=k..n-1
        if lag > 0:
            for j in range(n - k):
                g[j] += (1.0 - h[j + k])
            for j in range(k, n):
                g[j] += (-h[j - k])
        else:
            for j in range(k, n):
                g[j] += (1.0 - h[j - k])
            for j in range(0, n - k):
                g[j] += (-h[j + k])
    return g, k_max, lag

def optimize_sa(h_init, n, t_limit, use_gradient=True):
    """Simulated annealing with gradient-aware moves."""
    h = np.asarray(h_init, dtype=float).copy()
    h = make_feasible(h, n)
    c5_cur = eval_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    
    T = 0.04
    T_min = 1e-8
    alpha_cool = 0.99992  # slow cooling
    n_restarts = 0
    stall_count = 0
    
    while time.time() < t_limit:
        # Adaptive temperature: increase if stalled
        if stall_count > 200:
            T = max(T, 0.03)
            stall_count = 0
            n_restarts += 1
            if n_restarts > 5:
                # Random restart from perturbed best
                h = h_best.copy()
                mask = np.random.random(n) < 0.15
                h[mask] = np.random.rand(np.sum(mask))
                h = make_feasible(h, n)
                c5_cur = eval_c5(h)
                n_restarts = 0
        
        # Choose move type
        r = np.random.random()
        h_new = h.copy()
        
        if use_gradient and r < 0.35:
            # Gradient step on active constraint
            g, _, _ = compute_active_gradient(h, n)
            # Project gradient: subtract mean to preserve sum
            g = g - np.mean(g)
            step = 0.05 * (0.5 + T * 10)
            h_new = h - step * g
        elif r < 0.5:
            # Two-point adjust (preserves sum)
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                j = (j + 1) % n
            delta = np.random.uniform(-0.2, 0.2)
            h_new[i] += delta
            h_new[j] -= delta
        elif r < 0.65:
            # Swap two values
            i, j = np.random.choice(n, 2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        elif r < 0.75:
            # Block invert (0-1 style)
            bs = np.random.randint(1, max(2, n // 6))
            s = np.random.randint(0, n - bs + 1)
            h_new[s:s+bs] = 1.0 - h_new[s:s+bs]
        elif r < 0.85:
            # Lerp with best (pull toward best solution)
            h_new = (1.0 - 0.3) * h + 0.3 * h_best
            h_new += np.random.randn(n) * 0.02
        elif r < 0.93:

            k_pert = np.random.randint(1, 4)
            idx = np.random.choice(n, k_pert, replace=False)
            h_new[idx] += np.random.randn(k_pert) * 0.06
        else:

            zeros = np.where(h < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                h_new[i] = 0.0
                h_new[j] = 1.0

        c5_new = eval_c5(h_new)

            else:

        else:

def polish(h_init, n, t_limit):
    """Fine polishing with very small steps."""
    h = np.asarray(h_init, dtype=float).copy()
    h = make_feasible(h, n)
    c5_cur = eval_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    
    while time.time() < t_limit:
        # Small gradient step
        g, _, _ = compute_active_gradient(h, n)
        g = g - np.mean(g)
        step = 0.01
        h_try1 = make_feasible(h - step * g, n)
        c5_try1 = eval_c5(h_try1)
        
        # Small random move
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.uniform(-0.03, 0.03)
        h_try2 = h.copy()
        h_try2[i] += delta
        h_try2[j] -= delta
        h_try2 = make_feasible(h_try2, n)
        c5_try2 = eval_c5(h_try2)
        
        # Pick best of the two
        if c5_try1 <= c5_try2:
            h_new, c5_new = h_try1, c5_try1
        else:
            h_new, c5_new = h_try2, c5_try2
        
        if c5_new < c5_cur or np.random.random() < np.exp(-(c5_new - c5_cur) / 0.01):
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
    
    return h_best, c5_best

    # Get initial solution
    h_best = None
    c5_best = float('inf')
    n_best = 0
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
        h_best = make_feasible(h0, n0)
        c5_best = eval_c5(h_best)
        n_best = n0
    except:
        h0 = None
        n0 = 0
    
    if h_best is None:
        # Fallback initial
        n_best = 48
        h_best = random_01(n_best)
        c5_best = eval_c5(h_best)
    
    # Resolutions to try
    base_res = [48, 50, 60, 72, 80, 100, 120, 144, 160, 200, 256, 320, 400, 500, 640]
    resolutions = []
    if n0 > 0:
        resolutions.append(n0)
    for r in base_res:
        if r not in resolutions and r >= 24:
            resolutions.append(r)
    resolutions.sort()
    
    # Phase 1: Broad search across resolutions (60% of time)
    t_phase1_end = t0 + budget_s * 0.55
    
    for n in resolutions:
        if time.time() > t_phase1_end:
            break
        remaining = t_phase1_end - time.time()
        if remaining < 8:
            break
        
        # Allocate time inversely proportional to n (smaller n gets more time per eval)
        t_alloc = min(remaining * 0.35, 60.0 * (50.0 / max(n, 25)))
        if t_alloc < 3:
            t_alloc = remaining * 0.2
        t_limit = time.time() + t_alloc
        
        # Multiple initializations
        inits = []
        if h_best is not None:
            inits.append(upsample(h_best, n))
        inits.append(random_01(n))
        inits.append(structured_init(n))
        
        # Add a "shifted" version of best for diversity
        if h_best is not None and len(inits) < 4:
            h_shift = np.roll(h_best, n // 4)
            h_shift = make_feasible(h_shift, n)
            inits.append(h_shift)
        
        for h_s in inits:
            if time.time() > t_limit - 2:
                break
            h_opt, c5_opt = optimize_sa(h_s, n, min(t_limit, time.time() + t_alloc * 0.4))
            if c5_opt < c5_best:
                c5_best = c5_opt
                h_best = h_opt.copy()
                n_best = n
    
    # Phase 2: Focused refinement on top solutions (30% of time)
    t_phase2_end = t0 + budget_s * 0.85
    if time.time() < t_phase2_end:
        # Refine best solution at its resolution
        n_ref = n_best
        t_ref_limit = time.time() + (t_phase2_end - time.time()) * 0.5
        h_opt, c5_opt = optimize_sa(h_best, n_ref, t_ref_limit, use_gradient=True)
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt.copy()
            n_best = n_ref
        
        # Try at nearby resolutions
        for dn in [10, -10, 20, -20, 50]:
            n_try = n_best + dn
            if n_try < 24 or time.time() > t_phase2_end:
                continue
            t_try_limit = time.time() + min(20, t_phase2_end - time.time())
            if t_try_limit <= time.time():
                break
            h_try = upsample(h_best, n_try)
            h_opt2, c5_opt2 = optimize_sa(h_try, n_try, t_try_limit)
            if c5_opt2 < c5_best:
                c5_best = c5_opt2
                h_best = h_opt2.copy()
                n_best = n_try
    
    # Phase 3: Final polish (15% of time)
    if time.time() < t0 + budget_s - 5 and h_best is not None:
        t_polish = t0 + budget_s - 3
        h_pol, c5_pol = polish(h_best, n_best, t_polish)
        if c5_pol < c5_best:
            c5_best = c5_pol
            h_best = h_pol.copy()
            n_best = len(h_pol)
    
    # Final feasibility enforcement
    h_best = make_feasible(h_best, n_best)
    c5_final = eval_c5(h_best)
    
    # Verify
    n_final = len(h_best)
    assert np.all(h_best >= -1e-10) and np.all(h_best <= 1 + 1e-10), "Bounds violated"
    assert abs(np.sum(h_best) - n_final / 2.0) < 1e-6, f"Sum violated: {np.sum(h_best)} vs {n_final/2}"
    
    return h_best, c5_final, n_final
