import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_parent = np.array(initial_h_values, dtype=float)

    def c5_value(h):
        n = len(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        return float(np.max(corr) * dx)

    def project_constraint(h):
        h = np.clip(h, 0.0, 1.0)
        n = len(h)
        target = n / 2.0
        for _ in range(500):
            s = np.sum(h)
            diff = target - s
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                mask = h < 1.0 - 1e-12
                if not np.any(mask):
                    break
                step = min(diff / np.sum(mask), (1.0 - h[mask]).min() * 0.9)
                h[mask] += step
            else:
                mask = h > 1e-12
                if not np.any(mask):
                    break
                step = min(-diff / np.sum(mask), h[mask].min() * 0.9)
                h[mask] += step
            h = np.clip(h, 0.0, 1.0)
        return h

    def smooth_obj(h, n, p=50.0):
        """Smooth max via log-sum-exp for better gradient info."""
        target_sum = n / 2.0
        pen = 1e4 * (np.sum(h) - target_sum) ** 2
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        vals = corr * dx
        m = np.max(vals)
        smooth_max = m + np.log(np.sum(np.exp(p * (vals - m)))) / p
        return smooth_max + pen

    def exact_obj(h, n, pen_weight=1e4):
        target_sum = n / 2.0
        pen = pen_weight * (np.sum(h) - target_sum) ** 2
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        return float(np.max(corr) * dx) + pen

    def optimize_lbfgsb(h0, n, max_iter=3000):
        bounds = [(0.0, 1.0)] * n
        obj = lambda x: exact_obj(x, n)
        result = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14, 'maxfun': max_iter * 5})
        return project_constraint(result.x)

    def local_search_binary_spread(h, n, rng, max_iters=5000):
        """Try small perturbations and accept improvements."""
        h = h.copy()
        c5 = c5_value(h)
        step = 0.02
        for it in range(max_iters):
            idx = rng.randint(n)
            delta = rng.uniform(-step, step)
            h_old = h[idx]
            h[idx] = np.clip(h_old + delta, 0.0, 1.0)
            h = project_constraint(h)
            c5_new = c5_value(h)
            if c5_new < c5:
                c5 = c5_new
                step = min(step * 1.05, 0.1)
            else:
                h[idx] = h_old
                h = project_constraint(h)
                if it > 100 and it % 200 == 0:
                    step *= 0.95
        return h, c5

    def interpolate_to(h_old, n_new):
        n_old = len(h_old)
        if n_new == n_old:
            return h_old.copy()
        x_old = np.linspace(0, 2, n_old, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_ext = np.concatenate([h_old, [h_old[0]]])
        h_interp = np.interp(x_new, x_old, h_ext)
        return project_constraint(h_interp)

    best_h = h_parent.copy()
    best_c5 = c5_value(best_h)
    best_n = len(best_h)
    print(f"Parent: n={best_n}, C5={best_c5:.6f}")

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Optimize at parent resolution
    if time_left() > 120:
        n = len(best_h)
        candidate = optimize_lbfgsb(best_h.copy(), n, max_iter=5000)
        c = c5_value(candidate)
        print(f"After L-BFGS at n={n}: C5={c:.6f}")
        if c < best_c5:
            best_h, best_c5 = candidate, c

    # Phase 2: Local search refinement
    if time_left() > 100:
        n = len(best_h)
        h_ls, c5_ls = local_search_binary_spread(best_h, n, rng, max_iters=3000)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls
            print(f"After local search n={n}: C5={c5_ls:.6f}")

    # Phase 3: Multiple perturbations at parent resolution
    if time_left() > 200:
        n = len(best_h)
        for trial in range(5):
            if time_left() < 40:
                break
            perturb = rng.randn(n) * 0.03
            h_pert = project_constraint(best_h + perturb)
            candidate = optimize_lbfgsb(h_pert, n, max_iter=2000)
            c = c5_value(candidate)
            if c < best_c5:
                best_h, best_c5 = candidate, c
                print(f"Trial {trial+1} n={n}: C5={c:.6f}")

    # Phase 4: Higher resolution attempts
    for n_new in [90, 180]:
        if time_left() < 150:
            break
        print(f"\nAttempting n={n_new}, time_left={time_left():.0f}s")
        h_interp = interpolate_to(best_h, n_new)
        candidate = optimize_lbfgsb(h_interp, n_new, max_iter=3000)
        c = c5_value(candidate)
        print(f"L-BFGS at n={n_new}: C5={c:.6f} (best={best_c5:.6f})")
        if c < best_c5:
            best_h, best_c5, best_n = candidate, c, n_new
            # Quick local search at new resolution
            if time_left() > 100:
                h_ls, c5_ls = local_search_binary_spread(best_h, n_new, rng, max_iters=2000)
                if c5_ls < best_c5:
                    best_h, best_c5, best_n = h_ls, c5_ls, n_new
                    print(f"After local search n={n_new}: C5={c5_ls:.6f}")
        else:
            best_n = len(best_h)

    # Phase 5: Final polish at best resolution
    if time_left() > 60:
        n = len(best_h)
        candidate = optimize_lbfgsb(best_h.copy(), n, max_iter=3000)
        c = c5_value(candidate)
        if c < best_c5:
            best_h, best_c5, best_n = candidate, c, n

    # Final verification
    best_h = project_constraint(best_h)
    best_c5 = c5_value(best_h)
    best_n = len(best_h)
    print(f"\nFinal: n={best_n}, C5={best_c5:.6f}")
    print(f"Sum check: {np.sum(best_h):.6f} vs {best_n/2:.6f}")
    print(f"Min/Max: {np.min(best_h):.6f}/{np.max(best_h):.6f}")

    return (best_h, best_c5, best_n)
