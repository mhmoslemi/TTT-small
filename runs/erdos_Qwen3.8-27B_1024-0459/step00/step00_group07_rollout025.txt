def compute_c5(h):
    """Compute C5 = max(corr) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr, dx

def smooth_max(corr, dx, alpha):
    """Smooth max of dx*corr via log-sum-exp"""
    vals = alpha * dx * corr
    m = np.max(vals)
    if m > 500:  # overflow protection
        vals = vals - m
    e = np.exp(vals)
    return (m + np.log(np.sum(e))) / alpha

def gradient_smooth_max(h, dx, alpha):
    """Gradient of smooth max of dx*corr"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * dx * corr
    m = np.max(vals)
    vals_shifted = vals - m
    e = np.exp(vals_shifted)
    w = e / np.sum(e)  # weights, shape (2n-1,)
    
    # Gradient: for each shift s = k-(n-1), k=0..2n-2
    # ∂output[k]/∂h[j] = (1-h[j-s])*1[0≤j-s≤n-1] - h[j+s]*1[0≤j+s≤n-1]

    for k in range(2*n-1):
        s = k - (n-1)
        w_k = w[k] * dx
        if w_k < 1e-15:
            continue
        # (1-h[j-s]) for valid j: j-s in [0,n-1] => j in [s, s+n-1] intersect [0,n-1]

        j_hi = min(n-1, s+n-1)

        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= w_k * h[j_lo2+s:j_hi2+1+s]
    
    return grad

    _, corr, _ = compute_c5(h)

    res = minimize(f, h_start, jac=g, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15,
                            'gtol': 1e-9, 'maxfun': 50000,
                            'iprint': -1})
    h_opt = res.x
    h_opt = fix_sum(h_opt, n)
    c5 = compute_c5(h_opt)[0]
    return h_opt, c5


def local_search_worst(h, n, deadline, max_rounds=2000):
    """Iteratively reduce the worst-lag correlation."""
    best_c5 = compute_c5(h)[0]
    h_cur = h.copy()
    step = 0.02
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        # Gradient of corr[k_star]
        gk = np.zeros(n)
        jlo = max(0, s)
        jhi = min(n - 1, s + n - 1)
        if jlo <= jhi:
            gk[jlo:jhi + 1] += (1.0 - h_cur[jlo - s:jhi + 1 - s])
        jlo2 = max(0, -s)
        jhi2 = min(n - 1, n - 1 - s)
        if jlo2 <= jhi2:
            gk[jlo2:jhi2 + 1] -= h_cur[jlo2 + s:jhi2 + 1 + s]
        # Also consider second worst
        corr_sorted = np.sort(corr)[::-1]
        if len(corr_sorted) > 1 and corr_sorted[0] - corr_sorted[1] < 0.5:
            k2 = int(np.argsort(corr)[::-1][1])
            s2 = k2 - (n - 1)
            gk2 = np.zeros(n)
            jlo = max(0, s2)
            jhi = min(n - 1, s2 + n - 1)
            if jlo <= jhi:
                gk2[jlo:jhi + 1] += (1.0 - h_cur[jlo - s2:jhi + 1 - s2])
            jlo2 = max(0, -s2)
            jhi2 = min(n - 1, n - 1 - s2)
            if jlo2 <= jhi2:
                gk2[jlo2:jhi2 + 1] -= h_cur[jlo2 + s2:jhi2 + 1 + s2]
            gk = 0.5 * (gk + gk2)
        # Step
        h_try = h_cur - step * gk
        h_try = fix_sum(h_try, n)
        c5_try = compute_c5(h_try)[0]
        if c5_try < best_c5 - 1e-10:
            h_cur = h_try
            best_c5 = c5_try
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.7
            if step < 1e-6:
                break
    return h_cur, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except (NameError, Exception):
        n_init = 80
        h_init = np.random.random(n_init)
        h_init = fix_sum(h_init, n_init)

    c5_init = compute_c5(h_init)[0]
    best_h = h_init.copy()
    best_c5 = c5_init
    best_n = n_init

    # Multi-resolution ladder
    resolutions = []
    for nr in [160, 256, 384, 512, 768, 1024, 1536]:
        if nr > n_init:
            resolutions.append(nr)

    current_h = h_init.copy()
    current_n = n_init

    for n_target in resolutions:
        if time.time() > deadline - 80:
            break

        # Upsample
        if n_target > current_n:
            scale = n_target / current_n
            current_h = zoom(current_h, scale, order=1)
            current_h = np.clip(current_h, 0.0, 1.0)
            current_h = fix_sum(current_h, n_target)
        current_n = n_target

        # Optimize with increasing alpha
        alpha_schedule = [30, 80, 150, 300, 600]
        lam = max(5.0, current_n / 200.0)

        for alpha in alpha_schedule:
            if time.time() > deadline - 40:
                break
            max_iter = 300 if n_target <= 512 else 150
            h_opt, c5_val = optimize_level(current_h, n_target, alpha, lam,
                                           max_iter, deadline)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n_target
            current_h = h_opt

        # Local search refinement
        if time.time() < deadline - 30:
            h_ls, c5_ls = local_search_worst(current_h, n_target, deadline,
                                             max_rounds=1500)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n_target

    # Additional refinements on the best solution found
    if best_h is not None and time.time() < deadline - 20:
        n_best = best_n
        # Higher alpha polish
        for alpha in [800, 1500]:
            if time.time() > deadline - 15:
                break
            h_opt, c5_val = optimize_level(best_h, n_best, alpha,
                                           max(5.0, n_best / 200.0),
                                           200, deadline)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()

        # Final local search
        if time.time() < deadline - 10:
            h_ls, c5_ls = local_search_worst(best_h, n_best, deadline,
                                             max_rounds=3000)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n_best

    # Ensure final constraints
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final)
    c5_final = compute_c5(best_h)[0]

    return best_h, c5_final, n_final
