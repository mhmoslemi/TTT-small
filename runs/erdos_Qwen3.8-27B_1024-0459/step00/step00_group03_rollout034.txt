from scipy.optimize import minimize

def compute_c5_exact(h, n):

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def soft_max_objective(h, n, temperature, lambda_pen=1000.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    max_g = np.max(g)
    log_sum = max_g + np.log(np.sum(np.exp(temperature * (g - max_g))))
    soft_max_val = log_sum / temperature
    penalty = lambda_pen * (np.sum(h) - n / 2.0) ** 2
    return soft_max_val + penalty

def soft_max_gradient(h, n, temperature, lambda_pen=1000.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    g = corr * dx
    n_corr = 2 * n - 1
    
    # Soft max weights
    max_g = np.max(g)
    exp_g = np.exp(temperature * (g - max_g))
    p = exp_g / np.sum(exp_g)
    
    grad = np.zeros(n)
    
    for m in range(n_corr):
        if p[m] < 1e-16:
            continue
        s = m - (n - 1)
        if s >= 0:
            # (1-h_{j-s}) valid for j in [s, n-1]
            if s < n:
                grad[s:] += p[m] * dx * (1 - h[:n-s])
            # -h_{j+s} valid for j in [0, n-1-s]
            grad[:n-s] -= p[m] * dx * h[s:n]
        else:
            sp = -s
            # (1-h_{j+sp}) valid for j in [0, n-1-sp]
            grad[:n-sp] += p[m] * dx * (1 - h[sp:n])
            # -h_{j-sp} valid for j in [sp, n-1]
            grad[sp:] -= p[m] * dx * h[:n-sp]
    
    # Constraint penalty gradient
    grad += 2.0 * lambda_pen * (np.sum(h) - n / 2.0)
    
    return grad

def project_to_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2 approximately."""
    for _ in range(20):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s < 1e-10:
            h = np.full(n, 0.5)
            break
        h = h * (n / 2.0) / s
        h = np.clip(h, 0, 1)
    # Final adjustment
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0, 1)
    return h

def try_random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0, 1, n)
    h = project_to_feasible(h, n)
    return h

def interpolate_init(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    h_dst = project_to_feasible(h_dst, n_dst)
    return h_dst

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 70
_g_T = 1.0
_g_lam = 1000.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _obj(x):
    n = _g_n
    T = _g_T
    lam = _g_lam
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    g = corr * dx
    max_g = np.max(g)
    exp_terms = np.exp((g - max_g) / T)
    soft_max = max_g + T * np.log(np.mean(exp_terms))
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return soft_max + penalty


def _grad(x):
    n = _g_n
    T = _g_T
    lam = _g_lam
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    g = corr * dx
    n_corr = 2 * n - 1
    max_g = np.max(g)
    exp_terms = np.exp((g - max_g) / T)
    sum_exp = np.sum(exp_terms)
    p = exp_terms / sum_exp

    grad = np.zeros(n)
    for m in range(n_corr):
        if p[m] < 1e-18:
            continue
        s = m - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += p[m] * dx * (1.0 - x[:n - s])
            if n - 1 - s >= 0:
                grad[:n - s] -= p[m] * dx * x[s:]
        else:
            sp = -s
            if n - 1 - sp >= 0:
                grad[:n - sp] += p[m] * dx * (1.0 - x[sp:])
            grad[sp:] -= p[m] * dx * x[:n - sp]

    grad += 2.0 * lam * (np.sum(x) - n / 2.0)
    return grad


def project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        if s < 1e-15:
            h = np.full(n, 0.5)
            break
        h = h * target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def try_random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0.0, 1.0, n)
    h = project_to_feasible(h, n)
    return h


def try_block_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    positions = rng.choice(n, size=n // 2, replace=False)
    h[positions] = 1.0
    h = project_to_feasible(h, n)
    return h


def interpolate_init(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_dst = np.interp(x_dst, x_src, h_src)
    h_dst = project_to_feasible(h_dst, n_dst)
    return h_dst


def optimize_single(h_init, n, deadline, temperatures, max_iter=300):
    global _g_n, _g_T, _g_lam
    _g_n = n
    _g_lam = 1000.0
    bounds = [(0.0, 1.0)] * n
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    for T in temperatures:
        if time.time() > deadline:
            break
        _g_T = T
        h_curr = best_h.copy()
        try:
            result = minimize(
                fun=_obj,
                x0=h_curr,
                jac=_grad,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-12, 'maxfun': 10000}
            )
            h_new = np.clip(result.x, 0.0, 1.0)
            h_new = project_to_feasible(h_new, n)
        except Exception:
            h_new = h_curr.copy()

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5
