    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        if s == 0:
            h = np.full(n, target / n)
            break
        h = h + (target - s) / n
    h = np.clip(h, 0, 1)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            # Add to smallest elements
            idx = np.argsort(h)
            for i in idx:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:

            idx = np.argsort(h)[::-1]
            for i in idx:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff > -1e-12:
                    break

    return float(np.max(corr) * dx)

def project_constraint(h, n):
    h = np.array(h, dtype=float)
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)

            break

            h[:] = target / n
            break

    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                if add <= 0: break
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-15: break

    bump[:q] = 1.0
    bump[n // 2:n // 2 + q] = 1.0
    starts.append(project_constraint(bump, n))
    ramp = np.linspace(1.0, 0.0, n)
    starts.append(project_constraint(ramp, n))
    # Staircase
    stair = np.zeros(n)
    for i in range(n):
        stair[i] = 1.0 if (i % (n // 7)) < (n // 14) else 0.0
    starts.append(project_constraint(stair, n))
    return starts

def optimize_single(h0, n, deadline):
    h0 = project_constraint(h0.copy(), n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with soft max (smooth objective)
    try:
        res = minimize(obj_softmax, h0.copy(), args=(n, 80.0, 5000.0),
                       method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-10})
        if time.time() > deadline:
            return best_h, best_c5
        h1 = project_constraint(res.x, n)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1
    except Exception:
        pass

    # Phase 2: L-BFGS-B with hard max (refinement)
    if time.time() < deadline - 2:
        try:
            res2 = minimize(obj_hard, best_h.copy(), args=(n, 5000.0),
                            method='L-BFGS-B',
                            bounds=[(0.0, 1.0)] * n,
                            options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-10})
            h2 = project_constraint(res2.x, n)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2
        except Exception:
            pass

    # Phase 3: Local coordinate search
    if time.time() < deadline - 1:
        for _ in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for delta in [0.01, 0.005, 0.02, -0.01, -0.005, -0.02]:
                    h_try = best_h.copy()
                    h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                    # Find j to adjust for constraint
                    diff = n / 2.0 - np.sum(h_try)
                    if abs(diff) < 1e-12:
                        h_try = project_constraint(h_try, n)
                        c_try = compute_c5(h_try)
                        if c_try < best_c5 - 1e-10:
                            best_h = h_try
                            best_c5 = c_try
                            improved = True
                            break
                    else:
                        # Adjust a free coordinate
                        if diff > 0:
                            candidates = np.where(h_try < 0.99)[0]
                        else:
                            candidates = np.where(h_try > 0.01)[0]
                        if len(candidates) > 0:
                            j = candidates[np.argmax(candidates != i)] if np.any(candidates != i) else candidates[0]
                            if j == i:
                                continue
                            h_try[j] = np.clip(h_try[j] + diff, 0.0, 1.0)
                            h_try = project_constraint(h_try, n)
                            c_try = compute_c5(h_try)
                            if c_try < best_c5 - 1e-10:
                                best_h = h_try
                                best_c5 = c_try
                                improved = True
                                break
            if not improved and _ > 50:
                break

    return best_h, best_c5

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new points using linear interpolation."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_constraint(h_new, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = np.inf
    best_h = None
    best_n = 0

    parent = None
    pn = 0
    try:
        parent = np.array(initial_h_values, dtype=float)
        pn = len(parent)
    except Exception:
        pass

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Grid sizes to try (prioritize parent and moderate sizes)
    grids = [60, 80, 100, 120, 150, 200]
    if pn > 0 and pn not in grids:
        grids.insert(0, pn)
    grids = sorted(set(grids))

    # Phase A: Coarse-to-fine optimization
    prev_h = None
    for gi, n in enumerate(grids):
        if remaining() < 50:
            break

        n_grids_left = len(grids[gi:])
        per_grid = min(remaining() / n_grids_left, 350.0)
        deadline = time.time() + per_grid

        starts = make_starts(n, rng, parent, pn)

        # Also try upsampled from previous best if available
        if prev_h is not None and len(prev_h) != n:
            starts.append(upsample(prev_h, len(prev_h), n))

        grid_best_h = None
        grid_best_c5 = np.inf

        for h0 in starts:
            if time.time() > deadline:
                break
            if remaining() < 10:
                break
            h_opt, c5_val = optimize_single(h0, n, deadline)
            if c5_val < grid_best_c5:
                grid_best_c5 = c5_val
                grid_best_h = h_opt.copy()
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n
                print(f"n={n} start={len(starts)}: C5 = {best_c5:.6f}")

        if grid_best_h is not None:
            prev_h = grid_best_h

        # Perturbation around grid best
        if grid_best_h is not None and time.time() < deadline:
            for pidx in range(5):
                if time.time() > deadline:
                    break
                h_pert = grid_best_h + rng.normal(0, 0.03 + 0.02 * pidx, n)
                h_pert = project_constraint(h_pert, n)
                h_opt, c5_val = optimize_single(h_pert, n, deadline)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
                    best_n = n
                    print(f"n={n} perturb{pidx}: C5 = {best_c5:.6f}")

    # Phase B: Final refinement with higher penalty and more iterations
    if best_h is not None and remaining() > 40:
        n = best_n
        deadline = time.time() + min(remaining() - 10, 120)
        try:
            res = minimize(obj_softmax, best_h.copy(), args=(n, 200.0, 50000.0),
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-12})
            h_ref = project_constraint(res.x, n)
            c_ref = compute_c5(h_ref)
            if c_ref < best_c5:
                best_c5 = c_ref
                best_h = h_ref
                print(f"Final refinement: C5 = {best_c5:.6f}")
        except Exception:
            pass

        # Hard max refinement
        if remaining() > 20 and best_h is not None:
            try:
                res2 = minimize(obj_hard, best_h.copy(), args=(n, 50000.0),
                                method='L-BFGS-B',
                                bounds=[(0.0, 1.0)] * n,
                                options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-12})
                h_ref2 = project_constraint(res2.x, n)
                c_ref2 = compute_c5(h_ref2)
                if c_ref2 < best_c5:
                    best_c5 = c_ref2
                    best_h = h_ref2
                    print(f"Hard max refinement: C5 = {best_c5:.6f}")
            except Exception:
                pass

    # Phase C: SLSQP with equality constraint for final polish
    if best_h is not None and remaining() > 30:
        n = best_n
        try:
            def obj_eq(h):
                return compute_c5(np.clip(h, 0.0, 1.0))
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
                     'jac': lambda h: np.ones(n)}]
            res = minimize(obj_eq, best_h.copy(), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-12})
            h_s = project_constraint(res.x, n)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s
                print(f"SLSQP: C5 = {best_c5:.6f}")
        except Exception:
            pass

    # Fallback
    if best_h is None:
        best_h = np.array([0.5, 0.5])
        best_n = 2
        best_c5 = compute_c5(best_h)

    # Final validation
    best_h = project_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
