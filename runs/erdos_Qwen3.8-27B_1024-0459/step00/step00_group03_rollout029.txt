import time

def compute_c5(h, dx):
    """Compute C5 = max of correlation values * dx"""
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_to_sum(h, target_sum):
    """Project h to satisfy sum constraint while keeping [0,1] bounds."""
    # Simple: clip and scale
    h_clipped = np.clip(h, 0, 1)
    s = np.sum(h_clipped)
    if s == 0:
        return np.full_like(h, 0.5)
    # Scale towards target
    scaled = h_clipped * (target_sum / s)
    # If any exceed 1, clip and redistribute
    for _ in range(100):
        over = scaled > 1.0
        if not np.any(over):
            break
        excess = np.sum(scaled[over] - 1.0)
        scaled[over] = 1.0
        under = ~over
        if np.sum(scaled[under]) == 0:
            break
        scaled[under] -= excess / np.sum(under)
        scaled = np.clip(scaled, 0, 1)
    return np.clip(scaled, 0, 1)

    max_corr = np.max(corr)
    shifted = corr - max_corr
    log_sum_exp = max_corr + eta * np.log(np.sum(np.exp(shifted / eta)))

    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)

            h[j] = best_hj
        h = project_feasible(h, target_sum)
        new_c5 = compute_c5(h, dx)
        if new_c5 >= current_c5:
            h[j] = best_hj
            break
        current_c5 = new_c5
        improved = True
        if not improved:
            break
    return h, current_c5

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    rng = np.random.RandomState(seed)

    # Determine grid sizes to try
    n_list = [70, 84, 100, 120, 140, 160]

    # Build starting points
    all_starts = {}
    for n in n_list:
        dx = 2.0 / n
        target_sum = n / 2.0
        starts = []
        # Constant
        starts.append(np.full(n, 0.5))
        # Sine patterns
        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [1, 2, 3]:
            h_s = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x)
            starts.append(project_feasible(h_s, target_sum))
        # Random
        for _ in range(4):
            h_r = rng.uniform(0.2, 0.8, size=n)
            starts.append(project_feasible(h_r, target_sum))
        # Block patterns
        h_b = np.zeros(n)
        h_b[:n//3] = 1.0
        h_b[2*n//3:] = 0.3
        starts.append(project_feasible(h_b, target_sum))
        # Triangular wave
        t_wave = np.zeros(n)
        for i in range(n):
            phase = (i / n) * 2.0
            t_wave[i] = 1.0 - abs(2.0 * phase - np.round(phase))
        starts.append(project_feasible(t_wave, target_sum))
        all_starts[n] = starts

    # Add parent as start for n=70
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        if len(h_parent) == 70:
            h_parent = project_feasible(h_parent, 35.0)
            all_starts[70].insert(0, h_parent)

    # Time allocation: spend more time on smaller grids first
    # Total usable time
    t_end = t_start + budget_s
    # Reserve 10% for final polishing
    opt_deadline = t_start + 0.85 * budget_s

    # Process grids in order, allocating time proportionally
    # Smaller grids get less time per start (faster), larger grids more
    time_weights = {}
    for i, n in enumerate(n_list):
        time_weights[n] = (i + 1) ** 1.5

    total_weight = sum(time_weights.values())
    time_budget_per_grid = {}
    for n in n_list:
        time_budget_per_grid[n] = (opt_deadline - t_start) * time_weights[n] / total_weight

    for n in n_list:
        grid_start = time.time()
        grid_deadline = min(grid_start + time_budget_per_grid[n], opt_deadline)
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        starts = all_starts[n]

        # Sort starts: parent first (if present), then by initial C5
        def init_score(h):
            return compute_c5(h, dx)
        starts.sort(key=init_score)

        # Limit number of starts based on available time
        max_starts = min(len(starts), max(3, int((grid_deadline - grid_start) / 30)))
        starts_to_try = starts[:max_starts]

        for si, h_init in enumerate(starts_to_try):
            if time.time() > grid_deadline:
                break
            h_opt = optimize_from_start(h_init, n, dx, bounds, grid_deadline, seed_offset=si)
            h_opt = project_feasible(h_opt, n / 2.0)
            c5_val = compute_c5(h_opt, dx)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

    # Final polishing phase
    polish_deadline = t_end - 10
    if best_h is not None and time.time() < polish_deadline:
        n_polish = best_n
        dx_polish = 2.0 / n_polish
        # Coordinate descent polish
        h_polished, c5_polished = coordinate_polish(best_h, n_polish, dx_polish,
                                                     max_rounds=30, deadline=polish_deadline)
        if c5_polished < best_c5:
            best_c5 = c5_polished
            best_h = h_polished.copy()

        # Try small perturbations
        for _ in range(20):
            if time.time() > polish_deadline:
                break
            h_pert = random_perturbation(best_h, best_n, rng, scale=0.05)
            h_pert = project_feasible(h_pert, best_n / 2.0)
            c5_pert = compute_c5(h_pert, best_n / 2.0)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
                # Re-polish
                h_polished, c5_polished = coordinate_polish(best_h, best_n, best_n and dx_polish,
                                                             max_rounds=10, deadline=polish_deadline)
                best_h = h_polished
                best_c5 = c5_polished

    # Fallback
    if best_h is None:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_c5 = compute_c5(best_h, 2.0 / best_n)
        else:
            best_h = np.full(70, 0.5)
            best_n = 70
            best_c5 = compute_c5(best_h, 2.0 / best_n)

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, best_c5, best_n
