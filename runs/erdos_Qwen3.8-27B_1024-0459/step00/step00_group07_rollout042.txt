```python
import numpy as np
import time

def compute_corr(h):
    """Compute full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def subgradient_for_lag(h, lag):
    """
    Subgradient of C(lag) at h.
    C(lag) = sum over j of (1-h[j]) * h[j-lag] (h=0 outside [0,n-1])
    lag corresponds to the offset in the full correlation.
    In np.correlate(h, 1-h, mode='full'), the index k corresponds to lag = k - (n-1).
    """
    n = len(h)
    grad = np.zeros(n)
    
    # Term 1: (1-h[i+lag]) for valid i+lag
    # i+lag in [0, n-1] => i in [-lag, n-1-lag]
    start1 = max(0, -lag)
    end1 = min(n, n - lag)
    if start1 < end1:
        grad[start1:end1] += 1.0 - h[start1 + lag : end1 + lag]
    
    # Term 2: -h[i-lag] for valid i-lag
    # i-lag in [0, n-1] => i in [lag, lag+n-1]
    start2 = max(0, lag)
    end2 = min(n, lag + n)
    if start2 < end2:
        grad[start2:end2] -= h[start2 - lag : end2 - lag]
    
    return grad

def project_constraint(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2} using bisection."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_constraint(h_new, n_new)

def subgradient_optimize(h, n_steps, initial_lr, final_lr, budget_s, start_time):
    """Optimize h using subgradient method."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break
        
        corr = compute_corr(h)
        max_val = np.max(corr)
        
        # Find active lags (within 1% of max)
        threshold = max_val * 0.99
        active_lags = np.where(corr >= threshold)[0]
        
        # Average subgradient over active lags
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(corr)])
        
        sg = np.zeros(n)
        for lag_idx in active_lags:
            lag = int(lag_idx) - (n - 1)
            sg += subgradient_for_lag(h, lag)
        sg /= len(active_lags)
        
        # Adaptive step size
        progress = step / max(1, n_steps)
        lr = initial_lr * (final_lr / initial_lr) ** progress
        
        # Update
        h_new = h - lr * sg
        h_new = project_constraint(h_new, n)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5
```

**Idea: Use a "smoothed" max and then do gradient descent.**

Replace max_k C(k) with a smooth approximation: log-sum-exp of C(k)/τ for small τ. This gives a smooth approximation to the max that we can differentiate.

The gradient of (1/τ) * log(∑ exp(τ*C(k))) with respect to h is:
(∑ exp(τ*C(k)) * ∂C(k)/∂h) / (∑ exp(τ*C(k)))

This is a weighted average of the subgradients of the individual C(k), weighted by their "importance" (how close they are to the max).

This should converge faster than the hard max subgradient.

```python
    n = len(h)
    corr = compute_corr(h)  # shape (2n-1,)
    
    # Smooth max: (1/tau) * log(sum(exp(tau * corr)))
    # Gradient: sum(exp(tau*corr) * dC(k)/dh) / sum(exp(tau*corr))
    # For numerical stability, subtract max
    corr_shifted = corr - np.max(corr)
    weights = np.exp(tau * corr_shifted)  # these are the weights
```

```python
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def lag_subgradient(h, m):
    n = len(h)
    grad = np.zeros(n)
    s1 = max(0, -m)
    e1 = min(n, n - m)
    if s1 < e1:
        grad[s1:e1] += 1.0 - h[s1+m:e1+m]
    s2 = max(0, m)
    e2 = min(n, m + n)
    if s2 < e2:
        grad[s2:e2] -= h[s2-m:e2-m]
    return grad

def project_constraint(v, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_constraint(h_new, n_new)

def downsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_constraint(h_new, n_new)

    n = len(h)
    corr = compute_corr(h)
    corr_max = np.max(corr)
    corr_shifted = corr - corr_max
    weights = np.exp(tau * corr_shifted)
    total = np.sum(weights)
    weights /= total
    
    sg = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-10:
            continue
        m = k - (n - 1)
        s1 = max(0, -m)
        e1 = min(n, n - m)
        if s1 < e1:
            sg[s1:e1] += wk * (1.0 - h[s1+m:e1+m])
        s2 = max(0, m)
        e2 = min(n, m + n)
        if s2 < e2:
            sg[s2:e2] -= wk * h[s2-m:e2-m]
    return sg

def optimize(h, budget_s, start_time, target_time_fraction=1.0):
    """Optimize h using smooth subgradient with momentum."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    momentum = np.zeros(n)
    beta = 0.9  # momentum coefficient
    
    end_time = start_time + budget_s * target_time_fraction
    step = 0
    no_improve = 0
    max_no_improve = 5000
    
    # Adaptive parameters
    lr = 0.01
    tau = 1.0  # smoothness parameter
    
    while time.time() < end_time and no_improve < max_no_improve:
        step += 1
        
        # Anneal tau: decrease over time
        progress = min(1.0, step / (max_no_improve * 2))
        tau = max(0.1, 1.0 * (1.0 - progress) + 0.1)
        
        # Compute subgradient
        sg = smooth_subgradient(h, tau)
        
        # Momentum update
        momentum = beta * momentum - lr * sg
        h_new = h + momentum
        h_new = project_constraint(h_new, n)
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Periodic random perturbation
        if no_improve % 2000 == 1000 and no_improve > 1000:
            noise = np.random.normal(0, 0.05, n)
            h = project_constraint(h + noise, n)
            # Reset momentum
            momentum = np.zeros(n)
            no_improve = 0
        
        # Reduce learning rate if stuck
        if no_improve > 3000:
            lr *= 0.5
            momentum = np.zeros(n)
    
    return best_h, best_c5

    # Phase 3: Final refinement at best resolution

    # Final evaluation
    final_c5 = compute_c5(best_h)
```

```python
def lag_subgrad(h, m):

def project(v, n):

        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def upsample(h, n_new):

    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)
```
