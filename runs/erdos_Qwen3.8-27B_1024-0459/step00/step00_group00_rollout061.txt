from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(v, n):
    """Project v onto {0 ≤ x ≤ 1, sum(x) = n/2}."""
    target = n / 2.0
    # Binary search for λ such that sum(clamp(v - λ, 0, 1)) = target
    lo, hi = -v.max() - 1, -v.min() + 1  # ensure bracketing
    # Actually: if λ is very negative, clamp gives 1s; if very positive, gives 0s
    lo, hi = v.min() - 1, v.max() + 1

    # If v - λ < 0, result is 0. If v - λ > 1, result is 1.
    # sum(clamp(v-λ, 0, 1)) is decreasing in λ.
    # When λ → -∞: sum → n (all 1s)
    # When λ → +∞: sum → 0 (all 0s)
    
    def sum_clamp(lam):
        return np.sum(np.clip(v - lam, 0.0, 1.0))

    lo = v.min() - 1.0  # sum_clamp(lo) >= n (all 1s)
