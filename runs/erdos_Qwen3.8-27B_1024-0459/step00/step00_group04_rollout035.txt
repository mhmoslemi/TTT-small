corr[m] = Σ_{valid i} h[i] * (1-h[i+m])

For a fixed m, the partial derivative with respect to h[p]:
∂corr[m]/∂h[p] = Σ_{valid i} [δ_{i,p}(1-h[i+m]) + h[i] * (-δ_{i+m,p})]
= (1-h[p+m]) * [p valid for the sum] - h[p-m] * [p-m valid for the sum]

More precisely:
- The term where i = p: contributes (1-h[p+m]) if p+m is valid (0 ≤ p+m < n)
- The term where i+m = p, i.e., i = p-m: contributes -h[p-m] if p-m is valid (0 ≤ p-m < n)

So:
∂corr[m]/∂h[p] = (1-h[p+m]) * 1_{0≤p+m<n} - h[p-m] * 1_{0≤p-m<n}

(Note that we also need p to satisfy 0≤p<n, which it does since we're taking the gradient.)

- For terms where i+m=b (i=b-m): was h[b-m](1-h[b]) = h[b-m], now 0. Change: -h[b-m] if 0≤b-m<n

```python
def swap_update(h, corr, a, b):
    """Swap h[a] 1->0 and h[b] 0->1, update corr in place. Returns delta of max."""
    n = len(h)
    
    # Compute delta for each m using OLD h values
    for m_idx in range(len(corr)):
        m = m_idx - (n - 1)
        delta = 0.0
        # a: 1 -> 0
        pm_a = a + m
        if 0 <= pm_a < n:
            delta -= (1.0 - h[pm_a])
        pmm_a = a - m
        if 0 <= pmm_a < n:
            delta += h[pmm_a]
        # b: 0 -> 1
        pm_b = b + m
        if 0 <= pm_b < n:
            delta += (1.0 - h[pm_b])
        pmm_b = b - m
        if 0 <= pmm_b < n:
            delta -= h[pmm_b]
        corr[m_idx] += delta
    
    # Actually update h
    h[a] = 0.0
    h[b] = 1.0
    
    return corr
```

```python
    n = len(h)
    shifts = np.arange(-(n-1), n)  # 2n-1 values, index 0 corresponds to shift -(n-1)
    
    # For a: 1 -> 0
    # Term 1: -(1-h[a+m]) for m where 0 <= a+m < n
    #   a+m in [0, a], wait: a+m >= 0 => m >= -a, a+m < n => m < n-a
    #   m range: max(-(n-1), -a) to min(n-1, n-a-1)
    m_valid_a1 = np.arange(max(-(n-1), -a), min(n, n-a))
    if len(m_valid_a1) > 0:
        idx_a1 = m_valid_a1 + (n-1)  # index into corr
        corr[idx_a1] -= (1.0 - h[a + m_valid_a1])
    
    # Term 2: +h[a-m] for m where 0 <= a-m < n
    #   a-m >= 0 => m <= a, a-m < n => m >= a-n+1
    #   m range: max(-(n-1), a-n+1) to min(n-1, a)
    m_valid_a2 = np.arange(max(-(n-1), a-n+1), min(n, a+1))
    if len(m_valid_a2) > 0:
        idx_a2 = m_valid_a2 + (n-1)
        corr[idx_a2] += h[a - m_valid_a2]
    
    # For b: 0 -> 1
    # Term 3: +(1-h[b+m]) for m where 0 <= b+m < n
    m_valid_b1 = np.arange(max(-(n-1), -b), min(n, n-b))
    if len(m_valid_b1) > 0:
        idx_b1 = m_valid_b1 + (n-1)
        corr[idx_b1] += (1.0 - h[b + m_valid_b1])
    
    # Term 4: -h[b-m] for m where 0 <= b-m < n
    m_valid_b2 = np.arange(max(-(n-1), b-n+1), min(n, b+1))
    if len(m_valid_b2) > 0:
        idx_b2 = m_valid_b2 + (n-1)
        corr[idx_b2] -= h[b - m_valid_b2]
    
    # Update h
    h[a] = 0.0
    h[b] = 1.0
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def c5_gradient(h):
    """Gradient of C5 = dx * max_k correlate(h, 1-h, mode='full')[k]"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Find the shift with max correlation (could be multiple, take first)
    max_val = np.max(corr)
    mask = corr >= max_val - 1e-10  # All active constraints
    active_indices = np.where(mask)[0]
    m_values = active_indices - (n - 1)
    
    grad = np.zeros(n)
    for m in m_values:
        for p in range(n):
            val = 0.0
            pm = p + m
            if 0 <= pm < n:
                val += (1.0 - h[pm])
            pmm = p - m
            if 0 <= pmm < n:
                val -= h[pmm]
            grad[p] += dx * val
    grad /= len(m_values)
    return grad
```

```python
def c5_gradient_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = np.argmax(corr)
    m = max_idx - (n - 1)
    
    # For shift m:
    # grad[p] = (1-h[p+m]) * 1[0≤p+m<n] - h[p-m] * 1[0≤p-m<n]
    
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    
    if m >= 0:
        # p+m in [0, n-1] => p in [0, n-1-m]
        term1[:n-m] = 1.0 - h[m:]
        # p-m in [0, n-1] => p in [m, n-1]
        term2[m:] = h[:n-m]
    else:
        # m is negative, let m' = -m
        mp = -m
        # p+m in [0, n-1] => p in [-m, n-1] = [mp, n-1]
        term1[mp:] = 1.0 - h[:n-mp]
        # p-m in [0, n-1] => p+m' in [0, n-1] => p in [0, n-1-m'] = [0, n-1-mp]
        term2[:n-mp] = h[mp:]
    
    grad = dx * (term1 - term2)
    return grad
```

```python
def local_search_01(h_init, n, t_max, seed=42):
    """0/1 local search to minimize C5."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    
    # Start from rounded version
    h = (h_init >= 0.5).astype(float)
    # Fix sum to be exactly n/2
    # If too many 1s, flip some to 0; if too many 0s, flip some to 1
    while h.sum() > n/2:
        idx = rng.choice(np.where(h == 1)[0])
        h[idx] = 0
    while h.sum() < n/2:
        idx = rng.choice(np.where(h == 0)[0])
        h[idx] = 1
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    best_corr = max_corr
    best_h = h.copy()
    
    ones_list = np.where(h == 1)[0]
    zeros_list = np.where(h == 0)[0]
    
    no_improve = 0
    max_no_improve = 5000 * n  # Scale with problem size
    
    while time.time() < t_max:
        if no_improve > max_no_improve:
            break
        
        a = ones_list[rng.randint(len(ones_list))]
        b = zeros_list[rng.randint(len(zeros_list))]
        
        h[a] = 0
        h[b] = 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = np.max(corr)
        
        if new_max <= max_corr:
            # Accept
            if new_max < best_corr:
                best_corr = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            max_corr = new_max
            ones_list = np.where(h == 1)[0]
            zeros_list = np.where(h == 0)[0]
        else:
            # Revert
            h[a] = 1
            h[b] = 0
            no_improve += 1
    
    return best_h, best_corr * dx
```
