    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def enforce_constraints(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Adjust interior points
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
        else:

            if diff > 0:
                idx = np.argmax(1.0 - h)
            else:
                idx = np.argmin(h)
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    # Exact final correction
    residual = target - np.sum(h)
    if abs(residual) > 0:
        if residual > 0:
            candidates = h < 1.0
        else:

            else:

def c5_gradient(h):
    """Compute gradient of C5(h) = max_k corr_k * dx at h"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    k_star = np.argmax(corr)

    # = sum_i h[i] - sum_i h[i] * h[i + j - (n-1)]

    grad = np.zeros(n)

def compute_c5(h):

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
        else:
            if diff > 0:
                idx = np.argmax(1.0 - h)
            else:
                idx = np.argmin(h)
            h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        if residual > 0:
            candidates = h < 1.0
        else:
            candidates = h > 0.0
        if np.any(candidates):

            h = np.clip(h, 0.0, 1.0)
