        h_r, c5_r = optimize_at_resolution(h_interp, n_r, dx_r, maxiter=maxiter_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_r
        
        elapsed = time.time() - t0
        time_left = budget_s - elapsed
    
    # Phase 3: Random restarts at various resolutions
    elapsed = time.time() - t0
    time_left = budget_s - elapsed
    
    restart_attempts = 0
    while time_left > 15 and restart_attempts < 20:
        restart_attempts += 1
        # Pick a resolution
        n_start = rng.choice([best_n, min(best_n * 2, 360), 45, 90, 120])
        n_start = int(n_start)
        dx_s = 2.0 / n_start
        target_s = n_start / 2.0
        
        # Random initialization
        if rng.rand() < 0.4:
            h_start = interpolated_init(best_h, best_n, n_start)
            h_start += rng.randn(n_start) * 0.1
        elif rng.rand() < 0.7:
            h_start = random_binary_init(n_start, rng)
        else:
            h_start = np.full(n_start, 0.5) + rng.randn(n_start) * 0.1
        
        h_start = project_to_constraint(h_start, target_s)
        
        maxiter_s = min(1500, max(100, int(time_left * 0.3 * 5)))
        h_s, c5_s = optimize_at_resolution(h_start, n_start, dx_s, maxiter=maxiter_s, power=30)
        
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
            best_n = n_start
        
        elapsed = time.time() - t0
        time_left = budget_s - elapsed
    
    # Phase 4: Local search refinement at best resolution
    elapsed = time.time() - t0
    time_left = budget_s - elapsed
    if time_left > 10 and best_h is not None:
        dx_best = 2.0 / best_n
        h_ls, c5_ls = local_search(best_h, best_n, dx_best,
                                    max_iters=min(5000, int(time_left * 200)), rng=rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Phase 5: Try one more SLSQP polish at the best solution
    elapsed = time.time() - t0
    time_left = budget_s - elapsed
    if time_left > 10 and best_h is not None:
        dx_best = 2.0 / best_n
        h_pol, c5_pol = optimize_at_resolution(best_h.copy(), best_n, dx_best, maxiter=1000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Final validation
    if best_h is None:
        n_fb = 45
        best_h = np.full(n_fb, 0.5)
        best_c5 = compute_c5_value(best_h, 2.0 / n_fb)
        best_n = n_fb
    
    best_h = project_to_constraint(best_h, best_n / 2.0)
    best_c5 = compute_c5_value(best_h, 2.0 / best_n)
    
    return best_h, best_c5, best_n
