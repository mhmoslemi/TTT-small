def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr), corr, dx

def project(h, target_sum):
    x = h.copy()
    s = np.sum(x)
    if abs(s - target_sum) < 1e-10:
        return np.clip(x, 0.0, 1.0)
    
    def f(lam):
        return np.sum(np.clip(x - lam, 0.0, 1.0)) - target_sum
    
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    
    if f(lo) <= 0:
        result = np.clip(x - lo, 0.0, 1.0)
        return result
    if f(hi) >= 0:
        result = np.clip(x - hi, 0.0, 1.0)
        return result
    
    lam = brentq(f, lo, hi, xtol=1e-13)
    return np.clip(x - lam, 0.0, 1.0)

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    max_scaled = np.max(scaled)
    log_sum = np.log(np.sum(np.exp(scaled - max_scaled)))
    obj = (max_scaled + log_sum) / beta * dx
    
    weights = np.exp(scaled - max_scaled)
    weights /= np.sum(weights)
    
    grad_h = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)
        scale = w * beta * dx
        j1s = max(0, -k)
        j1e = min(n, n - k)
        if j1e > j1s:
            grad_h[j1s:j1e] += scale * (1.0 - h[j1s+k:j1e+k])
        j2s = max(0, k)
        j2e = n
        if j2e > j2s:
            grad_h[j2s:j2e] -= scale * h[j2s-k:j2e-k]
    
    return obj, grad_h

    start = time.time()
    n = len(h)
    h = h.copy()

    best_c5, _, _ = compute_c5(h)

    num_phases = 8
    total_steps = 5000
    
    for phase in range(num_phases):
        if time.time() - start > time_limit:
            break
        beta = 5.0 * (10 ** (phase / 3.0))  # 5, ~14, ~40, ~110, ~300, ~800, ~2200, ~6000
        # Cap beta
        beta = min(beta, 5000.0)
        
        steps = max(100, total_steps // num_phases)
        lr = 1.0 / (beta * 0.5 + 1)  # Scale inversely with beta
        
        for step in range(steps):

            h_new = project(h_new, target_sum)

                best_c5 = c5_new

            if c5_new < best_c5:

                best_c5 = c5_new

    dx_old = 2.0 / n_old
