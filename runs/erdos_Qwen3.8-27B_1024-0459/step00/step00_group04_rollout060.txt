```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_obj(h, target_sum, t=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    # For numerical stability
    corr_shifted = corr - m
    obj = m + np.log(np.sum(np.exp(t * corr_shifted))) / t
    # Penalty for constraint violation
    pen = 1e8 * (np.sum(h) - target_sum) ** 2
    return obj + pen

def smooth_max_grad(h, target_sum, t=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    corr_shifted = corr - m
    # Softmax weights
    exp_vals = np.exp(t * corr_shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Now compute gradient of sum_k weights[k] * corr[k] wrt h
    # (The 1/t * log(sum exp) gradient is sum_k p_k * d corr_k / d h)
    
    # corr[k] where k is index in full correlation (0 to 2n-2)
    # shift for index k: s = k - (n-1)
    # corr[k] = sum_j h[j+s] * (1-h[j]) for valid j
    
    # d corr[k] / d h[i] = (1-h[i-s])*I[0<=i-s<n] - h[i+s]*I[0<=i+s<n]
    
    # gradient[i] = sum_k weights[k] * [(1-h[i-s_k])*I[0<=i-s_k<n] - h[i+s_k]*I[0<=i+s_k<n]]
    # where s_k = k - (n-1)
    
    grad = np.zeros(n)
    # s_k ranges from -(n-1) to (n-1)
    for k in range(2*n - 1):
        if weights[k] < 1e-15:
            continue
        s = k - (n - 1)
        # (1-h[i-s])*I[0<=i-s<n]: i-s in [0,n) means i in [s, s+n)
        # h[i+s]*I[0<=i+s<n]: i+s in [0,n) means i in [-s, n-s)
        
        # First term: for i in [max(0,s), min(n, s+n)):
        i_start1 = max(0, s)
        i_end1 = min(n, s + n)
        if i_start1 < i_end1:
            idx = np.arange(i_start1, i_end1)
            grad[i_start1:i_end1] += weights[k] * (1.0 - h[idx - s])
        
        # Second term: for i in [max(0,-s), min(n, n-s)):
        i_start2 = max(0, -s)
        i_end2 = min(n, n - s)
        if i_start2 < i_end2:
            idx = np.arange(i_start2, i_end2)
            grad[i_start2:i_end2] -= weights[k] * h[idx + s]
    
    grad *= dx
    
    # Penalty gradient
    pen_grad = 2e8 * (np.sum(h) - target_sum)
    grad += pen_grad * np.ones(n)
    
    return grad
```

def compute_c5(h_values):
    n = len(h_values)
    dx = 2.0 / n
    corr = correlate(h_values, 1.0 - h_values, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            add_per = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + add_per, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            rem_per = (-diff) / np.sum(mask)
            h[mask] = np.maximum(h[mask] - rem_per, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def obj_func(h, target_sum, t):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    exp_vals = np.exp(t * shifted)
    obj = m + np.log(np.sum(exp_vals)) / t
    penalty = 1e8 * (np.sum(h) - target_sum) ** 2
    return obj + penalty


def grad_func(h, target_sum, t):

    grad = np.zeros(n)

        if wk < 1e-13:
            continue
        s = k - (n - 1)
        i1a = max(0, s)
        i1b = min(n, s + n)
        if i1a < i1b:
