def project_box_sum(x, m):
    x = np.asarray(x,dtype=float).copy()
    n=x.size
    if n==0: return x
    m = float(m)
    x = np.clip(x,0.0,1.0)
    s = x.sum()
    if abs(s-m) < 1e-13:
       return x
    # Euclidean projection via bisection on lambda
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    # f(lambda)=sum clip(x-lambda)
    for _ in range(60):
       mid=(lo+hi)*0.5
       s_mid = np.sum(np.clip(x-mid,0.0,1.0))
       if s_mid > m: lo=mid
       else: hi=mid
    proj = np.clip(x-(lo+hi)*0.5,0.0,1.0)
    # fix residual proportionally as before
    diff = m - proj.sum()
    if diff > 0:
       head = 1.0-proj; tot=head.sum()
       if tot > 0 and diff <= tot + 1e-9:
          proj += head*(diff/tot)
    elif diff < 0:
       mass = proj; tot=mass.sum()
       if tot > 0 and -diff <= tot + 1e-9:
          proj -= mass*((-diff)/tot)
    proj = np.clip(proj,0,1)
    # final residual loop
    diff = m - proj.sum()
    if abs(diff) > 1e-14:
       if diff > 0:
          order=np.argsort(1.0-proj)[::-1]
          rem=diff
          for idx in order:
             if rem <= 1e-15: break
             add = min(1.0-proj[idx], rem)
             proj[idx] += add; rem -= add
       else:
          order=np.argsort(proj)[::-1]
          rem=-diff
          for idx in order:
             if rem <= 1e-15: break
             sub = min(proj[idx], rem)
             proj[idx] -= sub; rem -= sub
    return proj

def erdos_G(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / n
    if n == 1:
        return np.array([dx * h[0] * (1.0 - h[0])])
    conv = np.convolve(h, h[::-1], mode='full')
    B = conv[n-1:2*n-1]
    pref = np.empty(n+1, dtype=float)
    pref[0]=0.0
    pref[1:] = np.cumsum(h)
    A_plus = pref[n:0:-1]
    A_minus = pref[n] - pref[:n]
    G_plus = dx * (A_plus - B)
    G_minus = dx * (A_minus - B)
    G = np.concatenate((G_plus, G_minus[1:]))
    G = np.maximum(G, 0.0)
    return G

import time
import numpy as np
from scipy import optimize

_SL_BETA = 1.0


def project_box_sum(x, m):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    m = float(m)
    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if abs(s - m) < 1e-13:
        return x

    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s_mid > m:
            lo = mid
        else:
            hi = mid

    proj = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    diff = m - float(proj.sum())
    if diff > 0.0:
        head = 1.0 - proj
        tot = float(head.sum())
        if tot > 0.0 and diff <= tot + 1e-9:
            proj += head * (diff / tot)
        elif tot == 0.0:
            proj = np.full(proj.shape, m / n, dtype=float)
    elif diff < 0.0:
        mass = proj
        tot = float(mass.sum())
        if tot > 0.0 and -diff <= tot + 1e-9:
            proj -= mass * ((-diff) / tot)
        elif tot == 0.0:
            proj = np.zeros(proj.shape, dtype=float)

    proj = np.clip(proj, 0.0, 1.0)

    diff = m - float(proj.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            order = np.argsort(1.0 - proj)[::-1]
            rem = diff
            for idx in order:
                if rem <= 1e-15:
                    break
                add = min(1.0 - float(proj[idx]), rem)
                proj[idx] += add
                rem -= add
        else:
            order = np.argsort(proj)[::-1]
            rem = -diff
            for idx in order:
                if rem <= 1e-15:
                    break
                sub = min(float(proj[idx]), rem)
                proj[idx] -= sub
                rem -= sub
    return proj


def _sl_prepare(x):
    x = np.asarray(x, dtype=float)
    n = x.size
    y = np.clip(x, 0.0, 1.0).copy()
    m = 0.5 * float(n)
    diff = m - float(y.sum())
    if diff > 0.0:
        head = 1.0 - y
        tot = float(head.sum())
        if tot > 0.0:
            y += head * (diff / tot)
    elif diff < 0.0:
        mass = y
        tot = float(mass.sum())
        if tot > 0.0:
            y -= mass * ((-diff) / tot)
    return np.clip(y, 0.0, 1.0)


def erdos_G(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    if n == 1:
        return np.array([dx * h[0] * (1.0 - h[0])], dtype=float)

    conv = np.convolve(h, h[::-1], mode="full")
    B = conv[n - 1:2 * n - 1]

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)

    A_plus = pref[n:0:-1]
    A_minus = pref[n] - pref[:n]

    G_plus = dx * (A_plus - B)
    G_minus = dx * (A_minus - B)
    G = np.concatenate((G_plus, G_minus[1:]))
    return np.maximum(G, 0.0)


def erdos_c5_direct(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1.0
    if n == 1:
        return float(2.0 * h[0] * (1.0 - h[0]))
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / float(n))


def resample_piecewise(h, m):
    h = np.asarray(h, dtype=float)
    n = h.size
    m = int(m)
    if m == n:
        return project_box_sum(h, 0.5 * m)

    dxo = 2.0 / float(n)
    dxm = 2.0 / float(m)

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)

    L = np.arange(m, dtype=float) * dxm
    R = L + dxm
    if R.size:
        R[-1] = 2.0

    iL = np.floor(L / dxo).astype(np.int64)
    iL = np.clip(iL, 0, n - 1)
    remL = L - iL * dxo
    areaL = pref[iL] * dxo
    if n > 0:
        areaL = areaL + h[iL] * remL

    iR = np.floor(R / dxo).astype(np.int64)
    iR = np.clip(iR, 0, n)
    remR = R - iR * dxo
    areaR = pref[iR] * dxo
    maskR = iR < n
    if maskR.any():
        areaR = areaR.astype(float, copy=True)
        areaR[maskR] += h[iR[maskR]] * remR[maskR]

    out = (areaR - areaL) / dxm
    out = np.clip(out, 0.0, 1.0)
    return project_box_sum(out, 0.5 * m)


def slq_constraint(x):
    x = np.asarray(x, dtype=float)
    return np.array([float(x.sum()) - 0.5 * float(x.size)], dtype=float)


def slq_objective(x):
    h = _sl_prepare(x)
    G = erdos_G(h)
    beta = float(_SL_BETA)
    Gmax = float(np.max(G))
    diff = beta * (G - Gmax)
    expv = np.exp(np.clip(diff, -700.0, 700.0))
    s = float(expv.sum())
    if s <= 0.0:
        return float(beta * Gmax + np.log(float(G.size)))
    return float(beta * Gmax + np.log(s))


def slq_gradient(x):
    h = _sl_prepare(x)
    n = h.size
    dx = 2.0 / float(n)
    G = erdos_G(h)
    beta = float(_SL_BETA)
    Gmax = float(np.max(G))
    if Gmax <= 0.0:
        return np.zeros(n, dtype=float)

    diff = beta * (G - Gmax)
    expv = np.exp(np.clip(diff, -700.0, 700.0))
    s = float(expv.sum())
    if s <= 0.0:
        return np.zeros(n, dtype=float)

    w = expv / s

    if n == 1:
        grad_unit = w[0] * (1.0 - 2.0 * h)
    else:
        A = np.zeros(n, dtype=float)
        B = np.zeros(n, dtype=float)
        A[1:] = w[1:n]
        B[1:] = w[n:2 * n - 1]

        prefA = np.cumsum(A)[::-1]
        prefB = np.cumsum(B)

        T = A[1:] + B[1:]
        K = np.zeros(2 * n - 1, dtype=float)
        K[0:n - 1] = T[::-1]
        K[n:] = T

        C = np.convolve(h, K, mode="full")[n - 1:2 * n - 1]
        grad_unit = w[0] * (1.0 - 2.0 * h) + prefA + prefB - C

    return dx * grad_unit


def slsqp_refine(h, beta, maxiter):
    global _SL_BETA
    h0 = project_box_sum(h, 0.5 * float(h.size))
    _SL_BETA = float(beta)
    try:
        bounds = [(0.0, 1.0)] * h0.size
        res = optimize.minimize(
            slq_objective,
            h0,
            jac=slq_gradient,
            method="SLSQP",
            bounds=bounds,
            constraints=[{"type": "eq", "fun": slq_constraint}],
            options={"maxiter": int(maxiter), "ftol": 1e-9, "disp": False},
        )
        if res is not None and hasattr(res, "x"):
            h1 = project_box_sum(res.x, 0.5 * float(h0.size))
        else:
            h1 = h0
    except Exception:
        h1 = h0
    return h1


def add_coord_to_arrays(aplus, amin, h, p, factor, n):
    L = n - 1
    if L <= 0:
        return

    a = n - 1 - p
    if a > 0:
        seg = h[p + 1:p + a + 1]
        aplus[:a] += factor * (1.0 - seg)
        amin[:a] += factor * (-seg)

    b = p
    if b > 0:
        seg = h[p - 1::-1]
        aplus[:b] += factor * (-seg)
        amin[:b] += factor * (1.0 - seg)


def make_pair_lfull(lfull, h, p, q, n, dx):
    lfull.fill(0.0)
    lfull[0] = 2.0 * dx * (h[q] - h[p])
    if n > 1:
        add_coord_to_arrays(lfull[1:n], lfull[n:], h, p, dx, n)
        add_coord_to_arrays(lfull[1:n], lfull[n:], h, q, -dx, n)


def pair_hill_climb(h, n, rng, stop_at, max_no_improve=None, recompute_every=1200):

            if total % recompute_every == 0:

        sdist = abs(p - q)

        cnt += 1

        cnt += 1

                tmp[i2] += qd * d2

            if val < best_val:

def make_resolution_list(base_n, budget_s):
    base = int(base_n)
    cands = [base]
    if base % 2 == 1:
        cands.append(base + 1)
    for k in (2, 3, 4, 6, 8):
        cands.append(int(round(float(base) * float(k))))
    for f in (64, 96, 128, 192, 256, 384, 512):
        cands.append(f)

    out = []
    for x in sorted(set(cands)):
        if x < 16 or x > 512:
            continue
        if x < base:
            continue
        if x != base and x < base + 4:
            continue
        if out and x - out[-1] < max(1, int(0.12 * out[-1])):
            continue
        out.append(int(x))

    if not out:
        out = [base]

    if budget_s < 150:
        return out[:1]
    if budget_s < 400:
        return out[:2]
    if budget_s < 800:
        return out[:3]
    if budget_s < 1300:
        return out[:4]
    return out[:5]


def simple_start(n, seed):
    rng = np.random.RandomState(seed)
    n = int(n)
    h = 0.5 + 0.25 * rng.standard_normal(n)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.perf_counter()
    budget = float(budget_s)
    if budget <= 0:
        budget = 1.0
    reserve = 8.0 if budget < 300.0 else 18.0
    deadline = t_start + max(2.0, budget - reserve)

    try:
        parent = np.array(initial_h_values, dtype=float, copy=True)
    except Exception:
        parent = None

    if parent is None or parent.size == 0:

    parent_c = erdos_c5_direct(h_parent)

    best_h = h_parent.copy()
    best_c = float(parent_c)

    rng = np.random.RandomState(int(seed))

    if time.perf_counter() + 30.0 > deadline:
        final_h = best_h.copy()
        final_c = best_c
        return final_h, float(final_c), int(final_h.size)

    slq_set = set()
    for x in (n_parent, 64, 96, 128):
        if x >= n_parent and x <= 128:
            slq_set.add(int(x))

    res_list = make_resolution_list(n_parent, budget)

    for pi, n in enumerate(res_list):
        now = time.perf_counter()
        if now + 20.0 > deadline:
            break

        remaining = max(0.0, deadline - now)
        left_phases = max(1, len(res_list) - pi)
        phase_alloc = max(30.0, min(320.0, (remaining / float(left_phases)) * 1.15))
        phase_stop = min(deadline - 10.0, time.perf_counter() + phase_alloc)

        if best_h.size == n:
            h = best_h.copy()
        else:
            h = resample_piecewise(best_h, n)

        if int(n) in slq_set and n <= 128 and time.perf_counter() < phase_stop - 70.0:
            c_in = erdos_c5_direct(h)
            maxiter = 42 if n > 80 else 58
            for beta in (12.0, 35.0, 90.0):
                if time.perf_counter() > phase_stop - 45.0:
                    break
                h_ref = slsqp_refine(h, beta, maxiter)
                c_ref = erdos_c5_direct(h_ref)
                if c_ref < c_in - 1e-13:
                    c_in = c_ref
                    h = h_ref
                if c_ref < best_c - 1e-13:
                    best_c = c_ref
                    best_h = h_ref.copy()
        elif int(n) in slq_set and n <= 128 and time.perf_counter() < phase_stop - 60.0:
            c_in = erdos_c5_direct(h)
            h_ref = slsqp_refine(h, 25.0, 35)
            c_ref = erdos_c5_direct(h_ref)
            if c_ref < c_in - 1e-13:
                h = h_ref
            if c_ref < best_c - 1e-13:
                best_c = c_ref
                best_h = h_ref.copy()

        if time.perf_counter() < phase_stop - 5.0:
            h2, c2 = pair_hill_climb(h, int(n), rng, phase_stop)
            if c2 < best_c - 1e-13:
                best_c = c2
                best_h = h2.copy()

    if time.perf_counter() + 35.0 < deadline:
        hfin = best_h.copy()
        hfin, cfin = pair_hill_climb(hfin, int(hfin.size), rng, deadline - 8.0)
        if cfin < best_c - 1e-13:
            best_c = cfin
            best_h = hfin.copy()

    if time.perf_counter() + 55.0 < deadline:
        n_extra = min(512, max(int(best_h.size) * 2, 128))
        if n_extra != int(best_h.size):
            hextra = resample_piecewise(best_h, n_extra)
            hextra, cextra = pair_hill_climb(hextra, int(n_extra), rng, deadline - 8.0)
            if cextra < best_c - 1e-13:
                best_c = cextra
                best_h = hextra.copy()

    final_h = project_box_sum(best_h, 0.5 * float(best_h.size))
    final_c = erdos_c5_direct(final_h)

    if parent_c <= final_c:
        final_h = h_parent
        final_c = parent_c
    elif final_c < best_c:
        best_c = float(final_c)

    return final_h, float(final_c), int(final_h.size)
