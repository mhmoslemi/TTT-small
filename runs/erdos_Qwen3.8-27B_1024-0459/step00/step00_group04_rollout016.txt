    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones

        order = np.argsort(h[ones_idx])[::-1]

        ones = np.where(h_bin == 1)[0]

import numpy as np
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h):
    n = len(h)
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        m = np.sum(mask)
        if m == 0:
            break
        h[mask] += diff / m
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def upsample_linear(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return project_feasible(h.copy())
    if n_new < n_old:
        idx = (np.arange(n_new) * n_old // n_new)
        return project_feasible(h[idx].copy())
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new)


def random_feasible(n, rng):
    h = rng.random(n)
    return project_feasible(h)


def sa_loop(h, n_iter, t0, t1, pert_frac, pert_mag, rng, deadline):
    h = h.copy()
    n = len(h)
    c5_cur = evaluate_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    n_p = max(1, int(pert_frac * n))
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t = t0 * (t1 / t0) ** (i / max(n_iter - 1, 1))
        idx = rng.choice(n, n_p, replace=False)
        h_new = h.copy()
        h_new[idx] += rng.uniform(-pert_mag, pert_mag, n_p)
        h_new = project_feasible(h_new)
        c5_new = evaluate_c5(h_new)
        if c5_new <= c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            ratio = np.exp(-(c5_new - c5_cur) / max(t, 1e-15))
            if rng.random() < ratio:
                h = h_new
                c5_cur = c5_new
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h.copy()
    return h_best, c5_best


def subgrad_polish(h, n_steps, lr0, lr1, rng, deadline):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    c5_cur = evaluate_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    for s in range(n_steps):
        if time.time() > deadline:
            break
        lr = lr0 * (lr1 / lr0) ** (s / max(n_steps - 1, 1))
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)
        grad = np.ones(n)
        if lag == 0:
            grad -= 2.0 * h
        elif lag > 0:
            grad[:n - lag] -= h[lag:]
            grad[lag:] -= h[:n - lag]
        else:
            l = -lag
            grad[l:] -= h[:n - l]
            grad[:n - l] -= h[l:]
        h -= lr * dx * grad
        h = project_feasible(h)
        c5_new = evaluate_c5(h)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()
            c5_cur = c5_new
    return h_best, c5_best


def binary_local_search(h, n_iter, rng, deadline, t_start=0.01):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])[::-1]
        for i in range(n_ones - target_ones):
            h_bin[ones_idx[order[i]]] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(h[zeros_idx])
        for i in range(target_ones - n_ones):
            h_bin[zeros_idx[order[i]]] = 1.0
    h_bin = project_feasible(h_bin)
    c5_cur = evaluate_c5(h_bin)
    h_best = h_bin.copy()
    c5_best = c5_cur
    t = t_start
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t_start * 0.999
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_new = h_bin.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        c5_new = evaluate_c5(h_new)
        if c5_new <= c5_cur or (c5_new - c5_cur) < t * 0.01:
            h_bin = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_bin.copy()
    return h_best, c5_best
