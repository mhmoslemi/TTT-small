import numpy as np
from scipy.optimize import minimize
import time

def project_h(h, n):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12:
        h = h * (n / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
        # Iterative projection
        for _ in range(50):
            s = np.sum(h)
            target = n / 2.0
            if abs(s - target) < 1e-12:
                break
            h = h + (target - s) / n
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def c5_smooth(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) * beta
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    log_sum = m * beta + np.log(np.sum(np.exp(shifted)))
    return log_sum / beta

def corr_grad_at_shift(h, s, n):
    g = np.zeros(n)
    if s > 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] += -h[s:]
    elif s < 0:
        sp = -s
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] += -h[:n - sp]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def c5_smooth_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) * beta
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-15:
        return np.zeros(n)
    weights = exp_shifted / total
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        s = k - (n - 1)
        g = corr_grad_at_shift(h, s, n)
        grad += w * dx * g
    return grad

def constraint_fun(h, n):
    return np.sum(h) - n / 2.0

def constraint_jac(h, n):
    return np.ones(n)

def generate_random_h(n, rng):
    h = rng.rand(n)
    h = project_h(h, n)
    return h

def generate_binary_h(n, rng):
    h = np.zeros(n)
    half = n // 2
    if n % 2 == 1:
        extra = n - 2 * half
        half = n // 2
        extra = n - 2 * half
    indices = rng.choice(n, size=half, replace=False)
    h[indices] = 1.0
    if n % 2 == 1:
        remaining = np.where(h == 0)[0]
        if len(remaining) > 0:
            h[remaining[0]] = 1.0
    h = project_h(h + 0.1 * rng.rand(n), n)
    return h

def generate_structured_h(n, rng):
    # Alternating blocks
    num_blocks = rng.randint(2, max(3, n // 5) + 1)
    block_sizes = np.maximum(1, rng.multinomial(n, np.ones(num_blocks) / num_blocks))
    h = np.zeros(n)
    pos = 0
    for bi, bs in enumerate(block_sizes):
        if bi % 2 == 0:
            h[pos:pos + bs] = 1.0
        pos += bs
    h = project_h(h + 0.05 * rng.rand(n), n)
    return h

def generate_smooth_h(n, rng):
    freq = rng.choice([1, 2, 3, 4])
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0, 1)
    h = project_h(h, n)
    return h

def optimize_h_slsqp(h0, n, time_limit, beta_start=20.0):
    """Optimize using SLSQP with adaptive smoothing."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac, 'args': (n,)}]
    
    h = h0.copy()
    h = project_h(h, n)
    best_local = h.copy()
    best_local_c5 = c5_value(h)
    
    start = time.time()
    beta = beta_start
    
    for iteration in range(5):
        elapsed = time.time() - start
        if elapsed > time_limit * 0.85:
            break
        
        sub_time = time_limit * 0.12
        maxiter = max(50, int(sub_time * 20))
        
        try:
            result = minimize(
                c5_smooth, h,
                args=(n, beta),
                jac=c5_smooth_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-12, 'maxls': 50}
            )
            h_new = result.x
            h_new = project_h(h_new, n)
            c5_new = c5_value(h_new)
            if c5_new < best_local_c5:
                best_local_c5 = c5_new
                best_local = h_new.copy()
            h = h_new
        except Exception:
            pass
        
        elapsed = time.time() - start
        if elapsed > time_limit * 0.9:
            break
        
        # Increase beta for sharper max approximation
        beta = min(beta * 2.0, 200.0)
        
        # Add small perturbation if stuck
        if not result.success and hasattr(result, 'success'):
            h = project_h(h + 0.01 * np.random.RandomState(int(time.time()*1000) % 100000).rand(n), n)
    
    return best_local, best_local_c5

def optimize_h_nocipher(h0, n, time_limit):
    """Stochastic local search with binary-ish moves."""
    h = h0.copy()
    h = project_h(h, n)
    best = h.copy()
    best_c5 = c5_value(h)
    rng = np.random.RandomState(42)
    start = time.time()
    temp = 0.01
    dx = 2.0 / n
    
    while time.time() - start < time_limit:
        if time.time() - start > time_limit * 0.95:
            break
        # Random perturbation
        mask = rng.rand(n) < 0.1
        delta = (rng.rand(n) - 0.5) * 0.3 * mask
        h_new = project_h(h + delta, n)
        c5_new = c5_value(h_new)
        
        if c5_new < best_c5 or rng.rand() < np.exp(-(c5_new - best_c5) / temp):
            h = h_new
            if c5_new < best_c5:
                best = h_new.copy()
                best_c5 = c5_new
        temp *= 0.999
        temp = max(temp, 0.001)
    
    return best, best_c5

def upscale_h(h, n_new):
    """Upscale h from its current length to n_new using linear interpolation."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    h_new = project_h(h_new, n_new)
    return h_new
