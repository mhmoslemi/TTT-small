    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    shift_offset = n - 1
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-14:
            continue
        m_val = idx - shift_offset
        lo1 = max(0, -m_val)
        hi1 = min(n - 1, n - 1 - m_val)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * (1.0 - h[lo1 + m_val:hi1 + m_val + 1])
        lo2 = max(0, m_val)
        hi2 = min(n - 1, n - 1 + m_val)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * h[lo2 - m_val:hi2 - m_val + 1]
    grad *= dx
    return grad

def _project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _slsqp_optimize(h, n, dx, temp, max_iter=400):
    target = n / 2.0
    h = _project(h, target)
    
    def obj(hv):
        return _obj_smooth(hv, dx, temp)
    
    def grad(hv):
        return _grad_smooth(hv, dx, temp)
    
    cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - target,
             'jac': lambda hv: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    if res.x is not None:
        res.x = np.clip(res.x, 0.0, 1.0)
        res.x = _project(res.x, target)
    return res.x

def _round_step(h, dx, n):
    target = n / 2.0
    h_best = h.copy()
    c5_best = compute_c5(h_best, dx)
    for i in range(n):
        for trial in [0.0, 1.0]:
            if abs(h[i] - trial) < 1e-6:
                continue
            h_test = h_best.copy()
            h_test[i] = trial
            h_test = _project(h_test, target)
            c5_test = compute_c5(h_test, dx)
            if c5_test < c5_best - 1e-12:
                h_best = h_test
                c5_best = c5_test
    return h_best, c5_best

def _random_perturb(h, n, strength=0.3):
    h_p = h.copy()
    n_ones = np.sum(h_p > 0.5)
    idx = np.random.choice(n, size=max(1, int(n * 0.1)), replace=False)
    for i in idx:
        h_p[i] = np.random.choice([0.0, 1.0])
    h_p = _project(h_p, n / 2.0)
    return h_p

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def time_up():
        return (time.time() - t0) >= budget_s - 10
    
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    
    h0 = _project(h0, target0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)
    best_n = n0
    
    # Phase 1: SLSQP with decreasing temperature
    for temp in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if time_up():
            break
        h_new = _slsqp_optimize(best_h, n0, dx0, temp, max_iter=300)
        c5_new = compute_c5(h_new, dx0)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 2: Rounding steps
    if not time_up():
        for _ in range(5):
            best_h, best_c5 = _round_step(best_h, dx0, n0)
            if time_up():
                break
    
    # Phase 3: Random restarts on parent resolution
    if not time_up():
        n_restarts = 0
        while not time_up() and n_restarts < 30:
            h_pert = _random_perturb(best_h, n0, 0.4)
            h_opt = _slsqp_optimize(h_pert, n0, dx0, 0.01, max_iter=200)
            c5_opt = compute_c5(h_opt, dx0)
            if c5_opt < best_c5 - 1e-10:
                best_c5 = c5_opt
                best_h = h_opt
            n_restarts += 1
    
    # Phase 4: Try slightly different resolutions
    for trial_n in [94, 96, 100, 92]:
        if time_up():
            break
        dx_t = 2.0 / trial_n
        target_t = trial_n / 2.0
        # Interpolate best_h to new resolution
        if trial_n == n0:
            h_t = best_h.copy()
        else:
            x_old = np.linspace(0, 2, n0)
            x_new = np.linspace(0, 2, trial_n)
            # Use step function: assign each new point to nearest old point
            h_t = np.zeros(trial_n)
            for j in range(trial_n):
                idx = min(n0 - 1, int(x_new[j] / dx_t * 1 + 0.5) * (n0 - 1) / (trial_n - 1) if trial_n > 1 else 0)
                idx = int(np.argmin(np.abs(x_old - x_new[j])))
                h_t[j] = h0[idx]
            h_t = _project(h_t, target_t)
        
        h_opt = _slsqp_optimize(h_t, trial_n, dx_t, 0.02, max_iter=200)
        c5_opt = compute_c5(h_opt, dx_t)
        if c5_opt < best_c5 - 1e-10:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = trial_n
        if not time_up():
            h_opt2 = _slsqp_optimize(h_opt, trial_n, dx_t, 0.005, max_iter=200)
            c5_opt2 = compute_c5(h_opt2, dx_t)
            if c5_opt2 < best_c5 - 1e-10:
                best_c5 = c5_opt2
                best_h = h_opt2
                best_n = trial_n
    
    # Phase 5: Final polishing on best solution
    if not time_up():
        dx_b = 2.0 / best_n
        for temp in [0.003, 0.001]:
            if time_up():
                break
            h_pol = _slsqp_optimize(best_h, best_n, dx_b, temp, max_iter=200)
            c5_pol = compute_c5(h_pol, dx_b)
            if c5_pol < best_c5 - 1e-10:
                best_c5 = c5_pol
                best_h = h_pol
        # One more round step
        if not time_up():
            best_h, best_c5 = _round_step(best_h, dx_b, best_n)
            if not time_up():
                best_h, best_c5 = _round_step(best_h, dx_b, best_n)
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    
    return best_h, best_c5, best_n
