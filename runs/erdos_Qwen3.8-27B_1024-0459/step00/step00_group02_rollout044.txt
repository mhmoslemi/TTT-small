def project_to_feasible(h, n):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2} using bisection on offset."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_c = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_c)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_c = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune: adjust by tiny amount to hit target exactly
    diff = target - np.sum(h_c)
    if abs(diff) > 1e-10:
        # Distribute the difference among entries not at bounds
        mask = (h_c > 1e-8) & (h_c < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h_c[mask] += diff / np.sum(mask)
            h_c = np.clip(h_c, 0.0, 1.0)
    return h_c

def compute_subgradient(h, n):
    """Compute subgradient of max_k corr[k]*dx w.r.t. h."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_val = np.max(corr)
    # Active set: all k within tolerance of max
    tol = 1e-10
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            lo, hi = s, n - 1 - s
        else:
            lo, hi = 0, n - 1 + s
        
        # For j in [lo, hi]: grad[j] += (1 - h[j-s])
        # For j where j+s in [lo, hi]: grad[j] -= h[j+s], i.e., j in [lo-s, hi-s]
        
        idx1 = np.arange(lo, hi + 1)
        if len(idx1) > 0:
            grad[idx1] += (1.0 - h[idx1 - s])
        
        lo2, hi2 = lo - s, hi - s
        lo2 = max(0, lo2)
        hi2 = min(n - 1, hi2)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            grad[idx2] -= h[idx2 + s]
    
    grad *= dx / len(active)
    return grad, active, corr

def optimize_h(h_init, n, max_iter=500, lr_init=0.1, verbose=False):
