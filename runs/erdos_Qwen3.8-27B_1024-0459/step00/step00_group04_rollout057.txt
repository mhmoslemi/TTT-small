import time

def compute_objective(h):
    """Compute C5 value."""
    n = len(h)
    dx = 2.0 / n
    if hasattr(h, 'shape') and h.shape[0] > 1:
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * dx)
    return 0.0

def obj_and_grad(h, n, beta=50.0):
    """Smoothed max of correlations, and its gradient."""
    dx = 2.0 / n
    h = np.asarray(h)
    u = 1.0 - h
    corr = np.correlate(h, u, mode='full')  # length 2n-1
    vals = corr * dx
    
    # Smoothed max
    v_max = np.max(vals)
    shifted = beta * (vals - v_max)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = (v_max + np.log(sum_exp)) / beta
    
    # Weights for smoothed gradient
    w = exp_vals / sum_exp
    
    # Gradient: sum_p w[p] * dx * dc[p]/dh
    # For each p, k = p - (n-1)
    # dc[p]/dh[m] = I(j_lo ≤ m ≤ j_hi) - h[m-k]*I(j_lo ≤ m ≤ j_hi) - h[m+k]*I(i_lo ≤ m ≤ i_hi)
    # where i_lo = max(0,-k), i_hi = min(n-1, n-1-k), j_lo = i_lo+k, j_hi = i_hi+k
    
    grad = np.zeros(n)
    
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-12:
            continue
        k = p - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo > i_hi:
            continue
        j_lo = i_lo + k
        j_hi = i_hi + k
        
        # First term: I(j_lo ≤ m ≤ j_hi)
        grad[j_lo:j_hi+1] += wp * dx
        
        # Second term: -h[m-k] for m in [j_lo, j_hi]
        grad[j_lo:j_hi+1] -= wp * dx * h[j_lo - k:j_hi + 1 - k]
        
        # Third term: -h[m+k] for m in [i_lo, i_hi]
        grad[i_lo:i_hi+1] -= wp * dx * h[i_lo + k:i_hi + 1 + k]
    
    return obj, grad
