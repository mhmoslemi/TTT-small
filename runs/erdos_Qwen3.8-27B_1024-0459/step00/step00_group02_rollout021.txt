    return float(np.max(corr))

def smooth_c5(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    if temp < 1e-10:
        return m, corr
    shifted = (corr - m) / temp
    # For numerical stability
    log_sum = temp * (np.max(shifted) + np.log(np.sum(np.exp(shifted - np.max(shifted)))))
    return m + log_sum, corr

        f1, _ = smooth_c5(h, n, temp)
        h[j] -= eps
        grad[j] = (f1 - f0) / eps

        # Final SLSQP with very low temperature

        try_update(h_final, n_best)
        # Final coordinate descent
        h_final = coordinate_descent(h_final, n_best, 10000, min(60, time_remaining() * 0.3), rng)
        try_update(h_final, n_best)

    # Use evaluate_erdos_solution if available for final check
    if best_h is not None:
        try:
            eval_result = evaluate_erdos_solution(best_h, len(best_h))
            if eval_result is not None:
                if isinstance(eval_result, tuple):
                    best_c5 = min(best_c5, float(eval_result[0]))
                else:
                    best_c5 = min(best_c5, float(eval_result))
        except Exception:
            pass

    if best_h is None:
        n = 45
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
