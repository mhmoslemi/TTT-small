def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        if diff > 0:
            h[mask] = np.minimum(h[mask] + adj, 1.0)
        else:
            h[mask] = np.maximum(h[mask] + adj, 0.0)
    return h

def booth_sequence(n):
    """Generate a Booth-like sequence (low autocorrelation)."""
    # Use a simple greedy construction
    h = np.zeros(n)
    # Place n/2 ones using a quadratic pattern
    if n % 2 == 0:
        half = n // 2
        positions = [(i * i) % n for i in range(half)]
        positions = sorted(set(positions))
        # Fill in if we don't have enough unique positions
        if len(positions) < half:
            for i in range(n):
                if i not in positions and len(positions) < half:
                    positions.append(i)
        for p in positions[:half]:
            h[p] = 1.0
    return h

def random_binary(n):
    """Random binary sequence with exactly n/2 ones."""
    h = np.zeros(n)

        ones = np.random.choice(n, n//2, replace=False)

    target = n / 2.0
    h = repair_sum(h.copy(), target)
    current = compute_c5(h)
    best = current
    best_h = h.copy()
    step = 0.03

        max_idx = np.argmax(ov)
        k = max_idx - (n - 1)

        if k >= 0:
            i_vals = np.arange(n - k)
        else:
            i_vals = np.arange(-k, n)
        
        j_vals = i_vals + k
        contribs = h[i_vals] * (1.0 - h[j_vals])
        
        # Try worst contributors
        order = np.argsort(contribs)[::-1]
        improved = False
        for idx in order[:15]:
            i, j = i_vals[idx], j_vals[idx]
            c = contribs[idx]
            if c < 1e-8:
                break

            # Try moving from i to j
            if h[i] - d >= 0 and h[j] + d <= 1:
                h[i] -= d; h[j] += d
                nc = compute_c5(h)
                if nc < current:
                    current = nc
                    if nc < best:
                        best = nc; best_h = h.copy()

                current = best

def smooth_optimize(h, n, alpha=50.0, max_iters=200):
    """Use scipy to optimize a smoothed version of the max."""
    dx = 2.0 / n
    target = n / 2.0
    
    def objective(h_flat):
        h_use = np.clip(h_flat, 0, 1)
        corr = np.correlate(h_use, 1.0 - h_use, mode="full") * dx
        m = np.max(corr)
        shifted = alpha * (corr - m)
        return m + np.log(np.sum(np.exp(np.clip(shifted, -500, 500)))) / alpha

    )
    h_opt = repair_sum(result.x, target)
    return h_opt, compute_c5(h_opt)

`np.correlate(h, 1-h, mode="full")` for h of length n gives an array of length 2n-1.

The k-th element (0-indexed) is:
c[k] = sum_{i} h[i] * (1-h)[i - k + n - 1]

For k = n-1 (the middle element):
c[n-1] = sum_{i=0}^{n-1} h[i] * (1-h)[i] = sum h[i]*(1-h[i])

This is the "self-overlap" at lag 0.

For k = 0:
c[0] = h[0] * (1-h)[n-1]

For k = 2n-2:
c[2n-2] = h[n-1] * (1-h)[0]

So the shifts go from -(n-1) to +(n-1), with shift 0 at index n-1.

The max over all these (times dx) is our C₅.

Note that at shift 0: c[n-1] = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]².
If h is binary, this is 0. If h is at 0.5, this is 0.5*n*0.25 = n/8.

So for binary h, the shift-0 overlap is 0, and we're only concerned with non-zero shifts.

For non-binary h, the shift-0 overlap can be significant, which pushes C₅ up. This suggests that pushing h towards binary values (0 or 1) is beneficial.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += d / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def overlaps_arr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def to_binary(h, n):
    n1 = n // 2
    idx = np.argsort(h)[::-1][:n1]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def local_search_continuous(h, n, max_iters, t_start, time_limit):

    h = repair_sum(h.copy(), target)
    current = compute_c5(h)
    best = current
    best_h = h.copy()
    step = 0.04
    no_improve = 0

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        ov = overlaps_arr(h)
        max_idx = np.argmax(ov)
        k = max_idx - (n - 1)
        if k >= 0:
            iv = np.arange(n - k)
        else:
            iv = np.arange(-k, n)
        jv = iv + k
        contribs = h[iv] * (1.0 - h[jv])
        improved = False
        top_count = min(12, len(contribs))
        order = np.argsort(contribs)[::-1][:top_count]
        for oi in order:
            i, j = iv[oi], jv[oi]
            d = step * np.random.uniform(0.4, 1.0)
            if h[i] - d >= 0 and h[j] + d <= 1:
                h[i] -= d; h[j] += d
                nc = compute_c5(h)
                if nc < current - 1e-12:
                    current = nc
                    if nc < best:
                        best = nc; best_h = h.copy()
                    improved = True

                h[i] += d; h[j] -= d

def make_quadratic_binary(n):
    h = np.zeros(n)
    half = n // 2
    pos = sorted(set([(i * i) % n for i in range(n)]))
    for p in pos[:half]:
        h[p] = 1.0
    if np.sum(h) < half:
        for p in range(n):
            if h[p] == 0 and np.sum(h) < half:
                h[p] = 1.0
    return h

def make_phase_binary(n):
    h = np.zeros(n)
    half = n // 2
    placed = 0
    step_size = max(1, n // (2 * half + 1))
    idx = 0
    while placed < half and idx < n * 2:
        p = idx % n
        if h[p] == 0:
            h[p] = 1.0
            placed += 1
        idx += step_size
    if placed < half:
        for p in range(n):
            if h[p] == 0 and placed < half:
                h[p] = 1.0
                placed += 1
    return h
