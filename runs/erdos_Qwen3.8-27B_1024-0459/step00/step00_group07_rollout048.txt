```python
import numpy as np
from scipy import fft, optimize
import time

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    n = len(h)
    dx = 2.0 / n
    # Use FFT for correlation
    nfft = 1
    while nfft < 2*n-1:
        nfft *= 2
    H = fft.rfft(h, nfft)
    G = fft.rfft(1.0 - h, nfft)
    c = np.real(fft.irfft(H * np.conj(G), nfft)[:2*n-1])
    return np.max(c) * dx, c * dx

def compute_gradient(h, active_indices):
    """Compute gradient of max(c) w.r.t. h, averaged over active indices."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    grad = np.zeros(n)
    for m in active_indices:
        s = m - (n-1)
        # c[m] = sum_i h[i] * g[i+s] for valid i
        # dc[m]/dh[j] = g[j+s] (if valid) - h[j-s] (if valid)
        lo = max(0, -s)
        hi = min(n, n - s)
        # For j in [lo, hi): term g[j+s] contributes
        # For j in [max(0,s), min(n, n+s)): term -h[j-s] contributes
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += g[lo1+s:hi1+s]
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2-s:hi2-s]
    grad /= len(active_indices)
    grad *= dx
    return grad
```

```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy import optimize
import time

def compute_correlation(h):
    """Compute full cross-correlation of h with 1-h using FFT."""
    n = len(h)
    nfft = 1
    while nfft < 2 * n - 1:
        nfft *= 2
    H = rfft(h, nfft)
    G = rfft(1.0 - h, nfft)
    c = irfft(H * np.conj(G), nfft, s=nfft)[:2*n-1]
    return c

def compute_c5(c, n):
    """Compute C5 from correlation array."""
    dx = 2.0 / n
    return np.max(c) * dx

def project_onto_constraint(h, n):
    """Project h onto {sum(h) = n/2, 0 <= h <= 1}."""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    # Distribute the difference
    if diff > 0:
        # Need to add
        can_add = (1.0 - h) > 0
        if np.sum(can_add) > 0:
            h[can_add] += diff / np.sum(can_add)
            h = np.clip(h, 0, 1)
    elif diff < 0:
        can_sub = h > 0
        if np.sum(can_sub) > 0:
            h[can_sub] += diff / np.sum(can_sub)
            h = np.clip(h, 0, 1)
    return h

def upscale(h, n_new):
    """Upscale h from n_old to n_new points using linear interpolation."""
    n_old = len(h)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def compute_gradient_active(h, n, top_k=20):
    """Compute subgradient of max(c) averaged over top-k active lags."""
    c = compute_correlation(h)
    dx = 2.0 / n
    # Find top-k indices
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for m in top_idx:
        s = m - (n - 1)
        # Term 1: (1-h[j+s]) for j in [max(0,-s), min(n, n-s))
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
        # Term 2: -h[j-s] for j in [max(0,s), min(n, n+s))
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2-s:hi2-s]
    grad /= k
    grad *= dx
    return grad, c
```

```python
def project_constraint(h, n):
    """Project onto {x: sum(x)=n/2, 0≤x≤1}."""
    target = n / 2.0
    # Bisection for lambda
    lo = -np.max(h)
    hi = 1.0 - np.min(h)
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - mid, 0, 1)
    return x
```
