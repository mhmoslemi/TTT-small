    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def enforce_sum_constraint(h, target_sum):
    """Adjust h to have sum = target_sum while keeping 0<=h<=1"""
    h = h.copy()
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add mass; add to elements with room
        order = np.argsort(h)  # lowest first (most room)
        remaining = diff
        for idx in order:
            space = 1.0 - h[idx]
            add = min(space, remaining)
            h[idx] += add
            remaining -= add
            if remaining < 1e-12:
                break
    else:

        order = np.argsort(-h)  # highest first
        remaining = -diff
        for idx in order:
            rem = min(h[idx], remaining)
            h[idx] -= rem
            remaining -= rem
            if remaining < 1e-12:

        # Coordinate descent
        if time_left() > 30:
            if n <= 120:
                h, c5 = coordinate_descent_full(h, dx, target_sum, max_passes=25,
                                                rng=rng, time_check=start_time,
                                                time_limit=budget_s * 0.82)
            else:
                h, c5 = coordinate_descent_random(h, dx, target_sum, max_passes=20,
                                                  rng=rng, time_check=start_time,
                                                  time_limit=budget_s * 0.82)

        h = enforce_constraint(h, target_sum)
        c5 = compute_c5(h, dx)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Final refinement phase
    if best_h is not None and time_left() > 40:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        h = best_h.copy()
        c5 = best_c5

        # Another round of L-BFGS-B with higher smoothness parameter
        for p_try in [80, 120]:
            if time_left() < 20:
                break
            lam = 100.0
            def obj_final(x):
                return objective_fn(x, dx, target_sum, lam, p_try)
            try:
                result3 = minimize(obj_final, h, method='L-BFGS-B',
                                  bounds=[(0, 1)] * n,
                                  options={'maxiter': 200, 'ftol': 1e-14})
                if result3.x is not None:
                    h_test = enforce_constraint(result3.x, target_sum)
                    c5_test = compute_c5(h_test, dx)
                    if c5_test < c5:
                        h = h_test
                        c5 = c5_test
            except Exception:
                pass

        # Final coordinate descent
        if n <= 120:
            h, c5 = coordinate_descent_full(h, dx, target_sum, max_passes=20,
                                            rng=rng, time_check=start_time,
                                            time_limit=budget_s * 0.96)
        else:
            h, c5 = coordinate_descent_random(h, dx, target_sum, max_passes=15,
                                              rng=rng, time_check=start_time,
                                              time_limit=budget_s * 0.96)

        h = enforce_constraint(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Final validation
    if best_h is None:
        best_h = np.full(95, 0.5)
        best_n = 95
        best_h = enforce_constraint(best_h, 47.5)
        best_c5 = compute_c5(best_h, 2.0 / 95)

    best_h = enforce_constraint(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / best_n)

    return best_h, final_c5, best_n
