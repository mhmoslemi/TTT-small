        if pm < 1e-15:

        i_end = min(n, l + n)
        if i_start < i_end:

            grad[i_start:i_end] += pm * (1 - h[idx - l])
        # -h[i+l]: valid for 0 <= i+l < n, i.e., -l <= i < n-l

        i_end2 = min(n, n - l)
        if i_start2 < i_end2:

            grad[i_start2:i_end2] -= pm * h[idx2 + l]
    
    return dx * grad
