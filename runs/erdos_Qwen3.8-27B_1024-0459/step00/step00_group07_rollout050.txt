    n = len(h)

    corr = np.correlate(h, g, mode='full') * dx

    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    weights = np.exp(shifted)

def _sinusoidal_init(n, freq, phase):

    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    return _fix_constraint(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.ones(80) * 0.5

    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _compute_c5(best_h)

    # Phase 1: Interpolate to moderate resolution and optimize
    n1 = 400
    h1 = _interp_h(h_parent, n1)
    c5_1 = _compute_c5(h1)
    if c5_1 < best_c5:
        best_h = h1.copy()
        best_c5 = c5_1

    time_left = budget_s - (time.time() - t0) - 40
    if time_left > 30:
        h_opt1 = _projected_grad_descent(h1, n1, 25.0, min(time_left, 180), max_iters=30000)
        c5_opt1 = _compute_c5(h_opt1)
        if c5_opt1 < best_c5:
            best_h = h_opt1.copy()
            best_c5 = c5_opt1

    time_left = budget_s - (time.time() - t0) - 30
    if time_left > 30:
        h_slq = _slsqp_optimize(best_h, n1, 40.0, min(time_left, 120))
        c5_slq = _compute_c5(h_slq)
        if c5_slq < best_c5:
            best_h = h_slq.copy()
            best_c5 = c5_slq

    # Phase 2: Try alternative starts
    time_left = budget_s - (time.time() - t0) - 30
    if time_left > 60:
        for trial in range(3):
            if time.time() - t0 > budget_s - 30:
                break
            if trial == 0:
                h_alt = _random_binary_init(n1, rng)
            elif trial == 1:
                h_alt = _structured_init(n1, rng)
            else:
                h_alt = _sinusoidal_init(n1, 3 + trial, rng.uniform(0, 5))

            h_alt_adj = _interp_h(best_h, n1)
            h_alt = 0.7 * h_alt + 0.3 * h_alt_adj
            h_alt = _fix_constraint(h_alt, n1)

            tl = min(time_left - trial * 40, 80)
            if tl < 15:
                break
            h_opt_alt = _projected_grad_descent(h_alt, n1, 25.0, tl, max_iters=20000)
            c5_alt = _compute_c5(h_opt_alt)
            if c5_alt < best_c5:
                best_h = h_opt_alt.copy()
                best_c5 = c5_alt
            time_left = budget_s - (time.time() - t0) - 30

    # Phase 3: Upscale for higher precision
    time_left = budget_s - (time.time() - t0) - 20
    if time_left > 60:
        n2 = 800
        h2 = _interp_h(best_h, n2)
        c5_2 = _compute_c5(h2)

        tl2 = min(time_left - 10, 200)
        if tl2 > 30:
            h_opt2 = _projected_grad_descent(h2, n2, 30.0, tl2, max_iters=20000)
            c5_opt2 = _compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_h = h_opt2.copy()
                best_c5 = c5_opt2
            else:
                if c5_2 < best_c5:
                    best_h = h2.copy()
                    best_c5 = c5_2

    # Phase 4: SLSQP polish on best solution
    time_left = budget_s - (time.time() - t0) - 10
    if time_left > 20 and len(best_h) <= 500:
        h_polish = _slsqp_optimize(best_h, len(best_h), 50.0, min(time_left, 60))
        c5_polish = _compute_c5(h_polish)
        if c5_polish < best_c5:
            best_h = h_polish.copy()
            best_c5 = c5_polish

    # Final validation
    best_h = _fix_constraint(best_h, len(best_h))
    final_c5 = _compute_c5(best_h)
    final_n = len(best_h)

    return best_h, final_c5, final_n
