  Contribution: ∂/∂h[j] [h[j] * (1-h[j-d])] = (1-h[j-d])

- j appears as "i" when i = j: valid if 0 ≤ j < n (always) and 0 ≤ j+d < n.
  Contribution: ∂/∂h[j] [h[j+d] * (1-h[j])] = -h[j+d]

So:
∂c/∂h[j] = (1-h[j-d]) * [0 ≤ j-d < n] - h[j+d] * [0 ≤ j+d < n]

Let me verify with a simple example. n=4, h = [1,0,1,0], d=1.
c = h[1]*(1-h[0]) + h[2]*(1-h[1]) + h[3]*(1-h[2]) = 0*0 + 1*1 + 0*0 = 1.

∂c/∂h[0]: j=0, j-d=-1 (invalid), j+d=1 (valid). = 0 - h[1] = 0 - 0 = 0.
∂c/∂h[1]: j=1, j-d=0 (valid), j+d=2 (valid). = (1-h[0]) - h[2] = 0 - 1 = -1.
∂c/∂h[2]: j=2, j-d=1 (valid), j+d=3 (valid). = (1-h[1]) - h[3] = 1 - 0 = 1.
∂c/∂h[3]: j=3, j-d=2 (valid), j+d=4 (invalid). = (1-h[2]) - 0 = 0.

Check: if I increase h[1] by ε: h = [1,ε,1,0]. 
c = ε*(1-1) + 1*(1-ε) + 0*(1-1) = 0 + 1-ε + 0 = 1-ε.
So ∂c/∂h[1] = -1. ✓

If I increase h[2] by ε: h = [1,0,1+ε,0]. But h must be in [0,1], so this is a directional derivative in the feasible direction.
c = 0*(1-1) + (1+ε)*(1-0) + 0*(1-(1+ε)) = 0 + 1+ε + 0 = 1+ε.

Now, the subgradient of C₅(h) = max_m c[m]*dx is the gradient of the active c[m*]*dx (where m* = argmax). If multiple m are tied at the max, the subgradient is the average of their gradients (or any convex combination).

For the projected gradient method:
1. Compute g = gradient of C₅ at h
2. h_new = h - α * g (gradient step, with step size α)
3. Project h_new onto {h: 0≤h≤1, ∑h=n/2}

The projection onto {0≤h≤1} is just clipping. The projection onto ∑h=n/2 after clipping requires a water-filling algorithm.

def project_feasible(h, n):
    """Project h onto {h: 0≤h≤1, ∑h=n/2}."""
    h = np.clip(h, 0, 1)
    # Adjust sum to n/2
    target = n / 2.0
    current = np.sum(h)
    diff = target - current
    if abs(diff) < 1e-10:
        return h
    # Water-filling: distribute diff to elements not at bounds
    if diff > 0:
        # Need to increase sum: add to elements < 1
        mask = h < 1.0
        count = np.sum(mask)
        if count == 0:
            return h  # Can't increase
        add = min(diff / count, 1.0 - h[mask]).max() if count > 0 else 0
        # Actually, proper water filling:
        # Add the same amount to all non-saturated elements
        step = diff / count
        step = min(step, (1.0 - h[mask]).min()) if count > 0 else 0
        h[mask] += step
        # If still not at target (some became saturated), iterate
        # For simplicity, just do one pass
    else:
        mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            return h
        step = diff / count
        step = max(step, h[mask].min()) if count > 0 else 0
        h[mask] += step
    return np.clip(h, 0, 1)

import numpy as np
import time

def compute_c5(h):
    """Compute C5 = max_k dx * correlate(h, 1-h, full)[k]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def local_search_binary(h, max_time=60.0, seed=42):
    """Local search for binary h using swap moves (best improvement + restarts)."""
    rng = np.random.RandomState(seed)
    start = time.time()
    h = h.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    no_improve = 0
    
    while time.time() - start < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_i, best_j = -1, -1
        best_val = current_c5
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                val = compute_c5(h)
                h[i] = 1
                h[j] = 0
                if val < best_val:
                    best_val = val
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            current_c5 = best_val
            if current_c5 < best_c5 - 1e-15:
                best_c5 = current_c5
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 5:
                # Perturb: random 1-3 swaps
                n_pert = rng.randint(1, 4)
                for _ in range(n_pert):

                        j = zeros[rng.randint(len(zeros))]
                        h[i] = 0
                        h[j] = 1
                current_c5 = compute_c5(h)
                no_improve = 0
    
    return best_h, best_c5

    """Project h onto {h: 0<=h<=1, sum(h)=n/2}."""

def interpolate_to_resolution(h, n_new):
    """Interpolate h to a new resolution."""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)  # Non-periodic
    return np.clip(h_new, 0.0, 1.0)

def make_binary_feasible(h, n):
    """Convert h to a binary feasible solution."""
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2

        # Remove the ones with lowest "importance" (closest to 0.5 in original h)

        for idx in order[:n_ones - target]:
            h_bin[ones_idx[idx]] = 0
    elif n_ones < target:
