Since C₅ = max_k f_k, and each f_k is bilinear in (h[i], h[j]), the new f_k as a function of δ is at most quadratic. The max of quadratic functions is convex in δ (actually, the max of any set of functions is convex if each function is convex, but quadratic functions that aren't convex... hmm).

Actually, f_k as a function of δ:
- f_k depends on h[i] and h[j] (and possibly on h[i+k] or h[i-k] and h[j+k] or h[j-k])
- The terms involving both h[i] and h[j] would be h[i]*h[j] or similar, which after substitution becomes (h[i]+δ)(h[j]-δ) = h[i]h[j] - h[i]δ + h[j]δ - δ², which is quadratic in δ.
- Most terms are linear in δ.

So each f_k(δ) is at most quadratic, and C₅(δ) = max_k f_k(δ) is the max of quadratic functions. Finding the minimum of this in 1D can be done by:
1. Evaluating at a grid of δ values
2. Or more cleverly, finding the breakpoints where the active set changes

For simplicity, I'll use a fine grid in δ.

from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_simplex(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    h = h.copy()
    # First clip to [0,1]
    h = np.clip(h, 0, 1)
    # Then adjust to meet the sum constraint
    diff = target_sum - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase some values
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        # Need to decrease some values
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)

def pair_swap_optimize(h, i, j, dx, n):
    """Find optimal δ for swapping between h[i] and h[j]."""
    # h[i] += δ, h[j] -= δ
    # Valid range: max(-h[i], h[j]-1) <= δ <= min(1-h[i], h[j])
    delta_min = max(-h[i], h[j] - 1)
    delta_max = min(1 - h[i], h[j])
    
    if delta_min >= delta_max:
        return h, 0.0
    
    # Sample δ values
    num_samples = 50
    deltas = np.linspace(delta_min, delta_max, num_samples)
    
    best_c5 = compute_c5(h)
    best_delta = 0.0
    
    h[i] += 0  # no-op, just to make sure we work on a copy
    h_orig_i = h[i]
    h_orig_j = h[j]
    
    for d in deltas:
        h[i] = h_orig_i + d
        h[j] = h_orig_j - d
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_delta = d
    
    h[i] = h_orig_i + best_delta
    h[j] = h_orig_j - best_delta
    return h, best_delta

The soft-max (LogSumExp) approximation:
f_t(h) = t * log(∑_k exp(f_k(h)/t))

As t → 0, this approaches max_k f_k(h). For small t (like 0.01), it's a good smooth approximation.

The gradient of f_t with respect to h[i] is:
∂f_t/∂h[i] = ∑_k (exp(f_k/t) / ∑_j exp(f_j/t)) * ∂f_k/∂h[i]

This is a weighted average of the gradients of the individual f_k's.

For f_k(h) = ∑_m h[m] * (1-h[m+k]) (for the appropriate range):
∂f_k/∂h[i] = (1-h[i-k]) * [i-k is valid] + (-h[i+k]) * [i+k is valid]

Wait, I need to be more careful. Let me define:
f_k = ∑_{m} h[m] * (1-h[m+k]) where the sum is over valid m (0 ≤ m ≤ n-1, 0 ≤ m+k ≤ n-1).

∂f_k/∂h[i]:
- Terms where h[i] appears as h[m]: m = i, valid if 0 ≤ i+k ≤ n-1
  Contribution: (1-h[i+k])
- Terms where h[i] appears as h[m+k]: m+k = i, so m = i-k, valid if 0 ≤ i-k ≤ n-1
  Contribution: -h[i-k]

So ∂f_k/∂h[i] = (1-h[i+k]) * 1[0≤i+k≤n-1] - h[i-k] * 1[0≤i-k≤n-1]

This is easy to compute!

def lse_c5(h, t=0.005):
    """LogSumExp approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    log_sum = max_corr + t * np.log(np.sum(np.exp((corr - max_corr) / t)))
    return log_sum * dx

def lse_c5_grad(h, t=0.005):
    """Gradient of LogSumExp approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    weights = np.exp((corr - max_corr) / t)
    weights /= np.sum(weights)
    
    # Gradient of corr[k] with respect to h[i]
    # corr has length 2n-1, indexed 0 to 2n-2
    # corr[n-1+l] = sum_m h[m]*(1-h[m+l]) for valid m
    # d(corr[n-1+l])/dh[i] = (1-h[i+l])*1[0≤i+l≤n-1] - h[i-l]*1[0≤i-l≤n-1]
    
    grad = np.zeros(n)
    for l in range(-(n-1), n):
        k = n - 1 + l  # index in corr array
        w = weights[k]
        # (1-h[i+l]) for valid i+l
        if l >= 0:
            # i+l ∈ [0, n-1] means i ∈ [0, n-1-l]
            # -h[i-l] for valid i-l means i-l ∈ [0,n-1] always (since i ∈ [0,n-1], l ≥ 0)
            # Actually i-l ∈ [0,n-1] means i ∈ [l, n-1]
            # Combining: i ∈ [0, n-1-l] for first term, i ∈ [l, n-1] for second
            # First term: for i in [0, n-1-l], grad[i] += w * (1-h[i+l])
            grad[:n-l] += w * (1 - h[l:l+n-l])  # h[i+l] for i in [0, n-1-l]
            # Second term: for i in [l, n-1], grad[i] += w * (-h[i-l])
            grad[l:] -= w * h[:n-l]  # h[i-l] for i in [l, n-1]
        else:
            s = -l  # s > 0
            # i+l = i-s ∈ [0, n-1] means i ∈ [s, n-1]
            # i-l = i+s ∈ [0, n-1] means i ∈ [0, n-1-s]
            # First term: for i in [s, n-1], grad[i] += w * (1-h[i-s])
            grad[s:] += w * (1 - h[:n-s])  # h[i-s] for i in [s, n-1]
            # Second term: for i in [0, n-1-s], grad[i] += w * (-h[i+s])
            grad[:n-s] -= w * h[s:s+n-s]  # h[i+s] for i in [0, n-1-s]
    
    return grad * dx

Then ∫₀^{2-k} h(x)(1-h(x+k)) dx = A(k) - C(k)
And ∫₀^{2-k} h(x+k)(1-h(x)) dx = B(k) - C(k)

So C₅ = max(½, max_{k>0} (A(k) - C(k)), max_{k>0} (B(k) - C(k)))

Note that A(k) + B(k) = ∫₀^{2-k} h(x) dx + ∫_k^2 h(x) dx = ∫₀^k h(x) dx + ∫_k^2 h(x) dx = ∫₀² h(x) dx = 1.

So A(k) + B(k) = 1, which means one of them is ≥ 1/2.

Therefore, max(A(k)-C(k), B(k)-C(k)) ≥ max(A(k), B(k)) - C(k) ≥ 1/2 - C(k).

Also, A(k) - C(k) ≥ A(k) - A(k) = 0 (trivially, since C(k) ≤ min(A(k), B(k)) by Cauchy-Schwarz or just by h² ≤ h).

The maximum possible value of A(k) - C(k) is A(k) (when C(k) = 0, i.e., h and its k-shift have disjoint support in the overlap region).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def lse_objective(h, t=0.003):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    val = m + t * np.log(np.exp(corr - m).sum())
    return val * dx

def lse_gradient(h, t=0.003):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp((corr - m) / t)
    w /= w.sum()
    
    grad = np.zeros(n)
    # For each lag l (offset from center), compute contribution
    # corr[n-1+l] = sum_{m: 0<=m<=n-1, 0<=m+l<=n-1} h[m]*(1-h[m+l])
    # d/dh[i] of corr[n-1+l]:
    #   Term 1: i appears as m, needs 0<=i+l<=n-1: contributes (1-h[i+l])
    #   Term 2: i appears as m+l, so m=i-l, needs 0<=i-l<=n-1: contributes -h[i-l]
    
    for l in range(-(n-1), n):
        idx = n - 1 + l
        wl = w[idx]
        if wl < 1e-15:
            continue
        if l >= 0:
            # Term 1: i in [0, n-1-l], h[i+l]
            grad[:n-l] += wl * (1.0 - h[l:l+n-l])
            # Term 2: i in [l, n-1], h[i-l]
            grad[l:] -= wl * h[:n-l]
        else:
            s = -l
            # Term 1: i in [s, n-1], h[i-s]
            grad[s:] += wl * (1.0 - h[:n-s])
            # Term 2: i in [0, n-1-s], h[i+s]
            grad[:n-s] -= wl * h[s:s+n-s]
    
    return grad * dx

def pair_swap(h, i, j, n_deltas=30):
    """Optimize h[i] += d, h[j] -= d."""
    hi, hj = h[i], h[j]
    d_lo = max(-hi, hj - 1.0)
    d_hi = min(1.0 - hi, hj)
    
    if d_hi - d_lo < 1e-14:
        return h, hi, hj
    
    deltas = np.linspace(d_lo, d_hi, n_deltas)
    best_obj = np.inf
    best_d = 0.0
    
    for d in deltas:
        h[i] = hi + d
        h[j] = hj - d
        obj = np.max(np.correlate(h, 1.0 - h, mode='full'))
        if obj < best_obj:
            best_obj = obj
            best_d = d
    
    h[i] = hi + best_d
    h[j] = hj - best_d
    return h, hi + best_d, hj - best_d

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    def elapsed():
        return time.time() - start_time
    
    def time_left():
        return budget_s - elapsed()
    
    # Start from parent
    h = initial_h_values.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    # Ensure valid
    h = np.clip(h, 0, 1)
    # Fix sum
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute evenly among non-saturated
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0, 1)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_n = n
    
    # Phase 1: Gradient-based optimization (LogSumExp)
    if time_left() > 300:
        t_param = 0.005
        constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                         'jac': lambda x: np.ones(n)}]
        bounds = [(0, 1)] * n
        
        res = minimize(
            lse_objective, h,
            jac=lse_gradient,
            args=(t_param,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-12}
        )
        
        if res.success or res.fun < best_c5:
            h_res = np.clip(res.x, 0, 1)
            # Fix sum
            diff = target - h_res.sum()
            if abs(diff) > 1e-8:
                if diff > 0:
                    mask = h_res < 1.0
                else:
                    mask = h_res > 0.0
                if mask.sum() > 0:
                    h_res[mask] += diff / mask.sum()
                    h_res = np.clip(h_res, 0, 1)
            
            c5_res = compute_c5(h_res)
