```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    """Compute C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_c5_all(h):
    """Compute all overlap values for given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def binary_local_search(h, rng, time_limit):
    """Local search on binary h. Returns (h, c5)."""
    n = len(h)
    best_c5 = compute_c5(h)
    start = time.time()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    no_improve = 0
    best_no_improve_reset = 10000
    kick_count = 0
    
    while time.time() - start < time_limit:
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        
        new_c5 = compute_c5(h)
        
        if new_c5 <= best_c5 + 1e-15:
            best_c5 = new_c5
            no_improve = 0
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
        else:
            h[i] = 1
            h[j] = 0
            no_improve += 1
        
        # If stuck, do a kick
        if no_improve > best_no_improve_reset + kick_count * 5000:
            # Accept a slightly worse solution to escape local optimum
            for _ in range(min(5, no_improve // 1000)):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                if new_c5 < best_c5 + 0.02:
                    best_c5 = new_c5
                    ones.remove(i)
                    zeros.remove(j)
                    ones.append(j)
                    zeros.append(i)
                    break
            no_improve = 0
            kick_count += 1
            if kick_count > 20:

def continuous_refine(h, rng, time_limit):
    """Refine h using projected gradient descent with log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    beta = 20.0  # Smoothing parameter
    lr = 0.01  # Learning rate
    
    while time.time() - start < time_limit:
        # Compute all overlap values
        c_vals = compute_c5_all(h)
        c_max = np.max(c_vals)
        
        # Log-sum-exp weights
        shifted = beta * (c_vals - c_max)
        shifted = np.clip(shifted, -50, 0)
        weights = np.exp(shifted)
        w_sum = np.sum(weights)
        if w_sum < 1e-30:
            break
        weights /= w_sum
        
        # Compute gradient (weighted average of individual gradients)
        grad = np.zeros(n)
        for k in range(len(c_vals)):
            if weights[k] < 1e-10:
                continue
            s = k - (n - 1)
            w = weights[k] * dx
            if s == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                i_min = max(0, -s)
                i_max = min(n - 1, n - 1 - s)
                if i_min <= i_max:
                    # d/dh[i] h[i]*(1-h[i+s]) = (1-h[i+s])
                    grad[i_min:i_max+1] += w * (1.0 - h[i_min+s:i_max+s+1])
                    # d/dh[i+s] h[i]*(1-h[i+s]) = -h[i]
                    grad[i_min+s:i_max+s+1] += w * (-h[i_min:i_max+1])
        
        # Project gradient descent step
        h_new = h - lr * grad
        # Project onto {0 <= h <= 1, sum(h) = n/2}
        h_new = project_to_polytope(h_new, n / 2.0)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            h = h_new
            # Decrease learning rate slightly
            lr *= 0.999
        else:
            lr *= 0.9
            if lr < 1e-6:
                break
        
        # Increase beta
        beta = min(beta * 1.01, 200.0)
    
    return h, best_c5

def project_to_polytope(h, target_sum):
    """Project h onto {x: 0 <= x <= 1, sum(x) = target_sum}."""
    h = h.copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            return np.clip(h, 0, 1)
        
        # Bisection on the threshold
        lo, hi = -5.0, 5.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            projected = np.clip(h - mid, 0.0, 1.0)
            s = np.sum(projected)
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    
    return np.clip(h, 0, 1)
```

```python
        h_new = h - lr * grad
        h_new = project_to_polytope(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-14:
            best_c5 = new_c5
            h = h_new
            stall = 0
        else:
            h = h_new
            lr *= 0.85
            stall += 1
            if stall > 30 or lr < 1e-10:
                break
        beta = min(beta * 1.01, 150.0)
    return h, best_c5

def make_binary(n, rng, mode=0, parent=None):
    target = n // 2
    if mode == 0 and parent is not None:
        x_old = np.linspace(0, 2, len(parent))
        x_new = np.linspace(0, 2, n)
        hc = np.interp(x_new, x_old, parent)
        h = (hc >= 0.5).astype(float)
    elif mode == 1:
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    elif mode == 2:
        h = np.zeros(n)
        state = (1 << 8) - 1
        for i in range(n):
            h[i] = float(state & 1)
            bit = ((state >> 0) ^ (state >> 3) ^ (state >> 5) ^ (state >> 7)) & 1
            state = (state >> 1) | (bit << 8)
    else:
        h = np.zeros(n)
        h[:target] = 1.0
        rng.shuffle(h)
    diff = target - int(np.sum(h))
    if diff > 0:
        zeros = list(np.where(h == 0)[0])
        rng.shuffle(zeros)
        h[zeros[:diff]] = 1.0
    elif diff < 0:
        ones = list(np.where(h == 1)[0])
        rng.shuffle(ones)
        h[ones[:-diff]] = 0.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tr():
        return max(0.0, budget_s - (time.time() - start_time) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    
    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
        pc5 = compute_c5(parent)
        best_c5 = pc5
        best_h = parent.copy()
    
    # Phase 1: Binary local search across n values and modes
    n_values = [96, 128, 160, 192, 256]
    
    for n in n_values:
        if tr() < 20:
            break
        n_attempts = 8
        alloc = tr() / (len(n_values) * n_attempts + 1)
        for attempt in range(n_attempts):
            if tr() < 8:
                break
            mode = attempt % 4
            h = make_binary(n, rng, mode, parent)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            tl = max(4.0, min(20.0, tr() / 5.0))
            h, c5 = binary_local_search(h, rng, tl)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Phase 2: Continuous refinement from best binary
    if best_h is not None and tr() > 25:
        n = len(best_h)
        for _ in range(6):
            if tr() < 12:
                break
            h = best_h.astype(float).copy()
            tl = min(35.0, tr() / 3.0)
            h, c5 = continuous_refine(h, tl)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Phase 3: More local search on best (if binary)
    if best_h is not None and tr() > 20:
        h = best_h.copy()
        if np.allclose(h, np.round(h), atol=0.01):
            h = np.round(h).astype(float)
            tl = min(50.0, tr() / 2.0)
            h, c5 = binary_local_search(h, rng, tl)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Phase 4: Final continuous polish
    if best_h is not None and tr() > 15:
        h = best_h.astype(float).copy()
        tl = min(30.0, tr() / 2.0)
        h, c5 = continuous_refine(h, tl)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    if best_h is None:
        n = 96
        best_h = np.zeros(n)
        best_h[:48] = 1.0
        rng.shuffle(best_h)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```
