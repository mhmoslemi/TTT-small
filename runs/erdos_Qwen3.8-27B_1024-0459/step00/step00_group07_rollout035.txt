import numpy as np

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h_parent = initial_h_values
    n_parent = len(h_parent)
    
    # ===== Helper functions =====
    
    def project_sum(h, target):
        """Project onto {0<=h<=1, sum=target}"""
        h = h.copy()
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            return h
        # Bisection for Lagrange multiplier
        lo, hi = -2.0, 2.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            h_t = np.clip(h - mid, 0.0, 1.0)
            s_t = np.sum(h_t)
            if s_t > target:
                lo = mid
            else:
                hi = mid
        t = (lo + hi) * 0.5
        result = np.clip(h - t, 0.0, 1.0)
        # Fix residual
        diff = target - np.sum(result)
        if diff > 1e-12:
            order = np.argsort(1.0 - result)
            for idx in order:
                room = 1.0 - result[idx]
                if room < 1e-15:
                    continue
                add = min(room, diff)
                result[idx] += add
                diff -= add
                if diff < 1e-12:
                    break
        elif diff < -1e-12:
            order = np.argsort(result)
            for idx in order:
                av = result[idx]
                if av < 1e-15:
                    continue
                sub = min(av, -diff)
                result[idx] -= sub
                diff += sub
                if diff > -1e-12:
                    break
        return result

        dx = 2.0 / n
        def eval_c5(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            return float(np.max(corr) * dx)

            corr = np.correlate(h, 1.0 - h, mode='full')
            return corr * dx

            add = min(room, diff)
            result[int(idx)] += add
            diff -= add
            if diff < 1e-12:
                break
    elif diff < -1e-12:
        order = np.argsort(result)
        for idx in order:
            av = result[idx]
            if av < 1e-14:
                continue
            sub = min(av, -diff)
            result[int(idx)] -= sub
            diff += sub
            if diff > -1e-12:
                break
    return result

def eval_c5(h):
    """Compute C5 value"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def eval_c5_all(h):
    """Compute all correlation values times dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_subgradient(h, active_indices):
    """Compute subgradient of max_k f_k at h"""
    n = len(h)
    grad = np.zeros(n)
    for k in active_indices:
        m = int(k) - (n - 1)
        if m >= 0:
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]
        else:
            am = -m
            grad[am:] += (1.0 - h[:n-am])
            grad[:n-am] -= h[am:]
    grad /= max(len(active_indices), 1)
    return grad

def optimize_subgradient(h, target, max_time, start_time, rng, initial_alpha=0.1):
    """Run subgradient optimization"""
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(best_h)
    alpha = initial_alpha
    stall = 0
    last_improve = start_time
    
    while time.time() < max_time - 3:
        all_f = eval_c5_all(best_h)
        c5_max = float(np.max(all_f))
        
        # Active set
        threshold = c5_max - 0.003 * c5_max
        active = np.where(all_f >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(all_f))])
        if len(active) > 30:
            active = active[np.argsort(all_f[active])[::-1][:30]]
        
        g = compute_subgradient(best_h, active)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        
        h_new = project_sum(best_h - alpha * g, target)
        new_c5 = eval_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            stall = 0
            last_improve = time.time()
            alpha = min(alpha * 1.02, 0.5)
        else:
            alpha = max(alpha * 0.8, 0.0001)
            stall += 1
        
        # Restart
        if stall > 2000 or time.time() - last_improve > 20:
            h_p = best_h + rng.uniform(-0.015, 0.015, n)
