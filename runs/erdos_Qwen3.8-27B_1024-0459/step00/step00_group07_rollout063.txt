def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    """Upsample h from len(h) points to new_n points on [0,2]"""
    old_n = len(h)
    x_old = np.arange(old_n) / old_n * 2.0  # positions in [0, 2)
    x_new = np.arange(new_n) / new_n * 2.0
    # Linear interpolation (no wrap-around since it's not periodic)
    # But we need to handle the edges
    h_ext = np.concatenate([[h[0]], h, [h[-1]]])
    x_ext = np.concatenate([[-2.0/old_n], x_old, [2.0 + 2.0/old_n]])
    h_new = np.interp(x_new, x_ext, h_ext)
    h_new = project(h_new, new_n)
    return h_new

    """Gradient of (1/tau)*log(sum(exp(tau*c_j))) where c_j are correlation values"""
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    c_max = c.max()

            grad[sp:] += w[j] * (-h[:n-sp])

def optimize_slsqp(h0, n, time_limit=300.0, t_start=0):
    """SLSQP optimization for final polish"""
    tau = 100.0
    
    def obj(h):
        return smooth_max_fun(h, tau)
    
    def obj_grad(h):
        return gradient_smooth_max(h, tau)
    
    constraints = [{'type': 'eq', 'fun': lambda h: h.sum() - n/2.0,
                    'jac': lambda h: np.ones(n)}]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))


def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).astype(np.float64).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return project(h.copy(), new_n)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    x_ext = np.concatenate([[-1.0 / old_n], x_old, [2.0 + 1.0 / old_n]])
    h_ext = np.concatenate([[h[0]], h, [h[-1]]])
    h_new = np.interp(x_new, x_ext, h_ext)
    return project(h_new, new_n)


def gradient_smooth_max(h, tau):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    c = c_raw * dx
    c_max = c.max()
    logits = tau * (c - c_max)
    exp_l = np.exp(logits)
    w = exp_l / exp_l.sum()
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        s = j - (n - 1)
        if s == 0:
            grad += w[j] * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w[j] * (1.0 - h[:n - s])
            grad[:n - s] += w[j] * (-h[s:])
        else:
            sp = -s
            grad[:n - sp] += w[j] * (1.0 - h[sp:])
            grad[sp:] += w[j] * (-h[:n - sp])
    grad *= dx
    return grad, c_max


def smooth_max_fun(h, tau):

    dx = 2.0 / n

    c_max = c.max()
    val = c_max + np.log(np.sum(np.exp(tau * (c - c_max)))) / tau
    return val
