    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full")
    max_val = np.max(conv)

    # Subgradient of g_k(h) = sum_i h[i] * (1-h[i+k_shift])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full")
    return float(np.max(conv) * dx)

def project_onto_constraint(h, n):
    target = n / 2.0
    def sum_clipped(t):
        return np.sum(np.clip(h - t, 0, 1))
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if sum_clipped(mid) > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    proj = np.clip(h - t, 0, 1)
    # Final correction for numerical issues
    s = proj.sum()
    if abs(s - target) > 1e-10:
        # Small adjustment
        diff = target - s
        mask = (proj > 1e-8) & (proj < 1 - 1e-8)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            proj[mask] += adj
            proj = np.clip(proj, 0, 1)
    return proj

def smooth_max_c5(h, n, T=30.0):
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full") * dx
    max_c = conv.max()
    shifted = (conv - max_c) / T
    return max_c + T * np.log(np.mean(np.exp(shifted)))

def smooth_max_grad(h, n, T=30.0):
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full") * dx
    max_c = conv.max()
    shifted = (conv - max_c) / T
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        shift = k - (n - 1)
        # g_k = sum_i h[i] * (1-h[i+shift]) for valid i
        # d(g_k)/d(h[j]) = (1 - h[j+shift]) - h[j-shift] (with boundary)
        i_min = max(0, -shift)
        i_max = min(n, n - shift)
        for i in range(i_min, i_max):
            grad[i] += wk * (1.0 - h[i + shift])
            js = i - shift
            if 0 <= js < n:
                grad[i] -= wk * h[js]
    return grad

def projected_gradient_step(h, n, lr, T):
    """One step of projected gradient descent on smooth max C5"""
    grad = smooth_max_grad(h, n, T)
    h_new = h - lr * grad
    h_new = project_onto_constraint(h_new, n)
    return h_new

def run_subgradient(h0, n, budget_s, T_init=30.0, lr_init=0.05):
    """Run subgradient optimization"""
    h = h0.copy().astype(float)
    h = project_onto_constraint(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t_start = time.time()
    
    lr = lr_init
    T = T_init
    stalled = 0
    
    for iteration in range(100000):

        h_new = projected_gradient_step(h, n, lr, T)
        c5_new = compute_c5(h_new, n)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stalled = 0
            # Slightly increase lr on improvement
            lr = min(lr * 1.02, 0.5)
        else:
            h = h_new  # Still take the step
            stalled += 1
            if stalled > 50:

                T *= 1.05
                stalled = 0

        if iteration % 100 == 0 and iteration > 0:

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full")
    return float(np.max(conv) * dx)

def project_onto_constraint(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Bisection for water level
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    proj = np.clip(h - t, 0.0, 1.0)
    # Numerical fix
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def get_subgradient(h, n):
    """Get (sub)gradient of C5 and the max value using smooth max"""
    T = 20.0  # Temperature
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full") * dx
    max_c = conv.max()
    shifted = (conv - max_c) / T
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j1_lo = max(0, -s)
        j1_hi = min(n, n - s)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += wk * (1.0 - h[j1_lo + s:j1_hi + s])
        j2_lo = max(0, s)
        j2_hi = min(n, n + s)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] -= wk * h[j2_lo - s:j2_hi - s]
    
    return grad, max_c

def adam_projected_step(h, n, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam step with projection"""
    grad, _ = get_subgradient(h, n)
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = project_onto_constraint(h_new, n)
    return h_new, m, v

def optimize_smooth(h0, n, budget_s, T0=20.0):
    """Adam optimization with projected steps"""
    h = h0.copy().astype(float)
    h = project_onto_constraint(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 1
    lr = 0.02
    
    t_start = time.time()
    last_improve = 0
    patience = 0
    
    while time.time() - t_start < budget_s:
        h_new, m, v = adam_projected_step(h, n, m, v, t, lr=lr)
        t += 1
        c5_new = compute_c5(h_new, n)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improve = t
            patience = 0
        else:
            patience += 1
        
        h = h_new
        
        # Adaptive learning rate
        if patience > 200:
            lr *= 0.9
            patience = 0
        elif patience > 50:
            lr *= 0.95
        
        if t % 500 == 0:
            lr = max(lr * 0.999, 0.001)
    
    return best_h, best_c5

def interpolate_h(h_src, n_src, n_dst):
    """Linearly interpolate h from resolution n_src to n_dst"""
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return h_dst

            h_opt, c5_opt = optimize_adam(h_init, n_try, t_use)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_try
    # Phase 3: Final polish
    remaining = budget_s - (time.time() - t0) - 5.0
    if remaining > 20:
        h_pol, c5_pol = optimize_adam(best_h, best_n, remaining, T0=10.0, T_end=2.0, lr_init=0.02)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
