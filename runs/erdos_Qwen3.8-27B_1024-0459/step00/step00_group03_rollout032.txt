import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_constraint(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target_sum) < 1e-14:
        return h.copy()
    if s > target_sum:
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            p = np.clip(h - mid, 0.0, 1.0)
            if float(np.sum(p)) > target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    else:
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            p = np.clip(h + mid, 0.0, 1.0)
            if float(np.sum(p)) < target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


class C5Objective:
    def __init__(self, n, target_sum):
        self.n = n
        self.dx = 2.0 / n
        self.target_sum = target_sum

    def __call__(self, x):
        h = project_constraint(x, self.target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * self.dx)


class C5ObjSmooth:
    def __init__(self, n, target_sum, temperature):
        self.n = n
        self.dx = 2.0 / n
        self.target_sum = target_sum
        self.temperature = temperature

    def __call__(self, x):
        h = project_constraint(x, self.target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full') * self.dx
        shifted = corr - np.max(corr)
        logsumexp = np.max(corr) + (1.0 / self.temperature) * np.log(np.sum(np.exp(self.temperature * shifted)))
        return float(logsumexp * self.dx / self.dx)


def run_lbbfgs(h0, n, maxiter, maxfun, ftol, gtol):
    target_sum = n / 2.0
    h0 = project_constraint(h0, target_sum)
    obj = C5Objective(n, target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol,
                                'maxfun': maxfun, 'disp': False})
        h_opt = project_constraint(res.x, target_sum)
        c5 = compute_c5(h_opt)
        return h_opt, c5
    except Exception:
        return None, np.inf


def make_binary_init(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h


def make_sturmian_init(n, rng):
    m = n // 2
    h = np.zeros(n)
    step = (n - 1) / m if m > 0 else 1.0
    positions = np.round(np.arange(m) * step).astype(int)
    positions = np.clip(positions, 0, n - 1)
    positions = np.unique(positions)
    if len(positions) < m:
        remaining = n - len(positions)
        extra = rng.choice(remaining, m - len(positions), replace=False)
        positions = np.concatenate([positions, extra + 1])
        positions = np.clip(positions, 0, n - 1)
    h[positions[:m]] = 1.0
    if np.sum(h) != m:
        h = project_constraint(h, m)
    return h


def make_block_init(n, rng, n_blocks):
    h = np.zeros(n)
    n_ones = n // 2
    block_len = n_ones // n_blocks
    offsets = np.sort(rng.choice(n, n_blocks, replace=False))
    for i, off in enumerate(offsets):
        s = off
        e = min(s + block_len, n)
        h[s:e] = 1.0
    if np.sum(h) < n_ones:
        remaining = n_ones - int(np.sum(h))
        zeros = np.where(h == 0)[0]
        extra = rng.choice(zeros, min(remaining, len(zeros)), replace=False)
        h[extra] = 1.0
    h = project_constraint(h, n / 2.0)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = np.inf
    best_n = 70

    def tl():
        return budget_s - 8 - (time.time() - t0)

    # Phase 1: Optimize parent if available
    if 'initial_h_values' in dir() and initial_h_values is not None:
        ihv = np.asarray(initial_h_values, dtype=float)
        n_par = len(ihv)
        if tl() > 30:
            h_opt, c5 = run_lbbfgs(ihv.copy(), n_par, 4000, 12000, 1e-12, 1e-8)
            if h_opt is not None and c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, n_par
        # Interpolate to a couple of nearby resolutions
        for n_try in [n_par + 10, n_par + 30]:
            if tl() < 20:
                break
            x_old = np.linspace(0, 2, n_par, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, ihv)
            h_interp = project_constraint(h_interp, n_try / 2.0)
            h_opt, c5 = run_lbbfgs(h_interp, n_try, 2500, 8000, 1e-12, 1e-8)
            if h_opt is not None and c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, n_try

    # Phase 2: Binary and structured initializations
    for n in [60, 70, 80, 90, 100, 120]:
        if tl() < 25:
            break
        for attempt in range(4):
            if tl() < 8:
                break
            h_bin = make_binary_init(n, rng)
            h_opt, c5 = run_lbbfgs(h_bin, n, 2000, 6000, 1e-12, 1e-8)
            if h_opt is not None and c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, n

    # Phase 3: Sturmian / block structures
    for n in [80, 100, 120, 140]:
        if tl() < 20:
            break
        for nb in [2, 3, 4, 5]:
            if tl() < 8:
                break
            h_blk = make_block_init(n, rng, nb)
            h_opt, c5 = run_lbbfgs(h_blk, n, 2000, 6000, 1e-12, 1e-8)
            if h_opt is not None and c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, n
        h_st = make_sturmian_init(n, rng)
        h_opt, c5 = run_lbbfgs(h_st, n, 2000, 6000, 1e-12, 1e-8)
        if h_opt is not None and c5 < best_c5:
            best_c5, best_h, best_n = c5, h_opt, n

    # Phase 4: Random continuous starts
    for n in [70, 100, 140]:
        if tl() < 20:
            break
        for attempt in range(4):
            if tl() < 8:
                break
            h_rand = rng.uniform(0, 1, n)
            h_rand = project_constraint(h_rand, n / 2.0)
            h_opt, c5 = run_lbbfgs(h_rand, n, 2000, 6000, 1e-12, 1e-8)
            if h_opt is not None and c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, n

    # Phase 5: Local perturbation / hill-climbing on best
    if best_h is not None and tl() > 15:
        h_cur = best_h.copy()
        n_cur = len(h_cur)
        for it in range(50):
            if tl() < 8:
                break
            sigma = 0.02 + 0.03 * (1.0 - it / 50.0)
            h_pert = h_cur + rng.normal(0, sigma, n_cur)
            h_opt, c5 = run_lbbfgs(h_pert, n_cur, 800, 2500, 1e-12, 1e-7)
            if h_opt is not None and c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, n_cur
                h_cur = best_h.copy()

    # Phase 6: Try smooth-max refinement on the best
    if best_h is not None and tl() > 30:
        n_cur = len(best_h)
        h_smooth = best_h.copy()
        for temp in [50.0, 20.0, 10.0]:
            if tl() < 15:
                break
            target_sum = n_cur / 2.0
            obj_sm = C5ObjSmooth(n_cur, target_sum, temp)
            bounds = [(0.0, 1.0)] * n_cur
            try:
                res = minimize(obj_sm, h_smooth, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 1000, 'ftol': 1e-12, 'gtol': 1e-8, 'maxfun': 3000})
                h_sm_opt = project_constraint(res.x, target_sum)
                c5_sm = compute_c5(h_sm_opt)
                if c5_sm < best_c5:
                    best_c5, best_h, best_n = c5_sm, h_sm_opt, n_cur
                    h_smooth = h_sm_opt.copy()
            except Exception:
                pass

    # Phase 7: Higher resolution refinement
    if best_h is not None and tl() > 40:
        n_high = min(len(best_h) * 2, 280)
        if n_high > len(best_h):
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_high, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_constraint(h_interp, n_high / 2.0)
            h_opt, c5 = run_lbbfgs(h_interp, n_high, 2000, 6000, 1e-12, 1e-8)
            if h_opt is not None and c5 < best_c5:
                best_c5, best_h, best_n = c5, h_opt, n_high

    # Phase 8: Final local polish
    if best_h is not None and tl() > 10:
        h_opt, c5 = run_lbbfgs(best_h.copy(), len(best_h), 2000, 6000, 1e-14, 1e-10)
        if h_opt is not None and c5 < best_c5:

    # Fallback

        n = 70
        best_h = np.ones(n) / 2.0
        best_c5 = compute_c5(best_h)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_constraint(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
