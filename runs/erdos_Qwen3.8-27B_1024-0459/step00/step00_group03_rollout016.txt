    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")

def project_simplex(v, s):
    v = np.clip(v.astype(float), 0.0, 1.0)
    for _ in range(500):
        d = s - v.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            m = v < 1.0
        else:
            m = v > 0.0
        if not m.any():
            break
        v[m] += d / m.sum()
        v = np.clip(v, 0.0, 1.0)
    # Final fix
    d = s - v.sum()
    if abs(d) > 1e-10:
        m = (v > 1e-8) & (v < 1.0 - 1e-8)
        if m.any():
            v[m] -= d / m.sum()
            v = np.clip(v, 0, 1)
    return v

def compute_grads(h):
    """Compute ∂corr_k/∂h[m] for k=0,...,n-1, m=0,...,n-1."""
    n = len(h)
    grads = np.zeros((n, n))
    for k in range(n):
        for m in range(n):
            val = 0.0
            if k <= m:  # m-k >= 0
                i = m - k
                if i <= n - 1 - k:  # valid i
                    val += (1.0 - h[i])
            if m <= n - 1 - k:  # m+k < n
                val -= h[m + k]
            grads[k, m] = val
    return grads
