import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def lse_objective(h, dx, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = corr * dx / t
    m = s.max()
    return t * (m + np.log(np.sum(np.exp(s - m))))

def lse_gradient(h, dx, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = corr * dx / t
    m = s.max()
    e = np.exp(s - m)
    p = e / e.sum()
    
    one_minus_h = 1.0 - h
    conv1 = np.convolve(one_minus_h, p, mode='full')
    corr_p_h = np.correlate(p, h, mode='full')
    
    grad = dx * (conv1[n-1:2*n-1] - corr_p_h[n-1::-1])
    return grad

def lse_obj_grad(h, dx, t):
    return lse_objective(h, dx, t), lse_gradient(h, dx, t)

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint and bounds approximately."""

        try:
            res = minimize(
                _hard_obj,
                best_local,
                args=(dx, target_sum),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-14}
            )
            if np.all(res.x >= -1e-10) and np.all(res.x <= 1 + 1e-10):
                cand = np.clip(res.x, 0.0, 1.0)
                cand += (target_sum - np.sum(cand)) / n
                cand = np.clip(cand, 0.0, 1.0)
                cand_c5 = _compute_c5(cand, dx)
                if cand_c5 < best_c5_local:
                    best_c5_local = cand_c5
                    best_local = cand.copy()
        except Exception:
            pass

    return best_local, best_c5_local


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Use initial_h_values if available
    try:
        global initial_h_values
        if initial_h_values is not None:
            h_init = np.array(initial_h_values, dtype=float)
            n_init = len(h_init)
            dx_init = 2.0 / n_init
            c5_init = _compute_c5(h_init, dx_init)
            best_h = h_init.copy()
            best_c5 = c5_init
            best_n = n_init
    except Exception:
        pass

    # N values to try
    n_list = []
    if best_n > 0:
        n_list.append(best_n)
    for n in [50, 60, 80, 100, 120, 150, 200, 250, 300, 40, 30, 47, 55]:
        if n not in n_list:
            n_list.append(n)

    for n in n_list:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 30
        if remaining < 30:
            break

        dx = 2.0 / n
        target_sum = n / 2.0

        # Generate starting points
        starts = []
        if best_h is not None:
            starts.append(_upsample(best_h, best_n, n, rng))
        starts.append(_make_initial_structured(n, rng, best_h))

        # Add a random perturbation
        if best_h is not None and len(best_h) == n:
            pert = best_h.copy() + rng.randn(n) * 0.08
            pert = np.clip(pert, 0.0, 1.0)
            pert += (target_sum - np.sum(pert)) / n
            pert = np.clip(pert, 0.0, 1.0)
            starts.append(pert)

        for h0 in starts:
            if time.time() - start_time > budget_s - 30:
                break
            h_opt, c5_opt = _optimize_at_n(h0, n, dx, target_sum, budget_s - 20 - start_time, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

    # Final intensive refinement at the best n
    if best_h is not None and time.time() - start_time < budget_s - 40:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': _constraint_eq, 'args': (target_sum,)}]

        # Very low temperature L-BFGS-B
        for temp in [0.003, 0.001, 0.0005, 0.0002]:
            if time.time() - start_time > budget_s - 15:
                break
            try:
                res = minimize(
                    _lse_func,
                    best_h,
                    args=(dx, temp),
                    method='L-BFGS-B',
                    jac=_lse_grad_func,
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 5000, 'ftol': 1e-16, 'gtol': 1e-11}
                )
                cand_c5 = _compute_c5(res.x, dx)
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = res.x.copy()
            except Exception:
                pass

        # Multiple SLSQP restarts
        for trial in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            try:
                h_start = best_h.copy()
                if trial > 0:
                    h_start = h_start + rng.randn(n) * 0.01
                    h_start = np.clip(h_start, 0.0, 1.0)
                    h_start += (target_sum - np.sum(h_start)) / n
                    h_start = np.clip(h_start, 0.0, 1.0)
                res = minimize(
                    _hard_obj,
                    h_start,
                    args=(dx, target_sum),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 3000, 'ftol': 1e-15}
                )
                cand = np.clip(res.x, 0.0, 1.0)
                cand += (target_sum - np.sum(cand)) / n
                cand = np.clip(cand, 0.0, 1.0)
                cand_c5 = _compute_c5(cand, dx)
                if cand_c5 < best_c5:
                    best_c5 = cand_c5
                    best_h = cand.copy()
            except Exception:
                pass

    # Ensure constraint is satisfied
    if best_h is not None:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h += (target_sum - np.sum(best_h)) / n
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = _compute_c5(best_h, dx)

    return best_h, best_c5, best_n
