def make_obj_grad_smooth(n, alpha):
    def obj(params):
        h = project_feasible(params, n)
        return smooth_objective(h, alpha)
    def grad(params):
        h = project_feasible(params, n)
        n_len = len(h)
        dx = 2.0 / n_len
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        exp_vals = np.exp(alpha * (corr - m))
        weights = exp_vals / np.sum(exp_vals)
        # d(corr_k)/d(h_i) for each k
        # corr_k = dx * sum_i h[i+k-*(n-1)]*(1-h[i]) or similar
        # Use numerical gradient for reliability
        grad_out = np.zeros(n_len)
        eps = 1e-7
        for i in range(n_len):
            h_plus = h.copy()
            h_plus[i] += eps
            corr_plus = np.correlate(h_plus, 1.0 - h_plus, mode='full') * dx
            m_plus = np.max(corr_plus)
            exp_plus = np.exp(alpha * (corr_plus - m_plus))
            val_plus = m_plus + np.log(np.sum(exp_plus)) / alpha
            h_minus = h.copy()
            h_minus[i] = max(h[i] - eps, 0.0)
            corr_minus = np.correlate(h_minus, 1.0 - h_minus, mode='full') * dx
            m_minus = np.max(corr_minus)
            exp_minus = np.exp(alpha * (corr_minus - m_minus))
            val_minus = m_minus + np.log(np.sum(exp_minus)) / alpha
            grad_out[i] = (val_plus - val_minus) / (2.0 * eps)
        return grad_out
    def obj_grad(params):
        return obj(params), grad(params)
    return obj, obj_grad

def try_nm(h_start, n, max_iter, time_limit, rng):
    """Run Nelder-Mead optimization."""
    obj = make_obj_nm(n)
    res = minimize(obj, h_start, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-10,
                            'fatol': 1e-12, 'adaptive': True})
    h_out = project_feasible(res.x, n)
    c5_out = compute_c5(h_out)
    return h_out, c5_out, res

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Start from parent
    h_par = np.array(initial_h_values, dtype=float).copy()
    n_par = len(h_par)
    h_par = project_feasible(h_par, n_par)
    c5_par = compute_c5(h_par)
    
    best_h = h_par.copy()
    best_c5 = c5_par
    best_n = n_par
    
    # Stage 1: Nelder-Mead from parent at original resolution
    if time_left() > 200:
        h_try, c5_try, _ = try_nm(h_par, n_par, 15000, 150, rng)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            best_n = n_par
    
    # Stage 2: Smoothed SLSQP refinement
    if time_left() > 150:
        for alpha in [100, 300, 800, 2000]:
            if time_left() < 50:
                break
            obj, obj_grad = make_obj_grad_smooth(best_n, alpha)
            bounds = [(0.0, 1.0)] * best_n
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - best_n / 2.0,
                           'jac': lambda x: np.ones(best_n)}]
            res = minimize(obj_grad, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 300, 'ftol': 1e-14})
            h_try = project_feasible(res.x, best_n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
    
    # Stage 3: More NM iterations from current best
    if time_left() > 120:
        h_try, c5_try, _ = try_nm(best_h, best_n, 10000, 100, rng)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            best_n = best_n
    
    # Stage 4: Try higher resolution (200)
    if time_left() > 200:
        n_hi = 200
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_hi, endpoint=False)
        h_hi = np.interp(x_new, x_old, best_h)
        h_hi = project_feasible(h_hi, n_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_h = h_hi
            best_c5 = c5_hi
            best_n = n_hi
        # Optimize at higher resolution
        if time_left() > 120:
            h_try, c5_try, _ = try_nm(h_hi, n_hi, 8000, 100, rng)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                best_n = n_hi
    
    # Stage 5: Random perturbations + local optimization
    if time_left() > 100:
        for attempt in range(15):
            if time_left() < 30:
                break
            scale = 0.03 + 0.02 * (attempt // 5)
            noise = rng.randn(best_n) * scale
            h_pert = project_feasible(best_h + noise, best_n)
            h_try, c5_try, _ = try_nm(h_pert, best_n, 5000, 30, rng)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                best_n = best_n
    
    # Stage 6: Final polish with many NM iterations
    if time_left() > 60:
        h_try, c5_try, _ = try_nm(best_h, best_n, 20000, int(time_left()) - 10, rng)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            best_n = best_n
    
    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Ensure exact feasibility
    assert np.all(best_h >= -1e-12), "h has negative values"
    assert np.all(best_h <= 1.0 + 1e-12), "h exceeds 1"
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8, "sum constraint violated"
    best_h = np.clip(best_h, 0.0, 1.0)
    
    return best_h, best_c5, best_n
