    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def compute_correlations(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    exp_vals = np.exp(t * (c - c_max))

    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Fix rounding to get exact sum
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:

            continue
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] += -w * h[:n - s]
        else:
            r = -s
            grad[r:] += w * (1.0 - h[:n - r])
            grad[:n - r] += -w * h[r:]

    return grad


def compute_grad_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    k_star = int(np.argmax(c))
    s = k_star - (n - 1)

    grad = np.zeros(n)
    if s == 0:
        grad += dx * (1.0 - 2.0 * h)
    elif s > 0:
        grad[:n - s] += dx * (1.0 - h[s:])
        grad[s:] += -dx * h[:n - s]
    else:
        r = -s
        grad[r:] += dx * (1.0 - h[:n - r])
        grad[:n - r] += -dx * h[r:]

    return grad


def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s_val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        mask = (result > 1e-6) & (result < 1.0 - 1e-6)
        n_adj = mask.sum()
        if n_adj > 0:
            indices = np.where(mask)[0]
            step = diff / n_adj
            result[indices] += step
            result = np.clip(result, 0.0, 1.0)
    return result


def upsample_linear(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new


def upsample_nearest(h, n_new):
    n_old = len(h)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        idx = int(i * n_old / n_new)
        if idx >= n_old:
            idx = n_old - 1
        h_new[i] = h[idx]
    return h_new


def optimize_pgd(h_start, t_start, t_end, max_iter, time_limit, use_subgrad_frac=0.3):
    n = len(h_start)
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    t0 = time.time()
    lr = 0.5
    patience = 0

    for it in range(max_iter):
        elapsed = time.time() - t0
        if elapsed > time_limit:
            break
        if patience > 800:
            break

        progress = min(1.0, it / (max_iter * 0.7))
        t = t_start + (t_end - t_start) * progress

        if it % 3 < int(3 * use_subgrad_frac):
            grad = compute_grad_subgrad(h)
        else:
            grad = compute_grad_softmax(h, t)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break

        lr_try = lr
        accepted = False
        for ls in range(15):
            h_new = h - lr_try * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new <= best_c5 + 1e-12:
                accepted = True
                break
            lr_try *= 0.4

        if not accepted or lr_try < 1e-10:
            lr *= 0.5
            patience += 5
            if lr < 1e-8:
                break
            continue

        h = h_new
        c5_cur = compute_c5(h)

        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h.copy()
            patience = 0
            lr = min(lr * 1.1, 3.0)
        else:
            patience += 1
            lr *= 0.85

    return best_h, best_c5


def make_slsqp_objective(t_smooth):
    def obj(h):
        corr = compute_correlations(h)
        c_max = np.max(corr)
        exp_vals = np.exp(t_smooth * (corr - c_max))
        return c_max + (1.0 / t_smooth) * np.log(np.sum(exp_vals))
    return obj


def make_slsqp_constraint(n):
    def con(h):
        return np.sum(h) - n / 2.0
    return con


def polish_slsqp(h_init, n, time_limit):
    def obj_func(h):
        corr = compute_correlations(h)
        c_max = np.max(corr)
        exp_vals = np.exp(40.0 * (corr - c_max))
        return c_max + (1.0 / 40.0) * np.log(np.sum(exp_vals))

    def con_func(h):
        return np.sum(h) - n / 2.0

    constraints = [{'type': 'eq', 'fun': con_func}]
    bounds = [(0.0, 1.0)] * n

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    try:
        result = minimize(obj_func, h_init, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-10, 'iprint': 0})
        if result.x is not None:
            h_pol = project_feasible(result.x, n)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol
    except Exception:
        pass

    return best_h, best_c5


def random_perturb(h, n, sigma, rng):
    h_pert = h + sigma * rng.standard_normal(n)
    h_pert = project_feasible(h_pert, n)
    return h_pert


def binary_local_search(h, n, max_flips=50):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    idx = np.argsort(h)

    for attempt in range(max_flips):
        improved = False
        for i in idx[:n // 2]:
            if h[i] < 0.5:
                trial = best_h.copy()
                trial[i] = 1.0
                trial = project_feasible(trial, n)
                c5_t = compute_c5(trial)
                if c5_t < best_c5 - 1e-12:
                    best_h = trial
                    best_c5 = c5_t
                    improved = True
                    break
        for i in idx[n // 2:][::-1]:
            if h[i] > 0.5:
                trial = best_h.copy()
                trial[i] = 0.0
                trial = project_feasible(trial, n)
                c5_t = compute_c5(trial)
                if c5_t < best_c5 - 1e-12:
                    best_h = trial
                    best_c5 = c5_t
                    improved = True
                    break
        if not improved:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    if h0 is None:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    best_h = project_feasible(h0.copy(), n0)
    best_c5 = compute_c5(best_h)

    elapsed = time.time() - start_time
    remain = budget_s - elapsed - 20

    if remain > 90:
        time_alloc1 = min(remain * 0.45, 400)
        h_opt, c5_opt = optimize_pgd(best_h.copy(), 10.0, 80.0, 8000, time_alloc1, 0.3)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        print(f"After PGD at n={n0}: C5={best_c5:.6f}")

    # Try perturbation restarts
    elapsed = time.time() - start_time
    remain = budget_s - elapsed - 60
    for restart in range(3):
        if remain < 60:
            break
        h_pert = random_perturb(best_h, n0, 0.15, rng)
        time_alloc_r = min(remain * 0.2, 150)
        h_opt, c5_opt = optimize_pgd(h_pert, 15.0, 100.0, 4000, time_alloc_r, 0.25)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            print(f"Restart {restart}: C5={best_c5:.6f}")
        remain = budget_s - (time.time() - start_time) - 30

    # Try higher resolutions
    for factor in [2, 3, 4]:
        n_new = n0 * factor
        elapsed = time.time() - start_time
        remain = budget_s - elapsed - 50
        if remain < 150:
            break

        h_up = upsample_linear(best_h, n_new)
        h_up = project_feasible(h_up, n_new)

        time_alloc2 = min(remain * 0.5, 350)
        h_opt, c5_opt = optimize_pgd(h_up, 10.0, 70.0, 5000, time_alloc2, 0.25)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            print(f"Resolution n={n_new}: C5={best_c5:.6f}")

    # SLSQP polish
    elapsed = time.time() - start_time
    remain = budget_s - elapsed - 15
    if remain > 40 and len(best_h) <= 120:
        time_alloc3 = min(remain * 0.6, 120)
        h_pol, c5_pol = polish_slsqp(best_h.copy(), len(best_h), time_alloc3)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
            print(f"After SLSQP: C5={best_c5:.6f}")

    # Local binary perturbation search
    elapsed = time.time() - start_time
    remain = budget_s - elapsed - 10
    if remain > 30:
        h_b, c5_b = binary_local_search(best_h, len(best_h), max_flips=30)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b

    # Final feasibility enforcement
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
