def repair_sum(h, n):
    """Adjust h to have sum = n/2 while staying in [0,1]."""
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    h = h + diff / n
    h = np.clip(h, 0, 1)
    # Final adjustment
    s2 = np.sum(h)
    if abs(s2 - target) > 1e-10:
        # Manual fix
        h = np.clip(h, 0, 1)
        diff2 = target - np.sum(h)
        if diff2 > 0:
            mask = h < 1.0
            h[mask] = np.minimum(h[mask] + diff2 / mask.sum(), 1.0)
        else:
            mask = h > 0.0
            h[mask] = np.maximum(h[mask] + diff2 / mask.sum(), 0.0)

        zeros = np.where(h < 0.001)[0]
