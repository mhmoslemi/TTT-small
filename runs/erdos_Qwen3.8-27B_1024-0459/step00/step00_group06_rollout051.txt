from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def smoothed_max(corr, dx, alpha):
    """Smoothed maximum of dx*corr using log-sum-exp"""
    vals = dx * corr
    m = np.max(vals)
    if alpha * m > 700:
        # Avoid overflow
        vals = vals - m
        return m + np.log(np.sum(np.exp(alpha * vals))) / alpha
    return np.log(np.sum(np.exp(alpha * vals))) / alpha

    corr = np.correlate(h, 1 - h, mode='full')
    return smoothed_max(corr, dx, alpha)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def logsumexp_smoothed(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * dx * corr
    m = np.max(vals)
    return (m + np.log(np.sum(np.exp(vals - m)))) / alpha

def lse_jacobian(h, dx, alpha):
    """Analytic gradient of log-sum-exp smoothed max."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * dx * corr
    m = np.max(vals)
    exp_vals = np.exp(vals - m)
    p = exp_vals / np.sum(exp_vals)  # p[k] for k=0,...,2n-2
    
    # For each i, compute d/dh[i] of the smoothed max
    # = dx * sum_k p[k] * d(corr[k])/dh[i]
    
    grad = np.zeros(n)
    
    for i in range(n):
        # First factor: k in [i, i+n-1], contribution 1-h[n-1-(k-i)]
        # k ranges from i to i+n-1
        k_start = i
        k_end = i + n - 1
        # j = n-1-(k-i) = n-1-k+i, ranges from n-1 to 0 as k goes from i to i+n-1
        # contribution: 1-h[j] where j = n-1-(k-i)
        # So contribution array for k in [i, i+n-1] is: 1-h[n-1], 1-h[n-2], ..., 1-h[0]
        first = 1.0 - h[::-1]  # [1-h[n-1], 1-h[n-2], ..., 1-h[0]]
        # p[k] for k in [i, i+n-1]
        pk = p[k_start:k_end+1]

        # Second factor: k in [n-1-i, 2n-2-i], contribution -h[k-n+1+i]
        j_start = 0  # when k = n-1-i, j = k-n+1+i = 0

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_value(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * dx * corr
    m = np.max(vals)
    return float(m + np.log(np.sum(np.exp(vals - m)))) / alpha

def lse_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = alpha * dx * corr
    m = np.max(vals)
    exp_vals = np.exp(vals - m)
    p = exp_vals / float(np.sum(exp_vals))
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    rev_one_minus_h = one_minus_h[::-1]
    
    for i in range(n):
        # Term 1: k in [i, i+n-1]
        # contribution: sum_{k=i}^{i+n-1} p[k] * (1-h[n-1-(k-i)])
        # Let j = n-1-(k-i), so j = n-1-i, n-2-i, ..., 0-i = -i ... 
        # Actually j = n-1-(k-i) where k goes from i to i+n-1
        # j = n-1-k+i: when k=i, j=n-1; when k=i+n-1, j=0
        # So the vector is [1-h[n-1], 1-h[n-2], ..., 1-h[0]] = one_minus_h[::-1]
        pk1 = p[i:i+n]
        grad[i] += dx * np.dot(pk1, rev_one_minus_h)
        
        # Term 2: k in [n-1-i, 2n-2-i]
        # contribution: sum_{k=n-1-i}^{2n-2-i} p[k] * (-h[k-(n-1-i)])
        # Let j = k-(n-1-i), j goes from 0 to n-1
        # So the vector is [-h[0], -h[1], ..., -h[n-1]] = -h
        pk2 = p[n-1-i:2*n-1-i]
        grad[i] += dx * np.dot(pk2, -h)
    
    return grad

    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""

    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    # Bisection for lambda
    if diff > 0:
        lo, hi = -1.0, 0.0
    else:
        lo, hi = 0.0, 1.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s_proj = float(np.sum(proj))
        if diff > 0:
            if s_proj > target_sum:
                lo = mid
            else:
                hi = mid
        else:
            if s_proj < target_sum:
                lo = mid
            else:
                hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)
