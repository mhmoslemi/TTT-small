```python
def compute_gradient(h, n, t=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx
    m = corr_scaled.max()
    w = np.exp((corr_scaled - m) / t)
    w /= w.sum()
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[n-1+k] * dx
        if wk < 1e-12:
            continue
        if k >= 0:
            # (1-h[j-k]) for j in [k, n-1]
            j_range = np.arange(k, n)
            grad[j_range] += wk * (1 - h[j_range - k])
            # -h[j+k] for j in [0, n-1-k]
            j_range2 = np.arange(0, n-k)
            grad[j_range2] -= wk * h[j_range2 + k]
        else:
            p = -k
            # (1-h[j+p]) for j in [0, n-1-p]
            j_range = np.arange(0, n-p)
            grad[j_range] += wk * (1 - h[j_range + p])
            # -h[j-p] for j in [p, n-1]
            j_range2 = np.arange(p, n)
            grad[j_range2] -= wk * h[j_range2 - p]
    
    return grad
```
