import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(2000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            room = 1.0 - h[mask]
            total = room.sum()
            if total < 1e-14:
                break
            h[mask] += diff * room / total
        else:
            mask = h > 0.0
            if not mask.any():
                break
            room = h[mask]
            total = room.sum()
            if total < 1e-14:
                break
            h[mask] += diff * room / total
        h = np.clip(h, 0.0, 1.0)
    return h

class ObjFunc:
    def __init__(self, n, dx):
        self.n = n
        self.dx = dx
    def __call__(self, x):
        h = project_feasible(x, self.n)
        return compute_c5(h, self.dx)

def optimize_slsqp(h0, n, dx, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    obj = ObjFunc(n, dx)
    h0 = project_feasible(h0, n)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new, dx)
        return h_new, c5_new
    except Exception:
        return h0, compute_c5(h0, dx)
