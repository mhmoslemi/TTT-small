        res_deadline = min(start_time + budget_s - 15, time_per_res)
        if res_deadline < res_start + 10:
            res_deadline = start_time + budget_s - 15
            if time.time() > start_time + budget_s - 30:
                break
        
        # Create initial solutions
        n_solutions = 5
        solutions = []
        
        for sol_idx in range(n_solutions):
            if sol_idx == 0 and h_parent is not None:
                h_init = _init_binary(h_parent, n_pts, rng)
            else:
                h_init = _init_binary(None, n_pts, rng)
            c5_init = _compute_c5(h_init, n_pts)
            solutions.append((h_init, c5_init))
        
        # Greedy improvement
        for sol_idx, (h_sol, c5_sol) in enumerate(solutions):
            if time.time() > res_deadline:
                break
            sub_deadline = min(res_deadline, time.time() + max(5, (res_deadline - res_start) // n_solutions))
            h_sol, c5_sol = _greedy_improve(h_sol, n_pts, rng, max_rounds=100,
                                             time_limit=sub_deadline, start_time=start_time)
            solutions[sol_idx] = (h_sol, c5_sol)
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
        
        # Simulated annealing on best few
        solutions.sort(key=lambda x: x[1])
        for sol_idx in range(min(3, len(solutions))):
            if time.time() > res_deadline:
                break
            h_sol, c5_sol = solutions[sol_idx]
            sa_deadline = min(res_deadline, time.time() + max(3, (res_deadline - time.time()) // (min(3, len(solutions)) - sol_idx)))
            h_sol, c5_sol = _simulated_annealing(h_sol, n_pts, n_iter=100000,
                                                  T_start=0.3, T_end=0.005,
                                                  rng=rng, time_limit=sa_deadline, start_time=start_time)
            if c5_sol < best_c5:
                best_c5 = c5_sol
                best_h = h_sol.copy()
        
        # Greedy again after SA
        h_best_res, c5_best_res = best_h.copy(), best_c5
        if time.time() < res_deadline - 5:
            h_best_res, c5_best_res = _greedy_improve(h_best_res, n_pts, rng, max_rounds=50,
                                                       time_limit=res_deadline, start_time=start_time)
            if c5_best_res < best_c5:
                best_c5 = c5_best_res
                best_h = h_best_res.copy()
    
    # Final continuous refinement with remaining time
    remaining = deadline - time.time()
    if remaining > 10 and best_h is not None:
        n_final = len(best_h)
        h_final, c5_final = _continuous_refine(
            best_h, n_final, max_iter=5000,
            time_limit=remaining - 5, start_time=start_time, rng=rng
        )
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Evaluate and validate final solution
    if best_h is not None:
        n_final = len(best_h)
        final_c5 = _compute_c5(best_h, n_final)
        best_c5 = min(best_c5, final_c5)
        # Ensure constraint satisfaction
        dx = 2.0 / n_final
        total = np.sum(best_h) * dx
        if abs(total - 1.0) > 1e-8:
            # Fix: rescale
            target_sum = n_final / 2.0
            current_sum = np.sum(best_h)
            if current_sum > 0:
                best_h = best_h * (target_sum / current_sum)
                best_h = np.clip(best_h, 0, 1)
                # Fine-tune
                diff = target_sum - np.sum(best_h)
                best_h += diff / n_final
                best_h = np.clip(best_h, 0, 1)
            final_c5 = _compute_c5(best_h, n_final)
            best_c5 = final_c5
    else:
        # Fallback: create a simple solution
        n_final = 90
        best_h = _project_binary(rng.random(n_final), n_final, rng)
        best_c5 = _compute_c5(best_h, n_final)
    
    # Final evaluation
    n_points = len(best_h)
    c5_bound = _compute_c5(best_h, n_points)
    
    return (best_h, c5_bound, n_points)
