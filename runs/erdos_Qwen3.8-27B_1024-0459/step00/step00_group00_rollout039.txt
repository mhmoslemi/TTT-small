import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    m = np.max(corr)
    exp_terms = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_terms)
    weights = exp_terms / sum_exp

    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        s = j - (n - 1)
        if s >= 0:
            fp = np.zeros(n)
            fp[s:] = 1.0
            sp_arr = np.zeros(n)
            sp_arr[s:] += h[:n - s]
            sp_arr[:n - s] += h[s:]
            grad += w * (fp - sp_arr)
        else:
            p = -s
            fp = np.zeros(n)
            fp[:n - p] = 1.0
            sp_arr = np.zeros(n)
            sp_arr[:n - p] += h[p:]
            sp_arr[p:] += h[:n - p]
            grad += w * (fp - sp_arr)

    return grad * dx

def make_penalized(alpha, n, penalty):
    def obj(h):
        return smooth_objective(h, alpha) + penalty * (np.sum(h) - n / 2.0) ** 2
    def grad(h):
        return smooth_gradient(h, alpha) + 2.0 * penalty * (np.sum(h) - n / 2.0)
    return obj, grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-15:
        # Adjust the element closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += residual
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    old_pos = np.arange(n_old) * (2.0 / n_old)
    new_pos = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(new_pos, old_pos, h_old)
    return project_feasible(h_new, n_new)

def optimize_at_resolution(h_start, n, alpha_list, max_iter, time_limit):
    penalty = 500.0
    current = h_start.copy()
    for alpha in alpha_list:
        if time.time() - optimize_at_resolution.t0 > time_limit:
            break
        obj_fn, grad_fn = make_penalized(alpha, n, penalty)
        bounds = [(0.0, 1.0)] * n
        try:
            result = minimize(obj_fn, current, jac=grad_fn,
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': max_iter, 'ftol': 1e-10, 'gtol': 1e-8})
            if result.fun < obj_fn(current) + 1e-10:
                current = project_feasible(result.x, n)
        except Exception:
            pass
    return current

def generate_random_start(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0, 1, n)
    return project_feasible(h, n)

def generate_gear_start(n, seed):
    rng = np.random.RandomState(seed)
    n_teeth = rng.randint(2, n // 4 + 1)
    tooth_width = n // (2 * n_teeth)
    h = np.zeros(n)
    for k in range(n_teeth):
        start = k * (n // n_teeth)
        end = min(start + tooth_width, n)
        h[start:end] = 1.0
    return project_feasible(h, n)

def generate_ramp_start(n, seed):
    rng = np.random.RandomState(seed)
    # Make a triangular or ramp-like pattern
    t = np.linspace(0, 1, n)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * t + phase)
    return project_feasible(h, n)

def generate_periodic_start(n, seed):
    rng = np.random.RandomState(seed)
    freq = rng.randint(1, n // 6 + 1)
    t = np.linspace(0, 1, n)
    h = 0.5 + 0.5 * np.cos(2 * np.pi * freq * t + phase)
    return project_feasible(h, n)

    remaining = [budget_s]

        return budget_s - (time.time() - t_start)

    # Get initial solution
