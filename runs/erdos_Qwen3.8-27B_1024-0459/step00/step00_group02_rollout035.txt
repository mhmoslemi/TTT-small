    lo, hi = -1, 1
    for _ in range(100):

        val = np.clip(h - mid, 0, 1).sum()
        if val > target_sum:
            lo = mid
        else:
            hi = mid

        else:
            mask = h > 0
        count = mask.sum()
        if count > 0:

import time

def project_onto_constraints(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} using bisection."""
    h = np.array(h, dtype=float).copy()
    # If already valid
    if h.min() >= -1e-12 and h.max() <= 1 + 1e-12 and abs(h.sum() - target_sum) < 1e-10:
        h = np.clip(h, 0, 1)
        return h
    
    # Bisection on lambda: find lambda s.t. sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        val = np.clip(h - mid, 0, 1).sum()
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    
    # Fix residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add to elements that are < 1
            room = 1.0 - h
            mask = room > 1e-10
            if mask.sum() > 0:
                frac = np.minimum(room[mask] / room[mask].sum(), 1.0) if room[mask].sum() > 0 else 1.0
                # Simpler: distribute equally
                per = residual / mask.sum()
                h[mask] = np.minimum(h[mask] + per, 1.0)
        else:
            # Need to subtract from elements that are > 0
            mask = h > 1e-10
            if mask.sum() > 0:
                per = residual / mask.sum()
                h[mask] = np.maximum(h[mask] + per, 0.0)

    #          = 1 - h[j-lag] - h[j+lag] (for valid indices)
    
    for j in range(n):

        j_plus = j + lag
        if 0 <= j_plus < n:
            val -= h[j_plus]
        j_minus = j - lag
        if 0 <= j_minus < n:
            val -= h[j_minus]
        grad[j] = val

    # For shift>0: valid j in [0, n-1-l]
    # For shift<0: valid j in [l, n-1] (since j+shift=j-l≥0 means j≥l)

        # (1-h[j-shift]) for j in [shift, n-1]
        grad[shift:] += 1.0 - h[:n-l]  # h[j-shift] for j in [shift, n-1]

import numpy as np
import time


def project(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} via bisection on L2 projection."""
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target_sum) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = np.clip(h - mid, 0.0, 1.0).sum()
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(h - lam, 0.0, 1.0)
    resid = target_sum - h.sum()
    if abs(resid) > 1e-10:
        if resid > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += resid / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def subgrad_lag(h, shift):
    """Subgradient of np.correlate(h, 1-h, mode='full')[shift + n - 1] w.r.t. h."""
    n = len(h)
    if shift == 0:
        return 1.0 - 2.0 * h
    shift = int(shift)
    l = abs(shift)
    if l >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift > 0:
        grad[:n - l] -= h[shift:n]
        grad[shift:] += 1.0 - h[:n - l]
    else:
        grad[:n - l] += 1.0 - h[l:]
        grad[l:] -= h[:n - l]
    return grad


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def get_smooth_grad(h, beta):
    """Smoothed (log-sum-exp) gradient of the C5 objective (before *dx)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = beta * (corr - max_c)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        k_star = np.argmax(corr)
        return subgrad_lag(h, k_star - (n - 1))
    w /= s
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        shift = k - (n - 1)
        grad += w[k] * subgrad_lag(h, shift)
    return grad


def upsample(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return np.interp(new_x, old_x, h)


def local_optimize(h, n, rng, t0, deadline):
    """Run local optimization for h of length n until deadline."""
    target = n / 2.0
    h = project(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Smoothed gradient descent with increasing beta
    betas = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024]
    for beta in betas:
        if time.time() - t0 > deadline:
            break
        step = 0.04
        n_iter = 150
        for it in range(n_iter):

                break
            grad = get_smooth_grad(h, beta)
            h_new = h - step * grad
            h_new = project(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
            step *= 0.995

    step = 0.02
    n_iter = 300

        h_new = h - step * grad
        h_new = project(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        step = max(0.002, step * 0.995)

def try_binary_push(h, target):
    """Try pushing h toward binary (0/1) values and check improvement."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for thresh in [0.2, 0.3, 0.4, 0.5]:
        h_bin = h.copy()
        h_bin[h_bin < thresh] = 0.0

            best_c5 = c5_bin
