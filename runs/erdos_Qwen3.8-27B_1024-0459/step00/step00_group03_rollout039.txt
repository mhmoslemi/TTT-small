    """Smooth max and its gradient (softmax weights)"""
    # val is array of length 2n-1
    m = np.max(val)
    exp_val = np.exp(val - m)
    sum_exp = np.sum(exp_val)
    weights = exp_val / sum_exp

```python
import numpy as np
import time

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, 'full') * dx"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum}"""
    n = len(h)
    if n == 0:
        return h
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h - mid, 0.0, 1.0)
        s = hp.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def get_weights(vals, alpha):
    """Softmax weights for smooth max gradient"""
    scaled = alpha * vals
    m = scaled.max()
    e = np.exp(scaled - m)
    s = e.sum()
    if s == 0:
        e = np.ones_like(e) / len(e)
        return e
    return e / s

def get_gradient(h, dx, weights):
    """Gradient of smooth max of correlation values w.r.t. h"""
    n = len(h)
    A = 1.0 - h
    B = h
    w = weights  # length 2n-1
    P = np.zeros(n)
    Q = np.zeros(n)
    for i in range(n):
        P[i] = np.dot(A, w[n-1-i : 2*n-1-i])
        Q[i] = np.dot(B, w[i : i+n][::-1])
    return dx * (P - Q)

def optimize_h(h, dx, target_sum, max_time, alpha0=30.0, lr0=0.02, seed=42):
    """Optimize h using projected gradient with momentum"""
    rng = np.random.RandomState(seed)
    t_start = time.time()
    n = len(h)
    
    h = h.copy()
    h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    alpha = alpha0
    lr = lr0
    momentum = 0.9
    v = np.zeros(n)
    
    stall_count = 0
    max_stall = 500
    
    while time.time() - t_start < max_time:
        vals = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = vals.max()
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > max_stall:
            # Restart with perturbation
            h = project_feasible(best_h + 0.05 * rng.randn(n), target_sum)
            v = np.zeros(n)
            lr = lr0
            alpha = alpha0
            stall_count = 0
            continue
        
        w = get_weights(vals, alpha)
        grad = get_gradient(h, dx, w)
        
        # Nesterov-style momentum
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5:
            h = h_new
            alpha = min(alpha * 1.002, 150.0)
        else:
            lr *= 0.8
            v *= 0.5
            if lr < 1e-7:
                lr = lr0 * 0.5
                v = np.zeros(n)
    
    return best_h, best_c5
