def fix_sum_binary(h):
    """Ensure sum(h) = len(h)//2 for a binary array."""
    n = len(h)
    target = n // 2
    current_sum = int(np.sum(h))

        np.random.shuffle(zeros)

        need = current_sum - target
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        h[ones[:need]] = 0
    return h

    # For the full correlation index k (0 to L-1), the actual lag is lags[k] = k - (n-1)
