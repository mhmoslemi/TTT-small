import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(vals, tau):
    mv = np.max(vals)
    if tau < 1e-6:
        return float(mv)
    e = np.exp(tau * (vals - mv))
    s = e.sum()
    if s < 1e-300:
        return float(mv)
    return float(mv + np.log(s) / tau)

def interpol_h(h_coarse, n_fine):
    nc = len(h_coarse)
    if nc == n_fine:
        return h_coarse.copy()
    x_c = (np.arange(nc) + 0.5) / nc
    x_f = (np.arange(n_fine) + 0.5) / n_fine
    h_f = np.interp(x_f, x_c, h_coarse, left=h_coarse[0], right=h_coarse[-1])
    return np.clip(h_f, 0.0, 1.0)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        rsum = room.sum()
        if rsum > 1e-12:
            h += (diff / rsum) * room
    else:
        rsum = h.sum()
        if rsum > 1e-12:
            h += (diff / rsum) * h
    h = np.clip(h, 0.0, 1.0)
    residual = target - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + residual)
    return h

def sa_binary(h_init, n, target_sum, max_time):
    h = (h_init > 0.5).astype(float).copy()
    if h.sum() > target_sum:
        order = np.argsort(-h_init)
        for i in order:
            if h.sum() <= target_sum:
                break
            h[i] = 0.0
    elif h.sum() < target_sum:
        order = np.argsort(-h_init)
        for i in order:
            if h.sum() >= target_sum:
                break
            h[i] = 1.0
    h = fix_sum(h, target_sum)
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    T = 0.05
    T_min = 0.0005
    t_end = time.time() + max_time
    while time.time() < t_end and T > T_min:
        i = int(np.random.randint(n))
        old_val = h[i]
        h[i] = 1.0 - old_val
        if h.sum() != target_sum:
            if h.sum() > target_sum:
                ones = np.where(h == 1)[0]
                if len(ones) > 0:
                    h[int(np.random.choice(ones))] = 0.0
            else:
                zeros = np.where(h == 0)[0]
                if len(zeros) > 0:
                    h[int(np.random.choice(zeros))] = 1.0
        h = fix_sum(h, target_sum)
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h[i] = old_val
            h = fix_sum(h, target_sum)
            c5_cur = compute_c5(h)
        T *= 0.998
    return best_h, best_c5
