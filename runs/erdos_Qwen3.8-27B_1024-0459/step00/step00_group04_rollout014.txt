    try:
        res = minimize(obj, h0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': xatol, 'fatol': fatol,
                                'adaptive': True})
        return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def try_update(best_h, best_c5, candidate_h, n_cand):
    candidate_h = project_feasible(candidate_h, n_cand)
    c5_cand = compute_c5(candidate_h)
    if c5_cand < best_c5:
        return candidate_h, c5_cand
    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def make_binary(h, n):
    """Create a binary (0/1) vector with exactly n/2 ones, closest to h."""
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    k = n // 2
    for i in range(k):
        h_bin[sorted_idx[i]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    def remaining_frac():
        return max(0.0, 1.0 - elapsed() / budget_s)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    h_init = project_feasible(h_init, n)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at initial resolution
    if remaining_frac() > 0.75:
        h_candidate = slsqp_optimize(h_init, n, maxiter=1000, ftol=1e-15)
        best_h, best_c5 = try_update(best_h, best_c5, h_candidate, n)

    # Phase 2: Coordinate descent
    if remaining_frac() > 0.6:
        time_left = budget_s * remaining_frac() * 0.15
        h_cd = coord_descent(best_h, n, n_sweeps=100,
                            time_limit=time_left, start_time=start_time)
        h_cd = project_feasible(h_cd, n)
        best_h, best_c5 = try_update(best_h, best_c5, h_cd, n)

    # Phase 3: Nelder-Mead refinement
    if remaining_frac() > 0.5:
        h_nm = nelder_mead_optimize(best_h, n, maxiter=5000, xatol=1e-10, fatol=1e-14)
        best_h, best_c5 = try_update(best_h, best_c5, h_nm, n)

    # Phase 4: More coordinate descent
    if remaining_frac() > 0.4:
        time_left = budget_s * remaining_frac() * 0.1
        h_cd = coord_descent(best_h, n, n_sweeps=100,
                            time_limit=time_left, start_time=start_time)
        h_cd = project_feasible(h_cd, n)
        best_h, best_c5 = try_update(best_h, best_c5, h_cd, n)

    # Phase 5: Binary-ification attempt
    if remaining_frac() > 0.3:
        h_bin = make_binary(best_h, n)
        h_bin_opt = slsqp_optimize(h_bin.astype(float), n, maxiter=500)
        h_bin_opt = coord_descent(h_bin_opt, n, n_sweeps=30,
                                  time_limit=budget_s * 0.03, start_time=start_time)
        h_bin_opt = project_feasible(h_bin_opt, n)
        best_h, best_c5 = try_update(best_h, best_c5, h_bin_opt, n)

    # Phase 6: Random restarts at original resolution
    if remaining_frac() > 0.25:
        n_restarts = 5
        for r in range(n_restarts):
            if elapsed() > budget_s * 0.7:
                break
            h_rand = np.random.rand(n)
            h_rand = project_feasible(h_rand, n)
            h_r = slsqp_optimize(h_rand, n, maxiter=300)
            time_left = budget_s * 0.02
            h_r = coord_descent(h_r, n, n_sweeps=30,
                               time_limit=time_left, start_time=start_time)
            h_r = project_feasible(h_r, n)
            best_h, best_c5 = try_update(best_h, best_c5, h_r, n)

    # Phase 7: Higher resolution
    for mult in [2, 3, 4, 6]:
        if remaining_frac() < 0.15:
            break
        n_new = n * mult
        if n_new > 500:
            n_new = 500
        if n_new == n:
            continue

        h_interp = interpolate_h(best_h, len(best_h), n_new)
        h_interp = project_feasible(h_interp, n_new)

        # SLSQP
        if n_new <= 300 and remaining_frac() > 0.2:
            h_opt = slsqp_optimize(h_interp, n_new, maxiter=300)
            best_h, best_c5 = try_update(best_h, best_c5, h_opt, n_new)

        # Coordinate descent
        if remaining_frac() > 0.18:
            time_left = budget_s * 0.03
            h_cd = coord_descent(h_interp if n_new > len(best_h) else best_h, n_new,
                                n_sweeps=30, time_limit=time_left, start_time=start_time)
            h_cd = project_feasible(h_cd, n_new)
            best_h, best_c5 = try_update(best_h, best_c5, h_cd, n_new)

    # Phase 8: Final refinement at best resolution
    if remaining_frac() > 0.05:
        n_best = len(best_h)
        time_left = budget_s * 0.04
        h_cd = coord_descent(best_h, n_best, n_sweeps=50,
                            time_limit=time_left, start_time=start_time)
        h_cd = project_feasible(h_cd, n_best)
        best_h, best_c5 = try_update(best_h, best_c5, h_cd, n_best)

    # Final SLSQP polish
    if remaining_frac() > 0.03:
        n_best = len(best_h)
        h_pol = slsqp_optimize(best_h, n_best, maxiter=200, ftol=1e-15)
        best_h, best_c5 = try_update(best_h, best_c5, h_pol, n_best)

    # Ensure final feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
