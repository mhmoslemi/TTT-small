def project_feasible(h, n):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float) * (2.0 / n_old)
    x_new = np.arange(n_new, dtype=float) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
