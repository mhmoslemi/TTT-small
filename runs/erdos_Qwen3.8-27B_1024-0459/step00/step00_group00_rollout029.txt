from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def interpolate_to_resolution(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    diff = target - h.sum()
    h += diff / n
    h = np.clip(h, 0, 1)
    # If clipping changed the sum, do a second pass
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0, 1)
    return h

def optimize_slsqp(h0, n, time_limit, start_time):
    """Optimize h to minimize max correlation using SLSQP."""
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Parameters: [h_0, ..., h_{n-1}, t]
    n_total = n + 1
    
    def objective(params):
        if time.time() - start_time > time_limit:
            return 1e10
        return params[n]  # t
    
    def constraint_sum(params):
        return params[:n].sum() - target_sum
    
    def constraint_corr(params):
        if time.time() - start_time > time_limit:
            return np.ones(2*n-1)
        h = np.clip(params[:n], 0, 1)
        t = params[n]
        corr = np.correlate(h, 1 - h, mode="full")
        return t - corr * dx
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': constraint_corr}
    ]
    
    bounds = [(0, 1)] * n + [(0, 1)]  # h in [0,1], t in [0,1]
    
    x0 = np.concatenate([h0, [evaluate_c5(h0)]])
    
    result = minimize(
        objective, x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-10}
    )

        h_opt = np.clip(result.x[:n], 0, 1)
        h_opt = fix_sum(h_opt, n)

Actually wait, if h only takes values 0 and 1, then h(x)(1-h(x+k)) is 1 only when h(x)=1 and h(x+k)=0. So the integral is just the measure of {x: h(x)=1 and h(x+k)=0} times dx... no, the integral of a step function.

For a step function with h ∈ {0,1}:
∫ h(x)(1-h(x+k)) dx = measure of {x ∈ [0,2]: h(x)=1 and h(x+k)=0}

In discrete: sum_i [h[i]=1 and h[i+k mod n]=0] * dx (if we wrap around) or without wrap.

With np.correlate (no wrap), it's: sum over valid i of h[i]*(1-h[i+shift]) * dx.

If h ∈ {0,1}, then h[i]*(1-h[i+shift]) is 1 if h[i]=1 and h[i+shift]=0, else 0.

So the objective becomes: max_k (count_k * dx) where count_k = number of i with h[i]=1 and h[i+k]=0 (for valid i).

This is a purely combinatorial problem! And the "smooth" relaxation (h ∈ [0,1]) is what we're optimizing.

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0, 1)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        # Distribute the difference
        h += diff / n
        h = np.clip(h, 0, 1)
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Distribute to entries not at boundary
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
        else:
            h += diff / n
            h = np.clip(h, 0, 1)
    return h

    target_sum = n / 2.0
    c5_init = evaluate_c5(h0)
    
    def obj(params):
        if time.time() - start_time > max_time:
            return 1e10
        t = params[n]
        return t
    
    def con_eq(params):
        if time.time() - start_time > max_time:
            return 0

    def con_ineq(params):
        if time.time() - start_time > max_time:
            return np.ones(2 * n - 1)
        h = params[:n]
        t = params[n]
        corr = np.correlate(h, 1 - h, mode="full")
        return t - corr * dx
    
    constraints = [
        {'type': 'eq', 'fun': con_eq},
        {'type': 'ineq', 'fun': con_ineq}
    ]
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    x0 = list(h0) + [c5_init + 1e-6]
    
    result = minimize(
        obj, x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
    )
    
    if result is not None:
        h_opt = np.clip(result.x[:n], 0, 1)
        h_opt = project_constraint(h_opt, n)
        c5 = evaluate_c5(h_opt)
        if c5 < c5_init:
            return h_opt, c5
    return None, c5_init

def optimize_gradient(h0, n, max_time, start_time, lr=0.01):
    """Simple projected gradient descent on smooth max."""
    h = h0.copy()
    target_sum = n / 2.0
    dx = 2.0 / n
    best = evaluate_c5(h)
    best_h = h.copy()
    
    temperature = 50
    
    while time.time() - start_time < max_time:
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode="full") * dx
        # Smooth max
        f = np.log(np.sum(np.exp(temperature * corr))) / temperature
        
        # Numerical gradient
        grad = np.zeros(n)
        eps = 1e-5
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] = min(1, h[i] + eps)
            h_minus = h.copy()
            h_minus[i] = max(0, h[i] - eps)
            f_plus = np.log(np.sum(np.exp(temperature * np.correlate(h_plus, 1-h_plus, mode="full")*dx))) / temperature
            f_minus = np.log(np.sum(np.exp(temperature * np.correlate(h_minus, 1-h_minus, mode="full")*dx))) / temperature

        # Update

        h = project_constraint(h, n)
        
        c5 = evaluate_c5(h)
        if c5 < best:
            best = c5
            best_h = h.copy()
    
    return best_h, best

        h_interp = np.interp(x_new, x_old, best_h)

Actually, let me compute the analytical gradient of the max correlation.

For c_k(h) = sum_i h[i] * (1-h[i+s_k]) * dx (where s_k is the shift for the k-th correlation value):

dc_k/dh[j] = dx * [(1-h[j+s_k]) * delta(j as first index) + h[j] * (-1) * delta(j as second index)]
          = dx * [(1-h[j+s_k]) - h[j-s_k]] (when both terms are valid)

But this is for a specific k. The max over k is non-smooth.

For the smooth max: f(h) = (1/T) log(sum_k exp(T * c_k(h)))

df/dh[j] = (1/T) * sum_k exp(T*c_k) / sum_m exp(T*c_m) * T * dc_k/dh[j]
         = sum_k p_k * dc_k/dh[j]

where p_k = exp(T*c_k) / sum_m exp(T*c_m) (softmax weights).

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0, 1)
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def interpolate_h(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def optimize_slsqp(h0, n, max_time, start_time):
    dx = 2.0 / n
    target_sum = n / 2.0
    c5_init = evaluate_c5(h0)
    n_corr = 2 * n - 1

    def obj(params):
        if time.time() - start_time > max_time:
            return 1e10
        return params[n]

    def con_eq(params):
        return params[:n].sum() - target_sum

    def con_ineq(params):
        if time.time() - start_time > max_time:
            return np.ones(n_corr)
        h = params[:n]
        t = params[n]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t - corr * dx

    constraints = [
        {'type': 'eq', 'fun': con_eq},
        {'type': 'ineq', 'fun': con_ineq}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    x0 = np.concatenate([h0, [c5_init + 1e-8]])

    try:
        result = minimize(
            obj, x0, method='SLSQP', bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300, 'ftol': 1e-12, 'disp': False}
        )
        if result is not None and len(result.x) >= n + 1:
            h_opt = np.clip(result.x[:n], 0, 1)
            h_opt = project_constraint(h_opt, n)
            c5 = evaluate_c5(h_opt)
            if c5 < c5_init - 1e-12:
                return h_opt, c5
    except Exception:
        pass
    return None, c5_init

def local_search(h0, n, max_time, start_time, rng):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    target_sum = n / 2.0

    step = 0.1
    iters = 0
    while time.time() - start_time < max_time:
        iters += 1
        i = rng.randint(n)
        delta = rng.uniform(-step, step)
        h_new = h.copy()
        h_new[i] = np.clip(h[i] + delta, 0, 1)
        actual_delta = h_new[i] - h[i]
        if abs(actual_delta) < 1e-15:
            continue
        j = (i + 1 + rng.randint(n - 1)) % n
        h_new[j] = np.clip(h[j] - actual_delta, 0, 1)
        actual_correction = -(h_new[j] - h[j])
        h_new[i] = np.clip(h[i] + actual_delta + actual_correction, 0, 1)
        h_new = project_constraint(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
            step = min(step * 1.1, 0.5)
        else:
            step = max(step * 0.9, 0.01)
        if iters % 500 == 0:
            step = max(step * 0.5, 0.005)

    return best_h, best_c5

    indices = rng.choice(n, size=n // 2, replace=False)
