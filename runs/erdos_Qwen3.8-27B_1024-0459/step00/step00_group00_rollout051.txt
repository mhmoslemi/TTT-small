            h[i], h[j] = 1.0, 0.0
            stall += 1

    return h, best_max * 2.0 / n

def continuous_optimize(h_init, n, time_limit=20.0):
    x0 = h_init.astype(np.float64).copy()
    x0 = np.clip(x0, 0.0, 1.0)

    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        return float(np.max(corr)) * dx

    def grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        max_val = np.max(corr)
        # Subgradient: gradient of the argmax component
        max_idx = np.argmax(corr)
        lag = max_idx - n + 1
        i_min = max(0, -lag)
        i_max = min(n - 1, n - 1 - lag)
        g = np.zeros(n)
        for i in range(i_min, i_max + 1):
            # corr contribution: x[i+lag]*(1-x[i])
            # d/dx[i+lag] = (1-x[i]) * dx
            # d/dx[i] = -x[i+lag] * dx
            g[i + lag] += (1.0 - x[i])
            g[i] -= x[i + lag]
        return g * dx

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    start = time.time()
    try:
        res = minimize(obj, x0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 500, 'ftol': 1e-12})
        if time.time() - start < time_limit and res.fun < obj(x0):
            return res.x.copy(), float(res.fun)
    except Exception:
        pass
    return x0, obj(x0)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = initial_h_values.copy()
    n_parent = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search at multiple sizes
    sizes = [40, 50, 60, 80, 100, 120, 160, 200, 300, 400]
    # Weight smaller sizes more (they're faster)
    for n in sizes:
        if n % 2 != 0:
            continue
        remaining = budget_s - (time.time() - t0) - 60
        if remaining < 10:
            break

        # Budget for this size
        if n <= 80:
            n_trials = min(8, max(2, int(remaining / 30)))
            t_per = max(1.0, remaining / n_trials)
        elif n <= 160:
            n_trials = min(4, max(2, int(remaining / 50)))
            t_per = max(1.0, remaining / n_trials)
        else:
            n_trials = max(1, min(2, int(remaining / 80)))
            t_per = max(2.0, remaining / n_trials)

        for trial in range(n_trials):
            if time.time() - t0 > budget_s - 60:
                break

            #多样化初始化
            if trial == 0:
                # Resample parent
                x_old = np.linspace(0, 2, n_parent, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h_init = np.interp(x_new, x_old, h_parent)
                h_init = (h_init > 0.5).astype(np.float64)
            elif trial == 1:
                # Block pattern
                h_init = np.zeros(n)
                block = max(4, n // 25)
                for k in range(0, n, block * 2):
                    h_init[k:k + block] = 1.0
            else:
                # Random
                h_init = np.zeros(n)
                h_init[rng.choice(n, n // 2, replace=False)] = 1.0

            h_opt, c5_val = local_search_binary(h_init, rng, time_limit=t_per)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()

    # Phase 2: Refine best binary solution with extra local search
    remaining = budget_s - (time.time() - t0) - 30
    if remaining > 10 and best_h is not None:
        n_best = len(best_h)
        if n_best % 2 == 0:
            h_refined, c5_ref = local_search_binary(best_h, rng, time_limit=min(remaining, 60.0))
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_refined.copy()

    # Phase 3: Continuous optimization from best solution
    remaining = budget_s - (time.time() - t0) - 15
    if remaining > 20:
        # Try at a few sizes near the best
        for n_try in [min(n_parent, 40), 50, 60]:
            if time.time() - t0 > budget_s - 15:
                break
            if len(best_h) >= n_try:
                h_cont_init = best_h[:n_try]
            else:
                x_old = np.linspace(0, 2, len(best_h), endpoint=False)
                x_new = np.linspace(0, 2, n_try, endpoint=False)
                h_cont_init = np.interp(x_new, x_old, best_h)

            h_cont_opt, c5_cont = continuous_optimize(h_cont_init, n_try, time_limit=remaining / 3)
            if c5_cont < best_c5:
                best_c5 = c5_cont
                best_h = h_cont_opt.copy()

    # Phase 4: Final verification and fix constraints
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = n_final / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-10:
        # Distribute correction
        if diff > 0:
            # Need to add: bump up smallest values
            order = np.argsort(best_h)
            for idx in order:
                add = min(diff, 1.0 - best_h[idx])
                if add > 1e-10:
                    best_h[idx] += add
                    diff -= add
                    if abs(diff) < 1e-10:
                        break
        else:
            # Need to subtract: reduce largest values
            order = np.argsort(-best_h)
            for idx in order:
                sub = min(-diff, best_h[idx])
                if sub > 1e-10:
                    best_h[idx] -= sub
                    diff += sub
                    if abs(diff) < 1e-10:
                        break

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    # If continuous opt broke something, revert
    if final_c5 > best_c5 + 1e-8:
        best_c5 = final_c5

    return best_h, best_c5, n_final
