import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    else:
        Lp = -L
        g[Lp:] += (1.0 - h[:n-Lp])
        g[:n-Lp] -= h[Lp:]
    return g

def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def subgrad_optimize(h0, max_iters, top_k=3, alpha0=0.05, name=""):
    n = len(h0)
    target_sum = 0.5 * n
    dx = 2.0 / n
    h = project_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    # Adaptive step size
    alpha = alpha0
    
    for t in range(max_iters):
        corr = compute_corr(h)
        # Get top-k
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        
        # Average gradient
        g = np.zeros(n)
        for idx in top_idx:
            L = idx - (n - 1)
            g += grad_lag(h, L)
        g /= top_k
        
        # Step
        h_new = project_feasible(h - alpha * g, target_sum)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_val:
            best_val = c5_new
            best_h = h_new.copy()
            alpha *= 1.01  # increase slightly on improvement
        else:
            alpha *= 0.95  # decrease on non-improvement
        
        h = h_new
        # Keep alpha in bounds
        alpha = max(alpha, 1e-6)
        alpha = min(alpha, 0.5)
    
    return best_h, best_val

def resample(h_old, n_new):
    """Resample h from old resolution to new resolution."""
    n_old = len(h_old)
    # Create centers
    x_old = np.arange(n_old) * (2.0 / n_old) + 1.0 / n_old
    x_new = np.arange(n_new) * (2.0 / n_new) + 1.0 / n_new
    # Interpolate
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def smooth_max_corr(h, beta):
    """Smooth max of correlation * dx using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    x = beta * corr
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(x - x_max))) / beta

def smooth_max_grad(h, beta):
    """Gradient of smooth max of correlation * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    x = beta * corr
    x_max = np.max(x)
    weights = np.exp(x - x_max)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for i, w in enumerate(weights):
        if w > 1e-8:
            L = i - (n - 1)
            g += w * grad_lag(h, L)
    g *= dx
    return g

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
    else:
        Lp = -L
        if Lp < n:
            g[Lp:] += (1.0 - h[:n-Lp])
            g[:n-Lp] -= h[Lp:]
    return g

def project_feasible(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    return np.clip(h - mu, 0.0, 1.0)

    """One subgradient step. Returns (h_new, improvement_flag)."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    g = np.zeros(n)
    for idx in top_idx:
        L = idx - (n - 1)
        g += grad_lag(h, L)
    g /= top_k
    return g

    x_max = np.max(x)

    """Gradient of overlap_L = sum_n h[n]*(1-h[n-L]) w.r.t. h.
    L ranges from -(n-1) to (n-1)."""
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        # valid1: j in [L, n-1]: +(1-h[j-L])
        g[L:] += (1.0 - h[:n-L])
        # valid2: j in [0, n-1-L]: -h[j+L]
        g[:n-L] -= h[L:]
    else:
        Lp = -L
        # valid1: j in [0, n-1-Lp]: +(1-h[j+Lp])
        g[:n-Lp] += (1.0 - h[Lp:])
        # valid2: j in [Lp, n-1]: -h[j-Lp]
        g[Lp:] -= h[:n-Lp]
    return g

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n

def grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L >= 0:
        if L < n:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
    else:
        Lp = -L
        if Lp < n:
            g[:n-Lp] += (1.0 - h[Lp:])
            g[Lp:] -= h[:n-Lp]
    return g

        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    return np.clip(h - mu, 0.0, 1.0)

def resample(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return np.clip(h_new, 0.0, 1.0)
