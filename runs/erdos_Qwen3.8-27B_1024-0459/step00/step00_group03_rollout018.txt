    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def smooth_max(correlation, dx, tau):
    """Smooth max of correlation * dx using log-sum-exp"""
    vals = correlation * dx
    # Scale for numerical stability
    max_val = np.max(vals)
    exp_vals = np.exp(tau * (vals - max_val))
    return max_val + np.log(np.sum(exp_vals)) / tau, exp_vals

        lo1 = max(0, -s)
        hi1 = min(n-1, n-1-s)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += weights[k] * (1.0 - h[lo1+s:hi1+s+1])
        
        # For j-s in [0,n-1]: j in [max(0,s), min(n-1, s+n-1)]
        lo2 = max(0, s)
        hi2 = min(n-1, s+n-1)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= weights[k] * h[lo2-s:hi2-s+1]
    
    grad *= dx
    return grad

    new_x = np.linspace(0, 2, new_n + 1)[:-1] + 1.0/new_n

The second and third terms can be computed as weighted sums of h with shifted indices, which is like a weighted correlation.

    target = n / 2.0
    if abs(h.sum() - target) < 1e-10 and h.min() >= 0 and h.max() <= 1:
        return h.copy()
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:

        else:

    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    vals = corr * dx
    
    max_val = np.max(vals)

    log_w = np.clip(log_w, -300, 300)
    w = np.exp(log_w)

def nelder_mead_optimize(h0, n, max_iter=3000, time_limit=60.0):
    start_time = time.time()
    
    def objective(x):
        h = np.clip(x, 0.0, 1.0)

        h = project_h(h, n)
