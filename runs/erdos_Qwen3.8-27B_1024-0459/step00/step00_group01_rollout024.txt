import numpy as np
import time
import math

def thue_morse(n):
    result = np.zeros(n, dtype=float)
    for i in range(n):
        v = 0
        x = i
        while x:
            v ^= 1
            x &= x - 1
        result[i] = v
    return result

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def normalize_sum(h, target):
    h = h.astype(float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        for _ in range(h.size * 20):
            room = 1.0 - h
            i = np.argmax(room)
            if room[i] < 1e-14:
                break
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        for _ in range(h.size * 20):
            i = np.argmax(h)
            if h[i] < 1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff > -1e-12:
                break
    return h

        old_j = h[j]

                stall_count = 0
