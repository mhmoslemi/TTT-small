def analytic_grad(h, T, dx):

    corr = np.correlate(h, 1-h, mode='full')

    c -= c.max()

        # j-s in [0,n-1] AND j in [0,n-1]
        lo1 = max(0, s)
        hi1 = min(n-1, n-1+s)  # j <= n-1+s, j <= n-1, so j <= n-1
        # Actually: j-s >= 0 → j >= s; j-s <= n-1 → j <= n-1+s; j >= 0; j <= n-1
        lo1 = max(0, s)
        hi1 = min(n-1, n-1+s)

        # For s<0: lo1=0, hi1=n-1+s = n-1-t where t=-s

        lo2 = max(0, -s)
        hi2 = min(n-1, n-1-s)
        # For s>=0: lo2=0, hi2=n-1-s
        # For s<0: lo2=-s=t, hi2=n-1

To minimize this, we want to maximize ∫h² dx, subject to ∫h dx = 1, 0≤h≤1.
∫h² dx ≥ (∫h dx)²/2 = 1/2 (by Cauchy-Schwarz, since the domain has measure 2).
Equality when h is constant = 1/2. But then ∫h(1-h) dx = 1 - 1/4*2... wait.
If h = 1/2 everywhere: ∫h² dx = ∫1/4 dx over [0,2] = 1/2. So ∫h(1-h) dx = 1 - 1/2 = 1/2.
And dx * corr[0] (for n points, corr[0] = sum h[i](1-h[i]) = n/4 when h[i]=1/2) * dx = n/4 * 2/n = 1/2.

But we can also make h binary (0 or 1), in which case corr[0] = 0 (since h(1-h)=0 for binary h). The challenge is the other shifts.

For binary h with n/2 ones and n/2 zeros:
corr_s = number of i where h[i+s]=1 and h[i]=0 (for shift s)
= |{i: i in A^c, i+s in A}| where A = {i: h[i]=1}

We want to minimize max_s |{i: i in A^c, i+s in A}|.

The total number of (i, s) pairs with i in A^c and i+s in A is (n/2)*(n/2) = n²/4 (counting all s for each i).
Spread over 2n-1 shifts, average is n²/(4(2n-1)) ≈ n/8.

So C₅ ≈ dx * n/8 = (2/n) * n/8 = 1/4 = 0.25.

import numpy as np
import time

def project(y, S):
    """Project y onto {h: 0<=h_i<=1, sum(h_i)=S}"""
    n = len(y)
    lo = -np.max(y) - 2.0
    hi = np.max(y) + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def analytic_grad(h, T, dx):
    """Gradient of dx * T * log(sum exp(corr/T)) w.r.t. h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr / T
    c -= c.max()
    w = np.exp(c)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            L = n - s
            grad[s:] += wm * (1.0 - h[:L])
            grad[:L] -= wm * h[s:]
        else:
            t = -s
            L = n - t
            grad[:L] += wm * (1.0 - h[t:])
            grad[t:] -= wm * h[:L]
    
    return dx * grad

def pgd_optimize(h0, n, T_start, T_end, lr_init, n_iter, time_budget, verbose=False):
    """Projected gradient descent"""
    dx = 2.0 / n
    S = n / 2
    h = project(h0.copy(), S)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    t0 = time.time()
    lr = lr_init
    T = T_start
    stalled = 0
    
    for it in range(n_iter):
        if time.time() - t0 > time_budget:
            break
        
        grad = analytic_grad(h, T, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Step
        for attempt in range(20):
            h_new = project(h - lr * grad, S)
            c5_new = compute_c5(h_new, n)
            if c5_new <= best_c5 + 1e-6:
                break
            lr *= 0.5
        else:
            break
        
        h = h_new
        c5_cur = compute_c5(h, n)
        
        if c5_cur < best_c5 - 1e-10:
            best_h = h.copy()
            best_c5 = c5_cur
            stalled = 0
            lr = min(lr * 1.1, lr_init)
        else:
            stalled += 1
        
        # Anneal temperature
        T = max(T * 0.995, T_end)
        
        # Random perturbation if stalled
        if stalled > 50:
            h = project(h + 0.05 * np.random.randn(n), S)
            lr = lr_init * 0.5
            stalled = 0
    
    return best_h, best_c5

def upsample(h, factor):
    n_old = len(h)
    n_new = n_old * factor
    S = n_new / 2.0
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, S)

def random_binary_init(n, seed=0):
    rng = np.random.RandomState(seed)
    h = rng.randint(0, 2, n).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        to_flip = rng.choice(ones_idx, n_ones - target, replace=False)
        h[to_flip] = 0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zeros_idx, target - n_ones, replace=False)
        h[to_flip] = 1
    return h

def periodic_init(n, period, shift):
    """Create a periodic binary pattern with given period and shift"""
    h = np.zeros(n)
    pattern = np.zeros(period)
    pattern[:period//2] = 1.0
    # Apply shift
    pattern = np.roll(pattern, shift)
    h = np.tile(pattern, n // period + 1)[:n]
    return project(h, n / 2.0)
