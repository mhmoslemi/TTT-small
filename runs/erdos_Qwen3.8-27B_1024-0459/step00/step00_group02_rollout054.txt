import time

def compute_c5(h):
    """Compute the C5 objective value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_correlations(h):
    """Compute all correlation values."""
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode='full')

def project_to_set(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}."""
    n = len(h)
    target_sum = float(target_sum)
    
    def f_of_lam(lam):
        return np.sum(np.clip(h + lam, 0.0, 1.0)) - target_sum
    
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if f_of_lam(mid) < 0:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)

def compute_subgradient(h):
    """Compute subgradient of the max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    
    grad = np.zeros(n)
    for j in range(n):
        # From h[j] * (1-h[j+lag]) term: d/dh[j] = (1-h[j+lag]) if 0<=j+lag<n
        val = 0.0
        if 0 <= j + lag < n:
            val += (1.0 - h[j + lag])
        # From -h[j] * h[j-lag] term (i.e., -(1-h[j-lag])*h[j] contribution from i=j-lag):
        # Correlation term for i: h[i]*(1-h[i+lag])
        # d/dh[j] of h[i]*(1-h[i+lag]) when i+lag=j: -h[i] = -h[j-lag] if 0<=j-lag<n
        if 0 <= j - lag < n:
            val -= h[j - lag]
        grad[j] = val
    
    return dx * grad

def smooth_max_gradient(h, beta=50.0):
    """Compute gradient of smooth max approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = dx * np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights
    corr_shifted = corr - np.max(corr)
    weights = np.exp(beta * corr_shifted)
    weights /= np.sum(weights)
    
    # Gradient is weighted average of individual gradients
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for m in range(n_full):
        if weights[m] < 1e-10:
            continue
        lag = m - (n - 1)
        for j in range(n):
            val = 0.0
            if 0 <= j + lag < n:
                val += (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                val -= h[j - lag]
            grad[j] += weights[m] * dx * val
    
    return grad

def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = dx * np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_vals = np.exp(beta * (corr - max_c))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for m in range(n_full):
        w = weights[m]
        if w < 1e-12:
            continue
        k = (n - 1) - m
        ak = abs(k)
        if ak >= n:
            continue
        idx1 = np.arange(n - ak)
        grad[idx1] += w * (1.0 - h[idx1 + ak])
        idx2 = np.arange(ak, n)
        if k >= 0:
            grad[idx2] -= w * h[idx2 - ak]
        else:
            grad[idx2] -= w * h[idx2 - ak]
    return grad

def optimize_smooth(h, target_sum, max_time, beta_init=40.0, lr_init=0.005):
    n = len(h)
    h = h.copy()
    h = project_to_set(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = lr_init
    beta = beta_init
    no_improve = 0
    stall_count = 0
    
    t0 = time.time()
    while time.time() - t0 < max_time:
        if time.time() - t0 > max_time - 1:
            break
        grad = smooth_max_grad(h, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        # Adaptive step size with simple line search
        accepted = False
        for attempt in range(5):
            h_new = h - lr * grad
            h_new = project_to_set(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                no_improve = 0
                lr *= 1.1
                beta = min(beta * 1.03, 500.0)
                accepted = True
                break
            elif c5_new < compute_c5(h) + 1e-8:
                # Accept even neutral moves to escape plateaus occasionally
                if no_improve > 15 and np.random.random() < 0.3:
                    h = h_new
                    no_improve += 1
                    lr *= 0.8
                    accepted = True
                    break
            lr *= 0.6
        
        if not accepted:
            no_improve += 1
            stall_count += 1
            if stall_count > 30:
                break
            if no_improve % 10 == 0:
                beta = min(beta * 1.5, 800.0)
                # Try a random kick
                if beta > 300:
                    h = h + 0.02 * np.random.randn(n)
                    h = project_to_set(h, target_sum)
    
    return best_h, best_c5

def push_to_binary(h, target_sum):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = int(round(target_sum))
    
    if n_ones > n_target:
        ones_idx = np.where(h_bin == 1)[0]
        sorted_ones = ones_idx[np.argsort(h[ones_idx])]
        for idx in sorted_ones[:n_ones - n_target]:
            h_bin[idx] = 0
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0)[0]
        sorted_zeros = zeros_idx[np.argsort(-h[zeros_idx])]
        for idx in sorted_zeros[:n_target - n_ones]:
            h_bin[idx] = 1
    
    return h_bin

def binary_local_search(h_bin, max_time):
    n = len(h)
    best_c5 = compute_c5(h_bin)
    h_curr = h_bin.copy()
    t0 = time.time()
    
    ones = list(np.where(h_curr == 1)[0])
    zeros = list(np.where(h_curr == 0)[0])
    
    improved = True
    while improved and time.time() - t0 < max_time:
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        for i in ones[:20]:
            for j in zeros[:20]:
                h_trial = h_curr.copy()
                h_trial[i] = 0
                h_trial[j] = 1
                c5_try = compute_c5(h_trial)
                if c5_try < best_c5 - 1e-10:
                    h_curr = h_trial
                    best_c5 = c5_try
                    improved = True
                    break
            if improved:
                break
        if time.time() - t0 > max_time - 2:
            break
    
    return h_curr, best_c5

def upscale(h, factor):
    return np.repeat(h, factor)

def structured_init(n, seed=0):
    """Create a structured initial guess with periodic pattern."""
    rng = np.random.RandomState(seed)
    # Try a pattern based on golden ratio spacing
    phi = (1 + 5**0.5) / 2
    h = np.zeros(n)
    count = n // 2
    last = -1
    for i in range(count):
        pos = int((i * phi) % n)
        if pos == last and i > 0:
            pos = (pos + 1) % n
        h[pos] = 1.0
        last = pos
    # Ensure sum constraint
    n_ones = int(np.sum(h))
    if n_ones < count:
        for i in range(n):
            if h[i] == 0 and n_ones < count:
                h[i] = 1.0
                n_ones += 1
    elif n_ones > count:
        idxs = np.where(h == 1)[0]
        for idx in idxs[:n_ones - count]:
            h[idx] = 0
    return project_to_set(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def elapsed():
        return time.time() - t_start
    
    # Get initial solution
    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    
    if h_init is None:
        h_init = np.full(45, 0.5)
    
    n_init = len(h_init)
    h_init = project_to_set(h_init, n_init / 2.0)
    c5_init = compute_c5(h_init)
    
    best_h = h_init.copy()
    best_c5 = c5_init
    best_n = n_init
    
    print(f"Initial: n={n_init}, C5={c5_init:.6f}", flush=True)
    
    # Phase 1: Optimize at initial resolution
    time_left = budget_s - elapsed() - 600
    if time_left > 20:
        h_opt, c5_opt = optimize_smooth(h_init, n_init / 2.0, time_left, beta_init=40.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_init
            print(f"Phase 1 n={n_init}: C5={c5_opt:.6f}", flush=True)
    
    # Phase 2: Try binary at same resolution (only if n is even)
    if best_n % 2 == 0 and elapsed() < budget_s - 500:
        h_bin = push_to_binary(best_h, best_n / 2.0)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            # Local search
            time_left = budget_s - elapsed() - 500
            if time_left > 5:
                h_bin2, c5_bin2 = binary_local_search(h_bin, min(time_left, 60))
                if c5_bin2 < best_c5:
                    best_c5 = c5_bin2
                    best_h = h_bin2.copy()
            print(f"Phase 2 binary n={best_n}: C5={best_c5:.6f}", flush=True)
    
    # Phase 3: Multi-resolution optimization
    for target_n in [90, 180, 360]:
        if best_n > target_n:
            continue
        if elapsed() > budget_s - 200:
            break
        if target_n % 2 != 0:
            continue
        
        time_left = budget_s - elapsed() - 150
        if time_left < 40:
            break
        
        # Upscale
        if best_n != target_n:
            factor = target_n // best_n
            if factor * best_n != target_n:
                # Interpolated upscaling
                h_old = best_h
                h_new = np.zeros(target_n)
                for i in range(target_n):
                    pos_old = i * best_n / target_n
                    idx_lo = int(pos_old)
                    idx_hi = min(idx_lo + 1, best_n - 1)
                    frac = pos_old - idx_lo
                    h_new[i] = (1 - frac) * h_old[idx_lo] + frac * h_old[idx_hi]
            else:
                h_new = upscale(best_h, factor)
        else:
            h_new = best_h.copy()
        
        h_new = project_to_set(h_new, target_n / 2.0)
        
        # Optimize
        t_opt = time_left * 0.55
        h_opt, c5_opt = optimize_smooth(h_new, target_n / 2.0, t_opt, beta_init=30.0, lr_init=0.004)
        
        # Try binary
        h_bin_new = push_to_binary(h_opt, target_n / 2.0)
        c5_bin_new = compute_c5(h_bin_new)
        if c5_bin_new < c5_opt:
            c5_opt = c5_bin_new
            h_opt = h_bin_new
            # Binary local search
            t_bs = min(budget_s - elapsed() - 150, 45)
            if t_bs > 3:
                h_opt, c5_opt = binary_local_search(h_opt, t_bs)
        
        # Further refine
        t_refine = budget_s - elapsed() - 150
        if t_refine > 20:
            h_opt2, c5_opt2 = optimize_smooth(h_opt, target_n / 2.0, t_refine, beta_init=80.0, lr_init=0.003)
            if c5_opt2 < c5_opt:
                c5_opt = c5_opt2
                h_opt = h_opt2
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = target_n
            print(f"Phase 3 n={target_n}: C5={c5_opt:.6f}", flush=True)
    
    # Phase 4: Try structured initializations at best resolution
    if elapsed() < budget_s - 100:
        n_try = best_n if best_n % 2 == 0 else best_n + 1
        if n_try > 720:
            n_try = 360
        for s in range(5):
            if elapsed() > budget_s - 50:
                break
            h_struct = structured_init(n_try, seed=seed + 100 + s)
            t_sub = min(budget_s - elapsed() - 50, 40)
            if t_sub < 10:
                break
            h_opt, c5_opt = optimize_smooth(h_struct, n_try / 2.0, t_sub, beta_init=50.0)
            if n_try % 2 == 0:
                h_b = push_to_binary(h_opt, n_try / 2.0)
                c5_b = compute_c5(h_b)
                if c5_b < c5_opt:
                    c5_opt = c5_b
                    h_opt = h_b
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try
                print(f"Structured {s} n={n_try}: C5={c5_opt:.6f}", flush=True)
    
    # Phase 5: Random perturbations around best
    if elapsed() < budget_s - 40:
        n_r = best_n
        n_attempts = max(1, int((budget_s - elapsed() - 40) / 12))
        for r in range(n_attempts):
            if elapsed() > budget_s - 30:
                break
            h_pert = best_h + 0.05 * rng.randn(best_n)
            h_pert = project_to_set(h_pert, n_r / 2.0)
            t_sub = min(budget_s - elapsed() - 30, 12)
            if t_sub < 3:
                break
            h_opt, c5_opt = optimize_smooth(h_pert, n_r / 2.0, t_sub, beta_init=60.0, lr_init=0.003)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                print(f"Perturb {r}: C5={c5_opt:.6f}", flush=True)
    
    # Final: one more polish
    if elapsed() < budget_s - 10:
        t_final = budget_s - elapsed() - 5
        h_pol, c5_pol = optimize_smooth(best_h, best_n / 2.0, t_final, beta_init=200.0, lr_init=0.001)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_set(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h)
    
    # Safety: if somehow worse than initial, use initial
    if final_c5 > c5_init + 1e-9:
        best_h = h_init.copy()
        final_c5 = c5_init
        best_n = n_init
        print(f"Fall back to initial: C5={final_c5:.6f}", flush=True)
    else:
        print(f"FINAL: n={best_n}, C5={final_c5:.6f}", flush=True)
    
    return (best_h, final_c5, best_n)
