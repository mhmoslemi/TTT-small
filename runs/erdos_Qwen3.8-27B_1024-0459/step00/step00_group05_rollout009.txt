    return h

def _c5_tup(h):
    return c5_obj_and_grad(h, _GLOBAL_N)

def _sum_eq(h):
    return float(np.sum(h)) - _GLOBAL_N / 2.0

def _sum_jac(h):
    return np.ones(_GLOBAL_N)

def random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def binary_local_search(h, n, rng, max_iters=300):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    zeros = list(np.where(best == 0.0)[0])
    ones = list(np.where(best == 1.0)[0])
    temp = 0.1
    for it in range(max_iters):
        temp = max(0.001, temp * 0.97)
        improved = False
        n_tries = 30
        for _ in range(n_tries):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            if i == j:
                continue
            best[i], best[j] = best[j], best[i]
            c5 = compute_c5(best, n)
            delta = c5 - best_c5
            if delta < 0 or rng.rand() < np.exp(-delta / temp):
                if delta < 0:
                    best_c5 = c5
                    improved = True
                zeros = list(np.where(best == 0.0)[0])
                ones = list(np.where(best == 1.0)[0])
            else:
                best[i], best[j] = best[j], best[i]
        if not improved and temp < 0.01:
            break
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _GLOBAL_N
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.full(70, 0.5)

    n_init = len(h0)
    _GLOBAL_N = n_init
    h0 = project_feasible(h0, n_init)
    best_c5 = compute_c5(h0, n_init)
    best_h = h0.copy()

    constraints = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    bounds = [(0.0, 1.0)] * n_init

    # Phase 1: SLSQP from initial
    result = optimize.minimize(
        _c5_tup, h0, method='SLSQP',
        bounds=bounds, constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-14}
    )
    c5_val = compute_c5(result.x, n_init)
    if c5_val < best_c5:
        best_h = project_feasible(result.x, n_init)
        best_c5 = c5_val

    # Phase 2: SLSQP from multiple perturbations
    for i in range(60):
        if time.time() - t0 > budget_s * 0.40:
            break
        if i % 4 == 0:
            h_rand = best_h + rng.normal(0, 0.12, n_init)
        elif i % 4 == 1:
            h_rand = best_h + rng.uniform(-0.2, 0.2, n_init)
        elif i % 4 == 2:
            h_rand = rng.uniform(0, 1, n_init)
        else:
            h_rand = random_binary(n_init, rng)
        h_rand = project_feasible(h_rand, n_init)
        result = optimize.minimize(
            _c5_tup, h_rand, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': 400, 'ftol': 1e-14}
        )
        c5_val = compute_c5(result.x, n_init)
        if c5_val < best_c5:
            best_h = project_feasible(result.x, n_init)
            best_c5 = c5_val

    # Phase 3: Binary simulated annealing
    if time.time() - t0 < budget_s * 0.55:
        n_trials = 80
        for trial in range(n_trials):
            if time.time() - t0 > budget_s * 0.65:
                break
            h_bin = random_binary(n_init, rng)
            h_bin, c5_bin = binary_local_search(h_bin, n_init, rng, max_iters=200)
            if c5_bin < best_c5:
                best_h = h_bin.copy()
                best_c5 = c5_bin

    # Phase 4: SLSQP from best binary / current best with small noise
    if time.time() - t0 < budget_s * 0.35:
        for attempt in range(10):
            if time.time() - t0 > budget_s * 0.80:
                break
            h_start = best_h + rng.normal(0, 0.03, n_init)
            h_start = project_feasible(h_start, n_init)
            result = optimize.minimize(
                _c5_tup, h_start, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-15}
            )
            c5_val = compute_c5(result.x, n_init)
            if c5_val < best_c5:
                best_h = project_feasible(result.x, n_init)
                best_c5 = c5_val

    # Phase 5: Try doubled resolution
    if time.time() - t0 < budget_s * 0.30:
        n_fine = n_init * 2
        _GLOBAL_N = n_fine
        h_fine = np.repeat(best_h, 2)
        h_fine = project_feasible(h_fine, n_fine)
        constraints_fine = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
        bounds_fine = [(0.0, 1.0)] * n_fine
        result = optimize.minimize(
            _c5_tup, h_fine, method='SLSQP',
            bounds=bounds_fine, constraints=constraints_fine,
            options={'maxiter': 500, 'ftol': 1e-13}
        )
        c5_val = compute_c5(result.x, n_fine)
        if c5_val < best_c5:
            best_h = project_feasible(result.x, n_fine)
            best_c5 = c5_val
            n_init = n_fine

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
