def grad_correlation_k(h, k):
    """Gradient of corr_k with respect to h.
    corr_k = sum_i h[i]*(1-h[i+k]) for valid i (0 <= i, i+k < n)
    """
    n = len(h)
    g = np.zeros(n)
    # For k >= 0: corr_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
    # d/dh[j] = (1-h[j+k]) [if j+k < n] - h[j-k] [if j >= k]
    if k >= 0:
        for i in range(n - k):
            g[i] += (1.0 - h[i+k])
            g[i+k] -= h[i]
    else:
        # corr_{-k} = sum_{i=-k}^{n-1} h[i]*(1-h[i+k]) = sum_{i=0}^{n-1+k} h[i+k]*... 
        # Actually for negative k: corr_{-|k|} = sum_{i=|k|}^{n-1} h[i]*(1-h[i-|k|])
        ak = -k
        for i in range(ak, n):
            g[i] += (1.0 - h[i-ak])
            g[i-ak] -= h[i]
    return g

def subgradient_step(h, lr, n):
    """One subgradient step on max_k corr_k."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = np.argmax(corr)  # index in full correlation
    # Convert to shift: shift = active_k - (n-1)
    k = active_k - (n - 1)
    g = grad_correlation_k(h, k)
    h_new = h - lr * g
    # Project to [0,1]^n with sum = n/2
    h_new = project_onto_feasible(h_new, n)
    return h_new, g

def project_onto_feasible(x, n):

    target = n / 2.0

def compute_c5(h):
    """Compute C5 = max_k correlate(h, 1-h, mode='full') * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    """Project x onto {h in [0,1]^n : sum(h) = n/2}."""
    target = n / 2.0
    lo, hi = -2.0, 2.0

        if np.sum(np.clip(x + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid

    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    eval_count = 0
    n_iter = len(ones) * len(zeros)
    
    for it in range(max_evals):
        if time.time() - t_start > 500:  # internal time limit
            break
        if eval_count > max_evals:
            break
        
        # Temperature
        frac = it / max_evals
        temp = t_init * (t_final / t_init) ** frac
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        ones.remove(i)
        zeros.remove(j)
        ones.append(j)
        zeros.append(i)
        
        new_c5 = compute_c5(h)
        eval_count += 1
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:

            delta = new_c5 - current_c5
            if delta > 0 and rng.random() < np.exp(-delta / max(temp, 1e-10)):
                current_c5 = new_c5

                h[j] = 0
                ones.remove(j)
                zeros.remove(i)

                zeros.append(j)

            break
        
        # Find active k
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argmax(corr)
        k = k_idx - (n - 1)
        
        # Gradient of corr_k
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        else:
            ak = -k
            g[ak:] += (1.0 - h[:n-ak])
            g[:n-ak] -= h[ak:]
        
        # Update
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Try smaller step
            h_new = h - lr * 0.1 * g
            h_new = project_feasible(h_new, n)
            new_c5 = compute_c5(h_new)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            lr *= 0.95
        lr *= 0.9999
    
    return best_h, best_c5

def fine_local_search(h, n, max_evals=200000, seed=42):
    """Fine local search with small perturbations."""
    rng = np.random.RandomState(seed)
    t_start = time.time()
    target = n / 2.0
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    for it in range(max_evals):
        if time.time() - t_start > 200:
            break
        
        # Pick two random indices and transfer small amount
        i, j = rng.choice(n, 2, replace=False)
        delta = rng.random() * 0.02
        
        if h[i] - delta < 0 or h[j] + delta > 1:
            continue
        
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5 - 1e-12:
            h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return best_h, best_c5
