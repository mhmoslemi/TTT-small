from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corrs) * dx)

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_c5(h, p=4):
    """Smooth approximation to C5 using p-norm."""
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)
    return float((np.sum(np.power(corrs, p)) ** (1.0/p)) * dx)

def smooth_c5_grad(h, p=4):
    """Gradient of smooth C5."""
    n = len(h)
    dx = 2.0 / n
    corrs = compute_corrs(h)  # length 2n-1
    
    pt_norm = (np.sum(np.power(corrs, p))) ** (1.0/p)
    
    # d(pt_norm)/d(corrs[m]) = corrs[m]^(p-1) / pt_norm^(p-1)
    weights = np.power(corrs, p-1) / (pt_norm ** (p-1))
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        l = m - (n-1)
        w = weights[m] * dx
        if abs(w) < 1e-12:
            continue
        if l == 0:
            grad += w * (1 - 2*h)
        elif l > 0:
            grad[0:n-l] += w * (1 - h[l:n])
            grad[l:n] -= w * h[0:n-l]
        else:
            p_abs = -l
            grad[p_abs:n] += w * (1 - h[0:n-p_abs])
            grad[0:n-p_abs] -= w * h[p_abs:n]
    
    return grad

def project_onto_feasible(h, n):
    """Project h onto {0≤h≤1, sum=h n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    current_sum = np.sum(h)
    diff = target - current_sum
    
    if abs(diff) < 1e-12:
        return h
    
    if diff > 0:
        # Need to increase sum by diff
        # Increase elements that have room
        room = 1.0 - h
        # Distribute proportionally to room
        total_room = np.sum(room)
        if total_room < diff:
            return np.ones(n)  # Shouldn't happen
        h += diff * room / total_room
        h = np.minimum(h, 1.0)
        # Fix any rounding
        remaining = target - np.sum(h)
        if remaining > 1e-10:
            idx = np.argmax(1 - h)
            h[idx] = min(1.0, h[idx] + remaining)
    else:
        # Need to decrease sum by |diff|
        space = h
        total_space = np.sum(space)
        if total_space < -diff:
            return np.zeros(n)
        h += diff * space / total_space
        h = np.maximum(h, 0.0)
        remaining = np.sum(h) - target
        if remaining > 1e-10:
            idx = np.argmin(h)
            h[idx] = max(0.0, h[idx] - remaining)
    
    return h

Wait, the projection is tricky. The constraint set is {h : 0≤h≤1, sum(h)=n/2}. This is a "sandwich" of a hyperplane and a box. The projection is:
1. First project onto the box: clip to [0,1]
2. Then project onto the hyperplane sum=n/2 (while staying in the box)

The projection onto the hyperplane while staying in the box is:
- If sum > n/2: need to decrease. Redistribute the excess among elements > 0, weighted by their current value (or uniformly).
- If sum < n/2: need to increase. Redistribute among elements < 1.

A cleaner approach: use water-filling.
- To increase: start by adding uniformly to all elements < 1. If some hit 1, exclude them and add uniformly to the rest.
- To decrease: start by subtracting uniformly from all elements > 0. If some hit 0, exclude them and subtract from the rest.

1. Clip to [0,1]
2. Compute the error in sum
3. Add/subtract the error/n to all elements (or to elements with room)
4. Clip again
5. If still off, do one more correction

This might not be a perfect projection but it's good enough for optimization.
