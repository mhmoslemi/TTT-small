import time

def project_onto_constraint(h, target_sum):
    n = len(h)
    if target_sum <= 1e-10:
        return np.zeros(n, dtype=np.float64)
    if target_sum >= n - 1e-10:
        return np.ones(n, dtype=np.float64)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def compute_gradient_smoothed(h, T=30.0):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    scaled = T * conv * dx
    sm = np.max(scaled)
    exp_s = np.exp(scaled - sm)
    Z = np.sum(exp_s)
    w = exp_s / Z
    # Gradient = dx * sum_k w[k] * grad_conv[k]
    grad = np.zeros(n, dtype=np.float64)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        offset = k - n + 1
        # Term 1: (1-h[j+offset]) for j in [max(0,-offset), min(n, n-offset))
        t1_lo = max(0, -offset)
        t1_hi = min(n, n - offset)
        if t1_lo < t1_hi:
            grad[t1_lo:t1_hi] += wk * (1.0 - h[t1_lo + offset:t1_hi + offset])
        # Term 2: -h[j-offset] for j in [max(0,offset), min(n, n+offset))
        t2_lo = max(0, offset)
        t2_hi = min(n, n + offset)
        if t2_lo < t2_hi:
            grad[t2_lo:t2_hi] -= wk * h[t2_lo - offset:t2_hi - offset]
    grad *= dx
    c5_smooth = (1.0 / T) * (np.log(Z) + sm)
    return c5_smooth, grad
