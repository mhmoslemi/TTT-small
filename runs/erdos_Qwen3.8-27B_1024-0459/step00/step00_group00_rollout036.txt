import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_gradient(h, T=50.0):
    """Compute f(h) and gradient using soft-max approximation of max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx  # the actual integrals
    
    # Soft max
    m = np.max(c)
    exp_vals = np.exp(T * (c - m))
    sum_exp = np.sum(exp_vals)
    f = m + np.log(sum_exp) / T
    
    # Weights
    p = exp_vals / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(len(corr)):
        pi = p[idx]
        if pi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[0:n-k] += pi * (1.0 - h[k:n])
            grad[k:n] -= pi * h[0:n-k]
        else:
            m_val = -k
            grad[m_val:n] += pi * (1.0 - h[0:n-m_val])
            grad[0:n-m_val] -= pi * h[m_val:n]
    
    grad *= dx
    return f, grad

def sum_constraint(h):
    n = len(h)
    return np.sum(h) - n / 2.0

def sum_constraint_jacobian(h):
    n = len(h)
    return np.ones(n)

def optimize_at_resolution(h0, n, budget_remaining, T_start=10.0, T_end=100.0):
    """Optimize h at given resolution using annealed soft-max."""
    dx = 2.0 / n
    h = h0.copy()
    h = np.clip(h, 0, 1)
    # Ensure sum constraint approximately
    h = h - (np.sum(h) - n/2.0) / n
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    constraints = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jacobian}]
    bounds = [(0, 1)] * n
    
    # Anneal T
    n_steps = 5
    Ts = np.geomspace(T_start, T_end, n_steps)
    
    for T in Ts:
        if time.time() > budget_remaining:
            break
        
        def obj(h):
            f, _ = c5_gradient(h, T=T)
            return f
        
        def obj_grad(h):
            f, g = c5_gradient(h, T=T)
            return f, g
        
        try:
            result = minimize(
                obj, h, method='SLSQP',
                jac=obj_grad,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception as e:
            pass
    
    return best_h, best_c5

import time

def soft_max_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exp_vals = np.exp(T * (c - m))
    sum_exp = np.sum(exp_vals)
    f = m + np.log(sum_exp) / T
    p = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        pi = p[idx]
        if pi < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[0:n-k] += pi * (1.0 - h[k:n])
                grad[k:n] -= pi * h[0:n-k]
        else:
            m_val = -k
            if n - m_val > 0:
                grad[m_val:n] += pi * (1.0 - h[0:n-m_val])
                grad[0:n-m_val] -= pi * h[m_val:n]
    grad *= dx
    return f, grad

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0, 1)
    # Adjust sum
    diff = np.sum(h) - n / 2.0
    if diff > 0:
        # Need to decrease sum; reduce elements that are > 0
        mask = h > 0
        if np.sum(mask) > 0:
            h[mask] -= diff / np.sum(mask)
    else:
        # Need to increase sum; increase elements that are < 1
        mask = h < 1
        if np.sum(mask) > 0:
            h[mask] -= diff / np.sum(mask)  # diff is negative, so this increases
    return np.clip(h, 0, 1)

def optimize_gradient_descent(h0, n, max_iter=1000, lr_start=0.01, T_start=10, T_end=500):
    """Annealed soft-max gradient descent with projection."""
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        T = T_start * (T_end / T_start) ** (it / max_iter)
        
        f, grad = soft_max_gradient(h, T)
        
        # Gradient step
        h_new = h - lr_start * grad
        
        # Project to feasible set
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

    one_h = 1.0 - h

    c = corr * dx

    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max_f_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = T * (c - m)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    f = m + np.log(sum_exp) / T
    p = exp_vals / sum_exp
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        pi = p[idx]
        if pi < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[:nk] += pi * one_h[k:]
                grad[k:] -= pi * h[:nk]
        else:
            mk = -k
            nm = n - mk
            if nm > 0:
                grad[mk:] += pi * one_h[:nm]
                grad[:nm] -= pi * h[mk:]
    grad *= dx
    return f, grad

def sum_eq(h, n):
    return np.sum(h) - n / 2.0

def sum_eq_jac(h, n):
    return np.ones(n)

def project_feasible(h, n):

    diff = np.sum(h) - n / 2.0
    if diff > 1e-12:
        mask = h > 0
        cnt = np.sum(mask)

            h[mask] -= diff / cnt
    elif diff < -1e-12:
        mask = h < 1
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] -= diff / cnt
    return np.clip(h, 0.0, 1.0)

        return soft_max_f_and_grad(h, T)

    result = minimize(obj, h0, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
    h_new = np.clip(result.x, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new

    h = h0.astype(float).copy()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    Ts = np.geomspace(T_lo, T_hi, n_T)
    
    for i, T in enumerate(Ts):
        if time.time() > time_limit:
            break
        mi = min(maxiter * 2, 300) if i < n_T // 2 else maxiter
        try:
            h_new = run_slsqp(h, n, T, mi)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5

def coordinate_descent_pass(h, n, iterations=200):
    """Local refinement: for each active shift, try adjusting one coordinate."""
    h = h.astype(float).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(iterations):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
