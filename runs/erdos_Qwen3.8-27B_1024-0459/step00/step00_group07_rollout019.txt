def project_onto_constraint(h, n):
    target = n / 2.0

        x = np.clip(h - mu, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mu
        else:
            hi = mu
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_subgradient(h, n, top_k=8):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    
    # Get top_k correlations
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    vals = corr[top_idx]
    max_val = vals.max()
    
    # Weight: exponential of (val - max_val) / temperature
    temps = max_val * 0.01
    weights = np.exp((vals - max_val) / temps)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            kp = -k
            grad[:n-kp] += w * (1.0 - h[kp:])
            grad[kp:] -= w * h[:n-kp]
    
    grad *= dx
    return c5, grad

Total ones: sum over all valid m of [m ∈ S] = n/2.

The "overlap" at lag k is: D_k = |{m ∈ S : m+k ∈ S, both valid}| = |S ∩ {valid}| - C_k

And C_k = |S ∩ {valid}| - D_k.

The objective is max_k C_k.

Sum_k D_k = sum_k |{m ∈ S : m+k ∈ S}| = sum_{m∈S} (number of k such that m+k ∈ S and both valid)
