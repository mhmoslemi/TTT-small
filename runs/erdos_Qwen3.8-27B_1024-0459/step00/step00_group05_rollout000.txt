import numpy as np
import time
import math
from scipy.optimize import minimize

_RUN_DEADLINE = None
_GLOBAL_DEADLINE = None


class BudgetError(Exception):

def check_deadline():
    if _RUN_DEADLINE is not None:

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0 or (not np.all(np.isfinite(h))):
        return 1e6
    h = np.clip(h, 0.0, 1.0)
    conv = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(conv) * (2.0 / n))
    if not math.isfinite(c):
        return 1e6
    return c


def is_valid_candidate(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(h.min()) < -1e-7 or float(h.max()) > 1.0 + 1e-7:
        return False
    if abs(float(h.sum()) - 0.5 * n) > 1e-5:
        return False
    return True


def _adjust_residual(h, target):
    h = np.asarray(h, dtype=float).copy()
    guard = 0
    diff = target - float(h.sum())
    n = h.size
    while abs(diff) > 1e-12 and guard <= n + 5:
        guard += 1
        if diff > 0.0:
            mask = h < 1.0
            if not np.any(mask):

            cnt = int(np.count_nonzero(mask))
            per = diff / float(cnt)

        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            tmp = np.zeros(n, dtype=float)
            m = min(n, h.size)
            if m > 0:
                tmp[:m] = h[:m]
            h = tmp
    if not np.all(np.isfinite(h)):
        zeros = np.zeros(2 * n - 1, dtype=float)
        zeros[0] = 1e6
        return zeros, 1e6
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, h, mode="full")
    ac = corr[n - 1:n - 1 + n]
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    scores = np.empty(2 * n - 1, dtype=float)
    scores[0] = total - ac[0]
    for d in range(1, n):
        scores[2 * d - 1] = prefix[n - d] - ac[d]
        scores[2 * d] = total - prefix[d] - ac[d]
    m = float(np.max(scores))
    if not math.isfinite(m):
        scores[:] = 1e6
        m = 1e6
    return scores, m


def slsqp_objective(x, alpha):
    check_deadline()
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return 1e6
    if not np.all(np.isfinite(x)):
        return 1e6
    h = np.clip(x, 0.0, 1.0)
    scores, m = score_max(h, n)
    if alpha > 1e-14:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if not math.isfinite(A) or A <= 0.0:
            val = m
        else:
            val = m + alpha * math.log(A)
    else:
        val = m
    if not math.isfinite(val):
        return 1e6
    return float(val) * (2.0 / n)


def add_lag_grad(grad, idx, coef, h, n):
    if coef == 0.0:
        return
    if idx == 0:
        grad += coef * (1.0 - 2.0 * h)
        return
    if idx % 2 == 1:
        d = (idx + 1) // 2
    else:
        d = idx // 2
    m = n - d
    if m <= 0:
        return
    if idx % 2 == 1:
        grad[:m] += coef * (1.0 - h[d:])
        grad[d:] += coef * (-h[:m])
    else:
        grad[:m] += coef * (-h[d:])
        grad[d:] += coef * (1.0 - h[:m])


def slsqp_gradient(x, alpha):
    check_deadline()
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return np.zeros(0, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    h = np.clip(x, 0.0, 1.0)
    scores, m = score_max(h, n)
    grad = np.zeros(n, dtype=float)
    if alpha <= 1e-14:
        tol = max(1e-8, 1e-7 * max(1.0, m))
        active = []
        for i, s in enumerate(scores):
            if s >= m - tol:
                active.append(i)
        if not active:
            active = [int(np.argmax(scores))]
        coef = 1.0 / float(len(active))
        for i in active:
            add_lag_grad(grad, i, coef, h, n)
    else:
        w = np.exp((scores - m) / alpha)
        A = float(np.sum(w))
        if math.isfinite(A) and A > 0.0:
            thresh = 1e-10
            for i, wi in enumerate(w):
                coef = wi / A
                if coef >= thresh:
                    add_lag_grad(grad, i, coef, h, n)
    grad *= 2.0 / n
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=float)
    return grad


def constraint_eq(x, *args):
    n = max(1, int(x.size))
    return (float(np.sum(x)) - 0.5 * float(n)) / float(n)


def constraint_jac(x, *args):
    n = max(1, int(x.size))
    return np.full((1, n), 1.0 / float(n), dtype=float)


class OptimizerState:
    def __init__(self, h, c, n):
        self.h = np.array(h, dtype=float)
        self.c = float(c)
        self.n = int(n)

    def update(self, h, c, n):
        if h is None:
            return
        h = np.asarray(h, dtype=float)
        if h.ndim != 1 or h.size == 0:
            return
        if n is None:
            n = h.size
        n = int(n)
        if h.size != n:
            if h.size > n:
                h = h[:n]
            else:
                return
        try:
            h2 = as_feasible(h, n)
            c2 = compute_c5(h2)
        except Exception:
            return
        if math.isfinite(c2) and c2 < self.c - 1e-15:
            self.h = h2
            self.c = float(c2)
            self.n = n


def get_parent():
    try:
        p = np.asarray(initial_h_values, dtype=float)
        if p.ndim != 1 or p.size < 2:
            return None
        return p.copy()
    except Exception:
        return None


def upsample_linear(parent, n):
    n0 = int(parent.size)
    if n == n0:
        return parent.copy()
    if n0 <= 0:
        return np.full(n, 0.5)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    return np.interp(x, x0, parent.astype(float)).astype(float)


def smooth1(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    out = h.copy()
    if n == 0:
        return out
    if n == 1:
        return out
    if n == 2:
        return np.array([0.75 * h[0] + 0.25 * h[1],
                         0.25 * h[0] + 0.75 * h[1]], dtype=float)
    out[0] = 0.75 * h[0] + 0.25 * h[1]
    out[-1] = 0.25 * h[-2] + 0.75 * h[-1]
    out[1:-1] = 0.25 * h[:-2] + 0.5 * h[1:-1] + 0.25 * h[2:]
    return out


def synthetic_h(n, rng):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    best_h = as_feasible(np.full(n, 0.5), n)
    best_c = compute_c5(best_h)
    for f in (1, 2, 3, 4):
        for ph in (0.0, 0.5 * math.pi):
            h = 0.5 + 0.5 * np.cos((math.pi * f * x) + ph)
            h = as_feasible(h, n)
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h
    hb = make_binary_area(None, n, rng).astype(float)
    h = smooth1(hb)
    h = as_feasible(h, n)
    c = compute_c5(h)
    if c < best_c:
        best_c = c
        best_h = h
    return best_h


def cand_key(item):
    return item[1]


def is_same_candidate(a, b):
    if a.size != b.size:
        return False
    return bool(np.allclose(a, b, rtol=1e-7, atol=1e-7))


def candidate_not_duplicate(cands, h):
    for hh, cc in cands:
        if is_same_candidate(hh, h):
            return False
    return True


def make_candidates(n, parent, starts, rng, state, limit=3):
    n = int(n)
    cands = []
    if n in starts:
        for h in starts[n]:
            try:
                h2 = as_feasible(h, n)
                c2 = compute_c5(h2)
                if math.isfinite(c2) and candidate_not_duplicate(cands, h2):
                    cands.append((h2, c2))
            except Exception:
                pass
    if parent is not None:
        n0 = int(parent.size)
        if n != n0:
            if n0 > 0 and n % n0 == 0:
                r = n // n0
                try:
                    h = np.repeat(parent, r)
                    h = as_feasible(h, n)
                    c = compute_c5(h)
                    if math.isfinite(c) and candidate_not_duplicate(cands, h):
                        cands.append((h, c))
                except Exception:
                    pass
            try:
                h = upsample_linear(parent, n)
                h = as_feasible(h, n)
                c = compute_c5(h)
                if math.isfinite(c) and candidate_not_duplicate(cands, h):
                    cands.append((h, c))
            except Exception:
                pass
    if state is not None and state.h is not None:
        hn = int(state.h.size)
        try:
            if hn == n:
                h = as_feasible(state.h, n)
                c = compute_c5(h)
            else:
                h = upsample_linear(state.h, n)
                h = as_feasible(h, n)
                c = compute_c5(h)
            if math.isfinite(c) and candidate_not_duplicate(cands, h):
                cands.append((h, c))
        except Exception:
            pass
    if parent is None or len(cands) < 2:
        try:
            h = synthetic_h(n, rng)
            c = compute_c5(h)
            if math.isfinite(c) and candidate_not_duplicate(cands, h):
                cands.append((h, c))
        except Exception:
            pass
    if len(cands) == 0:
        try:
            h = as_feasible(np.full(n, 0.5), n)
            c = compute_c5(h)
            cands.append((h, c))
        except Exception:
            pass
    try:
        cands.sort(key=cand_key)
    except Exception:
        pass
    if limit > 0 and len(cands) > limit:
        cands = cands[:limit]
    return cands


def run_continuous(h, n, state, time_limit):
    global _RUN_DEADLINE
    if state is None:
        return
    n = int(n)
    if h is None or h.size != n:
        return
    if float(time_limit) <= 5.0:
        return
    if _RUN_DEADLINE is not None and time.time() + 15.0 >= _RUN_DEADLINE:
        return

    h_start = as_feasible(h, n)
    state.update(h_start, None, n)
    start_t = time.time()

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": constraint_eq, "jac": constraint_jac}]

    alpha = max(0.2, 0.002 * float(n))
    maxiter1 = int(120.0 * (70.0 / float(n)))
    maxiter1 = max(20, min(300, maxiter1))

    stage_deadline = start_t + max(10.0, 0.45 * float(time_limit))
    if _RUN_DEADLINE is not None:
        stage_deadline = min(stage_deadline, _RUN_DEADLINE)
    old_deadline = _RUN_DEADLINE
    _RUN_DEADLINE = stage_deadline
    res1 = None
    try:
        res1 = minimize(
            slsqp_objective,
            h_start,
            args=(alpha,),
            jac=slsqp_gradient,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": maxiter1, "ftol": 1e-9, "disp": 0},
        )
    except BudgetError:
        res1 = None
    except Exception:
        res1 = None
    finally:
        _RUN_DEADLINE = old_deadline

    if res1 is not None and getattr(res1, "x", None) is not None:
        h1 = as_feasible(res1.x, n)
    else:
        h1 = h_start
    state.update(h1, None, n)

    try:
        if state.n == n and state.c <= compute_c5(h1):
            h1 = state.h.copy()
    except Exception:
        h1 = as_feasible(h1, n)

    if _RUN_DEADLINE is None or time.time() + 15.0 < _RUN_DEADLINE:
        maxiter2 = int(260.0 * (70.0 / float(n)))
        maxiter2 = max(30, min(400, maxiter2))
        exact_deadline = start_t + max(20.0, float(time_limit))
        if _RUN_DEADLINE is not None:
            exact_deadline = min(exact_deadline, _RUN_DEADLINE)
        old_deadline = _RUN_DEADLINE
        _RUN_DEADLINE = exact_deadline
        res2 = None
        try:
            res2 = minimize(
                slsqp_objective,
                h1,
                args=(0.0,),
                jac=slsqp_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter2, "ftol": 1e-9, "disp": 0},
            )
        except BudgetError:
            res2 = None
        except Exception:
            res2 = None
        finally:
            _RUN_DEADLINE = old_deadline
        if res2 is not None and getattr(res2, "x", None) is not None:
            state.update(res2.x, None, n)

    state.update(h1, None, n)


def make_binary_area(start, n, rng):
    n = int(n)
    target = n // 2
    h = np.zeros(n, dtype=np.int8)
    if target <= 0:
        return h
    if start is not None and np.size(start) == n:
        vals = np.asarray(start, dtype=float).copy()
        vals = np.nan_to_num(vals, nan=0.0, posinf=1.0, neginf=0.0)
        vals = np.clip(vals, 0.0, 1.0)
        order = np.argsort(vals)
        h[order[target:]] = 1
    else:
        idx = rng.choice(n, size=target, replace=False)
        h[idx] = 1
    return h


def binary_scores(h, n):
    n = int(n)
    scores = np.zeros(2 * n - 1, dtype=float)
    hf = h.astype(float)
    for d in range(1, n):
        scores[2 * d - 1] = float(np.dot(hf[:n - d], 1.0 - hf[d:]))
        scores[2 * d] = float(np.dot(hf[d:], 1.0 - hf[:n - d]))
    return scores


def binary_flip_update(scores, h, idx, new_val, n):
    idx = int(idx)
    new_val = int(new_val)
    i = idx
    for d in range(1, n):
        pidx = 2 * d - 1
        midx = 2 * d
        if i - d >= 0:
            hj = int(h[i - d])
            if new_val == 1:
                scores[pidx] -= hj
                scores[midx] += 1.0 - hj
            else:
                scores[pidx] += hj
                scores[midx] -= 1.0 - hj
        if i + d < n:
            hj = int(h[i + d])
            if new_val == 1:
                scores[pidx] += 1.0 - hj
                scores[midx] -= hj
            else:
                scores[pidx] -= 1.0 - hj
                scores[midx] += hj
    h[i] = new_val


def binary_swap_apply(scores, h, i, j, n):
    i = int(i)
    j = int(j)
    if i == j:
        return False
    hi = int(h[i])
    hj = int(h[j])
    if hi == hj:
        return False
    if hi == 0 and hj == 1:
        binary_flip_update(scores, h, i, 1, n)
        binary_flip_update(scores, h, j, 0, n)
    else:
        binary_flip_update(scores, h, i, 0, n)
        binary_flip_update(scores, h, j, 1, n)
    return True


def propose_binary_swap(h, scores, cur, rng, n):
    num_active = 0.7
    if rng.random() < num_active:
        active = np.flatnonzero(scores >= cur - 1e-9)
        if active.size > 0:
            idx = int(active[int(rng.integers(active.size))])
            if idx != 0:
                if idx % 2 == 1:
                    d = (idx + 1) // 2
                    mask = (h[:n - d] == 1) & (h[d:] == 0)
                else:
                    d = idx // 2
                    mask = (h[:n - d] == 0) & (h[d:] == 1)
                cand = np.nonzero(mask)[0]
                if cand.size > 0:
                    pos = int(cand[int(rng.integers(cand.size))])
                    return pos, pos + d

    zeros = np.flatnonzero(h == 0)
    ones = np.flatnonzero(h == 1)
    if zeros.size == 0 or ones.size == 0:
        return -1, -1
    i = int(zeros[int(rng.integers(zeros.size))])
    j = int(ones[int(rng.integers(ones.size))])
    return i, j


def binary_local_once(start_h, n, state, rng, deadline, max_ops, stall_limit):
    n = int(n)
    h = make_binary_area(start_h, n, rng).astype(np.int8)
    scores = binary_scores(h, n)
    cur = float(np.max(scores))
    best = cur
    best_h = h.copy()
    if state is not None:
        state.update(h.astype(float), best * (2.0 / n), n)

    ops = 0
    stall = 0
    while ops < max_ops and time.time() < deadline:
        ops += 1
        if _RUN_DEADLINE is not None and time.time() > _RUN_DEADLINE:
            break
        i, j = propose_binary_swap(h, scores, cur, rng, n)
        if i < 0:
            stall += 50
            continue
        if not binary_swap_apply(scores, h, i, j, n):
            continue
        new_max = float(np.max(scores))
        if new_max + 1e-9 < cur:
            cur = new_max
            stall = 0
            if cur < best:
                best = cur
                best_h = h.copy()
                if state is not None:
                    state.update(h.astype(float), best * (2.0 / n), n)
        elif new_max <= cur + 1e-9:
            if stall > 20 and rng.random() < 0.07:
                stall += 1
            else:
                binary_swap_apply(scores, h, i, j, n)
                stall += 1
        else:
            binary_swap_apply(scores, h, i, j, n)
            stall += 1
        if stall > stall_limit:
            break

    final_c = cur * (2.0 / n)
    if state is not None:
        state.update(h.astype(float), final_c, n)
    if final_c < best + 1e-12:
        best_h = h.copy()
        best = final_c
    return best_h.astype(float), float(best * (2.0 / n))


def binary_hill_search(start_h, n, state, deadline, time_limit, rng):
    n = int(n)
    if n % 2 != 0:
        return None, 1e9
    stage_deadline = time.time() + max(1.0, float(time_limit))
    if deadline is not None:
        stage_deadline = min(stage_deadline, float(deadline))
    if _RUN_DEADLINE is not None:
        stage_deadline = min(stage_deadline, _RUN_DEADLINE)

    best_h = None
    best_c = 1e9
    max_restarts = 3 if n <= 100 else 2
    max_ops = 120000 if n <= 100 else 40000
    stall_limit = 8000 if n <= 100 else 3000
    restart_time = max(5.0, float(time_limit) / float(max_restarts))

    done = 0
    while done < max_restarts and time.time() < stage_deadline - 2.0:
        if done == 0:
            start = start_h
        else:
            start = None
        rl = time.time() + restart_time
        if rl > stage_deadline:
            rl = stage_deadline
        try:
            h, c = binary_local_once(start, n, state, rng, rl, max_ops, stall_limit)
        except Exception:
            h, c = None, 1e9
        if h is not None and math.isfinite(c) and c < best_c:
            best_c = c
            best_h = h
        done += 1

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _RUN_DEADLINE, _GLOBAL_DEADLINE

    t0 = time.time()
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 1300.0
    if not math.isfinite(budget_s) or budget_s <= 0.0:
        budget_s = 30.0
    reserve = 20.0 if budget_s > 60.0 else max(2.0, 0.1 * budget_s)
    global_deadline = t0 + max(1.0, budget_s - reserve)
    _GLOBAL_DEADLINE = global_deadline
    _RUN_DEADLINE = global_deadline

    rng = np.random.default_rng(seed)

    parent = get_parent()
    if parent is not None and parent.size > 0:
        n0 = int(parent.size)
        h0 = parent.copy()
        c0 = compute_c5(h0)
        valid0 = is_valid_candidate(h0)
    else:
        n0 = 70
        h0 = synthetic_h(n0, rng)
        c0 = compute_c5(h0)
        valid0 = True

    if valid0:
        state = OptimizerState(h0.copy(), c0, n0)
    else:
        hf = as_feasible(h0, n0)
        cf = compute_c5(hf)
        state = OptimizerState(hf, cf, n0)

    initial_h = h0.copy()
    initial_c = c0
    initial_n = n0
    initial_valid = valid0

    starts = {}
    starts[n0] = [h0.copy()]

    trial_ns = []
    for mult in (1.0, 1.5, 2.0, 3.0, 4.0):
        nn = int(round((mult * float(n0) / 2.0)) * 2)
        if nn >= 4 and nn <= 8 * n0:
            if nn not in trial_ns:
                trial_ns.append(nn)
    if n0 not in trial_ns:
        trial_ns.append(n0)

    if n0 % 2 == 0 and _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 90.0:
        try:
            if parent is not None:
                bin_start = upsample_linear(parent, n0)
            else:
                bin_start = state.h
            blim = min(60.0, max(5.0, float(_RUN_DEADLINE - time.time())))
            bh, bc = binary_hill_search(bin_start, n0, state, _RUN_DEADLINE, blim, rng)
            if bh is not None:
                bhf = as_feasible(bh, n0)
                if n0 not in starts:
                    starts[n0] = []
                starts[n0].append(bhf)
        except Exception:
            pass

    if n0 % 2 == 0 and _RUN_DEADLINE is not None and time.time() < _RUN_DEADLINE - 70.0:
        n2 = 2 * n0
        if n2 not in trial_ns and n2 <= 8 * n0:
            trial_ns.append(n2)
        try:
            if parent is not None:
                bin_start2 = upsample_linear(parent, n2)
            else:
                bin_start2 = None
            blim2 = min(40.0, max(5.0, float(_RUN_DEADLINE - time.time())))
            bh2, bc2 = binary_hill_search(bin_start2, n2, state, _RUN_DEADLINE, blim2, rng)
            if bh2 is not None:
                bhf2 = as_feasible(bh2, n2)
                if n2 not in starts:
                    starts[n2] = []
                starts[n2].append(bhf2)
        except Exception:
            pass

    ordered_ns = [n0]
    extra = []
    seen = set([n0])
    for nn in trial_ns:
        if nn != n0 and nn not in seen:
            extra.append(nn)
            seen.add(nn)
    extra.sort()

    for nn in ordered_ns + extra:
        if _RUN_DEADLINE is None or time.time() >= _RUN_DEADLINE - 50.0:
            break
        remaining = float(_RUN_DEADLINE - time.time())
        if remaining < 70.0:
            break
        if nn == n0:
            maxtime = 360.0
        elif nn <= 2 * n0:
            maxtime = 300.0
        else:
            maxtime = 220.0
        maxtime = min(maxtime, max(30.0, remaining - 60.0))
        if maxtime < 35.0:
            break
        cand_limit = 3 if remaining > 800.0 else 2
        cands = make_candidates(nn, parent, starts, rng, state, cand_limit)
        if not cands:
            continue
        k = max(1, len(cands))
        base = max(20.0, maxtime / float(k))
        for h, dd in cands:
            if _RUN_DEADLINE is None or time.time() >= _RUN_DEADLINE - 30.0:
                break
            rem = float(_RUN_DEADLINE - time.time())
            lt = min(base, max(20.0, rem - 30.0))
            if lt <= 20.0:
                break
            try:
                run_continuous(h, nn, state, lt)
            except Exception:
                pass

    if state is not None and state.h is not None:
        h_final = as_feasible(state.h, state.n)
        c_final = compute_c5(h_final)
        n_final = int(state.n)
    else:
        h_final = as_feasible(initial_h, initial_n)
        c_final = compute_c5(h_final)
        n_final = int(initial_n)

    if initial_valid and math.isfinite(initial_c) and initial_c <= c_final + 1e-12:
        h_final = initial_h
        c_final = initial_c
        n_final = initial_n

    if not is_valid_candidate(h_final):
        h_final = as_feasible(h_final, h_final.size)
        c_final = compute_c5(h_final)
        n_final = int(h_final.size)

    return h_final, float(c_final), int(n_final)
