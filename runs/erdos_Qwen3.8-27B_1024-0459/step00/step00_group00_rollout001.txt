def gradient_from_weights(h,w,n):
    x = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    n = int(n)
    c = n - 1
    if n <= 0: return np.zeros(0)
    if n <= 200:
        B = np.convolve(x, w, mode='full')[c:c+n]
        wr = w[::-1]
        A = np.convolve(x, wr, mode='full')[c:c+n]
    else:
        conv_len = n + (2*n - 1) - 1
        m = 1
        while m < conv_len:
            m <<= 1
        fx = np.fft.rfft(x, m)
        fw = np.fft.rfft(w, m)
        B = np.fft.irfft(fx * fw, m)[c:c+n]
        fw_r = np.fft.rfft(w[::-1], m)
        A = np.fft.irfft(fx * fw_r, m)[c:c+n]
    cum = np.empty(2*n, dtype=float)
    cum[0] = 0.0
    cum[1:] = np.cumsum(w)
    j = np.arange(n)
    starts = c - j
    ends = 2*c - j
    D = cum[ends + 1] - cum[starts]
    return D - A - B

import numpy as np
import math
import time
from scipy.optimize import minimize, Bounds, differential_evolution

_CACHE = {}


def _clear_cache():
    _CACHE.clear()


def project_box_sum(h, target):
    x = np.asarray(h, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    target = float(target)

    diff = target - float(np.sum(x))
    if abs(diff) > 1e-7:
        lo = -1.0
        hi = 1.0
        for _ in range(80):
            mid = 0.5 * (lo + hi)
            s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
            if s < target:
                lo = mid
            else:
                hi = mid
        x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(8):
        diff = target - float(np.sum(x))
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            slack = 1.0 - x
            total = float(np.sum(slack))
            if total <= 1e-15:
                break
            add = min(diff, total)
            x += (add / total) * slack
            x = np.clip(x, 0.0, 1.0)
        else:
            mass = x
            total = float(np.sum(mass))
            if total <= 1e-15:
                break
            rem = min(-diff, total)
            x -= (rem / total) * mass
            x = np.clip(x, 0.0, 1.0)

    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        if diff > 0.0:
            idx = int(np.argsort(1.0 - x)[-1])
            x[idx] = min(1.0, x[idx] + diff)
        else:
            idx = int(np.argsort(x)[0])
            x[idx] = max(0.0, x[idx] + diff)
    return x


def compute_c5(h):
    x = np.asarray(h, dtype=float)
    n = int(x.size)
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    return float(np.max(c))


def smooth_lse_values(h, n, dx, tau):
    x = np.asarray(h, dtype=float)
    n = int(n)
    dx = float(dx)
    tau = float(tau)

    if not np.all(np.isfinite(x)):
        w = np.full(2 * n - 1, 1.0 / max(1, 2 * n - 1), dtype=float)
        return 1e100, w

    cached = _CACHE.get("x")
    if cached is not None and _CACHE.get("n") == n and _CACHE.get("dx") == dx and _CACHE.get("tau") == tau:
        if cached is x or np.array_equal(cached, x):
            return _CACHE["F"], _CACHE["w"]

    v = 1.0 - x
    corr = np.correlate(x, v, mode="full") * dx
    z = corr / tau
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(np.sum(e))

    if s <= 0.0 or not math.isfinite(s):
        w = np.zeros_like(corr, dtype=float)
        w[int(np.argmax(corr))] = 1.0
        F = m
    else:
        w = e / s
        F = (m + math.log(s)) * tau

    _CACHE["x"] = x.copy()
    _CACHE["n"] = n
    _CACHE["dx"] = dx
    _CACHE["tau"] = tau
    _CACHE["F"] = F
    _CACHE["w"] = w
    return F, w


def smooth_objective(h, n, dx, tau):
    F, _ = smooth_lse_values(h, n, dx, tau)
    return float(F)


def gradient_from_weights(h, w, n):
    x = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float)

    c = n - 1
    B = np.convolve(x, w, mode="full")[c:c + n]
    A = np.convolve(x, w[::-1], mode="full")[c:c + n]

    cum = np.empty(2 * n, dtype=float)
    cum[0] = 0.0
    cum[1:] = np.cumsum(w)

    j = np.arange(n, dtype=float)
    starts = c - j
    ends = 2 * c - j
    D = cum[ends + 1] - cum[starts]

    return D - A - B


def smooth_gradient(h, n, dx, tau):
    _, w = smooth_lse_values(h, n, dx, tau)
    return gradient_from_weights(h, w, n)


def penalty_objective(h, n, dx, tau, rho):
    F, _ = smooth_lse_values(h, n, dx, tau)
    e = float(np.sum(h) - 0.5 * n)
    return float(F + 0.5 * rho * e * e)


def penalty_gradient(h, n, dx, tau, rho):
    _, w = smooth_lse_values(h, n, dx, tau)
    g = gradient_from_weights(h, w, n)
    e = float(np.sum(h) - 0.5 * n)
    return g + rho * e

def pairwise_polish(h0, n, deadline, rng):
    n = int(n)
    target = 0.5 * float(n)
    x = project_box_sum(h0, target)
    best = x.copy()
    best_c = compute_c5(x)
    if not np.isfinite(best_c):
        return best, float(best_c)

    steps = np.array(
        [0.0, 0.03, -0.03, 0.07, -0.07, 0.12, -0.12, 0.20, -0.20, 0.35, -0.35],
        dtype=float,
    )
    no_improve = 0

    while time.time() < deadline - 0.5:
        i = int(rng.integers(n))
        j = int(rng.integers(n - 1))
        if j >= i:
            j += 1

        a = float(x[i])
        b = float(x[j])
        improved = False

        for s in steps:
            ni = a + float(s)
            if ni < -1e-12 or ni > 1.0 + 1e-12:
                continue
            ni = float(np.clip(ni, 0.0, 1.0))
            nj = b - (ni - a)
            if nj < -1e-12 or nj > 1.0 + 1e-12:

            if np.isfinite(c) and c < best_c - 1e-13:
                best_c = float(c)
                best = x.copy()
                a = ni
                b = nj
                improved = True
            else:
                x[i] = a
                x[j] = b
        x[i] = a
        x[j] = b

        if abs(a - b) > 1e-12:
            x[i] = b
            x[j] = a
            c = compute_c5(x)
            if np.isfinite(c) and c < best_c - 1e-13:
                best_c = float(c)
                best = x.copy()
                a, b = b, a
                improved = True
            else:
                x[i] = a
                x[j] = b
        x[i] = a
        x[j] = b

        if not improved and no_improve > 0:
            s = float(rng.normal(0.0, 0.2))
            ni = a + s
            if 0.0 <= ni <= 1.0:
                nj = b - (ni - a)
                if 0.0 <= nj <= 1.0:
                    x[i] = ni
                    x[j] = nj
                    c = compute_c5(x)
                    if np.isfinite(c) and c < best_c - 1e-13:
                        best_c = float(c)
                        best = x.copy()
                        improved = True
                    else:
                        x[i] = a
                        x[j] = b
            x[i] = a
            x[j] = b

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget_s = float(max(60.0, float(budget_s)))
    end = t0 + budget_s
    rng = np.random.default_rng(seed)
    _clear_cache()

    parent = None
    try:
        p = globals().get("initial_h_values", None)
        if p is not None:
            parent = np.asarray(p, dtype=float)
            if parent.size == 0:
                parent = None
    except Exception:
        parent = None

    best_h = None
    best_c = float("inf")
    best_n = None

    if parent is not None:
        h = project_box_sum(parent, 0.5 * parent.size)
        c = compute_c5(h)
        best_h = h.copy()
        best_c = float(c) if np.isfinite(c) else 1e100
        best_n = int(h.size)
    else:
        h = random_start(45, rng, 0)
        best_h = h.copy()
        best_c = compute_c5(h)
        if not np.isfinite(best_c):
            best_c = 1e100
        best_n = int(h.size)

    plan = [
        (9, 0.02, 3),
        (15, 0.03, 3),
        (21, 0.04, 3),
        (31, 0.05, 3),
        (45, 0.20, 4),
        (89, 0.12, 3),
        (179, 0.12, 3),
        (361, 0.14, 2),
        (513, 0.08, 2),
    ]

    for n, frac, n_rand in plan:
        time_left = budget_s - (time.time() - t0)
        if time_left < 35.0:
            break

        seg_budget = min(frac * budget_s, time_left * 0.90)
        if seg_budget < 18.0:
            break

        starts = []

        if best_h is not None:
            if int(np.asarray(best_h).size) == int(n):
                starts.append(np.asarray(best_h, dtype=float).copy())
            else:
                starts.append(resample_h(best_h, n))

        if parent is not None:
            if int(parent.size) != int(n):
                starts.append(resample_h(parent, n))

        if n in (15, 21, 31, 45) and time_left > 70.0:
            try:
                hde = de_search(n, int(seed * 100000 + n * 7 + 11), maxiter=10, popsize=14)
                if hde is not None:
                    starts.append(hde)
            except Exception:
                pass

        for i in range(int(n_rand)):
            starts.append(random_start(n, rng, i % 3))

        if not starts:
            starts.append(random_start(n, rng, 0))

        candidate_budget = max(6.0, seg_budget / float(max(1, len(starts))))

        for sh in starts:
            rem = budget_s - (time.time() - t0)
            if rem < 30.0:
                break
            candidate_deadline = time.time() + candidate_budget
            try:
                h, c = optimize_until_deadline(sh, n, candidate_deadline, rng)
            except Exception:
                h = project_box_sum(sh, 0.5 * int(n))
                c = compute_c5(h)
            if np.isfinite(c) and c < best_c:
                best_c = float(c)
                best_h = h.copy()
                best_n = int(h.size)

    rem = budget_s - (time.time() - t0)
    if best_h is not None and rem > 120.0:
        fn = int(best_n)
        if fn <= 250 or rem > 300.0:
            deadline = min(end - 40.0, time.time() + max(40.0, min(200.0, rem - 80.0)))
            try:
                h, c = slsqp_refine(best_h, fn, deadline)
            except Exception:
                h, c = best_h, best_c
            if np.isfinite(c) and c < best_c:
                best_c = float(c)
                best_h = h.copy()
                best_n = fn

    rem = budget_s - (time.time() - t0)
    if best_h is not None and rem > 60.0:
        deadline = min(end - 20.0, time.time() + max(20.0, min(160.0, rem - 40.0)))
        try:
            h, c = pairwise_polish(best_h, best_n, deadline, rng)
        except Exception:
            h, c = best_h, best_c
        if np.isfinite(c) and c < best_c:
            best_c = float(c)
            best_h = h.copy()
            best_n = int(h.size)

    final_h = project_box_sum(best_h, 0.5 * int(best_h.size))
    final_c = compute_c5(final_h)
    return final_h, float(final_c), int(final_h.size)
