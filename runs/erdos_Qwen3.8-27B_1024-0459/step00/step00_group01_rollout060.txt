import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def project_onto_feasible(h, n):
    """Project h onto the feasible set: 0 <= h <= 1, sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute evenly
        h += diff / n
        h = np.clip(h, 0, 1)
        # If still not exact, do iterative adjustment
        for _ in range(100):
            diff = target - np.sum(h)
            if abs(diff) < 1e-10:
                break
            mask = (h < 0.999) if diff > 0 else (h > 0.001)
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

def subgradient_c5(h, n):
    """Compute subgradient of C5 at h, and the value of C5."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    c5 = np.max(corr) * dx
    # Find all k where corr[k] is close to max
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-6)[0]
    
    # Compute gradient for each active k
    # corr[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
    # d/dh[j] of this = sum over valid i: d/dh[j] [h[i+k]*(1-h[i])]
    #   j = i+k: (1-h[i]) = (1-h[j-k])
    #   j = i: -h[i+k] = -h[j+k]
    
    grad = np.zeros(n)
    for k_idx in active:
        shift = k_idx - (n-1)  # This is the actual shift
        if shift == 0:

            # d/dh[j] = (1-h[j]) - h[j] = 1 - 2*h[j]
            grad += (1 - 2*h)

            # d/dh[j]: j appears as i+shift (i=j-shift, need j-shift>=0 and i<=n-1-shift)

            for j in range(n):

                    g += (1 - h[j - shift])
                if j <= n - 1 - shift:

                    g += (1 - h[j + s])
                if j - s >= 0 and (j - s) <= n - 1 - s:

                grad[j] += g

    for i in range(len(top5)):
        if budget_s - (time.time() - t0) < 50:
            break
        c5_cand, h_cand, n_cand = top5[i]
        h_sg, c5_sg = subgrad_opt(h_cand, n_cand, n_steps=80000, lr0=0.03,
                                   time_limit=min(60, max(10, int(budget_s - (time.time() - t0) - 40))))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
            best_n = n_cand
        if budget_s - (time.time() - t0) < 20:
            break
        h_opt, c5_opt = slsqp_opt(h_sg, n_cand, maxiter=300, ftol=1e-12)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_cand

    # Phase 3: Try more SLSQP on other resolutions with best structure
    if best_h is not None:
        n_list = [64, 96, 128, 192, 256]
        for n_try in n_list:
            if budget_s - (time.time() - t0) < 35:
                break
            if abs(n_try - best_n) < 10:
                continue
            h_tr = upsample_h(best_h, n_try)
            c5_tr = compute_c5(h_tr)
            if c5_tr < best_c5:
                best_c5 = c5_tr
                best_h = h_tr
                best_n = n_try
            if budget_s - (time.time() - t0) < 15:
                break
            h_opt, c5_opt = slsqp_opt(h_tr, n_try, maxiter=200, ftol=1e-12)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 4: Upsample to higher resolution
    if best_h is not None:
        for n_hi in [300, 400, 500, 700]:
            if budget_s - (time.time() - t0) < 35:
                break
            if n_hi <= best_n:
                continue
            h_up = upsample_h(best_h, n_hi)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up
                best_n = n_hi
            if budget_s - (time.time() - t0) < 15:
                break
            h_opt, c5_opt = slsqp_opt(h_up, n_hi, maxiter=150, ftol=1e-12)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_hi

    # Phase 5: Perturb and re-optimize
    if best_h is not None:
        rng_f = np.random.RandomState(seed + 4242)
        for attempt in range(5):
            if budget_s - (time.time() - t0) < 20:
                break
            scale = 0.015 + 0.01 * attempt
            h_pert = best_h + rng_f.normal(0, scale, best_n)
            h_pert = project_feasible(h_pert, best_n)
            h_opt, c5_opt = slsqp_opt(h_pert, best_n, maxiter=250, ftol=1e-13)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = best_n  # same n

    # Phase 6: Final SLSQP polish
    if best_h is not None and budget_s - (time.time() - t0) > 12:
        h_final, c5_final = slsqp_opt(best_h, best_n, maxiter=500, ftol=1e-14)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Ensure feasibility of final result
    if best_h is not None:
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
