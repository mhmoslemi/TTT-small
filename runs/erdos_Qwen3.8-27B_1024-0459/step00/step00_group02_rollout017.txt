def smooth_max_c5(h, n, T=0.05):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") / T
    # log-sum-exp for numerical stability
    max_c = np.max(corr)
    return T * max_c + T * np.log(np.sum(np.exp(corr - max_c))) * dx

def smooth_max_c5_grad(h, n, T=0.05):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr_raw = np.correlate(h, one_minus_h, mode="full")
    corr = corr_raw / T
    max_c = np.max(corr)
    weights = np.exp(corr - max_c)
    weights /= np.sum(weights)
    
    # Gradient of each c[k] with respect to h
    # c[k] = sum_i h[i] * (1-h[i+k_offset])
    # This is complex to compute analytically, so let's use finite differences or a trick
    
    # Actually, let's use the fact that:
    # d/dh[m] of c[k] = d/dh[m] of sum_i h[i]*(1-h[i+k_off])
    # For a specific shift k, the offset is k-(n-1)
    # d/dh[m] = (1-h[m+k_off]) if m+k_off in [0,n-1]
    #           - h[m-k_off] if m-k_off in [0,n-1]
    #           (sum of both terms if both valid)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        offset = k - (n-1)
        w = weights[k]
        if offset >= 0:
            # h[i] * (1-h[i+offset]) for i = 0 to n-1-offset
            for i in range(n - offset):
                grad[i] += w * (1 - h[i + offset])
                grad[i + offset] += w * (-h[i])
        else:
            # h[i] * (1-h[i+offset]) where offset < 0, so i+offset >= 0 means i >= -offset
            # h[i] * (1-h[i-offset]) for i = -offset to n-1
            for i in range(-offset, n):
                grad[i] += w * (1 - h[i + offset])
                grad[i + offset] += w * (-h[i])
    
    return grad * dx

    if abs(diff) > 1e-12:

            if total_can > 0:

                h = np.clip(h, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, n):
    """Project h to [0,1]^n with sum = n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-12:
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total > 1e-15:
                h = h + room * (diff / total)
        else:
            total = h.sum()
            if total > 1e-15:
                h = h + h * (diff / total)
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    residual = target - h.sum()
    if abs(residual) > 1e-15:
        idxs = np.where((residual > 0) & (h < 1) | (residual < 0) & (h > 0))[0]
        if len(idxs) > 0:
            for idx in idxs:
                step = np.clip(residual, -(h[idx]) if residual < 0 else (1-h[idx]), 
                               (1-h[idx]) if residual > 0 else h[idx])
                h[idx] += step
                residual -= step
                if abs(residual) < 1e-15:
                    break
    return h

def local_search(h, n, rng, time_limit):
    """Random local search with pairwise exchanges"""
    h = project_h(h.copy(), n)
    current = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current
    start = time.time()
    
    while time.time() - start < time_limit:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        
        delta_max = min(1.0 - h[i], h[j])
        delta_min = max(-h[i], h[j] - 1.0)
        
        if delta_max - delta_min < 1e-12:
            continue
        
        delta = rng.uniform(delta_min, delta_max)
        h[i] += delta
        h[j] -= delta
        
        new_c5 = compute_c5(h, n)
        if new_c5 < current:
            current = new_c5
            if current < best_c5:
                best_c5 = current
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
    
    return best_h, best_c5

def targeted_local_search(h, n, rng, time_limit):
    """Local search targeting the maximum correlation shift"""
    h = project_h(h.copy(), n)
    current = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current
    start = time.time()
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        
        # Determine offset for this shift
        offset = k_star - (n - 1)
        
        # Find valid indices for this shift
        if offset >= 0:
            idxs = range(offset, n)
        else:
            idxs = range(0, n + offset)
        
        idxs = list(idxs)
        if len(idxs) < 2:
            # Fall back to random
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            delta = rng.uniform(-0.05, 0.05)
            h[i] = np.clip(h[i] + delta, 0, 1)
            h[j] = np.clip(h[j] - delta, 0, 1)
            h = project_h(h, n)
            new_c5 = compute_c5(h, n)
            if new_c5 < current:
                current = new_c5
                if current < best_c5:
                    best_c5 = current
                    best_h = h.copy()
            else:
                # Undo
                h = project_h(h, n)  # Just re-clip
            continue
        
        # Pick two indices from the active set
        i = idxs[rng.integers(len(idxs))]
        j = idxs[rng.integers(len(idxs))]
        if i == j:
            i = (i + 1) % n
        
        delta_max = min(1.0 - h[i], h[j])
        delta_min = max(-h[i], h[j] - 1.0)
        if delta_max - delta_min < 1e-12:

                best_c5 = current
                best_h = h.copy()
        else:
            h[i] -= delta

    return best_h, best_c5

def smooth_gradient_descent(h, n, rng, time_limit, T=0.01):
    """Gradient descent with smooth max approximation"""
    h = project_h(h.copy(), n)
    current = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = current
    start = time.time()
    
    lr = 0.05
    
    while time.time() - start < time_limit:
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_c = np.max(corr)
        # Smooth weights
        weights = np.exp((corr - max_c) / T)
        weights /= weights.sum()
        
        # Compute gradient of sum_k weights[k] * corr[k] * dx
        # corr[k] = sum_j h[j] * (1 - h[j + offset_k]) for valid j
        # where offset_k = k - (n-1)
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            offset = k - (n - 1)
            w = weights[k]
            if abs(w) < 1e-10:
                continue
            if offset >= 0:
                # corr[k] = sum_{j=offset}^{n-1} h[j] * (1-h[j-offset])
                for j in range(offset, n):
                    grad[j] += w * (1.0 - h[j - offset])
                    grad[j - offset] += w * (-h[j])
            else:
                d = -offset
                # corr[k] = sum_{j=0}^{n-1-d} h[j] * (1-h[j+d])
                for j in range(0, n - d):
                    grad[j] += w * (1.0 - h[j + d])
                    grad[j + d] += w * (-h[j])
        
        grad *= dx
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_h(h_new, n)
        
        new_c5 = compute_c5(h_new, n)
        if new_c5 < current:
            h = h_new
            current = new_c5
            if current < best_c5:
                best_c5 = current
                best_h = h.copy()
            # Decay learning rate slowly
            lr *= 0.999
            if lr < 1e-6:
                lr = 0.05
        else:
            lr *= 0.5
            if lr < 1e-6:
                # Random restart
                lr = 0.05
                pert = rng.normal(0, 0.01, n)
                h_new = project_h(h + pert, n)
                h = h_new
                current = compute_c5(h, n)
    
    return best_h, best_c5
