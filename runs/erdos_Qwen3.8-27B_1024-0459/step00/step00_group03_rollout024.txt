    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def local_search_binary(h, dx, max_iters=100000, timeout=None, start_time=None):
    """Local search for binary sequences using swaps."""
    if start_time is None:
        start_time = time.time()
    n = len(h)

    zeros = np.where(h == 0)[0]

        best_swap = None
        best_val = best_c5

            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]

            h[i] = 1
            h[j] = 0
