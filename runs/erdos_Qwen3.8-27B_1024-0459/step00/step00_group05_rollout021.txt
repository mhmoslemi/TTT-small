import time

def compute_c5(h):

    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full")
    return float(np.max(conv) * dx)

def compute_c5_smooth(h, beta=50):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full") * dx
    # log-sum-exp for numerical stability
    m = np.max(conv)
    return m + np.log(np.sum(np.exp(beta * (conv - m)))) / beta

def compute_gradient(h, beta=50):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full") * dx
    
    # Softmax weights
    m = np.max(conv)
    weights = np.exp(beta * (conv - m))
    weights /= np.sum(weights)
    
    # Gradient of conv[k] w.r.t. h[i]
    # For lag m_idx in the full output (0 to 2n-2), shift = m_idx - (n-1)
    # conv[m_idx] = sum_j h[j] * (1-h[j+shift]) for valid j
    # d(conv[m_idx])/d(h[i]) = (1-h[i+shift]) if i+shift valid - h[i-shift] if i-shift valid
    
    grad = np.zeros(n)
    for m_idx in range(2*n-1):
        w = weights[m_idx]
        if w < 1e-15:
            continue
        shift = m_idx - (n-1)
        # d/dh[i] of sum_j h[j]*(1-h[j+shift])
        # = (1-h[i+shift]) - h[i-shift] (for valid indices)
        for i in range(n):
            # term 1: (1-h[i+shift]) if 0 <= i+shift < n
            val = 0.0
            j1 = i + shift
            if 0 <= j1 < n:
                val += (1.0 - h[j1])
            j2 = i - shift
            if 0 <= j2 < n:
                val -= h[j2]
            grad[i] += w * val
    
    return grad

    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1-h, mode="full") * dx
    
    m = np.max(conv)
    weights = np.exp(beta * (conv - m))
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for shift in range(-(n-1), n):
        w = weights[shift + (n-1)]
        if w < 1e-15:
            continue
        # term1: (1-h[i+shift]) for 0<=i+shift<n
        # This is: for i in range(max(0,-shift), min(n, n-shift))
        # grad[i] += w * (1-h[i+shift])
        if shift >= 0:
            # i+shift from 0 to n-1 → i from 0 to n-1-shift
            grad[:n-shift] += w * (1-h[shift:])
            # i-shift from 0 to n-1 → i from shift to n-1
            grad[shift:] -= w * h[:n-shift]
        else:
            s = -shift
            # i+shift = i-s, valid for i from s to n-1
            grad[s:] += w * (1-h[:n-s])
            # i-shift = i+s, valid for i from 0 to n-1-s
            grad[:n-s] -= w * h[s:]
    
    return grad

                return h

            room = 1.0 - h[mask]
            total_room = np.sum(room)
            if total_room < 1e-15:
                return np.clip(h, 0, 1)
            h[mask] += diff * room / total_room
            h = np.clip(h, 0, 1)

            h[mask] += diff * room / total_room
            h = np.clip(h, 0, 1)

        t_polish_end = min(t0 + budget_s - 10, time.time() + 60)
        if t_polish_end > time.time() + 15:
            h_final, c5_final = optimize_at_resolution(
                h_best, n_final, time.time(), t_polish_end,
                beta_init=80.0, top_k=50, seed=seed + 5000
            )
            if c5_final < c5_best:
                c5_best = c5_final
                h_best = h_final

    # Final verification and cleanup
    h_best = project_feasible(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    # Verify feasibility
    assert 0.0 <= np.min(h_best), "h has negative values"
    assert np.max(h_best) <= 1.0 + 1e-12, "h exceeds 1"
    assert abs(np.sum(h_best) - len(h_best) / 2.0) < 1e-8, "sum constraint violated"

    return h_best, c5_final, len(h_best)
