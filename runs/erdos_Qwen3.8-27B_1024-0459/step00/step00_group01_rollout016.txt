import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(1.0 - h))
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if abs(diff) > 1e-14:
                idx2 = int(np.argmin(1.0 - h))
                h[idx2] += min(diff, 1.0 - h[idx2])
        else:
            idx = int(np.argmax(h))
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if abs(diff) > 1e-14:
                idx2 = int(np.argmax(h))
                h[idx2] -= min(-diff, h[idx2])
    return h

        j = int(rng.integers(0, n))

            continue

        mag = float(rng.uniform(0.001, 0.05))

            si = curr[i]
