def project(h, n):

    target = n / 2.0
    
    # Bisection on alpha: find alpha s.t. sum(clip(h+alpha, 0, 1)) = target
    lo = -h.max() - 1.0
    hi = 1.0 - h.min()
    
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

import time

def compute_C5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, -1.0, 2.0)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) * 0.5
    return np.clip(h + alpha, 0.0, 1.0)

def soft_max_gradient(h, n_points, alpha):
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    scaled = conv * dx
    max_val = np.max(scaled)
    exp_vals = np.exp(alpha * (scaled - max_val))
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-9:
            continue
        shift = m - n + 1
        if shift == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif shift > 0:
            grad[:n - shift] += wm * (1.0 - h[shift:])
            grad[shift:] -= wm * h[:n - shift]
        else:
            t = -shift
            grad[t:] += wm * (1.0 - h[:n - t])
            grad[:n - t] -= wm * h[t:]
    return grad * dx

def gradient_descent_step(h, n, alpha, lr, rng):
    n_iter = 0
    c5_curr = compute_C5(h, n)
    improved_total = 0
    while True:
        grad = soft_max_gradient(h, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step_dir = grad / gn
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_new = h - trial_lr * step_dir
            h_new = project(h_new, n)
            c5_new = compute_C5(h_new, n)
            if c5_new < c5_curr - 1e-12:
                h = h_new
                c5_curr = c5_new
                improved = True
                improved_total += 1
                break
        if not improved:
            n_iter += 1
            if n_iter > 200:
                break

            h_new = h + noise
            h_new = np.clip(h_new, 0.0, 1.0)

            c5_new = compute_C5(h_new, n)
            if c5_new < c5_curr * 1.001:
                h = h_new
                c5_curr = c5_new
    return h, c5_curr, improved_total
