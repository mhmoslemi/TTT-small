import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def evaluate_c5_detail(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h_try = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_try) > target_sum:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    diff = target_sum - np.sum(h_out)
    if abs(diff) > 1e-9:
        interior = (h_out > 1e-8) & (h_out < 1.0 - 1e-8)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h_out[interior] += diff / cnt
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def compute_gradient_for_lags(h, lags):
    """Compute gradient of sum_{k in lags} g_k(h)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in lags:
        L = k - (n - 1)
        if L >= 0:
            nv = n - L
            if nv <= 0: continue
            grad[L:] += (1.0 - h[:nv])
            grad[:nv] += -h[L:]
        else:
            M = -L
            nv = n - M
            if nv <= 0: continue
            grad[M:] += -h[:nv]
            grad[:nv] += (1.0 - h[M:])
    return grad * dx

def subgradient_step(h, n, corr, step_size, top_m, target_sum):
    idx = np.argpartition(corr, -top_m)[-top_m:]
    grad = compute_gradient_for_lags(h, idx)
    h_new = h - step_size * grad
    return project_feasible(h_new, target_sum)
