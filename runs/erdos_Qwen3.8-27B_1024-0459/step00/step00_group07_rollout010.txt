    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum to n/2
    target = n / 2.0
    current = np.sum(h)
    if current == target:
        return h
    diff = target - current
    # Distribute the correction
    # If diff > 0, need to increase some values (those < 1)
    # If diff < 0, need to decrease some values (those > 0)

            h[h < 1.0] = 1.0

            # Simpler: just add diff/n to all, then clip
            h += diff / n
            h = np.clip(h, 0.0, 1.0)

def compute_c5(h):

def upsample(h_parent, n_new):
    """Upsample from n_parent to n_new using linear interpolation"""
    n_old = len(h_parent)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_parent)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def binary_local_search(h, n_swaps, random_state=42):
    """Local search by random swaps for binary h"""
    n = len(h)
    rng = np.random.RandomState(random_state)
    c5 = compute_c5(h)
    h = h.copy()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for _ in range(n_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        h[a], h[b] = h[b], h[a]
        new_c5 = compute_c5(h)

import time

def compute_corr(h):
    """Compute full correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

def subgradient_c5(h):
    """Compute subgradient of C5 = max_k corr_k * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_val = np.max(corr)
    tol = 1e-8
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    shifts = np.arange(len(corr)) - (n - 1)  # -n+1, ..., n-1
    
    for k in active:
        s = shifts[k]
        if s >= 0:
            if s < n:
                grad[:n-s] += (1.0 - h[s:n])
                grad[s:] -= h[:n-s]
        else:
            s2 = -s
            if s2 < n:
                grad[s2:] += (1.0 - h[:n-s2])
                grad[:n-s2] -= h[s2:]
    
    grad *= dx / len(active)
    return grad

def project_onto_set(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(h+lambda, 0, 1)) = target
    lo = -1.0 - np.max(h)
    hi = 1.0 - np.min(h)
    for _ in range(60):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:

    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old, period=n_old)

    """Convert to binary with exactly n/2 ones"""
    h = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h))

        # Remove extras

        h[to_remove] = 0

        to_add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h[to_add] = 1
    return h

    """Local search for binary h by random swaps"""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    start = time.time()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_c5 = c5
    best_h = h.copy()
    
    while time.time() - start < max_time:
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        h[a] = 0
        h[b] = 1
        new_c5 = compute_c5(h)
        
        if new_c5 < c5:
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            # Update tracking
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[a] = 1
            h[b] = 0
    
    return best_h, best_c5

def continuous_refine(h, max_time, rng, alpha=100.0):
    """Projected subgradient descent for continuous h"""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    h = project_onto_set(h, n)
    
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    start = time.time()
    
    lr = 0.1  # Learning rate
    t = 0
    
    while time.time() - start < max_time:
        t += 1
        # Smooth max approximation
        corr = compute_corr(h)
        scaled = alpha * corr * dx
        max_scaled = np.max(scaled)
        weights = np.exp(scaled - max_scaled)
        weights /= np.sum(weights)
        
        # Gradient of smooth max
        shifts = np.arange(len(corr)) - (n - 1)
        grad = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] < 1e-6:
                continue
            s = shifts[k]
            if s >= 0:
                if s < n:
                    grad[:n-s] += weights[k] * (1.0 - h[s:n])
                    grad[s:] -= weights[k] * h[:n-s]
            else:
                s2 = -s
                if s2 < n:
                    grad[s2:] += weights[k] * (1.0 - h[:n-s2])
                    grad[:n-s2] -= weights[k] * h[s2:]
        
        grad *= dx
        
        # Step
        lr_t = lr / (1.0 + 0.01 * t)
        h = h - lr_t * grad
        h = project_onto_set(h, n)
        
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return best_h, best_c5

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_onto_set(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def make_binary(h, n, rng):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        to_remove = rng.choice(ones_idx, n_ones - target, replace=False)
        h_bin[to_remove] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        to_add = rng.choice(zeros_idx, target - n_ones, replace=False)
        h_bin[to_add] = 1.0

def upsample_linear(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    h_new = np.interp(x_new, x_old, h_old, period=n_old)
    return np.clip(h_new, 0.0, 1.0)
