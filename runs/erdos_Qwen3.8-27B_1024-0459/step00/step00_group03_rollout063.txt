    # Simple scaling

    # If still off, do iterative clip+scale
    for _ in range(20):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-10:

        if diff > 0:

            mask = h < 1.0
            if not np.any(mask):

            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)

    start = time.time()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute correlation
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        
        # Find max
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        
        # Subgradient
        grad = np.zeros(n)
        # First term: (1-h[i+lag]) for valid i+lag
        if lag >= 0:
            # i + lag in [0, n-1] => i in [0, n-1-lag]
            grad[:-lag] += (1.0 - h[lag:])  # h[i+lag] for i in [0, n-1-lag]
        else:

        # Second term: -h[i-lag] for valid i-lag
        if lag >= 0:
            # i - lag in [0, n-1] => i in [lag, n-1]

        # Project
        h = project_to_feasible(h, n)
        
        # Evaluate
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive step size
        lr *= 0.999
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute C5 = max_k corr[k] * dx"""
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    return np.max(corr)

def project_to_feasible(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on shift
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_p) > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return h_p

def subgradient_step(h, n):
    """One subgradient step: find max lag, compute subgradient, return grad"""
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    
    grad = np.zeros(n)
    # First term: (1-h[i+lag]) for valid i+lag
    if lag >= 0:
        # i in [0, n-1-lag], i+lag in [lag, n-1]
        grad[:n-lag] += (1.0 - h[lag:])
    else:
        # i in [-lag, n-1], i+lag in [0, n-1+lag]
        lneg = -lag
        grad[lneg:] += (1.0 - h[:n-lneg])
    
    # Second term: -h[i-lag] for valid i-lag
    if lag >= 0:
        # i-lag in [0, n-1], i in [lag, n-1]
        grad[lag:] -= h[:n-lag]
    else:
        # i-lag in [0, n-1], i in [0, n-1+lag]
        lneg = -lag
        grad[:n-lneg] -= h[lneg:]
    
    return grad

def coordinate_descent(h, n, dx, n_passes=50, time_limit=None, start_time=None):
    """Coordinate descent: for each i, optimize h[i] in [0,1]"""
    h = h.copy()
    
    for p in range(n_passes):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break

def compute_c5(h, dx):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')

def project_to_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid

def compute_c5(h, dx):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def coordinate_descent(h, n, dx, n_passes=50, time_limit_s=120, start_time=None):

            h[i] = 0.0

def subgradient_optimize(h, n, dx, n_steps=500, time_limit_s=60, start_time=None):
    h = h.copy()

        start_time = time.time()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    lr = 0.03

        if time.time() - start_time > time_limit_s:
            break
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)

        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
