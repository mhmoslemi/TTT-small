def smooth_max(x, alpha):
    """Smooth approximation to max(x) using log-sum-exp."""
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(alpha * (x - x_max)))) / alpha

def objective(h, n, alpha, penalty):
    """Objective function: smooth max of correlation + penalty for sum constraint."""
    h_clipped = np.clip(h, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode="full")
    values = corr * (2.0 / n)
    obj = smooth_max(values, alpha)
    sum_constraint = np.sum(h_clipped) - n / 2.0
    obj += penalty * sum_constraint ** 2
    return obj

    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(alpha * (x - x_max)))) / alpha

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    target = n / 2.0

        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        h -= diff / n

    return h
