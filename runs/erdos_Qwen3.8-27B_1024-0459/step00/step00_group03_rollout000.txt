def _project_box_sum(v, target):
    v=np.asarray(v,dtype=float)
    n=len(v)
    if n==0: return v
    target=float(target)
    if target <= 0.0:
       return np.zeros(n)
    if target >= n:
       return np.ones(n)
    v=np.clip(v, -100.0, 100.0)
    lo=-100.0; hi=100.0
    # Ensure at lo sum>=target, hi<=target
    # since v-(-100)=v+100 clipped to 1 -> all 1 if all v>-101, sum n>=target. hi all 0.
    for _ in range(70):
        mid=0.5*(lo+hi)
        s=np.sum(np.clip(v-mid,0.0,1.0))
        if s > target:
            lo=mid
        else:
            hi=mid
    h=np.clip(v-0.5*(lo+hi),0.0,1.0)
    diff=target-h.sum()
    # correction using headroom/tail
    if diff > 0.0:
        head=1.0-h
        hsum=head.sum()
        if hsum > 0.0:
            h += head*(diff/hsum)
        else:
            h.fill(1.0); 
    elif diff < 0.0:
        tail=h
        tsum=tail.sum()
        if tsum > 0.0:
            h -= tail*((-diff)/tsum)
    h=np.clip(h,0.0,1.0)
    diff=target-h.sum()
    if diff > 1e-14:
       idx=int(np.argmax(1.0-h)); add=min(1.0-h[idx], diff); h[idx]+=add
    elif diff < -1e-14:
       idx=int(np.argmax(h)); sub=min(h[idx], -diff); h[idx]-=sub

def _calibrate_corr_direction():
    global _CORR_FIRST_NEG
    if _CORR_FIRST_NEG is None:
        a = np.array([1.0, 2.0, 3.0, 5.0])

    while m < need:
        m <<= 1
    H = np.fft.rfft(h, m)
    B = np.fft.rfft(b, m)
    c = np.fft.irfft(H * np.conj(B), m)
    vals = np.zeros(2 * n - 1, dtype=float)

    if first_neg:
        vals[n - 1:n - 1 + n] = c[:n]
        if n - 1 > 0:
            vals[:n - 1] = c[m - n + 1:m]
    else:
        vals[:n] = c[:n][::-1]
        if n - 1 > 0:
            vals[n:] = c[m - n + 1:m][::-1]

    vals = np.where(vals > 0.0, vals, 0.0)
    return vals


def _full_counts(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n <= 1:
        return np.array([float(h[0] * (1.0 - h[0]))]), np.array([0])
    first_neg = _calibrate_corr_direction()
    if n <= 900:
        vals = np.asarray(np.correlate(h, 1.0 - h, mode="full"), dtype=float)
    else:
        vals = _fft_full_counts(h, n, first_neg)
    vals = np.where(vals > 0.0, vals, 0.0)
    lags = _lags_for(n)
    return vals, lags


def _objective_from_counts(counts, n):
    if counts.size == 0:
        return 1.0
    return float(np.max(counts)) * (2.0 / float(n))


def _objective_float(h):
    h = np.asarray(h, dtype=float)
    counts, _ = _full_counts(h)
    return _objective_from_counts(counts, len(h))


def _project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n = v.size
    if n == 0:
        return v.copy()
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    v = np.clip(v, -100.0, 100.0)
    lo = -100.0
    hi = 100.0

    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(np.sum(h))
    if diff > 0.0:
        head = 1.0 - h
        hs = float(np.sum(head))
        if hs > 0.0:
            h += head * (diff / hs)
        else:
            h.fill(1.0)
    elif diff < 0.0:
        tail = h
        ts = float(np.sum(tail))
        if ts > 0.0:
            h -= tail * ((-diff) / ts)
    h = np.clip(h, 0.0, 1.0)

    diff = target - float(np.sum(h))
    if diff > 1.0e-14:
        idx = int(np.argmax(1.0 - h))
        add = min(1.0 - h[idx], diff)
        h[idx] += add
    elif diff < -1.0e-14:
        idx = int(np.argmax(h))
        sub = min(h[idx], -diff)
        h[idx] -= sub

    return h


def _snap_near_binary(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return None
    s = float(np.sum(h))
    s_target = int(round(s))
    if abs(s - s_target) > 1.0e-6:
        return None
    h2 = np.where(h < 0.5, 0.0, 1.0)
    if int(np.sum(h2)) != s_target:
        return None
    if float(np.max(np.abs(h - h2))) < 1.0e-3:
        return h2
    return None


def _candidate_stats(h):
    if h is None:
        return None, 1.0
    h = np.asarray(h, dtype=float)
    if h.ndim != 1:
        h = h.ravel()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if n == 0:
        return np.array([0.5]), 1.0
    h = _project_box_sum(h, 0.5 * n)
    c = _objective_float(h)
    hs = _snap_near_binary(h)
    if hs is not None:
        cs = _objective_float(hs)
        if cs < c:
            return hs, cs
    return h, c


def _grad_lag(h, k):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if k == 0:
        return 1.0 - 2.0 * h
    if k > 0:
        g[k:] += 1.0 - h[:-k]
        g[:-k] -= h[k:]
    else:
        m = -k
        g[:n - m] += 1.0 - h[m:]
        g[m:] -= h[:n - m]
    return g


def _sum_constraint(x, target):
    return float(np.sum(x) - target)


def _sum_jacobian(x, target):
    return np.ones(len(x), dtype=float)


def _slsqp_objective(x, beta, n, topk, aux_weight):
    x = np.clip(x, 0.0, 1.0)
    counts, _ = _full_counts(x)
    dx = 2.0 / float(n)
    s = counts * dx
    smax = float(np.max(s))
    z = beta * (s - smax)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    soft = smax + float(np.log(float(np.sum(ez)))) / beta
    aux = aux_weight * float(np.sum(x * (1.0 - x))) * dx
    return float(soft + aux)


def _slsqp_jacobian(x, beta, n, topk, aux_weight):
    x = np.clip(x, 0.0, 1.0)
    counts, lags = _full_counts(x)
    dx = 2.0 / float(n)
    s = counts * dx
    smax = float(np.max(s))
    z = beta * (s - smax)
    z = np.clip(z, -50.0, 50.0)
    ez = np.exp(z)
    w = ez / float(np.sum(ez))

    grad = np.zeros(n, dtype=float)
    L = len(w)
    if topk and topk < L:
        order = np.argpartition(-s, topk)[:topk]
        w = w[order]
        lags = lags[order]
    for wt, lag in zip(w, lags):
        if wt > 1.0e-12:
            grad += wt * _grad_lag(x, lag)
    grad *= dx
    if aux_weight:
        grad += aux_weight * dx * (1.0 - 2.0 * x)
    return grad


def _slsqp_one(h, stop_time, beta, topk, aux_weight):
    h0 = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h0)
    target = 0.5 * n
    h0 = _project_box_sum(h0, target)
    c0 = _objective_float(h0)

    tleft = max(0.0, stop_time - time.time())
    maxiter = int(min(160, max(30, 30 + 5.0 * tleft)))

    bounds = [(0.0, 1.0)] * n
    constraints = [
        {
            "type": "eq",
            "fun": _sum_constraint,
            "jac": _sum_jacobian,
            "args": (target,),
        }
    ]

    try:
        res = sopt.minimize(
            _slsqp_objective,
            h0,
            args=(beta, n, topk, aux_weight),
            jac=_slsqp_jacobian,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": maxiter, "ftol": 1.0e-12, "disp": False, "maxls": 50},
        )
        x = np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
        x = _project_box_sum(x, target)
        c = _objective_float(x)
        if c < c0:
            return x
    except Exception:
        pass
    return h0.copy()


def _slsqp_sequence(h, stop_time, beta_list, aux_list, topk=0):
    current = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for beta, aux in zip(beta_list, aux_list):
        if time.time() > stop_time - 5.0:
            break
        cand = _slsqp_one(current, stop_time, beta, topk, aux)
        current = cand
    return current


def _projected_descent(h, stop_time, seed=0, beta=25.0, topk=40, aux_weight=0.003):
    rng = np.random.default_rng(seed)
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = 0.5 * n
    h = _project_box_sum(h, target)

    best = h.copy()
    best_c = _objective_float(h)
    cur_c = best_c
    alpha = max(1.0, float(n) / 8.0)
    failed = 0
    stall = 0

    while time.time() < stop_time:
        counts, lags = _full_counts(h)
        dx = 2.0 / float(n)
        s = counts * dx
        cur_c = _objective_from_counts(counts, n)
        if cur_c < best_c:
            best = h.copy()
            best_c = cur_c
            stall = 0

        L = len(s)
        if topk and topk < L:
            idx = np.argpartition(-s, topk)[:topk]
        else:
            idx = np.arange(L)
        idx = idx[np.argsort(-s[idx])]

        obj = cur_c
        z = beta * (s[idx] - obj)
        z = np.clip(z, -50.0, 0.0)
        ez = np.exp(z)
        w = ez / float(np.sum(ez))

        grad = np.zeros(n, dtype=float)
        for wt, lag in zip(w, lags[idx]):
            if wt > 1.0e-12:
                grad += wt * _grad_lag(h, lag)
        grad *= dx
        if aux_weight:
            grad += aux_weight * dx * (1.0 - 2.0 * h)

        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1.0e-15:
            break
        direction = -grad

        moved = False
        for fac in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = _project_box_sum(h + fac * alpha * direction, target)
            c = _objective_float(cand)
            if c < obj - 1.0e-10:
                h = cand
                cur_c = c
                alpha = min(alpha * 1.08, float(n))
                moved = True
                failed = 0
                break
        if moved:
            if cur_c < best_c:
                best = h.copy()
                best_c = cur_c
                stall = 0
        else:
            alpha = max(1.0, alpha * 0.6)
            failed += 1
            if failed >= 12:
                failed = 0
                stall += 1
                if stall > 4:
                    alpha = max(1.0, float(n) / 12.0)
                    h = _project_box_sum(best + 0.08 * rng.standard_normal(n), target)
                    cur_c = _objective_float(h)
                    stall = 0

    return best, best_c


def _pair_polish(h, stop_time, seed=0):
    rng = np.random.default_rng(seed)
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = len(h)
    target = 0.5 * n
    h = _project_box_sum(h, target)

    best = h.copy()
    best_c = _objective_float(h)
    cur_c = best_c
    stale = 0
    noise = 0.12

    while time.time() < stop_time:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        lo = -h[i]
        hi = 1.0 - h[j]
        span = hi - lo
        if span < 1.0e-12:
            continue

        candidates = []
        candidates.append(0.5 * span)
        candidates.append(rng.uniform(0.0, span))
        candidates.append(rng.normal(0.0, max(0.02, 0.15 * span)))
        candidates.append(rng.uniform(0.0, span))

        accepted = False
        for d in candidates:
            d = float(d)
            if d <= lo + 1.0e-13 or d >= hi - 1.0e-13:
                d = min(max(d, lo + 1.0e-13), hi - 1.0e-13)
            cand = h.copy()
            cand[i] += d
            cand[j] -= d
            c = _objective_float(cand)
            if c < cur_c - 1.0e-11:
                h = cand
                cur_c = c
                if cur_c < best_c:
                    best = h.copy()
                    best_c = cur_c
                stale = 0
                accepted = True
                break
        if not accepted:
            stale += 1
        if stale > 3000:
            stale = 0
            h = _project_box_sum(best + noise * rng.standard_normal(n), target)
            cur_c = _objective_float(h)
            noise = max(0.02, noise * 0.8)

    return best, best_c


def _binary_search(n, stop_time, seed=0, start_float=None):
    if n % 2 != 0:
        return None, 1.0
    rng = np.random.default_rng(seed)
    target = n // 2

    if start_float is not None and len(start_float) == n:
        start = np.asarray(start_float, dtype=float)
        order = np.argsort(-start)
        idx = order[:target]
    else:
        idx = rng.choice(n, size=target, replace=False)

    pos = np.zeros(n, dtype=np.int64)
    pos[idx] = 1

    def _bin_obj(vec):
        b = 1 - vec
        return int(np.max(np.correlate(vec, b, mode="full")))

    cur = _bin_obj(pos)
    best = pos.copy()
    best_cur = cur
    T = 4.0
    min_T = 1.0e-3
    no_impr = 0

    while time.time() < stop_time:
        a = int(rng.integers(n))
        for _ in range(200):
            if pos[a] == 1:
                break
            a = int(rng.integers(n))
        b = int(rng.integers(n))
        for _ in range(200):
            if pos[b] == 0:
                break
            b = int(rng.integers(n))
        if a == b or pos[a] != 1 or pos[b] != 0:
            continue

        pos[a] = 0
        pos[b] = 1
        new = _bin_obj(pos)
        delta = new - cur

        accept = delta < 0
        if (not accept) and T > min_T:
            p = 0.0
            if delta < 50.0 * T:
                p = float(np.exp(-float(delta) / T))
            accept = bool(rng.random() < p)

        if accept:
            cur = new
            T = max(min_T, T * 0.9992)
            if cur < best_cur:
                best_cur = cur
                best = pos.copy()
                no_impr = 0
            else:
                no_impr += 1
        else:
            pos[a] = 1
            pos[b] = 0
            no_impr += 1

        if no_impr > 2500:
            no_impr = 0
            pos = best.copy()
            cur = best_cur
            T = max(min_T, T * 1.4)
            for _ in range(3):
                aa = int(rng.integers(n))
                for __ in range(100):
                    if pos[aa] == 1:
                        break
                    aa = int(rng.integers(n))
                bb = int(rng.integers(n))
                for __ in range(100):
                    if pos[bb] == 0:
                        break
                    bb = int(rng.integers(n))
                if aa != bb and pos[aa] == 1 and pos[bb] == 0:
                    pos[aa] = 0
                    pos[bb] = 1
            cur = _bin_obj(pos)

    return best.astype(float), float(best_cur) * (2.0 / float(n))


def _upsample(parent, n1):
    parent = np.asarray(parent, dtype=float)
    n0 = len(parent)
    if n0 == n1:
        return parent.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n1) + 0.5) * (2.0 / n1)
    h = np.interp(x1, x0, parent, left=parent[0], right=parent[-1])
    return _project_box_sum(h, 0.5 * n1)


def _random_smooth(n, seed):
    rng = np.random.default_rng(seed)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    u = np.zeros(n, dtype=float)
    for m in range(1, 8):
        amp = rng.standard_normal(1)[0] / (0.5 * float(m))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        u += amp * np.cos(m * np.pi * x + ph)
    u -= float(np.mean(u))
    amp = float(np.percentile(np.abs(u), 95))
    if amp > 1.0e-10:
        u = 0.95 * u / amp
    else:
        u = rng.standard_normal(n)
        u -= float(np.mean(u))
    h = 0.5 + 0.5 * u
    return _project_box_sum(h, 0.5 * n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1300.0
    deadline = t0 + budget - min(30.0, max(5.0, 0.03 * budget))
    seed_i = int(seed)

    best_h = None
    best_c = 1.0
    best_n = 0

    par = None
    n_par = 0
    try:
        parent_arr = np.asarray(initial_h_values, dtype=float)
        if parent_arr.ndim != 1:
            parent_arr = parent_arr.ravel()
        if parent_arr.size > 0:
            par = np.clip(parent_arr, 0.0, 1.0)
            n_par = len(par)
    except Exception:
        par = None

    if par is None:
        n_par = 80
        x = (np.arange(n_par) + 0.5) * (2.0 / n_par)
        par = 0.5 + 0.5 * np.cos(np.pi * x)
        par = _project_box_sum(par, 0.5 * n_par)
    else:
        par = _project_box_sum(par, 0.5 * n_par)

    cand_list = [par, 0.5 * (par + par[::-1]), 1.0 - par]
    for cand in cand_list:
        h, c = _candidate_stats(cand)
        if h is not None and c < best_c:
            best_h = h
            best_c = c
            best_n = len(h)

    if best_h is None:
        best_h, best_c = _candidate_stats(par)
        best_n = len(best_h)

    base_n = int(best_n)
    for sh in range(0, base_n, max(1, base_n // 10)):
        h, c = _candidate_stats(np.roll(best_h, sh))
        if c < best_c:
            best_h = h
            best_c = c
            best_n = len(h)

    if best_n <= 220 and time.time() + 60.0 < deadline:
        stop = min(deadline, time.time() + 200.0)
        h1, c1 = _candidate_stats(
            _slsqp_sequence(
                best_h,
                stop,
                [8.0, 25.0],
                [0.02, 0.005],
                topk=0,
            )
        )
        if c1 < best_c:
            best_h = h1
            best_c = c1
            best_n = len(h1)

    if time.time() + 50.0 < deadline:
        stop = min(deadline, time.time() + 120.0)
        topk = max(24, best_n // 2)
        h1, c1 = _candidate_stats(
            _projected_descent(
                best_h,
                stop,
                seed=seed_i + 1,
                beta=25.0,
                topk=topk,
                aux_weight=0.003,
            )
        )
        if c1 < best_c:
            best_h = h1
            best_c = c1
            best_n = len(h1)

    if time.time() + 30.0 < deadline:
        stop = min(deadline, time.time() + 70.0)
        h1, c1 = _candidate_stats(_pair_polish(best_h, stop, seed=seed_i + 2))
        if c1 < best_c:
            best_h = h1
            best_c = c1
            best_n = len(h1)

    if best_n <= 90 and time.time() + 80.0 < deadline:
        stop = min(deadline, time.time() + 180.0)
        hb, cb = _binary_search(best_n, stop, seed=seed_i + 3, start_float=best_h)
        hb, cb = _candidate_stats(hb)
        if cb < best_c:
            best_h = hb
            best_c = cb
            best_n = len(hb)

        if time.time() + 60.0 < deadline and best_n <= 220:
            stop = min(deadline, time.time() + 100.0)
            h1, c1 = _candidate_stats(
                _slsqp_sequence(
                    best_h,
                    stop,
                    [15.0, 40.0],
                    [0.01, 0.002],
                    topk=0,
                )
            )
            if c1 < best_c:
                best_h = h1
                best_c = c1
                best_n = len(h1)

    base = best_h.copy()
    base_n = int(len(base))
    if time.time() + 120.0 < deadline:
        for mult in (2, 3, 4):
            n1 = base_n * mult
            if n1 <= base_n or n1 > 480:
                continue
            if time.time() + 90.0 > deadline:
                break

            h1 = _upsample(base, n1)
            if n1 <= 180 and time.time() + 90.0 < deadline:
                stop = min(deadline, time.time() + 110.0)
                h1, _ = _candidate_stats(
                    _slsqp_sequence(
                        h1,
                        stop,
                        [10.0, 30.0],
                        [0.01, 0.003],
                        topk=0,
                    )
                )
            else:
                stop = min(deadline, time.time() + 110.0)
                topk = min(60, max(30, n1 // 4))
                h1, _ = _candidate_stats(
                    _projected_descent(
                        h1,
                        stop,
                        seed=seed_i + 50 + mult,
                        beta=25.0,
                        topk=topk,
                        aux_weight=0.003,
                    )
                )

            if time.time() + 30.0 < deadline:
                stop = min(deadline, time.time() + 40.0)
                h1, _ = _candidate_stats(_pair_polish(h1, stop, seed=seed_i + 80 + mult))

            h1, c1 = _candidate_stats(h1)
            if c1 < best_c:
                best_h = h1
                best_c = c1
                best_n = len(h1)

    if time.time() + 140.0 < deadline:
        for r in range(3):
            if time.time() + 80.0 > deadline:
                break
            hrand, _ = _candidate_stats(_random_smooth(best_n, seed=seed_i + 200 + r))
            stop = min(deadline, time.time() + 70.0)
            hrand, _ = _candidate_stats(
                _projected_descent(
                    hrand,
                    stop,
                    seed=seed_i + 300 + r,
                    beta=25.0,
                    topk=max(24, best_n // 2),
                    aux_weight=0.002,
                )
            )
            if time.time() + 20.0 < deadline:
                stop = min(deadline, time.time() + 25.0)
                hrand, _ = _candidate_stats(_pair_polish(hrand, stop, seed=seed_i + 400 + r))

            hrand, crand = _candidate_stats(hrand)
            if crand < best_c:
                best_h = hrand
                best_c = crand
                best_n = len(hrand)

    if time.time() + 40.0 < deadline:
        stop = min(deadline, time.time() + 90.0)
        h1, c1 = _candidate_stats(_pair_polish(best_h, stop, seed=seed_i + 999))
        if c1 < best_c:
            best_h = h1
            best_c = c1
            best_n = len(h1)

    h_final, c_final = _candidate_stats(best_h)
    if h_final is None:
        h_final = best_h
        c_final = best_c

    return h_final, float(c_final), int(len(h_final))
