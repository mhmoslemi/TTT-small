def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)

    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def grad_corr_m(h, m, n):
    """Compute gradient of correlation at output index m with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    
    if m == n - 1:  # zero lag
        grad[:] = 1.0 - 2.0 * h
    else:
        k = abs(m - (n - 1))
        # Forward term: h[j] appears as h[i+k] when j = i+k, i.e., j in [k, n-1]
        # Contribution: (1 - h[j-k])
        if k <= n - 1:
            grad[k:] += (1.0 - h[:n-k])
        # Backward term: h[j] appears in (1-h[i]) when j = i, i.e., j in [0, n-1-k]
        # Contribution: -h[j+k]
        if k <= n - 1:
            grad[:n-k] += (-h[k:])
    return grad * dx

def obj_grad(h, tau=100.0):
    """Compute objective and gradient using soft-max smoothing."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')

    scaled_max = np.max(scaled)
    weights = np.exp(scaled - scaled_max)

    obj = (scaled_max / tau + np.log(np.sum(weights)) / tau)  # log-sum-exp

def resample_h(h, new_n):

    if old_n == new_n:
        return h.copy()
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)


def do_gradient_descent(h, n, time_limit, rng, init_tau, init_alpha):
    tau = init_tau
    alpha = init_alpha
    c5 = compute_c5(h)
    stalled = 0
    for _ in range(200000):
        if time.time() > time_limit:
            break
        _, grad = smooth_obj_and_grad(h, tau)
        h_new = project_to_set(h - alpha * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-12:
            h = h_new
            c5 = c5_new
            stalled = 0
            alpha = min(alpha * 1.02, 0.3)
        else:
            stalled += 1
            alpha = max(alpha * 0.7, 0.005)
            if stalled > 40:
                h = project_to_set(h + rng.normal(0, 0.04, n), n)
                c5 = compute_c5(h)
                tau = float(rng.choice([20.0, 40.0, 80.0, 150.0, 300.0, 500.0]))
                alpha = 0.03
                stalled = 0
    return h, c5


def do_random_search(h, n, time_limit, rng, n_trials):
    c5 = compute_c5(h)
    for _ in range(n_trials):
        if time.time() > time_limit:

        sigma = float(rng.choice([0.01, 0.02, 0.03, 0.05, 0.08]))
        h_new = project_to_set(h + rng.normal(0, sigma, n), n)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5


def do_structured_moves(h, n, time_limit, rng):
