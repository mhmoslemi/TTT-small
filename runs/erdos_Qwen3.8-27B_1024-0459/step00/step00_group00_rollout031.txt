            grad[0:n+s] -= w * h[-s:n]
    
    grad *= dx
    return grad

def project_to_feasible(h, target_sum):
    """Project h to [0,1] with sum = target_sum"""
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if diff != 0:
        # Distribute evenly
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    return h

def optimize_at_resolution(h0, n, time_limit):

    h_best = h0.copy()

            diff = target_sum - x.sum()
            x = x + diff / n
            x = np.clip(x, 0, 1)
            return smooth_max_obj_val(x, alpha)

            # Add constraint gradient
            g += (x.sum() - target_sum) / n

            {'type': 'eq', 'fun': lambda x: x.sum() - target_sum,

            if res.success or res.fun < best_score:

                    best_score = s
                    h_best = h_new
        except:
            pass

    if time.time() - start < time_limit - 5:
        h_best = coordinate_polish(h_best, n, time_limit - (time.time() - start) - 2)

def coordinate_polish(h, n, time_limit):
    """Coordinate descent to polish the solution"""
    start = time.time()
    target_sum = n / 2.0
    h = h.copy()
    
    for iteration in range(100):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                    break
            old_val = h[j]
            # Try small perturbations
            best_j = old_val
            best_f = eval_c5(h)
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 
                         -0.1, 0.1, -0.2, 0.2]:
                new_val = old_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[j] = new_val
                h = project_to_feasible(h, target_sum)
                f = eval_c5(h)
                if f < best_f:
                    best_f = f
                    best_j = new_val
            h[j] = best_j
            h = project_to_feasible(h, target_sum)
            if abs(h[j] - old_val) > 1e-10:
                improved = True
        if not improved:
            break
    
    return h

def upsample(h, new_n):
    """Upsample h to new_n points using periodic interpolation"""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2)
    # Project to feasible
    return project_to_feasible(h_new, new_n / 2.0)
