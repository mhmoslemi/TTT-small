def softmax_weights(x, temp):
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted / temp)
    return exp_x / np.sum(exp_x)

def compute_softerr(h, dx, temp):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    x_max = np.max(corr)
    w = softmax_weights(corr - x_max, temp)  # weights for softmax
    sm = x_max + temp * np.log(np.sum(np.exp((corr - x_max) / temp)))
    return sm, w

def gradient_softerr(h, dx, w):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        for j in range(j_lo, j_hi + 1):
            grad[j + s] += wk * (1.0 - h[j])
            grad[j] -= wk * h[j + s]
    grad *= dx
    return grad

    sm, w = compute_softerr(h, dx, temp)

    return sm + pen, w
