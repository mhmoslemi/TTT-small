def random_binary_h(n,rng):
    h = rng.integers(0,2,size=n).astype(float)
    target=0.5*n
    # if exact integer halfway for even; for odd approximate

def sturmian_h(n, alpha, beta):

def resample_nearest(h, n_new):
    h = np.asarray(h,float)
    old_n = h.size
    if old_n == n_new: return h.copy()
    if old_n ==1:
       out = np.full(n_new, float(h[0])); return project_box_sum(out,n_new)
    if n_new > old_n and n_new % old_n == 0:
       out = np.repeat(h, n_new//old_n)
       if abs(float(np.sum(out)) - 0.5*n_new) < 1e-10:
          return np.clip(out,0,1)
       return project_box_sum(out,n_new)
    idx = (np.arange(n_new) * old_n) // n_new
    idx = np.minimum(idx, old_n - 1)
    out = h[idx]
    return project_box_sum(out,n_new)

def pair_eval_max(plus, minus, cost0, Vplus, Vminus, cross, hi, hj, deltas):
    g = deltas.size
    D = deltas.reshape(g,1)
    D2 = D * D
    cross_term = cross * D2
    term1 = plus + D * Vplus + cross_term
    term2 = minus + D * Vminus + cross_term

    rowmax = np.max(rowmax, axis=1)
    v0 = cost0 - 2.0 * deltas * (hi - hj) - 2.0 * deltas * deltas

    idx = int(np.argmin(rowmax))
    return float(rowmax[idx]), idx

import time

def project_box_sum(x, n):
    n = int(n)
    x = np.asarray(x, dtype=float).reshape(-1).copy()
    if x.size != n:
        x = x[:n]
        while x.size < n:
            x = np.concatenate((x, np.zeros(n - x.size, dtype=float)))
    x = np.clip(x, 0.0, 1.0)
    target = 0.5 * n
    s = float(np.sum(x))
    if abs(s - target) < 1e-10:
        diff = target - s
        if abs(diff) > 1e-12:
            if diff > 0:
                idx = int(np.argmax(1.0 - x))
                x[idx] = min(1.0, x[idx] + diff)
            else:
                idx = int(np.argmax(x))
                x[idx] = max(0.0, x[idx] + diff)
        return np.clip(x, 0.0, 1.0)

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x - hi, 0.0, 1.0)
    diff = target - float(np.sum(h))

    if diff > 1e-12:
        order = np.argsort(1.0 - h)[::-1]
        for idx in order:
            if diff <= 1e-14:
                break

            if add > 0.0:
                h[idx] += add
                diff -= add
    elif diff < -1e-12:
        order = np.argsort(h)[::-1]
        for idx in order:
            if diff >= -1e-14:

    h = np.clip(h, 0.0, 1.0)

def compute_c5_exact(h):

    n = h.size
    if n == 0:

def compute_costs_direct(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        cost0 = float(np.dot(h, 1.0 - h)) if n > 0 else 0.0
        return np.empty(0, dtype=float), np.empty(0, dtype=float), cost0, cost0

    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]
    plus = np.empty(n - 1, dtype=float)
    minus = np.empty(n - 1, dtype=float)

    for k in range(1, n):
        a = h[:-k]
        b = h[k:]
        ab = float(np.dot(a, b))
        plus[k - 1] = (total - prefix[k]) - ab
        minus[k - 1] = prefix[n - k] - ab

    cost0 = float(total - np.dot(h, h))
    raw = float(max(cost0, float(np.max(plus)), float(np.max(minus))))
    return plus, minus, cost0, raw


def ensure_zero(deltas, dlo, dhi):
    if deltas.size == 0:
        return deltas
    if dlo < 0.0 < dhi:
        if abs(float(deltas[0])) < 1e-13 or abs(float(deltas[-1])) < 1e-13:
            return deltas
        pos = int(np.searchsorted(deltas, 0.0))
        if pos < deltas.size and abs(float(deltas[pos])) < 1e-13:
            return deltas
        if pos > 0 and abs(float(deltas[pos - 1])) < 1e-13:
            return deltas
        return np.insert(deltas, pos, 0.0)
    return deltas


def make_deltas(dlo, dhi, size):
    if dhi - dlo < 1e-14:
        return np.empty(0, dtype=float)
    deltas = np.linspace(dlo, dhi, max(2, int(size)))
    return ensure_zero(deltas, dlo, dhi)


def pair_eval_max(plus, minus, cost0, vplus, vminus, cross, hi, hj, deltas):
    deltas = np.asarray(deltas, dtype=float).reshape(-1)
    g = deltas.size
    if g == 0:
        return float("inf"), 0
    D = deltas.reshape(g, 1)
    D2 = D * D
    cross_term = cross * D2
    term1 = plus + D * vplus + cross_term
    term2 = minus + D * vminus + cross_term
    rowmax = np.maximum(term1, term2)
    rowmax = np.max(rowmax, axis=1)
    v0 = cost0 - 2.0 * deltas * (hi - hj) - 2.0 * deltas * deltas
    rowmax = np.maximum(rowmax, v0)
    idx = int(np.argmin(rowmax))
    return float(rowmax[idx]), idx

def try_pair_move(h, plus, minus, cost0, max_raw, i, j, grid_size, idx_arr):
    old_max = float(max_raw)
    n = h.size
    hi = float(h[i])
    hj = float(h[j])

    if j > 0:
        a[:j] -= h[:j][::-1]

    vplus = suf - a
    vminus = pref - a

    newm, bidx = pair_eval_max(plus, minus, cost0, vplus, vminus, cross, hi, hj, deltas)

    chosen = 0.0
    if newm < old_max - 1e-12 and abs(float(deltas[bidx])) > 1e-12:
        step = (dhi - dlo) / max(1, int(deltas.size) - 1)
        center = float(deltas[bidx])
        lo_r = max(dlo, center - 2.0 * step)
        hi_r = min(dhi, center + 2.0 * step)
        deltas2 = np.linspace(lo_r, hi_r, 11)
        deltas2 = ensure_zero(deltas2, lo_r, hi_r)
        if deltas2.size >= 2:
            newm2, bidx2 = pair_eval_max(plus, minus, cost0, vplus, vminus, cross, hi, hj, deltas2)
            if newm2 < newm - 1e-14:
                newm = newm2
                deltas = deltas2
                bidx = bidx2
        chosen = float(deltas[bidx])

    if newm < old_max - 1e-12 and abs(chosen) > 1e-12:
        delta = chosen
        d2 = delta * delta
        h[i] += delta
        h[j] -= delta
        plus += delta * vplus + cross * d2
        minus += delta * vminus + cross * d2
        cost0 = cost0 - 2.0 * delta * (hi - hj) - 2.0 * d2
        if cost0 < 0.0 and cost0 > -1e-10:
            cost0 = 0.0

        actual = max(float(np.max(plus)), float(np.max(minus)), float(cost0))
        if actual >= old_max - 1e-11:
            h[i] -= delta
            h[j] += delta
            plus -= delta * vplus + cross * d2
            minus -= delta * vminus + cross * d2
            cost0 += 2.0 * delta * (hi - hj) + 2.0 * d2
            if cost0 < 0.0 and cost0 > -1e-10:
                cost0 = 0.0
            return old_max, False, 0.0

        return actual, True, delta

    return old_max, False, 0.0


def pair_search(h_start, n, time_limit, rng):
    n = int(n)
    if n < 2 or time_limit < 1.0:
        h = enforce_sum(h_start)
        c, _ = compute_c5_exact(h)
        return h, float(c), n

    t_end = time.perf_counter() + max(1.0, float(time_limit))
    h = project_box_sum(np.asarray(h_start, dtype=float), n)
    plus, minus, cost0, max_raw = compute_costs_direct(h)
    best_h = h.copy()
    best_raw = float(max_raw)

    recompose_freq = 250 if n <= 120 else 500 if n <= 200 else 1000
    recompute_count = 0
    noise_std = 0.08
    noise_max = 0.35
    noise_incr = 1.35

    if n <= 90:
        grid_size = 81
    elif n <= 160:
        grid_size = 49
    else:
        grid_size = 33

    batch_size = int(max(80, min(600, 40000.0 / max(n, 1))))
    idx_arr = np.arange(n - 1, dtype=float)

    full_passes = 0
    if n <= 90 and t_end - time.perf_counter() > 60.0:
        full_passes = 2
    elif n <= 140 and t_end - time.perf_counter() > 120.0:
        full_passes = 1

    stall_batches = 0

    while time.perf_counter() < t_end:
        if full_passes > 0:
            passed = False
            stop_outer = False
            for i in range(n - 1):
                for j in range(i + 1, n):
                    if time.perf_counter() >= t_end:
                        stop_outer = True
                        break
                    newm, imp, _ = try_pair_move(h, plus, minus, cost0, max_raw, i, j, grid_size, idx_arr)
                    if imp:
                        passed = True
                        max_raw = newm
                        recompute_count += 1
                        if max_raw < best_raw - 1e-12:
                            best_raw = max_raw
                            best_h = h.copy()
                        if recompute_count >= recompose_freq:
                            plus, minus, cost0, max_raw = compute_costs_direct(h)
                            recompute_count = 0
                            if max_raw < best_raw - 1e-12:
                                best_raw = max_raw
                                best_h = h.copy()
                if stop_outer:
                    break

            if stop_outer:
                break

            full_passes -= 1
            if not passed:
                if t_end - time.perf_counter() > 30.0:
                    h = project_box_sum(best_h + rng.normal(0.0, noise_std, n), n)
                    plus, minus, cost0, max_raw = compute_costs_direct(h)
                    if max_raw < best_raw - 1e-12:
                        best_raw = max_raw
                        best_h = h.copy()
                    noise_std = min(noise_max, noise_std * noise_incr + 0.01)
                else:
                    break

        else:
            if t_end - time.perf_counter() <= 0.0:
                break
            batch = make_batch(h, plus, minus, cost0, max_raw, n, rng, batch_size)
            batch_imp = False
            for i, j in batch:
                if time.perf_counter() >= t_end:
                    break
                newm, imp, _ = try_pair_move(h, plus, minus, cost0, max_raw, i, j, grid_size, idx_arr)
                if imp:
                    batch_imp = True
                    max_raw = newm
                    recompute_count += 1
                    if max_raw < best_raw - 1e-12:
                        best_raw = max_raw
                        best_h = h.copy()
                    if recompute_count >= recompose_freq:
                        plus, minus, cost0, max_raw = compute_costs_direct(h)
                        recompute_count = 0
                        if max_raw < best_raw - 1e-12:
                            best_raw = max_raw
                            best_h = h.copy()

            if not batch_imp:
                stall_batches += 1
                if stall_batches >= 3:
                    if t_end - time.perf_counter() > 25.0:
                        h = project_box_sum(best_h + rng.normal(0.0, noise_std, n), n)
                        plus, minus, cost0, max_raw = compute_costs_direct(h)
                        if max_raw < best_raw - 1e-12:
                            best_raw = max_raw
                            best_h = h.copy()
                        noise_std = min(noise_max, noise_std * noise_incr + 0.01)
                        stall_batches = 0
                    else:
                        break
            else:
                stall_batches = 0

    best_h = enforce_sum(best_h)
    c, _ = compute_c5_exact(best_h)
    return best_h, float(c), n


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.perf_counter()
    budget_s = float(budget_s)
    rng = np.random.default_rng(int(seed))

    try:
        parent = globals().get("initial_h_values")
        candidates = []
        default_n = 70

        if parent is not None:
            arr = np.asarray(parent, dtype=float)
            if arr.size > 0:
                arr = project_box_sum(arr, arr.size)
                c, _ = compute_c5_exact(arr)
                candidates.append((arr.copy(), float(c), int(arr.size)))

        if not candidates:
            arr = random_binary_h(default_n, rng)
            c, _ = compute_c5_exact(arr)
            candidates.append((arr.copy(), float(c), default_n))

        best_h = candidates[0][0].copy()
        best_n = int(candidates[0][2])
        best_c = float(candidates[0][1])

        base_n = max(2, best_n)
        resolutions = []
        for m in (1, 2, 3, 4):
            nn = base_n * m
            if nn < 2:
                continue
            if nn > 280:
                break
            if nn not in resolutions:
                resolutions.append(nn)
        if not resolutions:
            resolutions = [base_n]

        if len(resolutions) == 4:
            weights = [0.32, 0.28, 0.24, 0.16]
        elif len(resolutions) == 3:
            weights = [0.35, 0.32, 0.33]
        elif len(resolutions) == 2:
            weights = [0.55, 0.45]
        else:
            weights = [1.0]

        total_alloc = max(60.0, budget_s - 20.0)

        for idx, n_res in enumerate(resolutions):
            elapsed = time.perf_counter() - t0
            remaining = budget_s - elapsed - 10.0
            if remaining < 35.0:
                break
            slice_s = int(total_alloc * weights[idx])
            if slice_s > remaining - 5.0:
                slice_s = int(max(10.0, remaining - 5.0))
            if slice_s < 20:
                slice_s = int(max(10.0, remaining - 5.0))
            if slice_s <= 0:
                break

            if best_h.size == n_res:
                h_start = best_h.copy()
            else:
                h_start = resample_nearest(best_h, n_res)

            h_ret, c_ret, _ = pair_search(h_start, n_res, slice_s, rng)
            if float(c_ret) < best_c - 1e-12:
                best_h = np.asarray(h_ret, dtype=float).copy()
                best_n = int(n_res)
                best_c = float(c_ret)

        elapsed = time.perf_counter() - t0
        remaining = budget_s - elapsed - 10.0
        if remaining > 60.0:
            h_ret, c_ret, _ = pair_search(best_h.copy(), best_n, int(remaining * 0.8), rng)
            if float(c_ret) < best_c - 1e-12:
                best_h = np.asarray(h_ret, dtype=float).copy()
                best_n = int(best_n)
                best_c = float(c_ret)

        elapsed = time.perf_counter() - t0
        remaining = budget_s - elapsed - 10.0
        if remaining > 220.0:
            alt_n = int(min(140, max(base_n, 70)))
            kinds = [1, 0, 2]
            for kind in kinds:
                elapsed = time.perf_counter() - t0
                remaining = budget_s - elapsed - 10.0
                if remaining < 45.0:
                    break
                slice_s = int(min(80, remaining * 0.35))
                if slice_s <= 0:
                    break
                h_start = template_start(alt_n, kind, rng)
                h_ret, c_ret, _ = pair_search(h_start, alt_n, slice_s, rng)
                if float(c_ret) < best_c - 1e-12:
                    best_h = np.asarray(h_ret, dtype=float).copy()
                    best_n = int(alt_n)
                    best_c = float(c_ret)

        final_h = enforce_sum(best_h)
        final_c, final_n = compute_c5_exact(final_h)
        return final_h, float(final_c), int(final_n)

    except Exception:
        parent = globals().get("initial_h_values")
        if parent is not None:
            h = np.asarray(parent, dtype=float)
            if h.size > 0:
                h = project_box_sum(h, h.size)
                c, n = compute_c5_exact(h)
                return h, float(c), int(n)
        h = random_binary_h(70, np.random.default_rng(int(seed)))
        c, n = compute_c5_exact(h)
        return h, float(c), int(n)
