import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_objective_and_grad(h, n, alpha):
    """Compute soft-max approximation of max(conv*dx) and its gradient."""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft max
    m = np.max(conv)
    shifted = alpha * (conv - m)
    e = np.exp(shifted)
    s = e.sum()
    val = m + np.log(s) / alpha
    w = e / s
    # Gradient via correlation trick
    grad = dx * np.convolve(1.0 - 2.0 * h, w[::-1], mode='full')[n-1:2*n-1]
    return val, grad
