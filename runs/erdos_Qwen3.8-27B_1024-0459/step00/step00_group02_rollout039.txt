    dx = 2.0 / n

def compute_objective(h):

    return np.max(compute_corr(h))

def project_feasible(x, n):
    """Project x onto {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0
    # Binary search for lambda such that sum(clip(x - lambda, 0, 1)) = target
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0, 1)
    # Fine-tune: adjust to get exactly sum = target
    # Add small correction
    diff = target - h.sum()
    if diff > 0:
        # Need to increase some entries
        mask = h < 1
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    else:
        mask = h > 0
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def compute_subgradient(h, n, active_indices):
    """Compute subgradient of max corr at active indices."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in active_indices:
        d = m - (n - 1)
        # grad[j] += (1-h[j+d]) * 1_{0<=j+d<n} + h[j-d] * 1_{0<=j-d<n}
        if d >= 0:
            # h[j+d] is valid for j = 0, ..., n-1-d
            # Term 1: (1-h[j+d]) for j = 0, ..., n-1-d
            grad[:n-d] += (1 - h[d:])
            # Term 2: h[j-d] for j = d, ..., n-1
            grad[d:] += h[:n-d]
        else:

            # Term 2: h[j+d_abs] for j = 0, ..., n-1-d_abs
            grad[:n-d_abs] += h[d_abs:]
