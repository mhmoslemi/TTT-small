def compute_gradient(h, top_k=50):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    
    # Find active lags (top_k largest values)

    max_val = np.max(corr)

    active = np.where(corr >= threshold)[0]

        # ∂c_k/∂h[m] = (1-h[m-k])[0≤m-k<n] - h[m+k][0≤m+k<n]

            if 0 <= m-k < n:
                g += (1.0 - h[m-k])
            if 0 <= m+k < n:
                g -= h[m+k]
            grad[m] += g

    return np.max(corr) * dx

def compute_gradient(h, n_active=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Find lags close to max
    max_val = np.max(corr)
    # Use top n_active lags or those within 5% of max
    threshold = 0.95 * max_val
    active_mask = corr >= threshold
    active_indices = np.where(active_mask)[0]
    
    if len(active_indices) > n_active:
        # Select top n_active
        top_idx = np.argsort(-corr)[:n_active]
        active_indices = top_idx
    
    grad = np.zeros(n)
    
    for k_idx in active_indices:
        k = k_idx - (n - 1)
        
        # First term: (1-h[m-k]) for m where 0≤m-k<n
        # m range: [max(0,k), min(n, k+n))
        ms = max(0, k)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def upscale_h(h_parent, n_target):
    n_parent = len(h_parent)
    if n_parent == n_target:
        return h_parent.copy()
    x_old = np.arange(n_parent, dtype=float)
    x_new = np.arange(n_target, dtype=float) * (n_parent / n_target)
    h_new = np.interp(x_new, x_old, h_parent)
    h_new = np.clip(h_new, 0.0, 1.0)
    target_sum = n_target / 2.0
    _project_sum(h_new, target_sum)
    return h_new

def _project_sum(h, target_sum):
    for _ in range(50):
        deficit = target_sum - np.sum(h)
        if abs(deficit) < 1e-12:
            break
        if deficit > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        if not np.any(mask):
            break
        per = deficit / np.sum(mask)
        h[mask] += per
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_obj(h, tau, penalty):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    dx = 2.0 / n
    obj = (c_max + tau * np.log(np.sum(np.exp((corr - c_max) / tau)))) * dx
    sum_h = np.sum(h)
    obj += penalty * (sum_h - n / 2.0) ** 2
    return obj

def smooth_grad(h, tau, penalty):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    weights = np.exp((corr - c_max) / tau)
    weights /= weights.sum()
    dx = 2.0 / n
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        ms = max(0, k)
        me = min(n, k + n)
        if ms < me:
            grad[ms:me] += w * (1.0 - h[ms - k:me - k])
        ms2 = max(0, -k)
        me2 = min(n, n - k)
        if ms2 < me2:
            grad[ms2:me2] -= w * h[ms2 + k:me2 + k]
    
    grad *= dx
    constraint_val = penalty * (np.sum(h) - n / 2.0)
    grad += 2.0 * constraint_val
    return grad

def vectorized_local_search(h, max_time=300, max_sweeps=50):

def _update_effect(effects, m, h, n):
    n_corr = 2 * n - 1

def round_to_binary(h, n):
    target_sum = n / 2.0
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:int(target_sum)]] = 1.0
    # If target_sum is not integer (shouldn't be for even n), handle
    remainder = target_sum - np.sum(h_bin)
    if abs(remainder) > 1e-10:
        # There's a fractional element
        next_idx = sorted_idx[int(target_sum)]
        h_bin[next_idx] = remainder
    return h_bin

def refine_fractional(h, n, max_time=60):
    """Refine near-boundary values using local gradient descent."""
    start = time.time()
    h = h.astype(float).copy()
    target_sum = n / 2.0
    
    for _ in range(200):
        if time.time() - start > max_time:
            break
        corr = compute_corr(h)
        c_max = np.max(corr)
        # Find active lags
        threshold = c_max * 0.98
        active = np.where(corr >= threshold)[0]
        
        # Gradient of true max (subgradient)
        grad = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)
            ms = max(0, k)
            me = min(n, k + n)
            if ms < me:
                grad[ms:me] += (1.0 - h[ms - k:me - k])
            ms2 = max(0, -k)
            me2 = min(n, n - k)
            if ms2 < me2:
                grad[ms2:me2] -= h[ms2 + k:me2 + k]
        grad /= max(len(active), 1)
        
        # Update: move against gradient, project to [0,1] and sum constraint
        lr = 0.01
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        _project_sum(h_new, target_sum)
        
        if compute_c5(h_new) <= compute_c5(h) + 1e-10:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    
    return h
