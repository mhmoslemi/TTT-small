                options={'maxiter': 100, 'maxfun': 200,
                         'ftol': 1e-10, 'gtol': 1e-7}
            )
            elapsed += time.time() - t0
            if res.fun < 1e10:
                h_try = project_feasible(res.x, n)
                v = compute_c5(h_try)
                if v < best_val:
                    best_val = v
                    best_h = h_try
        except Exception:
            elapsed += time.time() - t0

    return best_h, best_val


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    # Get initial solution
    best_h = None
    best_c5 = float('inf')

    try:
        parent = np.array(initial_h_values, dtype=float)
        n_parent = len(parent)
        best_h = project_feasible(parent, n_parent)
        best_c5 = compute_c5(best_h)
    except (NameError, UnboundLocalError, TypeError):
        # No parent available - create initial guess
        n_parent = 80
        x = np.linspace(0, 2, n_parent, endpoint=False)
        # Try a few constructions
        candidates = [
            np.full(n_parent, 0.5),
            project_feasible(np.clip(0.5 + 0.4 * np.sin(np.pi * x), 0, 1), n_parent),
            project_feasible(np.clip(0.5 + 0.3 * np.cos(np.pi * x), 0, 1), n_parent),
        ]
        best_h = min(candidates, key=compute_c5)
        best_c5 = compute_c5(best_h)

    # Stage 1: Optimize at parent resolution
    if budget_s - elapsed() > 100:
        tb = min(budget_s - elapsed() - 80, 250)
        h_opt, v_opt = optimize_at_resolution(best_h, len(best_h), tb, seed=seed)
        if v_opt < best_c5:
            best_c5 = v_opt
            best_h = h_opt

    # Stage 2: Try random perturbations at current resolution
    if budget_s - elapsed() > 80:
        tb = min(budget_s - elapsed() - 60, 120)
        h_opt, v_opt = try_random_starts(best_h, len(best_h), tb, n_tries=3, seed=seed+1)
        if v_opt < best_c5:
            best_c5 = v_opt
            best_h = h_opt

    # Stage 3: Multi-resolution refinement
    target_resolutions = [84, 120, 168, 240, 336]
    for n_target in target_resolutions:
        rem = budget_s - elapsed()
        if rem < 120:
            break
        n_curr = len(best_h)
        if n_target <= n_curr:
            continue

        h_interp = interpolate_h(best_h, n_target)
        tb = min(rem * 0.4, 300)

        h_opt, v_opt = optimize_at_resolution(h_interp, n_target, tb, seed=seed + n_target)
        if v_opt < best_c5:
            best_c5 = v_opt
            best_h = h_opt

        # Quick random restart at this resolution
        rem2 = budget_s - elapsed()
        if rem2 > 60:
            tb2 = min(rem2 * 0.3, 100)
            h_opt2, v_opt2 = try_random_starts(best_h, n_target, tb2, n_tries=2, seed=seed + n_target + 100)
            if v_opt2 < best_c5:
                best_c5 = v_opt2
                best_h = h_opt2

    # Final polish at the best resolution found
    rem = budget_s - elapsed()
    if rem > 30:
        n_final = len(best_h)
        tb = min(rem - 5, 100)
        h_opt, v_opt = optimize_at_resolution(best_h, n_final, tb, seed=seed + 999)
        if v_opt < best_c5:
            best_c5 = v_opt
            best_h = h_opt

    # Ensure final solution is valid
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
