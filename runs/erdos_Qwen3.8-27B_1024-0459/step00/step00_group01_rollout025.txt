import numpy as np
from scipy.optimize import minimize
import time


def _c5_of(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _smooth_max_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c_vals = c_raw * dx
    mx = np.max(c_vals)
    cs = c_vals - mx
    e = np.exp(beta * cs)
    se = np.sum(e)
    smax = mx + np.log(se) / beta
    w = e / se

    grad_raw = np.zeros(n)
    for m in range(2 * n - 1):
        wi = w[m]
        if wi < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            L = n - k
            if L > 0:
                grad_raw[k:] += wi * (1.0 - h[:L])
                grad_raw[:L] -= wi * h[k:]
        else:
            ev = -k
            L = n - ev
            if L > 0:
                grad_raw[:L] += wi * (1.0 - h[ev:])
                grad_raw[ev:] -= wi * h[:L]

    return smax, dx * grad_raw


def _project_sum(h, target):
    h = h.copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        nfree = np.sum(free)
        if nfree == 0:
            break
        step = diff / nfree
        h[free] = np.clip(h[free] + step, 0.0, 1.0)
    return h


def _slsqp_opt(h0, n, beta, maxiter=300):
    dx = 2.0 / n

    def obj(h):
        v, _ = _smooth_max_obj_grad(h, beta)
        return v

    def grad(h):
        _, g = _smooth_max_obj_grad(h, beta)
        return g

    def cfun(h):
        return np.sum(h) - n / 2.0

    def cjac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_out = np.clip(res.x, 0.0, 1.0)
        h_out = _project_sum(h_out, n / 2.0)
        return h_out
    except Exception:
        return None


def _pgd_opt(h0, n, beta, max_iter=3000, lr=0.005):
    h = _project_sum(h0.copy(), n / 2.0)
    best_h = h.copy()
    best_c5 = _c5_of(h)
    for i in range(max_iter):
        _, g = _smooth_max_obj_grad(h, beta)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_sum(h_new, n / 2.0)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if i % 500 == 499:
            lr *= 0.7
    return best_h


def _interp_to(h_src, n_dst):
    n_src = len(h_src)
    xs = np.linspace(0.0, 2.0, n_src + 1)
    xd = np.linspace(0.0, 2.0, n_dst)
    xd = xd + 1.0 / (n_dst * 2)
    h_dst = np.interp(xd, xs, np.array([h_src[0]] + list(h_src)),
                      left=h_src[0], right=h_src[-1])
    return h_dst


def _bin_local_search(h_bin, n, max_rounds=50):
    h = h_bin.copy()
    c5_val = _c5_of(h)
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5_try = _c5_of(h)
                if c5_try < c5_val - 1e-10:
                    c5_val = c5_try
                    improved = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, c5_val


def _make_bin(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 0:
        za = np.where(h == 0)[0]
        n_flip = min(int(np.ceil(diff)), len(za))
        h[rng.choice(za, n_flip, replace=False)] = 1.0
    elif diff < 0:
        oa = np.where(h == 1)[0]
        n_flip = min(int(np.ceil(-diff)), len(oa))
        h[rng.choice(oa, n_flip, replace=False)] = 0.0
    h = _project_sum(h, target)
    return h
