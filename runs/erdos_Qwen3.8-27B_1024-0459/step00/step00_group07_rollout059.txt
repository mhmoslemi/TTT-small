```python
import numpy as np
import time

def evaluate_c5(h):
    """Compute C5 = max_k correlate(h, 1-h, 'full')[k] * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h):
    """Convert to binary maintaining sum = n/2"""
    n = len(h)
    target = n // 2
    # Sort by value, threshold
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def local_search(h, time_limit):
    """Binary local search with random swaps"""
    n = len(h)
    h = h.copy()
    t0 = time.time()
    
    current_c5 = evaluate_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    # Simulated annealing parameters
    T = 0.01  # Initial "temperature"
    T_min = 1e-6
    cooling = 0.999
    
    no_improve = 0
    max_no_improve = 2000
    
    while time.time() - t0 < time_limit and no_improve < max_no_improve:
        no_improve += 1
        
        # Pick a random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        if i == j:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = evaluate_c5(h)
        
        delta = new_c5 - current_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            current_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        # Cool
        T = max(T * cooling, T_min)
    
    # Final greedy pass
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        for i in range(n):
            if h[i] == 1.0:
                for j in range(n):
                    if h[j] == 0.0:
                        h[i] = 0.0
                        h[j] = 1.0
                        new_c5 = evaluate_c5(h)
                        if new_c5 < current_c5:
                            current_c5 = new_c5
                            improved = True
                        else:
                            h[i] = 1.0
                            h[j] = 0.0
    
    return h, current_c5
```

```python
    """Compute C5 = max_k np.correlate(h, 1-h, 'full')[k] * dx"""

    """Upscale parent to resolution n"""
    n_par = len(parent)
    if n == n_par:
        return parent.copy()
    scale = n / n_par
    idx = np.minimum((np.arange(n) * scale).astype(int), n_par - 1)
    h = parent[idx].copy()
    _fix_sum(h, n)
    return h

def _fix_sum(h, n):
    """Ensure sum(h) = n/2 by adjusting"""
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    # Simple proportional fix
    h = h * (target / s)
    h = np.clip(h, 0, 1)
    # Fine fix
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute
        idx = np.argsort(h)
        step = 1.0 / n
        remaining = abs(diff)
        if diff > 0:
            for i in range(len(idx)):
                j = idx[len(idx) - 1 - i]
                add = min(step, 1.0 - h[j], remaining)
                h[j] += add
                remaining -= add
                if remaining < 1e-12:
                    break
        else:
            for i in range(len(idx)):
                j = idx[i]
                sub = min(step, h[j], remaining)
                h[j] -= sub
                remaining -= sub
                if remaining < 1e-12:
                    break
    return h
```
