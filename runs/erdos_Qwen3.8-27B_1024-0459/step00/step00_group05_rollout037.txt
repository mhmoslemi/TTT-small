import numpy as np

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    for _ in range(40):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            cap = 1.0 - h
            tc = np.sum(cap)
            if tc < 1e-13:
                break
            h = h + (diff / tc) * cap
        else:
            th = np.sum(h)
            if th < 1e-13:
                break
            h = h + (diff / th) * h
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample_linear(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    if new_n % old_n == 0:
        r = new_n // old_n
        return np.repeat(h, r)
    x_old = np.arange(float(old_n))
    x_new = np.linspace(0.0, float(old_n), new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def downsample_linear(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    if old_n % new_n == 0:
        r = old_n // new_n
        return h.reshape(new_n, r).mean(axis=1)
    x_old = np.arange(float(old_n))
    x_new = np.linspace(0.0, float(old_n), new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def make_move(h, n, rng, move_type):
    h_new = h.copy()
    if move_type == 0:
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            d = rng.uniform(-0.05, 0.05)
            h_new[i] += d
            h_new[j] -= d
    elif move_type == 1:
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            d = rng.uniform(-0.3, 0.3)
            h_new[i] += d
            h_new[j] -= d
    elif move_type == 2:
        start = rng.randint(n)
        length = rng.randint(3, max(4, n // 6))
        for p in range(length):
            idx = (start + p) % n
            h_new[idx] += rng.uniform(-0.03, 0.03)
    elif move_type == 3:
        shift = rng.randint(1, max(2, n // 12))
        h_new = np.roll(h, shift)
    elif move_type == 4:
        h_new = h + rng.normal(0, 0.02, n)
    else:
        k = rng.randint(2, 6)
        idx = rng.choice(n, size=k, replace=False)
        d = rng.uniform(-0.08, 0.08, k)
        d = d - d.mean()
        h_new[idx] += d
    return h_new


def fix_constraints(h, n):

    """Polish using subgradient of the max correlation."""
    cur_h = h.copy()
    cur_c5 = compute_c5(h, n)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    step_size = 0.05

        elapsed = time.time() - t0
        if budget_s - elapsed < min_time_left:
            break

        conv = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        k_star = int(np.argmax(conv))

        # Build subgradient direction
        grad = np.zeros(n)
        # c[k_star] = sum_j h[j + k_star] * (1 - h[j]) for valid j
        # 
        # For k_star >= 0 (laid out as index n-1+k_star in full correlation):
        # Actually, in np.correlate(a, v, "full"), the output at index p is:
        # sum_n a[n] * v[n - p + (len(v)-1)]  ... this is getting confusing.
        # Let me just use the fact that we want to decrease c[k_star].
        #
        # Simpler: just do a small perturbation biased against the worst lag.
        # Find positions that contribute most to c[k_star].
        
        # Identify the lag offset
        # In "full" mode, output[p] = sum_i a[i] * v[i - p + (len(v)-1)]
        # With a=h, v=1-h, len(v)=n:
        # output[p] = sum_i h[i] * (1-h[i-p+n-1])
        # Let q = p - (n-1), so output[n-1+q] = sum_i h[i] * (1-h[i-q])
        # when i-q is in [0, n-1]
        
        q = k_star - (n - 1)
        # q >= 0 means we're looking at h[i] * (1-h[i-q])
        # q < 0 means we're looking at h[i] * (1-h[i+|q|])
        
        if q >= 0:
            # c = sum_{i=q}^{n-1} h[i] * (1 - h[i-q])
            for i in range(q, n):
                # d(c)/d(h[i]) = (1-h[i-q]) - h[i-q] = 1 - 2*h[i-q]  (from first term)
                #                           - h[i-q]  (from second term where i appears as i-q... no)
                # Wait: c = sum_{i=q}^{n-1} h[i] - sum_{i=q}^{n-1} h[i]*h[i-q]
                # d(c)/d(h[i]) = 1 - 2*h[i-q] for i in [q, n-1]
                # But also h[i] appears in the product h[i]*h[i-q] as the first factor,
                # and in h[i+q]*h[i] as the second factor (when i+q < n)
                # So: d(c)/d(h[i]) = 1 - h[i-q] - h[i-q] ... no
                # Let me be more careful:
                # c = sum_{i=q}^{n-1} h[i] * (1 - h[i-q])
                #   = sum_{i=q}^{n-1} h[i] - sum_{i=q}^{n-1} h[i]*h[i-q]
                # d(c)/d(h[i]) for a specific i:
                #   From sum h[i]: contributes 1
                #   From -sum h[i]*h[i-q]: contributes -h[i-q] (derivative wrt h[i] as first factor)
                #   Also, if i-q >= 0 and i is in range [q, n-1] as the "second factor"
                #   h[j]*h[j-q] where j-q = i, i.e., j = i+q, and j must be in [q, n-1]
                #   So if i+q < n and i >= 0: contributes -h[i+q] (derivative wrt h[i] as second factor)
                # Combined: d(c)/d(h[i]) = 1 - h[i-q] - (h[i+q] if i+q < n and i >= 0 else 0)
                
                if i - q < n:
                    grad[i] += (1.0 - h_val_safe(cur_h, i - q))
                if i + q < n and i >= 0:
                    grad[i] -= h_val_safe(cur_h, i + q)
        else:
            q_abs = -q
            # c = sum_{i=0}^{n-1-q_abs} h[i] * (1 - h[i+q_abs])
            # d(c)/d(h[i]) = 1 - h[i+q_abs] - (h[i-q_abs] if i>=q_abs else 0) ... 
            # c = sum_{i=0}^{n-1-q_abs} h[i] - sum_{i=0}^{n-1-q_abs} h[i]*h[i+q_abs]
            for i in range(n - q_abs):
                grad[i] += 1.0 - h_val_safe(cur_h, i + q_abs)
                if i - q_abs >= 0:
                    grad[i] -= h_val_safe(cur_h, i - q_abs)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break

        # Move in negative gradient direction (to decrease c[k_star])
        direction = -grad / gnorm
        cur_h = cur_h + step_size * direction
        cur_h = fix_constraints(cur_h, n)
        c5_new = compute_c5(cur_h, n)

                best_h = cur_h.copy()
        else:
            step_size *= 0.7

        if step_size < 0.001:
            break

    return best_h, n, best_c5
