           free = p < 1.0

           free = p > 0.0

        for _ in range(60):
            mid = 0.5 * (lo + hi)
            s = float(np.sum(np.clip(p + mid, 0.0, 1.0)))
            if s < target:
                lo = mid
            else:
                hi = mid
        lam = 0.5 * (lo + hi)
        p = np.clip(p + lam, 0.0, 1.0)

    p = enforce_sum_only(p, target)
    return p


def prepare_feasible(h0, n):
    h = np.asarray(h0, dtype=float).copy()
    n = int(h.size)
    target = 0.5 * float(n)
    if np.all(h >= -1e-9) and np.all(h <= 1.0 + 1e-9) and abs(float(np.sum(h)) - target) <= 1e-7:
        h = np.clip(h, 0.0, 1.0)
        h = enforce_sum_only(h, target)
    else:
        h = project_box_sum(h, n)
    return h


def counts_ab(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    N = get_fft_len(n)

    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real
    I = ac[:n]
    I = np.where(I < 0.0, 0.0, I)

    csum = np.empty(n + 1, dtype=float)
    csum[0] = 0.0
    np.cumsum(h, out=csum[1:])
    total = csum[n]

    pre = csum[n:0:-1]
    suff = total - csum[:n]

    A = pre - I
    B = suff - I

    A = np.where(A < 0.0, 0.0, A)
    B = np.where(B < 0.0, 0.0, B)
    return A, B


def max_count_objective(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n <= 96:
        full = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(full))
    A, B = counts_ab(h)
    return float(max(np.max(A), np.max(B)))


def c5_exact(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    full = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(full) * dx)


def add_grad_A(g, s, h, w):
    n = int(h.size)
    if s == 0:
        g += w * (1.0 - 2.0 * h)
    else:
        g[:n - s] += w
        g[s:] -= w * h[:n - s]


def add_grad_B(g, s, h, w):
    n = int(h.size)
    if s == 0:
        g += w * (1.0 - 2.0 * h)
    else:
        g[s:] += w
        g[:n - s] -= w * h[s:]


def linear_direction(g, n):
    n = int(n)
    target = 0.5 * float(n)
    order = np.argsort(g, kind="mergesort")
    v = np.zeros(n, dtype=float)
    k = int(math.floor(target + 1e-15))
    if k > 0:
        v[order[:k]] = 1.0
    frac = target - float(k)
    if frac > 1e-12 and k < n:
        v[order[k]] = frac
    return v


def metropolis_accept(delta, temperature, rng):
    delta = float(delta)
    temperature = float(temperature)
    if delta <= 1e-12:
        return True
    if temperature <= 1e-12:
        return False
    if delta > 50.0 * temperature:
        return False
    p = math.exp(-delta / temperature)
    return float(rng.random()) < p


def random_binary(n, rng):
    n = int(n)
    v = np.zeros(n, dtype=float)
    if n % 2 == 0:
        k = n // 2
        if k > 0:
            idx = rng.choice(n, size=k, replace=False)
            v[idx] = 1.0
    else:
        k = n // 2
        if k > 0:
            idx = rng.choice(n, size=k, replace=False)
            v[idx] = 1.0
        rem = np.where(v == 0.0)[0]
        if rem.size > 0:
            v[int(rng.choice(rem.size))] = 0.5
    return project_box_sum(v, n)


def random_smooth(n, rng):
    n = int(n)
    x = np.arange(n, dtype=float)
    v = np.zeros(n, dtype=float)
    for _ in range(3):
        freq = rng.uniform(1.0, max(2.0, 0.25 * float(n)))
        amp = rng.uniform(0.3, 1.0)
        phase = rng.uniform(0.0, 2.0 * math.pi)
        v += amp * np.sin(2.0 * math.pi * freq * x / float(n) + phase)
    v = 0.5 + 0.45 * v
    return project_box_sum(v, n)


def upsample_h(h, nnew):
    n = int(h.size)
    nnew = int(nnew)
    if nnew == n:
        return project_box_sum(h, nnew)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    y = (np.arange(nnew, dtype=float) + 0.5) * (2.0 / float(nnew))
    v = np.interp(y, x, h, left=float(h[0]), right=float(h[-1]))
    return project_box_sum(v, nnew)


def candidate_from_best(best_h, nnew):
    n = int(best_h.size)
    nnew = int(nnew)
    if nnew == n:
        return project_box_sum(best_h, nnew)
    if nnew > n and nnew % n == 0:
        v = np.tile(best_h, nnew // n)
        return project_box_sum(v, nnew)
    if nnew < n and n % nnew == 0:
        blocks = best_h.reshape(nnew, n // nnew)
        v = np.mean(blocks, axis=1)
        return project_box_sum(v, nnew)
    return upsample_h(best_h, nnew)


def polish_local(h0, time_limit, seed):
    n = int(h0.size)
    if n < 2:
        h = prepare_feasible(h0, n)
        return h, max_count_objective(h)

    time_limit = float(time_limit)
    if time_limit <= 0.0:
        h = prepare_feasible(h0, n)
        return h, max_count_objective(h)

    rng = np.random.default_rng(int(seed))
    t0 = time.time()

    h = prepare_feasible(h0, n)
    cur = max_count_objective(h)
    best_h = h.copy()
    best_cur = cur

    gamma_list = (1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.03, 0.015, 0.008)
    step_list = (0.12, 0.06, 0.03, 0.015, 0.008, 0.004)

    temperature = max(1e-4, 0.01 * max(1.0, best_cur) + 0.005)
    sigma = 0.15
    no_improve = 0
    it = 0
    max_iters = 1000000

    while (time.time() - t0) < time_limit and it < max_iters:
        it += 1

        A, B = counts_ab(h)
        mc = max(float(np.max(A)), float(np.max(B)))

        if mc < cur:
            cur = mc

        if mc < best_cur - 1e-12:
            best_cur = mc
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        tol = max(1e-4, 0.002 * max(1.0, mc) + 1e-4)
        if no_improve > 2000:
            tol = max(tol, 0.02)

        g = np.zeros(n, dtype=float)
        sw = 0.0

        active_A = np.where(A >= mc - tol)[0]
        for s in active_A:
            s = int(s)
            val = float(A[s])
            w = 1.0 if val >= mc else math.exp((val - mc) / tol)
            if w > 1e-8:
                add_grad_A(g, s, h, w)
                sw += w

        active_B = np.where(B >= mc - tol)[0]
        for s in active_B:
            s = int(s)
            if s == 0:
                continue
            val = float(B[s])
            w = 1.0 if val >= mc else math.exp((val - mc) / tol)
            if w > 1e-8:
                add_grad_B(g, s, h, w)
                sw += w

        moved = False

        if sw > 0.0:
            g *= (1.0 / sw)
            ng = float(np.linalg.norm(g))
        else:
            ng = 0.0

        if ng > 1e-12:
            gp = g + rng.normal(0.0, 1e-10, n)
            v = linear_direction(gp, n)

            cand_h = None
            cand_c = math.inf
            for gamma in gamma_list:
                if time.time() - t0 > time_limit:
                    break
                h2 = (1.0 - gamma) * h + gamma * v
                c2 = max_count_objective(h2)
                if c2 < cand_c:
                    cand_c = c2
                    cand_h = h2

            if cand_h is not None:
                if cand_c < best_cur or metropolis_accept(cand_c - cur, temperature, rng):
                    h = cand_h.copy()
                    cur = cand_c
                    if cur < best_cur:
                        best_cur = cur
                        best_h = h.copy()
                        no_improve = 0
                    moved = True
                    if cand_c > best_cur:
                        temperature = max(1e-4, temperature * 0.92)
                    else:
                        temperature = max(1e-4, temperature * 0.985)

        if (not moved) and ng > 1e-12:
            g_unit = g / ng
            cand_h = None
            cand_c = math.inf
            for step in step_list:
                if time.time() - t0 > time_limit:
                    break
                h2 = project_box_sum(h - step * g_unit, n)
                c2 = max_count_objective(h2)
                if c2 < cand_c:
                    cand_c = c2
                    cand_h = h2

            if cand_h is not None:
                if cand_c < best_cur or metropolis_accept(cand_c - cur, temperature, rng):
                    h = cand_h.copy()
                    cur = cand_c
                    if cur < best_cur:
                        best_cur = cur
                        best_h = h.copy()
                        no_improve = 0
                    moved = True
                    if cand_c > best_cur:
                        temperature = max(1e-4, temperature * 0.92)
                    else:
                        temperature = max(1e-4, temperature * 0.985)

        if not moved:
            if no_improve > 400:
                temperature = max(1e-4, temperature * 0.90)
            if no_improve > 200:
                sigma = max(0.03, sigma * 0.95)
            elif no_improve < 80:
                sigma = min(0.30, sigma * 1.02)

            k = max(1, int(0.25 * n))
            h2 = h.copy()
            idx = rng.integers(0, n, size=k)
            h2[idx] += rng.normal(0.0, sigma, size=k)
            h2 = project_box_sum(h2, n)
            c2 = max_count_objective(h2)

            if c2 < best_cur or metropolis_accept(c2 - cur, temperature, rng):
                h = h2.copy()
                cur = c2
                if cur < best_cur:
                    best_cur = cur
                    best_h = h.copy()
                    no_improve = 0
                moved = True
            else:
                for _ in range(5):
                    if time.time() - t0 > time_limit:
                        break
                    i, j = rng.integers(0, n, size=2)
                    i = int(i)
                    j = int(j)
                    if i == j:
                        continue

                    if h[i] >= h[j]:
                        d = min(float(rng.uniform(0.01, 0.20)), float(h[i]), float(1.0 - h[j]))
                        if d <= 0.0:
                            continue
                        h3 = h.copy()
                        h3[i] -= d
                        h3[j] += d
                    else:
                        d = min(float(rng.uniform(0.01, 0.20)), float(h[j]), float(1.0 - h[i]))
                        if d <= 0.0:
                            continue
                        h3 = h.copy()
                        h3[i] += d
                        h3[j] -= d

                    c3 = max_count_objective(h3)
                    if c3 < best_cur or metropolis_accept(c3 - cur, temperature, rng):
                        h = h3.copy()
                        cur = c3
                        if cur < best_cur:
                            best_cur = cur
                            best_h = h.copy()
                            no_improve = 0
                        moved = True
                        break

        if no_improve == 0:
            temperature = max(1e-4, temperature * 0.990)
        else:
            temperature = max(1e-4, temperature * 0.995)

    best_h = prepare_feasible(best_h, n)
    best_cur = max_count_objective(best_h)
    return best_h, best_cur


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1300.0
    if total_budget <= 0.0:
        total_budget = 1.0
    deadline = start + total_budget * 0.96

    rng = np.random.default_rng(int(seed))

    best_h = None
    best_c = math.inf
    best_n = 0

    init = None
    try:
        init = initial_h_values
    except NameError:
        init = None

    if init is not None:
        try:
            harr = np.asarray(init, dtype=float).ravel()
            if harr.size >= 4 and np.all(np.isfinite(harr)):
                n0 = int(harr.size)
                h0 = prepare_feasible(harr, n0)
                c0 = c5_exact(h0)
                best_h = h0
                best_c = float(c0)
                best_n = n0
        except Exception:
            best_h = None

    if best_h is None:
        n0 = 42
        h0 = random_binary(n0, rng)
        best_h = prepare_feasible(h0, n0)
        best_c = c5_exact(best_h)
        best_n = int(best_h.size)

    polished = {int(best_n)}

    remaining = deadline - time.time()
    if remaining > 15.0:
        tlim = min(240.0, remaining * 0.25)
        if tlim > 10.0:
            hp, cp = polish_local(best_h, tlim, int(seed) + 1)
            hp = prepare_feasible(hp, int(hp.size))
            cp_exact = c5_exact(hp)
            if cp_exact < best_c:
                best_h = hp
                best_c = float(cp_exact)
                best_n = int(hp.size)
            polished.add(int(best_n))

    base = int(best_n)
    if base < 2:
        base = 42

    resolutions = []
    for m in (1, 2, 3, 4, 6, 8):
        r = base * m
        if r <= 384 and r % 2 == 0:
            resolutions.append(int(r))
    for r in (42, 84, 168, 252, 336):
        if r <= 384:
            resolutions.append(int(r))

    seen = set()
    uniq_res = []
    for r in resolutions:
        r = int(r)
        if r >= 2 and r not in seen:
            seen.add(r)
            uniq_res.append(r)

    for r in uniq_res:
        if r > 384:
            continue
        remaining = deadline - time.time()
        if remaining < 30.0:
            break
        if r in polished and r == int(best_n):
            continue

        cand = candidate_from_best(best_h, r)
        c_cand = c5_exact(cand)

        tlim = min(220.0, remaining * 0.40)
        if tlim < 15.0:
            break

        hp, cp = polish_local(cand, tlim, int(seed) + r + 100)
        hp = prepare_feasible(hp, int(hp.size))
        cp_exact = c5_exact(hp)
        if cp_exact < best_c:
            best_h = hp
            best_c = float(cp_exact)
            best_n = int(hp.size)
        polished.add(int(r))

        remaining = deadline - time.time()
        if r <= 126 and remaining > 100.0:
            tlim2 = min(70.0, remaining * 0.20)
            if tlim2 > 10.0:
                rnd = random_binary(r, rng)
                hp2, cp2 = polish_local(rnd, tlim2, int(seed) + r + 777)
                hp2 = prepare_feasible(hp2, int(hp2.size))
                cp2_exact = c5_exact(hp2)
                if cp2_exact < best_c:
                    best_h = hp2
                    best_c = float(cp2_exact)
                    best_n = int(hp2.size)

    remaining = deadline - time.time()
    if remaining > 80.0:
        tlim = min(140.0, remaining * 0.60)
        if tlim > 10.0:
            hp, cp = polish_local(best_h, tlim, int(seed) + 99999)
            hp = prepare_feasible(hp, int(hp.size))
            cp_exact = c5_exact(hp)
            if cp_exact < best_c:
                best_h = hp
                best_c = float(cp_exact)
                best_n = int(hp.size)

    best_h = prepare_feasible(best_h, int(best_h.size))
    best_c = float(c5_exact(best_h))

    return best_h, best_c, int(best_h.size)
