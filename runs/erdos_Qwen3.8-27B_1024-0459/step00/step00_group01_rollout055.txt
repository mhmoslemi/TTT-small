import numpy as np

from scipy.optimize import minimize

def compute_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return float(np.max(corr) * dx)

def repair_h(h, n):
    """Project h onto the feasible set: 0<=h<=1, sum(h)=n/2."""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    
    # Bisection on the Lagrange multiplier for sum constraint
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return h

def make_objective(n):
    """Create the objective function for optimization."""
    def obj(params):
        h = np.clip(params, 0.0, 1.0)
        base = compute_c5(h)
        # Penalty for sum constraint (light, since bounds handle most of it)
        target = n / 2.0
        penalty = 10.0 * (np.sum(h) - target) ** 2
        return base + penalty
    return obj

def optimize_local(h, n, maxiter=500, ftol=1e-12):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    def obj(x):
        return compute_c5(np.clip(x, 0.0, 1.0))
    try:

                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5(h_out)

        return h, compute_c5(h)

            grad[i + k] -= h[i]

            grad[i] -= h[i + kk]
    h_new = h - lr * grad
