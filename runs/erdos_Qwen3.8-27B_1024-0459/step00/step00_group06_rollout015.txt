```python
import numpy as np
from scipy.optimize import minimize
import time

def get_c5_values(h, n):
    """Return all correlation values (before multiplying by dx)."""
    return np.correlate(h, 1.0 - h, mode="full")

def obj_soft(h, n, beta):
    """Soft-max objective."""
    c = get_c5_values(h, n)
    scaled = beta * c
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * (2.0 / n)

def obj_soft_grad(h, n, beta):
    """Soft-max objective and gradient."""
    c = get_c5_values(h, n)
    scaled = beta * c
    m = np.max(scaled)
    w = np.exp(scaled - m)
    w /= np.sum(w)
    obj = (m + np.log(np.sum(w))) / beta * (2.0 / n)
    
    dx = 2.0 / n
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k_idx in range(n_corr):
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= wk * h[lag:]
        else:
            grad[:n + lag] += wk * (1.0 - h[-lag:])
            if -lag < n:
                grad[-lag:] -= wk * h[:n + lag]
    
    return obj, grad * dx

def obj_hard(h, n):
    """Hard max objective."""
    c = get_c5_values(h, n)
    return np.max(c) * (2.0 / n)

def obj_hard_grad(h, n):
    """Hard max objective and gradient (subgradient)."""
    c = get_c5_values(h, n)
    k_max = np.argmax(c)
    obj = c[k_max] * (2.0 / n)
    lag = k_max - (n - 1)
    
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:] += (1.0 - h[:n - lag])
        if n - lag > 0:
            grad[:n - lag] -= h[lag:]
    else:
        grad[:n + lag] += (1.0 - h[-lag:])
        if -lag < n:
            grad[-lag:] -= h[:n + lag]
    
    return obj, grad * dx

def project_to_feasible(h, n):
    """Project h to satisfy box and sum constraints."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if diff > 0:
        # Need to decrease sum by diff
        # Decrease where h is largest
        h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    else:
        h += (-diff) / n
        h = np.clip(h, 0.0, 1.0)
    # After clipping, re-adjust
    for _ in range(10):
        diff = h.sum() - target
        if abs(diff) < 1e-10:
            break
        mask = (h > 0) & (h < 1) if diff > 0 else (h > 0) & (h < 1)
        if diff > 0:
            to_decrease = diff / mask.sum()
            h[mask] -= to_decrease
        else:
            to_increase = -diff / mask.sum()
            h[mask] += to_increase
        h = np.clip(h, 0.0, 1.0)
    return h

    # Get initial values
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        n = 95
        h_init = np.full(n, 0.5)
    
    # Ensure feasibility
    h_init = project_to_feasible(h_init, n)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n)
    
    def time_exhausted():
        return time.time() - t0 > budget_s - 30

    for beta in [5.0, 20.0, 100.0]:

        h_curr = best_h.copy()

            fun=lambda x: obj_soft(x, n, beta),

            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': 500, 'ftol': 1e-10}
        )
        
        if result.success or result.fun < best_c5:
            h_new = np.clip(result.x, 0.0, 1.0)
            h_new = project_to_feasible(h_new, n)

        )
        if result.fun < best_c5:

            h_new = project_to_feasible(h_new, n)
            c5_new = compute_c5(h_new, n)

                fun=lambda x: obj_hard(x, n),
```

```python
            h_alt_init = project_feasible(h_alt_init, n_alt)

            bounds_alt = [(0.0, 1.0)] * n_alt
            cons_alt = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n_alt,)}]

            try:
                res = minimize(
                    fun=obj_hard,
                    x0=h_alt_init,
                    jac=obj_hard_grad_only,
                    args=(n_alt,),
                    method='SLSQP',
                    bounds=bounds_alt,
                    constraints=cons_alt,
                    options={'maxiter': 800, 'ftol': 1e-12}
                )
                h_alt_new = project_feasible(np.clip(res.x, 0.0, 1.0), n_alt)
                c5_alt = compute_c5(h_alt_new, n_alt)
                if c5_alt < best_c5:
                    best_h = h_alt_new
                    best_c5 = c5_alt
                    n = n_alt
                    bounds = bounds_alt
                    cons = cons_alt
            except Exception:
                pass

    # Final refinement on the best found solution
    if time.time() - t0 < budget_s - 15:
        try:
            res = minimize(
                fun=obj_hard,
                x0=best_h,
                jac=obj_hard_grad_only,
                args=(n,),
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-14}
            )
            h_final = project_feasible(np.clip(res.x, 0.0, 1.0), n)
            c5_final = compute_c5(h_final, n)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    # Validate with evaluator if available
    try:
        val = evaluate_erdos_solution(best_h, n)
        if val is not None and 0.0 < val < 1.0:
            best_c5 = min(best_c5, float(val))
    except Exception:
        pass

    return (best_h, best_c5, n)
```
