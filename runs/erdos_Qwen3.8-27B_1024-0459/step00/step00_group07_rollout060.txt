def project_to_feasible(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h) = target_sum}."""
    n = len(h)
    # Binary search on the shift
    lo, hi = -max(h) - 1, -min(h) + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        h_p = np.clip(h - mid, 0, 1)
        if h_p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine adjustment
    diff = target_sum - h_p.sum()
    if diff > 0:
        # Add diff to the element with the most room
        idx = np.argmax(1 - h_p)
        h_p[idx] = min(1, h_p[idx] + diff)
    elif diff < 0:
        idx = np.argmin(h_p)
        h_p[idx] = max(0, h_p[idx] + diff)
