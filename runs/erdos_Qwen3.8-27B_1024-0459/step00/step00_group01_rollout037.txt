import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def compute_c5(h):
    """Compute C5 = max_k correlate(h, 1-h, full) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    """Compute full correlation array (scaled)"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def repair(h, n):
    """Ensure sum(h) = n/2 and 0 <= h[i] <= 1"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
    return h

def upsample(h, n_new):
    """Upsample h from len(h) to n_new points maintaining integral"""
    n_old = len(h)
    old_x = (np.arange(n_old) + 0.5) / n_old * 2.0
    new_x = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(new_x, old_x, h)
    return repair(h_new, n_new)

def downdsample(h, n_new):
    """Downsample h from len(h) to n_new points"""
    n_old = len(h)
    old_x = (np.arange(n_old) + 0.5) / n_old * 2.0
    new_x = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(new_x, old_x, h)
    return repair(h_new, n_new)

def sa_optimize(h_init, n, time_limit, seed=0):
    """Simulated annealing with pair-transfer moves"""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    start = time.time()

    T0, T1 = 0.04, 0.0003
    step_base = 0.06

    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        T = T0 * (T1 / T0) ** (elapsed / time_limit)
        step = step_base * (0.3 + 0.7 * T / T0)

        # Process in micro-batches for speed
        batch = 500
        end_batch = start + time_limit
        t_batch = time.time()
        while time.time() - t_batch < 0.02:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(-step, step)

            h[i] += delta
            h[j] -= delta

            if h[i] < 0.0 or h[i] > 1.0 or h[j] < 0.0 or h[j] > 1.0:
                h[i] -= delta
                h[j] += delta
                continue

            new_c5 = compute_c5(h)
            if new_c5 <= cur_c5:
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                accept_prob = np.exp(-(new_c5 - cur_c5) / T)
                if rng.random() < accept_prob:
                    cur_c5 = new_c5
                else:
                    h[i] -= delta
                    h[j] += delta

    return best_h, best_c5

def local_search(h_init, n, time_limit, seed=0):
    """Greedy local search with diverse moves"""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    no_improve = 0
    max_no_improve = 50000

    while time.time() - start < time_limit and no_improve < max_no_improve:
        improved = False
        delta = 0.01 + 0.05 * (1.0 - (time.time() - start) / time_limit)

        # Try multiple random moves
        for _ in range(2000):
            if time.time() - start > time_limit:
                break
            move_type = rng.random()
            if move_type < 0.6:
                # Pair transfer
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = rng.choice([-delta, delta, -2*delta, 2*delta])
                h[i] += d
                h[j] -= d
                if 0 <= h[i] <= 1 and 0 <= h[j] <= 1:
                    new_c5 = compute_c5(h)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] -= d
                        h[j] += d
                else:
                    h[i] -= d
                    h[j] += d
            elif move_type < 0.8:
                # Single point set + repair
                i = rng.randint(n)
                old_val = h[i]
                h[i] = rng.choice([0.0, 0.25, 0.5, 0.75, 1.0])
                h = repair(h, n)
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()
            else:
                # Binary push: move a value toward 0 or 1
                i = rng.randint(n)
                if h[i] > 0.5:
                    h[i] = min(1.0, h[i] + 0.1)
                elif h[i] < 0.5:
                    h[i] = max(0.0, h[i] - 0.1)
                else:
                    h[i] = rng.choice([0.0, 1.0])
                h = repair(h, n)
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()

        if improved:
            no_improve = 0
        else:
            no_improve += 2000

    return best_h, best_c5

def scipy_polish(h_init, n, time_limit, seed=0):
    """Use scipy SLSQP to polish the solution"""
    start = time.time()
    target = n / 2.0
    dx = 2.0 / n

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    for beta in [30, 60, 100, 200]:
        if time.time() - start > time_limit * 0.8:
            break

        def obj_smooth(x):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            c_max = np.max(corr)
            shifted = beta * (corr - c_max)
            shifted = np.clip(shifted, -500, 500)
            return c_max + np.log(np.sum(np.exp(shifted))) / beta

        def grad_smooth(x):
            corr = np.correlate(x, 1.0 - x, mode='full') * dx
            c_max = np.max(corr)
            shifted = beta * (corr - c_max)
            shifted = np.clip(shifted, -500, 500)
            p = np.exp(shifted)
            p /= np.sum(p)
            # Gradient of corr[m] w.r.t. x[j]
            # corr[m] = sum_i x[i] * (1-x[i+m-n+1]) * dx
            # shift[m] = m - n + 1
            shifts = np.arange(len(corr)) - n + 1
            # For each m, grad_x[m] = sum_j p[m] * d_corr[m]/dx[j]
            # d_corr[m]/dx[j] = dx * [(1-x[j+shift[m]])_valid - x[j-shift[m]]_valid]
            g = np.zeros(n)
            for m in range(len(corr)):
                if p[m] < 1e-10:
                    continue
                s = shifts[m]
                # First term: (1-x[j+s]) for valid j+s
                idx1 = np.arange(max(0, -s), min(n, n - s))
                if len(idx1) > 0:
                    g[idx1] += p[m] * dx * (1.0 - x[idx1 + s])
                # Second term: -x[j-s] for valid j-s
                idx2 = np.arange(max(0, s), min(n, n + s))
                if len(idx2) > 0:
                    g[idx2] -= p[m] * dx * x[idx2 - s]
            return g

        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}
        ]
        bounds = [(0.0, 1.0)] * n

        try:
            result = scipy_minimize(obj_smooth, best_h, method='SLSQP',
                                     jac=grad_smooth, bounds=bounds,
                                     constraints=constraints,
                                     options={'maxiter': 200, 'ftol': 1e-10})
            if result.success or result.fun < best_c5:
                h_new = repair(result.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass

    return best_h, best_c5
