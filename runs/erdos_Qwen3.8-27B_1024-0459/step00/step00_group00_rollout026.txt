    n = len(h)
    dx = 2.0 / n
    corr = get_correlations(h)
    return float(np.max(corr) * dx)

def max_corr_raw(h, n_pts):
    h_clip = np.clip(h, 0.0, 1.0)

    return h.sum() - n_pts / 2.0

    h[:half] = h_half
    if n_pts % 2 == 0:
        h[half:] = 1.0 - h_half[::-1]
    else:
        h[half:] = 1.0 - h_half[:half - (n_pts - 2*half)][::-1]
        h[half] = 0.5
    h = fix_sum(h, n_pts)
    return h

def local_search(h, n_pts, rng, n_trials, time_limit):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_sizes = [0.01, 0.005, 0.002, 0.001]
    for trial in range(n_trials):
        if time.time() - t_start > time_limit:
            break
        for ss in step_sizes:
            h_trial = best_h + rng.randn(n_pts) * ss
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = fix_sum(h_trial, n_pts)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
    return best_h, best_c5

def perturb_init(parent, n_pts, rng, strength):
    h = upsample_parent(parent, n_pts)
    h += rng.randn(n_pts) * strength
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, n_pts)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get parent solution
    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
        parent = np.clip(parent, 0.0, 1.0)
    
    best_h = None
    best_c5 = float('inf')
    
    # Stage 1: Multi-resolution optimization
    resolutions = [45, 60, 90, 120, 180, 240, 360]
    
    for n_pts in resolutions:
        if time.time() - start_time > budget_s - 180:
            break
        
        time_for_this = min(200, int(budget_s * 0.12))
        
        # Multiple starting points
        starts = []
        if parent is not None:
            starts.append(upsample_parent(parent, n_pts))
            starts.append(perturb_init(parent, n_pts, rng, 0.05))
            starts.append(perturb_init(parent, n_pts, rng, 0.1))
        starts.append(random_init(n_pts, rng))
        starts.append(symmetric_init(n_pts, rng))
        
        for h_init in starts:
            if time.time() - start_time > budget_s - 60:
                break
            h_init = fix_sum(h_init, n_pts)
            c5_init = compute_c5(h_init)
            
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = h_init.copy()
            
            # SLSQP with direct max
            h_opt, c5_opt, elapsed = try_slsqp(
                h_init, n_pts, maxiter=80, ftol=1e-8,
                time_limit=time_for_this
            )
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            
            if time.time() - start_time > budget_s - 60:
                break
        
        if time.time() - start_time > budget_s - 120:
            break
    
    # Stage 2: Refine best solution with soft max at higher precision
    if best_h is not None and time.time() - start_time < budget_s - 60:
        n_ref = len(best_h)
        time_left = int(budget_s - 60 - (time.time() - start_time))
        if time_left > 30:
            h_ref, c5_ref, elapsed = try_soft_max_slsqp(
                best_h, n_ref,
                tau_iter=[0.05, 0.02, 0.01, 0.005],
                maxiter=60,
                time_limit=min(time_left, 120)
            )
            if h_ref is not None and c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
    
    # Stage 3: Local search / random restarts from best
    if time.time() - start_time < budget_s - 30:
        n_ls = len(best_h)
        time_left = int(budget_s - 30 - (time.time() - start_time))
        n_trials = min(200, time_left * 50)
        h_ls, c5_ls = local_search(
            best_h, n_ls, rng, n_trials=n_trials,
            time_limit=min(time_left, 60)
        )
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Stage 4: Try a different resolution from the best solution found
    if time.time() - start_time < budget_s - 40:
        n_up = min(len(best_h) * 2, 480)
        time_left = int(budget_s - 40 - (time.time() - start_time))
        if time_left > 60 and n_up > len(best_h):
            h_up = upsample_parent(best_h, n_up)
            h_up = fix_sum(h_up, n_up)
            h_opt2, c5_opt2, _ = try_slsqp(
                h_up, n_up, maxiter=100, ftol=1e-9,
                time_limit=min(time_left, 100)
            )
            if h_opt2 is not None and c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
    
    # Final fixup and verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    # Ensure valid return
    if best_h is None or not np.all(np.isfinite(best_h)):
        n_fallback = 90
        best_h = np.full(n_fallback, 0.5)
        best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
