import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * dx), c * dx


def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_grad(h, penalty=5000.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    mi = int(np.argmax(c))
    lag = mi - (n - 1)
    c5 = c[mi] * dx

    j = np.arange(n)
    g = np.zeros(n)
    i1 = j - lag
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + lag
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    g *= dx

    cv = h.sum() - n / 2.0
    val = c5 + penalty * cv * cv
    g = g + 2.0 * penalty * cv
    return val, g


def smooth_obj_grad(h, penalty=5000.0, tau=0.02):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    cs = c / tau
    cs_max = cs.max()
    shifts = cs - cs_max
    exps = np.exp(shifts)
    logsum = cs_max + np.log(np.sum(exps))
    c5_smooth = logsum * tau * dx

    weights = exps / np.sum(exps)

    j = np.arange(n)
    g = np.zeros(n)
    for l in range(len(c)):
        lag = l - (n - 1)
        i1 = j - lag
        m1 = (i1 >= 0) & (i1 < n)
        i2 = j + lag
        m2 = (i2 >= 0) & (i2 < n)
        gl = np.zeros(n)
        gl[m1] += (1.0 - h[i1[m1]])
        gl[m2] -= h[i2[m2]]
        g += weights[l] * gl
    g *= dx

    cv = h.sum() - n / 2.0
    val = c5_smooth + penalty * cv * cv
    g = g + 2.0 * penalty * cv
    return val, g


def optimize_stage(h, n, penalty=5000.0, maxiter=500, method="L-BFGS-B"):
    h0 = project_feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    try:
        if method == "smooth":
            res = minimize(smooth_obj_grad, h0, method='L-BFGS-B',
                           bounds=bounds, args=(penalty, 0.02),
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        else:
            res = minimize(obj_grad, h0, method='L-BFGS-B',
                           bounds=bounds, args=(penalty,),
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)


def interpolate_up(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.arange(n_old, dtype=float) * (2.0 / n_old)
    xn = np.arange(n_new, dtype=float) * (2.0 / n_new)
    return np.interp(xn, xo, h)


def make_random_01(n, rng):
    h = np.zeros(n)
    ones_count = n // 2
    idx = rng.choice(n, size=ones_count, replace=False)
    h[idx] = 1.0
    return h
