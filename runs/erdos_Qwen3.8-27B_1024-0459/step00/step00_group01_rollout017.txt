        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-shift:hi2-shift]
    
    grad *= dx * scale
    return grad

def optimize_pg(h, n, max_iters=3000, power=50, time_limit=None, start_t=None):
    """Optimize using projected gradient with smooth max"""
    h = project_box_sum(h, n).copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    alpha = 1.0 / (2.0 * n) * 10  # initial step size
    rho = 0.99  # decay
    
    for it in range(max_iters):
        if time_limit is not None and time.time() - start_t > time_limit:
            break
        
        f, = [smooth_max_value(h, n, power)]
        grad = smooth_max_grad(h, n, power)
        
        # Gradient step + projection
        h_new = project_box_sum(h - alpha * grad, n)
        c5_new = compute_c5(h_new, n)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Adaptive step size
        f_new = smooth_max_value(h_new, n, power)
        if f_new <= f:
            alpha = min(alpha * 1.1, 10.0)
        else:
            alpha *= rho
        
        h = h_new
        
        # Check convergence
        if it > 100 and abs(f_new - f) / (f + 1e-10) < 1e-8:
            break
    
    return best_h, best_c5

def optimize_slsqp(h, n, maxiter=500):
    """Optimize using SLSQP (for small n)"""
    h = project_box_sum(h, n).copy()
    n_var = n + 1  # h_0..h_{n-1}, t
    
    x0 = np.concatenate([h, [compute_c5(h, n) + 0.05]])
    
    def objective(x):
        return x[n]
    
    def obj_grad(x):
        return np.concatenate([np.zeros(n), [1.0]])
    
    constraints = []
    constraints.append({
        'type': 'eq',
        'fun': lambda x: np.sum(x[:n]) - n / 2.0,
        'jac': lambda x: np.concatenate([np.ones(n), [0.0]])
    })
    
    # Add correlation constraints (limit to reduce SLSQP overhead)
    dx = 2.0 / n
    corrs_init = np.correlate(h, 1.0 - h, mode="full") * dx
    # Add all constraints for small n, top 50 for larger
    if n <= 60:
        lags = range(2 * n - 1)
    else:
        top_idx = np.argsort(corrs_init)[-50:]
        lags = top_idx
    
    for m in lags:
        def corr_fun(x, m=m):
            hh = x[:n]
            c = np.correlate(hh, 1.0 - hh, mode="full") * dx
            return x[n] - c[m]
        constraints.append({'type': 'ineq', 'fun': corr_fun})
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        result = minimize(objective, x0, method='SLSQP', bounds=bounds,
                         constraints=constraints, jac=obj_grad,
                         options={'maxiter': maxiter, 'ftol': 1e-10})
        if result.x is not None:
            h_opt = np.clip(result.x[:n], 0.0, 1.0)
            h_opt = project_box_sum(h_opt, n)
            c5_opt = compute_c5(h_opt, n)
            return h_opt, c5_opt
    except Exception:
        pass
    
    return h, compute_c5(h, n)

def upsample(h, n_from, n_to):
    """Upsample or downsample h using linear interpolation"""
    if n_from == n_to:
        return h.copy()
    x_from = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_to = np.linspace(0.0, 2.0, n_to, endpoint=False)
    h_new = np.interp(x_to, x_from, h)
    return np.clip(h_new, 0.0, 1.0)

    h_init = initial_h_values.copy().astype(float)
    n_init = len(h_init)
    h_init = np.clip(h_init, 0.0, 1.0)
    h_init = project_box_sum(h_init, n_init)
    
    best_c5 = compute_c5(h_init, n_init)
    best_h = h_init.copy()
    best_n = n_init
    
    def update_best(h, n):
        nonlocal best_c5, best_h, best_n
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        return c5
    
    print(f"Initial C5 = {best_c5:.6f} at n={n_init}")
    
    # Phase 1: SLSQP at initial resolution
    if time_ok():
        h_s, c5_s = optimize_slsqp(h_init, n_init, maxiter=300)
        c5_s_final = update_best(h_s, n_init)
        if c5_s_final < best_c5:
