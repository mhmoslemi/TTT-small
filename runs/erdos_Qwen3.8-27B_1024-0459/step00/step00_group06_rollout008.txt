    corr = np.correlate(h, 1-h, mode="full")

```python
def compute_grad_full(h, beta):
    """Compute gradient of smooth max of correlations w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = corr / beta - np.max(corr) / beta
    p = np.exp(shifted)
    p /= np.sum(p)
    
    # s_m = n-1-m, m = 0, ..., 2n-2
    s = (n - 1) - np.arange(2 * n - 1)
    grad = np.zeros(n)
    
    for j in range(n):
        # d(corr_m)/d(h_j) = (1 - h[j+s_m]) * [0 <= j+s_m < n] - h[j-s_m] * [0 <= j-s_m < n]
        js = j + s
        mask1 = (js >= 0) & (js < n)
        if np.any(mask1):
            grad[j] += np.dot(p[mask1], 1.0 - h[js[mask1]])
        jms = j - s
        mask2 = (jms >= 0) & (jms < n)
        if np.any(mask2):
            grad[j] -= np.dot(p[mask2], h[jms[mask2]])
    
    return grad
```
