def _eval_c5(h):
    """Compute C5 for a (binary or continuous) h of length N."""
    n = len(h)
    dx = 2.0 / n
    cc = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cc) * dx)

def _fix_sum(h, N):
    """Clip to [0,1] and adjust sum to be N/2."""
    h = np.clip(h, 0, 1).copy()
    s = h.sum()
    target = N / 2
    if abs(s - target) < 1e-9:
        return h
    # adjust: add/remove mass
    if s < target:
        # add to "low" values (max gain in h per cost)
        order = np.argsort(h)  # lowest first
        gap = target - s
        for i in order:
            if gap <= 1e-9:
                break
            add = min(1.0 - h[i], gap)
            h[i] += add
            gap -= add
    else:
        order = np.argsort(-h)  # highest first
        gap = s - target
        for i in order:
            if gap <= 1e-9:
                break
            rem = min(h[i], gap)
            h[i] -= rem
            gap -= rem
    return h

import numpy as np
import time

def _c5_eval(h):
    """Compute C5 = max over all d of (2/N) * c(d)."""
    n = len(h)
    h = np.asarray(h, dtype=float)
    dx = 2.0 / n
    cc = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cc) * dx)

def _fix_sum(h, N):
    """Clip h to [0,1] and ensure sum = N/2."""
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    s = h.sum()
    tgt = float(N) / 2.0
    if abs(s - tgt) <= 1e-12:
        return h
    if s < tgt:
        gap = tgt - s
        order = np.argsort(h)   # add to lowest first
        for i in order:
            if gap <= 1e-10: break
            add = min(1.0 - h[i], gap)
            h[i] += add; gap -= add
    else:
        gap = s - tgt
        order = np.argsort(-h)  # remove from highest first
        for i in order:
            if gap <= 1e-10: break
            rem = min(h[i], gap)
            h[i] -= rem; gap -= rem
    return h

def _snap_bin(h, N):
    """Snap to binary and adjust count to N//2 (N even)."""
    h = (np.asarray(h, dtype=float) >= 0.5).astype(float).copy()
    ct = int(h.sum())
    tgt = N // 2
    if ct > tgt:
        o = np.where(h == 1.0)[0]
        o = o[np.random.default_rng(12345).permutation(len(o))]
        h[o[:ct - tgt]] = 0.0

def c5_val(h):
    """C5 value (max of correlate) * dx. h: 1D array."""
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / n
    cc = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(cc) * dx)

def clip_fix(h, N):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    s = float(h.sum())

        gap = tgt - s
        order = np.argsort(h)
        for i in order:
            if gap <= 1e-9:
                break
            add = min(1.0 - h[i], gap)
            h[i] += add
            gap -= add
    elif s > tgt:

    try:

            _obj, base_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * Nf,
            options={'maxiter': maxiter, 'ftol': ftol, 'gtol': ftol},
        )
        hc = _fix_sum(res.x, Nf)
        return hc, _c5_eval(hc)
    except Exception:
        return base_h.copy(), _c5_eval(base_h)


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s
    best_h = None
    best_c = float("inf")

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        p = np.asarray(parent, dtype=float)
        p = _fix_sum(p, len(p))
        best_h = p
        best_c = _c5_eval(p)
        print(f"Parent C5 (n={len(best_h)}): {best_c:.6f}")

    N_list = [40, 60, 80, 100, 120, 150, 200, 250]
    inits = [_init_sine, _init_cluster, _init_rand_bin, _init_group, _init_saw, _init_rand_frac]

    for N in N_list:
        if N % 2:
            N += 1
        if time.time() > deadline - 60:
            break
        for init_fn in inits:
            if time.time() > deadline - 20:
                break
            h0 = np.asarray(init_fn(N, rng), dtype=float)
            hb = _snap_bin(h0, N)
            t1 = time.time()
            h_r, c_r = _ran_local(hb, rng, max_steps=60000, time_budget_s=14.0)
            dt = time.time() - t1
            if c_r < best_c:
                best_c = c_r
                best_h = h_r.copy()
            print(f"N={N}, {init_fn.__name__}: c5={c_r:.4f} (best {best_c:.4f}, {dt:.1f}s)")

    # Continuous refinements
    if time.time() < deadline - 90:
        Nf_vals = [95, 100, 120, 150]
        if parent is not None:
            Nf_vals.insert(0, len(parent))
        for Nf in Nf_vals:
            if Nf % 2:
                Nf += 1
            if time.time() > deadline - 30:
                break
            base = best_h.copy()
            if len(base) != Nf:
                xb = np.linspace(0, 1, Nf)
                xo = np.linspace(0, 1, len(base))
                base = np.interp(xb, xo, base)
            base = _fix_sum(base, Nf)
            hc, cc = _slsqp_solve(base, maxiter=100 if Nf <= 100 else 60)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()
                print(f"SLSQP Nf={Nf}: c5={cc:.4f} (best {best_c:.4f})")

    # Final
    if best_h is None:
        best_h = _fix_sum(_init_sine(100), 100)
    best_h = _fix_sum(np.asarray(best_h, dtype=float), len(best_h))
    best_c = _c5_eval(best_h)
    return best_h, float(best_c), int(len(best_h))
