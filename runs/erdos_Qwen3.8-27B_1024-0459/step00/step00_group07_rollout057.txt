n/2 - sum h[i]^2 <= t/dx

This is a convex constraint (lower bound on convex function)! So if only the center lag were binding, we'd have a convex problem.

                j = rng.integers(n // 2) * 2 + 1
                h[i], h[j] = h[j], h[i]
        candidates.append(h)
    # Block pattern
    for _ in range(count - 2 * (count // 3)):
        block_size = rng.integers(2, max(3, n // 20))

        count_ones = 0

            h[pos:pos+take] = 1.0

            pos += take
            pos += block_size  # skip

        self.limit = 120.0
        self.n = 80
        self.beta = 50.0
    
    def set_params(self, t0, limit, n, beta):

        self.beta = beta

    m = np.max(vals)
    shifted = vals - m
    return m + np.log(np.sum(np.exp(shifted))) / beta

    dx = 2.0 / n

    h = np.clip(h, 0.0, 1.0)
    diff = n * 0.5 - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)

            ones = np.where(h == 1)[0]
            to_zero = rng.choice(ones, n_ones - n // 2, replace=False)
            h[to_zero] = 0
        else:
            zeros = np.where(h == 0)[0]
            to_one = rng.choice(zeros, n // 2 - n_ones, replace=False)
            h[to_one] = 1
    current = _compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_trials = min(300, len(ones) * len(zeros))
        for _t in range(n_trials):
            i = int(rng.integers(len(zeros)))
            j = int(rng.integers(len(ones)))
            h[zeros[i]], h[ones[j]] = 1.0, 0.0
            c5_new = _compute_c5(h)
            if c5_new < current - 1e-14:
                current = c5_new
                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                break
            else:
                h[zeros[i]], h[ones[j]] = 0.0, 1.0
        if not improved:
            break
    return h, current
