def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h onto {sum(h)=N/2, 0<=h<=1} using bisection on Lagrange multiplier."""
    target = n_points / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-12:
            break
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_objective(h, n_points, alpha=80.0):
    """Smooth approximation of max of correlation*dx using log-sum-exp."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    vals = corr * dx
    vmax = np.max(vals)
    shifted = alpha * (vals - vmax)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    smax = vmax + np.log(np.sum(np.exp(shifted))) / alpha
    return smax

    """Construct h from Fourier coefficients: h = 1/2 + sum (a_j cos + b_j sin)."""
    i = np.arange(n_points)
    h = np.full(n_points, 0.5)
    for j in range(1, n_harmonics + 1):
        a = params[2 * (j - 1)]
        b = params[2 * (j - 1) + 1]
        h += a * np.cos(2 * np.pi * j * i / n_points) + b * np.sin(2 * np.pi * j * i / n_points)
    return h
