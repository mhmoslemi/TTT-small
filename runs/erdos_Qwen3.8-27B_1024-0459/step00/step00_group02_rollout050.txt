import time

def compute_corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    return dx * np.max(corr)

def project_feasible(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    if s > target:
        excess = s - target
        # Water-fill down: reduce largest values first
        order = np.array(sorted(range(len(h)), key=lambda i: -h[i]))
        remaining = excess
        for i in order:
            reduce = min(remaining, h[i])
            if reduce > 0:
                h[i] -= reduce
                remaining -= reduce
            if remaining <= 1e-14:
                break
    else:

        # Water-fill up: increase smallest values first
        order = np.array(sorted(range(len(h)), key=lambda i: h[i]))

            increase = min(remaining, 1.0 - h[i])
            if increase > 0:
                h[i] += increase
                remaining -= increase
            if remaining <= 1e-14:
                break

def compute_corr(corr_input):

    return np.correlate(h, one_minus_h, mode='full')

    return dx * np.max(corr)

def project_feasible(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = s - target
    if diff > 1e-12:
        # Need to reduce
        excess = diff
        order = np.argsort(-h)
        remaining = excess
        for i in order:
            if remaining <= 1e-14:
                break
            reduce = min(remaining, h[i])
            h[i] -= reduce
            remaining -= reduce
    elif diff < -1e-12:
        deficit = -diff
        order = np.argsort(h)
        remaining = deficit
        for i in order:
            if remaining <= 1e-14:
                break
            incr = min(remaining, 1.0 - h[i])
            h[i] += incr
            remaining -= incr
    return h

def compute_subgrad(h, n, top_k, beta):

        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5_val(h_new)
        
        if c5_new < best_c5 - 1e-8:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
            lr = min(lr * 1.02, 0.3)
        else:
            no_improve += 1
            lr *= 0.98
            h_new = h  # Don't accept
        
        h = h_new
        
        # Periodic random perturbation
        if no_improve > 80 and rng.random() < 0.1:
            h = h + rng.randn(n) * 0.02
            h = project_feasible(h, target)
            v = np.zeros(n)
            no_improve = 0
    
    return best_h, best_c5

def structured_init(n, rng):
    """Create structured initializations"""
    target = n / 2.0
    methods = []
    
    # Method 1: Periodic two-level
    period = max(2, n // 10)
    h = np.zeros(n)
    for i in range(n):
        h[i] = 0.7 if (i % (2 * period)) < period else 0.3
    methods.append(project_feasible(h, target))
    
    # Method 2: Smooth bump
    x = np.linspace(0, 2, n)
    center = 1.0
    width = 0.6
    h = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    h = h / h.max()
    h = project_feasible(h, target)
    methods.append(h)
    
    # Method 3: Nearly uniform with Fourier perturbation
    k = rng.randint(3, 20)
    h = 0.5 + 0.15 * np.sin(2 * np.pi * k * x / 2.0)
    h = np.clip(h, 0, 1)
    h = project_feasible(h, target)
    methods.append(h)
    
    # Method 4: Sawtooth
    h = np.zeros(n)
    for i in range(n):
        frac = (i % period) / period
        h[i] = 0.5 + 0.3 * (1 if frac < 0.5 else -1)
    h = np.clip(h, 0, 1)
    h = project_feasible(h, target)
    methods.append(h)
    
    # Method 5: Random
    h = rng.uniform(0, 1, n)
    h = project_feasible(h, target)
    methods.append(h)
    
    return methods

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get initial values
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except (NameError, FileNotFoundError):
        h_parent = None
    
    # Base resolution
    if h_parent is not None:
        n_base = len(h_parent)
        h_base = project_feasible(h_parent.copy(), n_base / 2.0)
        c5_base = compute_c5_val(h_base)
    else:
        n_base = 90
        h_base = np.full(n_base, 0.5)
        h_base = project_feasible(h_base, n_base / 2.0)
        c5_base = compute_c5_val(h_base)
    
    best_h = h_base.copy()
    best_c5 = c5_base
    best_n = n_base
    
    # Phase 1: Optimize at base resolution with structured inits
    deadline1 = start_time + min(60, budget_s * 0.1)
    
    inits = [h_base]
    if h_parent is None:
        inits.extend(structured_init(n_base, rng))
    
    for h_init in inits:
        if time.time() >= deadline1:
            break
        h_opt, c5_opt = optimize_single(h_init, n_base, rng, deadline1, momentum=0.5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_base
    
    if h_parent is not None:
        # Re-optimize from parent
        if time_left() > 30:
            h_opt, c5_opt = optimize_single(h_parent.copy(), n_base, rng,
                                            start_time + budget_s * 0.12, momentum=0.6)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_base
    
    # Phase 2: Progressive resolution increase
    resolutions = []
    for r in [120, 180, 300, 450, 600]:
        if r > n_base:
            resolutions.append(r)
    
    for target_n in resolutions:
        if time_left() < 40:
            break
        
        # Allocate time based on remaining and number of resolutions left
        remaining_res = len([r for r in resolutions if r >= target_n])
        time_alloc = max(30, min(time_left() * 0.7 / remaining_res, 300))
        deadline = min(start_time + budget_s * 0.88, time.time() + time_alloc)
        
        # Upsample best solution
        h_up = upsample(best_h, target_n)
        
        # Also try structured inits at this resolution
        h_struct = structured_init(target_n, rng)
        
        candidates = [h_up] + h_struct[:2]
        
        for h_init in candidates:
            if time.time() >= deadline:
                break
            h_opt, c5_opt = optimize_single(h_init, target_n, rng, deadline, momentum=0.65)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = target_n
    
    # Phase 3: Final polish at best resolution found
    if time_left() > 20:
        h_polish = upsample(best_h, best_n)
        h_polish, c5_polish = optimize_single(h_polish, best_n, rng,
                                              start_time + budget_s * 0.95, momentum=0.8)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5_val(best_h)
    
    return (best_h, best_c5, best_n)
