def compute_correlations(h, dx):
    """Compute all correlation values: dx * np.correlate(h, 1-h, mode='full')"""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return corr * dx

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h, dx)
    return np.max(corr)

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""

    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h + mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + mid, 0, 1)

    if abs(diff) > 1e-10:

        free = (h_p > 1e-6) & (h_p < 1 - 1e-6)
        if np.any(free):
            h_p[free] += diff / np.sum(free)
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def compute_gradient_smoothed(h, corr, dx, eps_smooth):
    """Compute gradient of smoothed max objective"""
    n = len(h)
    # Smoothed weights
    log_vals = eps_smooth * corr / dx  # Work in unscaled values
    log_vals -= np.max(log_vals)  # Numerical stability
    weights = np.exp(log_vals)
    weights /= np.sum(weights)
    
    # Gradient of each correlation w.r.t. h
    # For shift s (0-indexed: s = k - (n-1) for k in 0..2n-2):
    # c_k = dx * sum_{i valid} h[i]*(1-h[i+s])
    # ∂c_k/∂h[j] = dx*(1-h[j+s])*1_{valid} - dx*h[j-s]*1_{valid}
    
    grad = np.zeros(n)
    
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)  # shift
        # Valid i range:
        if s >= 0:
            i_lo, i_hi = 0, n - 1 - s
        else:
            i_lo, i_hi = -s, n - 1
        
        # Contribution from h[i] term: dx*(1-h[i+s]) for i in [i_lo, i_hi]
        # This adds to grad[i] for i in [i_lo, i_hi]
        grad[i_lo:i_hi+1] += weights[k] * dx * (1.0 - h[i_lo+s:i_hi+s+1])
        
        # Contribution from -h[i+s] term: -dx*h[i] for i in [i_lo, i_hi]
        # This adds to grad[i+s] for i in [i_lo, i_hi]
        grad[i_lo+s:i_hi+s+1] -= weights[k] * dx * h[i_lo:i_hi+1]
    
    return grad

def optimize_h(h_init, n, rng, max_time=60.0, initial_lr=0.01):
    """Optimize h using smoothed max + projected subgradient"""
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    
    eps_smooth = 1.0  # Initial smoothing
    lr = initial_lr
    no_improve = 0
    
    start = time.time()
    iteration = 0
    
    while time.time() - start < max_time:
        iteration += 1
        
        # Compute objective and gradient
        c5_val, corr = compute_c5(h)
        grad = compute_gradient_smoothed(h, corr, dx, eps_smooth)
        
        # Take step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        # Evaluate
        c5_new, _ = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
            h = h_new
        elif c5_new < c5_val:
            h = h_new
            no_improve += 1
        else:
            no_improve += 1
            # Random perturbation every few failed iterations
            if no_improve % 10 == 0:
                perturbation = rng.normal(0, 0.05, n)
                h_pert = h + perturbation
                h_pert = project_feasible(h_pert, target_sum)
                c5_pert, _ = compute_c5(h_pert)
                if c5_pert < c5_val:
                    h = h_pert
        
        # Adaptive parameters
        if iteration % 50 == 0:
            lr *= 0.8
            eps_smooth = max(0.1, eps_smooth * 0.9)
        
        if no_improve > 200:
            # Restart with perturbation
            no_improve = 0
            eps_smooth = max(0.1, eps_smooth * 0.5)
            h = h + rng.normal(0, 0.1, n)
            h = project_feasible(h, target_sum)
            lr = initial_lr * 0.5
    
    return best_h, best_c5

def upscale(h, n_new):
    """Upscale h from n_old to n_new points"""
    n_old = len(h)
    factor = n_new // n_old
    if factor * n_old != n_new:
        # Non-uniform upscaling - use interpolation
        from scipy.interpolate import interp1d
        x_old = np.linspace(0, 2, n_old, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
        h_new = f(x_new)
    else:
        h_new = np.repeat(h, factor)
    # Fix the sum
    target_sum = n_new / 2.0
    h_new = project_feasible(h_new, target_sum)
    return h_new

    return best_h, final_c5, len(best_h)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr), corr

def project_feasible(h, target_sum):
    n = len(h)
    lo = -float(np.max(h)) - 0.01
    hi = 1.0 - float(np.min(h)) + 0.01
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h_p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h_p)
    if abs(diff) > 1e-9:
        free = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        if np.any(free) and np.sum(free) > 0:
            h_p[free] += diff / np.sum(free)
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def compute_gradient_smoothed(h, corr_vals, dx, n, log_weights):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if log_weights[k] < -20.0:
            continue
        s = k - (n - 1)
        if s >= 0:
            n_lo = 0
            n_hi = n - 1 - s
        else:
            n_lo = -s
            n_hi = n - 1
        if n_hi < n_lo:
            continue
        w = log_weights[k]
        grad[n_lo + s:n_hi + s + 1] += w * dx * (1.0 - h[n_lo:n_hi + 1])
        grad[n_lo:n_hi + 1] -= w * dx * h[n_lo + s:n_hi + s + 1]
    return grad

def softmax_weights(corr_vals, eps):
    scaled = eps * corr_vals
    scaled -= np.max(scaled)
    w = np.exp(scaled)
    total = np.sum(w)
    if total < 1e-300:
        w = np.ones(len(w)) / len(w)
    else:
        w /= total
    return w

def optimize_h(h_init, n, rng, max_time, initial_lr, eps_start):
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    lr = initial_lr
    eps_smooth = eps_start
    no_improve = 0
    start = time.time()
    iteration = 0
    while time.time() - start < max_time:
        iteration += 1
        c5_val, corr = compute_c5(h)
        if iteration % 50 == 0:
            lr = max(1e-6, initial_lr * (0.97 ** (iteration // 50)))
            eps_smooth = max(0.05, eps_start * (0.95 ** (iteration // 100)))
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        diff = c5_val - best_c5
        if no_improve > 300:
            h = h + rng.normal(0, 0.15, n)
            h = project_feasible(h, target_sum)
            no_improve = 0
            lr = initial_lr
            eps_smooth = max(0.05, eps_smooth * 0.5)
        if no_improve > 150:
            h = h + rng.normal(0, 0.05, n)
            h = project_feasible(h, target_sum)
            no_improve = 0
        log_w = softmax_weights(corr, eps_smooth)
        grad = compute_gradient_smoothed(h, corr, dx, n, log_w)
        if np.max(np.abs(grad)) > 1e12:
            break
        step = lr * grad
        h_new = h - step
        h_new = project_feasible(h_new, target_sum)
        c5_new, _ = compute_c5(h_new)
        if c5_new < c5_val:
            h = h_new
        elif c5_new < best_c5 + 0.005:
            if rng.rand() < 0.3:
                h = h_new
        else:
            h = h_new
            h = h + rng.normal(0, lr * 2.0, n)
            h = project_feasible(h, target_sum)
    c5_final, _ = compute_c5(best_h)
    return best_h, c5_final

def random_binary_init(n, rng):
    target = n // 2
    h = np.zeros(n)
    indices = rng.choice(n, size=target, replace=False)
    h[indices] = 1.0
    h = h + rng.uniform(0, 0.3, n)
    h = project_feasible(h, target)
    return h

def random_smooth_init(n, rng):
    h = np.sort(rng.uniform(0, 1, n))
    target = n / 2.0
    mean_val = np.mean(h)
    h = h - mean_val + target / n
    h = project_feasible(h, target)
    return h

def upscale(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = project_feasible(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.ones(45) * 0.5

    n_parent = len(h_init)
    target_parent = n_parent / 2.0
    h_init = project_feasible(h_init, target_parent)

    best_h = h_init.copy()
    best_c5, _ = compute_c5(h_init)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Optimize at parent resolution
    t1_time = min(200, remaining() * 0.25)
    if t1_time > 10:
        h_r, c5_r = optimize_h(h_init.copy(), n_parent, rng, t1_time, 0.02, 2.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 2: Random binary initialization at parent resolution
    if remaining() > 50:
        for _ in range(3):
            h_rb = random_binary_init(n_parent, rng)
            h_r, c5_r = optimize_h(h_rb, n_parent, rng, min(60, remaining() * 0.1), 0.015, 1.5)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            if remaining() < 60:
                break

    # Phase 3: Upscale to 2x and optimize
    if remaining() > 80:
        n2 = n_parent * 2
        h_up = upscale(best_h, n2)
        h_r, c5_r = optimize_h(h_up, n2, rng, min(300, remaining() * 0.25), 0.01, 1.5)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Random init at higher resolution
    if remaining() > 60:
        n2 = len(best_h)
        h_rb2 = random_binary_init(n2, rng)
        h_r, c5_r = optimize_h(h_rb2, n2, rng, min(120, remaining() * 0.15), 0.01, 1.5)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 5: 4x upscaled
    if remaining() > 100:
        n4 = n_parent * 4
        h_up4 = upscale(best_h, n4)
        h_r, c5_r = optimize_h(h_up4, n4, rng, min(200, remaining() * 0.25), 0.008, 1.0)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Final polish at best resolution

        if c5_r < best_c5:
            best_c5 = c5_r

    # Phase 7: Quick local search with very small steps

        h_fine = best_h.copy()
        t_fine = min(60, remaining() * 0.4)

        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    final_c5, _ = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
