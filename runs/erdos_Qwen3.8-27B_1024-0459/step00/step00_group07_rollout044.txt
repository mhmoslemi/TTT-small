def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def upscale(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    f = interp1d(x_old, h_old, kind='linear', fill_value='extrapolate')
    h_new = f(x_new)
    return np.clip(h_new, 0, 1)

def normalize_sum(h, n):

    h_adj = h.copy()

        h_adj = np.clip(h_adj, 0, 1)

Δr[-kp] = -(1-h[i-kp])*[i ≥ kp] + h[i+kp]*[i < n-kp] + (1-h[j-kp])*[j ≥ kp] - h[j+kp]*[j < n-kp]

def binary_local_search_sa(h, n, time_limit=60.0, seed=42):
    """Simulated annealing local search on binary sequence."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = np.max(r)
    best_max = current_max
    best_h = h.copy()
    
    T_start = 1.0
    T_end = 0.01
    n_swaps = int(time_limit * 3000)  # estimate
    # Actually, let's just iterate until time is up
    
    deadline = time.time() + time_limit
    step = 0
    no_improve = 0
    max_no_improve = 20000
    
    while time.time() < deadline:
        step += 1
        # Simulated annealing temperature
        progress = min(step / max(1, n_swaps), 1.0)
        T = T_start * (T_end / T_start) ** progress
        
        # Choose random swap: pick a random 1 and a random 0
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        
        if i == j:
            continue
        
        # Compute delta in correlations
        old_r = r
        # Apply swap
        h[i] = 0
        h[j] = 1
        # Recompute r (for simplicity; optimization would update incrementally)
        new_r = np.correlate(h, 1.0 - h, mode='full').astype(float)
        new_max = np.max(new_r)
        
        delta = new_max - current_max
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            current_max = new_max
            r = new_r
            ones.discard(i)
            ones.add(j)
            zeros.add(i)
            zeros.discard(j)
            
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            h[i] = 1
            h[j] = 0
            no_improve += 1
        
        if no_improve > max_no_improve:
            break
        
        # Periodic check
        if step % 10000 == 0 and time.time() > deadline:
            break
    
    return best_h, best_max * dx, step

def compute_swap_delta(h, i, j, n):
    """Compute the change in correlation array when swapping h[i] (1→0) and h[j] (0→1)."""
    delta = np.zeros(2*n-1)

        k = m - (n-1)
        d = 0.0

            # r[m] = sum_{t=0}^{n-1-k} h[t]*(1-h[t+k])
            # Effect of h[i] going 1→0:
            if i <= n-1-k:
                # h[i] term: was 1*(1-h[i+k]), now 0 → change = -(1-h[i+k]) = -1 + h[i+k]
                d += -1.0 + h[i+k]
            if i-k >= 0 and i-k <= n-1-k:
                # (1-h[t+k]) term where t+k=i, t=i-k: was h[i-k]*(1-1)=0, now h[i-k]*1 → change = +h[i-k]
                d += h[i-k]
            # Effect of h[j] going 0→1:
            if j <= n-1-k:
                d += 1.0 - h[j+k]
            if j-k >= 0 and j-k <= n-1-k:
                d += -h[j-k]
        else:
            kp = -k
            # r[m] = sum_{t=kp}^{n-1} h[t]*(1-h[t-kp])
            if i >= kp and i <= n-1:
                d += -1.0 + h[i-kp]
            if i+kp >= kp and i+kp <= n-1:
                d += h[i+kp]
            if j >= kp and j <= n-1:
                d += 1.0 - h[j-kp]
            if j+kp >= kp and j+kp <= n-1:
                d += -h[j+kp]
        delta[m] = d
    
    return delta
