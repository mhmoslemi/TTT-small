```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upsample_to_n(h_old, n_new):
    """Upsample or downsample h to length n_new using interpolation, then binarize."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_interp = np.interp(x_new, x_old, h_old)
    h_new = (h_interp > 0.5).astype(float)
    # Adjust to have exactly n_new//2 ones
    target = n_new // 2
    current = int(h_new.sum())
    if current < target:
        # Need more ones
        scores = h_interp  # use interpolation values as "softness"
        idx = np.argsort(-scores)
        for i in idx:
            if h_new[i] == 0:
                h_new[i] = 1
                current += 1
                if current == target:
                    break
    elif current > target:
        idx = np.argsort(scores)
        for i in idx:
            if h_new[i] == 1:
                h_new[i] = 0
                current -= 1
                if current == target:
                    break
    return h_new

def swap_c5(h, i, j):
    """Compute new C5 after swapping h[i] and h[j] (h[i]=1, h[j]=0)."""
    n = len(h)
    dx = 2.0 / n
    h_new = h.copy()
    h_new[i] = 0.0
    h_new[j] = 1.0
    return compute_c5(h_new)

def local_search(h, n_iter, rng):
    """Hill climbing with 2-bit swaps."""
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for _ in range(n_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new <= c5:
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def simulated_annealing(h, n_iter, rng, T0=0.1, T_final=0.001):
    """SA with 2-bit swaps."""
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    n_sweeps = max(1, n_iter // max(1, len(ones) * len(zeros)))
    for sweep in range(n_iter):
        T = T0 * (T_final / T0) ** (sweep / max(1, n_iter - 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.arange(n_new, dtype=float) * (n_old - 1) / (n_new - 1)
    h_interp = np.interp(x_new, x_old, h_old)
    # Binarize
    h_bin = (h_interp >= 0.5).astype(float)
    target = n_new // 2
    current = int(h_bin.sum())
    if current != target:
        if current < target:
            # Need more ones: pick highest interpolation values that are 0
            idx0 = np.where(h_bin == 0)[0]
            order = idx0[np.argsort(-h_interp[idx0])]
            for i in order[:target - current]:
                h_bin[i] = 1.0
        else:
            idx1 = np.where(h_bin == 1)[0]
            order = idx1[np.argsort(h_interp[idx1])]
            for i in order[:current - target]:
                h_bin[i] = 0.0
    return h_bin

def full_scan_ls(h, max_scans=100, rng=None):
    """Full scan local search for binary h."""
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    no_improve = 0
    
    for _ in range(max_scans):
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        best_delta = 0
        best_i, best_j = -1, -1
        
        for i in ones_list:
            h[i] = 0
        # Vectorized: for each i, compute the effect...
        # Actually, let's just loop for clarity
        for i in ones_list:
            h[i] = 1  # restore
        
        # Try all swaps
        for i in ones_list:
            for j in zeros_list:
                h[i] = 0
                h[j] = 1
                new_c5 = compute_c5(h)
                delta = new_c5 - c5
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h[i] = 1
                h[j] = 0
        
        if best_i < 0:
            no_improve += 1
            if no_improve >= 2:
                break
            continue
        
        h[best_i] = 0
        h[best_j] = 1
        c5 += best_delta
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

But random swaps are inefficient for finding improvements. A better approach: for each 1 in h, try swapping with a few random 0s, and keep the best swap overall. This is a "first improvement" or "best improvement" heuristic.

Actually, let me think about this more carefully. The issue with pure random swaps is that the acceptance rate drops as we approach the local minimum. We need many trials per actual improvement.

A better approach for moderate n (up to ~200):
1. Compute all pair effects at once (vectorized).
2. Find the best swap.
3. Apply it.
4. Repeat.

For step 1, I can vectorize the computation. Given h, the correlation is:
corr = np.correlate(h, 1-h, "full")

If I swap h[i] (1→0) and h[j] (0→1), how does corr change?
- h changes: h[i] -= 1, h[j] += 1
- (1-h) changes: (1-h)[i] += 1, (1-h)[j] -= 1
- corr changes by: (Δh) correlated with (1-h) + h correlated with (Δ(1-h))
  = (-e_i + e_j) * (1-h) + h * (e_i - e_j)
  where e_i is the unit vector at position i

The first term: (-e_i + e_j) correlated with (1-h)
= -correlate(e_i, 1-h) + correlate(e_j, 1-h)

correlate(e_i, v) is just v shifted: it's v[:, i] in the "full" output... actually, let me think.

np.correlate(e_i, v, "full")[k] = Σ_n e_i[n+k] * v[n] = v[i-k] (if 0 ≤ i-k < n, else 0)

So correlate(e_i, v, "full") is v reversed and shifted. Specifically, it's the vector where position (n-1+i-k) has value v[k]... this is getting confusing.

Let me just use the direct formula. For the full correlation of length 2n-1:
correlate(a, b, "full")[p] = Σ_n a[n] * b[n - (p - (n-1))]... 

I'll just stick with the simple approach: compute_c5 for each candidate swap. For n up to 200, this is fast enough.

Let me now write the final, clean code. I'll focus on:
1. Multiple n values (even, for binary)
2. Full-scan LS for n ≤ 120
3. Random-swap LS + ILS for n up to 300
4. Time management

Let me also think about whether the "full" correlation includes the self-correlation (lag 0). For binary h, the lag-0 term is Σ h[i]*(1-h[i]) = 0, so it doesn't contribute to the max. Good.

One more thought: I should also try the "reverse" correlation. The problem says max_k ∫ h(x)(1-h(x+k)) dx. In the discrete correlate, this includes both positive and negative k (i.e., both h before 1-h and 1-h before h). The np.correlate(h, 1-h, "full") gives both directions. So we're correctly capturing the max over all shifts.

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old, dtype=float)
    x_new = np.arange(n_new, dtype=float) * (n_old - 1) / max(1, n_new - 1)
    h_interp = np.interp(x_new, x_old, h_old)
    h_bin = (h_interp >= 0.5).astype(float)
    target = n_new // 2
    current = int(h_bin.sum())
    if current != target:
        if current < target:
            idx0 = np.where(h_bin == 0)[0]
            order = idx0[np.argsort(-h_interp[idx0])]
            for i in order[:target - current]:
                h_bin[i] = 1.0
        else:
            idx1 = np.where(h_bin == 1)[0]
            order = idx1[np.argsort(h_interp[idx1])]
            for i in order[:current - target]:
                h_bin[i] = 0.0
    return h_bin

def full_scan_ls(h, max_scans=200):
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for _ in range(max_scans):
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        best_delta = 0
        best_i, best_j = -1, -1
        for i in ones_list:
            h[i] = 0
            for j in zeros_list:
                h[j] = 1
                new_c5 = compute_c5(h)
                delta = new_c5 - c5
                if delta < best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                h[j] = 0
            h[i] = 1
        if best_i < 0:
            break
        h[best_i] = 0
        h[best_j] = 1
        c5 += best_delta
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def random_swap_ls(h, n_trials, rng):
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    ones_list = np.where(h == 1)[0]
    zeros_list = np.where(h == 0)[0]
    for _ in range(n_trials):
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        if new_c5 <= c5:
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            ones_list = np.where(h == 1)[0]
            zeros_list = np.where(h == 0)[0]
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5

def beam_search_ls(h, n_candidates=200, n_rounds=200, rng=None):
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for _ in range(n_rounds):
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        if rng is None:
            rng = np.random.RandomState(0)
        best_delta = 0
        best_i, best_j = -1, -1
        n_cand = min(n_candidates, len(ones_list) * len(zeros_list))
        for _ in range(n_cand):
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            h[i] = 0
            h[j] = 1
            new_c5 = compute_c5(h)
            delta = new_c5 - c5
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
            h[i] = 1
            h[j] = 0
        if best_i < 0:
            break
        h[best_i] = 0
        h[best_j] = 1
        c5 += best_delta
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def simulated_annealing(h, n_trials, rng, T0=0.05, T_end=0.001):
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for t in range(n_trials):
        T = T0 * (T_end / T0) ** (t / max(1, n_trials - 1))
        ones_list = np.where(h == 1)[0]
        zeros_list = np.where(h == 0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')

    def elapsed():
        return time.time() - t_start

    def remaining_frac():
        return max(0.0, 1.0 - elapsed() / budget_s)

    # Use parent if available
    h_parent = None
    n_parent = 0
    if 'initial_h_values' in dir() and _check_initial():
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
        c5_p = compute_c5(h_parent)
        best_c5 = c5_p
        best_h = h_parent.copy()

    # Phase 1: Optimize at parent resolution if odd (continuous)
    if h_parent is not None and n_parent % 2 == 1 and elapsed() < budget_s * 0.15:
        # Do a quick continuous local search
        h_cont = h_parent.copy()
        c5_cont = compute_c5(h_cont)
        # Try small perturbations
        for _ in range(2000):
            if elapsed() > budget_s * 0.15:
                break
            i = rng.randint(n_parent)
            j = rng.randint(n_parent)
            if i == j:
                continue
            old_i, old_j = h_cont[i], h_cont[j]
            delta = rng.uniform(-0.1, 0.1)
            h_cont[i] = np.clip(old_i + delta, 0, 1)
            h_cont[j] = np.clip(old_j - delta, 0, 1)
            new_c5 = compute_c5(h_cont)
            if new_c5 <= c5_cont:
                c5_cont = new_c5
                if c5_cont < best_c5:
                    best_c5 = c5_cont
                    best_h = h_cont.copy()
            else:
                h_cont[i] = old_i
                h_cont[j] = old_j

    # Phase 2: Binary optimization at various even n
    n_list = [48, 50, 52, 56, 60, 64, 72, 80, 96, 100, 112, 120, 144, 160, 192, 200, 256, 320]

    for n in n_list:
        if remaining_frac() < 0.05:
            break
        time_alloc = remaining_frac() * budget_s / max(1, len(n_list) - n_list.index(n))
        time_alloc = min(time_alloc, 120.0)

        # Initialize
        if h_parent is not None:
            h_init = upsample(h_parent, n)
        else:
            h_init = np.zeros(n)
            idx = rng.choice(n, n // 2, replace=False)
            h_init[idx] = 1.0

        c5_cur = compute_c5(h_init)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_init.copy()

        # Full-scan LS for small n
        if n <= 120 and elapsed() < t_start + time_alloc * 0.5:
            h_ls, c5_ls = full_scan_ls(h_init.copy(), max_scans=min(150, int(time_alloc * 5)))
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
            h_init = h_ls.copy()
        elif n <= 320 and elapsed() < t_start + time_alloc * 0.5:
            n_cand = min(300, (n // 2) * (n // 2))
            n_rnd = min(500, int(time_alloc * 50))
            h_ls, c5_ls = beam_search_ls(h_init.copy(), n_candidates=n_cand, n_rounds=n_rnd, rng=rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
            h_init = h_ls.copy()

        # Random swap LS
        n_rs = min(30000, int(time_alloc * 30))
        if n_rs > 100 and elapsed() < t_start + time_alloc * 0.8:
            h_rs, c5_rs = random_swap_ls(h_init.copy(), n_rs, rng)
            if c5_rs < best_c5:
                best_c5 = c5_rs
                best_h = h_rs.copy()
            h_init = h_rs.copy() if c5_rs < compute_c5(h_init) else h_init.copy()

        # ILS: perturb + local search
        n_ils = max(5, int(time_alloc / max(0.5, n * 0.001)))
        for _ in range(n_ils):
            if remaining_frac() < 0.08:
                break
            h_pert = h_init.copy()
            K = max(3, n // 15)
            ones_l = np.where(h_pert == 1)[0]
            zeros_l = np.where(h_pert == 0)[0]
            for _ in range(K):
                if len(ones_l) == 0 or len(zeros_l) == 0:
                    break
                ii = ones_l[rng.randint(len(ones_l))]
                jj = zeros_l[rng.randint(len(zeros_l))]
                h_pert[ii] = 0
                h_pert[jj] = 1
                ones_l = np.where(h_pert == 1)[0]
                zeros_l = np.where(h_pert == 0)[0]

            if n <= 120:
                h_try, c5_try = full_scan_ls(h_pert, max_scans=50)
            else:
                h_try, c5_try = beam_search_ls(h_pert, n_candidates=200, n_rounds=100, rng=rng)

            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_init = h_try.copy()

        # SA
        if remaining_frac() > 0.1:
            n_sa = min(20000, int(time_alloc * 20))
            h_sa, c5_sa = simulated_annealing(h_init.copy(), n_sa, rng, T0=0.04, T_end=0.0005)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 3: Final refinement on best solution found
    if best_h is not None and remaining_frac() > 0.05:
        n_best = len(best_h)
        if n_best % 2 == 0:
            h_bin = upsample(best_h, n_best)
            c5_cand = compute_c5(h_bin)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_bin.copy()
            if n_best <= 120:
                h_fin, c5_fin = full_scan_ls(h_bin.copy(), max_scans=100)
            else:
                h_fin, c5_fin = beam_search_ls(h_bin.copy(), n_candidates=300, n_rounds=500, rng=rng)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    if best_h is None:
        n = 100
        best_h = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        best_h[idx] = 1.0
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)

def _check_initial():
    try:
        return initial_h_values is not None
    except NameError:
        return False
```
