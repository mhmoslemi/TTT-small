def corr_values(h):
    n = int(h.size)
    if n == 0: return np.empty(0)
    h = np.asarray(h, dtype=float)
    total = float(np.sum(h))
    nfft = get_nfft(n)

```python
import numpy as np
import time
from scipy.optimize import minimize

_NFFT_CACHE = {}


def get_nfft(n):
    if n in _NFFT_CACHE:
        return _NFFT_CACHE[n]
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    _NFFT_CACHE[n] = m
    return m


def top_indices(vals, topk):
    m = vals.size
    if topk <= 0:
        return np.empty(0, dtype=int)
    if topk >= m:
        return np.arange(m)
    k = m - topk
    return np.argpartition(vals, k)[k:]


def corr_values(h):
    n = int(h.size)
    if n == 0:
        return np.empty(0, dtype=float)
    h = np.asarray(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.0)
    total = float(np.sum(h))
    nfft = get_nfft(n)
    F = np.fft.rfft(h, nfft)
    ac = np.fft.irfft(F * np.conjugate(F), nfft).real
    A = ac[:n]
    pre = np.empty(n + 1, dtype=float)
    pre[0] = 0.0
    np.cumsum(h, out=pre[1:])
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = total - A[0]
    if n > 1:
        A1 = A[1:]
        vals[1:n] = pre[n - 1:0:-1] - A1
        vals[n:] = total - pre[1:n] - A1
    vals[vals < 0.0] = 0.0
    return vals


def fft_c5(h):
    n = int(h.size)
    if n == 0:
        return 0.0
    vals = corr_values(h)
    return float((2.0 / float(n)) * float(np.max(vals)))


def exact_c5(h):
    n = int(h.size)
    if n == 0:
        return 0.0
    if n <= 6000:
        c = np.correlate(h, 1.0 - h, mode="full")
        return float((2.0 / float(n)) * float(np.max(c)))
    return fft_c5(h)


def project_simplex(z, target):
    z = np.asarray(z, dtype=float)
    n = z.size
    if n == 0:
        return z
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)
    z = np.where(np.isfinite(z), z, 0.0)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(48):
        mu = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mu, 0.0, 1.0)))
        if s > target:
            lo = mu
        else:
            hi = mu
    mu = 0.5 * (lo + hi)
    p = np.clip(z - mu, 0.0, 1.0)
    for _ in range(4):
        diff = target - float(np.sum(p))
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            mask = p < 1.0 - 1e-14
        else:
            mask = p > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        p[mask] = np.clip(p[mask] + diff / float(cnt), 0.0, 1.0)
    diff = target - float(np.sum(p))
    if abs(diff) > 1e-13:
        if diff > 0:
            idxs = np.where(p < 1.0)[0]
            if idxs.size:
                cap = 1.0 - p[idxs]
                j = int(np.argmax(cap))
                p[idxs[j]] = min(1.0, p[idxs[j]] + diff)
        else:
            idxs = np.where(p > 0.0)[0]
            if idxs.size:
                cap = p[idxs]
                j = int(np.argmin(cap))
                p[idxs[j]] = max(0.0, p[idxs[j]] + diff)
    return p


def make_feasible(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        old = h.size
        if old == 0:
            h = np.zeros(n, dtype=float)
        else:
            if old == 1:
                h = np.full(n, float(h[0]))
            else:
                x_old = np.arange(old, dtype=float)
                x_new = np.linspace(0.0, float(old - 1), n)
                h = np.interp(x_new, x_old, h)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return project_simplex(h, 0.5 * float(n))


def quantize_exact(h, n, p=30):
    n = int(n)
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = interpolate_to_n(h, n)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    scale = 1 << p
    float_scale = float(scale)
    q = np.rint(h * float_scale + 1e-12).astype(np.int64)
    q = np.clip(q, 0, scale).astype(np.int64)
    if n % 2 == 0:
        target = (n // 2) * scale
    else:
        target = int(round(float(n) * 0.5 * float_scale))
    diff = int(target - int(q.sum()))
    if diff != 0:
        if abs(diff) > 100000:
            if diff > 0:
                act = np.where(q < scale)[0]
                if act.size:
                    base = diff // act.size
                    q[act] = np.minimum(scale, q[act] + base)
            else:
                act = np.where(q > 0)[0]
                if act.size:
                    base = (-diff) // act.size
                    q[act] = np.maximum(0, q[act] - base)
            diff = int(target - int(q.sum()))
        if diff > 0:
            act = np.where(q < scale)[0]
            if act.size:
                order = act[np.argsort(q[act], kind="stable")]
                idx = 0
                while diff > 0 and order.size:
                    q[order[idx % order.size]] += 1
                    diff -= 1
                    idx += 1
        elif diff < 0:
            act = np.where(q > 0)[0]
            if act.size:
                order = act[np.argsort(-q[act], kind="stable")]
                idx = 0
                while diff < 0 and order.size:
                    q[order[idx % order.size]] -= 1
                    diff += 1
                    idx += 1
    return q / float_scale


def interpolate_to_n(h, n_new):
    h = np.asarray(h, dtype=float)
    n = h.size
    n_new = int(n_new)
    if n_new == n:
        return h.copy()
    if n == 0 or n_new == 0:
        return np.zeros(n_new, dtype=float)
    if n == 1:
        return np.full(n_new, float(h[0]))
    if n_new % n == 0:
        return np.repeat(h, n_new // n)
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0.0, float(n - 1), n_new)
    return np.interp(x_new, x_old, h)


def random_binary(n, rng):
    n = int(n)
    m = n // 2
    b = np.zeros(n, dtype=float)
    if m > 0 and n > 0:
        idx = rng.choice(n, size=m, replace=False)
        b[idx] = 1.0
    return b


def round_to_binary(h, n, rng):
    h = np.asarray(h, dtype=float)
    if h.size != int(n):
        h = interpolate_to_n(h, int(n))
    n = int(h.size)
    m = n // 2
    b = np.zeros(n, dtype=float)
    if m <= 0 or n <= 0:
        return b
    key = h + rng.uniform(0.0, 1e-6, n)
    key = np.where(np.isfinite(key), key, 0.0)
    idx = np.argpartition(-key, m)[:m]
    b[idx] = 1.0
    return b


def soft_value_grad(x, n, beta, topk):
    n = int(n)
    h = np.asarray(x, dtype=float)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.size), mode="constant")
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(n)
    vals = corr_values(h)
    if not np.all(np.isfinite(vals)):
        return 1e100, np.zeros(n, dtype=float)
    idxs = top_indices(vals, int(topk))
    if idxs.size == 0:
        return float(dx * float(np.max(vals))), np.zeros(n, dtype=float)
    tv = vals[idxs].astype(float, copy=False)
    vmax = float(np.max(tv))
    b = max(1e-6, float(beta))
    e = np.exp(b * (tv - vmax))
    wsum = float(np.sum(e))
    if wsum <= 0.0 or not np.isfinite(wsum):
        soft = vmax
        w = np.zeros(idxs.size, dtype=float)
        w[int(np.argmax(tv))] = 1.0
    else:
        soft = vmax + (1.0 / b) * float(np.log(wsum))
        w = e / wsum

    W0 = 0.0
    Wp = np.zeros(n, dtype=float)
    Wm = np.zeros(n, dtype=float)
    for k in range(idxs.size):
        wi = float(w[k])
        if wi <= 0.0:
            continue
        idx = int(idxs[k])
        if idx == 0:
            W0 += wi
        elif idx <= n - 1:
            Wm[idx] += wi
        else:
            Wp[idx - n + 1] += wi

    grad = W0 * (1.0 - 2.0 * h)
    if n > 1:
        if np.any(Wp):
            cum = np.empty(n, dtype=float)
            cum[0] = 0.0
            np.cumsum(Wp[1:n], out=cum[1:])
            grad = grad + cum
        if np.any(Wm):
            cumm = np.cumsum(Wm[1:n])
            rev = cumm[::-1]
            grad[:n - 1] += rev
        Wcorr = Wp + Wm
        ds = np.nonzero(Wcorr)[0]
        for dd in ds:
            wc = float(Wcorr[dd])
            if wc == 0.0:
                continue
            d = int(dd)
            if d <= 0 or d >= n:
                continue
            grad[d:] -= wc * h[:-d]
            grad[:-d] -= wc * h[d:]
    return float(dx * soft), dx * grad


def obj_func(x, n, beta, topk):
    val, _ = soft_value_grad(x, int(n), float(beta), int(topk))
    return val


def jac_func(x, n, beta, topk):
    _, grad = soft_value_grad(x, int(n), float(beta), int(topk))
    return grad


def eq_fun(x):
    x = np.asarray(x, dtype=float)
    return float(np.sum(x) - 0.5 * float(x.size))


def eq_jac(x):
    return np.ones(np.asarray(x).shape, dtype=float)


def optimize_stage_slsqp(h0, n, maxiter, beta, topk):
    n = int(n)
    h0 = make_feasible(h0, n)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": eq_fun, "jac": eq_jac}]
        res = minimize(
            obj_func,
            h0,
            args=(n, float(beta), int(topk)),
            jac=jac_func,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            maxiter=maxiter,
            ftol=1e-8,
        )
    except Exception:
        return None
    x = getattr(res, "x", None)
    if x is None:
        return None
    h = np.asarray(x, dtype=float)
    return make_feasible(h, n)


def projected_gradient(h0, n, deadline, beta, topk, seed):
    n = int(n)
    h0 = make_feasible(h0, n)
    rng = np.random.default_rng(seed)
    cur_c = fft_c5(h0)
    best_h = h0.copy()
    best_c = cur_c
    it = 0
    no = 0
    escapes = 0
    tlist = (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    while time.time() < deadline:
        it += 1
        if it > 20000:
            break
        val, grad = soft_value_grad(h0, n, beta, topk)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-12:
            break
        dirn = -grad / gnorm
        trial_best = cur_c
        trial_y = None
        for t in tlist:
            if time.time() + 1.0 > deadline:
                break
            y = h0 + t * dirn
            y = make_feasible(y, n)
            c = fft_c5(y)
            if c < trial_best - 1e-11:
                trial_best = c
                trial_y = y.copy()
        if trial_y is not None:
            h0 = trial_y
            cur_c = trial_best
            if cur_c < best_c - 1e-11:
                best_h = h0.copy()
                best_c = cur_c
            no = 0
        else:
            no += 1
            if no == 50 or no == 150 or no == 300:
                scale = 0.02 + 0.03 * (float(no) / 100.0)
                y = best_h + rng.normal(0.0, scale, n)
                y = make_feasible(y, n)
                c = fft_c5(y)
                if c < cur_c + 0.001:
                    h0 = y.copy()
                    cur_c = c
                    if c < best_c:
                        best_h = h0.copy()
                        best_c = c
                    no = 10
            if no > 600:
                escapes += 1
                if escapes > 25:
                    break
                y = best_h + rng.normal(0.0, 0.1, n)
                y = make_feasible(y, n)
                h0 = y.copy()
                cur_c = fft_c5(y)
                no = 0
    return best_h, best_c, h0


def binary_local_search(b0, n, deadline, seed):
    n = int(n)
    rng = np.random.default_rng(seed)
    if b0 is None:
        b = random_binary(n, rng)
    else:
        b = round_to_binary(b0, n, rng)
    b = np.clip(b, 0.0, 1.0)
    cur_c = fft_c5(b)
    best = b.copy()
    best_c = cur_c
    no = 0
    escapes = 0
    topk = 12
    while time.time() < deadline:
        if no > 800:
            break
        vals = corr_values(b)
        idxs = top_indices(vals, topk)
        cands = []
        if idxs.size:
            for idx in idxs:
                idx = int(idx)
                if idx == 0:
                    continue
                if idx <= n - 1:
                    d = idx
                    kind = 0
                else:
                    d = idx - n + 1
                    kind = 1
                m = n - d
                if m <= 0:
                    continue
                if kind == 0:
                    mask = (b[:m] == 1.0) & (b[d:] == 0.0)
                else:
                    mask = (b[d:] == 1.0) & (b[:m] == 0.0)
                cnt = int(np.count_nonzero(mask))
                if cnt == 0:
                    continue
                if cnt == 1:
                    poslist = [int(np.flatnonzero(mask)[0])]
                else:
                    poslist = []
                    for _ in range(min(2, cnt)):
                        pos = int(rng.integers(0, m))
                        if mask[pos]:
                            poslist.append(pos)
                for pos in poslist:
                    if kind == 0:
                        cands.append((pos, pos + d))
                    else:
                        cands.append((pos + d, pos))
        ones = np.flatnonzero(b == 1.0)
        zeros = np.flatnonzero(b == 0.0)
        if ones.size and zeros.size:
            for _ in range(12):
                p = int(ones[int(rng.integers(0, ones.size))])
                q = int(zeros[int(rng.integers(0, zeros.size))])
                if p != q:
                    cands.append((p, q))
        accepted = False
        for p, q in cands:
            if p == q or b[p] != 1.0 or b[q] != 0.0:
                continue
            b[p] = 0.0
            b[q] = 1.0
            c = fft_c5(b)
            if c < cur_c - 1e-11:
                cur_c = c
                accepted = True
                if c < best_c - 1e-11:
                    best = b.copy()
                    best_c = c
                break
            else:
                b[p] = 1.0
                b[q] = 0.0
        if accepted:
            no = 0
        else:
            no += 1
            if no > 100 and no % 20 == 0:
                ones = np.flatnonzero(b == 1.0)
                zeros = np.flatnonzero(b == 0.0)
                if ones.size and zeros.size:
                    p = int(ones[int(rng.integers(0, ones.size))])
                    q = int(zeros[int(rng.integers(0, zeros.size))])
                    if p != q:
                        b[p] = 0.0
                        b[q] = 1.0
                        cur_c = fft_c5(b)
                        if cur_c < best_c:
                            best = b.copy()
                            best_c = cur_c
                        no = 0
                        escapes += 1
                        if escapes > 30:
                            break
    best_exact = exact_c5(best)
    return best, float(best_exact)


def get_parent_initial(seed):
    try:
        h = np.array(initial_h_values, dtype=float, copy=True)
        h = np.where(np.isfinite(h), h, 0.0)
        if h.size > 0:
            return h
    except Exception:
        pass
    rng = np.random.default_rng(seed)
    return random_binary(42, rng)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    total_budget = max(1.0, float(budget_s))
    deadline = start_time + max(1.0, total_budget - 5.0)
    try:
        parent = get_parent_initial(seed)
        base_n = int(parent.size)
        if base_n < 2 or base_n % 2 != 0:
            if base_n > 0:
                parent = interpolate_to_n(parent, 42)
            else:
                parent = random_binary(42, np.random.default_rng(seed))
            base_n = 42
        rng = np.random.default_rng(int(seed) + 1)

        best_h = make_feasible(parent, base_n)
        best_n = base_n
        best_c = exact_c5(best_h) if total_budget > 30.0 else fft_c5(best_h)

        stages = []
        nc = base_n
        while len(stages) < 7 and nc <= 1400:
            stages.append(nc)
            nc *= 2

        stage_slices = [40, 160, 260, 380, 340, 280]

        for si, n in enumerate(stages):
            if time.time() >= deadline:
                break
            remaining = deadline - time.time()
            if remaining <= 2.0:
                break
            slice_s = stage_slices[si] if si < len(stage_slices) else 80.0
            slice_s = min(float(slice_s), remaining)
            stage_end = time.time() + slice_s

            if best_n == n:
                start_h = best_h.copy()
            elif best_h is not None:
                start_h = interpolate_to_n(best_h, n)
            else:
                start_h = random_binary(n, rng)
            start_h = make_feasible(start_h, n)

            if n <= 200 and time.time() + 20.0 < stage_end:
                maxiter = 80 if n <= 60 else (50 if n <= 120 else 30)
                topk_s = min(2 * n - 1, max(32, min(120, n // 3)))
                h1 = optimize_stage_slsqp(start_h, n, maxiter, 20.0, topk_s)
                if h1 is not None:
                    h1 = make_feasible(h1, n)
                    c1_fft = fft_c5(h1)
                    c0_fft = fft_c5(start_h)
                    if c1_fft < c0_fft:
                        start_h = h1
                    if c1_fft < best_c + 0.01:
                        c1e = exact_c5(h1)
                        if c1e < best_c - 1e-12:
                            best_h = h1
                            best_c = c1e
                            best_n = n

            if time.time() < stage_end - 5.0:
                c_start = fft_c5(start_h)
                noise_scale = 0.005 if n <= 84 else 0.01
                noisy = start_h + rng.normal(0.0, noise_scale, n)
                noisy = make_feasible(noisy, n)
                c_noisy = fft_c5(noisy)
                if c_noisy < c_start:
                    start_h = noisy

            if time.time() + 5.0 < stage_end:
                topk_pg = min(2 * n - 1, max(16, min(64, n // 4)))
                h2, c2_fft, _ = projected_gradient(start_h, n, stage_end, 25.0, topk_pg, int(seed) + n)
                if h2 is not None and c2_fft < best_c + 0.01:
                    h2 = make_feasible(h2, n)
                    c2e = exact_c5(h2)
                    if c2e < best_c - 1e-12:
                        best_h = h2
                        best_c = c2e
                        best_n = n

            if n <= 252 and time.time() + 40.0 < stage_end:
                if best_n == n:
                    bin_start = best_h
                elif best_h is not None:
                    bin_start = interpolate_to_n(best_h, n)
                else:
                    bin_start = start_h
                bh, bc = binary_local_search(bin_start, n, min(stage_end, deadline), int(seed) * 10000 + n)
                if bh is not None and bc < best_c - 1e-12:
                    bh = make_feasible(bh, n)
                    bc2 = exact_c5(bh)
                    if bc2 < best_c - 1e-12:
                        best_h = bh
                        best_c = bc2
                        best_n = n

        if best_h is None:
            best_h = make_feasible(parent, base_n)
            best_n = base_n
            best_c = exact_c5(best_h)

        if time.time() + 30.0 < deadline:
            topk_final = min(2 * best_n - 1, max(16, min(64, best_n // 4)))
            h3 = make_feasible(best_h, best_n)
            h3 = h3 + rng.normal(0.0, 0.05, best_n)
            h3 = make_feasible(h3, best_n)
            h4, c4_fft, _ = projected_gradient(h3, best_n, deadline, 25.0, topk_final, int(seed) + 99999)
            if h4 is not None and c4_fft < best_c + 0.01:
                h4 = make_feasible(h4, best_n)
                c4e = exact_c5(h4)
                if c4e < best_c - 1e-12:
                    best_h = h4
                    best_c = c4e
                    best_n = best_n

        final_h = make_feasible(best_h, best_n)
        c_raw = exact_c5(final_h)
        qh = quantize_exact(final_h, best_n, 30)
        c_q = exact_c5(qh)
        sum_dev = abs(float(np.sum(final_h)) - 0.5 * float(best_n))

        if sum_dev < 1e-10 and c_raw < c_q - 1e-7:
            final_h = final_h
            c_final = c_raw
        else:
            final_h = qh
            c_final = c_q

        raw2 = make_feasible(best_h, best_n)
        cr2 = exact_c5(raw2)
        q2 = quantize_exact(raw2, best_n, 30)
        cq2 = exact_c5(q2)
        if cr2 < c_final and abs(float(np.sum(raw2)) - 0.5 * float(best_n)) < 1e-8:
            final_h = raw2
            c_final = cr2
        elif cq2 < c_final:
            final_h = q2
            c_final = cq2

        return (np.ascontiguousarray(final_h, dtype=float), float(c_final), int(best_n))

    except Exception:
        try:
            p = get_parent_initial(seed)
            n = int(p.size)
            if n < 2 or n % 2 != 0:
                if n > 0:
                    p = interpolate_to_n(p, 42)
                else:
                    p = random_binary(42, np.random.default_rng(seed))
                n = 42
            h = make_feasible(p, n)
            c = exact_c5(h)
            q = quantize_exact(h, n, 30)
            cq = exact_c5(q)
            if cq <= c + 1e-8:
                return (np.ascontiguousarray(q, dtype=float), float(cq), int(n))
            return (np.ascontiguousarray(h, dtype=float), float(c), int(n))
        except Exception:
            h = np.full(42, 0.5, dtype=float)
            return (h, 0.5, 42)
```
