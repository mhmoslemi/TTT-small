import numpy as np
from scipy.optimize import minimize
import time

def compute_C5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def compute_all_overlaps(h, n):
    """Compute f_s = dx * sum h[i+s]*(1-h[i]) for all valid s"""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv * dx  # length 2n-1

def gradient_f_s(h, n, s):
    """Gradient of f_s with respect to h, where s is the shift index in correlate output
    In correlate output, index k corresponds to shift s = k - (n-1)"""
    dx = 2.0 / n
    shift = s - (n - 1)  # actual shift
    grad = np.zeros(n)
    
    if shift >= 0:
        # I_s = {i : 0 ≤ i ≤ n-1-shift}
        # ∂f/∂h[m] = dx * [(m-shift)*(1-h[m-shift]) - (m ≤ n-1-shift)*h[m+shift]]
        for m in range(n):
            g = 0.0
            if m >= shift and m - shift <= n - 1 - shift:
                # m-shift is valid i, contributes (1-h[m-shift])
                # But need m-shift ≤ n-1-shift, which is m ≤ n-1, always true
                g += (1.0 - h[m - shift])
            if m <= n - 1 - shift:
                # m is valid i, contributes -h[m+shift]
                g -= h[m + shift]
            grad[m] = dx * g
    else:
        # shift < 0, let t = -shift
        t = -shift
        # I_s = {i : t ≤ i ≤ n-1}
        # ∂f/∂h[m] = dx * [(m ≤ n-1-t)*(1-h[m+t]) - (m ≥ t)*h[m-t]]
        for m in range(n):
            g = 0.0
            if m <= n - 1 - t:
                # m+t is valid, contributes (1-h[m+t])
                # Need m+t ≤ n-1, i.e., m ≤ n-1-t
                g += (1.0 - h[m + t])
            if m >= t:
                # m is valid i, contributes -h[m-t]
                g -= h[m - t]
            grad[m] = dx * g
    
    return grad

def softmax_gradient(h, n, beta):
    """Compute gradient of soft-max: (1/beta)*log(sum exp(beta*f_s))"""
    f = compute_all_overlaps(h, n)
    # Softmax weights
    f_shifted = f - np.max(f)  # for numerical stability
    weights = np.exp(beta * f_shifted)
    weights /= np.sum(weights)
    
    # Gradient = sum_k weights[k] * grad(f_k)
    grad = np.zeros(n)
    dx = 2.0 / n
    for k in range(2*n-1):
        if weights[k] < 1e-15:
            continue
        grad += weights[k] * gradient_f_s(h, n, k)
    return grad

def project_feasible(h, n):
    """Project to 0≤h≤1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(50):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        # Need to increase or decrease
        diff = target - s
        if diff > 0:
            # Increase h where possible
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room < 1e-12:
                break
            h += diff * room / total_room
        else:
