import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def _grad_c5(h):
    """Gradient of C5 = max_k correlate(h,1-h,'full')[k] * dx with respect to h"""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    j = int(np.argmax(conv))
    offset = j - (n - 1)
    
    # result[j] = sum_i h[i+offset] * (1-h[i]) for valid i
    i_lo = max(0, -offset)
    i_hi = min(n, n - offset)  # exclusive
    if i_lo >= i_hi:
        return np.zeros(n)
    
    idx = np.arange(i_lo, i_hi)
    g = np.zeros(n)
    # d/dh[i+offset]: contributes (1-h[i])
    g[idx + offset] += (1.0 - h[idx])
    # d/dh[i]: contributes -h[i+offset]
    g[idx] += (-h[idx + offset])
    return g * dx

def _fix_constraint(h, n):
    """Ensure sum(h) = n/2 and 0 <= h <= 1"""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-10:
        h = h * (n / 2.0) / s
    else:
        h = np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute evenly among elements not at boundary
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _fix(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _grad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    j = int(np.argmax(conv))
    off = j - (n - 1)
    lo = max(0, -off)
    hi = min(n, n - off)
    g = np.zeros(n)
    if lo < hi:
        idx = np.arange(lo, hi)
        g[idx + off] += 1.0 - h[idx]
        g[idx] -= h[idx + off]
    return g * dx

def _cons(h):
    return np.sum(h) - len(h) / 2.0

def _cons_jac(h):
    return np.ones(len(h))

def _sm_c5(h, tau=40.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    return float(m + np.log(np.sum(np.exp(tau * (c - m)))) / tau)

def _sm_grad(h, tau=40.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = c * dx
    m = np.max(cs)
    e = np.exp(tau * (cs - m))
    w = e / np.sum(e)
    g = np.zeros(n)
    for j in range(len(c)):
        off = j - (n - 1)
        lo = max(0, -off)
        hi = min(n, n - off)
        if lo >= hi:
            continue
        idx = np.arange(lo, hi)
        g[idx + off] += w[j] * (1.0 - h[idx])
        g[idx] -= w[j] * h[idx + off]
    val = m + np.log(np.sum(e)) / tau
    return float(val), g * dx

def _sm_fun(h):
    return _sm_c5(h)

def _sm_jac(h):
    return _sm_grad(h)[1]

def run(seed=42, budget_s=1300, **kwargs):

    t0 = time.time()
    
    # Get starting point

        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)
    n = len(h0)
    h0 = _fix(h0, n)
    
    best_h = h0.copy()
    best_c5 = _c5(best_h)
    best_n = n
    print(f"Init: C5={best_c5:.6f}, n={n}", flush=True)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}]
    
    # SLSQP with exact objective
    res = minimize(_c5, h0, jac=_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 5000, 'ftol': 1e-16})
    if res.x is not None:
        h_new = _fix(res.x, n)
        c5_new = _c5(h_new)
        if c5_new < best_c5:

    print(f"SLSQP: C5={best_c5:.6f} ({time.time()-t0:.1f}s)", flush=True)

        if res2.x is not None:

            c5_new = _c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
    print(f"Smooth: C5={best_c5:.6f} ({time.time()-t0:.1f}s)", flush=True)
    
    # Restarts

    if time.time() - t0 < budget_s - 50:

            x_old = np.linspace(0, 2, n)

            h_hi = _fix(np.interp(x_new, x_old, best_h), n2)
