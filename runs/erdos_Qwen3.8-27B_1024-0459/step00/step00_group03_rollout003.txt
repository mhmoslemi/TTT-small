def compute_separates(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / max(1, n)
    g = 1.0 - h
    Splus = np.empty(n, dtype=float)

        Splus[m] = float(np.dot(h[:sl], g[m:]))

def lse_weights(Splus,Sminus,gamma):
    n=Splus.size
    if n==0: return 0.0,0.0,np.array([0.0])
    if n==1:
        vals=Splus
    else:
        vals=np.concatenate((Splus, Sminus[1:]))
    smax=float(np.max(vals))
    if gamma <= 1e-6:
        w = np.full(vals.size, 1.0/vals.size)
        F = float(np.mean(vals))
        return F, smax, w
    z = gamma * (vals - smax)
    z = np.where(z < -745.0, -745.0, z)
    ez = np.exp(z)
    se = float(np.sum(ez))
    if se <= 0.0 or not math.isfinite(se):
        # fallback to max-heuristic
        w = np.full(vals.size, 0.0)
        max_idx = int(np.argmax(vals)); w[max_idx]=1.0
        return smax, smax, w
    w = ez / se
    F = smax + math.log(se) / gamma
    return float(F), smax, w

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize as scipy_minimize

def _sanitize_array(arr, value=0.5):
    a = np.array(arr, dtype=float, copy=True)
    if a.size == 0:
        return a
    bad = ~np.isfinite(a)
    if np.any(bad):
        a[bad] = float(value)
    return a


def normalize_sum(h, target):
    h = _sanitize_array(h, 0.5)
    h = np.clip(h, 0.0, 1.0)
    if h.size == 0:
        return h
    diff = float(target) - float(np.sum(h))
    if not math.isfinite(diff) or abs(diff) < 1e-14:
        return h
    n = h.size
    inc = diff / float(n)

    if inc > 0.0:
        if h.max() + inc <= 1.0 + 1e-12:
            h = h + inc
            return np.clip(h, 0.0, 1.0)
    else:
        if inc < 0.0 and h.min() + inc >= -1e-12:
            h = h + inc
            return np.clip(h, 0.0, 1.0)

    rem = float(diff)
    if rem > 0.0:
        slack = 1.0 - h
        order = np.argsort(-slack)
        for idx in order:
            if rem <= 1e-15:
                break
            take = min(float(slack[idx]), rem)
            h[idx] += take
            rem -= take
    else:
        rem = -rem
        order = np.argsort(h)
        for idx in order:
            if rem <= 1e-15:
                break
            take = min(float(h[idx]), rem)
            h[idx] -= take
            rem -= take
    return np.clip(h, 0.0, 1.0)


def project_box_sum(v, target, iters=35):
    v = _sanitize_array(v, 0.5)
    target = float(target)
    if v.size == 0:
        return v

    clipped = np.clip(v, 0.0, 1.0)
    if abs(float(np.sum(clipped)) - target) <= 1e-12:
        return normalize_sum(clipped, target)

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

    for _ in range(max(10, int(iters))):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return normalize_sum(h, target)


def compute_separates(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float), np.empty(0, dtype=float)

    dx = 2.0 / float(n)
    g = 1.0 - h

    sp = np.empty(n, dtype=float)
    sm = np.empty(n, dtype=float)

    v0 = float(np.dot(h, g))
    sp[0] = v0
    sm[0] = v0

    for m in range(1, n):
        sl = n - m
        sp[m] = float(np.dot(h[:sl], g[m:]))
        sm[m] = float(np.dot(h[m:], g[:sl]))

    return sp * dx, sm * dx


def separable_c5(sp, sm):
    n = sp.size
    if n == 0:
        return 1e100
    c = float(np.max(sp))
    if n > 1:
        c2 = float(np.max(sm[1:]))
        if c2 > c:
            c = c2
    if not math.isfinite(c) or c < 0.0:
        return 0.0
    return c


def lse_weights(sp, sm, gamma):
    n = sp.size
    if n == 0:
        return 0.0, 0.0, np.array([0.0], dtype=float)

    if n == 1:
        vals = sp
    else:
        vals = np.concatenate((sp, sm[1:]))

    smax = float(np.max(vals))
    if gamma <= 1e-6:
        w = np.full(vals.size, 1.0 / vals.size, dtype=float)
        return float(np.mean(vals)), smax, w

    z = float(gamma) * (vals - smax)
    z = np.where(z < -745.0, -745.0, z)
    ez = np.exp(z)
    se = float(np.sum(ez))

    if se <= 0.0 or not math.isfinite(se):
        w = np.zeros(vals.size, dtype=float)
        w[int(np.argmax(vals))] = 1.0
        return smax, smax, w

    w = ez / se
    F = smax + math.log(se) / float(gamma)
    return float(F), smax, w


def active_lse_weights(sp, sm, tol):
    n = sp.size
    if n == 0:
        return 0.0, np.array([0.0], dtype=float)
    if n == 1:
        vals = sp
    else:
        vals = np.concatenate((sp, sm[1:]))

    smax = float(np.max(vals))
    active = vals >= (smax - float(tol))
    cnt = int(np.count_nonzero(active))

    if cnt == 0:
        active = np.zeros(vals.size, dtype=bool)
        active[int(np.argmax(vals))] = True
        cnt = 1

    w = np.zeros(vals.size, dtype=float)
    w[active] = 1.0 / float(cnt)
    return smax, w


def gradient_from_weights(h, P, Q, dx):
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float)

    grad = np.zeros(n, dtype=float)

    for m in range(n):
        pm = float(P[m]) if m < P.size else 0.0
        if pm <= 1e-14:
            continue
        if m == 0:
            grad[:] += pm * (1.0 - 2.0 * h)
        else:
            sl = n - m
            grad[:sl] += pm * (1.0 - h[m:])
            grad[m:] += -pm * h[:sl]

    for m in range(1, min(n, Q.size)):
        qm = float(Q[m])
        if qm <= 1e-14:
            continue
        sl = n - m
        grad[m:] += qm * (1.0 - h[:sl])
        grad[:sl] += -qm * h[m:]

    grad *= float(dx)
    return grad


def exact_c5_numpy(h):
    h = _sanitize_array(h, 0.5)
    n = h.size
    if n == 0:
        return 1.0
    fc = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(fc)) * (2.0 / float(n))
    if not math.isfinite(c) or c < 0.0:
        return 0.0
    return c


def make_dyadic_exact(h, n, q=20):
    h = _sanitize_array(h, 0.5)
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)

    if h.size != n:
        if h.size == 0:
            h = np.full(n, 0.5, dtype=float)
        else:
            x0 = (np.arange(h.size, dtype=float) + 0.5) * (2.0 / float(h.size))
            x1 = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
            h = np.interp(x1, x0, h)

    target = 0.5 * float(n)
    h = project_box_sum(h, target, iters=50)

    Q = 1 << q
    target_q = (int(n) * Q) // 2

    r = h * float(Q)
    w = np.rint(r).astype(np.int64)
    w = np.clip(w, 0, Q)

    s = int(np.sum(w, dtype=np.int64))
    diff = int(target_q) - s

    guard = 0
    while diff != 0 and guard < 100000:
        guard += 1
        if diff > 0:
            valid = np.nonzero(w < Q)[0]
            if valid.size == 0:
                break
            frac = r - w.astype(float)
            idx = valid[int(np.argmax(frac[valid]))]
            w[idx] += 1
            diff -= 1
        else:
            valid = np.nonzero(w > 0)[0]
            if valid.size == 0:
                break
            frac = r - w.astype(float)
            idx = valid[int(np.argmin(frac[valid]))]
            w[idx] -= 1
            diff += 1

    if diff != 0:
        # Fallback: preserve feasibility by clipping and redistributing a little.
        out = np.clip(w.astype(float) / float(Q), 0.0, 1.0)
        out = project_box_sum(out, target, iters=60)
        return out

    return np.clip(w.astype(float) / float(Q), 0.0, 1.0)


def upsample_to(h0, n_target):
    h0 = np.asarray(h0, dtype=float)
    n_target = int(n_target)
    if n_target <= 0:
        return np.empty(0, dtype=float)
    if h0.size == 0:
        return np.full(n_target, 0.5, dtype=float)

    n0 = h0.size
    target = 0.5 * float(n_target)

    if n0 == n_target:
        return project_box_sum(h0, target, iters=45)

    if n_target % n0 == 0:
        h = np.repeat(h0, n_target // n0)
    else:
        x0 = (np.arange(n0, dtype=float) + 0.5) * (2.0 / float(n0))
        x1 = (np.arange(n_target, dtype=float) + 0.5) * (2.0 / float(n_target))
        h = np.interp(x1, x0, h0)

    return project_box_sum(h, target, iters=45)


def symmetrize_reverse(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return h
    h = 0.5 * (h + h[::-1])
    return project_box_sum(h, 0.5 * float(n), iters=45)


def exactify_candidates(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    he = make_dyadic_exact(h, n)
    ce = exact_c5_numpy(he)
    try:
        hs = symmetrize_reverse(h)
        he2 = make_dyadic_exact(hs, n)
        ce2 = exact_c5_numpy(he2)
        if ce2 < ce:
            return he2, ce2
    except Exception:
        pass
    return he, ce


def get_initial_parent():
    try:
        parent = initial_h_values
    except NameError:
        return None
    if parent is None:
        return None
    arr = np.array(parent, dtype=float, copy=True).ravel()
    if arr.size == 0:
        return None
    bad = ~np.isfinite(arr)
    if np.any(bad):
        arr[bad] = 0.5
    return np.clip(arr, 0.0, 1.0)


def make_initial_h(n, seed):
    n = int(n)
    rng = np.random.default_rng(seed)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    a = 0.9 * np.sin(math.pi * x)
    a += 0.1 * (rng.random(n) - 0.5)
    mx = float(np.max(np.abs(a)))
    if mx > 1.0:
        a = a / mx
    h = 0.5 + 0.5 * a
    return project_box_sum(h, 0.5 * float(n), iters=60)


def make_random_smooth(n, seed, guide=None):
    n = int(n)
    rng = np.random.default_rng(seed)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    a = np.zeros(n, dtype=float)

    for _ in range(5):
        f = int(rng.integers(1, 5))
        amp = float(rng.uniform(0.20, 0.80))
        ph = float(rng.uniform(0.0, 2.0 * math.pi))
        a += amp * np.sin(math.pi * float(f) * x + ph)

    mx = float(np.max(np.abs(a)))
    if mx > 1.0:
        a = a / mx

    h = 0.5 + 0.5 * a
    h = project_box_sum(h, 0.5 * float(n), iters=60)

    if guide is not None and guide.size == n:
        h = project_box_sum(0.75 * guide + 0.25 * h, 0.5 * float(n), iters=60)
    return h


def optimize_from(h0, n, end_time, seed):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float), 1.0

    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    h = project_box_sum(h0, target, iters=45)
    rng = np.random.default_rng(seed)

    sp, sm = compute_separates(h)
    best_c = separable_c5(sp, sm)
    best_h = h.copy()

    now = time.time()
    if now >= float(end_time) - 2.0:
        return best_h, best_c

    total = max(0.0, float(end_time) - now)
    if total < 3.0:
        return best_h, best_c

    if total > 80.0:
        soft_deadline = now + 0.68 * total
        active_deadline = float(end_time)
    else:
        soft_deadline = float(end_time)
        active_deadline = None

    if soft_deadline < now + 5.0:
        soft_deadline = min(float(end_time), now + max(2.0, 0.5 * total))
        active_deadline = None

    soft_time = max(0.0, soft_deadline - time.time())
    if soft_time > 180.0:
        gammas = [1.0, 5.0, 25.0, 120.0, 600.0]
    elif soft_time > 80.0:
        gammas = [2.0, 15.0, 80.0, 300.0]
    elif soft_time > 30.0:
        gammas = [5.0, 40.0, 220.0]
    else:
        gammas = [20.0, 200.0]

    # Smooth softmax-projected gradient phase.
    for gi, gamma in enumerate(gammas):
        now = time.time()
        if now >= soft_deadline - 2.0:
            break

        remaining = max(1, len(gammas) - gi)
        sub_end = soft_deadline - 2.0
        sub_end = min(sub_end, now + max(2.0, max(0.0, soft_deadline - now - 2.0) / float(remaining)))
        sub_end = min(sub_end, float(end_time) - 2.0)

        step = min(15.0, 5.0 / max(0.5, float(gamma) * dx))
        step = max(0.02, step)
        max_step = 30.0
        min_step = 1e-6

        fail_count = 0
        no_improve = 0
        iters = 0

        while time.time() < sub_end - 1.0 and iters < 30000:
            iters += 1
            h = np.clip(h, 0.0, 1.0)
            sp, sm = compute_separates(h)
            ccur = separable_c5(sp, sm)

            if ccur < best_c - 1e-12:
                best_c = ccur
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            F0, _, w = lse_weights(sp, sm, gamma)
            if not math.isfinite(F0):
                h = project_box_sum(h + 0.05 * (rng.random(n) - 0.5), target, iters=35)
                step = min(15.0, 5.0 / max(0.5, float(gamma) * dx))
                continue

            P = np.zeros(n, dtype=float)
            Q = np.zeros(n, dtype=float)
            if w.size >= n:
                P[:] = w[:n]
            if w.size > n:
                Q[1:] = w[n:]

            grad = gradient_from_weights(h, P, Q, dx)
            gnrm = float(np.linalg.norm(grad))
            if not math.isfinite(gnrm) or gnrm < 1e-15:
                h = project_box_sum(h + 0.03 * (rng.random(n) - 0.5), target, iters=35)
                step = min(15.0, 5.0 / max(0.5, float(gamma) * dx))
                continue

            accepted = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                s = step * factor
                if s < min_step:
                    break
                htrial = project_box_sum(h - s * grad, target, iters=30)
                stp, stn = compute_separates(htrial)
                Ftrial, _, _ = lse_weights(stp, stn, gamma)
                ctrial = separable_c5(stp, stn)

                if math.isfinite(Ftrial) and (Ftrial < F0 - 1e-13 or ctrial < ccur - 1e-12):
                    h = htrial
                    if Ftrial < F0 - 1e-13:
                        step = min(max_step, step * 1.22)
                    else:
                        step = max(min_step, step * 0.9)
                    accepted = True
                    break

            if not accepted:
                fail_count += 1
                step = max(min_step, step * 0.62)
                if fail_count > 25 or step < 1e-5:
                    noise = 0.03 if gamma > 20.0 else 0.06
                    h = project_box_sum(h + noise * (rng.random(n) - 0.5), target, iters=35)
                    step = min(15.0, 5.0 / max(0.5, float(gamma) * dx))
                    fail_count = 0

            if no_improve > 120:
                noise = 0.02 if gamma > 20.0 else 0.05
                h = project_box_sum(h + noise * (rng.random(n) - 0.5), target, iters=35)
                step = min(15.0, 5.0 / max(0.5, float(gamma) * dx))
                no_improve = 0

    # Direct max-reduction phase using nearby-active lag subgradients.
    if active_deadline is not None and time.time() < active_deadline - 3.0:
        active_tol = 1e-4
        step = min(6.0, 3.0 / max(0.5, dx * 10.0))
        step = max(0.02, step)
        max_step = 20.0
        min_step = 1e-6
        fail_count = 0
        no_improve = 0
        iters = 0

        while time.time() < active_deadline - 2.0 and iters < 30000:
            iters += 1
            h = np.clip(h, 0.0, 1.0)
            sp, sm = compute_separates(h)
            ccur = separable_c5(sp, sm)

            if ccur < best_c - 1e-12:
                best_c = ccur
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            _, w = active_lse_weights(sp, sm, active_tol)
            P = np.zeros(n, dtype=float)
            Q = np.zeros(n, dtype=float)
            if w.size >= n:
                P[:] = w[:n]
            if w.size > n:
                Q[1:] = w[n:]

            grad = gradient_from_weights(h, P, Q, dx)
            gnrm = float(np.linalg.norm(grad))
            if not math.isfinite(gnrm) or gnrm < 1e-15:
                h = project_box_sum(h + 0.025 * (rng.random(n) - 0.5), target, iters=35)
                step = min(6.0, 3.0 / max(0.5, dx * 10.0))
                continue

            accepted = False
            for factor in (1.0, 0.5, 0.25, 0.125):
                s = step * factor
                if s < min_step:
                    break
                htrial = project_box_sum(h - s * grad, target, iters=30)
                stp, stn = compute_separates(htrial)
                ctrial = separable_c5(stp, stn)

                if ctrial < ccur - 1e-14:
                    h = htrial
                    step = min(max_step, step * 1.18)
                    accepted = True
                    break

            if not accepted:
                fail_count += 1
                step = max(min_step, step * 0.6)
                if fail_count > 25 or step < 1e-5:
                    h = project_box_sum(h + 0.025 * (rng.random(n) - 0.5), target, iters=35)
                    step = min(6.0, 3.0 / max(0.5, dx * 10.0))
                    fail_count = 0

            if no_improve > 90:
                h = project_box_sum(h + 0.03 * (rng.random(n) - 0.5), target, iters=35)
                step = min(6.0, 3.0 / max(0.5, dx * 10.0))
                no_improve = 0

    return best_h, best_c


def slsqp_objective(x, gamma):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0 or not np.all(np.isfinite(x)):
        return 1e100
    h = np.clip(x, 0.0, 1.0)
    sp, sm = compute_separates(h)
    F, _, _ = lse_weights(sp, sm, float(gamma))
    return float(F)


def slsqp_gradient(x, gamma):
    x = np.asarray(x, dtype=float)
    n = x.size
    if n == 0:
        return np.empty(0, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)

    h = np.clip(x, 0.0, 1.0)
    sp, sm = compute_separates(h)
    F, _, w = lse_weights(sp, sm, float(gamma))
    if not math.isfinite(F):
        return np.zeros(n, dtype=float)

    P = np.zeros(n, dtype=float)
    Q = np.zeros(n, dtype=float)
    if w.size >= n:
        P[:] = w[:n]
    if w.size > n:
        Q[1:] = w[n:]

    grad = gradient_from_weights(h, P, Q, 2.0 / float(n))
    if not np.all(np.isfinite(grad)):
        return np.zeros(n, dtype=float)
    return grad


def slsqp_eq_fun(x):
    return np.array([float(np.sum(x) - 0.5 * len(x))], dtype=float)


def slsqp_eq_jac(x):
    return np.ones((1, len(x)), dtype=float)


def polish_slsqp(h0, n, end_time):
    if scipy_minimize is None:
        return None
    n = int(n)
    if n <= 0 or n > 280:
        return None

    best = np.asarray(h0, dtype=float).copy()
    sp, sm = compute_separates(best)
    best_c = separable_c5(sp, sm)

    for gamma in (30.0, 200.0, 800.0):
        rem = float(end_time) - time.time()
        if rem < 22.0:
            break
        maxiter = int(max(6, min(25, max(1.0, rem / 8.0))))
        try:
            res = scipy_minimize(
                slsqp_objective,
                best,
                method="SLSQP",
                bounds=[(0.0, 1.0)] * n,
                args=(gamma,),
                jac=slsqp_gradient,
                constraints={"type": "eq", "fun": slsqp_eq_fun, "jac": slsqp_eq_jac},
                options={"maxiter": maxiter, "ftol": 1e-10, "disp": False}
            )
            xx = res.x
            if xx is not None and np.all(np.isfinite(xx)):
                hh = project_box_sum(xx, 0.5 * float(n), iters=50)
                c = separable_c5(*compute_separates(hh))
                if c < best_c - 1e-12:
                    best = hh
                    best_c = c
        except Exception:
            pass

    return best


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    if budget_s is None:
        budget = 1300.0
    else:
        budget = float(budget_s)
    if not math.isfinite(budget) or budget < 20.0:
        budget = max(20.0, float(budget) if math.isfinite(budget) else 20.0)

    seed = int(seed) if seed is not None else 42
    deadline = t0 + budget - 4.0

    parent = get_initial_parent()
    if parent is None:
        base_n = 70
        base_h = make_initial_h(base_n, seed)
    else:
        base_n = int(parent.size)
        if base_n <= 0:
            base_n = 70
            base_h = make_initial_h(base_n, seed)
        else:
            target0 = 0.5 * float(base_n)
            if abs(float(np.sum(parent)) - target0) <= 1e-8:
                base_h = np.clip(parent, 0.0, 1.0)
            else:
                base_h = project_box_sum(parent, target0, iters=60)

    best_h, best_c = exactify_candidates(base_h, base_n)
    best_n = base_n

    # Also try a reverse-symmetrized version of the starting point.
    sh, sc = exactify_candidates(symmetrize_reverse(base_h), base_n)
    if sc < best_c - 1e-12:
        best_h, best_c, best_n = sh, sc, base_n

    N_done = set()
    N_list = []
    for f in (1, 2, 3, 4, 5, 6, 8):
        N = base_n * f
        if 0 < N <= 1200 and N not in N_done:
            N_list.append(N)
            N_done.add(N)

    for idx, N in enumerate(N_list):
        now = time.time()
        if now > deadline - 30.0:
            break

        rem_count = max(1, len(N_list) - idx)
        remaining = max(0.0, deadline - now - 10.0)
        stage_time = min(350.0, max(20.0, remaining / float(rem_count)))
        stage_end = min(deadline - 5.0, now + stage_time)
        if stage_end <= now + 3.0:
            break

        init = upsample_to(best_h, N)
        h1, c1 = optimize_from(init, N, stage_end, seed + 1000 * idx + 7)
        h1e, c1e = exactify_candidates(h1, N)
        if c1e < best_c - 1e-12:
            best_h, best_c, best_n = h1e, c1e, N

        leftover = max(0.0, stage_end - time.time())
        if leftover > 45.0 and N > base_n:
            end_rand = min(deadline - 8.0, time.time() + 0.75 * leftover)
            if end_rand > time.time() + 25.0:
                guide = upsample_to(best_h, N)
                init_rand = make_random_smooth(N, seed + 2000 * idx + 11, guide=guide)
                h2, c2 = optimize_from(init_rand, N, end_rand, seed + 3000 * idx + 31)
                h2e, c2e = exactify_candidates(h2, N)
                if c2e < best_c - 1e-12:
                    best_h, best_c, best_n = h2e, c2e, N

    # Final SLSQP polish on the current best resolution if feasible.
    polish_time = deadline - 5.0 - time.time()
    if scipy_minimize is not None and polish_time > 35.0 and best_n <= 280:
        end_pol = min(deadline - 4.0, time.time() + min(120.0, 0.6 * polish_time))
        hp = polish_slsqp(best_h, best_n, end_pol)
        if hp is not None:
            hpe, cpe = exactify_candidates(hp, best_n)
            if cpe < best_c - 1e-12:
                best_h, best_c, best_n = hpe, cpe, best_n

    # Optional polish at the original resolution if it is much smaller.
    extra = deadline - 5.0 - time.time()
    if scipy_minimize is not None and extra > 45.0 and base_n <= 210 and base_n != best_n:
        init_base = upsample_to(best_h, base_n)
        end_base = min(deadline - 4.0, time.time() + min(90.0, 0.5 * extra))
        hp2 = polish_slsqp(init_base, base_n, end_base)
        if hp2 is not None:
            hpe2, cpe2 = exactify_candidates(hp2, base_n)
            if cpe2 < best_c - 1e-12:
                best_h, best_c, best_n = hpe2, cpe2, base_n

    # Final exact short restarts from the best feasible candidate.
    restarts = 0
    while time.time() < deadline - 20.0 and restarts < 3:
        now = time.time()
        end_r = min(deadline - 8.0, now + min(80.0, 0.5 * max(0.0, deadline - now)))
        if end_r <= now + 15.0:
            break
        hr, cr = optimize_from(best_h, best_n, end_r, seed + 9000 + restarts * 137)
        hre, cre = exactify_candidates(hr, best_n)
        if cre < best_c - 1e-12:
            best_h, best_c, best_n = hre, cre, best_n
            restarts += 1
        else:
            break

    # Ensure final return is exactly feasible in the strong dyadic sense.
    final_h = make_dyadic_exact(best_h, best_n)
    final_c = exact_c5_numpy(final_h)

    alt_h, alt_c = exactify_candidates(symmetrize_reverse(best_h), best_n)
    if alt_c < final_c - 1e-12:
        final_h, final_c = alt_h, alt_c

    return final_h, float(final_c), int(best_n)
