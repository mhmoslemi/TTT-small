from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, full) * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def interpolate_h(h_old, n_new):
    """Interpolate h from old resolution to new resolution."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2.0, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    # Adjust to satisfy constraint
    target_sum = n_new / 2.0
    h_new += (target_sum - np.sum(h_new)) / n_new
    h_new = np.clip(h_new, 0, 1)
    # Fix constraint again after clipping
    if np.sum(h_new) != target_sum:
        diff = target_sum - np.sum(h_new)
        # Distribute evenly
        h_new += diff / n_new
        h_new = np.clip(h_new, 0, 1)
    return h_new

def local_search(h, start_c5, rng, time_limit, max_delta=0.5):
    """Local search with simulated annealing."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = start_c5
    curr_c5 = start_c5
    
    start = time.time()
    temp = 0.005
    cool_rate = 0.9995
    min_temp = 1e-7

        j = rng.integers(n)

            j = rng.integers(n)
        
        delta = rng.uniform(0.1, max_delta)
        direction = 1 if rng.random() < 0.5 else -1
        delta *= direction
        
        new_ci = h[i] + delta
        new_cj = h[j] - delta
        
        if new_ci < 0 or new_ci > 1 or new_cj < 0 or new_cj > 1:
            continue
        
        h[i] = new_ci
        h[j] = new_cj
        new_c5 = compute_c5(h)

        else:
