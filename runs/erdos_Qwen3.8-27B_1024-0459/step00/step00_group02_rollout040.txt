    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def coord_step(h, j):
    """Optimize coordinate j in place. Returns the new value."""
    n = len(h)
    old = h[j]
    h[j] = 0.0
    omh = 1.0 - h
    c_base = np.correlate(h, omh, mode="full")
    
    # Precompute L and Q vectors
    L = np.zeros(2*n-1)
    Q = np.zeros(2*n-1)
    jm = j  # j+m, m from -(n-1) to n-1
    # For output index k, shift m = k - (n-1)
    # j+m valid: 0 <= j + k - (n-1) < n → n-1-j <= k <= n-1+j
    # j-m valid: 0 <= j - k + (n-1) < n → j-(n-1) <= k <= j+(n-1)
    
    k_lo = max(0, n-1-j)
    k_hi = min(2*n-2, n-1+j)
    for k in range(k_lo, k_hi+1):
        m = k - (n-1)
        L[k] += (1.0 - h[j+m])
        Q[k] += -h[j+m]
    
    k_lo2 = max(0, j-(n-1))
    k_hi2 = min(2*n-2, j+(n-1))
    for k in range(k_lo2, k_hi2+1):
        m = k - (n-1)
        L[k] += -h[j-m]
    
    # Minimize max(c_base + L*x + Q*x^2) over x in [0,1]
    def obj(x):
        return np.max(c_base + L*x + Q*x*x)
    
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded', options={'xatol': 1e-12})
    x_opt = res.x
    h[j] = x_opt
    return x_opt

def coordinate_descent(h, n_sweeps=200, tol=1e-12):
    h = h.copy()
    n = len(h)
    best = compute_c5(h)

        c5 = compute_c5(h)
        if c5 < best:
            best = c5
        if max_change < tol:
            break
    return h, best
