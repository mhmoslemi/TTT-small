def _exact_c(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n == 0:
        return float("inf")
    r = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(r) * (2.0 / n))


def _project_box_sum(z, target):
    z = np.asarray(z, dtype=float).reshape(-1)
    n = z.size
    if n == 0:
        return z
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)

    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if val > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(z - tau, 0.0, 1.0)
    diff = target - float(np.sum(out))
    if abs(diff) > 1e-10:
        out = np.clip(out + diff / n, 0.0, 1.0)
    return out

    h = np.clip(h, 0.0, 1.0)
    if abs(float(np.sum(h)) - target) > 1e-7:
        h = _project_box_sum(h, target)
    c = _exact_c(h)
    if not math.isfinite(c):
        return
    if c < _BEST_C - 1e-12:
        _BEST_C = c
        _BEST_H = h.copy()
        _BEST_N = int(n)


def _tent_h(n):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    h = 2.0 * np.minimum(t, 1.0 - t)
    return _project_box_sum(h, 0.5 * n)


def _random_smooth_h(n, rng):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n, dtype=float)
    for f in range(1, 9):
        amp = rng.uniform(0.04, 0.45) / max(1.0, float(f))
        ph1 = rng.uniform(0.0, 2.0 * math.pi)
        ph2 = rng.uniform(0.0, 2.0 * math.pi)
        a += amp * (np.sin(2.0 * math.pi * f * t + ph1) +
                    1.5 * np.cos(2.0 * math.pi * f * t + ph2))
    h = 0.5 + a
    return _project_box_sum(h, 0.5 * n)


def _make_smooth_starts(parent, n, rng, max_starts=6):
    n = int(n)
    starts = []
    if parent is not None:
        starts.append(_upsample(parent, n))
        h = _upsample(parent, n)
        starts.append(_project_box_sum(h + rng.normal(0.0, 0.03, n), 0.5 * n))

    starts.append(_tent_h(n))
    t = np.linspace(0.0, 1.0, n)
    for amp in (1.0, 1.35, 1.7):
        h = 1.0 - amp * np.abs(2.0 * t - 1.0)
        starts.append(_project_box_sum(h, 0.5 * n))

    for f in (1, 2, 3):
        phase = rng.uniform(0.0, 2.0 * math.pi)
        h = 0.5 + 0.5 * np.sin(2.0 * math.pi * f * t + phase)
        starts.append(_project_box_sum(h, 0.5 * n))

    starts.append(_random_smooth_h(n, rng))
    starts.append(_random_smooth_h(n, rng))
    return starts[:max_starts]


def _upsample(h, n):
    n = int(n)
    if h is None:
        return _tent_h(n)
    h = np.asarray(h, dtype=float).reshape(-1)
    m = h.size
    if m == 0:
        return _tent_h(n)
    if m == 1:
        out = np.full(n, float(h[0]), dtype=float)
    elif m == n:
        out = h.copy()
    elif n > m:
        if n % m == 0:
            out = np.repeat(h, n // m)
        else:
            xo = np.linspace(0.0, 1.0, m)
            xn = np.linspace(0.0, 1.0, n)
            out = np.interp(xn, xo, h)
    else:
        xo = np.linspace(0.0, 1.0, m)
        xn = np.linspace(0.0, 1.0, n)
        out = np.interp(xn, xo, h)
    return _project_box_sum(out, 0.5 * n)


def _fix_binary_sum(a, n=None):
    a = np.asarray(a, dtype=float).reshape(-1)
    n = a.size if n is None else int(n)
    a = np.where(a >= 0.5, 1, 0).astype(np.int32)
    target = n // 2
    s = int(a.sum())
    if s < target:
        idx = np.flatnonzero(a == 0)[: target - s]
        a[idx] = 1
    elif s > target:
        idx = np.flatnonzero(a == 1)[: s - target]
        a[idx] = 0
    return a


def _binary_from_h(h, n, rng=None):
    n = int(n)
    h = _upsample(h, n)
    if rng is not None:
        h = h + rng.normal(0.0, 0.01, n)
    idx = np.argsort(h)[- (n // 2):]
    a = np.zeros(n, dtype=np.int32)
    a[idx] = 1
    return a


def _binary_pattern(n, rng, f=0):
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    if f == 0:
        val = rng.normal(0.0, 1.0, n)
    else:
        phase = rng.uniform(0.0, 2.0 * math.pi)
        val = 0.5 + 0.5 * np.sin(2.0 * math.pi * f * t + phase)
    idx = np.argsort(val)[- (n // 2):]
    a = np.zeros(n, dtype=np.int32)
    a[idx] = 1
    return a


def _autocorr_linear(x):
    n = x.size
    if n == 0:
        return np.empty(0, dtype=float)
    nfft = 1
    need = 2 * n
    while nfft < need:
        nfft <<= 1
    X = np.fft.rfft(x, n=nfft)
    r = np.fft.irfft(X * np.conjugate(X), n=nfft)[:n]
    return np.real(r)


def _binary_lags(a):
    a = np.asarray(a, dtype=np.int32).reshape(-1)
    n = a.size
    P = np.zeros(n, dtype=np.int32)
    Q = np.zeros(n, dtype=np.int32)
    if n <= 2048:
        for m in range(1, n):
            x = a[:-m]
            y = a[m:]
            P[m] = int(np.dot(x, 1 - y))
            Q[m] = int(np.dot(1 - x, y))
    else:
        af = a.astype(np.float64)
        r = _autocorr_linear(af)
        prefix = np.empty(n + 1, dtype=float)
        prefix[0] = 0.0
        np.cumsum(af, out=prefix[1:])
        m = np.arange(n)
        P = np.rint(prefix[n - m] - r).astype(np.int32)
        Q = np.rint((prefix[n] - prefix[m]) - r).astype(np.int32)
        P = np.maximum(P, 0)
        Q = np.maximum(Q, 0)
    if n > 0:
        P[0] = 0
        Q[0] = 0
    return P, Q


def _apply_flip(a, P, Q, r, d):
    n = a.size
    if r > 0:
        other = a[r - 1::-1]
        if d == 1:
            P[1:r + 1] -= other
            Q[1:r + 1] += 1
            Q[1:r + 1] -= other
        else:
            P[1:r + 1] += other
            Q[1:r + 1] -= 1
            Q[1:r + 1] += other

    if r < n - 1:
        stop = n - r
        other = a[r + 1:]
        if d == 1:
            P[1:stop] += 1
            P[1:stop] -= other
            Q[1:stop] -= other
        else:
            P[1:stop] -= 1
            P[1:stop] += other
            Q[1:stop] += other

    a[r] = 1 if d == 1 else 0


def _binary_search(a0, budget_s=60.0, deadline=None, seed=0, max_initial_c=0.47):
    a = _fix_binary_sum(a0)
    n = a.size
    if n < 4 or (n & 1):
        return a.astype(float), float("inf")

    P, Q = _binary_lags(a)
    M = int(max(int(P.max()), int(Q.max())))
    c = 2.0 * M / float(n)
    budget_s = float(max(0.0, budget_s))
    if c > max_initial_c and budget_s > 20.0:
        budget_s = 20.0

    best_a = a.copy()
    best_M = M
    best_c = c

    t_start = time.time()
    t_end = t_start + budget_s
    if deadline is not None:
        t_end = min(t_end, float(deadline))
    if t_end <= t_start + 1e-3:
        return best_a.astype(float), best_c

    dur = max(1e-3, t_end - t_start)
    T0 = max(1.0, 0.15 * float(M))
    Tmin = 0.1
    rng = np.random.default_rng(seed + 17)

    while time.time() < t_end - 1e-3:
        frac = max(0.0, 1.0 - (time.time() - t_start) / dur)
        T = max(Tmin, T0 * frac * frac)

        p1 = -1
        q0 = -1
        for _ in range(10):
            p = int(rng.integers(0, n))
            q = int(rng.integers(0, n))
            if a[p] != a[q]:
                if a[p] == 1:
                    p1 = p
                    q0 = q
                else:
                    p1 = q
                    q0 = p
                break
        if p1 < 0:
            continue

        _apply_flip(a, P, Q, p1, -1)
        _apply_flip(a, P, Q, q0, 1)
        Mnew = max(int(P.max()), int(Q.max()))
        delta = Mnew - M

        accept = False
        if delta <= 0:
            accept = True
        elif delta > 0 and T > 0.0:
            if delta < 60 * T:
                accept = (rng.random() < math.exp(-float(delta) / T))

        if accept:
            M = Mnew
            if M < best_M:
                best_M = M
                best_c = 2.0 * M / float(n)
                best_a = a.copy()
        else:
            _apply_flip(a, P, Q, q0, -1)
            _apply_flip(a, P, Q, p1, 1)

    return best_a.astype(float), best_c


def _sum_constraint(x):
    return float(np.sum(x) - 0.5 * len(x))


def _sum_constraint_jac(x):
    return np.ones(len(x), dtype=float)


def _smooth_value(x):
    h = np.asarray(x, dtype=float).reshape(-1)
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    if n == 0:
        return float("inf")
    beta = float(_SMOOTH_BETA)
    R = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(R)):
        return 1.0e100
    Rmax = float(np.max(R))
    if beta <= 1e-12:
        return Rmax * (2.0 / n)
    z = beta * (R - Rmax)
    s = float(np.sum(np.exp(z)))
    if s <= 0.0 or not math.isfinite(s):
        return Rmax * (2.0 / n)
    return float((Rmax + math.log(s) / beta) * (2.0 / n))


def _smooth_gradient(x):
    h = np.asarray(x, dtype=float).reshape(-1)
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float)
    beta = float(_SMOOTH_BETA)
    R = np.correlate(h, 1.0 - h, mode="full")
    if not np.all(np.isfinite(R)):
        return np.zeros(n, dtype=float)
    Rmax = float(np.max(R))
    if beta <= 1e-12:
        idx = int(np.argmax(R))
        w = np.zeros_like(R)
        w[idx] = 1.0
    else:
        z = beta * (R - Rmax)
        w = np.exp(z)
        s = float(w.sum())
        if s <= 0.0 or not math.isfinite(s):
            idx = int(np.argmax(R))
            w = np.zeros_like(R)
            w[idx] = 1.0
        else:
            w = w / s

    g = np.zeros(n, dtype=float)
    c = n - 1
    w0 = float(w[c])
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)

    for m in range(1, n):
        wp = float(w[c - m])
        wq = float(w[c + m])
        if wp < 1e-15 and wq < 1e-15:
            continue
        a1 = h[:n - m]
        a2 = h[m:]
        if wp > 1e-15:
            g[:n - m] += wp * (1.0 - a2)
            g[m:] -= wp * a1
        if wq > 1e-15:
            g[:n - m] -= wq * a2
            g[m:] += wq * (1.0 - a1)

    return (2.0 / n) * g


def _smooth_slsqp(h0, n, betas=(8.0, 25.0, 80.0), maxiter_per_beta=80, time_limit=40.0):
    if _SO is None:
        h0 = _project_box_sum(np.asarray(h0, dtype=float).reshape(-1), 0.5 * n)
        return h0, _exact_c(h0)

    x = _project_box_sum(np.asarray(h0, dtype=float).reshape(-1), 0.5 * n)
    best_x = x.copy()
    best_c = _exact_c(x)
    bounds = [(0.0, 1.0)] * int(n)
    cons = [{"type": "eq", "fun": _sum_constraint, "jac": _sum_constraint_jac}]
    t_start = time.time()

    for beta in betas:
        if time.time() - t_start >= float(time_limit) - 1e-3:
            break
        global _SMOOTH_BETA
        _SMOOTH_BETA = float(beta)
        try:
            res = _SO.minimize(
                _smooth_value,
                x,
                jac=_smooth_gradient,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(maxiter_per_beta), "ftol": 1e-8},
            )
        except Exception:
            res = None
        if res is not None and getattr(res, "x", None) is not None:
            y = _project_box_sum(np.asarray(res.x, dtype=float).reshape(-1), 0.5 * n)
            cy = _exact_c(y)
            if math.isfinite(cy) and cy < best_c:
                best_c = cy
                best_x = y.copy()
                x = y.copy()
    return best_x, float(best_c)


def _pg_refine(h0, n, beta=40.0, time_limit=60.0, max_iter=2000, seed=0):
    n = int(n)
    h = _project_box_sum(np.asarray(h0, dtype=float).reshape(-1), 0.5 * n)
    best_h = h.copy()
    best_c = _exact_c(h)
    if not math.isfinite(best_c):
        return best_h, float("inf")

    global _SMOOTH_BETA
    _SMOOTH_BETA = float(beta)
    t_start = time.time()
    t_end = t_start + float(time_limit)
    rng = np.random.default_rng(seed + 101)
    eta = 0.2

    for it in range(int(max_iter)):
        if time.time() >= t_end - 1e-3:
            break
        g = _smooth_gradient(h)
        if not np.all(np.isfinite(g)):
            break
        g -= float(np.mean(g))
        norm = float(np.linalg.norm(g))
        if norm < 1e-12:
            break

        accepted = False
        for scale in (0.25, 0.5, 1.0, 2.0, 4.0):
            cand = _project_box_sum(h - eta * scale * g, 0.5 * n)
            c = _exact_c(cand)
            if math.isfinite(c) and c < best_c - 1e-13:
                h = cand
                best_h = cand.copy()
                best_c = c
                eta = min(1.0, eta * 1.3)
                accepted = True
                break

        if not accepted:
            eta *= 0.72
            if it % 12 == 11:
                cand = _project_box_sum(h + 0.004 * rng.normal(0.0, 1.0, n), 0.5 * n)
                c = _exact_c(cand)
                if math.isfinite(c):
                    if c < best_c:
                        h = cand
                        best_h = cand.copy()
                        best_c = c
                    elif c < best_c + 0.005:
                        h = cand
            if eta < 1e-5:
                break

    return best_h, float(best_c)


def _mip_feasible(n, M_limit, timeout):
    if _CP is None:
        return None
    n = int(n)
    M_limit = int(M_limit)
    timeout = max(1.0, float(timeout))
    if n < 8 or (n & 1) or M_limit < 0:
        return None

    try:
        try:
            _CP.settings.quiet = True
        except Exception:
            pass

        x = _CP.Variable(n, boolean=True)
        cons = [_CP.sum(x) == (n // 2)]

        for m in range(1, n):
            y = _CP.Variable(n - m, nonneg=True)
            idx1 = np.arange(n - m)
            idx2 = idx1 + m

            cons += [
                y <= x[idx1],
                y <= x[idx2],
                _CP.sum(x[: n - m]) - _CP.sum(y) <= M_limit,
                _CP.sum(x[m:]) - _CP.sum(y) <= M_limit,
            ]

        prob = _CP.Problem(_CP.Minimize(_CP.sum(x)), cons)

        installed = set()
        try:
            installed = set(_CP.installed_solvers())
        except Exception:
            installed = set()

        desired = ["CBC", "SCIP", "GUROBI", "XPRESS", "GLPK", "MOSEK", "CPLEX"]
        solvers = []
        for name in desired:
            if name in installed and hasattr(_CP, name):
                solvers.append(name)

        infeasible = False
        for name in solvers:
            kwseq = [
                {"output_flag": 0, "time_limit": timeout},
                {"output_flag": 0, "timeLimit": timeout},
                {"time_limit": timeout},
                {"timeLimit": timeout},
                {"output_flag": 0},
                {},
            ]
            used_exact_solver = False
            for kw in kwseq:
                used_exact_solver = True
                try:
                    prob.solve(solver=name, **kw)
                    break
                except Exception:
                    continue

            stat = getattr(prob, "status", "")
            if stat == "infeasible":
                infeasible = True
                break
            if stat in ("optimal", "optimal_inaccurate", "solved", "solved_inaccurate") and x.value is not None:
                xv = np.asarray(x.value, dtype=float).reshape(-1)
                if xv.size == n and np.all(np.abs(xv - np.round(xv)) < 1e-5):
                    return _fix_binary_sum(xv, n)
            if used_exact_solver and x.value is not None:
                xv = np.asarray(x.value, dtype=float).reshape(-1)
                if xv.size == n and np.all(np.abs(xv - np.round(xv)) < 1e-5):
                    return _fix_binary_sum(xv, n)

        if infeasible:
            return None
    except Exception:
        return None

    return None


def _get_parent():
    try:
        p = globals().get("initial_h_values", None)
    except Exception:
        p = None
    if p is None:
        return None
    try:
        p = np.asarray(p, dtype=float).reshape(-1)
    except Exception:
        return None
    if p.size == 0:
        return None
    target = 0.5 * p.size
    if abs(float(np.sum(p)) - target) > 1e-7:
        p = _project_box_sum(p, target)
    return p


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    budget = max(1.0, budget)
    deadline = t0 + budget - 2.0
    rng = np.random.default_rng(seed)

    parent = _get_parent()
    if parent is None:
        parent = _tent_h(80)
        _update_best(parent, _exact_c(parent), parent.size)
    else:
        p_raw = np.asarray(parent, dtype=float).reshape(-1)
        target = 0.5 * p_raw.size
        if abs(float(np.sum(p_raw)) - target) <= 1e-7:
            _update_best(p_raw, _exact_c(p_raw), p_raw.size)
        else:
            p_raw = _project_box_sum(p_raw, target)
            _update_best(p_raw, _exact_c(p_raw), p_raw.size)
        parent = _project_box_sum(p_raw, target)

    # Small continuous/SLSQP refinement around the parent scale.
    if _SO is not None and time.time() < deadline - 60.0:
        n80 = 80
        max_starts = 5
        starts = _make_smooth_starts(parent, n80, rng, max_starts)
        for i, h in enumerate(starts):
            if time.time() > deadline - 25.0:
                break
            remaining_starts = max(1, max_starts - i)
            budget_here = min(
                45.0,
                max(5.0, (deadline - time.time()) / float(remaining_starts) - 5.0),
            )
            h2, c2 = _smooth_slsqp(
                h,
                n80,
                betas=(6.0, 20.0, 70.0),
                maxiter_per_beta=90,
                time_limit=budget_here,
            )
            _update_best(h2, c2, n80)

    # MILP feasibility gap for a strictly better 0/1 construction.
    mip_seed = None
    if _CP is not None and _BEST_C > 0.379 and time.time() < deadline - 120.0:
        attempts = []
        if time.time() < deadline - 160.0:
            attempts.append(80)
        if time.time() < deadline - 90.0:
            attempts.append(128)
        for nmip in attempts:
            if time.time() > deadline - 60.0:
                break
            Mlim = int(math.floor(0.38085 * nmip / 2.0 - 1e-9))
            if Mlim <= 0:
                continue
            timeout = min(70.0, max(15.0, deadline - time.time() - 45.0))
            sol = _mip_feasible(nmip, Mlim, timeout)
            if sol is not None:
                fsol = sol.astype(float)
                csol = _exact_c(fsol)
                if math.isfinite(csol):
                    _update_best(fsol, csol, nmip)
                    if csol < 0.47:
                        mip_seed = sol

    # Binary local search at higher resolutions.
    if time.time() < deadline - 60.0:
        rem = max(0.0, deadline - time.time())
        if rem > 520.0:
            n_bins = [1024, 512]
        elif rem > 260.0:
            n_bins = [512, 256]
        else:
            n_bins = [256]

        seen_bin = set()
        for nb in n_bins:
            if time.time() > deadline - 30.0:
                break
            seeds = []
            if _BEST_H is not None:
                seeds.append(_binary_from_h(_BEST_H, nb, rng))
            if mip_seed is not None:
                seeds.append(_binary_from_h(mip_seed, nb, rng))
            seeds.append(_binary_pattern(nb, rng, 1))
            seeds.append(_binary_pattern(nb, rng, 2))
            seeds.append(_binary_pattern(nb, rng, 0))

            used = 0
            for a in seeds:
                if used >= 2:
                    break
                key = a.tobytes()
                if key in seen_bin:
                    continue
                seen_bin.add(key)
                used += 1

                if time.time() > deadline - 25.0:
                    break
                budget_here = min(
                    180.0,
                    max(20.0, (deadline - time.time()) - 40.0),
                )
                af, cf = _binary_search(
                    a,
                    budget_s=budget_here,
                    deadline=deadline,
                    seed=int(seed) + int(nb),
                )
                if math.isfinite(cf):
                    _update_best(af, cf, nb)
            if _BEST_C <= 0.376:
                break

    # Final continuous refinement on best known fractional / binary seeds.
    if time.time() < deadline - 25.0:
        rem = max(0.0, deadline - time.time())
        if rem > 220.0:
            nc = 512
        elif rem > 110.0:
            nc = 256
        else:
            nc = 128

        cont_starts = []
        if _BEST_H is not None:
            cont_starts.append(_upsample(_BEST_H, nc))
        if mip_seed is not None:
            cont_starts.append(_upsample(mip_seed, nc))
        cont_starts.append(_tent_h(nc))
        cont_starts.append(_random_smooth_h(nc, rng))

        for i, h in enumerate(cont_starts[:4]):
            if time.time() > deadline - 15.0:
                break
            budget_here = min(
                90.0,
                max(8.0, (deadline - time.time()) - 20.0),
            )
            if nc <= 256 and _SO is not None:
                h2, c2 = _smooth_slsqp(
                    h,
                    nc,
                    betas=(15.0, 60.0),
                    maxiter_per_beta=70,
                    time_limit=budget_here,
                )
            else:
                h2, c2 = _pg_refine(
                    h,
                    nc,
                    beta=50.0,
                    time_limit=budget_here,
                    max_iter=1500,
                    seed=int(seed) + 1000 + i,
                )
            if math.isfinite(c2):
                _update_best(h2, c2, nc)

    if _BEST_H is None:
        h = _tent_h(80)
        c = _exact_c(h)
        n = 80
    else:
        h = np.asarray(_BEST_H, dtype=float).reshape(-1)
        n = int(h.size)
        c = _BEST_C
        target = 0.5 * n
        h = np.clip(h, 0.0, 1.0)
        if abs(float(np.sum(h)) - target) > 1e-8:
            h = _project_box_sum(h, target)
            c = _exact_c(h)
        else:
            c = min(float(c), _exact_c(h))

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    c = float(c)
    if not math.isfinite(c):
        c = float("inf")
    return h, c, int(n)
