def project_box(x, target):
    x = np.asarray(x, dtype=float)
    n = x.shape[0]
    target = float(target)
    if target <= 0: return zeros; if target >= n: return ones
    # Quick if already in box and sum close: return with exact adjust
    # But if x has out-of-box, need projection.
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(30):
       mid=(lo+hi)*0.5
       s=np.sum(np.clip(x-mid,0.0,1.0))
       if s > target: lo=mid
       else: hi=mid
    lam=(lo+hi)*0.5
    y=np.clip(x-lam,0.0,1.0)
    diff=target-float(np.sum(y))
    # adjustment loop; if diff large due bisection tolerance, adjust.
    if diff > 0:
       order=np.argsort(1.0-y, kind='mergesort')
    else:
       order=np.argsort(y, kind='mergesort')
    for idx in order:
       if diff > 0:
          add = min(1.0-y[idx], diff)
          if add <= 1e-15: continue

       else:
          sub = min(y[idx], -diff)
          if sub <= 1e-15: continue
          y[idx] -= sub; diff += sub
       if abs(diff) <= 1e-14: break
    return y

_PERM_CACHE = {}

def default_permutation(n):
    a = np.arange(n - 1, -1, -1, dtype=int)

    return np.concatenate((a, b))

def manual_values(h):
    n = len(h)
    if n == 0:
        return np.array([], dtype=float)
    b = 1.0 - h
    p = np.empty(n, dtype=float)
    q = np.zeros(n, dtype=float)
    for d in range(n):
        m = n - d
        p[d] = np.dot(h[:m], b[d:])
        if d > 0:
            q[d] = np.dot(h[d:], b[:m])
    return np.concatenate((p, q[1:]))

def get_permutation(n):

        return _PERM_CACHE[n]

def project_box(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    target = float(target)

        return x.copy()

    if float(np.max(x)) <= 1.0 + 1e-12 and float(np.min(x)) >= -1e-12:

        return 1.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * (2.0 / float(n))

def soft_max_weights(h, tau_obj):
    n = len(h)
    scale = 2.0 / float(n)
    full = np.correlate(h, 1.0 - h, mode="full")
    perm = get_permutation(n)
    v = full[perm]
    y = v * scale
    m = float(np.max(y))
    if tau_obj <= 0.0:
        tau = 1e-4
    else:
        tau = float(tau_obj)
    z = (y - m) / tau
    z = np.clip(z, -55.0, 0.0)
    w = np.exp(z)
    s = float(np.sum(w))
    if not np.isfinite(s) or s == 0.0:
        w = np.zeros_like(w)
        w[int(np.argmax(y))] = 1.0
    else:
        w = w / s
    return w, m

def gradient_from_weights(h, weights):
    n = len(h)
    scale = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    b = 1.0 - h

    wp = weights[:n]
    nz = np.nonzero(wp > 1e-7)[0]
    for d in nz:
        w = float(wp[d])
        m = n - int(d)
        grad[:m] += w * b[int(d):]
        grad[int(d):] -= w * h[:m]

    wn = weights[n:]
    nz = np.nonzero(wn > 1e-7)[0]
    for idx in nz:
        d = int(idx) + 1
        w = float(wn[idx])
        m = n - d
        grad[d:] += w * b[:m]
        grad[:m] -= w * h[d:]

    grad -= float(np.mean(grad))
    grad *= scale
    return grad

def try_pair_move(best_h, best_c, i, j, scale, grid, rng_state):
    n = len(best_h)
    hi = float(best_h[i])
    hj = float(best_h[j])
    lo = max(-hi, hj - 1.0)
    hi_bound = min(1.0 - hi, hj)
    if hi_bound <= lo + 1e-14:
        return best_h, best_c, False, 0

    h_test = best_h.copy()
    q_test = 1.0 - h_test
    local_best = float(best_c)
    local_best_h = None
    evals = 0

    width = hi_bound - lo
    num = int(grid)
    if width < 1e-3:
        num = max(4, num // 3)
    deltas = np.linspace(lo, hi_bound, num)

    if width > 0.12:
        small = min(0.06, width / 4.0)
        if small > 1e-6:
            s_lo = max(lo, -small)
            s_hi = min(hi_bound, small)
            if s_hi > s_lo:
                small_deltas = np.linspace(s_lo, s_hi, max(3, num // 4))
                deltas = np.concatenate((deltas, small_deltas))

    if rng_state is not None:
        cnt = max(2, num // 4)
        rd = rng_state.random(cnt) * width
        deltas = np.concatenate((deltas, lo + rd))

    for delta in deltas:
        nd = float(delta)
        ni = hi + nd
        nj = hj - nd
        if ni < 0.0:
            ni = 0.0
        elif ni > 1.0:
            ni = 1.0
        if nj < 0.0:
            nj = 0.0
        elif nj > 1.0:
            nj = 1.0
        h_test[i] = ni
        h_test[j] = nj
        q_test[i] = 1.0 - ni
        q_test[j] = 1.0 - nj
        corr = np.correlate(h_test, q_test, mode="full")
        c2 = float(np.max(corr)) * scale
        evals += 1
        if c2 < local_best - 1e-15:
            local_best = c2
            local_best_h = h_test.copy()

    if local_best_h is not None:
        return local_best_h, local_best, True, evals
    return best_h, best_c, False, evals

def pair_random_search(h, c, time_limit, rng, max_evals=None, grid=24):
    if max_evals is None:
        max_evals = 1000000000
    start = time.time()
    end = start + max(0.0, float(time_limit))
    n = len(h)
    if n < 2:
        return h.copy(), float(c)
    scale = 2.0 / float(n)
    best_h = h.copy()
    best_c = float(c)
    evals = 0
    no_improve = 0

    while time.time() < end and evals < max_evals and no_improve < 4000:
        i = int(rng.integers(n))
        j = (i + int(rng.integers(1, n))) % n
        new_h, new_c, improved, add_evals = try_pair_move(
            best_h, best_c, i, j, scale, grid, rng
        )
        evals += add_evals
        if improved:
            best_h = new_h
            best_c = new_c
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_c

def pair_sweep(h, c, time_limit, rng=None, max_evals=None, grid=None):
    if max_evals is None:
        max_evals = 1000000000
    n = len(h)
    if n < 2:
        return h.copy(), float(c)
    if grid is None:
        if n <= 90:
            grid = 14
        elif n <= 160:
            grid = 10
        else:
            grid = 8
    start = time.time()
    end = start + max(0.0, float(time_limit))
    scale = 2.0 / float(n)
    best_h = h.copy()
    best_c = float(c)
    evals = 0
    target = 0.5 * float(n)
    max_passes = 3 if time_limit > 30.0 else 1

    for _pass in range(max_passes):
        pass_improved = False
        stop_early = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() >= end or evals >= max_evals:
                    stop_early = True
                    break
                new_h, new_c, improved, add_evals = try_pair_move(
                    best_h, best_c, i, j, scale, int(grid), None
                )
                evals += add_evals
                if improved:
                    best_h = new_h
                    best_c = new_c
                    pass_improved = True
            if stop_early:
                break
        if not pass_improved:
            break
        best_h = project_box(best_h, target)
        best_c = eval_c(best_h)

    return best_h, best_c

def adam_optimize(h, c, time_limit, rng, max_steps=3000,
                  tau_start=0.035, tau_end=0.002):
    n = len(h)
    if n < 1:
        return h.copy(), float(c)
    target = 0.5 * float(n)
    start = time.time()
    end = start + max(0.0, float(time_limit))
    best_h = h.copy()
    best_c = float(c)
    local_best = best_c
    cur_h = h.copy()
    m1 = np.zeros(n, dtype=float)
    m2 = np.zeros(n, dtype=float)
    t = 0
    steps = 0
    lr = 0.15
    no_best = 0

    while time.time() < end and steps < max_steps:
        tau = tau_start + (tau_end - tau_start) * min(1.0, float(steps) / 800.0)
        w, _ = soft_max_weights(cur_h, tau)
        g = gradient_from_weights(cur_h, w)

        t += 1
        m1 = 0.9 * m1 + 0.1 * g
        m2 = 0.999 * m2 + 0.001 * (g * g)
        b1 = 1.0 - 0.9 ** t
        b2 = 1.0 - 0.999 ** t
        step = lr * (m1 / b1) / (np.sqrt(m2 / b2) + 1e-8)
        step = np.clip(step, -0.45, 0.45)

        cand1 = project_box(cur_h - step, target)
        c1 = eval_c(cand1)
        accepted = False
        step_improved = False
        temp = 0.002 + 0.008 * math.exp(-float(steps) / 700.0)

        if c1 < local_best:
            local_best = c1
            best_h = cand1.copy()
            best_c = local_best
            step_improved = True

        if c1 <= local_best + temp:
            cur_h = cand1
            accepted = True
        elif c1 <= local_best + 0.025:
            cand2 = project_box(cur_h - 0.5 * step, target)
            c2 = eval_c(cand2)
            if c2 < local_best:
                local_best = c2
                best_h = cand2.copy()
                best_c = local_best
                step_improved = True
            if c2 <= local_best + temp:
                cur_h = cand2
                accepted = True

        if accepted:
            lr = min(0.35, lr * 1.02)
        else:
            lr = max(0.02, lr * 0.75)

        if step_improved:
            no_best = 0
        else:
            no_best += 1

        if no_best > 320:
            sig = 0.045 + 0.015 * (no_best % 4)
            perturbed = cur_h + rng.normal(0.0, sig, n)
            perturbed = project_box(perturbed, target)
            cp = eval_c(perturbed)
            if cp <= local_best + 0.02:
                cur_h = perturbed
            else:
                cur_h = best_h.copy()
            if cur_h is best_h:
                pass
            c_cur = eval_c(cur_h)
            if c_cur < local_best:
                local_best = c_cur
                best_h = cur_h.copy()
                best_c = local_best
            m1 *= 0.5
            m2 *= 0.5
            no_best = 0
            if lr < 0.12:
                lr = 0.2

        steps += 1

    return best_h, best_c

def optimize_measure(h_init, c_init, n, time_limit, rng):
    if len(h_init) != n:
        h_init = parent_to_n(h_init, n)
    target = 0.5 * float(n)
    h = project_box(h_init, target)
    c = eval_c(h)
    best_h = h.copy()
    best_c = float(c)
    start = time.time()
    end = start + max(0.0, float(time_limit))
    phase = 0
    attempt = 0

    while time.time() < end - 5.0:
        if phase % 2 == 0:
            rem = end - time.time()
            adam_t = min(120.0, max(15.0, 0.45 * rem))
            if rem - adam_t < 5.0:
                adam_t = max(5.0, rem - 5.0)
            if adam_t > 5.0:
                rate = 50.0 if n <= 80 else 25.0
                steps = max(50, int(adam_t * rate))
                steps = min(steps, 6000)
                tau_s = max(0.005, 0.035 / float(attempt + 1))
                h2, c2 = adam_optimize(
                    h, c, adam_t, rng, max_steps=steps,
                    tau_start=tau_s, tau_end=0.002
                )
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2
                h = best_h.copy()
                if time.time() < end - 5.0:
                    sig = 0.03 + 0.03 * float(attempt)
                    if sig > 0.12:
                        sig = 0.12
                    h = project_box(h + rng.normal(0.0, sig, n), target)
                    c = eval_c(h)
                    if c > best_c + 0.05:
                        h = best_h.copy()
                        c = best_c
                attempt += 1
        else:
            rem = end - time.time()
            pair_t = min(35.0, max(8.0, 0.25 * rem))
            if n <= 100:
                h2, c2 = pair_sweep(h, c, pair_t, rng,
                                    max_evals=int(pair_t * 800.0), grid=14)
            else:
                h2, c2 = pair_random_search(h, c, pair_t, rng,
                                            max_evals=int(pair_t * 500.0), grid=20)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
            if time.time() < end - 5.0:
                h2, c2 = pair_random_search(
                    h, c, max(1.0, min(10.0, end - time.time() - 5.0)),
                    rng, max_evals=4000, grid=18
                )
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2
            h = best_h.copy()
        phase += 1

    rem = end - time.time()
    if rem > 3.0:
        h = best_h.copy()
        if n <= 100:
            h2, c2 = pair_sweep(h, c, rem, rng,
                                max_evals=int(rem * 700.0), grid=12)
        else:
            h2, c2 = pair_random_search(h, c, rem, rng,
                                        max_evals=int(rem * 400.0), grid=16)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    return best_h, best_c

def parent_to_n(parent, n):
    p = np.asarray(parent, dtype=float).reshape(-1)
    n0 = p.size
    if n0 == 0:
        return _default_h(n)
    if n == n0:
        return p.copy()
    if n > n0 and n % n0 == 0:
        return np.repeat(p, n // n0)
    x_old = np.linspace(0.0, float(n0), n0, endpoint=False)
    x_new = np.linspace(0.0, float(n), n, endpoint=False)
    out = np.interp(x_new, x_old, p)
    return out

def construct_sine_init(n, seed):
    if n <= 0:
        return np.array([], dtype=float), 1.0
    target = 0.5 * float(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    best_h = None
    best_c = 2.0

    amps = (0.25, 0.5, 0.75, 1.0, 1.25)
    phases = (0.0, 0.4, 0.9, 1.5, 2.1, 2.8, 3.4)
    for a in amps:
        for ph in phases:
            h = 0.5 + a * np.sin(math.pi * x + ph)
            h = project_box(h, target)
            c = eval_c(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    a1s = (0.4, 0.6, 0.8)
    a2s = (-0.30, -0.10, 0.10, 0.30)
    phs = (0.0, 1.0, 2.0)
    for a1 in a1s:
        for a2 in a2s:
            for ph in phs:
                h = 0.5 + a1 * np.sin(math.pi * x + ph) + a2 * np.sin(2.0 * math.pi * x)
                h = project_box(h, target)
                c = eval_c(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()

    for a1 in (0.5, 0.8):
        for a2 in (0.2, 0.4):
            h = 0.5 + a1 * np.sin(math.pi * x) + a2 * np.sin(2.0 * math.pi * x) \
                + 0.1 * np.sin(3.0 * math.pi * x)
            h = project_box(h, target)
            c = eval_c(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()

    if best_h is None:
        best_h = _default_h(n, seed)
        best_c = eval_c(best_h)
    return best_h, float(best_c)

def make_exact_zerosum(h, scale=1 << 24):
    n = len(h)
    scale_float = float(scale)
    q = np.rint(h * scale_float).astype(np.int64)
    q = np.clip(q, 0, scale)
    if n % 2 == 0:
        target = int((n // 2) * scale)
    else:
        target = int(round(0.5 * float(n) * scale_float))
    need = target - int(q.sum())
    if need != 0:
        if need > 0:
            order = np.argsort(scale - q, kind="mergesort")
        else:
            order = np.argsort(q, kind="mergesort")
        for idx in order:
            if need == 0:
                break
            if need > 0:
                room = scale - int(q[idx])
                if room > 0:
                    take = min(room, need)
                    q[idx] += take
                    need -= take
            else:
                if int(q[idx]) > 0:
                    take = min(int(q[idx]), -need)
                    q[idx] -= take
                    need += take

        if need != 0:
            q = np.floor(h * scale_float).astype(np.int64)
            q = np.clip(q, 0, scale)
            rem = target - int(q.sum())
            frac = h * scale_float - q.astype(float)
            if rem > 0:
                order = np.argsort(-frac, kind="mergesort")
            else:
                order = np.argsort(frac, kind="mergesort")
            k = 0
            max_iter = max(1, 3 * n + 10)
            while rem != 0 and k < max_iter:
                idx = int(order[k % n])
                if rem > 0:
                    if q[idx] < scale:
                        q[idx] += 1
                        rem -= 1
                else:
                    if q[idx] > 0:
                        q[idx] -= 1
                        rem += 1
                k += 1
    return q.astype(float) / scale_float

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    total_budget = float(budget_s)
    try:
        parent = initial_h_values
    except NameError:
        parent = None

    if parent is None:
        p_arr = np.array([], dtype=float)
    else:
        p_arr = np.asarray(parent, dtype=float).reshape(-1)

    if p_arr.size < 8:
        n0 = 70
        h0 = _default_h(n0, seed)
    else:
        n0 = int(p_arr.size)
        if n0 % 2 == 1:
            if n0 > 1:
                n0 -= 1
                p_arr = p_arr[:n0]
            else:
                n0 = 70
                p_arr = np.array([], dtype=float)
        h0 = p_arr.copy()
        h0 = project_box(h0, 0.5 * float(n0))

    c0 = eval_c(h0)
    best_h = h0.copy()
    best_c = float(c0)
    best_n = n0

    if p_arr.size > 0:
        h0 = project_box(p_arr.copy(), 0.5 * float(n0))
        c0 = eval_c(h0)
        if c0 < best_c:
            best_h = h0.copy()
            best_c = c0
            best_n = n0

    if time.time() < start + total_budget - 100.0:
        hs, cs = construct_sine_init(n0, seed + 7)
        if cs < best_c:
            best_h = hs.copy()
            best_c = cs
            best_n = n0
        h_alt = best_h.copy()
        sig = 0.06
        h_alt = project_box(h_alt + np.random.default_rng(seed + 11).normal(0.0, sig, n0), 0.5 * float(n0))
        c_alt = eval_c(h_alt)
        if c_alt < best_c + 0.02 and c_alt < best_c + 0.005:
            pass

    main_limit = max(10.0, total_budget - 140.0)
    main_limit = min(main_limit, 1050.0)
    rng = np.random.default_rng(seed)
    if main_limit > 5.0:
        h_best, c_best = optimize_measure(best_h, best_c, best_n, main_limit, rng)
        if c_best < best_c:
            best_h = h_best.copy()
            best_c = c_best
            best_n = best_n

    alt_candidates = []
    for cand in (best_n - 10, best_n + 10, best_n + 20, 2 * best_n, max(10, best_n // 2)):
        if cand < 8 or cand > 300:
            continue
        cand -= cand % 2
        if cand < 2:
            continue
        if cand not in alt_candidates:
            alt_candidates.append(cand)
    alt_candidates.sort()

    safe_end = start + total_budget - 80.0
    for cand in alt_candidates:
        left = safe_end - time.time()
        if left < 70.0:
            break
        limit = min(90.0, max(15.0, left / 3.0))
        source = best_h if best_h.size > 0 else p_arr
        h_init = parent_to_n(source, cand)
        c_init = eval_c(h_init)
        h_sine, c_sine = construct_sine_init(cand, seed + cand)
        if c_sine < c_init:
            h_init = h_sine
            c_init = c_sine
        h_best, c_best = optimize_measure(h_init, c_init, cand, limit, rng)
        if c_best < best_c:
            best_h = h_best.copy()
            best_c = c_best
            best_n = cand

    hard_end = start + total_budget - 5.0
    if time.time() < hard_end - 3.0:
        h_precise = make_exact_zerosum(best_h, 1 << 24)
        c_precise = eval_c(h_precise)
        fit = min(45.0, max(0.0, hard_end - time.time() - 3.0))
        if fit > 3.0 and best_n <= 220:
            if best_n <= 90:
                h_precise, c_precise = pair_sweep(
                    h_precise, c_precise, fit, rng,
                    max_evals=int(fit * 700.0), grid=12
                )
            else:
                h_precise, c_precise = pair_random_search(
                    h_precise, c_precise, fit, rng,
                    max_evals=int(fit * 450.0), grid=16
                )
            if time.time() < hard_end - 2.0:
                h_precise = make_exact_zerosum(h_precise, 1 << 24)
                c_precise = eval_c(h_precise)
        return h_precise, float(c_precise), int(best_n)

    return best_h, float(best_c), int(best_n)
