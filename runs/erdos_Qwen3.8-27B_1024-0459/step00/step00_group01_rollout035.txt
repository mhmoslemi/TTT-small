import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def resample_to_n(h_src, n_src, n_dst):
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    h_dst = np.interp(x_dst, x_src, h_src)
    h_dst = np.clip(h_dst, 0.0, 1.0)
    target = n_dst / 2.0
    diff = target - np.sum(h_dst)
    if abs(diff) > 1e-12:
        h_dst += diff / n_dst
        h_dst = np.clip(h_dst, 0.0, 1.0)
    # Second pass for precision
    diff = target - np.sum(h_dst)
    if abs(diff) > 1e-12:
        h_dst += diff / n_dst
        h_dst = np.clip(h_dst, 0.0, 1.0)
    return h_dst


def repair_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def sa_optimize(h_init, n, seed, t_end):

    best_c5 = cur_c5
    
    T = 0.01
    alpha = 0.999999
    min_T = 1e-9
    no_improve = 0
    
    while time.time() < t_end:
        # Batch of 1000 iterations
        for _ in range(1000):
            if time.time() > t_end:
                break

            new_h = None

            else:

def slsqp_polish(h_init, n, t_end):
    target = n / 2.0
    h0 = repair_sum(h_init.copy(), n)
    
    def objective(x):
        return compute_c5(x, n)
    
    def obj_grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full")
        c5 = np.max(corr) * dx
        max_val = np.max(corr)
        tol = max(1e-8, 1e-4 * max_val)
        active = np.where(corr >= max_val - tol)[0]
        grad = np.zeros(n)
        for m in active:
            lag = m - (n - 1)
            g = np.zeros(n)
            # c[m] = sum_{j valid} x[j] * (1 - x[j+lag])
            # where valid: 0 <= j < n and 0 <= j+lag < n
            if lag >= 0:
                j_max = n - lag
            else:
                j_max = n
            j_min = 0 if lag >= 0 else -lag
            
            for j in range(j_min, j_max):
                g[j] += 1.0  # from d/dx[j] of x[j]
                g[j] -= x[j + lag]  # from d/dx[j] of -x[j]*x[j+lag]
                if j != j + lag:
                    g[j + lag] -= x[j]  # from d/dx[j+lag] of -x[j]*x[j+lag]
            grad += g / len(active)
        grad *= dx
        return c5, grad
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(objective, h0, method='SLSQP',
                       jac=obj_grad, constraints=constraints,
                       bounds=bounds, options={'maxiter': 200,
                                               'ftol': 1e-10,
                                               'disp': False})
        if res.success or res.fun < compute_c5(h0, n):
            h_res = repair_sum(res.x, n)
            c5_res = compute_c5(h_res, n)
            return h_res, c5_res
    except Exception:
        pass
    
    return h0, compute_c5(h0, n)

Wait, I have a problem with the SLSQP gradient computation. The loop over j for each active m is O(n), and there could be O(n) active m's, making it O(n²) per gradient evaluation. For n=200, that's 40,000 operations per gradient, which is fine.

But actually, for SLSQP, the gradient is called many times. Let me make it faster.

Actually, let me just use a vectorized approach for the gradient. For each lag l:
c[m] = sum_{j: valid} x[j](1 - x[j+l])

∂c[m]/∂x[i] for the given l:
If i is in the valid j range: +1 (from d/dx_i of x[i]) and -x[i+l] (from d/dx_i of -x[i]x[i+l])
If i-l is in the valid j range and i-l ≠ i: -x[i-l] (from d/dx_i of -x[i-l]x[i])

So:
∂c[m]/∂x[i] = 1_{valid_j(i)} - x[i+l]·1_{valid(i,l)} - x[i-l]·1_{valid(i-l)}

Where valid_j(i) means i is in the valid range for j
valid(i,l) means 0 ≤ i+l < n
valid(i-l) means 0 ≤ i-l < n

Let me vectorize this:

def compute_c5_grad_fast(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(corr) * dx
    max_val = np.max(corr)
    tol = max(1e-8, 1e-4 * max_val)
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    
    for m in active:
        lag = m - (n - 1)
        g = np.zeros(n)
        
        # Valid j range
        if lag >= 0:
            j_start, j_end = 0, n - lag
        else:
            j_start, j_end = -lag, n
        
        g[j_start:j_end] = 1.0  # First term
        
        # Second term: -x[i+lag] for valid i+lag
        if lag >= 0:
            g[:n-lag] -= h[lag:]
        else:
            g[-lag:] -= h[:n+lag]
        
        # Third term: -x[i-lag] for valid i-lag
        if lag >= 0:
            g[lag:] -= h[:n-lag]
        else:
            g[:n+lag] -= h[-lag:]
        
        # But wait, the second and third terms should only apply where j is valid
        # Actually, the correction is:
        # ∂c[m]/∂x[i] = 1_{j_start ≤ i < j_end} - h[i+lag]*1_{0≤i+lag<n} - h[i-lag]*1_{0≤i-lag<n}
        # BUT the h[i+lag] term should only be in the range where j=i is valid
        # AND the h[i-lag] term should only be where j=i-lag is valid
        
        # Let me redo this more carefully:
        # c[m] = sum_{j=j_start}^{j_end-1} x[j](1-x[j+lag])
        #      = sum_{j=j_start}^{j_end-1} x[j] - sum_{j=j_start}^{j_end-1} x[j]x[j+lag]
        
        # d/dx_i of first sum = 1 if j_start ≤ i ≤ j_end-1
        # d/dx_i of second sum = x[i+lag] (from j=i) + x[i-lag] (from j=i-lag if ≠ i)
        
        g2 = np.zeros(n)
        g2[j_start:j_end] = 1.0
        
        # d/dx_i of sum x[j]x[j+lag]:
        # For j=i: contributes x[i+lag] if j_start ≤ i ≤ j_end-1 and 0 ≤ i+lag < n
        # For j+lag=i, i.e., j=i-lag: contributes x[i-lag] if j_start ≤ i-lag ≤ j_end-1 and 0 ≤ i < n and i-lag ≠ i
        # Note: the "0 ≤ i < n" is always true since i is in [0,n-1]
        
        # Term 1: x[i+lag] where j_start ≤ i < j_end
        if lag >= 0:
            # i+lag = i+lag, need 0 ≤ i+lag < n → i < n-lag
            # Also need j_start ≤ i < j_end, i.e., 0 ≤ i < n-lag
            # So: i in [0, n-lag)
            g2[:n-lag] -= h[lag:]  # g2[i] -= h[i+lag]
        else:
            # lag < 0, i+lag = i+lag, need 0 ≤ i+lag < n → i ≥ -lag
            # Also need j_start=-lag ≤ i < n
            # So: i in [-lag, n)
            g2[-lag:] -= h[:n+lag]  # g2[i] -= h[i+lag] where i+lag = i-|lag|
        
        # Term 2: x[i-lag] where j_start ≤ i-lag < j_end
        if lag >= 0:
            # i-lag ≥ j_start=0 → i ≥ lag
            # i-lag < j_end = n-lag → i < n
            # So: i in [lag, n)
            # And i-lag ≠ i (lag ≠ 0, but if lag=0, these terms combine)
            if lag != 0:
                g2[lag:] -= h[:n-lag]  # g2[i] -= h[i-lag]
            else:
                # lag = 0: x[i]x[i], d/dx_i = 2x[i]
                # We already have g2[i] -= h[i] from term 1
                # We need g2[i] -= h[i] again (for the other factor)
                g2[j_start:j_end] -= h[j_start:j_end]
        else:

            # i-lag = i+L ≥ j_start = L → i ≥ 0 (always)
            # i+L < j_end = n → i < n-L
            # So: i in [0, n-L)
            if lag != 0:
                g2[:n+lag] -= h[-lag:]  # g2[i] -= h[i-lag] = h[i+L]
        
        grad += (g2) / len(active)
    
    grad *= dx
    return c5, grad

Coordinate descent:
For each i, find the optimal h[i] given all other h's fixed. The objective is max_k corr[k], and corr[k] is linear in h[i] (since h[i] appears in only a few terms of each correlation).

Wait, corr[k] = sum_j h[j](1-h[j+l]). When we vary h[i]:
- h[i] appears in corr[k] via:
  a) j=i: contributes h[i](1-h[i+l])
  b) j+l=i, i.e., j=i-l: contributes h[i-l](1-h[i])
  (if these indices are in range)

So corr[k] as a function of h[i] is:
corr[k](h[i]) = A_k + h[i](1-h[i+l_k]) - h[i-l_k]·h[i] (if applicable)
              = A_k + h[i] - h[i]·h[i+l_k] - h[i]·h[i-l_k] (where terms are 0 if out of range)
              = A_k + h[i](1 - h[i+l_k] - h[i-l_k])

This is a linear function of h[i]! So the max of linear functions of h[i] is a convex piecewise-linear function. We can minimize it by checking the breakpoints.

The breakpoints are where two correlations are equal. For each pair of active correlations, solve for h[i]. The minimum of the max is at one of these breakpoints or at a boundary (0 or 1).

This is a 1D problem for each i! I can solve it by:
1. For each active correlation, compute the slope w.r.t. h[i]
2. The max of these linear functions is minimized where the "top" lines cross
3. Check all pairwise intersections and the boundaries

This gives an exact coordinate descent step! And it's very fast.

    corr = np.correlate(h, 1.0 - h, mode="full")

def repair_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
