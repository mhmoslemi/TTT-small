import numpy as np
import time

def project(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    N = len(h)
    # Use bisection on shift t: h_proj = clip(h - t, 0, 1)
    # Find t such that sum(clip(h-t, 0, 1)) = target_sum
    
    lo = h.min() - 1
    hi = h.max() + 1
    
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_c5(h, N):
    dx = 2.0 / N
    c = np.correlate(h, 1 - h, mode='full')
    return c.max() * dx, c, dx

def subgradient(h, N, c, top_k=10, temp=0.01):
    """Compute smoothed subgradient of max(c) * dx"""
    dx = 2.0 / N
    c_max = c.max()
    # Softmax weights over all lags (or top-k)
    # For efficiency, use top-k
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[-k:]
    top_vals = c[top_idx]
    # Temperature-scaled softmax
    weights = np.exp((top_vals - top_vals[0]) / temp)
    weights /= weights.sum()
    
    grad = np.zeros(N)
    for w, m in zip(weights, top_idx):
        lag = m - (N - 1)
        # Term 1: (1-h[j-lag]) for 0 <= j-lag < N
        j1_lo = max(0, lag)
        j1_hi = min(N, N + lag)
        if j1_lo < j1_hi:

            grad[idx] += w * dx * (1 - h[idx - lag])

        j2_lo = max(0, -lag)
        j2_hi = min(N, N - lag)
        if j2_hi > j2_lo:
            idx = np.arange(j2_lo, j2_hi)
            grad[idx] -= w * dx * h[idx + lag]
    
    return grad

def optimize(h, N, max_iters=5000, lr_init=0.01, time_limit=100):

    dx = 2.0 / N

    best_c5 = compute_c5(h, N)[0]

        if c5 < best_c5:

        h = h_new

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_N = target_N

    # Phase 3: Binary local search on the best solution
    if best_N > 0 and _remaining() > 80:
        h_bin = (best_h > 0.5).astype(np.float64)
        # Fix sum: ensure exactly N/2 ones
        n_ones = int(h_bin.sum())
        if n_ones > best_N // 2:
            excess = n_ones - best_N // 2
            idx = np.where(h_bin > 0.5)[0]
            to_flip = idx[rng.choice(len(idx), excess, replace=False)]
            h_bin[to_flip] = 0.0
        elif n_ones < best_N // 2:
            deficit = best_N // 2 - n_ones
            idx = np.where(h_bin < 0.5)[0]
            to_flip = idx[rng.choice(len(idx), deficit, replace=False)]
            h_bin[to_flip] = 1.0

        t_alloc = min(_remaining() - 60, 150)
        h_opt, c5_opt = _binary_local_search(h_bin, best_N, max(t_alloc, 10), rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_N = best_N

    # Phase 4: Re-optimize continuously from binary solution
    if best_N > 0 and _remaining() > 60:
        t_alloc = min(_remaining() - 40, 120)
        h_opt, c5_opt = _optimize_sg(best_h, best_N, max(t_alloc, 10), rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_N = best_N

    # Phase 5: Higher resolution fine-tuning
    if _remaining() > 100 and best_N > 0:
        N_fine = best_N * 2
        if N_fine <= 1000:
            x_old = np.linspace(0, 2.0, best_N, endpoint=False)
            x_new = np.linspace(0, 2.0, N_fine, endpoint=False)
            h_fine = np.interp(x_new, x_old + np.array([0.0]), best_h)
            h_fine = _project(h_fine, N_fine / 2.0)
            t_alloc = min(_remaining() - 30, _remaining() * 0.6)
            h_opt, c5_opt = _optimize_sg(h_fine, N_fine, max(t_alloc, 10), rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_N = N_fine

    # Phase 6: Final binary polish
    if best_N > 0 and _remaining() > 30:
        h_bin = (best_h > 0.5).astype(np.float64)
        n_ones = int(h_bin.sum())
        if n_ones > best_N // 2:
            excess = n_ones - best_N // 2
            idx = np.where(h_bin > 0.5)[0]
            to_flip = idx[rng.choice(len(idx), excess, replace=False)]
            h_bin[to_flip] = 0.0
        elif n_ones < best_N // 2:
            deficit = best_N // 2 - n_ones
            idx = np.where(h_bin < 0.5)[0]
            to_flip = idx[rng.choice(len(idx), deficit, replace=False)]
            h_bin[to_flip] = 1.0
        t_alloc = min(_remaining() - 5, 40)
        h_opt, c5_opt = _binary_local_search(h_bin, best_N, max(t_alloc, 5), rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_N = best_N

    # Phase 7: Multi-restart at final resolution
    if _remaining() > 60 and best_N > 0:
        for r in range(2):
            if _remaining() < 30:
                break
            h_init = best_h + rng.randn(best_N) * 0.02
            h_init = _project(h_init, best_N / 2.0)
            t_alloc = min(_remaining() - 10, 50)
            h_opt, c5_opt = _optimize_sg(h_init, best_N, max(t_alloc, 5), rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_N = best_N

    # Final: ensure validity and compute exact value
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, best_N / 2.0)
    final_c = _correlation(best_h, best_N)
    final_c5 = _c5_value(final_c, best_N)

    return best_h, final_c5, best_N
