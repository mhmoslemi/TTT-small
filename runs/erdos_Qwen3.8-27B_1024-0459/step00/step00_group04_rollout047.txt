import numpy as np
from scipy.optimize import minimize
import time

def compute_correlation(h):
    return np.correlate(h, 1 - h, mode="full")

def objective_smoothed(h, dx, alpha):
    cor = compute_correlation(h) * dx
    # Log-sum-exp for numerical stability
    m = np.max(cor)
    return (1.0 / alpha) * np.log(np.sum(np.exp(alpha * (cor - m)))) + m

def objective_grad(h, dx, alpha):
    """Compute objective and gradient using finite differences"""
    n = len(h)
    f = objective_smoothed(h, dx, alpha)
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] = min(1.0, h[i] + eps)
        h_minus = h.copy()
        h_minus[i] = max(0.0, h[i] - eps)
        f_plus = objective_smoothed(h_plus, dx, alpha)
        f_minus = objective_smoothed(h_minus, dx, alpha)
        if h[i] + eps > 1.0:
            grad[i] = (f_plus - f) / (h_plus[i] - h[i])
        elif h[i] - eps < 0.0:
            grad[i] = (f - f_minus) / (h[i] - h_minus[i])
        else:
            grad[i] = (f_plus - f_minus) / (2 * eps)
    return f, grad

def project_to_feasible(h, n):
    """Project h to satisfy 0 <= h <= 1 and sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    diff = n / 2.0 - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    # Increase where h < 1, decrease where h > 0
    if diff > 0:
        room = 1.0 - h
        total_room = np.sum(room)
        if total_room > 0:
            h = h + diff * room / total_room
    else:
        space = h
        total_space = np.sum(space)
        if total_space > 0:
            h = h + diff * space / total_space
    return np.clip(h, 0, 1)

def interpolate_h(h, n_old, n_new):
    """Interpolate h from n_old points to n_new points"""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old  # midpoints
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h, period=2.0)  # periodic? or not?
    # Actually, h is defined on [0,2], not periodic. Let's use regular interp.
    # But we need to be careful at boundaries.
    h_new = np.interp(x_new, x_old, h)
    return h_new
