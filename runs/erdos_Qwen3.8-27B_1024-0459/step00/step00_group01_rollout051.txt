from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def ensure_constraint(h, n):
    """Ensure sum(h) = n/2 and 0 <= h <= 1"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, rng):
    """Convert to exact binary with n/2 ones."""
    target = n // 2
    h_bin = (h >= 0.5).astype(np.float64)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        flip = rng.choice(ones, n_ones - target, replace=False)
        h_bin[flip] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        flip = rng.choice(zeros, target - n_ones, replace=False)
        h_bin[flip] = 1
    return h_bin

    if rng is None:
        rng = np.random.default_rng()
    
    target = n // 2
    h = np.array(h_init, dtype=np.float64)
    h_bin = make_binary(h, n, rng)
    
    current_score = compute_c5(h_bin, n)
    best_score = current_score
    best_h = h_bin.copy()
    
    ones_idx = np.where(h_bin == 1)[0]
    zeros_idx = np.where(h_bin == 0)[0]

        frac = i / max(n_iter - 1, 1)
        temp = t_start * (t_end / t_start) ** frac

        if rng.random() < 0.5:
            idx = ones_idx[rng.integers(len(ones_idx))]
            h_bin[idx] = 0
            ones_idx = np.delete(ones_idx, np.where(ones_idx == idx))
            zeros_idx = np.append(zeros_idx, idx)
        else:

            h_bin[idx] = 1
            zeros_idx = np.delete(zeros_idx, np.where(zeros_idx == idx))
            ones_idx = np.append(ones_idx, idx)
        
        new_score = compute_c5(h_bin, n)
        
        if new_score <= current_score or rng.random() < np.exp((current_score - new_score) / temp):

            if new_score < best_score:

        # Single flip search
        for i in range(n):
            h[i] = 1.0 - h[i]
            score = compute_c5(h, n)
            h[i] = 1.0 - h[i]
            delta = current - score
            if delta > best_delta:
                best_delta = delta
                best_single = i
        
        # Two-flip search (only if no single flip helps, or as escape)
        if best_single < 0:
            found = False
            ones_list = np.where(h == 1)[0]
            zeros_list = np.where(h == 0)[0]
            if len(ones_list) > 0 and len(zeros_list) > 0:
                # Sample a subset of 2-flips for speed
                n_try = min(200, len(ones_list) * len(zeros_list))
                for _ in range(n_try):
                    i = ones_list[rng.integers(len(ones_list))]
                    j = zeros_list[rng.integers(len(zeros_list))]
                    if i == j:
                        continue
                    h[i] = 0.0
                    h[j] = 1.0
                    score = compute_c5(h, n)
                    delta = current - score
                    h[i] = 1.0
                    h[j] = 0.0
                    if delta > best_delta:
                        best_delta = delta
                        best_pair = (int(i), int(j))
                        found = True
        
        applied = False
        if best_single >= 0:
            h[best_single] = 1.0 - h[best_single]
            current -= best_delta
            applied = True
        elif found and best_pair[0] >= 0:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            current -= best_delta
            applied = True
        
        if not applied:
            break
        
        if current < best:
            best = current
            best_h = h.copy()
    
    return best_h, best

def sa_binary(h_init, n, n_iter, t_start, t_end, rng):
    h = make_binary(h_init, n, rng)
    current = compute_c5(h, n)
    best = current
    best_h = h.copy()
    
    for i in range(n_iter):
        frac = float(i) / max(n_iter - 1, 1)
        temp = t_start * (t_end / t_start) ** frac
        
        idx = int(rng.integers(n))
        h[idx] = 1.0 - h[idx]
        new_score = compute_c5(h, n)
        
        delta = new_score - current
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            current = new_score
            if current < best:
                best = current
                best_h = h.copy()
        else:
            h[idx] = 1.0 - h[idx]
    
    return best_h, best

def upsample_h(h, n_src, n_dst):
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h, period=2.0)
    target = n_dst / 2.0
    h_dst += (target - h_dst.sum()) / n_dst
    return np.clip(h_dst, 0.0, 1.0)

def local_search_continuous(h, n, rng, max_iter=5000, step=0.08):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    h += (n / 2.0 - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    
    current = compute_c5(h, n)
    best = current
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iter):
        k_pert = int(rng.integers(1, 6))
        idxs = rng.choice(n, k_pert, replace=False)
        h_old = h.copy()
        h[idxs] += rng.normal(0, step, k_pert)
        h = np.clip(h, 0.0, 1.0)
        
        diff = n / 2.0 - h.sum()
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
        
        new_score = compute_c5(h, n)
        if new_score < current - 1e-14:
            current = new_score
            if current < best:
                best = current
                best_h = h.copy()
            no_improve = 0
        else:
            h = h_old
            no_improve += 1
            if no_improve > 500:
                break
    
    return best_h, best

def structured_start(n, period, fraction, rng):
    """Create a structured periodic binary pattern."""
    h = np.zeros(n)
    for i in range(n):
        phase = (i % period) / period
        h[i] = 1.0 if phase < fraction else 0.0
    h = make_binary(h, n, rng)
    return h

def try_scipy_minimize(h_init, n, rng, max_iter=500):
    """Use scipy minimize with penalty for max objective."""
    h = np.array(h_init, dtype=np.float64)
    n_ineq = 2 * n + 1
    
    def objective(vars_):
        hv = vars_[:n]
        t = vars_[n]
        return t
    
    def constraint(t_vars):
        h = t_vars[:n]
        t = t_vars[n]
        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        # All corr_k * dx <= t
        n_constr = len(corr)
        c = np.empty(n_constr + n + 1)
        c[:n_constr] = t - corr * dx
        c[n_constr:n_constr + n] = h
        c[n_constr + n:n_constr + n + 1] = h.sum() - n / 2.0 + (1.0 - (n/2.0 - h.sum())) * 0  # equality handled separately
        return c
    
    # Use SLSQP with a simpler formulation
    x0 = np.append(h, compute_c5(h, n) + 0.05)
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    def obj_func(x):
        return x[n]
    
    cons = []
    
    def eq_constr(x):
        return x[:n].sum() - n / 2.0
    
    cons.append({'type': 'eq', 'fun': eq_constr})
    
    # Inequality: t - corr_k * dx >= 0 for all k
    # This is too many constraints for SLSQP with n=47 (93 + others)
    # Instead use a smooth max approximation
    def smooth_max_obj(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        vals = corr * dx
        # Smooth max
        alpha = 50.0
        smax = np.logsumexp(alpha * vals) / alpha
        # Penalty for constraint violation
        pen = 0.0
        if h.sum() != n / 2.0:
            pen += 100.0 * (h.sum() - n / 2.0) ** 2
        if np.any(h < 0) or np.any(h > 1):
            pen += 100.0 * (np.sum(np.maximum(0, -h)) ** 2 + np.sum(np.maximum(0, h - 1)) ** 2)
        return smax + pen
    
    def smooth_max_grad(x):
        h = x[:n]
        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        vals = corr * dx
        alpha = 50.0
        weights = np.exp(alpha * vals)
        weights /= weights.sum()
        grad_h = np.zeros(n)
        # d/dh of corr_k * dx is complex; use numerical gradient fallback
        # Skip gradient, let SLSQP approximate
        return None
    
    x0 = np.append(h, compute_c5(h, n) + 0.1)
    res = minimize(lambda x: smooth_max_obj(x), x0[:n], method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': 200, 'ftol': 1e-8})
    
    if res.success and res.x is not None:
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt += (n / 2.0 - h_opt.sum()) / n
        h_opt = np.clip(h_opt, 0.0, 1.0)
        c5_opt = compute_c5(h_opt, n)
        return h_opt, c5_opt
    return h_init, compute_c5(h_init, n)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15.0
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Phase 0: Evaluate parent
    n_p = len(initial_h_values)
    h_parent = np.array(initial_h_values, dtype=np.float64)
    c5_parent = compute_c5(h_parent, n_p)
    best_h = h_parent.copy()
    best_c5 = c5_parent
    best_n = n_p
    
    def still_time():
        return time.time() < deadline
    
    # Phase 1: Binary local search at parent resolution
    if still_time():
        h_bin = make_binary(h_parent, n_p, rng)
        h_ls, c5_ls = local_search_binary(h_bin, n_p, rng, max_iter=3000)
        if c5_ls < best_c5:
            best_h = h_ls.copy()
            best_c5 = c5_ls
            best_n = n_p
        else:
            c5_bin = compute_c5(h_bin, n_p)
            if c5_bin < best_c5:
                best_h = h_bin.copy()
                best_c5 = c5_bin
                best_n = n_p
    
    # Phase 2: Multiple SA restarts at parent resolution
    n_sa_restarts = 8
    sa_iters_per = 15000
    if still_time():
        for r in range(n_sa_restarts):
            if not still_time():
                break
            if r == 0:
                h_init = best_h.copy()
            elif r == 1:
                h_init = make_binary(rng.random(n_p), n_p, rng)
            elif r == 2:
                h_init = structured_start(n_p, 7, 0.5, rng)
            elif r == 3:
                h_init = structured_start(n_p, 9, 0.5, rng)
            elif r == 4:
                h_init = structured_start(n_p, 5, 0.5, rng)
            elif r == 5:
                h_init = structured_start(n_p, 11, 0.5, rng)
            else:
                h_init = make_binary(rng.random(n_p), n_p, rng)
            
            h_sa, c5_sa = sa_binary(h_init, n_p, sa_iters_per, 0.5, 0.02, rng)
            if c5_sa < best_c5:
                best_h = h_sa.copy()
                best_c5 = c5_sa
                best_n = n_p
            
            # Polish with local search
            h_ls2, c5_ls2 = local_search_binary(h_sa, n_p, rng, max_iter=1500)
            if c5_ls2 < best_c5:
                best_h = h_ls2.copy()
                best_c5 = c5_ls2
                best_n = n_p
    
    # Phase 3: Continuous refinement at parent resolution
    if still_time():
        h_cont = best_h.copy()
        h_c, c5_c = local_search_continuous(h_cont, n_p, rng, max_iter=8000, step=0.06)
        if c5_c < best_c5:
            best_h = h_c.copy()
            best_c5 = c5_c
            best_n = n_p
    
    # Phase 4: Try scipy minimize on the smooth approximation
    if still_time():
        h_sc, c5_sc = try_scipy_minimize(best_h, n_p, rng)
        if c5_sc < best_c5:
            best_h = h_sc.copy()
            best_c5 = c5_sc
            best_n = n_p
        # Polish scipy result
        h_p2, c5_p2 = local_search_continuous(h_sc, n_p, rng, max_iter=3000, step=0.04)
        if c5_p2 < best_c5:
            best_h = h_p2.copy()
            best_c5 = c5_p2
            best_n = n_p
    
    # Phase 5: Multi-resolution optimization
    resolutions = [n_p + 1, n_p + 2, int(n_p * 1.5), n_p * 2, n_p * 2 + 1]
    for n_dst in resolutions:
        if not still_time():
            break
        h_up = upsample_h(best_h, best_n, n_dst)
        
        # Continuous local search
        h_opt, c5_opt = local_search_continuous(h_up, n_dst, rng, max_iter=6000, step=0.05)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n = n_dst
        
        # Binary search at new resolution
        if still_time():
            h_b = make_binary(h_opt, n_dst, rng)
            h_opt2, c5_opt2 = local_search_binary(h_b, n_dst, rng, max_iter=1000)
            if c5_opt2 < best_c5:
                best_h = h_opt2.copy()
                best_c5 = c5_opt2
                best_n = n_dst
        
        # Quick SA at new resolution
        if still_time():
            h_opt3, c5_opt3 = sa_binary(h_opt, n_dst, 8000, 0.3, 0.01, rng)
            if c5_opt3 < best_c5:
                best_h = h_opt3.copy()
                best_c5 = c5_opt3
                best_n = n_dst
            h_opt4, c5_opt4 = local_search_binary(h_opt3, n_dst, rng, max_iter=800)
            if c5_opt4 < best_c5:
                best_h = h_opt4.copy()
                best_c5 = c5_opt4
                best_n = n_dst
    
    # Phase 6: Additional structured searches at parent resolution
    if still_time():
        periods_to_try = [3, 4, 5, 6, 7, 8, 10, 12, 13, 14, 15, 17, 19, 23, 29]
        for p in periods_to_try:
            if not still_time():
                break
            if p >= n_p:
                continue
            for frac_num in range(1, p):
                h_s = structured_start(n_p, p, frac_num / float(p), rng)
                c5_s = compute_c5(h_s, n_p)
                if c5_s < best_c5:
                    h_ls3, c5_ls3 = local_search_binary(h_s, n_p, rng, max_iter=2000)
                    if c5_ls3 < best_c5:
                        best_h = h_ls3.copy()
                        best_c5 = c5_ls3
                        best_n = n_p
                    elif c5_s < best_c5:
                        best_h = h_s.copy()
                        best_c5 = c5_s
                        best_n = n_p
    
    # Phase 7: Final polish on best solution
    if still_time():
        h_final, c5_final = local_search_continuous(best_h, best_n, rng, max_iter=5000, step=0.03)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
            best_n = best_n
        
        # Binary polish on the best
        if best_n == n_p or still_time():
            h_fb = make_binary(best_h, best_n, rng)
            h_fp, c5_fp = local_search_binary(h_fb, best_n, rng, max_iter=2000)
            if c5_fp < best_c5:
                best_h = h_fp.copy()
                best_c5 = c5_fp
    
    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    # Final constraint check
    target_sum = best_n / 2.0
    if abs(best_h.sum() - target_sum) > 1e-10:
        best_h += (target_sum - best_h.sum()) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
    
    best_c5 = compute_c5(best_h, best_n)
    
    return (best_h, best_c5, best_n)
