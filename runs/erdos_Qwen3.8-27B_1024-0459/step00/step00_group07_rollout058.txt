def compute_gradient(h, T):
    n = len(h)
    h1 = 1 - h
    corr = np.correlate(h, h1, mode='full')
    
    # Soft-max weights
    max_corr = np.max(corr)
    exp_vals = np.exp((corr - max_corr) / T)
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)

        if wm < 1e-14:

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def soft_max_obj(h, T):
    corr = np.correlate(h, 1-h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / T)
    return max_c + T * np.log(np.sum(exp_c))

def soft_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / T)
    w = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        offset = m - (n-1)
        lo = max(0, -offset)
        hi = min(n-1, n-1-offset)
        
        # Term 1
        idx = np.arange(lo, hi+1)
        grad[idx] += wm * (1 - h[idx + offset])
        
        # Term 2
        j_lo = max(0, lo + offset)
        j_hi = min(n-1, hi + offset)
        if j_lo <= j_hi:
            idx2 = np.arange(j_lo, j_hi+1)
            grad[idx2] -= wm * h[idx2 - offset]
    
    return grad

def upsolve(h, n_new):
    """Upscale h to n_new points via interpolation."""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # Fix sum
    h_new *= (n_new / 2.0) / h_new.sum()
    h_new = np.clip(h_new, 0, 1)
    return h_new

def fix_sum(h, n):
    """Fix the sum to n/2 while maintaining bounds."""
    h = h.copy()
    diff = n/2.0 - h.sum()
    h += diff / n
    h = np.clip(h, 0, 1)
    return h

def optimize_h(init_h, max_time, T_start=1.0, T_end=0.05):
    """Optimize h using SLSQP with soft-max objective."""
    n = len(init_h)
    h = init_h.copy()
    best_h = h.copy()
    best_obj = compute_c5(h)
    start = time.time()
    
    T = T_start
    num_anneal_steps = 5
    
    for step in range(num_anneal_steps):
        T = T_start * (T_end/T_start) ** (step / max(num_anneal_steps-1, 1))
        
        def obj(h):
            return soft_max_obj(h, T)
        
        def grad(h):
            return soft_max_grad(h, T)
        
        def constraint(sum_h, n):
            return sum_h - n/2.0
        
        cons = [{'type': 'eq', 'fun': lambda h: h.sum() - n/2.0, 'jac': lambda h: np.ones(n)}]
        bounds = [(0, 1)] * n
        
        max_iter = max(20, int((max_time / num_anneal_steps - (time.time()-start)) / 2))
        
        result = optimize.minimize(
            obj, h, jac=grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-10}
        )
        
        if result.success and result.fun < 1e10:
            h_new = np.clip(result.x, 0, 1)
            c5_new = compute_c5(h_new)
            if c5_new < best_obj:
                best_h = h_new.copy()
                best_obj = c5_new
            h = h_new
    
    return best_h, best_obj

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = np.ones(80) * 0.5
    
    n_parent = len(h_parent)
    c5_parent = compute_c5(h_parent)
    best_h = h_parent.copy()
    best_c5 = c5_parent
    
    # Strategy: try multiple resolutions and random restarts
    resolutions = []
    for n in [n_parent, n_parent*2, n_parent*4, n_parent*8]:
        if 2*n-1 < 10000:  # Keep correlation computation manageable
            resolutions.append(n)
    
    # Phase 1: Optimize at each resolution
    for n in resolutions:
        if time.time() - start_time > budget_s * 0.5:
            break
        
        if n == n_parent:
            h_init = h_parent.copy()
        else:
            h_init = upsolve(h_parent, n)
        
        time_per_n = (budget_s * 0.5 - (time.time() - start_time)) / max(len(resolutions) - resolutions.index(n), 1)
        
        h_opt, c5_opt = optimize_h(h_init, time_per_n)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Phase 2: Random restarts at best resolution
    n_best = len(best_h)
    remaining = budget_s - 60 - (time.time() - start_time)
    if remaining > 30:
        num_restarts = int(remaining / 30)
        for _ in range(min(num_restarts, 20)):
            if time.time() - start_time > budget_s - 30:
                break
            
            # Generate random perturbation
            h_restart = best_h + rng.randn(n_best) * 0.1
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = fix_sum(h_restart, n_best)
            
            h_opt, c5_opt = optimize_h(h_restart, 25)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    # Phase 3: Binary projection search
    # Try rounding to binary and see if we get improvement
    if time.time() - start_time < budget_s - 10:
        h_binary = (best_h > 0.5).astype(float)
        # Fix sum
        ones = h_binary.sum()
        if ones > n_best//2:
            # Remove excess ones (those closest to 0.5 in original h)
            idx_ones = np.where(h_binary > 0.5)[0]
            sorted_idx = idx_ones[np.argsort(best_h[idx_ones])]
            for idx in sorted_idx[:int(ones - n_best//2)]:
                h_binary[idx] = 0
        elif ones < n_best//2:
            idx_zeros = np.where(h_binary <= 0.5)[0]
            sorted_idx = idx_zeros[np.argsort(-best_h[idx_zeros])]
            for idx in sorted_idx[:int(n_best//2 - ones)]:
                h_binary[idx] = 1
        
        if abs(h_binary.sum() - n_best//2) < 1e-10:
            c5_bin = compute_c5(h_binary)
            if c5_bin < best_c5:
                best_h = h_binary
                best_c5 = c5_bin
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, len(best_h)

import numpy as np
from scipy import optimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_value(corr, T):
    max_c = np.max(corr)
    if T < 1e-12:
        return max_c, np.zeros_like(corr)
    shifted = corr - max_c
    exp_c = np.exp(shifted / T)
    s = np.sum(exp_c)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = exp_c / s
    return max_c + T * np.log(s), w

def objective_and_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, w = soft_max_value(corr, T)
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        wm = w[m]
        if wm < 1e-13:
            continue
        offset = m - (n - 1)
        lo = max(0, -offset)
        hi = min(n - 1, n - 1 - offset)
        
        idx1 = np.arange(lo, hi + 1)
        grad[idx1] += wm * (1.0 - h[idx1 + offset])
        
        j_lo = max(0, lo + offset)
        j_hi = min(n - 1, hi + offset)
        if j_lo <= j_hi:
            idx2 = np.arange(j_lo, j_hi + 1)
            grad[idx2] -= wm * h[idx2 - offset]
    
    return val, grad

    n_old = len(h)
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    # Fix sum
    target = n_new / 2.0
    curr = h_new.sum()
    if curr > 0:

    h_new = np.clip(h_new, 0.0, 1.0)

        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    for T in T_schedule:
        if time.time() - t0 > max_time:
            break
        
        def obj(x):
            v, _ = objective_and_grad(x, T)
            return v
        
        def jac(x):
            _, g = objective_and_grad(x, T)
            return g
        
        def con_fun(x):
            return x.sum() - n / 2.0
        
        def con_jac(x):
            return np.ones(n)
        
        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        bounds = [(0.0, 1.0)] * n
        
        try:
            res = optimize.minimize(
                obj, h, jac=jac, method='SLSQP',
                bounds=bounds, constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            if res.x is not None:
                h_new = np.clip(res.x, 0.0, 1.0)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def coordinate_descent(h, max_time):

    n = len(h)
    h = h.copy()
    t0 = time.time()
    curr_c5 = compute_c5(h)
    
    while time.time() - t0 < max_time:
        improved = False
        for _ in range(50):
            if time.time() - t0 > max_time:
                break

            j = np.random.randint(n)
            if i == j:
                continue

            h_new[i] = np.clip(h[i] + delta, 0.0, 1.0)

                curr_c5 = c5_new
                improved = True
        
        if not improved:
            break
    
    return h

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_vals(corr, T):
    max_c = np.max(corr)
    if T < 1e-14:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        return max_c, w
    shifted = (corr - max_c) / T
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    w = exp_s / s
    return max_c + T * np.log(s), w

def obj_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, w = soft_max_vals(corr, T)
    grad = np.zeros(n)

        if j_lo <= j_hi:
            idx2 = np.arange(j_lo, j_hi + 1)
