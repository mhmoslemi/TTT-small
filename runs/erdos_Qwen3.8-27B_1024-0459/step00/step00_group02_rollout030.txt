def lse_eval(h, beta):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    max_c = np.max(c)
    exp_vals = np.exp(np.clip((c - max_c) / beta, -50, 50))
    S = np.sum(exp_vals)
    f = max_c + beta * np.log(S)

    w = exp_vals / S
    grad = np.zeros(n)
