import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 value for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_value(corr, temperature):
    """Compute smooth maximum of corr array."""
    scaled = corr / temperature
    max_s = np.max(scaled)
    shifted = np.exp(scaled - max_s)
    return temperature * (max_s + np.log(np.sum(shifted)))

def smooth_max_gradient(corr, temperature, h):
    """Compute gradient of smooth max of corr w.r.t. h."""
    n = len(h)
    scaled = corr / temperature
    max_s = np.max(scaled)
    weights = np.exp(scaled - max_s)
    total_w = np.sum(weights)
    weights /= total_w
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:

        shift = k - (n - 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, T):
    scaled = corr / T
    m = np.max(scaled)
    return T * (m + np.log(np.sum(np.exp(scaled - m))))

def smooth_max_weights(corr, T):
    scaled = corr / T
    m = np.max(scaled)
    w = np.exp(scaled - m)
    return w / np.sum(w)

def grad_smooth_max(corr, T, h):
    n = len(h)
    w = smooth_max_weights(corr, T)
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-10:
            continue
        shift = k - (n-1)
        # d(corr[k])/d(h[m]) = (1-h[m+shift])*[valid] - h[m-shift]*[valid]
        # First term: m+shift in [0,n) => m in [-shift, n-1-shift] ∩ [0,n-1]
        m1_start = max(0, -shift)
        m1_end = min(n, n-shift)
        if m1_end > m1_start:
            idx = np.arange(m1_start, m1_end)
            grad[idx] += wk * (1.0 - h[idx + shift])
        # Second term: m-shift in [0,n) => m in [shift, n-1+shift] ∩ [0,n-1]
        m2_start = max(0, shift)
        m2_end = min(n, n+shift)
        if m2_end > m2_start:
            idx = np.arange(m2_start, m2_end)
            grad[idx] -= wk * h[idx - shift]
    return grad

def project_onto_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    n = len(h)
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if diff > 0:
            projected = np.clip(h + mid, 0, 1)
        else:
            projected = np.clip(h + mid, 0, 1)
        s = np.sum(projected)
        if (diff > 0 and s < target_sum) or (diff < 0 and s > target_sum):
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h + mid, 0, 1)

def slsqp_optimize(h0, n, T, max_iter=200):
    dx = 2.0 / n
    
    def obj(h):
        corr = np.correlate(h, 1.0-h, mode='full')
        return smooth_max(corr, T) * dx
    
    def grad(h):
        corr = np.correlate(h, 1.0-h, mode='full')
        return grad_smooth_max(corr, T, h) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x, res.fun, res.success

def projected_gradient(h0, n, T, lr, steps):
    dx = 2.0 / n
    h = h0.copy()
    for s in range(steps):
        corr = np.correlate(h, 1.0-h, mode='full')
        g = grad_smooth_max(corr, T, h) * dx
        h = h - lr * g
        h = project_onto_constraint(h, n/2.0)
    return h

def binary_local_search(h0, n, passes=3):
    """Try flipping individual elements (0<->1) or small adjustments."""
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)

        for i in range(n):

                    improved = True
                    # Re-project
                    h = project_onto_constraint(h, n/2.0)
                else:
                    h[i] = h_old
        if not improved:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, T):
    scaled = corr / T
    m = np.max(scaled)
    return T * (m + np.log(np.sum(np.exp(scaled - m))))

def smooth_max_weights(corr, T):
    scaled = corr / T
    m = np.max(scaled)
    w = np.exp(scaled - m)
    s = np.sum(w)
    if s < 1e-300:
        return np.full(len(corr), 1.0/len(corr))
    return w / s

def grad_smooth_max(corr, T, h):
    n = len(h)
    w = smooth_max_weights(corr, T)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        shift = k - (n - 1)
        m1s = max(0, -shift)
        m1e = min(n, n - shift)
        if m1e > m1s:
            idx = np.arange(m1s, m1e)
            grad[idx] += wk * (1.0 - h[idx + shift])
        m2s = max(0, shift)
        m2e = min(n, n + shift)
        if m2e > m2s:
            idx = np.arange(m2s, m2e)
            grad[idx] -= wk * h[idx - shift]
    return grad

def project_sum(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if diff > 0:
            if s < target:
                lo = mid
            else:
                hi = mid
        else:
            if s > target:
                lo = mid
            else:
                hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h + mid, 0.0, 1.0)

def slsqp_run(h0, n, T, max_iter=200):
    dx = 2.0 / n
    def obj(h):
        return smooth_max(get_corr(h), T) * dx
    def grad(h):
        return grad_smooth_max(get_corr(h), T, h) * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0) for _ in range(n)]
    res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return res.x, res.fun, res.success

def proj_grad_run(h0, n, T, lr, steps):
    dx = 2.0 / n
    h = h0.copy()
    for _ in range(steps):
        c = get_corr(h)
        g = grad_smooth_max(c, T, h) * dx
        h = h - lr * g
        h = project_sum(h, n / 2.0)
    return h

def local_search_binary(h0, n, passes=3):
    dx = 2.0 / n
    h = h0.copy()
    best = compute_c5(h)
    for _ in range(passes):
        improved = False
        idx = np.arange(n)
        for i in idx:
            if time.time() > 20000:
                break
            old = h[i]
            if old > 0.05:
                h[i] = 0.0
                c = compute_c5(h)
                if c < best - 1e-12:
                    h = project_sum(h, n / 2.0)
                    best = c
                    improved = True
                else:
                    h[i] = old
            if h[i] < 0.95:
                old2 = h[i]
                h[i] = 1.0
                c = compute_c5(h)
                if c < best - 1e-12:
                    h = project_sum(h, n / 2.0)
                    best = c
                    improved = True
                else:
                    h[i] = old2
        if not improved:
            break
    return h, best

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_sum(h_new, n_new / 2.0)
