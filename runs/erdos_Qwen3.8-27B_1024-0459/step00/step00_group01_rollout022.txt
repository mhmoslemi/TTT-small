A key insight for local search: if I can identify which lags are "hot" (near the max), I can target modifications that reduce those specific lags.

For a swap of positions i and j (1↔0), the effect on lag k is:
- Changes corr at lags related to |i-j|
- Specifically, moving a 1 from position j to position i affects lags that involve positions near i and j

This is complex to track, so I'll just recompute the full correlation after each swap.

from scipy.optimize import minimize, dual_annealing
import time

def compute_c5(h, n):
    """Compute C5 = max(corr) * dx for step function h with n points."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, n):
    """Ensure h is in [0,1] and sum(h) = n/2."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute correction
        if diff > 0:
            # Add to elements < 1
            mask = h < 1.0
            count = np.sum(mask)
            if count > 0:
                add = min(diff / count, (1.0 - h[mask]).max() * 0.5)
                h[mask] += add
                diff -= add * count
            if diff > 1e-12:
                # Force into specific elements
                idx = np.argsort(h)[:max(1, int(diff))]
                for i in idx:
                    room = 1.0 - h[i]
                    add = min(room, diff)
                    h[i] += add
                    diff -= add
                    if diff < 1e-12:
                        break
        else:
            # Remove from elements > 0
            mask = h > 0.0
            count = np.sum(mask)
            if count > 0:
                rem = min(-diff / count, h[mask].min() * 0.5)
                h[mask] -= rem
                diff += rem * count
            if diff < -1e-12:
                idx = np.argsort(h)[-max(1, int(-diff)):]
                for i in idx:
                    take = min(h[i], -diff)
                    h[i] -= take
                    diff += take
                    if diff > -1e-12:
                        break
    # Final tiny adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15 and count > 0:
        h[np.argmax(h) if diff < 0 else np.argmin(h)] += diff
    return h

                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:

            # Swap values
            old_i, old_j = current[i], current[j]
            current[i], current[j] = current[j], current[i]

            if new_c5 < current_c5 - 1e-15:
                current_c5 = new_c5
                improved_in_round = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best = current.copy()
                # Update lists
                if current[i] > 0.5 and current[j] < 0.5:

def grad_descent(h, n, max_time, start_time, rng):
    """Subgradient descent for the max correlation."""
    current = h.copy()
    current_c5 = compute_c5(current, n)
    best = current.copy()
    best_c5 = current_c5
    target = n / 2.0
    lr = 0.01
    
    while time.time() - start_time < max_time:
        corr = np.correlate(current, 1.0 - current, mode="full")
        k_max = np.argmax(corr)
        shift = k_max - (n - 1)
        
        # Gradient of corr[k_max] w.r.t. current
        grad = np.zeros(n)
        if shift >= 0:
            grad[shift:n] += (1.0 - current[0:n-shift])
            grad[0:n-shift] -= current[shift:n]
        else:
            s = -shift
            grad[s:n] += (1.0 - current[0:n-s])
            grad[0:s] -= current[s:n]
        
        # Project to sum-zero
        grad -= np.mean(grad)
        
        # Step
        new_h = current - lr * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = repair_h(new_h, n)
        
        new_c5 = compute_c5(new_h, n)
        if new_c5 < current_c5:
            current = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best = current.copy()
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-6:
                # Random restart
                current = best.copy() + rng.randn(n) * 0.02
                current = repair_h(current, n)
                current_c5 = compute_c5(current, n)
                lr = 0.01
    
    return best, best_c5

    x_old = np.linspace(0, 1, n_old, endpoint=False) + 0.5/n_old  # centers

    h_new = np.interp(x_new, x_old, h, period=1)  # periodic interpolation

    h_new = np.clip(h_new, 0, 1)
    h_new = repair_h(h_new, n_new)
