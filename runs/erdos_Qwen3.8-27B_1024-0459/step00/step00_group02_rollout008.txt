def soft_max_c5(h, tau=50.0):
    c5_arr = compute_c5_array(h)
    c5_max = np.max(c5_arr)
    shifted = tau * (c5_arr - c5_max)
    shifted = np.clip(shifted, -50, 50)
    return c5_max + (1.0 / tau) * np.log(np.sum(np.exp(shifted)))


def swap_local_search(h, target_sum, rng, max_iters=20000, time_limit=200):
    n = len(h)
    h = h.copy()
    current_c5 = compute_c5(h)
    start_t = time.time()
    deltas = [0.02, 0.05, 0.1, 0.15, 0.3, 0.5, 0.8]
    deltas = deltas + [-d for d in deltas]
    for _ in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            h_trial = h.copy()
            h_trial[i] -= d
            h_trial[j] += d
            if h_trial[i] < -1e-12 or h_trial[i] > 1 + 1e-12:
                continue
            if h_trial[j] < -1e-12 or h_trial[j] > 1 + 1e-12:
                continue
            h_trial = np.clip(h_trial, 0, 1)
            if abs(h_trial.sum() - target_sum) > 1e-9:
                h_trial = make_feasible(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < current_c5 - 1e-12:
                h = h_trial
                current_c5 = c5_trial
                break
    return h


def correlate_gradient(h, k_max):
    n = len(h)
    lag = k_max - (n - 1)
    j_lo = max(0, -lag)
    j_hi = min(n - 1, n - 1 - lag)
    grad = np.zeros(n)
    for j in range(j_lo, j_hi + 1):
        i = j + lag
        grad[j] -= h[i]
        grad[i] += (1.0 - h[j])
    return grad


def multi_active_subgradient(h, target_sum, alpha, top_k=3):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr_raw)[::-1][:top_k]
    grad = np.zeros(n)
    for k in top_idx:
        grad += correlate_gradient(h, int(k))
    grad *= dx / top_k
    h_new = h - alpha * grad
    return make_feasible(h_new, target_sum)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init = 0.5 * np.ones(45)

    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n

    h_init = make_feasible(h_init, target_sum)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    bounds = [(0.0, 1.0)] * n
    sum_constraint = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                      'jac': lambda x: np.ones(n)}

    # Phase 1: SLSQP from initial
    if time_left() > 120:
        try:
            res = minimize(compute_c5, best_h, method='SLSQP',
                          bounds=bounds, constraints=[sum_constraint],
                          options={'maxiter': 3000, 'ftol': 1e-14})
            h_try = make_feasible(res.x, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try
        except Exception:
            pass

    # Phase 2: Multi-active subgradient descent
    if time_left() > 300:
        alpha = 0.02
        patience = 0
        for step in range(10000):
            if time.time() - start_time > budget_s - 200:
                break
            h_new = multi_active_subgradient(best_h, target_sum, alpha, top_k=3)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                patience = 0
            else:
                patience += 1
                if patience > 200:
                    alpha *= 0.85
                    patience = 0
                    if alpha < 1e-6:
                        alpha = 0.01

    # Phase 3: SLSQP from improved point
    if time_left() > 150:
        try:
            res = minimize(compute_c5, best_h, method='SLSQP',
                          bounds=bounds, constraints=[sum_constraint],
                          options={'maxiter': 3000, 'ftol': 1e-14})
            h_try = make_feasible(res.x, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try
        except Exception:
            pass

    # Phase 4: Smooth max SLSQP
    if time_left() > 200:
        for tau in [20.0, 50.0, 100.0]:
            if time_left() < 100:
                break
            try:
                res = minimize(lambda x: soft_max_c5(x, tau=tau), best_h,
                             method='SLSQP', bounds=bounds,
                             constraints=[sum_constraint],
                             options={'maxiter': 2000, 'ftol': 1e-12})
                h_try = make_feasible(res.x, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h, best_c5 = h_try, c5_try
            except Exception:
                pass

    # Phase 5: Swap-based local search
    if time_left() > 80:
        best_h = swap_local_search(best_h, target_sum, rng,
                                   max_iters=50000, time_limit=min(time_left() - 40, 300))
        best_c5 = compute_c5(best_h)

    # Phase 6: Random restarts with local optimization
    restart_count = 0
    while time_left() > 180 and restart_count < 50:
        # Perturb best solution
        noise_scale = 0.08 + 0.04 * restart_count * 0.5
        noise = rng.normal(0, noise_scale, n)
        h_start = make_feasible(best_h + noise, target_sum)

        try:
            res = minimize(compute_c5, h_start, method='SLSQP',
                          bounds=bounds, constraints=[sum_constraint],
                          options={'maxiter': 800, 'ftol': 1e-12})
            h_try = make_feasible(res.x, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                # Polish with swap search
                h_try = swap_local_search(h_try, target_sum, rng,
                                         max_iters=5000, time_limit=30)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h, best_c5 = h_try, c5_try
        except Exception:
            pass
        restart_count += 1

    # Phase 7: Final SLSQP polish
    if time_left() > 60:
        try:
            res = minimize(compute_c5, best_h, method='SLSQP',
                          bounds=bounds, constraints=[sum_constraint],
                          options={'maxiter': 5000, 'ftol': 1e-15})
            if res.fun < best_c5:
                h_try = make_feasible(res.x, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_h, best_c5 = h_try, c5_try
        except Exception:
            pass

    # Phase 8: Final swap polish
    if time_left() > 40:
        best_h = swap_local_search(best_h, target_sum, rng,
                                   max_iters=30000, time_limit=min(time_left() - 15, 120))
        best_c5 = compute_c5(best_h)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = make_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
