from scipy.optimize import minimize
import time

_PARAMS = {'beta': 100.0, 'penalty': 1000.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_func(h):
    n = len(h)
    dx = 2.0 / n
    beta = _PARAMS['beta']
    penalty = _PARAMS['penalty']
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    values = corr * dx
    bv = beta * values
    max_bv = np.max(bv)
    shifted = bv - max_bv
    soft_max = max_bv / beta + np.log(np.sum(np.exp(shifted))) / beta
    sum_h = np.sum(h)
    target = n / 2.0
    p = penalty * (sum_h - target) ** 2
    return soft_max + p

def grad_func(h):
    n = len(h)
    dx = 2.0 / n
    beta = _PARAMS['beta']
    penalty = _PARAMS['penalty']
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    values = corr * dx
    bv = beta * values
    max_bv = np.max(bv)
    shifted = bv - max_bv
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        # Second term: -h[m+s] for m where m+s in [0, n-1]

def upsample_func(h_src, n_dst):
    """Upsample step function from n_src to n_dst points."""
    n_src = len(h_src)
    dx_src = 2.0 / n_src
    x_src = (np.arange(n_src) + 0.5) * dx_src
    x_dst = np.linspace(0, 2.0, n_dst, endpoint=False) + (1.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    h_dst = np.clip(h_dst, 0, 1)
    # Roughly enforce integral
    target_sum = n_dst / 2.0
    current_sum = np.sum(h_dst)
    if current_sum > 0 and abs(current_sum - target_sum) > 1e-6:
        h_dst += (target_sum - current_sum) / n_dst
        h_dst = np.clip(h_dst, 0, 1)
    return h_dst

    """Fix h so that sum(h) = n/2 and 0 <= h <= 1."""
    h = h.copy()

            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': 300000, 'disp': False}
        )
        h_opt = result.x
        elapsed = time.time() - t_start
    except Exception:
        return None
    
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = fix_sum(h_opt, n)
    c5 = compute_c5(h_opt)
    return h_opt, c5

def local_search_binary(h, time_limit):
    """Local search on binary solutions by swapping 1s and 0s."""
    h = h.astype(float).copy()
    h[h < 0.5] = 0.0
    h[h >= 0.5] = 1.0
    n = len(h)
    
    # Ensure correct number of 1s
    num_ones = int(np.sum(h))
    target_ones = n // 2
    if num_ones > target_ones:
        ones_idx = np.where(h == 1)[0]
        extra = np.random.choice(ones_idx, num_ones - target_ones, replace=False)
        h[extra] = 0
    elif num_ones < target_ones:
        zeros_idx = np.where(h == 0)[0]
        needed = np.random.choice(zeros_idx, target_ones - num_ones, replace=False)
        h[needed] = 1
    
    c5_curr = compute_c5(h)
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    rng = np.random.RandomState(42)
    
    t_start = time.time()
    no_improve = 0
    while time.time() - t_start < time_limit and no_improve < 200:
        # Try random swaps
        improved = False
        for _ in range(500):
            if time.time() - t_start > time_limit:
                break
            i = rng.randint(len(ones_idx))
            j = rng.randint(len(zeros_idx))
            a = ones_idx[i]
            b = zeros_idx[j]
            
            h[a] = 0
            h[b] = 1
            c5_new = compute_c5(h)
            
            if c5_new < c5_curr - 1e-12:
                c5_curr = c5_new
                ones_idx[i] = b
                zeros_idx[j] = a
                improved = True
                no_improve = 0
            else:
                h[a] = 1
                h[b] = 0
        
        if not improved:
            no_improve += 1
    
    return h, c5_curr

def random_binary_init(n, seed_offset=0):
    """Create a random binary vector with exactly n/2 ones."""
    rng = np.random.RandomState(42 + seed_offset)
    vec = np.zeros(n)
    ones_pos = rng.choice(n, n // 2, replace=False)
    vec[ones_pos] = 1.0
    # Apply some block structure for better starting point
    return vec

def block_init(n, n_blocks, seed_offset=0):
    """Create a block-structured binary initialization."""
    rng = np.random.RandomState(42 + seed_offset * 100)
    # Divide into blocks, assign 1 or 0 to each block to get n/2 total ones
    block_size = n // n_blocks
    remainder = n % n_blocks
    blocks = []
    for b in range(n_blocks):
        sz = block_size + (1 if b < remainder else 0)
        blocks.append(sz)
    
    # Choose which blocks get more 1s
    num_full = (n // 2) // block_size
    num_partial = (n // 2) - num_full * block_size
    
    block_vals = []
    full_count = 0
    for sz in blocks:
        if full_count < num_full and sz <= block_size:
            block_vals.append(sz)
            full_count += 1
        else:
            block_vals.append(0)
    
    # Distribute remaining
    remaining = n // 2 - sum(block_vals)
    nonfull = [i for i in range(n_blocks) if block_vals[i] < blocks[i]]
    rng.shuffle(nonfull)
    for i in nonfull:
        if remaining <= 0:
            break
        add = min(remaining, blocks[i] - block_vals[i])
        block_vals[i] += add
        remaining -= add
    
    # Build the vector
    vec = np.zeros(n)
    pos = 0
    for i in range(n_blocks):
        n_ones_in_block = block_vals[i]
        # Randomly place ones within block
        if n_ones_in_block > 0:
            positions = rng.choice(range(pos, pos + blocks[i]), n_ones_in_block, replace=False)
            vec[positions] = 1.0
        pos += blocks[i]
    
    # Fix count
    if np.sum(vec) != n // 2:
        diff = n // 2 - int(np.sum(vec))
        if diff > 0:
            zeros = np.where(vec == 0)[0]
            vec[rng.choice(zeros, diff, replace=False)] = 1
        else:
            ones = np.where(vec == 1)[0]
            vec[rng.choice(ones, -diff, replace=False)] = 0
    
    return vec

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Phase 0: Evaluate parent if available
    try:
        if initial_h_values is not None:
            h_parent = np.array(initial_h_values, dtype=float)
            c5_parent = compute_c5(h_parent)
            best_h = h_parent.copy()
            best_c5 = c5_parent
    except NameError:
        h_parent = None
    
    if h_parent is None:
        # Create a default starting point
        n = 256
        h_parent = np.full(n, 0.5)
        # Add some structure
        for i in range(n):
            h_parent[i] = 1.0 if (i % 4 < 2) else 0.0
        h_parent = fix_sum(h_parent, n)
        best_h = h_parent.copy()
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at parent resolution
    nparent = len(h_parent)
    if time_left() > 30:
        result = optimize_lbfgsb(h_parent, beta=80, penalty=500,
                                  max_iter=3000, time_limit=min(time_left()-15, 60))
        if result is not None and result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0]
    
    # Phase 2: Optimize at higher resolutions with multiple inits
    resolutions = [256, 512, 1024]
    
    for n in resolutions:
        if time_left() < 60:
            break
        
        # Create multiple initializations
        inits = []
        
        # Init 1: Upsampled parent
        h_up = upsample_func(h_parent, n)
        inits.append(h_up)
        
        # Init 2: Upsampled best so far (if different from parent)
        if best_h is not None and len(best_h) != nparent:
            h_up2 = upsample_func(best_h, n)
            inits.append(h_up2)
        
        # Init 3: Random binary
        inits.append(random_binary_init(n, seed_offset=1))
        
        # Init 4: Block-structured binary
        inits.append(block_init(n, n_blocks=8, seed_offset=2))
        
        # Init 5: Block-structured binary (different)
        inits.append(block_init(n, n_blocks=16, seed_offset=3))
        
        for init_idx, h_init in enumerate(inits):
            if time_left() < 40:
                break
            
            time_per_run = min(time_left() - 20, max(30, 300.0 * (512.0 / n)))
            
            # Run L-BFGS-B
            result = optimize_lbfgsb(h_init, beta=100, penalty=1000,
                                      max_iter=5000, time_limit=time_per_run)
            if result is not None and result[1] < best_c5:
                best_c5 = result[1]
                best_h = result[0]
            
            # If solution is near-binary, try local search
            if best_h is not None and len(best_h) == n and time_left() > 30:
                binary_test = (best_h > 0.4).sum()
                if abs(binary_test - n/2) < 5:
                    h_bin, c5_bin = local_search_binary(best_h, time_limit=min(time_left()-15, 30))
                    if c5_bin < best_c5:
                        best_c5 = c5_bin
                        best_h = h_bin
    
    # Phase 3: Final refinement at best resolution
    if best_h is not None and time_left() > 30:
        n_best = len(best_h)
        # Higher beta for sharper max approximation
        result = optimize_lbfgsb(best_h, beta=200, penalty=2000,
                                  max_iter=5000, time_limit=min(time_left()-15, 200))
        if result is not None and result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0]
        
        # Another refinement with even higher beta
        if time_left() > 30:
            result = optimize_lbfgsb(best_h, beta=400, penalty=5000,
                                      max_iter=3000, time_limit=min(time_left()-10, 100))
            if result is not None and result[1] < best_c5:
                best_c5 = result[1]
                best_h = result[0]
    
    # Phase 4: One more pass - try to improve with n=512 specifically
    if time_left() > 30 and len(best_h) != 512:
        h_try = upsample_func(best_h, 512)
        result = optimize_lbfgsb(h_try, beta=150, penalty=1000,
                                  max_iter=4000, time_limit=min(time_left()-10, 120))
        if result is not None and result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0]
    
    # Final: ensure constraints
    n_final = len(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = fix_sum(best_h, n_final)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_final
