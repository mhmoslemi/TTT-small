def _c5_from_h(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        return 0.0
    dx = 2.0 / float(n)
    b = 1.0 - h
    if n <= 700:
        corr = np.correlate(h, b, mode="full")
        return float(corr.max() * dx)
    m = 1
    need = 2 * n - 1
    while m < need:
        m *= 2
    H = np.fft.rfft(h, m)
    B = np.fft.rfft(b, m)
    c = np.fft.irfft(H * np.conjugate(B), m)
    pos = c[:n]
    neg = c[m - n + 1:m]
    mx = max(float(pos.max()), float(neg.max()))
    return mx * dx


def random_balanced(n, rng):
    h = np.zeros(n, dtype=np.uint8)
    t = n // 2
    if t > 0:
        choices = rng.choice(n, t, replace=False)
        h[choices] = 1
    return h


def upsample_binary(parent, n, rng):
    p = np.asarray(parent, dtype=float)
    if p.size == 0:
        return random_balanced(n, rng)
    if p.size == n:
        vals = p.copy()
    else:
        x_old = np.linspace(0.0, 1.0, p.size)
        x_new = np.linspace(0.0, 1.0, n)
        vals = np.interp(x_new, x_old, p)
    vals = vals + rng.normal(0.0, 1e-3, n)
    h = np.zeros(n, dtype=np.uint8)
    t = n // 2
    if t > 0:
        idx = np.argpartition(vals, t - 1)[t:]
        h[idx] = 1
    return h


def structured_binary(n, variant, rng):

def compute_D_int(h):
    n = int(h.size)
    hi = h.astype(np.int32)
    bi = 1 - hi
    D = np.zeros(2 * n - 1, dtype=np.int32)

        D[off + s] = int(np.dot(hi[:n - s], bi[s:]))
        D[off - s] = int(np.dot(hi[s:], bi[:n - s]))
    return D


def apply_swap(h, b, D, p, q, n, off):
    if p == q:
        return
    if int(h[p]) == int(h[q]):
        return

    v = int(h[p])
    sb = off - p
    sh = off - (n - 1 - p)
    if v == 0:
        D[sb:sb + n] += b
        D[sh:sh + n] -= h[::-1]
    else:
        D[sb:sb + n] -= b
        D[sh:sh + n] += h[::-1]
    D[off] -= 1
    h[p] = 1 - v
    b[p] = v

    v = int(h[q])
    sb = off - q
    sh = off - (n - 1 - q)
    if v == 0:
        D[sb:sb + n] += b
        D[sh:sh + n] -= h[::-1]
    else:
        D[sb:sb + n] -= b
        D[sh:sh + n] += h[::-1]
    D[off] -= 1
    h[q] = 1 - v
    b[q] = v

    pp = []

    for ix in idx:
        s = int(ix - off)
        if s == 0:
            continue
        k = -s if s < 0 else s
        if s > 0:
            i = np.flatnonzero((h[:n - k] == 1) & (h[k:] == 0))
        else:

        pp = pp[::step][:limit]
        qq = qq[::step][:limit]
    arr = np.empty((len(pp), 2), dtype=np.int32)
    arr[:, 0] = pp
    arr[:, 1] = qq
    return arr

    ub_arr = np.ones(m)
    ub_arr[tcol] = float(n)
    bounds = Bounds(lb=np.zeros(m), ub=ub_arr)

    c = np.zeros(m)
    c[tcol] = 1.0

    res = None
    for opts in (
        {"time_limit": int(timeout), "mip_rel_gap": 1e-6, "threads": 1},
        {"time_limit": int(timeout), "mip_rel_gap": 1e-6},
        {"time_limit": int(timeout)},
    ):
        try:
            res = milp(c, constraints=constraints, integrality=integrality, bounds=bounds, options=opts)
            break
        except Exception:
            res = None

    if res is None:
        return None
    x = getattr(res, "x", None)
    if x is None or not np.all(np.isfinite(x)):
        return None

    hval = np.rint(x[:n]).astype(np.uint8)
    target = n // 2
    s = int(hval.sum())
    if s != target:
        rngx = np.random.default_rng(123456)
        if s > target:
            ones = np.flatnonzero(hval)
            excess = s - target
            if ones.size > 0:
                take = min(excess, ones.size)
                rem = rngx.choice(ones.size, take, replace=False)
                hval[ones[rem]] = 0
        else:
            zeros = np.flatnonzero(1 - hval)
            need = target - s
            if zeros.size > 0:
                take = min(need, zeros.size)
                rem = rngx.choice(zeros.size, take, replace=False)
                hval[zeros[rem]] = 1

    D = compute_D_int(hval)
    return hval, int(D.max())


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 5.0
    if deadline <= t0 + 10.0:
        deadline = t0 + max(5.0, budget - 1.0)

    base_seed = int(seed)
    parent = globals().get("initial_h_values", None)

    rng0 = np.random.default_rng(base_seed)
    if parent is not None:
        parent = np.asarray(parent, dtype=float)
        best_h = parent.copy()
    else:
        best_h = random_balanced(48, rng0).astype(float)

    best_c5 = _c5_from_h(best_h)

    schedule = [(48, 0.22), (64, 0.18), (96, 0.15), (128, 0.12), (160, 0.08)]

    for n, w in schedule:
        if time.time() >= deadline - 40.0:
            break
        allocated = budget * w
        allocated = min(allocated, deadline - time.time())
        if allocated < 45.0:
            continue

        starts = []
        if parent is not None and parent.size > 0:
            starts.append(upsample_binary(parent, n, np.random.default_rng(base_seed + n)))
        starts.append(random_balanced(n, np.random.default_rng(base_seed + n + 7)))
        if n <= 64:
            starts.append(structured_binary(n, 1, np.random.default_rng(base_seed + n + 13)))
        else:
            if parent is not None and parent.size > 0:
                starts.append(upsample_binary(parent, n, np.random.default_rng(base_seed + n + 17)))
            else:
                starts.append(random_balanced(n, np.random.default_rng(base_seed + n + 17)))

        sub_budget = allocated / float(max(1, len(starts)))
        for i, st in enumerate(starts):
            if time.time() >= deadline - 30.0:
                break
            phase_time = min(sub_budget, deadline - time.time())
            if phase_time < 20.0:
                continue
            phase_deadline = time.time() + phase_time
            try:
                hbest, dmax = search_binary(st, n, phase_deadline, base_seed + n * 100 + i)
                ccheck = _c5_from_h(hbest.astype(float))
                if ccheck < best_c5 - 1e-12:
                    best_c5 = ccheck
                    best_h = hbest.astype(float)
            except Exception:
                pass

    if time.time() < deadline - 260.0:
        timeout = min(deadline - time.time() - 20.0, 250.0)
        try:
            res = milp_attempt(48, timeout)
            if res is not None:
                hbest, dmax = res
                ccheck = _c5_from_h(hbest.astype(float))
                if ccheck < best_c5 - 1e-12:
                    best_c5 = ccheck
                    best_h = hbest.astype(float)
        except Exception:
            pass

    if time.time() < deadline - 170.0:
        timeout = min(deadline - time.time() - 20.0, 180.0)
        try:
            res = milp_attempt(64, timeout)
            if res is not None:
                hbest, dmax = res
                ccheck = _c5_from_h(hbest.astype(float))
                if ccheck < best_c5 - 1e-12:
                    best_c5 = ccheck
                    best_h = hbest.astype(float)
        except Exception:
            pass

    remaining = deadline - time.time()
    if remaining > 170.0:
        n = 200
        rlarge = np.random.default_rng(base_seed + 999)
        if best_h.size > 0:
            start = upsample_binary(best_h, n, rlarge)
        else:
            start = random_balanced(n, rlarge)
        try:
            hbest, dmax = search_binary(start, n, deadline - 10.0, base_seed + 997)
            ccheck = _c5_from_h(hbest.astype(float))
            if ccheck < best_c5 - 1e-12:
                best_c5 = ccheck
                best_h = hbest.astype(float)
        except Exception:
            pass

    best_c5 = _c5_from_h(best_h)
    return best_h.astype(float), float(best_c5), int(best_h.size)
