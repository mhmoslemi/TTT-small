```python
import time

def compute_score(h, dx):
    """Compute C5 for a given h"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_subgradient(h, dx, smooth_param=1.0):
    """Compute gradient of smoothed max of correlation"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Smooth max using log-sum-exp
    max_corr = np.max(corr)
    shifted = corr - max_corr
    if smooth_param > 0:
        log_weights = shifted / smooth_param
        log_weights -= np.max(log_weights)
        weights = np.exp(log_weights)
        weights /= np.sum(weights)
    else:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
    
    # Compute gradient: d/dh[j] of Σ_k w_k * S_k
    # where S_k = Σ_i h[i] * (1-h[i+m]) for valid (i, i+m)
    # m = k - (n-1)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        m = k - (n - 1)  # shift in units of dx
        if m >= 0:
            # S = Σ_{i=0}^{n-1-m} h[i] * (1-h[i+m])
            # dS/dh[j] = (1-h[j+m]) for j ∈ [0, n-1-m]

            for j in range(n - m):
                grad[j] += w * (1.0 - h[j + m])
```

```python
def project_onto_constraint(h, target_sum):
    """Project h onto {0<=h[i]<=1, sum(h)=target_sum}"""
    n = len(h)
    # Bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0, 1)
    return projected
```

```python
    t_budget2 = min(t_budget2, remaining2)
    if t_budget2 > 5:
        if best_n != n2:
            h2_start = upsample_to(best_h, best_n, n2)
        else:
            h2_start = best_h.copy()
        h2, c5_2 = optimize_inner(h2_start, n2, t_budget2, seed=seed + 90)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2

    # Phase 3: Optimize at n=180
    n3 = 180
    t_budget3 = budget_s * 0.20
    remaining3 = budget_s * 0.95 - (time.time() - t0)
    t_budget3 = min(t_budget3, remaining3)
    if t_budget3 > 5:
        if best_n != n3:
            h3_start = upsample_to(best_h, best_n, n3)
        else:
            h3_start = best_h.copy()
        h3, c5_3 = optimize_inner(h3_start, n3, t_budget3, seed=seed + 180)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, n3

    # Phase 4: Polish at best resolution
    remaining4 = budget_s * 0.95 - (time.time() - t0)
    if remaining4 > 5:
        h4, c5_4 = optimize_inner(best_h, best_n, remaining4, seed=seed + 500)
        if c5_4 < best_c5:
            best_h, best_c5, best_n = h4, c5_4, best_n

    # Try binary rounding
    remaining5 = budget_s * 0.98 - (time.time() - t0)
    if remaining5 > 2 and best_h is not None:
        dx_b = 2.0 / best_n
        if best_n % 2 == 0:
            h_bin = binary_round(best_h, best_n)
            h_bin = project(h_bin, best_n / 2.0)
            c5_bin = compute_score(h_bin, dx_b)
            if c5_bin < best_c5:
                best_h, best_c5 = h_bin, c5_bin

    # Final verification
    final_dx = 2.0 / best_n
    final_c5 = compute_score(best_h, final_dx)

    # Safety: ensure feasibility
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_score(best_h, final_dx)

    return best_h, final_c5, best_n
```
