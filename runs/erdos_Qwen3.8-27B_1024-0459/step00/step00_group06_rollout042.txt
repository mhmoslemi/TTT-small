from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_active_lags(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-10)[0]
    return max_val * dx, active

def c5_subgradient(h, n_points, active_lags=None):
    """Compute subgradient of C5 at h"""
    dx = 2.0 / n_points
    n = n_points
    
    if active_lags is None:
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        active_lags = np.where(corr >= max_val - 1e-10)[0]
    
    grad = np.zeros(n)
    for m in active_lags:
        tau = m - (n - 1)
        g = np.zeros(n)
        if tau >= 0:
            g[:n-tau] += (1 - h[tau:n])
            g[tau:] -= h[:n-tau]
        else:
            t = -tau
            g[t:] += (1 - h[:n-t])
            g[:n-t] -= h[t:n]
        grad += g
    grad /= len(active_lags)
    return grad * dx

    """Project v onto {0 <= h <= 1, sum(h) = target_sum}"""

    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_details(h, n_points):
    """Returns (c5_value, active_lags)"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')

    active = np.where(corr >= max_val - 1e-10)[0]
    return max_val * dx, active

def c5_subgradient(h, n_points, active_lags):
    """Compute subgradient of C5 at h using given active lags"""
    dx = 2.0 / n_points
    n = n_points
    grad = np.zeros(n)
    for m in active_lags:
        tau = m - (n - 1)
        if tau >= 0:
            grad[:n-tau] += (1 - h[tau:n])
            grad[tau:] -= h[:n-tau]
        else:
            t = -tau
            grad[t:] += (1 - h[:n-t])
            grad[:n-t] -= h[t:n]

def pair_optimize(h, j, k, n, target_sum):
    """Optimize the pair (h[j], h[k]) with all other h's fixed"""
    # h[k] = S - h[j] where S = target_sum - sum(h[i] for i != j, k)
    others_sum = np.sum(h) - h[j] - h[k]
    S = target_sum - others_sum
    # Constraints: 0 <= h[j] <= 1, 0 <= S - h[j] <= 1
    # So: max(0, S-1) <= h[j] <= min(1, S)
    lo = max(0.0, S - 1.0)
    hi = min(1.0, S)
    if lo >= hi:
        return h
    
    # Minimize max_l c_l(h[j]) where h[k] = S - h[j]
    # Use golden section search
    def f(x_j):
        h_temp = h.copy()
        h_temp[j] = x_j
        h_temp[k] = S - x_j
        h_temp = np.clip(h_temp, 0, 1)  # safety
        return compute_c5(h_temp, n)
    
    # Golden section
    ratio = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - ratio * (b - a)
    d = a + ratio * (b - a)
    fc = f(c)
    fd = f(d)
    
    for _ in range(60):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - ratio * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + ratio * (b - a)
            fd = f(d)
    
    x_opt = (a + b) / 2
    h_new = h.copy()
    h_new[j] = x_opt
    h_new[k] = S - x_opt
    h_new = np.clip(h_new, 0, 1)
    return h_new

            h_new = project_feasible(h - step * grad, n, target_sum)
            f_new = compute_c5(h_new, n)
            if f_new < f_curr - 1e-14:
                break
            step *= 0.5
        else:
            h_new = project_feasible(h - step * grad, n, target_sum)
            f_new = compute_c5(h_new, n)

        h = h_new
        if f_new < best_f:
            best_h = h.copy()
            best_f = f_new
            no_improve = 0
            alpha = min(alpha * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                alpha *= 0.7

    return best_h, best_f

def subgradient_method(h, n, target_sum, max_iter, deadline):
    best_h = h.copy()
    best_f = compute_c5(h, n)
    alpha = 0.03

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = float(np.max(corr))
        active = np.where(corr >= mx - 1e-10)[0]
        grad = np.zeros(n)
        for m in active:
            tau = m - (n - 1)
            _lag_grad_into(h, n, tau, grad)
        grad /= len(active)
        grad *= (2.0 / n)

        h = h - alpha * grad
        h = project_feasible(h, n, target_sum)
        f_cur = compute_c5(h, n)
        if f_cur < best_f:
            best_h = h.copy()
            best_f = f_cur
        if it % 10000 == 9999:
            alpha *= 0.85

    return best_h, best_f

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 95
        h0 = np.full(n, 0.5)

    target_sum = n / 2.0
    h0 = project_feasible(h0, n, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, n)

    # Phase 1: Projected gradient with smooth objective from parent
    deadline1 = start_time + budget_s * 0.25
    h_p, c5_p = projected_gradient(h0.copy(), n, target_sum, beta=40.0, max_iter=200000, deadline=deadline1)
    if c5_p < best_c5:
        best_h = h_p
        best_c5 = c5_p

    # Phase 2: Subgradient for diversity
    deadline2 = start_time + budget_s * 0.40
    h_s, c5_s = subgradient_method(best_h.copy(), n, target_sum, max_iter=200000, deadline=deadline2)
    if c5_s < best_c5:
        best_h = h_s
        best_c5 = c5_s

    # Phase 3: More projected gradient from current best with different beta
    deadline3 = start_time + budget_s * 0.55
    h_p2, c5_p2 = projected_gradient(best_h.copy(), n, target_sum, beta=80.0, max_iter=100000, deadline=deadline3)
    if c5_p2 < best_c5:
        best_h = h_p2
        best_c5 = c5_p2

    # Phase 4: Multi-start with perturbations
    n_starts = 40
    for attempt in range(n_starts):
        deadline4 = start_time + budget_s * 0.95
        if time.time() > deadline4:
            break

        # Various perturbation strategies
        strat = attempt % 4
        if strat == 0:
            scale = 0.05 + 0.05 * rng.rand()
            h_try = best_h + rng.randn(n) * scale
        elif strat == 1:
            # Flip some values
            h_try = best_h.copy()
            idx = rng.choice(n, size=n // 10, replace=False)
            h_try[idx] = 1.0 - h_try[idx]
        elif strat == 2:
            # Sine perturbation
            freq = rng.choice([2, 3, 5, 7])
            phase = rng.rand() * 2 * np.pi
            pert = 0.1 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
            h_try = best_h + pert
        else:
            # Larger random restart
            h_try = best_h + rng.randn(n) * 0.15

        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n, target_sum)

        # Quick projected gradient
        dl = start_time + budget_s * 0.97
        h_opt, c5_opt = projected_gradient(h_try, n, target_sum, beta=50.0, max_iter=30000, deadline=dl)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 5: Try different resolution if time permits
    if time.time() < start_time + budget_s * 0.75:
        for n_alt in [100, 128, 150]:
            if time.time() > start_time + budget_s * 0.90:
                break
            n_alt_ts = n_alt / 2.0
            # Interpolate best_h to new resolution
            old_x = np.linspace(0, 2, n, endpoint=False)
            new_x = np.linspace(0, 2, n_alt, endpoint=False)
            h_interp = np.interp(new_x, old_x, best_h, period=2.0) if False else np.interp(new_x, old_x, best_h)
            h_interp = np.clip(h_interp, 0.0, 1.0)
            h_interp = project_feasible(h_interp, n_alt, n_alt_ts)

            dl = start_time + budget_s * 0.92
            h_new, c5_new = projected_gradient(h_interp, n_alt, n_alt_ts, beta=50.0, max_iter=50000, deadline=dl)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n = n_alt
                target_sum = n_alt_ts

    # Final polish
    if time.time() < start_time + budget_s * 0.97:
        h_new, c5_new = projected_gradient(best_h.copy(), n, target_sum, beta=100.0, max_iter=50000, deadline=start_time + budget_s * 0.98)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Ensure feasibility
    best_h = project_feasible(best_h, n, target_sum)
    best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, n
