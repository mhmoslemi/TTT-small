def project_to_feasible(h, target_sum):
    """Project h onto {x: 0 <= x <= 1, sum(x) = target_sum}."""
    h = np.asarray(h, dtype=float).copy()

        s = h_clipped.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:

        h = h_clipped + diff / len(h)

def solve_lp_step(g, N, target_sum):
    """Given g (fixed complement), solve LP for optimal h."""
    n_corr = 2 * N - 1
    n_ineq = n_corr + 2 * N
    
    # Build correlation constraint matrix
    # For output index m (0 to 2N-2), lag = m - (N-1)
    # c[m] = sum_j h[j] * g[j - (m-(N-1))] for valid j-(m-(N-1))
    # Constraint: c[m] <= T, i.e., sum_j h[j] * g[j-lag_m] - T <= 0
    
    A_corr = np.zeros((n_corr, N))
    for m in range(n_corr):
        lag = m - (N - 1)
        # j ranges over [0, N-1], and j - lag must be in [0, N-1]
        # So j in [max(0, lag), min(N-1, N-1+lag)]
        j_min = max(0, lag)
        j_max = min(N - 1, N - 1 + lag)
        for j in range(j_min, j_max + 1):
            jj = j - lag
            A_corr[m, j] = g[jj]
    
    # Full constraint matrix
    A_ub = np.zeros((n_ineq, N + 1))
    b_ub = np.zeros(n_ineq)
    
    # Correlation constraints: A_corr @ h - T <= 0
    A_ub[:n_corr, :N] = A_corr
    A_ub[:n_corr, N] = -1.0
    
    # h >= 0: -h <= 0
    A_ub[n_corr:n_corr+N, :N] = -np.eye(N)
    b_ub[n_corr:n_corr+N] = 0.0
    
    # h <= 1: h <= 1
    A_ub[n_corr+N:, :N] = np.eye(N)
    b_ub[n_corr+N:] = 1.0
    
    # Equality: sum(h) = target_sum
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([target_sum])
    
    # Objective: minimize T
    c_obj = np.zeros(N + 1)
    c_obj[N] = 1.0
    
    bounds = [(0.0, 1.0)] * N + [(0.0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    return None, None

import time

def compute_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, target):
    h = np.array(h, dtype=float)
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:

        h = h + (target - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def build_lp_matrices(g, N):
    """Build LP matrices for the correlation minimization problem."""
    n_corr = 2 * N - 1
    
    # A_corr[m, j] = g[j - lag_m] if valid, else 0
    # where lag_m = m - (N-1)
    A_corr = np.zeros((n_corr, N))
    for m in range(n_corr):
        lag = m - (N - 1)
        j_start = max(0, lag)
        j_end = min(N, N + lag)  # exclusive
        for j in range(j_start, j_end):
            jj = j - lag
            if 0 <= jj < N:

    A_ub[n_corr+N:, :N] = np.eye(N)
