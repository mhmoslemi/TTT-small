from scipy.optimize import minimize
import time

_target_sum = 0.0
_dx_val = 0.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def enforce_constraint(h, n_pts):
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    target = n_pts / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        h += diff / n_pts
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-13:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.any():
                h[free] += diff / float(free.sum())
                h = np.clip(h, 0.0, 1.0)
    return h

def obj_func(h):
    return compute_c5(h, _dx_val)

def con_func(h):
    return h.sum() - _target_sum

def local_search(h_best, best_c5, n_pts, dx, rng, n_iter, deadline, step_init=0.05):
    step = step_init
    for it in range(n_iter):
        if time.time() > deadline:
            break
        n_pert = rng.randint(1, 4)
        idx = rng.choice(n_pts, size=n_pert, replace=False)
        h_try = h_best.copy()
        h_try[idx] += rng.uniform(-step, step, size=n_pert)
        h_try = enforce_constraint(h_try, n_pts)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5:
            h_best = h_try
            best_c5 = c5_try
        if it % 3000 == 2999 and step > 0.001:
            step *= 0.92
    return h_best, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:

        h_try = enforce_constraint(res.x, n_pts)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 and abs(h_try.sum() - n_pts/2.0) < 1e-6:

        return (best_h, compute_c5(best_h, dx), n_pts)
