import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def upsample_nn(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    idx = np.floor(np.arange(n_new) * n_old / n_new).astype(np.int64)
    idx = np.clip(idx, 0, n_old - 1)
    return h_old[idx].copy()


def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') & 1
    return project_feasible(h, n)


def sa_search(h_init, n, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    T = 0.006
    T_min = 0.0002
    step = 0.04
    stagnant = 0
    patience = 4000

    while time.time() < deadline:
        r = rng.random()
        h_new = h.copy()

        if r < 0.35:
            idx = rng.randint(n)
            h_new[idx] += rng.randn() * step
        elif r < 0.75:
            blen = rng.randint(2, max(3, n // 12))
            s = rng.randint(0, n - blen)
            h_new[s:s + blen] += rng.randn(blen) * step * 0.6
        else:
            blen = rng.randint(max(2, n // 8), max(3, n // 3))
            s = rng.randint(0, max(1, n - blen))

        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        d = c5_new - c5

        if d < 0 or (T > T_min and rng.random() < np.exp(-d / T)):
            h = h_new
            c5 = c5_new
            if c5_new < best_c5 - 1e-12:
                best_h = h.copy()
                best_c5 = c5_new
                stagnant = 0
            else:
                stagnant += 1
        else:
            stagnant += 1

        T = max(T_min, T * 0.9996)

        if stagnant >= patience:

            step = max(0.015, step * 0.75)
            stagnant = 0
            h = best_h.copy()
            c5 = best_c5
        elif stagnant < patience // 4:
            step = min(0.1, step * 1.003)

    return best_h, best_c5

    best = h.copy()
