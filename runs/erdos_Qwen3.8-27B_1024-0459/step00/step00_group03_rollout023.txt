import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute C5 = max(corr) * dx"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum} using bisection on lambda"""
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient_step(h, dx, n):
    """Compute subgradient of C5 at h"""
    corr = np.correlate(h, 1-h, mode="full")
    k_max = np.argmax(corr)
    lag = k_max - (n - 1)
    # grad of g_lag(h) = sum_i h[i]*(1-h[i+lag])
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        # h[i] appears as h[i] in sum: contributes (1 - h[i+lag]) if i+lag in [0,n)
        if 0 <= i + lag < n:
            g += (1 - h[i + lag])
        # h[i] appears as h[i+lag] where i+lag = i, i.e., original index = i-lag
        if 0 <= i - lag < n:
            g -= h[i - lag]
        grad[i] = g
    return grad * dx

def opt_subgradient(h, dx, n, max_iter, budget_s, start_time):
    """Optimize h using subgradient method"""
    target_sum = n / 2.0
    alpha = dx * n * 0.1  # initial step size
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
        grad = subgradient_step(h, dx, n)
        h_new = h - alpha * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        alpha *= 0.995  # decay
    
    return best_h, best_c5

For the gradient computation, I need to be careful about the boundaries. Let me re-derive:

The actual shift is: lag = k - (n-1)
c[k] = sum_{i} h[i + lag] * (1-h[i]) where the sum is over valid i (where both indices are in [0, n-1])

Equivalently: c[k] = sum_{j} h[j] * (1-h[j - lag]) where j = i + lag, valid when 0≤j<n and 0≤j-lag<n

For the gradient ∂c[k]/∂h[m]:
- h[m] appears in the first factor: when j=m and 0≤m-lag<n → contributes (1-h[m-lag])
- h[m] appears in the second factor: when j-lag=m, i.e., j=m+lag, and 0≤m+lag<n → contributes -h[m+lag]

So: ∂c[k]/∂h[m] = [0≤m-lag<n ? (1-h[m-lag]) : 0] - [0≤m+lag<n ? h[m+lag] : 0]

    """Return (objective_value, gradient) for smoothed max objective"""
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")  # shape (2n-1,)
    # Softmax weights (numerically stable)
    corr_max = np.max(corr)
    exp_corr = np.exp(tau * (corr - corr_max))
    w = exp_corr / np.sum(exp_corr)
    
    # Objective value (smoothed): (1/tau) * log(sum exp(tau*corr)) * dx
    obj = (corr_max + np.log(np.sum(exp_corr)) / tau) * dx

        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        wk = w[k]
        # ∂c[k]/∂h[m] = [m-lag in [0,n) ? (1-h[m-lag]) : 0] - [m+lag in [0,n) ? h[m+lag] : 0]
        # Vectorized:
        # Term 1: for m where 0 <= m-lag < n, i.e., lag <= m < n+lag, intersect [0,n)
        m_start1 = max(0, lag)
        m_end1 = min(n, n + lag)
        if m_end1 > m_start1:

        # Term 2: for m where 0 <= m+lag < n, i.e., -lag <= m < n-lag, intersect [0,n)

        m_end2 = min(n, n - lag)
        if m_end2 > m_start2:
            grad[m_start2:m_end2] -= wk * h[m_start2 + lag:m_end2 + lag]
    
    return obj, grad * dx

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    # Bisection on the dual variable
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)

    return np.correlate(h, 1.0 - h, mode="full")

    """Gradient of c_lag(h) = sum_i h[i]*(1-h[i+lag]) with valid indices"""
    grad = np.zeros(n)
    # Term 1: h[m] in first position, needs m+lag valid (or m-lag... let me re-derive)

    # c(lag) = sum_{m: 0<=m<n, 0<=m-lag<n} h[m] * (1-h[m-lag])
    #         = sum_{m: 0<=m<n, 0<=m+lag<n} h[m] * (1-h[m+lag])  [replacing m with m+lag in the second form]
    
    # ∂/∂h[m] c(lag):
    # m appears in h[m] (first factor): gives (1-h[m-lag]) if 0<=m<n and 0<=m-lag<n
    #                 OR (1-h[m+lag]) if 0<=m<n and 0<=m+lag<n [from the other form]
    # m appears in h[m±lag] (second factor): gives -h[m] ... 
    
    # Let me just use: c(lag) = sum_i h[i] * (1-h[i+lag]) for valid i
    # where valid means 0<=i<n and 0<=i+lag<n
    # ∂c/∂h[m] = (1-h[m+lag]) * [0<=m+lag<n] * [0<=m<n]  (from h[m] in first factor, i=m)
    #          + sum_i [i+lag=m] * h[i] * (-1) * [0<=i<n, 0<=i+lag<n]
    #          = (1-h[m+lag]) * [valid(m, m+lag)] - h[m-lag] * [valid(m-lag, m)]
    
    # valid(a,b) means 0<=a<n and 0<=b<n

    m1s = max(0, -lag)
    m1e = min(n, n - lag)
    if m1e > m1s:

        # h[m+lag] for m in [m1s, m1e) is h[max(0,-lag)+lag : min(n,n-lag)+lag]
        idx1 = slice(m1s + lag, m1e + lag)
        grad[m1s:m1e] += (1.0 - h[idx1])
    
    # Term 2: m-lag in [0,n) and m in [0,n): m in [max(0,lag), min(n, n+lag))
    m2s = max(0, lag)
    m2e = min(n, n + lag)
    if m2e > m2s:
        # grad[m] -= h[m-lag] for m in [m2s, m2e)
        idx2 = slice(m2s - lag, m2e - lag)
        grad[m2s:m2e] -= h[idx2]
    
    return grad

    corr = get_corr(h)  # shape (2n-1,)
    corr_shifted = corr - np.max(corr)
    exp_vals = np.exp(tau * corr_shifted)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def gradient_at_lag(h, lag, n):
    grad = np.zeros(n)
    m1s = max(0, -lag)
    m1e = min(n, n - lag)
    if m1e > m1s:
        grad[m1s:m1e] += (1.0 - h[m1s + lag:m1e + lag])
    m2s = max(0, lag)
    m2e = min(n, n + lag)
    if m2e > m2s:
        grad[m2s:m2e] -= h[m2s - lag:m2e - lag]
    return grad

def smoothed_grad(h, n, tau):
    corr = get_corr(h)
    cmax = np.max(corr)
    exp_vals = np.exp(tau * (corr - cmax))
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = (n - 1) - k
        grad += w[k] * gradient_at_lag(h, lag, n)
    obj = (cmax + np.log(np.sum(exp_vals)) / tau)
    return obj, grad
