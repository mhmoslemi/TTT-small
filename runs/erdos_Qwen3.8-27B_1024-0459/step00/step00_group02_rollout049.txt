        h45, c5_45 = optimize_one_start(h45, n45, min(remaining() * 0.15, 150), time.time(), seed)
        h45, c5_45 = local_search(h45, n45, min(remaining() * 0.05, 50), time.time(), rng)
        h45, c5_45 = binary_refine(h45, n45, min(remaining() * 0.05, 40), time.time(), rng)
        overall_best_h = h45.copy()
        overall_best_c5 = c5_45
        overall_best_n = n45

    # Multi-start at n=45
    n_starts = max(3, int(remaining() / 80))
    for s in range(n_starts):
        if remaining() < 30:
            break
        h_init = random_init(n45, np.random.RandomState(seed + 1000 + s))
        h_s, c5_s = optimize_one_start(h_init, n45, min(remaining() * 0.08, 100), time.time(), seed + 1000 + s)
        h_s, c5_s = local_search(h_s, n45, min(remaining() * 0.02, 20), time.time(), rng)
        if c5_s < overall_best_c5:
            overall_best_c5 = c5_s
            overall_best_h = h_s.copy()
            overall_best_n = n45
            h45 = h_s.copy()

    # Phase 2: Try higher resolutions
    for n_try in n_list:
        if n_try <= n45:
            continue
        if remaining() < 120:
            break
        # Upsample from best solution
        h_up = upsample(overall_best_h, overall_best_n, n_try)
        # Also try random init
        h_rand = random_init(n_try, np.random.RandomState(seed + n_try))

        # Optimize upsampled
        time_alloc = min(remaining() * 0.25, 300)
        h_opt, c5_opt = optimize_one_start(h_up, n_try, time_alloc * 0.6, time.time(), seed + n_try)
        h_opt, c5_opt = local_search(h_opt, n_try, time_alloc * 0.25, time.time(), rng)
        h_opt, c5_opt = binary_refine(h_opt, n_try, time_alloc * 0.15, time.time(), rng)
        if c5_opt < overall_best_c5:
            overall_best_c5 = c5_opt
            overall_best_h = h_opt.copy()
            overall_best_n = n_try

        # Optimize random init if time allows
        if remaining() > 60:
            h_opt2, c5_opt2 = optimize_one_start(h_rand, n_try, min(remaining() * 0.1, 60), time.time(), seed + 5000 + n_try)
            h_opt2, c5_opt2 = local_search(h_opt2, n_try, min(remaining() * 0.03, 20), time.time(), rng)
            if c5_opt2 < overall_best_c5:
                overall_best_c5 = c5_opt2
                overall_best_h = h_opt2.copy()
                overall_best_n = n_try

    # Phase 3: Final refinement on the best solution found
    if remaining() > 20:
        final_h = overall_best_h.copy()
        final_n = overall_best_n
        final_c5 = overall_best_c5
        final_h, final_c5 = local_search(final_h, final_n, min(remaining() * 0.4, 120), time.time(), rng)
        final_h, final_c5 = binary_refine(final_h, final_n, min(remaining() * 0.3, 80), time.time(), rng)
        if final_c5 < overall_best_c5:
            overall_best_c5 = final_c5
            overall_best_h = final_h.copy()
            overall_best_n = final_n

    # Final time-check optimization
    if remaining() > 15:
        final_h = overall_best_h.copy()
        final_n = overall_best_n
        final_h, final_c5 = optimize_one_start(final_h, final_n, min(remaining() * 0.7, 100), time.time(), seed + 99999)
        if final_c5 < overall_best_c5:
            overall_best_c5 = final_c5
            overall_best_h = final_h.copy()
            overall_best_n = final_n

    # Validate
    hn = overall_best_n
    hx = np.clip(overall_best_h, 0.0, 1.0)
    hx = project(hx, hn)
    final_c5 = compute_c5(hx, hn)
    overall_best_c5 = final_c5

    return hx, overall_best_c5, hn
