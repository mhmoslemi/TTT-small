def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    if v.size == 0:
        return v
    v = np.clip(v, 0.0, 1.0)
    # bisection over lambda
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    out = np.clip(v + 0.5*(lo+hi), 0.0, 1.0)
    diff = target - out.sum()
    if abs(diff) > 1e-13:
        idx = np.where((out > 1e-12) & (out < 1.0 - 1e-12))[0]
        if idx.size:
            for i in idx:
                if diff <= 0:
                    add = max(-out[i], min(diff, -out[i])) # wait if diff negative, add negative between -out and diff
                else:
                    add = min(1.0 - out[i], diff)
                out[i] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        if abs(diff) > 1e-12:
            i = int(np.argmax((out > 1e-12) & (out < 1.0 - 1e-12))) if idx.size else 0
            out[i] = float(np.clip(out[i] + diff, 0.0, 1.0))
    else:
        # still remove tiny -0
        out[out < 0.0] = 0.0; out[out > 1.0] = 1.0

import time
import math
import numpy as np
from scipy.optimize import minimize


class _OptTimeout(Exception):
    pass


_OPT_N = 80
_OPT_SUM = 40.0
_OPT_EPS = 1e-3
_OPT_PEN = 0.0
_OPT_BOUNDS = []
_OPT_STOP_TIME = None
_OPT_ACTIVE_MAX = 256
_OPT_ACTIVE_THRESH = 1e-7
_OPT_CACHE = None


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    if v.size == 0:
        return v
    v = np.clip(v, 0.0, 1.0)
    target = float(target)

    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid

    out = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(out.sum())

    if abs(diff) > 1e-13:
        interior = np.where((out > 1e-12) & (out < 1.0 - 1e-12))[0]
        if interior.size:
            for i in interior:
                if diff <= 0.0:
                    add = max(-out[i], diff)
                else:
                    add = min(1.0 - out[i], diff)
                out[i] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        if abs(diff) > 1e-12:
            i = 0
            out[i] = float(np.clip(out[i] + diff, 0.0, 1.0))

    out = np.clip(out, 0.0, 1.0)
    return out


def sanitize_h(h):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1:
        h = h.ravel()
    n = h.size
    if n == 0:
        return h
    target = 0.5 * n
    if not np.all(np.isfinite(h)):
        return project_box_sum(np.full(n, 0.5), target)
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - target) <= 1e-11:
        return h
    return project_box_sum(h, target)


def _autocorr(h):
    n = int(h.size)
    if n <= 128:
        out = np.empty(n, dtype=float)
        for p in range(n):
            m = n - p
            if m > 0:
                out[p] = float(np.dot(h[:m], h[p:]))
            else:
                out[p] = 0.0
        return out

    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    F = np.fft.rfft(h, L)
    ac = np.fft.irfft(F * F.conj(), L)
    return ac[:n].astype(float, copy=False)


def _fast_values(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return 0.0, np.empty(0, dtype=float), np.empty(0, dtype=float)

    dx = 2.0 / n
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])

    P = _autocorr(h)
    O0 = total - float(P[0])

    if n == 1:
        return float(max(0.0, O0)) * dx, np.empty(0, dtype=float), np.empty(0, dtype=float)

    A = pref[n - 1:0:-1]
    B = total - pref[1:n]
    Pp = P[1:n]

    Op = A - Pp
    On = B - Pp

    Op = np.maximum(Op, 0.0)
    On = np.maximum(On, 0.0)

    return float(max(0.0, O0)) * dx, Op * dx, On * dx


def _values_array(h):
    O0, Op, On = _fast_values(h)
    return np.concatenate((np.array([O0], dtype=float), Op, On))


def _fast_c5(h):
    vals = _values_array(h)
    if vals.size == 0:
        return float("inf")
    return float(np.max(vals))


def _exact_c5(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return float("inf")
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _lag_from_index(idx, n):
    idx = int(idx)
    if idx == 0:
        return 0
    if idx < n:
        return idx
    return -(idx - (n - 1))


def _add_lag_grad(g, h, lag):
    n = int(h.size)
    lag = int(lag)
    if lag == 0:
        g += 1.0 - 2.0 * h
        return

    p = abs(lag)
    if p >= n:
        return
    m = n - p
    if m <= 0:
        return

    if lag > 0:
        g[:m] += 1.0 - h[p:]
        g[p:] -= h[:m]
    else:
        g[p:] += 1.0 - h[:m]
        g[:m] -= h[p:]


def _ensure_cache(h):
    global _OPT_CACHE

    harr = np.asarray(h, dtype=float)
    if _OPT_CACHE is not None:
        cached_h, ww, idx, smooth = _OPT_CACHE
        if cached_h.shape == harr.shape and np.array_equal(cached_h, harr):
            return ww, idx

    vals = _values_array(harr)
    if vals.size == 0:
        ww = np.array([1.0], dtype=float)
        idx = np.array([0], dtype=int)
        _OPT_CACHE = (harr.copy(), ww, idx, 0.0)
        return ww, idx

    m = float(np.max(vals))
    eps = float(_OPT_EPS)

    if eps <= 1e-12:
        tol = max(1e-11, 1e-9 * max(1.0, m))
        idx = np.where(vals >= m - tol)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(vals))], dtype=int)
        elif idx.size > _OPT_ACTIVE_MAX:
            idx = np.argsort(vals)[::-1][:_OPT_ACTIVE_MAX].astype(int)
            idx = np.sort(idx)
        ww = np.full(idx.size, 1.0 / float(idx.size), dtype=float)
        smooth = m
    else:
        shifted = (vals - m) / eps
        shifted = np.clip(shifted, -745.0, 0.0)
        w = np.exp(shifted)
        s = float(w.sum())

        if s <= 0.0 or not math.isfinite(s):
            idx = np.array([int(np.argmax(vals))], dtype=int)
            ww = np.array([1.0], dtype=float)
            smooth = m
        else:
            ww_all = w / s
            idx = np.where(ww_all >= _OPT_ACTIVE_THRESH)[0]
            if idx.size == 0:
                idx = np.array([int(np.argmax(vals))], dtype=int)
            elif idx.size > _OPT_ACTIVE_MAX:
                idx = np.argsort(ww_all)[::-1][:_OPT_ACTIVE_MAX].astype(int)
                idx = np.sort(idx)

            ww = ww_all[idx].astype(float, copy=False)
            sw = float(ww.sum())
            if sw <= 0.0 or not math.isfinite(sw):
                idx = np.array([int(np.argmax(vals))], dtype=int)
                ww = np.array([1.0], dtype=float)
            else:
                ww = ww / sw

            smooth = m + eps * math.log(s)

    _OPT_CACHE = (harr.copy(), ww.astype(float, copy=False), idx.astype(int, copy=False), float(smooth))
    return _OPT_CACHE[1], _OPT_CACHE[2]


def _smooth_objective(h):
    if _OPT_STOP_TIME is not None and time.time() > _OPT_STOP_TIME:
        raise _OptTimeout
    _ensure_cache(h)
    return float(_OPT_CACHE[3])


def _smooth_gradient(h):
    if _OPT_STOP_TIME is not None and time.time() > _OPT_STOP_TIME:
        raise _OptTimeout

    harr = np.asarray(h, dtype=float)
    ww, idx = _ensure_cache(harr)
    n = int(harr.size)
    g = np.zeros(n, dtype=float)

    for k in range(idx.size):
        w = float(ww[k])
        if w > 1e-12:
            lag = _lag_from_index(int(idx[k]), n)
            _add_lag_grad(g, harr, lag)

    return (2.0 / n) * g


def _sum_eq(h):
    return _OPT_SUM - float(np.sum(h))


def _sum_jac(h):
    return np.ones(_OPT_N, dtype=float)


def _get_constraints():
    return ({"type": "eq", "fun": _sum_eq, "jac": _sum_jac},)


def _optimize_slsqp(h0, n, eps, maxfun, stop_time):
    global _OPT_N, _OPT_SUM, _OPT_EPS, _OPT_PEN, _OPT_BOUNDS, _OPT_CACHE, _OPT_STOP_TIME

    n = int(n)
    _OPT_N = n
    _OPT_SUM = 0.5 * n
    _OPT_EPS = float(eps)
    _OPT_PEN = 0.0
    _OPT_BOUNDS = [(0.0, 1.0)] * n
    _OPT_CACHE = None
    _OPT_STOP_TIME = float(stop_time) if stop_time is not None else None

    x0 = np.asarray(h0, dtype=float).copy()
    x0 = project_box_sum(x0, _OPT_SUM)

    mf = int(max(50, int(maxfun)))
    mi = int(min(800, max(50, mf)))

    try:
        res = minimize(
            _smooth_objective,
            x0,
            jac=_smooth_gradient,
            method="SLSQP",
            bounds=_OPT_BOUNDS,
            constraints=_get_constraints(),
            options={"maxiter": mi, "maxfun": mf, "ftol": 1e-8, "iprint": 0},
        )
        if hasattr(res, "x") and res.x is not None:
            x = np.asarray(res.x, dtype=float)
        else:
            x = x0
    except Exception:
        x = x0

    _OPT_STOP_TIME = None
    return project_box_sum(x, _OPT_SUM)


def _pen_objective(h):
    if _OPT_STOP_TIME is not None and time.time() > _OPT_STOP_TIME:
        raise _OptTimeout
    _ensure_cache(h)
    val = float(_OPT_CACHE[3])
    s = float(np.sum(h))
    d = s - _OPT_SUM
    val += _OPT_PEN * d * d
    return val


def _pen_jac(h):
    if _OPT_STOP_TIME is not None and time.time() > _OPT_STOP_TIME:
        raise _OptTimeout
    g = _smooth_gradient(h)
    if _OPT_PEN != 0.0:
        s = float(np.sum(h))
        g = g + (2.0 * _OPT_PEN) * (s - _OPT_SUM) * np.ones(_OPT_N, dtype=float)
    return g


def _optimize_lbfgs(h0, n, eps, maxfun, penalty, stop_time):

def _active_grad_exact(h, tol_rel=1e-7, cap=32):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    vals = _values_array(h)
    m = float(np.max(vals))
    tol = max(1e-10, tol_rel * max(1.0, m))

    idx = np.where(vals >= m - tol)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(vals))], dtype=int)
    elif idx.size > cap:
        idx = np.argsort(vals)[::-1][:cap].astype(int)

    g = np.zeros(n, dtype=float)
    dx = 2.0 / n
    for i in idx:
        lag = _lag_from_index(int(i), n)
        _add_lag_grad(g, h, lag)
    return dx * (g / float(max(1, idx.size)))


def _polish_subgrad(h, iterations, stop_time):
    h = np.asarray(h, dtype=float).copy()
    n = int(h.size)
    if n == 0:
        return h

    target = 0.5 * n
    h = project_box_sum(h, target)
    F = _fast_c5(h)
    alpha = 0.06
    stuck = 0

    for _ in range(int(iterations)):
        if stop_time is not None and time.time() > stop_time:
            break

        g = _active_grad_exact(h, tol_rel=1e-7, cap=32)
        nrm = float(np.linalg.norm(g))
        if nrm < 1e-10:
            break

        d = -g / nrm
        accepted = False
        a = alpha

        for _ in range(25):
            cand = project_box_sum(h + a * d, target)
            Fn = _fast_c5(cand)
            if Fn <= F + 1e-12:
                h = cand
                F = Fn
                alpha = min(0.25, a * 1.3)
                accepted = True
                stuck = 0
                break
            a *= 0.5

        if not accepted:
            alpha *= 0.5
            stuck += 1
            if stuck > 40 or alpha < 1e-6:
                break

    return h


def _quick_hillclimb(h, iterations, stop_time, rng):
    if rng is None:
        rng = np.random.default_rng(0)

    h = np.asarray(h, dtype=float).copy()
    n = int(h.size)
    if n == 0:
        return h

    target = 0.5 * n
    h = project_box_sum(h, target)
    F = _fast_c5(h)
    step = 0.05

    for _ in range(int(iterations)):
        if stop_time is not None and time.time() > stop_time:
            break

        scale = step * (0.5 + float(rng.random()))
        cand = project_box_sum(h + scale * rng.standard_normal(n), target)
        Fn = _fast_c5(cand)

        if Fn < F - 1e-12:
            h = cand
            F = Fn
            step = min(0.2, step * 1.1)
        else:
            step = max(0.008, step * 0.9)

    return h


def _merged_update(candidate, best_c, best_h, local_c, local_h):
    if candidate is None:
        return best_c, best_h, local_c, local_h

    cand = np.asarray(candidate, dtype=float)
    if cand.ndim != 1:
        cand = cand.ravel()
    if cand.size == 0 or not np.all(np.isfinite(cand)):
        return best_c, best_h, local_c, local_h

    cand = sanitize_h(cand)
    c = _exact_c5(cand)
    if not math.isfinite(c):
        return best_c, best_h, local_c, local_h

    c = float(c)

    if local_h is None or c < float(local_c) - 1e-12:
        local_h = cand.copy()
        local_c = c

    if c < float(best_c) - 1e-12:
        best_h = cand.copy()
        best_c = c

    return best_c, best_h, local_c, local_h


def _upsample_repeat(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if h.size == n:
        return project_box_sum(h, 0.5 * n)

    n0 = int(h.size)
    if n0 == 0:
        return project_box_sum(np.full(n, 0.5), 0.5 * n)

    if n % n0 == 0:
        out = np.repeat(h, n // n0)
    else:
        out = _upsample_linear(h, n)

    return project_box_sum(out, 0.5 * n)


def _upsample_linear(h, n):
    h = np.asarray(h, dtype=float)
    n = int(n)

    n0 = int(h.size)
    if n0 == n:
        return project_box_sum(h, 0.5 * n)
    if n0 == 0:
        return project_box_sum(np.full(n, 0.5), 0.5 * n)
    if n0 == 1:
        return project_box_sum(np.full(n, float(h[0])), 0.5 * n)

    x_old = (np.arange(n0, dtype=float) + 0.5) * (2.0 / n0)
    x_new = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    v = np.interp(x_new, x_old, h)
    return project_box_sum(v, 0.5 * n)


def _random_wave(n, rng):
    n = int(n)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / n)
    maxf = max(2, n // 3)

    f1 = int(rng.integers(1, maxf + 1))
    p1 = float(rng.uniform(0.0, 2.0 * math.pi))
    w = np.sin(math.pi * f1 * x + p1)

    r = float(rng.random())
    if r < 0.5:
        f2 = int(rng.integers(1, max(1, maxf // 2 + 1)))
        p2 = float(rng.uniform(0.0, 2.0 * math.pi))
        w = w + 0.5 * np.sin(math.pi * f2 * x + p2)
    elif r < 0.8:
        w = w - 0.35 * np.cos(math.pi * f1 * x)

    return w


def _random_blocks(n, rng):
    n = int(n)
    h = np.empty(n, dtype=float)
    pos = 0
    val = int(rng.integers(0, 2))

    min_len = max(1, n // 32)
    max_len = max(min_len + 1, n // 6)
    hi = max_len + 1

    while pos < n:
        if hi > min_len:
            length = int(rng.integers(min_len, hi))
        else:
            length = min_len
        end = min(n, pos + length)
        h[pos:end] = float(val)
        pos = end
        if pos < n:
            val = 1 - val

    return project_box_sum(h, 0.5 * n)


def _random_start(base, n, rng, scale):
    n = int(n)
    base_arr = np.asarray(base, dtype=float)
    if base_arr.size != n:
        base_arr = _upsample_repeat(base_arr, n)
    else:
        base_arr = base_arr.copy()

    r = float(rng.random())
    if r < 0.5:
        h = base_arr + float(scale) * rng.standard_normal(n)
    elif r < 0.8:
        wave = _random_wave(n, rng)
        h = base_arr + 0.5 * float(scale) * (rng.standard_normal(n) + wave)
    else:
        wave = _random_wave(n, rng)
        h = base_arr + float(scale) * wave

    return project_box_sum(h, 0.5 * n)


def _synthetic_start(n, seed):

def _stage_optimize(h_start, n, c_best, h_best, h_alt, rng, t_end):
    n = int(n)
    h_start = np.asarray(h_start, dtype=float)
    if h_start.size != n:
        h_start = _upsample_repeat(h_start, n)

    c_best, h_best, local_c, local_h = _merged_update(h_start, c_best, h_best, float("inf"), None)

    if h_alt is not None:
        h_alt = np.asarray(h_alt, dtype=float)
        if h_alt.size == n:
            c_best, h_best, local_c, local_h = _merged_update(h_alt, c_best, h_best, local_c, local_h)

    if n > 64 and t_end is not None and time.time() <= t_end - 80:
        stop = min(t_end - 20, time.time() + 45)
        h_lb = _optimize_lbfgs(local_h, n, 0.004, 400, 1000.0, stop)
        c_best, h_best, local_c, local_h = _merged_update(h_lb, c_best, h_best, local_c, local_h)

    if n <= 128:
        main = [
            (0.008, 800, 60),
            (0.0025, 1000, 70),
            (0.0008, 1000, 70),
            (0.00025, 800, 60),
        ]
    elif n <= 256:
        main = [
            (0.006, 500, 55),
            (0.002, 600, 60),
            (0.0007, 500, 55),
            (0.0002, 400, 50),
        ]
    elif n <= 512:
        main = [
            (0.005, 300, 60),
            (0.0018, 350, 60),
            (0.0006, 300, 50),
        ]
    else:
        main = [
            (0.004, 200, 80),
            (0.0015, 200, 80),
            (0.0005, 150, 60),
        ]

    for eps, mf, tsec in main:
        if t_end is not None and time.time() > t_end - max(50, tsec + 20):
            break
        stop = min(t_end - 20, time.time() + tsec)
        h_opt = _optimize_slsqp(local_h, n, eps, mf, stop)
        c_best, h_best, local_c, local_h = _merged_update(h_opt, c_best, h_best, local_c, local_h)

    if t_end is not None and time.time() <= t_end - 80:
        h_pol = _polish_subgrad(local_h, 2000, t_end - 30)
        c_best, h_best, local_c, local_h = _merged_update(h_pol, c_best, h_best, local_c, local_h)

    if n <= 128:
        scales = [0.04, 0.08, 0.15, 0.25, 0.40, 0.65, 1.0]
        count = 2
        r_eps = 0.0015
        r_mf = 250
        r_tsec = 25
        use_lb = False
    elif n <= 256:
        scales = [0.06, 0.12, 0.22, 0.38, 0.60]
        count = 1
        r_eps = 0.001
        r_mf = 150
        r_tsec = 25
        use_lb = True
    elif n <= 512:
        scales = [0.10, 0.20, 0.40]
        count = 1
        r_eps = 0.0008
        r_mf = 100
        r_tsec = 20
        use_lb = True
    else:
        scales = []
        count = 0
        r_eps = 0.0
        r_mf = 0
        r_tsec = 0
        use_lb = True

    for scale in scales:
        for _ in range(count):
            if t_end is not None and time.time() > t_end - 70:
                break
            h_r = _random_start(local_h, n, rng, scale)
            stop = min(t_end - 20, time.time() + r_tsec)

            if use_lb and n > 128:
                h_opt = _optimize_lbfgs(h_r, n, r_eps, r_mf, 1000.0, stop)
            else:
                h_opt = _optimize_slsqp(h_r, n, r_eps, r_mf, stop)

            if t_end is not None and time.time() <= t_end - 60:
                h_opt2 = _optimize_slsqp(h_opt, n, 0.0005, min(250, r_mf), min(t_end - 20, time.time() + 25))
                c_best, h_best, local_c, local_h = _merged_update(h_opt2, c_best, h_best, local_c, local_h)

    return c_best, h_best, local_c, local_h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0

    if budget <= 10.0:
        h_best = None
        c_best = float("inf")
        try:
            parent = initial_h_values
        except Exception:
            parent = None
        if parent is not None:
            h_best = sanitize_h(parent)
            c_best = _exact_c5(h_best)
        if h_best is None:
            c_best, h_best = _synthetic_start(80, seed)
        h_best = sanitize_h(h_best)
        c_best = _exact_c5(h_best)
        return h_best, float(c_best), int(h_best.size)

    deadline = t0 + budget
    safe_end = deadline - 45.0

    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = initial_h_values
    except Exception:
        parent = None

    h_best = None
    c_best = float("inf")

    if parent is not None:
        h_parent = np.asarray(parent, dtype=float)
        if h_parent.size > 0:
            h_best = sanitize_h(h_parent)
            c_best = _exact_c5(h_best)

    if h_best is None:
        c_best, h_best = _synthetic_start(80, seed)

    c_best = float(c_best)
    h_best = np.asarray(h_best, dtype=float).copy()
    work_h = h_best.copy()

    base_n = int(work_h.size)
    if base_n < 2:
        base_n = 80

    n_list = []
    m = max(base_n, 80) if base_n < 80 else base_n
    while m <= 1024 and len(n_list) < 4:
        if m % 2 == 1:
            m += 1
        n_list.append(int(m))
        m *= 2

    if budget < 600:
        n_list = n_list[:2]
    elif budget < 1000:
        n_list = n_list[:3]
    else:
        n_list = n_list[:4]

    for n_target in n_list:
        if time.time() > safe_end - 80:
            break

        if work_h.size == n_target:
            h_rep = work_h.copy()
        else:
            h_rep = _upsample_repeat(work_h, n_target)

        if h_rep.size == n_target:
            h_lin = h_rep.copy()
        else:
            h_lin = _upsample_linear(work_h, n_target)
            h_lin = _upsample_linear(h_rep, n_target)

        c_best, h_best, local_c, local_h = _stage_optimize(
            h_rep, n_target, c_best, h_best, h_lin, rng, safe_end
        )

        if local_h is not None:
            work_h = np.asarray(local_h, dtype=float).copy()

    if time.time() < safe_end - 80:
        h_q = _quick_hillclimb(h_best, 5000, safe_end - 40, rng)
        c_best, h_best, _, _ = _merged_update(h_q, c_best, h_best, float("inf"), None)

    if time.time() < safe_end - 50:
        h_pol = _polish_subgrad(h_best, 2000, safe_end - 25)
        c_best, h_best, _, _ = _merged_update(h_pol, c_best, h_best, float("inf"), None)

    h_best = sanitize_h(h_best)
    c_best = _exact_c5(h_best)

    if not math.isfinite(c_best):
        h_best = project_box_sum(np.full(80, 0.5), 40.0)
        c_best = _exact_c5(h_best)

    return h_best, float(c_best), int(h_best.size)
