    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def compute_gradients(h, n, top_frac=0.08):
    """Compute smoothed subgradient of C5"""
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    n_lags = len(conv)
    n_top = max(1, int(n_lags * top_frac))
    top_indices = np.argsort(conv)[-n_top:]

        k = ki - n + 1

            # result[k] = sum_{u=k}^{n-1} h[u]*(1-h[u-k])
            js = np.arange(k, n)
            g[js] += w * (1.0 - h[js - k])
            js2 = np.arange(0, n - k)
            g[js2] -= w * h[js2 + k]
        else:

            g[js] += w * (1.0 - h[js + m])
            js2 = np.arange(m, n)
            g[js2] -= w * h[js2 - m]

        if time.time() - t0 > time_budget:
            break
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 70
        h_init = rng.random(n_init)
        h_init = project_feasible(h_init, n_init)
    
    best_h = h_init.copy()
    best_c5 = compute_C5(h_init, n_init)
    n_best = n_init
    
    def elapsed():
        return time.time() - t_start
    
    def remaining():
        return budget_s - elapsed()
    
    # Phase 1: Subgradient descent from initial (n~70), ~150s
    phase1_time = min(150.0, budget_s * 0.12)
    h1, c5_1 = subgradient_optimize(h_init, n_init, rng, phase1_time, alpha_init=0.008)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
        n_best = n_init
    
    # Phase 2: Local search refinement, ~100s
    if remaining() > 50:
        phase2_time = min(100.0, remaining() * 0.15)
        h2, c5_2 = local_search(best_h, n_best, rng, max_iter=500000, step_size_init=0.04)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
    
    # Phase 3: Bimodal push + local search, ~80s
    if remaining() > 50:
        h3 = bimodal_push(best_h, n_best, rng, strength=0.15)
        h3, c5_3 = local_search(h3, n_best, rng, max_iter=300000, step_size_init=0.03)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
    
    # Phase 4: Multi-resolution search
    resolutions = []
    for nr in [2*n_best, 3*n_best, int(4*n_best)]:
        if nr <= 600:
            resolutions.append(nr)
    if not resolutions:
        resolutions = [140, 280, 420]
    
    for n_trial in resolutions:
        if remaining() < 60:
            break
        
        # Upsample current best
        h_up = upsample(best_h, n_best, n_trial)
        c5_up = compute_C5(h_up, n_trial)
        
        # Optimize at this resolution
        phase_time = min(remaining() * 0.2, 120.0)
        h_opt, c5_opt = subgradient_optimize(h_up, n_trial, rng, phase_time, alpha_init=0.004)
        
        # Also try local search
        if remaining() > 30:
            h_ls, c5_ls = local_search(h_opt, n_trial, rng, max_iter=100000, step_size_init=0.03)
            if c5_ls < c5_opt:
                c5_opt = c5_ls
                h_opt = h_ls
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            n_best = n_trial
    
    # Phase 5: Random restarts at original resolution
    if remaining() > 40:
        n_restart = max(50, min(n_best, 100))
        for trial in range(20):
            if remaining() < 15:
                break
            h_r = rng.random(n_restart)
            h_r = project_feasible(h_r, n_restart)
            # Quick local search
            h_r, c5_r = local_search(h_r, n_restart, rng, max_iter=20000, step_size_init=0.1)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r
                n_best = n_restart
    
    # Phase 6: Final polish
    if remaining() > 20:
        h_polish, c5_polish = subgradient_optimize(best_h, n_best, rng, remaining() * 0.7, alpha_init=0.003)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_polish
    
    # Final validation and correction
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, n_best)
    final_c5 = compute_C5(best_h, n_best)
    
    # Ensure sum constraint
    assert abs(np.sum(best_h) - n_best / 2.0) < 1e-6, f"Sum constraint violated: {np.sum(best_h)} vs {n_best/2}"
    assert np.min(best_h) >= -1e-10
    assert np.max(best_h) <= 1.0 + 1e-10
    
    return best_h, final_c5, n_best
