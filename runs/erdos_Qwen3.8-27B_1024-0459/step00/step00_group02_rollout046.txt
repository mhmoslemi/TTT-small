import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return dx * np.max(corr)

def compute_all_crosscorrs(h, n_points):
    """Compute all cross-correlation values"""
    one_minus_h = 1.0 - h
    return np.correlate(h, one_minus_h, mode='full')

def project_to_feasible(h, n_points):
    """Project h onto the feasible set: sum(h) = n/2, 0 <= h <= 1"""
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-10:
        h = np.full(n_points, 0.5)
        return h
    # Scale to get correct sum
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fine adjustment: distribute remaining mass
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute to elements not at bounds
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        if np.sum(mask) > 0:
            per = diff / np.sum(mask)
            h[mask] += per
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(values, mu):
    """Smooth maximum using log-sum-exp"""
    max_val = np.max(values)
    if mu < 1e-10:
        return max_val
    shifted = values / mu
    shifted = shifted - np.max(shifted)
    return max_val + mu * np.log(np.sum(np.exp(shifted)))

def compute_smooth_c5(h, n_points, mu):
    """Compute smoothed C5"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return smooth_max(corr, mu)

def compute_smooth_gradient(h, n_points, mu):
    """Compute gradient of smoothed C5 with respect to h"""
    dx = 2.0 / n_points
    n = n_points
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Weights for soft-max
    max_corr = np.max(corr)
    weights = np.exp((corr - max_corr) / mu)
    weights /= np.sum(weights)
    
    # Gradient of each corr[k] with respect to h
    # corr[k] for output index k (0 to 2n-2), shift = k - (n-1)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if weights[idx] < 1e-12:
            continue
        k = idx - (n - 1)  # shift
        if k >= 0:
            # corr = sum_{i=k}^{n-1} h[i] * (1-h[i-k])
            # d/dh[j]: 
            #   j in [k, n-1]: contributes (1 - h[j-k])
            #   j in [0, n-1-k]: contributes (-h[j+k])
            for i in range(k, n):
                grad[i] += weights[idx] * dx * (1.0 - h[i - k])
            for i in range(0, n - k):
                grad[i] -= weights[idx] * dx * h[i + k]
        else:
            # k < 0, let k' = -k
            kp = -k
            # corr = sum_{i=0}^{n-1-kp} h[i] * (1-h[i+kp])
            # d/dh[j]:
            #   j in [0, n-1-kp]: contributes (1 - h[j+kp])
            #   j in [kp, n-1]: contributes (-h[j-kp])
            for i in range(0, n - kp):
                grad[i] += weights[idx] * dx * (1.0 - h[i + kp])
            for i in range(kp, n):
                grad[i] -= weights[idx] * dx * h[i - kp]
    
    return grad
