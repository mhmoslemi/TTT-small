Let me also think about whether continuous solutions can beat binary ones. 

For the continuous case, h[i] can be in [0,1]. The objective is:
max_k sum_i h[i]*(1-h[i+k]) * dx

This is a quadratic function of h. For a fixed k, it's:
sum_i h[i] - sum_i h[i]*h[i+k]

The first term is fixed (sum h[i] = n/2). So we want to maximize sum_i h[i]*h[i+k] for the worst k.

This is like maximizing the minimum autocorrelation. For continuous values, we might do better than binary by "smoothing out" the correlations.

- Allow small perturbations
- Use projected gradient descent
- Or formulate as a QP (even though non-convex, we can try)

This is a non-convex QCQP (quadratically constrained quadratic program) because the constraints have h[i]*h[i+k] terms (which are indefinite quadratic forms).

However, we can use the fact that at the optimum, many h[i] will be at the boundaries (0 or 1). So a good approach is:
1. Find a good binary solution
2. Try to "relax" it slightly to see if we can reduce the max
3. Use local perturbation

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def pair_flip_search(h, dx, max_iters, seed, use_sa=True, t0=0.005, cooling=0.9995):
    rng = np.random.RandomState(seed)
    h = (h >= 0.5).astype(float).copy()
    n = len(h)
    
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    temp = t0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, dx)
        
        delta = new_c5 - current_c5
        if delta <= 0 or (use_sa and rng.random() < np.exp(-delta / (temp + 1e-10))):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            current_c5 = new_c5
            # Update arrays
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        if use_sa:
            temp *= cooling
    
    return best_h, best_c5

def continuous_refine(h, dx, max_iters=500, step_size=0.1, seed=42):

    rng = np.random.RandomState(seed)
    h = h.copy()
    n = len(h)
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    for it in range(max_iters):
        # Pick a random subset of indices and perturb
        num_perturb = rng.randint(1, 5)
        indices = rng.choice(n, num_perturb, replace=False)
        
        h_new = h.copy()
        for idx in indices:
            h_new[idx] += rng.uniform(-step_size, step_size)
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Renormalize: adjust to maintain sum = n/2
        target_sum = n / 2.0
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-10:
            # Adjust uniformly
            diff = target_sum - current_sum
            h_new += diff / n
            h_new = np.clip(h_new, 0, 1)
            # If clipping changed the sum, adjust again
            diff = target_sum - np.sum(h_new)
            mask = (h_new > 0) & (h_new < 1)
            if np.sum(mask) > 0:
                h_new[mask] += diff / np.sum(mask)
                h_new = np.clip(h_new, 0, 1)
        
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5:
            current_c5 = new_c5
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        # else: don't update h
    
    return best_h, best_c5

    rng = np.random.RandomState(seed)
    h = h.copy().astype(float)
    n = len(h)
    target_sum = n / 2.0
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h *= target_sum / np.sum(h)
    h = np.clip(h, 0, 1)
    # Adjust sum
    diff = target_sum - np.sum(h)
    mask = (h > 0) & (h < 1)
    if np.sum(mask) > 0:
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for it in range(max_iters):
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        
        # Compute gradient of g_{k_star}
        # g = sum_{valid i} h[i] * (1-h[i+shift]) where shift = k_star - (n-1)
        shift = k_star - (n - 1)
        grad = np.zeros(n)
        
        # First term: h[i] * (1-h[i+shift]) → ∂/∂h[j] = (1-h[j+shift]) if j+shift valid
        # Second term: h[i-shift] * (1-h[j]) → ∂/∂h[j] = -h[j-shift] if j-shift valid
        
        for j in range(n):
            g = 0.0
            if 0 <= j + shift < n:
                g += (1 - h[j + shift])
            if 0 <= j - shift < n:
                g -= h[j - shift]
            grad[j] = g
        
        # Gradient step (minimize g_{k_star})
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        diff = target_sum - np.sum(h_new)
        mask = (h_new > 1e-6) & (h_new < 1 - 1e-6)
        if np.sum(mask) > 0:
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0, 1)
        
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            # Decrease learning rate
            lr *= 0.999
        else:
            lr *= 0.95
            if lr < 1e-6:
                break
    
    return best_h, best_c5
