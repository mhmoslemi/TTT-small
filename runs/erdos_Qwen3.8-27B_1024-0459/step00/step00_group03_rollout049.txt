    return m + np.log(np.sum(np.exp(beta * (vals - m)))) / beta

def subgradient_optimize(h, dx, n, max_iter=5000, alpha0=0.01):
    """Subgradient descent with active set."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha0
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = corr * dx
        max_idx = np.argmax(vals)
        c5_val = vals[max_idx]
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Numerical subgradient of the active term
        # c = sum_i h[i] * (1-h[i+s]) for s determined by max_idx
        # Use central differences
        grad = np.zeros(n)
        eps = 1e-7
        for j in range(n):
            h[j] += eps
            c_plus = np.correlate(h, 1.0 - h, mode='full')[max_idx] * dx
            h[j] -= 2 * eps
            c_minus = np.correlate(h, 1.0 - h, mode='full')[max_idx] * dx
            h[j] += eps
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        h = h - alpha * grad
        h = project_h(h, n)
        
        # Adaptive step size
        curr_c5 = compute_c5(h, dx)
        if curr_c5 > best_c5 * 1.1:
            alpha *= 0.9
        elif (it % 100 == 0):
            alpha = max(alpha * 0.99, 0.001)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        nmask = int(np.sum(mask))
        if nmask == 0:
            break
        adj = diff / nmask
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def smoothed_c5(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = beta * (vals - m)
    # Clip for numerical safety
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def active_gradient(h, max_idx, n):
    """Compute gradient of the active correlation term."""
    s = max_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        # overlap = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
        for j in range(n):
            g = 0.0
            if j <= n - 1 - s:
                g += 1.0 - h[j + s]
            if j >= s and j - s <= n - 1 - s:
                g -= h[j - s]
            grad[j] = g
    else:
        # shift is negative, s = -|s|
        ss = -s
        # overlap = sum_{i=ss}^{n-1} h[i]*(1-h[i-ss])
        for j in range(n):
            g = 0.0
            if j >= ss and j <= n - 1:
                g += 1.0 - h[j - ss]
            if j - ss >= 0 and j - ss <= n - 1 and j >= 0:
                g -= h[j + ss]
            grad[j] = g
    return grad

        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

            lr *= 0.8
            patience = 0
            if lr < 1e-5:
                break
        
        grad = active_gradient(h, max_idx, n)
        h = h - lr * grad
        h = project_h(h, n)
        
        if it % 200 == 0:
            lr = max(lr * 0.97, 1e-6)
    
    return best_h, best_c5
