def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-14:
        diff = target - s
        if diff > 0:
            room = 1.0 - h
            total = np.sum(room)
            if total > 1e-14:
                h += diff * room / total
            else:
                # All at 1, can't increase
                pass
        else:
            room = h.copy()
            total = np.sum(room)
            if total > 1e-14:
                h += diff * room / total
            else:
                # All at 0, can't decrease
                pass
    h = np.clip(h, 0.0, 1.0)
    # Final correction: adjust the free variables
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        # Split among coordinates with room
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total = np.sum(room)
            if total > 1e-14:
                h = h + diff * room / total
            else:
                h = np.clip(h, 0, 1)
        else:
            room = h.copy()
            total = np.sum(room)
            if total > 1e-14:
                h = h + diff * room / total
            else:
                h = np.clip(h, 0, 1)
    h = np.clip(h, 0.0, 1.0)
    return h

def cd_sweep(h, n, candidates):
    for i in range(n):
        old_val = h[i]
        h[i] = old_val
        best_obj = evaluate_c5(h, n)
        best_val = old_val
        for v in candidates:
            if abs(v - old_val) < 1e-14:
                continue
            h[i] = v
            obj = evaluate_c5(h, n)
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_val = v
        h[i] = best_val
    return h

def multi_start_cd(h, n, candidates, n_sweeps, deadline):
    for _ in range(n_sweeps):
        if time.time() > deadline:
            break
        cd_sweep(h, n, candidates)
        h = project_feasible(h, n)
    return h
