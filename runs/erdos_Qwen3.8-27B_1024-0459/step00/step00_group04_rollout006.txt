    if max_lags <= 1:

        s = np.array([idx - len(cor)//2], dtype=int)

        return s,w,C
    idx_all = np.argsort(-cor)
    vals = cor[idx_all]
    tol = 1e-9 + 1e-5 * C
    keep = vals >= C - tol

def compute_our_cor_values(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.array([])
    dx = 2.0 / n
    one_h = 1.0 - h
    vals = np.empty(2*n-1, dtype=float)
    r0 = n-1
    vals[r0] = float(np.dot(h, one_h))
    for s in range(1, n):
        L = n - s
        vals[r0 + s] = float(np.dot(h[:L], one_h[s:]))
        vals[r0 - s] = float(np.dot(h[s:], one_h[:L]))
    return vals * dx

def active_lags(cor, max_lags=32):
    if cor.size == 0:
       return np.array([]), np.array([]), float('inf')
    C = float(np.max(cor))
    r0 = (cor.size - 1)//2
    if max_lags <= 1 or cor.size == 1:
        idx = int(np.argmax(cor))
        s = np.array([idx - r0], dtype=float)
        w = np.array([1.0])
        return s,w,C
    idx_all = np.argsort(-cor)
    vals = cor[idx_all]
    tol = 1e-10 + 1e-5*abs(C)
    keep = vals >= C - tol
    if not np.any(keep):
       idx = idx_all[:1]
    else:
       idx = idx_all[keep]
    if len(idx) > max_lags:
       idx = idx[:max_lags]
    s_vals = (idx.astype(int) - r0).astype(float)
    vals = cor[idx]
    temp = 1e-4
    w = np.exp((vals - C) / temp)
    sw = float(w.sum())
    if sw <= 0 or not np.isfinite(sw):
       w = np.ones(len(idx)) / len(idx)
    else:
       w /= sw
    return s_vals, w, C

import numpy as np
import math
import time


def make_full_from_p(p):
    p = np.asarray(p, dtype=float).ravel()
    m = len(p)
    h = np.empty(2 * m, dtype=float)
    h[:m] = p
    h[m:] = p[::-1]
    return h


def eval_full(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * dx)


def project_box_sum(z, target):
    z = np.asarray(z, dtype=float).ravel()
    n = z.size
    if n == 0:
        return z
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)
    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=1.0, neginf=0.0)
    z = np.clip(z, -1e100, 1e100)

    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    if hi <= lo:
        hi = lo + 1.0

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    x = np.clip(z - lam, 0.0, 1.0)

    for _ in range(20):
        diff = target - float(np.sum(x))
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            mask = x < 1.0 - 1e-15
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            add = diff / float(cnt)
            max_add = float(np.max(1.0 - x[mask]))
            if add <= max_add + 1e-15:
                x[mask] += add
            else:
                idx = int(np.argmax(1.0 - x))
                x[idx] = 1.0
        else:
            mask = x > 1e-15
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            sub = -diff / float(cnt)
            min_sub = float(np.min(x[mask]))
            if sub <= min_sub + 1e-15:
                x[mask] -= sub
            else:
                idx = int(np.argmin(x))
                x[idx] = 0.0

    diff = target - float(np.sum(x))
    if abs(diff) > 0.0:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - x))
            add = min(diff, 1.0 - x[idx])
            if add > 0.0:
                x[idx] += add
        else:
            idx = int(np.argmin(x))
            add = max(-x[idx], diff)
            if add < 0.0:
                x[idx] += add
    return x


def project_state(z, target, use_sym):
    q = project_box_sum(z, target)
    if use_sym:
        q = 0.5 * (q + q[::-1])
    return q


def repair_sym(h, target):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0 or (n % 2 != 0):
        return project_box_sum(h, target)
    m = n // 2
    p = 0.5 * (h[:m] + h[m:][::-1])
    p = project_box_sum(p, target * 0.5)
    return make_full_from_p(p)


def compute_our_cor_values(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.array([], dtype=float)
    dx = 2.0 / float(n)
    one_h = 1.0 - h
    vals = np.empty(2 * n - 1, dtype=float)
    r0 = n - 1
    vals[r0] = float(np.dot(h, one_h))
    for s in range(1, n):
        L = n - s
        vals[r0 + s] = float(np.dot(h[:L], one_h[s:]))
        vals[r0 - s] = float(np.dot(h[s:], one_h[:L]))
    return vals * dx


def active_lags(cor, max_lags=32):
    if cor.size == 0:
        return np.array([], dtype=float), np.array([], dtype=float), float("inf")
    C = float(np.max(cor))
    r0 = (cor.size - 1) // 2

    if max_lags <= 1 or cor.size == 1:
        idx = int(np.argmax(cor))
        s = np.array([idx - r0], dtype=float)
        w = np.array([1.0], dtype=float)
        return s, w, C

    idx_all = np.argsort(-cor)
    vals = cor[idx_all]
    tol = 1e-10 + 1e-5 * abs(C)
    keep = vals >= (C - tol)
    if not np.any(keep):
        idx = idx_all[:1]
    else:
        idx = idx_all[keep]

    if len(idx) > max_lags:
        idx = idx[:max_lags]

    s_vals = (idx.astype(int) - r0).astype(float)
    vals = cor[idx]
    temp = 1e-4
    w = np.exp((vals - C) / temp)
    sw = float(np.sum(w))
    if sw <= 0.0 or not math.isfinite(sw):
        w = np.ones(len(idx), dtype=float) / float(len(idx))
    else:
        w = w / sw
    return s_vals, w, C


def gradient_step(h, C_cur, alpha0, max_lags, use_sym, target):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return None, None

    cor = compute_our_cor_values(h)
    C = float(C_cur)
    Co = float(np.max(cor)) if cor.size else 0.0
    if Co > C:
        C = Co

    s_vals, w, _ = active_lags(cor, max_lags)
    if s_vals.size == 0:
        return None, None

    g = np.zeros(n, dtype=float)
    for sv, wt in zip(s_vals, w):
        s = int(round(float(sv)))
        if s >= 0:
            L = n - s
            if L <= 0:
                continue
            g[:L] += wt * (1.0 - h[s:])
            g[s:] -= wt * h[:L]
        else:
            m = -s
            L = n - m
            if L <= 0:
                continue
            g[m:] += wt * (1.0 - h[:L])
            g[:L] -= wt * h[m:]

    if use_sym:
        g = 0.5 * (g + g[::-1])

    if float(np.linalg.norm(g)) <= 1e-14:
        return None, None

    a = float(alpha0)
    for _ in range(18):
        if a <= 1e-8:
            break
        z = h - a * g
        q = project_state(z, target, use_sym)
        Cq = eval_full(q)
        if Cq < C - 1e-12:
            return q, Cq
        a *= 0.5
    return None, None


def propose_pair_move(h, use_sym, rng, step_scale, target):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n < 2:
        return None

    for _ in range(8):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        if rng.random() < 0.5:
            maxd = min(1.0 - h[i], h[j])
            if maxd <= 1e-15:
                continue
            frac = float(rng.uniform(0.0, min(1.0, float(step_scale))))
            d = maxd * frac
        else:
            maxd = min(h[i], 1.0 - h[j])
            if maxd <= 1e-15:
                continue
            frac = float(rng.uniform(0.0, min(1.0, float(step_scale))))
            d = -maxd * frac

        if abs(d) <= 1e-15:
            continue

        q = h.copy()
        q[i] += d
        q[j] -= d

        if (q[i] < -1e-12 or q[i] > 1.0 + 1e-12 or
                q[j] < -1e-12 or q[j] > 1.0 + 1e-12):
            q = np.clip(q, 0.0, 1.0)
            q = project_box_sum(q, target)
            if use_sym:
                q = 0.5 * (q + q[::-1])
        elif use_sym:
            q = 0.5 * (q + q[::-1])
        return q

    return None


def anneal_full(h_in, C_in, deadline, use_sym, rng):
    h = np.asarray(h_in, dtype=float).copy().ravel()
    n = h.size
    if n == 0:
        return h, float(C_in), h, float(C_in)

    target = 0.5 * float(n)
    C = float(C_in)
    best_h = h.copy()
    best_C = C

    T = 0.01
    step_scale = 0.30
    no_improve = 0
    window = 0
    accepted = 0

    while time.time() < deadline:
        q = propose_pair_move(h, use_sym, rng, step_scale, target)
        if q is None:
            break

        Cq = eval_full(q)
        if Cq <= best_C:
            best_h = q.copy()
            best_C = float(Cq)
            no_improve = 0
        else:
            no_improve += 1

        if Cq <= C:
            accept = True
        else:
            diff = Cq - C
            if diff <= 50.0 * T:
                ratio = diff / T
                if ratio >= 700.0:
                    accept = False
                else:
                    accept = bool(rng.random() < math.exp(-ratio))
            else:
                accept = False

        if accept:
            h = q
            C = float(Cq)
            accepted += 1

        window += 1
        if window >= 1000:
            rate = float(accepted) / float(window)
            if rate > 0.75:
                step_scale = min(0.80, step_scale * 1.10)
            elif rate < 0.15:
                step_scale = max(0.02, step_scale * 0.85)
            window = 0
            accepted = 0

        T *= 0.9999
        if T < 1e-5:
            T = 1e-5

        if no_improve > 10000:
            h = best_h.copy()
            C = best_C
            z = h + rng.normal(0.0, 0.05, n)
            q = project_state(z, target, use_sym)
            Cq = eval_full(q)
            if Cq <= best_C + 0.02:
                h = q
                C = float(Cq)
            no_improve = 0
            T = 0.05
            step_scale = 0.50

        if C > best_C + 0.05:
            h = best_h.copy()
            C = best_C
            no_improve = 0
            T = 0.03

    return best_h, best_C, h, C


def optimize_state(h_in, C_in, use_sym, deadline, rng):
    h = np.asarray(h_in, dtype=float).copy().ravel()
    if h.size == 0:
        return h, float(C_in)

    C = float(C_in)
    best_h = h.copy()
    best_C = C
    n = h.size
    target = 0.5 * float(n)
    grad_alpha = 0.05
    fail_count = 0

    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        g_dur = min(20.0, max(2.0, 0.15 * remaining))
        g_dead = min(deadline, time.time() + g_dur)

        if g_dead > time.time():
            while time.time() < g_dead:
                improved = False
                for ml in (1, 4, 16, 32):
                    q, Cq = gradient_step(h, C, grad_alpha, ml, use_sym, target)
                    if q is not None:
                        h = q
                        C = float(Cq)
                        if C < best_C - 1e-12:
                            best_h = h.copy()
                            best_C = C
                        grad_alpha = min(0.20, grad_alpha * 1.05)
                        fail_count = 0
                        improved = True
                        break
                if not improved:
                    fail_count += 1
                    grad_alpha = max(0.004, grad_alpha * 0.75)
                    if fail_count >= 6 or grad_alpha <= 0.005:
                        break

        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        if remaining <= 0.0:
            break

        a_dur = min(180.0, max(5.0, 0.35 * remaining))
        a_dead = min(deadline, time.time() + a_dur)

        if a_dead > time.time():
            bh, bc, h, C = anneal_full(h, C, a_dead, use_sym, rng)
            if bc < best_C - 1e-12:
                best_h = np.asarray(bh, dtype=float).copy()
                best_C = float(bc)
            if C > best_C + 0.05:
                h = best_h.copy()
                C = best_C
            grad_alpha = max(0.03, min(grad_alpha * 0.9, 0.10))

    return best_h, best_C


def upsample_full(h, N, use_sym=True):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if N == n:
        return h.copy()
    if n == 0 or N <= 0:
        return h.copy()

    if N % n == 0:
        h2 = np.repeat(h, N // n)
    else:
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, N)
        h2 = np.interp(x_new, x_old, h)

    if use_sym and (N % 2 == 0):
        return repair_sym(h2, float(N) / 2.0)
    return project_box_sum(h2, float(N) / 2.0)

    return make_full_from_p(p)


def feasible_full(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return False
    target = 0.5 * float(n)
    if np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return False
    return abs(float(np.sum(h)) - target) <= 1e-9


def finalize_state(h, use_sym):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return h, float("inf")
    target = 0.5 * float(n)
    candidates = []

    if feasible_full(h):
        candidates.append(h.copy())
    if use_sym and (n % 2 == 0):
        candidates.append(repair_sym(h, target))
    candidates.append(project_box_sum(h, target))

    best = None
    best_c = float("inf")
    for q in candidates:
        cq = eval_full(q)
        if cq < best_c:
            best_c = float(cq)
            best = q
    if best is None:
        best = project_box_sum(h, target)
        best_c = eval_full(best)
    return best, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    total = float(budget_s) if budget_s is not None else 1300.0
    if total <= 0.0:
        total = 1300.0

    margin = min(20.0, 0.03 * total) if total > 10.0 else 0.0
    deadline = start + total - margin
    if deadline <= start:
        deadline = start + total

    rng = np.random.default_rng(seed)

    init = globals().get("initial_h_values")
    if init is not None:
        try:
            h0 = np.asarray(init, dtype=float).copy().ravel()
        except Exception:
            h0 = fallback_initial()
    else:
        h0 = fallback_initial()

    if h0.size == 0:
        h0 = fallback_initial()

    n0 = h0.size
    target0 = 0.5 * float(n0)

    h_full = project_box_sum(h0, target0)
    C_full = eval_full(h_full)

    best_h = h_full.copy()
    best_C = float(C_full)
    best_sym = bool(n0 % 2 == 0 and np.max(np.abs(h_full - h_full[::-1])) < 1e-7)

    h_sym = None
    C_sym = float("inf")
    if n0 % 2 == 0:
        h_sym = repair_sym(h_full, target0)
        C_sym = eval_full(h_sym)
        if C_sym < C_full - 1e-12:
            best_h = h_sym.copy()
            best_C = float(C_sym)
            best_sym = True

    if n0 % 2 == 0:
        main_h = h_sym
        main_C = float(C_sym)
        main_sym = True
    else:
        main_h = h_full
        main_C = float(C_full)
        main_sym = False

    rem = deadline - time.time()
    if rem > 20.0:
        dur = min(rem, max(5.0, 0.45 * rem))
        dl = min(deadline, time.time() + dur)
        mh, mC = optimize_state(main_h, main_C, main_sym, dl, rng)
        if mC < best_C - 1e-12:
            best_h = np.asarray(mh, dtype=float).copy()
            best_C = float(mC)
            best_sym = bool(main_sym)

    rem = deadline - time.time()
    if n0 % 2 == 0 and rem > 20.0 and C_full <= best_C + 0.02:
        dur = min(rem, max(5.0, 0.12 * rem))
        dl = min(deadline, time.time() + dur)
        fh, fC = optimize_state(h_full, C_full, False, dl, rng)
        if fC < best_C - 1e-12:
            best_h = np.asarray(fh, dtype=float).copy()
            best_C = float(fC)
            best_sym = False

    base_n = int(n0)
    cap_n = int(min(800, max(336, base_n * 8)))
    next_n = 2 * base_n
    refinements = 0

    while next_n <= cap_n and next_n > int(best_h.size) and refinements < 4:
        rem = deadline - time.time()
        if rem <= 20.0:
            break
        hup = upsample_full(best_h, next_n, True)
        Cup = eval_full(hup)
        if Cup <= best_C + 0.03:
            dur = min(rem, max(5.0, 0.35 * rem))
            dl = min(deadline, time.time() + dur)
            uh, uC = optimize_state(hup, Cup, True, dl, rng)
            if uC < best_C - 1e-12:
                best_h = np.asarray(uh, dtype=float).copy()
                best_C = float(uC)
                best_sym = True
        next_n *= 2
        refinements += 1

    rem = deadline - time.time()
    if rem > 20.0:
        dur = min(rem, max(5.0, 0.50 * rem))
        dl = min(deadline, time.time() + dur)
        ph, pC = optimize_state(best_h, best_C, best_sym, dl, rng)
        if pC < best_C - 1e-12:
            best_h = np.asarray(ph, dtype=float).copy()
            best_C = float(pC)

    final_h, final_C = finalize_state(best_h, best_sym)
    return final_h, float(final_C), int(final_h.size)
