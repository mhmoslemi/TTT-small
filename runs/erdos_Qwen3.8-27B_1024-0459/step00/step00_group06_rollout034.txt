_n_global = 100
_beta_global = 30.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def obj_raw(x):
    n = _n_global
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx

    n = _n_global
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _beta_global
    exps = np.exp(beta * (corr - m))
    return (m + np.log(np.sum(exps)) / beta) * dx

def con_eq(x):
    return np.sum(x) - _n_global / 2.0

def project_onto_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

    h = np.clip(h, 0.0, 1.0)
