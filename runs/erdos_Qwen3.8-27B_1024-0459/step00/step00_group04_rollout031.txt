                    obj_func, h_opt,
                    jac=obj_grad,
                    args=(T2, lam2, n_target),
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n,
                    options={
                        'maxiter': 150,
                        'ftol': 1e-12,
                        'gtol': 1e-7,
                        'maxfun': 500,
                        'iprint': -1
                    }
                )
                h_opt2 = np.clip(result2.x, 0.0, 1.0)
                h_opt2 = project_sum(h_opt2)
                c5_2 = compute_c5(h_opt2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h_opt2.copy()
                    best_n = n
                    print(f"  Refined at n={n}: C5={best_c5:.6f}")
            except Exception:
                pass

        print(f"  n={n}: best C5={best_c5:.6f}, elapsed={time.time()-t_start:.1f}s")

    # Random restarts for diversity
    remaining = budget_s - (time.time() - t_start)
    if remaining > 80:
        n_try = best_n
        n_target = n_try // 2
        n_trials = int(remaining / 40)
        n_trials = max(1, min(n_trials, 6))

        for trial in range(n_trials):
            if time.time() - t_start > budget_s - 40:
                break

            # Different random perturbations
            if trial % 3 == 0:
                h_pert = best_h + 0.15 * np.random.randn(n_try)
            elif trial % 3 == 1:
                h_pert = np.clip(1.0 - best_h, 0.0, 1.0)
                h_pert += 0.05 * np.random.randn(n_try)
            else:
                h_pert = best_h.copy()
                idx = np.random.choice(n_try, size=n_try // 3, replace=False)
                h_pert[idx] = np.random.uniform(0, 1, size=len(idx))

            h_pert = np.clip(h_pert, 0.0, 1.0)
            h_pert = project_sum(h_pert)

            T = 40.0
            lam = 400.0
            try:
                result = minimize(
                    obj_func, h_pert,
                    jac=obj_grad,
                    args=(T, lam, n_target),
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0)] * n_try,
                    options={
                        'maxiter': 200,
                        'ftol': 1e-12,
                        'gtol': 1e-6,
                        'maxfun': 600,
                        'iprint': -1
                    }
                )
                h_r = np.clip(result.x, 0.0, 1.0)
                h_r = project_sum(h_r)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_n = n_try
                    print(f"  Trial {trial}: C5={best_c5:.6f}")
            except Exception:
                pass

    # Final projection and evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h)
    best_c5 = compute_c5(best_h)

    print(f"Final: n={best_n}, C5={best_c5:.6f}, time={time.time()-t_start:.1f}s")
    return best_h, best_c5, best_n
