def obj_and_grad(h, n, t, pen_lambda):
    """Compute smooth objective and its gradient."""
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1-h, mode="full")
    vals = corr * dx
    m = np.max(vals)
    
    scaled = t * (vals - m)
    if np.max(scaled) - np.min(scaled) > 700:
        # Protect against overflow
        exp_scaled = np.exp(scaled - np.max(scaled))
        logsumexp = np.max(scaled) + np.log(np.sum(exp_scaled))
        p = exp_scaled / np.sum(exp_scaled)
    else:
        exp_scaled = np.exp(scaled)
        sum_exp = np.sum(exp_scaled)
        p = exp_scaled / sum_exp
        logsumexp = np.log(sum_exp)
    
    f_obj = m + logsumexp / t + pen_lambda * (np.sum(h) - target)**2
    
    # Gradient
    grad = np.zeros(n)
    # p is indexed by k_full in [0, 2n-2], shift = k_full - (n-1)

        w = p[k_full]
        if w < 1e-12:

    target = n / 2.0

        h = np.clip(h, 0, 1)
        new_diff = target - np.sum(h)
        if abs(new_diff) < 1e-12:
            break
        diff = new_diff
    return h
