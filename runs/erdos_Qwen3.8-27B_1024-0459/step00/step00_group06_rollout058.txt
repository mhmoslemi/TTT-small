    x_old = (np.arange(n_old) + 0.5) / float(n_old)

    corr = np.correlate(h, 1.0 - h, mode="full")

        gm2 = np.zeros(n)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / float(n_old)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new)

def smooth_max_and_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    vmax = np.max(vals)
    shifted = T * (vals - vmax)
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    s = np.sum(exps)
    smooth_val = vmax + (max_shifted + np.log(s)) / T
    weights = exps / s
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for k in range(n_full):
        w = weights[k]
        if w < 1e-9:
            continue
        wdx = w * dx
        lag = k - (n - 1)
        if lag == 0:
            grad += wdx * (1.0 - 2.0 * h)
        elif lag > 0:
            sa = lag
            grad[sa:n] += wdx * (1.0 - h[0:n - sa])

            sa = -lag
            grad[0:n - sa] += wdx * (1.0 - h[sa:n])
            grad[sa:n] += wdx * (-h[0:n - sa])
    return smooth_val, grad

def random_feasible(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = rng.random(n)
    h = np.where(h > 0.3, 1.0, 0.0)
    return project_feasible(h, n)

def structured_init(n, mode, seed_val):
    rng = np.random.RandomState(seed_val)
    if mode == 0:
        h = np.tile([1.0, 0.0], n // 2)
        if n % 2 == 1:
            h = np.append(h, 0.0)

        h = np.where(h > 0.5, 1.0, 0.0)

def try_slsqp(h, n, time_budget):
    start = time.time()
    def obj(hv):
        if time.time() - start > time_budget:
            return 1e6
        return compute_c5(hv)
    cons = [{'type': 'eq', 'fun': lambda hv: np.sum(hv) - n / 2.0,
             'jac': lambda hv: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-8})
        if time.time() - start > time_budget:
            return h, compute_c5(h)
        h_opt = project_feasible(res.x, n)
        return h_opt, compute_c5(h_opt)
    except:
        return h, compute_c5(h)
