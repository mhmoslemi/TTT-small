        return z
    target = float(target)

    n = z.size
    if target <= 0.0:
       np.clip(z,0,1)?? target 0 -> zeros
    if target >= n:
       all ones
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5*(lo+hi)
        vals = np.clip(z - mid, 0.0, 1.0)
        s = float(np.sum(vals))
        if s > target:
            lo = mid
        else:
            hi = mid
    vals = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)

    for _ in range(20):

        if abs(diff) < 1e-14:

_GRAD_REVERSE = False
_GRAD_SIGN = 1.0
_TAU = 0.05
_PENALTY = 0.0

_CURRENT_H = None
_CURRENT_G = None

_FOURIER_N = 0
_FOURIER_COS = None

_BLOCK_N = 0
_BLOCK_COUNT = 0
_BLOCK_LEN = 1


def project_to_simplex(z, target):
    z = np.asarray(z, dtype=float).copy()
    if z.size == 0:
        return z
    target = float(target)
    z[~np.isfinite(z)] = 0.5
    n = z.size
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        vals = np.clip(z - mid, 0.0, 1.0)
        s = float(np.sum(vals))
        if s > target:
            lo = mid
        else:
            hi = mid

    vals = np.clip(z - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(25):
        diff = target - float(np.sum(vals))
        if abs(diff) < 1e-14:
            break
        mask = (vals > 1e-12) & (vals < 1.0 - 1e-12)
        idx = np.nonzero(mask)[0]
        if idx.size > 0:
            slack = np.minimum(vals[idx], 1.0 - vals[idx])
            tot = float(np.sum(slack))
            if tot > 1e-15:
                vals[idx] += diff * (slack / tot)
            else:
                vals[idx[0]] += diff
        else:
            vals = vals + diff / float(n)
        vals = np.clip(vals, 0.0, 1.0)
    return vals


def c5_values(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.array([], dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))


def c5_value(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return float("inf")
    c = c5_values(h)
    if not np.all(np.isfinite(c)):
        return float("inf")
    return float(np.max(c))


def symmetrize_h(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return h.copy()
    hs = 0.5 * (h + h[::-1])
    return project_to_simplex(hs, h.size / 2.0)


def weighted_grad_with(W, h):
    n = h.size
    if n == 0:
        return np.zeros(0, dtype=float)
    off = n - 1
    W = np.asarray(W, dtype=float)
    g = 1.0 - h
    conv_g = np.convolve(W, g, mode="full")[off:off + n]
    Wneg = W[::-1]
    conv_h = np.convolve(Wneg, h, mode="full")[off:off + n]
    return (conv_g - conv_h) * (2.0 / float(n))


def weighted_grad(h, w):
    if _GRAD_REVERSE:
        W = w[::-1]
    else:
        W = w
    return _GRAD_SIGN * weighted_grad_with(W, h)


def weighted_correlate_sum(h, w):
    c = c5_values(h)
    if not np.all(np.isfinite(c)):
        return float("inf")
    return float(np.dot(w, c))


def calibrate_gradient():
    global _GRAD_REVERSE, _GRAD_SIGN
    n = 16
    rng = np.random.RandomState(123456)
    h = 0.20 + 0.50 * rng.rand(n)
    c = c5_values(h)
    tau = 0.10
    z = c - float(np.max(c))
    w = np.exp(z / tau)
    w = w / float(np.sum(w))

    eps = 1e-6
    fd = np.zeros(n, dtype=float)
    for i in range(n):
        hp = h.copy()
        hp[i] += eps
        fp = weighted_correlate_sum(hp, w)
        hp[i] -= 2.0 * eps
        fm = weighted_correlate_sum(hp, w)
        hp[i] += eps
        fd[i] = (fp - fm) / (2.0 * eps)

    best_err = float("inf")
    best_rev = False
    best_sign = 1.0
    for rev in (False, True):
        W = w[::-1] if rev else w
        G = weighted_grad_with(W, h)
        for sign in (1.0, -1.0):
            err = float(np.linalg.norm(G * sign - fd))
            if err < best_err:
                best_err = err
                best_rev = rev
                best_sign = sign

    _GRAD_REVERSE = best_rev
    _GRAD_SIGN = best_sign


def soft_fun_jac(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return (0.0, np.zeros(0, dtype=float))

    tau = float(_TAU)
    c = c5_values(h)
    if not np.all(np.isfinite(c)):
        return (1e12, np.zeros(n, dtype=float))

    m = float(np.max(c))
    z = c - m
    w = np.exp(z / tau)
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros_like(c)
        ww = np.nonzero(c == m)[0]
        if ww.size:
            w[ww[0]] = 1.0
            s = 1.0
    f_soft = m + tau * float(np.log(s))
    w = w / s
    g_soft = weighted_grad(h, w)

    f = f_soft
    g = g_soft.copy()
    if _PENALTY != 0.0:
        err = float(np.mean(h) - 0.5)
        f += 0.5 * float(_PENALTY) * err * err
        g += (float(_PENALTY) / float(n)) * err * np.ones(n, dtype=float)
    return (float(f), g)


def soft_fun(h):
    global _CURRENT_H, _CURRENT_G
    h = np.asarray(h, dtype=float)
    f, g = soft_fun_jac(h)
    _CURRENT_H = h.copy()
    _CURRENT_G = g
    return f


def soft_jac(h):
    global _CURRENT_H, _CURRENT_G
    h = np.asarray(h, dtype=float)
    if _CURRENT_H is not None and h.shape == _CURRENT_H.shape:
        if np.array_equal(_CURRENT_H, h):
            return _CURRENT_G.copy()
    _, g = soft_fun_jac(h)
    _CURRENT_H = h.copy()
    _CURRENT_G = g
    return g.copy()


def lbfgs_local(h0, tau, maxfun, penalty):
    h0 = project_to_simplex(np.asarray(h0, dtype=float), h0.size / 2.0)
    global _TAU, _PENALTY
    _TAU = float(tau)
    _PENALTY = float(penalty)

    n = h0.size
    if n == 0 or int(maxfun) <= 5:
        return h0.copy()

    bounds = [(0.0, 1.0) for _ in range(n)]
    try:
        res = minimize(
            soft_fun,
            h0,
            method="L-BFGS-B",
            jac=soft_jac,
            bounds=bounds,
            options={
                "maxfun": int(maxfun),
                "maxiter": int(maxfun),
                "ftol": 1e-12,
                "gtol": 1e-10,
                "maxls": 20,
                "disp": False,
            },
        )
        x = res.x
        if x is not None and np.all(np.isfinite(x)):
            return project_to_simplex(x, n / 2.0)
    except Exception:
        pass
    return h0.copy()


def refine_h(h0, time_limit, seed=42):
    h0 = project_to_simplex(np.asarray(h0, dtype=float), h0.size / 2.0)
    if h0.size == 0:
        return h0, float("inf"), 0

    hs = symmetrize_h(h0)
    c0 = c5_value(h0)
    cs = c5_value(hs)
    if cs < c0:
        h0 = hs
        c0 = cs

    best_h = h0.copy()
    best_c = c0
    n = best_h.size

    if time_limit <= 0.0 or n == 0:
        return best_h, best_c, n

    t0 = time.time()
    if n <= 120:
        base = 900
    elif n <= 240:
        base = 500
    else:
        base = 240

    taus = [0.30, 0.15, 0.075, 0.04, 0.02, 0.01]
    for tau in taus:
        if time.time() - t0 > time_limit * 0.78:
            break
        h2 = lbfgs_local(best_h, tau, base, 10000.0)
        c2 = c5_value(h2)
        if c2 < best_c - 1e-13:
            hs2 = symmetrize_h(h2)
            c2s = c5_value(hs2)
            if c2s < c2 - 1e-13:
                best_h = hs2
                best_c = c2s
            else:
                best_h = h2
                best_c = c2

    if time.time() - t0 < time_limit * 0.80:
        h2 = lbfgs_local(best_h, 0.008, max(60, base // 3), 10000.0)
        c2 = c5_value(h2)
        if c2 < best_c - 1e-13:
            best_h = h2
            best_c = c2

    rng = np.random.RandomState(int(seed) % (2**31 - 1))
    alphas = [0.02, 0.05, 0.10, 0.15]
    for alpha in alphas:
        if time.time() - t0 > time_limit * 0.97:
            break
        h2 = project_to_simplex(best_h + alpha * rng.normal(size=n), n / 2.0)
        c2 = c5_value(h2)
        if c2 < best_c + 0.02:
            h3 = lbfgs_local(h2, 0.05, max(60, base // 3), 10000.0)
            c3 = c5_value(h3)
            if c3 < best_c - 1e-13:
                best_h = h3
                best_c = c3

    best_h = project_to_simplex(best_h, n / 2.0)
    final_c = c5_value(best_h)
    if np.isfinite(final_c):
        best_c = final_c
    return best_h, best_c, n


def setup_fourier(n, center=True):
    global _FOURIER_N, _FOURIER_COS
    n = int(n)
    if n <= 0:
        _FOURIER_N = 1
        _FOURIER_COS = np.ones((1, _MAX_MODES))
        return
    if center:
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
    else:
        x = np.linspace(0.0, 2.0, n, endpoint=False)
    cosx = np.empty((n, _MAX_MODES), dtype=float)
    for m in range(1, _MAX_MODES + 1):
        cosx[:, m - 1] = np.cos(np.pi * m * x)
    _FOURIER_N = n
    _FOURIER_COS = cosx


def fourier_obj(params):
    if _FOURIER_COS is None:
        return 1e6
    p = np.asarray(params, dtype=float)
    if p.size < _MAX_MODES:
        pp = np.zeros(_MAX_MODES, dtype=float)
        pp[:p.size] = p
        p = pp
    else:
        p = p[:_MAX_MODES].copy()
    if not np.all(np.isfinite(p)):
        return 1e6
    h = 0.5 + _FOURIER_COS.dot(p)
    h = project_to_simplex(h, _FOURIER_N / 2.0)
    c = c5_value(h)
    if not np.isfinite(c):
        return 1e6
    return float(c)


def run_fourier_search(n, seed, time_limit):
    setup_fourier(int(n), True)
    t0 = time.time()
    rng = np.random.RandomState(int(seed) % (2**31 - 1))
    best_h = None
    best_c = float("inf")
    best_n = int(n)

    starts = [
        [0.30, 0.00, 0.00, 0.00],
        [0.30, 0.15, 0.05, 0.00],
        [0.25, 0.10, -0.08, 0.00],
        [0.35, -0.10, 0.08, 0.05],
        [0.20, 0.20, 0.00, -0.10],
        [0.25, -0.15, 0.15, 0.00],
    ]
    for _ in range(3):
        s = [0.18 + 0.25 * rng.rand() for _ in range(4)]
        s[0] = abs(s[0])
        starts.append(s)

    for p0 in starts:
        if time.time() - t0 > time_limit * 0.92:
            break
        p0 = np.asarray(p0, dtype=float)
        p0 = p0 + 0.015 * rng.normal(_MAX_MODES)
        try:
            res = minimize(
                fourier_obj,
                p0,
                method="Nelder-Mead",
                options={
                    "maxiter": 160,
                    "xatol": 1e-5,
                    "fatol": 1e-8,
                    "disp": False,
                },
            )
            p = res.x
            if p is None or not np.all(np.isfinite(p)):
                p = p0
        except Exception:
            p = p0

        p = np.asarray(p, dtype=float)
        if p.size < _MAX_MODES:
            pp = np.zeros(_MAX_MODES, dtype=float)
            pp[:p.size] = p
            p = pp
        else:
            p = p[:_MAX_MODES]

        h = 0.5 + _FOURIER_COS.dot(p)
        h = project_to_simplex(h, _FOURIER_N / 2.0)
        hs = symmetrize_h(h)
        c = c5_value(h)
        cs = c5_value(hs)
        if cs < c:
            h = hs
            c = cs
        if np.isfinite(c) and c < best_c:
            best_h = h
            best_c = c

    if best_h is not None:
        return best_h, best_c, best_n
    return None, float("inf"), best_n


def setup_block(n, blocks):
    global _BLOCK_N, _BLOCK_COUNT, _BLOCK_LEN
    _BLOCK_N = int(n)
    _BLOCK_COUNT = int(blocks)
    _BLOCK_LEN = max(1, int(n // blocks))


def block_obj(vals):
    if _BLOCK_N <= 0 or _BLOCK_COUNT <= 0:
        return 1e6
    vals = np.asarray(vals, dtype=float)
    vals = project_to_simplex(vals, _BLOCK_COUNT / 2.0)
    h = np.repeat(vals, _BLOCK_LEN)
    if h.size != _BLOCK_N:
        h = h[:_BLOCK_N]
    h = project_to_simplex(h, _BLOCK_N / 2.0)
    c = c5_value(h)
    if not np.isfinite(c):
        return 1e6
    return float(c)


def run_block_search(n, blocks, seed, time_limit, maxiter=25, popsize=15):
    n = int(n)
    blocks = int(blocks)
    best_n = n
    if n <= 0 or blocks <= 0 or (n % blocks) != 0:
        return None, float("inf"), best_n
    if time_limit <= 10.0:
        return None, float("inf"), best_n

    setup_block(n, blocks)
    bounds = [(0.0, 1.0) for _ in range(blocks)]
    best_h = None
    best_c = float("inf")

    try:
        res = differential_evolution(
            block_obj,
            bounds,
            seed=int(seed) % (2**31 - 1),
            maxiter=int(maxiter),
            popsize=int(popsize),
            tol=1e-6,
            mutation=(0.5, 1.5),
            recombine=0.7,
            init="latinhypercube",
            polish=False,
        )
        vals = res.x
        if vals is not None:
            vals = project_to_simplex(vals, blocks / 2.0)
            h = np.repeat(vals, n // blocks)
            h = project_to_simplex(h, n / 2.0)
            hs = symmetrize_h(h)
            c = c5_value(h)
            cs = c5_value(hs)
            if cs < c:
                h = hs
                c = cs
            if np.isfinite(c):
                best_h = h
                best_c = c
    except Exception:
        pass

    if best_h is not None:
        return best_h, best_c, best_n
    return None, float("inf"), best_n


def resample_h(h_old, n):
    h_old = np.asarray(h_old, dtype=float)
    n_old = h_old.size
    n = int(n)
    if n <= 0:
        return h_old.copy()
    if n_old == n:
        return project_to_simplex(h_old, n / 2.0)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n) + 0.5) * (2.0 / float(n))
    hnew = np.interp(x_new, x_old, h_old)
    return project_to_simplex(hnew, n / 2.0)


def make_simple_h(n, seed=0):
    n = int(n)
    if n <= 0:
        n = 95
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 0.5
    h += 0.28 * np.cos(np.pi * x)
    h += 0.12 * np.cos(2.0 * np.pi * x)
    h -= 0.05 * np.cos(3.0 * np.pi * x)
    h = project_to_simplex(h, n / 2.0)
    hs = symmetrize_h(h)
    if c5_value(hs) < c5_value(h):
        h = hs
    return h


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 1300.0
    if total_budget <= 0.0:
        total_budget = 10.0

    end_time = t0 + max(5.0, total_budget - 15.0)
    seed_int = int(seed) % (10**6)

    calibrate_gradient()

    best_h = None
    best_c = float("inf")
    best_n = 0

    def update_candidate(h, c, n):
        nonlocal best_h, best_c, best_n
        if h is None or not np.isfinite(c):
            return
        h = np.asarray(h, dtype=float)
        h = project_to_simplex(h, h.size / 2.0)
        hs = symmetrize_h(h)
        cs = c5_value(hs)
        ch = c5_value(h)
        if cs < ch - 1e-13:
            h = hs
            c = cs
        else:
            c = ch
        if np.isfinite(c) and c < best_c - 1e-15:
            best_h = h
            best_c = c
            best_n = int(h.size)

    parent = globals().get("initial_h_values", None)
    if parent is not None:
        h = np.asarray(parent, dtype=float)
        if h.ndim != 1:
            h = h.ravel()
        if h.size > 0:
            h = project_to_simplex(h, h.size / 2.0)
            c = c5_value(h)
            update_candidate(h, c, h.size)

    if best_h is None:
        h = make_simple_h(95, seed_int)
        c = c5_value(h)
        update_candidate(h, c, h.size)

    if best_h is not None:
        rem = end_time - time.time()
        if rem > 80.0:
            tl = min(300.0, rem * 0.45)
            h2, c2, n2 = refine_h(best_h, tl, seed_int)
            update_candidate(h2, c2, n2)

    rem = end_time - time.time()
    if rem > 220.0:
        tl = min(180.0, rem * 0.30)
        hf, cf, nf = run_fourier_search(120, seed_int + 1, tl)
        update_candidate(hf, cf, nf)
        if hf is not None and cf < best_c + 0.02:
            rem = end_time - time.time()
            if rem > 120.0:
                h2, c2, n2 = refine_h(hf, min(120.0, rem * 0.25), seed_int + 2)
                update_candidate(h2, c2, n2)

    rem = end_time - time.time()
    if rem > 260.0:
        tl = min(240.0, rem * 0.30)
        hb, cb, nb = run_block_search(160, 16, seed_int + 3, tl, maxiter=22, popsize=15)
        update_candidate(hb, cb, nb)
        if hb is not None and cb < best_c + 0.02:
            rem = end_time - time.time()
            if rem > 120.0:
                h2, c2, n2 = refine_h(hb, min(120.0, rem * 0.25), seed_int + 4)
                update_candidate(h2, c2, n2)

    rem = end_time - time.time()
    if best_h is not None and rem > 180.0:
        n_cur = int(best_h.size)
        n_next = int(round(n_cur * 1.5))
        if n_next % 2 != 0:
            n_next += 1
        if n_next > n_cur and n_next <= 360:
            h3 = resample_h(best_h, n_next)
            c3 = c5_value(h3)
            if c3 < best_c + 0.02:
                h4, c4, n4 = refine_h(h3, min(180.0, rem * 0.35), seed_int + 5)
                update_candidate(h4, c4, n4)

    rem = end_time - time.time()
    if best_h is not None and rem > 40.0:
        n = int(best_h.size)
        if n <= 120:
            base = 900
        elif n <= 240:
            base = 500
        else:
            base = 240
        h5 = lbfgs_local(best_h, 0.02, max(60, base // 3), 10000.0)
        c5 = c5_value(h5)
        update_candidate(h5, c5, h5.size)

    if best_h is None or not np.isfinite(best_c):
        best_h = make_simple_h(95, seed_int)
        best_c = c5_value(best_h)
        best_n = int(best_h.size)

    best_h = project_to_simplex(np.asarray(best_h, dtype=float), int(best_n) / 2.0)
    hs = symmetrize_h(best_h)
    cs = c5_value(hs)
    cb = c5_value(best_h)
    if cs < cb - 1e-13:
        best_h = hs
        best_c = cs
    else:
        best_c = cb

    if not np.isfinite(best_c):
        best_c = 1.0
    return best_h, float(best_c), int(best_h.size)
