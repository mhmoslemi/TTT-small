def grad_weighted(f,w,h):
   n=len(h); L=2*n-1
   G=np.zeros(n)
   # To avoid zero-weight loops, get indices where w > tol
   tol=1e-12
   idxs=np.nonzero(w>tol)[0]
   # ensure sum weights not zero
   for idx in idxs:
      k = idx - (n-1)
      val = w[idx]
      if val <= tol: continue
      if k >=0:
         r = n - k
         G[k:n] += val * (1.0 - h[0:r])
         G[0:r] -= val * h[k:n]
      else:
         q = -k
         r = n - q
         G[0:r] += val * (1.0 - h[q:n])
         G[q:n] -= val * h[0:r]
   return G

def project_feasible(z):

    if z.size == 0:
        return z
    np.clip(z, 0.0, 1.0, out=z)
    target = 0.5 * z.size
    s = float(np.sum(z))
    if abs(s - target) <= 1e-14:
        return z
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z))
    # if hi-lo small
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.clip(z - mid, 0.0, 1.0)))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    out = np.clip(z - 0.5*(lo+hi), 0.0, 1.0)
    # fix residual
    for _ in range(20):
        diff = target - float(np.sum(out))
        if abs(diff) <= 1e-12:
            break
        interior = (out > 1e-13) & (out < 1.0 - 1e-13)
        if not np.any(interior):
            # fallback distribute to all if possible? If all at bounds, this shouldn't happen unless target infeasible.
            allm = np.ones_like(out, dtype=bool)
        else:
            allm = interior
        cnt = int(np.count_nonzero(allm))
        if cnt == 0:
            break
        step = diff / cnt
        # Clipping after adding may clip some; loop
        out[allm] += step
        np.clip(out, 0.0, 1.0, out=out)
    # Final exact: if still off, adjust one interior greedily
    diff = target - float(np.sum(out))
    if abs(diff) > 1e-13:
        # Try all indices with room
        for idx in range(z.size):

            if diff > 0:

                if step > 0:
                    out[idx] += step
                    diff -= step

                room = out[idx]
                step = max(diff, -room)
                if step < 0:
                    out[idx] += step

def project_feasible(z):

def eval_full(h, fft_threshold=None):
    if fft_threshold is None:
        fft_threshold = FFT_THRESHOLD
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.zeros(0, dtype=float)
    if n < fft_threshold:
        return np.correlate(1.0 - h, h, mode='full').astype(float, copy=False)

    m = 1 << (2 * n - 1).bit_length()
    a = 1.0 - h
    F = np.fft.rfft(a, m)
    G = np.fft.rfft(h, m)

def grad_weighted(w, h):
    n = len(h)
    G = np.zeros(n, dtype=float)
    if n == 0:
        return G

    w = np.asarray(w, dtype=float)
    idxs = np.nonzero(w > 1e-14)[0]
    if idxs.size > 256:
        idxs = np.argpartition(w, -256)[-256:]

    for idx in idxs:
        val = float(w[idx])
        if val <= 1e-14:
            continue
        k = int(idx - (n - 1))
        if k >= 0:
            r = n - k
            if r > 0:
                G[k:n] += val * (1.0 - h[0:r])
                G[0:r] -= val * h[k:n]
        else:

            r = n - q
            if r > 0:
                G[0:r] += val * (1.0 - h[q:n])
                G[q:n] -= val * h[0:r]
    return G


def alpha_max_for_direction(h, d):

    return amax

def zero_sum_noise(rng, n, sigma):

        return np.zeros(n, dtype=float)

        norm = float(np.linalg.norm(v))
        if norm > 1e-14:
            return sigma * math.sqrt(float(n)) * v / norm
    return np.zeros(n, dtype=float)


def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    if n <= 0:
        return np.zeros(0, dtype=float)
    if n % 2 != 0:
        return np.full(n, 0.5, dtype=float)
    x = np.zeros(n, dtype=float)

    rng.shuffle(x)
    return x


def nearest_binary(h, target=None):

        return np.ones(n, dtype=float)
    idx = np.argpartition(h, -target)[-target:]
    x = np.zeros(n, dtype=float)
    x[idx] = 1.0
    return x


def upsample(h, n):

    old = h.size
    if n <= 0:

def triangular_start(n):

def harmonic_start(n, seed):

        return np.zeros(0, dtype=float)
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.5, dtype=float)

def binary_lists(x):
    ones = list(np.nonzero(x)[0].astype(int))
    zeros = list(np.nonzero(1 - x)[0].astype(int))
    return ones, zeros


def eval_binary_points(x):
    x = np.asarray(x, dtype=np.int16)
    if x.size == 0:
        return 0.0
    y = (1 - x).astype(np.int16, copy=False)
    f = np.correlate(y, x, mode='full')
    return float(np.max(f))


def binary_apply_random_swaps(x, ones, zeros, rng, count):
    for _ in range(count):
        if len(ones) == 0 or len(zeros) == 0:
            break
        ii = int(rng.randint(len(ones)))
        jj = int(rng.randint(len(zeros)))
        i = ones[ii]
        j = zeros[jj]
        x[i] = 0
        x[j] = 1
        ones[ii] = j
        zeros[jj] = i
    return ones, zeros


def binary_optimize(x0, seed, time_limit, target_c=TARGET_C):

    if n == 0:

    if int(np.sum(x)) != target:

    if time_limit <= 0.0:

    if len(ones) == 0 or len(zeros) == 0:
        f = eval_binary_points(x)
        return x.astype(float), f, f * (2.0 / n)

    t_end = time.time() + max(0.0, float(time_limit))

    rng = np.random.RandomState(seed)

    current_f = eval_binary_points(x)
    best_f = current_f
    best_x = x.copy()
    stall = 0

    while time.time() < t_end:
        rem = max(0.0, t_end - time.time())
        frac = rem / total_for_decay
        T = 0.02 + 1.5 * frac * frac

        ii = int(rng.randint(len(ones)))
        jj = int(rng.randint(len(zeros)))
        i = ones[ii]
        j = zeros[jj]

        x[j] = 1
        newf = eval_binary_points(x)
        delta = newf - current_f

        if delta <= 0.0:
            accept = True
        else:
            p = math.exp(-float(delta) / T) if T > 1e-12 else 0.0
            accept = (rng.random() < p)

        if accept:
            ones[ii] = j
            zeros[jj] = i
            current_f = newf
            if newf < best_f:
                best_f = newf

            if stall > 6000:

        else:
            x[i] = 1
            x[j] = 0
            stall += 1
            if stall > 6000:
                x[:] = best_x
                ones, zeros = binary_lists(x)
                ones, zeros = binary_apply_random_swaps(x, ones, zeros, rng, 12)
                current_f = eval_binary_points(x)
                stall = 0

    best_f = float(eval_binary_points(best_x))
    best_c = best_f * (2.0 / n)
    return best_x.astype(float), best_f, best_c

                    elif h[j] >= h[i] + 1e-9:
                        dd = np.zeros(n, dtype=float)
                        dd[i] = 1.0
                        dd[j] = -1.0
                    else:
                        dd = np.zeros(n, dtype=float)
                        dd[i] = 1.0
                        dd[j] = -1.0

                    dd -= dd.mean()
                    nd = float(np.linalg.norm(dd))
                    if nd <= 1e-12:
                        continue
                    dd /= nd
                    amax = alpha_max_for_direction(h, dd)
                    if amax <= 1e-9:
                        continue
                    h_new, f_new = line_search_direction(h, dd, amax, fmax, fft_threshold)
                    if math.isfinite(f_new) and f_new < fmax - 1e-12:
                        h = h_new
                        accepted = True
                        break

        if not accepted and stall > 25:
            h = project_feasible(best_h.copy() + zero_sum_noise(rng, n, 0.05))
            stall = 0

        if stall > 15:
            K = 1
            power = 1.0
        elif it % 25 == 0:
            K = int(min(max(8, K + max(2, K // 4)), max(8, (2 * n - 1) // 3, 200)))
            power = min(8.0, power + 0.5)

    f = eval_full(h, fft_threshold)
    fmax = float(np.max(f)) if f.size else 0.0
    if math.isfinite(fmax) and fmax < best_f:
        best_f = fmax
        best_h = h.copy()

    best_h = project_feasible(best_h)
    best_f = eval_max(best_h, fft_threshold)
    best_c = best_f * (2.0 / n)
    return best_h, best_f, best_c


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    total = float(budget_s)

        h0 = random_binary(70, seed)

    best_h = h0.copy()
    best_c = eval_c5(best_h)
    if not math.isfinite(best_c) or best_c < 0.0:
        best_c = 1.0

    if (n0 % 2) == 0:
        rem = total - (time.time() - t0) - 80.0
        if rem > 20.0:
            tb = min(160.0, max(10.0, 0.15 * rem))
            tb = min(tb, total - (time.time() - t0) - 70.0)
            if tb > 5.0:
                xb = nearest_binary(h0, n0 // 2)
                xbest, _fb, cbest = binary_optimize(xb, seed + 1, tb, TARGET_C)
                cbest = eval_c5(xbest)
                if math.isfinite(cbest) and cbest < best_c:
                    best_c = cbest
                    best_h = np.array(xbest, dtype=float, copy=True)

    rem = total - (time.time() - t0) - 40.0
    if rem > 20.0:
        reserve_hi = 0.0
        if rem > 300.0:
            reserve_hi = min(400.0, 0.30 * rem)
        local_budget = max(10.0, rem - reserve_hi - 20.0)
        local_budget = min(local_budget, total - (time.time() - t0) - 30.0)

        starts = []
        starts.append((best_h.copy(), seed + 10, 0.35))

                       seed + 12, 0.20))

            starts.append((random_binary(n0, seed + 14), seed + 15, 0.12))

        starts.append((harmonic_start(n0, seed + 17), seed + 18, 0.08))

        for item in starts:

            if max_slice <= 10.0:
                break
            sh, ss, weight = item
            t_slice = local_budget * weight
            t_slice = max(5.0, t_slice)
            t_slice = min(t_slice, max_slice)
            if t_slice <= 1.0:
                break
            sh, _fb, cb = local_refine(sh, ss, t_slice)
            cb = eval_c5(sh)
            if math.isfinite(cb) and cb < best_c:
                best_c = cb
                best_h = np.array(sh, dtype=float, copy=True)

        rem2 = total - (time.time() - t0) - 50.0
        if rem2 > 30.0:
            t_slice = min(120.0, rem2)
            sh, _fb, cb = local_refine(best_h, seed + 19, t_slice)
            cb = eval_c5(sh)
            if math.isfinite(cb) and cb < best_c:
                best_c = cb
                best_h = np.array(sh, dtype=float, copy=True)

    rem = total - (time.time() - t0) - 40.0
    if rem > 100.0:
        n1 = int(min(1400, max(2 * n0, 2)))
        if n1 % 2 == 0 and n1 != n0:
            remb = total - (time.time() - t0) - 40.0
            if n1 <= 280 and remb > 150.0:

                tb = min(50.0, remb * 0.25)
                if tb > 10.0:

                        best_h = np.array(xbest, dtype=float, copy=True)

                (upsample(best_h, n1), seed + 30, 0.55),
                (upsample(h0, n1), seed + 31, 0.25),
                (triangular_start(n1), seed + 32, 0.20),

            for item in hi_starts:
                max_slice = total - (time.time() - t0) - 30.0
                if max_slice <= 10.0:
                    break
                sh, ss, weight = item
                t_slice = hi_budget * weight
                t_slice = max(5.0, t_slice)
                t_slice = min(t_slice, max_slice)
                if t_slice <= 1.0:

                sh, _fb, cb = local_refine(sh, ss, t_slice)
                cb = eval_c5(sh)
                if math.isfinite(cb) and cb < best_c:

                    best_h = np.array(sh, dtype=float, copy=True)

    rem = total - (time.time() - t0) - 50.0
    if rem > 150.0:
        n2 = int(min(2200, max(4 * n0, 4)))
        if n2 % 2 == 0:
            hi_budget = min(250.0, max(30.0, rem - 30.0))
            hi_starts = [
                (upsample(best_h, n2), seed + 40, 0.70),
                (triangular_start(n2), seed + 41, 0.30),

            for item in hi_starts:
                max_slice = total - (time.time() - t0) - 30.0

                    break
                sh, ss, weight = item

                t_slice = max(5.0, t_slice)

        cb = eval_c5(sh)
        if math.isfinite(cb) and cb < best_c:
            best_c = cb
            best_h = np.array(sh, dtype=float, copy=True)

    rem = total - (time.time() - t0) - 5.0
    nf = len(best_h)
    if rem > 30.0 and (nf % 2) == 0:

            best_c = cbest
            best_h = np.array(xbest, dtype=float, copy=True)

        final_c = float(np.max(f)) * (2.0 / float(n))
    else:

    if not math.isfinite(final_c):
        final_c = best_c
    if final_c < 0.0:
        final_c = 0.0

    return best_h, final_c, n
