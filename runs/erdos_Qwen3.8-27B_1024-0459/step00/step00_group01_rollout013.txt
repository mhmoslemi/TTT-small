Wait, but is corr_k linear in h? Let me double-check.

corr_k = ∑_{i=0}^{n-1-k} h[i] · (1 - h[i+k])  [for k ≥ 0]

Expand: = ∑_{i} h[i] - ∑_{i} h[i]·h[i+k]

The second term ∑_{i} h[i]·h[i+k] is NOT linear in h! It's bilinear (quadratic).

= ... + h[i] - h[i]h[i+k] - h[i-k]h[i] + h[i-k] + ...
= ... + h[i](1 - h[i+k] - h[i-k]) + h[i-k] + ...

With all other h's fixed, this IS linear in h[i]! The coefficient of h[i] is (1 - h[i+k] - h[i-k]).

So the problem is: each corr_k is a quadratic function of h, but it's linear in each individual coordinate (given the others fixed). This means coordinate descent could work!

For coordinate descent:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_and_argmax(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    idx = int(np.argmax(vals))
    return float(vals[idx]), idx

def resample(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.arange(old_n) / old_n  # normalized [0,1)
    new_x = np.arange(new_n) / new_n
    h_interp = np.interp(new_x, old_x, h)
    # Fix integral
    target = new_n / 2.0
    diff = target - np.sum(h_interp)
    if np.abs(diff) > 1e-9:
        mask = (h_interp > 0.001) & (h_interp < 0.999)
        if np.sum(mask) > 0:
            h_interp[mask] += diff / np.sum(mask)
        h_interp = np.clip(h_interp, 0, 1)
        # Final fix
        diff2 = target - np.sum(h_interp)
        if np.abs(diff2) > 1e-9:
            # Add to the most free position
            free = np.argmax((h_interp < 1 - np.abs(diff2/n)) * (h_interp > np.abs(diff2/n)))
            h_interp[int(free)] += diff2
            h_interp = np.clip(h_interp, 0, 1)
    return h_interp

    idx = np.argsort(h)[::-1]

    test_ns = sorted(test_ns)

    phase1_budget = budget_s * 0.55
    n_res = len(test_ns)

    for n in test_ns:
        elapsed = time.time() - t_start
        if elapsed > phase1_budget:
            break
        per_res_time = min(30.0, max(5.0, (phase1_budget - elapsed) / max(1, n_res)))

        if got_initial and best_n > 0:
            h_start = resample_h(best_h, n) if best_n != n else best_h.copy()
        else:
            h_start = rng.random(n)
            s = np.sum(h_start)
            if s > 1e-10:
                h_start = np.clip(h_start * (n / 2.0) / s, 0, 1)

        if n % 2 == 0:
            h_b = binarize(h_start, n)
            alloc_bin = min(per_res_time * 0.55, 25)
            if alloc_bin > 1:
                h_b, c5_b = sa_binary(h_b, n, alloc_bin, rng)
            if compute_c5(h_b) < best_c5:
                best_h = h_b.copy()
                best_c5 = compute_c5(h_b)
                best_n = n
            alloc_ls = min(8, (phase1_budget - (time.time() - t_start)))
            if alloc_ls > 1:
                h_b, c5_b = local_search_binary(h_b, n, budget_s=alloc_ls)
                if c5_b < best_c5:
                    best_h = h_b.copy()
                    best_c5 = c5_b
                    best_n = n

        remaining_p1 = phase1_budget - (time.time() - t_start)
        if remaining_p1 > 3:
            h_c, c5_c = sa_continuous(h_start, n, min(per_res_time * 0.4, remaining_p1), rng)
            if c5_c < best_c5:
                best_h = h_c.copy()
                best_c5 = c5_c
                best_n = n

    # Phase 2: focused refinement
    phase2_remain = budget_s - (time.time() - t_start)
    if phase2_remain > 15 and best_n > 0:
        n_best = best_n
        for restart in range(20):
            remaining2 = budget_s - (time.time() - t_start)
            if remaining2 < 5:
                break
            if n_best % 2 == 0:
                h_b = binarize(best_h, n_best)
                ones = np.where(h_b == 1.0)[0]
                zeros = np.where(h_b == 0.0)[0]
                n_swap = rng.integers(1, 6)
                for _ in range(n_swap):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    ci = ones[rng.integers(len(ones))]
                    cj = zeros[rng.integers(len(zeros))]
                    h_b[ci], h_b[cj] = h_b[cj], h_b[ci]
                    ones = np.where(h_b == 1.0)[0]
                    zeros = np.where(h_b == 0.0)[0]
                alloc_sa = min(remaining2 * 0.35, 80)
                alloc_ls = min(remaining2 * 0.1, 12)
                if alloc_sa > 1:
                    h_b, c5_b = sa_binary(h_b, n_best, alloc_sa, rng)
                if alloc_ls > 1:
                    h_b, c5_b = local_search_binary(h_b, n_best, budget_s=alloc_ls)
                if c5_b < best_c5:
                    best_h = h_b.copy()
                    best_c5 = c5_b
                    best_n = n_best

        remaining3 = budget_s - (time.time() - t_start)
        if remaining3 > 10 and best_n > 0:
            h_c, c5_c = sa_continuous(best_h, best_n, int(remaining3 * 0.85), rng)
            if c5_c < best_c5:
                best_h = h_c.copy()
                best_c5 = c5_c
                best_n = best_n

    # Final: try to improve at nearby even resolutions with remaining time
    remaining4 = budget_s - (time.time() - t_start)
    if remaining4 > 10 and best_n > 0:
        for n_try in [best_n - 2, best_n + 2, best_n - 4, best_n + 4]:
            if n_try < 20 or n_try > 120 or n_try % 2 != 0:
                continue
            if time.time() - t_start > budget_s - 5:
                break
            h_r = resample_h(best_h, n_try)
            h_b = binarize(h_r, n_try)
            alloc = min(30, (budget_s - (time.time() - t_start)) * 0.6)
            if alloc > 2:
                h_b, c5_b = sa_binary(h_b, n_try, alloc, rng)
                alloc2 = min(8, (budget_s - (time.time() - t_start)) * 0.4)
                if alloc2 > 1:
                    h_b, c5_b = local_search_binary(h_b, n_try, budget_s=alloc2)
                if c5_b < best_c5:
                    best_h = h_b.copy()
                    best_c5 = c5_b
                    best_n = n_try

    if best_h is None:
        n_fb = 48
        best_h = random_binary(n_fb, rng)
        best_c5 = compute_c5(best_h)
        best_n = n_fb

    return best_h, best_c5, best_n
