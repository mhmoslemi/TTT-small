def evaluate_objective(h_vec):
    """Compute the max overlap using numpy correlation"""
    n_nodes = len(h_vec)
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
    dx = 2.0 / n_nodes
    return np.max(corr) * dx
