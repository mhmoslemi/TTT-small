from scipy.optimize import minimize
import time

def compute_c5(h, n):
    """Compute C5 for a given h vector of length n."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0, 1)
    diff = n/2 - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the correction
        if diff > 0:
            # Need to increase sum
            mask = h < 1.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
        else:
            # Need to decrease sum
            mask = h > 0.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
    return h

def objective_c5(h_flat, n, beta=100.0):
    """Smooth approximation of max correlation."""
    h = h_flat
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx

    max_val = np.max(corr)

    shifted = corr * beta
    max_shifted = np.max(shifted)
    return (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / beta

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final exact adjustment
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-15:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = h[idx] + diff2
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h, n_from, n_to):
    x_from = (np.arange(n_from) + 0.5) * (2.0 / n_from)
    x_to = (np.arange(n_to) + 0.5) * (2.0 / n_to)
    return np.interp(x_to, x_from, h)

def _con_eq(x):
    return np.sum(x) - _g_n / 2.0

def _obj_smooth(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(corr)
    s = _g_beta * corr
    sm = np.max(s)
    return (sm + np.log(np.sum(np.exp(s - sm)))) / _g_beta

def _obj_exact(x):
    return compute_c5(x, _g_n)

def _obj_penalty(x):
    n = _g_n
    dx = 2.0 / n
    corr = np.correlate(x, 1 - x, mode='full') * dx
    m = np.max(corr)
    s = _g_beta * corr
    sm = np.max(s)
    base = (sm + np.log(np.sum(np.exp(s - sm)))) / _g_beta
    penalty = _g_mu * (np.sum(x) - n / 2.0) ** 2
    return base + penalty

def _run_slsqp(h_start, n, beta, maxiter, ftol):
    global _g_n, _g_beta
    _g_n = n
    _g_beta = beta
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq}]
    res = minimize(_obj_smooth, h_start, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    _g_beta = 100.0
    return res

def _run_lbfgsb(h_start, n, beta, mu, maxiter, ftol):
    global _g_n, _g_beta, _g_mu
    _g_n = n
    _g_beta = beta
    _g_mu = mu
    bnds = [(0.0, 1.0)] * n
    res = minimize(_obj_penalty, h_start, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-6})
    _g_mu = 1e4
    return res

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_beta
    t0 = time.time()
    np.random.seed(seed)

    h0 = np.array(initial_h_values, dtype=float).flatten()
    n0 = len(h0)
    best_h = h0.copy()
    best_n = n0
    best_c5 = compute_c5(best_h, best_n)

    # Phase 1: Optimize at initial resolution with SLSQP
    if budget_s - (time.time() - t0) > 60:
        res = _run_slsqp(best_h, n0, 120.0, 400, 1e-13)
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            hx = project_feasible(res.x, n0)
            cx = compute_c5(hx, n0)
            if cx < best_c5:
                best_h, best_c5 = hx, cx

    # Also try a few random perturbations at n0
    for seed2 in [1, 7, 42, 123]:
        if budget_s - (time.time() - t0) < 40:
            break
        np.random.seed(seed2)
        h_rand = best_h + 0.05 * np.random.randn(n0)
        h_rand = project_feasible(h_rand, n0)
        res = _run_slsqp(h_rand, n0, 100.0, 200, 1e-12)
        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            hx = project_feasible(res.x, n0)
            cx = compute_c5(hx, n0)
            if cx < best_c5:
                best_h, best_c5 = hx, cx

    # Phase 2: Multi-resolution optimization
    cur_n = best_n
    cur_h = best_h.copy()

    for level in range(6):
        elapsed = time.time() - t0
        if budget_s - elapsed < 50:
            break
        tgt_n = cur_n * 2
        h_interp = interp_h(cur_h, cur_n, tgt_n)
        h_interp = project_feasible(h_interp, tgt_n)
        c5_int = compute_c5(h_interp, tgt_n)

        if c5_int < best_c5:
            best_h, best_c5, best_n = h_interp.copy(), c5_int, tgt_n

        remaining = budget_s - (time.time() - t0)
        if remaining < 40:
            cur_h, cur_n = h_interp, tgt_n
            continue

        # Choose optimizer based on size
        if tgt_n <= 200:
            maxit = min(200, max(50, int(20.0) * 30 // (tgt_n / 42)))
            res = _run_slsqp(h_interp, tgt_n, 100.0, maxit, 1e-12)
        else:
            maxit = min(300, max(50, int(30.0 * 200.0 / tgt_n)))
            res = _run_lbfgsb(h_interp, tgt_n, 80.0, 1e6, maxit, 1e-10)

        if hasattr(res, 'x') and res.x is not None and np.all(np.isfinite(res.x)):
            hx = project_feasible(res.x, tgt_n)
            cx = compute_c5(hx, tgt_n)
            if cx < best_c5:
                best_h, best_c5, best_n = hx, cx, tgt_n
                cur_h, cur_n = hx, tgt_n
            else:
                cur_h, cur_n = h_interp, tgt_n
        else:
            cur_h, cur_n = h_interp, tgt_n

        # Second pass with refined beta if time allows
        if tgt_n <= 200 and budget_s - (time.time() - t0) > 80:
            res2 = _run_slsqp(cur_h, tgt_n, 200.0, 100, 1e-13)
            if hasattr(res2, 'x') and res2.x is not None and np.all(np.isfinite(res2.x)):
                hx2 = project_feasible(res2.x, tgt_n)
                cx2 = compute_c5(hx2, tgt_n)
                if cx2 < best_c5:
                    best_h, best_c5, best_n = hx2, cx2, tgt_n
                    cur_h, cur_n = hx2, tgt_n

    # Phase 3: Truncated Newton / coordinate descent refinement
    if budget_s - (time.time() - t0) > 20:
        n = best_n
        h = best_h.copy()
        remaining = budget_s - (time.time() - t0)
        use_slsqp = n <= 300
        if use_slsqp:
            res3 = _run_slsqp(h, n, 300.0, min(100, int(remaining / 1.5)), 1e-14)
        else:
            res3 = _run_lbfgsb(h, n, 200.0, 1e8, min(200, int(remaining / 0.5)), 1e-12)
        if hasattr(res3, 'x') and res3.x is not None and np.all(np.isfinite(res3.x)):
            hx3 = project_feasible(res3.x, n)
            cx3 = compute_c5(hx3, n)
            if cx3 < best_c5:
                best_h, best_c5 = hx3, cx3

    # Phase 4: Coordinate descent polish
    if budget_s - (time.time() - t0) > 10:
        n = best_n
        h = best_h.copy()
        dx = 2.0 / n
        remaining = budget_s - (time.time() - t0)
        max_cd_iters = int(min(200, remaining / 0.5))

        for it in range(max_cd_iters):
            improved = False
            for i in range(n):
                old_val = h[i]
                # Optimize h[i] in [0,1] keeping others fixed
                # Objective as function of t = h[i]
                # Try a few candidate values
                best_local = compute_c5(h, n)
                candidates = [0.0, 1.0, old_val]
                # Add values around old_val
                for delta in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
                    v = old_val + delta
                    if 0.0 < v < 1.0:
                        candidates.append(v)
                # Bisection-like: try to reduce the max
                c5_before = compute_c5(h, n)
                best_t = old_val
                best_c5_local = c5_before
                for t_cand in candidates:
                    h[i] = t_cand
                    c_val = compute_c5(h, n)
                    if c_val < best_c5_local:
                        best_c5_local = c_val
                        best_t = t_cand
                h[i] = best_t
                if h[i] != old_val:
                    improved = True
            # Re-project for sum constraint
            h = project_feasible(h, n)
            c5_now = compute_c5(h, n)
            if c5_now < best_c5:
                best_h, best_c5 = h.copy(), c5_now
            if not improved and it > 2:
                break
            if budget_s - (time.time() - t0) < 5:
                break

    # Final verification
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
