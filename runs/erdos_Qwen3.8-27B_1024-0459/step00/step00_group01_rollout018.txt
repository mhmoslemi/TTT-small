    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')

def c5_gradient(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        # j+s < n => j < n-s, contribution: (1-h[j+s])
        if n - s > 0:

            h = np.concatenate([h, t[:rem]])
    else:
        h = rng.random(n)
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, int(n))
    return h


def run_slsqp(h_init, n, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    res = minimize(
        obj_wrapper, h_init, jac=jac_wrapper, method='SLSQP',
        bounds=bounds, constraints=cons,
        options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0}
    )
    h_opt = project_sum(res.x, n)
    return h_opt, res.niter


def projected_gradient(h, n, steps, lr_init, decay):
    h = h.copy()
    lr = lr_init
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    for it in range(steps):
        obj, grad = c5_gradient(h), compute_c5(h, n)
        h = h - lr * grad
        h = project_sum(h, n)
        c5_val = compute_c5(h, n)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        lr *= decay
    return best_h, best_c5


def simulated_annealing_binary(h, n, t_start, t_end, n_iter, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    # Round to binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = project_sum(h, n)
    # Make truly binary
    n_ones = int(round(dh_sum(h)))
    h = np.zeros(n)
    ones_idx = np.where(h > 0.5)[0]
    if len(ones_idx) >= n_ones:
        h[ones_idx[:n_ones]] = 1.0
    else:
        h[ones_idx] = 1.0
        need = n_ones - len(ones_idx)
        zeros_idx = np.where(h == 0.0)[0]
        if len(zeros_idx) >= need:
            h[zeros_idx[:need]] = 1.0
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    temp = t_start
    for it in range(n_iter):
        temp = t_start * (t_end / t_start) ** (it / max(n_iter - 1, 1))
        # Random swap: flip one 1 to 0 and one 0 to 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return best_h, best_c5


def dh_sum(h):
    return h.sum()


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    # Evaluate parent
    parent_h = initial_h_values.copy().astype(np.float64)
    parent_n = len(parent_h)
    parent_c5 = compute_c5(parent_h, parent_n)
    print(f"Parent: n={parent_n}, C5={parent_c5:.6f}")

    best_h = parent_h.copy()
    best_n = parent_n
    best_c5 = parent_c5

    # Also evaluate simple constructions at parent resolution
    for ptype in ['half', 'alternating', 'periodic3', 'random_binary']:
        h_test = make_bin_init(parent_n, ptype, seed)
        c5_test = compute_c5(h_test, parent_n)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test
            best_n = parent_n
            print(f"  Simple {ptype} at n={parent_n}: C5={c5_test:.6f}")

    # Main optimization across resolutions
    resolutions = [100, 150, 200, 300, 500]
    patterns = ['half', 'alternating', 'random_binary', 'periodic3', 'random_continuous']

    for n_target in resolutions:
        if elapsed() > budget_s - 120:
            break
        print(f"\n--- n={n_target} ---")

        # Build initializations
        inits = []
        # 1. Upsampled best so far
        if best_n < n_target:
            h_up = upsample_step(best_h, best_n, n_target)
            inits.append(('upsampled', h_up))
        else:
            # Downsample
            h_dn = upsample_step(best_h, best_n, n_target)
            inits.append(('downsampled', h_dn))

        # 2. Pattern inits
        for ptype in patterns:
            h_p = make_bin_init(n_target, ptype, seed + n_target + hash(ptype) % 1000)
            inits.append((ptype, h_p))

        # 3. Upsampled parent
        h_par_up = upsample_step(parent_h, parent_n, n_target)
        inits.append(('parent_up', h_par_up))

        # Run SLSQP on each init
        time_per_opt = max(30, min(180, (budget_s - 120 - elapsed()) / max(len(inits), 1)))
        max_iter = int(min(500, time_per_opt * 10))

        for name, h_init in inits:
            if elapsed() > budget_s - 60:
                break
            try:
                h_opt, niter = run_slsqp(h_init, n_target, max_iter, 1e-12)
                c5_opt = compute_c5(h_opt, n_target)
                print(f"  {name}: C5={c5_opt:.6f} (iter={niter})")
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_target
                    print(f"  *** NEW BEST: {best_c5:.6f} at n={n_target} ***")
            except Exception as e:
                print(f"  {name}: Error {e}")

        # Also try projected gradient refinement on the best at this n
        if elapsed() < budget_s - 60:
            h_pg, c5_pg = projected_gradient(h_opt if 'h_opt' in dir() else h_init, n_target,
                                             steps=2000, lr_init=0.05, decay=0.998)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                best_n = n_target
                print(f"  PG refine: C5={c5_pg:.6f}")

    # Final refinement at best resolution
    if elapsed() < budget_s - 30:
        print(f"\n--- Final refinement at n={best_n} ---")
        remaining = budget_s - 30 - elapsed()
        steps = int(min(50000, remaining * 200))
        h_fin, c5_fin = projected_gradient(best_h, best_n,
                                           steps=steps, lr_init=0.02, decay=0.9995)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            print(f"  Final PG: C5={c5_fin:.6f}")

        # One more SLSQP polish
        if elapsed() < budget_s - 20:
            try:
                h_pol, _ = run_slsqp(best_h, best_n, 200, 1e-14)
                c5_pol = compute_c5(h_pol, best_n)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    print(f"  SLSQP polish: C5={c5_pol:.6f}")
            except Exception:
                pass

    # Ensure final solution is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, best_n)
    final_c5 = compute_c5(best_h, best_n)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"\nFinal: n={best_n}, C5={best_c5:.6f}")
    return best_h, best_c5, best_n
