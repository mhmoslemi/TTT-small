    """Project h onto {0 <= x <= 1, sum(x) = target_sum} using bisection."""
    n = len(h)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction (tiny adjustments)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        # Distribute evenly
        mask = (x > 0) & (x < 1)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_corr(h, n):
    """Compute full correlation of h with 1-h, scaled by dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(corr_vals):
    return np.max(corr_vals)

def soft_max(vals, t):
    """Smooth max: (1/t) log sum exp(t * vals)"""
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(t * (vals - m)))) / t

    """Gradient of soft_max w.r.t. vals: t * softmax(t*(vals - max))"""
    m = np.max(vals)
    e = np.exp(t * (vals - m))
    return t * e / np.sum(e)

def corr_and_grad(h, n):
    """
    Compute corr values and the gradient matrix.
    Returns: corr (array of 2n-1), G (matrix (2n-1) x n) where G[k,j] = d(corr_k)/d(h_j)
    """

    corr = corr_raw * dx

    G = np.zeros((2*n-1, n))

    G[n-1, :] = (1.0 - 2.0*h) * dx

        # d/dh[j]: (1-h[j+k]) if j <= n-1-k; -h[j-k] if j >= k

        part1 = (1.0 - h[k:n])  # h[k], h[k+1], ..., h[n-1], length n-k
        G[idx, 0:upper+1] = part1[:upper+1]
        # Part 2: for j in [k, n-1]: -h[j-k]
        part2 = -h[0:n-k]  # -h[0], -h[1], ..., -h[n-1-k], length n-k
        G[idx, k:n] += part2
        G[idx, :] *= dx
    
    # Negative lags k = 1, ..., n-1: idx = n-1-k (represents lag -k)
    for k in range(1, n):
        idx = n - 1 - k
        # corr_{-k} = sum_{i=k}^{n-1} h[i]*(1-h[i-k]) * dx

        part2 = -h[k:n]  # -h[k], ..., -h[n-1], length n-k
        G[idx, 0:upper+1] += part2
        G[idx, :] *= dx
    
    return corr, G

    obj = soft_max(corr, t)
    weights = soft_max_grad(corr, t)  # shape (2n-1,)

            n_starts = 1

        if n > 200:
            n_starts = max(1, n_starts - 1)

        for si in range(n_starts):
            if time_left() < 20:
                break

            # Create starting point
            if si == 0:
                h_init = upsample_to(h0, n)
            elif si == 1:
                h_init = np.full(n, 0.5)
                h_init += 0.08 * np.random.randn(n)
            elif si == 2:
                h_init = np.full(n, 0.5)
                # Block-like perturbation
                h_init[:n // 4] += 0.15
                h_init[n // 4:n // 2] -= 0.1
                h_init[n // 2:3 * n // 4] += 0.1
                h_init[3 * n // 4:] -= 0.15
            else:
                h_init = np.full(n, 0.5)
                h_init += 0.15 * np.random.randn(n)

            h_init = project_box_sum(h_init, n / 2.0)

            # Warm-start with smooth max, anneal temperature
            t_schedule = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
            # For very tight t, use fewer iterations
            h_curr = h_init.copy()

            for t in t_schedule:
                if time_left() < 10:
                    break

                # Number of iterations
                if n <= 100:
                    n_iter = 300
                elif n <= 200:
                    n_iter = 150
                else:
                    n_iter = 80

                if t >= 40:
                    n_iter = max(30, n_iter // 3)

                lr_init = 40.0 / (t * n)
                lr = lr_init
                prev_obj = obj_and_grad(h_curr, n, t)[0]

                for it in range(n_iter):
                    if time_left() < 8:
                        break

                    obj, grad = obj_and_grad(h_curr, n, t)

                    # Simple gradient step with step size decay
                    step_norm = np.linalg.norm(grad)
                    if step_norm < 1e-12:
                        break

                    # Adaptive step size
                    alpha = lr / (step_norm + 1e-10)
                    # Clamp to reasonable step
                    alpha = min(alpha, 2.0 / step_norm)

                    h_try = h_curr - alpha * grad
                    h_try = project_box_sum(h_try, n / 2.0)

                    obj_try = obj_and_grad(h_try, n, t)[0]

                    if obj_try < obj:
                        h_curr = h_try
                        lr *= 1.05
                    else:
                        # Try smaller step
                        found_smaller = False
                        for back in range(5):
                            alpha *= 0.5
                            h_try = h_curr - alpha * grad
                            h_try = project_box_sum(h_try, n / 2.0)
                            obj_try = obj_and_grad(h_try, n, t)[0]
                            if obj_try < obj:
                                h_curr = h_try
                                found_smaller = True
                                break
                        if not found_smaller:
                            lr *= 0.5
                            if lr < 1e-10:
                                break

                # Add small perturbation at each temperature to escape local minima
                if t < 40 and time_left() > 30:
                    h_curr = project_box_sum(h_curr + 0.02 * np.random.randn(n), n / 2.0)

            # Evaluate this solution
            c5_curr = compute_c5(h_curr, n)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

            # If we found a good solution, update h0 for interpolation in later resolutions
            if c5_curr < compute_c5(h0, n0):
                h0 = upsample_to(h_curr, n0) if n != n0 else h_curr.copy()

    # Final polishing: local search around best solution
    if time_left() > 20:
        n_polish = best_n
        h_polish = best_h.copy()
        eps_values = [0.01, 0.005, 0.002, 0.001]

        for eps in eps_values:
            if time_left() < 10:
                break
            for trial in range(50):
                if time_left() < 8:
                    break
                # Random coordinate perturbation
                idx = np.random.choice(n_polish, size=max(1, n_polish // 10), replace=False)
                delta = eps * (np.random.rand(len(idx)) * 2 - 1)
                h_try = h_polish.copy()
                h_try[idx] += delta
                h_try = project_box_sum(h_try, n_polish / 2.0)
                c5_try = compute_c5(h_try, n_polish)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h_polish = h_try.copy()

    # Final verification
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
