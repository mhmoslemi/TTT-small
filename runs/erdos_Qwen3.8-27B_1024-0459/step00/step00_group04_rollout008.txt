from scipy.optimize import minimize, minimize_scalar
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx

def project_to_feasible(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}."""
    h = np.array(h, dtype=float)
    target = n / 2.0
    
    # Bisection on Lagrange multiplier
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h + mid, 0, 1)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    h_proj = np.clip(h + (lo+hi)/2, 0, 1)
    # Final adjustment (should be very small)
    diff = target - np.sum(h_proj)
    if diff > 0:
        # Need to add more mass
        mask = h_proj < 1.0
        if np.sum(mask) > 0:
            add = diff / np.sum(mask)
            h_proj[mask] = np.minimum(h_proj[mask] + add, 1.0)
    else:
        mask = h_proj > 0.0
        if np.sum(mask) > 0:
            sub = -diff / np.sum(mask)
            h_proj[mask] = np.maximum(h_proj[mask] - sub, 0.0)
    
    return np.clip(h_proj, 0, 1)

def objective(h, n):
    """Object for optimization: C5 + penalty for constraint violation."""
    h_proj = project_to_feasible(h, n)
    c5 = evaluate_c5(h_proj)
    # Small penalty for deviation from exact constraint
    penalty = 1000 * (np.sum(h_proj) - n/2.0)**2
    return c5 + penalty

def smooth_objective(h, n, alpha=50):
    h_proj = project_to_feasible(h, n)
    dx = 2.0 / n
    c = np.correlate(h_proj, 1 - h_proj, mode="full") * dx
    # Smooth max via log-sum-exp
    max_c = np.max(c)
    shifted = c - max_c
    smooth_max = max_c + np.log(np.sum(np.exp(alpha * shifted))) / alpha

    n = _g_n
    alpha = _g_alpha
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx

    sh = np.maximum(sh, -50.0)
    logsum = m + np.log(np.sum(np.exp(sh))) / alpha
    pen = 100.0 * (np.sum(h) - 0.5 * n) ** 2

        else:
