from scipy.signal import fftconvolve
import time

def softmax_max(x, beta):
    """Smooth approximation to max(x)"""
    # x is a 1D array
    x_max = np.max(x)
    x_shifted = x - x_max
    return x_max + np.log(np.mean(np.exp(beta * x_shifted))) / beta

def compute_c5(h, dx):
    """Compute C5 = max(correlate(h, 1-h, full)) * dx"""
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx

def objective_softmax(h, n_points, beta):
    """Objective with soft-max for gradient computation"""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1-h, mode='full')
    # Soft-max of corr * dx
    return softmax_max(corr * dx, beta)

def project_to_integral(h, n_points):
    """Project h onto sum(h) = n_points/2 while maintaining 0<=h<=1"""
    target = n_points / 2.0
    diff = target - np.sum(h)
    h = h + diff / n_points
    return np.clip(h, 0, 1)

def _fast_corr(h, n):
    """Compute full correlation of h with 1-h using FFT"""
    m = 2 * n - 1
    fh = np.fft.rfft(h, m)
    f1mh = np.fft.rfft(1.0 - h, m)
    return np.fft.irfft(fh * np.conj(f1mh), m)

def _c5_from_correlation(corr, dx):
    return np.max(corr) * dx

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = _fast_corr(h, n)
    return np.max(corr) * dx

def _soft_max(x, beta=40.0):
    mx = np.max(x)
    return mx + np.log(np.mean(np.exp(beta * (x - mx)))) / beta

def _c5_soft(h, beta=40.0):
    n = len(h)
    dx = 2.0 / n
    corr = _fast_corr(h, n) * dx
    return _soft_max(corr, beta)

        return h

    h = h + diff / len(h)

def _corr_full(h):
    """Full correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    """Compute C5 value for a given h"""
    n = len(h)
    dx = 2.0 / n
    corr = _corr_full(h)
    return np.max(corr) * dx

def _soft_max_c5(h, beta=50.0):
    """Soft-max version for smooth optimization"""
    n = len(h)
    dx = 2.0 / n
    corr = _corr_full(h) * dx
    mx = np.max(corr)
    return mx + np.log(np.mean(np.exp(beta * (corr - mx)))) / beta

def _project_sum(h, target_sum):
    """Project h to satisfy sum constraint and bounds"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def _generate_structured(n, rng, pattern_type=0):
    """Generate structured initial candidates"""
    h = np.zeros(n)

        x = np.linspace(0, 2, n, endpoint=False)
        h = (0.5 + 0.4 * np.sin(2*np.pi*5*x)).clip(0, 1)
    
    return h
