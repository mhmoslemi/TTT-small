import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_constrained(h, N):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = N / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_func(params, N, dx, tau, target_sum, penalty_w):
    h = np.clip(params, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(corr)
    shifted = tau * (corr - cmax)
    # Stable log-sum-exp
    log_sum = np.log(np.sum(np.exp(shifted)))
    soft = (cmax + log_sum / tau) * dx
    pen = penalty_w * (np.sum(h) - target_sum) ** 2
    return soft + pen


def make_gradient(params, N, dx, tau, target_sum, penalty_w):
    h = np.clip(params, 0.0, 1.0)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    cmax = np.max(corr)
    shifted = tau * (corr - cmax)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp / tau
    grad_corr = np.zeros(N)
    full_len = 2 * N - 1
    for k in range(full_len):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        # corr[k] = sum_i h[i] * (1-h[i+k-N+1]) for valid i
        # d(corr[k])/d(h[j]) involves terms where j=i or j=i+k-N+1
        offset = k - (N - 1)

        # Term: h[j-offset] * (1-h[j]) -> derivative w.r.t h[j] is -h[j-offset]
        lo = max(0, offset)
        hi = min(N, N + offset)
        for j in range(lo, hi):
            grad_corr[j] += w * (1.0 - h[j + offset])
        lo2 = max(0, -offset)
        hi2 = min(N, N - offset)
        for j in range(lo2, hi2):
            grad_corr[j] -= w * h[j - offset]
    pen_grad = 2.0 * penalty_w * (np.sum(h) - target_sum)
    return grad_corr + pen_grad
