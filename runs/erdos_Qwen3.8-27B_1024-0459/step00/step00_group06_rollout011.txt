    return h


class ObjFunc:
    def __init__(self, n):
        self.n = n
        self.dx = 2.0 / self.n

    def __call__(self, x):
        x_clipped = np.clip(x, 0.0, 1.0)
        return compute_c5(x_clipped, self.dx)


class SumConstraint:
    def __init__(self, n):
        self.n = n

    def __call__(self, x):
        return np.sum(x) - self.n / 2.0


def slsqp_optimize(h_init, n, maxiter=300, ftol=1e-10):
    dx = 2.0 / n
    h_init = project_to_feasible(h_init.copy(), n)
    c5_init = compute_c5(h_init, dx)

    obj = ObjFunc(n)
    constr = SumConstraint(n)
    constraints = [{'type': 'eq', 'fun': constr.__call__}]
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj, h_init, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': ftol})
    h_opt = project_to_feasible(res.x, n)
    c5_opt = compute_c5(h_opt, dx)
    if c5_opt > c5_init:
        return h_init, c5_init
    return h_opt, c5_opt


def coordinate_descent_polish(h_init, n, max_rounds=50, lr=0.05):
    dx = 2.0 / n
    h = project_to_feasible(h_init.copy(), n)
    c5_best = compute_c5(h, dx)

    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            for delta in [lr, -lr, 0.5*lr, -0.5*lr, 0.25*lr, -0.25*lr]:
                h_trial = h.copy()
                h_trial[i] = np.clip(h[i] + delta, 0.0, 1.0)
                h_trial = project_to_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial, dx)
                if c5_trial < c5_best - 1e-12:
                    h = h_trial
                    c5_best = c5_trial
                    improved = True
                    break
        if not improved:
            break
    return h, c5_best


def inspect_func_time(start_time, budget_s, threshold=0.85):
    return (time.time() - start_time) > budget_s * threshold


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_c5 = np.inf
    best_n = 0

    def record_trial(h, n, label):
        nonlocal best_h, best_c5, best_n
        dx = 2.0 / n
        h = project_to_feasible(h.copy(), n)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            print(f"  [{label}] C5 = {c5:.6f}, n = {n}")
        return c5

    # Phase 1: Try the parent construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
        print(f"Parent: n={n_parent}")
        record_trial(h_parent, n_parent, "parent_init")

        if not inspect_func_time(start_time, budget_s):
            h_opt, c5_opt = slsqp_optimize(h_parent, n_parent, maxiter=200, ftol=1e-9)
            record_trial(h_opt, n_parent, "parent_slsqp")

    # Phase 2: Try n=100 (even, allows pure binary)
    if not inspect_func_time(start_time, budget_s):
        print("Trying n=100...")
        h100 = binary_initial(100, seed)
        record_trial(h100, 100, "n100_binary")
        h_opt, c5_opt = slsqp_optimize(h100, 100, maxiter=200, ftol=1e-9)
        record_trial(h_opt, 100, "n100_slsqp")

    # Phase 3: Try n=95 binary
    if not inspect_func_time(start_time, budget_s):
        print("Trying n=95...")
        h95 = binary_initial(95, seed + 100)
        record_trial(h95, 95, "n95_binary")
        h_opt, c5_opt = slsqp_optimize(h95, 95, maxiter=200, ftol=1e-9)
        record_trial(h_opt, 95, "n95_slsqp")

    # Phase 4: Try n=128 (even, binary)
    if not inspect_func_time(start_time, budget_s):
        print("Trying n=128...")
        h128 = binary_initial(128, seed + 200)
        record_trial(h128, 128, "n128_binary")
        h_opt, c5_opt = slsqp_optimize(h128, 128, maxiter=150, ftol=1e-9)
        record_trial(h_opt, 128, "n128_slsqp")

    # Phase 5: Coordinate descent polish on best
    if not inspect_func_time(start_time, budget_s) and best_h is not None:
        print(f"Polishing n={best_n}...")
        hx, cx = coordinate_descent_polish(best_h, best_n, max_rounds=30, lr=0.08)
        record_trial(hx, best_n, "polish")

    # Phase 6: Random restarts from best
    if not inspect_func_time(start_time, budget_s) and best_h is not None:
        n_best = best_n
        for attempt in range(8):
            if inspect_func_time(start_time, budget_s, 0.8):
                break
            rng = np.random.RandomState(seed + 1000 + attempt * 7)
            scale = 0.15 / (1 + attempt * 0.3)
            h_pert = best_h + scale * rng.randn(n_best)
            h_pert = project_to_feasible(h_pert, n_best)
            h_opt, c5_opt = slsqp_optimize(h_pert, n_best, maxiter=100, ftol=1e-8)
            record_trial(h_opt, n_best, f"restart{attempt}")

    # Phase 7: Polish again
    if not inspect_func_time(start_time, budget_s, 0.85) and best_h is not None:
        hx, cx = coordinate_descent_polish(best_h, best_n, max_rounds=20, lr=0.04)
        record_trial(hx, best_n, "final_polish")

    # Fallback if nothing worked
    if best_h is None:
        n_fb = 100
        best_h = np.ones(n_fb) * 0.5
        best_c5 = compute_c5(best_h, 2.0 / n_fb)
        best_n = n_fb
        print(f"  [fallback] C5 = {best_c5:.6f}")

    elapsed = time.time() - start_time
    print(f"\nDONE: C5 = {best_c5:.6f}, n = {best_n}, time = {elapsed:.1f}s")

    return (best_h, best_c5, best_n)
