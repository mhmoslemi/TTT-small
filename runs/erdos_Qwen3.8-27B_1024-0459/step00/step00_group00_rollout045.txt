import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max(vals, t=50.0):
    m = np.max(vals)
    return float(m + np.log(np.mean(np.exp((vals - m) * t))) / t)

def upsample_linear(h, new_n):
    n = len(h)
    old_x = (np.arange(n) + 0.5) * (2.0 / n)
    new_x = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    new_h = np.interp(new_x, old_x, h)
    return np.clip(new_h, 0.0, 1.0)

def obj_lbfgs(params, n, t, pc):
    h = np.clip(params, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    sm = soft_max(corr, t)
    penalty = pc * (np.sum(h) - n * 0.5) ** 2
    return sm + penalty

def fix_sum_exact(h, n):
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    order = np.argsort(-np.abs(h - 0.5))
    remaining = diff
    for i in order:
        if remaining > 0:
            adj = min(remaining, 1.0 - h[i])
            h[i] += adj
            remaining -= adj
        elif remaining < 0:
            adj = max(remaining, -h[i])
            h[i] += adj
            remaining -= adj
        if abs(remaining) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def local_search(h, n, max_iters=300, time_limit=30.0):
    import time
    t0 = time.time()
    h = h.copy()
    current_c5 = compute_c5(h)
    target = n * 0.5
    trial_vals = [0.0, 1.0, 0.25, 0.5, 0.75]
    
    for _ in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            for nv in trial_vals:
                if abs(nv - h[i]) < 1e-6:
                    continue
                h_new = h.copy()
                h_new[i] = nv
                h_new = fix_sum_exact(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < current_c5 - 1e-10:
                    h = h_new
                    current_c5 = c5_new
                    improved = True
                    break
            if improved:
                break
        # Also try complement
        if not improved:
            h_comp = 1.0 - h
            h_comp = fix_sum_exact(h_comp, n)
            c5_comp = compute_c5(h_comp)
            if c5_comp < current_c5 - 1e-10:
                h = h_comp
                current_c5 = c5_comp
                improved = True
        if not improved:
            break
    return h, current_c5
