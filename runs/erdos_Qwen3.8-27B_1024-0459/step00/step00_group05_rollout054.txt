    return h_cur, c5_cur

def nelder_mead_optimize(h0, n, time_fn, min_time):
    """Run Nelder-Mead with time guard."""
    def obj(h):
        return compute_c5(project_h(h, n), n)
    try:
        res = optimize.minimize(obj, h0, method='Nelder-Mead',
                                options={'maxiter': 3000, 'adaptive': True,
                                         'xatol': 1e-9, 'fatol': 1e-10})
        h_res = project_h(res.x, n)
        return h_res, compute_c5(h_res, n)
    except Exception:
        return h0, compute_c5(h0, n)

def random_binary_init(n, rng):
    """Create a random binary-ish initialization."""
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # Add small noise to break symmetry
    h += rng.uniform(-0.05, 0.05, n)
    return project_h(h, n)

def structured_init(n, rng):
    """Try various structured inits."""
    method = rng.randint(5)
    if method == 0:
        # Alternating blocks
        h = np.tile([1, 0], n // 2)
        if n % 2:
            h = np.append(h, 0)
    elif method == 1:
        # Random permutation of 35 ones
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 1
    elif method == 2:
        # Periodic pattern
        period = max(2, rng.randint(3, 20))
        pattern = (np.arange(period) < period // 2).astype(float)
        h = np.tile(pattern, n // len(pattern) + 1)[:n]
    elif method == 3:
        # Smooth wave
        t = np.linspace(0, 2 * np.pi, n)
        h = 0.5 + 0.49 * np.sin(t + rng.uniform(0, 2 * np.pi))
    else:
        # Gaussian bump shifted
        center = rng.uniform(0.3, 1.7)
        width = rng.uniform(0.3, 0.7)
        t = np.linspace(0, 2, n)
        h = np.exp(-0.5 * ((t - center) / width) ** 2)
    return project_h(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    # Get initial solution
    try:
        n = len(initial_h_values)
        h_init = project_h(np.array(initial_h_values, dtype=float), n)
    except NameError:
        n = 70
        h_init = project_h(np.full(n, 0.5), n)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, n)
    best_n = n

    # Phase 1: Nelder-Mead from initial
    if tl() > 200:
        h_nm, c5_nm = nelder_mead_optimize(h_init.copy(), n, tl, 100)
        if c5_nm < best_c5:
            best_h, best_c5, best_n = h_nm, c5_nm, n

    # Phase 2: Aggressive random hill climbing from best
    if tl() > 200:
        h_cur = best_h.copy()
        c5_cur = best_c5
        for step_i, (s_init, s_min) in enumerate([(1.0, 0.001), (0.5, 0.001), (0.2, 0.0005), (0.1, 0.0005), (0.05, 0.0002)]):
            if tl() < 100:
                break
            max_iter = 300000
            h_cur, c5_cur = local_search(h_cur, n, rng, max_iter, s_init, s_min, tl, 100)
            if c5_cur < best_c5:
                best_h, best_c5, best_n = h_cur.copy(), c5_cur, n

    # Phase 3: Nelder-Mead from polished solution
    if tl() > 150:
        h_nm, c5_nm = nelder_mead_optimize(best_h.copy(), n, tl, 80)
        if c5_nm < best_c5:
            best_h, best_c5, best_n = h_nm, c5_nm, n

    # Phase 4: More hill climbing
    if tl() > 100:
        h_cur, c5_cur = local_search(best_h.copy(), n, rng, 500000, 0.3, 0.0005, tl, 80)
        if c5_cur < best_c5:
            best_h, best_c5, best_n = h_cur.copy(), c5_cur, n

    # Phase 5: Random restarts with local search
    if tl() > 120:
        for restart in range(10):
            if tl() < 60:
                break
            # Mix of random and structured inits
            h_r = structured_init(n, rng)
            h_r, c5_r = local_search(h_r, n, rng, 100000, 0.5, 0.001, tl, 60)
            if c5_r < best_c5:
                best_h, best_c5, best_n = h_r.copy(), c5_r, n

    # Phase 6: Higher resolution
    if tl() > 180:
        for res_mult in [2, 3]:
            if tl() < 120:
                break
            n2 = best_n * res_mult
            ts_old = np.linspace(0, 2, best_n, endpoint=False)
            ts_new = np.linspace(0, 2, n2, endpoint=False)
            h2 = np.interp(ts_new, ts_old, best_h)
            h2 = project_h(h2, n2)
            c5_2 = compute_c5(h2, n2)

            # Optimize at higher res
            if tl() > 120:
                h2, c5_2 = local_search(h2, n2, rng, 200000, 0.3, 0.001, tl, 60)
            if tl() > 80:
                h_nm, c5_nm = nelder_mead_optimize(h2.copy(), n2, tl, 40)
                if c5_nm < c5_2:
                    h2, c5_2 = h_nm, c5_nm

            if c5_2 < best_c5:
                best_h, best_c5, best_n = h2.copy(), c5_2, n2

    # Phase 7: Final polishing
    if tl() > 60:
        h_final, c5_final = local_search(best_h.copy(), best_n, rng, 300000, 0.1, 0.0003, tl, 30)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    # Ensure validity
    best_h = project_h(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
