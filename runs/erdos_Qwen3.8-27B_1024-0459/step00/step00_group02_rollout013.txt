    vals_lo = A + B * x_lo
    k_dominant = np.argmax(vals_lo)

        return x_lo

    x = x_lo

        k_max = np.argmax(vals)

        # (A[k_max] - A[j]) / (B[j] - B[k_max]) for B[j] > B[k_max]
