    fm = compute_c5(h_minus, n)
    g_fd = (fp - fm) / (2.0 * eps)
    return abs(g_analytic[idx] - g_fd)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n_parent = len(initial_h_values)
    h0 = project_constrained(np.array(initial_h_values, dtype=float), n_parent)
    c5_0 = compute_c5(h0, n_parent)

    best_h = h0.copy()
    best_c5 = c5_0
    best_n = n_parent

    # Phase 1: Projected gradient descent with smooth max at parent resolution
    n = best_n
    h = best_h.copy()
    c5 = best_c5
    T = 25.0
    lr = 0.015
    patience = 0
    max_p1 = 8000

    for it in range(max_p1):
        if time.time() - t0 > budget_s * 0.40:
            break

        g = gradient_c5_smoothed(h, n, T)
        g_proj = g - np.mean(g)

        # Adaptive line search
        improved = False
        for am in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * am
            h_trial = h - step * g_proj
            h_trial = project_constrained(h_trial, n)
            c5_trial = compute_c5(h_trial, n)
            if c5_trial < c5 - 1e-9:
                h = h_trial
                c5 = c5_trial
                improved = True
                patience = 0
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
                    best_n = n
                break

        if not improved:
            patience += 1
            if patience % 50 == 0:
                lr *= 0.9
                T = max(5.0, T * 0.95)
            if patience > 500:
                # Escape: random perturbation
                h = h + rng.normal(0, 0.08, n)
                h = project_constrained(h, n)
                c5 = compute_c5(h, n)
                patience = 0
                lr = 0.02
                T = 25.0
            if patience > 2000:
                break

        # Occasionally reduce T to focus on worst
        if it % 500 == 499:
            T = max(10.0, T * 0.9)

    # Phase 2: Multiple random restarts with short gradient descent
    if time.time() - t0 < budget_s * 0.65:
        n_restarts = 8
        for r in range(n_restarts):
            if time.time() - t0 > budget_s * 0.65:
                break
            scale = 0.05 + 0.1 * rng.random()
            h = best_h + rng.normal(0, scale, best_n)
            h = project_constrained(h, best_n)
            c5 = compute_c5(h, best_n)

            for it in range(800):
                if time.time() - t0 > budget_s * 0.65:
                    break
                g = gradient_c5_smoothed(h, best_n, 25.0)
                g_proj = g - np.mean(g)
                h_trial = h - 0.01 * g_proj
                h_trial = project_constrained(h_trial, best_n)
                c5_trial = compute_c5(h_trial, best_n)
                if c5_trial < c5 - 1e-9:
                    h = h_trial
                    c5 = c5_trial
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5
                        best_n = best_n

    # Phase 3: Coordinate descent fine-tuning
    if time.time() - t0 < budget_s * 0.85:
        h = best_h.copy()
        c5 = best_c5
        n = best_n
        max_cd = 5000
        for it in range(max_cd):
            if time.time() - t0 > budget_s * 0.87:
                break
            idx = it % n
            improved = False
            for delta in [0.004, 0.008, 0.015, 0.03, -0.004, -0.008, -0.015, -0.03]:
                h_trial = h.copy()
                h_trial[idx] += delta
                h_trial = project_constrained(h_trial, n)
                c5_trial = compute_c5(h_trial, n)
                if c5_trial < c5 - 1e-9:
                    h = h_trial
                    c5 = c5_trial
                    improved = True
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5
                    break
            if not improved and it % 200 == 199:
                # Small random kick
                kick_idx = rng.randint(n)
                h[kick_idx] += rng.choice([-0.02, 0.02])
                h = project_constrained(h, n)
                c5 = compute_c5(h, n)

    # Phase 4: Try slightly different resolutions
    if time.time() - t0 < budget_s * 0.92:
        for n_try in [n_parent + 10, n_parent - 10, n_parent + 20]:
            if time.time() - t0 > budget_s * 0.92:
                break
            if n_try < 30:
                continue
            # Upsample/downsample best solution
            x_old = np.linspace(0, 2, best_n, endpoint=False) + 1.0 / best_n
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_try = np.interp(x_new, x_old, best_h, period=2.0)
            h_try = project_constrained(h_try, n_try)
            c5_try = compute_c5(h_try, n_try)

            # Short gradient descent
            T2 = 20.0
            for it in range(400):
                if time.time() - t0 > budget_s * 0.93:
                    break
                g = gradient_c5_smoothed(h_try, n_try, T2)
                g_proj = g - np.mean(g)
                h_trial = h_try - 0.01 * g_proj
                h_trial = project_constrained(h_trial, n_try)
                c5_trial = compute_c5(h_trial, n_try)
                if c5_trial < c5_try - 1e-9:
                    h_try = h_trial
                    c5_try = c5_trial

            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
                best_n = n_try

    # Final validation
    best_h = project_constrained(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    # Ensure valid
    if np.min(best_h) < -1e-9 or np.max(best_h) > 1.0 + 1e-9:
        best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = best_n / 2.0
    if abs(np.sum(best_h) - target_sum) > 1.0:
        best_h = project_constrained(best_h, best_n)
        best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
