    # Scale to meet constraint
    if abs(s - target) > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        # Second pass if needed
        s2 = np.sum(h)

            if diff > 0:

                cnt = np.sum(mask)

            else:
                mask = h > 0.0
                cnt = np.sum(mask)
                if cnt > 0:
                    h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

def binary_local_search(h, n, max_sweeps=100, time_limit=60):

    start = time.time()
    c5, _, _ = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0
        best_swap = None
        
        # Try all swaps (this is O(n²) which for n=70 is 4900)

            for j in zeros:
                if i == j:

                h[i] = 0.0
                h[j] = 1.0
                c5_new, _, _ = compute_c5(h)
                delta = c5_new - best_c5
                if delta < best_delta - 1e-12:

                    best_swap = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_swap is not None:

            h[j] = 1.0
            best_c5 += best_delta
            improved = True

            no_improve = 0
        else:
            no_improve += 1
            # Try random perturbation to escape
            if rng is not None and no_improve < 3:
                for _ in range(5):
                    o = rng.choice(len(ones))
                    z = rng.choice(len(zeros))
                    h[ones[o]] = 0.0
                    h[zeros[z]] = 1.0
                    c = compute_c5(h)
                    h[ones[o]] = 1.0
                    h[zeros[z]] = 0.0
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
                        no_improve = 0
                        break
    
    return best_h, best_c5

def projected_gradient(h, n, time_limit=60.0, rng=None):
    """Projected subgradient method for non-binary h"""
    t0 = time.time()
    h = project_feasible(h.copy(), n)
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    dx = 2.0 / n
    lr = 0.05  # initial learning rate
    
    for step in range(10000):
        if time.time() - t0 > time_limit:
            break
        
        # Compute correlation and find maximizing lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        
        # Compute gradient of f_lag at h
        # f_lag = sum_i h[i+lag]*(1-h[i]) for valid i
        # df/dh[j] = (1-h[j-lag]) if j-lag valid - h[j+lag] if j+lag valid
        grad = np.zeros(n)
        
        # For the term (1 - h[j-lag]):
        if lag >= 0:
            # j-lag valid when j >= lag, j-lag < n → j < n+lag
            # Also need j-lag >= 0 and j-lag <= n-1
            # So j ranges from lag to n-1 (since j < n and j-lag >= 0 means j >= lag)
            # And j-lag <= n-1 means j <= n-1+lag (always true since j < n and lag >= 0)
            grad[lag:n] += 1.0
            grad[lag:n] -= h[0:n-lag]  # h[j-lag] for j in [lag, n-1]
        else:
            # j-lag valid when j >= 0 and j-lag < n → j < n+lag = n-|lag|
            # Also j-lag >= 0 means j >= lag (but lag < 0, so j >= 0 is enough)
            # Wait: lag < 0, so j-lag = j+|lag|
            # Valid when 0 <= j+|lag| <= n-1, i.e., -|lag| <= j <= n-1-|lag|
            # Since j >= 0: 0 <= j <= n-1-|lag|
            neg_lag = -lag
            grad[0:n-neg_lag] += 1.0
            grad[0:n-neg_lag] -= h[neg_lag:n]  # h[j+neg_lag] for j in [0, n-1-neg_lag]
        
        # For the term -h[j+lag]:
        if lag >= 0:
            # j+lag valid when 0 <= j+lag <= n-1, i.e., -lag <= j <= n-1-lag
            # j >= 0: 0 <= j <= n-1-lag
            grad[0:n-lag] -= h[lag:n]  # h[j+lag] for j in [0, n-1-lag]
        else:
            # j+lag = j-|lag| valid when 0 <= j-|lag| <= n-1, i.e., |lag| <= j <= n-1+|lag|
            # j < n: |lag| <= j <= n-1
            neg_lag = -lag
            grad[neg_lag:n] -= h[0:n-neg_lag]  # h[j-neg_lag] for j in [neg_lag, n-1]
