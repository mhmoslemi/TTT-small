    """p is the softmax weights (length 2n-1) for each lag."""
    dx = 2.0 / n

    # term1[j] = sum_i p[i] * (1-h[j-lags[i]]) for valid j-lags[i] in [0,n-1]
    # term2[j] = sum_i p[i] * h[j+lags[i]] for valid j+lags[i] in [0,n-1]
    
    # Vectorize: create index arrays
    # For term1: j - lags[i] is valid when 0 <= j - (i-(n-1)) <= n-1
    # i.e., j <= i <= j + n - 1 (and 0<=i<=2n-2)
    
    # I'll compute using for-loop over j with vectorized inner product
    # For n=800, this is 800 iterations of ~800-element dot products
    
    one_minus_h = 1.0 - h
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    
    for j in range(n):
        # term1: i ranges from max(0,j) to min(2n-2, j+n-1)
        i_lo = max(0, j)

            h_idx = j + n - 1 - idx  # = j - lags[idx]
            term1[j] = np.dot(p[i_lo:i_hi+1], one_minus_h[h_idx])
        
        # term2: j + lags[i] = j + i - (n-1) valid when 0 <= j+i-(n-1) <= n-1
        # => (n-1)-j <= i <= 2(n-1)-j, and 0<=i<=2n-2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(x, target_sum):
    """Project x onto {0<=xi<=1, sum(x)=target_sum}."""
    x = np.array(x, dtype=float)
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    if abs(s - target_sum) < 1e-12:
        return x
    # Binary search for offset
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if np.sum(y) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, tau):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx  # the actual C5 contributions
    
    # Smooth max: (1/tau) * log(sum exp(tau * f))
    m = np.max(tau * f)
    exp_f = np.exp(tau * f - m)
    sum_exp = np.sum(exp_f)
    c5_smooth = (m + np.log(sum_exp)) / tau
    
    # Softmax weights
    p = exp_f / sum_exp
    
    # Gradient: sum_k p_k * (df_k/dh)
    # df_k/dh[j] = dx * (d corr_k/dh[j])
    # d corr_k/dh[j] = (1-h[j-lag_k]) * [valid1] - h[j+lag_k] * [valid2]
    # where lag_k = k - (n-1)
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for j in range(n):
        # Part 1: sum_i p[i] * (1-h[j-(i-(n-1))]) for 0<=j-(i-(n-1))<=n-1
        # j - i + n - 1 in [0, n-1] => j <= i <= j + n - 1
        i_lo = max(0, j)
        i_hi = min(2*n-2, j + n - 1)
        if i_lo <= i_hi:
            idx = i_lo + np.arange(i_hi - i_lo + 1)
            h_idx = j + n - 1 - idx
            grad[j] += np.dot(p[i_lo:i_hi+1], one_minus_h[h_idx])
        
        # Part 2: -sum_i p[i] * h[j+(i-(n-1))] for 0<=j+(i-(n-1))<=n-1
        # j + i - n + 1 in [0, n-1] => n-1-j <= i <= 2n-2-j
        i_lo2 = max(0, n-1-j)
        i_hi2 = min(2*n-2, 2*n-2-j)
        if i_lo2 <= i_hi2:
            idx2 = i_lo2 + np.arange(i_hi2 - i_lo2 + 1)
            h_idx2 = j + idx2 - (n-1)
            grad[j] -= np.dot(p[i_lo2:i_hi2+1], h[h_idx2])
    
    grad *= dx
    return c5_smooth, grad
