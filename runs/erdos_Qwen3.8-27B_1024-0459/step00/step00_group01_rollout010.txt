def project_feasible(h, n):
    """Project h onto {0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h + lam, 0.0, 1.0)
    # Fix rounding
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute the difference
        result[np.argmax(result)] += diff  # rough fix
        # Actually let's just re-project
    return result

def compute_grad_at_maxlag(h, corr_vals, n):
    """Compute gradient of the active (max) constraint at h"""
    dx = 2.0 / n
    k_star = np.argmax(corr_vals)
    # In full mode, the center is at index n-1
    lag = k_star - (n - 1)  # actual shift
    
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        # h[i] appears in h[n+lag] when n+lag = i, i.e., n = i-lag
        n_idx = i - lag
        if 0 <= n_idx < n:
            g += (1.0 - h[n_idx])
        # h[i] appears in h[n] when n = i
        n_plus_lag = i + lag
        if 0 <= n_plus_lag < n:
            g -= h[n_plus_lag]
        grad[i] = dx * g
    return grad

def binary_local_search(h_bin, n, max_iter=100000, seed=42):
    """Swap-based local search for binary h with exactly n/2 ones"""
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]

    best_c5 = c5
    best_h = h.copy()
    
    for it in range(max_iter):
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        
        new_c5, _ = evaluate_overlap(h, n)
        
        if new_c5 <= c5:

            ones = np.where(h == 1.0)[0]

                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    c5, _ = evaluate_overlap(h, n)
    best_c5 = c5
    best_h = h.copy()
