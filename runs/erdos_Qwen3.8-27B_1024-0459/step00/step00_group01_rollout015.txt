def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = 0.5 * n
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample_linear(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]))
    h_new = h_new[:n_new]
    return project_feasible(h_new, n_new)

def downsample_avg(h, n_new):
    n_old = len(h)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        start = int(i * n_old / n_new)
        end = int((i + 1) * n_old / n_new)
        end = max(end, start + 1)
        h_new[i] = h[start:end].mean()
    return project_feasible(h_new, n_new)

def nm_optimize(h0, n, max_iter=3000, t0=None, deadline=None):
    def obj(x):
        h = project_feasible(x, n)
        return compute_c5(h)
    best_h = h0.copy()
    best_val = obj(best_h)
    x = h0.copy()
    for it in range(max_iter):
        if t0 is not None and time.time() > deadline:
            break
        x_new, val = _nm_step(x, best_h, best_val, obj, n)
        if val < best_val:
            best_h = x_new.copy()
            best_val = val
        x = x_new
    return best_h, best_val

def _nm_step(x, best_h, best_val, obj, n):
    # Simple Nelder-Mead: try moving toward best reflection
    if np.allclose(x, best_h):
        # Initialize simplex direction
        direction = np.random.randn(n) * 0.05
        candidate = project_feasible(x + direction, n)
        return candidate, obj(candidate)
    # Move toward best with momentum
    alpha = 0.5
    candidate = project_feasible(best_h + alpha * (best_h - x) * 0.1 + np.random.randn(n) * 0.02, n)
    return candidate, obj(candidate)

def coordinate_descent(h0, n, max_rounds=200, t0=None, deadline=None):
    h = h0.copy()
    c5 = compute_c5(h)
    rng = np.random.RandomState(42)
    for rnd in range(max_rounds):
        if t0 is not None and time.time() > deadline:
            break
        idx = rng.permutation(n)
        improved = False
        for i in idx:
            old = h[i]
            for step in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01]:
                hv = h.copy()
                hv[i] = old + step
                if hv[i] < 0 or hv[i] > 1:
                    continue
                hv = project_feasible(hv, n)
                v = compute_c5(hv)
                if v < c5 - 1e-13:
                    h = hv
                    c5 = v
                    improved = True
                    break
        if not improved:
            # Try larger steps
            large_improved = False
            for i in idx[:n//2]:
                old = h[i]
                for step in [0.3, -0.3, 0.5, -0.5, 0.2, -0.2]:
                    hv = h.copy()
                    hv[i] = old + step
                    if hv[i] < 0 or hv[i] > 1:
                        continue
                    hv = project_feasible(hv, n)
                    v = compute_c5(hv)
                    if v < c5 - 1e-13:
                        h = hv
                        c5 = v
                        large_improved = True
                        break
                if large_improved:
                    break
            if not large_improved:
                break
    return h, c5

def binary_swap_search(h0, n, max_iters=50000, t0=None, deadline=None):
    h = (h0 > 0.5).astype(float)
    num_ones = int(h.sum())
    target = n // 2
    if n % 2 == 0:
        target = n // 2
    else:
        target = n // 2
    if num_ones > target:
        ones = np.where(h == 1)[0]
        h[ones[num_ones-target:]] = 0
    elif num_ones < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target-num_ones]] = 1
    c5 = compute_c5(h)
    rng = random
    stall = 0
    for it in range(max_iters):
        if t0 is not None and time.time() > deadline:
            break
        i = rng.randrange(n)
        j = rng.randrange(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 < c5 - 1e-14:
            c5 = new_c5
            stall = 0
        else:
            h[i], h[j] = h[j], h[i]
            stall += 1
        if stall > 5000:
            break
    return h, c5

def smooth_start(n, freq, phase, amplitude, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amplitude * np.sin(2*np.pi*freq*x + phase)
    return project_feasible(h, n)

def structured_start(n, num_blocks, rng):
    h = np.zeros(n)
    block_size = n // num_blocks
    block_vals = rng.rand(num_blocks)
    for b in range(num_blocks):
        start = b * block_size
        end = (b+1) * block_size
        h[start:end] = block_vals[b]
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.92  # Leave 8% buffer

    # Get initial solution
    init_h = None
    if 'initial_h_values' in dir() and initial_h_values is not None:
        init_h = np.array(initial_h_values, dtype=float)
    
    best_h_global = None
    best_c5_global = float('inf')
    best_n_global = 0

    def update_best(h, n):
        nonlocal best_h_global, best_c5_global, best_n_global
        c5 = compute_c5(h)
        if c5 < best_c5_global - 1e-12:
            best_h_global = h.copy()
            best_c5_global = c5
            best_n_global = n
            return True
        return False

    # Initialize from given values
    if init_h is not None:
        n0 = len(init_h)
        h_init = project_feasible(init_h.copy(), n0)
        update_best(h_init, n0)
        best_h_global = h_init.copy()
        best_c5_global = compute_c5(best_h_global)
        best_n_global = n0

    # Phase 1: Local optimization from initial (Nelder-Mead style + coordinate descent)
    if best_h_global is not None:
        n_op = best_n_global
        # Coordinate descent
        h_cd, c5_cd = coordinate_descent(best_h_global, n_op, max_rounds=100, t0=t0, deadline=deadline)
        if c5_cd < best_c5_global:
            best_h_global = h_cd.copy()
            best_c5_global = c5_cd
        # Binary search
        h_bin, c5_bin = binary_swap_search(best_h_global, n_op, max_iters=30000, t0=t0, deadline=deadline)
        if c5_bin < best_c5_global:
            best_h_global = h_bin.copy()
            best_c5_global = c5_bin
        # More coordinate descent from binary result
        h_cd2, c5_cd2 = coordinate_descent(best_h_global, n_op, max_rounds=100, t0=t0, deadline=deadline)
        if c5_cd2 < best_c5_global:
            best_h_global = h_cd2.copy()
            best_c5_global = c5_cd2

    # Phase 2: Try multiple starting points at various resolutions
    resolutions = [47, 60, 80, 100, 120, 160]
    rng_local = np.random.RandomState(seed + 100)

    for n_try in resolutions:
        if time.time() > deadline:
            break
        # Time budget per resolution
        if n_try <= 60:
            alloc = 120.0
        elif n_try <= 100:
            alloc = 100.0
        elif n_try <= 120:
            alloc = 80.0
        else:
            alloc = 60.0
        phase_deadline = min(time.time() + alloc, deadline)

        # Try several starting points
        for s in range(6):
            if time.time() > phase_deadline:
                break
            if s == 0:
                # Upsample from best
                h_start = upsample_linear(best_h_global, n_try)
            elif s == 1:
                # Sine wave
                h_start = smooth_start(n_try, rng_local.randint(1, 5), rng_local.rand()*2*np.pi,
                                       rng_local.uniform(0.2, 0.45), rng_local)
            elif s == 2:
                # Structured (random blocks)
                h_start = structured_start(n_try, rng_local.randint(4, 12), rng_local)
            elif s == 3:
                # High frequency oscillation
                h_start = smooth_start(n_try, rng_local.randint(3, 8), 0.0,
                                       rng_local.uniform(0.3, 0.5), rng_local)
            elif s == 4:
                # Random
                h_start = project_feasible(rng_local.rand(n_try), n_try)
            else:
                # Bezier-like smooth
                pts = 4 + rng_local.randint(0, 4)
                x_pts = np.sort(rng_local.rand(pts) * 2.0)
                y_pts = 0.3 + 0.4 * rng_local.rand(pts)
                x = np.linspace(0, 2, n_try, endpoint=False)
                h_start = np.interp(x, x_pts, y_pts)
                h_start = project_feasible(h_start, n_try)

            # Coordinate descent
            h_opt, c5_opt = coordinate_descent(h_start, n_try, max_rounds=80, t0=t0, deadline=phase_deadline)
            # Binary refinement
            h_bin2, c5_bin2 = binary_swap_search(h_opt, n_try, max_iters=20000, t0=t0, deadline=phase_deadline)
            # Final coordinate descent
            h_final, c5_final = coordinate_descent(h_bin2, n_try, max_rounds=50, t0=t0, deadline=phase_deadline)
            if c5_final < best_c5_global:
                best_h_global = h_final.copy()
                best_c5_global = c5_final
                best_n_global = n_try

    # Phase 3: Intensive refinement of best solution
    if best_h_global is not None:
        n_best = best_n_global
        refine_deadline = min(time.time() + 200.0, deadline)

        # Multiple coordinate descent passes
        for rep in range(10):
            if time.time() > refine_deadline:
                break
            h_ref, c5_ref = coordinate_descent(best_h_global, n_best, max_rounds=150, t0=t0, deadline=refine_deadline)
            if c5_ref < best_c5_global:
                best_h_global = h_ref.copy()
                best_c5_global = c5_ref

        # Binary search refinement
        h_b, c5_b = binary_swap_search(best_h_global, n_best, max_iters=50000, t0=t0, deadline=refine_deadline)
        if c5_b < best_c5_global:
            best_h_global = h_b.copy()
            best_c5_global = c5_b

        # Coordinate descent after binary
        h_f, c5_f = coordinate_descent(best_h_global, n_best, max_rounds=100, t0=t0, deadline=refine_deadline)
        if c5_f < best_c5_global:
            best_h_global = h_f.copy()
            best_c5_global = c5_f

    # Phase 4: Try higher resolution with best solution upsampled
    if time.time() < deadline:
        for n_hi in [200, 300, 400]:
            if time.time() > deadline:
                break
            h_up = upsample_linear(best_h_global, n_hi)
            h_opt, c5_opt = coordinate_descent(h_up, n_hi, max_rounds=40, t0=t0, deadline=min(time.time()+120, deadline))
            h_bin3, c5_bin3 = binary_swap_search(h_opt, n_hi, max_iters=30000, t0=t0, deadline=min(time.time()+80, deadline))
            h_final2, c5_final2 = coordinate_descent(h_bin3, n_hi, max_rounds=30, t0=t0, deadline=deadline)
            if c5_final2 < best_c5_global:
                best_h_global = h_final2.copy()
                best_c5_global = c5_final2
                best_n_global = n_hi

    # Phase 5: Final subgradient-inspired polishing
    if time.time() < deadline and best_h_global is not None:
        n_fin = best_n_global
        h_pol = best_h_global.copy()
        c5_pol = best_c5_global
        dx = 2.0 / n_fin
        lr = 0.005
        for s in range(5000):
            if time.time() > deadline:
                break
            overlaps = np.correlate(h_pol, 1.0 - h_pol, mode='full') * dx
            max_idx = int(np.argmax(overlaps))
            lag = max_idx - (n_fin - 1)
            grad = np.zeros(n_fin)
            if lag == 0:
                grad[:] = 1.0 - 2.0 * h_pol
            elif lag > 0:
                grad[:n_fin-lag] += (1.0 - h_pol[lag:])
                grad[lag:] += -h_pol[:n_fin-lag]
            else:
                m = -lag
                grad[m:] += (1.0 - h_pol[:n_fin-m])
                grad[:n_fin-m] += -h_pol[m:]
            grad *= dx
            h_new = h_pol - lr * grad
            h_new = project_feasible(h_new, n_fin)
            c5_new = compute_c5(h_new)
            if c5_new < c5_pol - 1e-14:
                h_pol = h_new
                c5_pol = c5_new
                lr = min(lr * 1.05, 0.05)
            else:
                lr *= 0.95
        if c5_pol < best_c5_global:
            best_h_global = h_pol.copy()
            best_c5_global = c5_pol

    # Final coordinate descent polish
    if best_h_global is not None and time.time() < deadline:
        h_last, c5_last = coordinate_descent(best_h_global, best_n_global, max_rounds=50, t0=t0, deadline=deadline)
        if c5_last < best_c5_global:
            best_h_global = h_last
            best_c5_global = c5_last

    # Ensure feasibility
    best_h_global = project_feasible(best_h_global, best_n_global)
    best_c5_global = compute_c5(best_h_global)

    return best_h_global, best_c5_global, best_n_global
