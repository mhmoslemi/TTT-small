                ja = k - a

                    d += 2.0 * h[ja] - 1.0
                jb = k - b
                if 0 <= jb < n and jb != b:
                    d += 1.0 - 2.0 * h[jb]
                c_raw[k] = c[k] + d
            c = c_raw
            h[a] = 0
            h[b] = 1
            ones_list.remove(a)
            zeros_list.remove(b)
            ones_list.append(b)
            zeros_list.append(a)
    
    return best_h, best_c5

def local_search_binary_vectorized(h_init, max_time, seed=42):
    """Faster version using numpy vectorization for delta computation."""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy().astype(float)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c5 = float(np.max(c) * dx)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    t0 = time.time()
    stagnation = 0
    
    for it in range(10000000):
        if time.time() - t0 > max_time:
            break
        if len(ones_list) < 1 or len(zeros_list) < 1:
            break
        
        a = ones_list[rng.randint(len(ones_list))]
        b = zeros_list[rng.randint(len(zeros_list))]
        
        # Vectorized delta computation
        # For positions k where 0<=k-a<n and k!=2a: delta += 2*h[k-a]-1
        # For positions k where 0<=k-b<n and k!=2b: delta += 1-2*h[k-b]
        delta = np.zeros(2 * n - 1)
        
        lo_a = a
        hi_a = a + n - 1
        if lo_a < 2*n-1:
            seg = np.arange(lo_a, min(hi_a+1, 2*n-1))
            ja = seg - a
            h_ja = h[ja]
            d_a = 2.0 * h_ja - 1.0
            # Remove diagonal
            diag_a = np.where(seg == 2*a)[0]
            if len(diag_a) > 0:
                d_a[diag_a] = 0.0
            delta[seg] += d_a
        
        lo_b = b
        hi_b = b + n - 1
        if lo_b < 2*n-1:
            seg = np.arange(lo_b, min(hi_b+1, 2*n-1))
            jb = seg - b
            h_jb = h[jb]
            d_b = 1.0 - 2.0 * h_jb
            diag_b = np.where(seg == 2*b)[0]
            if len(diag_b) > 0:
                d_b[diag_b] = 0.0
            delta[seg] += d_b
        
        new_c = c + delta
        new_max = np.max(new_c)
        new_c5 = new_max * dx
        
        if new_c5 < best_c5 - 1e-12:
            c = new_c
            h[a] = 0
            h[b] = 1
            ones_list.remove(a)
            zeros_list.remove(b)
            ones_list.append(b)
            zeros_list.append(a)
            best_c5 = new_c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 20000:
                # Random restart within current solution
                for _ in range(5):
                    ra = ones_list[rng.randint(len(ones_list))]
                    rb = zeros_list[rng.randint(len(zeros_list))]
                    delta2 = np.zeros(2 * n - 1)
                    lo_a2 = ra
                    hi_a2 = ra + n - 1
                    if lo_a2 < 2*n-1:
                        seg2 = np.arange(lo_a2, min(hi_a2+1, 2*n-1))
                        ja2 = seg2 - ra
                        d_a2 = 2.0 * h[ja2] - 1.0
                        diag = np.where(seg2 == 2*ra)[0]
                        if len(diag) > 0:
                            d_a2[diag] = 0.0
                        delta2[seg2] += d_a2
                    lo_b2 = rb
                    hi_b2 = rb + n - 1
                    if lo_b2 < 2*n-1:
                        seg2 = np.arange(lo_b2, min(hi_b2+1, 2*n-1))
                        jb2 = seg2 - rb
                        d_b2 = 1.0 - 2.0 * h[jb2]
                        diag = np.where(seg2 == 2*rb)[0]
                        if len(diag) > 0:
                            d_b2[diag] = 0.0
                        delta2[seg2] += d_b2
                    c2 = c + delta2
                    if np.max(c2) * dx < best_c5:
                        c = c2
                        h[ra] = 0
                        h[rb] = 1
                        ones_list.remove(ra)
                        zeros_list.remove(rb)
                        ones_list.append(rb)
                        zeros_list.append(ra)
                        best_c5 = np.max(c) * dx
                        best_h = h.copy()
                        stagnation = 0
                        break
                if stagnation > 50000:
                    break
    
    return best_h, best_c5

def projected_gradient_optimize(h_init, max_iters, seed=42):
    """Projected gradient descent with smooth max."""
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for beta in [30.0, 100.0, 300.0, 1000.0, 3000.0]:
        alpha = 2.0 / (beta * 4.0 + 1.0)
        for it in range(max_iters):
            obj, grad = smooth_obj_grad(h, beta)
            h_new = h - alpha * grad
            h_new = project_step(h_new, n, target_sum)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            h = h_new
            # Adaptive step
            if it % 500 == 499:
                obj2, grad2 = smooth_obj_grad(h, beta)
                if np.max(np.abs(grad2)) < 1e-6:
                    break
    
    return best_h, best_c5

def greedy_binary_init(n, seed=42):
    """Greedily construct a binary solution."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n // 2
    c = np.zeros(2 * n - 1)  # correlation values (all zero initially)
    
    for step in range(target):
        best_pos = -1
        best_max = 1e9
        for pos in range(n):
            # If we set h[pos] = 1, new c[k] changes
            # New terms: c_new[k] = c[k] + (1-h[k-pos]) for valid k-pos, and h[k-pos]*(1-0) for valid
            # Actually for a 0-1 h, adding h[pos]=1:
            # c_new[k] = c[k] + (1-h[k-pos])*[valid] + h[k-pos]*[k-pos valid, k-pos!=pos]
            # Since h is 0-1: = c[k] + (1-h[k-pos]) + h[k-pos] (when valid and not diagonal)
            # = c[k] + 1 (when valid and not diagonal) or c[k] + 0 (diagonal or invalid)
            # Wait: (1-h[k-pos]) + h[k-pos] = 1. So delta is +1 for all valid non-diagonal k.
            # Diagonal (k=2pos): term is h[pos]*(1-h[pos]) = 0, no change.
            
            # So new c[k] = c[k] + 1 for k in [pos, pos+n-1], k != 2pos
            # New max = max(c) considering the +1 additions
            
            # Quick check: new_max = max(c[k]+1 for k in [pos,pos+n-1], k!=2pos, c[k] for others)
            lo = pos
            hi = min(pos + n - 1, 2 * n - 2)
            if lo < 2*n-1:
                seg_c = c[lo:hi+1].copy()
                diag_idx = 2 * pos - lo
                if 0 <= diag_idx < len(seg_c):
                    seg_c[diag_idx] = seg_c[diag_idx] - 1e9  # exclude diagonal
                max_added = np.max(seg_c) + 1.0
            else:
                max_added = 0.0
            max_rest = c[:lo].max() if lo > 0 else -1e9
            max_rest2 = c[hi+1:].max() if hi+1 < 2*n-1 else -1e9
            cur_max = max(max_added, max_rest, max_rest2)
            
            if cur_max < best_max:
                best_max = cur_max
                best_pos = pos
        
        # Place 1 at best_pos
        h[best_pos] = 1.0
        lo = best_pos
        hi = min(best_pos + n - 1, 2 * n - 2)
        if lo < 2*n-1:
            c[lo:hi+1] += 1.0
            diag_idx = 2 * best_pos
            if 0 <= diag_idx < 2*n-1:
                c[diag_idx] -= 1.0  # undo the diagonal
    
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5.0
    
    best_overall_c5 = 1.0
    best_overall_h = None
    best_overall_n = 95
    
    # Phase 1: Optimize the parent construction (n=95) with projected gradient
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        target_sum0 = n0 / 2.0
        h0 = project_step(h0, n0, target_sum0)
        
        # Quick projected gradient
        time_remaining = budget_s - 60
        h_pg, c5_pg = projected_gradient_optimize(h0, max_iters=2000, seed=seed)
        if c5_pg < best_overall_c5:
            best_overall_c5 = c5_pg
            best_overall_h = h_pg
            best_overall_n = n0
        
        # Binarize the result and do local search
        if time.time() < deadline:
            for n_try in [n0 - 1, n0 + 1]:  # try even n neighbors
                if n_try < 4:
                    continue
                # Create initial binary for n_try
                # Interpolate h_pg to n_try points
                x_old = np.linspace(0, 2, n0)
                x_new = np.linspace(0, 2, n_try)
                h_interp = np.interp(x_new, x_old, h_pg)
                h_bin = binarize(h_interp, n_try)
                
                ls_time = min(60.0, (deadline - time.time()) * 0.3)
                h_ls, c5_ls = local_search_binary_vectorized(h_bin, ls_time, seed=seed)
                if c5_ls < best_overall_c5:
                    best_overall_c5 = c5_ls
                    best_overall_h = h_ls
                    best_overall_n = n_try
    
    # Phase 2: Try greedy + local search for various n
    for n_try in [94, 96, 98, 100, 92]:
        if time.time() > deadline:
            break
        h_greedy = greedy_binary_init(n_try, seed=seed)
        ls_time = min(80.0, (deadline - time.time()) * 0.25)
        if ls_time < 5:
            break
        h_ls, c5_ls = local_search_binary_vectorized(h_greedy, ls_time, seed=seed)
        if c5_ls < best_overall_c5:
            best_overall_c5 = c5_ls
            best_overall_h = h_ls
            best_overall_n = n_try
    
    # Phase 3: Try random binary + local search
    for n_try in [94, 96, 95, 98]:
        if time.time() > deadline:
            break
        rng = np.random.RandomState(seed + n_try)
        h_rand = np.zeros(n_try)
        pos = rng.choice(n_try, size=n_try // 2, replace=False)
        h_rand[pos] = 1.0
        ls_time = min(40.0, (deadline - time.time()) * 0.15)
        if ls_time < 3:
            break
        h_ls, c5_ls = local_search_binary_vectorized(h_rand, ls_time, seed=seed + 100)
        if c5_ls < best_overall_c5:
            best_overall_c5 = c5_ls
            best_overall_h = h_ls
            best_overall_n = n_try
    
    # Phase 4: More aggressive local search on best binary solution
    if best_overall_h is not None and time.time() < deadline - 10:
        n_best = best_overall_n
        # Check if it's binary
        if np.allclose(best_overall_h, np.round(best_overall_h)):
            ls_time = min(120.0, deadline - time.time() - 5)
            if ls_time > 10:
                h_ls2, c5_ls2 = local_search_binary_vectorized(
                    best_overall_h, ls_time, seed=seed + 999)
                if c5_ls2 < best_overall_c5:
                    best_overall_c5 = c5_ls2
                    best_overall_h = h_ls2
    
    # Phase 5: Try continuous optimization on the best binary (as soft starting point)
    if best_overall_h is not None and time.time() < deadline - 30:
        n_b = len(best_overall_h)
        h_soft = best_overall_h.astype(float).copy()
        # Add small noise
        rng = np.random.RandomState(seed + 555)
        h_soft += rng.uniform(-0.1, 0.1, n_b)
        h_soft = np.clip(h_soft, 0, 1)
        h_soft = project_step(h_soft, n_b, n_b / 2.0)
        
        remaining = deadline - time.time() - 5
        if remaining > 20:
            h_pg2, c5_pg2 = projected_gradient_optimize(h_soft, max_iters=3000, seed=seed+777)
            if c5_pg2 < best_overall_c5:
                best_overall_c5 = c5_pg2
                best_overall_h = h_pg2
                best_overall_n = n_b
    
    # Final: ensure feasibility
    if best_overall_h is None:
        # Fallback: return parent
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        best_overall_h = project_step(h0, n0, n0 / 2.0)
        best_overall_c5 = compute_c5(best_overall_h)
        best_overall_n = n0
    else:
        n_f = len(best_overall_h)
        best_overall_h = np.clip(best_overall_h, 0.0, 1.0)
        # Ensure sum constraint
        s = np.sum(best_overall_h)
        target = n_f / 2.0
        if abs(s - target) > 1e-6:
            best_overall_h = project_step(best_overall_h, n_f, target)
        best_overall_c5 = compute_c5(best_overall_h)
    
    return best_overall_h, best_overall_c5, len(best_overall_h)
