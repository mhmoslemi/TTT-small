def smooth_max(x, tau):
    x_max = np.max(x)
    return float(x_max + tau * np.log(np.sum(np.exp((x - x_max) / tau))))

def continuous_obj(x, n, tau, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    obj = smooth_max(corr, tau)
    s = x.sum() - n / 2.0
    obj += penalty * s * s
    return obj

def project_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        step = diff / float(np.sum(mask))
        h[mask] += step
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14 and np.isfinite(diff):
        idx = np.where(np.isfinite(h))[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

    h0 = project_sum(h_init.copy(), n / 2.0)
    best_h = h0.copy()

    for tau in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:

            break
        try:

                          method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'maxfun': 500, 'disp': False})
            h_new = project_sum(res.x, n / 2.0)

                best_c5 = c5_new
                best_h = h_new.copy()

        except Exception:
            pass
    return best_h, best_c5

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, Exception):
        h_init = np.zeros(42)
        h_init[::2] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Greedy constructions at initial resolution
    remaining = deadline - time.time()
    if remaining > 30:
        for s in range(5):
            if time.time() > deadline - 10:
                break
            h_g, c5_g = greedy_construction(n_init, seed_offset=seed + s * 100)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    # Phase 2: Local search (random swaps)
    remaining = deadline - time.time()
    if remaining > 60:
        tl = min(150.0, remaining * 0.30)
        h_ls, c5_ls = local_search_binary(best_h, time_limit=tl)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 3: Directed local search
    remaining = deadline - time.time()
    if remaining > 60:
        tl = min(100.0, remaining * 0.20)
        h_ds, c5_ds = local_search_directed(best_h, time_limit=tl)
        if c5_ds < best_c5:
            best_c5 = c5_ds
            best_h = h_ds.copy()

    # Phase 4: Continuous optimization
    remaining = deadline - time.time()
    if remaining > 40:
        tl = min(80.0, remaining * 0.25)
        h_co, c5_co = continuous_optimize(best_h, time_limit=tl)
        if c5_co < best_c5:
            best_c5 = c5_co
            best_h = h_co.copy()

    # Phase 5: Another round of local search on best
    remaining = deadline - time.time()
    if remaining > 40:
        tl = min(80.0, remaining * 0.30)
        h_ls2, c5_ls2 = local_search_binary(best_h, time_limit=tl)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 6: Try higher resolutions
    for n_try in [44, 46, 48, 60, 84]:
        if n_try % 2 != 0:
            continue
        remaining = deadline - time.time()
        if remaining < 50:
            break
        h_interp = interpolate_to_size(best_h, n_try)
        h_bin = to_binary(h_interp)
        tl1 = min(40.0, remaining * 0.35)
        if tl1 < 10:
            break
        h_nr, c5_nr = local_search_binary(h_bin, time_limit=tl1)
        remaining = deadline - time.time()
        if remaining > 20:
            tl2 = min(20.0, remaining * 0.2)
            h_nc, c5_nc = continuous_optimize(h_nr, time_limit=tl2)
            if c5_nc < c5_nr:
                h_nr = h_nc
                c5_nr = c5_nc
        if c5_nr < best_c5:
            best_c5 = c5_nr
            best_h = h_nr.copy()

    # Phase 7: Final local search polish
    remaining = deadline - time.time()
    if remaining > 30:
        h_bin_final = to_binary(best_h)
        tl = min(remaining - 10.0, 80.0)
        if tl > 10:
            h_pf, c5_pf = local_search_binary(h_bin_final, time_limit=tl)
            if c5_pf < best_c5:
                best_c5 = c5_pf
                best_h = h_pf.copy()

    # Ensure validity
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
