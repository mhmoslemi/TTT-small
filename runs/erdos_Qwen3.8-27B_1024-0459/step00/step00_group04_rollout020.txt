    return float(np.max(corr) * dx)

def smooth_objective(h, dx, n, alpha):
    """Smooth approximation of the max overlap."""
    h_clip = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_clip, 1.0 - h_clip, mode="full") * dx
    # Log-sum-exp for numerical stability
    max_corr = np.max(corr)
    if max_corr < 0:
        return max_corr
    shifted = corr - max_corr
    log_sum = max_corr + np.log(np.sum(np.exp(alpha * shifted))) / alpha

    sum_val = np.sum(h_clip)
    target = n / 2.0
    penalty = 1000.0 * (sum_val - target) ** 2
    return log_sum + penalty

    h_clip = np.clip(h, 0.0, 1.0)

                    print(f"  L-BFGS-B error n={n_r} a={alpha}: {e}")
    
    # Phase 3: Final local polish at best resolution
    elapsed = time.time() - t0
    if elapsed < budget_s - 30 and best_n <= 336:
        dx = 2.0 / best_n
        remaining = budget_s - elapsed - 10
        if best_n <= 168:
            maxiter = min(500, int(remaining * 30))
        else:
            maxiter = min(200, int(remaining * 20))
        if maxiter >= 30:
            try:
                if best_n <= 168:
                    res = minimize(
                        objective_cobyla, best_h, args=(dx, best_n),
                        method='COBYLA',
                        bounds=[(0, 1)] * best_n,
                        options={'maxiter': maxiter, 'rhobeg': 0.03, 'catol': 1e-11}
                    )
                else:
                    res = minimize(
                        smooth_max_obj2, best_h, args=(dx, best_n, 300),
                        method='L-BFGS-B',
                        bounds=[(0, 1)] * best_n,
                        options={'maxiter': maxiter, 'ftol': 1e-13, 'gtol': 1e-8}
                    )
                h_opt = project_to_feasible(res.x, best_n)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    print(f"  Polish: C5={c5:.6f}")
            except:
                pass
    
    # Final output
    h_ret = project_to_feasible(best_h, best_n)
    dx_ret = 2.0 / best_n
    c5_ret = compute_c5(h_ret, dx_ret)
    print(f"FINAL: n={best_n}, C5={c5_ret:.6f}, elapsed={time.time()-t0:.1f}s")
    return (h_ret, c5_ret, best_n)
