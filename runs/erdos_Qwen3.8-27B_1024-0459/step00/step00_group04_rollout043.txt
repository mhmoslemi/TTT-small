    current_c5 = c5_of(h)
    best_local = h.copy()
    best_local_c5 = current_c5
    temp = init_temp
    no_improve = 0
    while time.time() - t_start < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = c5_of(h)
        delta = new_c5 - current_c5
        if delta <= 0 or (temp > 1e-12 and rng.random() < np.exp(-delta / temp)):
            current_c5 = new_c5
            if new_c5 < best_local_c5 - 1e-12:
                best_local_c5 = new_c5
                best_local = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        temp *= 0.99995
        if no_improve > 200000:
            break
    return best_local, best_local_c5

def interpolate_double(h):
    n = len(h)
    h2 = np.zeros(2 * n)
    h2[0::2] = h
    h2[1::2] = h
    return h2

def interpolate_double_split(h, rng):
    """Interpolate by splitting 1s into [1,0] or [0,1] randomly."""
    n = len(h)
    m = n // 2
    # Count ones
    n_ones = int(np.sum(h > 0.5))
    # We need n_ones ones in the new array of length 2n
    # Each original 1 becomes either [1,1], [1,0], or [0,1]
    # Each original 0 becomes [0,0]
    # Total ones in new = n_ones (from [1,0] or [0,1]) or 2*n_ones (from [1,1])
    # We want exactly n_ones ones in the new array (since new length is 2n, need 2n/2 = n ones)
    # So we need n ones total. Each [1,1] contributes 2, each [1,0]/[0,1] contributes 1, each [0,0] contributes 0.
    # If all 1s become [1,1]: total = 2*n_ones = n. Perfect!
    # So [1,1] for all ones gives exactly the right count.
    # For variety, let's mix: some [1,1], some [1,0]/[0,1]
    # If k ones become [1,1] and (n_ones - k) become single: total = 2k + (n_ones - k) = k + n_ones
    # We need k + n_ones = n, so k = n - n_ones = n/2 (since n_ones = n/2 for the parent)
    # So half the ones become [1,1] and half become single [1,0] or [0,1]
    h2 = np.zeros(2 * n)
    ones_pos = np.where(h > 0.5)[0]
    # Choose which ones become double
    if len(ones_pos) > 0:
        n_double = n - len(ones_pos)  # number that become [1,1]
        n_double = min(n_double, len(ones_pos))
        if n_double > 0:
            double_idx = rng.choice(len(ones_pos), size=n_double, replace=False)
        else:
            double_idx = np.array([], dtype=int)
        for di in double_idx:
            pos = ones_pos[di]
            h2[2*pos] = 1.0
            h2[2*pos+1] = 1.0
        for pos in ones_pos:
            idx_in = np.where(ones_pos == pos)[0][0]
            if idx_in not in double_idx:
                if rng.random() < 0.5:
                    h2[2*pos] = 1.0
                else:
                    h2[2*pos+1] = 1.0
    return h2

def continuous_optimize(h_init, n, time_limit, t_start, rng):
    """Try continuous optimization using smooth max."""
    x0 = h_init.astype(float).copy()
    # Ensure sum constraint
    target_sum = n / 2.0
    x0 = np.clip(x0, 0.0, 1.0)
    diff = target_sum - np.sum(x0)
    if abs(diff) > 1e-8:
        free = (x0 > 0.05) & (x0 < 0.95)
        if np.sum(free) > 0:
            x0[free] += diff / np.sum(free)
        else:
            x0 += diff / n
        x0 = np.clip(x0, 0.0, 1.0)
    
    eps = 0.005
    
    def smooth_obj(x):
        dx = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        max_c = np.max(c)
        # Smooth max
        shifted = c - max_c
        return eps * (max_c + np.log(np.sum(np.exp(shifted / eps)))) + 50.0 * (np.sum(x) - n/2.0)**2
    
    try:
        result = minimize(smooth_obj, x0, method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': 200, 'maxls': 20,
                                  'ftol': 1e-10, 'gtol': 1e-8})
        x = np.clip(result.x, 0.0, 1.0)
        # Project to sum constraint
        diff = n/2.0 - np.sum(x)
        if abs(diff) > 1e-6:
            free = (x > 0.01) & (x < 0.99)
            if np.sum(free) > 0:
                x[free] += diff / np.sum(free)
            else:
                x += diff / n
            x = np.clip(x, 0.0, 1.0)
        c5_cont = c5_of(x)
        return x, c5_cont
    except Exception:
        return x0, c5_of(x0)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=float)
        if len(parent) == 0:
            parent = None
    except Exception:
        parent = None
    
    if parent is None:
        n_init = 42
        h = np.zeros(n_init)
        h[rng.choice(n_init, n_init//2, replace=False)] = 1.0
    else:
        h = parent.astype(float).copy()
    
    best_h = h.copy()
    best_c5 = c5_of(best_h)
    best_n = len(best_h)
    
    # Phase 1: Steepest descent at initial resolution
    if time_left() > 60:
        t_limit = min(time_left() * 0.25, 250.0)
        h_try, c5_try = steepest_descent_binary(best_h, rng, t_limit, t0)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
    
    # Phase 2: Annealing at initial resolution
    if time_left() > 60:
        t_limit = min(time_left() * 0.2, 150.0)
        init_temp = max(0.005, (best_c5 - 0.3) * 0.05)
        h_try, c5_try = anneal_binary(best_h, rng, t_limit, t0, init_temp=init_temp)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
    
    # Phase 3: Multi-resolution refinement
    current_h = best_h.copy()
    current_n = len(current_h)
    for _ in range(4):  # Try up to 4 doublings
        new_n = 2 * current_n
        if new_n > 512 or time_left() < 120:
            break
        
        # Try both interpolation strategies
        for strategy in ['repeat', 'split']:
            if time_left() < 60:
                break
            
            if strategy == 'repeat':
                h_fine = interpolate_double(current_h)
            else:
                h_fine = interpolate_double_split(current_h, rng)
            
            if len(h_fine) != new_n:
                continue
            
            c5_fine = c5_of(h_fine)
            
            # Steepest descent on fine grid
            t_limit = min(time_left() * 0.25, 200.0)
            h_try, c5_try = steepest_descent_binary(h_fine, rng, t_limit, t0)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = len(h_try)
            
            # Brief annealing
            if time_left() > 80 and c5_try <= c5_fine:
                t_limit = min(time_left() * 0.15, 100.0)
                h_try2, c5_try2 = anneal_binary(h_try, rng, t_limit, t0, init_temp=0.01)
                if c5_try2 < best_c5:
                    best_c5 = c5_try2
                    best_h = h_try2.copy()
                    best_n = len(h_try2)
                    current_h = h_try2.copy()
        
        current_n = new_n
        if time_left() < 120:
            break
    
    # Phase 4: Go back to best resolution and do more search
    if time_left() > 60:
        t_limit = min(time_left() * 0.3, 200.0)
        h_try, c5_try = steepest_descent_binary(best_h, rng, t_limit, t0)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = len(h_try)
    
    # Phase 5: Continuous optimization at best resolution
    if time_left() > 40:
        t_limit = min(time_left() * 0.4, 180.0)
        x_try, c5_try = continuous_optimize(best_h, best_n, t_limit, t0, rng)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = x_try.copy()
            best_n = len(x_try)
    
    # Phase 6: Final polish - try binary projection of continuous solution
    if time_left() > 30:
        h_bin = best_h.copy()
        n = len(h_bin)
        m = n // 2
        # Round: keep top m values as 1
        idx = np.argsort(-h_bin, kind='stable')
        h_bin2 = np.zeros(n)
        h_bin2[idx[:m]] = 1.0
        c5_bin = c5_of(h_bin2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin2.copy()
            best_n = n
        # Also try the binary with steepest descent
        if time_left() > 30:
            t_limit = min(time_left() * 0.5, 100.0)
            h_try, c5_try = steepest_descent_binary(h_bin2, rng, t_limit, t0)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = c5_of(best_h)
    if final_c5 < best_c5 + 1e-10:
        best_c5 = final_c5
    
    return best_h, best_c5, best_n
