```
def direct_terms_and_grad(h, dx, gamma, weight_tol=1e-9):
    n = h.shape[0]
    L = 2*n - 1
    oh = 1.0 - h
    terms = np.empty(L, dtype=float)
    # local for speed
    for idx in range(L):
        s = idx - (n-1)
        if s >= 0:
            m = n - s
            if m <= 0:
                terms[idx] = 0.0
            elif s == 0:
                terms[idx] = dx * np.dot(h, oh)
            else:
                terms[idx] = dx * np.dot(h[:m], oh[s:])
        else:
            q = -s
            m = n - q
            terms[idx] = dx * np.dot(h[q:], oh[:m])
    maxt = float(terms.max())
    # logsum weights
    z = gamma * (terms - maxt)
    # avoid overflow: z <=0
    ex = np.exp(z)
    sx = float(ex.sum())
    if sx == 0.0 or not np.isfinite(sx):
        # fallback active max only
        weights = np.zeros(L)
        weights[terms == maxt] = 1.0 / np.count_nonzero(terms == maxt)
    else:
        weights = ex / sx
    grad = np.zeros(n, dtype=float)
    # active indices
    # to reduce loop, get idxs where weight>tol
    idxs = np.nonzero(weights > weight_tol)[0]
    for idx in idxs:
        w = float(weights[idx])
        s = int(idx - (n-1))
        wx = w * dx
        if s >= 0:
            m = n - s
            if m > 0:
                grad[:m] += wx * oh[s:]
                grad[s:] -= wx * h[:m]
        else:
            q = -s
            m = n - q
            grad[q:] += wx * oh[:m]
            grad[:m] -= wx * h[q:]
    G = maxt + math.log(sx) / gamma if sx>0 else maxt
    return maxt, G, grad
```

def project_box_sum(v, target):
    v = np.asarray(v, dtype=float).copy()
    n = len(v)
    target = float(target)
    if target < 0: target=0.0
    if target > n: target=float(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    # If v already in [0,1] and sum target close, return after small adjust
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        sx = float(x.sum())
        if sx > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)*0.5, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(interior))
        if cnt > 0:
            x[interior] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        # Remaining due clips: adjust one variable with room
        rem = target - float(x.sum())
        if abs(rem) > 1e-13:
            if rem > 0:
                idx = int(np.argmax(1.0 - x))
                add = min(rem, 1.0 - x[idx])
                x[idx] += add
            else:
                idx = int(np.argmax(x))
                sub = min(-rem, x[idx])
                x[idx] -= sub
            x = np.clip(x, 0.0, 1.0)
    return x

```
    n = int(v.size)
    if n == 0: return v

    if target < 0.0: target = 0.0
    if target > float(n): target = float(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        x = np.clip(v - mid, 0.0, 1.0)
        sx = float(x.sum())

            lo = mid
        else:
            hi = mid
```

```python
import time
import math
import numpy as np


def sanitize_h(h):
    if h is None:
        return None
    try:
        a = np.asarray(h, dtype=float).ravel()
    except Exception:
        return None
    if a.size == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=1.0, neginf=0.0)
    return np.clip(a, 0.0, 1.0)


def get_parent():
    if "initial_h_values" not in globals():
        return None
    return sanitize_h(globals().get("initial_h_values", None))


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float).copy()
    n = int(v.size)
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > float(n):
        target = float(n)

    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        x = np.clip(v - mid, 0.0, 1.0)
        sx = float(x.sum())
        if sx > target:
            lo = mid
        else:
            hi = mid

    mid = 0.5 * (lo + hi)
    x = np.clip(v - mid, 0.0, 1.0)
    diff = target - float(x.sum())
    if abs(diff) > 1e-13:
        interior = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = int(np.count_nonzero(interior))
        if cnt > 0:
            x[interior] += diff / float(cnt)
            x = np.clip(x, 0.0, 1.0)
        rem = target - float(x.sum())
        if abs(rem) > 1e-13:
            if rem > 0.0:
                idx = int(np.argmax(1.0 - x))
                x[idx] = min(1.0, float(x[idx]) + rem)
            else:
                idx = int(np.argmax(x))
                x[idx] = max(0.0, float(x[idx]) + rem)
            x = np.clip(x, 0.0, 1.0)
    return x


def enforce_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)
    target = float(target)
    for _ in range(30):
        diff = target - float(h.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] += min(diff, 1.0 - float(h[idx]))
        else:
            idx = int(np.argmax(h))
            h[idx] += max(diff, -float(h[idx]))
        h = np.clip(h, 0.0, 1.0)
    return h


def exact_c5_raw(h, dx):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return 1e30
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * float(dx))


def consider_candidate(h, c, n, best_h, best_c, best_n):
    h = sanitize_h(h)
    if h is None:
        return best_h, best_c, best_n
    n = int(h.size)
    if n <= 0:
        return best_h, best_c, best_n
    h2 = project_box_sum(h, 0.5 * n)
    dx = 2.0 / float(n)
    c2 = exact_c5_raw(h2, dx)
    if not math.isfinite(c2):
        return best_h, best_c, best_n
    if c2 < best_c - 1e-12:
        return h2, float(c2), n
    return best_h, best_c, best_n


def make_random_continuous(n, rng):
    n = int(n)
    if n <= 0:
        n = 100
    x = np.linspace(0.0, 1.0, n)
    v = 0.5 * np.ones(n, dtype=float)
    for _ in range(3):
        amp = 0.10 + 0.15 * float(rng.random())
        v = v + amp * np.sin(2.0 * np.pi * float(rng.uniform(1.0, 8.0)) * x
                             + float(rng.uniform(0.0, 2.0 * np.pi)))
    v = v + 0.15 * rng.standard_normal(n)
    return project_box_sum(v, 0.5 * n)


def resample_h(h0, n_new):
    h0 = sanitize_h(h0)
    if h0 is None:
        return None
    n0 = int(h0.size)
    n_new = int(n_new)
    if n_new <= 0 or n0 <= 0:
        return None
    if n0 == n_new:
        return h0.copy()
    if n0 == 1:
        return np.full(n_new, float(h0[0]), dtype=float)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return np.interp(x1, x0, h0)


def binary_from_values(vals, n):
    n = int(n)
    arr = sanitize_h(vals)
    if arr is None:
        return None
    if arr.size != n:
        arr = resample_h(arr, n)
        if arr is None:
            return None
    m = n // 2
    if np.all((arr <= 1e-12) | (arr >= 1.0 - 1e-12)):
        b0 = (arr > 0.5).astype(np.int32)
        if int(b0.sum()) == m:
            return b0
    order = np.argsort(-arr)
    b = np.zeros(n, dtype=np.int32)
    b[order[:m]] = 1
    return b


def build_binary_lists(a):
    n = int(a.size)
    ones = []
    zeros = []
    pos_o = [-1] * n
    pos_z = [-1] * n
    for i in range(n):
        if int(a[i]) > 0:
            pos_o[i] = len(ones)
            ones.append(i)
        else:
            pos_z[i] = len(zeros)
            zeros.append(i)
    return ones, zeros, pos_o, pos_z


def swap_remove_add(ones, zeros, pos_o, pos_z, i, j):
    p_o = pos_o[i]
    last = ones[-1]
    ones[p_o] = last
    pos_o[last] = p_o
    del ones[-1]
    pos_o[i] = -1

    p_z = pos_z[j]
    last = zeros[-1]
    zeros[p_z] = last
    pos_z[last] = p_z
    del zeros[-1]
    pos_z[j] = -1

    pos_z[i] = len(zeros)
    zeros.append(i)
    pos_o[j] = len(ones)
    ones.append(j)


def find_max_shift_pair(h, dx):
    n = int(h.size)
    oh = 1.0 - h
    bestC = -1.0
    best_s = 0
    best_src = 0
    best_tgt = 0

    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        if s >= 0:
            m = n - s
            if m <= 0:
                continue
            c = float(dx) * float(np.dot(h[:m], oh[s:]))
            if c > bestC:
                bestC = c
                best_s = s
                contrib = h[:m] * oh[s:]
                p = int(np.argmax(contrib))
                best_src = p
                best_tgt = p + s
        else:
            q = -s
            m = n - q
            c = float(dx) * float(np.dot(h[q:], oh[:m]))
            if c > bestC:
                bestC = c
                best_s = s
                contrib = h[q:] * oh[:m]
                p = int(np.argmax(contrib))
                src = q + p
                best_src = src
                best_tgt = src + s
    return bestC, best_s, best_src, best_tgt


def active_pair_polish(h0, deadline, rng):
    h = sanitize_h(h0)
    if h is None:
        return None, 1e30, 0
    n = int(h.size)
    if n <= 0:
        return h, 1e30, n
    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    dx = 2.0 / float(n)
    C = exact_c5_raw(h, dx)
    best_h = h.copy()
    best_c = C
    stall = 0
    fracs = (0.005, 0.02, 0.05, 0.10, 0.25, 0.50, 0.90)

    while time.time() < deadline - 2.0 and stall < 60:
        c_cur, s, src, tgt = find_max_shift_pair(h, dx)
        if not math.isfinite(c_cur):
            break
        C = float(c_cur)
        if C < best_c - 1e-12:
            best_c = C
            best_h = h.copy()

        improved = False

        if s == 0 or src == tgt or src < 0 or tgt < 0 or src >= n or tgt >= n:
            for beta in (1.15, 1.50, 2.00):
                v = project_box_sum(0.5 + beta * (h - 0.5), target)
                cn = exact_c5_raw(v, dx)
                if cn < C - 1e-12:
                    h = v
                    C = cn
                    improved = True
                    if C < best_c - 1e-12:
                        best_c = C
                        best_h = h.copy()
                    break
        else:
            maxd = min(float(h[src]), float(1.0 - h[tgt]))
            if maxd > 1e-12:
                for fr in fracs:
                    d = maxd * float(fr)
                    if d <= 1e-14:
                        continue
                    h[src] -= d
                    h[tgt] += d
                    cn = exact_c5_raw(h, dx)
                    if cn < C - 1e-12:
                        C = cn
                        improved = True
                        if C < best_c - 1e-12:
                            best_c = C
                            best_h = h.copy()
                        break
                    else:
                        h[src] += d
                        h[tgt] -= d

        if not improved:
            for _ in range(30):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                if float(h[i]) >= float(h[j]):
                    s2 = i
                    t2 = j
                    maxd2 = min(float(h[i]), float(1.0 - h[j]))
                else:
                    s2 = j
                    t2 = i
                    maxd2 = min(float(h[j]), float(1.0 - h[i]))
                if maxd2 <= 1e-12:
                    continue
                d2 = maxd2 * (0.005 + 0.08 * float(rng.random()))
                h[s2] -= d2
                h[t2] += d2
                cn = exact_c5_raw(h, dx)
                if cn < C - 1e-12:
                    C = cn
                    improved = True
                    if C < best_c - 1e-12:
                        best_c = C
                        best_h = h.copy()
                    break
                else:
                    h[s2] += d2
                    h[t2] -= d2

            if not improved:
                stall += 1
                continue

        stall = 0

    best_h = project_box_sum(best_h, target)
    bc = exact_c5_raw(best_h, dx)
    if bc < best_c:
        best_c = bc
    return best_h, float(best_c), n


def direct_terms_and_grad(h, dx, gamma, weight_tol=1e-10):
    n = int(h.size)
    L = 2 * n - 1
    oh = 1.0 - h
    terms = np.empty(L, dtype=float)

    for idx in range(L):
        s = idx - (n - 1)
        if s >= 0:
            m = n - s
            if m <= 0:
                terms[idx] = 0.0
            elif s == 0:
                terms[idx] = dx * float(np.dot(h, oh))
            else:
                terms[idx] = dx * float(np.dot(h[:m], oh[s:]))
        else:
            q = -s
            m = n - q
            terms[idx] = dx * float(np.dot(h[q:], oh[:m]))

    maxt = float(terms.max()) if L > 0 else 0.0
    if maxt <= 1e-15:
        return maxt, np.zeros(n, dtype=float)

    z = gamma * (terms - maxt)
    ex = np.exp(z)
    sx = float(ex.sum())
    if sx <= 0.0 or not math.isfinite(sx):
        weights = np.zeros(L, dtype=float)
        mask = terms >= (maxt - 1e-13)
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            weights[mask] = 1.0 / float(cnt)
    else:
        weights = ex / sx

    grad = np.zeros(n, dtype=float)
    idxs = np.nonzero(weights > weight_tol)[0]
    for idx in idxs:
        w = float(weights[idx])
        s = int(idx) - (n - 1)
        wx = w * dx
        if s >= 0:
            m = n - s
            if m > 0:
                grad[:m] += wx * oh[s:]
                grad[s:] -= wx * h[:m]
        else:
            q = -s
            m = n - q
            grad[q:] += wx * oh[:m]
            grad[:m] -= wx * h[q:]
    return maxt, grad


def continuous_optimize(h0, deadline, rng, gamma_start=15.0):
    h = sanitize_h(h0)
    if h is None:
        return None, 1e30, 0
    n = int(h.size)
    if n <= 1:
        target = 0.5 * float(n)
        h = project_box_sum(h, target)
        return h, exact_c5_raw(h, 2.0 / float(n)), n

    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    dx = 2.0 / float(n)
    C = exact_c5_raw(h, dx)
    best_h = h.copy()
    best_c = C
    gamma = float(gamma_start)
    no_improve = 0
    alphas = (0.10, 0.05, 0.20, 0.40, 0.02, 0.80)

    while time.time() < deadline - 3.0:
        maxt, grad = direct_terms_and_grad(h, dx, gamma)
        C = float(maxt)
        if C < best_c - 1e-12:
            best_c = C
            best_h = h.copy()

        gnorm = float(np.linalg.norm(grad))
        improved = False
        if gnorm > 1e-12:
            gdir = grad / gnorm
            for a_step in alphas:
                h_try = project_box_sum(h - a_step * gdir, target)
                cn = exact_c5_raw(h_try, dx)
                if cn < C - 1e-12:
                    h = h_try
                    C = cn
                    improved = True
                    break

        if improved:
            if C < best_c - 1e-12:
                best_c = C
                best_h = h.copy()
            no_improve = 0
            gamma = min(450.0, gamma * 1.04)
        else:
            no_improve += 1
            moved = False
            for _ in range(12):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                if float(h[i]) >= float(h[j]):
                    s2 = i
                    t2 = j
                    maxd = min(float(h[i]), float(1.0 - h[j]))
                else:
                    s2 = j
                    t2 = i
                    maxd = min(float(h[j]), float(1.0 - h[i]))
                if maxd <= 1e-12:
                    continue
                d2 = maxd * (0.005 + 0.05 * float(rng.random()))
                h[s2] -= d2
                h[t2] += d2
                cn = exact_c5_raw(h, dx)
                if cn < C - 1e-12:
                    C = cn
                    moved = True
                    if C < best_c - 1e-12:
                        best_c = C
                        best_h = h.copy()
                    break
                else:
                    h[s2] += d2
                    h[t2] -= d2

            if moved:
                no_improve = 0
            else:
                if no_improve >= 25 and no_improve % 5 == 0:
                    scale = 0.01 + 0.003 * min(float(no_improve), 20.0)
                    h_try = project_box_sum(h + scale * rng.standard_normal(n), target)
                    cn = exact_c5_raw(h_try, dx)
                    if cn < C + 1e-3:
                        h = h_try
                        C = cn
                    gamma = max(5.0, gamma * 0.75)
                    no_improve = 0

    best_h = project_box_sum(best_h, target)
    bc = exact_c5_raw(best_h, dx)
    if bc < best_c:
        best_c = bc
    return best_h, float(best_c), n


def binary_descent(a, n, deadline, rng, sample_pairs=4000, all_pairs_limit=6000):
    a = binary_from_values(a, int(n))
    if a is None:
        return None, 1e30, 0
    n = int(a.size)
    ones, zeros, pos_o, pos_z = build_binary_lists(a)
    if len(ones) == 0 or len(zeros) == 0:
        return a, 0.0, n

    E = int(np.max(np.correlate(a, 1 - a, mode="full")))
    best_a = a.copy()
    best_E = E
    best_C = 2.0 * float(E) / float(n)
    stall = 0

    while time.time() < deadline - 2.0 and stall < 25:
        best_e = E
        best_i = -1
        best_j = -1
        total = len(ones) * len(zeros)

        if total <= int(all_pairs_limit):
            for ii in range(len(ones)):
                i = ones[ii]
                for jj in range(len(zeros)):
                    j = zeros[jj]
                    a[i] = 0
                    a[j] = 1
                    e = int(np.max(np.correlate(a, 1 - a, mode="full")))
                    a[i] = 1
                    a[j] = 0
                    if e < best_e:
                        best_e = e
                        best_i = i
                        best_j = j
        else:
            for _ in range(int(sample_pairs)):
                if time.time() > deadline - 2.0:
                    break
                i = ones[int(rng.integers(len(ones)))]
                j = zeros[int(rng.integers(len(zeros)))]
                a[i] = 0
                a[j] = 1
                e = int(np.max(np.correlate(a, 1 - a, mode="full")))
                a[i] = 1
                a[j] = 0
                if e < best_e:
                    best_e = e
                    best_i = i
                    best_j = j

        if best_i >= 0 and best_e < E:
            swap_remove_add(ones, zeros, pos_o, pos_z, best_i, best_j)
            a[best_i] = 0
            a[best_j] = 1
            E = best_e
            stall = 0
            if E < best_E:
                best_E = E
                best_a = a.copy()
                best_C = 2.0 * float(E) / float(n)
        else:
            stall += 1

    return best_a, float(best_C), n


def binary_sa(a, n, deadline, rng, T_start=1.8, T_min=0.03):
    a = binary_from_values(a, int(n))
    if a is None:
        return None, 1e30, 0
    n = int(a.size)
    ones, zeros, pos_o, pos_z = build_binary_lists(a)
    if len(ones) == 0 or len(zeros) == 0:
        return a, 0.0, n

    E = int(np.max(np.correlate(a, 1 - a, mode="full")))
    best_a = a.copy()
    best_E = E
    best_C = 2.0 * float(E) / float(n)
    T = float(T_start)
    no_improve = 0

    while time.time() < deadline - 3.0:
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[int(rng.integers(len(ones)))]
        j = zeros[int(rng.integers(len(zeros)))]

        a[i] = 0
        a[j] = 1
        e = int(np.max(np.correlate(a, 1 - a, mode="full")))

        if e < best_E:
            best_E = e
            best_a = a.copy()
            best_C = 2.0 * float(e) / float(n)

        if e <= E:
            swap_remove_add(ones, zeros, pos_o, pos_z, i, j)
            E = e
            no_improve = 0
        else:
            accept = False
            if T > 1e-9:
                if rng.random() < math.exp(-float(e - E) / T):
                    accept = True
            if accept:
                swap_remove_add(ones, zeros, pos_o, pos_z, i, j)
                E = e
                no_improve = 0
            else:
                a[i] = 1
                a[j] = 0
                no_improve += 1

        T = max(float(T_min), T * 0.99985)

        if E > best_E + 8:
            a = best_a.copy()
            E = best_E
            ones, zeros, pos_o, pos_z = build_binary_lists(a)
            no_improve = 0
            T = 1.0

        if no_improve > 2000:
            a = best_a.copy()
            E = best_E
            ones, zeros, pos_o, pos_z = build_binary_lists(a)
            no_improve = 0
            T = 1.5

    best_a = project_box_sum(best_a.astype(float), 0.5 * float(n))
    bc = exact_c5_raw(best_a, 2.0 / float(n))
    if bc < best_C:
        best_C = bc
    return best_a, float(best_C), n


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    try:
        B = float(budget_s)
    except Exception:
        B = 1300.0
    if not math.isfinite(B) or B <= 0.0:
        B = 1300.0
    deadline = t0 + max(5.0, B - 8.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = 1e30
    best_n = 0

    parent = get_parent()
    if parent is None or parent.size < 2:
        parent = make_random_continuous(100, rng)

    n_parent = int(parent.size)
    parent_feas = project_box_sum(parent, 0.5 * float(n_parent))

    best_h, best_c, best_n = consider_candidate(parent, None, n_parent,
                                                best_h, best_c, best_n)
    best_h, best_c, best_n = consider_candidate(parent_feas, None, n_parent,
                                                best_h, best_c, best_n)

    if time.time() < deadline - 5.0 and best_h is not None:
        d = min(deadline, time.time() + min(60.0, 0.05 * B))
        h, c, n = active_pair_polish(best_h.copy(), d, rng)
        best_h, best_c, best_n = consider_candidate(h, c, n, best_h, best_c, best_n)

    if time.time() < deadline - 10.0 and best_h is not None:
        d = min(deadline, time.time() + min(240.0, 0.20 * B))
        h, c, n = continuous_optimize(best_h.copy(), d, rng)
        best_h, best_c, best_n = consider_candidate(h, c, n, best_h, best_c, best_n)

    if best_h is not None:
        base_n = int(best_n)
        first = base_n if base_n % 2 == 0 else base_n + 1
        cand_ns = []
        for x in (first, 96, 100, 120, 150):
            if x >= 16 and (x % 2 == 0) and x not in cand_ns:
                cand_ns.append(x)
        if B > 900.0:
            cand_ns = cand_ns[:4]
        elif B > 500.0:
            cand_ns = cand_ns[:3]
        else:
            cand_ns = cand_ns[:2]
    else:
        cand_ns = [100]

    for nn in cand_ns:
        if time.time() > deadline - 40.0:
            break

        if int(best_n) == int(nn) and best_h is not None:
            h0 = best_h.copy()
        else:
            h0 = resample_h(best_h, int(nn))
            if h0 is None:
                continue

        best_h, best_c, best_n = consider_candidate(h0, None, int(nn),
                                                    best_h, best_c, best_n)

        if time.time() < deadline - 10.0:
            d = min(deadline, time.time() + min(50.0, 0.03 * B))
            h, c, n = active_pair_polish(h0.copy(), d, rng)
            best_h, best_c, best_n = consider_candidate(h, c, n,
                                                        best_h, best_c, best_n)

        a = binary_from_values(best_h, int(nn))
        ab = None
        ab2 = None

        if time.time() < deadline - 10.0 and a is not None:
            d = min(deadline, time.time() + min(70.0, 0.05 * B))
            ab, cb, n_ab = binary_descent(a, int(nn), d, rng)
            best_h, best_c, best_n = consider_candidate(ab, cb, n_ab,
                                                        best_h, best_c, best_n)

        if time.time() < deadline - 10.0 and ab is not None:
            d = min(deadline, time.time() + min(90.0, 0.05 * B))
            ab2, ca, n_ab2 = binary_sa(ab, int(nn), d, rng)
            best_h, best_c, best_n = consider_candidate(ab2, ca, n_ab2,
                                                        best_h, best_c, best_n)

        if time.time() < deadline - 10.0 and ab2 is not None:
            d = min(deadline, time.time() + min(50.0, 0.03 * B))
            h, c, n = active_pair_polish(ab2.copy(), d, rng)
            best_h, best_c, best_n = consider_candidate(h, c, n,
                                                        best_h, best_c, best_n)

        if time.time() < deadline - 30.0:
            if int(best_n) == int(nn):
                hcur = best_h.copy()
            else:
                hcur = resample_h(best_h, int(nn))
            if hcur is not None:
                d = min(deadline, time.time() + min(130.0, 0.10 * B))
                h, c, n = continuous_optimize(hcur, d, rng)
                best_h, best_c, best_n = consider_candidate(h, c, n,
                                                            best_h, best_c, best_n)

    if best_h is None:
        best_h = make_random_continuous(100, rng)
        best_h, best_c, best_n = consider_candidate(best_h, None, 100,
                                                    best_h, best_c, best_n)

    best_h = np.asarray(best_h, dtype=float)
    best_h = enforce_sum(best_h, 0.5 * float(best_h.size))
    best_c = exact_c5_raw(best_h, 2.0 / float(best_h.size))

    if (not math.isfinite(best_c)) or best_c > 1e20:
        fallback = make_random_continuous(100, rng)
        fallback = project_box_sum(fallback, 50.0)
        best_h = fallback
        best_c = exact_c5_raw(best_h, 0.02)

    return best_h, float(best_c), int(best_h.size)
```
