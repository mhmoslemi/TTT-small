import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_array(h, dx):
    return np.correlate(h, 1.0-h, mode="full") * dx

def project_to_constraint(v, target_sum):
    """Project v onto {0 <= h <= 1, sum(h) = target_sum}"""
    # Bisection on t: find t such that sum(clip(v-t, 0, 1)) = target_sum
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(v - t, 0.0, 1.0)

def gradient_active(h, m_star, dx):
