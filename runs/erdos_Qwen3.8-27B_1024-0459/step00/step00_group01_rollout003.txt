            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        diff = target - float(h.sum())
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            idx = np.where(h < 1.0 - 1e-15)[0]
            if idx.size == 0:
                break
            h[idx] += diff / float(idx.size)
        else:
            idx = np.where(h > 1e-15)[0]
            if idx.size == 0:
                break
            h[idx] += diff / float(idx.size)
    return np.clip(h, 0.0, 1.0)


def binary_project(h, n):
    n = int(n)
    target = int(round(0.5 * float(n)))
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    if target >= n:
        return np.ones(n, dtype=np.float64)
    h = np.asarray(h, dtype=np.float64)
    order = np.argsort(-h, kind="stable")
    b = np.zeros(n, dtype=np.float64)
    b[order[:target]] = 1.0
    return b


def circular_from_h(h, n, exact=False):
    n = int(n)
    N = 2 * n - 1
    h = np.asarray(h, dtype=np.float64)
    if exact:
        full = np.correlate(h, 1.0 - h, mode="full")
        c = np.empty(N, dtype=np.float64)
        c[:n] = full[n - 1:]
        c[n:] = full[n - 2::-1]
        return c
    hp = np.zeros(N, dtype=np.float64)
    vp = np.zeros(N, dtype=np.float64)
    hp[:n] = h
    vp[:n] = 1.0 - h
    fh = np.fft.fft(hp)
    fv = np.fft.fft(vp)
    c = np.fft.ifft(fh * np.conj(fv)).real
    return c


def exact_c5(h, n):
    h = np.asarray(h, dtype=np.float64)
    n = int(n)
    dx = 2.0 / float(n)
    if n <= 4096:
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr) * dx)
    c = circular_from_h(h, n, exact=False)
    return float(np.max(c) * dx)


def make_state(h, n, exact=False, binary=False):
    h = np.asarray(h, dtype=np.float64).copy()
    n = int(n)
    if binary:
        h = binary_project(project_sum(h, 0.5 * float(n)), n)
    else:
        h = project_sum(h, 0.5 * float(n))
    c = circular_from_h(h, n, exact=exact)
    if binary:
        c = np.rint(c)

    N = 2 * n - 1
    hp = np.zeros(N, dtype=np.float64)
    vp = np.zeros(N, dtype=np.float64)
    hp[:n] = h
    vp[:n] = 1.0 - h

    vneg = np.zeros(N, dtype=np.float64)
    vneg[0] = vp[0]
    idx = np.arange(1, N)
    vneg[idx] = vp[(-idx) % N]

    return h, c, hp, vp, vneg


def state_value(c, n):
    return float(np.max(c)) * (2.0 / float(int(n)))


def resample_vector(x, n):
    x = np.asarray(x, dtype=np.float64)
    m = int(x.size)
    n = int(n)
    if m == n:
        return x.copy()
    if m == 1:
        return np.full(n, float(x[0]), dtype=np.float64)
    xo = (np.arange(m, dtype=np.float64) + 0.5) * (2.0 / float(m))
    xn = (np.arange(n, dtype=np.float64) + 0.5) * (2.0 / float(n))
    return np.interp(xn, xo, x)


def random_binary(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=np.float64)
    if n <= 1:
        return h
    perm = rng.permutation(n)
    k = n // 2
    if k <= 0:
        return h
    h[perm[:k]] = 1.0
    return h


def thue_morse_binary(n, rng):
    n = int(n)
    t = np.array([0], dtype=np.int8)
    while t.size < n:
        t = np.concatenate((t, 1 - t))
    L = int(t.size)
    max_shift = max(1, L - n + 1)
    shift = int(rng.integers(0, max_shift))
    seg = t[shift:shift + n]
    if int(rng.integers(0, 2)) == 1:
        seg = 1 - seg
    return binary_project(seg.astype(np.float64), n)


def random_continuous(n, rng):
    n = int(n)
    return project_sum(rng.random(n, dtype=np.float64), 0.5 * float(n))


def parent_continuous_start(parent, n):
    n = int(n)
    if parent is None:
        r = np.random.default_rng(12345)
        return random_continuous(n, r)
    vals = resample_vector(parent, n)
    return project_sum(vals, 0.5 * float(n))


def parent_binary_start(parent, n, rng, noise=0.0):
    n = int(n)
    if parent is None:
        return random_binary(n, rng)
    vals = resample_vector(parent, n)
    vals = project_sum(vals, 0.5 * float(n))
    if noise > 0.0:
        vals = vals + rng.normal(0.0, float(noise), size=n)
    return binary_project(vals, n)


def pair_a_vector(vneg, hp, N, idx):
    idx = int(idx)
    return np.roll(vneg, idx) - np.roll(hp, -idx)


def binary_move_value_from_A(c, Ai, Aj, N, i, j, dx):
    arr = c + Ai - Aj
    arr[0] -= 2.0
    p1 = (int(i) - int(j)) % int(N)
    p2 = (int(j) - int(i)) % int(N)
    arr[p1] += 1.0
    arr[p2] += 1.0
    return float(np.max(arr)) * float(dx)


def apply_move_from_A(c, hp, vp, vneg, N, i, j, d, Ai, Aj):
    d = float(d)
    if d == 0.0:
        return
    i = int(i)
    j = int(j)
    N = int(N)
    lin = Ai - Aj
    c += d * lin
    d2 = d * d
    c[0] -= 2.0 * d2
    p1 = (i - j) % N
    p2 = (j - i) % N
    c[p1] += d2
    c[p2] += d2

    hp[i] += d
    hp[j] -= d
    vp[i] -= d
    vp[j] += d
    vneg[(-i) % N] -= d
    vneg[(-j) % N] += d


def sync_binary_state(c, hp, vp, vneg, N, n):
    n = int(n)
    N = int(N)
    h = np.rint(hp[:n])
    h = np.clip(h, 0.0, 1.0)
    target = int(round(0.5 * float(n)))
    if abs(float(h.sum()) - float(target)) > 1e-6:
        h = binary_project(h, n)

    c[:] = np.rint(c)
    hp[:n] = h
    hp[n:] = 0.0
    vp[:n] = 1.0 - h
    vp[n:] = 0.0
    vneg[0] = vp[0]
    idx = np.arange(1, N)
    vneg[idx] = vp[(-idx) % N]


def active_gradients(c, hp, vp, n, N, top_m):
    n = int(n)
    N = int(N)
    top_m = int(top_m)
    if top_m <= 0:
        return np.zeros(n, dtype=np.float64), 0
    if top_m >= N:
        idx = np.arange(N)
    else:
        pos = N - top_m
        idx = np.argpartition(c, pos)[-top_m:]
    G = np.zeros(n, dtype=np.float64)
    for kk in idx:
        k = int(kk)
        G += np.roll(vp, k)[:n] - np.roll(hp, -k)[:n]
    return G, int(idx.size)


def local_search_binary(h0, n, budget_s, seed):
    n = int(n)
    budget_s = float(budget_s)
    if n < 2 or n % 2 != 0:
        h = binary_project(project_sum(np.asarray(h0, dtype=np.float64), 0.5 * float(n)), n)
        val = exact_c5(h, n)
        return h, val

    if budget_s <= 3.0:
        h = binary_project(project_sum(np.asarray(h0, dtype=np.float64), 0.5 * float(n)), n)
        val = state_value(circular_from_h(h, n, exact=True), n)
        return h, val

    h, c, hp, vp, vneg = make_state(h0, n, exact=(n <= 1024), binary=True)
    N = 2 * n - 1
    dx = 2.0 / float(n)
    currentC = state_value(c, n)
    best_h = hp[:n].copy()
    best_val = currentC

    rng = np.random.default_rng(int(seed))
    t0 = time.time()
    t_end = t0 + budget_s - 1.5

    if n <= 256:
        P = 10
        R = 8
        M = 8
    elif n <= 1024:
        P = 8
        R = 6
        M = 6
    elif n <= 2048:
        P = 6
        R = 4
        M = 5
    else:
        P = 4
        R = 3
        M = 4

    temp_start = 3.0 * dx
    temp_floor = 0.1 * dx
    stall = 0
    applied = 0

    while time.time() < t_end:
        G, _ = active_gradients(c, hp, vp, n, N, M)
        zeros = np.where(hp[:n] < 0.5)[0]
        ones = np.where(hp[:n] > 0.5)[0]
        if zeros.size == 0 or ones.size == 0:
            break

        ki = min(P, int(zeros.size))
        kj = min(P, int(ones.size))
        Gz = G[zeros]
        pos_i = int(Gz.size) - ki
        zi = zeros[np.argpartition(Gz, pos_i)[-ki:]]
        Go = G[ones]
        pos_j = max(0, kj - 1)
        oj = ones[np.argpartition(Go, pos_j)[:kj]]

        ri_size = min(R, int(zeros.size))
        rj_size = min(R, int(ones.size))
        ri = zeros[rng.integers(0, zeros.size, size=ri_size)]
        rj = ones[rng.integers(0, ones.size, size=rj_size)]

        all_i = np.unique(np.concatenate((zi, ri)))
        all_j = np.unique(np.concatenate((oj, rj)))

        A_i = {}
        for ii in all_i:
            ii = int(ii)
            A_i[ii] = pair_a_vector(vneg, hp, N, ii)
        A_j = {}
        for jj in all_j:
            jj = int(jj)
            A_j[jj] = pair_a_vector(vneg, hp, N, jj)

        best_moveC = currentC + 1e6
        bestii = -1
        bestjj = -1

        for ii in all_i:
            ii = int(ii)
            if float(hp[ii]) > 0.5:
                continue
            Ai = A_i[ii]
            for jj in all_j:
                jj = int(jj)
                if float(hp[jj]) < 0.5:
                    continue
                Aj = A_j[jj]
                val = binary_move_value_from_A(c, Ai, Aj, N, ii, jj, dx)
                if val < best_moveC:
                    best_moveC = val
                    bestii = ii
                    bestjj = jj

        if bestii < 0:
            break

        deltaC = best_moveC - currentC
        frac = (time.time() - t0) / max(1e-9, budget_s)
        if frac > 1.0:
            frac = 1.0
        T = temp_floor + (temp_start - temp_floor) * (0.5 ** frac)

        accept = (best_moveC < currentC - 1e-12)
        if not accept:
            stall += 1
            if T > 1e-8 and deltaC >= -1e-12:
                ratio = deltaC / T
                if ratio < 700.0:
                    p = math.exp(-ratio)
                else:
                    p = 0.0
                if rng.random() < p:
                    accept = True
            if not accept and stall > 1500 and best_moveC < currentC + max(0.05, 3.0 * dx):
                accept = True
                stall = 0
            if not accept and stall > 6000:
                accept = True
                stall = 0

        if accept:
            Ai = A_i[bestii]
            Aj = A_j[bestjj]
            apply_move_from_A(c, hp, vp, vneg, N, bestii, bestjj, 1.0, Ai, Aj)
            currentC = state_value(c, n)
            if currentC < best_val - 1e-12:
                best_val = currentC
                best_h = hp[:n].copy()
            stall = 0
            applied += 1
            if applied % 200 == 0:
                sync_binary_state(c, hp, vp, vneg, N, n)
                currentC = state_value(c, n)
                if currentC < best_val - 1e-12:
                    best_val = currentC
                    best_h = hp[:n].copy()

    h_cand = binary_project(project_sum(hp[:n], 0.5 * float(n)), n)
    if n <= 1024 and time_left(t0, budget_s) > 2.0:
        val_cand = exact_c5(h_cand, n)
    else:
        val_cand = state_value(circular_from_h(h_cand, n, exact=False), n)

    if val_cand < best_val - 1e-12:
        best_h = h_cand.copy()
        best_val = val_cand
    return best_h.copy(), float(best_val)


def local_search_continuous(h0, n, budget_s, seed):
    n = int(n)
    budget_s = float(budget_s)
    if n < 1:
        return np.zeros(0, dtype=np.float64), 0.0
    if budget_s <= 3.0:
        h = project_sum(np.asarray(h0, dtype=np.float64), 0.5 * float(n))
        val = state_value(circular_from_h(h, n, exact=True), n)
        return h, val

    h, c, hp, vp, vneg = make_state(h0, n, exact=(n <= 1024), binary=False)
    N = 2 * n - 1
    dx = 2.0 / float(n)
    currentC = state_value(c, n)
    best_h = hp[:n].copy()
    best_val = currentC

    rng = np.random.default_rng(int(seed))
    t0 = time.time()
    t_end = t0 + budget_s - 1.5

    if n <= 200:
        P = 8
        R = 4
        M = 8
        S = 10
    else:
        P = 6
        R = 3
        M = 6
        S = 6

    temp_start = 0.03
    temp_floor = 1e-5
    stall = 0
    applied = 0

    while time.time() < t_end:
        G, _ = active_gradients(c, hp, vp, n, N, M)

        ki = min(P, n)
        kj = min(P, n)
        pos_i = n - ki
        hi = np.argpartition(G, pos_i)[-ki:]
        pos_j = max(0, kj - 1)
        lo = np.argpartition(G, pos_j)[:kj]

        hi_r = rng.integers(0, n, size=min(R, n))
        lo_r = rng.integers(0, n, size=min(R, n))
        all_high = np.unique(np.concatenate((hi, hi_r)))
        all_low = np.unique(np.concatenate((lo, lo_r)))

        A_high = {}
        for xx in all_high:
            xx = int(xx)
            A_high[xx] = pair_a_vector(vneg, hp, N, xx)
        A_low = {}
        for xx in all_low:
            xx = int(xx)
            A_low[xx] = pair_a_vector(vneg, hp, N, xx)

        candidates = []
        counter = 0
        for ii in all_high:
            ii = int(ii)
            hai = float(hp[ii])
            if hai >= 1.0:
                continue
            for jj in all_low:
                jj = int(jj)
                haj = float(hp[jj])
                dmax = min(1.0 - hai, haj)
                if dmax <= 1e-8:
                    continue
                score = (float(G[ii]) - float(G[jj])) * dmax
                if score <= 1e-10:
                    continue
                candidates.append((score, counter, ii, jj, dmax, A_high[ii], A_low[jj]))
                counter += 1

        if not candidates:
            break

        if len(candidates) > S:
            candidates.sort(reverse=True)
            candidates = candidates[:S]

        best_moveC = currentC + 1e6
        best_d = 0.0
        best_i = -1
        best_j = -1

        for _score, _ctr, ii, jj, dmax, Ai, Aj in candidates:
            diff = Ai - Aj
            p1 = (ii - jj) % N
            p2 = (jj - ii) % N
            for fr in _D_FRACS:
                d = float(fr) * float(dmax)
                if d <= 1e-9:
                    continue
                arr = c + d * diff
                d2 = d * d
                arr[0] -= 2.0 * d2
                arr[p1] += d2
                arr[p2] += d2
                val = float(np.max(arr)) * dx
                if val < best_moveC:
                    best_moveC = val
                    best_d = d
                    best_i = ii
                    best_j = jj

        if best_i < 0:
            break

        deltaC = best_moveC - currentC
        frac = (time.time() - t0) / max(1e-9, budget_s)
        if frac > 1.0:
            frac = 1.0
        T = temp_floor + (temp_start - temp_floor) * (0.5 ** frac)

        accept = (best_moveC < currentC - 1e-12)
        if not accept:
            stall += 1
            if T > 1e-8 and deltaC >= -1e-12:
                ratio = deltaC / T
                if ratio < 700.0:
                    p = math.exp(-ratio)
                else:
                    p = 0.0
                if rng.random() < p:
                    accept = True
            if not accept and stall > 2000 and best_moveC < currentC + 0.03:
                accept = True
                stall = 0
            if not accept and stall > 8000:
                accept = True
                stall = 0

        if accept:
            d = float(best_d)
            d = min(d, 1.0 - float(hp[best_i]), float(hp[best_j]))
            if d > 1e-9:
                Ai = pair_a_vector(vneg, hp, N, best_i)
                Aj = pair_a_vector(vneg, hp, N, best_j)
                apply_move_from_A(c, hp, vp, vneg, N, best_i, best_j, d, Ai, Aj)
                hp[:n] = np.clip(hp[:n], 0.0, 1.0)
                currentC = state_value(c, n)
                if currentC < best_val - 1e-12:
                    best_val = currentC
                    best_h = hp[:n].copy()
                stall = 0
                applied += 1

    h_cand = project_sum(hp[:n], 0.5 * float(n))
    if n <= 1024 and time_left(t0, budget_s) > 2.0:
        val_cand = exact_c5(h_cand, n)
    else:
        val_cand = state_value(circular_from_h(h_cand, n, exact=False), n)

    if val_cand < best_val - 1e-12:
        best_h = h_cand.copy()
        best_val = val_cand
    return best_h.copy(), float(best_val)


def _soft_c5_func(h, n, tau):
    h = np.asarray(h, dtype=np.float64)
    n = int(n)
    tau = float(tau)
    dx = 2.0 / float(n)
    c = circular_from_h(h, n, exact=True)
    c = np.maximum(c, 0.0)
    z = c * tau
    zm = float(np.max(z))
    return float(dx * (zm + float(np.log(np.sum(np.exp(z - zm))))) / tau)


def _soft_grad_func(h, n, tau):
    h = np.asarray(h, dtype=np.float64)
    n = int(n)
    tau = float(tau)
    dx = 2.0 / float(n)
    N = 2 * n - 1
    c = circular_from_h(h, n, exact=True)
    c = np.maximum(c, 0.0)
    z = c * tau
    zm = float(np.max(z))
    w = np.exp(z - zm)
    s = float(np.sum(w))
    if s <= 0.0:
        w[:] = 0.0
    else:
        w /= s

    hp = np.zeros(N, dtype=np.float64)
    vp = np.zeros(N, dtype=np.float64)
    hp[:n] = h
    vp[:n] = 1.0 - h

    G = np.zeros(n, dtype=np.float64)
    for k in range(N):
        wk = float(w[k])
        if wk <= 1e-12:
            continue
        G += wk * (np.roll(vp, k)[:n] - np.roll(hp, -k)[:n])
    return dx * G


def _slsqp_objective(x):
    return _soft_c5_func(np.asarray(x, dtype=np.float64), _SLSQP_N, _SLSQP_TAU)


def _slsqp_gradient(x):
    return _soft_grad_func(np.asarray(x, dtype=np.float64), _SLSQP_N, _SLSQP_TAU)


def _slsqp_sum(x):
    return float(np.sum(np.asarray(x, dtype=np.float64))) - 0.5 * float(_SLSQP_N)


def _slsqp_sum_jac(x):
    return np.ones(_SLSQP_N, dtype=np.float64)


def refine_with_slsqp(h0, n, budget_s, tau, maxiter):
    global _SLSQP_N, _SLSQP_TAU
    if _scipy_minimize is None:
        return None
    n = int(n)
    budget_s = float(budget_s)
    if n <= 0 or n > 220 or budget_s < 10.0:
        return None

    h0 = project_sum(np.asarray(h0, dtype=np.float64), 0.5 * float(n))
    val0 = _soft_c5_func(h0, n, float(tau))
    best_h = h0.copy()
    best_val = val0

    _SLSQP_N = n
    _SLSQP_TAU = float(tau)

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))

    constraints = []
    item = {}
    item["type"] = "eq"
    item["fun"] = _slsqp_sum
    item["jac"] = _slsqp_sum_jac
    constraints.append(item)

    options = {}
    options["maxiter"] = int(maxiter)
    options["ftol"] = 1e-10
    options["disp"] = False
    options["maxls"] = 49

    try:
        res = _scipy_minimize(
            _slsqp_objective,
            h0,
            method="SLSQP",
            jac=_slsqp_gradient,
            bounds=bounds,
            constraints=constraints,
            options=options,
        )
    except Exception:
        res = None

    if res is not None and getattr(res, "x", None) is not None:
        h1 = project_sum(np.clip(np.asarray(res.x, dtype=np.float64), 0.0, 1.0), 0.5 * float(n))
        val1 = _soft_c5_func(h1, n, float(tau))
        if val1 < best_val:
            best_h = h1
            best_val = val1

    return best_h.copy(), float(best_val)


def better_solution(h, n, best_h, best_val, best_n, t_start, budget_s):
    if h is None:
        return best_h, best_val, best_n
    n = int(n)
    if n <= 0:
        return best_h, best_val, best_n
    h = np.asarray(h, dtype=np.float64).copy()
    if h.size != n:
        return best_h, best_val, best_n

    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - 0.5 * float(n)) > 1e-8:
        h = project_sum(h, 0.5 * float(n))
        h = np.clip(h, 0.0, 1.0)

    if n <= 1024 and time_left(t_start, budget_s) > 5.0:
        val = exact_c5(h, n)
    else:
        c = circular_from_h(h, n, exact=False)
        val = float(np.max(c)) * (2.0 / float(n))

    if val < best_val - 1e-12:
        return h.copy(), float(val), int(n)
    return best_h, best_val, best_n


def finalize_best(best_h, best_n, best_val, t_start, budget_s):
    if best_h is None:
        rr = np.random.default_rng(0)
        h = random_binary(128, rr)
        n = 128
        val = exact_c5(h, n)
        return h, float(val), int(n)

    n = int(best_n)
    h = np.asarray(best_h, dtype=np.float64).copy()
    if h.size != n:
        h = np.clip(h, 0.0, 1.0)
        h = project_sum(h, 0.5 * float(h.size))
        n = int(h.size)

    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - 0.5 * float(n)) > 1e-8:
        h = project_sum(h, 0.5 * float(n))
        h = np.clip(h, 0.0, 1.0)

    if n <= 4096 and time_left(t_start, budget_s) > 2.0:
        val = exact_c5(h, n)
    else:
        c = circular_from_h(h, n, exact=False)
        val = float(np.max(c)) * (2.0 / float(n))

    if not np.isfinite(val) or val > best_val:
        val = float(best_val)
    return h, float(val), int(n)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    budget_s = float(budget_s)

    best_h = None
    best_val = math.inf
    best_n = 0

    parent = get_parent()
    if parent is not None:
        h_parent = np.asarray(parent, dtype=np.float64).copy()
        n_parent = int(h_parent.size)
        if abs(float(h_parent.sum()) - 0.5 * float(n_parent)) > 1e-8:
            h_parent = project_sum(h_parent, 0.5 * float(n_parent))
        best_h = h_parent.copy()
        best_val = exact_c5(h_parent, n_parent)
        best_n = n_parent

    if time_left(t_start, budget_s) < 15.0:
        return finalize_best(best_h, best_n, best_val, t_start, budget_s)

    base_budget = max(10.0, budget_s - 60.0)
    rng = np.random.default_rng(int(seed))
    seed_idx = 1

    if parent is not None and int(parent.size) <= 250 and time_left(t_start, budget_s) > 30.0:
        pn = int(parent.size)
        native_budget = min(0.05 * base_budget, time_left(t_start, budget_s) - 60.0)
        if native_budget > 8.0:
            h0 = parent_continuous_start(parent, pn)
            h_res, _ = local_search_continuous(h0, pn, native_budget, seed_idx)
            best_h, best_val, best_n = better_solution(h_res, pn, best_h, best_val, best_n, t_start, budget_s)
            seed_idx += 1

            if pn <= 100 and time_left(t_start, budget_s) > 60.0:
                sl_budget = min(35.0, time_left(t_start, budget_s) - 60.0)
                if sl_budget > 8.0:
                    sl_res = refine_with_slsqp(h_res, pn, sl_budget, 50.0, 40)
                    if sl_res is not None:
                        h_sl, _ = sl_res
                        best_h, best_val, best_n = better_solution(h_sl, pn, best_h, best_val, best_n, t_start, budget_s)
                    seed_idx += 1

    if parent is not None:
        pn = int(parent.size)
        targets = []
        for mult in (2, 4, 8, 16, 32, 64):
            n = int(pn) * int(mult)
            if n % 2 != 0:
                n += 1
            if n > 4096:
                break
            if n not in targets:
                targets.append(n)
        fractions = (0.13, 0.20, 0.23, 0.20, 0.13, 0.08)
    else:
        targets = (128, 256, 512, 1024, 2048, 4096)
        fractions = (0.12, 0.20, 0.24, 0.20, 0.13, 0.11)

    for idx_target in range(len(targets)):
        n = int(targets[idx_target])
        if time_left(t_start, budget_s) - 30.0 <= 15.0:
            break
        fr = float(fractions[idx_target]) if idx_target < len(fractions) else 0.05
        slice_budget = base_budget * fr
        rem = time_left(t_start, budget_s) - 30.0
        if slice_budget > rem:
            slice_budget = rem
        if slice_budget < 8.0:
            slice_budget = max(0.0, min(rem, 25.0))
            if slice_budget < 5.0:
                continue

        if parent is not None:
            task_specs = []
            task_specs.append((0.50, "parent"))
            if slice_budget > 50.0:
                task_specs.append((0.25, "noise"))
                task_specs.append((0.25, "random"))
            else:
                task_specs.append((0.50, "random"))
        else:
            task_specs = []
            task_specs.append((0.40, "random"))
            task_specs.append((0.30, "thue"))
            task_specs.append((0.30, "random2"))

        for fr_task, kind in task_specs:
            rem = time_left(t_start, budget_s) - 30.0
            if rem <= 10.0:
                break
            task_budget = slice_budget * float(fr_task)
            if task_budget < 4.0:
                continue
            if task_budget > rem:
                task_budget = rem

            h0 = None
            if kind == "parent":
                h0 = parent_binary_start(parent, n, rng, 0.0)
            elif kind == "noise":
                h0 = parent_binary_start(parent, n, rng, 0.08)
            elif kind == "thue":
                h0 = thue_morse_binary(n, rng)
            else:
                h0 = random_binary(n, rng)

            h_res, _ = local_search_binary(h0, n, task_budget, seed_idx)
            best_h, best_val, best_n = better_solution(h_res, n, best_h, best_val, best_n, t_start, budget_s)
            seed_idx += 1

            if n <= 188 and (kind == "parent" or kind == "noise") and time_left(t_start, budget_s) > 80.0:
                sl_budget = min(30.0, 0.25 * task_budget, time_left(t_start, budget_s) - 60.0)
                if sl_budget > 8.0:
                    sl_res = refine_with_slsqp(h_res, n, sl_budget, 50.0, 40)
                    if sl_res is not None:
                        h_sl, _ = sl_res
                        best_h, best_val, best_n = better_solution(h_sl, n, best_h, best_val, best_n, t_start, budget_s)
                    seed_idx += 1

        if n <= 188 and parent is not None and time_left(t_start, budget_s) > 80.0:
            h_res = parent_binary_start(parent, n, rng, 0.0)
            h_cont = parent_continuous_start(parent, n)
            cont_budget = min(25.0, 0.10 * slice_budget, time_left(t_start, budget_s) - 60.0)
            if cont_budget > 8.0:
                h_c, _ = local_search_continuous(h_cont, n, cont_budget, seed_idx)
                best_h, best_val, best_n = better_solution(h_c, n, best_h, best_val, best_n, t_start, budget_s)
                seed_idx += 1

    return finalize_best(best_h, best_n, best_val, t_start, budget_s)
