```python
import numpy as np
import time

def evaluate_c5(h, n):
    """Compute C5 for binary h of length n."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def count_ones(h):
    return int(np.sum(h > 0.5))

def local_search(h, n, rng, max_iters=10000, time_limit=10):
    """Local search by flipping bits."""
    start = time.time()
    current_c5 = evaluate_c5(h, n)
    best_c5 = current_c5
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        i = rng.integers(0, n)
        if h[i] > 0.5:
            if count_ones(h) <= n // 2:
                continue  # Can't remove a one
        else:
            if count_ones(h) >= n // 2:
                continue  # Can't add a one
        
        # Flip
        h[i] = 1.0 - h[i]
        new_c5 = evaluate_c5(h, n)
        
        if new_c5 <= current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Undo
            h[i] = 1.0 - h[i]
    
    return best_h, best_c5
```
