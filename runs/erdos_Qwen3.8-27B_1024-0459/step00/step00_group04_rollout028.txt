import numpy as np
import time

def evaluate_overlap(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def smooth_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shift = (corr - max_c) / tau
    return dx * (max_c + tau * np.log(np.sum(np.exp(shift))))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    exp_c = np.exp((corr - max_c) / tau)
    w = dx * exp_c / np.sum(exp_c)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        s = m - (n - 1)
        wm = w[m]
        if wm < 1e-10:
            continue
        lo1 = max(0, -s)
        hi1 = min(n - 1, n - 1 - s)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wm * (1.0 - h[lo1 + s:hi1 + s + 1])
        lo2 = max(0, s)
        hi2 = min(n - 1, n - 1 + s)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= wm * h[lo2 - s:hi2 - s + 1]
    return grad

def project(h, target_sum):
    lo = -1.5
    hi = 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def binary_round(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def local_search_binary(h_bin, budget_s, max_iter=None):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    best_val = evaluate_overlap(h)
    start = time.time()
    it_count = 0
    improved = True
    while improved and time.time() - start < budget_s:
        if max_iter is not None and it_count >= max_iter:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            if time.time() - start > budget_s:
                break
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                val = evaluate_overlap(h)
                if val < best_val - 1e-12:
                    best_val = val
                    improved = True
                    it_count += 1
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    return h, best_val

def optimize_continuous(h_init, budget_s, tau_start=0.08, tau_end=0.003):
    n = len(h_init)
    target = n / 2.0
    h = project(h_init.copy(), target)
    best_h = h.copy()
    best_val = evaluate_overlap(h)
    lr = 0.03
    start = time.time()
    it = 0
    last_improve = 0
    last_improve_val = best_val
    while time.time() - start < budget_s:
        remaining = budget_s - (time.time() - start)
        if remaining < 0.5:
            break
        progress = (time.time() - start) / budget_s
        tau = tau_start * (tau_end / tau_start) ** min(progress * 1.5, 3.0)
        obj = smooth_obj(h, tau)
        grad = smooth_grad(h, tau)
        accepted = False
        for _ in range(15):
            h_new = project(h - lr * grad, target)
            obj_new = smooth_obj(h_new, tau)
            if obj_new < obj:
                h = h_new
                accepted = True
                break
            lr *= 0.5
        if not accepted:

            if np.random.random() < 0.05:

        exact_val = evaluate_overlap(h)
        if exact_val < best_val - 1e-9:
            best_val = exact_val
            best_h = h.copy()
            last_improve = it
        if (it - last_improve) % 100 == 99:

            bin_val = evaluate_overlap(h_bin)

        bin_val = evaluate_overlap(h_bin)
