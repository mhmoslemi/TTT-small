import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    """Project h onto the feasible set: 0<=h_i<=1, sum(h)=n/2"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    # Iterative projection
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        # Distribute the correction
        if diff > 0:
            # Need to increase sum; add to those not at 1
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            add = diff / np.sum(mask)
            h[mask] += add
        else:
            # Need to decrease sum; subtract from those not at 0
            mask = h > 1e-10
            if not np.any(mask):
                break
            sub = (-diff) / np.sum(mask)
            h[mask] -= sub
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    global initial_h_values
    h_init_global = None
    try:
        h_init_global = np.array(initial_h_values, dtype=float)
    except:
        pass
    
    if h_init_global is not None:
        best_h = h_init_global.copy()
        n_best = len(best_h)
    else:
        n_best = 60
        best_h = np.ones(n_best) / 2.0
    
    best_c5 = compute_c5(best_h)
    
    # Resolutions to try
    resolutions = sorted(set([n_best, 60, 80, 100, 120, 150, 200]))
    
    best_overall_h = best_h.copy()
    best_overall_c5 = best_c5
    best_overall_n = n_best
    
    for n in resolutions:
        if time.time() - start_time > budget_s - 30:
            break
        
        # Create initial guesses for this resolution
        inits = []
        
        # From best solution (interpolated)
        if len(best_overall_h) >= 2:
            x_old = np.linspace(0, 2, len(best_overall_h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_overall_h)
            inits.append(make_feasible(h_interp, n))
        
        # From initial_h_values if different resolution
        if h_init_global is not None and len(h_init_global) != n:
            x_old = np.linspace(0, 2, len(h_init_global), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp2 = np.interp(x_new, x_old, h_init_global)
            inits.append(make_feasible(h_interp2, n))

            h_rand = np.random.rand(n) * 2.0
            inits.append(make_feasible(h_rand, n))

            h_struct = np.tile([1, 1, 0, 0], n // 4)

                    best_overall_h = h_final.copy()

    for idx in range(n_steps):
        beta = beta_start * (beta_end / beta_start) ** (idx / max(n_steps - 1, 1))
        def obj_smooth(x):
            return smooth_max_objective(x, n, beta)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: sum_constraint(x, n)}]
        try:
            res = minimize(obj_smooth, current, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1500, 'ftol': 1e-15})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
        except Exception:
            pass
    return current, best_c5

def local_perturbation(h, n, n_trials=2000, step_size=0.05):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        h_try = h.copy()
        n_mod = np.random.randint(1, 6)
        idxs = np.random.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += np.random.uniform(-step_size, step_size)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.005) & (h_try < 0.995)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            h = h_try
            best_c5 = c5_try
    return h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def generate_structured_init(n, rng):
    inits = []
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h1.copy())
        h2 = np.tile([1.0, 0.0, 1.0, 0.0], n // 4)
        inits.append(h2.copy())
    if n % 2 == 0:
        h3 = np.zeros(n)
        h3[:n // 4] = 1.0
        h3[n // 2:n // 2 + n // 4] = 1.0
        inits.append(h3.copy())
    # Thue-Morse inspired
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm, n))
    # Greedy balanced
    h4 = np.zeros(n)
    positions = sorted(range(n), key=lambda i: (i * 7919) % n)
    for j in range(n // 2):
        h4[positions[j]] = 1.0
    inits.append(make_feasible(h4, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    global initial_h_values
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 60
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: Low-resolution exploration for diverse patterns
    low_res = [12, 16, 20, 24, 30]
    low_res_solutions = []

    for n in low_res:
        if remaining() < 60:
            break
        inits = generate_structured_init(n, rng)
        for _ in range(3):
            inits.append(np.random.rand(n) * 2.0)

        for h0 in inits:
            if remaining() < 10:
                break
            h0 = make_feasible(h0, n)
            h_opt, c5_val = optimize_slsq(n, h0, max_iter=500)
            if h_opt is not None and c5_val < 0.55:
                low_res_solutions.append((h_opt.copy(), c5_val, n))

    # Sort by quality
    low_res_solutions.sort(key=lambda x: x[1])

    # Phase 2: Main optimization at various resolutions
    # Build list of initial guesses for each resolution
    resolutions = sorted(set([best_n]))
    for n_try in [best_n, 47, 50, 54, 60, 64, 80, 90, 100, 120, 140, 160, 200]:
        if n_try not in resolutions:
            resolutions.append(n_try)
    resolutions.sort()

    for n in resolutions:
        if remaining() < 30:
            break

        inits = []

        # From current best (interpolated)
        h_interp = interpolate_to_n(best_h, n)
        inits.append(make_feasible(h_interp, n))

        # From parent
        if h_parent is not None and len(h_parent) != n:
            h_p = interpolate_to_n(h_parent, n)
            h_p = make_feasible(h_p, n)
            if h_p is not None:
                inits.append(h_p)

        # From good low-res solutions (upsampled)
        for (h_lr, c5_lr, n_lr) in low_res_solutions[:3]:
            if n_lr != n:
                h_up = interpolate_to_n(h_lr, n)
                inits.append(make_feasible(h_up, n))

        # Structured
        for h_s in generate_structured_init(n, rng):
            inits.append(make_feasible(h_s, n))

        # Random
        n_rand = 2 if remaining() > 200 else 1
        for _ in range(n_rand):
            h_r = rng.rand(n) * 1.5
            inits.append(make_feasible(h_r, n))

        # Remove duplicates (approximately)
        unique_inits = [inits[0]]
        for h in inits[1:]:
            if all(np.max(np.abs(h - u)) > 0.05 for u in unique_inits):
                unique_inits.append(h)
        inits = unique_inits[:8]

        # Optimize each
        for h0 in inits:
            if remaining() < 15:
                break

            # Direct SLSQP
            if remaining() > 20:
                h_opt, c5_val = optimize_slsq(n, h0, max_iter=2000)
                if h_opt is not None and c5_val < best_c5 - 1e-10:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
                    best_n = n

            # Smooth annealing
            if remaining() > 20:
                h_opt2, c5_val2 = optimize_smooth_anneal(n, h0)
                if c5_val2 < best_c5 - 1e-10:
                    best_c5 = c5_val2
                    best_h = h_opt2.copy()
                    best_n = n

            # Local perturbation polish
            if remaining() > 10 and c5_val is not None and c5_val < best_c5 + 0.01:
                h_pol, c5_pol = local_perturbation(h0, n, n_trials=500, step_size=0.03)
                if c5_pol < best_c5 - 1e-10:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    best_n = n

    # Phase 3: Final polish on the best solution
    if remaining() > 15:
        # Tight SLSQP
        h_fin, c5_fin = optimize_slsq(best_n, best_h, max_iter=5000, ftol=1e-16)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 4: Local perturbation fine-tuning
    if remaining() > 20:
        n_trials = min(5000, int((remaining() - 10) / 0.003))
        best_h, best_c5 = local_perturbation(best_h, best_n,
                                            n_trials=n_trials, step_size=0.02)

    # Phase 5: Try Nelder-Mead as a final local optimizer
    if remaining() > 30:
        n = best_n
        def nm_obj(x):
            x_clip = np.clip(x, 0.0, 1.0)
            return compute_c5(x_clip)
        try:
            res_nm = minimize(nm_obj, best_h, method='Nelder-Mead',
                            options={'maxiter': 20000, 'xatol': 1e-12, 'fatol': 1e-14,
                                    'initial_simplex_size': 0.02})
            h_nm = make_feasible(res_nm.x, n)
            c5_nm = compute_c5(h_nm)
            if c5_nm < best_c5:
                best_c5 = c5_nm
                best_h = h_nm.copy()
        except Exception:
            pass

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
