from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def project_onto_feasible(h, n_points, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    # Need to adjust: find lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_mid = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Final adjustment
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute the difference
        # Add to elements that are strictly between 0 and 1
        free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if np.sum(free) > 0:
            h_proj[free] += diff / np.sum(free)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(dx * np.max(corr))

def project_feasible(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection to find lambda
    lo, hi = -3.0, 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_tmp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_tmp) > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        # Spread the difference
        n_free = np.sum((h_proj > 1e-15) & (h_proj < 1.0 - 1e-15))
        if n_free > 0:
            h_proj[(h_proj > 1e-15) & (h_proj < 1.0 - 1e-15)] += diff / n_free
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

```python
def subgrad_vec(h, n_points):
    dx = 2.0 / n_points
    h_c = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_c, 1.0 - h_c, mode="full")
    k_idx = int(np.argmax(corr))
    k = k_idx - (n_points - 1)
    grad = np.zeros(n_points)
    
    if k >= 0:
        # (1-h[i+k]) for i = 0..n-k-1
        grad[:n_points-k] += (1.0 - h_c[k:])
        # -h[i-k] for i = k..n-1
        grad[k:] -= h_c[:n_points-k]
    else:
        kp = -k
        # (1-h[i-kp]) for i = kp..n-1
        grad[kp:] += (1.0 - h_c[:n_points-kp])
        # -h[i+kp] for i = 0..n-kp-1
        grad[:n_points-kp] -= h_c[kp:]
    
    grad *= dx
    return grad, k, corr
```

Let me also think about the gradient approach more carefully. The issue with pure subgradient is that it only looks at the worst k, and might oscillate between different worst k values. A better approach might be:

1. Use a smooth approximation: C₅_smooth = τ * log(Σ exp(corr[k] * dx / τ))
2. The gradient is: ∇C₅_smooth = (1/τ) * Σ_k exp(corr[k]*dx/τ) * ∇(corr[k]*dx) / Σ_k exp(corr[k]*dx/τ)

This is equivalent to a weighted average of the gradients of all f_k, weighted by their "activeness" (softmax).

For small τ, this approaches the hard max (weight on the active k). For large τ, it's more uniform.

Let me implement this:

```python
def smooth_c5_grad(h, n, tau=0.1):
    dx = 2.0 / n
    h_c = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h_c, 1.0 - h_c, mode="full") * dx
    # Softmax weights
    max_corr = np.max(corr)
    weights = np.exp((corr - max_corr) / tau)
    weights /= np.sum(weights)
    
    # Weighted gradient
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-6:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[:n-k] += weights[k_idx] * (1.0 - h_c[k:])
            grad[k:] -= weights[k_idx] * h_c[:n-k]
        else:
            kp = -k
            grad[kp:] += weights[k_idx] * (1.0 - h_c[:n-kp])
            grad[:n-kp] -= weights[k_idx]
