import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def enforce_constraints(h, n_points):
    target = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-12:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_c5(h, n):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    max_val = np.max(corr)
    tol = 1e-10
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)

        tau = k_idx - (n - 1)

            grad[t:] += -dx * h[:n-t]

def subgradient_c5(h, n):

1. Identify the active set (shifts k where g_k is close to the maximum)
2. Minimize the average of the active g_k's (or use a smoothed max)
3. Use gradient descent on the smoothed objective

The smoothed max can be approximated by:
softmax(corr * dx / temperature) for small temperature

Or we can use the log-sum-exp trick:
log(Σ exp(g_k / T)) * T ≈ max g_k for small T

The gradient of the log-sum-exp is a weighted average of the individual gradients, where the weights are proportional to exp(g_k / T).

This gives a smooth approximation to the max, and we can use standard gradient descent on it.

Let me implement this:

```python
def smooth_c5(h, n, T=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    c = corr.max()
    return c + T * np.log(np.sum(np.exp((corr - c) / T)))

def smooth_grad_c5(h, n, T=0.01):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = corr.max()
    weights = np.exp((corr - c) / T)
    weights /= weights.sum()
    
    omh = 1.0 - h
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        t = k - (n - 1)
        if t >= 0:
            g[t:] += weights[k] * dx * omh[:n-t]
            g[:n-t] += weights[k] * (-dx) * h[t:]
        else:
            t = -t
            g[:n-t] += weights[k] * dx * omh[t:]
            g[t:] += weights[k] * (-dx) * h[:n-t]
    return g
```

This is O(n^2) per gradient evaluation (since we loop over all 2n-1 shifts). For n=45, that's about 88 * 45 = 3960 operations per gradient. For n=180, it's 359 * 180 = 64620. Still manageable.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def enforce(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if d > 0:
            m = h < 1.0 - 1e-12
        else:
            m = h > 1e-12
        if not np.any(m):
            break
        h[m] += d / np.sum(m)
        h = np.clip(h, 0.0, 1.0)
    # Final check
    if abs(h.sum() - target) > 1e-8:
        # Last resort: uniform adjustment
        h = h + (target - h.sum()) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_active(h, n, k_idx):
    """Gradient of g_k at h for a specific shift index k_idx in the full correlate."""
    dx = 2.0 / n
    t = k_idx - (n - 1)
    omh = 1.0 - h
    g = np.zeros(n)
    if t >= 0:
        g[t:] += dx * omh[:n-t]
        g[:n-t] -= dx * h[t:]
    else:
        t = -t
        g[:n-t] += dx * omh[t:]
        g[t:] -= dx * h[:n-t]
    return g

def smooth_opt(h0, n, time_limit):
    """Smooth (log-sum-exp) optimization."""
    h = enforce(h0.copy(), n)
    c5 = compute_c5(h, n)
    best_h, best_c5 = h.copy(), c5
    
    t0 = time.time()
    T = 0.03
    lr = 0.03
    
    while time.time() - t0 < time_limit:
        dx = 2.0 / n
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full') * dx
        c_max = corr.max()
        
        # Soft max and weights
        exp_vals = np.exp((corr - c_max) / T)
        w = exp_vals / exp_vals.sum()
        
        # Gradient (weighted sum of individual gradients)
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-6:
                continue
            t = k - (n - 1)
            if t >= 0:
                g[t:] += w[k] * dx * omh[:n-t]
                g[:n-t] -= w[k] * dx * h[t:]
            else:
                t = -t
                g[:n-t] += w[k] * dx * omh[t:]
                g[t:] -= w[k] * dx * h[:n-t]
        
        h_new = enforce(h - lr * g, n)
        c5_new = compute_c5(h_new, n)
        
        if c5_new < c5 - 1e-12:
            h, c5 = h_new, c5_new
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
            T = max(T * 0.995, 0.002)
            lr = min(lr * 1.01, 0.15)
        else:
            lr *= 0.6
            if lr < 0.002:
                lr = np.random.uniform(0.01, 0.05)
                T = max(T * 0.8, 0.005)
                h = enforce(h + np.random.randn(n) * 0.05, n)
                c5 = compute_c5(h, n)
    
    return best_h, best_c5

def local_opt(h0, n, time_limit):
    """Local search with structured perturbations."""
    h = enforce(h0.copy(), n)
    c5 = compute_c5(h, n)
