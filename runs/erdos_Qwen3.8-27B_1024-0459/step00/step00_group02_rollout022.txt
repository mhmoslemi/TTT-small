    if abs(diff) > 1e-10:

        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

                if len(idx) > 0:

    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to add mass; add to elements < 1
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        else:
            # Need to remove mass; remove from elements > 0
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_max(corr_vals, dx, tau):
    vals = corr_vals * dx / tau
    vmax = np.max(vals)
    return tau * (vmax + np.log(np.sum(np.exp(vals - vmax))))

def subgradient_c5(h, dx):
    """Compute subgradient of C5 at h (gradient of the active correlation)"""
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    max_idx = np.argmax(corr)
    # Gradient of corr[max_idx] w.r.t. h
    n = len(h)
    lag = max_idx - (n - 1)
    grad = np.zeros(n)
    # corr[k] = sum_j h[j] * (1-h[j+lag]) for valid j
    # d/dh[i] = (1-h[i+lag]) - h[i-lag] (if indices valid)
    for j in range(n):
        coeff = 0.0
        if 0 <= j + lag < n:
            coeff += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            coeff -= h[j - lag]
        grad[j] = dx * coeff
    return grad

    """PGD with projection onto constraint set"""
    target = n / 2.0
    best = h.copy()
    best_val = compute_c5(h, dx)

        # Compute subgradient
        grad = subgradient_c5(h, dx)
        
        # Take step

            if val < best_val:

If the counts are perfectly uniform, each lag would have n²/4 / (2n-1) ≈ n/8 pairs, giving C₅ ≈ (n/8) * 2/n = 0.25.

But we can't achieve perfect uniformity. The question is: what's the minimum possible maximum?

This is related to the "minimum maximum cross-correlation" of a binary sequence, which is a well-studied problem in coding theory.

For a binary sequence of length n with n/2 ones, the "aperiodic autocorrelation" is:
R(l) = sum_{j=0}^{n-1-l} (2h[j]-1)(2h[j+l]-1) for l ≥ 0

This is different from what we're computing. What we're computing is:
S(l) = sum_{j=0}^{n-1-l} h[j](1-h[j+l]) for l ≥ 0

And by symmetry consideration, S(l) + S(-l) = sum_j h[j](1-h[j+l]) + sum_j h[j](1-h[j-l])
= sum_j h[j] - sum_j h[j]h[j+l] + sum_j h[j] - sum_j h[j]h[j-l]
= 2*(n/2) - 2*sum_j h[j]h[j+l] (by symmetry of the last two sums)
= n - 2*A(l)

where A(l) = sum_j h[j]h[j+l] is the autocorrelation of the indicator.
