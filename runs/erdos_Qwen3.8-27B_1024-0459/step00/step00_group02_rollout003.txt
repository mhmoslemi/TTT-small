def resample_to(h, n_new):

_CGTGT = 0.0
_LX = None


def time_left():
    return max(0.0, float(_TDEAD - time.time()))


def c5_bound(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0 or not np.all(np.isfinite(h)):
        return float("inf")
    n = h.size
    dx = 2.0 / float(n)
    if float(np.min(h)) < -1e-12 or float(np.max(h)) > 1.0 + 1e-12:
        h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_to_mass(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > float(n):
        target = float(n)
    if n == 1:
        x[0] = min(1.0, max(0.0, target))
        return x

    low = float(np.min(x)) - target - 1.0
    high = float(np.max(x)) + target + 1.0
    for _ in range(70):
        mid = 0.5 * (low + high)
        q = np.clip(x - mid, 0.0, 1.0)
        s = float(np.sum(q))
        if s > target:
            low = mid
        else:
            high = mid

    tau = 0.5 * (low + high)
    q = np.clip(x - tau, 0.0, 1.0)
    res = target - float(np.sum(q))

    for _ in range(30):
        if abs(res) < 1e-14:
            break
        if res > 0.0:
            mask = q < 1.0 - 1e-13
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            room = (1.0 - q)[mask]
            step = min(res / float(cnt), float(np.min(room)))
            if step <= 0.0:
                break
            q[mask] += step
            res -= step * float(cnt)
        else:
            mask = q > 1e-13
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            room = q[mask]
            step = min(-res / float(cnt), float(np.min(room)))
            if step <= 0.0:
                break
            q[mask] -= step
            res += step * float(cnt)

    if abs(res) > 1e-12:
        if res > 0.0:
            idx = int(np.argmax(1.0 - q))
            q[idx] = min(1.0, q[idx] + res)
        else:
            idx = int(np.argmin(q))
            q[idx] = max(0.0, q[idx] + res)

    q = np.clip(q, 0.0, 1.0)
    return q


def prepare_and_eval(x):

        return None, float("inf")
    x = np.asarray(x, dtype=float).ravel()
    if x.size == 0:
        return None, float("inf")

def consider_candidate(x, best_h, best_c):

    if x is not None and math.isfinite(c) and (not math.isfinite(best_c) or c + 1e-12 < best_c):
        return x, float(c)
    return best_h, best_c


def compute_terms_clamped(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    om = 1.0 - h

    if n == 1:
        H = float(dx * np.dot(h, om))
        if H < 0.0:
            H = 0.0
        return H, np.empty(0, dtype=float), np.empty(0, dtype=float)

    H = float(dx * np.dot(h, om))
    if H < 0.0:
        H = 0.0

    F = np.empty(n - 1, dtype=float)
    G = np.empty(n - 1, dtype=float)
    for d in range(1, n):
        f = float(dx * np.dot(h[:n - d], om[d:]))
        g = float(dx * np.dot(h[d:], om[:n - d]))
        if f < 0.0:
            f = 0.0
        if g < 0.0:
            g = 0.0
        F[d - 1] = f
        G[d - 1] = g
    return H, F, G


def objective_and_grad(x, p, mu, target):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return 0.0, np.empty(0, dtype=float)
    if float(np.min(x)) < -1e-10 or float(np.max(x)) > 1.0 + 1e-10:
        x = np.clip(x, 0.0, 1.0)

    p = max(2, int(p))
    mu = float(mu)
    target = float(target)
    H, F, G = compute_terms_clamped(x)

    vals = np.concatenate(([H], F, G))
    S = float(np.sum(np.power(vals, p)))
    if S <= 0.0:
        obj = 0.0
        err = float(np.sum(x) - target)
        obj += 0.5 * mu * err * err
        g = mu * err * np.ones(n, dtype=float)
        return float(obj), g

    norm = S ** (1.0 / float(p))
    obj = norm
    grad = np.zeros(n, dtype=float)

    weights = np.power(vals / norm, p - 1)
    wH = float(weights[0])
    wF = weights[1:1 + n - 1]
    wG = weights[1 + n - 1:1 + 2 * (n - 1)]

    dx = 2.0 / float(n)
    if wH != 0.0:
        grad += wH * dx * (1.0 - 2.0 * x)

    if n > 1:
        for d in range(1, n):
            m = n - d
            wf = float(wF[d - 1]) * dx
            wg = float(wG[d - 1]) * dx
            if wf != 0.0:
                grad[:m] += wf * (1.0 - x[d:])
                grad[d:] -= wf * x[:m]
            if wg != 0.0:
                grad[d:] += wg * (1.0 - x[:m])
                grad[:m] -= wg * x[d:]

    err = float(np.sum(x) - target)
    if mu != 0.0:
        obj += 0.5 * mu * err * err
        grad += mu * err * np.ones(n, dtype=float)

    return float(obj), grad

    _CGTGT = target

        x1 = x0
    return project_to_mass(np.clip(x1, 0.0, 1.0), target)


def resample_to(h, n_new):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0 or n_new <= 0:
        return None
    n_new = int(n_new)
    if n == n_new:
        return project_to_mass(np.clip(h, 0.0, 1.0), float(n_new) / 2.0)
    oldx = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    newx = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    hnew = np.interp(newx, oldx, h, left=h[0], right=h[-1])
    return project_to_mass(np.clip(hnew, 0.0, 1.0), float(n_new) / 2.0)


def make_random_start(rng, n, kind=0):
    n = int(n)
    if n <= 0:
        return None
    target = float(n) / 2.0
    if kind == 1:
        r = rng.random(n)
        cut = float(np.quantile(r, target / float(n)))
        x = (r < cut).astype(float)
    elif kind == 2:
        t = np.arange(n, dtype=float)
        phase1 = float(rng.uniform(0.0, 2.0 * math.pi))
        phase2 = float(rng.uniform(0.0, 2.0 * math.pi))
        x = 0.5 + 0.40 * np.sin((2.0 * math.pi / float(n)) * t + phase1)
        x += 0.18 * np.sin((4.0 * math.pi / float(n)) * t + phase2)
    else:
        x = rng.random(n)
    return project_to_mass(np.clip(x, 0.0, 1.0), target)


def add_unique_start(starts, x, tol=1e-8):
    if x is None:
        return
    x = np.asarray(x, dtype=float).ravel()
    if x.size == 0 or not np.all(np.isfinite(x)):
        return
    for y in starts:
        if y.size == x.size and np.all(np.abs(x - y) <= tol):
            return
    starts.append(x.copy())


def active_gradient(h, C, tol):
    h = np.asarray(h, dtype=float)
    n = h.size
    H, F, G = compute_terms_clamped(h)
    if F.size:
        C = max(float(C), float(H), float(F.max()), float(G.max()))
    else:
        C = max(float(C), float(H))
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    cnt = 0.0

    if float(H) >= C - tol:
        grad += dx * (1.0 - 2.0 * h)
        cnt += 1.0

    if F.size:
        for d in range(1, n):
            idx = d - 1
            m = n - d
            if float(F[idx]) >= C - tol:
                grad[:m] += dx * (1.0 - h[d:])
                grad[d:] -= dx * h[:m]
                cnt += 1.0
            if float(G[idx]) >= C - tol:
                grad[d:] += dx * (1.0 - h[:m])
                grad[:m] -= dx * h[d:]
                cnt += 1.0

    if cnt == 0.0:
        return None
    grad /= cnt
    return grad


def subgrad_polish(h, target, max_iters=50, step0=0.02):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h, float("inf")
    n = h.size
    target = float(target)
    C = c5_bound(h)
    if not math.isfinite(C):
        return h, float("inf")
    step = float(step0)
    stall = 0

    for _ in range(int(max_iters)):
        if time_left() < 10.0:
            break
        H, F, G = compute_terms_clamped(h)
        Cdir = float(H)
        if F.size:
            Cdir = max(Cdir, float(F.max()), float(G.max()))
        Ceff = max(C, Cdir)
        tol = max(1e-6, 1e-5 * max(1.0, Ceff))
        g = active_gradient(h, Ceff, tol)
        if g is None:
            break
        ng = float(np.linalg.norm(g))
        if ng == 0.0 or not math.isfinite(ng):
            break
        g = g / ng

        improved = False
        for s in (step, step * 0.5, step * 0.25, step * 0.125, step * 0.0625):
            x = project_to_mass(h - s * g, target)
            Cx = c5_bound(x)
            if math.isfinite(Cx) and Cx < C - 1e-11:
                h = x
                C = Cx
                improved = True
                step = min(0.1, step * 1.15)
                stall = 0
                break
        if not improved:
            step *= 0.5
            stall += 1
            if step < 1e-5 or stall > 30:
                break

    return h, float(C)


def random_local_search(h, target, rng, iters, sigma):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    n = h.size
    target = float(target)
    C = c5_bound(h)
    if not math.isfinite(C):
        return h
    sig = float(sigma)
    stall = 0

    for _ in range(int(iters)):
        if time_left() < 5.0:
            break
        x = h + rng.normal(0.0, sig, n)
        x = project_to_mass(x, target)
        Cx = c5_bound(x)
        if math.isfinite(Cx) and Cx < C - 1e-12:
            h = x
            C = Cx
            stall = 0
            if sig < 0.02:
                sig *= 1.02
        else:
            stall += 1
            if math.isfinite(Cx) and Cx > C + 0.005:
                sig = max(1e-5, sig * 0.85)
            if stall > 800:
                sig = max(1e-5, sig * 0.7)
                stall = 0

    return h


def apply_delta_terms(h, H, F, G, i, new_val, dx):
    old = float(h[i])
    new_val = float(new_val)
    if old == new_val:
        return H
    delta = new_val - old
    H = float(H) + dx * (new_val * (1.0 - new_val) - old * (1.0 - old))
    n = h.size
    maxd = max(i, n - 1 - i)
    for d in range(1, maxd + 1):
        idx = d - 1
        j = i + d
        if j < n:
            hj = float(h[j])
            F[idx] += dx * (1.0 - hj) * delta
            G[idx] -= dx * hj * delta
        j = i - d
        if j >= 0:
            hj = float(h[j])
            F[idx] -= dx * hj * delta
            G[idx] += dx * (1.0 - hj) * delta
    h[i] = new_val
    if H < 0.0 and H > -1e-12:
        H = 0.0
    return H


def binary_local_search(n, rng, iters, margin):
    n = int(n)
    if n < 10:
        return None, float("inf")
    if n % 2 == 1:
        n -= 1
    target = n // 2
    best_h = None
    best_c = float("inf")
    dx = 2.0 / float(n)

    for restart in range(2):
        if time_left() < float(margin) + 5.0:
            break
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, target, replace=False).astype(int)
        h[idx] = 1.0
        H, F, G = compute_terms_clamped(h)
        C = max(float(H), float(F.max()), float(G.max()))
        ones = [int(i) for i in idx]
        zeros = [i for i in range(n) if h[i] < 0.5]
        noimp = 0

        for it in range(int(iters)):
            if time_left() < float(margin) + 5.0:
                break
            a = ones[int(rng.integers(len(ones)))] if len(ones) else None
            b = zeros[int(rng.integers(len(zeros)))] if len(zeros) else None
            if a is None or b is None or a == b:
                break

            hc = h.copy()
            Hc, Fc, Gc = H, F.copy(), G.copy()
            Hc = apply_delta_terms(hc, Hc, Fc, Gc, a, 0.0, dx)
            Hc = apply_delta_terms(hc, Hc, Fc, Gc, b, 1.0, dx)
            if Hc < 0.0:
                Hc = 0.0
            Cnew = max(Hc, float(Fc.max()), float(Gc.max()))

            if Cnew < C - 1e-13:
                h, H, F, G, C = hc, Hc, Fc, Gc, Cnew
                noimp = 0
                if C < best_c:
                    best_c = C
                    best_h = h.copy()
                ones = [i for i in range(n) if h[i] > 0.5]
                zeros = [i for i in range(n) if h[i] < 0.5]
                if it % 1000 == 0:
                    H, F, G = compute_terms_clamped(h)
                    C = max(float(H), float(F.max()), float(G.max()))
            else:
                noimp += 1
                if noimp > 8000:
                    break

    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    global _T0, _TDEAD
    _T0 = float(time.time())
    _TDEAD = _T0 + max(10.0, float(budget_s))
    rng = np.random.default_rng(int(seed))

    best_h = None
    best_c = float("inf")

    parent = None
    try:
        init = initial_h_values
        if init is not None:
            arr = np.asarray(init, dtype=float).ravel()
            if arr.size > 0:
                parent = arr
    except Exception:
        parent = None

    if parent is not None:
        best_h, best_c = consider_candidate(parent, best_h, best_c)

    if best_h is None:
        best_h, best_c = consider_candidate(make_random_start(rng, 45, 0), None, float("inf"))
        if best_h is None:
            best_h = project_to_mass(np.full(45, 0.5, dtype=float), 45.0 / 2.0)
            best_c = c5_bound(best_h)

    base_n = int(len(parent)) if parent is not None else int(len(best_h))
    base_n = max(1, base_n)

    multipliers = [1, 2, 4, 8]
    if float(budget_s) > 800.0 and base_n <= 100:
        multipliers.append(16)

    res = []
    for m in multipliers:
        nn = int(round(base_n * m))
        if nn < 1:
            nn = 1
        if nn > 720:
            nn = 720
        if nn not in res:
            res.append(nn)
    res = sorted(set(res))

    if float(budget_s) < 300.0:
        res = res[:2]
    elif float(budget_s) < 600.0:
        res = res[:3]

    for n in res:
        if time_left() < 25.0:
            break
        n = int(n)
        if n <= 0:
            continue

        starts = []
        if best_h is not None:
            add_unique_start(starts, resample_to(best_h, n))
        if parent is not None:
            add_unique_start(starts, resample_to(parent, n))
        if time_left() > 60.0:
            add_unique_start(starts, make_random_start(rng, n, 0))
        if time_left() > 90.0 and n <= 200:
            add_unique_start(starts, make_random_start(rng, n, 1))
        if time_left() > 120.0 and n <= 120:
            add_unique_start(starts, make_random_start(rng, n, 2))
        if time_left() > 150.0 and n % 2 == 0 and n <= 120:
            hb, _ = binary_local_search(n, rng, 5000, 15.0)
            add_unique_start(starts, hb)

        if n <= 60:
            it_large, it_small, it_slq = 180, 80, 40
        elif n <= 120:
            it_large, it_small, it_slq = 100, 70, 30
        elif n <= 240:
            it_large, it_small, it_slq = 60, 50, 20
        else:
            it_large, it_small, it_slq = 30, 20, 0

        for x0 in starts:
            if time_left() < 20.0:
                break

            x0, c0 = prepare_and_eval(x0)
            if x0 is None:
                continue

            if c0 < best_c - 1e-12:
                best_c = c0
                best_h = x0.copy()

            if c0 > best_c + 0.05:
                continue

            x1 = scipy_stage(x0, 16, 5000.0, "L-BFGS-B", it_large, 1e-9, 10.0)
            x1, c1 = prepare_and_eval(x1)
            if c1 < best_c - 1e-12:
                best_c = c1
                best_h = x1.copy()

            xcur = x1

            if time_left() > 30.0:
                x2 = scipy_stage(xcur, 32, 5000.0, "L-BFGS-B", it_small, 1e-9, 10.0)
                x2, c2 = prepare_and_eval(x2)
                if c2 < best_c - 1e-12:
                    best_c = c2
                    best_h = x2.copy()
                xcur = x2

            if n <= 240 and it_slq > 0 and time_left() > 40.0:
                x3 = scipy_stage(xcur, 64, 0.0, "SLSQP", it_slq, 1e-8, 10.0)
                x3, c3 = prepare_and_eval(x3)
                if c3 < best_c - 1e-12:
                    best_c = c3
                    best_h = x3.copy()
                xcur = x3

            if n <= 300 and time_left() > 20.0:
                x4, c4 = subgrad_polish(xcur, xcur.size / 2.0, max_iters=40, step0=0.02)
                x4, c4 = prepare_and_eval(x4)
                if c4 < best_c - 1e-12:
                    best_c = c4
                    best_h = x4.copy()
                xcur = x4

            if n <= 90 and time_left() > 30.0:
                xl = random_local_search(xcur, xcur.size / 2.0, rng, 3000, 0.01)
                xl, cl = prepare_and_eval(xl)
                if cl < best_c - 1e-12:
                    best_c = cl
                    best_h = xl.copy()
            elif n <= 240 and time_left() > 35.0:
                xl = random_local_search(xcur, xcur.size / 2.0, rng, 1000, 0.005)
                xl, cl = prepare_and_eval(xl)
                if cl < best_c - 1e-12:
                    best_c = cl
                    best_h = xl.copy()

    if best_c > 0.38085 + 1e-7 and time_left() > 60.0:
        for nb in (44, 46, 50, 64, 80):
            if time_left() < 30.0:
                break
            hb, cb = binary_local_search(nb, rng, 30000, 15.0)
            hb, cb = prepare_and_eval(hb)
            if cb < best_c - 1e-12:
                best_c = cb
                best_h = hb.copy()

    if parent is not None:
        best_h, best_c = consider_candidate(parent, best_h, best_c)

    best_h, best_c = prepare_and_eval(best_h)
    if best_h is None or not math.isfinite(best_c):
        best_h = project_to_mass(np.full(45, 0.5, dtype=float), 45.0 / 2.0)
        best_c = c5_bound(best_h)

    return best_h, float(best_c), int(best_h.size)
