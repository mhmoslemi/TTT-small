    n = len(h)
    s = np.arange(0, 2*n-1) - (n-1)  # shifts

def _compute_corr(h):
    """Full cross-correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_value(h):
    """Compute C5 objective."""
    n = len(h)
    return float(np.max(_compute_corr(h)) * 2.0 / n)

def _interp_up(h_src, n_src, n_dst):
    """Interpolate from n_src to n_dst points."""
    x_src = np.arange(n_src) * (2.0 / n_src)
    x_dst = np.arange(n_dst) * (2.0 / n_dst)

def _thue_morse(n):

def _fix_sum_binary(h, n, rng):
    """Ensure binary h has sum exactly n//2."""
    h = h.copy().astype(float)
    target = n // 2
    cur = int(np.round(h.sum()))
    if cur < target:

    elif cur > target:

            continue
        delta = float(rng.uniform(-step_size, step_size))
        h[i] = np.clip(h[i] + delta, 0.0, 1.0)
        h[j] = np.clip(h[j] - delta, 0.0, 1.0)
        # Fix sum

            step_size = min(step_size * 1.02, 0.1)

            h = best_h.copy()
