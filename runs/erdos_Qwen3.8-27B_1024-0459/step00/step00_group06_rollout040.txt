from scipy.optimize import NonlinearConstraint, LinearConstraint
import time

def compute_c5(h, n):

    corr = np.correlate(h, 1.0 - h, mode='full')

def correlation_vector(h, n):
    """Return the full correlation vector: corr[k] for k=0..2n-2"""
    return np.correlate(h, 1.0 - h, mode='full')

def correlation_jacobian(h, n):
    """Compute the Jacobian of the correlation vector with respect to h.
    Returns a (2n-1) x n matrix."""
    n_lags = 2 * n - 1
    jac = np.zeros((n_lags, n))
    
    for l in range(n_lags):
        shift = l - (n - 1)
        if shift >= 0:
            # corr = sum_{i=0}^{n-1-shift} h[i] * (1 - h[i+shift])
            for j in range(n):
                val = 0.0
                # j appears as h[i] when i=j, need j <= n-1-shift
                if j <= n - 1 - shift:
                    if j + shift < n:
                        val += (1.0 - h[j + shift])
                # j appears as h[i+shift] when i+shift=j, i=j-shift, need j-shift >= 0
                if j >= shift:
                    i = j - shift
                    if i <= n - 1 - shift:
                        val += -h[i]
                jac[l, j] = val
        else:
            s = -shift
            # corr = sum_{i=s}^{n-1} h[i] * (1 - h[i-s])
            for j in range(n):
                val = 0.0
                # j appears as h[i] when i=j, need j >= s
                if j >= s:
                    val += (1.0 - h[j - s])
                # j appears as h[i-s] when i-s=j, i=j+s, need j+s < n
                if j + s < n:
                    val += -h[j + s]
                jac[l, j] = val
    
    return jac
