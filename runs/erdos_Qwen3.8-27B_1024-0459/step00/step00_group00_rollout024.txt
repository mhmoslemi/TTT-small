                    break
        elif diff < -1e-10:
            idx = np.argsort(h)
            for i in idx:
                s = min(-diff, h[i])
                h[i] -= s
                diff += s
                if diff >= -1e-12:
                    break
        return h

    def grad_sub(h, n):
        dx = 2.0 / n
        corr = get_corr(h)
        m = int(np.argmax(corr))
        k = m - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            L = n - k
            g[0:L] += (1.0 - h[k:k+L]) * dx
            g[k:k+L] += (-h[0:L]) * dx
        else:
            kp = -k
            L = n - kp
            g[0:L] += (1.0 - h[kp:kp+L]) * dx
            g[kp:kp+L] += (-h[0:L]) * dx
        return g, float(np.max(corr))

    def time_left():
        return budget_s - (time.time() - start_time)

    def nesterov_optimize(h0, n, max_iters, time_budget):
        h = h0.copy()
        _, c5_init = grad_sub(h, n)
        best_local = c5_init
        best_h_local = h.copy()
        lr = 0.08
        mu = 0.92
        vel = np.zeros(n)
        y = h.copy()
        stall = 0
        for it in range(max_iters):
            if time_left() < time_budget:
                break
            h = y
            g, c5_val = grad_sub(h, n)
            if c5_val < best_local - 1e-9:
                best_local = c5_val
                best_h_local = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 400:
                lr *= 0.7
                stall = 0
                if lr < 1e-8:
                    break
            vel = mu * vel + g
            y = h - lr * vel
            h = proj(y, n)
        return best_h_local, best_local

    def local_search_binary(h_bin, n, max_rounds, time_budget):
        if n % 2 != 0:
            return h_bin, float(np.max(get_corr(h_bin)))
        ones = np.where(h_bin == 1.0)[0].tolist()
        zeros = np.where(h_bin == 0.0)[0].tolist()
        c5_cur = float(np.max(get_corr(h_bin)))
        for r in range(max_rounds):
            if time_left() < time_budget:
                break
            improved = False
            # Try random swaps
            n_try = 50
            for _ in range(n_try):
                if time_left() < time_budget:
                    break
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = int(rng.randint(0, len(ones)))
                j = int(rng.randint(0, len(zeros)))
                pos_i = ones[i]
                pos_j = zeros[j]
                h_bin[pos_i] = 0.0
                h_bin[pos_j] = 1.0
                c5_new = float(np.max(get_corr(h_bin)))
                if c5_new < c5_cur - 1e-12:
                    c5_cur = c5_new
                    ones[i] = pos_j
                    zeros[j] = pos_i
                    improved = True
                else:
                    h_bin[pos_i] = 1.0
                    h_bin[pos_j] = 0.0
            if not improved:
                # Try systematic: all pairs (expensive, limit)
                sys_improved = False
                for a in range(min(len(ones), 30)):
                    for b in range(min(len(zeros), 30)):
                        if time_left() < time_budget:
                            return h_bin, c5_cur
                        pos_i = ones[a]
                        pos_j = zeros[b]
                        h_bin[pos_i] = 0.0
                        h_bin[pos_j] = 1.0
                        c5_new = float(np.max(get_corr(h_bin)))
                        if c5_new < c5_cur - 1e-12:
                            c5_cur = c5_new
                            ones[a] = pos_j
                            zeros[a] = pos_j  # won't re-access this index
                            zeros[b] = pos_i
                            sys_improved = True
                        else:
                            h_bin[pos_i] = 1.0
                            h_bin[pos_j] = 0.0
                if not sys_improved and not improved:
                    break
        return h_bin, c5_cur

    # Resolutions to try
    res_list = []
    if init is not None:
        res_list.append(init_n)
    for n in [44, 46, 88, 90, 92, 132, 176, 180]:
        if n not in res_list:
            res_list.append(n)

    for n in res_list:
        if time_left() < 120:
            break

        # Build starting points
        starts = []

        if init is not None:
            if init_n == n:
                starts.append(proj(init.copy(), n))
            else:
                xs = np.linspace(0, 2, init_n, endpoint=False)
                xn = np.linspace(0, 2, n, endpoint=False)
                h_r = np.interp(xn, xs, np.concatenate([init, init[:1]]))
                starts.append(proj(h_r, n))

        # Thue-Morse
        h_tm = np.array([1.0 if bin(i).count('1') % 2 == 0 else 0.0 for i in range(n)])
        starts.append(proj(h_tm, n))

        # Alternating blocks
        for nblks in [5, 7, 9, 11]:
            blk_len = n // (2 * nblks)
            if blk_len < 1:
                continue
            h_b = np.zeros(n)
            for i in range(n):
                if (i // blk_len) % 2 == 0:
                    h_b[i] = 1.0
            starts.append(proj(h_b, n))

        # Random
        for _ in range(2):
            h_r2 = (rng.rand(n) < 0.5).astype(float)
            starts.append(proj(h_r2, n))

        # Sine threshold
        x_grid = np.linspace(0, 2, n, endpoint=False)
        h_sin = (np.sin(2 * np.pi * x_grid) > 0).astype(float)
        starts.append(proj(h_sin, n))

        for h0 in starts:
            if time_left() < 60:
                break

            # Nesterov gradient descent
            h_opt, c5_opt = nesterov_optimize(h0, n, 4000, 30)
            c5_eval = float(np.max(get_corr(h_opt)))
            if c5_eval < best_c5:
                best_c5 = c5_eval
                best_h = h_opt.copy()
                best_n = n

            # Binary rounding + local search
            if n % 2 == 0:
                h_bin = np.zeros(n)
                sorted_idx = np.argsort(-h_opt)
                h_bin[sorted_idx[:n//2]] = 1.0
                c5_bin = float(np.max(get_corr(h_bin)))
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin.copy()
                    best_n = n

                if time_left() > 45:
                    h_ls, c5_ls = local_search_binary(h_bin.copy(), n, 200, 30)
                    if c5_ls < best_c5:
                        best_c5 = c5_ls
                        best_h = h_ls.copy()
                        best_n = n

    # Additional refinement: try to improve best solution with more iterations
    if best_h is not None and time_left() > 60:
        n_cur = best_n
        h_refine = best_h.copy()
        h_refine, c5_refine = nesterov_optimize(h_refine, n_cur, 6000, 30)
        c5_refine_eval = float(np.max(get_corr(h_refine)))
        if c5_refine_eval < best_c5:
            best_c5 = c5_refine_eval
            best_h = h_refine

        if n_cur % 2 == 0:
            h_bin2 = np.zeros(n_cur)
            sorted_idx = np.argsort(-best_h)
            h_bin2[sorted_idx[:n_cur//2]] = 1.0
            if time_left() > 30:
                h_ls2, c5_ls2 = local_search_binary(h_bin2, n_cur, 300, 20)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2

    # Final fallback
    if best_h is None:
        n = 44
        best_h = (rng.rand(n) < 0.5).astype(float)
        best_h = proj(best_h, n)
        sorted_idx = np.argsort(-best_h)
        h_fin = np.zeros(n)
        h_fin[sorted_idx[:n//2]] = 1.0
        best_h = h_fin
        best_c5 = float(np.max(get_corr(best_h)))
        best_n = n

    return (best_h, best_c5, best_n)
