            idx = np.arange(start, start + win_size)
            t = np.linspace(0.0, 1.0, win_size)
            pert = 0.03 * np.sin(np.pi * t) * rng.choice([-1, 1])
            h_old_sum = h[idx].sum()
            h[idx] = np.clip(h[idx] + pert, 0.0, 1.0)
            diff = h_old_sum - h[idx].sum()
            if abs(diff) > 1e-12:
                others = np.setdiff1d(np.arange(n), idx)
                j = rng.choice(others)
                adj = -diff / max(1, len(others))
                h[j] = np.clip(h[j] + adj, 0.0, 1.0)
                h = _repair(h, n)

        else:
            # Pair move
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            d = rng.uniform(-0.04, 0.04)
            hi = h[i] + d
            hj = h[j] - d
            if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                it += 1
                continue
            h[i] = hi
            h[j] = hj

        nc = _eval_c5(h, n)
        dc = nc - cur

        if dc <= 0.0 or rng.random() < math.exp(-dc / T):
            cur = nc
            if cur < best_local:
                best_local = cur
                best_local_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Undo (approximate for window moves, exact for pairs)
            h = best_local_h.copy() if stall > 1000 else h
            cur = _eval_c5(h, n)
            stall += 1

        it += 1
        if it % 80000 == 0:
            T *= 0.97
            if T < 2e-5:
                T = 2e-5
        if stall > stall_limit:
            h = best_local_h.copy()
            cur = best_local
            T = max(T, T_init * 0.5)
            stall = 0

    return best_local, best_local_h


def _make_binary_init(n, rng):
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return _repair(h, n)


def _make_smooth_init(n, rng):
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2 * np.pi, n)
    h = 0.5 + 0.3 * np.sin(2 * x + phase) + 0.1 * np.sin(4 * x + 2 * phase)
    return _repair(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0) - 5.0

    # Get initial solution from parent
    try:
        h_parent = np.array(initial_h_values, dtype=np.float64)
    except:
        h_parent = np.full(47, 0.5)
    n_parent = len(h_parent)
    h_parent = _repair(h_parent, n_parent)

    best_c5 = _eval_c5(h_parent, n_parent)
    best_h = h_parent.copy()
    best_n = n_parent
    print(f"Parent: n={n_parent}, C5={best_c5:.6f}")

    # Phase 1: Intensive SA on parent resolution
    p1_end = t0 + min(400, budget_s * 0.35)
    c5_1, h_1 = _sa_optimize(h_parent, n_parent, rng, p1_end, 0.003, 0.05, 400000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h_1.copy()
        best_n = n_parent
    print(f"Phase1 (n={n_parent}): C5={best_c5:.6f}")

    # Phase 1b: Windowed SA on best
    if tleft() > 100:
        p1b_end = t0 + min(550, budget_s * 0.45)
        c5_1b, h_1b = _sa_window(h_1, n_parent, rng, p1b_end, 0.002, 5, 300000)
        if c5_1b < best_c5:
            best_c5 = c5_1b
            best_h = h_1b.copy()
            best_n = n_parent
        print(f"Phase1b (n={n_parent}): C5={best_c5:.6f}")

    # Phase 2: Try different resolutions
    resolutions_to_try = []
    for delta in [2, 3, 5, 8, 12, 20, 30, 50, 80, 150, 300]:
        n_try = n_parent + delta
        if n_try <= 600:
            resolutions_to_try.append(n_try)

    for n_try in resolutions_to_try:
        if tleft() < 40:
            break

        # Upsample from best solution
        x_old = (np.arange(n_parent) + 0.5) * (2.0 / n_parent)
        x_new = (np.arange(n_try) + 0.5) * (2.0 / n_try)
        h_start = np.interp(x_new, x_old, best_h)
        h_start = _repair(h_start, n_try)

        # Allocate time based on resolution
        if n_try < 100:
            alloc = min(120, tleft() * 0.2)
        elif n_try < 200:
            alloc = min(80, tleft() * 0.15)
        elif n_try < 400:
            alloc = min(50, tleft() * 0.1)
        else:
            alloc = min(30, tleft() * 0.08)

        p_end = time.time() + alloc

        if n_try <= 150:
            c5_try, h_try = _sa_optimize(h_start, n_try, rng, p_end, 0.002, 0.04, 300000)
        else:
            c5_try, h_try = _sa_optimize(h_start, n_try, rng, p_end, 0.001, 0.03, 200000)

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
            print(f"  Improvement at n={n_try}: C5={best_c5:.6f}")

    # Phase 3: Alternative inits at good resolutions
    n_opt = best_n
    for alt_seed in range(3):
        if tleft() < 30:
            break
        for alt_n in [n_opt, n_opt + 1, n_opt + 3]:
            if alt_n % 2 != 0 and alt_n > 4:
                continue
            h_alt = _make_binary_init(alt_n, rng)
            c5_alt = _eval_c5(h_alt, alt_n)
            if c5_alt < 0.42:
                alloc = min(60, tleft() * 0.15)
                p_end = time.time() + alloc
                c5_alt_opt, h_alt_opt = _sa_optimize(h_alt, alt_n, rng, p_end, 0.003, 0.05, 200000)
                if c5_alt_opt < best_c5:
                    best_c5 = c5_alt_opt
                    best_h = h_alt_opt.copy()
                    best_n = alt_n
                    print(f"  Binary init n={alt_n}: C5={best_c5:.6f}")

    # Phase 4: Fine-tune at best resolution
    if tleft() > 20:
        h_ft = best_h.copy()
        nf = best_n
        T_ft = 3e-5
        end_ft = time.time() + min(40, tleft() - 5)
        cur_ft = _eval_c5(h_ft, nf)
        it_ft = 0
        while time.time() < end_ft:
            i = rng.randint(nf)
            j = (i + rng.randint(1, nf)) % nf
            d = rng.uniform(-0.005, 0.005)
            hi = h_ft[i] + d
            hj = h_ft[j] - d
            if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
                continue
            h_ft[i] = hi
            h_ft[j] = hj
            nc = _eval_c5(h_ft, nf)
            dc = nc - cur_ft
            if dc <= 0.0 or rng.random() < math.exp(-dc / T_ft):
                cur_ft = nc
                if cur_ft < best_c5:
                    best_c5 = cur_ft
                    best_h = h_ft.copy()
            else:
                h_ft[i] = hi - d
                h_ft[j] = hj + d
            it_ft += 1

    # Final validation and repair
    best_h = _repair(best_h, best_n)
    final_c5 = _eval_c5(best_h, best_n)
    print(f"FINAL: n={best_n}, C5={final_c5:.6f}, budget_used={time.time()-t0:.1f}s")

    return (best_h, final_c5, best_n)
