import numpy as np
import time

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_subgrad_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    L = k_star - (n - 1)
    grad = np.zeros(n)
    if L > 0:
        idx1 = np.arange(n - L)
        grad[idx1] += (1.0 - h[idx1 + L])
        idx2 = np.arange(L, n)
        grad[idx2] += (-h[idx2 - L])
        grad *= dx
    elif L < 0:
        La = -L
        idx1 = np.arange(La, n)
        grad[idx1] += (1.0 - h[idx1 - La])
        idx2 = np.arange(n - La)
        grad[idx2] += (-h[idx2 + La])
        grad *= dx
    else:
        grad[:] = (1.0 - 2.0 * h) * dx
    return grad, L

def project_feasible(h, n):
    target = n / 2.0
    lo = -10.0
    hi = 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_optimize(h_init, time_limit, step0=0.5, decay=0.003, momentum=0.9):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5_value(h)
    t0 = time.time()
    it = 0
    while time.time() - t0 < time_limit:
        grad, L = compute_subgrad_vec(h)
        step = step0 / (1.0 + it * decay)
        v = momentum * v - step * grad
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c5 = compute_c5_value(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            v *= 0.5
        it += 1
    return best_h, best_c5

def binary_round(h):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    num_ones = int(h_bin.sum())
    target = n // 2
    if num_ones > target:
        ones = np.where(h_bin == 1)[0]
        np.random.shuffle(ones)
        for idx in ones[:num_ones - target]:
            h_bin[idx] = 0.0
    elif num_ones < target:
        zeros = np.where(h_bin == 0)[0]
        np.random.shuffle(zeros)
        for idx in zeros[:target - num_ones]:
            h_bin[idx] = 1.0
    return h_bin

def binary_local_search(h, time_limit):
    n = len(h)
    h_bin = binary_round(h)
    best_h = h_bin.copy()
    best_c5 = compute_c5_value(h_bin)
    t0 = time.time()
    temp = 0.05
    while time.time() - t0 < time_limit:
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[np.random.randint(len(ones))]
        zj = zeros[np.random.randint(len(zeros))]
        if oi == zj:
            continue
        old_oi, old_zj = h_bin[oi], h_bin[zj]
        h_bin[oi] = 0.0
        h_bin[zj] = 1.0
        c5 = compute_c5_value(h_bin)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()
        else:
            accept = np.random.random() < temp
            if not accept:
                h_bin[oi] = old_oi
                h_bin[zj] = old_zj
        temp = max(0.001, temp * 0.9995)
    return best_h, best_c5

def interpolate_up(h, target_n):
    n = len(h)
    n_new = target_n
    if n_new <= n:
        return h.copy()
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0, 1)
    return project_feasible(h_new, n_new)

def smooth_max_subgrad(h, alpha=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_corr = np.max(corr)
    weights = np.exp(alpha * (corr - max_corr))
    total_w = np.sum(weights)
    if total_w < 1e-30:
        k_star = int(np.argmax(corr))
        L = k_star - (n - 1)
        grad = np.zeros(n)
        if L > 0:
            idx1 = np.arange(n - L)
            grad[idx1] += (1.0 - h[idx1 + L])
            idx2 = np.arange(L, n)
            grad[idx2] += (-h[idx2 - L])
        elif L < 0:
            La = -L
            idx1 = np.arange(La, n)
            grad[idx1] += (1.0 - h[idx1 - La])
            idx2 = np.arange(n - La)
            grad[idx2] += (-h[idx2 + La])
        else:
            grad[:] = (1.0 - 2.0 * h)
        return grad * dx
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] / total_w < 1e-6:
            continue
        L = k_idx - (n - 1)
        wg = weights[k_idx] / total_w
        if L > 0:
            idx1 = np.arange(n - L)
            grad[idx1] += wg * (1.0 - h[idx1 + L])
            idx2 = np.arange(L, n)
            grad[idx2] += wg * (-h[idx2 - L])
        elif L < 0:
            La = -L
            idx1 = np.arange(La, n)
            grad[idx1] += wg * (1.0 - h[idx1 - La])
            idx2 = np.arange(n - La)
            grad[idx2] += wg * (-h[idx2 + La])
        else:
            grad[:] += wg * (1.0 - 2.0 * h)
    return grad * dx

def smooth_max_optimize(h_init, time_limit, step0=0.3, decay=0.002, momentum=0.9, alpha=50.0):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)

def make_periodic_candidate(n, m_periods=2, seed=42):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for k in range(n):
        phase = (x[k] * m_periods + rng.normal(0, 0.1)) % 1.0
        h[k] = 0.5 + 0.5 * np.cos(2.0 * np.pi * phase)
    h = np.clip(h, 0, 1)
    h = project_feasible(h, n)
    return h

def make_binary_candidate(n, seed=42):
    rng = np.random.RandomState(seed)
    perm = rng.permutation(n)
    h = np.zeros(n)
    h[perm[:n//2]] = 1.0
    return h
