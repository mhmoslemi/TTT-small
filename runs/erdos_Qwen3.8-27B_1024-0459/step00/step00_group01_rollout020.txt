from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    c = 1.0 - h
    corr = np.correlate(h, c, mode='full')
    return float(np.max(corr)) * dx, corr * dx

def proj(h, n):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    tgt = 0.5 * n
    for _ in range(2000):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h > 0) & (h < 1)
        if not m.any():
            # Need to unclip something
            if d > 0:
                idx = np.argmin(h)  # most room to increase
            else:
                idx = np.argmax(h)  # most room to decrease
            h[idx] += d
            h = np.clip(h, 0, 1)
        else:
            h[m] += d / m.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smo_max(x, b):
    if b < 1e-6:
        return float(x.max())
    m = x.max()
    return float(m + np.log(np.exp(b * (x - m)).sum()) / b)

def obj_fn(h, n, dx, beta, pen):
    _, cd = eval_c5(h, dx)
    v = smo_max(cd, beta)
    s = h.sum() - 0.5 * n
    return v + pen * s * s

        return obj_fn(x, n, dx, beta, pen)

                 bounds=[(0.0, 1.0)] * n,
                 options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
    return proj(r.x, n)

def subgrad_step(h, n, dx, step_size, top_k=5):
    """Take a subgradient step to reduce the max correlation"""
    _, cd = eval_c5(h, dx)
    # Find top k lags
    idx = np.argsort(cd)[::-1][:top_k]
    # Subgradient: average of gradients of top-k correlations
    # Gradient of corr[k] w.r.t. h[i] for the full correlate
    # Let's compute it numerically for the top lags
    eps = 1e-6
    grad = np.zeros(n)
    for k in idx:
        # corr[k] corresponds to lag = k - (n-1)
        # Compute numerical gradient
        for i in range(n):
            hp = h.copy()
            hp[i] = min(1.0, h[i] + eps)
            hp = proj(hp, n)
            _, cd_p = eval_c5(hp, dx)
            grad[i] += (cd_p[k] - cd[k]) / eps
    grad /= len(idx)
    # Step
    h_new = h - step_size * grad
    h_new = proj(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode='full')
    return float(np.max(corr)) * dx, corr * dx

def proj(h, n):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    tgt = 0.5 * n
    for _ in range(5000):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        if not m.any():
            if d > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + d, 0.0, 1.0)
        else:
            h[m] += d / m.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def lse_max(x, beta):
    m = x.max()
    if beta < 1e-6:
        return float(m)
    s = np.exp(beta * (x - m)).sum()
    return float(m + np.log(s) / beta)

def analytic_grad_max(h, n, dx, beta, top_k=10):
    """Compute gradient of softmax-max of correlations w.r.t. h using analytical form."""
    _, cd = eval_c5(h, dx)
    # Weights for softmax
    m = cd.max()
    weights = np.exp(beta * (cd - m))
    wsum = weights.sum()
    if wsum < 1e-300:
        wsum = 1.0
    # Use top-k for efficiency
    idx_top = np.argsort(cd)[::-1][:top_k]
    w_top = weights[idx_top] / wsum
    grad = np.zeros(n)
    for ki, k in enumerate(idx_top):
        lag = k - (n - 1)
        w = w_top[ki] * dx
        if lag >= 0:
            l = lag
            for j in range(l, n):
                grad[j] += w * (1.0 - h[j - l])
            for j in range(0, n - l):
                grad[j] -= w * h[j + l]
        else:
            m2 = -lag
            for j in range(m2, n):
                grad[j] += w * (1.0 - h[j - m2])
            for j in range(0, n - m2):
                grad[j] -= w * h[j + m2]
    return grad

def smooth_obj(h, n, dx, beta, pen):
    _, cd = eval_c5(h, dx)
    v = lse_max(cd, beta)
    s = h.sum() - 0.5 * n
    return v + pen * s * s

def lbf_opt(h0, n, dx, beta, pen, maxiter=400):
    h0 = proj(h0, n)
    def f_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        h = proj(h, n)
        val = smooth_obj(h, n, dx, beta, pen)
        g = analytic_grad_max(h, n, dx, beta, top_k=15)
        # Add penalty gradient
        s = h.sum() - 0.5 * n
        g = g + 2.0 * pen * s
        return val, g
    res = minimize(f_and_grad, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-14, 'maxls': 50})
    return proj(res.x, n)

def subgrad_polish(h, n, dx, n_iter=200, top_k=5):
    """Direct subgradient descent on the max correlation."""
    h = h.copy()
    best = h.copy()
    best_c, _ = eval_c5(h, dx)
    step = 0.02
    for it in range(n_iter):
        _, cd = eval_c5(h, dx)
        idx_top = np.argsort(cd)[::-1][:top_k]
        # Compute subgradient (average of gradients at active lags)
        grad = np.zeros(n)
        for k in idx_top:
            lag = k - (n - 1)
            if lag >= 0:
                l = lag
                for j in range(l, n):
                    grad[j] += (1.0 - h[j - l])
                for j in range(0, n - l):
                    grad[j] -= h[j + l]
            else:
                m2 = -lag
                for j in range(m2, n):
                    grad[j] += (1.0 - h[j - m2])
                for j in range(0, n - m2):
                    grad[j] -= h[j + m2]
        grad /= len(idx_top)
        # Stochastic gradient step with line search
        for trial_step in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_new = h - trial_step * grad
            h_new = proj(h_new, n)
            c_new, _ = eval_c5(h_new, dx)
            if c_new < best_c:
                best_c = c_new
                best = h_new.copy()
                h = h_new.copy()
                break
        step *= 0.995
    return best, best_c

def binary_search_c5(h_start, n, dx, n_trials=5000):
    """Optimize a binary (0/1) solution via pairwise swaps."""
    h = np.zeros(n)
    ones = np.sort(np.argsort(h_start))[:n // 2]
    h[ones] = 1.0
    c_best, _ = eval_c5(h, dx)
    best_h = h.copy()
    for trial in range(n_trials):
        # Pick random 1 and random 0
        pos1 = ones[np.random.randint(len(ones))]
        zeros = np.where(h == 0.0)[0]
        pos0 = zeros[np.random.randint(len(zeros))]
        if pos1 == pos0:
            continue
        h[pos1] = 0.0
        h[pos0] = 1.0
        c_new, _ = eval_c5(h, dx)
        if c_new < c_best:
            c_best = c_new
            best_h = h.copy()
        else:
            # Revert
            h[pos1] = 1.0
            h[pos0] = 0.0
    return best_h, c_best

def resample(h_old, n_old, n_new):
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    # Initialize from parent
    if 'initial_h_values' in globals():
        hb = np.array(initial_h_values, dtype=float).copy()
        nb = len(hb)
    else:
        nb = 47
        hb = np.full(nb, 0.5)
    hb = proj(hb, nb)
    db = 2.0 / nb
    cb, _ = eval_c5(hb, db)

    # Phase 1: Annealed L-BFGS-B optimization
    h_cur = hb.copy()
    betas = [2, 5, 10, 20, 40, 80, 160, 320, 640]
    for b in betas:
        if rem() < 30:
            break
        h_new = lbf_opt(h_cur, nb, db, b, 1e6, maxiter=500)
        c_new, _ = eval_c5(h_new, db)
        h_cur = h_new.copy()
        if c_new < cb:
            cb = c_new
            hb = h_new.copy()

    # Phase 2: Subgradient polishing
    if rem() > 20:
        h_pol, c_pol = subgrad_polish(hb, nb, db, n_iter=300, top_k=5)
        if c_pol < cb:
            cb = c_pol
            hb = h_pol.copy()

    # Phase 3: Multi-start with perturbations
    n_starts = min(40, int(rem() / 3)) if rem() > 30 else 0
    for s in range(n_starts):
        if rem() < 10:
            break
        scale = 0.03 + 0.08 * np.random.random()
        h_p = hb + np.random.normal(0, scale, nb)
        h_p = proj(h_p, nb)
        h_opt = lbf_opt(h_p, nb, db, 80, 1e6, maxiter=300)
        c_opt, _ = eval_c5(h_opt, db)
        if c_opt < cb:
            cb = c_opt
            hb = h_opt.copy()
            # Re-polish
            h_pol, c_pol = subgrad_polish(hb, nb, db, n_iter=100, top_k=5)
            if c_pol < cb:
                cb = c_pol
                hb = h_pol.copy()

    # Phase 4: Different resolutions
    resolutions = [48, 56, 64, 80, 96, 128, 160]
    for n_try in resolutions:
        if rem() < 60:
            break
        h_res = resample(hb, nb, n_try)
        h_res = proj(h_res, n_try)
        d_try = 2.0 / n_try
        h_opt = lbf_opt(h_res, n_try, d_try, 50, 1e6, maxiter=300)
        c_opt, _ = eval_c5(h_opt, d_try)
        if c_opt < cb:
            cb = c_opt
            hb = h_opt.copy()
            nb = n_try
            db = d_try
            # Polish at this resolution
            h_pol, c_pol = subgrad_polish(hb, nb, db, n_iter=150, top_k=5)
            if c_pol < cb:
                cb = c_pol
                hb = h_pol.copy()

    # Phase 5: Binary/semi-binary optimization (for even n)
    if rem() > 30:
        n_bin = nb if nb % 2 == 0 else nb + 1
        if n_bin % 2 == 0:
            d_bin = 2.0 / n_bin
            h_b = resample(hb, nb, n_bin)
            h_b = proj(h_b, n_bin)
            h_bin, c_bin = binary_search_c5(h_b, n_bin, d_bin, n_trials=min(50000, int(rem() * 100)))
            if c_bin < cb:
                cb = c_bin
                hb = h_bin.copy()
                nb = n_bin
                db = d_bin

    # Phase 6: Final fine-tuning with very high beta
    if rem() > 15:
        h_fin = lbf_opt(hb, nb, db, 2000, 1e6, maxiter=300)
        c_fin, _ = eval_c5(h_fin, db)
        if c_fin < cb:
            cb = c_fin
            hb = h_fin.copy()
        # One more subgradient pass
        h_pol, c_pol = subgrad_polish(hb, nb, db, n_iter=200, top_k=3)
        if c_pol < cb:
            cb = c_pol
            hb = h_pol.copy()

    # Final projection and verification
    hb = proj(hb, nb)
    cb, _ = eval_c5(hb, db)

    return hb, cb, nb
