    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    max_c = np.max(corr)
    scaled = T * (corr - max_c)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    obj = max_c + (1.0/T) * np.log(exp_scaled.sum())

        if np.any(valid2):
            grad[p] -= np.dot(w[valid2], h[p + d[valid2]])
    grad *= dx

from scipy.optimize import minimize
from scipy.ndimage import zoom
import time

def true_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    if T <= 0:
        return max_c
    shifted = T * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)  # max is 0
    logsum = np.log(np.exp(shifted).sum())
    return max_c + logsum / T

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    if T <= 0:
        # Subgradient of max: uniform over argmax
        w = np.zeros(2*n-1)
        w[np.argmax(corr)] = 1.0
    else:
        shifted = T * (corr - max_c)
        shifted = np.clip(shifted, -500, 0)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
    
    d = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for p in range(n):
        v1 = (p - d)
        valid1 = (v1 >= 0) & (v1 < n)
        if np.any(valid1):
            grad[p] += np.dot(w[valid1], 1.0 - h[v1[valid1]])
        v2 = (p + d)
        valid2 = (v2 >= 0) & (v2 < n)
        if np.any(valid2):
            grad[p] -= np.dot(w[valid2], h[v2[valid2]])
    grad *= dx
    return grad

def penalty_obj(h, T, M):
    n = len(h)
    base = smooth_obj(h, T)
    pen = M * (np.sum(h) - n/2.0)**2
    return base + pen

def penalty_grad(h, T, M):
    n = len(h)
    base_g = smooth_grad(h, T)
    pen_g = 2.0 * M * (np.sum(h) - n/2.0) * np.ones(n)
    return base_g + pen_g

def upsample(h, new_n):
    """Upsample h to new_n points using linear interpolation, then normalize."""
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # Renormalize
    h_new = h_new * (new_n / 2.0) / np.sum(h_new)
    h_new = np.clip(h_new, 0, 1)
    return h_new

def optimize_smoothed(h0, T_seq, M, max_iter=2000):
    """Optimize using L-BFGS-B with smoothed objective and penalty."""
    n = len(h0)
    x = h0.copy()
    best_x = x.copy()
    best_val = true_c5(x)
    
    for T in T_seq:
        res = minimize(
            penalty_obj, x,
            jac=penalty_grad,
            args=(T, M),
            method='L-BFGS-B',
            bounds=[(0, 1)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        x = res.x
        val = true_c5(x)
        if val < best_val:
            best_val = val
            best_x = x.copy()
    
    return best_x, best_val

def local_search_01(h0, n, iters=100000):
    """Local search on 0-1 valued h with exactly n/2 ones."""
    h = (h0 > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = np.sum(h)
    target = n // 2
    if n_ones > target:
        # Flip some 1s to 0s (start with the "least useful" ones)
        idx1 = np.where(h == 1)[0]
        rng = np.random.default_rng(42)
        rng.shuffle(idx1)
        for i in idx1[:int(n_ones - target)]:
            h[i] = 0
    elif n_ones < target:
        idx0 = np.where(h == 0)[0]
        rng = np.random.default_rng(42)
        rng.shuffle(idx0)
        for i in idx0[:int(target - n_ones)]:
            h[i] = 1
    
    current_c5 = true_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    rng = np.random.default_rng(42)
    
    for it in range(iters):
        # Try flipping two positions (one 1->0, one 0->1) to maintain sum constraint
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.integers(len(ones))
        j = rng.integers(len(zeros))
        pi = ones[i]
        pj = zeros[j]
        
        h[pi] = 0
        h[pj] = 1
        new_c5 = true_c5(h)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[pi] = 1
            h[pj] = 0
    
    return best_h, best_c5

Hmm, the local search with full C₅ recomputation at each step is O(n²) per iteration (for the correlation), so 100000 iterations * 32400 ops = 3.2 billion ops. Too slow for n=180.

Let me speed it up. For a single swap (flip positions pi from 1 to 0 and pj from 0 to 1), how does the correlation change?

corr[m] = Σ_i h[i+d_m] * (1-h[i])

After changing h[pi] from 1 to 0 and h[pj] from 0 to 1:
- For each m, the change in corr[m] is:
  - Terms involving h[pi]: 
    - h[pi] appeared as h[i+d_m] when i+d_m=pi, i=pi-d_m: contribution was (1-h[pi-d_m]), now 0. Change: -(1-h[pi-d_m])
    - h[pi] appeared in (1-h[i]) when i=pi: contribution was -h[pi+d_m]*h[pi] = -h[pi+d_m], now -h[pi+d_m]*0=0. Change: +h[pi+d_m]
  - Similarly for h[pj] going from 0 to 1:
    - h[pj] as h[i+d_m] when i+d_m=pj: contribution was 0, now (1-h[pj-d_m]). Change: +(1-h[pj-d_m])
    - h[pj] in (1-h[i]) when i=pj: contribution was -h[pj+d_m]*0=0, now -h[pj+d_m]*1. Change: -h[pj+d_m]

So Δcorr[m] = -(1-h[pi-d_m]) + h[pi+d_m] + (1-h[pj-d_m]) - h[pj+d_m]
             = h[pi+d_m] - h[pj+d_m] + (1-h[pj-d_m]) - (1-h[pi-d_m])
             = h[pi+d_m] - h[pj+d_m] + h[pi-d_m] - h[pj-d_m]

This is O(n) to compute for all m. And checking if max(corr+Δcorr) < current_c5 is also O(n).

So each swap evaluation is O(n), and 100000 iterations * 180 = 18 million ops. That's fast!

Wait, I need to double-check the delta formula. Let me re-derive.

Original: corr[m] = Σ_i h[i+d] * (1-h[i]) where d = m-(n-1).

After changing h[pi]: 1→0, h[pj]: 0→1.

New corr[m] = Σ_i h'[i+d] * (1-h'[i])

The difference comes only from terms where h' differs from h, i.e., at positions pi and pj.

Terms involving position pi:
- In h'[i+d] (first factor): i+d = pi → i = pi-d. Original: h[pi]*(1-h[pi-d]) = 1*(1-h[pi-d]). New: 0*(1-h[pi-d]) = 0. Change: -(1-h[pi-d]).
- In (1-h'[i]) (second factor): i = pi. Original: h[pi+d]*(1-h[pi]) = h[pi+d]*0 = 0. New: h[pi+d]*(1-0) = h[pi+d]. Change: +h[pi+d].

Terms involving position pj:
- In h'[i+d] (first factor): i+d = pj → i = pj-d. Original: h[pj]*(1-h[pj-d]) = 0. New: 1*(1-h[pj-d]) = 1-h[pj-d]. Change: +(1-h[pj-d]).
- In (1-h'[i]) (second factor): i = pj. Original: h[pj+d]*(1-h[pj]) = h[pj+d]*1 = h[pj+d]. New: h[pj+d]*(1-1) = 0. Change: -h[pj+d].

Total delta[m]:
Δcorr[m] = -(1-h[pi-d]) + h[pi+d] + (1-h[pj-d]) - h[pj+d]

where each term is 0 if the index is out of valid range [0, n-1].

Let me verify: d ranges from -(n-1) to n-1.
- h[pi-d]: index pi-d, valid if 0 ≤ pi-d ≤ n-1, i.e., d ≤ pi and d ≥ pi-(n-1)
- h[pi+d]: index pi+d, valid if 0 ≤ pi+d ≤ n-1, i.e., -pi ≤ d ≤ n-1-pi
- h[pj-d]: valid if 0 ≤ pj-d ≤ n-1
- h[pj+d]: valid if 0 ≤ pj+d ≤ n-1

So:
Δcorr[m] = -[0≤pi-d<n] * (1-h[pi-d]) + [0≤pi+d<n] * h[pi+d] + [0≤pj-d<n] * (1-h[pj-d]) - [0≤pj+d<n] * h[pj+d]

This is O(n) to compute for all m simultaneously (vectorized).

        delta = np.zeros(2*n-1)
        v = pi + d
        m = (v >= 0) & (v < n)
        delta[m] += h[v[m]]
        v = pj + d
        m = (v >= 0) & (v < n)
        delta[m] -= h[v[m]]
