        h_struct = make_structured_init(n3, rng, rng.randint(0, 3))
        s_struct = compute_C5(h_struct, n3)
        s_up3 = compute_C5(h_up3, n3)
        if s_struct < s_up3:
            h_work3 = h_struct
        else:
            h_work3 = h_up3
        t_sub = max(time.time(), t0 + budget_s * 0.72)
        t_sub_end = min(t_end3, t_sub + (t_end3 - t_sub) * 0.7)
        h3a, s3a = local_search(h_work3, n3, rng, t_sub_end, T_init=0.006)
        # Fine-tuning with structured perturbation
        t_sub2_end = t_end3
        if time.time() < t_sub2_end - 3:
            h3b, s3b = local_search(h3a, n3, rng, t_sub2_end, T_init=0.004)
            h3, s3 = (h3b, s3b) if s3b < s3a else (h3a, s3a)
        else:
            h3, s3 = h3a, s3a
        if s3 < best_score:
            best_h, best_score, n_best = h3, s3, n3

    # Phase 4: Final polish on best solution (remaining time)
    t_end4 = t0 + budget_s - 5
    if time.time() < t_end4 - 3:
        h_pol, s_pol = local_search(best_h, n_best, rng, t_end4, T_init=0.003)
        if s_pol < best_score:
            best_h, best_score, n_best = h_pol, s_pol, n_pol if 'n_pol' in dir() else n_best
            best_h, best_score, n_best = h_pol, s_pol, n_best

    # Final validation
    best_h = project_onto_feasible(best_h, n_best)
    best_score = compute_C5(best_h, n_best)

    # Sanity check
    if best_score > 0.5:
        # Fallback to constant
        best_h = np.full(n_best, 0.5)
        best_h = project_onto_feasible(best_h, n_best)
        best_score = compute_C5(best_h, n_best)

    return (best_h, best_score, n_best)
