    corr = np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

        if abs(diff) > 1e-12:
            if diff > 0:
                free = np.where(h < 1.0)[0]
                if len(free) > 0:
                    per = diff / len(free)
                    if per <= 1.0 - h[free[0]]:
                        h[free] += per
                        h = np.clip(h, 0.0, 1.0)

            print(f"After CD pass 2: C5 = {best_c5:.8f}")
    
    # Phase 4: Try different starting points
    elapsed = time.time() - start_time
    if elapsed < budget_s - 120:
        candidates = structured_initial(n, rng)
        for h_cand in candidates:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 40:
                break
            h_try = coordinate_descent_fast(h_cand, n, max_passes=min(100, int((budget_s - elapsed - 30) / 2.0)))
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c5_try
                print(f"After alt start: C5 = {best_c5:.8f}")
    
    # Phase 5: Try finer resolution
    elapsed = time.time() - start_time
    if elapsed < budget_s - 100:
        for n_alt in [94, 96, 100]:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 50:
                break
            dx_alt = 2.0 / n_alt
            target_alt = n_alt / 2.0
            
            # Interpolate parent to new resolution
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n_alt, endpoint=False) + 1.0 / n_alt
            h_interp = np.interp(x_new, x_old, h0, period=2.0)
            h_alt = project_to_sum(h_interp, target_alt)
            
            # Quick CD
            max_pass = max(1, min(50, int((budget_s - elapsed - 30) / 3.0)))
            h_opt = coordinate_descent_fast(h_alt, n_alt, max_passes=max_pass)
            h_opt = project_to_sum(h_opt, target_alt)
            c5_alt = compute_c5(h_opt, dx_alt)
            
            if c5_alt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_alt
                n = n_alt
                print(f"After n={n_alt} CD: C5 = {best_c5:.8f}")
    
    # Phase 6: Final polishing CD
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        h_final = coordinate_descent_fast(best_h, len(best_h), max_passes=100)
        h_final = project_to_sum(h_final, len(best_h) / 2.0)
        c5_final = compute_c5(h_final, 2.0 / len(best_h))
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    # Final validation
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, n_final / 2.0)
    final_c5 = compute_c5(best_h, dx_final)
    
    elapsed = time.time() - start_time
    print(f"FINAL: C5 = {final_c5:.8f}, n = {n_final}, time = {elapsed:.1f}s")
    
    return (best_h, final_c5, n_final)
