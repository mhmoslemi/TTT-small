        n0 = len(start_h)
    else:
        n0 = n_default
        start_h = np.full(n0, 0.5)

    # Evaluate initial
    try_update(start_h, n0)

    # Phase 1: Optimize at initial resolution with multiple starts
    _OPT_N = n0
    bounds_n = [(0.0, 1.0)] * n0

    n_starts = max(6, int(30 * (42.0 / n0)))
    for si in range(n_starts):
        if time.time() > deadline - 120:
            break
        if si == 0:
            h0 = start_h.copy()
        elif si == 1 and best_h is not None:
            h0 = interp_to_n(best_h, n0)
        elif si == 2:
            h0 = rand_binary(n0, rng)
        elif si == 3:
            h0 = rand_binary(n0, rng)
        else:
            h0 = project_feasible(best_h.copy() + rng.normal(0, 0.05, n0), n0)

        # L-BFGS-B (box constraints only, project after)
        try:
            res = minimize(_cb_obj, h0, jac=_cb_grad, method='L-BFGS-B',
                           bounds=bounds_n,
                           options={'maxiter': 1000, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 100})
            h_res = project_feasible(res.x, n0)
            try_update(h_res, n0)
        except Exception:
            pass

        # SLSQP with equality constraint (polishing)
        if time.time() > deadline - 90:
            break
        h_polish = project_feasible(res.x if 'res' in dir() and res is not None else h0, n0)
        try:
            res2 = minimize(_cb_obj, h_polish, jac=_cb_grad, method='SLSQP',
                           bounds=bounds_n,
                           constraints=[{'type': 'eq', 'fun': _cb_sum, 'jac': _cb_sum_jac}],
                           options={'maxiter': 500, 'ftol': 1e-15})
            try_update(res2.x, n0)
        except Exception:
            try:
            # Fallback without constraints
            pass
        except Exception:
            pass

    # Phase 2: Binary local search at initial resolution
    if time.time() < deadline - 180:
        _OPT_N = n0
        h_bin = rand_binary(n0, rng)
        h_bin, c5_bin = binary_swap_search(h_bin, n0, rng, max_iters=30000,
                                           timeout=deadline - 180, t_start=t0)
        try_update(h_bin, n0)

        # Try smoothing the binary solution
        h_smooth = smooth_binary(h_bin, n0, sigma=0.3)
        try_update(h_smooth, n0)

        # Optimize the smoothed version
        _OPT_N = n0
        try:
            res = minimize(_cb_soft_obj, h_smooth, jac=_cb_soft_grad, method='L-BFGS-B',
                           bounds=bounds_n,
                           options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-8})
            try_update(res.x, n0)
        except Exception:
            pass

    # Phase 3: Multiresolution - interpolate to higher resolutions and optimize
    resolutions = sorted(set([n0, n0*2, n0*3, n0*4, 84, 120, 168, 210, 252, 336, 420]))
    resolutions = [r for r in resolutions if r >= n0 and r % 2 == 0 and r <= 512]

    for n in resolutions:
        if time.time() > deadline - 60:
            break
        if n == n0:
            continue

        _OPT_N = n
        bounds_n = [(0.0, 1.0)] * n

        # Interpolate best solution
        h_interp = interp_to_n(best_h, n)
        try_update(h_interp, n)

        # Optimize
        n_starts_high = max(2, int(8 * (42.0 / n)))
        for si in range(n_starts_high):
            if time.time() > deadline - 50:
                break
            if si == 0:
                h0 = h_interp.copy()
            else:
                h0 = project_feasible(h_interp + rng.normal(0, 0.02, n), n)

            max_iter = max(200, int(3000 * (42.0 / n)))
            try:
                res = minimize(_cb_obj, h0, jac=_cb_grad, method='L-BFGS-B',
                               bounds=bounds_n,
                               options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 50})
                try_update(res.x, n)
            except Exception:
                pass

            # SLSQP polish
            if time.time() > deadline - 40:
                break
            h_pol = project_feasible(res.x, n) if 'res' in dir() else h0
            try:
                res2 = minimize(_cb_obj, h_pol, jac=_cb_grad, method='SLSQP',
                               bounds=bounds_n,
                               constraints=[{'type': 'eq', 'fun': _cb_sum, 'jac': _cb_sum_jac}],
                               options={'maxiter': 300, 'ftol': 1e-14})
                try_update(res2.x, n)
            except Exception:
                pass

        # Binary search at this resolution
        if time.time() < deadline - 40 and n <= 168:
            h_bin = rand_binary(n, rng)
            h_bin, _ = binary_swap_search(h_bin, n, rng, max_iters=10000,
                                          timeout=deadline - 40, t_start=t0)
            try_update(h_bin, n)
            h_sm = smooth_binary(h_bin, n, sigma=0.3)
            try_update(h_sm, n)

    # Phase 4: Final polishing at best resolution
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        _OPT_N = n_best
        bounds_n = [(0.0, 1.0)] * n_best
        try:
            res = minimize(_cb_soft_obj, best_h, jac=_cb_soft_grad, method='L-BFGS-B',
                           bounds=bounds_n,
                           options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-10})
            try_update(res.x, n_best)
        except Exception:
            pass
        try:
            h_pol = project_feasible(best_h, n_best)
            res2 = minimize(_cb_obj, h_pol, jac=_cb_grad, method='SLSQP',
                           bounds=bounds_n,
                           constraints=[{'type': 'eq', 'fun': _cb_sum, 'jac': _cb_sum_jac}],
                           options={'maxiter': 500, 'ftol': 1e-15})
            try_update(res2.x, n_best)
        except Exception:
            pass

    # Final validation
    if best_h is None:
        best_h = start_h
        best_c5 = _obj_val(best_h, n0)

    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final)
    best_c5 = _obj_val(best_h, n_final)

    return best_h, best_c5, n_final
