            except Exception:

        # --- Random restarts (if budget allows and n is small enough) ---
        if remaining(0.30) > 60 and n <= 200:
            for rs in range(3):
                if remaining(0.32) < 30:
                    break
                try:
                    h_r, c5_r = _random_restart_optimize(
                        n, seed=seed + rs * 1000 + n,
                        budget_s=min(40, remaining(0.33)))
                    if h_r is not None and c5_r < best_c5 - 1e-10:
                        best_c5 = c5_r
                        best_h = h_r.copy()
                        best_n = n
                except Exception:
                    pass

    # Final feasibility projection and evaluation
    best_h = _project_feasible(best_h, best_n)
    final_c5 = _compute_c5(best_h)

    # Verify constraints
    if np.any(best_h < -1e-9) or np.any(best_h > 1.0 + 1e-9):
        return parent.copy(), c5_par, n_par
    if abs(np.sum(best_h) - best_n / 2.0) > 0.05:
        return parent.copy(), c5_par, n_par

    if time.time() - t_start > budget_s * 0.97:
        return parent.copy(), c5_par, n_par

    return best_h, final_c5, best_n
