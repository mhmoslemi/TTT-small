def objective_jac(x):
    x = np.asarray(x, dtype=float)
    n=_N
    g=np.zeros(n, dtype=float)
    valsA=term_values(x)
    Cvals=valsA * _DX
    if _SCALE <= 0.0:
        maxc = float(np.max(Cvals))
        act = (maxc - Cvals) <= float(_ACTIVE_TOL)
        if not np.any(act):
            act = (Cvals == maxc)
        count = float(np.count_nonzero(act))
        weights = np.zeros(_NTERM, dtype=float)
        weights[act] = 1.0 / count
    else:
        maxc=float(np.max(Cvals))
        e=np.exp((Cvals - maxc)/float(_SCALE))
        z=float(e.sum())
        if z <= 0.0:
            weights=np.full(_NTERM, 1.0/_NTERM)
        else:
            weights=e/z
    thr = 1e-13
    m_arr=_TERM_M; s_arr=_TERM_SIGN
    for l in range(_NTERM):
       wl=float(weights[l])
       if wl <= thr: continue
       mm=int(m_arr[l]); sgn=int(s_arr[l])
       if sgn == 0:
          g += wl * (1.0 - 2.0*x)
       elif sgn > 0:
          if mm < n:
             g[:n-mm] += wl * (1.0 - x[mm:])
             g[mm:] -= wl * x[:n-mm]
       else:
          if mm < n:
             g[mm:] += wl * (1.0 - x[:n-mm])
             g[:n-mm] -= wl * x[mm:]
    g *= _DX
    if _PENALTY > 0.0:
       err = float(np.mean(x)) - 0.5
       g += (2.0 * float(_PENALTY) / float(n)) * err
    return g

def upsample(h, new_n):
    h=np.asarray(h,dtype=float).reshape(-1)
    old_n=int(h.size)
    new_n=int(new_n)
    if old_n == new_n:
       return project_sum(h,new_n)
    if old_n <=0:
       return np.full(new_n, 0.5)
    if new_n > 0 and old_n > 0 and new_n % old_n == 0:
       return project_sum(np.repeat(h, new_n // old_n), new_n)
    if old_n == 1:
       val = float(np.clip(h[0],0,1))
       return project_sum(np.full(new_n,val), new_n)

    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    hnew = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_sum(hnew, new_n)

def build_terms(n):
    global _N, _DX, _TERM_M, _TERM_SIGN, _NTERM
    n = int(n)
    if n != _N or _TERM_M is None or len(_TERM_M) != 2 * n - 1:
        _N = n
        _DX = 2.0 / float(n)
        total = 2 * n - 1
        m = np.zeros(total, dtype=np.int64)
        s = np.zeros(total, dtype=np.int64)
        idx = 1
        for k in range(1, n):
            m[idx] = k
            s[idx] = -1
            idx += 1
            m[idx] = k
            s[idx] = 1
            idx += 1

def project_sum(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    if h.size != n:
        if h.size == 0:
            h = np.full(n, 0.5, dtype=float)
        else:
            h = np.resize(h, n)
    h = np.clip(h.copy(), 0.0, 1.0)
    target = 0.5 * float(n)
    diff = target - float(np.sum(h))
    if diff > 1.0e-14:
        if float(np.max(h)) + diff / float(n) <= 1.0:
            h += diff / float(n)
        else:
            slack = 1.0 - h
            order = np.argsort(-slack)
            rem = diff
            for i in order:
                if rem <= 1.0e-15:
                    break
                add = min(float(slack[i]), rem)
                if add > 0.0:
                    h[i] += add
                    rem -= add
    elif diff < -1.0e-14:
        if float(np.min(h)) + diff / float(n) >= 0.0:
            h += diff / float(n)
        else:
            order = np.argsort(-h)
            rem = -diff
            for i in order:
                if rem <= 1.0e-15:
                    break
                dec = min(float(h[i]), rem)
                if dec > 0.0:
                    h[i] -= dec
                    rem -= dec
    return np.clip(h, 0.0, 1.0)


def finalize_h(h, n):
    n = int(n)
    h = project_sum(h, n)
    target = 0.5 * float(n)
    for _ in range(2):
        s = float(np.sum(h))
        err = target - s
        if abs(err) <= 1.0e-13:
            break
        req = target - (s - h)
        inside = (req >= 0.0) & (req <= 1.0)
        if np.any(inside):
            cand = np.flatnonzero(inside)
            d = np.abs(req - h)
            i = int(cand[int(np.argmin(d))])
            h[i] = float(np.clip(req[i], 0.0, 1.0))
        else:
            if err > 0.0:
                slack = 1.0 - h
                order = np.argsort(-slack)
                rem = err
                for idx in order:
                    if rem <= 1.0e-15:
                        break
                    add = min(float(slack[idx]), rem)
                    if add > 0.0:
                        h[idx] += add
                        rem -= add
            else:
                order = np.argsort(-h)
                rem = -err
                for idx in order:
                    if rem <= 1.0e-15:
                        break
                    dec = min(float(h[idx]), rem)
                    if dec > 0.0:
                        h[idx] -= dec
                        rem -= dec
        h = np.clip(h, 0.0, 1.0)
    if n > 1:
        val = target - float(np.sum(h[:-1]))
        if val >= -1.0e-12 and val <= 1.0 + 1.0e-12:
            h[-1] = float(np.clip(val, 0.0, 1.0))
    return h


def exact_c5(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(h.size)
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def upsample(h, new_n):
    h = np.asarray(h, dtype=float).reshape(-1)
    old_n = int(h.size)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([], dtype=float)
    if old_n == new_n:
        return project_sum(h, new_n)
    if old_n <= 0:
        return project_sum(np.full(new_n, 0.5, dtype=float), new_n)
    if old_n > 0 and new_n % old_n == 0:
        return project_sum(np.repeat(h, new_n // old_n), new_n)
    if old_n == 1:
        val = float(np.clip(h[0], 0.0, 1.0))
        return project_sum(np.full(new_n, val, dtype=float), new_n)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    hnew = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    return project_sum(hnew, new_n)


def downsample(h, new_n):
    h = np.asarray(h, dtype=float).reshape(-1)
    old_n = int(h.size)
    new_n = int(new_n)
    if new_n <= 0:
        return np.array([], dtype=float)
    if old_n <= new_n:
        return upsample(h, new_n)
    if old_n <= 0:
        return project_sum(np.full(new_n, 0.5, dtype=float), new_n)
    if old_n % new_n == 0:
        return project_sum(h.reshape(new_n, old_n // new_n).mean(axis=1), new_n)
    old_c = (np.arange(old_n, dtype=float) + 0.5) * (2.0 / float(old_n))
    new_c = (np.arange(new_n, dtype=float) + 0.5) * (2.0 / float(new_n))
    hnew = np.interp(new_c, old_c, h, left=h[0], right=h[-1])
    return project_sum(hnew, new_n)


def transformed_start(h, n, target_n):

    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:

def term_values(x):
    n = _N
    vals = np.empty(_NTERM, dtype=float)
    vals[0] = float(np.dot(x, 1.0 - x))
    idx = 1
    for k in range(1, n):
        a = x[:-k]
        b = x[k:]
        vals[idx] = float(np.dot(b, 1.0 - a))
        vals[idx + 1] = float(np.dot(a, 1.0 - b))
        idx += 2
    return vals


def soft_objective(x):
    x = np.asarray(x, dtype=float)
    vals = term_values(x) * _DX
    if vals.size == 0:
        return 0.0
    maxv = float(np.max(vals))
    if _SCALE <= 0.0:
        out = maxv
    else:
        e = np.exp((vals - maxv) / float(_SCALE))
        z = float(e.sum())
        if z <= 0.0:
            out = maxv
        else:
            out = maxv + float(_SCALE) * math.log(z)
    if _PENALTY > 0.0:
        err = float(np.mean(x)) - 0.5
        out += float(_PENALTY) * err * err
    return float(out)


def objective_jac(x):
    x = np.asarray(x, dtype=float)
    n = _N
    g = np.zeros(n, dtype=float)
    valsA = term_values(x)
    Cvals = valsA * _DX
    if Cvals.size == 0:
        return g

    if _SCALE <= 0.0:
        maxc = float(np.max(Cvals))
        act = (maxc - Cvals) <= float(_ACTIVE_TOL)
        if not np.any(act):
            act = (Cvals == maxc)
        count = float(np.count_nonzero(act))
        weights = np.zeros(_NTERM, dtype=float)
        weights[act] = 1.0 / count
    else:
        maxc = float(np.max(Cvals))
        e = np.exp((Cvals - maxc) / float(_SCALE))
        z = float(e.sum())
        if z <= 0.0:
            weights = np.full(_NTERM, 1.0 / float(_NTERM), dtype=float)
        else:
            weights = e / z

    thr = 1.0e-13
    m_arr = _TERM_M
    s_arr = _TERM_SIGN
    for l in range(_NTERM):
        wl = float(weights[l])
        if wl <= thr:
            continue
        mm = int(m_arr[l])
        sgn = int(s_arr[l])
        if sgn == 0:
            g += wl * (1.0 - 2.0 * x)
        elif sgn > 0:
            if mm < n:
                g[:n - mm] += wl * (1.0 - x[mm:])
                g[mm:] -= wl * x[:n - mm]
        else:
            if mm < n:
                g[mm:] += wl * (1.0 - x[:n - mm])
                g[:n - mm] -= wl * x[mm:]

    g *= _DX
    if _PENALTY > 0.0:
        err = float(np.mean(x)) - 0.5
        g += (2.0 * float(_PENALTY) / float(n)) * err
    return g


def constraint_fun(x):
    x = np.asarray(x, dtype=float)
    return float(np.mean(x) - 0.5)

def make_sinusoid(n, freq, phase, amp):
    n = int(n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + float(amp) * np.sin(2.0 * math.pi * float(freq) * x + float(phase))

def make_perturb(h, n, rng, sigma):
    n = int(n)
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size != n:
        if h.size == 0:

            h = upsample(h, n)

    noise = rng.normal(0.0, float(sigma), size=n)

def thue_morse_start(n):
    n = int(n)
    seq = [0.0]
    while len(seq) < n:
        seq.extend([1.0 - v for v in seq])
    vals = np.asarray(seq[:n], dtype=float)
    target = 0.5 * float(n)
    if abs(target - round(target)) < 1.0e-12:
        target_int = int(round(target))
        ones = np.flatnonzero(vals > 0.5)
        zeros = np.flatnonzero(vals < 0.5)
        if len(ones) > target_int:
            for i in ones[len(ones) - target_int:]:
                vals[i] = 0.0
        elif len(ones) < target_int:
            need = target_int - len(ones)
            for i in zeros[:need]:
                vals[i] = 1.0
        return np.clip(vals, 0.0, 1.0)
    return project_sum(vals, n)


def make_balanced_pattern(n, pattern):
    n = int(n)
    pat = np.asarray(pattern, dtype=float).reshape(-1)
    if pat.size == 0:
        return project_sum(np.full(n, 0.5, dtype=float), n)
    m = int(pat.size)
    reps = int(math.ceil(float(n) / float(m)))
    vals = np.tile(pat, reps)[:n]
    target = 0.5 * float(n)
    if abs(target - round(target)) < 1.0e-12:
        target_int = int(round(target))
        ones = np.flatnonzero(vals > 0.5)
        zeros = np.flatnonzero(vals < 0.5)
        if len(ones) > target_int:
            for i in ones[len(ones) - target_int:]:
                vals[i] = 0.0
        elif len(ones) < target_int:
            need = target_int - len(ones)
            for i in zeros[:need]:
                vals[i] = 1.0
        return np.clip(vals, 0.0, 1.0)

def stage_iters(requested, n, rem):
    n = int(n)
    cap = 120
    if n > 220:
        cap = 35
    elif n > 110:
        cap = 60
    elif n > 55:
        cap = 90
    if rem < 90:
        cap = min(cap, 12)
    elif rem < 180:
        cap = min(cap, 30)
    elif rem < 300:
        cap = min(cap, 55)
    return max(1, int(min(int(requested), cap)))


def run_slq_stage(x0, n, scale, penalty, maxiter, ftol=1.0e-7, deadline=None):
    global _SCALE, _PENALTY
    n = int(n)
    build_terms(n)
    _SCALE = float(scale)
    _PENALTY = float(penalty)

    x0 = project_sum(x0, n)
    cur = exact_c5(x0, n)
    if maxiter <= 0:
        return x0, cur, x0, cur

    rem = 1.0e12
    if deadline is not None:
        rem = float(deadline) - time.time()
        if rem < 35.0:
            return x0, cur, x0, cur
    iters = stage_iters(maxiter, n, rem)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": constraint_fun, "jac": constraint_jac}]
    opts = {"maxiter": iters, "ftol": float(ftol), "maxls": 60, "disp": False}

    try:
        res = scipy.optimize.minimize(
            soft_objective,
            x0,
            jac=objective_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options=opts,
        )
    except Exception:
        return x0, cur, x0, cur

    if res is not None and hasattr(res, "x") and res.x is not None:
        vn = np.asarray(res.x, dtype=float).reshape(-1)
        if vn.size == n and np.all(np.isfinite(vn)):
            xn = project_sum(vn, n)
            cn = exact_c5(xn, n)
        else:
            xn = x0
            cn = cur
    else:
        xn = x0
        cn = cur

    if cn < cur:
        return xn, cn, xn, cn
    return xn, cn, x0, cur

        nv = int(round(float(base_n) * float(mult)))
        if nv < 2:
            nv = 2
        if nv <= max_opt_n and nv not in cands:
            cands.append(nv)
    cands.sort()

    current_h_opt = best_start
    current_n_opt = base_n

    for target_n in cands:
        rem = float(deadline) - time.time()
        if rem < 320:
            break
        if target_n < 2:
            continue

        start_options = []
        s1 = transformed_start(h_best, n_best, target_n)
        if s1 is not None:
            start_options.append((exact_c5(s1, target_n), s1))
        s2 = transformed_start(current_h_opt, current_n_opt, target_n)
        if s2 is not None:
            start_options.append((exact_c5(s2, target_n), s2))

        if not start_options:
            h_start = project_sum(h_base, target_n)
        else:
            best_cand = None
            best_cand_c = 1.0e100
            for cc, ss in start_options:
                if cc < best_cand_c:
                    best_cand_c = cc
                    best_cand = ss
            h_start = best_cand

        h_opt, c_opt = optimize_local(h_start, target_n, deadline, rng)
        if c_opt < c_best - 1.0e-14:
            c_best = c_opt
            n_best = target_n
            h_best = h_opt.copy()
        current_h_opt = h_opt
        current_n_opt = target_n

    if int(n_best) <= 220:
        restarts = []
        restarts.append(make_perturb(h_best, n_best, rng, 0.14))
        restarts.append(thue_morse_start(n_best))
        restarts.append(make_balanced_pattern(n_best, [0.0, 0.0, 1.0, 1.0]))
        restarts.append(make_random(n_best, rng))
        restarts.append(make_sinusoid(n_best, 1.5, float(rng.uniform(0.0, 6.28318)), 0.5))
        restarts.append(make_sinusoid(n_best, 2.5, float(rng.uniform(0.0, 6.28318)), 0.5))
        for rs in restarts:
            if float(deadline) - time.time() < 120:
                break
            rq_x, rq_c = quick_optimize(rs, n_best, deadline - 15.0)
            if rq_c < c_best - 1.0e-14:
                c_best = rq_c
                n_best = int(n_best)
                h_best = rq_x.copy()

    h_out = finalize_h(h_best, int(n_best))
    c_out = exact_c5(h_out, int(n_best))

    return h_out, float(c_out), int(n_best)
