def interpolate_h(h_old, n_new):

    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)

            if n_ones > target_ones:
                idx = np.where(h == 1)[0]
                to_remove = idx[-(n_ones - target_ones):]
                h[to_remove] = 0
            elif n_ones < target_ones:
                idx = np.where(h == 0)[0]
                to_add = idx[:target_ones - n_ones]
                h[to_add] = 1
        else:
            # Scale
            h = h / max(np.sum(h), 1e-10) * target
            h = np.clip(h, 0, 1)
            # Fix sum
            diff = target - np.sum(h)
            if diff > 0:
                mask = h < 1.0
                h[mask] += diff / max(mask.sum(), 1)
            elif diff < 0:
                mask = h > 0.0
                h[mask] += diff / max(mask.sum(), 1)
            h = np.clip(h, 0, 1)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def fine_tune(h_init, budget=120):
    n = len(h_init)
    t_start = time.time()
    t_end = t_start + budget
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    deltas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.001, -0.1, -0.05, -0.02, -0.01, -0.005, -0.001]
    
    converged = False
    while not converged and (time.time() < t_end - 2):
        converged = True
        for i in range(n):
            if time.time() >= t_end - 2:
                break
            for delta in deltas:
                h_try = best_h.copy()
                h_try[i] += delta
                if h_try[i] < 0 or h_try[i] > 1:
                    continue
                # Fix sum by adjusting neighbor
                adjustments = []
                for j in [(i+1) % n, (i-1) % n, (i+5) % n, (i+n//2) % n]:
                    if j == i:
                        continue
                    h_try[j] = best_h[j] - delta
                    if 0 <= h_try[j] <= 1:
                        adjustments.append(h_try.copy())
                        break
                if not adjustments:
                    continue
                h_try = adjustments[0]
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-12:
                    best_h = h_try
                    best_c5 = c5_try
                    converged = False
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    
    global initial_h_values
    best_h = None
    best_c5 = np.inf
    
    # Get initial values from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        # Fix constraint
        target = n0 / 2.0
        diff = target - np.sum(h0)
        if abs(diff) > 1e-10:
            if diff > 0:
                mask = h0 < 1.0
            else:
                mask = h0 > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h0[mask] += diff / cnt
            h0 = np.clip(h0, 0, 1)
        best_h = h0
        best_c5 = compute_c5(h0)
    else:
        # Create starting point
        n0 = 95
        h0 = np.zeros(n0)
        h0[:n0//2] = 1.0
        best_h = h0
        best_c5 = compute_c5(h0)
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    
    # Stage 1: Local optimization on parent grid
    if remaining > 60:
        stage_budget = min(remaining * 0.35, 300)
        h_opt, c5_opt = local_optimize_nm(best_h.copy(), budget=stage_budget)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    
    # Stage 2: Try higher resolution (interpolation + optimize)
    if remaining > 120:
        n_current = len(best_h)
        for n_new in [2 * n_current, 3 * n_current]:
            if time.time() - t_start > budget_s - 90:
                break
            h_interp = interpolate_h(best_h, n_new)
            c5_interp = compute_c5(h_interp)
            if c5_interp < best_c5:
                best_h = h_interp
                best_c5 = c5_interp
            # Optimize at higher resolution (shorter budget)
            opt_budget = min(remaining * 0.25, 120)
            if opt_budget > 20:
                h_opt, c5_opt = local_optimize_nm(h_interp, budget=opt_budget)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
            elapsed = time.time() - t_start
            remaining = budget_s - elapsed
    
    # Stage 3: Fine-tune with coordinate descent
    if time.time() - t_start < budget_s - 30:
        ft_budget = min(budget_s - (time.time() - t_start) - 20, 120)
        if ft_budget > 10:
            best_h, best_c5 = fine_tune(best_h, budget=ft_budget)
    
    # Stage 4: Try structured initializations
    if time.time() - t_start < budget_s - 40:
        n_current = len(best_h)
        for n_try in [95, 190, n_current]:
            if time.time() - t_start > budget_s - 40:
                break
            h_struct, c5_struct = structured_init(n_try, seed=seed + 1)
            if c5_struct < best_c5:
                best_h = h_struct
                best_c5 = c5_struct
                # Quick optimize
                if time.time() - t_start < budget_s - 30:
                    h_opt, c5_opt = local_optimize_nm(h_struct, budget=40)
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
    
    # Stage 5: Final fine-tuning
    if time.time() - t_start < budget_s - 10:
        ft_budget = budget_s - (time.time() - t_start) - 10
        if ft_budget > 5:
            best_h, best_c5 = fine_tune(best_h, budget=ft_budget)
    
    # Final: Nelder-Mead polish
    if time.time() - t_start < budget_s - 15:
        polish_budget = min(budget_s - (time.time() - t_start) - 10, 60)
        if polish_budget > 10:
            h_opt, c5_opt = local_optimize_nm(best_h.copy(), budget=polish_budget)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    # Ensure final constraint satisfaction
    n_final = len(best_h)
    target = n_final / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = best_h < 1.0
        else:
            mask = best_h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            best_h[mask] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
