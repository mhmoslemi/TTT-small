```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def make_binary(h, n):
    """Ensure h is binary with exactly n//2 ones."""
    h = np.where(h >= 0.5, 1, 0).astype(float)
    ones_count = int(np.sum(h))
    target = n // 2
    if ones_count > target:
        # Flip (ones_count - target) ones to zeros
        one_idx = np.where(h == 1)[0]
        to_flip = np.random.choice(one_idx, ones_count - target, replace=False)
        h[to_flip] = 0
    elif ones_count < target:
        zero_idx = np.where(h == 0)[0]
        to_flip = np.random.choice(zero_idx, target - ones_count, replace=False)
        h[to_flip] = 1
    return h

def local_search(h, n, budget_s, start_time, rng):
    """Local search with pair swaps."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    
    while time.time() - start_time < budget_s:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap_c5 = best_c5
        best_swap = None
        found_improvement = False
        
        # Try random swaps
        n_trials = min(500, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c5 = compute_c5(h)
            h[i], h[j] = h[j], h[i]
            if c5 < best_swap_c5 - 1e-10:
                best_swap_c5 = c5
                best_swap = (i, j)
                found_improvement = True
        
        if found_improvement:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            best_c5 = best_swap_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Random perturbation every few iterations
            if no_improve > 10:
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h[i], h[j] = h[j], h[i]
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def ensure_binary(h, n, rng):
    """Ensure h is binary with exactly n//2 ones."""
    h = np.array(h, dtype=float)
    # Round to binary
    h = np.where(h >= 0.5, 1.0, 0.0)
    ones_count = int(np.sum(h))
    target = n // 2
    if ones_count > target:
        one_idx = np.where(h == 1)[0]
        to_flip = rng.choice(one_idx, ones_count - target, replace=False)
        h[to_flip] = 0.0
    elif ones_count < target:
        zero_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zero_idx, target - ones_count, replace=False)
        h[to_flip] = 1.0
    return h

def local_search(h, n, start_time, time_limit, rng):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnation = 0
    
    while time.time() - start_time < start_time + time_limit:
        if time.time() - start_time > start_time + time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        improved = False
        # Try up to 200 random swaps
        n_trials = min(300, len(ones) * len(zeros))
        best_trial_c5 = best_c5
        best_trial = None
        
        for t in range(n_trials):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c5 = compute_c5(h)
            if c5 < best_trial_c5:
                best_trial_c5 = c5
                best_trial = (i, j)
            else:
                h[i], h[j] = h[j], h[i]
        
        if best_trial is not None:
            i, j = best_trial
            h[i], h[j] = h[j], h[i]
            best_c5 = best_trial_c5
            best_h = h.copy()
            improved = True
            stagnation = 0
        else:
            stagnation += 1
            # Escape: random double swap
            if stagnation >= 5:
                for _ in range(min(3, len(ones))):
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                    h[i], h[j] = h[j], h[i]
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                stagnation = 0
        
        # Check budget
        if time.time() - start_time > start_time + time_limit:
            break
    
    return best_h, best_c5
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary_feasible(h, n, rng):
    """Make h binary with exactly n//2 ones."""
    h = np.array(h, dtype=float)
    h = np.where(h >= 0.5, 1.0, 0.0)
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        flip = rng.choice(idx, ones - target, replace=False)
        h[flip] = 0.0
    elif ones < target:
        idx = np.where(h == 0)[0]
        flip = rng.choice(idx, target - ones, replace=False)
        h[flip] = 1.0
    return h

def full_neighborhood_search(h, n, rng, time_limit, start_time):
    """Local search trying all pair swaps."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0
    
    while time.time() - start_time < start_time + time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap_c5 = best_c5
        best_swap = None
        
        # Try all pairs (for n <= ~150)
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if c5 < best_swap_c5 - 1e-12:
                    best_swap_c5 = c5
                    best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_swap_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 3:
                # Random perturbation
                for _ in range(2):
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                    h[i], h[j] = h[j], h[i]
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                patience = 0
    
    return best_h, best_c5

def random_sampling_search(h, n, rng, time_limit, start_time):
    """Local search with random pair swaps (for larger n)."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnation = 0
    
    while time.time() - start_time < start_time + time_limit:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap_c5 = best_c5
        best_swap = None
        
        n_trials = 500
        for _ in range(n_trials):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c5 = compute_c5(h)
            if c5 < best_swap_c5 - 1e-12:
                best_swap_c5 = c5
                best_swap = (i, j)
            else:
                h[i], h[j] = h[j], h[i]
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = h[j], h[i]
            best_c5 = best_swap_c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation >= 5:
                # Double swap to escape
                for _ in range(min(3, len(ones), len(zeros))):
                    i = ones[rng.integers(len(ones))]
                    j = zeros[rng.integers(len(zeros))]
                    h[i], h[j] = h[j], h[i]
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                stagnation = 0
    
    return best_h, best_c5
```
