def project_feasible(h, target):
    n = len(h)

    for _ in range(80):

        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on lambda."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_corrs(h):
    """Compute correlation values times dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def compute_gradient(h, weights, n):
    """Compute gradient of smooth-max objective w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    wc = weights[n - 1]
    if wc > 1e-15:
        grad += wc * dx * (1.0 - 2.0 * h)
    for s in range(1, n):
        w = weights[n - 1 + s]
        if w < 1e-14:
            continue
        wd = w * dx
        grad[:n - s] += wd * (1.0 - h[s:])
        grad[s:] -= wd * h[:n - s]
    for s in range(1, n):
        w = weights[n - 1 - s]
        if w < 1e-14:
            continue
        wd = w * dx
        grad[s:] += wd * (1.0 - h[:n - s])
        grad[:n - s] -= wd * h[s:]
    return grad
