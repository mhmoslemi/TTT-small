            current_val = cand_val
        else:
            candidate2 = _project_sum(np.clip(current_h + 0.02 * rng.randn(n), 0.0, 1.0), n)
            res_x2 = _optimize_stage(candidate2, n, beta, lam, max(30, n_iter // 2), 1e-12, 1e-10)
            candidate2 = _project_sum(np.clip(res_x2, 0.0, 1.0), n)
            cand_val2 = _objective_no_penalty(candidate2, n, 100.0)
            if cand_val2 < current_val:
                current_h = candidate2
                current_val = cand_val2

    # Final polish with very high beta
    elapsed = time.time() - start_time
    if elapsed < budget_s - 15:
        res_x = _optimize_stage(current_h, n, 500.0, 50.0, 100, 1e-14, 1e-12)
        candidate = _project_sum(np.clip(res_x, 0.0, 1.0), n)
        if _objective_no_penalty(candidate, n, 500.0) < _objective_no_penalty(current_h, n, 500.0):
            current_h = candidate

    return _project_sum(current_h, n)


def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values, evaluate_erdos_solution
    start_time = time.time()
    np.random.seed(seed)

    parent_h = np.array(initial_h_values, dtype=float)
    n_parent = len(parent_h)
    parent_h = np.clip(parent_h, 0.0, 1.0)
    parent_h = _project_sum(parent_h, n_parent)

    try:
        best_c5 = float(evaluate_erdos_solution(parent_h))
    except Exception:
        best_c5 = _compute_c5(parent_h)
    best_h = parent_h.copy()

    # Phase 1: Optimize at parent resolution
    elapsed = time.time() - start_time
    if elapsed < budget_s - 60:
        h_opt = _run_optimization(best_h, n_parent, budget_s, start_time, seed)
        c5 = _compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

    # Phase 2: Try higher resolutions
    elapsed = time.time() - start_time
    for n_try in [2 * n_parent, 4 * n_parent]:
        if time.time() - start_time > budget_s - 60:
            break
        h_init = _upsample_to(best_h, n_try)
        h_opt = _run_optimization(h_init, n_try, budget_s, start_time, seed + 7)
        c5 = _compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

    # Phase 3: Random restarts
    elapsed = time.time() - start_time
    attempt = 0
    while time.time() - start_time < budget_s - 20 and attempt < 30:
        attempt += 1
        n_try = n_parent
        rng = np.random.RandomState(seed + 1000 * attempt)
        noise_scale = 0.08 + 0.04 * (attempt % 3)
        h_rand = best_h + rng.normal(0, noise_scale, n_try)
        h_rand = np.clip(h_rand, 0.0, 1.0)
        h_rand = _project_sum(h_rand, n_try)

        h_opt = _run_optimization(h_rand, n_try, budget_s, start_time, seed + 100 * attempt)
        c5 = _compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

    # Phase 4: If at higher resolution, also try optimizing downsampled
    elapsed = time.time() - start_time
    if len(best_h) > n_parent and elapsed < budget_s - 30:
        h_dn = _project_sum(_upsample_to(best_h, n_parent), n_parent)
        # But actually downsample by averaging
        ratio = len(best_h) // n_parent
        if ratio >= 1 and len(best_h) % ratio == 0:
            h_dn = best_h.reshape(n_parent, ratio).mean(axis=1)
            h_dn = _project_sum(h_dn, n_parent)
            h_opt = _run_optimization(h_dn, n_parent, budget_s, start_time, seed + 500)
            c5 = _compute_c5(h_opt)
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5

    # Final: ensure constraints and evaluate
    best_h = _project_sum(best_h, len(best_h))
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5 = _compute_c5(best_h)

    # Verify sum constraint
    n_final = len(best_h)
    target_sum = 0.5 * n_final
    current_sum = np.sum(best_h)
    if abs(current_sum - target_sum) > 1e-10:
        best_h = _project_sum(best_h, n_final)

    return best_h, final_c5, n_final
