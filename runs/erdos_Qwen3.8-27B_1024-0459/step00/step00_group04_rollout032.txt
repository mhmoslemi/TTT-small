import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    one_minus = 1.0 - h
    corr = np.correlate(h, one_minus, mode="full")
    return float(np.max(corr * dx))

def fix_constraint(h, n):
    """Fix the sum constraint and box constraints."""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            if diff > 0:
                mask = h < 1
            else:
                mask = h > 0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
    h = np.clip(h, 0, 1)
    # Final tiny correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def upsample(h, factor=2):
    """Upsample h by factor using linear interpolation."""
    n = len(h)
    n_new = n * factor
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]))
    return fix_constraint(h_new, n_new)

def slsqp_optimize(h0, n, maxiter=500, ftol=1e-12, timeout=60.0):
    """Run SLSQP optimization on h0 of length n."""
    target_sum = n / 2.0
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    start_t = time.time()
    def obj(x):
        return compute_c5(x, n)
    
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < compute_c5(h0, n):
            h_opt = fix_constraint(res.x, n)
            return h_opt, compute_c5(h_opt, n)
    except Exception:
        pass
    return h0.copy(), compute_c5(h0, n)

def slsqp_optimize_v2(h0, n, maxiter=200, ftol=1e-10):
    """SLSQP with explicit gradient using finite differences on the active term."""
    target_sum = n / 2.0
    h = h0.copy()
    
    for iteration in range(maxiter):
        c5_val = compute_c5(h, n)
        dx = 2.0 / n
        one_minus = 1.0 - h
        corr = np.correlate(h, one_minus, mode="full") * dx
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-9)[0]
        
        # Build gradient of the active term(s)
        grad = np.zeros(n)
        for idx in active:
            # idx in [0, 2n-2], lag k = idx - (n-1)
            k = idx - (n - 1)
            if k >= 0:
                # corr = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
                # d/dh[j]: (1-h[j+k]) if j <= n-1-k; -h[j-k] if j >= k
                for j in range(n - k):
                    grad[j] += (1.0 - h[j + k]) / len(active)
                for j in range(k, n):
                    grad[j] += (-h[j - k]) / len(active)
            else:
                k = -k
                # corr = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
                # d/dh[j]: (1-h[j-k]) if j >= k; -h[j+k] if j <= n-1-k
                for j in range(k, n):
                    grad[j] += (1.0 - h[j - k]) / len(active)
                for j in range(n - k):
                    grad[j] += (-h[j + k]) / len(active)
        
        grad *= dx
        
        # Project gradient onto the constraint manifold (sum h = n/2)
        grad_proj = grad - np.mean(grad)
        
        # Line search
        alpha = 0.1
        for _ in range(30):
            h_trial = h - alpha * grad_proj
            h_trial = fix_constraint(h_trial, n)
            c5_trial = compute_c5(h_trial, n)
            if c5_trial < c5_val:
                h = h_trial
                c5_val = c5_trial
                break
            alpha *= 0.5
        
        if alpha < 1e-15:
            break
    
    return h, compute_c5(h, n)

def simulated_annealing(h0, n, n_steps, T_init, T_min, timeout):
    """Run simulated annealing."""
    rng = np.random.default_rng(42)
    h = h0.copy()
    c5_cur = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5_cur
    T = T_init
    start_t = time.time()
    
    for step in range(n_steps):
        if time.time() - start_t > timeout:
            break
        n_pert = rng.integers(1, min(5, n // 3) + 1)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_trial = h.copy()
        h_trial[idx] += rng.uniform(-0.15, 0.15, size=n_pert)
        h_trial = fix_constraint(h_trial, n)
        c5_trial = compute_c5(h_trial, n)
        
        delta = c5_trial - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            h = h_trial
            c5_cur = c5_trial
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        
        T = max(T_min, T * (T_min / T_init) ** (1.0 / max(n_steps, 1)))
    
    return best_h, best_c5

def structured_start(n, rng, seed):

        step = n // (n // 4 + 1)

            h[i:i + step] = 1.0
        h = fix_constraint(h, n)
    else:
        h = np.ones(n) * 0.5
        h = fix_constraint(h, n)
    return fix_constraint(h, n)
