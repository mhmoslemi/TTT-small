import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_all(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    """Project h onto {0 <= h_i <= 1, sum(h_i) = target_sum} using bisection on shift."""
    h = np.asarray(h, dtype=float)
    if target_sum < 0:
        target_sum = 0
    if target_sum > len(h):
        target_sum = len(h)
    
    def sum_clip(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0))
    
    # Find tau such that sum_clip(tau) = target_sum
    # sum_clip is decreasing in tau
    # At tau -> -inf, sum_clip -> n
    # At tau -> +inf, sum_clip -> 0
    
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = sum_clip(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def smooth_max_gradient(h, n, alpha):
    """Compute gradient of dx * (1/alpha) * log(sum exp(alpha * C_k))"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights (avoid overflow)
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        ell = idx - (n - 1)
        if ell >= 0:
            # First term: (1-h[m-ell]) for m in [ell, n-1]
            m_range = np.arange(ell, n)
            grad[m_range] += w * (1.0 - h[m_range - ell])
            # Second term: -h[m+ell] for m in [0, n-1-ell]
            m_range2 = np.arange(0, n - ell)
            grad[m_range2] += w * (-h[m_range2 + ell])
        else:
            ell_neg = -ell
            # First term: (1-h[m+ell_neg]) for m in [0, n-1-ell_neg]
            m_range = np.arange(0, n - ell_neg)
            grad[m_range] += w * (1.0 - h[m_range + ell_neg])
            # Second term: -h[m-ell_neg] for m in [ell_neg, n-1]
            m_range2 = np.arange(ell_neg, n)
            grad[m_range2] += w * (-h[m_range2 - ell_neg])
    
    return dx * grad

def optimize_smooth(h_init, n, max_iter, alpha_start, alpha_end, lr_start, lr_end, time_limit):
    """Optimize h using smooth max gradient descent with projection."""
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    start_time = time.time()
    lr = lr_start
    alpha = alpha_start
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        # Adaptive alpha (increase over iterations for sharper max)
        progress = it / max(max_iter - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute gradient
        grad = smooth_max_gradient(h, n, alpha)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project onto feasible set
        h_new = project_feasible(h_new, target_sum)
        
        # Evaluate
        c5_new = compute_c5(h_new, n)
        
        # Simple line search if objective worsened
        if c5_new > best_c5 * 1.01:
            # Try smaller step
            for shrink in [0.5, 0.25, 0.1]:
                h_try = h - lr * shrink * grad
                h_try = project_feasible(h_try, target_sum)
                c5_try = compute_c5(h_try, n)
                if c5_try < c5_new:
                    h_new = h_try
                    c5_new = c5_try
                    break
            else:
                # Keep old value
                h_new = h
                c5_new = best_c5
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def optimize_subgradient(h_init, n, max_iter, lr_start, lr_end, time_limit):

    target_sum = n / 2.0

    best_h = h.copy()

            best_c5 = best_c5_curr
            best_h = h.copy()

            else:

                m_range = np.arange(0, n - ell_neg)
                grad[m_range] += (1.0 - h[m_range + ell_neg])
                m_range2 = np.arange(ell_neg, n)
                grad[m_range2] += (-h[m_range2 - ell_neg])
        
        grad /= top_k
        grad *= dx
        
        # Adaptive learning rate
        progress = it / max(max_iter - 1, 1)
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        c5_new = compute_c5(h_new, n)

        if c5_new > best_c5_curr * 1.01:

                h_try = project_feasible(h_try, target_sum)

                    c5_new = c5_try
                    break
            else:
                h_new = h.copy()
                c5_new = best_c5_curr
        
        h = h_new

def try_n_points(h_best, n_best, n_target, n_best_points):
    """Resample from n_best_points to n_target."""
    n_old = n_best_points
    n_new = n_target
    
    # Linear interpolation
    x_old = np.linspace(0, 2, n_old, endpoint=False)

    # Enforce feasibility
    h_new = np.clip(h_new, 0, 1)
    h_new = project_feasible(h_new, n_new / 2.0)
    
    return h_new

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target_sum = np.clip(target_sum, 0.0, float(n))
    def sum_clip(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0))
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = sum_clip(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_c))
    w_sum = np.sum(exp_vals)
    if w_sum < 1e-300:
        return np.zeros(n), corr * dx
    weights = exp_vals / w_sum
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-10:
            continue
        ell = idx - (n - 1)
        if ell >= 0:
            if ell < n:
                m1 = np.arange(ell, n)
                grad[m1] += w * (1.0 - h[m1 - ell])
                m2 = np.arange(0, n - ell)
                grad[m2] += w * (-h[m2 + ell])
        else:
            en = -ell
            if en < n:
                m1 = np.arange(0, n - en)
                grad[m1] += w * (1.0 - h[m1 + en])
                m2 = np.arange(en, n)
                grad[m2] += w * (-h[m2 - en])
    return dx * grad, corr * dx

def topk_grad(h, n, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx
    top_idx = np.argsort(c_vals)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        ell = idx - (n - 1)
        if ell >= 0:
            if ell < n:
                m1 = np.arange(ell, n)
                grad[m1] += (1.0 - h[m1 - ell])
                m2 = np.arange(0, n - ell)
                grad[m2] += (-h[m2 + ell])
        else:
            en = -ell
            if en < n:
                m1 = np.arange(0, n - en)
                grad[m1] += (1.0 - h[m1 + en])
                m2 = np.arange(en, n)
                grad[m2] += (-h[m2 - en])
    grad /= top_k
    return dx * grad, c_vals

    target_sum = n / 2.0
    h = project_feasible(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    start_t = time.time()
    no_improve = 0
    patience = 300
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        progress = it / max(max_iter - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        if use_topk:

def resample_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n_new / 2.0)
    return h_new

def random_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = rng.uniform(0.2, 0.8, n)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = np.inf
    best_n = 0
    
    h_parent = None
    n_parent = 0
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    except Exception:
        pass
    
    if h_parent is None or n_parent == 0:
        n_parent = 90
        h_parent = random_init(n_parent, seed)
    
    h_parent = np.clip(h_parent, 0.0, 1.0)
    h_parent = project_feasible(h_parent, n_parent / 2.0)
    c5_p = compute_c5(h_parent, n_parent)
    best_h = h_parent.copy()
    best_c5 = c5_p
    best_n = n_parent
    
    # Phase 1: Refine at parent resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        t_lim = min(remaining * 0.25, 300)
        h_opt, c5_opt = optimize_h(h_parent, n_parent, 8000, 20.0, 300.0, 0.04, 0.0005, t_lim, False, 0)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_parent
    
    remaining = budget_s - (time.time() - t0)
    if remaining > 30 and best_n == n_parent:
        t_lim = min(remaining * 0.2, 200)
        h_opt, c5_opt = optimize_h(best_h, best_n, 5000, 10.0, 100.0, 0.03, 0.0003, t_lim, True, 5)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Phase 2: Multi-resolution optimization
    n_candidates = sorted(set([n_parent, n_parent*2, n_parent*3, 60, 90, 120, 180, 240, 360]))
    n_candidates = [x for x in n_candidates if 30 <= x <= 500]
    
    for n_try in n_candidates:
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        if n_try == best_n:
            h_init = best_h.copy()
        else:
            h_init = resample_h(best_h, best_n, n_try)
        
        n_attempts = 2 if remaining > 120 else 1
        for att in range(n_attempts):
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            if att == 0:
                h_try = h_init.copy()
            else:
                h_try = h_init + rng.normal(0, 0.04, n_try)
                h_try = project_feasible(h_try, n_try / 2.0)
            
            t_lim = min(remaining * 0.3, 180)
            h_opt, c5_opt = optimize_h(h_try, n_try, 6000, 15.0, 200.0, 0.03, 0.0003, t_lim, False, 0)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_try
        
        remaining = budget_s - (time.time() - t0)
        if remaining > 20 and best_n == n_try:
            t_lim = min(remaining * 0.3, 120)
            h_opt, c5_opt = optimize_h(best_h, best_n, 4000, 50.0, 500.0, 0.02, 0.0002, t_lim, True, 3)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
    
    # Phase 3: Random restarts
    remaining = budget_s - (time.time() - t0)
    n_restarts = min(6, int(remaining / 40))
    for r in range(n_restarts):
        remaining = budget_s - (time.time() - t0)
        if remaining < 10:
            break
        n_r = best_n
        h_rand = random_init(n_r, seed + 1000 + r)
        t_lim = min(remaining * 0.25, 80)
        h_opt, c5_opt = optimize_h(h_rand, n_r, 3000, 10.0, 100.0, 0.02, 0.0005, t_lim, False, 0)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_r
    
    # Phase 4: Fine-tune best solution
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        t_lim = min(remaining * 0.8, 300)
        h_opt, c5_opt = optimize_h(best_h, best_n, 10000, 200.0, 1000.0, 0.01, 0.00005, t_lim, True, 3)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Final: try a few more topk refinements
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        t_lim = min(remaining * 0.8, 200)
        h_opt, c5_opt = optimize_h(best_h, best_n, 8000, 100.0, 500.0, 0.005, 0.00003, t_lim, True, 5)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    
    # Final verification
    dx = 2.0 / best_n
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, best_n)
    best_c5 = final_c5
    
    return best_h, best_c5, best_n
