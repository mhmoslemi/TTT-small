def logsumexp_value_weights(c, beta):
    c = np.asarray(c, dtype=float)
    if c.size == 0: return 0.0, np.array([])
    m = float(np.max(c))
    if beta <= 0.0:
       p=np.full(c.size, 1.0/c.size)
       return m,p
    z = beta* (c - m)
    zmax = float(np.max(z))
    z = z - zmax
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300:
       p = np.zeros(c.size); p[int(np.argmax(c))] = 1.0
    else:
       p = e / s
    f = (m + float(np.log(s))/beta)
    return f,p

def _fix_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n == 0: return h
    target = float(target)
    s = float(h.sum())
    diff = target - s
    if abs(diff) <= 1e-12:
        return np.clip(h,0.0,1.0)
    for _ in range(10):
       s=float(h.sum()); diff=target-s
       if abs(diff) <= 1e-12: break
       if diff > 0:
          mask = h < 1.0 - 1e-14
          if not np.any(mask): break
          slack = 1.0 - h[mask]
          cap = float(slack.sum())
          if cap <= 1e-15: break
          add = min(diff, cap)
          h[mask] += add * (slack / cap)
       else:
          mask = h > 1e-14
          if not np.any(mask): break
          mass = h[mask]
          cap = float(mass.sum())
          if cap <= 1e-15: break
          sub = min(-diff, cap)
          h[mask] -= sub * (mass / cap)
       h = np.clip(h, 0.0, 1.0)
    # final
    s=float(h.sum()); diff=target-s
    if abs(diff) > 1e-13 and n>0:
       if diff > 0:
          idx = int(np.argmax(1.0 - h))
          add = min(1.0 - h[idx], diff)
          h[idx] += add
       else:
          idx = int(np.argmax(h))
          sub = min(h[idx], -diff)
          h[idx] -= sub
    h = np.clip(h,0.0,1.0)
    return h
```

def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n=v.size
    if n==0: return v
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=0.0)
    target=float(target)
    if target <= 0.0: return np.zeros(n)
    if target >= n: return np.ones(n)
    # quick clip sum
    h = np.clip(v, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) <= 1e-11:
       return _fix_sum(h, target)
    # scaling path
    if s > 1e-12:
       factor = target / s
       if factor >= 0.0 and factor <= 1.0 + 1e-12:
           h2 = h * factor
           h2 = _fix_sum(h2, target)
           if float(np.min(h2)) >= -1e-12 and float(np.max(h2)) <= 1.0 + 1e-12:
               return np.clip(h2,0.0,1.0)
       if factor > 1.0 and factor < 10.0: # maybe scale up possible
           if float(np.max(h)) * factor <= 1.0 + 1e-12:
               h2 = h * factor
               h2 = _fix_sum(h2, target)
               if float(np.min(h2)) >= -1e-12 and float(np.max(h2)) <= 1.0 + 1e-12:
                   return np.clip(h2,0.0,1.0)
    # bisection
    vmin = float(np.min(v)); vmax = float(np.max(v))
    lo = vmin - 1.0
    hi = vmax + 1.0
    if not (lo < hi):
       lo = -1.0; hi = 2.0
    for _ in range(45):
       lam = 0.5 * (lo + hi)
       s = float(np.sum(np.clip(v - lam, 0.0, 1.0)))
       if s > target:
          lo = lam
       else:
          hi = lam
    lam = 0.5*(lo+hi)
    h = np.clip(v - lam, 0.0, 1.0)
    h = _fix_sum(h, target)
    return h

def effective_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1.0
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def full_correlation(h):
    return np.correlate(h, 1.0 - h, mode="full")


def logsumexp_value_weights(c, beta):
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return 0.0, np.zeros(0, dtype=float)
    m = float(np.max(c))
    if beta <= 0.0:
        p = np.full(c.size, 1.0 / float(c.size), dtype=float)
        return m, p
    z = beta * (c - m)
    zmax = float(np.max(z))
    z = z - zmax
    e = np.exp(z)
    s = float(e.sum())
    if s <= 1e-300:
        p = np.zeros(c.size, dtype=float)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s
    f = (m + float(np.log(s)) / beta)
    return f, p


def weighted_gradient(h, p):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    if p is None or len(p) != 2 * n - 1:
        return g
    p = np.asarray(p, dtype=float)
    idx = n - 1
    p0 = float(p[idx])
    if p0 != 0.0:
        g += p0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        pp = float(p[idx + k])
        pn = float(p[idx - k])
        q = pp + pn
        if q <= 1e-300:
            continue
        left = n - k
        g[:left] += pn - q * h[k:]
        g[k:] += pp - q * h[:left]
    return g


def _fix_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    if n == 0:
        return h
    target = float(target)
    for _ in range(10):
        s = float(h.sum())
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            slack = 1.0 - h[mask]
            cap = float(slack.sum())
            if cap <= 1e-15:
                break
            add = min(diff, cap)
            h[mask] += add * (slack / cap)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            mass = h[mask]
            cap = float(mass.sum())
            if cap <= 1e-15:
                break
            sub = min(-diff, cap)
            h[mask] -= sub * (mass / cap)
        h = np.clip(h, 0.0, 1.0)
    s = float(h.sum())
    diff = target - s
    if abs(diff) > 1e-13 and n > 0:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(1.0 - h[idx], diff)
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(h[idx], -diff)
            h[idx] -= sub
    return np.clip(h, 0.0, 1.0)

    n = v.size
    if n == 0:
        return v

    target = float(target)

    if target >= n:

    h = np.clip(v, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) <= 1e-11:
        return _fix_sum(h, target)

        else:

def upsample(h, m):
    h = np.asarray(h, dtype=float)
    n = h.size
    m = int(m)
    if n == 0 or m <= 0:
        return np.zeros(0, dtype=float)

        return h.copy()

    new = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    new = project_box_sum(new, 0.5 * m)
    return new

def optimize_projected(h, limit_s, hard_deadline, rng=None,
                       beta_schedule=(20.0, 50.0, 100.0), base_step=0.05):

        return best_h, best_c

def smooth_obj(x, beta, target, deadline):
    if time.time() > deadline:
        raise OptTimeout()
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
        h = np.clip(h, 0.0, 1.0)
    minv = float(np.min(h))
    maxv = float(np.max(h))
    used = h
    if minv < -1e-7 or maxv > 1.0 + 1e-7:
        used = np.clip(h, 0.0, 1.0)
    c = np.correlate(used, 1.0 - used, mode="full")
    f, p = logsumexp_value_weights(c, beta)
    g = weighted_gradient(used, p)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(used)
    return f, g


def slsqp_polish(h, limit_s, beta, hard_deadline, maxiter=200):
    h = make_feasible(h).astype(float).copy()
    n = h.size
    best_h = h.copy()
    best_c = effective_c5(h)
    if minimize is None or limit_s <= 0.0 or n == 0:
        return best_h, best_c

    end = time.time() + max(0.0, float(limit_s))
    if hard_deadline is not None:
        end = min(end, float(hard_deadline) - 0.5)
    if end - time.time() < 5.0:
        return best_h, best_c

    if n > 250:
        maxiter = min(int(maxiter), 40)

    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    cons = [
        {
            "type": "eq",
            "fun": sum_eq_fun,
            "jac": sum_eq_jac,
            "args": (target,),
        }
    ]
    res = None
    try:
        res = minimize(
            smooth_obj,
            h,
            method="SLSQP",
            jac=True,
            args=(float(beta), target, end),
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-8, "disp": False},
        )
    except OptTimeout:
        pass
    except Exception:
        pass

    if res is not None:
        cand = np.asarray(res.x, dtype=float)
        for cand in (cand, make_feasible(cand)):
            if cand.size != n:
                continue
            if not np.all(np.isfinite(cand)):
                cand = np.nan_to_num(cand, nan=0.0, posinf=1.0, neginf=0.0)
            cand = np.clip(cand, 0.0, 1.0)
            c = effective_c5(cand)
            if c < best_c:
                best_c = c
                best_h = cand.copy()
    return best_h, best_c


def pair_refine(h, limit_s, rng, hard_deadline):
    h = make_feasible(h).astype(float).copy()
    n = h.size
    if n < 2 or limit_s <= 0.0:
        return h, effective_c5(h)
    dx = 2.0 / n
    end = time.time() + max(0.0, float(limit_s))
    if hard_deadline is not None:
        end = min(end, float(hard_deadline) - 0.5)
    best_h = h.copy()
    best_c = effective_c5(h)
    if rng is None:
        rng = np.random.default_rng(1234567)
    no_improve = 0

    while time.time() < end:
        c = full_correlation(h)
        C = float(np.max(c)) * dx
        f, p = logsumexp_value_weights(c, 50.0)
        g = weighted_gradient(h, p)
        d = -g
        pos = np.where(d > 1e-8)[0]
        neg = np.where(d < -1e-8)[0]
        improved = False

        if pos.size > 0 and neg.size > 0:
            max_top = 12
            pos_top = pos[np.argsort(d[pos])[-max_top:]] if pos.size > max_top else pos
            neg_top = neg[np.argsort(d[neg])[:max_top]] if neg.size > max_top else neg
            for _ in range(18):
                i = int(pos_top[rng.integers(pos_top.size)])
                j = int(neg_top[rng.integers(neg_top.size)])
                if i == j:
                    continue
                maxa = min(1.0 - float(h[i]), float(h[j]))
                if maxa <= 1e-8:
                    continue
                best_a = 0.0
                best_val = C
                for frac in (0.1, 0.25, 0.5, 0.75, 0.95):
                    a = frac * maxa
                    if a <= 0.0:
                        continue
                    h2 = h.copy()
                    h2[i] += a
                    h2[j] -= a
                    h2[i] = float(np.clip(h2[i], 0.0, 1.0))
                    h2[j] = float(np.clip(h2[j], 0.0, 1.0))
                    val = effective_c5(h2)
                    if val < best_val - 1e-12:
                        best_val = val
                        best_a = a
                if best_a > 0.0:
                    lo = max(0.0, best_a * 0.6)
                    hi = min(maxa, best_a * 1.4)
                    if hi > lo:
                        for aa in np.linspace(lo, hi, 9):
                            a = float(aa)
                            if a <= 1e-12 or a >= maxa:
                                continue
                            h2 = h.copy()
                            h2[i] += a
                            h2[j] -= a
                            h2[i] = float(np.clip(h2[i], 0.0, 1.0))
                            h2[j] = float(np.clip(h2[j], 0.0, 1.0))
                            val = effective_c5(h2)
                            if val < best_val - 1e-12:
                                best_val = val
                                best_a = a
                    h[i] = min(1.0, float(h[i] + best_a))
                    h[j] = max(0.0, float(h[j] - best_a))
                    C = best_val
                    improved = True

        if not improved:
            for _ in range(12):
                if improved:
                    break
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j:
                    continue
                maxa = min(float(h[i]), 1.0 - float(h[j]))
                if maxa <= 1e-8:
                    maxa = min(float(h[j]), 1.0 - float(h[i]))
                    i, j = j, i
                if maxa <= 1e-8:
                    continue
                for frac in (0.1, 0.3, 0.7):
                    a = frac * maxa
                    if a <= 0.0:
                        continue
                    h2 = h.copy()
                    h2[i] -= a
                    h2[j] += a
                    h2[i] = float(np.clip(h2[i], 0.0, 1.0))
                    h2[j] = float(np.clip(h2[j], 0.0, 1.0))
                    val = effective_c5(h2)
                    if val < C - 1e-11:
                        h = h2
                        C = val
                        improved = True
                        break

        if improved:
            if C < best_c - 1e-12:
                best_c = C
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_h, effective_c5(best_h)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    budget = max(1.0, float(budget_s))
    hard_deadline = start + max(1.0, budget - 3.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")

    parent = globals().get("initial_h_values")
    if parent is not None:
        p = np.asarray(parent, dtype=float)
        if p.size > 0:
            p = make_feasible(p)
            c = effective_c5(p)
            best_h = p
            best_c = c

    if best_h is None:
        n_start = 70
        cand_list = []
        for kind in range(6):
            cand_list.append(trig_h(n_start, kind))
        cand_list.append(smoothed_random(n_start, seed + 1, 3))
        cand_list.append(smoothed_random(n_start, seed + 2, 5))
        for cnd in cand_list:
            vv = effective_c5(cnd)
            if vv < best_c:
                best_c = vv
                best_h = cnd.copy()
        if best_h is None:
            best_h = trig_h(70, 0)
            best_c = effective_c5(best_h)

    n0 = int(best_h.size)
    start_h = best_h.copy()

    if time.time() < hard_deadline - 5.0:
        cands = quick_candidates_from_h(best_h)
        best_cand = None
        best_cand_val = float("inf")
        for cand in cands:
            cv = effective_c5(cand)
            if cv < best_cand_val - 1e-12:
                best_cand_val = cv
                best_cand = cand
        if best_cand is not None:
            start_h = best_cand
            if best_cand_val < best_c:
                best_c = best_cand_val
                best_h = best_cand.copy()

    if best_h.size <= 200:
        rem = hard_deadline - time.time()
        if rem > 60:
            alloc = min(300.0, 0.35 * rem)
            if alloc > 10:
                h1, c1 = optimize_projected(
                    start_h,
                    alloc,
                    hard_deadline,
                    rng,
                    (20.0, 50.0, 100.0),
                    0.06 if n0 <= 100 else 0.04,
                )
                if c1 < best_c:
                    best_c = c1
                    best_h = h1.copy()

        rem = hard_deadline - time.time()
        if rem > 50 and minimize is not None:
            alloc = min(60.0, 0.12 * rem)
            if alloc > 5:
                h2, c2 = slsqp_polish(
                    best_h,
                    alloc,
                    60.0,
                    hard_deadline,
                    maxiter=200 if best_h.size <= 100 else 100,
                )
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()

    rem = hard_deadline - time.time()
    if rem > 40 and best_h.size <= 280:
        alloc = min(45.0, 0.10 * rem)
        if alloc > 5:
            h3, c3 = pair_refine(best_h, alloc, rng, hard_deadline)
            if c3 < best_c:
                best_c = c3
                best_h = h3.copy()

    max_up = min(max(4 * n0, 560), 700)
    targets = []
    for mult in (2, 4, 8):
        nt = int(n0 * mult)
        if nt > int(best_h.size) + 5 and nt <= max_up and nt not in targets:
            targets.append(nt)

    for nt in targets:
        rem = hard_deadline - time.time()
        if rem < 120:
            break
        prev_c = best_c
        h_up = upsample(best_h, nt)
        base_step = 0.05 if nt <= 140 else 0.035
        alloc = min(240.0, 0.25 * rem)
        if alloc <= 10:
            break
        h_t, c_t = optimize_projected(
            h_up,
            alloc,
            hard_deadline,
            rng,
            (20.0, 60.0),
            base_step,
        )
        if c_t < best_c:
            best_c = c_t
            best_h = h_t.copy()

        rem = hard_deadline - time.time()
        if rem > 35 and nt <= 180 and minimize is not None:
            sl = min(35.0, 0.10 * rem)
            if sl > 5:
                h_s, c_s = slsqp_polish(best_h, sl, 60.0, hard_deadline, maxiter=80)
                if c_s < best_c:
                    best_c = c_s
                    best_h = h_s.copy()

        rem = hard_deadline - time.time()
        if rem > 25 and nt <= 180:
            pl = min(25.0, 0.08 * rem)
            if pl > 3:
                h_p, c_p = pair_refine(best_h, pl, rng, hard_deadline)
                if c_p < best_c:
                    best_c = c_p
                    best_h = h_p.copy()

        if best_c >= prev_c - 1e-9:
            break

    rem = hard_deadline - time.time()
    if rem > 70 and best_h.size <= 280:
        n_alt = int(best_h.size)
        alternatives = []
        for kind in range(6):
            alternatives.append(trig_h(n_alt, kind))
        alternatives.append(smoothed_random(n_alt, seed + 101, 3))
        alternatives.append(smoothed_random(n_alt, seed + 202, 5))

        scored = []
        for a in alternatives:
            scored.append((effective_c5(a), a))
        chosen = []
        remaining = scored
        for _ in range(min(3, len(remaining))):
            bestv = float("inf")
            besti = 0
            for idx in range(len(remaining)):
                if remaining[idx][0] < bestv:
                    bestv = remaining[idx][0]
                    besti = idx
            chosen.append(remaining[besti])
            del remaining[besti]

        for pair in chosen:
            rem = hard_deadline - time.time()
            if rem < 30:
                break
            alloc = min(60.0, 0.15 * rem)
            if alloc <= 5:
                break
            h_a, c_a = optimize_projected(
                pair[1],
                alloc,
                hard_deadline,
                rng,
                (20.0, 60.0),
                0.05,
            )
            if c_a < best_c:
                best_c = c_a
                best_h = h_a.copy()

            rem = hard_deadline - time.time()
            if rem > 25 and best_h.size <= 200 and minimize is not None:
                sl = min(25.0, 0.10 * rem)
                if sl > 3:
                    h_b, c_b = slsqp_polish(best_h, sl, 60.0, hard_deadline, maxiter=80)
                    if c_b < best_c:
                        best_c = c_b
                        best_h = h_b.copy()

    best_h = make_feasible(best_h)
    best_c = effective_c5(best_h)

    return np.asarray(best_h, dtype=float), float(best_c), int(best_h.size)
