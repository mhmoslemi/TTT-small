import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fix_sum(h, n):
    target = n * 0.5
    h = h.copy()
    for _ in range(500):
        np.clip(h, 0.0, 1.0, out=h)
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Spread to non-bound elements
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
    return np.clip(h, 0.0, 1.0)

def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    old_idx = np.arange(n_old) / (n_old - 1)
    new_idx = np.arange(n_new) / (n_new - 1)
    h_new = np.interp(new_idx, old_idx, h)
    return fix_sum(h_new, n_new)

def local_search_binary(h, n, budget, rng):
    """Local search on binary (or near-binary) h."""
    start = time.time()
    h = h.copy()
    h = fix_sum(h, n)
    best_c5 = compute_c5(h)
    no_improve = 0
    
    while time.time() - start < budget:
        # Find worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s_star = k_star - (n - 1)
        
        # Try moves
        improved = False
        num_tries = min(20, max(5, int((budget - (time.time()-start)) * 50)))
        
        for _ in range(num_tries):
            trial = h.copy()
            s = abs(s_star)
            
            if s == 0:
                # Binary: push a fractional value to 0 or 1
                frac = (h > 0.01) & (h < 0.99)
                if np.sum(frac) > 0:
                    idx = np.where(frac)[0]
                    i = idx[rng.integers(len(idx))]
                    trial[i] = 1.0 if h[i] > 0.3 else 0.0
                else:
                    # Already binary, try a flip pair
                    i = rng.integers(n)
                    trial[i] = 1.0 - h[i]
            else:
                # Try a swap at distance s
                i = rng.integers(0, max(1, n - s))
                j = i + s
                # Swap or adjust
                mode = rng.random()
                if mode < 0.3:
                    # Full swap
                    trial[i], trial[j] = trial[j], trial[i]
                elif mode < 0.6:
                    # Partial swap
                    delta = (trial[i] - trial[j]) * rng.uniform(0.3, 1.0)
                    trial[i] -= delta
                    trial[j] += delta
                else:
                    # Flip one
                    if h[i] > 0.5:
                        trial[i] = min(trial[i], 0.0)
                    else:
                        trial[i] = max(trial[i], 1.0)
            
            trial = fix_sum(trial, n)
            trial_c5 = compute_c5(trial)
            
            if trial_c5 < best_c5:
                h = trial
                best_c5 = trial_c5
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Random restart
                if rng.random() < 0.5:
                    # Small perturbation
                    h += rng.standard_normal(n) * 0.1

                no_improve = 0
                if compute_c5(h) > best_c5:
                    # Reset to best
                    pass  # Keep the perturbed version for exploration
    
    h = fix_sum(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5
