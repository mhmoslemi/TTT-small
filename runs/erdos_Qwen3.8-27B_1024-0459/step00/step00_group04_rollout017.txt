import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _smooth_max(corr, T):
    m = np.max(corr)
    scaled = (corr - m) * T
    scaled = np.clip(scaled, -700, 700)
    logsum = m + np.log(np.sum(np.exp(scaled)) / len(corr)) / T
    return float(logsum)

def _project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute correction
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    n_mask = np.sum(mask)
    if n_mask == 0:
        return h
    per = diff / n_mask
    h_new = h.copy()
    h_new[mask] += per
    h_new = np.clip(h_new, 0.0, 1.0)
    # Final adjustment
    residual = target - np.sum(h_new)
    if abs(residual) > 1e-10:
        idx = np.argsort(np.abs(h_new - 0.5))[:abs(int(residual))]
        if residual > 0:
            for i in idx:
                h_new[i] = min(1.0, h_new[i] + residual / max(1, len(idx)))
        else:
            for i in idx:
                h_new[i] = max(0.0, h_new[i] + residual / max(1, len(idx)))
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _slsqp_optimize(h_init, n, T, max_iter):
    target = n / 2.0

    def objective(h):
        corr = _compute_corr(h)
        return _smooth_max(corr, T)

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n

    h_start = _project_to_feasible(h_init.copy(), n)
    res = minimize(objective, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    if res.x is not None:
        h_out = _project_to_feasible(res.x, n)
        return h_out
    return None

    target = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(-h)[:target]
    h_bin[idx] = 1.0
    return h_bin

    best_val = best_corr

        temp = temperature_init * (cooling ** it)

        h_bin[j] = 1.0

            h_sa = np.zeros(n_init)
            idx = rng.choice(n_init, size=n_init // 2, replace=False)
            h_sa[idx] = 1.0
        
        max_iter = 200000
        t_init = max(1.0, 2.0 / n_init)
        cooling = (0.001) ** (1.0 / max_iter)
        
        h_sa_out, best_sa_val = _local_search_binary(h_sa, n_init, max_iter, t_init, cooling)
        c5_sa = best_sa_val * (2.0 / n_init)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa_out.copy()
            best_n = n_init
            print(f"  SA restart {restart}: C5={c5_sa:.6f}")
    
    # Phase 3: Steepest descent local search on best binary
    print("Phase 3: Steepest descent...")
    if best_n == n_init and np.allclose(best_h, np.round(best_h)):
        h_sd = best_h.copy()
    else:
        h_sd = _to_binary_preserve_sum(best_h, n_init)
    h_sd_out, best_sd_val = _local_search_steepest(h_sd, n_init, max_iter=30000)
    c5_sd = best_sd_val * (2.0 / n_init)
    if c5_sd < best_c5:
        best_c5 = c5_sd
        best_h = h_sd_out.copy()
        best_n = n_init
        print(f"  Steepest descent: C5={c5_sd:.6f}")
    
    # Phase 4: More SA with different parameters
    print("Phase 4: Extended SA...")
    for trial in range(8):
        if time_left() < 200:
            break
        # Start from best binary or random
        if trial % 2 == 0:
            h_sa = _to_binary_preserve_sum(best_h, n_init)
            # Perturb: flip 3 random pairs
            for _ in range(3):
                ones = np.where(h_sa == 1)[0]
                zeros = np.where(h_sa == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    h_sa[ones[rng.integers(len(ones))]] = 0
                    h_sa[zeros[rng.integers(len(zeros))]] = 1
        else:
            h_sa = np.zeros(n_init)
            idx = rng.choice(n_init, size=n_init // 2, replace=False)
            h_sa[idx] = 1.0
        
        max_iter = 300000
        t_init = 1.5
        cooling = (0.001) ** (1.0 / max_iter)
        
        h_sa_out, best_sa_val = _local_search_binary(h_sa, n_init, max_iter, t_init, cooling)
        c5_sa = best_sa_val * (2.0 / n_init)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa_out.copy()
            best_n = n_init
            print(f"  SA trial {trial}: C5={c5_sa:.6f}")
        
        # Follow up with steepest descent
        h_sd2, sd_val2 = _local_search_steepest(h_sa_out, n_init, max_iter=20000)
        c5_sd2 = sd_val2 * (2.0 / n_init)
        if c5_sd2 < best_c5:
            best_c5 = c5_sd2
            best_h = h_sd2.copy()
            best_n = n_init
            print(f"  SA+SD trial {trial}: C5={c5_sd2:.6f}")
    
    # Phase 5: Try higher resolution n=84
    print("Phase 5: Higher resolution n=84...")
    if time_left() > 300:
        n84 = 84
        h84 = _upsample(best_h, n84)
        h84 = _project_to_feasible(h84, n84)
        c5_84 = _compute_c5(h84)
        print(f"  Upsampled n=84: C5={c5_84:.6f}")
        
        # SLSQP on n=84
        for T in [50, 100]:
            if time_left() < 200:
                break
            try:
                h84_opt = _slsqp_optimize(h84, n84, T, max_iter=150)
                if h84_opt is not None:
                    c5_84_opt = _compute_c5(h84_opt)
                    if c5_84_opt < best_c5:
                        best_c5 = c5_84_opt
                        best_h = h84_opt.copy()
                        best_n = n84
                        print(f"  SLSQP n=84 T={T}: C5={c5_84_opt:.6f}")
            except Exception as e:
                print(f"  SLSQP n=84 failed: {e}")
        
        # Binary SA on n=84
        if time_left() > 200:
            h84_bin = _to_binary_preserve_sum(h84, n84)
            c5_84b = _compute_c5(h84_bin)
            if c5_84b < best_c5:
                best_c5 = c5_84b
                best_h = h84_bin.copy()
                best_n = n84
            
            for sa_trial in range(3):
                if time_left() < 150:
                    break
                h84_sa = _to_binary_preserve_sum(best_h, n84)
                if sa_trial > 0:
                    h84_sa = np.zeros(n84)
                    idx = rng.choice(n84, size=n84 // 2, replace=False)
                    h84_sa[idx] = 1.0
                
                max_iter = 200000
                t_init = 1.0
                cooling = (0.001) ** (1.0 / max_iter)
                h84_out, best_sa_val = _local_search_binary(h84_sa, n84, max_iter, t_init, cooling)
                c5_84sa = best_sa_val * (2.0 / n84)
                if c5_84sa < best_c5:
                    best_c5 = c5_84sa
                    best_h = h84_out.copy()
                    best_n = n84
                    print(f"  SA n=84 trial {sa_trial}: C5={c5_84sa:.6f}")
    
    # Phase 6: Final polish - SLSQP from best solution
    print("Phase 6: Final polish...")
    if time_left() > 60:
        try:
            h_final = _slsqp_optimize(best_h, best_n, T=300, max_iter=300)
            if h_final is not None:
                c5_final = _compute_c5(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
                    print(f"  Final polish: C5={c5_final:.6f}")
        except Exception as e:
            print(f"  Final polish failed: {e}")
    
    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    # Ensure sum constraint is met exactly in floating point
    target_sum = best_n / 2.0
    current_sum = np.sum(best_h)
    if abs(current_sum - target_sum) > 1e-10:
        best_h = _project_to_feasible(best_h, best_n)
    
    best_c5 = _compute_c5(best_h)
    print(f"Final: n={best_n}, C5={best_c5:.6f}")
    
    return best_h, best_c5, best_n
