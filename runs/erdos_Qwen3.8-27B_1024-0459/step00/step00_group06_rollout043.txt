def fix_sum(h, n):
    """Fix h so that sum(h) = n/2 and 0 <= h[i] <= 1."""
    target = n / 2.0

    return m + float(np.log(np.sum(np.exp(overlaps - m)))) / T

def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(h)
        for j in order:
            add = min(diff, 1.0 - h[j])
            h[j] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        order = np.argsort(-h)
