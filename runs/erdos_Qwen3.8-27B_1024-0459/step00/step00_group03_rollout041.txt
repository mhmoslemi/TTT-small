import numpy as np
import time

def project_feasible(y, target_sum):
    n = len(y)
    # Bisection for lambda
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * dx

def compute_gradient(h, dx, tau):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    # For each shift s (indexed by idx = s + n - 1, s from -(n-1) to (n-1)):
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        s = idx - (n - 1)
        # dc_s/dh[j] = (1-h[j-s]) * I(0<=j-s<n) - h[j+s] * I(0<=j+s<n)
        # First term: valid j are max(0, s) to min(n-1, n-1+s)
        j_lo = max(0, s)
        j_hi = min(n - 1, n - 1 + s)
        if j_hi >= j_lo:
            grad[j_lo:j_hi+1] += wi * (1.0 - h[j_lo - s:j_hi + 1 - s])
        
        # Second term: valid j are max(0, -s) to min(n-1, n-1-s)
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_hi2 >= j_lo2:
            grad[j_lo2:j_hi2+1] += -wi * h[j_lo2 + s:j_hi2 + 1 + s]
    
    return grad * dx

    best_c5 = compute_c5(h)
    
    alpha = 0.5 / max(tau_end, 1.0)  # initial step size
    min_alpha = 1e-6
    max_alpha = 10.0
    
    for it in range(max_iters):
        tau = tau_start * (tau_end / tau_start) ** (it / max(1, max_iters))
        
        grad = compute_gradient(h, dx, tau)
        
        # Gradient descent

def project_feasible(y, target_sum):
    """Project y onto {x: 0 <= x_i <= 1, sum(x) = target_sum}."""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    """Compute C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_gradient_smoothed(h, dx, tau):
    """Compute gradient of softmax-smoothed C5."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    c_shifted = c - c.max()
    exp_c = np.exp(tau * c_shifted)
    w = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-15:
            continue
        s = idx - (n - 1)
        # First term: (1-h[j-s]) for valid j
        j_lo = max(0, s)
        j_hi = min(n - 1, n - 1 + s)
        if j_hi >= j_lo:
            grad[j_lo:j_hi+1] += wi * (1.0 - h[j_lo - s:j_hi + 1 - s])
        # Second term: -h[j+s] for valid j
        j_lo2 = max(0, -s)
        j_hi2 = min(n - 1, n - 1 - s)
        if j_hi2 >= j_lo2:

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = project_feasible(h, target)
    return h


def make_block_init(n, n_blocks, seed=42):
    """Create block-structured initialization."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    block_size = n // n_blocks
    remainder = n - block_size * n_blocks
    num_on = n_blocks // 2
    on_blocks = set(rng.choice(n_blocks, num_on, replace=False))
    pos = 0
    for b in range(n_blocks):
        sz = block_size + (1 if b < remainder else 0)
        if b in on_blocks:
            h[pos:pos + sz] = 1.0
        pos += sz
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Get initial solution
    try:
        h_parent = np.array(initial_h_values, dtype=float)
    except Exception:
        h_parent = np.ones(70) * 0.5

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    n_parent = len(h_parent)

    # Phase 1: Optimize at parent resolution with PGD
    if remaining() > 30:
        dx = 2.0 / n_parent
        ts = n_parent / 2.0
        iters1 = min(8000, int(remaining() * 20))
        h_opt, c5_opt = optimize_pgd(h_parent, iters1, dx, ts, tau_start=5.0, tau_end=100.0,
                                     alpha_init=0.3, seed=seed)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 2: Try L-BFGS-B at parent resolution
    if remaining() > 60:
        dx = 2.0 / n_parent
        ts = n_parent / 2.0
        h_opt, c5_opt = optimize_lbfgsb(best_h, 300, dx, ts, tau=30.0, penalty=1e5)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: Multi-resolution strategy
    for factor in [2, 4, 8, 16, 32]:
        if remaining() < 80:
            break
        n_new = n_parent * factor
        if n_new > 2500:
            break
        dx_new = 2.0 / n_new
        ts_new = n_new / 2.0

        # From upsampled best
        h_init = upsample_linear(best_h, n_new)
        h_init = project_feasible(h_init, ts_new)

        # Scale iterations based on cost
        cost_factor = (n_new / n_parent) ** 1.5
        n_iters = max(500, min(5000, int(60 * 500 / cost_factor)))

        h_opt, c5_opt = optimize_pgd(h_init, n_iters, dx_new, ts_new,
                                     tau_start=10.0, tau_end=150.0, alpha_init=0.2, seed=seed + factor)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

        # L-BFGS-B at this resolution (if fast enough)
        if remaining() > 50 and n_new <= 400:
            h_opt2, c5_opt2 = optimize_lbfgsb(best_h, 200, dx_new, ts_new, tau=50.0, penalty=1e5)
            if c5_opt2 < best_c5:
                best_h = h_opt2
                best_c5 = c5_opt2

        # Random restart
        if remaining() > 50:
            h_rand = np.random.uniform(0, 1, n_new)
            h_rand = project_feasible(h_rand, ts_new)
            n_iters_r = max(300, n_iters // 4)
            h_opt_r, c5_opt_r = optimize_pgd(h_rand, n_iters_r, dx_new, ts_new,
                                             tau_start=10.0, tau_end=80.0, seed=seed + 100 + factor)
            if c5_opt_r < best_c5:
                best_h = h_opt_r
                best_c5 = c5_opt_r

    # Phase 4: Structured initializations
    n_best = len(best_h)
    for n_try in [n_best, max(n_parent, n_best // 2), min(n_best * 2, 1200)]:
        if remaining() < 40:
            break
        dx_try = 2.0 / n_try
        ts_try = n_try / 2.0

        # Sinusoidal inits
        for freq in [1, 2, 3, 4]:
            if remaining() < 25:
                break
            for amp in [0.3, 0.5, 0.7]:
                h_sin = make_sinusoidal_init(n_try, freq, amp, np.pi / 4)
                c5_sin = compute_c5(h_sin)
                if c5_sin < best_c5 + 0.05:
                    n_iters_s = max(500, min(2000, int(remaining() * 3 / (n_try / 70))))
                    h_opt_s, c5_opt_s = optimize_pgd(h_sin, n_iters_s, dx_try, ts_try,
                                                     tau_start=10.0, tau_end=100.0, seed=seed + freq * 10)
                    if c5_opt_s < best_c5:
                        best_h = h_opt_s
                        best_c5 = c5_opt_s

        # Block inits
        for nb in [4, 6, 8, 10, 12]:
            if remaining() < 25:
                break
            h_blk = make_block_init(n_try, nb, seed=seed + nb * 7)
            c5_blk = compute_c5(h_blk)
            if c5_blk < best_c5 + 0.05:
                n_iters_b = max(300, min(1500, int(remaining() * 2 / (n_try / 70))))
                h_opt_b, c5_opt_b = optimize_pgd(h_blk, n_iters_b, dx_try, ts_try,
                                                 tau_start=10.0, tau_end=80.0, seed=seed + nb * 13)
                if c5_opt_b < best_c5:
                    best_h = h_opt_b
                    best_c5 = c5_opt_b

    # Phase 5: Final polish with high tau
    if remaining() > 20:
        n_final = len(best_h)
        dx_final = 2.0 / n_final
        ts_final = n_final / 2.0
        n_iters_final = max(200, int(remaining() * 8 / max(1, (n_final / 70) ** 1.5)))
        h_polished, c5_polished = optimize_pgd(best_h, n_iters_final, dx_final, ts_final,
                                               tau_start=50.0, tau_end=500.0,
                                               alpha_init=0.05, topk_transition=0.3, seed=seed + 999)
        if c5_polished < best_c5:
            best_h = h_polished
            best_c5 = c5_polished

    # Final validation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
