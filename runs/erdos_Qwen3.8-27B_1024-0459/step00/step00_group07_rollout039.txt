        # First part: j ranges from k to n-1, grad[j] += (1 - h[j-k])
        grad[k:] += (1.0 - h[:m])

        grad[:m] += -h[k:]

        m = n - kp
        grad[:m] += (1.0 - h[kp:])
        grad[kp:] += -h[:m]
    
    return grad

def project_feasible(h, n):
    """Project h to satisfy 0 <= h <= 1 and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    s = h.sum()
    target = n / 2.0
    if s != target:
        diff = target - s
        # Adjust: if diff > 0, increase values that are < 1
        # If diff < 0, decrease values that are > 0
        if diff > 0:
            # Find entries < 1 and distribute
            room = 1.0 - h  # available room
            total_room = room.sum()
            if total_room > 0:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:
            headroom = h  # available to decrease
            total_headroom = headroom.sum()
            if total_headroom > 0:
                h -= headroom * (-diff / total_headroom)
            else:
                h += diff / n
    h = np.clip(h, 0, 1)
    # Final correction
    s = h.sum()
    if abs(s - target) > 1e-10:
        # Proportional adjustment
        h += (target - s) / n
        h = np.clip(h, 0, 1)
    return h

Let me also consider: what if I use a "greedy" construction instead of local search?

Greedy construction:
1. Start with an empty sequence
2. At each step, place a 1 (or 0) at the position that minimizes the increase in max out[k]
3. Repeat until we have n/2 ones

This is O(n³) total (n steps, each trying O(n) positions, each requiring O(n) to evaluate). For n=200, that's 8M operations, very fast.

But greedy construction tends to get stuck in bad local optima. Local search from a good start is usually better.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    out = np.correlate(h, 1.0 - h, mode="full")
    return out.max() * (2.0 / n)

def local_search(h, max_time, rng, n_trials_large=5000):
    n = len(h)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    start = time.time()
    stall = 0
    
    while time.time() - start < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap = None
        best_c5_try = c5
        
        if n <= 250:
            # Full search
            for i in ones:
                h[i] = 0
                for j in zeros:
                    h[j] = 1
                    c5_try = compute_c5(h)
                    h[j] = 0
                    if c5_try < best_c5_try - 1e-15:
                        best_c5_try = c5_try
                        best_swap = (i, j)
                        # Don't break - find the best
                h[i] = 1
        else:
            # Random trials
            for _ in range(n_trials_large):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                c5_try = compute_c5(h)
                h[i] = 1
                h[j] = 0
                if c5_try < best_c5_try - 1e-15:
                    best_c5_try = c5_try
                    best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            c5 = best_c5_try
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            # Try 2-flips to escape
            if len(ones) >= 2 and len(zeros) >= 2:
                tried = 0
                while tried < 100:
                    i1, i2 = rng.choice(ones, 2, replace=False)
                    j1, j2 = rng.choice(zeros, 2, replace=False)
                    h[i1], h[i2] = 0, 0
                    h[j1], h[j2] = 1, 1
                    c5_try = compute_c5(h)
                    if c5_try < c5:
                        c5 = c5_try
                        if c5 < best_c5 - 1e-15:
                            best_c5 = c5
                            best_h = h.copy()
                        stall = 0
                        break
                    h[i1], h[i2] = 1, 1
                    h[j1], h[j2] = 0, 0
                    tried += 1
            if stall > 10:
                break
    
    return best_h, best_c5
