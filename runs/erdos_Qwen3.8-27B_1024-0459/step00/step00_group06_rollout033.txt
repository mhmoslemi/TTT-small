    m_start = max(0, k)  # m-k >= 0 → m >= k (if k>0)

    # If k < 0: m in [0, N+k-1] (since m >= k is always true for m>=0, and m < N+k)

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _proj(v, T):
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.clip(v - m, 0, 1).sum() > T:
            lo = m
        else:
            hi = m
    p = np.clip(v - 0.5 * (lo + hi), 0, 1)
    d = T - p.sum()

        g[:N-k] -= h[k:]

import numpy as np
import time


def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def _proj(v, T):
    lo, hi = float(v.min()) - 1.0, float(v.max()) + 1.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if float(np.clip(v - m, 0.0, 1.0).sum()) > T:
            lo = m
        else:
            hi = m
    p = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    d = T - float(p.sum())
    if abs(d) > 1e-14:
        ms = (p > 1e-12) & (p < 1.0 - 1e-12)
        cnt = int(ms.sum())
        if cnt > 0:
            p[ms] += d / cnt
            np.clip(p, 0.0, 1.0, out=p)
    return p


def _gsh(h, k):
    N = len(h)
    g = np.zeros(N)
    if k >= 0:
        g[k:] += 1.0 - h[:N - k]
        g[:N - k] -= h[k:]
    else:
        g[:N + k] += 1.0 - h[-k:]
        g[-k:] -= h[:N + k]
    return g


def _grad_step(h, dx, beta, top):
    N = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(-c)[:top]
    w = np.exp(beta * (c[idx] - c[idx[0]]))
    ws = float(w.sum())
    if ws < 1e-300:
        return None
    w = w / ws
    g = np.zeros(N)
    for j in range(len(idx)):
        if w[j] > 1e-5:
            g += w[j] * _gsh(h, int(idx[j]) - (N - 1))
    return g * dx


def _grad_opt(h, dx, T, t_end, beta_init, beta_max, top):

    best = _c5(h, dx)
    bh = h.copy()

                if beta > beta_max:
                    break
    return bh, best


def _interp_h(h_src, N_src, N_dst):
    x_src = np.linspace(0.0, 2.0, N_src, endpoint=False) + 1.0 / N_src
    x_dst = np.linspace(0.0, 2.0, N_dst, endpoint=False) + 1.0 / N_dst
    h_ext = np.concatenate([h_src[-1:], h_src, h_src[:1]])
    x_ext = np.concatenate([[-1.0 / N_src], x_src, [2.0 + 1.0 / N_src]])
    h_new = np.interp(x_dst, x_ext, h_ext)
    return h_new
