from scipy.optimize import minimize
import time

def compute_c5(h, n_points):
    """Compute C5 value for given h."""
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return dx * np.max(corr)

def compute_c5_grad(h, n_points):
    """Compute C5 value and subgradient."""
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    c5 = dx * np.max(corr)
    peak_idx = np.argmax(corr)
    s = peak_idx - (n_points - 1)  # shift
    
    # Compute subgradient of the peak correlation term
    # f = dx * sum_i h[i] * (1 - h[i+s]) over valid i
    grad = np.zeros(n_points)
    # For each valid i: term is h[i]*(1-h[i+s])
    # d/du[h[i]*(1-h[i+s])] at u=h[i] = 1-h[i+s]
    # d/du[h[i]*(1-h[i+s])] at u=h[i+s] = -h[i]
    if s >= 0:
        # i+s < n means i < n-s, and i >= 0
        i_range = range(n_points - s)  # i from 0 to n-1-s
        for i in i_range:
            grad[i] += dx * (1 - h[i+s])
            grad[i+s] += -dx * h[i]
    else:
        ss = -s
        # i - ss >= 0 means i >= ss, and i < n
        i_range = range(ss, n_points)  # i from ss to n-1
        for i in i_range:
            grad[i] += dx * (1 - h[i-ss])
            grad[i-ss] += -dx * h[i]
    
    return c5, grad

def project_onto_constraint(h, target, n):
    """Project h onto {x: sum x = target, 0 ≤ x ≤ 1}."""
    # Bisection on lambda
    lo, hi = -h.max() - 1, -h.min() + 1  # lambda range
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - mid, 0, 1)
    # Fine adjustment
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        # Distribute remaining
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

import time

def compute_c5_value(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def compute_subgrad(h, n_points):
    """Compute subgradient of max correlation term."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = dx * np.max(corr)
    peak = np.argmax(corr)
    s = peak - (n_points - 1)
    
    grad = np.zeros(n_points)
    if s >= 0:
        for i in range(n_points - s):
            grad[i] += dx * (1.0 - h[i + s])
            grad[i + s] += -dx * h[i]
    else:
        ss = -s
        for i in range(ss, n_points):
            grad[i] += dx * (1.0 - h[i - ss])
            grad[i - ss] += -dx * h[i]
    
    return c5, grad

def project_onto_constraint(h, target):
    """Project h onto {x: sum x = target, 0 <= x <= 1}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)

        if s > target:

        else:
            hi = mid

        else:

        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

            lr *= 0.7
            no_improve = 0
        if no_improve > 2000:
            break
    
    return best_h, best_c5

def slsqp_optimize(h, n_points, time_limit=60):
    """Optimize using SLSQP with smooth max."""
    target = n_points / 2.0
    dx = 2.0 / n_points
    start_t = time.time()
    
    def objective(x):
        if time.time() - start_t > time_limit:
            return 1e10
        corr = np.correlate(x, 1.0 - x, mode='full')
        # Smooth max approximation
        tau = 0.01
        v = corr * dx
        v_max = v.max()
        soft = v_max + tau * np.log(np.sum(np.exp((v - v_max) / tau)))
        return soft
    
    def constraint_eq(x):
        return [x.sum() - target]
    
    bounds = [(0.0, 1.0)] * n_points

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * corr.max()

def get_active_gradients(h, n, top_m=1):
    """Get gradients of the top-m correlation terms."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = dx * corr.max()
    
    # Find top-m indices
    if top_m >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argsort(corr)[-top_m:]
    
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad[i] += dx * (1.0 - h[i + s])
                grad[i + s] += -dx * h[i]
        else:
            ss = -s
            for i in range(ss, n):
                grad[i] += dx * (1.0 - h[i - ss])
                grad[i - ss] += -dx * h[i]
    
    grad /= len(top_idx)
    return c5, grad

def project(h, target):
    """Project onto {x: sum(x)=target, 0<=x<=1}."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        xp = np.clip(h - mid, 0.0, 1.0)
        if xp.sum() > target:
            lo = mid
        else:
            hi = mid
    xp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    d = target - xp.sum()
    if abs(d) > 1e-12:
        m = xp < (1.0 - 1e-12) if d > 0 else xp > 1e-12
        if m.any():
            xp[m] += d / m.sum()
            xp = np.clip(xp, 0.0, 1.0)
    return xp

    best_c5 = compute_c5(h, n)
    t0 = time.time()

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        step = lr / (gn + 1e-6)
        h_new = project(h - step * g, target)
        c5_new = compute_c5(h_new, n)

            best_c5 = c5_new

                stall = 0
            elif stall > 500:

        if best_c5 < 0.35:
            break

def optimize_slsqp(h, n, time_limit=60):
    target = n / 2.0
    dx = 2.0 / n
    t0 = time.time()
    
    def obj(x):
        if time.time() - t0 > time_limit:
            return 1e6
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        # Smooth max
        tau = 0.05
        cm = c.max()
        return cm + tau * np.log(np.sum(np.exp((c - cm) / tau)))
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        x = np.clip(res.x, 0.0, 1.0)
        x = project(x, target)
        return x, compute_c5(x, n)
    except:
        return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def get_top_gradients(h, n, top_m=3):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = dx * corr.max()
    if top_m >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s >= 0:
            for i in range(n - s):
                grad[i] += dx * (1.0 - h[i + s])
                grad[i + s] += -dx * h[i]
        else:
            ss = -s
            for i in range(ss, n):
                grad[i] += dx * (1.0 - h[i - ss])
                grad[i - ss] += -dx * h[i]
    grad /= float(len(top_idx))
    return c5, grad


def project_onto_set(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        xp = np.clip(h - mid, 0.0, 1.0)
        if xp.sum() > target:
            lo = mid
        else:
            hi = mid
    xp = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - xp.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            m = xp < 1.0 - 1e-13
        else:
            m = xp > 1e-13
        if m.any():
            xp[m] += diff / m.sum()
            xp = np.clip(xp, 0.0, 1.0)
    return xp


def optimize_subgrad(h, n, time_limit, top_m=3):
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    lr = 1.0
    stall = 0
    while time.time() - t0 < time_limit:
        c5, g = get_top_gradients(h, n, top_m)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = lr / (gn + 1e-8)
        h_new = project_onto_set(h - step * g, target)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.92
                stall = 0
            elif stall > 500:
                break
        h = h_new
        if best_c5 < 0.34:
            break
    return best_h, best_c5


def optimize_slsqp(h, n, time_limit=60):
    target = n / 2.0
    dx = 2.0 / n
    t0 = time.time()

    def obj(x):
        if time.time() - t0 > time_limit:
            return 1e6
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        tau = 0.02
        cm = c.max()
        shifted = (c - cm) / tau
        shifted = np.clip(shifted, -50, 50)
        return cm + tau * np.log(np.sum(np.exp(shifted)) + 1e-30)

    def eq_fun(x):
        return x.sum() - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 400, 'ftol': 1e-14})
        x = np.clip(res.x, 0.0, 1.0)
        x = project_onto_set(x, target)
        c5v = compute_c5(x, n)
        return x, c5v
    except Exception:
        return h, compute_c5(h, n)


def optimize_local_search(h, n, time_limit=60):
    """Coordinate-wise local search."""
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    n_pass = 0
    while time.time() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            for delta in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
                h_try = best_h.copy()
                h_try[i] = h_try[i] + delta
                if h_try[i] < 0.0 or h_try[i] > 1.0:
                    continue
                h_try = project_onto_set(h_try, target)
                c5_try = compute_c5(h_try, n)
                if c5_try < best_c5 - 1e-12:
                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        n_pass += 1
        if not improved:
            break
    return best_h, best_c5


def upscale(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    hi = np.interp(x_new, x_old, h)
    hi = np.clip(hi, 0.0, 1.0)
    hi = project_onto_set(hi, n_new / 2.0)
    return hi

    h = np.clip(h, 0.0, 1.0)
    h = project_onto_set(h, n / 2.0)
    return h
