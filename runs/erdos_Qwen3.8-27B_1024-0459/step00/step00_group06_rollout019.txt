                      constraints=[{'type': 'eq', 'fun': con_func}],
                      options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None:
            h_out = project_sum(res.x, target)
            return h_out
    except Exception:
        pass
    return h_init

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # Evaluate parent construction
    global initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.array(initial_h_values, dtype=np.float64)
        c5_p = compute_c5(hp)
        best_h = hp.copy()
        best_c5 = c5_p
        best_n = len(hp)
    
    # Try binary local search at multiple resolutions
    resolutions = [100, 150, 200]
    
    for n in resolutions:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining < 60:
            break
        
        target = n // 2
        dx = 2.0 / n
        
        # Determine time per local search
        time_per_ls = min(remaining / 3, 80.0)
        
        n_starts = 4
        if n == 200:
            time_per_ls = min(remaining / 4, 60.0)
            n_starts = 3
        
        for s in range(n_starts):
            elapsed = time.time() - t_start
            remaining = budget_s - elapsed
            if remaining < 30:
                break
            
            # Create initial binary sequence
            rng_init = np.random.default_rng(seed + s * 1000 + n)
            h_init = np.zeros(n)
            idx_ones = rng_init.choice(n, target, replace=False)
            h_init[idx_ones] = 1.0
            
            # If we have a parent, also try interpolation-based start
            if s == 0 and best_h is not None and best_n != n:
                x_parent = np.linspace(0, 2, best_n, endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                h_interp = np.interp(x_new, x_parent, best_h)
                h_init = (h_interp > 0.5).astype(np.float64)
                n_ones = int(np.sum(h_init))
                if n_ones != target:
                    if n_ones > target:
                        ones_idx = np.where(h_init == 1.0)[0]
                        rng_init.shuffle(ones_idx)
                        h_init[ones_idx[target:]] = 0.0
                    else:
                        zeros_idx = np.where(h_init == 0.0)[0]
                        rng_init.shuffle(zeros_idx)
                        h_init[zeros_idx[:target - n_ones]] = 1.0
            
            # Run local search
            ls_time = min(time_per_ls, budget_s - (time.time() - t_start) - 20)
            if ls_time < 5:
                continue
            
            h_ls, c5_ls = binary_local_search(h_init, ls_time, seed + s)
            
            if c5_ls < best_c5:
                best_h = h_ls.copy()
                best_c5 = c5_ls
                best_n = n
        
        # Continuous refinement on best binary for this n
        elapsed = time.time() - t_start
        if elapsed < budget_s - 120 and best_n == n:
            h_ref = continuous_refine(best_h, float(target), max_iter=200)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_h = h_ref
                best_c5 = c5_ref
                best_n = n
    
    # Final: try one more round with the best configuration if time allows
    elapsed = time.time() - t_start
    if elapsed < budget_s - 100 and best_h is not None:
        n_best = best_n
        target_best = n_best / 2.0
        # Perturb and re-run local search from best
        rng_final = np.random.default_rng(seed + 99999)
        h_pertub = best_h.copy()
        n_perturb = min(5, n_best // 10)
        ones_idx = np.where(h_pertub >= 0.5)[0]
        zeros_idx = np.where(h_pertub < 0.5)[0]
        if len(ones_idx) >= n_perturb and len(zeros_idx) >= n_perturb:
            swap_o = rng_final.choice(ones_idx, n_perturb, replace=False)
            swap_z = rng_final.choice(zeros_idx, n_perturb, replace=False)
            h_pertub[swap_o] = 1.0 - h_pertub[swap_o]
            h_pertub[swap_z] = 1.0 - h_pertub[swap_z]
        h_pertub = np.clip(h_pertub, 0, 1)
        
        ls_time = min(120, budget_s - (time.time() - t_start) - 30)
        if ls_time > 10:
            h_final, c5_final = binary_local_search(h_pertub, ls_time, seed + 77777)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                best_n = n_best
        
        # SLSQP refinement
        elapsed = time.time() - t_start
        if elapsed < budget_s - 60:
            h_ref2 = continuous_refine(best_h, best_n / 2.0, max_iter=300)
            c5_ref2 = compute_c5(h_ref2)
            if c5_ref2 < best_c5:
                best_h = h_ref2
                best_c5 = c5_ref2
    
    # Fallback
    if best_h is None:
        best_h = np.ones(100) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 100
    
    # Ensure feasibility
    target_final = best_n / 2.0
    best_h = project_sum(best_h, target_final)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, best_n
