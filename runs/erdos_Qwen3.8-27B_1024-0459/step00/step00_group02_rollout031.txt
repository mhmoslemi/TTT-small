from scipy.optimize import minimize
import time

def shift_grad(h, s):
    """Gradient of corr at shift s w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        t = -s
        g[t:] += (1.0 - h[:n-t])
        g[:n-t] -= h[t:]
    return g

def c5_objective(h):
    n = len(h)
    dx = 2.0 / n
    return dx * np.max(np.correlate(h, 1.0 - h, mode='full'))

def c5_gradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    s = k_star - (n - 1)
    return dx * shift_grad(h, s)

def smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr - np.max(corr)
    m = np.max(corr) + eps * np.log(np.sum(np.exp(c / eps)))

    dx = 2.0 / n

            grad += w[k] * shift_grad(h, s)

def _slsqp_optimize(h0, n, maxiter=400, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_c5, h0, jac=_c5_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = _feasible(res.x, n)
        if _c5(h_new) < _c5(h0) + 1e-14:
            return h_new
    except Exception:
        pass
    return h0


def _smooth_anneal(h0, n, base_iter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h = h0.copy()
    eps_list = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
    for eps in eps_list:
        def obj_e(x):
            return _smooth_obj(x, eps)
        def obj_e_grad(x):
            return _smooth_grad(x, eps)
        try:
            res = minimize(obj_e, h, jac=obj_e_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': base_iter, 'ftol': 1e-10})
            h_new = _feasible(res.x, n)
            if _c5(h_new) < _c5(h):
                h = h_new
        except Exception:
            pass
    return h


def _local_search_01(h0, n, rng, n_iter=1000):
    target = n / 2.0
    h = np.zeros(n)
    sorted_idx = np.argsort(h0)[::-1]
    full = int(target)
    frac = target - full
    h[sorted_idx[:full]] = 1.0
    if frac > 1e-10:
        h[sorted_idx[full]] = frac
    c5_cur = _c5(h)
    temp = 0.01
    for it in range(n_iter):
        temp = max(1e-4, temp * 0.995)
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c5_new = _c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            c5_cur = c5_new
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h, c5_cur


def _greedy_swaps(h0, n, n_rounds=5):
    h = h0.copy()
    c5_cur = _c5(h)
    for _ in range(n_rounds):
        improved = False
        ones = np.where(h >= 0.999)[0]
        zeros = np.where(h <= 0.001)[0]
        best_delta = 0
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5_try = _c5(h)
                delta = c5_try - c5_cur
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_pair is not None:
            h[best_pair[0]] = 0.0
            h[best_pair[1]] = 1.0
            c5_cur += best_delta
            improved = True
        if not improved:
            break
    return h, c5_cur
