import numpy as np
from scipy.optimize import minimize
import time

def _c5_value(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _c5_value_fast(h, v):
    """Compute C5 where v = 1-h is precomputed"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, v, mode='full')
    return np.max(corr) * dx

def _upsample(h, n_new):
    """Upsample h to n_new points using linear interpolation"""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def _fix_integral(h, n):
    """Adjust h to satisfy sum(h) = n/2"""
    target = n / 2.0
    current = np.sum(h)
    diff = target - current
    # Distribute evenly, then clip
    h_adj = h + diff / n
    h_adj = np.clip(h_adj, 0, 1)
    # If clipping caused issues, iterate
    for _ in range(10):
        current = np.sum(h_adj)
        if abs(current - target) < 1e-10:
            break
        diff = target - current
        free = (h_adj > 0) & (h_adj < 1)
        if not np.any(free):
            break
        h_adj[free] += diff / np.sum(free)
        h_adj = np.clip(h_adj, 0, 1)
    return h_adj

def _project_box_sum(h, n):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Bisection to find shift
    lo, hi = -1, 1
    for _ in range(60):
        mid = (lo + hi) / 2
        h_shift = np.clip(h + mid, 0, 1)
        s = np.sum(h_shift)
        if s > target:
            hi = mid
        else:
            lo = mid
    h_shift = np.clip(h + (lo + hi) / 2, 0, 1)
    return h_shift

def _grad_c5(h):
    """Compute subgradient of C5 w.r.t. h"""
    n = len(h)
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    max_val = np.max(corr)
    # Subgradient: average of gradients at all maximizing lags
    active = (corr >= max_val - 1e-10)
    # Gradient of c_k w.r.t. h[j]:
    # c_k = sum_i h[i+k]*(1-h[i]) where k is the lag
    # dc_k/dh[j] = (1-h[j-k])*I[valid j-k] - h[j+k]*I[valid j+k]
    # where lag k maps to index k + n - 1 in the correlate array
    
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        k = k_idx - (n - 1)  # actual lag
        # h[j] appears in h[i+k] when i = j-k, valid when 0 <= j-k < n
        # contribution: (1 - h[j-k])
        # h[j] appears in (1-h[i]) when i = j, valid when 0 <= j+k < n
        # contribution: -h[j+k]
        
        # First term: (1-h[j-k]) for j where 0 <= j-k < n, i.e., k <= j < k+n
        # Since 0 <= j < n and k can be negative or positive:
        # j ranges from max(0, k) to min(n-1, k+n-1)
        j_lo = max(0, k)
        j_hi = min(n-1, k + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            # h[j-k] where j-k ranges from j_lo-k to j_hi-k
            temp_idx = idx - k
            grad[idx] += (1.0 - h[temp_idx])
        
        # Second term: -h[j+k] for j where 0 <= j+k < n, i.e., -k <= j < n-k
        j_lo2 = max(0, -k)
        j_hi2 = min(n-1, n - k - 1)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            temp_idx2 = idx2 + k
            grad[idx2] -= h[temp_idx2]
    
    n_active = np.sum(active)
    if n_active > 0:
        grad /= n_active
    
    return grad * (2.0 / n)  # multiply by dx

def _optimize_slsqp(h, n, max_time):
    """Optimize using SLSQP"""
    start = time.time()
    x0 = h.copy()
    
    def objective(x):
        return _c5_value(x)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                    'jac': lambda x: np.ones(n) / n}]
    bounds = [(0, 1)] * n
    
    result = minimize(objective, x0, method='SLSQP', constraints=constraints,
                      bounds=bounds, options={'maxiter': 200, 'ftol': 1e-10})
    
    if result.success or result.fun < objective(x0):
        return result.x, result.fun
    return x0, objective(x0)

def _optimize_pgd(h, n, max_time):

    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _fix_sum(h, n):
    target = n * 0.5
    diff = target - np.sum(h)
    h = h + diff / n
    return np.clip(h, 0.0, 1.0)

def _upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def _project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n * 0.5
    # Bisection on the level shift
    lo, hi = -1.0, 1.0
    for _ in range(100):

        h_proj = np.clip(h + mid, 0.0, 1.0)

        if s > target:

        else:

    return h_proj

def _subgrad(h):
    """Subgradient of C5 (using the max lag)"""
    n = len(h)
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    mx = np.max(corr)
    # Find active lags (within tolerance)
    tol = 1e-8
    active = corr >= mx - tol
    n_act = np.count_nonzero(active)
    if n_act == 0:
        return np.zeros(n)
    
    grad = np.zeros(n)
    lags = np.where(active)[0]
    for li in lags:
        k = li - (n - 1)  # lag value
        # Positive part: h[j] in h[i+k] slot → (1-h[j-k]) for valid j-k
        j_lo = max(0, k)
        j_hi = min(n - 1, n - 1 + k)
        if j_lo <= j_hi:

        j_lo2 = max(0, -k)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(h, n):
    target = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _up(h, n_new):
    n_old = len(h)
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xn, xo, h)

def _sg(h):
    """Subgradient of C5"""
    n = len(h)
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    mx = np.max(corr)
    # Use only the argmax for a cleaner gradient
    k_best = np.argmax(corr)
    k = k_best - (n - 1)
    grad = np.zeros(n)
    # positive part
    jlo = max(0, k)
    jhi = min(n - 1, n - 1 + k)
    if jlo <= jhi:
        idx = np.arange(jlo, jhi + 1)
        grad[idx] += 1.0 - h[idx - k]
    # negative part
    jlo2 = max(0, -k)
    jhi2 = min(n - 1, n - 1 - k)
    if jlo2 <= jhi2:
        idx2 = np.arange(jlo2, jhi2 + 1)
        grad[idx2] -= h[idx2 + k]
    return grad * (2.0 / n)

def _pgd(h, n, t_limit):
    """Projected gradient with backtracking"""
    t0 = time.time()
    x = _proj(h, n)
    best = x.copy()
    best_c5 = _c5(x)
    alpha = 0.1
    stall = 0
    
    while time.time() - t0 < t_limit:
        g = _sg(x)
        for _ in range(20):  # backtracking
            x_new = _proj(x - alpha * g, n)
            c5_new = _c5(x_new)
            if c5_new < best_c5 - 1e-12 or c5_new < _c5(x) - 1e-12:
                break
            alpha *= 0.5
        else:
            alpha *= 0.5
            stall += 1
            if stall > 50:

            continue
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best = x_new.copy()
            alpha = min(alpha * 1.1, 2.0)
            stall = 0
        else:
            alpha *= 0.7
        x = x_new
    
    return best, best_c5

    t0 = time.time()
    x = h.copy()
    best_c5 = _c5(x)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for epoch in range(200):

        for j in range(n):
