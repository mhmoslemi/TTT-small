def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Water filling: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    # Binary search on lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)

        # Add/subtract from entries not at bounds
        mask = (proj > 1e-10) & (proj < 1 - 1e-10)
        if mask.sum() > 0:

            proj = np.clip(proj, 0, 1)

    g = np.zeros(n)
    # First term: (1-h[i-s]) for i where 0 <= i-s < n, i.e., s <= i < s+n

import numpy as np
import time
import math

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def _get_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _project(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=np.float64)
    lo, hi = h.min() - 2, h.max() + 2
    for _ in range(70):
        mid = (lo + hi) / 2
        p = np.clip(h - mid, 0, 1)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    p = np.clip(h - (lo+hi)/2, 0, 1)
    d = target - p.sum()
    if abs(d) > 1e-12:
        m = (p > 1e-12) & (p < 1-1e-12)
        if m.sum() > 0:
            p[m] += d / m.sum()
            p = np.clip(p, 0, 1)
    return p

def _grad_shift(h, n, s):
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    else:
        sn = -s
        if sn < n:
            g[:n-sn] += (1.0 - h[sn:])
            g[sn:] -= h[:n-sn]
    return g * dx

def _softmax_c5_grad(h, T):
    n = len(h)
    c = _get_c(h)
    cmax = c.max()
    ec = np.exp(T * (c - cmax))
    w = ec / ec.sum()
    g = np.zeros(n)
    for j in range(len(c)):
        if w[j] > 1e-10:
            s = j - (n - 1)
            g += w[j] * _grad_shift(h, n, s)
    obj = cmax + np.log(ec.sum()) / T
    return obj, g

def _optimize(h, n, time_limit, rng, soft_T=50):
    best_h = h.copy()
    best_c5 = _compute_c5(h)
    h = h.copy()
    lr = 0.5
    no_imp = 0
    
    while time.time() < time_limit:
        obj, g = _softmax_c5_grad(h, soft_T)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        direction = -g / gn
        
        # Backtracking
        a = lr
        accepted = False
        for _ in range(25):
            hc = h + a * direction
            hc = _project(hc, n)
            c5c = _compute_c5(hc)
            if c5c < _compute_c5(h) - 1e-12:
                h = hc
                if c5c < best_c5:
                    best_c5 = c5c
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
                accepted = True
                break
            a *= 0.5
            if a < 1e-10:
                break
        if not accepted:
            no_imp += 1
            lr *= 0.5
            if no_imp > 50:
                h = _project(h + 0.15 * rng.standard_normal(n), n)
                no_imp = 0
                lr = 0.5
        else:
            lr = min(lr * 1.05, 3.0)
        
        # Adapt T
        if no_imp % 20 == 0:
            soft_T = max(soft_T * 1.2, 200)
    
    return best_h, best_c5

def _upsample(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
