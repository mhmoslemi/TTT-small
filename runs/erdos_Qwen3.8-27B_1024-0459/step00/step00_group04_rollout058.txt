def smooth_obj_grad(h, n, t, lam):
    dx = 2.0 / n
    # Compute c(s) for s = -(n-1), ..., n-1
    # np.correlate(h, 1-h, 'full') gives output[k] for k=0,...,2n-2
    # where output[k] = c(k-(n-1))
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # corr[k] = c(k-(n-1)) for k = 0, ..., 2n-2
    # So c(s) = corr[s+(n-1)]
    
    c_max = np.max(corr)
    exp_t = np.exp(t * (corr - c_max))
    w = exp_t / np.sum(exp_t)  # w[k] = weight for shift s = k-(n-1)
    
    # Now compute gradient
    # w_shift[s] = w[s+(n-1)] for s = -(n-1), ..., n-1
    # P = Part A, Q = Part B (negative of what we subtract)
    
    grad = np.zeros(n)
    
    # For s = 0 to n-1 (positive shifts in my convention):
    for s in range(0, n):
        ws = w[s + n - 1]
        if ws < 1e-15:
            continue
        # Part A: j from 0 to n-1-s, grad[j] += dx * ws * (1-h[j+s])
        if n - s > 0:
            grad[:n-s] += dx * ws * (1.0 - h[s:])
        # Part B: j from s to n-1, grad[j] += dx * ws * (-h[j-s])
        if n - s > 0:
            grad[s:] += dx * ws * (-h[:n-s])
    
    # For s = -(n-1) to -1 (negative shifts, r = -s from 1 to n-1):
    for r in range(1, n):
        ws = w[-r + n - 1]  # = w[n-1-r]
        if ws < 1e-15:
            continue
        # Part A: j from r to n-1, grad[j] += dx * ws * (1-h[j-r])
        if n - r > 0:
            grad[r:] += dx * ws * (1.0 - h[:n-r])
        # Part B: j from 0 to n-1-r, grad[j] += dx * ws * (-h[j+r])
        if n - r > 0:
            grad[:n-r] += dx * ws * (-h[r:])
    
    # Penalty gradient
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    
    return grad
