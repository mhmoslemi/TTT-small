import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def smooth_objective(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    cmax = np.max(corr)
    shifted = corr - cmax
    return cmax + np.log(np.sum(np.exp(alpha * shifted))) / alpha

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    cmax = np.max(corr)
    weights = np.exp(alpha * (corr - cmax))
    weights /= np.sum(weights)
    # weights has length 2n-1, indexed by s = k-(n-1), k=0..2n-2
    # w[k] corresponds to shift s = k-(n-1)
    
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        for k in range(2*n-1):
            s = k - (n-1)
            # dC_s/dh[j] = dx * [(1-h[j-s]) * 1_{0<=j-s<n} - h[j+s] * 1_{0<=j+s<n}]
            term1 = 0.0
            if 0 <= j - s < n:
                term1 = 1.0 - h[j-s]
            term2 = 0.0
            if 0 <= j + s < n:
                term2 = h[j+s]
            g += weights[k] * dx * (term1 - term2)
        grad[j] = g

    h = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        count = 0
        for i in idx:
            if count >= n_ones - target:
                break
            h[i] = 0
            count += 1
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        count = 0
        for i in idx:
            if count >= target - n_ones:
                break
            h[i] = 1
            count += 1
    
    best_c5 = compute_c5(h, n)
    no_improve = 0
    n_swaps_per_iter = min(30, n // 2)
    
    for it in range(max_iters):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in ones[:n_swaps_per_iter]:
            for j in zeros[:n_swaps_per_iter]:
                trial = h.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                c5 = compute_c5(trial, n)
                if c5 < best_c5 - 1e-12:
                    h = trial
                    best_c5 = c5
                    improved = True
                    break
            if improved:
                break
        no_improve += 1 if not improved else 0
        if no_improve > 200:
            break
    
    return h, best_c5

def hill_climb_binary(h, n, max_iters=20000):
    h = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        for i in range(n_ones - target):
            h[idx[i]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        for i in range(target - n_ones):
            h[idx[i]] = 1
    
    best_c5 = compute_c5(h, n)
    
    for it in range(max_iters):
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            for j in zeros:
                trial = h.copy()
                trial[i] = 0.0
                trial[j] = 1.0
                c5 = compute_c5(trial, n)
                if c5 < best_c5 - 1e-12:
                    h = trial
                    best_c5 = c5
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s
    
    def time_left():
        return deadline - time.time()
    
    best_h = None
    best_c5 = 1.0
    best_n = 70
    
    # Check for initial_h_values
    init_vals = None
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        try:
            init_vals = np.array(initial_h_values, dtype=float)
        except Exception:
            init_vals = None
    
    n_values = [70, 60, 50, 80, 100, 40]
    
    for n in n_values:
        if time_left() < 80:
            break
        
        bounds = [(0.0, 1.0)] * n
        target = n / 2.0
        
        # Initialization
        if n == 70 and init_vals is not None and len(init_vals) == 70:
            h0 = np.clip(init_vals.copy(), 0.0, 1.0)
            h0 = project_constraint(h0, n)
        else:
            # Mixed initialization: try a few patterns
            candidates = []
            # Random
            h_rand = np.random.rand(n)
            h_rand = project_constraint(h_rand, n)
            candidates.append(h_rand)
            # Two blocks
            h_blk = np.zeros(n)
            h_blk[:n//2] = 1.0
            candidates.append(h_blk)
            # Four blocks
            h_4blk = np.zeros(n)
            q = n // 4
            h_4blk[:q] = 1.0
            h_4blk[2*q:3*q] = 1.0
            candidates.append(project_constraint(h_4blk, n))
            # Alternating
            h_alt = np.zeros(n)
            h_alt[0::2] = 1.0
            if int(np.sum(h_alt)) > n // 2:
                h_alt[-1] = 0.0
            candidates.append(h_alt)
            
            # Pick the best candidate
            h0 = min(candidates, key=lambda x: compute_c5(x, n))
        
        # Gradient-based optimization with increasing alpha
        for alpha in [20.0, 40.0, 80.0, 150.0]:
            if time_left() < 30:
                break
            lam = 500.0
            try:
                res = minimize(
                    smooth_max_obj, h0.copy(),
                    jac=smooth_max_grad,
                    args=(n, alpha, lam),
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 150, 'ftol': 1e-12, 'gtol': 1e-7, 'iprint': -1}
                )
                h_opt = project_constraint(res.x, n)
                c5 = compute_c5(h_opt, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n
                h0 = h_opt
            except Exception:
                pass
        
        # Binary local search from the gradient result
        if time_left() < 30:
            break
        
        h_bin = (h0 >= 0.5).astype(float)
        try:
            h_sw, c5_sw = swap_local_search(h_bin, n, max_iters=30000, rng=np.random.RandomState(seed + n))
            if c5_sw < best_c5:
                best_c5 = c5_sw
                best_h = h_sw.copy()
                best_n = n
            h0 = h_sw
        except Exception:
            pass
        
        # Also try binary search from a random starting point
        if time_left() < 30:
            break
        h_rand_bin = np.zeros(n)
        idx = np.random.choice(n, size=n//2, replace=False)
        h_rand_bin[idx] = 1.0
        try:
            h_rb, c5_rb = swap_local_search(h_rand_bin, n, max_iters=15000, rng=np.random.RandomState(seed + n * 7))
            if c5_rb < best_c5:
                best_c5 = c5_rb
                best_h = h_rb.copy()
                best_n = n
        except Exception:
            pass
    
    # Final polishing of the best solution
    if best_h is not None and time_left() > 30:
        n = best_n
        bounds = [(0.0, 1.0)] * n
        for alpha in [200.0, 500.0]:
            if time_left() < 15:
                break
            lam = 5000.0
            try:
                res = minimize(
                    smooth_max_obj, best_h.copy(),
                    jac=smooth_max_grad,
                    args=(n, alpha, lam),
                    method='L-BFGS-B',
                    bounds=bounds,
                    options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-9, 'iprint': -1}
                )
                h_final = project_constraint(res.x, n)
                c5 = compute_c5(h_final, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_final
            except Exception:
                pass
        
        # One more binary polish
        if time_left() > 20:
            h_bin = (best_h >= 0.5).astype(float)
            try:
                h_bp, c5_bp = swap_local_search(h_bin, n, max_iters=20000, rng=np.random.RandomState(seed + 999))
                if c5_bp < best_c5:
                    best_c5 = c5_bp
                    best_h = h_bp
            except Exception:
                pass
    
    # Fallback
    if best_h is None:
        n = 70
        best_h = np.zeros(n)
        best_h[:n//2] = 1.0
        best_n = n
        best_c5 = compute_c5(best_h, n)
    
    # Final validation and constraint enforcement
    best_h = project_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    
    return (best_h, best_c5, best_n)
