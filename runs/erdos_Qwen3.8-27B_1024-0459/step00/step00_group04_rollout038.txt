def compute_overlap(h, dx):
    """Compute the overlap array and C5 value"""
    corr = np.correlate(h, 1-h, mode='full')
    overlaps = corr * dx
    return overlaps, np.max(overlaps)

def project_onto_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection"""
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_overlap(h, n):
    """Compute gradient of max overlap w.r.t. h (subgradient of the max)"""
    # Returns gradient of the active constraint
    corr = np.correlate(h, 1-h, mode='full')
    active = np.argmax(corr)
    s = active - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j + s < n:
            val += 1.0 - h[j + s]
        if 0 <= j - s < n:
            val -= h[j - s]
        grad[j] = val
    return grad, corr
