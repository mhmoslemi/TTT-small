```
def project_sum_box(y, target):
    y = np.asarray(y, dtype=float)
    n = y.size
    if n == 0: return y.copy()
    target = float(target)
    if target <= 0.0: return np.zeros(n)
    if target >= float(n): return np.ones(n)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    # If all y equal? binary search still.
    for _ in range(70):
        mid = 0.5*(lo+hi)
        z = y - mid
        # np.where clips
        s = np.clip(z, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    z = y - 0.5*(lo+hi)
    val = np.clip(z, 0.0, 1.0)
    # fix exact sum
    err = target - val.sum()
    if abs(err) > 1e-12:
        if err > 0:
            order = np.argsort(1.0 - val)[::-1]
        else:
            order = np.argsort(val)[::-1]
        rem = err
        for idx in order:
            if rem > 0:
                room = 1.0 - val[idx]
                add = min(room, rem)
                if add > 0:
                    val[idx] += add; rem -= add
                    if abs(rem) <= 1e-14: break
            else:
                room = val[idx]
                sub = min(room, -rem)
                if sub > 0:
                    val[idx] -= sub; rem += sub
                    if abs(rem) <= 1e-14: break
    return val
```

```python
import numpy as np
import time
import math


def fix_sum_exact(h, target):
    h = np.array(h, dtype=float).copy()
    err = float(target) - float(h.sum())
    if abs(err) < 1e-14:
        return h
    if err > 0:
        order = np.argsort(1.0 - h)[::-1]
        rem = err
        for idx in order:
            room = 1.0 - h[idx]
            if room <= 0.0:
                continue
            add = room if room < rem else rem
            if add > 0.0:
                h[idx] += add
                rem -= add
                if abs(rem) <= 1e-14:
                    break
    else:
        order = np.argsort(h)[::-1]
        rem = -err
        for idx in order:
            room = h[idx]
            if room <= 0.0:
                continue
            sub = room if room < rem else rem
            if sub > 0.0:
                h[idx] -= sub
                rem -= sub
                if rem <= 1e-14:
                    break
    return h


def project_sum_box(y, target):
    y = np.asarray(y, dtype=float)
    n = y.size
    target = float(target)
    if n == 0:
        return y.copy()
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(y - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    val = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    return fix_sum_exact(val, target)


def sanitize_h(h):
    h = np.asarray(h, dtype=float).ravel().copy()
    if h.size == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    target = h.size / 2.0
    diff = target - float(h.sum())
    if abs(diff) > 1e-9:
        h = project_sum_box(h, target)
    else:
        h = fix_sum_exact(h, target)
    return np.clip(h, 0.0, 1.0)


def eval_exact(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return float("inf"), 0.0, 0.0
    g = 1.0 - h
    c = np.correlate(h, g, "full")
    f = float(c.max())
    c5 = f * (2.0 / n)
    return c5, f, float(h.sum())


def resample_h(h, n_new):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.zeros(int(n_new), dtype=float)
    if n == int(n_new):
        return h.copy()

    n_new = int(n_new)
    dx_old = 2.0 / n
    dx_new = 2.0 / n_new
    old_edges = np.arange(n + 1, dtype=float) * dx_old
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = dx_old * np.cumsum(h)

    new_edges = np.arange(n_new + 1, dtype=float) * dx_new
    pref_new = np.interp(new_edges, old_edges, pref, left=pref[0], right=pref[-1])
    new_h = (pref_new[1:] - pref_new[:-1]) / dx_new
    return sanitize_h(new_h)


def buffer_initial(init, n_new):
    init = np.asarray(init, dtype=float).ravel()
    n0 = init.size
    n_new = int(n_new)
    if n0 == n_new:
        return init.copy()
    if n0 > 0 and n_new % n0 == 0:
        h = np.repeat(init, n_new // n0)
        return sanitize_h(h)
    return resample_h(init, n_new)


def make_binary_from_values(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    out = np.zeros(n, dtype=float)
    if n == 0:
        return out
    order = np.argsort(h)[::-1]
    if n % 2 == 0:
        m = n // 2
        if m > 0:
            out[order[:m]] = 1.0
    else:
        m = (n - 1) // 2
        if m > 0:
            out[order[:m]] = 1.0
        if m < n:
            out[order[m]] = 0.5
    return fix_sum_exact(out, n / 2.0)


def make_random_binary(rng, n):
    n = int(n)
    h = np.zeros(n, dtype=float)
    if n % 2 == 0:
        m = n // 2
        h[:m] = 1.0
    else:
        m = (n - 1) // 2
        h[:m] = 1.0
        if m < n:
            h[m] = 0.5
    if n > 1:
        rng.shuffle(h)
    return h


def make_msequence_h(n):
    n = int(n)
    if n <= 0:
        return None
    seq = []
    state = 1
    for _ in range(127):
        seq.append(float(state & 1))
        new_bit = (state & 1) ^ ((state >> 6) & 1)
        state = ((state << 1) & 0x7f) | new_bit

    if n <= 127:
        h = np.asarray(seq[:n], dtype=float)
    else:
        reps = (n + 126) // 127
        h = np.tile(np.asarray(seq, dtype=float), reps)[:n]
    return sanitize_h(h)


def register_best(h, c5, best_c5, best_h, best_n):
    if h is None:
        return best_c5, best_h, best_n
    if not np.isfinite(c5):
        return best_c5, best_h, best_n
    if best_h is None or c5 < best_c5 - 1e-12:
        return float(c5), np.array(h, dtype=float).copy(), int(h.size)
    return best_c5, best_h, best_n


def sub_deadline(deadline, requested_seconds):
    now = time.time()
    if now >= deadline:
        return now
    rem = deadline - now
    req = max(0.0, float(requested_seconds))
    return min(deadline, now + min(req, rem))


def fill_transfer_delta(tmp, h, g_rev, n, i, j, d, k0):
    tmp.fill(0.0)
    if d > 0.0:
        tmp[i:i + n] += d * g_rev
        tmp[k0 - i:k0 - i + n] -= d * h
        tmp[j:j + n] -= d * g_rev
        tmp[k0 - j:k0 - j + n] += d * h
        qq = d * d
    elif d < 0.0:
        nd = -d
        tmp[i:i + n] -= nd * g_rev
        tmp[k0 - i:k0 - i + n] += nd * h
        tmp[j:j + n] += nd * g_rev
        tmp[k0 - j:k0 - j + n] -= nd * h
        qq = nd * nd
    else:
        return
    tmp[k0] -= 2.0 * qq
    tmp[k0 + i - j] += qq
    tmp[k0 + j - i] += qq


def fill_swap_delta(tmp, h, g_rev, n, i, j, k0):
    tmp.fill(0.0)
    tmp[i:i + n] += g_rev
    tmp[k0 - i:k0 - i + n] -= h
    tmp[j:j + n] -= g_rev
    tmp[k0 - j:k0 - j + n] += h
    tmp[k0] -= 2.0
    tmp[k0 + i - j] += 1.0
    tmp[k0 + j - i] += 1.0


def eval_pair_new_f(counts, h, g_rev, n, i, j, d, tmp, sumtmp):
    k0 = n - 1
    absd = abs(d)
    if absd > 1.0 - 1e-12:
        if d > 0.0:
            fill_swap_delta(tmp, h, g_rev, n, i, j, k0)
        else:
            fill_swap_delta(tmp, h, g_rev, n, j, i, k0)
    else:
        fill_transfer_delta(tmp, h, g_rev, n, i, j, d, k0)
    np.add(counts, tmp, out=sumtmp)
    return float(sumtmp.max())


def local_search_swap(h, n_expected, rng, deadline, max_sweeps=1000, best_mode=False, stop_f=0.0):
    h = np.array(h, dtype=float).copy()
    n = h.size
    if n <= 1:
        if n == 1:
            c5, f, _s = eval_exact(h)
            return h, f
        return h, 0.0

    g = 1.0 - h
    g_rev = g[::-1]
    counts = np.correlate(h, g, "full")
    f = float(counts.max())
    if f <= stop_f:
        return h, f

    k0 = n - 1
    length = 2 * n - 1
    tmp = np.zeros(length, dtype=float)
    sumtmp = np.zeros(length, dtype=float)

    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    mp = len(pairs)
    sweep = 0
    check_mask = 1023

    while sweep < max_sweeps:
        if time.time() >= deadline:
            break
        order = rng.permutation(mp)

        if best_mode:
            best_new = f + 1e-12
            best_i = -1
            best_j = -1
            best_d = 0.0

            for pos in range(mp):
                if (pos & check_mask) == 0 and time.time() >= deadline:
                    break
                i, j = pairs[order[pos]]
                d = h[j] - h[i]
                if abs(d) < 1e-12:
                    continue
                new_f = eval_pair_new_f(counts, h, g_rev, n, i, j, d, tmp, sumtmp)
                if new_f < best_new:
                    best_new = new_f
                    best_i = i
                    best_j = j
                    best_d = d

            if time.time() >= deadline:
                break
            if best_i < 0:
                break

            eval_pair_new_f(counts, h, g_rev, n, best_i, best_j, best_d, tmp, sumtmp)
            h[best_i] += best_d
            h[best_j] -= best_d
            g[best_i] -= best_d
            g[best_j] += best_d
            counts += tmp
            f = best_new

            if f <= stop_f:
                break

            g = 1.0 - h
            g_rev = g[::-1]
            counts = np.correlate(h, g, "full")
            f = float(counts.max())
            sweep += 1
            continue

        improved = False
        for pos in range(mp):
            if (pos & check_mask) == 0 and time.time() >= deadline:
                break
            i, j = pairs[order[pos]]
            d = h[j] - h[i]
            if abs(d) < 1e-12:
                continue
            new_f = eval_pair_new_f(counts, h, g_rev, n, i, j, d, tmp, sumtmp)
            if new_f < f - 1e-9:
                h[i] += d
                h[j] -= d
                g[i] -= d
                g[j] += d
                counts += tmp
                f = new_f
                improved = True
                if f <= stop_f:
                    break

        if time.time() >= deadline:
            break
        if not improved:
            break

        g = 1.0 - h
        g_rev = g[::-1]
        counts = np.correlate(h, g, "full")
        f = float(counts.max())
        sweep += 1

    return h, f


def gradient_search(h, deadline, rng, beta=5.0, max_iters=1500):
    h = sanitize_h(h)
    n = h.size
    if n <= 1 or time.time() >= deadline:
        return h, float(np.max(np.correlate(h, 1.0 - h, "full"))) if n > 0 else 0.0

    target = n / 2.0
    g = 1.0 - h
    counts = np.correlate(h, g, "full")
    f = float(counts.max())
    grad = np.zeros(n, dtype=float)
    beta = max(1.0, min(float(beta), 80.0))
    stag = 0
    k0 = n - 1

    for _it in range(max_iters):
        if time.time() >= deadline:
            break
        if f <= 0.0:
            break

        z = (counts - f) * beta
        z -= float(z.max())
        w = np.exp(z)
        sw = float(w.sum())
        if sw < 1e-15:
            break
        w /= sw

        grad.fill(0.0)
        for idx, wk in enumerate(w):
            if wk < 1e-8:
                continue
            kk = idx - k0
            if kk == 0:
                grad += wk * (g - h)
            elif kk > 0:
                grad[kk:] += wk * g[:-kk]
                grad[:n - kk] -= wk * h[kk:]
            else:
                p = -kk
                grad[:n - p] += wk * g[p:]
                grad[p:] -= wk * h[:-p]

        gm = float(np.max(np.abs(grad))) if n else 0.0
        if gm < 1e-12:
            beta = min(80.0, beta * 1.5)
            if beta > 80.0:
                break
            continue

        base = 0.1 * n / (gm + 1e-12)
        improved = False
        for fac in (0.05, 0.15, 0.45, 1.35):
            step = base * fac
            y = h - step * grad
            h2 = project_sum_box(y, target)
            g2 = 1.0 - h2
            c2 = np.correlate(h2, g2, "full")
            f2 = float(c2.max())
            if f2 + 1e-10 < f:
                h = h2
                g = g2
                counts = c2
                f = f2
                improved = True
                stag = 0
                break

        if not improved:
            stag += 1
            beta = min(80.0, beta * 1.08)
            if stag > 30 or beta > 80.0 and stag > 10:
                break
    return h, f


def anneal_binary(h, n_expected, rng, deadline, start_T=2.0, min_T=0.25, max_moves=8000000):
    h = np.array(h, dtype=float).copy()
    n = h.size
    if n == 0:
        return h, 0.0
    if n % 2 != 0:
        _c5, f, _s = eval_exact(sanitize_h(h))
        return sanitize_h(h), f

    m = n // 2
    h = (h >= 0.5).astype(float)
    if int(round(float(h.sum()))) != m:
        order = np.argsort(h)[::-1]
        h.fill(0.0)
        h[order[:m]] = 1.0

    g = 1.0 - h
    g_rev = g[::-1]
    counts = np.correlate(h, g, "full")
    f = float(counts.max())

    best_h = h.copy()
    best_f = f
    k0 = n - 1
    length = 2 * n - 1
    tmp = np.zeros(length, dtype=float)
    sumtmp = np.zeros(length, dtype=float)

    T = float(start_T)
    cool = 0.9995
    moves_since_best = 0
    accepted = 0
    check_mask = 1023

    for it in range(int(max_moves)):
        if (it & check_mask) == 0 and time.time() >= deadline:
            break

        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if a == b or h[a] == h[b]:
            continue

        if h[a] > h[b]:
            i = b
            j = a
        else:
            i = a
            j = b

        fill_swap_delta(tmp, h, g_rev, n, i, j, k0)
        np.add(counts, tmp, out=sumtmp)
        new_f = float(sumtmp.max())

        accept = False
        if new_f < f - 1e-12:
            accept = True
        else:
            delta = new_f - f
            if delta <= 20.0 * T:
                accept = (rng.random() < math.exp(-delta / T))

        if accept:
            h[i] = 1.0
            h[j] = 0.0
            g[i] = 0.0
            g[j] = 1.0
            counts += tmp
            f = new_f
            accepted += 1

            if f < best_f - 1e-12:
                best_f = f
                best_h = h.copy()
                moves_since_best = 0
            else:
                moves_since_best += 1

            if accepted % 5000 == 0:
                g = 1.0 - h
                g_rev = g[::-1]
                counts = np.correlate(h, g, "full")
                f = float(counts.max())
                if f < best_f - 1e-12:
                    best_f = f
                    best_h = h.copy()
                    moves_since_best = 0
        else:
            moves_since_best += 1

        T *= cool
        if T < min_T:
            T = min_T
            if moves_since_best > 30000:
                h = best_h.copy()
                g = 1.0 - h
                g_rev = g[::-1]
                counts = np.correlate(h, g, "full")
                f = float(counts.max())
                if f > best_f:
                    f = best_f
                T = float(start_T)
                moves_since_best = 0

    return best_h, best_f


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(max(1.0, budget_s))
    deadline = t0 + 0.985 * budget
    scale = max(1e-6, budget / 1300.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c5 = float("inf")
    best_n = None

    try:
        init_vals = initial_h_values
    except NameError:
        init_vals = None

    init_h = None
    if init_vals is not None:
        h0 = np.asarray(init_vals, dtype=float).ravel().copy()
        if h0.size > 0:
            h0 = sanitize_h(h0)
            c0, _f0, _s0 = eval_exact(h0)
            if np.isfinite(c0):
                best_c5, best_h, best_n = register_best(h0, c0, best_c5, best_h, best_n)
                init_h = h0

    if init_h is None:
        # Fallback if no parent construction is supplied.
        fb_n = 128 if budget > 300 else 94
        hm = make_msequence_h(fb_n)
        if hm is not None:
            cm, _fm, _sm = eval_exact(sanitize_h(hm))
            best_c5, best_h, best_n = register_best(hm, cm, best_c5, best_h, best_n)
            init_h = sanitize_h(hm)
        if init_h is None:
            hr = make_random_binary(rng, fb_n)
            cr, _fr, _sr = eval_exact(hr)
            best_c5, best_h, best_n = register_best(hr, cr, best_c5, best_h, best_n)
            init_h = hr

    parent_n = int(init_h.size) if init_h is not None else -1

    # Local polish at the parent resolution, if it exists.
    if init_h is not None and time.time() < deadline:
        req = 110.0 * scale
        dl = sub_deadline(deadline, req)
        if dl > time.time() + 0.5:
            hp1, _fp1 = local_search_swap(init_h, parent_n, rng, dl, max_sweeps=300, best_mode=False)
            hp1 = sanitize_h(hp1)
            c1, _f1, _s1 = eval_exact(hp1)
            best_c5, best_h, best_n = register_best(hp1, c1, best_c5, best_h, best_n)

            reqg = 45.0 * scale
            dlg = sub_deadline(deadline, reqg)
            if dlg > time.time() + 0.5:
                hg1, _fg1 = gradient_search(hp1, dlg, rng, beta=5.0, max_iters=800)
                hg1 = sanitize_h(hg1)
                cg1, _fgg1, _sg1 = eval_exact(hg1)
                best_c5, best_h, best_n = register_best(hg1, cg1, best_c5, best_h, best_n)

    # Candidate resolutions: exact multiples of the parent plus a useful nearby even size.
    cand_ns = []
    if init_h is not None:
        for mult in (2, 4):
            N = int(parent_n * mult)
            if 24 <= N <= 210:
                cand_ns.append(N)
    for N in (94, 128):
        if 24 <= N <= 210:
            cand_ns.append(N)

    unique_cands = []
    for N in sorted(set(cand_ns)):
        if N == parent_n:
            continue
        unique_cands.append(int(N))

    for N in unique_cands:
        if deadline - time.time() < 45.0 * scale:
            break

        # Base continuous candidate.
        if init_h is not None:
            h_cont = buffer_initial(init_h, N)
        elif best_h is not None and best_h.size > 0:
            h_cont = resample_h(best_h, N)
        else:
            h_cont = project_sum_box(rng.random(N), N / 2.0)

        h_cont = sanitize_h(h_cont)
        cc, _fc, _sc = eval_exact(h_cont)
        best_c5, best_h, best_n = register_best(h_cont, cc, best_c5, best_h, best_n)

        if N > 150:
            req_cont = 18.0 * scale
            req_bin = 70.0 * scale
            req_an = 45.0 * scale
            req_pol = 18.0 * scale
            req_rand = 20.0 * scale
            req_rand_an = 28.0 * scale
            req_rand_pol = 14.0 * scale
        else:
            req_cont = 30.0 * scale
            req_bin = 90.0 * scale
            req_an = 70.0 * scale
            req_pol = 30.0 * scale
            req_rand = 42.0 * scale
            req_rand_an = 50.0 * scale
            req_rand_pol = 26.0 * scale

        # Continuous value-swap local search.
        dl = sub_deadline(deadline, req_cont)
        h_work = h_cont
        if dl > time.time() + 0.5:
            h_work, _fw = local_search_swap(h_work, N, rng, dl, max_sweeps=250, best_mode=False)
            h_work = sanitize_h(h_work)
            cw, _fw2, _sw = eval_exact(h_work)
            best_c5, best_h, best_n = register_best(h_work, cw, best_c5, best_h, best_n)

        # Binary threshold from the current continuous arrangement.
        h_bin = make_binary_from_values(h_work)
        cb, _fb, _sb = eval_exact(h_bin)
        best_c5, best_h, best_n = register_best(h_bin, cb, best_c5, best_h, best_n)

        h_bin_work = h_bin
        dl = sub_deadline(deadline, req_bin)
        if dl > time.time() + 0.5:
            h_bin_work, _fbin = local_search_swap(h_bin_work, N, rng, dl, max_sweeps=700, best_mode=False)
            h_bin_work = sanitize_h(h_bin_work)
            cbin, _fbin2, _sbin = eval_exact(h_bin_work)
            best_c5, best_h, best_n = register_best(h_bin_work, cbin, best_c5, best_h, best_n)

        if N % 2 == 0:
            dl = sub_deadline(deadline, req_an)
            h_anneal = h_bin_work
            if dl > time.time() + 0.5:
                h_anneal, _fan = anneal_binary(h_bin_work, N, rng, dl)
                h_anneal = sanitize_h(h_anneal)
                can, _fan2, _san = eval_exact(h_anneal)
                best_c5, best_h, best_n = register_best(h_anneal, can, best_c5, best_h, best_n)

            dlp = sub_deadline(deadline, req_pol)
            if dlp > time.time() + 0.5:
                h_pol, _fpol = local_search_swap(h_anneal, N, rng, dlp, max_sweeps=120, best_mode=True)
                h_pol = sanitize_h(h_pol)
                cpol, _fpol2, _spol = eval_exact(h_pol)
                best_c5, best_h, best_n = register_best(h_pol, cpol, best_c5, best_h, best_n)

        # Random restart for this resolution.
        if deadline - time.time() > 35.0 * scale:
            hrnd = make_random_binary(rng, N)
            crnd, _frnd, _srnd = eval_exact(hrnd)
            best_c5, best_h, best_n = register_best(hrnd, crnd, best_c5, best_h, best_n)

            dl = sub_deadline(deadline, req_rand)
            h_rnd_work = hrnd
            if dl > time.time() + 0.5:
                h_rnd_work, _frnd2 = local_search_swap(hrnd, N, rng, dl, max_sweeps=250, best_mode=False)
                h_rnd_work = sanitize_h(h_rnd_work)
                crnd2, _frnd3, _srnd2 = eval_exact(h_rnd_work)
                best_c5, best_h, best_n = register_best(h_rnd_work, crnd2, best_c5, best_h, best_n)

            if N % 2 == 0 and deadline - time.time() > 25.0 * scale:
                dlr = sub_deadline(deadline, req_rand_an)
                h_rnd_an = h_rnd_work
                if dlr > time.time() + 0.5:
                    h_rnd_an, _frnd4 = anneal_binary(h_rnd_work, N, rng, dlr)
                    h_rnd_an = sanitize_h(h_rnd_an)
                    crnd4, _frnd5, _srnd3 = eval_exact(h_rnd_an)
                    best_c5, best_h, best_n = register_best(h_rnd_an, crnd4, best_c5, best_h, best_n)

                    dlrp = sub_deadline(deadline, req_rand_pol)
                    if dlrp > time.time() + 0.5:
                        h_rnd_pol, _frnd6 = local_search_swap(h_rnd_an, N, rng, dlrp,
                                                             max_sweeps=80, best_mode=True)
                        h_rnd_pol = sanitize_h(h_rnd_pol)
                        crnd6, _frnd7, _srnd4 = eval_exact(h_rnd_pol)
                        best_c5, best_h, best_n = register_best(h_rnd_pol, crnd6,
                                                                 best_c5, best_h, best_n)

    if best_h is None:
        best_h = make_random_binary(rng, 94)
        best_c5, _f, _s = eval_exact(best_h)
        best_n = len(best_h)

    best_h = sanitize_h(best_h)
    final_c5, _final_f, _final_s = eval_exact(best_h)
    final_c5 = float(final_c5)
    best_n = int(best_h.size)
    return best_h, final_c5, best_n
```
