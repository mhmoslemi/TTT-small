import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_constraint(h, n_points):
    target = n_points / 2.0
    lo, hi = h.min() - 1.5, h.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def upsample(h, n_from, n_to):
    """Upsample h from n_from to n_to points using linear interpolation."""
    x_from = np.linspace(0, 2, n_from, endpoint=False)
    x_to = np.linspace(0, 2, n_to, endpoint=False)
    h_new = np.interp(x_to, x_from, h)
    h_new = np.clip(h_new, 0, 1)
    h_new = project_onto_constraint(h_new, n_to)
    return h_new

def coordinate_descent(h, n_points, max_sweeps=100, time_limit=None, start_time=None):
    """Coordinate descent to minimize C5."""
    h = h.copy()
    h = project_onto_constraint(h, n_points)
    best_h = h.copy()
    best_obj = compute_c5(h, n_points)
    
    dx = 2.0 / n_points
    L_values = np.arange(2*n_points-1) - (n_points-1)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        # Shuffle coordinates for stochastic CD
        order = np.random.permutation(n_points)
        
        for j in order:
            # Compute a and b for coordinate j
            corr = np.correlate(h, 1-h, mode="full")
            
            a = np.zeros(2*n_points-1)
            for idx, L in enumerate(L_values):
                coeff = 0.0
                if 0 <= j+L < n_points:
                    coeff += (1.0 - h[j+L])
                if 0 <= j-L < n_points:
                    coeff -= h[j-L]
                a[idx] = coeff
            
            b = corr - h[j] * a
            
            # Minimize max(a*x + b) for x in [0,1] via golden section
            def f_1d(x):
                return np.max(a * x + b)
            
            phi = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, 1.0
            x1 = hi - phi * (hi - lo)
            x2 = lo + phi * (hi - lo)
            f1, f2 = f_1d(x1), f_1d(x2)
            for _ in range(40):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - phi * (hi - lo)
                    f1 = f_1d(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + phi * (hi - lo)
                    f2 = f_1d(x2)
            
            h[j] = (lo + hi) / 2.0
        
        # Project onto constraint
        h = project_onto_constraint(h, n_points)
        obj = compute_c5(h, n_points)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    
    return best_h, best_obj

One optimization: instead of recomputing the full correlation for each coordinate, I can update it incrementally. When h[j] changes from old to new, only the correlation values that involve h[j] change.

c[m] = ∑_i h[i] * (1-h[i+L])

The terms involving h[j] are:
- h[j] * (1-h[j+L]) for the term where i=j (if valid)
- h[j-L] * (1-h[j]) for the term where i=j-L (if valid)

So when h[j] changes by δ:
Δc[m] = δ * (1-h[j+L_m]) [valid] + h[j-L_m] * (-δ) [valid]
= δ * (1-h[j+L_m]) [0≤j+L_m<n] - δ * h[j-L_m] [0≤j-L_m<n]

This is O(1) per m, so O(n) total per update. Much faster than O(n²) for recomputing the full correlation!

def coordinate_descent_fast(h, n_points, max_sweeps=200, time_limit=None, start_time=None):
    h = project_onto_constraint(h.copy(), n_points)
    best_h = h.copy()
    best_obj = compute_c5(h, n_points)
    
    dx = 2.0 / n_points
    n_lags = 2 * n_points - 1
    L_values = np.arange(n_lags) - (n_points - 1)
    
    # Precompute for each j, which lags involve h[j] and with what coefficient
    # For coordinate j and lag L:
    #   coeff = (1-h[j+L])*[0<=j+L<n] - h[j-L]*[0<=j-L<n]
    
    # Current correlation values
    corr = np.correlate(h, 1-h, mode="full").copy()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        order = np.random.permutation(n_points)
        
        for j in order:
            # a[m] = coefficient of h[j] in c[m]
            # a[m] = (1-h[j+L_m])*[0<=j+L_m<n] - h[j-L_m]*[0<=j-L_m<n]
            a = np.zeros(n_lags)
            for idx, L in enumerate(L_values):
                c = 0.0
                if 0 <= j + L < n_points:
                    c += (1.0 - h[j + L])
                if 0 <= j - L < n_points:
                    c -= h[j - L]
                a[idx] = c
            
            # b[m] = c[m] - h[j] * a[m]
            b = corr - h[j] * a
            
            # Find x* in [0,1] minimizing max(a*x + b)
            # Binary search / golden section
            def f_1d(x):
                return np.max(a * x + b)
            
            # Since f is convex and piecewise linear, use bisection on subgradient
            # Or just golden section (40 iterations gives precision ~ (0.618)^40 ≈ 1e-9)
            phi = (np.sqrt(5.0) - 1.0) / 2.0
            lo, hi = 0.0, 1.0
            x1 = hi - phi * (hi - lo)
            x2 = lo + phi * (hi - lo)
            f1, f2 = f_1d(x1), f_1d(x2)
            for _ in range(40):
                if f1 < f2:
                    hi = x2; x2 = x1; f2 = f1

                else:
                    lo = x1; x1 = x2; f1 = f2
                    x2 = lo + phi * (hi - lo); f2 = f_1d(x2)

            delta = new_val - h[j]

                h[j] = new_val
                # Update corr incrementally
                # Δc[m] = delta * a[m] (since a[m] is the coefficient of h[j] in c[m])
                corr += delta * a
        
        h = project_onto_constraint(h, n_points)
        obj = compute_c5(h, n_points)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    
    return best_h, best_obj

import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr)) * dx

def project_onto_constraint(h, n_points):
    target = n_points / 2.0
    # Bisection to find tau
    lo = h.min() - 1.5
    hi = h.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def upsample_linear(h, n_to):
    n_from = len(h)
    x_from = np.arange(n_from, dtype=float)
    x_to = np.arange(n_to, dtype=float) * (n_from / n_to)
    h_new = np.interp(x_to, x_from, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_onto_constraint(h_new, n_to)
    return h_new

def upsample_replicate(h, n_to):
    n_from = len(h)
    ratio = n_to / n_from
    h_new = np.zeros(n_to)
    for i in range(n_from):
        start = int(i * ratio)
        end = int((i + 1) * ratio)
        h_new[start:end] = h[i]
    # Fill any gaps
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_onto_constraint(h_new, n_to)
    return h_new

def coordinate_descent(h, n_points, max_sweeps=500, time_budget=60.0, start_time=None, perturbation=0.0):
    """Coordinate descent with optional perturbation for escaping local optima."""
    h = project_onto_constraint(h.copy(), n_points)
    best_h = h.copy()
    best_obj = compute_c5(h, n_points)
    no_improve_count = 0
    
    dx = 2.0 / n_points
    n_lags = 2 * n_points - 1
    L_values = np.arange(n_lags) - (n_points - 1)
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    
    corr = np.correlate(h, 1 - h, mode="full").astype(float).copy()
    
    for sweep in range(max_sweeps):
        if start_time is not None and time.time() - start_time > time_budget:
            break
        
        order = np.random.permutation(n_points)
        
        for j in order:
            # Compute a[m] = coefficient of h[j] in c[m]
            a = np.zeros(n_lags)
            hj = h[j]
            for idx in range(n_lags):
                L = L_values[idx]
                c_val = 0.0
                jp = j + L
                jm = j - L
                if 0 <= jp < n_points:
                    c_val += (1.0 - h[jp])
                if 0 <= jm < n_points:
                    c_val -= h[jm]
                a[idx] = c_val
            
            b = corr - hj * a
            
            # Golden section search for x in [0,1] minimizing max(a*x + b)
            lo_g, hi_g = 0.0, 1.0
            x1_g = hi_g - phi * (hi_g - lo_g)

    """Compute a[m] = coefficient of h[j] in c[m] for all m."""

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining < 20:
            break

    # Phase 5: Final gradient polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        h_opt, c5_opt = gradient_polish(best_h, len(best_h),
                                        time_budget=min(remaining * 0.5, 60.0),
                                        start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: One more CD pass on the best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10.0:
        h_opt, c5_opt = coordinate_descent(best_h, len(best_h), max_sweeps=1000,
                                           time_budget=min(remaining * 0.8, 120.0),
                                           start_time=start_time,
                                           perturbation=0.0, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Ensure feasibility
    best_h = project_onto_constraint(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
