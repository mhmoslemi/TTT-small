import time

import numpy as np


def c5_values(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c5(h):
    h = np.asarray(h, dtype=float)
    return float(np.max(c5_values(h)))


def project_capped_simplex(v, target):
    v = np.asarray(v, dtype=float)
    n = v.size
    target = float(target)
    if n == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=float)
    if target >= float(n):
        return np.ones_like(v, dtype=float)

    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    z = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(z.sum())

    if abs(diff) > 1e-14:
        interior = (z > 1e-12) & (z < 1.0 - 1e-12)
        if interior.any():
            z = z.copy()
            z[interior] += diff / float(interior.sum())
            z = np.clip(z, 0.0, 1.0)
            diff2 = target - float(z.sum())
            if abs(diff2) > 1e-12:
                interior2 = (z > 1e-12) & (z < 1.0 - 1e-12)
                if interior2.any():
                    z[interior2] += diff2 / float(interior2.sum())
                    z = np.clip(z, 0.0, 1.0)
        else:
            z = z.copy()
            if diff > 0.0:
                idx = int(np.argmax(z))
                room = 1.0 - z[idx]
                if room > 1e-12:
                    z[idx] += min(diff, room)
            else:
                idx = int(np.argmin(z))
                room = z[idx]
                if room > 1e-12:
                    z[idx] -= min(-diff, room)
    return z


def round_to_set(h, n):
    h = np.asarray(h, dtype=float)
    out = np.zeros(int(n), dtype=float)

    if n % 2 == 0:
        k = n // 2
        if k == 0:
            return out
        if k == n:
            out.fill(1.0)
            return out
        idx = np.argpartition(h, n - k)[n - k:]
        out[idx] = 1.0
    else:
        k = (n - 1) // 2
        if k > 0:
            idx = np.argpartition(h, n - k)[n - k:]
            out[idx] = 1.0
        rem = np.where(out == 0.0)[0]
        if rem.size > 0:
            loc = int(np.argmax(h[rem]))
            out[rem[loc]] = 0.5
    return out


def raw_weighted_gradient(h, weights):
    n = h.size
    center = n - 1
    m = np.arange(n)

    cum = np.empty(weights.size + 1, dtype=float)
    cum[0] = 0.0
    cum[1:] = np.cumsum(weights)

    t1 = cum[center + m + 1] - cum[m]
    conv_a = np.convolve(h, weights, mode="full")
    conv_b = np.convolve(h, weights[::-1], mode="full")
    grad = t1 - conv_a[center:center + n] - conv_b[center:center + n]
    return grad


def softmax_lag_weights(c, beta):
    cmax = float(np.max(c))
    z = beta * (c - cmax)
    ex = np.exp(z)
    s = float(np.sum(ex))
    if not math.isfinite(s) or s <= 0.0:
        active = (c >= cmax - 1e-8)
        if not active.any():
            active = (c == cmax)
        w = np.zeros_like(c, dtype=float)
        w[active] = 1.0 / float(active.sum())
        return w
    return ex / s


def active_lag_weights(c, alpha, cmax):
    if alpha <= 1e-12:
        active = (c >= cmax - 1e-6)
        if not active.any():
            active = (c == cmax)
        w = np.zeros_like(c, dtype=float)
        w[active] = 1.0 / float(active.sum())
        return w

    z = (c - cmax) / alpha
    ex = np.exp(z)
    s = float(np.sum(ex))
    if not math.isfinite(s) or s <= 0.0:
        tol = max(1e-6, 2.0 * alpha)
        active = (c >= cmax - tol)
        if not active.any():
            active = (c == cmax)
        w = np.zeros_like(c, dtype=float)
        w[active] = 1.0 / float(active.sum())
        return w
    return ex / s

def make_block_binary(n, rng):
    blocks = int(rng.integers(2, 7))
    p = max(1, n // (2 * blocks))
    pat_len = 2 * p

def make_wave_seed(n, f, phase, scale):

def make_quadratic_seed(n, rng):
    best_h = make_random_binary(n, rng)
    best_c = exact_c5(best_h)
    n_one = n // 2 if n % 2 == 0 else (n - 1) // 2

    mods = [max(3, int(0.7 * n)), max(3, int(0.9 * n)), max(3, int(1.1 * n)), max(3, int(1.3 * n))]
    idx = np.arange(n)

        noise = 1e-9 * rng.random(n)

        c = exact_c5(h)
        if c < best_c:

            best_c = c
    return best_h


def resample_feasible(h, n):
    h = np.asarray(h, dtype=float)
    m = h.size

    vals = np.interp(x_new, x_old, h, left=h[0], right=h[-1])

                    no_improve = 0
                    step = base_step

    return best_h, float(best_obj)


def seed_eval_key(item):
    return item[0]


def run(seed=42, budget_s=1300, **kwargs):
    budget = max(1.0, float(budget_s))
    start = time.time()
    leave = min(5.0, 0.1 * budget)
    hard_deadline = start + budget - leave
    rng = np.random.default_rng(int(seed))

    best_h = None
    best_c = 1e99
    best_n = 0

    try:
        init = np.asarray(initial_h_values, dtype=float)
        if init.size > 0:
            init = project_capped_simplex(init, init.size / 2.0)
            cinit = exact_c5(init)
            if cinit < best_c:
                best_h = init.copy()
                best_c = cinit
                best_n = int(init.size)
        else:
            init = None
    except Exception:
        init = None

    if time.time() >= hard_deadline - 0.5:
        if best_h is None:
            best_n = 60
            best_h = make_random_binary(best_n, rng)
            best_c = exact_c5(best_h)
        best_h = project_capped_simplex(best_h, best_n / 2.0)
        return best_h, float(best_c), best_n

    n_specs = []
    if init is not None:
        n_specs.append((int(init.size), 1.8))
    for nn, ww in [(64, 0.7), (90, 1.4), (128, 1.7), (256, 1.2), (384, 0.7)]:
        present = False
        for ex_n, _ in n_specs:
            if ex_n == nn:
                present = True
                break
        if not present:
            n_specs.append((nn, ww))

    total_w = 0.0
    for _, ww in n_specs:
        total_w += ww
    if total_w <= 0.0:
        n_specs = [(90, 1.0)]
        total_w = 1.0

    alloc_total = max(5.0, min(hard_deadline - time.time(), 0.92 * budget))
    remaining_alloc = alloc_total

    for nn, weight in n_specs:
        if time.time() >= hard_deadline - 2.0 or remaining_alloc <= 2.0:
            break
        limit = alloc_total * (weight / total_w)
        if limit > remaining_alloc:
            limit = remaining_alloc
        limit = max(8.0, limit)

        seeds = []
        if init is not None:
            seeds.append(resample_feasible(init, nn))

        seeds.append(make_random_binary(nn, rng))
        seeds.append(make_block_binary(nn, rng))
        seeds.append(make_thue_morse_seed(nn, rng))
        if nn <= 300:
            seeds.append(make_quadratic_seed(nn, rng))

        phases = [0.0, 1.3, 3.7]
        for ph in phases:
            phase = float(ph) + float(rng.uniform(-0.3, 0.3))
            seeds.append(make_wave_seed(nn, 2.0, phase, 1.0))
            seeds.append(make_wave_seed(nn, 3.0, phase, 0.85))

        mid_f = max(2, nn // 8)
        phase = float(rng.uniform(0.0, 2.0 * math.pi))
        seeds.append(make_wave_seed(nn, float(mid_f), phase, 1.0))

        info = []
        for sh in seeds:
            sh = project_capped_simplex(sh, nn / 2.0)
            sc = exact_c5(sh)
            info.append((sc, len(info)))

        info.sort(key=seed_eval_key)

        if limit < 35.0:
            max_choice = 1
        elif limit < 110.0:
            max_choice = 2
        else:
            max_choice = 3

        used_for_n = 0.0
        for rank in range(min(max_choice, len(info))):
            if time.time() >= hard_deadline - 2.0:
                break
            _, idx = info[rank]
            if rank == 0:
                seed_limit = limit * 0.55
            elif rank == 1:
                seed_limit = limit * 0.30
            else:
                seed_limit = limit * 0.15
            seed_limit = max(5.0, seed_limit)
            if seed_limit > hard_deadline - time.time():
                seed_limit = hard_deadline - time.time() - 0.5
            if seed_limit <= 0.0:
                break

            hopt, copt = optimize_seed(seeds[idx], seed_limit, hard_deadline, rng)
            if copt < best_c:
                best_h = hopt.copy()
                best_c = copt
                best_n = int(hopt.size)

            used_for_n += seed_limit
            if used_for_n >= limit - 2.0:
                break

        used_for_n = min(limit, used_for_n)
        remaining_alloc -= used_for_n

    if best_h is None:
        best_n = 90
        best_h = make_random_binary(best_n, rng)
        best_c = exact_c5(best_h)

    if time.time() < hard_deadline - 15.0:
        hr = round_to_set(best_h, best_n)
        cr = exact_c5(hr)
        if cr <= best_c + 1e-7:
            best_h = hr
            best_c = cr

        for eps in (0.03, 0.08, 0.15):
            hp = perturb_feasible(best_h, best_n / 2.0, eps, rng)
            cp = exact_c5(hp)
            if cp < best_c - 1e-12:
                best_h = hp
                best_c = cp

        double_n = best_n * 2
        if 0 < double_n <= 600 and time.time() < hard_deadline - 60.0:
            hr2 = resample_feasible(best_h, double_n)
            cr2 = exact_c5(hr2)
            if cr2 < best_c or time.time() < hard_deadline - 120.0:
                lim = max(10.0, min(180.0, hard_deadline - time.time() - 10.0))
                hopt, copt = optimize_seed(hr2, lim, hard_deadline, rng)
                if copt < best_c:
                    best_h = hopt.copy()
                    best_c = copt
                    best_n = int(hopt.size)

        rem = hard_deadline - time.time()
        if rem > 30.0:
            hopt, copt = optimize_seed(best_h, max(10.0, rem - 5.0), hard_deadline, rng)
            if copt < best_c:
                best_h = hopt.copy()
                best_c = copt
                best_n = int(hopt.size)

    best_h = project_capped_simplex(best_h, best_n / 2.0)

    hr = round_to_set(best_h, best_n)
    cr = exact_c5(hr)
    if cr <= best_c + 1e-7:
        best_h = hr
        best_c = cr

    best_c = exact_c5(best_h)
    best_h = project_capped_simplex(best_h, best_n / 2.0)
    best_c = exact_c5(best_h)

    return best_h, float(best_c), int(best_n)
