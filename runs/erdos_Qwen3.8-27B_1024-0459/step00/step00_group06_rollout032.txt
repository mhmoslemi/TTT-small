import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    h = np.strip(h) if hasattr(np, 'strip') else h
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room < 1e-14:
                break
            h = h + diff * room / total_room
        else:
            room = h.copy()
            total_room = np.sum(room)
            if total_room < 1e-14:
                break
            h = h + diff * room / total_room
    return np.clip(h, 0.0, 1.0)

def penalized_obj(h, n_points, target_sum, penalty_coeff):
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    s_err = np.sum(h) - target_sum
    return c5 + penalty_coeff * s_err * s_err

def h_from_harmonics(params, n_points):
    M = len(params) // 2
    x = np.linspace(0.0, 2.0, n_points, endpoint=False)
    h = np.full(n_points, 0.5)
    for m in range(1, M + 1):
        h += params[2 * (m - 1)] * np.cos(np.pi * m * x)
        h += params[2 * (m - 1) + 1] * np.sin(np.pi * m * x)
    return h

def gradient_of_c5(h):
    n = len(h)
    dx = 2.0 / n
    one_h = 1.0 - h
    corr = np.correlate(h, one_h, mode='full')
    max_val = np.max(corr)
    # Subgradient: average over all max-achieving lags
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            for i in range(n - k):
                grad[i] += (1.0 - h[i + k]) - h[i + k] * 0  # dh[i]/dh[i] = 1-h[i+k]
                if i + k < n:
                    grad[i + k] -= h[i] * 1.0  # dh[i+k] in (1-h[i+k]) gives -h[i]
        else:
            kk = -k
            for i in range(n - kk):
                grad[i + kk] += (1.0 - h[i])  # h[i+kk] * (1-h[i]), dh[i+kk] = 1-h[i]
                grad[i] -= h[i + kk]  # (1-h[i]), d(1-h[i])/dh[i] = -1, times h[i+kk]
    return dx * grad / len(active)

def frank_wolfe_step(grad, target_sum, n):
    c = -grad
    sorted_idx = np.argsort(c)
    v = np.zeros(n)
    remaining = target_sum
    for i in sorted_idx:
        if remaining <= 1e-12:
            break
        if remaining >= 1.0:
            v[i] = 1.0
            remaining -= 1.0
        else:
            v[i] = remaining
            remaining = 0.0
    return v

def projected_grad_descent(h0, n, target_sum, lr, max_iter, tol=1e-10):
    h = h0.copy()

        # Gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, target_sum)
        if np.max(np.abs(h_new - h)) < tol:
            h = h_new
            break
        # Simple line search
        for _ in range(10):
            obj_new = compute_c5(h_new)
            obj_old = compute_c5(h)
            if obj_new <= obj_old + 1e-12:
                break
            h_new = 0.5 * (h + h_new)
            h_new = project_sum(h_new, target_sum)
        h = h_new
    return h

    t = np.zeros(n, dtype=float)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t
