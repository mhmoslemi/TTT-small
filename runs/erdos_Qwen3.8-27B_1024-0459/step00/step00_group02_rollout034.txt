```python
def gradient_c5(h, n):
    """Compute gradient of max_k c[k] w.r.t. h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find all lags that achieve the max (subgradient)
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    
    # corr has length 2n-1, indexed 0 to 2n-2
    # corr[k] corresponds to lag k-(n-1)
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # actual lag
        for j in range(n):
            # ∂c[k]/∂h[j] = (1-h[j-k]) * [0≤j-k<n] - h[j+k] * [0≤j+k<n]
            g = 0.0
            jk = j - k
            if 0 <= jk < n:
                g += 1.0 - h[jk]
            jk2 = j + k
            if 0 <= jk2 < n:
                g -= h[jk2]
            grad[j] += g / len(active)  # average over active lags (subgradient)
    
    return grad
```

```python
def pgd_step(h, lr, n, target_sum):
    grad = gradient_c5(h, n)
    h_new = h - lr * grad
    # Project: clip to [0,1] and adjust sum
    h_new = np.clip(h_new, 0, 1)
    diff = target_sum - h_new.sum()
    h_new += diff / n
    h_new = np.clip(h_new, 0, 1)
    # Second-order correction for sum
    residual = target_sum - h_new.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h_new)  # most room first
        else:
            idx = np.argsort(h_new)  # least first
        h_new[idx[0]] += residual
    return h_new
```

```python
import numpy as np
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_max_corr(h, n):
    corr = compute_corr(h)
    max_val = np.max(corr)
    tol = 1e-8
    active = np.where(np.abs(corr - max_val) <= tol)[0]
    grad = np.zeros(n)
    n_active = len(active)
    for k_idx in active:
        k = k_idx - (n - 1)
        for j in range(n):
            g = 0.0
            jm = j - k
            if 0 <= jm < n:
                g += 1.0 - h[jm]
            jp = j + k
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += g
    grad /= n_active
    return grad

def project_feasible(h, n, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12:
        h += (target_sum - s) / n
        h = np.clip(h, 0.0, 1.0)
        resid = target_sum - h.sum()
        if abs(resid) > 1e-14:
            if resid > 0:
                idx = np.argmin(h)
            else:
                idx = np.argmax(h)
            h[idx] += resid
            h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def pgd_continuous(h0, n, max_time, lr_init=0.05, lr_min=1e-5, decay=0.9995):
    target_sum = n / 2.0
    h = h0.copy()
    h = project_feasible(h, n, target_sum)
    curr_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    lr = lr_init
    start = time.time()
    stagnant = 0
    while time.time() - start < max_time:
        grad = gradient_max_corr(h, n)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n, target_sum)
        new_c5 = eval_c5(h_new)
        if new_c5 < curr_c5:
            curr_c5 = new_c5
            h = h_new
            stagnant = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            stagnant += 1
            if stagnant > 200:
                break
            lr *= 0.95
        lr = max(lr, lr_min)
        lr *= decay
    return best_h, best_c5

def local_search_01(h01, n, max_time, seed=42):
    rng = np.random.RandomState(seed)
    h = h01.astype(float).copy()
    curr_c5 = eval_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    start = time.time()
    stall = 0
    while time.time() - start < max_time:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_pairs = len(ones) * len(zeros)
        n_try = min(n_pairs, 800)
        best_new_c5 = curr_c5
        best_move = None
        for _ in range(n_try):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = eval_c5(h)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_move = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            curr_c5 = best_new_c5

def make_01_from_cont(h_cont, n):
    h01 = np.zeros(n)
    order = np.argsort(h_cont)[::-1]
    h01[order[:n // 2]] = 1.0
    return h01

def upsample(parent, n_new):
    n_old = len(parent)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, parent)
```
