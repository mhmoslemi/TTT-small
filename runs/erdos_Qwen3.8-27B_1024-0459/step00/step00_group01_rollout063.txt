    # Phase 2: SLSQP with higher alpha (sharper max)
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        h2 = slsqp_optimize(best_h, alpha=150, maxiter=300, ftol=1e-11)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_h = h2
            best_c5 = c2

    # Phase 3: Cycling method at n0
    elapsed = time.time() - start_time
    if budget_s - elapsed > 90:
        h3, c3 = cycling_optimize(best_h, start_time, budget_s, max_cycles=10)
        if c3 < best_c5:
            best_h = h3
            best_c5 = c3

    # Phase 4: Try higher resolutions
    for n_new in [94, 150, 250]:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 90:
            break
        best_h, best_c5 = try_resolution(best_h, best_c5, n_new, start_time, budget_s)

    # Phase 5: Random restarts at moderate resolution
    elapsed = time.time() - start_time
    for rs in range(4):
        if budget_s - elapsed < 50:
            break
        h_rs, c_rs = random_restart_optimize(n0, seed + 100 + rs, start_time, budget_s)
        if c_rs < best_c5:
            best_h = h_rs
            best_c5 = c_rs

    # Phase 6: Local search (simulated annealing)
    elapsed = time.time() - start_time
    if budget_s - elapsed > 30:
        h_ls, c5_ls = local_search(best_h, seed + 500, 20000, start_time, budget_s)
        if c5_ls < best_c5:
            best_h = h_ls
            best_c5 = c5_ls

    # Phase 7: Final polish with SLSQP
    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        h_fin = slsqp_optimize(best_h, alpha=100, maxiter=200, ftol=1e-12)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_h = h_fin
            best_c5 = c_fin

    # Ensure constraint is satisfied
    n_final = len(best_h)
    best_h = renormalize(best_h, n_final)

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    if final_c5 <= best_c5:
        best_c5 = final_c5

    return (best_h, best_c5, n_final)
