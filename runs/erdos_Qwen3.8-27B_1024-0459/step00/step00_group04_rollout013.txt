        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # All at bounds, do best effort
            if diff > 0:
                h[h < 1.0] += min(diff / max(1, (h < 1.0).sum()), 1.0 - h[h < 1.0])
            else:
                h[h > 0.0] += diff / max(1, (h > 0.0).sum())
                h = np.clip(h, 0.0, 1.0)
    
    return h

def optimize_pg(h_init, n, max_iter=20000, time_limit=60):
    """Projected gradient optimization."""
    start = time.time()

            h_new = project_feasible(y, n)

            if c5_new >= c5_curr:

                lr *= 0.5
                if lr < 1e-10:

        h = h_new
        c5_curr = c5_new

            best_c5 = c5_curr
            best_h = h.copy()
        
        # Adapt lr
        if improved:
            lr = min(lr * 1.1, 5.0 / n)
        else:
            lr *= 0.7
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def compute_c5_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_val = dx * np.max(corr)
    max_corr = corr.max()
    tol = max(1e-12, 1e-3 * max_corr)
    active = np.where(corr >= max_corr - tol)[0]
    grad = np.zeros(n)
    for k in active:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_hi < j_lo:
            continue
        j_vals = np.arange(j_lo, j_hi + 1)
        m_vals = j_vals - k + n - 1
        grad[j_vals] += (1.0 - h[m_vals])
        grad[m_vals] -= h[j_vals]
    if len(active) > 1:
        grad /= len(active)
    return c5_val, dx * grad

def project_feasible(y, n):
    target = n / 2.0
    if target <= 0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(y - theta, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)

            adj = max(min(adj, 1.0 - h[free].min() - 1e-12), -h[free].max() + 1e-12)

        x = np.clip(x, 0, 1)
