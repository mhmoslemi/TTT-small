def compute_corr(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_val(corr, beta):
    c = np.max(corr)
    if c == 0:
        return 0.0
    shifted = beta * (corr - c)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    return c + np.log(np.sum(np.exp(shifted))) / beta

def compute_gradient(h, dx, beta):
    n = len(h)
    corr = compute_corr(h, dx)
    weights = np.exp(beta * (corr - np.max(corr)))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    shifts = np.arange(-(n-1), n)
    
    for m in range(n):
        # Term 1: (1-x[m-shift]) for valid
        idx1 = m - shifts
        valid1 = (idx1 >= 0) & (idx1 < n)
        if valid1.any():
            grad[m] += np.dot(weights[valid1], (1.0 - h[idx1[valid1]]))
        
        # Term 2: -x[m+shift] for valid
        idx2 = m + shifts
        valid2 = (idx2 >= 0) & (idx2 < n)
        if valid2.any():
            grad[m] -= np.dot(weights[valid2], h[idx2[valid2]])
    
    grad *= dx
    return grad

        h += diff / n
        return np.clip(h, 0, 1)
    h[mask] += diff / mask.sum()
    return np.clip(h, 0, 1)
