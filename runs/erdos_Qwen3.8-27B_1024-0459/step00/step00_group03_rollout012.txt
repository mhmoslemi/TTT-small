                h[idx] -= diff
        else:
            mask = (h > 1e-10) & (h < 1 - 1e-10)
            if mask.any():
                n_inner = mask.sum()
                adj = -diff / n_inner
                h[mask] += adj
                h = np.clip(h, 0.0, 1.0)
            else:
                idx = np.argmin(h)
                h[idx] -= diff
    return h

def gradient_softmax(h, alpha):
    """Compute gradient of softmax(correlate(h, 1-h, 'full'), alpha) * dx w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    
    # Soft max weights
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-16:
            continue
        lag = k - (n - 1)
        wk = weights[k]
        for j in range(n):
            d = 0.0
            # j in S: 0<=j<n and 0<=j+lag<n
            if 0 <= j + lag < n:
                d += 1.0 - h[j + lag]
            # j in S': 0<=j<n and 0<=j-lag<n (j appears as i+lag, i=j-lag)
            if 0 <= j - lag < n:
                d -= h[j - lag]
            grad[j] += wk * d
    
    grad *= dx
    return grad

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

def grad_softmax(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        lag = k - (n - 1)
