def repair_sum(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    if len(h)==0: return h
    if target is None: target = len(h)/2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    # distribute several times
    for _ in range(5):
        if abs(diff) < 1e-13: break
        if diff > 0:
            cands = np.where(h < 1.0 - 1e-12)[0]
            if len(cands)==0: break
            delta = diff / len(cands)
            h[cands] = np.minimum(1.0, h[cands] + delta)
        else:
            cands = np.where(h > 1e-12)[0]
            if len(cands)==0: break
            delta = diff / len(cands)
            h[cands] = np.maximum(0.0, h[cands] + delta)
        diff = target - float(np.sum(h))
    # final exact-ish adjust one variable
    if abs(diff) > 0:
       if diff >0: cands=np.where(h<1)[0]
       else: cands=np.where(h>0)[0]
       if len(cands)>0:
          h[cands[0]] = float(np.clip(h[cands[0]] + diff,0,1))
    return h

def repair_sum(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    if h.size == 0:
        return h
    if target is None:
        target = 0.5 * len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = float(target) - float(np.sum(h))
    for _ in range(6):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            cands = np.flatnonzero(h < 1.0 - 1e-12)
            if cands.size == 0:
                break
            delta = diff / float(cands.size)
            h[cands] = np.minimum(1.0, h[cands] + delta)
        else:
            cands = np.flatnonzero(h > 1e-12)
            if cands.size == 0:
                break
            delta = diff / float(cands.size)
            h[cands] = np.maximum(0.0, h[cands] + delta)
        diff = float(target) - float(np.sum(h))
    if abs(diff) > 0.0:
        if diff > 0:
            cands = np.flatnonzero(h < 1.0)
        else:
            cands = np.flatnonzero(h > 0.0)
        if cands.size > 0:
            idx = int(cands[0])
            h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def periodic_bits(n, p, shift=0, reverse=False):
    if p <= 1:
        return np.ones(n, dtype=int)
    ones_count = p // 2
    pattern = np.zeros(p, dtype=int)
    pattern[:ones_count] = 1
    if reverse:
        pattern = pattern[::-1].copy()
    if shift:
        pattern = np.roll(pattern, int(shift % p))
    reps = n // p + 2
    return np.tile(pattern, reps)[:n]

def add_var_to_pair_arrays(h, A, B, L0, L1, r, f, n):
    lo = max(L0, -r)
    hi = min(L1, n - 1 - r)
    if lo <= hi:
        i0 = lo - L0
        i1 = hi - L0 + 1
        A[i0:i1] += f * (1.0 - h[r + lo:r + hi + 1])

    lo = max(L0, -(n - 1 - r))
    hi = min(L1, r)
    if lo <= hi:
        i0 = lo - L0
        i1 = hi - L0 + 1
        A[i0:i1] += f * (-h[r - hi:r - lo + 1][::-1])


def pair_candidate_arrays(h, T, i, j):
    n = len(h)
    ri = max(i, n - 1 - i)
    rj = max(j, n - 1 - j)
    R = ri if ri > rj else rj
    L0 = -R
    L1 = R
    s0 = n - 1 + L0
    s1 = n - 1 + L1
    mtot = len(T)
    if s0 < 0:
        s0 = 0
    if s1 >= mtot:
        s1 = mtot - 1
    m = s1 - s0 + 1
    if m <= 0:
        A = np.zeros(1, dtype=float)
        B = np.zeros(1, dtype=float)
        Ts = np.array([float(np.max(T))])
        return A, B, Ts, -np.inf, 0, 0

    A = np.zeros(m, dtype=float)
    B = np.zeros(m, dtype=float)

    add_var_to_pair_arrays(h, A, B, L0, L1, i, -1.0, n)
    add_var_to_pair_arrays(h, A, B, L0, L1, j, 1.0, n)

    zero_off = 0 - L0
    if 0 <= zero_off < m:
        B[zero_off] -= 2.0

    dist = j - i
    off = dist - L0
    if L0 <= dist <= L1 and 0 <= off < m:
        B[off] += 1.0

    off = -dist - L0
    if L0 <= -dist <= L1 and 0 <= off < m:
        B[off] += 1.0

    Ts = T[s0:s1 + 1].copy()
    un = -np.inf
    if s0 > 0:
        un = float(np.max(T[:s0]))
    if s1 + 1 < mtot:
        un = max(un, float(np.max(T[s1 + 1:])))
    return A, B, Ts, un, s0, s1


def try_pair_move(h, T, i, j, dx, current_c, grid=33, refine=91, tol=1e-12):
    n = len(h)
    if i == j:
        return None
    hi = float(h[i])
    hj = float(h[j])
    dlo = -min(hj, 1.0 - hi)

        return None

    ds = np.linspace(dlo, dhi, grid, dtype=float)
    if dlo < 0.0 < dhi:
        ds = np.concatenate((ds, np.array([0.0], dtype=float)))

    q = ds[:, None]
    vals = Ts[None, :] + A[None, :] * q + B[None, :] * (q * q)
    maxvals = np.max(vals, axis=1)
    if un > -np.inf:
        maxvals = np.maximum(maxvals, un)
    cvals = maxvals * dx
    k = int(np.argmin(cvals))
    best_d = float(ds[k])
    best_c = float(cvals[k])

    if best_c < current_c - 1e-10:
        half = (dhi - dlo) / float(max(grid - 1, 1))
        for mult in (3.0, 1.0, 0.25):
            lo = max(dlo, best_d - half * mult)
            hi = min(dhi, best_d + half * mult)
            if hi - lo <= 1e-12:
                continue
            ds2 = np.linspace(lo, hi, refine, dtype=float)
            if dlo < 0.0 < dhi:
                ds2 = np.concatenate((ds2, np.array([0.0], dtype=float)))
            q2 = ds2[:, None]
            vals2 = Ts[None, :] + A[None, :] * q2 + B[None, :] * (q2 * q2)
            maxvals2 = np.max(vals2, axis=1)
            if un > -np.inf:
                maxvals2 = np.maximum(maxvals2, un)
            cvals2 = maxvals2 * dx
            k2 = int(np.argmin(cvals2))
            if cvals2[k2] < best_c:
                best_d = float(ds2[k2])
                best_c = float(cvals2[k2])

    h2 = h.copy()
    ni = hi - best_d
    nj = hj + best_d
    if ni < 0.0 and ni > -1e-10:
        ni = 0.0
    if nj < 0.0 and nj > -1e-10:
        nj = 0.0
    if ni > 1.0 and ni < 1.0 + 1e-10:
        ni = 1.0
    if nj > 1.0 and nj < 1.0 + 1e-10:
        nj = 1.0
    h2[i] = float(np.clip(ni, 0.0, 1.0))
    h2[j] = float(np.clip(nj, 0.0, 1.0))

        h[:] = h2
        Tnew = np.correlate(h, 1.0 - h, mode="full")
        return c2, Tnew
    return None


def compute_gradient_active(h, T, top_k=24):
    n = len(h)
    maxT = float(np.max(T))
    if maxT <= 0.0:
        return np.zeros(n, dtype=float)
    top = np.argsort(T)[::-1][:top_k]
    grad = np.zeros(n, dtype=float)
    w = 1.0 / float(max(1, len(top)))
    for s in top:
        L = int(s) - (n - 1)
        if L > 0:
            a = n - L
            if a <= 0:
                continue
            grad[:a] += w * (1.0 - h[L:L + a])
            grad[L:] += -w * h[:a]
        elif L < 0:
            a = -L
            if a >= n:
                continue
            grad[a:] += w * (1.0 - h[:n - a])
            grad[:n - a] += -w * h[a:]
        else:
            grad[:] += w * (1.0 - 2.0 * h)
    return grad


def proposals_from_gradient(h, T, grad, max_prop=48, rng=None):
    n = len(h)
    props = []
    seen = set()

    def add_prop(a, b):
        if a == b:
            return
        key = (int(a), int(b))
        if key not in seen:
            seen.add(key)
            props.append(key)

    top_lags = np.argsort(T)[::-1][:18]
    for s in top_lags:
        L = int(s) - (n - 1)
        if L == 0:
            continue
        if L > 0:
            a = n - L
            if a <= 0:
                continue
            contrib = h[:a] * (1.0 - h[L:L + a])
        else:
            a = -L
            if a >= n:
                continue
            contrib = h[a:] * (1.0 - h[:n - a])
        if contrib.size == 0:
            continue
        mx = float(np.max(contrib))
        if mx <= 1e-12:
            continue
        order = np.argsort(contrib)[::-1][:4]
        for u in order:
            u = int(u)
            if L > 0:
                add_prop(u, u + L)
            else:
                add_prop(a + u, u)
        if len(props) >= max_prop:
            break

    if len(props) < max_prop and len(grad) > 1:
        order = np.argsort(grad)
        low = order[:10]
        high = order[-10:]
        pairs = []
        for x in high:
            xi = int(x)
            for y in low:
                yi = int(y)
                if xi != yi:
                    pairs.append((float(grad[xi] - grad[yi]), xi, yi))
        pairs.sort(reverse=True)
        for _, a, b in pairs:
            add_prop(a, b)
            if len(props) >= max_prop:
                break

    if len(props) == 0:
        hi_idx = np.argsort(h)[-5:]
        lo_idx = np.argsort(h)[:5]
        for a in hi_idx:
            for b in lo_idx:
                add_prop(int(a), int(b))
                if len(props) >= max_prop:
                    break
            if len(props) >= max_prop:
                break

    return props[:max_prop]


def random_perturb_continuous(h, dx, n, rng, curr, atol=1e-4):
    for _ in range(60):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        hi = float(h[i])
        hj = float(h[j])
        lo = -min(hj, 1.0 - hi)
        hi_b = min(hi, 1.0 - hj)
        if hi_b <= lo + 1e-12:
            continue
        d = float(rng.uniform(lo, hi_b))
        h2 = h.copy()
        h2[i] = hi - d
        h2[j] = hj + d
        if h2.min() < -1e-9 or h2.max() > 1.0 + 1e-9:
            continue
        if abs(float(np.sum(h2)) - 0.5 * n) > 1e-8:
            continue
        c2 = evaluate_c5(h2, dx)
        if c2 < curr + atol:
            return h2, c2
    return None


def continuous_local_search(h, deadline, rng):
    h = repair_sum(h.copy(), 0.5 * len(h))
    n = len(h)
    dx = 2.0 / float(n)
    T = np.correlate(h, 1.0 - h, mode="full")
    curr = float(np.max(T) * dx)
    best_h = h.copy()
    best_c = curr
    no_improve = 0
    perturbations = 0
    max_perturb = 4

    while time.time() < deadline - 2.0 and no_improve < 4 and perturbations < max_perturb:
        grad = compute_gradient_active(h, T, top_k=28)
        props = proposals_from_gradient(h, T, grad, max_prop=56, rng=rng)
        improved = False
        for a, b in props:
            if time.time() >= deadline - 2.0:
                break
            res = try_pair_move(h, T, a, b, dx, curr, grid=33, refine=91)
            if res is not None:
                curr, T = res
                if curr < best_c - 1e-12:
                    best_c = curr
                    best_h = h.copy()
                improved = True
                no_improve = 0
                break
        if not improved:

            p = random_binary_perturb(h, inv, dx, rng, atol=1e-4, tries=80)
            if p is not None:
                h, inv, curr = p
                if curr < best_c:
                    best_c = curr
                    best_h = h.copy()
                    best_inv = inv.copy()
                stall = 0

    return best_h, best_c


def optimize_candidate(h, deadline, max_slice, rng):
    start = time.time()
    end = min(deadline - 2.0, start + max(1.0, max_slice))
    h0 = repair_sum(np.asarray(h, dtype=float).copy(), 0.5 * len(h))
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    best_n = len(best_h)
    c0 = best_c

    remaining = end - time.time()
    if remaining < 3.0:
        return best_h, best_c, best_n

    if len(best_h) <= 210:
        hbin = binarize_or_half(h0)
        cbin = evaluate_c5(hbin)
        if cbin < best_c:
            best_c = cbin
            best_h = hbin.copy()
        tbin = min(remaining * 0.38, 160.0)
        if tbin > 2.0:
            bh, bc = binary_local_search(hbin, time.time() + tbin, rng)
            if bc < best_c:
                best_c = bc
                best_h = bh.copy()
                best_n = len(best_h)
        remaining = end - time.time()
        if remaining > 3.0:
            if best_h is not None and is_close_binary_or_half(best_h) and best_c <= c0:
                cont_seed = best_h.copy()
            else:
                cont_seed = h0.copy()
            ch, cc = continuous_local_search(cont_seed, time.time() + remaining - 1.0, rng)
            if cc < best_c:
                best_c = cc
                best_h = ch.copy()
                best_n = len(best_h)

    remaining = end - time.time()
    if remaining > 8.0 and len(best_h) <= 210:
        hbin2 = binarize_or_half(best_h)
        cbin2 = evaluate_c5(hbin2)
        if cbin2 < best_c + 1e-6:
            bh2, bc2 = binary_local_search(hbin2, time.time() + max(2.0, remaining * 0.45), rng)
            if bc2 < best_c:
                best_c = bc2
                best_h = bh2.copy()
                best_n = len(best_h)

    return best_h, best_c, best_n


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    budget = float(budget_s)
    deadline = start + max(1.0, budget - 6.0)
    rng = np.random.RandomState(int(seed) % (2 ** 31 - 1))

    musts = []
    others = []

    init = None
    try:
        init = initial_h_values
    except Exception:
        init = None

    init_len = None
    if init is not None:
        hinit = np.asarray(init, dtype=float).reshape(-1)
        hinit = hinit[np.isfinite(hinit)]
        if hinit.size > 1:
            hinit = repair_sum(hinit, 0.5 * hinit.size)
            init_len = int(hinit.size)
            musts.append(hinit)
            musts.append(binarize_or_half(hinit))
            h2 = upsample_repeat(hinit)
            musts.append(h2)
            musts.append(binarize_or_half(h2))
            if hinit.size <= 48:
                h4 = upsample_repeat(h2)
                musts.append(h4)
                musts.append(binarize_or_half(h4))

    ns = []
    if init_len is not None:
        for n in (init_len, 2 * init_len, 4 * init_len):
            n = int(n)
            if 2 <= n <= 210 and n not in ns:
                ns.append(n)
    for n in (45, 90, 180):
        if 2 <= n <= 210 and n not in ns:
            ns.append(n)

    for n in ns:
        if n <= 60:
            rcount = 22
        elif n <= 100:
            rcount = 12
        else:
            rcount = 7
        for _ in range(rcount):
            others.append(make_random_binary(n, rng))
        for _ in range(2):
            others.append(make_random_fractional(n, rng))

        bits = thue_bits(n)
        others.append(balanced := balance_binary_bits(bits, rng))
        others.append(balance_binary_bits(np.flip(bits), rng))
        bits = golden_bits(n)
        others.append(balance_binary_bits(bits, rng))
        others.append(balance_binary_bits(np.flip(bits), rng))
        for p in (4, 5, 6, 7, 8, 9, 10, 12):
            if p < n:
                others.append(balance_binary_bits(periodic_bits(n, p, 0, False), rng))
                others.append(balance_binary_bits(periodic_bits(n, p, 1, False), rng))
                others.append(balance_binary_bits(periodic_bits(n, p, 0, True), rng))

    scored_must = []
    counter = 0
    for h in musts:
        try:
            c = evaluate_c5(h)
            if np.isfinite(c):
                scored_must.append((c, counter, h))
        except Exception:
            pass
        counter += 1

    scored_other = []
    for h in others:
        try:
            c = evaluate_c5(h)
            if np.isfinite(c):
                n = len(h)
                if n < 2:
                    continue
                if (len(scored_other) + len(scored_must)) >= 160:
                    continue
                scored_other.append((c, counter, h))
        except Exception:
            pass
        counter += 1

    scored_must.sort()
    scored_other.sort()
    ordered = scored_must + scored_other

    best_h = None
    best_c = float("inf")
    best_n = 0

    if len(ordered) > 0:
        c, _, h = ordered[0]
        best_h = np.asarray(h, dtype=float).copy()
        best_c = float(c)
        best_n = len(best_h)
    else:
        h = make_random_binary(45, rng)
        best_h = h
        best_n = len(h)
        best_c = evaluate_c5(h)

    max_opt = 12
    if budget > 900:
        max_opt = 14
    elif budget < 500:
        max_opt = 8
    if len(ordered) > max_opt:
        ordered = ordered[:max_opt]

    weights = [14, 13, 12, 10, 9, 8, 7, 6, 6, 5, 5, 5, 5, 5]

    for idx in range(len(ordered)):
        if time.time() >= deadline - 15.0:
            break
        total_w = 0.0
        for ii in range(idx, len(ordered)):
            if ii < len(weights):
                total_w += float(weights[ii])
            else:
                total_w += 4.0
        if total_w <= 0.0:
            break
        allow = (deadline - time.time() - 10.0) * float(weights[idx] if idx < len(weights) else 4.0) / total_w
        allow = min(allow, 320.0)
        if allow < 4.0:
            break
        _, _, h = ordered[idx]
        try:
            hh, cc, nn = optimize_candidate(h, deadline, allow, rng)
            if cc < best_c:
                best_c = float(cc)
                best_h = np.asarray(hh, dtype=float).copy()
                best_n = int(nn)
        except Exception:
            pass

    if time.time() < deadline - 20.0 and best_h is not None:
        allow = deadline - time.time() - 6.0
        try:
            hh, cc, nn = optimize_candidate(best_h, deadline, allow, rng)
            if cc < best_c:
                best_c = float(cc)
                best_h = np.asarray(hh, dtype=float).copy()
                best_n = int(nn)
        except Exception:
            pass

    final_h = repair_sum(np.asarray(best_h, dtype=float).copy(), 0.5 * len(best_h))
    final_h = np.clip(final_h, 0.0, 1.0)
    final_c = evaluate_c5(final_h)
    if not np.isfinite(final_c):
        final_h = make_random_binary(45, rng)
        final_c = evaluate_c5(final_h)
    return final_h, float(final_c), int(len(final_h))
