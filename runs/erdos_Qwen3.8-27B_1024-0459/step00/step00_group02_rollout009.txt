def project_onto_feasible(x, target_sum):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}."""
    # Find alpha such that sum(clip(x - alpha, 0, 1)) = target_sum
    lo, hi = -10, 10
    for _ in range(60):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0, 1)
        s = np.sum(y)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2, 0, 1)
    # Fix any tiny numerical issues
    y = np.clip(y, 0, 1)
    diff = target_sum - np.sum(y)
    if abs(diff) > 1e-10:
        # Distribute the difference
        n = len(y)
        y = y + diff / n
        y = np.clip(y, 0, 1)
    return y
