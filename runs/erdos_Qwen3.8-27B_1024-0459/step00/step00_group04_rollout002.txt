def project_box_sum(v, target):

    n=len(v); target=float(target)

       if s > target: lo=mid # need increase tau
       else: hi=mid
    p = np.clip(v-(lo+hi)/2,0,1)

def project_box_sum(v, target):
    v=np.asarray(v,float).copy()
    n=len(v)
    if n==0: return v
    target=np.clip(float(target),0.0,float(n))
    v=np.clip(v,0.0,1.0)
    s=float(v.sum())
    if abs(s-target) <= 1e-12: return v
    if s > target:
       p=project_reduce_sorted(v,target)
    else:
       q=1.0-v
       qred=project_reduce_sorted(q, float(n)-target)
       p=1.0-qred
    p=np.clip(p,0,1)
    # adjust residual
    for _ in range(3):
       diff=target-p.sum()
       if abs(diff) <= 1e-12: break
       if diff>0:
          idx=int(np.argmax(1.0-p + 1e-30)) # most room
          # distribute maybe to all with room if single not enough
          idxs=np.argsort(-(1.0-p))[:min(n,int(abs(diff)*10)+1)]
          add=diff/len(idxs)
          for i in idxs:
             p[i]=min(p[i]+add,1.0)
       else:
          idxs=np.argsort(p)[:min(n,int(abs(diff)*10)+1)]
          sub= -diff/len(idxs)
          for i in idxs:
             p[i]=max(p[i]-sub,0.0)
    return p

def raw_corr_fft(h):
    h=np.asarray(h,float).ravel()
    n=len(h)
    size=1
    while size < 2*n-1:
       size <<=1
    a=np.fft.rfft(h, size)

    bb=np.fft.rfft(b, size)

    v=v[:2*n-1]
    raw=v[::-1]

def project_reduce_sorted(u, target):
    u = np.asarray(u, dtype=float).ravel()
    n = u.size
    if n == 0:
        return u.copy()
    target = float(target)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    total = float(u.sum())
    if target >= total - 1e-12:
        return u.copy()

    s = np.sort(u)[::-1]
    pref = np.cumsum(s)
    lam = (pref[-1] - target) / float(n)

    for k in range(1, n + 1):
        lam_k = (pref[k - 1] - target) / float(k)
        if lam_k < 0.0:
            lam_k = 0.0
        if k < n:
            if lam_k <= s[k - 1] + 1e-12 and lam_k >= s[k] - 1e-12:
                lam = lam_k
                break
        else:
            lam = lam_k
            break

    p = np.maximum(u - lam, 0.0)
    p = np.minimum(p, 1.0)
    return p


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float).ravel().copy()
    n = v.size
    if n == 0:
        return v
    target = float(min(max(target, 0.0), float(n)))

    v = np.where(np.isfinite(v), v, 0.5)
    v = np.clip(v, 0.0, 1.0)

    s = float(v.sum())
    if abs(s - target) <= 1e-12:
        return v

    if s > target:
        p = project_reduce_sorted(v, target)
    else:
        q = 1.0 - v
        q1 = project_reduce_sorted(q, float(n) - target)
        p = 1.0 - q1

    p = np.clip(p, 0.0, 1.0)

    for _ in range(4):
        diff = target - float(p.sum())
        if abs(diff) <= 1e-12:
            break
        if diff > 0.0:
            room = 1.0 - p
            rs = float(room.sum())
            if rs > 0.0:
                p = np.clip(p + room * (diff / rs), 0.0, 1.0)
            else:
                p = np.ones(n, dtype=float)
                p.fill(min(target / float(n), 1.0))
        else:
            bs = float(p.sum())
            if bs > 0.0:
                p = np.clip(p + p * (diff / bs), 0.0, 1.0)
            else:
                p = np.zeros(n, dtype=float)

    return p

def exact_c5(h):

    n = h.size
    if n == 0:

    if not np.all(np.isfinite(h)):

def raw_corr_direct(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.zeros(0, dtype=float)
    b = 1.0 - h
    L = 2 * n - 1
    out = np.empty(L, dtype=float)
    idx = 0
    for tau in range(-(n - 1), n):
        if tau >= 0:
            out[idx] = float(np.dot(h[:n - tau], b[tau:]))
        else:
            sg = -tau
            out[idx] = float(np.dot(h[sg:], b[:n - sg]))
        idx += 1
    return out


def raw_corr_fft(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return np.zeros(0, dtype=float)

    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1

    a = np.fft.rfft(h, size)
    b = 1.0 - h[::-1]
    bb = np.fft.rfft(b, size)
    v = np.fft.irfft(a * bb, size).real[:need]
    raw = v[::-1]

    if not np.all(np.isfinite(raw)):
        return np.zeros(need, dtype=float)

    np.maximum(raw, 0.0, out=raw)
    return raw

def smooth_obj_direct(h, epsC):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0, np.zeros(0, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100, np.zeros(n, dtype=float)

    b = 1.0 - h
    L = 2 * n - 1
    raw = np.empty(L, dtype=float)
    idx = 0
    for tau in range(-(n - 1), n):
        if tau >= 0:
            raw[idx] = float(np.dot(h[:n - tau], b[tau:]))
        else:
            sg = -tau
            raw[idx] = float(np.dot(h[sg:], b[:n - sg]))
        idx += 1

    dx = 2.0 / float(n)
    rc = raw * dx
    m = float(np.max(rc))
    epsC = max(float(epsC), 1e-8)
    z = (rc - m) / epsC
    e = np.exp(z)
    exp_sum = float(e.sum())

    if not math.isfinite(exp_sum) or exp_sum <= 0.0:
        w = np.zeros(L, dtype=float)
        w[int(np.argmax(rc))] = 1.0
        f = m
    else:
        f = m + epsC * math.log(exp_sum)
        w = e / exp_sum

    grad = np.zeros(n, dtype=float)
    nz = np.flatnonzero(w > 1e-13)
    for idx in nz:
        wv = float(w[idx])
        if wv <= 0.0:
            continue
        tau = int(idx) - (n - 1)
        if tau >= 0:
            seg = n - tau
            if seg > 0:
                grad[:seg] += wv * b[tau:tau + seg]
                grad[tau:tau + seg] -= wv * h[:seg]
        else:
            sg = -tau
            seg = n - sg
            if seg > 0:
                grad[sg:sg + seg] += wv * b[:seg]
                grad[:seg] -= wv * h[sg:sg + seg]

    grad *= dx
    return f, grad


def smooth_f_fft(h, epsC):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0
    if not np.all(np.isfinite(h)):

def smooth_obj_fft_topk(h, epsC, topk):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return 0.0, np.zeros(0, dtype=float)
    if not np.all(np.isfinite(h)):

    raw = raw_corr_fft(h)

    L = 2 * n - 1
    m = float(np.max(rc))
    epsC = max(float(epsC), 1e-8)
    z = (rc - m) / epsC
    e = np.exp(z)
    exp_sum = float(e.sum())

    if not math.isfinite(exp_sum) or exp_sum <= 0.0:
        w = np.zeros(L, dtype=float)
        am = int(np.argmax(rc))
        w[am] = 1.0
        f = m
    else:
        f = m + epsC * math.log(exp_sum)
        w = e / exp_sum

    if topk is None or int(topk) >= L:
        active = np.flatnonzero(w > 1e-13)
        if active.size == 0:
            active = np.array([int(np.argmax(w))], dtype=int)

    b = 1.0 - h

        if wv <= 0.0:

        tau = int(idx) - (n - 1)
        if tau >= 0:
            seg = n - tau
            if seg > 0:

            sg = -tau
            seg = n - sg
            if seg > 0:
                grad[sg:sg + seg] += wv * b[:seg]
                grad[:seg] -= wv * h[sg:sg + seg]

    grad *= dx
    return f, grad


def equal_active_gradient(h, active):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    b = 1.0 - h
    grad = np.zeros(n, dtype=float)
    active = np.asarray(active, dtype=int).ravel()
    if active.size == 0:
        return grad

    for idx in active:
        tau = int(idx) - (n - 1)
        if tau >= 0:
            seg = n - tau
            if seg > 0:
                grad[:seg] += b[tau:tau + seg]
                grad[tau:tau + seg] -= h[:seg]
        else:
            sg = -tau
            seg = n - sg
            if seg > 0:
                grad[sg:sg + seg] += b[:seg]
                grad[:seg] -= h[sg:sg + seg]

    grad *= (2.0 / float(n)) / float(active.size)
    return grad


def slq_obj(x):

def slq_eq_fun(x):

    return float(np.sum(x) - _SLQ_TARGET)


def slq_eq_jac(x):
    x = np.asarray(x, dtype=float).ravel()
    return np.ones_like(x, dtype=float)


def get_slq_constraints():

    if _SLQ_CONST is None:
        _SLQ_CONST = [
            {"type": "eq", "fun": slq_eq_fun, "jac": slq_eq_jac}
        ]

def slsqp_optimize(h0, n, eps, maxiter):
    global _SLQ_EPS, _SLQ_TARGET
    n = int(n)
    x0 = project_box_sum(h0, 0.5 * float(n))
    _SLQ_EPS = float(eps)
    _SLQ_TARGET = 0.5 * float(n)

    try:
        res = minimize(
            slq_obj,
            x0,
            method="SLSQP",
            jac=slq_grad,
            bounds=Bounds(0.0, 1.0),
            constraints=get_slq_constraints(),
            options={"maxiter": int(maxiter), "ftol": 1e-10, "iprint": 0}
        )
        x = np.asarray(res.x, dtype=float).ravel()
    except Exception:
        x = x0

    return project_box_sum(x, 0.5 * float(n))


def lbfgs_value(x):
    x = np.asarray(x, dtype=float).ravel()
    f = smooth_f_fft(x, _LB_EPS)
    s = float(np.sum(x) - _LB_TARGET)
    f = f + 0.5 * _LB_BETA * s * s
    if not math.isfinite(f):
        return 1e100
    return f


def lbfgs_gradient(x):

    _, g = smooth_obj_fft_topk(x, _LB_EPS, _LB_TOPK)
    s = float(np.sum(x) - _LB_TARGET)
    g = g + _LB_BETA * s
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(x, dtype=float)
    return g

def polish_subgrad(h, max_iters=50, time_limit=15.0):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return h, 0.0
    h = project_box_sum(h, 0.5 * float(n))
    best = exact_c5(h)
    t0 = time.time()

    use_direct_raw = n <= 800
    fast_exact = n <= 256
    k_active = int(min(24, max(5, int(math.sqrt(max(n, 4))))))
    steps = (5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 1e-5)

    for _ in range(int(max_iters)):
        if time.time() - t0 > float(time_limit):
            break

        if use_direct_raw:
            raw = raw_corr_direct(h)
        else:
            raw = raw_corr_fft(h)

        dx = 2.0 / float(n)
        rc = raw * dx
        m = float(np.max(rc))
        if m + 1e-11 <= best:
            break

        tol = 1e-8 + 1e-5 * (m + 1.0)
        active = np.flatnonzero(rc >= m - tol)
        if active.size == 0:
            active = np.array([int(np.argmax(rc))], dtype=int)
        elif active.size > k_active:
            kk = int(min(k_active, raw.size))
            part = np.argpartition(-rc, kk - 1)[:kk]
            active = part[np.argsort(-rc[part])]

        improved = False
        for trial in (0, 1):
            if trial == 0:
                act = active
            else:
                act = np.array([int(np.argmax(rc))], dtype=int)

            grad = equal_active_gradient(h, act)
            norm = float(np.linalg.norm(grad))
            if norm < 1e-14:
                break

            for a in steps:
                v = project_box_sum(h - (a / norm) * grad, 0.5 * float(n))

                if fast_exact:
                    cfast = exact_c5(v)
                else:
                    cfast = approx_c5_fft(v)
                    if cfast >= best - 1e-9:
                        continue
                    cfast = exact_c5(v)

                if cfast < best - 1e-11:
                    h = v
                    best = cfast
                    improved = True
                    break

            if improved:
                break

        if not improved:
            break

    h = project_box_sum(h, 0.5 * float(n))
    best = exact_c5(h)
    return h, best

def upscale(h, n_out):

    n = h.size
    n_out = int(n_out)
    if n == 0 or n_out <= 0:
        return np.zeros(max(n_out, 1), dtype=float)
    if n_out == n:
        return h.copy()
    if n_out % n == 0:
        out = np.repeat(h, n_out // n)
    else:
        idx = np.floor(np.arange(n_out, dtype=float) * n / float(n_out)).astype(int)
        idx = np.clip(idx, 0, n - 1)
        out = h[idx]
    return project_box_sum(out, 0.5 * float(n_out))

        + 0.12 * np.sin(6.0 * np.pi * x + 1.97)
    )
    h = np.clip(h, 0.0, 1.0)
    return project_box_sum(h, 0.5 * float(n))


def dedup_candidates(arrays, tol=1e-8):
    out = []
    for a in arrays:
        a = np.asarray(a, dtype=float).ravel()
        if a.size == 0:
            continue
        dup = False
        for b in out:
            if b.size == a.size and float(np.max(np.abs(a - b))) <= float(tol):
                dup = True
                break
        if not dup:
            out.append(a)
    return out


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    total = float(budget_s) if budget_s is not None else 1300.0
    total = max(30.0, total)

    parent = None
    try:
        iv = globals().get("initial_h_values")
        if iv is not None:
            arr = np.asarray(iv, dtype=float).ravel()
            if arr.size > 0 and np.all(np.isfinite(arr)):
                parent = arr
    except Exception:
        parent = None

    if parent is None:
        parent = analytic_seed(96)

    parent = sanitize_h_for_return(parent)
    base_n = int(parent.size)

    best_h = parent.copy()
    best_c = exact_c5(best_h)

    if total < 60.0:
        best_h = sanitize_h_for_return(best_h)
        best_c = exact_c5(best_h)
        return best_h, float(best_c), int(best_h.size)

    rng = np.random.RandomState(int(seed) % (2**31 - 1))

    small_time = min(max(25.0, total * 0.30), max(25.0, total * 0.55))
    small_deadline = t0 + small_time

    small_sizes = []
    for mult in (1, 2, 3, 4):
        n = base_n * mult
        if 1 <= n <= 256 and n not in small_sizes:
            small_sizes.append(n)
    small_sizes = sorted(set(small_sizes))

    for n in small_sizes:
        if time.time() >= small_deadline:
            break

        seeds = []
        seeds.append(upscale(parent, n))
        if len(best_h) != n:
            seeds.append(upscale(best_h, n))
        if n > base_n:
            seeds.append(upscale_linear(parent, n))
        if n <= 2 * base_n:
            seeds.append(smooth_seed(seeds[0]))
            seeds.append(add_noise(seeds[0], rng, 0.04))

        for seed in dedup_candidates(seeds, 1e-8):
            if time.time() >= small_deadline:
                break
            rem = small_deadline - time.time()
            tlim = min(45.0, max(8.0, rem * 0.5))
            if tlim < 6.0:
                break
            seeds.append(seed)
            try:
                h2, c2 = improve_small(seed, n, time_limit=tlim)
            except Exception:
                continue
            if c2 < best_c - 1e-12:
                best_h = project_box_sum(h2, 0.5 * float(n))
                best_c = exact_c5(best_h)

    high_deadline = min(t0 + total - 5.0, t0 + total * 0.93)
    high_deadline = max(high_deadline, t0 + 30.0)
    high_deadline = max(high_deadline, small_deadline + 10.0)

    current_n = int(best_h.size)
    cand = []
    ntry = max(256, current_n * 2)
    while ntry <= 8192:
        cand.append(int(ntry))
        ntry = int(round(ntry * 2.0))
        if ntry <= current_n:
            ntry = current_n + 1

    for m in (4, 8, 16, 32, 64, 96, 128, 192):
        nm = base_n * m
        if 256 < nm <= 8192:
            cand.append(int(nm))

    cand = sorted(set([x for x in cand if x > current_n and x <= 8192]))

    if len(cand) > 5:
        idxs = [0, len(cand) // 4, len(cand) // 2, (3 * len(cand)) // 4, len(cand) - 1]
        cand = [cand[i] for i in idxs]
        cand = sorted(set(cand))

    high_budget = max(0.0, high_deadline - time.time() - 30.0)
    if len(cand) > 0 and high_budget > 20.0:
        per_high = max(20.0, high_budget / float(len(cand)))
        for n in cand:
            if time.time() >= high_deadline - 25.0:
                break
            tlim = min(per_high, high_deadline - time.time() - 20.0)
            if tlim <= 12.0:
                break

            h0 = upscale(best_h, n)
            try:
                h2, c2 = improve_large(h0, n, time_limit=tlim)
            except Exception:
                h2, c2 = h0, exact_c5(h0)

            if c2 < best_c - 1e-12:
                best_h = project_box_sum(h2, 0.5 * float(n))
                best_c = exact_c5(best_h)

            if time.time() < high_deadline - 45.0 and n > base_n:
                h1 = upscale_linear(parent, n)
                if best_h.size == h1.size and float(np.max(np.abs(best_h - h1))) < 1e-8:
                    continue
                tlim2 = min(0.45 * tlim, high_deadline - time.time() - 20.0)
                if tlim2 > 18.0:
                    try:
                        h2b, c2b = improve_large(h1, n, time_limit=tlim2)
                    except Exception:
                        h2b, c2b = h1, exact_c5(h1)
                    if c2b < best_c - 1e-12:
                        best_h = project_box_sum(h2b, 0.5 * float(n))
                        best_c = exact_c5(best_h)

    final_deadline = t0 + total - 3.0
    rem = final_deadline - time.time()
    if rem > 10.0:
        try:
            best_h, best_c = polish_subgrad(best_h, max_iters=40, time_limit=rem)
        except Exception:
            pass

    best_h = sanitize_h_for_return(best_h)
    best_c = exact_c5(best_h)

    return best_h, float(best_c), int(best_h.size)
