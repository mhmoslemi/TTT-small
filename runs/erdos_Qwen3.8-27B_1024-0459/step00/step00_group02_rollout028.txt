def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for water-filling
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    # Exact fix
    d = target - h.sum()
    if abs(d) > 1e-14:
        m = (h > 1e-12) & (h < 1.0-1e-12)
        if m.any():
            h[m] += d / m.sum()
            h = np.clip(h, 0, 1)
    return h

    shifted = tau * (corr - corr.max())

    grad = np.zeros(n)

            grad[i] += (1.0-h[j]) * wk

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if m.any():
            h[m] += d / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, n, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = tau * (corr - np.max(corr))
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        jm = max(0, n - 1 - k)
        jx = min(n - 1, n - 1 + k)
        for j in range(jm, jx + 1):
            i = j + k - (n - 1)
            grad[j] -= h[i] * wk
            grad[i] += (1.0 - h[j]) * wk
    return grad * dx

def hard_grad(h, n, dx, active_k):
    grad = np.zeros(n)
    k = active_k
    jm = max(0, n - 1 - k)
    jx = min(n - 1, n - 1 + k)
    for j in range(jm, jx + 1):
        i = j + k - (n - 1)
        grad[j] -= h[i]
        grad[i] += (1.0 - h[j])
    return grad * dx

def optimize_smoothed(h0, n, dx, max_iters, time_limit, t0):
    h = h0.astype(float).copy()
    best_c5 = compute_c5(h, dx)
    best = h.copy()
    h_prev = h.copy()
    t_nest = 1.0
    step = 1.0 / (n * dx)
    no_imp = 0
    c5_cur = best_c5

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        progress = float(it) / max(1, max_iters)
        tau = 4.0 + 30.0 * progress

        if t_nest > 1.0:
            beta = (t_nest - 2.0) / (t_nest + 1.0)
            y = h + beta * (h - h_prev)
            y = project_feasible(y, n)
        else:
            y = h.copy()

        g = smooth_grad(y, n, dx, tau)
        h_new = project_feasible(y - step * g, n)
        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best = h_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        if c5_new > c5_cur + 1e-10:
            step *= 0.85
        elif no_imp > 200:
            step *= 0.95
            no_imp = 0
        else:
            step = min(step * 1.03, 3.0 / (n * dx))

        t_nest += 1.0
        h_prev = h.copy()
        h = h_new
        c5_cur = c5_new

    return best, best_c5

def optimize_hard(h0, n, dx, max_iters, time_limit, t0):
    h = h0.astype(float).copy()
    best_c5 = compute_c5(h, dx)
    best = h.copy()
    step = 0.3 / (n * dx)
    no_imp = 0

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corr_scaled = np.correlate(h, 1.0 - h, mode='full') * dx
        ak = int(np.argmax(corr_scaled))
        g = hard_grad(h, n, dx, ak)
        h_new = project_feasible(h - step * g, n)
        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best = h_new.copy()
            no_imp = 0
            step = min(step * 1.02, 2.0 / (n * dx))
        else:
            no_imp += 1
            if no_imp > 100:
                step *= 0.85
                no_imp = 0

        h = h_new

    return best, best_c5

def make_binary_like(n, rng, density_shift=0.0):
    h = np.zeros(n)
    target = n / 2.0
    ones_needed = int(target)
    frac = target - ones_needed
    idx = rng.permutation(n)
    for i in range(ones_needed):
        h[idx[i]] = 1.0
    if frac > 0.01:
        h[idx[ones_needed]] = frac
    h = project_feasible(h, n)
    return h

def make_smooth_pattern(n, rng):
    x = np.linspace(0, 2.0, n)
    freq = rng.integers(2, 6)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.4 * np.sin(freq * np.pi * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, n)
    return h

def interpolate_parent(parent, n):
    pn = len(parent)
    if pn == n:
        return project_feasible(parent.astype(float).copy(), n)
    xp = np.linspace(0.0, 2.0, pn) + 2.0 / pn / 2.0
    xn = np.linspace(0.0, 2.0, n) + 2.0 / n / 2.0
    h = np.interp(xn, xp, parent)
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    t_start = time.time()
    rng = np.random.default_rng(seed)
    time_limit_total = budget_s - 10.0

    parent = None
    try:
        if 'initial_h_values' in globals():
            parent = np.array(initial_h_values, dtype=float)
            parent = project_feasible(parent, len(parent))
    except Exception:
        parent = None

    best_h = None
    best_c5 = 1.0
    best_n = 45

    if parent is not None:
        pn = len(parent)
        dx_p = 2.0 / pn
        c5_p = compute_c5(parent, dx_p)
        best_h = parent.copy()
        best_c5 = c5_p
        best_n = pn

    resolutions = [45, 50, 60, 80, 100]
    if time_limit_total > 400:
        resolutions.append(120)

    for n in resolutions:
        if time.time() - t_start > time_limit_total:
            break
        dx = 2.0 / n
        remaining = time_limit_total - (time.time() - t_start)
        if remaining < 30:
            break

        inits = []
        if parent is not None:
            inits.append(interpolate_parent(parent, n))
        for _ in range(2):
            inits.append(make_binary_like(n, rng))
        for _ in range(2):
            inits.append(make_smooth_pattern(n, rng))
        inits.append(project_feasible(np.full(n, 0.5, dtype=float) + 0.3 * rng.standard_normal(n), n))

        n_inits = len(inits)
        time_per_init = max(5.0, (remaining - 10.0) / n_inits)
        max_iters = int(time_per_init * 80)

        for idx_init, h_init in enumerate(inits):
            if time.time() - t_start > time_limit_total - 20:
                break
            rem = time_limit_total - (time.time() - t_start)
            if rem < 10:
                break

            h_opt, c5_opt = optimize_smoothed(
                h_init, n, dx,
                max_iters=max(100, int(rem * 0.5 * 80)),
                time_limit=rem * 0.5,
                t0=t_start
            )

            rem = time_limit_total - (time.time() - t_start)
            if rem > 15 and c5_opt < best_c5:
                h_opt2, c5_opt2 = optimize_hard(
                    h_opt, n, dx,
                    max_iters=max(50, int(rem * 0.4 * 80)),
                    time_limit=rem * 0.4,
                    t0=t_start
                )
                if c5_opt2 < c5_opt:
                    h_opt, c5_opt = h_opt2, c5_opt2

            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n

    if best_h is not None and time.time() - t_start < time_limit_total - 5:
        dx_best = 2.0 / best_n
        rem = time_limit_total - (time.time() - t_start)
        h_final, c5_final = optimize_smoothed(
            best_h, best_n, dx_best,
            max_iters=max(50, int(rem * 0.7 * 100)),
            time_limit=rem * 0.7,
            t0=t_start
        )
        rem2 = time_limit_total - (time.time() - t_start)
        if rem2 > 5 and c5_final < best_c5:
            h_final2, c5_final2 = optimize_hard(
                h_final, best_n, dx_best,
                max_iters=max(20, int(rem2 * 0.7 * 100)),
                time_limit=rem2 * 0.7,
                t0=t_start
            )
            if c5_final2 < c5_final:
                h_final, c5_final = h_final2, c5_final2
        best_h = h_final
        best_c5 = c5_final

    if best_h is None:
        best_h = project_feasible(np.full(45, 0.5, dtype=float), 45)
        best_c5 = compute_c5(best_h, 2.0 / 45)
        best_n = 45

    return best_h, best_c5, best_n
