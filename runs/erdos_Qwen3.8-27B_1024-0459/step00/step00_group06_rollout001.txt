import numpy as np
import time

def objective_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.inf
    if not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr)) * dx
    if not np.isfinite(val):
        return np.inf
    return val

def project_box_sum(x, target):
    x = np.array(x, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if target <= 0.0:
        x.fill(0.0)
        return x
    if target >= float(n):
        x.fill(1.0)
        return x
    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if abs(s - target) <= 1e-12:
        return x
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        sy = float(y.sum())
        if sy > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        res = target - float(y.sum())
        if abs(res) <= 1e-12:
            break
        if res > 0.0:
            idx = int(np.argmin(y))
            add = min(res, 1.0 - y[idx])
        else:
            idx = int(np.argmax(y))
            add = max(res, y[idx])
        if abs(add) <= 1e-15:
            if res > 0.0:
                mask = (1.0 - y) > 1e-15
                cnt = int(np.count_nonzero(mask))
                if cnt == 0:
                    break
                y[mask] = np.minimum(1.0, y[mask] + res / float(cnt))
            else:
                mask = y > 1e-15
                cnt = int(np.count_nonzero(mask))
                if cnt == 0:
                    break
                y[mask] = np.maximum(0.0, y[mask] - (-res) / float(cnt))
            continue
        y[idx] += add
    return y

def quick_project_box_sum(x, target):
    y = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    n = y.size
    if n == 0:
        return y
    res = target - float(y.sum())
    if abs(res) <= 1e-12:
        return y
    step = res / float(n)
    if res > 0.0:
        if float(np.min(1.0 - y)) + 1e-15 >= step:
            y += step
            return y
    else:
        if float(np.min(y)) + 1e-15 >= -step:
            y += step
            return y
    return project_box_sum(y, target)

def make_parametric_h(n, kind='tent', seed=0):
    n = int(n)
    if n <= 0:
        n = 64
    target = 0.5 * float(n)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    if kind == 'inv':
        base = 1.0 - np.minimum(x, 2.0 - x)
    elif kind == 'cos':
        base = 0.5 - 0.5 * np.cos(np.pi * x)
    elif kind == 'ramp':
        base = 0.5 * x
    elif kind == 'ramp_down':
        base = 1.0 - 0.5 * x
    elif kind == 'combo':
        base = 0.5 * (np.minimum(x, 2.0 - x) + 0.5 - 0.5 * np.cos(np.pi * x))
    else:
        base = np.minimum(x, 2.0 - x)
    return project_box_sum(base, target)

def resize_h(h0, n):
    h0 = np.asarray(h0, dtype=float)
    m = h0.size
    if m == 0:
        return make_parametric_h(int(n), 'tent')
    if m == int(n):
        return project_box_sum(h0, 0.5 * float(int(n)))
    x_old = (np.arange(m, dtype=float) + 0.5) * (2.0 / float(m))
    x_new = (np.arange(int(n), dtype=float) + 0.5) * (2.0 / float(int(n)))
    y = np.interp(x_new, x_old, h0)
    return project_box_sum(y, 0.5 * float(int(n)))

def compute_S_raw(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    L = 2 * n - 1
    off = n - 1
    g = 1.0 - h
    S = np.empty(L, dtype=float)
    for di in range(L):
        d = di - off
        if d >= 0:
            m = n - d
            if m > 0:
                S[di] = np.dot(h[:m], g[d:d + m])
            else:
                S[di] = 0.0
        else:
            e = -d
            m = n - e
            S[di] = np.dot(h[e:e + m], g[:m])
    return S

def soft_c5_from_S(S, dx, alpha):
    if S.size == 0:
        return np.inf
    m = float(np.max(S))
    if alpha <= 1.0:
        return m * dx
    z = alpha * (S - m)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        return m * dx
    return float((m + np.log(s) / alpha) * dx)

def soft_weights(S, alpha):
    n = S.size
    if n == 0:
        return np.array([], dtype=float)
    m = float(np.max(S))
    if alpha <= 1.0:
        return np.full(n, 1.0 / float(n), dtype=float)
    z = alpha * (S - m)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=float)
    return e / s

def grad_S_raw(h, weights):
    h = np.asarray(h, dtype=float)
    n = h.size
    off = n - 1
    L = 2 * n - 1
    g = 1.0 - h
    G = np.zeros(n, dtype=float)
    for di in range(L):
        wd = float(weights[di])
        if wd == 0.0:
            continue
        d = di - off
        if d >= 0:
            m = n - d
            if m > 0:
                G[:m] += wd * g[d:d + m]
                G[d:] -= wd * h[:m]
        else:
            e = -d
            m = n - e
            G[e:] += wd * g[:m]
            G[:m] -= wd * h[e:e + m]
    return G

def local_search(h0, max_time, alphas=None):
    if alphas is None:
        alphas = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
    t0 = time.time()
    h = np.asarray(h0, dtype=float).copy()
    n = h.size
    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    h = project_box_sum(h, target)
    S = compute_S_raw(h)
    best_c = float(np.max(S)) * dx
    best_h = h.copy()
    for alpha in alphas:
        if time.time() > t0 + max_time - 5.0:
            break
        cur_soft = soft_c5_from_S(S, dx, alpha)
        stall = 0
        it = 0
        max_step = 0.25
        while it < 150 and stall < 30:
            if time.time() > t0 + max_time - 5.0:
                break
            weights = soft_weights(S, alpha)
            grad = grad_S_raw(h, weights) * dx
            grad = grad - np.mean(grad)
            norm = float(np.sqrt(np.mean(grad * grad)))
            if norm < 1e-12:
                break
            direction = -grad / norm
            adv = False
            step = max_step
            for _ in range(10):
                y = quick_project_box_sum(h + step * direction, target)
                Sy = compute_S_raw(y)
                softy = soft_c5_from_S(Sy, dx, alpha)
                if softy < cur_soft - 1e-12:
                    h = y
                    S = Sy
                    cur_soft = softy
                    adv = True
                    cy = float(np.max(Sy)) * dx
                    if cy < best_c - 1e-12:
                        best_c = cy
                        best_h = h.copy()
                    break
                step *= 0.5
            it += 1
            if adv:
                stall = 0
                max_step = min(0.6, max_step * 1.1)
            else:
                stall += 1
                max_step = max(0.005, max_step * 0.7)
    cval = objective_c5(best_h)
    if cval < best_c:
        best_c = cval
    return best_h, best_c

import numpy as np
import time


def objective_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.inf
    if not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = float(np.max(corr)) * dx
    if not np.isfinite(val):
        return np.inf
    return val


def project_box_sum(x, target):
    x = np.array(x, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if target <= 0.0:
        x.fill(0.0)
        return x
    if target >= float(n):
        x.fill(1.0)
        return x

    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if abs(s - target) <= 1e-12:
        return x

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        sy = float(y.sum())
        if sy > target:
            lo = mid
        else:
            hi = mid

    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(20):
        res = target - float(y.sum())
        if abs(res) <= 1e-12:
            break
        if res > 0.0:
            idx = int(np.argmin(y))
            add = min(res, 1.0 - y[idx])
        else:
            idx = int(np.argmax(y))
            add = max(res, y[idx])

        if abs(add) <= 1e-15:
            if res > 0.0:
                mask = (1.0 - y) > 1e-15
                cnt = int(np.count_nonzero(mask))
                if cnt == 0:
                    break
                y[mask] = np.minimum(1.0, y[mask] + res / float(cnt))
            else:
                mask = y > 1e-15
                cnt = int(np.count_nonzero(mask))
                if cnt == 0:
                    break
                y[mask] = np.maximum(0.0, y[mask] - (-res) / float(cnt))
            continue

        y[idx] += add

    return y


def quick_project_box_sum(x, target):
    y = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    n = y.size
    if n == 0:
        return y

    res = target - float(y.sum())
    if abs(res) <= 1e-12:
        return y

    step = res / float(n)
    if res > 0.0:
        if float(np.min(1.0 - y)) + 1e-15 >= step:
            y += step
            return y
    else:
        if float(np.min(y)) + 1e-15 >= -step:
            y += step
            return y

    return project_box_sum(y, target)


def make_parametric_h(n, kind='tent', seed=0):
    n = int(n)
    if n <= 0:
        n = 64

    target = 0.5 * float(n)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))

    if kind == 'inv':
        base = 1.0 - np.minimum(x, 2.0 - x)
    elif kind == 'cos':
        base = 0.5 - 0.5 * np.cos(np.pi * x)
    elif kind == 'ramp':
        base = 0.5 * x
    elif kind == 'ramp_down':
        base = 1.0 - 0.5 * x
    elif kind == 'combo':
        base = 0.5 * (np.minimum(x, 2.0 - x) + 0.5 - 0.5 * np.cos(np.pi * x))
    else:
        base = np.minimum(x, 2.0 - x)

    return project_box_sum(base, target)


def resize_h(h0, n):
    h0 = np.asarray(h0, dtype=float)
    m = h0.size
    n = int(n)

    if m == 0:
        return make_parametric_h(n, 'tent')
    if m == n:
        return project_box_sum(h0, 0.5 * float(n))

    x_old = (np.arange(m, dtype=float) + 0.5) * (2.0 / float(m))
    x_new = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    y = np.interp(x_new, x_old, h0)
    return project_box_sum(y, 0.5 * float(n))


def compute_S_raw(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    L = 2 * n - 1
    off = n - 1
    g = 1.0 - h
    S = np.empty(L, dtype=float)

    for di in range(L):
        d = di - off
        if d >= 0:
            m = n - d
            if m > 0:
                S[di] = np.dot(h[:m], g[d:d + m])
            else:
                S[di] = 0.0
        else:
            e = -d
            m = n - e
            S[di] = np.dot(h[e:e + m], g[:m])

    return S


def soft_c5_from_S(S, dx, alpha):
    if S.size == 0:
        return np.inf
    m = float(np.max(S))
    if alpha <= 1.0:
        return m * dx

    z = alpha * (S - m)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        return m * dx
    return float((m + np.log(s) / alpha) * dx)


def soft_weights(S, alpha):
    n = S.size
    if n == 0:
        return np.array([], dtype=float)

    m = float(np.max(S))
    if alpha <= 1.0:
        return np.full(n, 1.0 / float(n), dtype=float)

    z = alpha * (S - m)
    z = np.clip(z, -700.0, 700.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        return np.full(n, 1.0 / float(n), dtype=float)
    return e / s


def grad_S_raw(h, weights):
    h = np.asarray(h, dtype=float)
    n = h.size
    off = n - 1
    L = 2 * n - 1
    g = 1.0 - h
    G = np.zeros(n, dtype=float)

    for di in range(L):
        wd = float(weights[di])
        if wd == 0.0:
            continue

        d = di - off
        if d >= 0:
            m = n - d
            if m > 0:
                G[:m] += wd * g[d:d + m]
                G[d:] -= wd * h[:m]
        else:
            e = -d
            m = n - e
            G[e:] += wd * g[:m]
            G[:m] -= wd * h[e:e + m]

    return G


def local_search(h0, max_time, alphas=None):
    if alphas is None:
        alphas = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0]

    t0 = time.time()
    h = np.asarray(h0, dtype=float).copy()
    n = h.size
    target = 0.5 * float(n)
    dx = 2.0 / float(n)

    h = project_box_sum(h, target)
    S = compute_S_raw(h)
    best_c = float(np.max(S)) * dx
    best_h = h.copy()

    for alpha in alphas:
        if time.time() > t0 + max_time - 5.0:
            break

        cur_soft = soft_c5_from_S(S, dx, alpha)
        stall = 0
        it = 0
        max_step = 0.25

        while it < 150 and stall < 30:
            if time.time() > t0 + max_time - 5.0:
                break

            weights = soft_weights(S, alpha)
            grad = grad_S_raw(h, weights) * dx
            grad = grad - np.mean(grad)
            norm = float(np.sqrt(np.mean(grad * grad)))
            if norm < 1e-12:
                break

            direction = -grad / norm
            adv = False
            step = max_step

            for _ in range(10):
                y = quick_project_box_sum(h + step * direction, target)
                Sy = compute_S_raw(y)
                softy = soft_c5_from_S(Sy, dx, alpha)

                if softy < cur_soft - 1e-12:
                    h = y
                    S = Sy
                    cur_soft = softy
                    adv = True
                    cy = float(np.max(Sy)) * dx
                    if cy < best_c - 1e-12:
                        best_c = cy
                        best_h = h.copy()
                    break

                step *= 0.5

            it += 1

            if adv:
                stall = 0
                max_step = min(0.6, max_step * 1.1)
            else:
                stall += 1
                max_step = max(0.005, max_step * 0.7)

    cval = objective_c5(best_h)
    if cval < best_c:
        best_c = cval

    return best_h, best_c
