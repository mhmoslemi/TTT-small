    h_new = project_to_constraint(h_new, target)
    return h_new

def soft_max_objective_and_grad(h, beta, dx):
    n = len(h)
    corr = compute_corr(h)
    c = corr * dx
    m = np.max(c)
    exps = np.exp(np.clip(beta * (c - m), -500, 500))
    se = np.sum(exps)
    val = m + (1.0 / beta) * np.log(se)
    w = exps / se

    grad = np.zeros(n)
    for a in range(-(n - 1), n):
        w_a = w[a + n - 1]
        if w_a < 1e-14:
            continue
        m_lo = max(0, a)
        m_hi = min(n - 1, a + n - 1)
        j_lo = m_lo - a
        j_hi = m_hi - a
        grad[m_lo:m_hi + 1] += w_a * (1.0 - h[j_lo:j_hi + 1])

        m_lo2 = max(0, -a)
        m_hi2 = min(n - 1, n - 1 - a)
        j_lo2 = m_lo2 + a
        j_hi2 = m_hi2 + a
        grad[m_lo2:m_hi2 + 1] -= w_a * h[j_lo2:j_hi2 + 1]

    grad *= dx
    return val, grad

def optimize_resolution(h_start, n_pts, target_sum, dx, t_budget, time_fn):
    h = h_start.copy()
    best_c5_loc = compute_c5(h)
    
    bounds = [(0.0, 1.0)] * n_pts
    
    for beta in [5.0, 15.0, 40.0, 100.0, 250.0, 500.0]:
        elapsed = time_fn()
        if elapsed > t_budget:
            break
        
        def obj(x):
            v, _ = soft_max_objective_and_grad(x, beta, dx)
            return v
        
        def obj_grad(x):
            v, g = soft_max_objective_and_grad(x, beta, dx)
            return v, g
        
        maxiter = int(min(800, max(50, (t_budget - elapsed) * 1.5)))
        
        res = minimize(obj_grad, h, method='L-BFGS-B',
                       jac=True, bounds=bounds,
                       constraints=[{'type': 'eq',
                                     'fun': lambda x: np.sum(x) - target_sum,
                                     'jac': lambda x: np.ones(n_pts)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-9, 'maxcor': 20, 'maxls': 20})
        
        elapsed = time_fn()
        if elapsed > t_budget:
            break
        
        h_new = np.clip(res.x, 0, 1)
        h_new = project_to_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5_loc:
            best_c5_loc = c5_new
            h = h_new.copy()
    
    return h, best_c5_loc

def try_binary_h(n_pts):
    """Try to find a good binary solution via random search."""
    best_c5 = 1e9
    best_h = None
    target = n_pts // 2
    half = n_pts // 2
    
    for _ in range(50):
        h = np.zeros(n_pts)
        idx = np.random.choice(n_pts, size=half, replace=False)
        h[idx] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def time_fn():
        return time.time() - t0
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, AttributeError, TypeError):
        h_init = None
    
    if h_init is None or len(h_init) == 0:
        n = 48
        h_init = np.zeros(n)
        h_init[::2] = 1.0
    
    n_init = len(h_init)
    
    # Ensure valid
    h_init = np.clip(h_init, 0, 1)
    target_init = n_init / 2.0
    s_init = np.sum(h_init)
    if abs(s_init - target_init) > 1e-8:
        h_init = project_to_constraint(h_init, target_init)
    
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    best_n = n_init
    
    # Strategy 1: Optimize at the initial resolution
    if n_init >= 16:
        dx_init = 2.0 / n_init
        h_opt, c5_opt = optimize_resolution(h_init, n_init, n_init / 2.0, dx_init,
                                            budget_s * 0.3, time_fn)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_init
    
    # Strategy 2: Try various even resolutions
    resolutions = [48, 50, 52, 56, 60, 64, 72, 80, 96, 112, 128, 160, 200]
    # Deduplicate and sort
    resolutions = sorted(set(r for r in resolutions if r % 2 == 0 and r >= 32))
    
    for n_pts in resolutions:
        elapsed = time_fn()
        if elapsed > budget_s - 120:
            break
        
        target_sum = n_pts / 2.0
        dx = 2.0 / n_pts
        
        # Initialize: interpolate current best
        if n_pts == best_n:
            h_start = best_h.copy()
        else:
            h_start = interpolate_h(best_h, n_pts)
        
        h_opt, c5_opt = optimize_resolution(h_start, n_pts, target_sum, dx,
                                            budget_s - 100, time_fn)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_pts
    
    # Strategy 3: Try binary solutions at a few resolutions
    for n_b in [48, 56, 64, 80, 96]:
        elapsed = time_fn()
        if elapsed > budget_s - 60:
            break
        h_bin, c5_bin = try_binary_h(n_b)
        if h_bin is not None and c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
            best_n = n_b
    
    # Strategy 4: Local refinement of the best solution found
    elapsed = time_fn()
    if elapsed < budget_s - 60:
        n_best = best_n
        dx_best = 2.0 / n_best
        target_best = n_best / 2.0
        h_ref, c5_ref = optimize_resolution(best_h, n_best, target_best, dx_best,
                                            budget_s - 30, time_fn)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_n = n_best
    
    # Strategy 5: Try a structured initialization (blocks of 0 and 1)
    elapsed = time_fn()
    if elapsed < budget_s - 40:
        for n_s in [48, 64, 80, 96]:
            if time_fn() > budget_s - 30:
                break
            # Create block pattern
            h_blk = np.zeros(n_s)
            block_size = n_s // 8
            for b in range(0, n_s, block_size * 2):
                h_blk[b:b + block_size] = 1.0
            # Adjust to meet constraint
            h_blk = project_to_constraint(h_blk, n_s / 2.0)
            
            dx_s = 2.0 / n_s
            h_opt_s, c5_s = optimize_resolution(h_blk, n_s, n_s / 2.0, dx_s,
                                                budget_s - 20, time_fn)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_opt_s
                best_n = n_s
    
    # Final: ensure exact constraint satisfaction
    target_final = best_n / 2.0
    s_final = np.sum(best_h)
    if abs(s_final - target_final) > 1e-10:
        best_h = project_to_constraint(best_h, target_final)
    
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, best_n)
