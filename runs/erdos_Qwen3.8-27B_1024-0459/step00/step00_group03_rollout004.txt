def get_weights_and_grad(h, beta):
    n = len(h)
    V = lags_from_h(h)
    m = V.size
    if beta <= 1e-9:
        w = np.full(m, 1.0 / m)
    else:
        M = float(np.max(V))
        e = np.exp(beta * (V - M))
        Z = float(np.sum(e))
        if Z > 0.0 and np.isfinite(Z):
            w = e / Z
        else:
            w = np.zeros(m)
            w[int(np.argmax(V))] = 1.0
    A = np.zeros(n)
    B = np.zeros(n)
    A[0] = w[0]
    if n > 1:
        A[1:] = w[1::2]
        B[1:] = w[2::2]
    if n == 1:
        return A[0] * (1.0 - 2.0 * h)
    K = A + B
    K[0] = 0.0
    past = np.convolve(h, K, mode='full')[:n]
    future = np.convolve(h, K[::-1], mode='full')[n - 1:2 * n - 1]
    prefixA = np.zeros(n)
    prefixA[1:] = np.cumsum(A[1:])
    prefixB = np.zeros(n)
    prefixB[1:] = np.cumsum(B[1:])
    grad = A[0] - 2.0 * A[0] * h + prefixA + prefixB[n - 1::-1] - past - future
    return grad
```

import numpy as np
import time


def make_even(n):
    n = int(n)
    if n < 10:
        n = 10
    if n % 2 != 0:
        n += 1
    return n


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n = v.size
    target = float(target)
    if n == 0:
        return v
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    x = np.clip(v, 0.0, 1.0)
    s = float(x.sum())
    if abs(s - target) <= 64.0 * np.finfo(float).eps * max(1, n):
        return x

    mn = float(np.min(v))
    mx = float(np.max(v))
    if mn >= -1e-8 and mx <= 1.0 + 1e-8 and abs(s - target) <= 1e-5:
        r = target - s
        if abs(r) > 1e-12:
            interior = (x > 1e-12) & (x < 1.0 - 1e-12)
            if np.any(interior):
                x = x.copy()
                x[interior] += r / float(interior.sum())
                x = np.clip(x, 0.0, 1.0)
        return x

        if s > target:
            hi = mid
        else:
            lo = mid

    x = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-12:

def lags_from_h(h):
    n = len(h)
    full = np.correlate(h, 1.0 - h, mode="full")
    if n == 1:
        return np.array([full[0]], dtype=float)
    V = np.empty(1 + 2 * (n - 1), dtype=float)
    V[0] = full[n - 1]
    V[1::2] = full[n:]
    V[2::2] = full[n - 2::-1]
    return V


def objective_direct(h):
    try:
        n = len(h)
        if n == 0:
            return 0.0
        full = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(full) * (2.0 / float(n)))
    except Exception:
        return np.inf


def get_weights_and_grad(h, beta):
    n = len(h)
    V = lags_from_h(h)
    m = V.size

    if beta <= 1e-9:
        w = np.full(m, 1.0 / float(m))
    else:
        M = float(np.max(V))
        e = np.exp(beta * (V - M))
        Z = float(np.sum(e))
        if Z > 0.0 and np.isfinite(Z):
            w = e / Z
        else:
            w = np.zeros(m)
            w[int(np.argmax(V))] = 1.0

    A = np.zeros(n)
    B = np.zeros(n)
    A[0] = w[0]
    if n > 1:
        A[1:] = w[1::2]
        B[1:] = w[2::2]

    if n == 1:
        return A[0] * (1.0 - 2.0 * h)

    K = A + B
    K[0] = 0.0

    past = np.convolve(h, K, mode="full")[:n]
    future = np.convolve(h, K[::-1], mode="full")[n - 1:2 * n - 1]

    prefixA = np.zeros(n)
    prefixA[1:] = np.cumsum(A[1:])
    prefixB = np.zeros(n)
    prefixB[1:] = np.cumsum(B[1:])

    grad = A[0] - 2.0 * A[0] * h + prefixA + prefixB[n - 1::-1] - past - future
    return grad


def make_triangle(n):
    n = make_even(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 1.0 - np.abs(x - 1.0)
    return project_box_sum(h, 0.5 * n)


def make_wave(n, m, amp, phase):
    n = make_even(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 0.5 + float(amp) * np.cos(np.pi * float(m) * x + float(phase))

def make_wave_best(n, rng=None, phases=12, amp_values=None):

def random_swap_move(h, rng):

def random_blend_move(h, rng):

                    h = project_box_sum(h + rng.normal(0.0, 0.01, n), target)
                    cur = objective_direct(h)

    best_h = project_box_sum(best_h, target)
    return best_h, objective_direct(best_h)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    budget = max(1.0, float(budget_s))
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(initial_h_values, dtype=float)
    except NameError:
        parent = None

    if parent is None or parent.size == 0:
        parent = make_default_parent(80, rng)

    parent = np.asarray(parent, dtype=float)
    best_h = None
    best_val = np.inf

    hp = project_box_sum(parent, 0.5 * parent.size)
    vp = objective_direct(hp)
    if np.isfinite(vp):
        best_h = hp.copy()
        best_val = vp

    if budget < 15.0 or best_h is None:
        if best_h is None:
            best_h = make_triangle(80)
            best_val = objective_direct(best_h)
        return best_h, float(best_val), int(best_h.size)

    base = parent.size
    mults = (1, 2, 3, 4)
    if budget > 700.0:
        mults = (1, 2, 3, 4, 5)

    n_list = []
    for mult in mults:
        n = make_even(base * mult)
        if n <= 450 and n not in n_list:
            n_list.append(n)

    if not n_list:
        n_list = [80, 160, 240]

    if budget < 250.0 and len(n_list) > 2:
        n_list = n_list[:2]
    if budget < 120.0 and len(n_list) > 1:
        n_list = n_list[:1]

    plans = []
    for idx, n in enumerate(n_list):
        if idx == 0:
            w = 1.4
        elif idx == 1:
            w = 1.2
        elif idx == 2:
            w = 1.0
        else:
            w = 0.7
        plans.append((n, "base", w))

    if n_list:
        wave_n = n_list[min(1, len(n_list) - 1)]
        plans.append((wave_n, "wave", 0.9))

    if budget < 120.0:
        plans = [(n_list[0], "base", 1.0)]

    total_w = 0.0
    for plan in plans:
        total_w += plan[2]
    if total_w <= 0.0:
        total_w = 1.0

    cum = 0.0
    for plan in plans:
        n, typ, w = plan
        cum += w
        deadline = start + budget * (cum / total_w) - 3.0
        if time.time() >= max(start + 5.0, deadline - 2.0):
            break

        if typ == "base":
            if best_h is not None:
                start_h = interpolate_to(best_h, n)
            else:
                start_h = interpolate_to(parent, n)
        else:
            start_h, _ = make_wave_best(n, rng, phases=12, amp_values=(0.5, 0.45))

        start_h = project_box_sum(start_h, 0.5 * n)
        vs = objective_direct(start_h)
        if np.isfinite(vs) and vs < best_val:
            best_val = vs
            best_h = start_h.copy()

        h_opt, v_opt = refine(start_h, deadline, rng)
        if np.isfinite(v_opt) and v_opt < best_val:
            best_val = v_opt
            best_h = h_opt.copy()

    if best_h is None:
        best_h = make_triangle(80)
        best_val = objective_direct(best_h)

    best_h = project_box_sum(best_h, 0.5 * best_h.size)
    n = best_h.size

    if n % 2 == 0 and time.time() < start + budget - 5.0:
        target = int(n // 2)
        if 0 < target < n:
            hb = np.zeros(n, dtype=float)
            hb[np.argsort(-best_h)[:target]] = 1.0
            vb = objective_direct(hb)
            if np.isfinite(vb) and vb < best_val:
                best_val = vb
                best_h = hb.copy()
                best_h = project_box_sum(best_h, 0.5 * n)

    if time.time() < start + budget - 35.0:
        h_opt, v_opt = refine(
            best_h,
            start + budget - 1.0,
            rng,
            betas=(10.0, 30.0, 80.0, 200.0, 500.0),
        )
        if np.isfinite(v_opt) and v_opt < best_val:
            best_val = v_opt
            best_h = h_opt.copy()

    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    return best_h, float(best_val), int(len(best_h))
