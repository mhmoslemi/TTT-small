        else:
            print(f"N={N_new}: C5={c5_opt:.6f} (no improvement, best={best_c5:.6f})")

        # Update current for next interpolation
        h_current = h_opt
        N_current = N_new

    # Binarization attempt
    t_elapsed = time.time() - t_start
    if budget_s - t_elapsed > 20:
        h_bin = try_binarize(h_current, N_current)
        c5_bin = compute_c5(h_bin, N_current)
        if c5_bin < best_c5:
            # Polish the binary solution
            h_bin, c5_bin = coordinate_descent_polish(h_bin, N_current, max_passes=5, time_limit=15.0)
            if c5_bin < best_c5:
                best_h, best_c5, best_N = h_bin.copy(), c5_bin, N_current
                print(f"Binary + CD N={N_current}: C5={best_c5:.6f}")
        else:
            print(f"Binary N={N_current}: C5={c5_bin:.6f} (no improvement)")

    # Coordinate descent polish on best solution
    t_elapsed = time.time() - t_start
    if budget_s - t_elapsed > 15:
        h_cd, c5_cd = coordinate_descent_polish(best_h, best_N, max_passes=5, time_limit=10.0)
        if c5_cd < best_c5:
            best_h, best_c5, best_N = h_cd.copy(), c5_cd, best_N
            print(f"After CD N={best_N}: C5={best_c5:.6f}")

    # Try one more optimization pass at the best resolution if time allows
    t_elapsed = time.time() - t_start
    if budget_s - t_elapsed > 30:
        h_final, c5_final = optimize_level(best_h, best_N, min(40, budget_s - t_elapsed - 5), seed_offset=999)
        if c5_final < best_c5:
            best_h, best_c5, best_N = h_final.copy(), c5_final, best_N
            print(f"Final opt N={best_N}: C5={best_c5:.6f}")

    # Ensure sum constraint is exactly met
    best_h = project_feasible(best_h, best_N)
    final_c5 = compute_c5(best_h, best_N)
    print(f"FINAL: N={best_N}, C5={final_c5:.6f}")

    return (best_h, final_c5, best_N)
