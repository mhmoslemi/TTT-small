    def __init__(self, n, tau=0.05):

        self.dx = 2.0 / n

        return self._eval(h)[0]

    def gradient(self, h):
        return self._eval(h)[1]

    def constraint_fun(self, h):
        return np.sum(h) - self.n / 2.0

    def constraint_jac(self, h):
        return np.ones(self.n)

    def _eval(self, h):
        n = self.n
        dx = self.dx
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        max_val = np.max(corr)

        shifted = corr - max_val
        weights = np.exp(shifted / self.tau)
        tw = np.sum(weights)
        if tw < 1e-30:
            weights = np.zeros(2 * n - 1)
            weights[np.argmax(corr)] = 1.0

                method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-12}
            )
            h_opt = fix_constraint(res.x, n_best)
            c5_opt = compute_c5(h_opt, n_best)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = n_best
        except Exception:
            pass

    # Phase 5: Final polish - run SLSQP again from current best with tight tol
    if remaining() > 10:
        opt = ErdosOpt(best_n, tau=0.02)
        bounds = [(0.0, 1.0)] * best_n
        constraints = [{'type': 'eq', 'fun': opt.constraint_fun, 'jac': opt.constraint_jac}]
        try:
            res = minimize(
                fun=opt.objective, x0=best_h.copy(), jac=opt.gradient,
                method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-15}
            )
            h_opt = fix_constraint(res.x, best_n)
            c5_opt = compute_c5(h_opt, best_n)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = best_n
        except Exception:
            pass

    # Ensure valid output
    best_h = fix_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
