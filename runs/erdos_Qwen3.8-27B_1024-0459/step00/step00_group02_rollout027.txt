            h_init = interpolate_to(h_cur, n_cur, n)
        elif h_cur is not None:
            h_init = h_cur.copy()
        else:
            h_init = rng.rand(n)
        
        # Convert to binary
        h_bin = to_binary(h_init, n, rng)
        c5_init = compute_c5(h_bin)
        
        if c5_init < best_c5:
            best_c5 = c5_init
            best_h = h_bin.copy()
        
        # Multiple restarts within this phase
        remaining = phase_deadline - time.time()
        if remaining < 20.0:
            continue
        
        n_restarts = 1
        if remaining > 300:
            n_restarts = 3
        elif remaining > 150:
            n_restarts = 2
        
        time_per_restart = (remaining - 5.0) / n_restarts
        
        for restart in range(n_restarts):
            if time.time() > deadline - 10.0:
                break
            restart_deadline = min(time.time() + time_per_restart - 2.0, deadline - 10.0)
            if restart_deadline <= time.time():
                break
            
            if restart == 0:
                h_try = h_bin.copy()
            else:
                # Perturb the best from previous restart
                h_try = h_bin.copy()
                n_flip = max(5, n // 20)
                ones = np.where(h_try == 1)[0]
                zeros = np.where(h_try == 0)[0]
                f2z = rng.choice(ones, size=min(n_flip, len(ones)), replace=False)
                f2o = rng.choice(zeros, size=min(n_flip, len(zeros)), replace=False)
                h_try[f2z] = 0.0
                h_try[f2o] = 1.0
            
            # Single-swap local search (main workhorse)
            t_single_floor = restart_deadline - 0.1 * (restart_deadline - time.time())
            h_opt, c5_opt = local_search_binary(h_try, t_single_floor, rng)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            
            # Double-swap for additional improvement
            if time.time() < restart_deadline - 5.0:
                h_opt2, c5_opt2 = local_search_double_swap(h_opt, restart_deadline - 2.0, rng)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()
            
            h_bin = h_opt.copy()
        
        h_cur = h_bin.copy()
        n_cur = n
    
    # Final polish: try one more round at the best resolution
    if time.time() < deadline - 15.0 and best_h is not None:
        n_best = len(best_h)
        if n_best % 2 == 0:
            h_polish = to_binary(best_h, n_best, rng)
            h_final, c5_final = local_search_binary(h_polish, deadline - 5.0, rng)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    
    # Fallback
    if best_h is None:
        n = 180
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
