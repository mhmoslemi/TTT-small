import numpy as np
import time

def project_to_sum(h, target_sum):
    """Project h onto {h in [0,1]^n : sum(h) = target_sum} via water-filling."""
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        free = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.any(free):
            result[free] += diff / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
        else:
            idx = np.argmax(result) if diff > 0 else np.argmin(result)
            result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, k) * dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_gradient(h, alpha):
    """Compute gradient of smooth approximation to max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    total_w = np.sum(exp_vals)
    if total_w < 1e-300:
        weights = np.zeros(2 * n - 1)
        weights[int(np.argmax(corr))] = 1.0
    else:
        weights = exp_vals / total_w

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            L = n - shift
            if L <= 0:
                continue
            grad[:L] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:L]
        else:
            s = -shift
            L = n - s
            if L <= 0:
                continue
            grad[s:] += w * (1.0 - h[:L])
            grad[:L] -= w * h[s:]
    return grad

def optimize_gradient_descent(h, dx, target_sum, max_time, alpha):
    """Projected gradient descent on the overlap objective."""
    n = len(h)
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    lr = 0.08
    min_lr = 1e-7
    no_improve = 0
    max_no_improve = 800
    momentum = 0.0
    mu = 0.9

    for it in range(8000):
        if time.time() - start_time > max_time:
            break

        grad = compute_gradient(h, alpha)
        grad_proj = grad - np.mean(grad)

        momentum = mu * momentum - lr * grad_proj
        h_new = h + momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)

        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            no_improve += 1
            if c5_new > best_c5 + 0.02:
                h = best_h.copy()
                momentum = 0.0
            else:
                h = h_new
            lr *= 0.97
            mu *= 0.999

        if lr < min_lr or no_improve > max_no_improve:
            break

    return best_h, best_c5

def interpolate_h(h_old, n_new):
    """Interpolate h from old grid to new grid size."""
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def random_perturb(h, target_sum, sigma):
    """Add random noise and re-project."""
    n = len(h)
    noise = np.random.randn(n) * sigma
    h_pert = h + noise
    h_pert = np.clip(h_pert, 0.0, 1.0)
    h_pert = project_to_sum(h_pert, target_sum)
    return h_pert
