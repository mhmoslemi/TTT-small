def allocate_budget(remaining, ideal, frac, min_alloc):
    remaining = float(remaining)

def clean_array(h):
    h = np.asarray(h, dtype=float)
    h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    return h


def project_to_target(h, target):
    h = clean_array(h)
    target = float(target)
    n = h.size
    if n == 0:
        return h
    if target <= 0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)

    if np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12):
        hc = np.clip(h, 0.0, 1.0)
        if abs(float(hc.sum()) - target) <= 1e-10:
            res = target - float(hc.sum())
            if abs(res) > 1e-14:
                if res > 0:
                    mask = hc < 1.0
                else:
                    mask = hc > 0.0
                if np.any(mask):
                    hc[mask] += res / float(np.count_nonzero(mask))
                    hc = np.clip(hc, 0.0, 1.0)

def score_c5(h, return_corr=False):
    h = clean_array(h)
    n = h.size
    dx = 2.0 / float(n)
    g = 1.0 - h
    if n <= 200:
        corr = np.correlate(h, g, mode="full")
    else:
        need = 2 * n - 1
        L = 1
        while L < need:
            L <<= 1
        fa = np.fft.rfft(h, L)
        fb = np.fft.rfft(g[::-1], L)
        corr = np.fft.irfft(fa * fb, L)[:need]
    c = float(dx * np.max(corr))
    if return_corr:
        return c, dx * corr
    return c


def active_indices(A, kmax=20, tol_frac=0.15):
    A = np.asarray(A, dtype=float)
    m = A.size
    c = float(np.max(A))
    if not np.isfinite(c) or c <= 1e-14:
        return np.array([m // 2], dtype=int)
    tol = max(1e-10, tol_frac * c)
    idx = np.where(A >= c - tol)[0]
    if idx.size == 0:

    if idx.size > kmax:
        top = np.argsort(A)[::-1][:kmax]
        idx = np.asarray(top, dtype=int)
    else:
        idx = np.asarray(idx, dtype=int)
    return idx


def gradient_active(h, idxs, dx, weights=None):
    h = np.asarray(h, dtype=float)
    n = h.size
    center = n - 1
    grad = np.zeros(n, dtype=float)
    idxs = np.asarray(idxs, dtype=int)
    k = idxs.size
    if k == 0:
        return grad
    if weights is None:
        w = np.full(k, 1.0 / float(k), dtype=float)
    else:
        w = np.asarray(weights, dtype=float)
        sw = float(w.sum())
        if sw <= 0 or not np.isfinite(sw):
            w = np.full(k, 1.0 / float(k), dtype=float)
        else:
            w = w / sw
    for ii in range(k):
        idx = int(idxs[ii])
        if idx < 0 or idx >= 2 * n - 1:
            continue
        r = idx - center
        wi = float(w[ii])
        if wi == 0.0:
            continue
        s = wi * dx
        if r == 0:
            grad += s * (1.0 - 2.0 * h)
        elif r > 0:
            grad[r:] += s * (1.0 - h[:-r])
            grad[:-r] -= s * h[r:]
        else:
            t = -r
            grad[:n - t] += s * (1.0 - h[t:])
            grad[t:] -= s * h[:n - t]
    return grad


def weighted_gradient(h, A, active, nu):
    active = np.asarray(active, dtype=int)
    if active.size == 0:
        return np.zeros(h.size, dtype=float)
    aact = np.asarray(A[active], dtype=float)
    mx = float(np.max(aact))
    mn = float(np.min(aact))
    nu = max(1e-10, float(nu))
    if mx - mn <= 1e-12:
        w = np.full(active.size, 1.0 / float(active.size), dtype=float)
    else:
        y = (aact - mx) / nu
        y = y - float(np.max(y))
        e = np.exp(y)
        sw = float(e.sum())
        if sw <= 0.0 or not np.isfinite(sw):
            w = np.full(active.size, 1.0 / float(active.size), dtype=float)
        else:
            w = e / sw
    return gradient_active(h, active, 2.0 / float(h.size), w)


def lp_step(h, A, active, delta_cap, target):
    h = np.asarray(h, dtype=float)
    n = h.size
    active = np.asarray(active, dtype=int)
    active = active[(active >= 0) & (active < 2 * n - 1)]
    if active.size == 0:
        return None
    rows = []
    rhs = []
    dx = 2.0 / float(n)
    for idx in active:
        g = gradient_active(h, np.array([int(idx)], dtype=int), dx, None)
        row = np.empty(n + 1, dtype=float)
        row[0] = -1.0
        row[1:] = g
        rows.append(row)
        rhs.append(-float(A[idx]))
    if not rows:
        return None
    A_ub = np.vstack(rows)
    b_ub = np.asarray(rhs, dtype=float)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1, dtype=float)
    bounds = [(-2.0, 2.0)]
    cap = float(delta_cap)
    for i in range(n):
        hi = min(1.0 - float(h[i]), cap)
        lo = max(-float(h[i]), -cap)
        if hi < -1e-12 or lo > 1e-12:
            return None
        bounds.append((lo, hi))
    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[0] = 1.0
    try:
        try:
            res = sopt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                               b_eq=b_eq, bounds=bounds, method="highs",
                               options={"maxiter": 2000})
        except Exception:
            res = sopt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq,
                               b_eq=b_eq, bounds=bounds)
    except Exception:
        return None
    if getattr(res, "status", 1) != 0:
        return None
    delta = np.asarray(res.x[1:], dtype=float)
    if not np.all(np.isfinite(delta)):
        return None
    return project_to_target(h + delta, target)

def upsample_continuous(h, n_new):
    h = clean_array(h)
    n_old = h.size
    n_new = int(n_new)
    if n_new <= 0:
        return np.zeros(max(1, n_new), dtype=float)
    if n_old == n_new:
        return project_to_target(h, 0.5 * n_new)
    x_old = (np.arange(n_old, dtype=float) + 0.5) * (2.0 / float(n_old))
    x_new = (np.arange(n_new, dtype=float) + 0.5) * (2.0 / float(n_new))
    h_new = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return project_to_target(h_new, 0.5 * n_new)


def binary_from_float(h):
    h = clean_array(h)
    n = h.size
    target = 0.5 * n
    k = int(math.floor(target))
    b = np.zeros(n, dtype=float)
    if k > 0:
        order = np.argsort(-h)
        b[order[:k]] = 1.0
    if 2 * k != n:
        order = np.argsort(-h)
        if k < n:
            b[order[k]] = 0.5
    return project_to_target(b, target)


def binary_fix_counts(b, target, rng):
    b = np.array(clean_array(b), dtype=float, copy=True)
    target = int(target)
    n = b.size
    if target <= 0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)
    b[b > 0.5] = 1.0
    b[b < 0.5] = 0.0
    b[np.abs(b - 0.5) < 1e-12] = 0.0
    s = int(round(float(b.sum())))
    need = target - s
    if need > 0:
        zeros = np.where(b < 0.5)[0]
        if zeros.size >= need:
            idx = rng.choice(zeros, size=need, replace=False)
            b[idx] = 1.0
        else:
            b[zeros] = 1.0
            s = int(round(float(b.sum())))
            need = target - s
            if need > 0:
                order = np.argsort(-b)
                b[order[:need]] = 1.0
    elif need < 0:
        ones = np.where(b > 0.5)[0]
        if ones.size >= -need:
            idx = rng.choice(ones, size=-need, replace=False)
            b[idx] = 0.0
        else:
            b[ones] = 0.0
            s = int(round(float(b.sum())))
            need = target - s
            if need < 0:
                order = np.argsort(b)
                b[order[: -need]] = 0.0
    b = np.clip(b, 0.0, 1.0)
    res = target - float(b.sum())
    if abs(res) > 1e-12:
        if res > 0:
            order = np.argsort(1.0 - b)
            cnt = min(int(math.ceil(res)), b.size)
            b[order[:cnt]] = np.minimum(1.0, b[order[:cnt]] + 1.0)
        else:
            order = np.argsort(b)
            cnt = min(int(math.ceil(-res)), b.size)
            b[order[:cnt]] = np.maximum(0.0, b[order[:cnt]] - 1.0)
    return b


def thue_binary(n):
    b = np.zeros(n, dtype=float)
    for i in range(n):
        b[i] = 1.0 if (bin(i).count("1") & 1) else 0.0
    return b


def kronecker_binary(n, seed, alpha):
    n = int(n)
    if n <= 1:
        return np.zeros(n, dtype=float)
    shift = (float(int(seed) * 37) * 0.6180339887498949) % 1.0
    keys = np.mod(np.arange(n, dtype=float) * float(alpha) + shift, 1.0)
    order = np.argsort(keys)
    b = np.zeros(n, dtype=float)
    b[order[0::2]] = 1.0
    return b


def msequence_binary(n):
    n = int(n)
    if n <= 0:
        return None
    if (n & (n - 1)) != 0:
        return None
    m = n.bit_length() - 1
    if m < 2 or m > 12:
        return None
    taps = MSEQ_TAPS.get(m, [1])
    state = (1 << m) - 1
    total = (1 << m) - 1
    seq = []
    for _ in range(total):
        seq.append(float(state & 1))
        fb = 0
        for t in taps:
            fb ^= (state >> t) & 1
        state = (state >> 1) | (fb << (m - 1))
        if state == 0:
            break
    if len(seq) < total:
        for i in range(total - len(seq)):
            seq.append(1.0 if (i & 1) == 0 else 0.0)
    seq.append(0.0)
    return np.asarray(seq[:n], dtype=float)

def binary_swap_update(h, corr, a, b, i_all, n):
    a = int(a)
    b = int(b)
    da = 1.0 - 2.0 * h[a]
    db = 1.0 - 2.0 * h[b]
    center = n - 1
    mask = (i_all != a) & (i_all != b)
    if np.any(mask):
        im = i_all[mask]
        hm = h[im]
        one_minus_pm = 1.0 - hm
        idx = a - im + center
        corr[idx] += one_minus_pm * da
        idx = b - im + center
        corr[idx] += one_minus_pm * db
        idx = im - a + center
        corr[idx] -= hm * da
        idx = im - b + center
        corr[idx] -= hm * db

    ha0 = float(h[a])
    hb0 = float(h[b])
    ha1 = ha0 + float(da)
    hb1 = hb0 + float(db)

    idx = b - a + center
    corr[idx] += hb1 * (1.0 - ha1) - hb0 * (1.0 - ha0)
    idx = a - b + center
    corr[idx] += ha1 * (1.0 - hb1) - ha0 * (1.0 - hb0)

    h[a] = ha1
    h[b] = hb1


def choose_binary_move(b, inc, dec, n, rng):
    inc = np.asarray(inc, dtype=int)
    dec = np.asarray(dec, dtype=int)
    if inc.size > 0 and dec.size > 0:
        a = int(rng.choice(inc))
        d = int(rng.choice(dec))
        if 0 <= a < n and 0 <= d < n and b[a] != b[d]:
            return a, d
    for _ in range(25):
        a = int(rng.integers(n))
        d = int(rng.integers(n))
        if b[a] != b[d]:
            return a, d
    ones = np.where(b > 0.5)[0]
    zeros = np.where(b < 0.5)[0]
    if ones.size > 0 and zeros.size > 0:
        a = int(rng.choice(ones))
        d = int(rng.choice(zeros))
        return a, d
    return 0, 1

    b = binary_from_float(h0)
    b = binary_fix_counts(b, target, rng)
    corr = np.correlate(b, 1.0 - b, mode="full")
    cur_count = float(np.max(corr))
    best_count = cur_count
    best_b = b.copy()
    i_all = np.arange(n, dtype=int)
    T = max(1.0, 0.08 * cur_count)
    end = time.time() + max(1.0, float(budget))
    moves = 0
    moves_since_best = 0
    inc = np.zeros(0, dtype=int)
    dec = np.zeros(0, dtype=int)

    while time.time() < end - 0.2:
        moves += 1
        if moves % 2500 == 0:
            corr = np.correlate(b, 1.0 - b, mode="full")
            cur_count = float(np.max(corr))
        if moves % 25 == 0:
            Araw = corr
            active = active_indices(Araw, kmax=16, tol_frac=0.16)
            grad = gradient_active(b, active, 2.0 / float(n), None)
            inc = np.where((grad < -1e-7) & (b < 0.5))[0]
            dec = np.where((grad > 1e-7) & (b > 0.5))[0]

        a, d = choose_binary_move(b, inc, dec, n, rng)
        if a == d or b[a] == b[d]:
            ones = np.where(b > 0.5)[0]
            zeros = np.where(b < 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            a = int(rng.choice(ones))
            d = int(rng.choice(zeros))

        old_count = cur_count
        binary_swap_update(b, corr, a, d, i_all, n)
        new_count = float(np.max(corr))

        best_improved = False
        if new_count < best_count - 1e-9:
            best_count = new_count
            best_b = b.copy()
            best_improved = True
            T = max(0.25, T * 0.96)
        if not best_improved:
            moves_since_best += 1

        accepted = False
        if new_count < old_count - 1e-9:
            accepted = True
            T = max(0.25, T * 0.95)
        elif abs(new_count - old_count) <= 1e-7:
            if rng.random() < 0.18:
                accepted = True
        else:
            delta = float(new_count - old_count)
            if delta < 3.0 and rng.random() < math.exp(-delta / max(1e-3, T)):
                accepted = True

        if accepted:
            cur_count = new_count
        else:
            binary_swap_update(b, corr, a, d, i_all, n)
            cur_count = old_count

        if moves_since_best > max(2500, 8 * n) and cur_count > best_count + max(5.0, 0.25 * best_count):
            b = best_b.copy()
            corr = np.correlate(b, 1.0 - b, mode="full")
            cur_count = best_count
            moves_since_best = 0
            T = max(1.0, 0.10 * cur_count)

    best_c = float(best_count) * (2.0 / float(n))
    exact = score_c5(best_b)
    if exact < best_c + 1e-12:
        best_c = exact
    return best_b, best_c


def default_initial(n, seed):
    n = int(n)
    if n <= 0:
        return np.zeros(1, dtype=float)
    rng = np.random.default_rng((int(seed) * 53 + 7) & ((1 << 64) - 1))
    keys = rng.standard_normal(n)
    order = np.argsort(keys)
    h = np.zeros(n, dtype=float)
    k = int(math.floor(0.5 * n))
    if k > 0:
        h[order[:k]] = 1.0
    if 2 * k != n and k < n:
        h[order[k]] = 0.5
    return project_to_target(h, 0.5 * n)


def make_feasible(h):
    h = clean_array(h)
    target = 0.5 * h.size
    h = project_to_target(h, target)
    c = score_c5(h)
    if not np.isfinite(c):
        c = 0.9
    return h, float(c)


def update_best(store, h, c):
    if h is None:
        return store
    h = clean_array(h)
    try:
        c = float(c)
    except Exception:
        c = np.inf
    if h.size > 0 and np.isfinite(c) and c < store[1]:
        store[0] = h.copy()
        store[1] = c
    return store


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    t0 = time.time()
    deadline = t0 + max(30.0, float(budget_s) * 0.985)
    store = [None, np.inf]

    have_parent = False
    parent = None
    if "initial_h_values" in globals():
        try:
            parent = np.asarray(initial_h_values, dtype=float).copy()
        except Exception:
            parent = None
        if parent is not None and parent.size > 1:
            parent = project_to_target(parent, 0.5 * parent.size)
            c = score_c5(parent)
            store = update_best(store, parent, c)
            have_parent = True

    if not have_parent or parent is None:
        parent = default_initial(45, seed)
        c = score_c5(parent)
        store = update_best(store, parent, c)
        have_parent = True

    rem = seconds_left(deadline)
    if rem > 120:
        alloc = allocate_budget(rem, 110, 0.22, 30)
        if alloc >= 15:
            h, c = continuous_improve(parent, alloc, seed + 1, force_sym=False)
            store = update_best(store, h, c)

    rem = seconds_left(deadline)
    if rem > 120:
        psym = symmetrize_project(parent, 0.5 * parent.size)
        alloc = allocate_budget(rem, 90, 0.20, 25)
        if alloc >= 15:
            h, c = continuous_improve(psym, alloc, seed + 2, force_sym=True)
            store = update_best(store, h, c)

    ups = [(90, 95), (180, 70), (360, 50), (720, 28)]
    for idx, (nn, ideal) in enumerate(ups):
        rem = seconds_left(deadline)
        if rem < 90:
            break
        src = store[0] if store[0] is not None and store[0].size > 0 else parent
        u = upsample_continuous(src, nn)
        alloc = allocate_budget(rem, ideal, 0.18, 20)
        if alloc >= 12:
            h, c = continuous_improve(u, alloc, seed + 1000 * idx + nn)
            store = update_best(store, h, c)

        rem = seconds_left(deadline)
        if rem > 110 and ideal >= 60:
            us = symmetrize_project(u, 0.5 * nn)
            alloc2 = allocate_budget(rem, max(20, ideal - 35), 0.16, 18)
            if alloc2 >= 12:
                h, c = continuous_improve(us, alloc2, seed + 2000 * idx + nn, force_sym=True)
                store = update_best(store, h, c)

    binary_ns = [44, 50, 60, 80, 100, 120, 160, 200, 256, 320]
    binary_ideals = [95, 65, 55, 75, 95, 65, 85, 95, 105, 70]
    for idx, n in enumerate(binary_ns):
        rem = seconds_left(deadline)
        if rem < 70:
            break
        starts = best_binary_starts(n, seed + 101 * n, k=2)
        if store[0] is not None and store[0].size > 0:
            ub = binary_from_float(upsample_continuous(store[0], n))
            starts.append(ub)
        if not starts:
            continue

        cvals = []
        feasible_starts = []
        for b in starts:
            b = binary_fix_counts(b, n // 2, np.random.default_rng(seed + n + len(starts)))
            feasible_starts.append(b)
            cvals.append(score_c5(b))
        cvals = np.asarray(cvals, dtype=float)
        order = np.argsort(cvals)
        chosen = []
        for i in order[:2]:
            if 0 <= i < len(feasible_starts):
                chosen.append(feasible_starts[int(i)])
        for j, b0 in enumerate(chosen):
            rem = seconds_left(deadline)
            if rem < 45:
                break
            ideal = binary_ideals[idx] * (1.0 if j == 0 else 0.6)
            alloc = allocate_budget(rem, ideal, 0.22, 18)
            if alloc >= 10:
                hb, cb = binary_local_search(b0, alloc, seed + 5000 * idx + 7 * j)
                if hb is not None:
                    store = update_best(store, hb, cb)

    rem = seconds_left(deadline)
    if rem > 70 and store[0] is not None:
        alloc = allocate_budget(rem, 90, 0.35, 25)
        if alloc >= 15:
            hbase = store[0].copy()
            h, c = continuous_improve(hbase, alloc, seed + 99991, force_sym=False,
                                      max_lp_n=120)
            store = update_best(store, h, c)

    rem = seconds_left(deadline)
    if rem > 45 and store[0] is not None:
        alloc = allocate_budget(rem, 55, 0.40, 15)
        if alloc >= 10:
            hbase = store[0].copy()
            hs = symmetrize_project(hbase, 0.5 * hbase.size)
            h, c = continuous_improve(hs, alloc, seed + 99992, force_sym=True,
                                      max_lp_n=120)
            store = update_best(store, h, c)

    if store[0] is None:
        h = default_initial(44, seed)
        c = score_c5(h)
        store = update_best(store, h, c)

    h, c = make_feasible(store[0])
    store = update_best(store, h, c)
    h = clean_array(store[0])
    c = float(store[1])
    if not np.isfinite(c):
        c = 0.5
    if h.size == 0:
        h = default_initial(44, seed)
        c = score_c5(h)
    return h, float(c), int(h.size)
