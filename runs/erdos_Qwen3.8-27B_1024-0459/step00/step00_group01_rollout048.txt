            elif 0 <= im < n and h[im] > 0.1 and h[i] < 0.9:
                # Increase h[i], decrease h[im]
                j = im
                direction = 1
            else:
                continue
            if i == j:
                continue
            # Try transfer
            for eps in [0.02, 0.05, 0.1, 0.15]:
                h_new = h.copy()
                h_new[i] += direction * eps
                h_new[j] -= direction * eps
                if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                    continue
                h_new = np.clip(h_new, 0, 1)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-12:
                    h = h_new
                    c5 = c5_new
                    improved_this_round = True
                    break

        if not improved_this_round:
            # Try random perturbations
            for _ in range(20):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                eps = rng.uniform(0.01, 0.2) * rng.choice([-1, 1])
                h_new = h.copy()
                h_new[i] += eps
                h_new[j] -= eps
                if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
                    continue
                h_new = np.clip(h_new, 0, 1)
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-12:
                    h = h_new
                    c5 = c5_new
                    improved_this_round = True
                    break
            if not improved_this_round:
                break

    return h

def make_binary_like(n, rng):
    """Create a binary-like initial guess with good structure."""
    h = np.zeros(n)
    # Place 1's in a spread pattern
    n_ones = int(n / 2)
    if n % 2 == 0:
        n_zeros = n - n_ones
    else:
        n_zeros = n - n_ones  # n_ones = floor(n/2), will add fraction

    # Use a quasi-random spread (like low-discrepancy)
    golden = 0.618033988749895
    positions = sorted(((i * golden) % 1.0 * n).astype(int) % n for i in range(n_ones))
    positions = sorted(set(positions))
    # Fill remaining
    while len(positions) < n_ones:
        p = rng.randint(n)
        if p not in positions:
            positions.append(p)
            positions.sort()

    for p in positions[:n_ones]:
        h[p] = 1.0

    if n % 2 == 1:
        # Need half-integer sum, so set one position to 0.5
        if n_ones < n:
            h[rng.randint(n)] = 0.5
        # Redo: make sum = n/2
        while h.sum() > n / 2 + 1e-10:
            idx = np.where(h == 1.0)[0]
            if len(idx) == 0:
                break
            idx = idx[0]
            h[idx] -= (h.sum() - n / 2)
            h[idx] = max(h[idx], 0)
        while h.sum() < n / 2 - 1e-10:
            idx = np.where(h == 0.0)[0]
            if len(idx) == 0:
                break
            idx = idx[0]
            h[idx] = min(h[idx] + (n / 2 - h.sum()), 1.0)

    return project_feasible(h, n)

def make_wave(n, freq, phase, rng):
    """Create a sinusoidal initial guess."""
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase) + rng.normal(0, 0.05, n)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    # Get initial values
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        pass
    except Exception:
        pass

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def try_update(h):
        nonlocal best_h, best_c5, best_n
        n = len(h)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    # Initialize from parent
    if h_parent is not None:
        n_p = len(h_parent)
        hp = project_feasible(h_parent, n_p)
        try_update(hp)

    # Optimization loop over resolutions and initializations
    n_list = [47, 48, 50, 54, 60, 64, 72, 80]
    if h_parent is not None:
        if len(h_parent) not in n_list:
            n_list.insert(0, len(h_parent))

    for n in n_list:
        if time.time() - t0 > budget_s - 40:
            break

        # Multiple initializations
        inits = []

        # From parent (interpolated)
        if h_parent is not None:
            if n == len(h_parent):
                inits.append(h_parent.copy())
            else:
                x_old = np.linspace(0, 1, len(h_parent), endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                h_interp = np.interp(x_new, x_old, h_parent)
                inits.append(project_feasible(h_interp, n))

        # Binary-like
        inits.append(make_binary_like(n, rng))

        # Wave
        inits.append(make_wave(n, 1, 0, rng))
        inits.append(make_wave(n, 2, np.pi / 3, rng))

        # Near-uniform perturbation
        h_uni = np.full(n, 0.5)
        h_uni += rng.normal(0, 0.2, n)
        inits.append(project_feasible(h_uni, n))

        for h_init in inits:
            if time.time() - t0 > budget_s - 40:
                break
            h_init = project_feasible(h_init, n)
            try_update(h_init)

            # SLSQP optimization with increasing alpha
            for alpha in [100.0, 250.0, 500.0]:
                if time.time() - t0 > budget_s - 40:
                    break
                h_opt = slsqp_optimize(h_init, n, t0, budget_s, alpha=alpha, maxiter=150)
                h_init = h_opt
                try_update(h_init)

            # 2-opt refinement
            h_ref = two_opt_refine(h_init, n, t0, budget_s, max_rounds=100)
            try_update(h_ref)

    # Final: do more refinement on the best solution found
    if best_h is not None and time.time() - t0 < budget_s - 20:
        h_final = two_opt_refine(best_h, best_n, t0, budget_s, max_rounds=200)
        try_update(h_final)

        # One more SLSQP pass
        if time.time() - t0 < budget_s - 20:
            h_final2 = slsqp_optimize(h_final, best_n, t0, budget_s, alpha=500.0, maxiter=100)
            try_update(h_final2)
            h_final3 = two_opt_refine(h_final2, best_n, t0, budget_s, max_rounds=100)
            try_update(h_final3)

    if best_h is None:
        # Fallback: uniform
        n_fb = 47
        best_h = np.full(n_fb, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n_fb

    return best_h, best_c5, best_n
