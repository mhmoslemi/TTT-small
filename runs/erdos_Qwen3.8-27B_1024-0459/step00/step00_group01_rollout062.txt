def compute_c5(h):

    return float(np.max(correlation) * dx)

def compute_corr(h):
    n = len(h)
    correlation = np.correlate(h, 1.0 - h, mode='full')
    return correlation

def smooth_max(vals, alpha):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha

def repair_integral(h, n):
    """Adjust h to satisfy sum(h) = n/2 while maintaining 0 <= h <= 1."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add diff total. Add to entries < 1.
        mask = h < 1.0 - 1e-10
        count = mask.sum()
        if count == 0:
            return h
        add_per = diff / count
        h[mask] = np.clip(h[mask] + add_per, 0.0, 1.0)
    else:
        # Need to remove |diff| total. Remove from entries > 0.
        mask = h > 1e-10
        count = mask.sum()
        if count == 0:
            return h
        rem_per = -diff / count
        h[mask] = np.clip(h[mask] - rem_per, 0.0, 1.0)
    return h

def smooth_objective(x, alpha, n):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, alpha) * dx

        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')

    """Minimize max(base + delta * x) over x in [lo, hi]."""
    for _ in range(60):  # ~1e-18 precision
        mid = (lo + hi) / 2
        f_mid = base + delta * mid
        # Subgradient at mid: average of active slopes
        max_val = np.max(f_mid)
        active = np.abs(f_mid - max_val) < 1e-10
        grad = np.mean(delta[active])
        if grad < 0:
            lo = mid
        elif grad > 0:
            hi = mid
        else:
            return mid
    return (lo + hi) / 2

- f(x) = max_s (a_s + b_s * x)
- f is convex, piecewise linear
- f'(x+) (right derivative) = max{b_s : a_s + b_s * x = f(x)}
- f'(x-) (left derivative) = min{b_s : a_s + b_s * x = f(x)}
- f is minimized where f'(x-) ≤ 0 ≤ f'(x+)

So the condition for x to be optimal:
- min{b_s : active} ≤ 0 ≤ max{b_s : active}

In bisection:
- If f'(x+) < 0 (all active slopes < 0): move right (lo = mid)
- If f'(x-) > 0 (all active slopes > 0): move left (hi = mid)
- Otherwise: x is optimal

        max_val = np.max(vals)

        max_slope = np.max(delta[active])

            return mid

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time
import random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    correlation = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(correlation) * dx)

def repair_integral(h, n, max_iter=30):
    target = n / 2.0
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def min_1d_max_lines(base, delta):
    """Minimize max_s(base[s] + delta[s]*x) over x in [0,1] via subgradient bisection."""
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = base + delta * mid
        max_val = np.max(vals)
        active = vals >= max_val - 1e-10
        min_slope = float(np.min(delta[active]))
        max_slope = float(np.max(delta[active]))
        if min_slope <= 0.0 and max_slope >= 0.0:
            return mid
        elif max_slope < -1e-14:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def coordinate_descent(h, n, max_sweeps=200, time_limit=None, verbose=False):

    best_c5 = compute_c5(h)

            break
        improved = False
        for i in range(n):
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0

    h = repair_integral(h, n)
    final_c5 = compute_c5(h)
    if verbose:
        print(f"  CD done: {sweep+1} sweeps, C5={final_c5:.6f}")
    return h, final_c5

def smooth_max_corr(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

def l_bfgsb_optimize(h0, n, alpha=150, maxiter=5000):
    h0 = np.array(h0, dtype=float).copy()
    h0 = repair_integral(h0, n)

    def obj(x):
        h_c = np.clip(x, 0.0, 1.0)
        return smooth_max_corr(h_c, alpha)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    constraints = [{'type': 'eq', 'fun': con_fun}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h0, method='L-BFGS-B',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_opt = np.clip(res.x, 0.0, 1.0)
        h_opt = repair_integral(h_opt, n)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_random_01(n, rng):
