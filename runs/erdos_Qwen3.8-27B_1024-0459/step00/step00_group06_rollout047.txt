def project_to_box(x, n, dx):
    """Project x onto {0 <= x <= 1, sum(x)*dx = 1}"""
    x = np.clip(x, 0, 1)
    # Adjust sum to n/2
    diff = n/2 - sum(x)
    if diff > 0:
        # Need to increase some values
        mask = x < 1
        if mask.sum() > 0:

            x = np.clip(x, 0, 1)
    elif diff < 0:

        if mask.sum() > 0:

            x = np.clip(x, 0, 1)
    return x

def projection(h, n_pts, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using water-filling"""
    h = np.clip(h, 0, 1)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

def subgradient_step(h, dx, lr, n_pts):
    """One subgradient descent step"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)

    lag = k_star - (n_pts - 1)
    grad = corr_gradient_single_k(h, lag, n_pts)
    h_new = h - lr * grad / dx  # dx scaling
    h_new = projection(h_new, n_pts, n_pts / 2.0)
    return h_new, corr[k_star] * dx
