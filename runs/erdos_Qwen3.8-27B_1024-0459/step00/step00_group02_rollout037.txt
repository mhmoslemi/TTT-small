import time

def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def evaluate_c5_from_counts(counts):
    """Compute C5 from precomputed counts array."""
    n = len(counts) // 2  # counts has length 2n-1, so n = (len+1)//2
    # Actually counts has length 2*n-1
    dx = 2.0 / ((len(counts) + 1) // 2)
    return np.max(counts) * dx

def compute_all_counts(h):
    """Compute count_s for all shifts."""
    n = len(h)
    counts = np.zeros(2 * n - 1)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode="full")
    return corr
