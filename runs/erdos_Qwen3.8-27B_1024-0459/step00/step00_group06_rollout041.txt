    n = n_points
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = (corr - cmax) / eps
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp  # softmax weights, shape (2n-1,)
    smax = cmax + eps * np.log(sum_exp)
    
    val = smax * dx
    s = np.sum(h) - n / 2.0
    val += lam * s * s
    
    # Gradient
    grad = np.zeros(n)
    # For each lag k (0 to 2n-2), s_k = k - (n-1)
    # ∂c_k/∂h[j] = (1-h[j+s_k])*[0≤j+s_k<n] - h[j-s_k]*[0≤j-s_k<n]
    # Sum over k with weights
    
    # This is O(n * (2n-1)) = O(n²)
    for k in range(2 * n - 1):
        sk = k - (n - 1)
        wk = weights[k] * dx
        if sk >= 0:

            for j in range(n - sk):
                grad[j] += wk * (1.0 - h[j + sk])
            # Second term: j from sk to n-1, grad[j] -= wk * h[j-sk]
            for j in range(sk, n):
                grad[j] -= wk * h[j - sk]
        else:
            t = -sk
            # First term: j from t to n-1, grad[j] += wk * (1-h[j-t])
            for j in range(t, n):

            # Second term: j from 0 to n-t-1, grad[j] -= wk * h[j+t]
            for j in range(n - t):
                grad[j] -= wk * h[j + t]
    
    # Add penalty gradient
    grad += 2.0 * lam * s
    
    return val, grad
