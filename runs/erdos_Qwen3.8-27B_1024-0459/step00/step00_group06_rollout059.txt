def init_from_parent(parent, n_p, n, rng):
    x_p = np.linspace(0.0, 2.0, n_p, endpoint=False)
    x_n = np.linspace(0.0, 2.0, n, endpoint=False)
    h_cont = np.interp(x_n, x_p, parent)

def init_structured(n, rng, variant=0):
    h = np.zeros(n)
    half = n // 2
    if variant == 0:
        # Spread ones as evenly as possible
        positions = np.sort(rng.choice(n, size=half, replace=False))
        h[positions] = 1.0
    elif variant == 1:
        # Shift-based pattern
        shift = rng.randint(1, n)
        for i in range(n):
            if (i + shift) % (n // 2 + 1) < (n // 4):
                h[i] = 1.0
        # Fix count
        if np.sum(h) > half:
            excess = np.sum(h) - half
            ones = np.where(h == 1)[0]
            for i in range(excess):

        elif np.sum(h) < half:
            deficit = half - np.sum(h)
            zeros = np.where(h == 0)[0]
            for i in range(deficit):

    else:

        h[positions] = 1.0
    return h

            h_opt, c5_opt = iterated_local_search(h_init, n, time_per_init, rng)

            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n = n

    # Final intensive polish on the best solution
    elapsed = time.time() - t0
    time_left = budget_s * 0.97 - elapsed
    if time_left > 20 and best_h is not None:
        n_final = len(best_h)
        if n_final % 2 == 0:
            best_h, best_c5 = iterated_local_search(best_h, n_final, time_left, rng)
        else:
            # For odd n, try to convert to even
            n_even = n_final + 1 if n_final + 1 % 2 == 0 else n_final - 1
            x_odd = np.linspace(0, 2, n_final, endpoint=False)
            x_even = np.linspace(0, 2, n_even, endpoint=False)
            h_even_cont = np.interp(x_even, x_odd, best_h)
            h_even = np.zeros(n_even)
            h_even[np.argpartition(h_even_cont, n_even // 2)[:n_even // 2]] = 1.0
            h_polished, c5_polished = iterated_local_search(h_even, n_even, time_left, rng)
            if c5_polished < best_c5:
                best_h = h_polished
                best_c5 = c5_polished
                best_n = n_even

    # Final verification and cleanup
    n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    dx = 2.0 / n
    # Ensure sum is correct
    target = n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-8:
        if diff > 0:
            mask = best_h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                best_h[mask] = np.clip(best_h[mask] + diff / cnt, 0, 1)
        else:
            mask = best_h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                best_h[mask] = np.clip(best_h[mask] + diff / cnt, 0, 1)

    c5_final, _ = compute_c5(best_h)

    return (best_h, c5_final, n)
