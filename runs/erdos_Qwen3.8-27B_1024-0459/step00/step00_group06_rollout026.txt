def compute_c5(h):

    return float(np.max(corr) * dx)


def project_to_constraint(h, target_sum):
    n = len(h)
    hi_bound = 1.0 - float(np.max(h)) + 1.0
    lo_bound = float(np.min(h)) - 2.0
    lo = lo_bound
    hi = hi_bound
    for _ in range(100):

        clipped = np.clip(h - mid, 0.0, 1.0)
        s = float(np.sum(clipped))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_subgradient(h, tol_frac=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = float(np.max(corr))
    threshold = max_corr - tol_frac * n * dx
    active_indices = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for k in active_indices:
        s = int(k) - (n - 1)
        if s >= 0:
            if s == 0:
                grad += dx * (1.0 - 2.0 * h)
            else:
                grad[s:] += dx * (1.0 - h[:n - s])
                grad[:n - s] -= dx * h[s:]
        else:
            r = -s
            grad[:n - r] += dx * (1.0 - h[r:])
            grad[r:] -= dx * h[:n - r]
    grad /= max(1, len(active_indices))
    return grad, max_corr * dx


def optimize_subgradient(h0, target_sum, time_limit):

    patience = 0

        if c5_val < best_c5 - 1e-13:

            best_h = h.copy()
            patience = 0
        else:
            patience += 1
