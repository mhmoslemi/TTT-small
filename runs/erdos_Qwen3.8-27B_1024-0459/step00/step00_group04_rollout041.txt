    i_max = min(n-1, n-1-s)

    # j+s valid means 0 ≤ j+s ≤ n-1, i.e., -s ≤ j ≤ n-1-s
    # Combined with i_min ≤ j ≤ i_max:
    j1_lo = max(i_min, 0, -s)  # j ≥ -s means j+s ≥ 0
    j1_hi = min(i_max, n-1-s)  # j+s ≤ n-1 means j ≤ n-1-s
    if j1_lo <= j1_hi:
        grad[j1_lo:j1_hi+1] -= h[j1_lo+s:j1_hi+s+1]
    
    # Subtract h[j-s] for valid j
    # j-s valid means 0 ≤ j-s ≤ n-1, i.e., s ≤ j ≤ n-1+s
    # Combined with i_min ≤ j ≤ i_max:
    j2_lo = max(i_min, s)  # j ≥ s means j-s ≥ 0
    j2_hi = min(i_max, n-1)  # j-s ≤ n-1 means j ≤ n-1+s (always true since j ≤ n-1)
    if j2_lo <= j2_hi:
        grad[j2_lo:j2_hi+1] -= h[j2_lo-s:j2_hi-s+1]
    
    return grad
