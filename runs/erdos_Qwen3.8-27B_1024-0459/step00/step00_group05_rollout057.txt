    """Improve h via coordinate descent"""

    current_c5, _ = compute_c5(h)
    improved = True

                    current_c5 = trial_c5
                    improved = True
                    break
    return h, current_c5

import numpy as np
import time
import math

def project_h(h, n_points):
    """Project h onto {0<=h<=1, sum(h)=n/2} via bisection on threshold."""
    target = n_points / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    x = np.clip(h - t, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = np.where((x > 1e-8) & (x < 1.0 - 1e-8))[0]
        if len(free) > 0:
            x[free] += diff / len(free)
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_worst_shift(h):
    """Compute C5 and return the index of the worst shift."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx]) * dx, idx, corr * dx

def sa_optimize(h_init, n_points, max_iters, t0, t1, seed, step_scale=0.3):
    """Simulated annealing."""
    h = project_h(h_init.copy(), n_points)
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    denom = max(max_iters - 1, 1)
    for it in range(max_iters):
        temp = t0 * (t1 / t0) ** (it / denom)
        n_pert = max(1, int(n_points * 0.12 * rng.uniform(0.3, 2.5)))
        n_pert = min(n_pert, n_points)
        indices = rng.choice(n_points, n_pert, replace=False)

        h_trial = project_h(h_trial, n_points)
        trial_c5 = compute_c5(h_trial)
        delta = trial_c5 - cur_c5
        if delta < 0 or (temp > 1e-14 and rng.random() < math.exp(-delta / temp)):
            h = h_trial
            cur_c5 = trial_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5

def hill_climb(h_init, n_points, max_iters, seed, step0=0.06):
    """Hill climbing with adaptive step size."""
    h = project_h(h_init.copy(), n_points)
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    step = step0
    for it in range(max_iters):
        n_pert = max(1, rng.randint(1, max(2, n_points // 5) + 1))
        indices = rng.choice(n_points, n_pert, replace=False)
        h_trial = h.copy()
        h_trial[indices] += rng.uniform(-step, step, n_pert)
        h_trial = project_h(h_trial, n_points)
        trial_c5 = compute_c5(h_trial)

            h = h_trial
            cur_c5 = trial_c5
            best_h = h.copy()
            best_c5 = cur_c5
            step = min(0.25, step * 1.01)
        else:
            step = max(0.003, step * 0.97)
    return best_h, best_c5

def directed_hill_climb(h_init, n_points, max_iters, seed, step=0.04):
    """Hill climb targeting the worst shift."""
    h = project_h(h_init.copy(), n_points)
    cur_c5, _, _ = compute_c5_worst_shift(h)

    best_h = h.copy()

        h_trial = h.copy()
        try:

                valid_i = np.arange(shift, n_points)
                valid_j = np.arange(0, n_points - shift)

                valid_j = np.arange(-shift, n_points)

                gj = valid_j[ii]

    h = np.zeros(n_points)

def upsample_h(h, n_new):
    """Upsample h to n_new points via interpolation."""
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return np.interp(x_new, x_old, h)
