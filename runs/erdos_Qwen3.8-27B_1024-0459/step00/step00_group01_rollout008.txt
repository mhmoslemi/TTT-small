    # Bisection on lambda: h_proj = clip(h - lambda, 0, 1), find lambda s.t. sum = target
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        free = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_subgradient(h, n_points, k_star):
    """Compute subgradient of f_{k_star} w.r.t. h.
    k_star is the index in the correlate output (0 to 2n-2)."""
    lag = k_star - (n_points - 1)
    g = np.zeros(n_points)
    for i in range(n_points):
        j1 = i + lag
        if 0 <= j1 < n_points:
            g[i] += (1.0 - h[j1])
        j2 = i - lag
        if 0 <= j2 < n_points:
            g[i] -= h[j2]
    return g

def subgradient_descent(h, n_points, alpha0=0.01, n_iter=10000, verbose=False):
    """Run subgradient descent with projection."""
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr, n_points)
    best_h = h_curr.copy()
    
    for it in range(n_iter):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        k_star = int(np.argmax(corr))
        c5_val = corr[k_star] * (2.0 / n_points)
        
        if c5_val < best_c5 - 1e-9:
            best_c5 = c5_val
            best_h = h_curr.copy()
        
        g = compute_subgradient(h_curr, n_points, k_star)
        alpha = alpha0 / (1.0 + 0.01 * it)
        h_new = h_curr - alpha * g
        h_curr = project_to_constraints(h_new, n_points)
    
    # Final check
    c5_final = compute_c5(h_curr, n_points)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_curr.copy()
    
    return best_h, best_c5

def local_search(h, n_points, n_iterations=5000):
    """Local search: try small random perturbations and accept if better."""
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr, n_points)
    best_h = h_curr.copy()
    dx_step = 0.05
    rng = np.random.default_rng(12345)
    
    for it in range(n_iterations):
        # Pick 2-4 random coordinates to perturb
        n_perturb = rng.integers(2, 5)
        indices = rng.choice(n_points, size=n_perturb, replace=False)
        
        h_try = h_curr.copy()
        # Perturb and maintain sum
        delta = rng.normal(0, dx_step, size=n_perturb)
        delta_sum = np.sum(delta)
        delta -= delta_sum / n_perturb  # zero-sum perturbation
        h_try[indices] += delta
        h_try = project_to_constraints(h_try, n_points)
        
        c5_try = compute_c5(h_try, n_points)
        if c5_try < best_c5 - 1e-9:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_curr = h_try.copy()
            # Decrease step
            dx_step = max(0.01, dx_step * 0.98)
    
    return best_h, best_c5

def resample(h, old_n, new_n):
    """Resample h from old_n points to new_n points, maintaining integral."""
    old_x = np.linspace(0, 2, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0, 2, new_n, endpoint=False) + 1.0 / new_n
    # Step function interpolation
    h_new = np.interp(new_x, old_x, h, period=2.0)
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_constraints(h_new, new_n)
    return h_new

def coordinate_descent(h, n_points, n_sweeps=100):
    """Coordinate descent optimizing pairs of coordinates."""
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr, n_points)
    best_h = h_curr.copy()
    target = n_points / 2.0
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n_points - 1):
            # Try to optimize h[i] and h[i+1] while maintaining sum
            s_pair = h_curr[i] + h_curr[i+1]
            # In 1D: for t in [max(0, s_pair-1), min(1, s_pair)]
            # h[i] = t, h[i+1] = s_pair - t
            # f_lag(t) is a quadratic in t (since t appears in products)
            # Actually f_lag is linear in each h[j] individually, but the pair
            # (t, s_pair-t) gives f_lag = a*t + b*(s_pair-t) + c = (a-b)*t + b*s_pair + c
            # So it's still linear in t! And we want min max_lag f_lag(t)
            
            t_lo = max(0.0, s_pair - 1.0)
            t_hi = min(1.0, s_pair)
            
            if t_hi - t_lo < 1e-10:
                continue
            
            # Evaluate at a few points
            best_t = h_curr[i]
            best_val = compute_c5(h_curr, n_points)
            
            for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
                t = t_lo + frac * (t_hi - t_lo)
                h_try = h_curr.copy()
                h_try[i] = t
                h_try[i+1] = s_pair - t
                val = compute_c5(h_try, n_points)
                if val < best_val - 1e-12:
                    best_val = val
                    best_t = t
            
            if best_t != h_curr[i]:
                h_curr[i] = best_t
                h_curr[i+1] = s_pair - best_t
                improved = True
                if best_val < best_c5 - 1e-9:
                    best_c5 = best_val
                    best_h = h_curr.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

    Returns the optimal t = h[i] (and h[j] = s - t)."""

    # t range: max(0, s-1) ≤ t ≤ min(1, s)
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    
    if t_hi - t_lo < 1e-12:
        return t_lo  # no freedom
    
    # For all 2n-1 lags, compute A_lag, B_lag, C_lag
    # f_lag(t) = (A_lag - D_lag) * t + const

    dx = 2.0 / n
    lines = []  # (a_lag, b_lag) for f_lag(t) = a_lag * t + b_lag

        #   p=i-lag (if valid): h[i-lag]*(1-h[i])
