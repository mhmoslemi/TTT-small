def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(-h)
        remaining = diff
        for idx in order:
            dec = min(h[idx], remaining)
            h[idx] -= dec
            remaining -= dec
            if remaining <= 1e-14:
                break
    else:
        order = np.argsort(h)
        remaining = -diff
        for idx in order:
            inc = min(1.0 - h[idx], remaining)
            h[idx] += inc
            remaining -= inc
            if remaining <= 1e-14:
                break
    return h

def upsample(h, new_n):

    old_x = np.linspace(0, 2, n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, np.concatenate([h, [h[0]]]))

    deadline = start_time + budget_s - 10  # leave 10s buffer
    
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, TypeError):
        h_init = np.full(64, 0.5)
    
    # Target resolution
    N = 1024
    
    # Upsample
    h = upsample(h_init, N)
    target_sum = N / 2.0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    elapsed = lambda: time.time() - start_time
    
    # Coordinate descent
    golden_ratio_conjugate = (np.sqrt(5.0) - 1.0) / 2.0
    gs_iters = 25
    
    def coord_desc_pass(h):
        n = len(h)
        dx = 2.0 / n
        # Compute current correlation values
        curr_corr = np.correlate(h, 1-h, mode="full") * dx
        
        for j in range(n):
            old_val = h[j]
            # Compute A[k] = change in corr[k]*dx when h[j] goes from 0 to 1
            A = np.zeros(2*n-1)
            
            # Positive lags m = 0, 1, ..., n-1 (index k = m + n - 1)
            for m in range(n):
                k = m + n - 1
                if j <= n - 1 - m:
                    A[k] += (1.0 - h[j+m])
                if m <= j:
                    A[k] -= h[j-m]
            
            # Negative lags m' = 1, ..., n-1 (index k = n-1-m')
            for m_prime in range(1, n):
                k = n - 1 - m_prime
                if m_prime <= j:
                    A[k] += (1.0 - h[j-m_prime])
                if j <= n - 1 - m_prime:
                    A[k] -= h[j+m_prime]
            
            A *= dx
            
            # B[k] = curr_corr[k] - old_val * A[k]
            B = curr_corr - old_val * A
            
            # Golden section search on [0,1] to minimize max(A*t + B)
            lo, hi = 0.0, 1.0
            for _ in range(gs_iters):
                t1 = hi - golden_ratio_conjugate * (hi - lo)
                t2 = lo + golden_ratio_conjugate * (hi - lo)
                f1 = np.atleast_1d(np.max(A * t1 + B))
                f2 = np.atleast_1d(np.max(A * t2 + B))
                if f1[0] < f2[0]:
                    hi = t2
                else:
                    lo = t1
            
            h[j] = (lo + hi) / 2.0
            # Update curr_corr for the next coordinate
            # curr_corr[k] changes by (h[j] - old_val) * A[k]
            curr_corr += (h[j] - old_val) * A
        
        return project(h, target_sum)
    
    no_improve = 0
    patience = 200
    
    while elapsed() < deadline:
        h = coord_desc_pass(h)
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            # Perturb and restart
            h = best_h + np.random.randn(N) * 0.05
            h = project(h, target_sum)
            no_improve = 0
    
    return best_h, best_c5, N
