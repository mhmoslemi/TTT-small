    dx = 2.0 / n
    ones = 1.0 - h
    corr = np.correlate(h, ones, mode='full')
    c5 = np.max(corr) * dx

    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-6)[0]
    
    # Compute subgradient (average gradient over active lags)
    grad = np.zeros(n)
    for m_idx in active:
        # m_idx corresponds to lag m = m_idx - (n-1)
        m = m_idx - (n - 1)
        # corr[m_idx] = sum_n h[n+m] * (1-h[n]) for valid n
        # grad[m_idx][i] = (1-h[i-m]) if i-m valid as n - h[i+m] if i+m valid as n+m

            g = 0.0
            # Contribution from h[i] as h[n+m]: need n = i-m, check if valid
            n_val = i - m
            if 0 <= n_val < n:

            # Contribution from h[i] as h[n]: need n = i, check if i+m valid
            im = i + m
            if 0 <= im < n:
                g -= h[im]
            grad[i] += g
    
    grad /= len(active)
    
    # Step
    h_new = h - step_size * grad
    h_new = project(h_new, n)
    return h_new, c5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-14:
        # Bisect on uniform offset
        lo, hi = -1.0, 1.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            s = np.clip(h + mid, 0, 1).sum()
            if s < target:
                lo = mid
            else:

        if abs(diff) > 1e-12:

                idx = np.argsort(-h)[:max(1, int(abs(diff)))]
                for j in idx:
                    amt = min(diff, 1.0 - h[j])
                    h[j] += amt
                    diff -= amt
                    if abs(diff) < 1e-12:
                        break
            else:

                idx = np.argsort(h)[:max(1, int(diff)+1)]
                for j in idx:

                    h[j] -= amt
                    diff -= amt
                    if abs(diff) < 1e-12:
                        break
        h = np.clip(h, 0, 1)
    return h

def subgradient_direction(h, k):
    """Compute -∂corr_k/∂h (direction to decrease corr_k)"""
    n = len(h)
    n_max = len(h) - 1  # max valid lag magnitude
    # corr_k = sum_{n valid} h[n+k]*(1-h[n])
    # ∂corr_k/∂h[i] = (1-h[i-k])*[i-k in range] - h[i+k]*[i+k in range]
    # We want -∂corr_k/∂h[i] to decrease corr_k
    grad = np.zeros(n)
    lo_n = max(0, -k)
    hi_n = min(n-1, n-1-k)
    for n_val in range(lo_n, hi_n + 1):
        i = n_val + k  # index of h[n+k]
        # ∂/∂h[n+k] = (1-h[n])
        grad[i] -= (1.0 - h[n_val])  # this is -∂ for the first term
        # ∂/∂h[n] = -h[n+k]
        grad[n_val] += h[n_val + k]  # this is -∂ for the second term
    return grad

def local_search(h, n, budget, rng, verbose=False):
    h = h.copy()
    c5 = evaluate_c5(h)
    target = n / 2.0
    start = time.time()
    best_h = h.copy()
    best_c5 = c5
    no_improve = 0
    max_no_improve = 5000
    scale = 0.5
    
    while time.time() - start < budget:
        # Check time
        if time.time() - start > budget:
            break
        
        # Choose move type
        r = rng.random()
        h_try = h.copy()
        
        if r < 0.4:
            # Subgradient step at max lag
            corr = compute_corr(h)
            k_idx = int(np.argmax(corr))
            k = k_idx - (n - 1)
            if k != 0:
                grad = subgradient_direction(h, k)
                step = scale * 0.1
                h_try = h - step * grad
                h_try = project_to_feasible(h_try, n)
        elif r < 0.7:
            # Random perturbation of 1-3 coordinates
            n_pert = int(rng.integers(1, 4))
            idx = rng.integers(0, n, size=n_pert)
            for i in idx:
                h_try[i] += rng.normal(0, scale)
            h_try = project_to_feasible(h_try, n)
        elif r < 0.85:
            # Pair swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                h_try[i], h_try[j] = h_try[j], h_try[i]
                h_try = project_to_feasible(h_try, n)
        else:
            # Small region perturbation
            start_idx = int(rng.integers(0, n - 3))
            length = int(rng.integers(2, 5))
            h_try[start_idx:start_idx+length] += rng.normal(0, scale*0.5, size=length)
            h_try = project_to_feasible(h_try, n)
        
        c5_try = evaluate_c5(h_try)
        if c5_try < c5 - 1e-12:
            h = h_try
            c5 = c5_try
            no_improve = 0
            scale = min(1.0, scale * 1.01)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Restart with larger scale
                scale = min(1.0, scale * 1.5)
                no_improve = 0
                # Small random kick
                h = project_to_feasible(h + rng.normal(0, 0.1, n), n)
                c5 = evaluate_c5(h)
        
    return h, c5

def resample(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_feasible(h_new, n_new)
