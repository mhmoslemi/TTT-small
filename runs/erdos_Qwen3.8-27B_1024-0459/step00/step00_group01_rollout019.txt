```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_obj(h, theta, penalty_backoff=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    # Smooth max
    shifted = theta * vals
    shifted -= np.max(shifted)
    smooth = (np.max(shifted) + np.log(np.sum(np.exp(shifted)))) / theta
    # Penalty for sum constraint
    s = np.sum(h) - n / 2.0
    penalty = 1e4 * s * s
    return smooth + penalty

def smooth_max_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    # Weighted gradient
    shifted = theta * vals
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    # Compute gradient of each corr_k wrt h[j]
    # corr_k = sum_n h[n+offset(k)] * (1-h[n])
    # where offset depends on k in full mode
    
    # Actually, let me use a simpler approach:
    # grad of corr wrt h can be computed as:
    # grad[j] = sum_k w_k * ((1-h[j-k_off]) - h[j+k_off])
    # where k_off is the offset for the k-th correlation entry
    
    # For np.correlate(h, 1-h, 'full'), the k-th entry (0-indexed) has shift = k-(n-1)
    # corr[k] = sum_n h[n + k - (n-1)] * (1-h[n]) for valid indices
    # Let shift = k - (n-1)
    # corr[k] = sum_n h[n + shift] * (1-h[n])
    
    # ∂corr[k]/∂h[j] = (1-h[j-shift]) * I[0≤j-shift<n] - h[j+shift] * I[0≤j+shift<n]
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        shift = k - (n - 1)
        if w[k] < 1e-12:
            continue
        # (1-h[j-shift]) for j-shift in [0, n-1]
        # i.e., j in [shift, shift+n-1] intersected with [0, n-1]
        lo = max(0, shift)
        hi = min(n, shift + n)
        if hi > lo:
            grad[lo:hi] += w[k] * (1.0 - h[np.arange(lo - shift, hi - shift)])
        # -h[j+shift] for j+shift in [0, n-1]
        # i.e., j in [-shift, n-1-shift] intersected with [0, n-1]
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if hi2 > lo2:
            grad[lo2:hi2] -= w[k] * h[np.arange(lo2 + shift, hi2 + shift)]
    
    grad *= dx
    # Add penalty gradient
    s = np.sum(h) - n / 2.0
    grad += 2e4 * s * np.ones(n)
    
    return grad

def resample_linear(h_src, n_src, n_dst):
    """Resample h from n_src points to n_dst points using linear interpolation."""
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src  # centers
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0/n_dst  # centers
    # Actually, for piecewise constant, let's use midpoint integration
    h_dst = np.interp(x_dst, x_src, h_src)
    return np.clip(h_dst, 0, 1)

def project_sum(h, n):
    """Adjust h so that sum(h) = n/2, staying in [0,1]."""
    target = n / 2.0
    current = np.sum(h)
    diff = target - current
    # Distribute evenly
    h_new = h + diff / n
    # If any go out of bounds, do iterative projection
    for _ in range(100):
        h_new = np.clip(h_new, 0, 1)
        diff = target - np.sum(h_new)
        if abs(diff) < 1e-10:
            break
        mask = (h_new > 1e-10) & (h_new < 1 - 1e-10)
        if np.sum(mask) == 0:
            break
        h_new[mask] += diff / np.sum(mask)
    return h_new

def optimize_at_resolution(h_init, n, time_limit):
    """Optimize h to minimize C5 at given resolution n."""
    h_init = project_sum(h_init, n)
    h_init = np.clip(h_init, 0, 1)
    
    current_best = h_init.copy()
    current_best_c5 = compute_c5(h_init)
    end_time = time.time() + time_limit
    
    # Try multiple theta values (increasing sharpness)
    for theta in [20, 50, 100, 200, 400]:
        if time.time() > end_time:
            break
        
        def obj(x):
            return smooth_max_obj(x, theta)
        
        def grad(x):
            return smooth_max_grad(x, theta)
        
        try:
            result = minimize(
                obj, h_init, jac=grad,
                method='L-BFGS-B',
                bounds=[(0,1)]*n,
                options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 50}
            )
            h_opt = np.clip(result.x, 0, 1)
            h_opt = project_sum(h_opt, n)
            c5 = compute_c5(h_opt)
            if c5 < current_best_c5:
                current_best_c5 = c5
                current_best = h_opt
                h_init = h_opt.copy()
        except Exception:
            pass
    
    # Final polish with direct max (no smoothing)
    if time.time() < end_time - 5:
        # Use SLSQP for final step
        def obj_direct(x):
            return compute_c5(x)
        
        try:
            result = minimize(
                obj_direct, current_best,
                method='SLSQP',
                bounds=[(0,1)]*n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            h_opt = np.clip(result.x, 0, 1)
            h_opt = project_sum(h_opt, n)
            c5 = compute_c5(h_opt)
            if c5 < current_best_c5:
                current_best_c5 = c5
                current_best = h_opt
        except Exception:
            pass
    
    return current_best, current_best_c5
```

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_max_obj(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    shifted = theta * vals
    shifted -= np.max(shifted)
    smooth = (np.max(shifted) + np.log(np.sum(np.exp(shifted)))) / theta
    s = np.sum(h) - n / 2.0
    penalty = 1e5 * s * s
    return smooth + penalty


def smooth_max_grad(h, theta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    shifted = theta * vals
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        shift = k - (n - 1)
        wv = w[k]
        lo1 = max(0, shift)
        hi1 = min(n, shift + n)
        if hi1 > lo1:
            idx = np.arange(lo1 - shift, hi1 - shift)
            grad[lo1:hi1] += wv * (1.0 - h[idx])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if hi2 > lo2:
            idx2 = np.arange(lo2 + shift, hi2 + shift)
            grad[lo2:hi2] -= wv * h[idx2]

    grad *= dx
    s = np.sum(h) - n / 2.0
    grad += 2e5 * s * np.ones(n)
    return grad


def resample_linear(h_src, n_src, n_dst):
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return np.clip(h_dst, 0.0, 1.0)


def project_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) == 0:
            h = np.clip(h + diff / n, 0, 1)
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h


def local_search(h, n, max_iter=5000, step_sizes=None):

    best_c5 = compute_c5(best_h)
    if step_sizes is None:

        improved = True
        iters = 0
        while improved and iters < max_iter:
            improved = False
            iters += 1

        result = optimize_h(h_init, n, time_alloc)
        if result is not None:
            h_opt, c5_opt = result
            if c5_opt < best_c5 - 1e-10:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = n

        # Try a perturbed start for the best resolution found so far
        if n == best_n and time.time() - t0 < budget_s - 60:
            rng = np.random.RandomState(seed + n)
            h_pert = best_h + rng.normal(0, 0.05, best_n)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = project_sum(h_pert, best_n)
            time_left = budget_s - (time.time() - t0) - 20
            if time_left > 30:
                result2 = optimize_h(h_pert, best_n, min(time_left * 0.4, 60))
                if result2 is not None:
                    h_opt2, c5_opt2 = result2
                    if c5_opt2 < best_c5 - 1e-10:
                        best_h = h_opt2
                        best_c5 = c5_opt2
                        best_n = best_n

    # Final verification and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # One more local search if time permits
    if time.time() - t0 < budget_s - 10:
        h_ls, c5_ls = local_search(best_h, best_n, max_iter=3000)
        if c5_ls < best_c5 - 1e-12:
            best_h = h_ls
            best_c5 = c5_ls

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
