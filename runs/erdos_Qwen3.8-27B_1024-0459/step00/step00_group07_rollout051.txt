import numpy as np
import time

def _compute_C(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h):
    return float(np.max(_compute_C(h)) * 2.0 / len(h))

def _project(z, target_sum):
    n = len(z)
    lo, hi = z.min() - target_sum - 1.0, z.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_smoothed(h, beta):
    n = len(h)
    C = _compute_C(h)
    Cm = C - C.max()
    w = np.exp(beta * Cm)
    w /= w.sum()
    v = 1.0 - 2.0 * h
    return np.correlate(w, v, mode='valid')

def _optimize_smoothed(h0, n, deadline, n_restarts=2):

def _upscale(h_init, n_target):
    n_init = len(h_init)
    old_x = np.linspace(0.0, 2.0, n_init)
    new_x = np.linspace(0.0, 2.0, n_target)
    return np.interp(new_x, old_x, h_init)

def _random_binary(n):
    h = np.zeros(n)
    idx = np.random.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def _random_block(n, n_blocks=4):
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size

        ones_in_block = int(frac_ones)
        positions = np.random.choice(range(start, end), size=ones_in_block, replace=False)
        h[positions] = 1.0
    return h
