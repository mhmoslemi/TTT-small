    # Ensure constraints are met
    target_sum = best_n / 2.0
    s = np.sum(best_h)
    if abs(s - target_sum) > 0.5:
        # Re-project
        best_h = np.clip(best_h, 0, 1)
        if np.sum(best_h) > 1e-10:
            best_h *= target_sum / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)
    
    best_c5 = compute_C5(best_h, best_n)
    
    return (best_h, best_c5, best_n)
