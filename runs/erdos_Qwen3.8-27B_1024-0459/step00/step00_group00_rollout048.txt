    m = np.max(vals)
    if tau * (vals[-1] - m) < -700:

        n_best = len(best_h)
        if n_best % 2 != 0:
            best_h = upsample(best_h, n_best + 1)
            n_best += 1
        h_refined, c5_refined = binary_local_search(best_h, max_time=min(remaining * 0.4, 120))
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Phase 4: Try continuous refinement on best binary solution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 5
    if remaining > 20:
        try:
            h_cont2, c5_cont2 = continuous_optimize(best_h + np.random.randn(len(best_h)) * 0.1, max_time=min(remaining * 0.5, 60))
            h_cont2 = project_onto_constraint(h_cont2, len(h_cont2))
            if c5_cont2 < best_c5:
                best_h = h_cont2
                best_c5 = c5_cont2
        except Exception:
            pass
    
    # Final: ensure constraint satisfaction and clipping
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    target = n_final / 2.0
    diff = np.sum(best_h) - target
    if abs(diff) > 1e-10:
        best_h -= diff / n_final
        best_h = np.clip(best_h, 0.0, 1.0)
    
    # Final verification
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, n_final
