2. Find k* = argmax(conv)
3. Compute τ = k* - (n-1)
4. Compute gradient: grad[m] = (1 - h[m-τ]) - h[m+τ] (with boundary handling)
5. Step: h_new = h - α * grad
6. Project h_new to constraint set
7. Repeat

This is essentially a subgradient method (since we only use the active constraint).

For the projection to {h: 0 ≤ h ≤ 1, sum(h) = n/2}:

- We can use projected gradient or solve a small convex problem
- Simple approach: Euclidean projection onto the intersection of a box and a hyperplane

import numpy as np
import time

def compute_C5(h, n_points):
    """Compute C5 = max_k sum_i h[i]*(1-h[i+k]) * dx"""
    dx = 2.0 / n_points
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv)) * dx

def project_box_hplane(v, S):
    """Project v onto {0<=x<=1, sum(x)=S} using bisection on Lagrange multiplier."""
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(p)
        if s > S:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = S - np.sum(p)
    if abs(diff) > 1e-10:
        # Distribute the difference among non-saturated coordinates
        free = (p > 1e-10) & (p < 1.0 - 1e-10)
        if np.any(free):
            p[free] += diff / np.sum(free)
            p = np.clip(p, 0.0, 1.0)
    return p

def gradient_C5(h, n_points):
    """Compute subgradient of max_k conv_k at h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(conv))
    tau = k_star - (n_points - 1)
    
    # grad[m] = (1 - h[m-tau]) - h[m+tau] with boundary handling
    grad = np.zeros(n_points)
    
    # Term 1: (1 - h[m-tau]) for valid m-tau
    if tau >= 0:
        # m - tau >= 0 => m >= tau
        # m >= tau and m < n
        valid_m = range(tau, n_points)
        for m in valid_m:
            grad[m] += (1.0 - h[m - tau])
    else:
        # m - tau = m + |tau| < n => m < n - |tau|
        valid_m = range(0, n_points + tau)  # tau is negative
        for m in valid_m:
            grad[m] += (1.0 - h[m - tau])
    
    # Term 2: -h[m+tau] for valid m+tau
    if tau >= 0:
        # m + tau < n => m < n - tau
        valid_m = range(0, n_points - tau)
        for m in valid_m:
            grad[m] -= h[m + tau]
    else:
        # m + tau >= 0 => m >= -tau
        valid_m = range(-tau, n_points)
        for m in valid_m:
            grad[m] -= h[m + tau]
    
    return grad

def compute_C5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv)) * dx

def project_h(v, n):
    target = n * 0.5
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) < 1e-12:
        return v.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        p = np.clip(v - mid, 0, 1)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5*(lo+hi), 0, 1)

def make_binary(h, n, rng):
    """Convert h to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(np.float64)
    n_ones = h_bin.sum()
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        h_bin[rng.choice(ones, int(n_ones - target), replace=False)] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        h_bin[rng.choice(zeros, int(target - n_ones), replace=False)] = 1
    return h_bin

def local_search_binary(h, n, rng, max_time, c5_current):
    """Pair-swap local search. Returns (h, c5, improved)."""
    t_start = time.time()
    c5 = c5_current
    h = h.copy()
    
    while time.time() - t_start < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]

        if c5_new < c5:
            c5 = c5_new
        else:
            h[i], h[j] = h[j], h[i]
