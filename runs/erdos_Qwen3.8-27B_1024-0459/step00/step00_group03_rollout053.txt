import numpy as np
from scipy.optimize import minimize
import math
import time

def evaluate_c5(h, dx):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    c = np.correlate(h, 1-h, mode='full')
    return np.max(c) * dx

def project_binary(h):
    """Project h to a binary vector with sum = len(h)/2"""
    n = len(h)
    target = n // 2
    # Sort by value, keep top n/2
    idx = np.argsort(-h)
    binary = np.zeros(n)
    binary[idx[:target]] = 1.0
    return binary

def objective_continuous(x, n):
    """Objective for continuous optimization: max correlation + constraint penalty"""
    h = x
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    max_c = np.max(c)
    # Penalty for constraint violation
    constraint_viol = (np.sum(h) - n/2.0) ** 2
    return max_c * dx + 100.0 * constraint_viol

def gradient_continuous(x, n):
    """Gradient of the continuous objective"""
    h = x
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    k_star = np.argmax(c)
    # k_star is index in full correlation, corresponding to lag = k_star - n + 1
    lag = k_star - n + 1
    
    grad = np.zeros(n)
    # c_k = sum_i h[i+k]*(1-h[i]) for valid i
    # ∂c_k/∂h[j] = (1-h[j-k]) if valid - h[j+k] if valid
    for j in range(n):
        g = 0.0
        # j = i+k => i = j-k
        if 0 <= j - lag < n:
            g += (1 - h[j - lag])
        # j = i => i+k = j+k
        if 0 <= j + lag < n:
            g -= h[j + lag]
        grad[j] = g
    
    return dx * grad + 200.0 * (np.sum(h) - n/2.0) * 1.0

def local_search_binary(h, n, max_iters=50000, T_init=1.0, T_final=0.01, rng=None):
    """Simulated annealing on binary vectors"""
    if rng is None:
        rng = np.random.default_rng()
    
    h = h.copy()
    dx = 2.0 / n
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    n_iter = 0
    while n_iter < max_iters:
        T = T_init * (T_final / T_init) ** (n_iter / max_iters)
        # Pick random one to remove and zero to add
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        if i == j:
            n_iter += 1
            continue
        
        # Try the flip
        h[i] = 0
        h[j] = 1
        new_c5 = evaluate_c5(h, dx)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            # Update sets
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        n_iter += 1
    
    return best_h, best_c5

        if time_left() > 30 and best_n == n:
            try:
                h_cont_init = best_h.copy()
                # Blend with sturmian
                h_s = sturmian_init(n)
                h_blend = 0.7 * h_cont_init + 0.3 * h_s
                h_blend = np.clip(h_blend, 0, 1)
                # Fix sum
                h_blend = h_blend * (n / 2.0) / h_blend.sum()
                h_blend = np.clip(h_blend, 0, 1)
                
                h_opt, c5_opt = continuous_slsqp(h_blend, n, max_iter=100)
                update_best(h_opt, n)
                
                # Also try from pure binary
                h_opt2, c5_opt2 = continuous_slsqp(h_cont_init, n, max_iter=80)
                update_best(h_opt2, n)
            except:
                pass
    
    # Final: if no good solution, make one
    if best_h is None or best_c5 > 0.5:
        n_f = 80
        attempt = 0
        while time_left() > 5 and best_c5 > 0.45 and attempt < 5:
            h_f = np.zeros(n_f)
            h_f[rng.permutation(n_f)[:n_f//2]] = 1.0
            h_f, c5_f = sa_binary(h_f, n_f, 50000, 0.3, 0.01, rng,
                                    time_limit=time_left() - 5, t_start=time.time())
            update_best(h_f, n_f)
            attempt += 1
    
    if best_h is None:
        n_f = 70
        best_h = np.zeros(n_f)
        best_h[:n_f//2] = 1.0
        best_n = n_f
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, 2.0 / best_n)
    
    return (best_h, final_c5, best_n)
