        return m

    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

    """Smooth max of corr * dx using log-sum-exp."""
    vals = corr * dx
    m = np.max(vals)
    if not np.isfinite(m):
        return m
    shifted = vals - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

    Gradient of smooth max of corr * dx w.r.t. h.
    corr = np.correlate(h, 1-h, mode='full')

    # corr is indexed from 0 to 2n-2, corresponding to lags -(n-1) to n-1

        if w < 1e-14:
            continue
        if k >= n - 1:
            lag = k - (n - 1)
            if lag >= n:
                continue
            grad[:n-lag] += w * (1 - h[lag:n])
            grad[lag:] -= w * h[:n-lag]
        else:
            m_lag = n - 1 - k
            if m_lag >= n:
                continue
            grad[m_lag:] += w * (1 - h[:n-m_lag])
            grad[:n-m_lag] -= w * h[m_lag:]
    
    s = np.sum(h) - target_sum
    grad += 2 * lam * s * np.ones(n)
    
    return obj + lam * s * s, grad

def project_h(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum."""
    h = np.clip(h, 0, 1)
    for _ in range(20):
        s = np.sum(h)
        if abs(s - target_sum) < 1e-12:
            break
        h = h * target_sum / s
        h = np.clip(h, 0, 1)
        # If clipping changed the sum, rescale
        h2 = h * target_sum / np.sum(h)
        if np.max(h2) > 1 or np.min(h2) < -1e-10:
            h = np.clip(h2, 0, 1)
        else:
            h = h2
            break
    return h

import numpy as np
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

def smooth_max_obj(h, dx, beta, lam, target_sum):
    corr = np.correlate(h, 1 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = vals - m
    max_shifted = np.max(shifted)
    if max_shifted > 500:
        return m + max_shifted / beta + lam * (np.sum(h) - target_sum) ** 2
    exps = np.exp(beta * shifted)
    result = m + np.log(np.sum(exps)) / beta
    s = np.sum(h) - target_sum
    result += lam * s * s
    return result

def smooth_max_grad(h, dx, beta, lam, target_sum):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    shifted = vals - m
    max_shifted = np.max(shifted)
    if max_shifted > 500:
        weights = np.zeros_like(vals)
        weights[shifted == max_shifted] = 1.0 / np.sum(shifted == max_shifted)
    else:
        exps = np.exp(beta * shifted)
        weights = exps / np.sum(exps)
        obj_val = m + np.log(np.sum(exps)) / beta
    grad = np.zeros(n)

        w = weights[k] * dx

            continue

                continue
            grad[m_lag:] += w * (1 - h[:n - m_lag])
            grad[:n - m_lag] -= w * h[m_lag:]
    s = np.sum(h) - target_sum

    return h

def exact_max_grad_step(h, dx, target_sum, step_size=0.01):

    vals = corr * dx

    grad = np.zeros(n)
    if k_star >= n - 1:
        lag = k_star - (n - 1)
        if lag >= n:
            return h
        grad[:n - lag] += (1 - h[lag:n])
        grad[lag:] -= h[:n - lag]
    else:
        m_lag = n - 1 - k_star
        if m_lag >= n:
            return h
        grad[m_lag:] += (1 - h[:n - m_lag])
        grad[:n - m_lag] -= h[m_lag:]
    h_new = h - step_size * grad
    h_new = project_h(h_new, target_sum)
    return h_new
