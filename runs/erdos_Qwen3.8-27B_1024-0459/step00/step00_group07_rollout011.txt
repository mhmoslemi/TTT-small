import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5, corr

def get_gradient_for_lag(h, d):
    """
    Gradient of correlate(h, 1-h)[n-1+d] with respect to h.
    d can be positive, negative, or zero.
    """
    n = len(h)
    grad = np.zeros(n)
    
    if d >= 0:
        # result = sum_{i=d}^{n-1} h[i] * (1-h[i-d])
        # d/dh[j]:
        #   j from d to n-1: (1-h[j-d])
        #   j from 0 to n-1-d: -h[j+d]
        if d == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[d:] += (1.0 - h[:n-d])
            grad[:n-d] += (-h[d:])
    else:
        # result = sum_{i=0}^{n-1+d} h[i] * (1-h[i-d]) where -d > 0
        # Let e = -d > 0
        # result = sum_{i=0}^{n-1-e} h[i] * (1-h[i+e])
        # d/dh[j]:
        #   j from 0 to n-1-e: (1-h[j+e])
        #   j from e to n-1: -h[j-e]
        e = -d
        grad[:n-e] += (1.0 - h[e:])
        grad[e:] += (-h[:n-e])
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def lag_gradient(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        if d == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[d:] += (1.0 - h[:n-d])
            grad[:n-d] += (-h[d:])
    else:
        e = -d
        grad[:n-e] += (1.0 - h[e:])
        grad[e:] += (-h[:n-e])
    return grad

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    wsum = np.sum(exp_c)
    weights = exp_c / wsum
    obj = c_max + np.log(wsum) / alpha
    m = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] > 1e-10:
            d = k - m
            grad += weights[k] * lag_gradient(h, d) * dx
    return obj, grad

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5, _ = evaluate_c5(h)
    
    obj, grad = smooth_obj_grad(h, alpha_schedule[0])
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 500 == 0:
            c5_cur, _ = evaluate_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        
        # Anneal alpha based on progress
        frac = min(1.0, it / (max_iter * 0.5))
        alpha_idx = min(int(frac * len(alpha_schedule)), len(alpha_schedule) - 1)
        alpha = alpha_schedule[alpha_idx]
        
        obj, grad = smooth_obj_grad(h, alpha)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        
        # Step size: diminish slowly
        lr = lr_init / (1.0 + 0.005 * it)
        step = lr / (g_norm + 1e-10)
        step = min(step, 0.5)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        
        # Verify improvement (cheap check every few iters)
        if it % 5 == 0:
            c5_new, _ = evaluate_c5(h_new)
            c5_cur, _ = evaluate_c5(h)
            if c5_new > c5_cur + 0.005:
                # Rejected, halve step
                step *= 0.3
                h_new = h - step * grad
                h_new = project_feasible(h_new, n)
            h = h_new
        else:
            h = h_new
    
    c5_final, _ = evaluate_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    
    return best_h, best_c5

def coordinate_descent_polish(h0, n, deadline, max_rounds=500):
    h = h0.copy().astype(float)
    best_h = h.copy()
    best_c5, _ = evaluate_c5(h)
    target = n / 2.0
    
    for r in range(max_rounds):

        improved = False
        c5_cur, corr = evaluate_c5(h)
        max_k = np.argmax(corr)

            if time.time() > deadline:
                break

                h_try = h.copy()
                h_try[i] += delta
                h_try[i] = np.clip(h_try[i], 0.0, 1.0)

                diff = target - np.sum(h_try)
                if abs(diff) > 1e-12:

                    elif n > 1:

                    else:
                        continue
                h_try = project_feasible(h_try, n)

                if c5_try < best_c5 - 1e-8:

                    best_c5 = c5_try
                    best_h = h.copy()
