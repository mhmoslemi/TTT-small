def _resample(parent, n):
    m=len(parent)

    x_new = np.linspace(0.5/n, 1-0.5/n, n)
    interp = np.interp(x_new, x_old, parent)
    target = n/2.0
    # adjust area to target: scale about 0.5
    for _ in range(10):
       shift = (target - interp.sum())/n
       interp = np.clip(interp + shift, 0,1)
    return interp

import numpy as np
import math
import time


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _eval_counts(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        zero = np.zeros(0, dtype=float)
        return zero, zero, 0.0

    s = float(h.sum())
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])

    need = 2 * n - 1
    L = _next_pow2(need)
    spec = np.fft.rfft(h, L)
    ac = np.fft.irfft(spec * np.conj(spec), n=L).real[:n]

    idx = n - np.arange(n, dtype=int)
    pos = prefix[idx] - ac
    neg = s - prefix[:n] - ac

    eps = 1e-12
    pos = np.where((pos < 0.0) & (pos > -eps), 0.0, pos)
    neg = np.where((neg < 0.0) & (neg > -eps), 0.0, neg)
    return pos, neg, s


def _c5_from_counts(pos, neg, dx):
    if pos.size == 0:
        return 0.0
    m1 = float(pos.max())
    m2 = float(neg.max())
    return dx * (m1 if m1 > m2 else m2)


def _direct_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    return float(corr.max()) * (2.0 / float(n))


def _make_half_exact(h, target_sum):
    h = np.asarray(h, dtype=float)
    n = len(h)
    h2 = np.rint(2.0 * h).astype(int)
    target = int(round(2.0 * target_sum))
    total = int(h2.sum())

    while total < target:
        mask = h2 < 2
        if not np.any(mask):
            break
        idxs = np.nonzero(mask)[0]
        vals = h[idxs]
        pick = idxs[int(np.argmax(vals))]
        h2[pick] += 1
        total += 1

    while total > target:
        mask = h2 > 0
        if not np.any(mask):
            break
        idxs = np.nonzero(mask)[0]
        vals = h[idxs]
        pick = idxs[int(np.argmin(vals))]
        h2[pick] -= 1
        total -= 1

    return 0.5 * h2.astype(float)


def _make_binary_from_half(h, target_sum):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = int(round(target_sum))
    ones = (h > 0.5).astype(int)
    count = int(ones.sum())

    if count > target:
        idxs = np.nonzero(ones == 1)[0]
        if idxs.size:
            order = np.argsort(h[idxs])
            for k in range(count - target):
                ones[idxs[order[k]]] = 0
    elif count < target:
        idxs = np.nonzero(ones == 0)[0]
        if idxs.size:
            order = np.argsort(-h[idxs])
            for k in range(target - count):
                ones[idxs[order[k]]] = 1

    return ones.astype(float)


def _resample_parent(parent, n):
    parent = np.asarray(parent, dtype=float)
    m = len(parent)
    if m == 0:
        return np.full(n, 0.5, dtype=float)
    if m == n:
        return parent.copy()

    x_old = np.linspace(0.5 / float(m), 1.0 - 0.5 / float(m), m)
    x_new = np.linspace(0.5 / float(n), 1.0 - 0.5 / float(n), n)
    h = np.interp(x_new, x_old, parent)
    target = n / 2.0

    for _ in range(30):
        shift = (target - float(h.sum())) / float(n)
        if abs(shift) < 1e-15:
            break
        h = np.clip(h + shift, 0.0, 1.0)

    return h


def _random_binary(n, seed):
    rng = np.random.default_rng(seed)
    s = n // 2
    h = np.zeros(n, dtype=float)
    if s:
        idx = rng.permutation(n)[:s]
        h[idx] = 1.0
    return h

    else:

    return _make_binary_from_half(h, n / 2.0)


def _best_rotation_binary(h, max_shifts=None):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n <= 1:
        return h.copy()
    if max_shifts is None:
        max_shifts = n if n <= 96 else 16
    max_shifts = int(max_shifts)
    if max_shifts <= 0:
        return h.copy()
    if max_shifts >= n:
        max_shifts = n

    best = h.copy()
    best_c = _direct_c5(h)

        rh = np.roll(h, k)
        c = _direct_c5(rh)
        if c < best_c:
            best_c = c
            best = rh
    return best


def _init_binary(parent, n, noise, seed):
    if parent is None or len(np.asarray(parent)) == 0:
        h = _random_binary(n, seed)
    else:
        h = _resample_parent(parent, n)
        if noise > 0.0:
            rng = np.random.default_rng(seed)
            rnd = rng.random(n)
            h = (1.0 - noise) * h + noise * rnd

        else:

def _candidate_thetas(h, p, q, step):
    if p > q:
        p, q = q, p
    u = float(h[p])
    v = float(h[q])

    if step == 0.0:
        th = v - u
        if abs(th) > 1e-12:
            return [th]
        return []

    lo = max(-u, v - 1.0)
    hi = min(1.0 - u, v)
    if hi <= lo + 1e-12:
        return []

    a = int(math.ceil(2.0 * lo - 1e-9))
    b = int(math.floor(2.0 * hi + 1e-9))

        nq = v - th
        if -1e-9 <= np_ <= 1.0 + 1e-9 and -1e-9 <= nq <= 1.0 + 1e-9:

                        if updates % 3000 == 0:

                            cur_c = _c5_from_counts(pos, neg, dx)
                            scratch1 = np.empty(n, dtype=float)
                            scratch2 = np.empty(n, dtype=float)
                            if cur_c < best_c - 1e-11:
                                best_c = cur_c
                                best_h = h.copy()

            if not improved:
                break

    if timef() < deadline:
        rng = np.random.default_rng(seed)
        T = 0.06
        Tmin = 0.0004
        no_improve = 0
        if n <= 64:
            max_no_improve = 5000
        elif n <= 256:
            max_no_improve = 12000
        else:
            max_no_improve = 20000
        restarts = 0
        max_restarts = 2

        while timef() < deadline - 1.0:
            if no_improve >= max_no_improve:
                if restarts < max_restarts:
                    h = best_h.copy()
                    pos, neg, _ = _eval_counts(h)
                    cur_c = _c5_from_counts(pos, neg, dx)
                    scratch1 = np.empty(n, dtype=float)
                    scratch2 = np.empty(n, dtype=float)
                    temp = np.empty(n, dtype=float)
                    restarts += 1
                    no_improve = 0
                    T = 0.09
                else:
                    break

            p = int(rng.integers(0, n - 1))
            q = int(rng.integers(p + 1, n))

            if step == 0.0:
                u = float(h[p])
                v = float(h[q])
                theta = v - u
                if abs(theta) < 1e-12:
                    continue
                c = _move_candidate_c5(pos, neg, h, p, q, theta, dx, scratch1, scratch2, temp)
            else:
                thetas = _candidate_thetas(h, p, q, 0.5)
                if not thetas:
                    continue
                theta = thetas[int(rng.integers(len(thetas)))]
                c = _move_candidate_c5(pos, neg, h, p, q, theta, dx, scratch1, scratch2, temp)

            if c < cur_c - 1e-11:
                delta = c - cur_c
                do_accept = True
            else:
                delta = c - cur_c
                do_accept = False
                if delta < 5.0 * T:
                    if rng.random() < math.exp(-delta / max(T, 1e-12)):
                        do_accept = True

            if do_accept:
                np_ = float(h[p]) + theta
                nq = float(h[q]) - theta
                h[p] = min(1.0, max(0.0, np_))
                h[q] = min(1.0, max(0.0, nq))
                oldpos, oldneg = pos, neg
                pos, neg = scratch1, scratch2
                scratch1, scratch2 = oldpos, oldneg
                cur_c = c
                no_improve = 0
                updates += 1
                if c < best_c - 1e-11:
                    best_c = c
                    best_h = h.copy()

                if delta <= 0.0:
                    T = max(Tmin, T * 0.995)
                else:
                    T = max(Tmin, T * 0.98)

                if updates % 3000 == 0:
                    pos, neg, _ = _eval_counts(h)
                    cur_c = _c5_from_counts(pos, neg, dx)
                    scratch1 = np.empty(n, dtype=float)
                    scratch2 = np.empty(n, dtype=float)
                    temp = np.empty(n, dtype=float)
                    if cur_c < best_c - 1e-11:
                        best_c = cur_c
                        best_h = h.copy()
            else:
                no_improve += 1
                T = max(Tmin, T * 0.9995)

    if step == 0.0:
        if n % 2 == 0:
            best_h = _make_binary_from_half(best_h, n / 2.0)
        else:
            best_h = _make_half_exact(best_h, n / 2.0)
    elif step == 0.5:
        best_h = _make_half_exact(best_h, n / 2.0)
    else:
        best_h = np.clip(best_h, 0.0, 1.0)

    c_out = _direct_c5(best_h)
    return best_h, c_out


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    t0 = time.time()
    deadline = t0 + budget_s

    try:
        parent = initial_h_values
    except NameError:
        parent = None

    if parent is not None:
        parent_arr = np.asarray(parent, dtype=float)
        parent_arr = np.clip(parent_arr, 0.0, 1.0)
    else:
        parent_arr = None

    best_h = None
    best_c = float("inf")
    best_step = None

    if parent_arr is not None:
        c0 = _direct_c5(parent_arr)
        best_h = parent_arr.copy()
        best_c = c0
        best_step = None

    if time.time() > deadline - 20.0:
        if best_h is None:
            best_h = _random_binary(48, seed)
            best_c = _direct_c5(best_h)
        return best_h, float(best_c), len(best_h)

    if parent_arr is not None:
        m = len(parent_arr)
        base_n = m if m % 2 == 0 else m + 1
        if base_n < 48:
            base_n = 48
    else:
        base_n = 48

    cases = []

    if parent_arr is not None and len(parent_arr) <= 128:
        hp = _make_half_exact(parent_arr, len(parent_arr) / 2.0)
        hp = _best_rotation_binary(hp, min(len(parent_arr), 64))
        cases.append((len(parent_arr), hp, 0.5, seed + 777))

    cases.append((base_n, _init_binary(parent_arr, base_n, 0.0, seed), 0.0, seed))
    cases.append((base_n, _init_binary(parent_arr, base_n, 0.35, seed + 1), 0.0, seed + 1))

    if parent_arr is None:
        cases.append((base_n, _pattern_binary(base_n, 1, seed + 2), 0.0, seed + 2))
        cases.append((base_n, _random_binary(base_n, seed + 3), 0.0, seed + 3))
    else:
        cases.append((base_n, _random_binary(base_n, seed + 2), 0.0, seed + 2))

    for n in (64, 96, 160, 256):
        if n == base_n:
            continue
        if time.time() > deadline - 60.0:
            break
        if parent_arr is None:
            init = random_h = _random_binary(n, seed + 20 + n)
            init = 0.7 * init + 0.3 * _pattern_binary(n, 1, seed + 30 + n)
            init = np.clip(init, 0.0, 1.0)
            init = _make_binary_from_half(init, n / 2.0)
            init = _best_rotation_binary(init, None)
        else:
            noise = 0.15 + 0.35 * min(1.0, float(n - base_n) / (float(base_n) * 2.0))
            init = _init_binary(parent_arr, n, noise, seed + 20 + n)
        cases.append((n, init, 0.0, seed + 20 + n))

    reserved = 30.0
    rem = deadline - time.time() - reserved
    if rem < 30.0 and cases:
        cases = cases[:1]
        rem = max(10.0, deadline - time.time() - 2.0)

    case_budget = max(15.0, rem / max(1, len(cases)))
    case_budget = min(260.0, case_budget)

    for n, init_h, step, sd in cases:
        if time.time() > deadline - reserved:
            break
        dl = min(deadline - reserved, time.time() + case_budget)
        if dl <= time.time() + 2.0:
            dl = time.time() + 2.0
        h_res, c_res = _search_once(init_h, dl, sd, step=step)
        if h_res is not None and c_res < best_c - 1e-12:
            best_h = h_res
            best_c = float(c_res)
            best_step = step

    if best_c > 0.38085 + 1e-8:
        extra = 0
        while best_c > 0.38085 + 1e-8 and time.time() < deadline - 60.0 and extra < 6:
            sd = seed + 1000 + 17 * extra
            if extra % 2 == 0:
                init = _random_binary(base_n, sd)
            else:
                init = _pattern_binary(base_n, 1 + (extra % 3), sd)
                init = _best_rotation_binary(init, None)
            dl = min(deadline - 30.0, time.time() + 28.0)
            h_res, c_res = _search_once(init, dl, sd, step=0.0)
            if h_res is not None and c_res < best_c - 1e-12:
                best_h = h_res
                best_c = float(c_res)
                best_step = 0.0
            extra += 1

    if best_h is not None and best_step is not None and time.time() < deadline - 45.0:
        dl = deadline - 10.0
        h_res, c_res = _search_once(best_h, dl, seed + 5000, step=best_step)
        if h_res is not None and c_res < best_c - 1e-12:
            best_h = h_res
            best_c = float(c_res)

    if best_h is None:
        best_h = _random_binary(48, seed)
        best_c = _direct_c5(best_h)

    best_h = np.asarray(best_h, dtype=float)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = _direct_c5(best_h)

    return best_h, float(best_c), len(best_h)
