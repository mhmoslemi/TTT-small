import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_hyperplane(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def objective_and_grad(x, n):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = np.max(corr) * dx
    # Gradient
    idx = np.argmax(corr)
    lag = idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag > 0:
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] += -h[lag:]
        else:
            grad[:] += (1.0 - h)
            grad[:] += -h
    else:
        m = -lag
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] += -h[:n-m]
    grad *= dx
    return val, grad

            grad[:m] += one_minus_h[m_abs:]
            grad[m_abs:] += -h[:m]
    grad *= dx
    return val, grad


def slsqp_constraints_eq(x, target_sum, n):
    return np.sum(x) - target_sum


def slsqp_constraints_jac(x, target_sum, n):
    return np.ones(n)


def optimize_slsqp(h0, n, time_limit):
    target_sum = n / 2.0
    best_val = evaluate_c5(h0)
    best_h = h0.copy()
    deadline = time.time() + time_limit

    def fun(x):
        return evaluate_c5(np.clip(x, 0.0, 1.0))

    def jac(x):
        _, g = c5_grad(np.clip(x, 0.0, 1.0), n)
        return g

    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': fun_eq, 'jac': fun_eq_jac}],
                       options={'maxiter': 2000, 'ftol': 1e-14})
        if time.time() < deadline and res.fun < best_val:
            best_h = project_box_hyperplane(res.x, target_sum)
            best_val = evaluate_c5(best_h)
    except Exception:
        pass
    return best_h, best_val


def fun_eq(x):
    n = len(x)
    return np.sum(x) - n / 2.0


def fun_eq_jac(x):
    return np.ones(len(x))


def optimize_projected_grad(h0, n, time_limit):
    target_sum = n / 2.0
    h = project_box_hyperplane(h0, target_sum)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    deadline = time.time() + time_limit
    lr = 0.05
    stale = 0

    while time.time() < deadline:
        val, grad = c5_grad(h, n)
        grad_proj = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_proj)
        if gnorm < 1e-12:
            break
        grad_proj = grad_proj / gnorm * gnorm

        improved = False
        for _ in range(40):
            h_new = project_box_hyperplane(h - lr * grad, target_sum)
            val_new = evaluate_c5(h_new)
            if val_new < best_val - 1e-14:
                best_val = val_new
                best_h = h_new.copy()
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            stale += 1
            lr *= 0.5
            if lr < 1e-14:
                break
            if stale > 100:
                break
        else:
            stale = 0
            lr = min(lr * 1.05, 0.5)

    return best_h, best_val


def interp_h(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    return project_box_hyperplane(h_new, target)


def optimize_ltrust(h0, n, time_limit):
    """Optimize using trust-constr method for medium n."""
    target_sum = n / 2.0
    best_val = evaluate_c5(h0)
    best_h = h0.copy()
    deadline = time.time() + time_limit

    def fun(x):
        return evaluate_c5(np.clip(x, 0.0, 1.0))

    def jac(x):
        _, g = c5_grad(np.clip(x, 0.0, 1.0), n)
        return g

    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 500, 'ftol': 1e-14, 'maxls': 50})
        h_res = project_box_hyperplane(res.x, target_sum)
        if evaluate_c5(h_res) < best_val:
            best_h = h_res.copy()
            best_val = evaluate_c5(best_h)
    except Exception:
        pass
    return best_h, best_val


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    # Get initial solution from parent
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.zeros(45)
        h_init[:23] = 1.0
        h_init[22] = 0.5
        h_init = project_box_hyperplane(h_init, 22.5)

    n_init = len(h_init)
    h_init = project_box_hyperplane(h_init, n_init / 2.0)
    best_h = h_init.copy()
    best_val = evaluate_c5(h_init)

    # Phase 1: SLSQP at initial resolution (~45)
    alloc1 = budget_s * 0.12
    if elapsed() < alloc1:
        best_h, best_val = optimize_slsqp(best_h, n_init, alloc1 - elapsed())

    # Phase 2: Go to 2x resolution
    n2 = n_init * 2
    alloc2 = budget_s * 0.40
    if elapsed() < alloc2:
        h2 = interp_h(best_h, n_init, n2)
        t_left = alloc2 - elapsed()
        best_h2, val2 = optimize_slsqp(h2, n2, t_left)
        if val2 < best_val:
            best_h, best_val = best_h2, val2

    # Phase 3: Go to 4x resolution
    n3 = n_init * 4
    alloc3 = budget_s * 0.65
    if elapsed() < alloc3:
        cur_n = len(best_h)
        h3 = interp_h(best_h, cur_n, n3)
        t_left = alloc3 - elapsed()
        if n3 <= 200:
            best_h3, val3 = optimize_slsqp(h3, n3, t_left)
        else:
            best_h3, val3 = optimize_projected_grad(h3, n3, t_left)
        if val3 < best_val:
            best_h, best_val = best_h3, val3

    # Phase 4: Try 6x or 8x resolution with projected gradient
    n4 = n_init * 6
    alloc4 = budget_s * 0.82
    if elapsed() < alloc4:
        cur_n = len(best_h)
        h4 = interp_h(best_h, cur_n, n4)
        t_left = alloc4 - elapsed()
        best_h4, val4 = optimize_projected_grad(h4, n4, t_left * 0.7)
        if val4 < best_val:
            best_h, best_val = best_h4, val4

    # Phase 5: Random perturbations at best resolution
    remaining = budget_s - elapsed() - 8
    if remaining > 15:
        n_cur = len(best_h)
        if n_cur <= 90:
            n_trials = int(remaining / 40)
        else:
            n_trials = int(remaining / 80)
        n_trials = max(1, min(n_trials, 12))

        for i in range(n_trials):
            if elapsed() > budget_s - 10:
                break
            sigma = 0.03 + 0.15 * (i / max(1, n_trials - 1))
            h_pert = best_h + sigma * np.random.randn(n_cur)
            h_pert = project_box_hyperplane(h_pert, n_cur / 2.0)
            t_trial = min(40.0, budget_s - 12 - elapsed())
            if t_trial < 3:
                break
            if n_cur <= 90:
                h_new, v_new = optimize_slsqp(h_pert, n_cur, t_trial)
            else:
                h_new, v_new = optimize_projected_grad(h_pert, n_cur, t_trial)
            if v_new < best_val:
                best_h, best_val = h_new, v_new

    # Phase 6: Final polish at best resolution
    remaining = budget_s - elapsed() - 3
    if remaining > 5:
        n_cur = len(best_h)
        if n_cur <= 90:
            best_h, best_val = optimize_slsqp(best_h, n_cur, remaining)
        else:
            best_h, best_val = optimize_projected_grad(best_h, n_cur, remaining)

    # Final validation and cleanup
    n_final = len(best_h)
    best_h = project_box_hyperplane(best_h, n_final / 2.0)
    # Ensure exact constraint satisfaction (numerical cleanup)
    diff = np.sum(best_h) - n_final / 2.0
    if abs(diff) > 1e-10:
        best_h = best_h - diff / n_final
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_box_hyperplane(best_h, n_final / 2.0)
    best_val = evaluate_c5(best_h)

    return best_h, best_val, n_final
