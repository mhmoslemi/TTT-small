            best_n = n_curr
        h_curr = h_new

    # Phase 2: Subgradient descent
    if remaining() > 30:
        h_sub, c5_sub = subgradient_descent(best_h, best_n, 30000, 0.05, remaining)
        if c5_sub < best_c5 - 1e-10:
            best_c5 = c5_sub
            best_h = h_sub.copy()
            best_n = best_n

    # Phase 3: Local perturbation search
    if remaining() > 20:
        h_loc, c5_loc = local_perturbation_search(best_h, best_n, remaining, rng)
        if c5_loc < best_c5 - 1e-10:
            best_c5 = c5_loc
            best_h = h_loc.copy()
            best_n = best_n

    # Phase 4: Another SLSQP pass on improved solution
    for alpha in [100, 300]:
        if remaining() < 30:
            break
        h_new, c5_new = slsqp_optimize(best_h, best_n, alpha, 200, remaining)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = best_n

    # Phase 5: Multi-resolution - interpolate to finer grids
    for factor in [2, 3]:
        if remaining() < 60:
            break
        n_fine = best_n * factor
        h_fine = interpolate_h(best_h, factor)
        h_fine = fix_sum(h_fine, n_fine)
        c5_fine = compute_c5(h_fine, n_fine)

        for alpha in [100, 300]:
            if remaining() < 30:
                break
            h_new, c5_new = slsqp_optimize(h_fine, n_fine, alpha, 200, remaining)
            if c5_new < c5_fine - 1e-10:
                c5_fine = c5_new
                h_fine = h_new

        if c5_fine < best_c5 - 1e-10:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            best_n = n_fine

    # Phase 6: Final refinement SLSQP
    if remaining() > 20:
        h_new, c5_new = slsqp_optimize(best_h, best_n, 500, 300, remaining)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = best_n

    # Phase 7: Final local search
    if remaining() > 10:
        h_loc, c5_loc = local_perturbation_search(best_h, best_n, remaining, rng)
        if c5_loc < best_c5 - 1e-10:
            best_c5 = c5_loc
            best_h = h_loc.copy()
            best_n = best_n

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = fix_sum(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return (best_h, best_c5, best_n)
