import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def generate_thue_morse(n):
    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    c = int(tm.sum())
    if c != n // 2:
        if c > n // 2:
            ones_idx = np.where(tm == 1)[0]
            tm[ones_idx[:c - n // 2]] = 0
        else:
            zeros_idx = np.where(tm == 0)[0]
            tm[zeros_idx[:n // 2 - c]] = 1
    return tm.astype(float)


def make_binary(h_init, n):
    target = n // 2
    idx = np.argsort(-h_init)
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result


def local_search(h, n, max_iters=1000, time_limit=60.0, rng=None):
    if rng is None:

    t0 = time.time()
    h = h.copy()
    c5 = compute_c5(h)

        if time.time() - t0 > time_limit:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_max = float(corr.max())

        best_delta = 0.0
        best_i = -1
        best_j = -1

                    h[i] = 1.0
                    h[j] = 0.0
                    nm = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                    h[i] = 0.0
                    h[j] = 1.0
                    d = curr_max - nm
                    if d > best_delta:
                        best_delta = d
                        best_i = i
                        best_j = j
        else:
