```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    return dx * np.max(corr), corr

def compute_lag_gradient(h, n, k):
    """Compute gradient of c[lag_index] w.r.t. h, where lag_index corresponds to shift k."""
    # k is the actual shift: output[k + n - 1]
    npts = n
    grad = np.zeros(npts)
    for j in range(npts):
        # Contribution from h[n+k] where n = j-k
        if 0 <= j - k < npts:
            grad[j] += (1.0 - h[j - k])
        # Contribution from -h[n+k]h[n] where n = j
        if 0 <= j + k < npts:
            grad[j] += -h[j + k]
    return grad

def project_polytope(v, target_sum):
    """Project v onto {h: 0 <= h_i <= 1, sum(h) = target_sum}"""
    n = len(v)
    # Bisection on threshold
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = 0.5 * (lo + hi)
    proj = np.clip(v - mid, 0.0, 1.0)
    # Exact correction
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-10:
        free = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nf = np.sum(free)
        if nf > 0:
            adj = diff / nf
            proj[free] = np.clip(proj[free] + adj, 0.0, 1.0)
            # Re-check
            diff2 = target_sum - np.sum(proj)
            if abs(diff2) > 1e-10 and nf > 0:
                # Force: adjust one free variable
                idx = np.where(free)[0][0]
                proj[idx] = np.clip(proj[idx] + diff2, 0.0, 1.0)
    return proj

def optimize_subgradient(h, n, max_iterations=5000, initial_lr=0.05, tol=1e-8):
    """Subgradient descent for minimizing max c[l]."""
    h = h.copy()
    target = n / 2.0
    best_c5 = compute_c5(h, n)[0]
    best_h = h.copy()
    
    lr = initial_lr
    no_improve = 0
    
    for it in range(max_iterations):
        # Compute all correlations
        dx = 2.0 / n
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode="full")
        current_c5 = dx * np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200 and lr < 1e-6:
            break
        
        # Find active lags (top few)
        max_corr = np.max(corr)
        active = np.where(corr > 0.9 * max_corr)[0]
        
        # Compute subgradient (average of gradients of active lags)
        grad = np.zeros(n)
        for lag_idx in active:
            k = lag_idx - (n - 1)  # actual shift
            g = np.zeros(n)
            for j in range(n):
                if 0 <= j - k < n:
                    g[j] += (1.0 - h[j - k])
                if 0 <= j + k < n:
                    g[j] += -h[j + k]
            grad += g / len(active)
        
        # Step
        h_new = h - lr * grad
        h = project_polytope(h_new, target)
        
        # Decay lr
        lr = max(initial_lr * (0.9995 ** it), 1e-7)
    
    return best_h, best_c5
```
