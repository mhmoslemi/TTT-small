def compute_subgradient(h, n_points, top_k=5):
    """Compute subgradient of C5/dx at h"""
    n = n_points
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    
    # Find top lags
    top_indices = np.argsort(conv)[::-1][:top_k]
    
    grad = np.zeros(n)
    for j in top_indices:
        l = j - (n - 1)  # lag
        if l >= 0:
            # df_l/dh[k] = (1 - h[k-l]) * [l ≤ k ≤ n-1] - h[k+l] * [k ≤ n-1-l]
            for k in range(l, n):
                grad[k] += (1.0 - h[k-l])
            for k in range(0, n-l):
                grad[k] -= h[k+l]
        else:
            m = -l
            # df_{-m}/dh[k] = (1 - h[k+m]) * [k ≤ n-1-m] - h[k-m] * [k ≥ m]
            for k in range(0, n-m):
                grad[k] += (1.0 - h[k+m])
            for k in range(m, n):
                grad[k] -= h[k-m]
    
    grad /= len(top_indices)
    return grad, float(np.max(conv) * dx)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def compute_gradient_lag(h, lag):
    """Compute gradient of the overlap at a specific lag."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        # f = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        # df/dh[k] = (1-h[k-lag]) for k in [lag, n-1]
        #           - h[k+lag] for k in [0, n-1-lag]
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] -= h[lag:]
    else:
        m = -lag
        # f = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
        # df/dh[k] = (1-h[k+m]) for k in [0, n-1-m]
        #           -h[k-m] for k in [m, n-1]
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    return grad

def compute_subgradient(h, top_k=5):
    n = len(h)
    conv = np.correlate(h, 1.0 - h, mode="full")

        lag = j - (n - 1)
        grad += compute_gradient_lag(h, lag)

def gradient_for_lag(h, lag):

    grad = np.zeros(n)

        grad += gradient_for_lag(h, lag)
