```
def select_top_weights(corr, tau=0.006, topk=12):
    m=corr.size
    if m==0: return np.array([],int), np.array([])
    maxc=corr.max()
    order=np.argsort(corr, kind='mergesort')[::-1]
    k=min(topk,m)
    order=order[:k]
    # also include any within tol? If topk not include due many equal? okay.
    z=(corr[order]-maxc)/tau
    # to avoid underflow: z - z.max()
    w=np.exp(np.minimum(z, 0.0)) # since z<=0
    s=w.sum()
    if s==0: w[0]=1
    else w/=s
    return order, w
```

```
def enforce_exact_sum(h,target):

    h=np.clip(h,0,1)
    s=float(h.sum())
    diff=target-s
    if abs(diff)<1e-12: return h
    if diff>0:
       order=np.argsort(h)  # lowest first
    else:
       order=np.argsort(-h)
    rem=diff
    for idx in order:
       if abs(rem)<1e-12: break
       if rem>0:
          add=min(rem,1.0-h[idx])
```

```python
def eval_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0, np.array([], dtype=float)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    return float(np.max(c)), c


def eval_score(h):
    c, _ = eval_c5(h)
    return c


def project_boxsum(v, target):
    v = np.asarray(v, dtype=float).copy()
    if v.size == 0:
        return v
    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > float(v.size):
        target = float(v.size)
    if abs(float(np.sum(v)) - target) <= 1e-12 and np.all(v >= -1e-12) and np.all(v <= 1.0 + 1e-12):
        return np.clip(v, 0.0, 1.0)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    delta = 0.5 * (lo + hi)
    return np.clip(v - delta, 0.0, 1.0)


def enforce_exact_sum(h, target):

def select_top_weights(corr, tau=0.004, topk=16):
    corr = np.asarray(corr, dtype=float)
    m = corr.size
    if m == 0:
        return np.array([], dtype=int), np.array([], dtype=float)
    k = min(int(topk), m)
    if k == 1:
        idx = np.array([int(np.argmax(corr))], dtype=int)
        return idx, np.array([1.0], dtype=float)
    idx = np.argpartition(corr, -k)[-k:]
    maxc = float(corr.max())
    z = (corr[idx] - maxc) / float(tau)
    z = z - float(np.max(z))
    z = np.minimum(z, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if not math.isfinite(s) or s <= 0.0:
        w = np.zeros(k, dtype=float)
        w[0] = 1.0
    else:
        w /= s
    return idx.astype(int, copy=False), w


def gradient_from_weights(h, indices, weights, dx):
    h = np.asarray(h, dtype=float)
    n = h.size
    grad = np.zeros(n, dtype=float)
    if indices.size == 0:
        return grad
    center = n - 1
    for ii in range(indices.size):
        idx = int(indices[ii])
        w = float(weights[ii])
        if w == 0.0:
            continue
        d = idx - center
        wd = w * float(dx)
        if d == 0:
            grad += wd * (1.0 - 2.0 * h)
        else:
            p = abs(d)
            m = n - p
            if d > 0:
                # numpy full index center + p is sum h[p:] * (1 - h[:m])
                grad[p:] += wd * (1.0 - h[:m])
                grad[:m] -= wd * h[p:]
            else:
                # numpy full index center - p is sum h[:m] * (1 - h[p:])
                grad[:m] += wd * (1.0 - h[p:])
                grad[p:] -= wd * h[:m]
    return grad

def quantize_exact(h, D=262144):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return h
    D = int(D)
    h = np.clip(h, 0.0, 1.0)
    scaled = h * float(D)
    v = np.rint(scaled).astype(np.int64)
    v = np.clip(v, 0, D)
    target = (n * D) // 2
    diff = int(target - int(v.sum()))
    frac = scaled - np.rint(scaled)
    if diff > 0:
        order = np.argsort(-frac, kind="mergesort")
        for idx in order:
            idx = int(idx)
            if diff <= 0:
                break
            if v[idx] < D:
                v[idx] += 1
                diff -= 1
        for idx in range(n):
            if diff <= 0:
                break
            if v[idx] < D:
                v[idx] += 1
                diff -= 1
    elif diff < 0:
        order = np.argsort(frac, kind="mergesort")
        for idx in order:
            idx = int(idx)
            if diff >= 0:
                break
            if v[idx] > 0:
                v[idx] -= 1
                diff += 1
        for idx in range(n):
            if diff >= 0:
                break
            if v[idx] > 0:
                v[idx] -= 1
                diff += 1
    out = v.astype(float) / float(D)
    return out

        cc = eval_score(ht)
        if math.isfinite(cc):
            cand_h.append(ht)
            cand_c.append(float(cc))

        for _ in range(6):
            hr = make_cos_candidate(n, rng, K=5, amp=0.38)
            cc = eval_score(hr)
            if math.isfinite(cc):
                cand_h.append(hr)
                cand_c.append(float(cc))

        if not cand_h:
            continue

        order = np.argsort(np.array(cand_c, dtype=float), kind="mergesort")
        top_count = 3 if limit > 80.0 else 2
        top_count = min(top_count, len(order))
        stage_left = limit
        per_limit = limit / float(max(1, top_count))
        for ti in range(top_count):
            if stage_left < 2.0:
                break
            idx = int(order[ti])
            use = min(per_limit, stage_left)
            t0 = time.perf_counter()
            h2, c2 = optimize_full(cand_h[idx], use, step=0.22, tau=0.004, topk=16, momentum=0.5, rng=rng)
            stage_left -= time.perf_counter() - t0
            if c2 < best_c:
                best_c = float(c2)
                best_h = h2

    for n, K, frac in ((192, 5, 0.07), (384, 5, 0.07)):
        remaining = deadline - time.perf_counter()
        if remaining <= 1.0:
            break
        limit = frac * usable
        if limit > remaining * 0.85:
            limit = remaining * 0.85
        if limit < 5.0:
            continue
        maxiter = max(4, min(15, int(6 + 9.0 * limit / 60.0)))
        h_de, c_de = de_global(n, K, int(seed) + n * 7 + 3, limit, maxiter=maxiter, popsize=10)
        if h_de is not None:
            if c_de < best_c:
                best_c = float(c_de)
                best_h = h_de.copy()
            elif c_de < best_c + 0.03:
                local_lim = min(0.03 * usable, (deadline - time.perf_counter()) * 0.5)
                if local_lim > 3.0:
                    h2, c2 = optimize_full(h_de, local_lim, step=0.20, tau=0.003, topk=14, momentum=0.5, rng=rng)
                    if c2 < best_c:
                        best_c = float(c2)
                        best_h = h2

    if best_h is not None:
        for target_n in (384, 512):
            remaining = deadline - time.perf_counter()
            if remaining <= 1.0:
                break
            if len(best_h) < target_n:
                limit = 0.035 * usable
                if limit > remaining * 0.45:
                    limit = remaining * 0.45
                if limit > 5.0:
                    hr = resample_full(best_h, target_n)
                    h2, c2 = optimize_full(hr, limit, step=0.18, tau=0.003, topk=14, momentum=0.5, rng=rng)
                    if c2 < best_c:
                        best_c = float(c2)
                        best_h = h2

    if best_h is not None:
        remaining = deadline - time.perf_counter()
        if remaining > 5.0:
            limit = min(0.05 * usable, remaining * 0.85)
            if limit > 3.0:
                h2, c2 = optimize_full(best_h, limit, step=0.12, tau=0.0025, topk=12, momentum=0.4, rng=rng)
                if c2 < best_c:
                    best_c = float(c2)
                    best_h = h2

    if best_h is None:
        best_h = make_triangle(96)
        best_c = eval_score(best_h)

    final_q = quantize_exact(best_h, 262144)
    c_q = eval_score(final_q)
    if math.isfinite(c_q):
        final_h = final_q
        final_c = float(c_q)
    else:
        final_h = enforce_exact_sum(best_h, 0.5 * float(best_h.size))
        final_c = eval_score(final_h)

    return final_h, float(final_c), int(final_h.size)
```
