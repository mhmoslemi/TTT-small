Actually, wait. Let me reconsider the gradient computation. I want to make sure it's correct.

f(h) = corr[k] = sum_{i valid} h[i]*(1-h[i+s])

∂f/∂h[m] = sum_{i valid} ∂(h[i]*(1-h[i+s]))/∂h[m]

For a fixed k (shift s = k-n+1):
∂(h[i]*(1-h[i+s]))/∂h[m] = δ(i,m)*(1-h[i+s]) + h[i]*∂(1-h[i+s])/∂h[m]
= δ(i,m)*(1-h[i+s]) - h[i]*δ(i+s,m)

Summing over valid i:
∂f/∂h[m] = (1-h[m+s]) * 1_{m valid and m+s valid} - sum_{i: i+s=m, i valid} h[i] * 1_{valid}
= (1-h[m+s]) * 1_{m+s valid} - h[m-s] * 1_{m-s valid}

where "m+s valid" means 0≤m+s≤n-1, and "m-s valid" means 0≤m-s≤n-1.

Note: "m valid" is always true since we're computing the gradient at m. And for the sum to contribute, we need i=m-s to be valid (in range [0,n-1]) AND (from the original sum) we need i valid and i+s valid. If i=m-s is valid and i+s=m is valid (which it is since m is in [0,n-1]), then it's included.

So: ∂f/∂h[m] = (1-h[m+s]) * 1_{0≤m+s≤n-1} - h[m-s] * 1_{0≤m-s≤n-1}

This matches what I had before. Good.

import numpy as np
import time


def c5_exact(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s_val = np.clip(h - mid, 0.0, 1.0).sum()
        if s_val > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def obj_grad_softmax(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    vmax = vals.max()
    exps = np.exp(T * (vals - vmax))
    wsum = exps.sum()
    weights = exps / wsum
    obj = (np.log(wsum) + T * vmax) / T

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - n + 1
        w = weights[k] * dx
        if w < 1e-12:
            continue
        m_min1 = max(0, -s)
        m_max1 = min(n - 1, n - 1 - s)
        if m_min1 <= m_max1:
            grad[m_min1:m_max1 + 1] += w * (1.0 - h[m_min1 + s:m_max1 + 1 + s])
        m_min2 = max(0, s)
        m_max2 = min(n - 1, n - 1 + s)
        if m_min2 <= m_max2:
            grad[m_min2:m_max2 + 1] -= w * h[m_min2 - s:m_max2 + 1 - s]

    return obj, grad


def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    if n_new < n_old:
        x_old = np.arange(n_old, dtype=np.float64)
        x_new = np.arange(n_new, dtype=np.float64)
        return np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    x_old = np.arange(n_old, dtype=np.float64)
    x_new = np.arange(n_new, dtype=np.float64)
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])


def pg_optimize(h0, dx, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = project(h0.copy())
    best_h = h.copy()
    best_c5 = c5_exact(h, dx)

    n = len(h)
    lr = 4.0 * dx
    no_improve = 0
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        T = 5.0 + 95.0 * min(1.0, iter_count / 400.0)

        obj, grad = obj_grad_softmax(h, dx, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        alpha = lr
        accepted = False
        for _ in range(20):
            h_new = project(h - alpha * grad)
            c5_new = c5_exact(h_new, dx)
            if c5_new <= obj + 0.5 * alpha * gnorm * 0.01:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                accepted = True
                break
            alpha *= 0.5

        if not accepted:
            no_improve += 1
            if no_improve > 30:
                h = project(h + 0.03 * rng.randn(n))
                c5_check = c5_exact(h, dx)
                if c5_check < best_c5:
                    best_c5 = c5_check
                    best_h = h.copy()
                no_improve = 0
                lr = 4.0 * dx
            elif no_improve > 150:
                break
        else:
            no_improve = 0

        lr = min(30.0 * dx, lr * 1.02)

    return best_h, best_c5


def binary_local_search(h0, dx, deadline, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = (h0 > 0.5).astype(np.float64)
    if h.sum() < n / 2:
        deficit = int(n / 2) - int(h.sum())
        zero_idx = np.where(h == 0)[0]
        rng.shuffle(zero_idx)
        h[zero_idx[:deficit]] = 1.0
    elif h.sum() > n / 2:
        excess = int(h.sum()) - int(n / 2)
        one_idx = np.where(h == 1)[0]
        rng.shuffle(one_idx)
        h[one_idx[:excess]] = 0.0

    best_h = h.copy()
    best_c5 = c5_exact(h, dx)
    no_improve = 0

    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        j1 = ones[rng.randint(len(ones))]
        j0 = zeros[rng.randint(len(zeros))]

        h[j1] = 0.0
        h[j0] = 1.0
        c5_new = c5_exact(h, dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            h[j1] = 1.0
            h[j0] = 0.0
            no_improve += 1
            if no_improve > 500:
                h = best_h.copy()
                no_improve = 0
                # Random restart on a few positions
                for _ in range(3):
                    oi = np.where(h == 1)[0]
                    zi = np.where(h == 0)[0]
                    h[oi[rng.randint(len(oi))]] = 0.0
                    h[zi[rng.randint(len(zi))]] = 1.0

    return best_h, best_c5


def random_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h


def structured_init(n, seed, variant=0):
    rng = np.random.RandomState(seed)
    x = np.arange(n, dtype=np.float64)
    if variant == 0:
        h = 0.5 + 0.4 * np.sin(2.0 * np.pi * x / n * 3.0)
    elif variant == 1:
        h = 0.5 + 0.45 * np.sin(2.0 * np.pi * x / n * 5.0)
    elif variant == 2:
        h = 0.5 + 0.4 * np.cos(2.0 * np.pi * x / n * 7.0)
    else:
        h = 0.5 + 0.3 * np.sin(2.0 * np.pi * x / n * 3.0) + 0.25 * np.sin(2.0 * np.pi * x / n * 8.0)
    h = project(h)
    return h
