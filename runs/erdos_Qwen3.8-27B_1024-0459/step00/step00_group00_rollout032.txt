    def update(h):
        nonlocal best_h, best_c5
        if h is None:
            return
        h = project_simplex(h, len(h) / 2.0)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Gather starting points at n=45
    n_base = 45
    starts = []

    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if len(h_parent) == n_base:
            starts.append(h_parent.copy())
        else:
            h_resized = upscale(h_parent, n_base)
            starts.append(h_resized)
    except Exception:
        pass

    for t in range(4):
        starts.append(make_random_binary(n_base, seed + 10 + t))
    for m in range(3):
        starts.append(make_smooth_start(n_base, m, seed + 40 + m))
    h_uniform = np.ones(n_base) * 0.5
    starts.append(h_uniform)

    # Phase 1: L-BFGS-B optimization at base resolution
    time_phase1 = budget_s * 0.45
    n_starts = len(starts)
    time_per_start = time_phase1 / (n_starts * 3)  # 3 alpha levels per start

    for si, sp in enumerate(starts):
        if time.time() - t_start > time_phase1:
            break
        for alpha in [30, 80, 200]:
            if time.time() - t_start > time_phase1:
                break
            remaining = time_phase1 - (time.time() - t_start)
            if remaining < 5:
                break
            t_limit = min(remaining * 0.9, time_per_start + 20)
            h_opt = try_optimize(sp, alpha, 200.0, 300, t_limit, t_start)
            if h_opt is not None:
                update(h_opt)
            if best_h is not None and len(best_h) == n_base:
                sp = best_h.copy()

    # Phase 2: Gradient refinement of best solution
    time_phase2 = budget_s * 0.25
    if best_h is not None and time.time() - t_start < budget_s * 0.75:
        t_p2_limit = budget_s * 0.75
        h_ref = gradient_step_refine(best_h, t_p2_limit, seed + 500, t_start)
        update(h_ref)

    # Phase 3: Try higher resolution
    time_phase3_end = budget_s * 0.92
    if best_h is not None:
        for n_new in [90, 180]:
            if time.time() - t_start > time_phase3_end:
                break
            h_up = upscale(best_h, n_new)
            update(h_up)
            if time.time() - t_start > time_phase3_end:
                break
            total_rem = time_phase3_end - (time.time() - t_start)
            if total_rem > 10:
                for alpha in [50, 150]:
                    if time.time() - t_start > time_phase3_end - 5:
                        break
                    rem = time_phase3_end - (time.time() - t_start)
                    if rem < 5:
                        break
                    h_opt = try_optimize(h_up, alpha, 200.0, 200, rem * 0.9, t_start)
                    if h_opt is not None:
                        update(h_opt)
                    if best_h is not None and len(best_h) == n_new:
                        h_up = best_h.copy()

                if time.time() - t_start < time_phase3_end - 10:
                    rem = time_phase3_end - (time.time() - t_start) - 5
                    if rem > 5 and best_h is not None and len(best_h) == n_new:
                        h_ref = gradient_step_refine(best_h, time_phase3_end - 5, seed + 900, t_start)
                        update(h_ref)

    # Final: ensure valid output
    if best_h is None:
        best_h = np.ones(n_base) * 0.5
    best_h = project_simplex(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
