            elif h[idx] > 1.0:
                h[idx] = 1.0

    res = target - float(np.sum(h))
    if abs(res) > 0.0:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)

    res = target - float(np.sum(h))
    if abs(res) > 0.0:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            idx = int(free[0])
            h[idx] += res
            h = np.clip(h, 0.0, 1.0)

    return h


def is_feasible(h, target, tol=1e-6):
    h = np.asarray(h, dtype=float)
    if h.size == 0:
        return False
    if np.any(h < -tol) or np.any(h > 1.0 + tol):
        return False
    if abs(float(np.sum(h)) - float(target)) > max(1e-8, tol * max(1, h.size)):
        return False
    return True


def vector_close(a, b):
    if a.size != b.size:
        return False
    return bool(np.allclose(a, b, rtol=0.0, atol=1e-7))


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def make_smooth_candidate(n, rng, mode):
    n = int(n)
    if n <= 1:
        return project_density(np.array([0.5], dtype=float), 0.5 * n)

    if mode == 1:
        u = rng.random(n)
        thresh = float(np.median(u))
        h = (u > thresh).astype(float)
    elif mode == 2:
        h = np.empty(n, dtype=float)
        levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
        pos = 0
        while pos < n:
            rem = n - pos
            length = int(rng.integers(1, rem + 1))
            val = float(rng.choice(levels))
            h[pos:pos + length] = val
            pos += length
    else:
        x = (np.arange(n) + 0.5) * (2.0 / float(n))
        h = np.full(n, 0.5, dtype=float)
        for m in range(1, 6):
            amp = float(rng.uniform(0.05, 0.35)) * (0.7 ** (m - 1))
            freq = int(max(1, np.rint(m * float(rng.uniform(0.5, 1.5)))))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)

    return project_density(h, 0.5 * n)


def set_context(n, alpha, target):
    global _NC, _ALPHA, _TARGET
    _NC = int(n)
    _ALPHA = float(alpha)
    _TARGET = float(target)


def smooth_objective(x):
    n = _NC
    alpha = _ALPHA
    if n <= 1:
        return 0.0

    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100

    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return 1e100

    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100

    return float((2.0 / float(n)) * (m + np.log(z) / alpha))


def smooth_jacobian(x):
    n = _NC
    alpha = _ALPHA
    if n <= 1:
        return np.zeros(max(1, n), dtype=float)

    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)

    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return np.zeros(n, dtype=float)

    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n, dtype=float)

    w = e / z
    idx = np.where(w > 1e-12)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))], dtype=int)

    g = np.zeros(n, dtype=float)
    center = n - 1

    for t in idx:
        t = int(t)
        p = float(w[t])
        if p <= 0.0:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
            else:
                g[:n - lag] += p
            g[:n - lag] -= p * h[lag:]
            g[lag:] -= p * h[:n - lag]

    return (2.0 / float(n)) * g


def sum_constraint(x):
    return float(np.sum(np.asarray(x, dtype=float)) - _TARGET)


def sum_constraint_jac(x):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)


def corr_gradient_indices(h, indices):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0 or len(indices) == 0:
        return g

    center = n - 1
    for t in indices:
        t = int(t)
        k = t - center
        if k == 0:
            g += 1.0 - 2.0 * h
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += 1.0
            else:
                g[:n - lag] += 1.0
            g[:n - lag] -= h[lag:]
            g[lag:] -= h[:n - lag]
    return g


def corr_gradient_average(h, indices):
    indices = list(indices)
    if len(indices) == 0:
        return np.zeros(len(h), dtype=float)
    return corr_gradient_indices(h, indices) / float(len(indices))


def polish_gradient(h, deadline, max_iter=300, active_k=4):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c

    k = min(int(active_k), 2 * n - 1)
    stall = 0

    for _ in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break

        corr = np.correlate(best, 1.0 - best, mode="full")
        if not np.all(np.isfinite(corr)):
            break

        idxs = np.argsort(corr)[-k:]
        g = corr_gradient_average(best, idxs)
        d = -g
        d -= float(np.mean(d))
        norm = float(np.linalg.norm(d))
        if norm < 1e-13:
            break
        d /= norm

        improved = False
        for step in (0.01, 0.03, 0.1, 0.3, 0.7):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-11:
                best = cand
                best_c = c
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break

    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def local_pair_search(h, deadline, max_iter=500, active_k=4):
    h = np.asarray(h, dtype=float)
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c

    k = min(int(active_k), 2 * n - 1)
    stall = 0

    for _ in range(int(max_iter)):
        if time.perf_counter() >= deadline:
            break

        corr = np.correlate(best, 1.0 - best, mode="full")
        if not np.all(np.isfinite(corr)):
            break

        idxs = np.argsort(corr)[-k:]
        lags = np.abs(idxs - (n - 1)).astype(int)
        g = corr_gradient_average(best, idxs)

        if g.size == 0:
            break
        if (float(np.max(g)) - float(np.min(g))) < 1e-11:
            break

        neg = np.argsort(g)[:3]
        pos = np.argsort(-g)[:3]
        found = False

        for ii in neg:
            ii = int(ii)
            for jj in pos:
                jj = int(jj)
                if ii == jj:
                    continue

                cap = min(1.0 - best[ii], best[jj])
                if cap <= 1e-12:
                    continue

                a = float(g[ii] - g[jj])
                if a >= -1e-11:
                    continue

                b = 0.0
                for lag in lags:
                    lag = int(lag)
                    if lag == 0:
                        b += -2.0
                    elif abs(ii - jj) == lag:
                        b += 1.0
                b /= float(k)

                deltas = []
                if b > 1e-12:
                    dv = -a / (2.0 * b)
                    if dv > 0.0:
                        if dv > cap:
                            dv = cap
                        deltas.append(float(dv))
                deltas.append(0.5 * cap)
                deltas.append(0.1 * cap)
                deltas.append(cap)

                uniq_d = []
                for d in deltas:
                    if d <= 1e-12:
                        continue
                    if d > cap:
                        d = cap
                    if d >= cap - 1e-15:
                        d = cap - 1e-15
                    if d <= 1e-12:
                        continue
                    close = False
                    for ud in uniq_d:
                        if abs(d - ud) < 1e-9:
                            close = True
                            break
                    if not close:
                        uniq_d.append(float(d))

                for d in uniq_d:
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d

                    if cand[ii] < 0.0:
                        cand[ii] = 0.0
                    elif cand[ii] > 1.0:
                        cand[ii] = 1.0
                    if cand[jj] < 0.0:
                        cand[jj] = 0.0
                    elif cand[jj] > 1.0:
                        cand[jj] = 1.0

                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-11:
                        best = cand
                        best_c = c
                        found = True

        if not found:
            d = -g
            d -= float(np.mean(d))
            norm = float(np.linalg.norm(d))
            if norm > 1e-12:
                d /= norm
                for step in (0.02, 0.07, 0.2):
                    cand = project_density(best + step * d, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-11:
                        best = cand
                        best_c = c
                        found = True
                        break

        if found:
            stall = 0
        else:
            stall += 1
            if stall >= 15:
                break

    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    return best, best_c


def optimize_one(h0, n, sub_deadline):
    n = int(n)
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    now = time.perf_counter()
    span = max(1.0, sub_deadline - now)
    polish_reserve = min(35.0, max(6.0, 0.20 * span))
    if polish_reserve > 0.6 * span:
        polish_reserve = 0.10 * span
    solve_deadline = sub_deadline - polish_reserve

    if n <= 90:
        alphas = (8.0, 25.0, 75.0, 175.0)
        maxiter = 150
    elif n <= 200:
        alphas = (10.0, 30.0, 90.0)
        maxiter = 100
    elif n <= 450:
        alphas = (10.0, 30.0)
        maxiter = 70
    else:
        alphas = (10.0, 30.0)
        maxiter = 45

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac},)

    for alpha in alphas:
        if time.perf_counter() >= solve_deadline:
            break

        set_context(n, alpha, target)
        try:
            res = minimize(
                smooth_objective,
                h,
                method="SLSQP",
                jac=smooth_jacobian,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-9, "disp": False},
            )
            xg = getattr(res, "x", None)
            if xg is not None:
                xg = np.asarray(xg, dtype=float)
                if np.all(np.isfinite(xg)):
                    cand = project_density(xg, target)
                    c = evaluate_erdos(cand)
                    if c < best_c:
                        best_h = cand
                        best_c = c
                    if c <= best_c + 0.02:
                        h = cand
                    else:
                        h = best_h
        except Exception:
            h = best_h

    if time.perf_counter() < sub_deadline - 1.0:
        if n <= 200 and (sub_deadline - time.perf_counter()) > 8.0:
            h2, c2 = local_pair_search(best_h, sub_deadline, max_iter=300, active_k=4)
            if c2 < best_c:
                best_h = h2
                best_c = c2

        if sub_deadline - time.perf_counter() > 3.0:
            h3, c3 = polish_gradient(best_h, sub_deadline, max_iter=200, active_k=4)
            if c3 < best_c:
                best_h = h3
                best_c = c3

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 12.0)

    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        nparent = int(parent.size)
        targetparent = 0.5 * nparent
        if is_feasible(parent, targetparent, 1e-6):
            cp = evaluate_erdos(parent)
            if cp < best_c:
                best_h = parent.copy()
                best_c = cp
                best_n = nparent
        pproj = project_density(parent, targetparent)
        cp = evaluate_erdos(pproj)
        if cp < best_c:
            best_h = pproj
            best_c = cp
            best_n = nparent

    if best_h is None:
        n0 = 90
        best_h = make_smooth_candidate(n0, rng, 0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    base_n = best_n if best_n > 1 else (int(parent.size) if parent is not None else 90)

    nset = set()
    for mult in (1, 2, 4, 8, 16):
        n = int(base_n * mult)
        if 2 <= n <= 720:
            nset.add(n)
    for n in (90, 180, 360):
        if base_n <= n <= 720:
            nset.add(n)
    nlist = sorted(nset)

    for ni in nlist:
        if deadline - time.perf_counter() < 25.0:
            break

        starts = []
        if best_h is not None:
            if int(best_h.size) == int(ni):
                starts.append(np.asarray(best_h, dtype=float).copy())
            else:
                starts.append(upsample_h(best_h, ni))

        if parent is not None:
            starts.append(upsample_h(parent, ni))

        if ni <= 90:
            nrand = 2
        elif ni <= 300:
            nrand = 1
        else:
            nrand = 1

        for mm in range(nrand):
            starts.append(make_smooth_candidate(ni, rng, mm % 3))

        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if vector_close(s, u):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        if not uniq:
            continue

        for idx_start, s in enumerate(uniq):
            rem = deadline - time.perf_counter()
            if rem < 12.0:
                break

            left_starts = max(1, len(uniq) - idx_start)
            if ni <= 90:
                cap = 120.0
            elif ni <= 240:
                cap = 80.0
            else:
                cap = 50.0
            per = min(rem / float(left_starts), cap)
            per = max(10.0, per)
            sub_deadline = time.perf_counter() + min(per, rem)

            h_out, c_out = optimize_one(s, ni, sub_deadline)
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * ni)
                c_out = evaluate_erdos(h_out)
                if c_out < best_c:
                    best_h = h_out
                    best_c = c_out
                    best_n = int(ni)

    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        nbest = int(best_h.size)
        if nbest <= 200:
            h2, c2 = local_pair_search(best_h, deadline, max_iter=1000, active_k=4)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = nbest

        if deadline - time.perf_counter() > 8.0:
            h3, c3 = polish_gradient(best_h, deadline, max_iter=500, active_k=4)
            if c3 < best_c:
                best_h = h3
                best_c = c3
                best_n = int(best_h.size)

    if best_h is None:
        fallback = project_density(np.full(45, 0.5, dtype=float), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
