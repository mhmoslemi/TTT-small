    n = len(h)

    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_box_sum(h, target_sum):
    n = len(h)
    # Project onto {h: 0 <= h_i <= 1, sum(h) = target_sum}
    # Using the beta parameter (uniform shift)
    lo, hi = -2.0, 2.0

import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def all_lags(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    h = np.asarray(h, dtype=float)
    
    def g(beta):
        return np.sum(np.minimum(1.0, np.maximum(0.0, h - beta))) - target_sum
    
    g_lo = g(lo)
    g_hi = g(hi)
    
    if g_lo <= 0:
        return np.clip(h - lo, 0, 1)
    if g_hi >= 0:
        return np.clip(h - hi, 0, 1)
    
    try:
        beta = optimize.brentq(g, lo, hi, xtol=1e-14, maxiter=200)
    except:
        beta = 0.0
    return np.clip(h - beta, 0, 1)

def subgradient_lag(h, k_star, n):
    sg = np.zeros(n)
    if k_star >= 0:
        k = k_star
        if k < n:
            sg[k:n] += (1.0 - h[0:n-k])
            sg[0:n-k] -= h[k:n]
    else:
        j = -k_star
        if j < n:
            sg[0:n-j] += (1.0 - h[j:n])
            sg[j:n] -= h[0:n-j]
    return sg

def smooth_objective(h, alpha, n):
    c = all_lags(h)
    c_max = np.max(c)
    shifted = c - c_max
    return (c_max + np.log(np.sum(np.exp(alpha * shifted))) / alpha) / 1.0

def smooth_grad(h, alpha, n):
    c = all_lags(h)
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    weights = exp_c / np.sum(exp_c)
    # Gradient of smooth objective = weighted sum of subgradients
    sg = np.zeros(n)
    for i, w in enumerate(weights):
        if w > 1e-10:
            k = i - (n - 1)
            sg += w * subgradient_lag(h, k, n)
    return sg, (c_max + np.log(np.sum(exp_c)) / alpha)

        c = all_lags(x)

    def obj_grad(x):
        alpha = 50.0
        c = all_lags(x)
        c_max = np.max(c)
        exp_c = np.exp(alpha * (c - c_max))
        val = c_max + np.log(np.sum(exp_c)) / alpha
        weights = exp_c / np.sum(exp_c)
        sg = np.zeros(n)
        for i, w in enumerate(weights):
            if w > 1e-8:
                k = i - (n - 1)
                sg += w * subgradient_lag(x, k, n)
        return val, sg
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n

def projected_subgradient(h, n, target_sum, t_max, rng, n_iter=2000):
