```python
def compute_c5_gradient(h, n):
    """Compute value and gradient of C5 = max_k (dx * correlate(h, 1-h, mode='full')[k])"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    
    # Find active indices (within tolerance of max)
    active = np.where(np.abs(corr - corr.max()) < 1e-6)[0]
    
    grad = np.zeros(n)
    for j in active:
        # c[j] = sum_m h[m] * (1-h[m+j-(n-1)])
        # dc[j]/dh[i] = (1-h[i+j-(n-1)]) [if in range] - h[i-j+(n-1)] [if in range]
        for i in range(n):
            # First term: m=i, need i+j-(n-1) in [0,n-1]
            idx1 = i + j - (n - 1)
            if 0 <= idx1 < n:
                grad[i] += (1 - h[idx1])
            # Second term: m+j-(n-1)=i, so m=i-j+(n-1), need m in [0,n-1]
            idx2 = i - j + (n - 1)
            if 0 <= idx2 < n:
                grad[i] += -h[idx2]
    
    return c5, grad * dx
```

```python
import numpy as np
import time

def project_to_feasible(h, n):
    """Project h onto {h: 0<=h[i]<=1, sum(h)=n/2}"""
    S = n / 2.0
    # Find t such that sum(clip(h - t, 0, 1)) = S
    # sum(clip(h-t, 0, 1)) = sum(max(0, min(1, h_i - t))) for i
    # This is linear and decreasing in t between min(h) and max(h)
    
    lo = h.min() - 1
    hi = h.max() + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if np.abs(s - S) < 1e-12:
            break
        if s > S:
            lo = mid
        else:
            hi = mid
    
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune to get exact sum
    diff = S - h_proj.sum()
    if np.abs(diff) > 1e-10:
        # Distribute among non-boundary elements
        free = (h_proj > 1e-10) & (h_proj < 1 - 1e-10)
        if free.sum() > 0:
            h_proj[free] += diff / free.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def compute_c5_and_grad(h, n):
    """Compute C5 value and gradient"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = corr.max() * dx
    
    max_val = corr.max()
    # Subgradient: average over active components
    tol = 1e-8
    active = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    for j in active:
        # dc[j]/dh[i] = (1-h[i+j-(n-1)]) - h[i-j+(n-1)] (with bounds)
        idx1 = np.arange(n) + j - (n - 1)
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += (1 - h[idx1[mask1]])
        
        idx2 = np.arange(n) - j + (n - 1)
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] += (-h[idx2[mask2]])
    
    grad /= len(active)
    return c5, grad * dx, corr.max() * dx, corr

def projected_gradient_descent(h0, n, lr_init=0.01, max_iter=5000, time_limit=None, start_time=None):
    """Run projected gradient descent"""
    h = h0.copy()
    h = project_to_feasible(h, n)
    best_c5 = compute_c5_and_grad(h, n)[0]
    best_h = h.copy()
    
    lr = lr_init
    n = len(h)
    
    for it in range(max_iter):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        c5, grad, _, corr = compute_c5_and_grad(h, n)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        # Check if we made progress
        c5_new = compute_c5_and_grad(h_new, n)[0]
        if c5_new > c5 + 1e-10:
            # Line search: reduce lr
            lr *= 0.5
            if lr < 1e-8:
                break
        else:
            h = h_new
            # Maybe increase lr slightly
            # lr = min(lr * 1.1, 0.1)
        
        # Adaptive: if gradient norm is very small, stop
        if np.max(np.abs(grad)) < 1e-10:
            break
    
    # Final check
    c5_final = compute_c5_and_grad(best_h, n)[0]
    return best_h, c5_final

def make_pattern(n, pattern_type='alternating'):
```

    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - h_p.sum()
    if abs(diff) > 1e-12:
        free = (h_p > 1e-9) & (h_p < 1.0 - 1e-9)
        nf = free.sum()
        if nf > 0:
            h_p[free] += diff / nf
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def compute_correlation(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return np.max(compute_correlation(h)) * dx

def compute_subgradient(h, n, corr, tol=1e-4):
    """Subgradient of max(corr)*dx using active components."""
    dx = 2.0 / n
    max_val = corr.max()
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    indices = np.arange(n)
    for j in active:
        idx1 = indices + j - (n - 1)
        m1 = (idx1 >= 0) & (idx1 < n)
        grad[m1] += (1.0 - h[idx1[m1]])
        idx2 = indices - j + (n - 1)
        m2 = (idx2 >= 0) & (idx2 < n)
        grad[m2] += (-h[idx2[m2]])
    grad /= len(active)
    return grad * dx

def pgd_optimize(h0, n, lr=0.01, max_iter=5000, time_limit=None, start_time=None):
    """Projected gradient descent with momentum and backtracking."""
    h = h0.copy().astype(float)
    h = project_to_feasible(h, n)
    c5 = compute_c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    momentum = np.zeros(n)
    mu = 0.85
    stall = 0
    
    for it in range(max_iter):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        corr = compute_correlation(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        grad = compute_subgradient(h, n, corr, tol=1e-4)
        momentum = mu * momentum - lr * grad
        h_trial = project_to_feasible(h + momentum, n)
        c5_trial = compute_c5(h_trial, n)
        if c5_trial <= c5 + 1e-10:
            h = h_trial
        else:
            h_trial = project_to_feasible(h - lr * grad, n)
            c5_trial = compute_c5(h_trial, n)
            if c5_trial <= c5 + 1e-10:
                h = h_trial
                momentum *= 0.5
            else:
                lr *= 0.5
                if lr < 1e-10:
                    break
        if it % 150 == 149:
            lr = min(lr * 1.5, 0.08)
        if np.max(np.abs(grad)) < 1e-11:
            break
        if stall > 2000:
            break
    
    final_c5 = compute_c5(best_h, n)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5

def make_structured(n, rng, style):
    x = np.linspace(0, 0, n)  # placeholder
    if style == 0:
        h = np.zeros(n)
        h[::2] = 1.0
        if n % 2 == 1:
            h[-1] = 0.0
    elif style == 1:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    elif style == 2:
        xx = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.35 * np.sin(np.pi * xx)
    elif style == 3:
        xx = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.2 * np.sin(np.pi * xx) + 0.15 * np.cos(2 * np.pi * xx)
    elif style == 4:
        h = np.zeros(n)
        idx = np.sort(rng.choice(n, size=n // 2, replace=False))
        h[idx] = 1.0
    elif style == 5:
        xx = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.3 * np.sin(np.pi * xx * 2)
    else:
        h = np.full(n, 0.5) + rng.randn(n) * 0.15
    return project_to_feasible(h, n)

def upscale_h(h, factor, rng):
    n = len(h)
    n_new = n * factor
    base = (np.arange(n_new).astype(float) * n / n_new).astype(int)
    base = np.clip(base, 0, n - 1)
    h_new = h[base].astype(float)
    h_new += rng.randn(n_new) * 0.03
    return project_to_feasible(h_new, n_new)

def local_polish_scipy(h, n, time_limit, start_time):
    """Use scipy L-BFGS-B with smoothed objective for local polish."""
    from scipy.optimize import minimize
    
    dx = 2.0 / n
    
    def smooth_obj(hv, tau=0.05):
        if time.time() - start_time > time_limit:
            return 999.0
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        shifted = corr / tau
        shifted -= shifted.max()
        log_sum_exp = tau * np.log(np.exp(shifted).sum()) + corr.max()
        return log_sum_exp * dx
    
    def smooth_grad(hv, tau=0.05):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        weights = np.exp(corr / tau - corr.max() / tau)
        weights /= weights.sum()
        grad = np.zeros(n)
        indices = np.arange(n)
        for j in range(len(corr)):
            if weights[j] < 1e-8:
                continue
            idx1 = indices + j - (n - 1)
            m1 = (idx1 >= 0) & (idx1 < n)
            grad[m1] += weights[j] * (1.0 - hv[idx1[m1]])
            idx2 = indices - j + (n - 1)
            m2 = (idx2 >= 0) & (idx2 < n)
            grad[m2] += -weights[j] * hv[idx2[m2]]
        return grad * dx
    
    res = minimize(smooth_obj, h, jac=smooth_grad, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': 200, 'time_limit': max(time_limit - (time.time() - start_time), 10)})
    if res.success or res.fun < compute_c5(h, n):
        h_res = project_to_feasible(res.x, n)
        return h_res
    return h
