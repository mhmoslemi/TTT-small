from scipy.optimize import minimize
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(vals, beta):
    v = np.asarray(vals, dtype=np.float64)
    vmax = np.max(v)
    if vmax < 1e-8:
        return max(0.0, vmax)
    shifted = beta * (v - vmax)
    shifted = np.clip(shifted, -500.0, 500.0)
    return vmax + np.log(np.sum(np.exp(shifted))) / beta

        s = np.sum(h) - n / 2.0

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Try multiple resolutions with multiple initializations
    all_ns = set()
    for n in range(max(30, n_parent - 15), n_parent + 45):
        all_ns.add(n)
    for extra in [32, 40, 44, 46, 48, 50, 52, 54, 56, 60, 64, 70, 75, 80, 90, 100, 110, 120, 128, 150, 200]:
        all_ns.add(extra)
    all_ns = sorted(x for x in all_ns if 30 <= x <= 250)

    n_res = len(all_ns)
    remaining = t_end - time.time() - 80
    time_per_res = max(4.0, remaining / max(1, n_res))

    for n in all_ns:
        if time.time() > t_end - 80:
            break
        t_n_end = min(t_end - 60, time.time() + time_per_res)

        starts = []
        starts.append(interp_h(parent_h, n))
        starts.append(random_h(n, rng))
        starts.append(near_binary_h(n, rng))
        starts.append(structured_h(n, rng))

        for h_init in starts:
            if time.time() > t_n_end:
                break
            # Quick subgradient pass
            t_sub_end = min(t_n_end - 2, time.time() + 5)
            h_sg, c5_sg = subgradient_optimize(h_init, n, t_sub_end, rng)
            # L-BFGS-B refinement
            h_opt, c5_opt = optimize_lbfgsb(h_sg, n, t_n_end)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n

    # Phase 3: Polishing - multiple perturbation + optimization from best
    if best_h is not None and time.time() < t_end - 20:
        for attempt in range(20):
            if time.time() > t_end - 10:
                break
            h_pert = perturb_h(best_h, best_n, rng, 0.15 - 0.01 * attempt)
            h_opt, c5_opt = optimize_lbfgsb(h_pert, best_n, min(t_end - 10, time.time() + 30))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 4: Try a few more resolutions near the best
    if best_h is not None and time.time() < t_end - 30:
        for dn in [-2, -1, 1, 2, 3, -3]:
            n_try = best_n + dn
            if n_try < 20 or n_try > 300:
                continue
            if time.time() > t_end - 15:
                break
            h_init = interp_h(best_h, n_try)
            h_opt, c5_opt = optimize_lbfgsb(h_init, n_try, min(t_end - 10, time.time() + 25))
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Final: one more subgradient polish
    if best_h is not None and time.time() < t_end - 5:
        h_sg, c5_sg = subgradient_optimize(best_h, best_n, t_end - 2, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Ensure valid output
    if best_h is not None:
        n_final = len(best_h)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = fix_sum(best_h, n_final)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
