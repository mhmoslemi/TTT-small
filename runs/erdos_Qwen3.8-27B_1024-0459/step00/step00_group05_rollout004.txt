def upsample(h, new_n):
    old_n=h.size
    if new_n == old_n: return h.copy()
    if new_n % old_n == 0:
       ratio = new_n // old_n
       new = np.repeat(h, ratio)
    else:
       # linear index interpolation
       old_idx = np.linspace(0, old_n-1, new_n)
       left = np.floor(old_idx).astype(int)
       right = np.minimum(left+1, old_n-1)
       frac = old_idx - left
       new = h[left]*(1-frac)+h[right]*frac
    return ensure_feasible(new)

    if n==0: return np.array([], dtype=float)
    g = 1.0 - h

import time
import numpy as np

try:
    import scipy.optimize as _optimize
except Exception:
    _optimize = None


_SLSQP_PARAM = {"beta": 40.0}
_USE_FD_GRAD = False
_GRAD_TESTED = False


def time_left(deadline):
    return max(0.0, float(deadline) - time.time())


def correlate_full_h(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return np.array([], dtype=float)

    g = 1.0 - h
    m = 2 * n - 1
    if n <= 256:
        return np.correlate(h, g, mode="full")

    L = 1
    while L < m:
        L <<= 1

    a = np.fft.rfft(h, L)
    b = np.fft.rfft(g[::-1], L)
    out = np.fft.irfft(a * b, L, n=L)
    out = np.asarray(out[:m], dtype=float)
    bad = out < 0.0
    if np.any(bad):
        out[bad] = 0.0
    return out


def eval_c5(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return 0.0
    c = correlate_full_h(h) * (2.0 / float(n))
    if c.size == 0:
        return 0.0
    if not np.all(np.isfinite(c)):
        return float("inf")
    return float(np.max(c))


def eval_c5_direct(h):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return 0.0
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
    if c.size == 0:
        return 0.0
    if not np.all(np.isfinite(c)):
        return float("inf")
    return float(np.max(c))


def lse_value_from_c(c, beta):
    c = np.asarray(c, dtype=float).ravel()
    if c.size == 0:
        return 0.0
    cmax = float(np.max(c))
    if not np.isfinite(cmax):
        return float("inf")
    if beta <= 1e-8:
        return cmax
    with np.errstate(under="ignore", over="ignore", invalid="ignore"):
        e = np.exp(beta * (c - cmax))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return cmax
    return cmax + float(np.log(s)) / float(beta)


def lse_value_from_h(h, beta):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return 0.0, np.array([], dtype=float)
    c = correlate_full_h(h) * (2.0 / float(n))
    return lse_value_from_c(c, beta), c


def weighted_gradient(h, w):
    h = np.asarray(h, dtype=float).ravel()
    w = np.asarray(w, dtype=float).ravel()
    n = int(h.size)
    g = np.zeros(n, dtype=float)
    if n == 0 or w.size == 0:
        return g

    dx = 2.0 / float(n)
    center = n - 1
    nz = np.nonzero(w > 1e-15)[0]

    for idx in nz:
        wd = float(w[idx])
        if wd <= 0.0:
            continue
        d = int(idx) - center
        if d == 0:
            g += wd * dx * (1.0 - 2.0 * h)
        elif d > 0:
            r = int(d)
            if r >= n:
                continue
            left = h[:-r]
            right = h[r:]
            dxw = wd * dx
            g[:-r] -= dxw * right
            g[r:] += dxw * (1.0 - left)
        else:
            r = int(-d)
            if r >= n:
                continue
            left = h[:-r]
            right = h[r:]
            dxw = wd * dx
            g[:-r] += dxw * (1.0 - right)
            g[r:] -= dxw * left

    return g


def lse_obj_grad(h, beta):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return np.zeros(0, dtype=float), 0.0, np.array([], dtype=float)

    if _USE_FD_GRAD and n <= 120:
        g, S, c = lse_fd_gradient(h, beta, 1e-5)
        return g, S, c

    dx = 2.0 / float(n)
    corr = correlate_full_h(h)
    c = corr * dx
    cmax = float(np.max(c)) if c.size else 0.0

    w = np.zeros(c.size, dtype=float)
    if c.size:
        if beta <= 1e-8:
            w[:] = 1.0 / float(c.size)
        else:
            with np.errstate(under="ignore", over="ignore", invalid="ignore"):
                e = np.exp(beta * (c - cmax))
            s = float(np.sum(e))
            if s > 0.0 and np.isfinite(s):
                w = e / s
            else:
                w[int(np.argmax(c))] = 1.0

    g = weighted_gradient(h, w)
    S = lse_value_from_c(c, beta)
    return g, S, c


def lse_fd_gradient(h, beta, eps=1e-5):
    h = np.asarray(h, dtype=float).ravel().copy()
    n = int(h.size)
    base, c = lse_value_from_h(h, beta)
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g, base, c

    for i in range(n):
        hp = h.copy()
        hm = h.copy()

        if h[i] + eps <= 1.0:
            hp[i] = min(1.0, h[i] + eps)
        else:
            hp[i] = max(0.0, h[i] - eps)

        if h[i] - eps >= 0.0:
            hm[i] = max(0.0, h[i] - eps)
        else:
            hm[i] = min(1.0, h[i] + eps)

        vp, _ = lse_value_from_h(hp, beta)
        vm, _ = lse_value_from_h(hm, beta)
        denom = float(hp[i] - hm[i])
        if abs(denom) < 1e-14:
            denom = 2.0 * eps
        val = (vp - vm) / denom
        if np.isfinite(val):
            g[i] = val
        else:
            g[i] = 0.0

    return g, base, c


def test_gradient():
    global _USE_FD_GRAD, _GRAD_TESTED
    if _GRAD_TESTED:
        return
    _GRAD_TESTED = True
    try:
        n = 6
        rng = np.random.default_rng(123)
        h = project_box_sum(rng.random(n), 0.5 * float(n))
        beta = 12.0
        g, _, _ = lse_obj_grad(h, beta)
        gfd, _, _ = lse_fd_gradient(h, beta, 1e-5)
        ng = float(np.linalg.norm(gfd))
        if ng > 1e-10:
            err = float(np.linalg.norm(g - gfd)) / ng
        else:
            err = 1.0
        if err > 5e-3:
            _USE_FD_GRAD = True
    except Exception:
        _USE_FD_GRAD = False


def adjust_sum(h, target):
    h = np.array(h, dtype=float, copy=True)
    if h.size == 0:
        return h
    target = float(target)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(4):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break

        if abs(diff) > 1e-7:
            if diff > 0.0:
                order = np.argsort(-(1.0 - h))
            else:
                order = np.argsort(h)
            rem = diff
            for idx in order:
                idx = int(idx)
                if rem > 0.0:
                    add = min(rem, 1.0 - h[idx])
                    if add > 0.0:
                        h[idx] += add
                        rem -= add
                        if abs(rem) <= 1e-14:
                            break
                else:
                    sub = min(-rem, h[idx])
                    if sub > 0.0:
                        h[idx] -= sub
                        rem += sub
                        if abs(rem) <= 1e-14:
                            break
        else:
            interior = (h > 1e-12) & (h < 1.0 - 1e-12)
            if interior.any():
                add = diff / float(np.count_nonzero(interior))
                vals = h[interior] + add
                if np.all(vals >= -1e-12) and np.all(vals <= 1.0 + 1e-12):
                    h[interior] = np.clip(vals, 0.0, 1.0)
                    continue
                if diff > 0.0:
                    order = np.argsort(-(1.0 - h))
                else:
                    order = np.argsort(h)
                rem = diff
                for idx in order:
                    idx = int(idx)
                    if rem > 0.0:
                        add = min(rem, 1.0 - h[idx])
                        if add > 0.0:
                            h[idx] += add
                            rem -= add
                            if abs(rem) <= 1e-14:
                                break
                    else:
                        sub = min(-rem, h[idx])
                        if sub > 0.0:
                            h[idx] -= sub
                            rem += sub
                            if abs(rem) <= 1e-14:
                                break
            else:
                break

    return h


def project_box_sum(x, target):
    x = np.asarray(x, dtype=float).ravel().copy()
    n = int(x.size)
    target = float(target)
    if n == 0:
        return x
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    x = np.nan_to_num(x, nan=0.0, posinf=1.0, neginf=0.0)
    x = np.clip(x, -10.0, 10.0)

    if np.all(x >= -1e-12) and np.all(x <= 1.0 + 1e-12):
        x = np.clip(x, 0.0, 1.0)
        if abs(float(np.sum(x)) - target) < 1e-9:
            return adjust_sum(x, target)

    xmin = float(np.min(x))
    xmax = float(np.max(x))
    lo = xmin - 1.0
    hi = xmax

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    val = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    return adjust_sum(val, target)


def ensure_feasible(h):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1:
        h = h.ravel()
    n = int(h.size)
    if n == 0:
        return h.copy()

    target = 0.5 * float(n)
    h = h.copy()
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, -10.0, 10.0)

    s = float(np.sum(h))
    if np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12) and abs(s - target) < 1e-8:
        return adjust_sum(np.clip(h, 0.0, 1.0), target)

    return project_box_sum(h, target)


def make_dyadic_feasible(h, q=30):
    h = np.asarray(h, dtype=float).ravel()
    n = int(h.size)
    if n == 0:
        return h.copy()

    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    h = np.clip(h, 0.0, 1.0)

    scale_int = 1 << int(q)
    scale_float = float(scale_int)
    target = 0.5 * float(n)
    target_units = int(round(target * scale_float))

    units = np.rint(h * scale_float).astype(np.int64)
    units = np.clip(units, 0, scale_int)
    cur_units = int(np.sum(units))
    diff = target_units - cur_units

    if diff > 0:
        order = np.argsort(-(scale_int - units))
    elif diff < 0:
        order = np.argsort(units)
    else:
        order = np.array([], dtype=int)

    rem = diff
    for idx in order:
        idx = int(idx)
        if rem > 0:
            add = min(rem, int(scale_int - units[idx]))
            if add > 0:
                units[idx] += add
                rem -= add
                if rem == 0:
                    break
        else:
            sub = min(-rem, int(units[idx]))
            if sub > 0:
                units[idx] -= sub
                rem += sub
                if rem == 0:
                    break

    out = units.astype(float) / scale_float
    return np.clip(out, 0.0, 1.0)


def upsample(h, new_n):
    h = np.asarray(h, dtype=float).ravel().copy()
    old_n = int(h.size)
    new_n = int(new_n)
    if old_n == 0:
        return np.array([], dtype=float)
    if new_n <= 0:
        return h.copy()
    if new_n == old_n:
        return ensure_feasible(h)

    if new_n % old_n == 0:
        ratio = new_n // old_n
        y = np.repeat(h, ratio)
    else:
        pos = np.linspace(0.0, float(old_n - 1), new_n)
        left = np.floor(pos).astype(int)
        right = np.minimum(left + 1, old_n - 1)
        frac = pos - left
        y = h[left] * (1.0 - frac) + h[right] * frac

    return ensure_feasible(y)


def smooth_pg(h, beta, deadline, rng=None, step_init=None, max_iterations=100000):
    h = ensure_feasible(h).copy()
    n = int(h.size)
    target = 0.5 * float(n)
    dx = 2.0 / float(n)

    c = correlate_full_h(h) * dx
    c5 = float(np.max(c)) if c.size else 0.0
    S = lse_value_from_c(c, beta)

    best_h = h.copy()
    best_c = c5

    if step_init is None:
        step = 0.01
    else:
        step = float(step_init)

    fail = 0
    factors = (0.25, 0.5, 1.0, 2.0, 4.0)
    it = 0

    while it < max_iterations:
        if time_left(deadline) < 0.25:
            break
        it += 1

        g, S, c = lse_obj_grad(h, beta)
        if not np.all(np.isfinite(g)) or not np.isfinite(S):
            break

        c5 = float(np.max(c)) if c.size else 0.0
        if c5 < best_c - 1e-12:
            best_c = c5
            best_h = h.copy()

        gn = float(np.linalg.norm(g))
        if gn < 1e-11:
            break

        if step is None or step <= 1e-12:
            step = min(0.05, max(1e-4, 0.1 / (gn + 1e-9)))

        use_c_primary = beta >= 50.0
        best_score = None
        best_tr = None
        best_S = None
        best_C = None
        best_factor = None

        for f in factors:
            st = step * f
            if st <= 0.0:
                continue
            y = h - st * g
            z = project_box_sum(y, target)
            if np.max(np.abs(z - h)) < 1e-15:
                continue

            c_tmp = correlate_full_h(z) * dx
            ct = float(np.max(c_tmp)) if c_tmp.size else 0.0
            St = lse_value_from_c(c_tmp, beta)
            if not (np.isfinite(ct) and np.isfinite(St)):
                continue

            if use_c_primary:
                score = (ct, St)
            else:
                score = (St, ct)

            if best_score is None or score < best_score:
                best_score = score
                best_tr = z
                best_S = St
                best_C = ct
                best_factor = f

        accepted = False
        if best_tr is not None and (best_S < S - 1e-11 or best_C < c5 - 1e-11):
            h = best_tr
            S = best_S
            c5 = best_C
            if c5 < best_c - 1e-12:
                best_c = c5
                best_h = h.copy()

            if best_factor is not None:
                if best_factor > 1.5:
                    step = min(0.5, step * 1.4)
                elif best_factor < 0.7:
                    step = max(1e-7, step * 0.7)
            fail = 0
            accepted = True

        if not accepted:
            step = max(1e-8, step * 0.5)
            fail += 1
            if step < 1e-8 or fail > 30:
                kicked = False
                if rng is not None:
                    for _ in range(20):
                        sigma = max(step, 0.005)
                        y = h + (rng.random(n) - 0.5) * (2.0 * sigma)
                        z = project_box_sum(y, target)
                        c_tmp = correlate_full_h(z) * dx
                        ct = float(np.max(c_tmp)) if c_tmp.size else 0.0
                        St = lse_value_from_c(c_tmp, beta)
                        if not (np.isfinite(ct) and np.isfinite(St)):
                            continue
                        if ct < c5 - 1e-11 or ct < best_c - 1e-11:
                            h = z
                            S = St
                            c5 = ct
                            if c5 < best_c - 1e-11:
                                best_c = c5
                                best_h = h.copy()
                            step = sigma
                            fail = 0
                            kicked = True
                            break
                if not kicked:
                    break

    return best_h, float(best_c), h.copy(), float(S)


def exact_pg(h, deadline, rng=None, active_tol=1e-5, max_active=40, step_init=5e-3):
    h = ensure_feasible(h).copy()
    n = int(h.size)
    target = 0.5 * float(n)
    dx = 2.0 / float(n)

    c = correlate_full_h(h) * dx
    c5 = float(np.max(c)) if c.size else 0.0
    best_h = h.copy()
    best_c = c5

    if step_init is None:
        step = 5e-3
    else:
        step = float(step_init)

    fail = 0
    factors = (0.25, 0.5, 1.0, 2.0, 4.0)
    it = 0

    while it < 100000:
        if time_left(deadline) < 0.25:
            break
        it += 1

        c = correlate_full_h(h) * dx
        c5 = float(np.max(c)) if c.size else 0.0
        if c5 < best_c - 1e-12:
            best_c = c5
            best_h = h.copy()

        active_tol_use = max(1e-6, float(active_tol))
        active = np.where(c >= c5 - active_tol_use)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c))], dtype=int)
        if active.size > max_active:
            active = active[np.argsort(c[active])[-max_active:]]

        w = np.zeros(c.size, dtype=float)
        w[active] = 1.0 / float(active.size)
        g = weighted_gradient(h, w)

        if not np.all(np.isfinite(g)):
            break
        gn = float(np.linalg.norm(g))
        if gn < 1e-11:
            break

        if step is None or step <= 1e-12:
            step = min(0.05, max(1e-4, 0.1 / (gn + 1e-9)))

        best_tr = None
        best_C = None
        best_factor = None

        for f in factors:
            st = step * f
            if st <= 0.0:
                continue
            y = h - st * g
            z = project_box_sum(y, target)
            if np.max(np.abs(z - h)) < 1e-15:
                continue
            c_tmp = correlate_full_h(z) * dx
            ct = float(np.max(c_tmp)) if c_tmp.size else 0.0
            if not np.isfinite(ct):
                continue
            if best_C is None or ct < best_C - 1e-15:
                best_tr = z
                best_C = ct
                best_factor = f

        accepted = False
        if best_tr is not None and best_C < c5 - 1e-11:
            h = best_tr
            c5 = best_C
            if c5 < best_c - 1e-12:
                best_c = c5
                best_h = h.copy()

            if best_factor is not None:
                if best_factor > 1.5:
                    step = min(0.5, step * 1.4)
                elif best_factor < 0.7:
                    step = max(1e-7, step * 0.7)
            fail = 0
            accepted = True

        if not accepted:
            step = max(1e-8, step * 0.5)
            fail += 1
            if step < 1e-8 or fail > 30:
                kicked = False
                if rng is not None:
                    for _ in range(20):
                        sigma = max(step, 0.004)
                        y = h + (rng.random(n) - 0.5) * (2.0 * sigma)
                        z = project_box_sum(y, target)
                        c_tmp = correlate_full_h(z) * dx
                        ct = float(np.max(c_tmp)) if c_tmp.size else 0.0
                        if not np.isfinite(ct):
                            continue
                        if ct < c5 - 1e-11 or ct < best_c - 1e-11:
                            h = z
                            c5 = ct
                            if c5 < best_c - 1e-11:
                                best_c = c5
                                best_h = h.copy()
                            step = sigma
                            fail = 0
                            kicked = True
                            break
                if not kicked:
                    break

    return best_h, float(best_c)


def _slsqp_objective(x):
    x = np.asarray(x, dtype=float).ravel()
    x = np.clip(x, 0.0, 1.0)
    beta = float(_SLSQP_PARAM.get("beta", 40.0))
    S, _ = lse_value_from_h(x, beta)
    if not np.isfinite(S):
        return 1e6
    return float(S)


def _slsqp_jacobian(x):
    x = np.asarray(x, dtype=float).ravel()
    x = np.clip(x, 0.0, 1.0)
    beta = float(_SLSQP_PARAM.get("beta", 40.0))
    return lse_obj_grad(x, beta)[0]


def _slsqp_constraint_fun(x):
    x = np.asarray(x, dtype=float).ravel()
    return float(np.sum(x) - 0.5 * float(x.size))


def _slsqp_constraint_jac(x):
    x = np.asarray(x, dtype=float).ravel()
    return np.ones_like(x, dtype=float)


def slsqp_smooth(h, beta, max_iter):
    if _optimize is None:
        return h, eval_c5(h)

    h = ensure_feasible(h).copy()
    n = int(h.size)
    c0 = eval_c5(h)
    if n == 0 or not np.isfinite(c0):
        return h, c0

    _SLSQP_PARAM["beta"] = float(beta)

    bounds = [(0.0, 1.0)] * n
    constraints = [
        {"type": "eq", "fun": _slsqp_constraint_fun, "jac": _slsqp_constraint_jac}
    ]

    try:
        res = _optimize.minimize(
            _slsqp_objective,
            h,
            method="SLSQP",
            jac=_slsqp_jacobian,
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(max_iter), "ftol": 1e-10, "disp": False},
        )
    except Exception:
        return h, c0

    if hasattr(res, "x") and res.x is not None:
        x = np.asarray(res.x, dtype=float).ravel()
        if x.size == n and np.all(np.isfinite(x)):
            x = ensure_feasible(x)
            c1 = eval_c5(x)
            if np.isfinite(c1) and c1 < c0 - 1e-12:
                return x, c1

    return h, c0


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    budget = float(budget_s)
    deadline = start + max(1.0, budget)
    rng = np.random.default_rng(seed)

    parent = globals().get("initial_h_values", None)
    h0 = None
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=float)
            if p.size > 0:
                h0 = p.ravel().copy()
        except Exception:
            h0 = None

    if h0 is None:
        h0 = np.full(70, 0.5, dtype=float)

    h0 = ensure_feasible(h0)
    best_h = h0
    best_c = eval_c5(h0)
    if not np.isfinite(best_c):
        best_c = 0.5
    base_n = int(best_h.size)

    if base_n == 0 or time_left(deadline) < 2.0:
        c_final = eval_c5_direct(best_h) if best_h.size > 0 else 0.0
        return best_h, float(c_final), int(best_h.size)

    test_gradient()

    if time_left(deadline) > 25.0:
        ex_dead = min(time.time() + 25.0, deadline - 20.0)
        if time.time() < ex_dead:
            hx, cx = exact_pg(best_h.copy(), ex_dead, rng)
            if cx < best_c:
                best_c = cx
                best_h = np.asarray(hx, dtype=float).copy()

    if _optimize is not None and time_left(deadline) > 120.0:
        slsqp_dead = min(time.time() + min(150.0, 0.12 * budget), deadline - 80.0)
        maxiter = 60 if best_h.size <= 100 else (40 if best_h.size <= 200 else 20)
        if time.time() < slsqp_dead:
            hs, cs = slsqp_smooth(best_h, 40.0, maxiter)
            if np.isfinite(cs) and cs < best_c:
                best_c = cs
                best_h = np.asarray(hs, dtype=float).copy()
        if time.time() < slsqp_dead - 10.0:
            hs, cs = slsqp_smooth(best_h, 80.0, max(10, maxiter // 2))
            if np.isfinite(cs) and cs < best_c:
                best_c = cs
                best_h = np.asarray(hs, dtype=float).copy()

    smooth_total = max(60.0, min(650.0, 0.35 * budget))
    smooth_dead = min(time.time() + smooth_total, deadline - 120.0)
    betas = (30.0, 60.0, 120.0, 200.0)
    cur = best_h.copy()

    for bi, beta in enumerate(betas):
        if time.time() >= smooth_dead or time_left(deadline) < 30.0:
            break
        remaining_sm = smooth_dead - time.time()
        if remaining_sm < 20.0:
            break
        stage = min(150.0, remaining_sm / float(max(1, len(betas) - bi)))
        stage_dead = time.time() + stage
        if stage_dead <= time.time():
            continue

        h_opt, c_opt, h_fin, _ = smooth_pg(cur.copy(), beta, stage_dead, rng)
        if np.isfinite(c_opt) and c_opt < best_c:
            best_c = c_opt
            best_h = np.asarray(h_opt, dtype=float).copy()

        c_fin = eval_c5(h_fin)
        if np.isfinite(c_fin) and c_fin <= best_c + 0.02:
            cur = np.asarray(h_fin, dtype=float).copy()
        else:
            cur = best_h.copy()

        if beta >= 120.0 and time_left(deadline) > 60.0 and time.time() < smooth_dead:
            ex_dead = min(time.time() + 40.0, smooth_dead - 5.0)
            if time.time() < ex_dead:
                hx, cx = exact_pg(cur.copy(), ex_dead, rng)
                if cx < best_c:
                    best_c = cx
                    best_h = np.asarray(hx, dtype=float).copy()

    if time_left(deadline) > 35.0:
        ex_dead = min(time.time() + min(120.0, 0.08 * budget), deadline - 60.0)
        if time.time() < ex_dead:
            hx, cx = exact_pg(best_h.copy(), ex_dead, rng)
            if cx < best_c:
                best_c = cx
                best_h = np.asarray(hx, dtype=float).copy()

    if time_left(deadline) > 150.0:
        max_n = int(min(2048, max(base_n, 16 * base_n)))
        upsrc = best_h.copy()
        seen = set([int(best_h.size)])
        up_steps = 0

        while up_steps < 4 and time_left(deadline) > 150.0:
            ncur = int(upsrc.size)
            new_n = min(2 * ncur, max_n)
            if new_n in seen or new_n <= ncur:
                break
            seen.add(new_n)

            h_up = upsample(upsrc, new_n)
            c_up = eval_c5(h_up)
            progressed = False

            if np.isfinite(c_up) and c_up < best_c:
                best_c = c_up
                best_h = np.asarray(h_up, dtype=float).copy()
                progressed = True

            if time_left(deadline) > 120.0 and np.isfinite(c_up) and c_up < best_c + 0.05:
                stage_dead = min(time.time() + min(200.0, 0.3 * time_left(deadline)), deadline - 80.0)
                if stage_dead > time.time():
                    h_opt, c_opt, h_fin, _ = smooth_pg(h_up, 40.0, stage_dead, rng)
                    if np.isfinite(c_opt) and c_opt < best_c:
                        best_c = c_opt
                        best_h = np.asarray(h_opt, dtype=float).copy()
                        progressed = True

                    c_fin = eval_c5(h_fin)
                    cand = h_fin if (np.isfinite(c_fin) and c_fin <= best_c + 0.03) else h_up

                    ex_dead = min(time.time() + 60.0, deadline - 40.0)
                    if time.time() < ex_dead:
                        hx, cx = exact_pg(cand.copy(), ex_dead, rng)
                        if np.isfinite(cx) and cx < best_c:
                            best_c = cx
                            best_h = np.asarray(hx, dtype=float).copy()
                            progressed = True
                            cand = hx

                    c_cand = eval_c5(cand)
                    if progressed or (np.isfinite(c_cand) and c_cand <= best_c + 0.03):
                        upsrc = np.asarray(cand, dtype=float).copy()
                    else:
                        upsrc = exec h_up if executed else upsrc  # never happens
            else:
                if progressed:
                    upsrc = best_h.copy()

            up_steps += 1
            if not progressed and (not np.isfinite(c_up) or c_up > best_c):
                break

    if time_left(deadline) > 120.0:
        random_dead = min(deadline - 10.0, time.time() + min(300.0, 0.20 * budget))
        n_rand = int(min(280, max(base_n, best_h.size)))
        if n_rand > 0:
            starts = []
            starts.append(np.full(n_rand, 0.5, dtype=float))

            t = np.linspace(0.0, 1.0, n_rand, endpoint=False)
            for f in (1, 2, 3, 5, 7, 11):
                for amp in (0.5, 0.8, 1.0):
                    starts.append(0.5 + amp * np.cos(2.0 * np.pi * float(f) * t))
                    starts.append(0.5 + amp * np.sin(2.0 * np.pi * float(f) * t + 0.7))

            for _ in range(8):
                starts.append(rng.random(n_rand))

            kbin = int(round(0.5 * n_rand))
            for _ in range(8):
                hb = np.zeros(n_rand, dtype=float)
                if kbin > 0:
                    idx = rng.choice(n_rand, size=kbin, replace=False)
                    hb[idx] = 1.0
                starts.append(hb)

            for y in starts:
                if time.time() >= random_dead - 5.0:
                    break
                yy = np.asarray(y, dtype=float).ravel()
                if yy.size != n_rand:
                    continue
                z = ensure_feasible(np.clip(yy, 0.0, 1.0))
                zc = eval_c5(z)
                if not np.isfinite(zc) or zc >= best_c + 0.02:
                    continue

                stage_dead = min(time.time() + 40.0, random_dead - 10.0)
                if stage_dead > time.time():
                    ho, co, hf, _ = smooth_pg(z, 50.0, stage_dead, rng)
                    if np.isfinite(co) and co < best_c:
                        best_c = co
                        best_h = np.asarray(ho, dtype=float).copy()

                    ex_dead = min(time.time() + 15.0, random_dead - 2.0)
                    if time.time() < ex_dead:
                        hx, cx = exact_pg(ho.copy(), ex_dead, rng)
                        if np.isfinite(cx) and cx < best_c:
                            best_c = cx
                            best_h = np.asarray(hx, dtype=float).copy()

    best_h = ensure_feasible(best_h)

    if best_h.size <= 2048 or time_left(deadline) > 0.7:
        c_final = eval_c5_direct(best_h)
    else:
        c_final = eval_c5(best_h)

    if np.isfinite(c_final) and c_final < best_c:
        best_c = c_final

    n_final = int(best_h.size)
    sum_err = abs(float(np.sum(best_h)) - 0.5 * float(n_final))
    if sum_err > 1e-9:
        hd = make_dyadic_feasible(best_h, 30)
        cd = eval_c5_direct(hd) if n_final <= 2048 or time_left(deadline) > 0.7 else eval_c5(hd)
        if np.isfinite(cd) and cd <= c_final + 1e-8:
            best_h = hd
            c_final = cd

    if not np.isfinite(c_final):
        c_final = 0.38085

    return np.asarray(best_h, dtype=float).copy(), float(c_final), int(n_final)
