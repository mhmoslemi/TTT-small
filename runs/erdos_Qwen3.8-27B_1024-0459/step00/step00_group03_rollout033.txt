import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))


def soft_max_obj(h, beta, dx, target_sum):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    bc = beta * c * dx
    bc_max = np.max(bc)
    exp_bc = np.exp(bc - bc_max)
    sum_exp = np.sum(exp_bc)
    f = (bc_max + np.log(sum_exp)) / beta
    p = exp_bc / sum_exp

    s = np.sum(h) - target_sum
    f += 500.0 * s * s

    n_full = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_full):
        wk = p[k] * beta * dx
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue
        idx = np.arange(lo, hi)
        grad[idx] += wk * (1.0 - h[idx + d])
        grad[idx + d] -= wk * h[idx]

    grad += 1000.0 * s * np.ones(n)
    return f, grad


def project_sum(h, target):
    h = h.copy()
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        count = np.sum(mask)
        if count == 0:
            h[:] = target / len(h)
            break
        h[mask] += diff / count
        np.clip(h, 0.0, 1.0, out=h)
    return h


def resample_h(h_in, n_out):
    n_in = len(h_in)
    if n_in == n_out:
        return h_in.copy()
    x_in = np.linspace(0.0, 1.0, n_in, endpoint=False)
    x_out = np.linspace(0.0, 1.0, n_out, endpoint=False)
    h_out = np.interp(x_out, x_in, h_in)
    return h_out


def binary_search(h_init, dx, target_sum, max_iters, seed):
    n = len(h_init)
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    order = np.argsort(h_init)[::-1]
    h[order[:n // 2]] = 1.0

    c5 = compute_c5(h, dx)
    best_c5 = c5
    best_h = h.copy()
    no_improve = 0
    patience = 800

    for it in range(max_iters):
        if no_improve > patience:
            break
        i = rng.randint(n)
        if h[i] == 1.0:
            zeros = np.where(h == 0)[0]
        else:
            zeros = np.where(h == 1)[0]
        if len(zeros) == 0:
            continue
        j = zeros[rng.randint(len(zeros))]

        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]

        c5_new = compute_c5(h, dx)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
            no_improve += 1

    return best_h, best_c5


def try_optimize(h_start, n, dx, target_sum, t0, budget_s, seed, n_starts):
    h_curr = h_start.copy()
    best_local_c5 = compute_c5(h_curr, dx)

    for beta in [3.0, 5.0, 8.0, 15.0, 30.0, 60.0]:
        if time.time() - t0 > budget_s - 20:
            break
        maxiter = 400 if n <= 90 else 250
        try:
            res = minimize(
                soft_max_obj, h_curr,
                args=(beta, dx, target_sum),
                method='L-BFGS-B',
                jac=True,
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
            )
            h_curr = project_sum(res.x, target_sum)
        except Exception:
            pass

        c5_curr = compute_c5(h_curr, dx)
        if c5_curr < best_local_c5:
            best_local_c5 = c5_curr

    # Binary refinement
    if time.time() - t0 > budget_s - 20:
        return h_curr, best_local_c5

    max_bi = 4000 if n <= 80 else 2500
    hb, c5_b = binary_search(h_curr, dx, target_sum, max_bi, seed + n_starts)
    if c5_b < best_local_c5:
        h_curr = hb
        best_local_c5 = c5_b

    return h_curr, best_local_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.ones(70) * 0.5
    n0 = len(h0)
    dx0 = 2.0 / n0
    c5_0 = compute_c5(h0, dx0)

    best_c5 = c5_0
    best_h = h0.copy()
    best_n = n0

    rng = np.random.RandomState(seed)

    # Resolutions to try (must be even for integer sum)
    n_candidates = [60, 70, 80, 84, 90, 96, 100, 108, 120, 140]
    n_list = sorted(set([n0] + [n for n in n_candidates if n % 2 == 0]))
    n_list = [n for n in n_list if 50 <= n <= 150]

    n_starts_counter = 0

    for n in n_list:
        if time.time() - t0 > budget_s - 100:
            break
        dx = 2.0 / n
        target = n / 2.0

        # Build start points
        starts = []

        # 1. Resample from original
        if n == n0:
            starts.append(h0.copy())
        else:
            hs = resample_h(h0, n)
            hs = project_sum(hs, target)
            starts.append(hs)

        # 2. Resample from best found so far
        if best_n != n:
            hs2 = resample_h(best_h, n)
            hs2 = project_sum(hs2, target)
            starts.append(hs2)
        elif len(starts) < 2:
            starts.append(h0.copy())

        # 3. Random start
        hr = np.clip(rng.normal(0.5, 0.25, n), 0, 1)
        hr = project_sum(hr, target)
        starts.append(hr)

        # 4. Structured "alternating blocks" start
        h_block = np.zeros(n)
        block_size = max(2, n // 10)
        for i in range(0, n - block_size, 2 * block_size):
            h_block[i:i + block_size] = 1.0
        h_block = project_sum(h_block, target)
        starts.append(h_block)

        # 5. Another random with different variance
        hr2 = np.clip(rng.normal(0.5, 0.15, n), 0, 1)
        hr2 = project_sum(hr2, target)
        starts.append(hr2)

        for si, h_start in enumerate(starts):
            if time.time() - t0 > budget_s - 50:
                break
            if np.max(h_start) <= 0.01 or np.min(h_start) >= 0.99:
                continue

            h_opt, c5_local = try_optimize(
                h_start, n, dx, target, t0, budget_s, seed, n_starts_counter
            )
            n_starts_counter += 1

            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_opt.copy()
                best_n = n

                # Also try binary refinement on the new best at this resolution
                if time.time() - t0 > budget_s - 30:
                    break
                hb2, c5_b2 = binary_search(h_opt, dx, target, 5000, seed + 1000 + n_starts_counter)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = hb2.copy()
                    best_n = n

    # Final: ensure constraint is met exactly
    dx_best = 2.0 / best_n
    best_h = project_sum(best_h, best_n / 2.0)
    final_c5 = compute_c5(best_h, dx_best)

    return best_h, final_c5, best_n
