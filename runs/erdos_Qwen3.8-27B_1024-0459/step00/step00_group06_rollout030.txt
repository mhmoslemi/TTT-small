import numpy as np
import time
import math

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def greedy_local_search(h, n, rng, max_rounds=200, time_limit=None, start_time=None):
    """Greedy: try all single swaps, take best improvement, repeat."""
    cur = compute_c5(h, n)
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    
    for r in range(max_rounds):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        best_delta = 0.0
        best_a, best_b = -1, -1
        no = len(ones_idx)
        nz = len(zeros_idx)
        # For small n, check all; for large n, sample
        total_pairs = no * nz
        if total_pairs <= 20000:
            pairs = [(ai, bi) for ai in range(no) for bi in range(nz)]
        else:
            pairs = [(int(rng.randint(no)), int(rng.randint(nz))) for _ in range(20000)]
        
        for ai, bi in pairs:
            a = int(ones_idx[ai])
            b = int(zeros_idx[bi])
            h[a], h[b] = h[b], h[a]
            new_c5 = compute_c5(h, n)
            delta = new_c5 - cur
            if delta < best_delta:
                best_delta = delta
                best_a, best_b = a, b
            h[a], h[b] = h[b], h[a]
        
        if best_a < 0:
            break
        h[best_a], h[best_b] = h[best_b], h[best_a]
        cur += best_delta
        # Update tracking
        ia = np.where(ones_idx == best_a)[0][0]
        ib = np.where(zeros_idx == best_b)[0][0]
        ones_idx[ia] = best_b
        zeros_idx[ib] = best_a
    return cur

def simulated_annealing(h, n, rng, duration_s, T_init=0.05, T_min=0.0005):
    """SA with swap moves, returns best found."""
    best_c5_local = compute_c5(h, n)
    cur_c5 = best_c5_local
    best_h_local = h.copy()
    
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    T = T_init
    end_time = time.time() + duration_s
    it = 0
    stall = 0
    stall_threshold = 50000
    
    while time.time() < end_time:
        i_one = rng.randint(len(ones_list))
        i_zero = rng.randint(len(zeros_list))
        a = ones_list[i_one]
        b = zeros_list[i_zero]
        
        h[a], h[b] = h[b], h[a]
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < math.exp(-max(delta, 0.0) / T):
            cur_c5 = new_c5
            ones_list[i_one] = b
            zeros_list[i_zero] = a
            if cur_c5 < best_c5_local:
                best_c5_local = cur_c5
                best_h_local = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[a], h[b] = h[b], h[a]
            stall += 1
        
        it += 1
        if it % 2000 == 0:
            T = max(T * 0.999, T_min)
        if stall > stall_threshold and T > T_min:
            T = max(T * 0.9, T_min)
            stall = 0
    
    return best_c5_local, best_h_local

def try_continuous_refine(h, n, time_limit=60):
    """Try to improve with continuous optimization (log-sum-exp smoothing)."""
    from scipy.optimize import minimize
    
    dx = 2.0 / n
    target_sum = n / 2.0
    
    def smoothed_obj(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode='full')
        # Log-sum-exp for smooth max
        tau = 0.01
        scaled = corr / (dx * tau)
        m = np.max(scaled)
        log_sum = m + np.log(np.sum(np.exp(scaled - m)))
        penalty = 1e6 * (np.sum(x) - target_sum) ** 2
        return float(dx * tau * log_sum + penalty)
    
    x0 = h.astype(float).copy()
    bounds = [(0.0, 1.0)] * n
    
    start_t = time.time()
    try:
        res = minimize(smoothed_obj, x0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8,
                                'maxfun': 500})
    except Exception:
        return h, compute_c5(h, n)
    
    if time.time() - start_t > time_limit:
        return h, compute_c5(h, n)
    
    h_new = np.clip(res.x, 0.0, 1.0)
    # Project to satisfy constraint approximately
    diff = target_sum - np.sum(h_new)
    if abs(diff) > 1e-6:
        # Adjust slightly
        h_new = h_new + diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
    
    c5_new = compute_c5(h_new, n)
    c5_old = compute_c5(h, n)
    if c5_new < c5_old:
        return h_new, c5_new
    return h, c5_old
