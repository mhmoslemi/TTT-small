def project_to_feasible(h, n):
    """Project h onto {0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
