def compute_gradient_vec(h, lam=100.0):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    scaled = lam * dx * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    weights = exp_s / np.sum(exp_s)
    
    # Build the gradient matrix
    # For each p (0..n-1) and k (0..2n-2):
    # s1 = p + k - (n-1), valid if 0<=s1<n
    # s2 = p - k + (n-1), valid if 0<=s2<n
    # contribution = weights[k] * ((1-h[s1])*I(0<=s1<n) - h[s2]*I(0<=s2<n))
    
    # Vectorize: create arrays of s1 and s2 for all (p,k) pairs
    p = np.arange(n)[:, None]  # shape (n, 1)
    k = np.arange(2*n-1)[None, :]  # shape (1, 2n-1)
    s1 = p + k - (n-1)  # shape (n, 2n-1)
    s2 = p - k + (n-1)  # shape (n, 2n-1)
    
    valid1 = (s1 >= 0) & (s1 < n)
    valid2 = (s2 >= 0) & (s2 < n)
    
    # For valid1: contribution is (1-h[s1])
    # For valid2: contribution is -h[s2]
    
    w = weights[None, :]  # shape (1, 2n-1)
    
    term1 = np.where(valid1, one_minus_h[s1], 0.0)  # (n, 2n-1)
    term2 = np.where(valid2, h[s2], 0.0)  # (n, 2n-1)
    
    grad = dx * np.sum(w * (term1 - term2), axis=1)
    
    return grad

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    """Compute C5 = max_k(correlate(h, 1-h, 'full') * dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def objective_and_grad(h, lam=500.0, mu=1000.0):
    """Objective (log-sum-exp + sum penalty) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    m = len(corr)  # 2n-1
    
    # Log-sum-exp of lam*dx*corr
    scaled = lam * dx * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    lse = (max_s + np.log(sum_exp)) / lam
    
    # Sum penalty
    s = np.sum(h)
    target = n / 2.0
    penalty = mu * (s - target) ** 2
    
    obj = lse + penalty
    
    # Softmax weights
    weights = exp_s / sum_exp
    
    # Gradient of LSE
    p_idx = np.arange(n)[:, None]
    k_idx = np.arange(m)[None, :]
    s1 = p_idx + k_idx - (n - 1)
    s2 = p_idx - k_idx + (n - 1)
    
    valid1 = (s1 >= 0) & (s1 < n)
    valid2 = (s2 >= 0) & (s2 < n)
    
    w_row = weights[None, :]
    
    term1 = np.where(valid1, 1.0 - h[np.clip(s1, 0, n-1)], 0.0)
    term2 = np.where(valid2, h[np.clip(s2, 0, n-1)], 0.0)
    
    grad_lse = dx * np.sum(w_row * (term1 - term2), axis=1)
    
    # Gradient of penalty
    grad_penalty = 2.0 * mu * (s - target) * np.ones(n)
    
    grad = grad_lse + grad_penalty
    return obj, grad
