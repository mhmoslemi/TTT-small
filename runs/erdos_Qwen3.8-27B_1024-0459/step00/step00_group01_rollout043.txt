def smooth_max_gradient(c, alpha):
    """Compute gradient of (1/alpha)*log(sum(exp(alpha*c))) with respect to c."""
    # c is the array of correlation values (already multiplied by dx)
    max_c = np.max(c)
    exp_c = np.exp(alpha * (c - max_c))
    p = exp_c / np.sum(exp_c)
    return p  # These are the "softmax" weights

def compute_gradient(h, alpha):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    c = corr * dx  # values including dx factor
    
    # Softmax weights
    max_c = np.max(c)
    exp_c = np.exp(alpha * (c - max_c))
    p = exp_c / np.sum(exp_c)
    
    # Gradient
    grad = np.zeros(n)
    # For each lag k (index m in corr, k = m - (n-1)):
    for m in range(2*n - 1):
        k = m - (n - 1)  # actual lag
        if k == 0:
            # c_0 = sum_i h[i]*(1-h[i])
            # dc_0/dh[j] = (1-h[j]) - h[j] = 1 - 2*h[j]

            for j in range(kp, n):
