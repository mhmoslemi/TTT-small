    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        js = j - s
        if 0 <= js < n:
            grad[j] += (1.0 - h[js])
        jp = j + s
        if 0 <= jp < n:
            grad[j] -= h[jp]
    return grad * dx

def smooth_objective(h, alpha=5.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - np.max(corr)
    return float(np.max(corr) + (1.0 / alpha) * np.log(np.sum(np.exp(alpha * shifted))))

def smooth_gradient(h, alpha=5.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    denom = np.sum(np.exp(alpha * corr))
    weights = np.exp(alpha * corr) / denom
    grad = np.zeros(n)
