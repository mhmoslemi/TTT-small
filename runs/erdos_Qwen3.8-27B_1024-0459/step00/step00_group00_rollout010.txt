        for it in range(max_iters):
            if time.time() > phase_end:
                break
            tau = tau_start * (tau_end / tau_start) ** (it / max(1, max_iters - 1))
            val, grad = smooth_max_value_grad(y, tau, n)

            # Adaptive step: use Lipschitz estimate
            L = 4.0 * (2 * n - 1) * (2.0 / n)
            step = 1.0 / (L + 1e-8)

            y_next = y - step * grad
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            y_next = y_next + ((t_fista - 1.0) / t_new) * (y_next - h)

            h_new = project_feasible(y_next, n)
            c5_new = true_c5(h_new)
            if c5_new < best_local:
                best_local = c5_new
            h = h_new
            y = h
            t_fista = t_new

        return h, best_local

    target_ns = [90, 135, 180, 270, 360, 540]
    for target_n in target_ns:
        if time.time() - start_time > budget_s * 0.80:
            break
        if target_n <= best_n:
            continue

        # Interpolate
        x_old = (np.arange(best_n) + 0.5) * 2.0 / best_n
        x_new = (np.arange(target_n) + 0.5) * 2.0 / target_n
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, target_n)

        # Run projected gradient with time budget
        remaining = budget_s * 0.80 - (time.time() - start_time)
        time_per = max(5.0, remaining / (len(target_ns) - target_ns.index(target_n)))
        max_iters = max(200, 200000 // target_n)
        h_opt, _ = projected_gradient(h_interp, target_n, max_iters, time_per)
        c5_exact = true_c5(h_opt)

        if c5_exact < best_c5:
            best_c5 = c5_exact
            best_h = h_opt.copy()
            best_n = target_n

    # Phase 3: Second round of projected gradient with finer tau
    if time.time() - start_time < budget_s * 0.85 and best_n <= 360:
        x_old = (np.arange(best_n) + 0.5) * 2.0 / best_n
        target_n2 = best_n * 2
        x_new2 = (np.arange(target_n2) + 0.5) * 2.0 / target_n2
        h_interp2 = np.interp(x_new2, x_old, best_h)
        h_interp2 = project_feasible(h_interp2, target_n2)
        remaining = budget_s * 0.85 - (time.time() - start_time)
        if remaining > 10:
            h_opt2, _ = projected_gradient(h_interp2, target_n2, max(200, 100000 // target_n2), remaining * 0.7)
            c5_2 = true_c5(h_opt2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h_opt2.copy()
                best_n = target_n2

    # Phase 4: Final SLSQP polish if resolution allows
    if best_n <= 150 and time.time() - start_time < budget_s * 0.88:
        tau_fin = 0.005
        n_fin = best_n

        def fin_fun(h):
            return smooth_max_value_grad(h, tau_fin, n_fin)[0]

        def fin_jac(h):
            return smooth_max_value_grad(h, tau_fin, n_fin)[1]

        eq_con_f = {'type': 'eq',
                    'fun': lambda h: np.sum(h) - n_fin / 2.0,
                    'jac': lambda h: np.ones(n_fin)}
        bnds_f = [(0.0, 1.0)] * n_fin
        try:
            res_f = minimize(fin_fun, best_h, jac=fin_jac, method='SLSQP',
                             bounds=bnds_f, constraints=[eq_con_f],
                             options={'maxiter': 1000, 'ftol': 1e-14})
            h_fin = project_feasible(res_f.x, n_fin)
            c5_fin = true_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        except Exception:
            pass

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n)
    final_c5 = true_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    assert 0 <= np.min(best_h) and np.max(best_h) <= 1.0
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6
    assert len(best_h) == best_n
    assert np.all(np.isfinite(best_h))

    return (best_h, best_c5, best_n)
