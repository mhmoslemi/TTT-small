```python
import numpy as np
from scipy.optimize import minimize
from scipy import correlate
import time

def eval_c5(h):
    """Evaluate the Erdős C5 objective for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def make_initial_binary(n):
    """Create an initial binary solution with n/2 ones."""
    h = np.zeros(n)
    # Place ones at alternating positions
    for i in range(0, n, 2):
        h[i] = 1.0
    return h

def objective_epigraph(params, n):
    """Objective: the epigraph variable t."""
    return params[n]

def constraint_t_corr(params, n, k, dx):
    """Inequality: t >= corr[k] * dx."""
    h = params[:n]
    t = params[n]
    corr = correlate.correlate(h, 1.0 - h, mode="full") * dx
    return t - corr[k]

def constraint_sum(params, n):
    """Equality: sum(h) = n/2."""
    return np.sum(params[:n]) - n / 2.0

def get_bounds(n):
    """Box constraints: 0 <= h[i] <= 1, 0 <= t <= 1."""
    return [(0.0, 1.0)] * n + [(0.0, 1.0)]

def get_constraints(n):
    """Get all constraints for SLSQP."""
    dx = 2.0 / n
    cons = []
    for k in range(2 * n - 1):
        cons.append({
            'type': 'ineq',
            'fun': constraint_t_corr,
            'args': (n, k, dx)
        })
    cons.append({
        'type': 'eq',
        'fun': constraint_sum,
        'args': (n,)
    })
    return cons

def optimize_slsqp(h_init, max_iter=300, timeout=None):
    """Optimize using SLSQP with epigraph reformulation."""
    n = len(h_init)
    initial_t = eval_c5(h_init) + 0.01
    x0 = np.concatenate([h_init, [min(initial_t, 1.0)]])
    
    result = minimize(
        objective_epigraph,
        x0,
        method='SLSQP',
        bounds=get_bounds(n),
        constraints=get_constraints(n),
        args=(n,),
        options={'maxiter': max_iter, 'ftol': 1e-10}
    )
    
    if result.success or result.fun < initial_t:
        h_opt = result.x[:n]
        # Snap to satisfy constraint exactly
        h_opt = np.clip(h_opt, 0, 1)
        diff = n / 2.0 - np.sum(h_opt)
        if abs(diff) > 1e-10:
            # Distribute the difference
            # Add to positions that have room
            for i in range(n):
                if diff > 0 and h_opt[i] < 1.0:
                    adj = min(diff, 1.0 - h_opt[i])
                    h_opt[i] += adj
                    diff -= adj
                elif diff < 0 and h_opt[i] > 0.0:
                    adj = min(-diff, h_opt[i])
                    h_opt[i] -= adj
                    diff += adj
    else:
        h_opt = h_init
```

```python
def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def make_binary_initial(n, seed=0):
    rng = np.random.default_rng(seed)
```
```python
    positions = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h

def snap_constraint(h, n):
```
```python
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for i in order:
        if diff > 0:
            adj = min(diff, 1.0 - h[i])
        else:
            adj = max(diff, -h[i])
        h[i] += adj
        diff -= adj
        if abs(diff) < 1e-12:
            break
    return h

def obj_epi(params, n):
    return params[n]

def constr_epi(params, n, k, dx):
    h = params[:n]
    t = params[n]
    corr = correlate.correlate(h, 1.0 - h, mode="full") * dx
    return t - corr[k]

def constr_eq(params, n):
    return np.sum(params[:n]) - n / 2.0

    c0 = eval_c5(h_init)
```
```python
    x0 = np.concatenate([h_init, [min(c0 + 0.02, 1.0)]])

    constraints = []

            'type': 'ineq',
            'fun': constr_epi,
            'args': (n, k, dx)
        })

        'type': 'eq',
        'fun': constr_eq,
        'args': (n,)
    })

        obj_epi, x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        args=(n,),
```

```python
        c5_bin = eval_c5(h_bin)
        if c5_bin < c5_new:
            h_new, c5_new = h_bin, c5_bin
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Coordinate polish on best
    if time.time() - start_time < budget_s - 10:
        best_h = coordinate_polish(best_h, 2)
        best_c5 = eval_c5(best_h)

    # Final: ensure constraint satisfied
    best_h = snap_constraint(best_h, len(best_h))
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
```
