from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_binary(h, n):
    """Project h to a binary vector with exactly n//2 ones."""
    # Sort values, take top n//2 as 1
    sorted_idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[sorted_idx[:n//2]] = 1.0
    return result

def simulated_annealing_binary(h, n, start_t, end_t, iterations, rng):
    """SA on binary vector with n//2 ones."""
    h = h.copy()
    # Ensure exactly n//2 ones
    ones_count = np.sum(h == 1)
    if ones_count != n // 2:
        h = project_to_binary(h, n)
    
    current_energy = compute_c5(h)
    best_energy = current_energy
    best_h = h.copy()
    
    T = start_t
    cooling = (end_t / start_t) ** (1.0 / max(iterations, 1))
    
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    for it in range(iterations):
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            break
        
        # Pick random swap
        i = ones_pos[rng.integers(len(ones_pos))]
        j = zeros_pos[rng.integers(len(zeros_pos))]
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        new_energy = compute_c5(h)
        
        delta = new_energy - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_energy = new_energy
            # Update position sets
            ones_pos = np.where(h == 1)[0]
            zeros_pos = np.where(h == 0)[0]
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
        
        T *= cooling
    
    return best_h, best_energy

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def make_binary(h, n, rng):
    target = n // 2
    if n % 2 != 0:
        target = (n - 1) // 2
    h = np.clip(h, 0.0, 1.0)
    if np.sum(h >= 0.5) >= target:
        idx = np.where(h >= 0.5)[0]
        vals = h[idx]
        order = np.argsort(vals)[::-1]
        result = np.zeros(n)
        chosen = idx[order[:target]]
        result[chosen] = 1.0
        return result
    else:
        idx_ones = np.where(h >= 0.5)[0]
        idx_zeros = np.where(h < 0.5)[0]
        result = np.zeros(n)
        result[idx_ones] = 1.0
        need = target - len(idx_ones)
        if need > 0:
            chosen = rng.choice(idx_zeros, size=need, replace=False)
            result[chosen] = 1.0
        return result

def sa_binary(h_init, n, n_iters, T0, T1, rng, deadline, shake_interval=50000):
    h = h_init.astype(np.float64).copy()
    target_ones = n // 2
    if np.count_nonzero(h > 0.5) != target_ones:
        h = make_binary(h, n, rng)
    
    E = compute_c5(h)
    E_best = E
    h_best = h.copy()
    
    T = T0
    if n_iters <= 1:
        return h_best, E_best
    cool = (T1 / T0) ** (1.0 / n_iters)
    
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    no_improve = 0
    max_no_improve = shake_interval
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if not ones or not zeros:
            break
        
        i_int = rng.integers(len(ones))
        j_int = rng.integers(len(zeros))
        i = ones[i_int]
        j = zeros[j_int]
        
        h[i] = 0.0
        h[j] = 1.0
        E_new = compute_c5(h)
        
        if E_new < E or (T > 1e-8 and rng.random() < np.exp(-(E_new - E) / T)):
            E = E_new
            ones[i_int] = j
            zeros[j_int] = i
            if E < E_best:
                E_best = E
                h_best = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        if no_improve >= max_no_improve:
            # Shake: do 5 random swaps
            for _ in range(5):
                if ones and zeros:
                    a = ones[rng.integers(len(ones))]
                    b = zeros[rng.integers(len(zeros))]
                    h[a] = 0.0
                    h[b] = 1.0
                    idx_a = ones.index(a)
                    idx_b = zeros.index(b)
                    ones[idx_a] = b
                    zeros[idx_b] = a
            E = compute_c5(h)
            no_improve = 0
            T = max(T, 0.02)
        
        T *= cool
    
    return h_best, E_best

def local_search_binary(h, n, rng, deadline, max_rounds=200):
    h = h.astype(np.float64).copy()
    E = compute_c5(h)
    improved = True
    rounds = 0
    
    while improved and rounds < max_rounds:
        if time.time() > deadline:
            break
        improved = False
        rounds += 1
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_swap_E = E
        best_i = -1
        best_j = -1
        
        # Try all swaps (for small n) or random subset (for large n)
        if len(ones) * len(zeros) <= 10000:
            pairs = [(i, j) for i in ones for j in zeros]
        else:
            npairs = 5000
            i_idx = rng.integers(len(ones), size=npairs)
            j_idx = rng.integers(len(zeros), size=npairs)
            pairs = [(ones[i_idx[k]], zeros[j_idx[k]]) for k in range(npairs)]
        
        for i, j in pairs:
            if time.time() > deadline:
                break
            h[i] = 0.0
            h[j] = 1.0
            E_new = compute_c5(h)
            if E_new < best_swap_E:
                best_swap_E = E_new
                best_i = i
                best_j = j
            h[i] = 1.0
            h[j] = 0.0
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            E = best_swap_E
            improved = True
    
    return h, E

def smooth_c5(h, t=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    shifted = corr - np.max(corr)
    return np.log(np.sum(np.exp(t * shifted))) / t + np.max(corr)

def continuous_polish(h_init, n, deadline, rng):
    h = h_init.astype(np.float64).copy()
    
    def objective(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s < 1e-10:
            return 1.0
        x = x * (n / 2.0) / s
        x = np.clip(x, 0, 1)
        return smooth_c5(x, t=80.0)
    
    def gradient(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s < 1e-10:
            return np.zeros(n)
        x = x * (n / 2.0) / s
        x = np.clip(x, 0, 1)
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        t = 80.0
        shifted = corr - np.max(corr)
        weights = np.exp(t * shifted)
        weights = weights / np.sum(weights)
        # grad of corr[k] w.r.t. x[j]
        # corr = np.correlate(x, 1-x)
        # d/dx[j] corr[k] involves x[j] and (1-x[j]) terms
        # For simplicity use numerical
        eps = 1e-5
        grad = np.zeros(n)
        for j in range(n):
            xp = x.copy()
            xp[j] = min(1.0, x[j] + eps)
            xn = x.copy()
            xn[j] = max(0.0, x[j] - eps)
            ep = smooth_c5(np.clip(xp * (n/2.0)/max(np.sum(xp),1e-10), 0, 1), t)
            en = smooth_c5(np.clip(xn * (n/2.0)/max(np.sum(xn),1e-10), 0, 1), t)
            grad[j] = (ep - en) / (2.0 * eps)
        return grad
    
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    E0 = objective(x0)
    
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 500, 'maxfun': 20000, 'ftol': 1e-10, 'gtol': 1e-8})
    
    if time.time() > deadline:
        return h, E0
    
    if result.success and result.fun < E0:
        x_opt = np.clip(result.x, 0, 1)
        s = np.sum(x_opt)
        if s > 1e-10:
            x_opt = x_opt * (n / 2.0) / s
            x_opt = np.clip(x_opt, 0, 1)
        E_opt = compute_c5(x_opt)
        if E_opt < E0:
            return x_opt, E_opt
    
    return h, E0
