def project_to_feasible(v, n):
    """Project v onto {x: 0 <= x_i <= 1, sum(x) = n/2}"""
    S = n / 2.0
    # Binary search on threshold
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fine-tune to get exact sum
    # Distribute residual
    residual = S - x.sum()

            x = np.clip(x, 0.0, 1.0)

    dx = 2.0 / n

    weights = np.exp((conv - conv.max()) / kappa)

        pass
    
    # Vectorized gradient computation
    # For lag k (output index = n-1+k):
    # k >= 0: g_k = sum_{n=0}^{n-1-k} h[n+k](1-h[n])
    # k < 0: g_{-m} = sum_{n=m}^{n-1} h[n-m](1-h[n])
    
    # Let's compute all gradients and then weight them
    # grad for lag k >= 0:
    #   ∂/∂h[i] = (1-h[i-k]) * [k<=i<n] + (-h[i+k]) * [0<=i<n-k]
    # grad for lag k < 0 (m=-k):
    #   ∂/∂h[i] = (1-h[i+m]) * [0<=i<n-m] + (-h[i-m]) * [m<=i<n]
    
    # For efficiency, compute the total weighted gradient
    w = weights  # length 2n-1
    center = n - 1
    
    # For each output index idx, compute its contribution to grad
    # idx corresponds to lag k = idx - center
    # If k >= 0 (idx >= center):
    #   grad contribution: w[idx] * [(1-h[i-k]) for i=k..n-1] + w[idx] * [-h[i+k] for i=0..n-1-k]
    # If k < 0 (idx < center), m = center - idx:
    #   grad contribution: w[idx] * [(1-h[i+m]) for i=0..n-1-m] + w[idx] * [-h[i-m] for i=m..n-1]
    
    # This is O(n²) which for n=280 is ~80000, fine.
    
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-15:
            continue
        k = idx - center
        if k >= 0:
            # i from k to n-1: (1-h[i-k])
            grad[k:] += w[idx] * (1.0 - h[:n-k])
            # i from 0 to n-1-k: -h[i+k]
            grad[:n-k] += w[idx] * (-h[k:])
        else:
            m = -k

            grad[:n-m] += w[idx] * (1.0 - h[m:])
            # i from m to n-1: -h[i-m]
            grad[m:] += w[idx] * (-h[:n-m])
    
    grad *= dx
    return grad, float(np.max(conv) * dx)

        opt_time = min(int(remaining * 0.7), 300)
        h_opt, c5_opt = subgradient_optimize(h_up, n, opt_time, rng, start_time + elapsed)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        
        # Local search
        elapsed2 = time.time() - start_time
        remaining2 = budget_s - elapsed2 - 30
        if remaining2 > 20:
            ls_time = min(30, int(remaining2 * 0.3))
            h_ls, c5_ls = local_search(best_h, n, ls_time, rng, start_time)
            c5_ls = compute_c5(h_ls)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
    
    # Phase 3: Random restarts at best resolution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 10
    if remaining > 20:
        n_best = len(best_h)
        n_restarts = min(5, int(remaining / 15))
        for r in range(n_restarts):
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed - 5
            if remaining < 10:
                break
            t = np.linspace(0, 2, n_best)
            h_r = best_h + rng.randn(n_best) * 0.03 * (r + 1) / 3.0
            h_r = project_feasible(h_r, n_best)
            sub_t = min(20, remaining - 2)
            if sub_t > 3:
                h_r, c5_r = subgradient_optimize(h_r, n_best, sub_t, rng, start_time)
            else:
                c5_r = compute_c5(h_r)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
    
    # Final local polish
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 2
    if remaining > 5:
        h_f, c5_f = local_search(best_h, len(best_h), remaining, rng, start_time)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
