    return best_h, best_c5

def coordinate_optimize(h, n, max_time, rng, start_time):
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    order = list(range(n))
    no_improve = 0
    
    while time.time() - start_time < max_time and no_improve < n * 5:
        rng.shuffle(order)
        improved = False
        for j in order:
            if time.time() - start_time > max_time:
                break
            # Try small perturbations of h[j]
            best_val = h[j]
            best_local = best_c5
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                h_new = h.copy()
                h_new[j] = np.clip(h_new[j] + delta, 0, 1)
                h_new = project_box_sum(h_new, target_sum)
                c5_new = compute_c5(h_new, n)
                if c5_new < best_local:
                    best_local = c5_new
                    best_val = h_new.copy()
            if best_local < best_c5 - 1e-10:
                h = best_val
                best_h = h.copy()
                best_c5 = best_local
                improved = True
                no_improve = 0
        if not improved:
            no_improve += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    results = []
    
    # Phase 1: Optimize at initial resolution (50% budget)
    n1 = len(initial_h_values)
    h1 = initial_h_values.copy()
    t1_budget = budget_s * 0.45
    
    h_opt1, c5_opt1 = subgradient_optimize(h1, n1, t1_budget, rng, start_time)
    results.append((h_opt1, c5_opt1, n1))
    
    # Also try from a random 0-1 start at same resolution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 20
    if remaining > 60:
        h_rand = random_zero_one(n1, rng)
        h_opt1b, c5_opt1b = subgradient_optimize(h_rand, n1, remaining * 0.5, rng, start_time)
        results.append((h_opt1b, c5_opt1b, n1))
    
    # Phase 2: Try n=90 (20% budget)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 20
    if remaining > 80:
        n2 = 90
        # Upsample best solution
        best_so_far = min(results, key=lambda x: x[1])
        h2 = upsample(best_so_far[0], best_so_far[2], n2)
        t2 = remaining * 0.6
        h_opt2, c5_opt2 = subgradient_optimize(h2, n2, t2, rng, start_time)
        results.append((h_opt2, c5_opt2, n2))
        
        # Also try random 0-1 at n=90
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 20
        if remaining > 60:
            h2_rand = random_zero_one(n2, rng)
            h_opt2b, c5_opt2b = subgradient_optimize(h2_rand, n2, remaining * 0.7, rng, start_time)
            results.append((h_opt2b, c5_opt2b, n2))
    
    # Phase 3: Try n=180 (remaining budget)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 15
    if remaining > 60:
        n3 = 180
        best_so_far = min(results, key=lambda x: x[1])
        h3 = upsample(best_so_far[0], best_so_far[2], n3)
        t3 = remaining * 0.8
        h_opt3, c5_opt3 = subgradient_optimize(h3, n3, t3, rng, start_time)
        results.append((h_opt3, c5_opt3, n3))
    
    # Final coordinate refinement on best result
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 5
    best_sol = min(results, key=lambda x: x[1])
    if remaining > 30:
        h_fin, c5_fin = coordinate_optimize(best_sol[0], best_sol[2], remaining, rng, start_time)
        results.append((h_fin, c5_fin, best_sol[2]))
    
    # Return best
    best_sol = min(results, key=lambda x: x[1])
    h_best, c5_best, n_best = best_sol
    
    # Validate
    h_best = np.clip(h_best, 0.0, 1.0)
    # Ensure sum constraint
    target = n_best / 2.0
    if abs(h_best.sum() - target) > 1e-6:
        h_best = project_box_sum(h_best, target)
    
    return h_best, c5_best, n_best
