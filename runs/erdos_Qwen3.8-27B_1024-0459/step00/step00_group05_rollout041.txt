import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo = -1.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    h_proj = np.clip(h - t, 0.0, 1.0)
    return h_proj

def subgradient(h, n, top_k=5):
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr_raw) * dx)
    m = len(corr_raw)
    if top_k >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(corr_raw, -top_k)[-top_k:]
    grad = np.zeros(n)
    for k_idx in idx:
        lag = int(k_idx) - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] += -h[:n - lag]
        else:
            L = -lag
            grad[L:] += (1.0 - h[:n - L])
            grad[:n - L] += -h[L:]
    grad /= float(len(idx))
    return grad * dx, c5

def upsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_to_feasible(h_new, n_new)

def binary_round(h, n):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def make_sawtooth(n, period):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.where((x % period) < period * 0.5, 1.0, 0.0)
    return project_to_feasible(h, n)

def optimize_h(h_init, n, deadline, c5_lb=0.30, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    T = 0.004
    T_min = 1e-7
    cooling = 0.9997
    no_improve = 0
    patience = 200
    top_k = 5
    step_scale = 1.0
    while time.time() < deadline:
        grad, c5_curr = subgradient(h, n, top_k=top_k)
        gnorm_sq = float(np.sum(grad * grad))
        if gnorm_sq > 1e-15:
            alpha = (c5_curr - c5_lb) * step_scale / gnorm_sq
            alpha = min(alpha, 0.3)
            alpha = max(alpha, 1e-5)
        else:
            alpha = 0.01
        h_cand = project_to_feasible(h - alpha * grad, n)
        c5_cand = compute_c5(h_cand, n)
        if c5_cand < best_c5:
            h = h_cand
            best_c5 = c5_cand
            best_h = h.copy()
            no_improve = 0
            step_scale = min(3.0, step_scale * 1.01)
        else:
            delta = c5_cand - best_c5
            if rng.random() < np.exp(-delta / max(T, T_min)):
                h = h_cand
            no_improve += 1
            step_scale = max(0.1, step_scale * 0.98)
        T = max(T_min, T * cooling)
        if no_improve > patience:
            r = rng.random()
            if r < 0.3:
                scale = 0.03 + 0.02 * rng.random()
                h = project_to_feasible(h + scale * rng.randn(n), n)
            elif r < 0.5:
                h = binary_round(h + 0.1 * rng.randn(n), n)
            elif r < 0.7:
                h = make_sawtooth(n, 2.0 * rng.randint(2, 7) / rng.randint(2, 6))
            else:
                h = project_to_feasible(h + 0.05 * rng.randn(n), n)
            no_improve = 0
            T = max(T, 0.003)
            top_k = max(1, min(15, top_k + rng.randint(-3, 4)))
            step_scale = 1.0
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        n_init = len(h_parent)
    else:
        n_init = 70
        h_parent = make_sawtooth(n_init, 1.0)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h, n_init)
    phases = [
        (n_init, 0.30),
        (n_init * 2, 0.30),
        (n_init * 4, 0.25),
        (n_init * 8, 0.13),
    ]
    current_n = n_init
    current_h = best_h.copy()
    elapsed_alloc = 0.0
    for pi, (phase_n, frac) in enumerate(phases):
        phase_deadline = start_time + elapsed_alloc + frac * budget_s
        if time.time() >= phase_deadline - 5:
            break
        if phase_n != current_n:
            current_h = upsample(current_h, current_n, phase_n)
            current_n = phase_n
        # Try multiple starts at this resolution
        starts = [current_h]
        if phase_n == n_init:
            starts.append(make_sawtooth(n_init, 0.5))
            starts.append(make_sawtooth(n_init, 2.0 / 3.0))
            starts.append(binary_round(np.full(n_init, 0.5) + 0.3 * np.random.RandomState(seed).randn(n_init), n_init))
        phase_time = phase_deadline - time.time()
        n_starts = len(starts)
        per_start_time = max(10, phase_time / n_starts)
        for si, s in enumerate(starts):
            sd = time.time() + per_start_time
            if sd > phase_deadline:
                sd = phase_deadline
            if time.time() >= sd:
                break
            h_try, c5_try = optimize_h(s, phase_n, sd, c5_lb=0.30, seed=seed + pi * 10 + si)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                current_h = h_try
        elapsed_alloc = time.time() - start_time
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        h_new, c5_new = optimize_h(current_h, current_n, start_time + budget_s - 3, c5_lb=0.30, seed=seed + 100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    final_n = len(best_h)
    best_h = project_to_feasible(best_h, final_n)
    final_c5 = compute_c5(best_h, final_n)
    # Safety: never return worse than parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent_c5 = compute_c5(h_parent, len(h_parent))
        if final_c5 > parent_c5:
            best_h = h_parent.copy()
            final_c5 = parent_c5
            final_n = len(h_parent)
    return best_h, final_c5, final_n
