import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (np.max(corr) * dx, corr * dx)

def upscale_to_resolution(h_parent, n_target):
    n_parent = len(h_parent)
    x_old = np.arange(n_parent) / n_parent * 2.0
    x_new = np.arange(n_target) / n_target * 2.0
    h_new = np.interp(x_new, x_old, h_parent)
    # Ensure integral constraint
    h_new = np.clip(h_new, 0, 1)
    current_sum = h_new.sum()
    target_sum = n_target / 2.0
    if current_sum > 0:
        # Scale and project
        h_new = h_new * (target_sum / current_sum)
        h_new = np.clip(h_new, 0, 1)
        # Fix sum by adjusting
        diff = target_sum - h_new.sum()
        if diff > 0:
            # Add to entries with most room
            headroom = 1.0 - h_new
            order = np.argsort(-headroom)
            i = 0
            while diff > 1e-10 and i < n_target:
                add = min(headroom[order[i]], diff)
                h_new[order[i]] += add
                headroom[order[i]] -= add
                diff -= add
                i += 1
        else:
            # Remove from entries with most value
            order = np.argsort(-h_new)
            i = 0
            while diff < -1e-10 and i < n_target:
                rem = min(h_new[order[i]], -diff)
                h_new[order[i]] -= rem
                diff += rem
                i += 1
    return h_new

def binary_local_search(h, n, max_iters=5000, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    c5, _ = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    for it in range(max_iters):
        # Pick a random 1 to turn off and random 0 to turn on
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        
        h[i] = 0
        h[j] = 1
        new_c5, _ = compute_c5(h)
        
        if new_c5 <= best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

            best_c5 = c5

    # Phase 2: Continuous optimization + rebinarize
    if time_left() > 60:
        # Try continuous optimization
        from scipy.optimize import minimize
        
        target_sum = n_target / 2.0
        bounds = [(0, 1)] * n_target
        
        def smooth_obj(x):
            c5_val, corr = compute_c5(x)
            # Add penalty for non-binary
            binary_penalty = 10 * np.sum(x * (1 - x))
            # Add constraint penalty
            sum_penalty = 100 * (np.sum(x) - target_sum) ** 2
            return c5_val + binary_penalty + sum_penalty
        
        res = minimize(smooth_obj, h, method='Nelder-Mead',

    import time
    
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    n = 160

    dx = 2.0 / n  # 0.0125

        h_p = np.array(initial_h_values, dtype=float)
        # Upscale to n
        x_old = np.linspace(0, 2, len(h_p))
        x_new = np.linspace(0, 2, n)
        h_cont = np.interp(x_new, x_old, h_p)
        # Binarize
        h = np.zeros(n)
        idx = np.argsort(h_cont)[::-1]
        h[idx[:target_sum]] = 1.0
    else:
        # Random initialization
        idx = rng.permutation(n)
        h = np.zeros(n)
        h[idx[:target_sum]] = 1.0
    
    # Initial evaluation
    def eval_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx, corr
    
    c5, corr = eval_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    # Simulated annealing
    T = 5.0  # initial temperature (in correlation units)
    cooling = 0.9995
    ones_set = set(np.where(h == 1)[0].tolist())
    
    while time.time() - start_time < budget_s - 30:
        # Pick a random 1 and a random 0
        ones_list = list(ones_set)
        i = ones_list[rng.randint(len(ones_list))]
        zeros_list = [j for j in range(n) if j not in ones_set]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        # Swap
        h[i] = 0
        h[j] = 1
        ones_set.remove(i)
        ones_set.add(j)
        
        new_c5, _ = eval_c5(h)
        delta = new_c5 - c5  # in C5 units
        
        if delta < 0 or rng.random() < np.exp(-delta / (T * dx)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

            # Revert
            h[i] = 1
            h[j] = 0
            ones_set.add(i)
            ones_set.remove(j)

import numpy as np
import time
from scipy.optimize import minimize

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def _upscale(h_parent, n_target):
    n_p = len(h_parent)
    x_old = np.linspace(0.0, 1.0, n_p)
    x_new = np.linspace(0.0, 1.0, n_target)
    h_new = np.interp(x_new, x_old, h_parent)
    return h_new

def _binarize(h_cont, n):
    h = np.zeros(n)
    idx = np.argsort(h_cont)[::-1]
    h[idx[:n//2]] = 1.0
    return h

def _sa_optimize(h_start, n, rng, max_time, T_init=5.0, cooling_factor=0.9999):
    """Simulated annealing with pair swaps."""
    target_sum = n // 2
    dx = 2.0 / n
    h = h_start.copy()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    best_c5 = c5
    best_h = h.copy()
    
    ones_list = list(np.where(h == 1)[0])
    T = T_init
    no_improve = 0
    patience = 30000
    
    start = time.time()
    eval_count = 0
    
    while time.time() - start < max_time:
        eval_count += 1
        if eval_count % 10000 == 0 and time.time() - start >= max_time:
            break
        
        if no_improve > patience:
            # Restart with random perturbation
            n_pert = rng.randint(3, 10)
            for _ in range(n_pert):
                ii = rng.randint(n)
                if h[ii] == 1:
                    jj = rng.randint(n)
                    while h[jj] == 1:
                        jj = rng.randint(n)
                    h[ii], h[jj] = h[jj], h[ii]
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr) * dx
            T = T_init * 0.5
            no_improve = 0
            ones_list = list(np.where(h == 1)[0])
        
        # Pick swap
        i = ones_list[rng.randint(len(ones_list))]
        attempts = 0
        while attempts < 20:
            j = rng.randint(n)
            if h[j] == 0:
                break
            attempts += 1
        else:
            continue
        
        # Perform swap
        h[i] = 0
        h[j] = 1
        
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = np.max(new_corr)
        delta_corr = new_max - np.max(corr)
        
        if delta_corr < 0 or np.log(rng.random()) < -delta_corr / T:
            c5 = new_max * dx
            corr = new_corr
            # Update ones_list
            ones_list[ones_list.index(i)] = j
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1
            h[j] = 0
        
        T *= cooling_factor
    
    return best_h, best_c5

def _local_search(h, n, max_iters=20000, rng=None):
    """Greedy local search: try all single swaps, keep best."""
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    best_c5 = c5
    best_h = h.copy()
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = [i for i in range(n) if h[i] == 0]
    
    for it in range(max_iters):
        improved = False
        # Try a random subset of swaps
        n_tries = min(len(ones_list) * 5, 500)
        tried = set()
        for _ in range(n_tries):
            i = rng.randint(len(ones_list))
            j = rng.randint(len(zeros_list))
            key = (ones_list[i], zeros_list[j])
            if key in tried:
                continue
            tried.add(key)
            
            a = ones_list[i]
            b = zeros_list[j]
            h[a] = 0
            h[b] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = np.max(new_corr) * dx
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                ones_list[i] = b
                zeros_list[j] = a
                improved = True
            else:
                h[a] = 1
                h[b] = 0
        
        if not improved:
            break
    
    return best_h, best_c5

    start_time = time.time()

        return budget_s - (time.time() - t0)

    h_parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)

    # Build initial candidates at different resolutions
    candidates = []
    for n in [160, 200, 240]:
        if h_parent is not None:
            h_up = _upscale(h_parent, n)
            candidates.append((_binarize(h_up, n), n, 'parent'))
        # Structured random: spread out
        h_struct = np.zeros(n)
        block_size = 1 + (n // 2) // 80
        positions = list(range(0, n, block_size))[:n // 2]
        h_struct[positions] = 1.0
        if h_struct.sum() != n // 2:
            h_struct = _binarize(h_struct + rng.random(n) * 0.1, n)
        candidates.append((h_struct, n, 'struct'))
        # Pure random
        h_rand = np.zeros(n)
        perm = rng.permutation(n)
        h_rand[perm[:n // 2]] = 1.0
        candidates.append((h_rand, n, 'rand'))

    best_h = np.zeros(160)
    best_h[rng.permutation(160)[:80]] = 1.0
    best_c5, _ = _eval_c5(best_h)
    best_n = 160

    # Phase 1: SA on each candidate
    for h_init, n, tag in candidates:
        if remaining() < 70:
            break
        alloc = min(remaining() - 30, 180)
        r_sub = np.random.RandomState(int(rng.randint(2**31)))
        h_opt, c5_opt = _sa_optimize(h_init, n, r_sub, alloc, T_init=12.0)
        if remaining() > 25:
            h_opt, c5_opt = _greedy_improve(h_opt, n, r_sub, max_iters=40000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

    # Phase 2: Multi-swap SA on best
    while remaining() > 60:
        alloc = min(remaining() - 20, 150)
        r_sub = np.random.RandomState(int(rng.randint(2**31)))
        h_opt, c5_opt = _multi_swap_sa(best_h, best_n, r_sub, alloc, T_init=12.0)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if remaining() > 20:
            h_opt, c5_opt = _greedy_improve(best_h, best_n,
                                            np.random.RandomState(7), max_iters=30000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Final single-swap polish
    while remaining() > 15:
        alloc = min(remaining() - 5, 80)
        r_sub = np.random.RandomState(int(rng.randint(2**31)))
        h_opt, c5_opt = _sa_optimize(best_h, best_n, r_sub, alloc, T_init=3.0, cool_rate=0.9999)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_opt, c5_opt = _greedy_improve(best_h, best_n,
                                        np.random.RandomState(42), max_iters=50000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Ensure validity
    n = best_n
    h_final = np.clip(best_h, 0.0, 1.0)
    target_sum = n / 2.0
    if abs(h_final.sum() - target_sum) > 0.05:
        h_final = _binarize(h_final, n)

    c5_final, _ = _eval_c5(h_final)
    return h_final, c5_final, n
