def active_gradient(h, lags, weights):

    for idx in range(len(lags)):

import time

def c5_value(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx


def c5_corr(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.array([0.0]), 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(corr)) * dx
    return corr, c


def exact_sum_adjust(h, target):
    h = np.asarray(h, dtype=float).copy()

        return h
    target = float(target)
    for _ in range(16):
        s = float(np.sum(h))
        r = float(target - s)
        if r == 0.0 or abs(r) < 1e-15:
            break

        return np.ones(n, dtype=float)
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)

    h = np.clip(y, 0.0, 1.0)
    s = float(np.sum(h))
    if abs(s - target) <= 1e-11:
        return exact_sum_adjust(h, target)

    pad = max(2.0, float(np.max(np.abs(y))) + 1.0)
    lo = float(np.min(y)) - pad
    hi = float(np.max(y)) + pad

    for _ in range(48):
        mid = 0.5 * (lo + hi)
        sm = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if sm > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    return exact_sum_adjust(h, target)


def active_gradient(h, lags, weights):
    h = np.asarray(h, dtype=float)
    n = h.size
    G = np.zeros(n, dtype=float)
    for p in range(len(lags)):
        w = float(weights[p])
        if w == 0.0:
            continue
        k = int(lags[p])
        if k == 0:
            G += w * (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                G[:-k] += w * (1.0 - h[k:])
                G[k:] += -w * h[:-k]
        else:
            p2 = -k
            if p2 < n:
                G[p2:] += w * (1.0 - h[:-p2])
                G[:-p2] += -w * h[p2:]
    return G


def top_indices(corr, k):
    corr = np.asarray(corr, dtype=float)
    m = corr.size
    if m == 0:
        return np.array([], dtype=int)
    k = int(k)
    if k <= 0:
        return np.array([], dtype=int)
    if k >= m:
        return np.argsort(corr)[::-1]
    idx = np.argpartition(corr, -k)[-k:]
    order = np.argsort(corr[idx])[::-1]
    return idx[order]


def update_best(state, h):
    if h is None:
        return state
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    if n == 0:
        return state
    target = 0.5 * float(n)
    h = project_box_sum(h, target)
    c = c5_value(h)
    if c < state[1] - 1e-15:
        state[0] = h
        state[1] = float(c)
        state[2] = int(n)
    return state


def remaining_time(start, budget, reserve):
    elapsed = time.perf_counter() - start
    return max(0.0, float(budget) - elapsed - float(reserve))


def random_perturbation(h, rng, sigma=0.05):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return h.copy()
    noise = rng.normal(0.0, float(sigma), size=n)
    noise -= float(np.mean(noise))
    return project_box_sum(h + noise, 0.5 * n)


def stochastic_pair_move(h, rng, target, moves=1, strength=0.1):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    if n < 2:
        return project_box_sum(h, target)
    for _ in range(int(moves)):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        oi = float(h[i])
        oj = float(h[j])
        lo = max(-oi, oj - 1.0)
        hi = min(1.0 - oi, oj)
        if hi - lo <= 1e-14:
            continue
        if rng.random() < 0.5:
            delta = float(rng.uniform(lo, hi))
        else:
            delta = float(rng.uniform(-float(strength), float(strength)))
            if delta < lo:
                delta = float(lo)
            elif delta > hi:
                delta = float(hi)
        h[i] = oi + delta
        h[j] = oj - delta
    return project_box_sum(h, target)


def binarize_h(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return h.copy()
    target = 0.5 * float(n)
    h = project_box_sum(h.copy(), target)
    out = np.zeros(n, dtype=float)
    order = np.argsort(h)[::-1]
    k = int(math.floor(target))
    if abs(target - float(k)) <= 1e-11:
        if k > 0:
            out[order[:k]] = 1.0
    else:
        if k > 0:
            out[order[:k]] = 1.0
        if k < n:
            out[order[k]] = target - float(k)
    return project_box_sum(out, target)


def thue_morse_h(n):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    bits = np.zeros(n, dtype=float)
    for i in range(n):
        v = i
        p = 0
        while v:
            p ^= (v & 1)
            v >>= 1
        bits[i] = float(p)
    h1 = project_box_sum(bits, 0.5 * n)
    h2 = project_box_sum(1.0 - bits, 0.5 * n)
    if c5_value(h2) < c5_value(h1):
        return h2
    return h1


def random_h(n, seed):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    rng = np.random.default_rng(int(seed))
    h = rng.random(size=n)
    p = float(rng.uniform(0.3, 4.0))
    h = np.power(h, p)
    if rng.random() < 0.5:
        x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
        amp = float(rng.uniform(0.0, 0.4))
        freq = int(rng.integers(1, 6))
        ph = float(rng.uniform(-math.pi, math.pi))
        h = h + amp * np.sin(freq * math.pi * x + ph)
    return project_box_sum(h, 0.5 * n)


def periodic_binary_h(n, seed, period=None):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    rng = np.random.default_rng(int(seed))
    if period is None or int(period) < 2:
        period = max(2, int(n // 4))
    period = int(period)
    half = min(period // 2, max(1, period // 2))
    pat = np.zeros(period, dtype=float)
    if half > 0:
        idx = rng.choice(period, size=half, replace=False)
        pat[idx] = 1.0
    reps = n // period + 1
    h = np.tile(pat, reps)[:n]
    h1 = project_box_sum(h, 0.5 * n)
    h2 = project_box_sum(1.0 - h, 0.5 * n)
    if c5_value(h2) < c5_value(h1):
        return h2
    return h1


def fourier_random_h(n, seed, m=4):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    rng = np.random.default_rng(int(seed))
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)
    for f in range(1, int(m) + 1):
        amp = float(rng.uniform(0.0, 0.45 / math.sqrt(float(f))))
        ph = float(rng.uniform(-math.pi, math.pi))
        h += amp * np.sin(float(f) * math.pi * x + ph)
    return project_box_sum(h, 0.5 * n)


def resample_h(h, n):
    h = np.asarray(h, dtype=float)
    m = h.size
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    if m == 0:
        return np.full(n, 0.5, dtype=float)
    if m == n:
        return project_box_sum(h.copy(), 0.5 * n)
    x_old = (np.arange(m, dtype=float) + 0.5) * (2.0 / float(m))
    x_new = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    h_new = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return project_box_sum(h_new, 0.5 * n)


def binary_local_search(h, seed, time_limit):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1 or time_limit <= 1e-6:
        return project_box_sum(h.copy(), 0.5 * max(1, n))
    target = 0.5 * float(n)
    cur = binarize_h(h)
    c = c5_value(cur)
    best = cur.copy()
    best_c = c
    rng = np.random.default_rng(int(seed))
    deadline = time.perf_counter() + max(0.0, float(time_limit))
    temp = 0.003
    no_improve = 0
    low_temp_fails = 0

    while time.perf_counter() < deadline:
        ones = np.flatnonzero(cur > 0.5)
        zeros = np.flatnonzero(cur < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        ii = int(rng.integers(0, ones.size))
        jj = int(rng.integers(0, zeros.size))
        i = int(ones[ii])
        j = int(zeros[jj])
        oi = float(cur[i])
        oj = float(cur[j])
        cur[i] = oj
        cur[j] = oi
        cnew = c5_value(cur)
        accepted = False
        if cnew < c - 1e-14:
            accepted = True
        else:
            delta = cnew - c
            if delta < 1e-12:
                accepted = True
            elif delta < 0.1:
                if rng.random() < math.exp(-delta / max(1e-12, temp)):
                    accepted = True
        if accepted:
            c = cnew
            no_improve = 0
            if c < best_c - 1e-14:
                best_c = c
                best = cur.copy()
        else:
            cur[i] = oi
            cur[j] = oj
            no_improve += 1

        if no_improve > 800:
            temp = max(1e-5, temp * 0.85)
            no_improve = 0
            if temp <= 1.0001e-5:
                low_temp_fails += 1
                if low_temp_fails >= 4:
                    break
    return project_box_sum(best, target)


def pair_hill_climb(h, seed, time_limit):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n < 2 or time_limit <= 1e-6:
        return project_box_sum(h.copy(), 0.5 * max(1, n))
    target = 0.5 * float(n)
    cur = project_box_sum(h.copy(), target)
    c = c5_value(cur)
    best = cur.copy()
    best_c = c
    rng = np.random.default_rng(int(seed))
    deadline = time.perf_counter() + max(0.0, float(time_limit))
    no_improve = 0
    max_no = 20000

    while time.perf_counter() < deadline and no_improve < max_no:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        oi = float(cur[i])
        oj = float(cur[j])
        if abs(oi - oj) < 1e-12:
            no_improve += 1
            continue

        accepted = False

        cur[i] = oj
        cur[j] = oi
        c1 = c5_value(cur)
        if c1 < c - 1e-13:
            c = c1
            accepted = True
        else:
            cur[i] = oi
            cur[j] = oj
            avg = 0.5 * (oi + oj)
            cur[i] = avg
            cur[j] = avg
            c2 = c5_value(cur)
            if c2 < c - 1e-13:
                c = c2
                accepted = True
            else:
                cur[i] = oi
                cur[j] = oj
                delta = -0.25 * (oi - oj)
                lo = max(-oi, oj - 1.0)
                hi = min(1.0 - oi, oj)
                if lo <= delta <= hi:
                    cur[i] = oi + delta
                    cur[j] = oj - delta
                    c3 = c5_value(cur)
                    if c3 < c - 1e-13:
                        c = c3
                        accepted = True
                if not accepted:
                    cur[i] = oi
                    cur[j] = oj
                    delta = float(rng.uniform(-0.3, 0.3))
                    if lo <= delta <= hi:
                        cur[i] = oi + delta
                        cur[j] = oj - delta
                        c4 = c5_value(cur)
                        if c4 < c - 1e-13:
                            c = c4
                            accepted = True

        if accepted:
            no_improve = 0
            if c < best_c - 1e-13:
                best_c = c
                best = cur.copy()
        else:
            cur[i] = oi
            cur[j] = oj
            no_improve += 1

    return project_box_sum(best, target)


def optimize_h(h, seed, time_limit):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return h.copy()
    target = 0.5 * float(n)
    if n <= 1:
        return project_box_sum(h.copy(), target)

    cur = project_box_sum(h.copy(), target)
    cur_corr, cur_c = c5_corr(cur)
    best = cur.copy()
    best_c = cur_c

    deadline = time.perf_counter() + max(0.0, float(time_limit))
    rng = np.random.default_rng(int(seed))
    step = 0.10
    topq = 8
    fails = 0
    iteration = 0
    m_corr = 2 * n - 1

    while time.perf_counter() < deadline:
        if cur_c < best_c - 1e-15:
            best_c = cur_c
            best = cur.copy()

        maxc = float(np.max(cur_corr))
        if maxc <= 1e-14:
            break

        if iteration % 5 == 0 or iteration % 5 == 1:
            q_use = 1
        elif iteration % 5 == 2:
            q_use = 3
        elif iteration % 5 == 3:
            q_use = topq
        else:
            q_use = max(5, topq)
        q_use = max(1, min(q_use, m_corr))

        idx = top_indices(cur_corr, q_use)
        if idx.size == 0:
            break
        lags = idx - (n - 1)

        if idx.size > 1:
            vals = cur_corr[idx]
            temp = max(1e-8, 0.03 * maxc + 1e-6)
            w = np.exp((vals - maxc) / temp)
            if rng.random() < 0.25:
                w = w * (rng.random(idx.size) > 0.35)
            ws = float(np.sum(w))
            if ws <= 1e-12:
                w = np.ones(idx.size, dtype=float) / float(idx.size)
            else:
                w = w / ws
        else:
            w = np.ones(1, dtype=float)

        G = active_gradient(cur, lags, w)
        G -= float(np.mean(G))
        scale = float(np.max(np.abs(G))) if G.size else 0.0

        if scale < 1e-12:
            idx2 = top_indices(cur_corr, min(3, m_corr))
            if idx2.size:
                G = active_gradient(cur, idx2 - (n - 1), np.ones(idx2.size, dtype=float) / float(idx2.size))
                G -= float(np.mean(G))
                scale = float(np.max(np.abs(G))) if G.size else 0.0

        if scale < 1e-12:
            cur = random_perturbation(best, rng, 0.05)
            cur_corr, cur_c = c5_corr(cur)
            fails += 1
            iteration += 1
            continue

        direction = -G / scale
        alphas = [step, 0.5 * step, 0.25 * step, 0.125 * step, 0.0625 * step, 0.03125 * step]
        accepted = False

        for a in alphas:
            if time.perf_counter() >= deadline:
                break
            y = cur + a * direction
            cand = project_box_sum(y, target)
            cand_corr, cand_c = c5_corr(cand)
            if cand_c < cur_c - 1e-13:
                cur = cand
                cur_corr = cand_corr
                cur_c = cand_c
                if cur_c < best_c - 1e-15:
                    best_c = cur_c
                    best = cur.copy()
                step = min(0.35, step * 1.15)
                fails = 0
                accepted = True
                break

        if not accepted:
            fails += 1
            step = max(0.005, step * 0.8)
            topq = min(40, topq + 1)
            if fails > 12:
                moves = int(rng.integers(1, 4))
                cur = stochastic_pair_move(cur, rng, target, moves=moves, strength=max(0.02, step * 2.0))
                cur_corr, cur_c = c5_corr(cur)
                fails = 0
                step = 0.08
                if cur_c > best_c + 0.02 and rng.random() < 0.7:
                    cur = random_perturbation(best, rng, 0.06)
                    cur_corr, cur_c = c5_corr(cur)
            elif fails > 50:
                cur = random_perturbation(best, rng, 0.10)
                cur_corr, cur_c = c5_corr(cur)
                fails = 0
                step = 0.10

        iteration += 1

    return project_box_sum(best, target)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    start = time.perf_counter()
    budget = float(budget_s)
    state = [None, float("inf"), 0]

    try:
        parent = None
        try:
            parent = initial_h_values
        except Exception:
            parent = None

        if parent is not None:
            state = update_best(state, parent)

        if state[0] is None:
            state = update_best(state, thue_morse_h(64))

        specs = []
        if parent is not None:
            ph = np.asarray(parent, dtype=float).copy()
            pn = int(ph.size)
            if pn > 0:
                specs.append((ph, 0.30, seed + 1, "cont"))
                n2 = max(2, min(int(2 * pn), 128))
                specs.append((resample_h(ph, n2), 0.16, seed + 2, "cont"))
                n4 = max(4, min(int(4 * pn), 196))
                specs.append((resample_h(ph, n4), 0.10, seed + 3, "cont"))
                specs.append((binarize_h(ph), 0.08, seed + 4, "binary"))
                for nn in (48,):
                    if nn != pn:
                        specs.append((resample_h(ph, nn), 0.05, seed + 40 + nn, "cont"))

        if state[0] is not None and not specs:
            specs.append((state[0].copy(), 0.25, seed, "cont"))

        specs.append((thue_morse_h(64), 0.08, seed + 10, "binary"))
        specs.append((periodic_binary_h(96, seed + 20, 32), 0.07, seed + 21, "binary"))
        specs.append((random_h(64, seed + 30), 0.05, seed + 31, "cont"))
        specs.append((fourier_random_h(64, seed + 40, 5), 0.05, seed + 41, "cont"))
        specs.append((thue_morse_h(128), 0.06, seed + 50, "binary"))
        specs.append((periodic_binary_h(128, seed + 60, 64), 0.05, seed + 61, "binary"))
        specs.append((random_h(47, seed + 70), 0.04, seed + 71, "cont"))

        for h0, frac, sd, kind in specs:
            rem = remaining_time(start, budget, 30.0)
            if rem < 3.0:
                break
            t = float(frac) * budget
            t = max(0.0, min(t, rem - 3.0))
            if t < 2.0:
                break

            h0 = np.asarray(h0, dtype=float).copy()
            h0 = project_box_sum(h0, 0.5 * float(h0.size))
            t0 = time.perf_counter()

            if kind == "binary":
                t_bin = max(0.0, min(0.5 * t, rem - 5.0))
                if t_bin > 0.5:
                    h1 = binary_local_search(h0, sd, t_bin)
                    state = update_best(state, h1)
                else:
                    h1 = h0

                rem1 = remaining_time(start, budget, 25.0)
                spent = time.perf_counter() - t0
                t_cont = max(0.0, min(t - spent, rem1))
                if t_cont > 0.5:
                    h2 = optimize_h(h1, sd + 1, t_cont)
                    state = update_best(state, h2)
            else:
                h1 = optimize_h(h0, sd, t)
                state = update_best(state, h1)

        if state[0] is not None:
            rem = remaining_time(start, budget, 8.0)
            if rem > 4.0:
                t_peak = max(0.0, min(rem, 0.20 * budget, 250.0))
                if t_peak > 1.0:
                    h2 = pair_hill_climb(state[0], seed + 555, t_peak)
                    state = update_best(state, h2)

            rem = remaining_time(start, budget, 5.0)
            if rem > 4.0:
                t_opt = max(0.0, min(rem, 0.35 * budget, 350.0))
                if t_opt > 1.0:
                    h3 = optimize_h(state[0], seed + 666, t_opt)
                    state = update_best(state, h3)

    except Exception:
        pass

    if state[0] is None:
        state = update_best(state, np.full(64, 0.5, dtype=float))

    h_final = project_box_sum(np.asarray(state[0], dtype=float).copy(), 0.5 * float(state[2]))
    c_final = c5_value(h_final)
    if c_final > state[1] and c_final < state[1] + 1e-8:
        c_final = state[1]

    return h_final, float(c_final), int(state[2])
