def compute_c5(h, dx):
    return float(np.max(compute_c5_values(h, dx)))

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_func(h, n, dx, alpha, penalty):
    vals = compute_c5_values(h, dx)
    m = np.max(vals)
    sm = m + np.log(np.sum(np.exp(alpha * (vals - m)))) / alpha
    pen = penalty * (h.sum() - n / 2.0) ** 2
    return sm + pen

def interp_to_n(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def optimize_lbfgsb(h0, n, dx, time_limit, start_time, rng):
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    alpha = 40.0
    penalty = 500.0
    bounds = [(0.0, 1.0)] * n
    n_restarts = 5
    
    for r in range(n_restarts):
        elapsed = time.time() - start_time
        if elapsed > time_limit or time_limit - elapsed < 3:
            break
        if r == 0:
            x0 = best_h.copy()
        else:
            scale = 0.05 / max(1.0, np.sqrt(n) / 7.0)
            x0 = best_h + scale * rng.randn(n)
            x0 = project_feasible(x0, n)
        
        res = minimize(obj_func, x0, args=(n, dx, alpha, penalty),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 300, 'maxfun': 5000, 'ftol': 1e-14})
        
        h_cand = project_feasible(res.x, n)
        c5_cand = compute_c5(h_cand, dx)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
    
    return best_h, best_c5

def local_search_binary(h, n, dx, n_rounds=300, n_tries_per_round=30):
    """Local search for even-n 0-1 solutions"""
    if n % 2 != 0:
        return h, compute_c5(h, dx)
    n_ones = n // 2
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:n_ones]] = 1.0
    c5 = compute_c5(h_bin, dx)
    
    for rnd in range(n_rounds):
        improved = False
        ones = list(np.where(h_bin == 1.0)[0])
        zeros = list(np.where(h_bin == 0.0)[0])
        for _ in range(n_tries_per_round):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_new = compute_c5(h_bin, dx)
            if c5_new < c5 - 1e-15:
                c5 = c5_new
                improved = True
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            break
    
    return h_bin, c5

Actually, wait. Let me reconsider the problem. The objective is:
max_k sum_i h[i] * (1 - h[i+k-m]) * dx

where m = n-1. This is a maximum of quadratic (bilinear) functions. The feasible set is a polytope (box ∩ hyperplane). The objective is not convex (it's the max of non-convex functions... well, each individual C_k is a bilinear function which is not convex or concave in general).

Actually, C_k = sum_i h[i] - sum_i h[i]h[i+m] (for valid i). The first term is linear, the second is a quadratic form with a specific structure. The Hessian of C_k w.r.t. h is:
-2 * (matrix with 1s at positions (i, i+m) and (i+m, i) for valid pairs)

This is indefinite (has both positive and negative eigenvalues), so C_k is not convex or concave. The max of these is also not convex.

So this is a genuinely non-convex problem. L-BFGS-B with smooth max is a reasonable heuristic.

def pair_swap_search(h, n, dx, n_steps=5000, rng=None):
    """Local search via pair mass transfers"""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    c5 = compute_c5(h, dx)
    
    for step in range(n_steps):
        # Pick two random indices
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try transferring from larger to smaller
        if h[i] < h[j]:
            i, j = j, i
        # Transfer amount
        eps = min(h[i], 1.0 - h[j]) * 0.5
        if eps < 1e-6:
            continue
        h[i] -= eps
        h[j] += eps
        c5_new = compute_c5(h, dx)
        if c5_new < c5:
            c5 = c5_new
        else:
            h[i] += eps
            h[j] -= eps
    
    return h, c5
