    """Compute C5 (soft or exact max) and its gradient. Vectorized."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    num_lags = len(corr)  # 2n-1
    
    if eps > 1e-12:
        scaled = corr / eps
        m = np.max(scaled)
        exps = np.exp(scaled - m)

        mask = np.isclose(corr, max_val, atol=1e-12)

        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            room = h.copy()
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
    h = np.clip(h, 0.0, 1.0)
    return h

def optimize_at_resolution(h0, n, eps_start=0.05, eps_end=0.001, max_iter=300, time_limit=180):
    """Optimize h at given resolution using SLSQP with soft-max."""
    dx = 2.0 / n
    target = n / 2.0
    h0 = project_to_constraint(h0, n)
    
    def objective(x):
        val, _ = c5_gradient(x, n, eps_start)
        return val
    
    def gradient(x):
        _, g = c5_gradient(x, n, eps_start)
        return g
    
    def constraint_fun(x):
        return np.sum(x) - target
    
    def constraint_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    
    for eps_curr in [eps_start, eps_start * 0.5, eps_start * 0.2, eps_end]:
        if time.time() - t0 > time_limit:
            break
        # Redefine with current eps
        def obj_cur(x, e=eps_curr):
            val, _ = c5_gradient(x, n, e)
            return val
        def grad_cur(x, e=eps_curr):
            _, g = c5_gradient(x, n, e)
            return g
        
        try:
            res = minimize(obj_cur, h0, jac=grad_cur, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': max_iter // 4, 'ftol': 1e-10})
            h0 = project_to_constraint(res.x, n)
        except Exception:
            pass
    
    return h0

def random_binary_init(n, rng):
    """Generate random binary sequence with n/2 ones."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    rng.shuffle(h)
    return h

def golden_ratio_pattern(n, rng):
    """Generate a Sturmian-like pattern based on golden ratio."""
    alpha = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = float(np.floor((i + 1) * alpha) - np.floor(i * alpha))
    # Adjust to get exactly n/2 ones
    while np.sum(h) > n / 2.0:
        idx = np.where(h == 1.0)[0]
        h[idx[rng.integers(len(idx))]] = 0.0
    while np.sum(h) < n / 2.0:
        idx = np.where(h == 0.0)[0]
        h[idx[rng.integers(len(idx))]] = 1.0
    return h

def local_search_binary(h, n, max_iter=2000, time_limit=30):
    """Local search: try swapping pairs of 1 and 0 positions."""
    dx = 2.0 / n
    h = h.copy()
    c5_cur = compute_c5(h, n)
    t0 = time.time()
    iterations = 0
    
    while iterations < max_iter and time.time() - t0 < time_limit:
        iterations += 1
        # Find a 1 and a 0 to swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try random swaps
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h, n)
        
        if c5_new < c5_cur - 1e-12:
            c5_cur = c5_new
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return h, c5_cur

def upsample(h, n_src, n_dst):
    """Upsample h from n_src to n_dst points."""
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    h_up = np.interp(x_dst, x_src, h)
    return h_up

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get initial solution from parent
    h_parent = initial_h_values.astype(float).copy()
    n_parent = len(h_parent)
    h_parent = np.clip(h_parent, 0.0, 1.0)
    # Fix constraint
    target_par = n_parent / 2.0
    diff = target_par - np.sum(h_parent)
    if abs(diff) > 1e-10:
        room = 1.0 - h_parent if diff > 0 else h_parent.copy()
        if np.sum(room) > 1e-10:
            h_parent += room * (diff / np.sum(room))
    h_parent = np.clip(h_parent, 0.0, 1.0)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h, n_parent)
    best_n = n_parent
    
    # Phase 1: Optimize at parent resolution
    if time_left() > 300:
        h_opt = optimize_at_resolution(best_h, n_parent, eps_start=0.03, eps_end=0.001,
                                       max_iter=500, time_limit=min(200, time_left() - 50))
        c5_opt = compute_c5(h_opt, n_parent)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
    
    # Phase 2: Try multiple resolutions with upsampled initial
    resolutions = []
    for mult in [2, 4, 8, 16]:
        n_try = n_parent * mult
        if n_try <= 720 and n_try % 2 == 0:
            resolutions.append(n_try)
    # Also add some fixed resolutions
    for nfix in [60, 100, 150, 200, 300, 450]:
        if nfix % 2 == 0 and nfix not in resolutions:
            resolutions.append(nfix)
    resolutions = sorted(set(resolutions))
    
    for n_try in resolutions:
        if time_left() < 120:
            break
        if n_try % 2 != 0:
            continue
        
        h_up = upsample(best_h, best_n, n_try)
        h_up = project_to_constraint(h_up, n_try)
        
        h_opt = optimize_at_resolution(h_up, n_try, eps_start=0.05, eps_end=0.001,
                                       max_iter=300, time_limit=min(150, time_left() - 30))
        c5_opt = compute_c5(h_opt, n_try)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n = n_try
    
    # Phase 3: Try different initializations at best resolution
    n_work = best_n
    for init_type in ['random_binary', 'golden', 'random_continuous']:
        if time_left() < 90:
            break
        
        if init_type == 'random_binary':
            h_init = random_binary_init(n_work, rng)
        elif init_type == 'golden':
            h_init = golden_ratio_pattern(n_work, rng)
        else:
            h_init = rng.rand(n_work)
            h_init = project_to_constraint(h_init, n_work)
        
        c5_try = compute_c5(h_init, n_work)
        
        h_opt = optimize_at_resolution(h_init, n_work, eps_start=0.1, eps_end=0.002,
                                       max_iter=400, time_limit=min(120, time_left() - 20))
        c5_opt = compute_c5(h_opt, n_work)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n = n_work
    
    # Phase 4: Binary local search refinement
    if time_left() > 60:
        h_bin = (best_h > 0.5).astype(float)
        # Fix to have exactly n/2 ones
        ones_count = int(np.sum(h_bin == 1.0))
        if ones_count < best_n // 2:
            zeros = np.where(h_bin == 0)[0]
            need = best_n // 2 - ones_count
            h_bin[zeros[:need]] = 1.0
        elif ones_count > best_n // 2:
            ones = np.where(h_bin == 1)[0]
            remove = ones_count - best_n // 2
            h_bin[ones[:remove]] = 0.0
        
        h_bin_opt, c5_bin = local_search_binary(h_bin, best_n, max_iter=5000,
                                                 time_limit=min(40, time_left() - 20))
        if c5_bin < best_c5:
            best_h = h_bin_opt.copy()
            best_c5 = c5_bin
    
    # Phase 5: Final polishing - try a few more SLSQP runs from current best
    if time_left() > 60:
        for eps_f in [0.005, 0.002]:
            if time_left() < 30:
                break
            h_pol = optimize_at_resolution(best_h, best_n, eps_start=eps_f, eps_end=eps_f * 0.1,
                                           max_iter=300, time_limit=min(60, time_left() - 10))
            c5_pol = compute_c5(h_pol, best_n)
            if c5_pol < best_c5:
                best_h = h_pol.copy()
                best_c5 = c5_pol
    
    # Final constraint enforcement
    best_h = project_to_constraint(best_h, best_n)
    # Verify
    final_c5 = compute_c5(best_h, best_n)
    
    return (best_h, final_c5, best_n)
