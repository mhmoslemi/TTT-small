import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def compute_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode="full") * dx

def soft_max_obj(h, alpha):
    vals = compute_corr_vals(h)
    m = np.max(vals)
    shifted = alpha * (vals - m)
    return m + (1.0/alpha) * np.log(np.sum(np.exp(shifted)))

def soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full") * dx
    m = np.max(corr)
    exponents = alpha * (corr - m)
    exponents -= np.max(exponents)
    weights = np.exp(exponents)
    weights /= weights.sum()
    
    lags = np.arange(2*n-1) - (n-1)
    
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for k in range(2*n-1):
        lag = lags[k]
        w = weights[k]
        if lag >= 0:
            # (1-h[i-lag]) for i in [lag, n-1]
            term1[lag:n] += w * (1 - h[0:n-lag])
            # h[i+lag] for i in [0, n-1-lag]
            term2[0:n-lag] += w * h[lag:n]
        else:
            # (1-h[i-lag]) for i in [0, n-1+lag]
            nlag = -lag
            term1[0:n-nlag] += w * (1 - h[nlag:n])
            # h[i+lag] for i in [nlag, n-1]
            term2[nlag:n] += w * h[0:n-nlag]
    
    return dx * (term1 - term2)

def fix_constraints(h, n):
    """Clip to [0,1] and fix sum to n/2."""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0, 1)
    return h

def thue_morse(n):
    """Generate Thue-Morse sequence of length n, mapped to [0,1]."""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    # t is 0/1, we want sum = n/2 approximately
    # Thue-Morse has roughly equal 0s and 1s
    return t.astype(float)

def upsample_to_n(h, n):
    """Upsample h to n points."""
    old_n = len(h)
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    return new_h

def make_binary_pattern(n, period):
    """Create binary pattern with given period (half period = 1s, half = 0s)."""
    half = period // 2
    pattern = np.concatenate([np.ones(half), np.zeros(half)])
    result = np.tile(pattern, n // len(pattern) + 1)[:n]
    return result

    np.random.seed(seed)
    t0 = time.time()
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    def update_best(h, n):
        nonlocal best_c5, best_h, best_n
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
            return c5
        return c5
    
    def elapsed():
        return time.time() - t0
    
    # Get initial values
    h_parent = None
    if 'initial_h_values' in globals():
        try:
            iv = initial_h_values
            if iv is not None:
                h_parent = np.array(iv, dtype=float)
        except:
            pass
    
    # Try various initializations at multiple resolutions
    ns_to_try = [100, 150, 200, 300]

_CONST_STATE = {'n': 0}

def _constraint_fun(x):
    return x.sum() - _CONST_STATE['n'] / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def soft_max_obj(h, alpha):
    vals = compute_corr_vals(h)
    m = np.max(vals)
    shifted = alpha * (vals - m)
    shifted -= np.max(shifted)
    return m + (1.0 / alpha) * np.log(np.sum(np.exp(shifted)))

def soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exponents = alpha * (corr - m)
    exponents -= np.max(exponents)
    weights = np.exp(exponents)
    ws = weights.sum()
    if ws < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights = weights / ws

    lags = np.arange(2 * n - 1) - (n - 1)
    term1 = np.zeros(n)
    term2 = np.zeros(n)

    for k in range(2 * n - 1):
        lag = int(lags[k])
        w = weights[k]
        if w < 1e-15:
            continue
        if lag >= 0:
            # (1-h[i-lag]) valid for i in [lag, n-1]
            term1[lag:n] += w * (1.0 - h[0:n - lag])
            # h[i+lag] valid for i in [0, n-1-lag]
            term2[0:n - lag] += w * h[lag:n]
        else:
            nlag = -lag
            # (1-h[i-lag]) = (1-h[i+nlag]) valid for i in [0, n-1-nlag]
            term1[0:n - nlag] += w * (1.0 - h[nlag:n])
            # h[i+lag] = h[i-nlag] valid for i in [nlag, n-1]
            term2[nlag:n] += w * h[0:n - nlag]

    return dx * (term1 - term2)

        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

        if s > target:
