```python
import time
import math
import numpy as np
import scipy.optimize as opt

_OPT = {"n": 0, "S": 0.0, "groups": [], "beta": 1.0}
_CAL_CACHE = {}
_MASK_CACHE = {}


def project_sum_box(y, S):
    y = np.array(y, dtype=float, copy=True)
    if y.size == 0:
        return y
    np.clip(y, 0.0, 1.0, out=y)
    s = float(y.sum())
    if abs(s - S) <= 1e-14:
        return y

    lo = -float(y.min())
    hi = 1.0 - float(y.max())

    s_lo = float(np.clip(y + lo, 0.0, 1.0).sum())
    s_hi = float(np.clip(y + hi, 0.0, 1.0).sum())
    if s_lo > S:
        hi = lo
    elif s_hi < S:
        lo = hi

    for _ in range(90):
        mid = 0.5 * (lo + hi)
        sm = float(np.clip(y + mid, 0.0, 1.0).sum())
        if sm > S:
            hi = mid
        else:
            lo = mid

    y = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - float(y.sum())

    if abs(diff) > 1e-13:
        order = np.argsort(np.abs(y - 0.5))
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0.0:
                avail = 1.0 - float(y[idx])
                if avail <= 0.0:
                    continue
                amt = diff if diff < avail else avail
            else:
                avail = -float(y[idx])
                if avail <= 0.0:
                    continue
                amt = -diff
                if amt > avail:
                    amt = avail
                amt = -amt
            y[idx] += amt
            diff -= amt

    return y


def max_count(h):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def evaluate_c5(h):
    h = np.asarray(h, dtype=float)
    return max_count(h) * (2.0 / float(h.size))


def get_masks(n):
    if n not in _MASK_CACHE:
        _MASK_CACHE[n] = [(1 << m) - 1 for m in range(n + 1)]
    return _MASK_CACHE[n]


def build_groups_mode(n, mode):
    m = n - 1
    L = 2 * n - 1
    groups = []
    for idx in range(L):
        if mode == "A_asc":
            k = idx - m
        elif mode == "A_desc":
            k = m - idx
        elif mode == "B_asc":
            k = idx - m
        else:
            k = m - idx

        if mode.startswith("A"):
            if k >= 0:
                i = np.arange(k, n, dtype=int)
                j = np.arange(0, n - k, dtype=int)
            else:
                i = np.arange(0, n + k, dtype=int)
                j = np.arange(-k, n, dtype=int)
        else:
            if k >= 0:
                i = np.arange(0, n - k, dtype=int)
                j = np.arange(k, n, dtype=int)
            else:
                i = np.arange(-k, n, dtype=int)
                j = np.arange(0, n + k, dtype=int)
        groups.append((i, j))
    return groups


def calibrate_pairs(n, seed=0):
    m = n - 1
    L = 2 * n - 1
    mode = "A_asc"

    if n > 1:
        rng = np.random.RandomState((seed + 101) % 2147483646 + 1)
        h = rng.uniform(0.1, 0.9, n)
        b = 1.0 - h
        q = np.correlate(h, b, mode="full")

        arrA = np.empty(L, dtype=float)
        arrB = np.empty(L, dtype=float)

        for idx, k in enumerate(range(-m, m + 1)):
            if k >= 0:
                Lk = n - k
                arrA[idx] = float(np.dot(h[k:], b[:Lk]))
                arrB[idx] = float(np.dot(h[:Lk], b[k:]))
            else:
                Lk = n + k
                arrA[idx] = float(np.dot(h[:Lk], b[-k:]))
                arrB[idx] = float(np.dot(h[-k:], b[:Lk]))

        rtol = 1e-12
        atol = 1e-12
        if np.allclose(q, arrA, rtol=rtol, atol=atol):
            mode = "A_asc"
        elif np.allclose(q, arrA[::-1], rtol=rtol, atol=atol):
            mode = "A_desc"
        elif np.allclose(q, arrB, rtol=rtol, atol=atol):
            mode = "B_asc"
        elif np.allclose(q, arrB[::-1], rtol=rtol, atol=atol):
            mode = "B_desc"
        else:
            mode = "A_asc"

    return build_groups_mode(n, mode)


def get_groups(n, seed=0):
    if n not in _CAL_CACHE:
        _CAL_CACHE[n] = calibrate_pairs(n, seed + 7)
    return _CAL_CACHE[n]


def set_opt_state(n, S, groups, beta):
    _OPT["n"] = int(n)
    _OPT["S"] = float(S)
    _OPT["groups"] = groups
    _OPT["beta"] = float(beta)


def _smooth_obj(x):
    st = _OPT
    n = int(st["n"])
    groups = st["groups"]
    beta = float(st["beta"])

    h = np.asarray(x, dtype=float).reshape(-1)
    if h.size != n:
        h = h[:n] if h.size > n else np.pad(h, (0, n - h.size), mode="constant")

    if not np.all(np.isfinite(h)):
        return 1.0e100, np.zeros(n, dtype=float)

    h = np.clip(h, 0.0, 1.0)
    q = np.correlate(h, 1.0 - h, mode="full")

    if q.size == 0:
        return 0.0, np.zeros(n, dtype=float)

    zmax = float(q.max())
    ez = np.exp(beta * (q - zmax))
    ez_sum = float(ez.sum())
    if ez_sum <= 0.0 or not math.isfinite(ez_sum):
        p = np.zeros(q.size, dtype=float)
        p[int(np.argmax(zmax * np.ones_like(q)))] = 1.0
        ez_sum = 1.0
    else:
        p = ez / ez_sum

    fval = (zmax + math.log(max(ez_sum, 1.0e-300))) / beta

    grad = np.zeros(n, dtype=float)
    for idx, w in enumerate(p):
        if w <= 1.0e-13:
            continue
        i, j = groups[idx]
        if i.size == 0:
            continue
        grad[i] += w * (1.0 - h[j])
        grad[j] -= w * h[i]

    return float(fval), grad


def _eq_fun(x):
    return float(np.sum(np.asarray(x, dtype=float).reshape(-1)) - float(_OPT["S"]))


def _eq_jac(x):
    return np.ones(int(_OPT["n"]), dtype=float)


def slsqp_refine(h0, deadline, seed):
    h = project_sum_box(h0, 0.5 * float(np.size(h0)))
    c = evaluate_c5(h)
    n = int(h.size)
    S = 0.5 * float(n)
    groups = get_groups(n, seed)

    best = h.copy()
    best_c = c
    cur = h.copy()

    left = float(deadline) - time.time()
    if left < 70.0 or n > 220:
        return best, best_c

    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{"type": "eq", "fun": _eq_fun, "jac": _eq_jac}]

    if left > 320.0:
        schedule = [(6, 50), (16, 60), (40, 80), (90, 80)]
    elif left > 180.0:
        schedule = [(8, 40), (30, 60), (80, 60)]
    elif left > 100.0:
        schedule = [(12, 30), (50, 45)]
    else:
        schedule = [(20, 25)]

    for beta, maxit in schedule:
        if time.time() > float(deadline) - 25.0:
            break
        set_opt_state(n, S, groups, beta)
        try:
            res = opt.minimize(
                _smooth_obj,
                cur.copy(),
                method="SLSQP",
                jac=True,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": int(maxit), "ftol": 1.0e-8, "disp": False, "iprint": 0},
            )
        except Exception:
            res = None

        if res is not None:
            x = np.asarray(res.x, dtype=float).reshape(-1)
            if x.size == n and np.all(np.isfinite(x)):
                x = project_sum_box(x, S)
                cx = evaluate_c5(x)
                cur = x
                if cx < best_c:
                    best = x.copy()
                    best_c = cx

    return best, best_c


def polish_pgd(h0, deadline, seed, groups=None, betas=(15, 45, 100), maxit_per=200):
    h = project_sum_box(h0, 0.5 * float(np.size(h0)))
    n = int(h.size)
    S = 0.5 * float(n)
    dx = 2.0 / float(n)

    if groups is None:
        groups = get_groups(n, seed)

    best = h.copy()
    best_c = evaluate_c5(h)
    cur = h.copy()
    cur_count = max_count(cur)

    if betas is None:
        betas = [40]
    for beta in betas:
        left = float(deadline) - time.time()
        if left < 10.0:
            break
        set_opt_state(n, S, groups, float(beta))
        rng = np.random.RandomState((int(seed) + int(beta)) % 2147483646 + 1)
        no_improve = 0

        itercap = int(maxit_per)
        if left < 60.0:
            itercap = min(itercap, 80)
        elif left < 150.0:
            itercap = min(itercap, 140)

        for it in range(itercap):
            if time.time() > float(deadline) - 2.0:
                break

            f, g = _smooth_obj(cur)
            gn = float(np.linalg.norm(g))
            if not math.isfinite(gn) or gn < 1.0e-10:
                break

            g = g / gn
            improved = False

            for base in (0.45, 0.25, 0.12, 0.06):
                for fac in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125):
                    cand = project_sum_box(cur - (base * fac) * g, S)
                    cc = max_count(cand)
                    if cc < cur_count - 1.0e-12:
                        cur = cand
                        cur_count = cc
                        improved = True
                        c5 = cc * dx
                        if c5 < best_c:
                            best_c = c5
                            best = cur.copy()
                        no_improve = 0
                        break
                if improved:
                    break

            if not improved:
                no_improve += 1
                if no_improve % 4 == 0:
                    scale = 0.045 if beta < 40.0 else 0.02
                    noise = rng.normal(0.0, scale, size=n)
                    cand = project_sum_box(cur + noise, S)
                    cc = max_count(cand)
                    if cc < cur_count - 1.0e-12:
                        cur = cand
                        cur_count = cc
                        c5 = cc * dx
                        if c5 < best_c:
                            best_c = c5
                            best = cur.copy()

                    cand = project_sum_box(cur - 0.08 * g + 0.5 * noise, S)
                    cc = max_count(cand)
                    if cc < cur_count - 1.0e-12:
                        cur = cand
                        cur_count = cc
                        c5 = cc * dx
                        if c5 < best_c:
                            best_c = c5
                            best = cur.copy()

                if no_improve > 70:
                    break

        cur = project_sum_box(cur, S)

    return best, best_c


def refine_continuous(h0, deadline, seed):
    h = project_sum_box(h0, 0.5 * float(np.size(h0)))
    c = evaluate_c5(h)
    best = h.copy()
    best_c = c

    left = time.time_to_death if False else float(deadline) - time.time()

    if left > 140.0:
        sl_deadline = time.time() + max(30.0, min(180.0, 0.36 * left))
        sl_best, sl_c = slsqp_refine(best, sl_deadline, seed)
        if sl_c < best_c:
            best = sl_best
            best_c = sl_c

    left = float(deadline) - time.time()
    if left > 60.0:
        if left > 260.0:
            betas = (15, 45, 100)
            mitt = 220
        elif left > 140.0:
            betas = (25, 80)
            mitt = 160
        else:
            betas = (50,)
            mitt = 90
        pg_best, pg_c = polish_pgd(best, float(deadline) - 2.0, seed + 17, betas=betas, maxit_per=mitt)
        if pg_c < best_c:
            best = pg_best
            best_c = pg_c

    return best, best_c


def support_count(A, n):
    return int((A & ((1 << n) - 1)).bit_count())


def normalize_support(A, n, K):
    A = int(A)
    A &= (1 << n) - 1
    cnt = support_count(A, n)

    i = 0
    while cnt > K and i < n:
        if (A >> i) & 1:
            A &= ~(1 << i)
            cnt -= 1
        i += 1

    i = 0
    while cnt < K and i < n:
        if ((A >> i) & 1) == 0:
            A |= 1 << i
            cnt += 1
        i += 1

    return A


def bits_from_indices(indices):
    A = 0
    for v in indices:
        A |= 1 << int(v)
    return A


def support_from_h(h, K):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if K <= 0:
        return 0
    order = np.argsort(h)[::-1]
    return normalize_support(bits_from_indices(order[:K]), n, K)


def beatty_support(n, K, alpha=0.6180339887498949, offset=0.0):
    if K <= 0:
        return 0
    i = np.arange(n, dtype=float)
    vals = np.mod(i * float(alpha) + float(offset), 1.0)
    order = np.argsort(vals)[:K]
    return normalize_support(bits_from_indices(order), n, K)


def random_support(rng, n, K):
    if K <= 0:
        return 0
    idx = rng.choice(n, size=K, replace=False)
    return normalize_support(bits_from_indices(idx), n, K)


def reverse_support(A, n):
    out = 0
    for i in range(n):
        if (A >> i) & 1:
            out |= 1 << (n - 1 - i)
    return out


def binary_max_count(A, n, masks):
    best = 0
    A &= (1 << n) - 1
    for d in range(1, n):
        m = n - d
        mask = masks[m]
        A_m = A & mask
        B = (A >> d) & mask
        not_A_m = mask ^ A_m
        not_B = mask ^ B
        df = int((A_m & not_B).bit_count())
        db = int((B & not_A_m).bit_count())
        if df > best:
            best = df
        if db > best:
            best = db
    return best


def sample_two_without_replacement(arr, rng):
    L = len(arr)
    a = int(rng.randint(L))
    b = int(rng.randint(L - 1))
    if b >= a:
        b += 1
    return arr[a], arr[b]


def propose_single(A, n, rng):
    A &= (1 << n) - 1
    i = int(rng.randint(n))
    while ((A >> i) & 1) == 0:
        i = int(rng.randint(n))
    j = int(rng.randint(n))
    while ((A >> j) & 1) != 0:
        j = int(rng.randint(n))
    return A ^ (1 << i) ^ (1 << j)


def propose_double(A, n, rng):
    A &= (1 << n) - 1
    ones = []
    zeros = []
    for i in range(n):
        if (A >> i) & 1:
            ones.append(i)
        else:
            zeros.append(i)

    if len(ones) < 2 or len(zeros) < 2:
        return propose_single(A, n, rng)

    i1, i2 = sample_two_without_replacement(ones, rng)
    j1, j2 = sample_two_without_replacement(zeros, rng)
    return A ^ (1 << i1) ^ (1 << i2) ^ (1 << j1) ^ (1 << j2)


def propose_move(A, n, rng):
    if rng.random_sample() < 0.76:
        return propose_single(A, n, rng)
    return propose_double(A, n, rng)


def binary_local_session(n, K, A, deadline, seed):
    masks = get_masks(n)
    rng = np.random.RandomState(seed % 2147483646 + 1)

    curA = normalize_support(A, n, K)
    curF = binary_max_count(curA, n, masks)

    bestA = curA
    bestF = curF

    T = 1.0
    no_improve = 0
    no_accept = 0
    batch = 80

    while time.time() < float(deadline):
        left = float(deadline) - time.time()
        if left <= 0.05:
            break

        bs = batch if left > 6.0 else max(1, int(min(18, left * 18)))
        old_bestF = bestF
        moveA = None
        moveF = 10**18

        for _ in range(bs):
            A2 = propose_move(curA, n, rng)
            F2 = binary_max_count(A2, n, masks)
            if F2 < moveF:
                moveF = F2
                moveA = A2
            if F2 < bestF:
                bestF = F2
                bestA = A2

        if moveA is None:
            continue

        if moveF < curF:
            accept = True
        else:
            if T > 0.0:
                delta = float(moveF - curF) / float(T)
                if delta < 700.0:
                    prob = math.exp(-delta)
                else:
                    prob = 0.0
                accept = rng.random_sample() < prob
            else:
                accept = False

        if accept:
            curA = moveA
            curF = moveF
            no_accept = 0
            if bestF < old_bestF:
                no_improve = 0
                T = max(0.05, 0.90 * T)
            else:
                no_improve += 1
        else:
            no_accept += 1
            no_improve += 1

        if not accept:
            if no_accept > 6 * bs:
                no_accept = 0
                T = max(0.25, min(1.2, float(bestF - curF) + 1.0))
                if rng.random_sample() < 0.45:
                    curA = bestA
                    curF = bestF

        if no_improve > 160:
            no_improve = 0
            T *= 0.65
            if T < 0.05:
                T = 1.0
                if rng.random_sample() < 0.72:
                    curA = bestA
                    curF = bestF
                else:
                    curA = random_support(rng, n, K)
                    curF = binary_max_count(curA, n, masks)

                for _ in range(int(rng.randint(1, 5))):
                    curA = propose_move(curA, n, rng)
                curF = binary_max_count(curA, n, masks)
                if curF < bestF:
                    bestF = curF
                    bestA = curA

    return bestA, bestF


def binary_search(n, K, starts, deadline, seed):
    masks = get_masks(n)
    bestA = None
    bestF = 10**18

    uniq = []
    seen = set()
    for A in starts:
        A = normalize_support(A, n, K)
        if A not in seen:
            seen.add(A)
            uniq.append(A)

    if not uniq:
        rng = np.random.RandomState(seed % 2147483646 + 1)
        uniq.append(random_support(rng, n, K))

    scored = []
    for A in uniq:
        F = binary_max_count(A, n, masks)
        if F < bestF:
            bestF = F
            bestA = A
        scored.append((F, A))

    scored.sort(key=lambda z: z[0])

    initial_left = float(deadline) - time.time()
    if initial_left <= 0.0:
        return bestA, bestF

    per_base = max(6.0, min(50.0, initial_left / max(1, len(scored)) * 0.72))

    for idx, (_, A) in enumerate(scored):
        if time.time() > float(deadline) - 2.0:
            break

        left = float(deadline) - time.time()
        if left <= 0.0:
            break

        per = min(per_base, max(5.0, 0.45 * left / max(1, len(scored) - idx)))
        if scored[idx][0] > bestF + 3.0:
            per = min(per, 12.0)

        local_deadline = time.time() + per
        A_loc, F_loc = binary_local_session(n, K, A, local_deadline, seed + 900 + 7 * idx)
        if F_loc < bestF:
            bestF = F_loc
            bestA = A_loc

    return bestA, bestF


def support_to_candidate(n, A):
    A = int(A)
    h = np.zeros(n, dtype=float)
    for i in range(n):
        if (A >> i) & 1:
            h[i] = 1.0

    if n % 2 == 0:
        h = project_sum_box(h, 0.5 * float(n))
        return h, evaluate_c5(h)

    K = (n - 1) // 2
    A = normalize_support(A, n, K)

    besth = None
    bestc = float("inf")

    for r in range(n):
        if ((A >> r) & 1) == 0:
            h2 = h.copy()
            for i in range(n):
                if (A >> i) & 1:
                    h2[i] = 1.0
                else:
                    h2[i] = 0.0
            h2[r] = 0.5
            c = evaluate_c5(h2)
            if c < bestc:
                bestc = c
                besth = h2

    if besth is None:
        besth = project_sum_box(h, 0.5 * float(n))
        bestc = evaluate_c5(besth)

    return besth, bestc
```
