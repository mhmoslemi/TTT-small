def compute_c5(h, dx):

def project_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h


def subgradient(h, n, dx):
    all_vals = np.correlate(h, 1 - h, mode="full") * dx
    max_val = np.max(all_vals)
    active = np.where(all_vals >= max_val - 1e-10)[0]
    grad = np.zeros(n)
    for lag_idx in active:
        k = lag_idx - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_end <= i_start:

        i_valid = np.arange(i_start, i_end)
        j_valid = i_valid + k
        grad[i_valid] += (1.0 - h[j_valid])
        grad[j_valid] += (-h[i_valid])
    grad /= max(len(active), 1)
    grad *= dx
    return grad

        best_h = h.copy()

    # Phase 3: More subgradient after local search
    h, c5_cur = subgradient_optimize(h, n, dx, target, max_iters=5000, time_limit=200, t0=t0)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h.copy()

    # Phase 4: Simulated annealing
    h, c5_cur = simulated_annealing(h, n, dx, target, rng, max_iters=80000, time_limit=300, t0=t0)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h.copy()

    # Phase 5: Local search again
    h, c5_cur = local_search_binary(h, n, dx, target, max_sweeps=10)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h.copy()

    # Phase 6: Try higher resolutions
    for n_try in [94, 120, 150, 200, 300]:
        if time_left() < 80:
            break
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        h_try = upsample(best_h, best_n, n_try)
        h_try = project_sum(h_try, target_try)
        c5_try = compute_c5(h_try, dx_try)

        # Subgradient at higher resolution
        remaining = time_left() - 20
        max_it = int(remaining * 20)
        h_try, c5_try = subgradient_optimize(h_try, n_try, dx_try, target_try,
                                             max_iters=max_it, time_limit=remaining * 0.4, t0=t0)
        # Local search
        if time_left() > 30:
            h_try, c5_try = local_search_binary(h_try, n_try, dx_try, target_try, max_sweeps=5)
        # SA
        if time_left() > 30:
            h_try, c5_try = simulated_annealing(h_try, n_try, dx_try, target_try, rng,
                                                max_iters=20000, time_limit=30, t0=t0)

        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            best_n = n_try
            n = n_try
            dx = dx_try
            target = target_try

    # Phase 7: Final polish at best resolution
    if time_left() > 30:
        h = best_h.copy()
        c5_cur = best_c5
        h, c5_cur = subgradient_optimize(h, len(h), 2.0 / len(h), len(h) / 2.0,
                                         max_iters=3000, time_limit=time_left() * 0.3, t0=t0)
        if time_left() > 15:
            h, c5_cur = local_search_binary(h, len(h), 2.0 / len(h), len(h) / 2.0, max_sweeps=5)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h

    # Final constraint check
    best_h = project_sum(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
