        h_cur = run_optimize(best_h, best_n, best_n / 2.0, beta=30.0, max_iter=400, ftol=1e-12)
        try_update(h_cur, best_n)
    
    # Step 1c: Sharp objective
    if elapsed() < budget_s * 0.6:
        h_cur = run_optimize(best_h, best_n, best_n / 2.0, beta=100.0, max_iter=300, ftol=1e-12)
        try_update(h_cur, best_n)
    
    # Phase 2: Try other resolutions
    for n_try in [100, 130, 170]:
        if elapsed() > budget_s * 0.78:
            break
        x_old = np.linspace(0, 2, n_parent, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        target_try = n_try / 2.0
        h_interp = project_sum(h_interp, target_try)
        
        if elapsed() > budget_s * 0.8:
            h_cur = run_optimize(h_interp, n_try, target_try, beta=20.0, max_iter=200, ftol=1e-10)
        else:
            h_cur = run_optimize(h_interp, n_try, target_try, beta=20.0, max_iter=400, ftol=1e-11)
        try_update(h_cur, n_try)
        
        if elapsed() > budget_s * 0.82:
            h_cur = run_optimize(best_h if best_n == n_try else h_interp,
                                best_n if best_n == n_try else n_try,
                                (best_n if best_n == n_try else n_try) / 2.0,
                                beta=80.0, max_iter=250, ftol=1e-12)
            try_update(h_cur, best_n if best_n == n_try else n_try)
    
    # Phase 3: Perturbation + reoptimize around best
    if elapsed() < budget_s * 0.92:
        n_best = best_n
        tgt = n_best / 2.0
        for attempt in range(4):
            if elapsed() > budget_s * 0.94:
                break
            noise_scale = 0.015 * (1 - attempt * 0.2)
            h_pert = best_h + np.random.randn(n_best) * noise_scale
            h_pert = project_sum(h_pert, tgt)
            h_cur = run_optimize(h_pert, n_best, tgt, beta=40.0, max_iter=300, ftol=1e-12)
            try_update(h_cur, n_best)
    
    # Phase 4: Final sharp optimization on best
    if elapsed() < budget_s * 0.98:
        n_best = best_n
        tgt = n_best / 2.0
        h_cur = run_optimize(best_h, n_best, tgt, beta=200.0, max_iter=200, ftol=1e-13)
        try_update(h_cur, n_best)
    
    # Final: ensure exact constraint
    n_final = best_n
    tgt_final = n_final / 2.0
    best_h = project_sum(best_h, tgt_final)
    
    # Verify sum is exact (within floating point)
    s = np.sum(best_h)
    if abs(s - tgt_final) > 1e-10:
        best_h[n_final // 2] += (tgt_final - s)
        best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    
    return (best_h, final_c5, best_n)
