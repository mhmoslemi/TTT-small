def evaluate(h):
    """Compute C5 value for a given h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def make_binary(h, n):
    """Convert h to a binary vector with exactly n/2 ones."""
    n_half = n // 2
    # Sort indices by h value, set top n/2 to 1
    sorted_idx = np.argsort(-h)
    binary = np.zeros(n)
    binary[sorted_idx[:n_half]] = 1
    return binary

def local_search_binary(h, n, max_iter=10000, temperature=0.01, cooling=0.999):
    """Simulated annealing local search on binary vectors."""
    n_half = n // 2
    h = make_binary(h, n).copy()
    best_h = h.copy()
    best_val = evaluate(h)
    curr_val = best_val
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        # Pick a random 1 and a random 0 to swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Swap
        h[i], h[j] = h[j], h[i]
        new_val = evaluate(h)
        
        if new_val < curr_val or np.random.random() < np.exp((curr_val - new_val) / temperature):
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Undo swap
            h[i], h[j] = h[j], h[i]
        
        temperature *= cooling
    
    return best_h, best_val

    for n in [64, 96, 128, 160, 192, 256, 288]:
        if n not in resolutions:
            resolutions.append(n)
    resolutions = sorted(set(resolutions))
    
    for n in resolutions:
        elapsed = time.time() - t0
        remaining = budget_s - elapsed
        if remaining < 60:
            break
        
        # Build multiple starting points
        starts = []
        
        # From best solution interpolated
        h_interp = interpolate_h(best_h, len(best_h), n)
        starts.append(h_interp)
        
        # Thue-Morse
        if n <= 512:
            starts.append(thue_morse(n))
        
        # Random
        starts.append(np.random.random(n))
        
        # Greedy (if time allows)
        if remaining > 300:
            for gs in range(3):
                h_g = greedy_construct(n, seed + gs * 100)
                starts.append(h_g)
        
        # Run multi-start SA
        time_per_res = min(remaining * 0.6, 300)
        h_new, c5_new = multi_start_sa(starts, n, time_per_res, seed_base=n)
        
        if h_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 3: Final SA refinement on best solution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 60:
        n_best = len(best_h)
        h_fin, c5_fin = sa_binary(best_h, n_best, 
                                   min(500000, int(remaining * 20000)), 
                                   0.3, 0.003, seed + 9999)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Phase 4: Continuous refinement
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 40 and len(best_h) <= 200:
        h_cont, c5_cont = continuous_refine(best_h, len(best_h), remaining - 10)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
    
    # Final: ensure feasibility
    n_final = len(best_h)
    best_h = np.clip(best_h, 0, 1)
    diff = n_final / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-6) & (best_h < 1 - 1e-6)
        if np.sum(mask) > 0:
            best_h[mask] += diff / np.sum(mask)
        best_h = np.clip(best_h, 0, 1)
    
    # If still not exactly n/2, snap to binary
    if abs(np.sum(best_h) - n_final / 2.0) > 1e-8:
        best_h = make_binary(best_h, n_final)
    
    final_c5 = evaluate_c5(best_h)
    
    return best_h, final_c5, n_final
