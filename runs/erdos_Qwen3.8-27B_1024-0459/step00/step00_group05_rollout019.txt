        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        # Transfer mass
        delta = np.random.uniform(-0.1, 0.1)
        new_hi = h[i] + delta
        new_hj = h[j] - delta
        if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:

        h[i] = new_hi
        h[j] = new_hj

            curr_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = new_hi - delta  # revert
            h[j] = new_hj + delta
        
        T *= 0.9999  # cooling
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def gradient_softmax(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / np.sum(e) / beta
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        s = k - (n-1)
        for i in range(n):
            g = 0.0
            if 0 <= i+s < n:
                g += 1.0 - h[i+s]
            if 0 <= i-s < n:
                g -= h[i-s]
            grad[i] += w[k] * dx * g
    return grad

def project(h, target_sum):
    h = np.clip(h, 0, 1)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-10:
            break
        h += diff / len(h)
        h = np.clip(h, 0, 1)
    return h

    best = h.copy()
    best_c5, _ = compute_c5(h)
    h = h.copy()
    step = 0.01
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # Anneal beta
        frac = min(1.0, (time.time() - start) / time_limit)
        beta = beta_start + (beta_end - beta_start) * frac
        grad = gradient_softmax(h, beta)
        # Projected gradient step

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_soft_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - (n - 1)
        for i in range(n):
            g = 0.0
            idx1 = i + s
            if 0 <= idx1 < n:
                g += 1.0 - h[idx1]
            idx2 = i - s
            if 0 <= idx2 < n:
                g -= h[idx2]
            grad[i] += w[k] * dx * g
    return grad

def optimize_gradient(h_init, n, time_limit=200, beta_start=10.0, beta_end=200.0):
    h = h_init.astype(float).copy()
    h = project(h, n / 2.0)
    c5_best, _ = compute_c5(h)
    h_best = h.copy()
    step = 1.0
    start = time.time()
    while time.time() - start < time_limit:
        frac = min(1.0, (time.time() - start) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** frac
        grad = grad_soft_max(h, beta)
        # Gradient descent step
        h_new = h - step * grad
        h_new = project(h_new, n / 2.0)
        c5_new, _ = compute_c5(h_new)
        c5_curr, _ = compute_c5(h)
        if c5_new < c5_curr:
            h = h_new
            step = min(step * 1.05, 5.0)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            step = max(step * 0.7, 0.01)
    return h_best, c5_best

def optimize_sa(h_init, n, time_limit=200, rng=None):

        h[j] = nj
