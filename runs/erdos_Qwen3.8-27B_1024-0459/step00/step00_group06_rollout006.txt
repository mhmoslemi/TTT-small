def evaluate_c5(h):

def compute_F(h):
    h=np.asarray(h,float).ravel()
    N=h.size
    R=np.correlate(h,h,mode='full')
    Rpos=R[N-1:N-1+N]
    pref=np.concatenate((np.array([0.0]), np.cumsum(h)))
    d=np.arange(N)
    pos=pref[N-d] - Rpos
    neg=(pref[-1] - pref[d]) - Rpos
    pos=np.where(np.isfinite(pos), pos, 0.0)
    neg=np.where(np.isfinite(neg), neg, 0.0)
    return pos, neg

def evaluate_c5(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size < 2:
        return float("inf")
    if not np.all(np.isfinite(h)):
        return float("inf")
    h = np.clip(h, 0.0, 1.0)
    dx = 2.0 / float(h.size)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    if not math.isfinite(m):
        return float("inf")
    if m < 0.0:
        m = 0.0
    return m * dx


def target_count_for_c5(N, c5):
    if N <= 0:
        return 0
    val = float(c5) * float(N) / 2.0
    return int(math.floor(val + 1e-12))


def project_box_sum(y, target):
    y = np.asarray(y, dtype=float).ravel()
    N = y.size
    if N == 0:
        return y
    y = np.where(np.isfinite(y), y, 0.5)
    if target <= 0.0:
        return np.zeros(N, dtype=float)
    if target >= float(N):
        return np.ones(N, dtype=float)

    ymax = float(np.max(y))
    lo = -2.0 - ymax
    hi = 2.0 + ymax

    for _ in range(30):
        s_lo = float(np.sum(np.clip(y - lo, 0.0, 1.0)))
        if s_lo < float(target):
            lo -= 10.0
        else:
            break

    for _ in range(30):
        s_hi = float(np.sum(np.clip(y - hi, 0.0, 1.0)))
        if s_hi > float(target):
            hi += 10.0
        else:
            break

        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y - mid, 0.0, 1.0)))
        if s > float(target):
            lo = mid
        else:
            hi = mid

    h = np.clip(y - hi, 0.0, 1.0)
    diff = float(target - float(h.sum()))

    if abs(diff) > 1e-10:
        if diff > 0.0:
            order = np.argsort(h)
            for idx in order:
                if abs(diff) <= 1e-12:
                    break
                room = 1.0 - h[idx]
                add = min(room, diff)
                if add > 0.0:

                    diff -= add

                room = h[idx]

                if dec > 0.0:

                    need -= dec

def compute_F(h):
    h = np.asarray(h, dtype=float).ravel()
    N = h.size
    if N == 0:
        return np.array([], dtype=float), np.array([], dtype=float)
    R = np.correlate(h, h, mode="full")
    Rpos = R[N - 1:N - 1 + N]
    pref = np.concatenate((np.array([0.0]), np.cumsum(h)))
    d = np.arange(N)
    pos = pref[N - d] - Rpos
    neg = (pref[-1] - pref[d]) - Rpos
    pos = np.where(np.isfinite(pos), pos, 0.0)
    neg = np.where(np.isfinite(neg), neg, 0.0)
    return pos, neg

                a[idx] -= subp
                need -= subp
        count = 0
        while need > 0 and count < 2 * N:
            idx = count % N
            subp = min(int(a[idx]), need)
            if subp > 0:
                a[idx] -= subp
                need -= subp
            count += 1

    return a.astype(float) / float(denom)


def compute_counts_binary(x):
    x = np.asarray(x).ravel()
    N = x.size
    if N == 0:
        return np.array([], dtype=np.int32), np.array([], dtype=np.int32)
    hx = x.astype(float)
    R = np.correlate(hx, hx, mode="full")
    Rpos = R[N - 1:N - 1 + N]
    pref = np.concatenate((np.array([0.0]), np.cumsum(hx)))
    d = np.arange(N)
    pos = pref[N - d] - Rpos
    neg = (pref[-1] - pref[d]) - Rpos
    pos = np.rint(pos).astype(np.int32)
    neg = np.rint(neg).astype(np.int32)
    pos = np.clip(pos, 0, N)
    neg = np.clip(neg, 0, N)
    if N > 0:
        pos[0] = 0
        neg[0] = 0
    return pos, neg


def count_score(pos, neg):
    c = np.maximum(pos, neg)
    if c.size == 0:
        return (0, 0, 0)
    m = int(np.max(c))
    exact = int(np.count_nonzero(c == m))
    if m > 0:
        near = int(np.count_nonzero(c >= m - 1))
    else:
        near = int(c.size)
    return (m, exact, near)


def apply_swap(x, pos, neg, a, b):
    if a == b:
        return
    xa = int(x[a])
    xb = int(x[b])
    if xa == xb:
        return

    N = len(x)
    s_a = 1 - 2 * xa
    s_b = 1 - 2 * xb

    if N - a > 1:
        end = N - a
        r = x[a + 1:end]
        pos[1:end] += s_a * (1 - r)
        neg[1:end] += -s_a * r
    if a > 0:
        l = x[:a]
        pos[1:a + 1] += -s_a * l
        neg[1:a + 1] += s_a * (1 - l)

    if N - b > 1:
        end = N - b
        r = x[b + 1:end]
        pos[1:end] += s_b * (1 - r)
        neg[1:end] += -s_b * r
    if b > 0:
        l = x[:b]
        pos[1:b + 1] += -s_b * l
        neg[1:b + 1] += s_b * (1 - l)

    d = abs(a - b)
    if 0 < d < N:
        pos[d] += 1
        neg[d] += 1

    x[a], x[b] = xb, xa


def choose_targeted_move(x, pos, neg, maxc, rng):
    N = len(x)
    if maxc <= 0 or N < 2:
        return None

    ap = np.where(pos == maxc)[0]
    an = np.where(neg == maxc)[0]
    if ap.size:
        ap = ap[ap > 0]
    if an.size:
        an = an[an > 0]

    orientation = None
    good = None

    if ap.size == 0 and an.size > 0:
        orientation = "neg"
        good = an
    elif an.size == 0 and ap.size > 0:
        orientation = "pos"
        good = ap
    elif ap.size > 0 and an.size > 0:
        gp = ap[neg[ap] < maxc]
        gn = an[pos[an] < maxc]
        if gp.size == 0 and gn.size == 0:
            if ap.size >= an.size:
                orientation = "pos"
                good = ap
            else:
                orientation = "neg"
                good = an
        else:
            if gp.size > gn.size:
                orientation = "pos"
                good = gp
            elif gn.size > gp.size:
                orientation = "neg"
                good = gn
            elif gp.size > 0 and gn.size > 0:
                if rng.random() < 0.5:
                    orientation = "pos"
                    good = gp
                else:
                    orientation = "neg"
                    good = gn
            elif gp.size > 0:
                orientation = "pos"
                good = gp
            else:
                orientation = "neg"
                good = gn
    else:
        return None

    if good is None or good.size == 0:
        return None

    d = int(good[rng.randrange(int(good.size))])
    if d <= 0 or d >= N:
        return None

    if orientation == "pos":
        pairs = np.flatnonzero((x[:-d] == 1) & (x[d:] == 0))
    else:
        pairs = np.flatnonzero((x[:-d] == 0) & (x[d:] == 1))

    if pairs.size == 0:
        return None

    i = int(pairs[rng.randrange(int(pairs.size))])
    return i, i + d


def random_swap_position(x, rng):
    ones = np.flatnonzero(x == 1)
    zeros = np.flatnonzero(x == 0)
    if ones.size == 0 or zeros.size == 0:
        return None
    a = int(ones[rng.randrange(int(ones.size))])
    b = int(zeros[rng.randrange(int(zeros.size))])
    return a, b


def local_search_binary(x0, deadline, seed):
    x0 = np.asarray(x0).ravel()
    N = x0.size
    if N < 2:
        return None, float("inf")

    x = x0.astype(np.int8).copy()
    pos, neg = compute_counts_binary(x)
    cur_score = count_score(pos, neg)
    cur_c = cur_score[0] * (2.0 / N)

    best_x = x.copy()
    best_pos = pos.copy()
    best_neg = neg.copy()
    best_score = cur_score
    best_c = cur_c

    rng = random.Random(seed)
    temp = 0.20
    ifails = 0
    it = 0

    while time.time() < deadline - 1.0:
        it += 1

        cur_c = cur_score[0] * (2.0 / N)
        if cur_c < best_c - 1e-12 or (abs(cur_c - best_c) <= 1e-12 and cur_score < best_score):
            best_c = cur_c
            best_score = cur_score
            best_x = x.copy()
            best_pos = pos.copy()
            best_neg = neg.copy()

        maxc = cur_score[0]
        if maxc == 0:
            break

        move = choose_targeted_move(x, pos, neg, maxc, rng)
        if move is None:
            move = random_swap_position(x, rng)
        if move is None:
            break

        a, b = move
        old_score = cur_score
        apply_swap(x, pos, neg, a, b)
        new_score = count_score(pos, neg)

        accept = False
        if new_score < old_score:
            accept = True
        elif new_score[0] <= old_score[0] + 1:
            accept = (rng.random() < temp)

        if accept:
            cur_score = new_score
            ifails = 0
            if new_score[0] < old_score[0]:
                temp = min(0.25, temp + 0.01)
        else:
            apply_swap(x, pos, neg, a, b)
            cur_score = old_score
            ifails += 1

            if ifails > 260:
                x = best_x.copy()
                pos = best_pos.copy()
                neg = best_neg.copy()
                cur_score = best_score
                ifails = 0
                perturb = rng.randint(1, 4)
                for _ in range(perturb):
                    mv = random_swap_position(x, rng)
                    if mv is None:
                        break
                    a2, b2 = mv
                    apply_swap(x, pos, neg, a2, b2)
                cur_score = count_score(pos, neg)
                cur_c2 = cur_score[0] * (2.0 / N)
                if cur_c2 < best_c - 1e-12 or (abs(cur_c2 - best_c) <= 1e-12 and cur_score < best_score):
                    best_c = cur_c2
                    best_score = cur_score
                    best_x = x.copy()
                    best_pos = pos.copy()
                    best_neg = neg.copy()

        if it % 500 == 0:
            temp *= 0.96
        if ifails > 8000:
            break

    final_c = evaluate_c5(best_x.astype(float))
    if not math.isfinite(final_c):
        final_c = float(best_c)
    return best_x, float(final_c)


def resample_values(h_old, N):
    h_old = np.asarray(h_old, dtype=float).ravel()
    oldN = h_old.size
    if N <= 0:
        return np.array([], dtype=float)
    if oldN == 0:
        return np.full(N, 0.5, dtype=float)
    if oldN == N:
        return h_old.copy()

    x_old = (np.arange(oldN) + 0.5) * (2.0 / oldN)
    x_new = (np.arange(N) + 0.5) * (2.0 / N)
    out = np.interp(x_new, x_old, h_old)
    out = np.where(np.isfinite(out), out, 0.5)
    return out


def probs_to_binary(p, N, seed):
    p = np.asarray(p, dtype=float).ravel()
    if p.size != N:
        p = np.resize(p, N)
    p = np.where(np.isfinite(p), p, 0.5)
    target = N // 2
    if target <= 0:
        return np.zeros(N, dtype=np.int8)
    order = np.argsort(-p, kind="mergesort")
    idx = order[:target]
    x = np.zeros(N, dtype=np.int8)
    x[idx] = 1
    return x


def random_balanced(N, seed):
    N = int(N)
    x = np.zeros(N, dtype=np.int8)
    if N < 1:
        return x
    rng = random.Random(seed)
    k = N // 2
    if k > 0 and k < N:
        idx = rng.sample(range(N), k)
        for i in idx:
            x[i] = 1
    return x


def thue_morse_balanced(N, seed):
    N = int(N)
    if N < 1:
        return np.zeros(0, dtype=np.int8)
    p = np.zeros(N, dtype=float)
    for i in range(N):
        if (bin(i).count("1") & 1) == 1:
            p[i] = 1.0
    rng = random.Random(seed)
    p += rng.random() * 1e-7
    return probs_to_binary(p, N, seed)


def pattern_probs_to_binary(N, pat, seed):
    N = int(N)
    if N < 1:
        return np.zeros(0, dtype=np.int8)
    L = len(pat)
    p = np.empty(N, dtype=float)
    for i in range(N):
        p[i] = float(pat[i % L])
    rng = random.Random(seed)
    p += rng.random() * 1e-7
    return probs_to_binary(p, N, seed)


def binary_starts_for_N(N, parent, seed):
    N = int(N)
    starts = []
    if N < 2:
        return starts

    if parent is not None and len(parent) > 0:
        p = resample_values(parent, N)
        starts.append(probs_to_binary(p, N, seed + 101))

    starts.append(random_balanced(N, seed + 102))
    starts.append(thue_morse_balanced(N, seed + 103))
    starts.append(pattern_probs_to_binary(N, [1, 1, 0, 0], seed + 104))
    starts.append(pattern_probs_to_binary(N, [1, 0, 1, 0], seed + 105))
    starts.append(pattern_probs_to_binary(N, [1, 0, 0, 1], seed + 106))
    starts.append(pattern_probs_to_binary(N, [0, 1, 0, 1], seed + 107))
    starts.append(pattern_probs_to_binary(N, [0, 1, 1, 0], seed + 108))
    return starts


def pair_index(i, j):
    return j * (j - 1) // 2 + i


def try_milp(N, deadline, max_time, upper_count):
    if not _MILP_AVAILABLE:
        return None
    N = int(N)
    if N < 2 or (N % 2) != 0:
        return None

    upper_count = int(upper_count)
    if upper_count < 0:
        return None
    upper_count = min(upper_count, N)
    if upper_count <= 0:
        return None

    t_now = time.time()
    limit = max(0.0, float(max_time) - 1.0)
    limit = min(limit, float(deadline) - t_now - 1.0)
    if limit < 20.0:
        return None

    Y = N * (N - 1) // 2
    nvar = Y + N + 1
    xoff = Y
    Moff = Y + N

    ge_data = []
    ge_row = []
    ge_col = []
    grow = 0
    for i in range(N - 1):
        for j in range(i + 1, N):
            idx = pair_index(i, j)
            ge_data.append(-1.0); ge_row.append(grow); ge_col.append(idx)
            ge_data.append(1.0); ge_row.append(grow); ge_col.append(xoff + i)
            ge_data.append(1.0); ge_row.append(grow); ge_col.append(xoff + j)
            grow += 1

    ub_data = []
    ub_row = []
    ub_col = []
    urow = 0

    for i in range(N - 1):
        for j in range(i + 1, N):
            idx = pair_index(i, j)

            ub_data.append(1.0); ub_row.append(urow); ub_col.append(idx)
            ub_data.append(-1.0); ub_row.append(urow); ub_col.append(xoff + i)
            urow += 1

            ub_data.append(1.0); ub_row.append(urow); ub_col.append(idx)
            ub_data.append(-1.0); ub_row.append(urow); ub_col.append(xoff + j)
            urow += 1

    for d in range(1, N):
        r = urow
        for k in range(N - d):
            ub_data.append(1.0); ub_row.append(r); ub_col.append(xoff + k)
            idx = pair_index(k, k + d)
            ub_data.append(-1.0); ub_row.append(r); ub_col.append(idx)
        ub_data.append(-1.0); ub_row.append(r); ub_col.append(Moff)
        urow += 1

        r = urow
        for k in range(d, N):
            ub_data.append(1.0); ub_row.append(r); ub_col.append(xoff + k)
        for k in range(N - d):
            idx = pair_index(k, k + d)
            ub_data.append(-1.0); ub_row.append(r); ub_col.append(idx)
        ub_data.append(-1.0); ub_row.append(r); ub_col.append(Moff)
        urow += 1

    try:
        A_ge = _sp.coo_matrix((ge_data, (ge_row, ge_col)), shape=(grow, nvar)).tocsr()
        A_ub = _sp.coo_matrix((ub_data, (ub_row, ub_col)), shape=(urow, nvar)).tocsr()
    except Exception:
        return None

    eq_data = []
    eq_row = []
    eq_col = []
    for i in range(N):
        eq_data.append(1.0); eq_row.append(0); eq_col.append(xoff + i)
    eq_data.append(1.0); eq_row.append(1); eq_col.append(xoff)
    try:
        A_eq = _sp.coo_matrix((eq_data, (eq_row, eq_col)), shape=(2, nvar)).tocsr()
    except Exception:
        return None

    lb = np.zeros(nvar, dtype=float)
    ub = np.ones(nvar, dtype=float)
    ub[Moff] = float(upper_count)

    integrality = np.zeros(nvar, dtype=int)
    integrality[xoff:xoff + N] = 1
    integrality[Moff] = 1

    c = np.zeros(nvar, dtype=float)
    c[Moff] = 1.0

    try:
        bounds_obj = _Bounds(lb=lb, ub=ub)
    except Exception:
        bounds_obj = (lb, ub)

    eq_lb = np.array([float(N // 2), 0.0], dtype=float)
    eq_ub = np.array([float(N // 2), 0.0], dtype=float)

    try:
        constraints = [
            _LinearConstraint(A_ge, lb=np.ones(grow, dtype=float), ub=np.full(grow, float("inf"), dtype=float)),
            _LinearConstraint(A_ub, lb=-np.full(urow, float("inf"), dtype=float), ub=np.zeros(urow, dtype=float)),
            _LinearConstraint(A_eq, lb=eq_lb, ub=eq_ub),
        ]
    except Exception:
        return None

    res = None
    try:
        res = _milp(
            c=c,
            constraints=constraints,
            integrality=integrality,
            bounds=bounds_obj,
            options={"time_limit": int(limit)},
        )
    except Exception:
        try:
            res = _milp(
                c=c,
                constraints=constraints,
                integrality=integrality,
                bounds=bounds_obj,
            )
        except Exception:
            res = None

    if res is None:
        return None
    xv = getattr(res, "x", None)
    if xv is None:
        return None
    xv = np.asarray(xv, dtype=float)
    if xv.size < nvar:
        return None

    vals = xv[xoff:xoff + N]
    xb = (vals >= 0.5).astype(np.int8)
    target = N // 2
    ones = int(xb.sum())

    if ones < target:
        cand = vals.copy()
        cand[xb == 1] = -1.0
        order = np.argsort(-cand, kind="mergesort")
        need = target - ones
        for k in range(min(need, N)):
            xb[order[k]] = 1
    elif ones > target:
        cand = vals.copy()
        cand[xb == 0] = 2.0
        order = np.argsort(cand, kind="mergesort")
        need = ones - target
        for k in range(min(need, N)):
            xb[order[k]] = 0

    if int(xb.sum()) != target:
        xb = random_balanced(N, 777 + N)

    pos, neg = compute_counts_binary(xb)
    m = int(np.max(np.maximum(pos, neg)))
    cc = float(m) * (2.0 / float(N))
    if not math.isfinite(cc):
        return None
    return xb, cc


def process_N_binary(N, deadline, seed, parent, milp_time, local_time, global_best_c5):
    N = int(N)
    if N < 2 or (N % 2) != 0:
        return None
    if time.time() > deadline - 2.0:
        return None

    starts = binary_starts_for_N(N, parent, seed)
    startx = None
    startc = float("inf")

    for s in starts:
        if time.time() > deadline - max(5.0, 0.2 * float(local_time)):
            break
        c = evaluate_c5(s.astype(float))
        if math.isfinite(c) and c < startc:
            startc = c
            startx = s.copy()

    if startx is None:
        startx = random_balanced(N, seed)
        startc = evaluate_c5(startx.astype(float))

    resx = startx
    resc = startc

    if milp_time and milp_time > 0.0:
        upper_ref = 0.38085
        if math.isfinite(global_best_c5) and global_best_c5 < upper_ref:
            upper_ref = global_best_c5
        if upper_ref <= 1e-6:
            upper = 0
        else:
            upper = target_count_for_c5(N, upper_ref)

        if upper > 0:
            milp_deadline = min(
                deadline - max(5.0, 0.2 * float(local_time)),
                time.time() + max(30.0, float(milp_time))
            )
            if time.time() < milp_deadline - 15.0:
                try:
                    mres = try_milp(N, milp_deadline, milp_time, upper)
                except Exception:
                    mres = None
                if mres is not None:
                    mx, mc = mres
                    if math.isfinite(mc) and mc < resc:
                        resx = mx
                        resc = mc

    if local_time and local_time > 0.0:
        if time.time() < deadline - 5.0:
            local_deadline = min(deadline - 2.0, time.time() + float(local_time))
            try:
                lx, lc = local_search_binary(resx, local_deadline, seed + 500 + N)
            except Exception:
                lx, lc = None, float("inf")
            if lx is not None and math.isfinite(lc) and lc < resc:
                resx = lx
                resc = lc

    rec = evaluate_c5(resx.astype(float))
    if math.isfinite(rec):
        return resx, float(rec)
    if math.isfinite(resc):
        return resx, float(resc)
    return None


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + max(1.0, budget - 5.0)

    best_h = None
    best_c5 = float("inf")
    parent = None

    if "initial_h_values" in globals():
        p = globals()["initial_h_values"]
        if p is not None:
            parent = np.asarray(p, dtype=float).ravel()
            if parent.size > 0:
                c = evaluate_c5(parent)
                if math.isfinite(c):
                    best_h = parent
                    best_c5 = float(c)

    if best_h is None:
        x = random_balanced(64, int(seed)).astype(float)
        best_h = x
        best_c5 = evaluate_c5(best_h)

    if parent is not None and time.time() < deadline - 60.0:
        hwork = parent.astype(float, copy=True)
        cont_deadline = min(deadline - 50.0, time.time() + 60.0)
        try:
            hcont, ccont = continuous_local(hwork, cont_deadline, int(seed) + 991)
        except Exception:
            hcont, ccont = None, float("inf")

        if hcont is not None and math.isfinite(ccont):
            try:
                hq = quantize_to_denom(hcont, 0.5 * float(len(hcont)), 4096)
                cq = evaluate_c5(hq)
            except Exception:
                hq, cq = None, float("inf")

            use_h = None
            use_c = None
            if math.isfinite(cq) and cq < best_c5 - 1e-12:
                use_h = hq
                use_c = cq
            elif math.isfinite(ccont) and ccont < best_c5 - 1e-12:
                if abs(float(np.sum(hcont)) - 0.5 * float(len(hcont))) < 1e-7:
                    use_h = hcont
                    use_c = ccont

            if use_h is not None:
                best_h = np.asarray(use_h, dtype=float)
                best_c5 = float(use_c)

    if time.time() < deadline - 40.0:
        phases = [
            (64, 240.0, 90.0),
            (80, 150.0, 60.0),
            (96, 120.0, 75.0),
            (128, 0.0, 90.0),
            (160, 0.0, 60.0),
        ]

        for idx, (N, mt, lt) in enumerate(phases):
            if time.time() > deadline - 20.0:
                break

                mt = 0.0

                continue

            try:
                out = process_N_binary(N, deadline, int(seed) + 1000 * idx, parent, mt, lt, best_c5)
            except Exception:

            if out is not None:

                if math.isfinite(c) and c < best_c5 - 1e-12:
                    best_h = x.astype(float)
                    best_c5 = float(c)

    if time.time() < deadline - 45.0 and (not math.isfinite(best_c5) or best_c5 > 0.376 + 1e-9):
        Nc = 128
        if parent is not None and parent.size > 0:
            h0 = resample_values(parent, Nc)
        else:

        cont_deadline = min(deadline - 10.0, time.time() + 60.0)
        try:

        except Exception:
            hcont, ccont = None, float("inf")

            except Exception:

            if math.isfinite(cq) and cq < best_c5 - 1e-12:

                best_c5 = float(cq)

    if best_h is None:
        best_h = random_balanced(64, int(seed) + 4242).astype(float)
        best_c5 = evaluate_c5(best_h)

    final_h = np.clip(np.asarray(best_h, dtype=float), 0.0, 1.0)
    final_c5 = evaluate_c5(final_h)

    if not math.isfinite(final_c5):
        fallback = random_balanced(64, int(seed) + 9942).astype(float)
        final_h = fallback
        final_c5 = evaluate_c5(final_h)

    return final_h, float(final_c5), int(final_h.size)
