import numpy as np
from scipy.optimize import minimize
import time

_opt_state = {}
_cb_state = {}

def smooth_max(corr, tau):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / tau

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    s = float(np.sum(h))
    diff = target - s
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(h)
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        order = np.argsort(h)[::-1]
        for idx in order:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
            if diff >= -1e-12:
                break
    return h

def resample_h(h_from, n_from, n_to):
    x_old = np.linspace(0.0, 2.0, n_from, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_to, endpoint=False)
    h_ext = np.concatenate([h_from, h_from, h_from])
    x_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_new = np.interp(x_new, x_ext, h_ext)
    return repair(h_new, n_to)

def lbfgs_obj_func(h_vec):
    tau = _opt_state['tau']
    n = _opt_state['n']
    pw = _opt_state['penalty_weight']
    h_vec = np.clip(h_vec, 0.0, 1.0)
    corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
    obj = smooth_max(corr, tau) * (2.0 / n)
    penalty = pw * (float(np.sum(h_vec)) - n / 2.0) ** 2
    return obj + penalty

def lbfgs_callback(k):
    if (time.time() - _cb_state['t_start']) > _cb_state['max_seconds']:
        raise RuntimeError("TIME_EXCEEDED")

def run_lbfgs(h_init, n, tau, penalty_weight, max_seconds):
    _opt_state['tau'] = tau
    _opt_state['n'] = n
    _opt_state['penalty_weight'] = penalty_weight
    _cb_state['t_start'] = time.time()
    _cb_state['max_seconds'] = max_seconds
    
    h_current = h_init.copy()
    try:
        result = minimize(
            lbfgs_obj_func, h_current,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            callback=lbfgs_callback,
            options={'maxiter': 3000, 'maxfun': 100000,
                     'ftol': 1e-15, 'gtol': 1e-10}
        )
        h_current = result.x
    except RuntimeError:
        pass
    except Exception:
        pass
    
    return repair(h_current, n)

def coordinate_descent(h_init, n, max_seconds, n_passes=50):
    h = h_init.copy()
    c5_cur = exact_c5(h)
    best = h.copy()
    best_c5 = c5_cur
    t_start = time.time()
    
    for p in range(n_passes):
        if (time.time() - t_start) > max_seconds:
            break
        improved = False
        step = 0.05 / (1.0 + 0.1 * p)
        for i in range(n):
            if (time.time() - t_start) > max_seconds:
                break
            for delta in [step, -step, step*0.5, -step*0.5]:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                h_try = repair(h_try, n)
                c5_try = exact_c5(h_try)
                if c5_try < c5_cur - 1e-12:
                    h = h_try
                    c5_cur = c5_try
                    improved = True
                    break
                if delta == -step*0.5 and c5_try < c5_cur:
                    break
        if not improved:
            step *= 0.7
    
    h = repair(h, n)
    c5_final = exact_c5(h)
    if c5_final < best_c5:
        best = h.copy()
        best_c5 = c5_final
    return best, best_c5

def simulated_annealing(h_init, n, max_seconds, cool_rate=0.9997):
    h = h_init.copy()
    c5_cur = exact_c5(h)
    best = h.copy()
    best_c5 = c5_cur
    t_start = time.time()
    T = 0.05
    nsteps = 0
    
    while (time.time() - t_start) < max_seconds:
        T *= cool_rate
        if T < 0.001:
            T = 0.001
        n_pert = np.random.randint(1, 4)
        h_try = h.copy()
        for _ in range(n_pert):
            idx = np.random.randint(0, n)
            delta = np.random.uniform(-0.2 * T - 0.01, 0.2 * T + 0.01)
            h_try[idx] = np.clip(h_try[idx] + delta, 0.0, 1.0)
        h_try = repair(h_try, n)
        c5_try = exact_c5(h_try)
        
        if c5_try < c5_cur:
            h = h_try
            c5_cur = c5_try
            if c5_try < best_c5:
                best = h_try.copy()
                best_c5 = c5_try
        else:
            accept_prob = np.exp(-(c5_try - c5_cur) / (T * 0.005))
            if np.random.random() < accept_prob:
                h = h_try
                c5_cur = c5_try
        nsteps += 1
        
        if nsteps % 200 == 0 and T < 0.01:
            h_cd, c5_cd = coordinate_descent(h, n, 5.0, n_passes=5)
            if c5_cd < best_c5:
                best = h_cd.copy()
                best_c5 = c5_cd
            h = h_cd.copy()
            c5_cur = c5_cd
    
    return best, best_c5

            break
        
        h_init = resample_h(best_h, best_n, n_target)
