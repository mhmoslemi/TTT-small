import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_simplex(h, n):
    target = 0.5 * n
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        h_p = np.clip(h - tau, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    h_p = np.clip(h - tau, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-13:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h_p[mask] += diff / cnt
        else:
            h_p += diff / n
        h_p = np.clip(h_p, 0.0, 1.0)
    return h_p


def penalty_objective(x, dx_val, n_val, lam_val):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx_val
    pen = lam_val * (np.sum(h) - 0.5 * n_val) ** 2
    return c5 + pen


def penalty_obj_grad(x, dx_val, n_val, lam_val):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx_val
    pen = lam_val * (np.sum(h) - 0.5 * n_val) ** 2
    return c5 + pen


def make_beatty(n, alpha, thresh):
    h = np.zeros(n)
    for i in range(n):
        frac = (i * alpha) % 1.0
        if frac < thresh:
            h[i] = 1.0
    return project_simplex(h, n)


def make_sinusoidal(n, freq, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x + phase)
    return project_simplex(h, n)


def make_blocks(n, nblocks):
    h = np.zeros(n)
    blen = float(n) / (2.0 * nblocks)
    for b in range(2 * nblocks):
        s = int(round(b * blen))
        e = int(round((b + 1) * blen))
        if b % 2 == 0:
            h[s:e] = 1.0
    return project_simplex(h, n)


def make_concatenation(n, pieces):
    h = np.zeros(n)
    idx = 0
    for (val, count) in pieces:
        end = min(idx + count, n)
        h[idx:end] = val
        idx = end
    return project_simplex(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = np.inf
    best_h = None
    best_n = 70

    def time_left():
        return budget_s - (time.time() - t0)

    has_parent = 'initial_h_values' in globals() and initial_h_values is not None
    parent_arr = None
    parent_n = 0
    if has_parent:
        parent_arr = np.asarray(initial_h_values, dtype=float)
        parent_n = len(parent_arr)

    n_sizes = [70, 80, 90, 100, 120, 150, 200, 250, 300, 400, 500]
    if has_parent and parent_n not in n_sizes:
        n_sizes.insert(0, parent_n)
    if has_parent and parent_n in n_sizes:
        idx0 = n_sizes.index(parent_n)
        n_sizes = [n_sizes[idx0]] + [x for x in n_sizes if x != parent_n]

    for n in n_sizes:
        if time_left() < 45:
            break

        dx = 2.0 / n
        lam = 500.0
        bounds = [(0.0, 1.0)] * n

        inits = []

        if has_parent and n == parent_n:
            inits.append(project_simplex(parent_arr, n))
        elif has_parent:
            xs = np.linspace(0, parent_n - 1, n)
            resized = np.interp(xs, np.arange(parent_n), parent_arr)
            inits.append(project_simplex(resized, n))

                    bounds=bounds,
                    options={'maxiter': n_iter_per_init, 'ftol': 1e-12, 'gtol': 1e-6, 'maxfun': n_iter_per_init * 20}
                )
                h_opt = project_simplex(res.x, n)
                c5_val = compute_c5(h_opt, dx)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
                    best_n = n
            except Exception:
                pass

        if best_n == n and best_h is not None and time_left() > 25:
            n_refine = min(10, int(time_left() / 5.0))
            for _ in range(n_refine):
                if time_left() < 8:
                    break
                sigma = 0.02 + 0.04 * rng.rand()
                h_pert = best_h + rng.normal(0, sigma, n)

                    res = minimize(

                        options={'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-6}
                    )
                    h_opt = project_simplex(res.x, n)
                    c5_val = compute_c5(h_opt, dx)

                    pass

    if best_h is None:

        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h, 2.0 / n)
        best_n = n

    return best_h, best_c5, best_n
