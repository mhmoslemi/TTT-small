    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Adjust sum to target
    diff = h.sum() - target
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to reduce sum by diff
        # Reduce from the largest elements first
        order = np.argsort(-h)
        h[order] -= diff / n  # uniform reduction

    lo = h.min() - 1
    hi = h.max() + 1

import time

def compute_c5(h, dx):

    conv = np.correlate(h, 1-h, mode='full')

def smooth_c5(h, dx, beta):

    conv = np.correlate(h, 1-h, mode='full')
    x = beta * conv * dx
    # For numerical stability
    x_max = np.max(x)
    if x_max > 500:
        return x_max / beta + np.log(np.sum(np.exp(x - x_max))) / beta

def smooth_c5_grad(h, dx, beta):

    conv = np.correlate(h, 1-h, mode='full')
    x = beta * conv * dx
    x_max = np.max(x)

            # c_lag = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])

            #   from h[i] term: (1-h[j+lag]) if j <= n-1-lag
            #   from (1-h[i+lag]) term: -h[j-lag] if j >= lag
            for j in range(n):

                if j <= n - 1 - lag:
                    dc += (1 - h[j + lag])
                if j >= lag:
                    dc -= h[j - lag]
                grad[j] += weights[l] * dc * dx
        else:  # lag < 0

                    dc += (1 - h[j - k])
                if j <= n - 1 - k:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def project_box_sum(h, target):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target}"""
    # Bisection on lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Verify and fix
    diff = result.sum() - target
    if abs(diff) > 1e-10:
        # Small correction
        mask = (result > 0) & (result < 1)
        if mask.sum() > 0:
            result[mask] -= diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

        h2 = _interpolate_to_resolution(best_h, n, n2)
        h2 = _project_box_sum(h2, target2)
        for beta in [50, 100, 200, 400]:
            if not time_ok(80):
                break
            h2 = _optimize_h_lbfgsb(h2, n2, dx2, target2, beta, penalty, maxiter=300)
            h2 = _project_box_sum(h2, target2)
            c5 = _compute_c5(h2, dx2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 3: Random restarts at best resolution
    n_best = len(best_h)
    dx_best = 2.0 / n_best
    target_best = n_best / 2.0
    scales = [0.02, 0.05, 0.1, 0.15, 0.08, 0.03, 0.12, 0.06]
    attempt = 0
    while time_ok(20) and attempt < 30:
        scale = scales[attempt % len(scales)]
        h_pert = best_h + np.random.randn(n_best) * scale
        h_pert = _project_box_sum(h_pert, target_best)
        h_pert = _optimize_h_lbfgsb(h_pert, n_best, dx_best, target_best, 150, penalty, maxiter=150)
        h_pert = _project_box_sum(h_pert, target_best)
        c5 = _compute_c5(h_pert, dx_best)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert.copy()
        attempt += 1

    # Final verification and cleanup
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _compute_c5(best_h, 2.0 / len(best_h))
    # Ensure sum constraint is met exactly
    best_h = _project_box_sum(best_h, len(best_h) / 2.0)
    final_c5 = _compute_c5(best_h, 2.0 / len(best_h))

    return best_h, final_c5, len(best_h)
