  Change in C_d: was h[a-d]*(1-1) = 0, now h[a-d]*(1-0) = h[a-d]

ΔC_d = -[1-h[a+d]] * I(-a ≤ d ≤ n-1-a) + h[a-d] * I(a-n+1 ≤ d ≤ a)

def swap_delta(C, h, a, b, n):
    """Compute delta C when swapping a(1->0) and b(0->1).
    a must be a 1, b must be a 0."""
    delta = np.zeros(2*n - 1)
    center = n - 1
    
    # a: 1 -> 0
    # As h[i] first factor: d in [-a, n-1-a], change = -(1-h[a+d])
    d_lo = max(-(n-1), -a)
    d_hi = min(n-1, n-1-a)

        a_plus_d = a + np.arange(d_lo, d_hi + 1)
        delta[idx] -= (1.0 - h[a_plus_d])

Actually, ∫h(1-h)dx = ∫h dx - ∫h² dx = 1 - ∫h² dx.
By Cauchy-Schwarz, ∫h² dx ≥ (∫h)²/2 = 1/2. So ∫h(1-h)dx ≤ 1 - 1/2 = 1/2.
Equality when h is constant = 1/2 everywhere.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)

def make_binary(h, n):
    """Make h binary with exactly n//2 ones, preserving order of h values."""
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:n // 2]] = 1.0
    return h_bin

def upscale(h, n_old, n_new):
    """Upscale h from n_old to n_new points using linear interpolation."""
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # cell centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    return np.interp(x_new, x_old, h)

def swap_delta(C, h, a, b, n):
    """Compute change in correlation when h[a]: 1->0 and h[b]: 0->1."""
    delta = np.zeros(2 * n - 1)
    center = n - 1
    
    # a: 1 -> 0, as first factor h[i], i=a
    # C_d += h[a]*(1-h[a+d]) for valid d; now h[a]=0, so change = -(1-h[a+d])
    d_lo = max(-(n-1), -a)
    d_hi = min(n-1, n-1-a)
    if d_lo <= d_hi:
        idx = np.arange(d_lo + center, d_hi + center + 1)
        delta[idx] -= (1.0 - h[a + np.arange(d_lo, d_hi + 1)])
    
    # a: 1 -> 0, as second factor (1-h[i]), i+d=a, i=a-d
    # C_d += h[a-d]*(1-h[a]) = h[a-d]*0 = 0; now (1-h[a])=1, change = +h[a-d]
    d_lo = max(-(n-1), a - n + 1)
    d_hi = min(n-1, a)
    if d_lo <= d_hi:
        idx = np.arange(d_lo + center, d_hi + center + 1)
        delta[idx] += h[a - np.arange(d_lo, d_hi + 1)]
    
    # b: 0 -> 1, as first factor h[i], i=b
    # C_d += h[b]*(1-h[b+d]) = 0; now h[b]=1, change = +(1-h[b+d])
    d_lo = max(-(n-1), -b)
    d_hi = min(n-1, n-1-b)
    if d_lo <= d_hi:
        idx = np.arange(d_lo + center, d_hi + center + 1)
        delta[idx] += (1.0 - h[b + np.arange(d_lo, d_hi + 1)])
    
    # b: 0 -> 1, as second factor (1-h[i]), i+d=b, i=b-d
    # C_d += h[b-d]*(1-h[b]) = h[b-d]*1; now (1-h[b])=0, change = -h[b-d]
    d_lo = max(-(n-1), b - n + 1)
    d_hi = min(n-1, b)
    if d_lo <= d_hi:
        idx = np.arange(d_lo + center, d_hi + center + 1)
        delta[idx] -= h[b - np.arange(d_lo, d_hi + 1)]
    
    return delta

def local_search_binary(h, n, max_passes, time_limit, rng):
    h = h.copy()
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(C.max())
    best_c5 = current_max * (2.0 / n)
    
    start = time.time()
    no_improve = 0
    
    for p in range(max_passes):
        if time_limit is not None and (time.time() - start) > time_limit:
            break
        
        improved = False
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        # Random permutation of swap pairs
        num_swaps = len(ones_idx) * len(zeros_idx)
        for _ in range(num_swaps):
            i = rng.randint(len(ones_idx))
            j = rng.randint(len(zeros_idx))
            a, b = int(ones_idx[i]), int(zeros_idx[j])
            
            delta = swap_delta(C, h, a, b, n)
            new_max = float((C + delta).max())
            if new_max < current_max - 1e-12:
                C += delta
                h[a] = 0.0
                h[b] = 1.0
                current_max = new_max
                best_c5 = current_max * (2.0 / n)
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return h, best_c5
