def project_box_sum(z, target):
    z = np.asarray(z, dtype=float)
    n = z.size
    if n == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=1.0, neginf=0.0)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)
    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    if hi <= lo:
        hi = lo + 1.0
    for _ in range(45):
        mid = 0.5*(lo+hi)
        x = z - mid
        np.clip(x, 0.0, 1.0, out=x)
        s = float(x.sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5*(lo+hi)
    x = z - lam
    np.clip(x, 0.0, 1.0, out=x)
    # residual adjust
    for _ in range(5):
        s = float(x.sum())
        res = target - s
        if abs(res) <= 1e-12:
            break
        if res > 0.0:
            can = x < 1.0
            cnt = int(can.sum())
            if cnt == 0:
                break
            add = min(res / float(cnt), float(np.max(1.0 - x[can])))
            x[can] += add
        else:
            can = x > 0.0
            cnt = int(can.sum())
            if cnt == 0:
                break
            dec = min(-res / float(cnt), float(np.max(x[can])))
            x[can] -= dec
    return x

def dyadic_feasible(h,n):
   h=np.asarray(h,dtype=float).reshape(-1)
   if h.size==0: return h
   if n <=0: return h
   scale=1<<30
   sf=float(scale)
   h=np.clip(h,0,1)
   target_int=int(round(0.5*n*sf))
   scaled=h*sf
   ints=np.rint(scaled).astype(np.int64)
   ints=np.clip(ints,0,scale)
   res=target_int-int(ints.sum())
   frac=scaled-np.floor(scaled)
   if res!=0:
      order=np.argsort(frac, kind='mergesort')
      if res>0:
         for idx in order[::-1]:
            if res<=0: break
            ii=int(idx)
            if ints[ii] < scale:
               ints[ii]+=1; res-=1
      else:
         for idx in order:
            if res>=0: break
            ii=int(idx)
            if ints[ii] > 0:
               ints[ii]-=1; res+=1
   out = ints.astype(float) * (1.0/sf)
   return out

        return np.empty(0, dtype=int), np.empty(0, dtype=float)
    if top <= 0:
        top = 1
    if top >= m:
        part = np.arange(m, dtype=int)
    else:
        part = np.argpartition(q, -top)[-top:]
        order = np.argsort(q[part])[::-1]
        part = part[order]
    part = part.astype(int, copy=False)
    qpart = q[part]
    if part.size == 1:
        weights = np.ones(1, dtype=float)
    else:
        gap = float(qpart[0] - qpart[-1])
        if not np.isfinite(gap) or gap <= 1e-12:
            weights = np.ones(part.size, dtype=float) / float(part.size)
        else:
            smooth = 0.25 * gap
            if smooth < 1e-12:
                smooth = 1e-12
            ex = (qpart - qpart[0]) / smooth
            np.clip(ex, -50.0, 0.0, out=ex)
            weights = np.exp(ex)
            wsum = float(weights.sum())
            if not np.isfinite(wsum) or wsum <= 0.0:
                weights = np.ones(part.size, dtype=float) / float(part.size)
            else:
                weights = weights / wsum
    return part, weights.astype(float, copy=False)

def compute_active_gradient(h, part, weights, dx):
    n = int(h.size)
    grad = np.zeros(n, dtype=float)
    offset = n - 1
    for ii in range(int(part.size)):
        idx = int(part[ii])
        w = float(weights[ii])
        if w == 0.0:
            continue
        k = idx - offset
        if k >= 0:
            m = k
            cnt = n - m
            if cnt <= 0:
                continue
            if m == 0:
                grad += w * dx * (1.0 - 2.0 * h)
            else:
                grad[m:m + cnt] += w * dx * (1.0 - h[:cnt])
                grad[:cnt] += -w * dx * h[m:m + cnt]
        else:
            m = -k
            cnt = n - m
            if cnt <= 0:
                continue
            grad[:cnt] += w * dx * (1.0 - h[m:m + cnt])
            grad[m:m + cnt] += -w * dx * h[:cnt]
    return grad

def continuous_refine(h_start, n, deadline_abs, rng, top_start=8, max_top=32):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float), float("inf")
    h = np.asarray(h_start, dtype=float).reshape(-1)
    if h.size != n:
        h = resample_h(h, n)
    h = prepare_feasible(h)
    if h.size != n:
        return h, c5_value(h)

    target = 0.5 * float(n)
    dx = 2.0 / float(n)
    current = c5_value(h)
    best_h = h.copy()
    best_c = current

    step = 0.08
    top = int(max(1, top_start))
    max_top = int(max(1, max_top))
    stuck = 0
    jumps = 0
    max_jumps = 14
    end = float(deadline_abs) - 2.0

    while safe_now() < end:
        q = correlate_full(h) * dx
        cmax = float(np.max(q))
        if cmax < current - 1e-12:
            current = cmax
            best_h = h.copy()
            best_c = cmax
            stuck = 0
        elif cmax < best_c - 1e-12:
            best_h = h.copy()
            best_c = cmax
            stuck = 0
        else:
            stuck += 1

        if stuck > 40:
            if top < max_top:
                top = min(max_top, top * 2)
                stuck = 0
            elif jumps < max_jumps:
                sigma = 0.12 / float(jumps + 1)
                h = project_box_sum(h + rng.standard_normal(n) * sigma, target)
                current = c5_value(h)
                jumps += 1
                stuck = 0
                top = int(max(1, top_start))
                step = 0.08
                if current < best_c:
                    best_h = h.copy()
                    best_c = current
                continue

        part, weights = select_active(q, top)
        grad = compute_active_gradient(h, part, weights, dx)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm <= 1e-14:
            if jumps < max_jumps:
                h = project_box_sum(h + rng.standard_normal(n) * 0.05, target)
                current = c5_value(h)
                jumps += 1
                continue
            break

        direction = -grad / gnorm
        accepted = False
        for fac in (1.0, 0.5, 0.25, 0.1):
            z = h + step * fac * direction
            zn = project_box_sum(z, target)
            ct = c5_value(zn)
            if ct < current - 1e-13 or ct < best_c - 1e-13:
                h = zn
                current = ct
                if ct < best_c - 1e-13:
                    best_h = h.copy()
                    best_c = ct
                step = min(step * 1.05, 0.25)
                accepted = True
                break
        if not accepted:
            step = max(step * 0.7, 1e-4)

    return best_h, best_c

def bits_from_array(arr, n):
    bits = 0
    for i in range(int(n)):
        if arr[i] > 0.5:
            bits |= (1 << i)
    return bits

def bits_to_array(bits, n):
    out = np.zeros(int(n), dtype=float)
    for i in range(int(n)):
        if (bits >> i) & 1:
            out[i] = 1.0
    return out

def bits_c5(bits, n, masks, allmask):
    n = int(n)
    if n <= 1:
        return 0.0
    zeros = allmask ^ bits
    best = 0
    for m in range(1, n):
        mask = masks[m]
        shifted = bits >> m
        c10 = (bits & (mask ^ shifted)).bit_count()
        c01 = (zeros & shifted).bit_count()
        if c10 > best:
            best = c10
        if c01 > best:
            best = c01
    return float(best) * (2.0 / float(n))

def binary_local_search(bits, n, deadline_abs, rng):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float), float("inf")
    if (n % 2) != 0:
        arr = bits_to_array(bits, n)
        arr = prepare_feasible(arr)
        return arr, c5_value(arr)

    target = n // 2
    if bits.bit_count() != target:
        ones = 0
        for i in range(n):
            if (bits >> i) & 1:
                ones += 1
        while ones < target:
            i = int(rng.integers(n))
            if not ((bits >> i) & 1):
                bits |= (1 << i)
                ones += 1
        while ones > target:
            i = int(rng.integers(n))
            if ((bits >> i) & 1):
                bits &= ~(1 << i)
                ones -= 1

    allmask = (1 << n) - 1
    masks = [0] * n
    for m in range(1, n):
        masks[m] = (1 << (n - m)) - 1

    cur = bits_c5(bits, n, masks, allmask)
    best_bits = bits
    best_c = cur
    temp = 1.0 / float(n)
    min_temp = 1e-6
    no_best = 0
    it = 0
    end = float(deadline_abs) - 2.0

    while safe_now() < end:
        it += 1
        if (it & 2047) == 0:
            if safe_now() >= end:
                break
            if no_best > 4096:
                temp = max(min_temp, temp * 0.9)
                no_best = 0

        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if ((bits >> i) & 1) == ((bits >> j) & 1):
            continue

        nb = bits ^ (1 << i) ^ (1 << j)
        nc = bits_c5(nb, n, masks, allmask)

        if nc < cur - 1e-15:
            bits = nb
            cur = nc
            if cur < best_c - 1e-15:
                best_bits = bits
                best_c = cur
                no_best = 0
            else:
                no_best += 1
        else:
            diff = nc - cur
            if diff <= 0.0:
                bits = nb
                cur = nc
            elif diff < 20.0 * temp:
                p = math.exp(-diff / temp)
                if rng.random() < p:
                    bits = nb
                    cur = nc
            no_best += 1

    return bits_to_array(best_bits, n), best_c

def dyadic_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    if n <= 0:
        return h
    scale = 1 << 30
    sf = float(scale)
    h = np.clip(h, 0.0, 1.0)
    target_int = int(round(0.5 * float(n) * sf))
    scaled = h * sf
    ints = np.rint(scaled).astype(np.int64)
    ints = np.clip(ints, 0, scale)
    res = target_int - int(ints.sum())

    if res != 0:
        frac = scaled - np.floor(scaled)
        order = np.argsort(frac, kind="mergesort")
        if res > 0:
            for idx in order[::-1]:
                if res <= 0:
                    break
                ii = int(idx)
                if ints[ii] < scale:
                    ints[ii] += 1
                    res -= 1
        else:
            for idx in order:
                if res >= 0:
                    break
                ii = int(idx)
                if ints[ii] > 0:
                    ints[ii] -= 1
                    res += 1

    out = ints.astype(float) * (1.0 / sf)
    return out

def finalize_candidate(h, n):
    if h is None:
        return None, float("inf"), 0
    h = np.asarray(h, dtype=float).reshape(-1)
    n = int(n)
    if h.size != n:
        h = resample_h(h, n)
    h = prepare_feasible(h)
    c = c5_value(h)
    sum_err = abs(float(np.sum(h)) - 0.5 * float(n))

    if sum_err < 1e-7:
        return h, c, n

    try:
        hq = dyadic_feasible(h, n)
        cq = c5_value(hq)
        if np.isfinite(cq):
            return hq, cq, n
    except Exception:
        pass
    return h, c, n

def run(seed=42, budget_s=1300, **kwargs):
    t0 = safe_now()
    deadline = t0 + float(budget_s) - 10.0
    rng = np.random.default_rng(seed)

    parent_h = None
    parent_n = 0
    parent_c = float("inf")

    try:
        raw = initial_h_values
    except NameError:
        raw = None

    if raw is not None:
        p = None
        try:
            p = np.asarray(raw, dtype=float)
        except Exception:
            try:
                p = np.asarray(raw[0], dtype=float)
            except Exception:
                p = None

        if p is not None:
            if p.ndim != 1:
                p = p.reshape(-1)
            pn = int(p.size)
            if pn > 0 and np.all(np.isfinite(p)):
                p = prepare_feasible(p)
                pc = c5_value(p)
                if np.isfinite(pc):
                    parent_h = p.copy()
                    parent_n = pn
                    parent_c = pc

    if parent_h is not None:
        best_h = parent_h.copy()
        best_c = parent_c
        best_n = parent_n
    else:
        base_n = 160
        best_h = random_balanced(base_n, rng)
        best_h = prepare_feasible(best_h)
        best_c = c5_value(best_h)
        best_n = base_n

    if deadline - safe_now() > 90 and (int(best_n) % 2 == 0):
        balloc = min(120.0, max(40.0, 0.10 * (deadline - safe_now())))
        end_bin = safe_now() + balloc
        th = threshold_binary(best_h)
        bits = bits_from_array(th, int(best_n))
        hb, cb = binary_local_search(bits, int(best_n), end_bin, rng)
        if np.isfinite(cb) and cb < best_c:
            best_h = hb
            best_c = cb
            best_n = int(best_n)

    if deadline - safe_now() > 90:
        base = int(best_n) if best_n > 0 else 160
        if base < 16:
            base = 80

        max_steps = 5 if base <= 128 else (4 if base <= 640 else 3)
        max_n = 1280 if base <= 128 else 2048

        ns = []
        v = base
        while len(ns) < max_steps and v > 0:
            ns.append(int(v))
            v2 = int(v) * 2
            if v2 <= v or v2 > max_n:
                break
            v = v2
        if len(ns) == 0:
            ns = [base]

        L = len(ns)
        if L == 1:
            weights = [1.0]
        elif L == 2:
            weights = [0.35, 0.65]
        elif L == 3:
            weights = [0.20, 0.30, 0.50]
        elif L == 4:
            weights = [0.15, 0.20, 0.28, 0.37]
        else:
            weights = [0.10, 0.15, 0.20, 0.25, 0.30]

        for ii, nn in enumerate(ns):
            if safe_now() >= deadline - 25:
                break
            rem_w = float(sum(weights[ii:]))
            rem_t = max(0.0, deadline - safe_now())
            alloc = rem_t * (float(weights[ii]) / rem_w if rem_w > 0 else 0.0)
            if alloc < 35:
                break
            t_iter = safe_now()
            n_iter_end = t_iter + alloc

            if int(best_h.size) == int(nn):
                start0 = prepare_feasible(best_h.copy())
            else:
                start0 = resample_h(best_h, int(nn))

            split0 = 0.68 if nn <= 512 else 0.75
            end0 = min(t_iter + alloc * split0, n_iter_end - 12)
            hb, cb = continuous_refine(start0, int(nn), end0, rng)
            if np.isfinite(cb) and cb < best_c:
                best_h = hb
                best_c = cb
                best_n = int(nn)

            remaining = n_iter_end - safe_now()
            if remaining > 70:
                if int(nn) % 2 == 0:
                    start1 = threshold_binary(start0)
                else:
                    start1 = random_balanced(int(nn), rng)
                    start1 = prepare_feasible(start1)
                use1 = min(0.32 * alloc, 0.6 * remaining)
                end1 = min(safe_now() + use1, n_iter_end)
                h1, c1 = continuous_refine(start1, int(nn), end1, rng)
                if np.isfinite(c1) and c1 < best_c:
                    best_h = h1
                    best_c = c1
                    best_n = int(nn)

                remaining = n_iter_end - safe_now()
                if remaining > 55:
                    start2 = gaussian_blur(start0, max(1.0, float(nn) / 32.0))
                    use2 = max(0.0, n_iter_end - safe_now())
                    h2, c2 = continuous_refine(start2, int(nn), n_iter_end - 5, rng)
                    if np.isfinite(c2) and c2 < best_c:
                        best_h = h2
                        best_c = c2
                        best_n = int(nn)

    candidates = []
    if parent_h is not None:
        candidates.append(finalize_candidate(parent_h, parent_n))
    if best_h is not None:
        candidates.append(finalize_candidate(best_h, best_n))

    chosen = None
    best_final_c = float("inf")
    for cand in candidates:
        ch, cc, cn = cand
        if ch is None or not np.isfinite(cc):
            continue
        if cc < best_final_c:
            best_final_c = cc
            chosen = (ch, cn)

    if chosen is None:
        fallback_n = 160
        fallback_h = random_balanced(fallback_n, rng)
        fallback_h = prepare_feasible(fallback_h)
        fallback_c = c5_value(fallback_h)
        return fallback_h, float(fallback_c), int(fallback_n)

    h_final, n_final = chosen
    if h_final.size != int(n_final):
        h_final = resample_h(h_final, int(n_final))
    else:
        h_final = np.clip(np.asarray(h_final, dtype=float), 0.0, 1.0)

    c_final = c5_value(h_final)
    return h_final, float(c_final), int(n_final)
