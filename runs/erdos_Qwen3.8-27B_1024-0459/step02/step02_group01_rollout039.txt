def binary_local_search(h, n, time_limit):
    # Convert to binary
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        # Remove excess ones
        ones = np.where(h_bin == 1)[0]
        # Remove the ones with lowest h value
        order = np.argsort(h[ones])[:n_ones - n_want]
        h_bin[ones[order]] = 0

                h[idx] -= rem

Let me also think about the binary local search approach more carefully:

For a binary sequence, the objective is:
C₅ = max_k (sum of h[i]*(1-h[i+k]) for valid i) * 2/n

- Compute all correlations: O(n²)
- Find the worst shift k*
- Propose a swap: pick i where h[i]=1 and j where h[j]=0
- The change in corr[k*] from this swap can be computed in O(1) (just check the 4 affected terms)
- Accept if it improves the overall max

So each swap evaluation is O(n), and we try O(n) candidates per iteration, giving O(n²) per iteration. For n=384, that's 147k per iteration. We can do about 10k iterations in 1300 seconds. That should be enough for local search.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nfree = np.sum(free)
        if nfree > 0:
            h[free] += diff / nfree
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                rem = min(-diff, h[idx])
                h[idx] -= rem
                diff += rem
            if abs(diff) < 1e-14:
                break
    return h

def smooth_objective(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    logsumexp = m + np.log(np.sum(np.exp(shifted))) / beta
    return logsumexp * (2.0 / n)

def smooth_gradient(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for ki in np.where(w > 1e-14)[0]:
        wi = w[ki]
        k = ki - (n - 1)
        if k >= 0:
            L = k
            grad[L:] += wi * (1.0 - h[:n-L])  # (1-h[j-L]) for j=L..n-1
            grad[:n-L] -= wi * h[L:]  # -h[j+L] for j=0..n-1-L
        else:
            M = -k
            grad[:n-M] += wi * (1.0 - h[M:])  # (1-h[j+M]) for j=0..n-1-M
            grad[M:] -= wi * h[:n-M]  # -h[j-M] for j=M..n-1
    
    return grad * (2.0 / n)
