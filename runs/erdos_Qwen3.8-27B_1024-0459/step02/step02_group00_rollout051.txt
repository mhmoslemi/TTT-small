def grad_corr_lag(h, n, lag):
    """Compute gradient of corr[lag] wrt h."""
    grad = np.zeros(n)
    if lag >= 0:
        n_pairs = n - lag
        # dc/dh[j] = (1-h[j-lag]) for j in [lag, n-1] minus h[j+lag] for j in [0, n_pairs-1]
        grad[lag:] = (1.0 - h[:n_pairs])  # first term: (1-h[j-lag]) for j=lag..n-1
        grad[:n_pairs] -= h[lag:lag+n_pairs]  # second term: -h[j+lag] for j=0..n_pairs-1

        n_pairs = n - a

        grad[:n_pairs] = (1.0 - h[a:a+n_pairs])  # first term
        grad[a:] -= h[:n_pairs]  # second term: -h[j-a] for j=a..n-1
    return grad
