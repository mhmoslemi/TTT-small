        inits.append(make_fourier_init(n, rng, nfreq=3))
        inits.append(make_fourier_init(n, rng, nfreq=8))
        inits.append(make_block_init(n, rng))
        inits.append(make_binary_init(n, rng))

        for h0 in inits:
            if remaining() < 25:
                break
            h0 = project_feasible(h0, n)
            c5_0 = compute_c5(h0)
            if c5_0 < best_c5:
                best_c5 = c5_0
                best_h = h0.copy()
                best_n = n
                print("  Init: n={}, C5={:.8f}".format(n, best_c5))

            # Phase 1: Momentum PGD (coarse to fine beta)
            if remaining() > 35:
                try:
                    n_iter = min(400, int(remaining() * 5))
                    h1, c5_1 = momentum_pgd(h0, n, beta_start=8, beta_end=80,
                                            max_iter=n_iter, momentum=0.8)
                    if c5_1 < best_c5:
                        best_c5 = c5_1
                        best_h = h1.copy()
                        best_n = n
                        print("  PGD1: n={}, C5={:.8f}".format(n, best_c5))
                except Exception:
                    pass

            # Phase 2: Finer PGD
            if remaining() > 30 and best_n == n:
                try:
                    n_iter = min(500, int(remaining() * 5))
                    h2, c5_2 = momentum_pgd(best_h, n, beta_start=80, beta_end=500,
                                            max_iter=n_iter, momentum=0.7)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()
                        best_n = n
                        print("  PGD2: n={}, C5={:.8f}".format(n, best_c5))
                except Exception:
                    pass

            # Phase 3: SLSQP polish
            if remaining() > 25 and best_n == n:
                try:
                    h3, c5_3 = slsqp_optimize(best_h, n, beta=300, max_iter=200)
                    if c5_3 < best_c5:
                        best_c5 = c5_3
                        best_h = h3.copy()
                        best_n = n
                        print("  SLSQP: n={}, C5={:.8f}".format(n, best_c5))
                except Exception:
                    pass

            # Phase 4: High-beta PGD
            if remaining() > 20 and best_n == n:
                try:
                    n_iter = min(400, int(remaining() * 4))
                    h4, c5_4 = momentum_pgd(best_h, n, beta_start=300, beta_end=2000,
                                            max_iter=n_iter, momentum=0.5)
                    if c5_4 < best_c5:
                        best_c5 = c5_4
                        best_h = h4.copy()
                        best_n = n
                        print("  PGD3: n={}, C5={:.8f}".format(n, best_c5))
                except Exception:
                    pass

        # Binary refinement at this resolution
        if remaining() > 30 and best_n == n:
            h_bin = round_to_binary(best_h, n)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n
                print("  Binary: n={}, C5={:.8f}".format(n, best_c5))

            if remaining() > 15:
                h_bl, c5_bl = binary_local_search(h_bin, n, max_rounds=2000, rng=rng)
                if c5_bl < best_c5:
                    best_c5 = c5_bl
                    best_h = h_bl.copy()
                    best_n = n
                    print("  BinLS: n={}, C5={:.8f}".format(n, best_c5))

    # Final polish: try a few more SLSQP rounds on the best
    if remaining() > 20:
        try:
            hp, cp = slsqp_optimize(best_h, best_n, beta=500, max_iter=300)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()
                print("  Final SLSQP: C5={:.8f}".format(best_c5))
        except Exception:
            pass

    if remaining() > 15 and best_n > 100:
        try:
            hp, cp = slsqp_optimize(best_h, best_n, beta=1000, max_iter=200)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()
                print("  Final SLSQP2: C5={:.8f}".format(best_c5))
        except Exception:
            pass

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
