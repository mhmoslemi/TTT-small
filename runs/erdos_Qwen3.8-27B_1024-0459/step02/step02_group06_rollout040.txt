def project_density(x, target):

    x = np.clip(x, 0.0, 1.0)

    # For lag < 0: similar with |lag|

    return corr

    n = h.size

    if lag == 0:

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-10:
                free2 = np.where((h > 0.0) & (h < 1.0))[0]
                if free2.size > 0:
                    h[free2[0]] += res2
                    h = np.clip(h, 0.0, 1.0)
    return h


def eval_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def grad_lag(h, lag):
    n = h.size
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n - lag]
    else:
        m = -lag
        g[m:] += 1.0 - h[:n - m]
        g[:n - m] -= h[m:]
    return g


def soft_max_grad(h, n_top=40, temp=0.003):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    max_c = float(np.max(corr))
    idx = np.argsort(corr)[-n_top:]
    w = np.exp((corr[idx] - max_c) / temp)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(w)
        w[np.argmax(corr[idx])] = 1.0
    else:
        w /= wsum
    g = np.zeros(n)
    for i, t in enumerate(idx):
        lag = int(t) - center
        g += w[i] * grad_lag(h, lag)
    return g, max_c * (2.0 / n)

    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))

def pair_swap_optimize(h, deadline, max_iter=10000, n_top=20):

def coordinate_descent(h, deadline, max_rounds=500):
    """Cyclic coordinate descent: for each coordinate, try optimal value"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target).copy()
    best_c = eval_c5(best)
    rng = np.random.default_rng(88888)
    
    order = np.arange(n)
    
    for rnd in range(max_rounds):
        if time.perf_counter() >= deadline:
            break
        rng.shuffle(order)
        improved = False
        for j in order:
            j = int(j)
            if time.perf_counter() >= deadline:
                break
            # Try several values for h[j]
            for trial_val in (0.0, 0.25, 0.5, 0.75, 1.0,
                              best[j] + 0.05, best[j] - 0.05,
                              best[j] + 0.1, best[j] - 0.1):
                if trial_val < 0 or trial_val > 1:
                    continue
                adj = trial_val - best[j]
                if abs(adj) < 1e-10:
                    continue
                # Find where to put the adjustment
                if abs(adj) > 1e-10:
                    # Need to compensate elsewhere
                    # For simplicity, try setting h[j] and projecting
                    cand = best.copy()
                    cand[j] = trial_val
                    cand = project_density(cand, target)
                    c = eval_c5(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        improved = True
                        break
        if not improved and rnd > 50:
            break
    
    return best, best_c


def multi_freq_init(n, rng):
    """Create diverse initializations based on frequency structure"""
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    mode = int(rng.integers(4))
    if mode == 0:
        k1, k2 = int(rng.integers(2, 6)), int(rng.integers(3, 8))
        h = 0.5 + 0.3 * np.sin(2 * np.pi * k1 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * k2 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        # Block structure
        nblocks = int(rng.integers(4, 12))
        seg = n // nblocks
        h = np.empty(n)
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = rng.uniform(0.0, 1.0)
    elif mode == 2:
        # Sparse spikes
        h = 0.3 * np.ones(n)
        nspikes = int(n * rng.uniform(0.1, 0.3))
        spike_idx = rng.choice(n, nspikes, replace=False)
        h[spike_idx] = rng.uniform(0.6, 1.0, nspikes)
    else:
        # Gaussian bumps
        h = 0.2 * np.ones(n)
        nbumps = int(rng.integers(3, 8))
        for _ in range(nbumps):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.2)
            amp = rng.uniform(0.3, 0.7)
            idx = np.arange(n)
            h += amp * np.exp(-0.5 * ((idx - center) / width) ** 2)
    return project_density(h, 0.5 * n)
