def project_density(x, target):
    """Project x onto {0<=x<=1, sum(x)=target}"""
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    x = np.clip(x, 0.0, 1.0)
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    # Bisection for lambda such that sum(clip(x+lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.clip(x+mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    # Fix any residual
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size > 0:
            # Distribute residual to free coordinates
            per = res / free.size
            h[free] += per
            h = np.clip(h, 0.0, 1.0)
    return h
