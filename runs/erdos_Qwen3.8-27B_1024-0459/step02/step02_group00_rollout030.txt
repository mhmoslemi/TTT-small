    # Phase 5: Nelder-Mead in gradient subspace
    if remaining_time() > 25:
        h5, c5_5 = do_nm_subspace(best_h, n, rng)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            print("  NM: C5={:.10f}".format(c5_5))
    
    # Phase 6: More gradient descent after NM
    if remaining_time() > 20:
        h6, c5_6 = do_gradient_descent(best_h, n, max_iters=400)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
            print("  GradDesc2: C5={:.10f}".format(c5_6))
    
    # Phase 7: Try higher resolution
    for n2 in [320, 256, 400]:
        if remaining_time() < 80:
            break
        h2_init = interp_to(best_h, best_n, n2)
        h2_init = project_feasible(h2_init, n2)
        c5_2 = compute_c5(h2_init)
        
        for beta_val in [1000, 5000]:
            if remaining_time() < 30:
                break
            h2_new, c5_2_new = do_slsqp(h2_init, n2, beta_val, maxiter=200)
            if c5_2_new < c5_2:
                c5_2 = c5_2_new
                h2_init = h2_new
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2_init.copy()
                best_n = n2
                print("  HiRes n={}, beta={}: C5={:.10f}".format(n2, beta_val, c5_2))
        
        # Gradient descent at higher res
        if remaining_time() > 30 and c5_2 <= best_c5 * 1.001:
            h2_g, c5_2g = do_gradient_descent(h2_init, n2, max_iters=200)
            if c5_2g < c5_2:
                c5_2 = c5_2g
                h2_init = h2_g
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2_init.copy()
                best_n = n2
                print("  HiRes GD n={}: C5={:.10f}".format(n2, c5_2))
    
    # Phase 8: Final polish - random perturbations + SLSQP
    if remaining_time() > 40:
        n = best_n
        _N = n
        _TGT = n / 2.0
        for trial in range(15):
            if remaining_time() < 15:
                break
            noise_scale = 0.005 * (1.0 - trial / 15.0)
            noise = rng.normal(0, noise_scale, n)
            h_try = project_feasible(best_h + noise, n)
            h_try, c5_try = do_slsqp(h_try, n, 10000, maxiter=150)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                print("  Restart {}: C5={:.10f}".format(trial, c5_try)
            # Also try gradient descent from perturbed point
            if remaining_time() > 20:
                h_try2, c5_try2 = do_gradient_descent(h_try, n, max_iters=100)
                if c5_try2 < best_c5:
                    best_c5 = c5_try2
                    best_h = h_try2.copy()
                    print("  Restart GD {}: C5={:.10f}".format(trial, c5_try2)
    
    # Phase 9: Final very-fine coordinate descent
    if remaining_time() > 15:
        h9, c5_9 = do_coordinate_descent(best_h, best_n, n_sweeps=20, rng=rng)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()
    
    # Final SLSQP polish
    if remaining_time() > 8:
        h_final, c5_final = do_slsqp(best_h, best_n, 50000, maxiter=200)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.10f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
