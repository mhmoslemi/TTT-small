            break

            # C_k = a_k * h[j] + b_k where a_k depends on neighbors
            k_fwd = j + ells  # indices j + k for each k
            k_bwd = ells - j  # indices k - j for each k (i.e., the i in h[i+k]=h[j])

            fwd_idx = j + ells
            valid_fwd = (fwd_idx >= 0) & (fwd_idx < n)
            a[valid_fwd] += (1.0 - h[fwd_idx[valid_fwd]])

            a[valid_bwd] += -h[bwd_idx2[valid_bwd]]

        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.count_nonzero(mask)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return make_feasible(np.interp(x_new, x_old, h_src), n_dst)

def coord_desc(h, n, n_passes, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ells = np.arange(-(n-1), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    
    for _ in range(n_passes):
        for j in range(n):
            hj = h[j]
            a = np.zeros(2*n-1)
            fi = j + ells
            vf = (fi >= 0) & (fi < n)
            if np.any(vf): a[vf] += 1.0 - h[fi[vf]]
            bi = j - ells
            vb = (bi >= 0) & (bi < n)
            if np.any(vb): a[vb] += -h[bi[vb]]
            b = corr - hj * a
            lo, hi = 0.0, 1.0
            for _ in range(25):
                mid = (lo+hi)*0.5
                v = a*mid + b
                mx = v.max()
                act = v >= mx - 1e-15
                sg = a[act].mean()
                if sg > 1e-14: hi = mid
                elif sg < -1e-14: lo = mid
                else: break
            nh = (lo+hi)*0.5
            h[j] = nh
            d = nh - hj
            if abs(d) > 1e-16:
                corr += d * a
        h = make_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if time.time() > deadline:
            break
    return best_h, best_c5

def qr_sequence(n):
    """Quadratic residue sequence of length n (approximately)."""
    # Find a prime p close to n
    def is_prime(m):
        if m < 2: return False
        for i in range(2, int(m**0.5)+1):
            if m % i == 0: return False
        return True
    p = n
    while p < 4*n and not is_prime(p):
        p += 1
    # Build QR sequence of length p
    qr = np.zeros(p)
    for j in range(1, p):
        qr[j] = 1.0 if pow(j, (p-1)//2, p) == 1 else 0.0
    # Adjust to have exactly n/2 ones
    if p >= n:
        h = qr[:n]
    else:
        # Tile or truncate
        rept = (n + p - 1) // p
        h = np.tile(qr, rept)[:n]
    return make_feasible(h, n)

def slsqp_polish(h, n, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0},)
    def obj(x):
        return np.max(np.correlate(x, 1.0-x, mode='full'))
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'eps': 1e-8})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def binary_local_search(h, n, deadline):
    h_bin = (h > 0.5).astype(float)
    deficit = n/2.0 - np.sum(h_bin)
    if deficit > 0:
        zeros = np.where(h_bin==0)[0]
        order = np.argsort(-h[zeros])[:int(deficit)]
        h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin==1)[0]
        order = np.argsort(h[ones])[:int(-deficit)]
        h_bin[ones[order]] = 0.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(7)
    for _ in range(50000):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, 2, replace=False)
        if h_bin[i] != h_bin[j]:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            c5 = compute_c5(h_bin)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_bin.copy()
            else:
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
