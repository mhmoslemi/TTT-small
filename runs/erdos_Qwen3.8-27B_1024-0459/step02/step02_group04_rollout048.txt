        top_idx = np.argpartition(vals, -k_eff)[-k_eff:]

    if l > 0:
        g[:m] += dx * (1.0 - h[L:])
        g[L:] += -dx * h[:m]
    else:
        g[L:] += dx * (1.0 - h[:m])
        g[:m] += -dx * h[L:]
    return g

        if time.time() > hard_end - 500.0:
            break
        if n_r != best_n and time.time() > hard_end - 400.0:
            break
        h2 = smooth_optimize(h_r, n_r, 30, maxiter=30, ftol=1e-11)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()
            best_n = n_r

    # Higher beta smooth max on best
    for beta in (80, 150, 300):
        if time.time() > hard_end - 350.0:
            break
        h2 = smooth_optimize(best_h, best_n, beta, maxiter=45, ftol=1e-12)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 2: Active set refinement
    time_left = hard_end - time.time()
    if time_left > 200.0:
        # Start with smaller active sets, expand
        if best_n <= 200:
            ac_seq = (12, 24, 48, 80)
        else:
            ac_seq = (10, 20, 40, 64)
        maxit_ac = 55 if best_n <= 200 else 40
        h2, c2 = refine_active_set(best_h, best_n, time.time() + 250.0,
                                   seq=ac_seq, maxiter=maxit_ac)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 3: Top-k projected descent
    if time.time() < hard_end - 150.0:
        slot = min(200.0, hard_end - time.time() - 80.0)
        h2, c2 = topk_descent(best_h, best_n, k=6, num_steps=8000,
                              lr0=0.006, decay=0.9997,
                              time_limit=time.time() + slot)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 4: Directed perturbation
    if time.time() < hard_end - 100.0:
        slot = min(120.0, hard_end - time.time() - 50.0)
        h2, c2 = directed_perturb(best_h, best_n, time.time() + slot, max_steps=5000)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 5: Local random refinement
    if time.time() < hard_end - 60.0:
        slot = min(100.0, hard_end - time.time() - 30.0)
        h2, c2 = local_refine(best_h, best_n, time.time() + slot, seed + 7777,
                              max_trials=12000)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 6: Final active set polish
    if time.time() < hard_end - 20.0:
        vals = all_lag_values(best_h)
        active = select_active_lags(vals, 16, margin=1e-9)
        t0 = float(np.max(vals)) + 1e-7
        h2, t2 = active_set_solve(best_h, best_n, active, t0, maxiter=80, ftol=1e-14)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)

    return (best_h, float(best_c), int(best_n))
