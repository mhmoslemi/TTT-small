    for idx in range(n_steps):
        t = idx / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        try:
            res = minimize(smooth_max_obj, current, args=(beta,), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.fun < 1e10:
                candidate = make_feasible(res.x, n)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    current = candidate
        except Exception:
            pass
    
    # Final polish with direct max
    try:
        res = minimize(_direct_obj, current, method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            candidate = make_feasible(res.x, n)
            c5_val = compute_c5(candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                current = candidate
    except Exception:
        pass
    
    return current, best_c5

def continuous_local_search(h_init, n, rng, max_trials=50000, timeout=60.0):
    h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    target = n / 2.0
    no_improve = 0
    stall_limit = 5000
    start_t = time.time()
    
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        
        h_try = h.copy()
        n_mod = rng.randint(1, 5)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-1, 1) * 0.05
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = target - np.sum(h_try)
        if abs(diff) > 1e-12:
            mask = (h_try > 1e-4) & (h_try < 1.0 - 1e-4)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
            no_improve = 0
        else:
            no_improve += 1
    
    return h, best_c5

def generate_inits(n, rng, h_parent=None):
    inits = []
    if h_parent is not None and len(h_parent) > 0:
        h_p = make_feasible(interp_h(h_parent, n), n)
        inits.append(h_p)
    inits.append(thue_morse_init(n))
    
    # Antisymmetric init
    h = np.zeros(n)
    half = n // 2
    for i in range(half):
        h[i] = 1.0 if bin(i * 3 + 7).count('1') % 2 == 0 else 0.0
        h[i + half] = 1.0 - h[i]
    inits.append(make_feasible(h, n))
    
    # Random binaries
    for _ in range(4):
        h = (rng.rand(n) > 0.5).astype(float)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        target = n // 2
        if len(ones) > target:
            excess = len(ones) - target
            to_flip = rng.choice(ones, excess, replace=False)
            h[to_flip] = 0.0
        elif len(ones) < target:
            deficit = target - len(ones)
            to_flip = rng.choice(zeros, deficit, replace=False)
            h[to_flip] = 1.0
        inits.append(h)
    
    # Phase-shifted Thue-Morse
    for shift in [1, 2, 3]:
        tm = np.zeros(n)
        for i in range(n):
            tm[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
        inits.append(make_feasible(tm, n))
    
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    n = 256
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None
    
    best_h = thue_morse_init(n)
    best_c5 = compute_c5(best_h)
    
    inits = generate_inits(n, rng, h_parent)
    
    # Phase 1: Simulated annealing from parent and a few inits
    for h0 in inits[:3]:
        if remaining() < 100:
            break
        h_sa, c5_sa = simulated_annealing(h0, n, rng, t0=5.0, cooling=0.99995,
                                          max_trials=300000, timeout=min(200, remaining()-50))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 2: SLSQP smooth max optimization
    for h0 in [best_h] + inits[:2]:
        if remaining() < 120:
            break
        h_opt, c5_val = slsqp_optimize(h0, n, beta_start=15.0, beta_end=400.0,
                                        n_steps=6, max_iter=400)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
