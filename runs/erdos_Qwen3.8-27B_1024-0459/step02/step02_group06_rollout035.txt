```python
import time
import numpy as np
from scipy.optimize import minimize

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    # Bisection to find the right shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-10 and free2.size > 0:
                h[free2[0]] += res2
                h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_grad(h, k):
    """Gradient of corr at lag k with respect to h"""
    n = h.size
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
            g[lag:] -= h[:n-lag]
            g[:n-lag] -= h[lag:]
        else:
            g[:n-lag] += 1.0
            g[:n-lag] -= h[lag:]
            g[lag:] -= h[:n-lag]
    return g
```

```python
def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-13:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-11:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)

            if abs(res2) > 1e-11:
                free2 = np.where((h > 0.0) & (h < 1.0))[0]
                if free2.size > 0:
                    h[free2[0]] += res2
                    h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_grad_one(h, k):
    n = h.size
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        j = -k
        g[:n - j] += (1.0 - h[j:])
        g[j:] -= h[:n - j]
    return g

def active_gradient_weighted(h, top_frac=0.08, min_k=8, temperature=2.0):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    top_idx = np.argsort(corr)[-k_count:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    w = np.exp(-dv / max(1e-10, temperature * (dv.max() + 1e-10)))
    w /= w.sum()
    g = np.zeros(n)
    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += w[i] * corr_grad_one(h, k)
    return g, corr

def project_gradient_step(h, grad, target, steps=(0.02, 0.01, 0.005, 0.002, 0.001)):
    best_c = evaluate_erdos(h)
    best = h.copy()
    for step in steps:
        cand = project_density(h - step * grad, target)
        c = evaluate_erdos(cand)
        if c < best_c - 1e-13:
            best = cand
            best_c = c
            break
    return best, best_c

def smooth_minimax_obj(x, h_cache, alpha):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    n = len(h)
    return (2.0 / float(n)) * (m + np.log(z) / alpha)

def smooth_minimax_grad(x, alpha):
    n = x.size
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    center = n - 1
    idx = np.where(w > 1e-8)[0]
    g = np.zeros(n)
    for t in idx:
        p = float(w[t])
        k = int(t) - center
        gk = corr_grad_one(h, k)
        g += p * gk
        dk_dm = -1.0
        dk_dm += 1.0
        g += p * dk_dm * gk
    dx = 2.0 / float(n)
    return dx * g

def slsqp_optimize(h, target, deadline, n, alphas=None, maxiter=None):
    if alphas is None:
        if n <= 200:
            alphas = [20.0, 60.0, 150.0]
        elif n <= 500:
            alphas = [15.0, 50.0, 120.0]
        else:
            alphas = [12.0, 40.0, 100.0]
    if maxiter is None:
        if n <= 200:
            maxiter = 250
        elif n <= 500:
            maxiter = 150
        else:
            maxiter = 80
    best = h.copy()
    best_c = evaluate_erdos(best)
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": lambda x: float(np.sum(x) - target),
                    "jac": lambda x: np.ones_like(x)},)
    for alpha in alphas:
        if time.perf_counter() >= deadline - 6.0:
            break
        try:
            res = minimize(
                smooth_minimax_obj, best,
                args=(None, alpha),
                method="SLSQP",
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-12, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best = cand
                    best_c = c
                else:
                    best = project_density(best, target)
        except Exception:
            pass
    return best, evaluate_erdos(best)

        h = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * np.arange(n) / n * 2)
        h = np.where(h > 0.5, 1.0, 0.0)
    else:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * rng.integers(2, 8) * np.arange(n) / n * 2 + rng.uniform(0, 6.28))
        h = np.where(h > 0.5, 1.0, 0.0)
    return project_density(h, 0.5 * n)

def upsample_to(parent, n_target):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n_target:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(n_target, 0.5), 0.5 * n_target)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n_target) + 0.5) * (2.0 / float(n_target))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n_target)

def optimize_at_resolution(n, h_init, deadline, seed=42):
    target = 0.5 * n
    h = project_density(h_init, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    el = deadline - time.perf_counter()
    if el < 10:
        return best_h, best_c

    # Phase A: SLSQP (use ~30% of time)
    sl_dl = time.perf_counter() + el * 0.28
    h_sl, c_sl = slsqp_optimize(best_h, target, sl_dl, n)
    if c_sl < best_c:
        best_h = h_sl
        best_c = c_sl

    # Phase B: Momentum gradient (use ~25% of time)
    el = deadline - time.perf_counter()
    if el > 8:
        mo_dl = time.perf_counter() + el * 0.25
        h_mo, c_mo = momentum_optimize(best_h, target, mo_dl, max_iter=300, top_frac=0.07, min_k=8, temp=1.5)
        if c_mo < best_c:
            best_h = h_mo
            best_c = c_mo
        # Also try from momentum result with different params
        el = deadline - time.perf_counter()
        if el > 5:
            h_mo2, c_mo2 = momentum_optimize(best_h, target, time.perf_counter() + el * 0.5, max_iter=200, top_frac=0.05, min_k=10, temp=0.8)
            if c_mo2 < best_c:
                best_h = h_mo2
                best_c = c_mo2

    # Phase C: Focused pair swap (use ~25% of time)
    el = deadline - time.perf_counter()
    if el > 8:
        ps_dl = time.perf_counter() + el * 0.28
        h_ps, c_ps = focused_pair_swap(best_h, ps_dl, max_iter=3000, top_frac=0.06, min_k=8, n_try=14)
        if c_ps < best_c:
            best_h = h_ps
            best_c = c_ps

    # Phase D: Fine perturbation (use ~15% of time)
    el = deadline - time.perf_counter()
    if el > 5:
        fp_dl = time.perf_counter() + el * 0.4
        h_fp, c_fp = fine_perturbation(best_h, fp_dl, max_iter=5000, n_perturb=2)
        if c_fp < best_c:
            best_h = h_fp
            best_c = c_fp

    # Phase E: Another momentum pass
    el = deadline - time.perf_counter()
    if el > 5:
        h_mo3, c_mo3 = momentum_optimize(best_h, target, time.perf_counter() + el * 0.7, max_iter=400, top_frac=0.05, min_k=12, temp=1.0)
        if c_mo3 < best_c:
            best_h = h_mo3
            best_c = c_mo3

    # Phase F: Final pair swap
    el = deadline - time.perf_counter()
    if el > 4:
        h_ps2, c_ps2 = focused_pair_swap(best_h, time.perf_counter() + el * 0.8, max_iter=1500, top_frac=0.04, min_k=10, n_try=10)
        if c_ps2 < best_c:
            best_h = h_ps2
            best_c = c_ps2

    best_h = project_density(best_h, target)
    return best_h, evaluate_erdos(best_h)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).copy()
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        pn = int(parent.size)
        pp = project_density(parent, 0.5 * pn)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = pn
    else:
        n0 = 450
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 0.7)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Resolutions to try (prioritize parent resolution and slightly higher)
    resolutions = []
    base_n = best_n
    for r in [base_n, base_n + 30, base_n + 60, base_n + 150, base_n - 60, base_n - 150]:
        if 40 <= r <= 1200 and r not in resolutions:
            resolutions.append(r)
    # Also add some specific ones
    for r in [300, 360, 450, 540, 600, 750, 900]:
        if 40 <= r <= 1200 and r not in resolutions:
            resolutions.append(r)
    resolutions = sorted(set(resolutions))

    # Allocate time: weight toward parent resolution and slightly higher
    total_avail = deadline - time.perf_counter()
    if total_avail < 60:
        # Very tight budget, just do local search at parent resolution
        h_out = project_density(best_h, 0.5 * best_n)
        c_out = evaluate_erdos(h_out)
        h_fin, c_fin = focused_pair_swap(h_out, deadline - 2.0, max_iter=2000)
        if c_fin < c_out:
            h_out = h_fin
            c_out = c_fin
        return h_out, c_out, int(h_out.size)

    # Weight allocations
    weights = []
    for r in resolutions:
        if r == base_n:
            weights.append(4.0)
        elif abs(r - base_n) <= 60:
            weights.append(3.0)
        elif abs(r - base_n) <= 150:
            weights.append(2.0)
        elif r > base_n:
            weights.append(1.5)
        else:
            weights.append(1.0)
    wsum = sum(weights)
    time_alloc = {}
    for r, w in zip(resolutions, weights):
        time_alloc[r] = max(20.0, total_avail * w / wsum)

    # Skip resolutions that would get too little time
    min_time = 25.0
    active_res = [(r, t) for r, t in zip(resolutions, time_alloc.values()) if t > min_time * len(resolutions) * 0.3]
    if not active_res:
        active_res = [(r, t) for r, t in zip(resolutions, time_alloc.values())][:5]

    # Sort by priority (parent first, then nearby)
    active_res.sort(key=lambda x: -time_alloc.get(x[0], 0))

    for ni, alloc in active_res:
        rem = deadline - time.perf_counter()
        if rem < 20.0:
            break
        alloc = min(alloc, rem - 5.0)
        alloc = max(15.0, alloc)

        # Build starting points
        starts = []
        # Parent upsampled
        s1 = upsample_to(best_h, ni)
        starts.append(s1)
        if parent is not None:
            s2 = upsample_to(parent, ni)
            if not np.allclose(s1, s2, atol=1e-8):
                starts.append(s2)
        # Binary structured starts
        if ni <= 600:
            s3 = make_binary_start(ni, rng, pattern=4)
            starts.append(s3)

        # Deduplicate
        uniq = [starts[0]]
        for s in starts[1:]:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-8):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        # If time allows, try 2 starts; otherwise just the best one
        if len(uniq) > 1 and alloc > 50:
            per_start = alloc / 2
        else:
            uniq = [uniq[0]]
            per_start = alloc

        for si, s in enumerate(uniq):
            rem2 = deadline - time.perf_counter()
            if rem2 < 12.0:
                break
            sub_alloc = per_start if si == 0 else min(per_start, rem2 - 5.0)
            sub_dl = time.perf_counter() + max(12.0, sub_alloc)
            h_out, c_out = optimize_at_resolution(ni, s, sub_dl, seed + si * 100)
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * ni)
                c_out = evaluate_erdos(h_out)
                if c_out < best_c:
                    best_h = h_out.copy()
                    best_c = c_out
                    best_n = ni
                    # Update other starts for subsequent resolutions
                    for orig_s in starts:
                        pass  # starts already defined

    # Final intensive polish on the best solution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        c_fin = evaluate_erdos(h_fin)

        # Intensive pair swap
        if rem > 20:
            dl = time.perf_counter() + rem * 0.3
            h2, c2 = focused_pair_swap(h_fin, dl, max_iter=5000, top_frac=0.05, min_k=10, n_try=16)
            if c2 < c_fin:
                h_fin = h2
                c_fin = c2

        # Fine perturbation
        rem2 = deadline - time.perf_counter()
        if rem2 > 10:
            dl = time.perf_counter() + rem2 * 0.35
            h3, c3 = fine_perturbation(h_fin, dl, max_iter=8000, n_perturb=2)
            if c3 < c_fin:
                h_fin = h3
                c_fin = c3

        # Momentum with very focused active set
        rem3 = deadline - time.perf_counter()
        if rem3 > 8:
            dl = time.perf_counter() + rem3 * 0.4
            h4, c4 = momentum_optimize(h_fin, target, dl, max_iter=500, top_frac=0.04, min_k=12, temp=0.5)
            if c4 < c_fin:
                h_fin = h4
                c_fin = c4

        # Final pair swap
        rem4 = deadline - time.perf_counter()
        if rem4 > 6:
            h5, c5 = focused_pair_swap(h_fin, time.perf_counter() + rem4 * 0.85, max_iter=3000, top_frac=0.03, min_k=12, n_try=10)
            if c5 < c_fin:
                h_fin = h5
                c_fin = c5

        # Final fine perturbation
        rem5 = deadline - time.perf_counter()
        if rem5 > 4:
            h6, c6 = fine_perturbation(h_fin, time.perf_counter() + rem5 * 0.9, max_iter=3000, n_perturb=1)
            if c6 < c_fin:
                h_fin = h6
                c_fin = c6

        h_fin = project_density(h_fin, 0.5 * int(h_fin.size))
        c_fin = evaluate_erdos(h_fin)
        if c_fin < best_c:
            best_h = h_fin
            best_c = c_fin
            best_n = int(h_fin.size)

    if best_h is None:
        fallback = project_density(np.full(100, 0.5), 50.0)
        return fallback, evaluate_erdos(fallback), 100

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
```
