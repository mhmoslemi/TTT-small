def compute_c5(h):

    h = np.array(h, dtype=float).copy()

    target = n / 2.0

def slsqp_smooth(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -50, 50)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def slsqp_polish(h_in, n, max_iter=2000):
    h0 = make_feasible(h_in.copy(), n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    def obj_eq(x):
        return np.sum(x) - target

    def obj_eq_jac(x):
        return np.ones(n)

    con = {'type': 'eq', 'fun': obj_eq, 'jac': obj_eq_jac}
    for beta in [50, 150, 400]:
        try:
            res = minimize(slsqp_smooth, h0, args=(n, beta), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.fun < 1e10:
                h0 = make_feasible(res.x, n)
        except Exception:
            pass
    return h0, compute_c5(h0)

def step_function_optimize(h_init, n, M, rng, timeout=60.0):
    """Parameterize h as M-step function and optimize with SLSQP."""
    h0 = make_feasible(h_init.copy(), n)
    seg_len = n // M
    if seg_len < 1:
        return h0, compute_c5(h0)
    # Build breakpoint and value params from h0
    breakpoints = np.zeros(M - 1)
    values = np.zeros(M)
    for i in range(M):
        start = i * seg_len
        end = min((i + 1) * seg_len, n)
        values[i] = np.mean(h0[start:end])
    for i in range(1, M):
        breakpoints[i - 1] = i * seg_len
    # Parameters: [bp_1,...,bp_{M-1}, v_1,...,v_M]
    # Use log-spacing for breakpoints to maintain ordering
    # Just use direct with bounds
    x0 = np.concatenate([breakpoints, values])
    n_params = 2 * M - 1
    bps_bounds = [(1.0, i * seg_len + seg_len - 1) for i in range(1, M)]
    vals_bounds = [(0.0, 1.0)] * M
    all_bounds = bps_bounds + vals_bounds

    def params_to_h(params, n, M):
        bps = params[:M - 1].astype(int)
        vals = params[M - 1:]
        bps = np.sort(bps)
        h = np.zeros(n)
        edges = np.concatenate([[0], bps, [n]])
        for i in range(M):
            h[edges[i]:edges[i + 1]] = vals[i]
        return make_feasible(h, n)

    def obj_func(params):
        h = params_to_h(params, n, M)
        corr = np.correlate(h, 1.0 - h, mode='full')
        beta = 200.0
        cmax = np.max(corr)
        shifted = beta * (corr - cmax)
        shifted = np.clip(shifted, -50, 50)
        return (cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))) * (2.0 / n)

    t0 = time.time()
    try:
        res = minimize(obj_func, x0, method='SLSQP', bounds=all_bounds,
                      options={'maxiter': 500, 'ftol': 1e-12})
        if res.fun < 10.0:
            h_opt = params_to_h(res.x, n, M)
            c5_opt = compute_c5(h_opt)
            if c5_opt < compute_c5(h0):
                return h_opt, c5_opt
    except Exception:
        pass
    return h0, compute_c5(h0)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    # Get parent
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None

    n = 512
    dx = 2.0 / n

    # Phase 1: Generate initial solutions
    inits = []
    if h_parent is not None:
        inits.append(make_feasible(interp_h(h_parent, n), n))
    inits.append(thue_morse(n, 0))
    inits.append(thue_morse(n, 1))
    inits.append(thue_morse(n, 3))
    inits.append(thue_morse(n, 7))
    for s in range(3):
        inits.append(random_binary(n, rng, seed_extra=s * 13))

    best_h = None
    best_c5 = float('inf')
    for h0 in inits:
        c5_val = compute_c5(h0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h0.copy()

    # Phase 2: Greedy local search on best init (binary)
    if time_left() > 60:
        alloc = min(40.0, time_left() / 3)
        h_b, c5_b = greedy_local_search(best_h, n, rng, max_trials=200000, timeout=alloc, stall=10000)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Targeted local search
    if time_left() > 80:
        alloc = min(60.0, time_left() / 3)
        h_t, c5_t = targeted_local_search(best_h, n, rng, max_trials=150000, timeout=alloc, stall=8000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 4: More greedy from best
    if time_left() > 100:
        alloc = min(80.0, time_left() / 3)
        h_b2, c5_b2 = greedy_local_search(best_h, n, rng, max_trials=500000, timeout=alloc, stall=20000)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 5: Binary SA
    if time_left() > 200:
        alloc = min(200.0, time_left() / 2.5)
        h_sa, c5_sa = binary_sa(best_h, n, rng, timeout=alloc, t_start=0.8, t_end=0.005)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Targeted search again
    if time_left() > 100:
        alloc = min(60.0, time_left() / 3)
        h_t2, c5_t2 = targeted_local_search(best_h, n, rng, max_trials=200000, timeout=alloc, stall=10000)
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()

    # Phase 7: Continuous SA
    if time_left() > 150:
        alloc = min(120.0, time_left() / 2.5)
        h_csa, c5_csa = simulated_annealing(best_h, n, rng, timeout=alloc, t_start=1.5, t_end=0.01)
        if c5_csa < best_c5:
            best_c5 = c5_csa
            best_h = h_csa.copy()

    # Phase 8: Continuous refine
    if time_left() > 60:
        alloc = min(45.0, time_left() / 3)
        h_cr, c5_cr = continuous_refine(best_h, n, rng, timeout=alloc, max_trials=80000)
        if c5_cr < best_c5:
            best_c5 = c5_cr
            best_h = h_cr.copy()

    # Phase 9: Step-function optimization
    if time_left() > 80:
        for M in [8, 12, 16, 24]:
            if time_left() < 20:
                break
            alloc = min(15.0, time_left() / 4)
            h_sf, c5_sf = step_function_optimize(best_h, n, M, rng, timeout=alloc)
            if c5_sf < best_c5:
                best_c5 = c5_sf
                best_h = h_sf.copy()

    # Phase 10: SLSQP polish
    if time_left() > 30:
        h_sp, c5_sp = slsqp_polish(best_h, n, max_iter=1000)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp.copy()

    # Phase 11: Final targeted search
    if time_left() > 30:
        alloc = min(25.0, time_left() - 5)
        h_f, c5_f = targeted_local_search(best_h, n, rng, max_trials=100000, timeout=alloc, stall=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 12: Final greedy
    if time_left() > 20:
        alloc = min(15.0, time_left() - 3)
        h_g, c5_g = greedy_local_search(best_h, n, rng, max_trials=100000, timeout=alloc, stall=5000)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Also check at n=256 (in case binary solution there is better)
    if h_parent is not None and time_left() > 10:
        h_256 = np.round(interp_h(best_h, 256))
        h_256 = make_feasible(h_256, 256)
        c5_256 = compute_c5(h_256)
        if c5_256 < best_c5:
            best_h = h_256
            best_c5 = c5_256
            n = 256

    return (best_h, best_c5, n)
