def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_descent(h, n, max_passes, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ells = np.arange(-(n - 1), n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            idx1 = j - ells
            idx2 = j + ells
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            a = np.zeros(2 * n - 1)
            if np.any(valid1):
                a[valid1] = 1.0 - h[idx1[valid1]]
            if np.any(valid2):
                a[valid2] = a[valid2] - h[idx2[valid2]]
            b = corr - h[j] * a
            lo, hi = 0.0, 1.0
            for _ in range(25):
                mid = 0.5 * (lo + hi)
                vals = a * mid + b
                mx = vals.max()
                active = vals >= mx - 1e-14
                sg = a[active].mean()
                if sg > 1e-12:
                    hi = mid
                elif sg < -1e-12:
                    lo = mid
                else:
                    break
            h[j] = 0.5 * (lo + hi)
        h = make_feasible(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5


def binary_from_float(h_float, n):
    h = (h_float > 0.5).astype(float)
    s = int(np.sum(h))
    target = n // 2
    if s > target:
        ones_idx = np.where(h == 1)[0]
        to_remove = ones_idx[np.argsort(h_float[ones_idx])[:s - target]]
        h[to_remove] = 0
    elif s < target:
        zeros_idx = np.where(h == 0)[0]
        to_add = zeros_idx[np.argsort(-h_float[zeros_idx])[:target - s]]
        h[to_add] = 1
    return h


def sa_binary(h_init, n, deadline, seed=42, max_pairs=4, T_init=0.03):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    T = T_init
    T_min = 1e-8
    cool = 0.99985
    stall_count = 0
    stall_limit = 500

    while time.time() < deadline:
        n_pairs = rng.randint(1, max_pairs + 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < n_pairs or len(zeros) < n_pairs:
            break
        off = rng.choice(ones, n_pairs, replace=False)
        on = rng.choice(zeros, n_pairs, replace=False)
        h[off] = 0.0
        h[on] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            stall_count = 0
        elif rng.random() < np.exp(-delta / max(T, 1e-10)):
            stall_count += 1
        else:
            h[off] = 1.0
            h[on] = 0.0
            stall_count += 1
        T *= cool
        if T < T_min:
            T = T_min
        if stall_count > stall_limit:
            h = best_h.copy()
            T = max(T, 0.01)
            stall_count = 0
    return best_h, best_c5


def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(tm, n)


def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)
