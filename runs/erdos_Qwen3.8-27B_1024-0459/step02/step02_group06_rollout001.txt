def erdos_exact(h):
    h=np.asarray(h,float)
    n=h.size
    if n==0: return float('inf')
    dx=2.0/float(n)
    corr=np.correlate(h,1.0-h,mode='full')
    if corr.size==0: return inf
    return float(np.max(corr)*dx)

def fft_corr_components(h):
    h=np.asarray(h,float)
    n=h.size
    if n==0: return 0.0,np.empty(0),np.empty(0)

def smooth_phi_grad(h):
    h=np.asarray(h,float)
    n=h.size
    if n==0: return 1e100,np.empty(0)
    if not np.all(np.isfinite(h)):
        return 1e100, np.zeros(n)
    alpha=float(_GLOBAL_ALPHA)
    if alpha <=0: alpha=1.0
    dx=2.0/n
    Fh,Nf,r0,pos,neg=fft_corr_with_Fh(h)
    cmax=r0
    if pos.size: cmax=max(cmax,float(pos.max()))
    if neg.size: cmax=max(cmax,float(neg.max()))
    if cmax <= 0:
        return 0.0, np.zeros(n)
    y0=alpha*(r0-cmax)
    ep=alpha*(pos-cmax) if pos.size else np.empty(0)
    em=alpha*(neg-cmax) if neg.size else np.empty(0)
    y0=np.clip(y0, -745.0, 0.0)
    ep=np.clip(ep, -745.0, 0.0)
    em=np.clip(em, -745.0, 0.0)
    e0=float(np.exp(y0))
    ep=np.asarray(ep,float); em=np.asarray(em,float)
    E0=e0; Ep=float(ep.sum()); Em=float(em.sum())
    Z=E0+Ep+Em
    if Z <= 1e-300 or not np.isfinite(Z):
        return dx*cmax, np.zeros(n)
    w0=E0/Z
    wpos=ep/Z if pos.size else np.empty(0)
    wneg=em/Z if neg.size else np.empty(0)
    logZ=float(np.log(Z))
    phi=dx*(cmax + logZ/alpha)
    if n==1:
       grad=dx*w0*(1.0-2.0*h)
       return phi, grad
    pp=np.empty(n); pp[0]=0.0; pp[1:]=np.cumsum(wpos)
    pn=np.empty(n); pn[0]=0.0; pn[1:]=np.cumsum(wneg)
    const = w0 + pp + pn[::-1]
    a=np.zeros(n); a[1:] = wpos + wneg
    Fa=np.fft.rfft(a,Nf)
    conv1=np.fft.irfft(Fh*Fa,Nf)[:n]
    corr2=np.fft.irfft(np.conj(Fh)*Fa,Nf)[:n]
    gradS = const - 2.0*w0*h - conv1 - corr2
    grad=dx*gradS
    if not np.all(np.isfinite(grad)):
       grad=np.nan_to_num(grad, nan=0.0, posinf=1e6, neginf=-1e6)
    return phi, grad

def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x

    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))

    if abs(res) > 1e-9:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
            res = target - float(np.sum(h))

        if abs(res) > 1e-9:
            free2 = np.where((h > 0.0) & (h < 1.0))[0]
            if free2.size > 0:
                for idx in free2:
                    if abs(res) <= 1e-10:
                        break
                    lo2 = -h[idx]
                    hi2 = 1.0 - h[idx]
                    adj = np.clip(res, lo2, hi2)
                    h[idx] += adj
                    res -= adj

    return h


def erdos_exact(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def fft_corr_components(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0, np.empty(0), np.empty(0)
    Nf = next_pow2(2 * n - 1)
    Fh = np.fft.rfft(h, Nf)
    b = 1.0 - h
    Fb = np.fft.rfft(b, Nf)
    r = np.fft.irfft(Fh * np.conj(Fb), Nf).real
    r = np.nan_to_num(r, nan=0.0, posinf=1e100, neginf=-1e100)
    r = np.where(r > 0.0, r, 0.0)
    r0 = float(r[0])
    if n > 1:
        pos = r[Nf - 1:Nf - n:-1].copy()
        neg = r[1:n].copy()
    else:
        pos = np.empty(0)
        neg = np.empty(0)
    return r0, pos, neg


def fft_corr_with_Fh(h):

def smooth_phi_grad(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e100, np.empty(0)
    if not np.all(np.isfinite(h)):
        return 1e100, np.zeros(n, dtype=float)

    alpha = float(_GLOBAL_ALPHA)
    if alpha <= 0.0 or not np.isfinite(alpha):
        alpha = 1.0
    dx = 2.0 / float(n)

    Fh, Nf, r0, pos, neg = fft_corr_with_Fh(h)
    cmax = r0
    if pos.size:
        cmax = max(cmax, float(np.max(pos)))
    if neg.size:
        cmax = max(cmax, float(np.max(neg)))

    if cmax <= 0.0:
        return 0.0, np.zeros(n, dtype=float)

    y0 = alpha * (r0 - cmax)
    if pos.size:
        yp = alpha * (pos - cmax)
    else:
        yp = np.empty(0)
    if neg.size:
        yn = alpha * (neg - cmax)
    else:
        yn = np.empty(0)

    y0 = np.clip(y0, -745.0, 0.0)
    yp = np.clip(np.asarray(yp, dtype=float), -745.0, 0.0)
    yn = np.clip(np.asarray(yn, dtype=float), -745.0, 0.0)

    e0 = float(np.exp(y0))
    ep = np.asarray(yp, dtype=float)
    em = np.asarray(yn, dtype=float)
    E0 = e0
    Ep = float(np.sum(ep))
    Em = float(np.sum(em))
    Z = E0 + Ep + Em

    if Z <= 1e-300 or not np.isfinite(Z):
        return dx * cmax, np.zeros(n, dtype=float)

    w0 = E0 / Z
    wpos = ep / Z if pos.size else np.empty(0)
    wneg = em / Z if neg.size else np.empty(0)

    logZ = float(np.log(Z))
    phi = dx * (cmax + logZ / alpha)

    if n == 1:
        g = dx * w0 * (1.0 - 2.0 * h)
        return phi, g

    pp = np.empty(n, dtype=float)
    pp[0] = 0.0
    pp[1:] = np.cumsum(wpos)

    pn = np.empty(n, dtype=float)
    pn[0] = 0.0
    pn[1:] = np.cumsum(wneg)

    const = w0 + pp + pn[::-1]

    a = np.zeros(n, dtype=float)
    a[1:] = wpos + wneg
    Fa = np.fft.rfft(a, Nf)
    conv1 = np.fft.irfft(Fh * Fa, Nf)[:n]
    corr2 = np.fft.irfft(np.conj(Fh) * Fa, Nf)[:n]

    gradS = const - 2.0 * w0 * h - conv1 - corr2
    grad = dx * gradS
    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=1e6, neginf=-1e6)
    return phi, grad


def active_gradient_smooth(h, K_frac=0.03, K_min=8, tau_frac=0.005):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return np.empty(0), float("inf")
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float), 1e100

    Fh, Nf, r0, pos, neg = fft_corr_with_Fh(h)
    total = 2 * n - 1

    flat = np.empty(total, dtype=float)
    flat[0] = r0
    if n > 1:
        flat[1:n] = pos
        flat[n:] = neg

    K = int(K_min)
    if K_frac > 0.0:
        KK = int(np.ceil(float(K_frac) * float(total)))
        if KK > K:
            K = KK
    if K > total:
        K = total
    if K < 1:
        K = 1

    if K >= total:
        idx = np.arange(total)
    else:
        idx = np.argpartition(flat, total - K)[-K:]

    top = flat[idx]
    topmax = float(np.max(top))

    if topmax <= 0.0:
        w = np.ones(K, dtype=float) / float(K)
    else:
        scale = max(1e-12, float(tau_frac) * topmax)
        w = np.exp(np.clip((top - topmax) / scale, -745.0, 0.0))
        s = float(np.sum(w))
        if s <= 1e-300 or not np.isfinite(s):
            w = np.ones(K, dtype=float) / float(K)
        else:
            w = w / s

    w0 = 0.0
    wpos = np.zeros(max(0, n - 1), dtype=float)
    wneg = np.zeros(max(0, n - 1), dtype=float)

    for jj in range(K):
        j = int(idx[jj])
        wt = float(w[jj])
        if j == 0:
            w0 += wt
        elif j < n:
            L = j
            wpos[L - 1] += wt
        else:
            L = j - (n - 1)
            wneg[L - 1] += wt

    ztot = w0 + float(np.sum(wpos)) + float(np.sum(wneg))
    if ztot > 0.0:
        w0 = w0 / ztot
        wpos = wpos / ztot
        wneg = wneg / ztot

    dx = 2.0 / float(n)

    if n == 1:
        g = dx * w0 * (1.0 - 2.0 * h)
        return g, dx * topmax

    pp = np.empty(n, dtype=float)
    pp[0] = 0.0
    pp[1:] = np.cumsum(wpos)

    pn = np.empty(n, dtype=float)
    pn[0] = 0.0
    pn[1:] = np.cumsum(wneg)

    const = w0 + pp + pn[::-1]

    a = np.zeros(n, dtype=float)
    a[1:] = wpos + wneg
    Fa = np.fft.rfft(a, Nf)
    conv1 = np.fft.irfft(Fh * Fa, Nf)[:n]
    corr2 = np.fft.irfft(np.conj(Fh) * Fa, Nf)[:n]

    gradS = const - 2.0 * w0 * h - conv1 - corr2
    grad = dx * gradS

    if not np.all(np.isfinite(grad)):
        grad = np.nan_to_num(grad, nan=0.0, posinf=1e6, neginf=-1e6)
    return grad, dx * topmax


def penalty_fun(x):
    x = np.asarray(x, dtype=float)
    n = x.size
    if not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n, dtype=float)

    phi, g = smooth_phi_grad(x)
    if not np.isfinite(phi):
        return 1e100, np.zeros(n, dtype=float)

    s = float(np.sum(x) - float(_GLOBAL_TARGET))
    f = phi + float(_GLOBAL_MU) * s * s

    if _GLOBAL_ONES is not None and _GLOBAL_ONES.size == n:
        g = g + (2.0 * float(_GLOBAL_MU) * s) * _GLOBAL_ONES
    else:
        g = g + (2.0 * float(_GLOBAL_MU) * s) * np.ones(n, dtype=float)

    if not np.all(np.isfinite(g)):
        g = np.nan_to_num(g, nan=0.0, posinf=1e6, neginf=-1e6)
    return f, g

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return project_density(p, 0.5 * float(n))
    if n0 <= 1:
        return project_density(np.full(max(1, int(n)), 0.5, dtype=float), 0.5 * float(n))

    x0 = (np.arange(n0, dtype=float) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * float(n))


def make_diverse(n, rng, mode):
    n = int(n)
    x = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    if rng is None:
        rng = np.random.default_rng(0)

    if mode == 0:
        f1 = int(rng.integers(1, 7))
        f2 = int(rng.integers(1, 12))
        p1 = float(rng.uniform(0.0, 6.283185307179586))
        p2 = float(rng.uniform(0.0, 6.283185307179586))
        h = 0.5 + 0.32 * np.sin(np.pi * f1 * x + p1) + 0.14 * np.sin(np.pi * f2 * x + p2)
    elif mode == 1:
        nseg = int(rng.integers(4, 12))
        h = np.empty(n, dtype=float)
        for i in range(nseg):
            a = i * n // nseg
            b = (i + 1) * n // nseg
            if b <= a:
                b = a + 1
            h[a:b] = float(rng.uniform(0.05, 0.95))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.80, 0.20).astype(float)
        h = h + 0.08 * rng.normal(0.0, 1.0, n)
    elif mode == 3:
        h = (rng.random(n) < 0.5).astype(float)
        h = h + 0.05 * rng.normal(0.0, 1.0, n)
    else:
        center = float(rng.uniform(0.2, 1.8))
        slope = float(rng.uniform(2.0, 10.0))
        h = 1.0 / (1.0 + np.exp(-slope * (x - center)))
        h = 0.20 + 0.60 * h

    return project_density(h, 0.5 * float(n))


def perturb_h(base, amp):
    base = np.asarray(base, dtype=float)
    n = base.size
    if _RNG is None:
        return project_density(base, 0.5 * float(n))
    r = _RNG.standard_normal(n).astype(float)
    r -= float(np.mean(r))
    sd = float(np.std(r))
    if sd > 1e-12:
        r = r / sd * float(amp)
    return project_density(base + r, 0.5 * float(n))


def try_alternatives(h, rng):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return h, float("inf")

    target = 0.5 * float(n)
    best_h = project_density(h, target)
    best_c = erdos_exact(best_h)

    alts = []
    alts.append(0.5 * (best_h + best_h[::-1]))
    if n > 1:
        alts.append(np.roll(best_h, n // 4))
        alts.append(np.roll(best_h, -n // 4))
        if n > 3:
            alts.append(np.roll(best_h, n // 3))
            alts.append(np.roll(best_h, -n // 3))
    if rng is not None and n > 4:
        alts.append(np.roll(best_h, int(rng.integers(n))))
        alts.append(np.roll(best_h, int(rng.integers(n))))

    alts.append((best_h > 0.5).astype(float))

    if rng is not None and n > 8:
        noise = rng.standard_normal(n).astype(float)
        noise -= float(np.mean(noise))
        sd = float(np.std(noise))
        if sd > 1e-12:
            alts.append(best_h + 0.06 * (noise / sd))

    for alt in alts:
        alt = project_density(alt, target)
        cf = search_cost(alt)
        if cf < best_c - 1e-8:
            ce = erdos_exact(alt)
            if ce < best_c - 1e-12:
                best_h = alt
                best_c = ce

    return best_h, best_c


def active_polish(h_initial, deadline, max_iter=1200, K_frac=0.03, K_min=8,
                  tau_frac=0.005,
                  step_sizes=(0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.00125)):
    if h_initial is None:
        return None, float("inf")

    h = np.asarray(h_initial, dtype=float)
    n = h.size
    if n < 2:
        return h, erdos_exact(h)

    target = 0.5 * float(n)
    inp = project_density(h, target)
    inp_c = erdos_exact(inp)

    best = inp.copy()
    best_fast = search_cost(best)
    fail = 0

    for _ in range(int(max_iter)):
        if time.perf_counter() >= float(deadline):
            break

        g, _ = active_gradient_smooth(best, K_frac=K_frac, K_min=K_min, tau_frac=tau_frac)
        if not np.all(np.isfinite(g)):
            break

        g = g - float(np.mean(g))
        norm = float(np.linalg.norm(g))
        if norm < 1e-13:
            break

        d = -g / norm
        accepted = False

        for step in step_sizes:
            cand = project_density(best + float(step) * d, target)
            cf = search_cost(cand)
            if cf < best_fast - 1e-8:
                best = cand
                best_fast = cf
                accepted = True
                fail = 0
                break

        if not accepted:
            fail += 1
            if fail == 30 and _RNG is not None:
                noise = _RNG.standard_normal(n).astype(float)
                noise -= float(np.mean(noise))
                cand = project_density(best + 0.02 * noise, target)
                cf = search_cost(cand)
                if cf < best_fast - 1e-10:
                    best = cand
                    best_fast = cf
                fail = 15
            if fail > 60:
                break

    final = project_density(best, target)
    final_c = erdos_exact(final)
    if final_c < inp_c - 1e-12:
        return final, final_c
    return inp, inp_c


def pair_swap_search(h_initial, deadline, max_iter=1000, npos=10, nneg=10):
    if h_initial is None:
        return None, float("inf")

    h = np.asarray(h_initial, dtype=float)
    n = h.size
    if n < 2:
        return h, erdos_exact(h)

    target = 0.5 * float(n)
    inp = project_density(h, target)
    inp_c = erdos_exact(inp)

    best = inp.copy()
    best_fast = search_cost(best)
    fail = 0

    nuse = max(4, min(int(npos), int(nneg), n // 3))
    if nuse < 2:
        return inp, inp_c

    for _ in range(int(max_iter)):
        if time.perf_counter() >= float(deadline):
            break

        g, _ = active_gradient_smooth(best, K_frac=0.012, K_min=6, tau_frac=0.003)
        if not np.all(np.isfinite(g)):
            break

        neg_idx = np.argpartition(g, nuse)[:nuse]
        pos_idx = np.argpartition(-g, nuse)[:nuse]
        found = False

        for ii in neg_idx:
            ii = int(ii)
            for jj in pos_idx:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-9:
                    continue
                ds = (cap, 0.5 * cap, 0.75 * cap, min(0.05, cap), min(0.015, cap))
                for dval in ds:
                    dval = float(dval)
                    if dval <= 1e-10 or dval > cap + 1e-12:
                        continue
                    cand = best.copy()
                    cand[ii] += dval
                    cand[jj] -= dval
                    if cand[ii] < -1e-10 or cand[ii] > 1.0 + 1e-10:
                        continue
                    if cand[jj] < -1e-10 or cand[jj] > 1.0 + 1e-10:
                        continue
                    cf = search_cost(cand)
                    if cf < best_fast - 1e-8:
                        best = cand
                        best_fast = cf
                        found = True
                        fail = 0
                        break
                if found:
                    break
            if found:
                break

        if not found and _RNG is not None:
            for _r in range(6):
                ii = int(_RNG.integers(n))
                jj = int(_RNG.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-9:
                    continue
                dval = (0.15 + 0.85 * float(_RNG.random())) * cap
                cand = best.copy()
                cand[ii] += dval
                cand[jj] -= dval
                if cand[ii] < -1e-10 or cand[ii] > 1.0 + 1e-10:
                    continue
                if cand[jj] < -1e-10 or cand[jj] > 1.0 + 1e-10:
                    continue
                cf = search_cost(cand)
                if cf < best_fast - 1e-8:
                    best = cand
                    best_fast = cf
                    found = True
                    fail = 0
                    break

        if not found:
            fail += 1
            if fail > 45:
                break

    final = project_density(best, target)
    final_c = erdos_exact(final)
    if final_c < inp_c - 1e-12:
        return final, final_c
    return inp, inp_c


def local_refine(h_initial, deadline, lite=False):
    global _GLOBAL_ALPHA, _GLOBAL_MU, _GLOBAL_TARGET, _GLOBAL_BOUNDS, _GLOBAL_ONES

    if h_initial is None:
        return None, float("inf")

    h = np.asarray(h_initial, dtype=float).copy()
    n = h.size
    if n < 2:
        return project_density(h, 0.5 * float(max(1, n))), erdos_exact(h)

    target = 0.5 * float(n)
    best_h, best_c = try_alternatives(h, _RNG)

    if time.perf_counter() < float(deadline) - 20.0 and n <= 4096:
        _GLOBAL_TARGET = target
        _GLOBAL_BOUNDS = [(0.0, 1.0)] * int(n)
        _GLOBAL_ONES = np.ones(n, dtype=float)

        if lite:
            if n <= 80:
                alphas = (6.0, 20.0, 60.0)
                maxiter = 40
            elif n <= 300:
                alphas = (8.0, 30.0)
                maxiter = 20
            else:
                alphas = (10.0, 40.0)
                maxiter = 12
            muls = (8.0,)
        else:
            if n <= 120:
                alphas = (6.0, 20.0, 60.0, 180.0)
                maxiter = 60
            elif n <= 500:
                alphas = (8.0, 30.0, 100.0, 300.0)
                maxiter = 45
            elif n <= 1200:
                alphas = (10.0, 40.0, 120.0, 300.0)
                maxiter = 30
            else:
                alphas = (12.0, 50.0, 150.0)
                maxiter = 20
            muls = (10.0, 0.8)

        x = best_h.copy()
        work_fast = search_cost(x)

        for alpha in alphas:
            if time.perf_counter() >= float(deadline) - 12.0:
                break
            for mu in muls:
                if time.perf_counter() >= float(deadline) - 10.0:
                    break
                _GLOBAL_ALPHA = float(alpha)
                _GLOBAL_MU = float(mu)
                opts = {
                    "maxiter": int(maxiter),
                    "ftol": 1e-11,
                    "gtol": 1e-8,
                    "maxls": 15,
                    "maxcor": 12,
                    "maxfun": int(max(800, 14 * maxiter)),
                    "disp": False,
                }
                try:
                    res = minimize(penalty_fun, x, method="L-BFGS-B",
                                   bounds=_GLOBAL_BOUNDS, jac=True, options=opts)
                except Exception:
                    res = None

                if res is None or getattr(res, "x", None) is None:
                    continue

                cand = project_density(np.asarray(res.x, dtype=float), target)
                cf = search_cost(cand)
                if cf < work_fast - 1e-10:
                    x = cand
                    work_fast = cf

            if time.perf_counter() >= float(deadline) - 6.0:
                break

            ce = erdos_exact(x)
            if ce < best_c - 1e-12:
                best_h = x
                best_c = ce

            if ce >= best_c + 0.003:
                x = best_h
            work_fast = search_cost(x)

        best_h, best_c = try_alternatives(best_h, _RNG)

    if time.perf_counter() < float(deadline) - 5.0:
        if lite:
            h2, c2 = active_polish(best_h, float(deadline), max_iter=300,
                                   K_frac=0.03, K_min=8, tau_frac=0.006)
            if c2 < best_c - 1e-12:
                best_h = h2
                best_c = c2
        else:
            rem = float(deadline) - time.perf_counter()
            act1_dl = max(time.perf_counter() + 8.0, float(deadline) - max(10.0, 0.35 * rem))
            h2, c2 = active_polish(best_h, act1_dl, max_iter=3000,
                                   K_frac=0.02, K_min=6, tau_frac=0.004,
                                   step_sizes=(0.08, 0.04, 0.02, 0.01, 0.005, 0.0025, 0.00125))
            if c2 < best_c - 1e-12:
                best_h = h2
                best_c = c2

            rem = float(deadline) - time.perf_counter()
            if n <= 2400 and time.perf_counter() < float(deadline) - 8.0:
                pair_dl = max(time.perf_counter() + 5.0, float(deadline) - max(8.0, 0.25 * rem))
                h3, c3 = pair_swap_search(best_h, pair_dl, max_iter=1500, npos=10, nneg=10)
                if c3 < best_c - 1e-12:
                    best_h = h3
                    best_c = c3

            if time.perf_counter() < float(deadline) - 4.0:
                h4, c4 = active_polish(best_h, float(deadline) - 2.0, max_iter=2000,
                                       K_frac=0.07, K_min=12, tau_frac=0.02,
                                       step_sizes=(0.05, 0.025, 0.0125, 0.006, 0.003, 0.0015))
                if c4 < best_c - 1e-12:
                    best_h = h4
                    best_c = c4

    final = project_density(best_h, target)
    final_c = erdos_exact(final)
    return final, final_c


def global_multistart(best_h, best_c, deadline, seed):
    global _DE_N

    if differential_evolution is not None and time.perf_counter() < float(deadline) - 60.0:
        _DE_N = 48
        bounds = [(0.0, 1.0)] * 48
        try:
            res = differential_evolution(
                _de_obj,
                bounds,
                seed=int(seed) + 17,
                maxiter=40,
                popsize=10,
                tol=1e-6,
                polish=False,
                updating="immediate",
                disp=False
            )
            h = project_density(np.asarray(res.x, dtype=float), 24.0)
            c = erdos_exact(h)
            if c < best_c - 1e-12:
                best_h = h
                best_c = c
            if time.perf_counter() < float(deadline) - 25.0:
                h2, c2 = local_refine(h, min(float(deadline) - 25.0, time.perf_counter() + 25.0),
                                      lite=True)
                if c2 < best_c - 1e-12:
                    best_h = h2
                    best_c = c2
        except Exception:
            pass

    small_ns = (60, 90, 120)
    for n in small_ns:
        if time.perf_counter() >= float(deadline) - 25.0:
            break

        cands = []
        cands.append(upsample_h(best_h, n))
        if _RNG is not None:
            cands.append(make_diverse(n, _RNG, 0))
            cands.append(make_diverse(n, _RNG, 3))
            cands.append(perturb_h(upsample_h(best_h, n), 0.07))

        uniq = []
        for s in cands:
            s = project_density(s, 0.5 * float(n))
            dup = False
            for u in uniq:
                if s.size == u.size and np.allclose(s, u, atol=1e-6):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        if not uniq:
            continue

        per = max(8.0, (float(deadline) - time.perf_counter() - 20.0) / float(max(1, len(uniq))))
        for s in uniq:
            if time.perf_counter() >= float(deadline) - 20.0:
                break
            h2, c2 = local_refine(s, min(float(deadline) - 20.0, time.perf_counter() + per),
                                  lite=True)
            if c2 < best_c - 1e-12:
                best_h = h2
                best_c = c2

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global _RNG

    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 25.0)
    _RNG = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    parent_h = None

    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            p = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            p = np.clip(p, 0.0, 1.0)
            pp = project_density(p, 0.5 * float(p.size))
            pc = erdos_exact(pp)
            parent_h = pp.copy()
            best_h = pp.copy()
            best_c = pc
    except Exception:
        parent_h = None

    if best_h is None:
        n0 = 120
        x = (np.arange(n0, dtype=float) + 0.5) * (2.0 / float(n0))
        h_init = 0.5 + 0.32 * np.sin(3.0 * np.pi * x) + 0.16 * np.sin(5.0 * np.pi * x + 0.7)
        best_h = project_density(h_init, 0.5 * float(n0))
        best_c = erdos_exact(best_h)

    best_h, best_c = try_alternatives(best_h, _RNG)

    total = max(1.0, budget - 25.0)

    phase1_end = start + min(180.0, max(45.0, 0.13 * total))
    phase1_end = max(start + 5.0, min(phase1_end, deadline - 5.0))
    if time.perf_counter() < phase1_end:
        best_h, best_c = local_refine(best_h, float(phase1_end), lite=False)

    reserve_high = max(220.0, 0.20 * total)
    phase2_end = min(deadline - reserve_high,
                     phase1_end + min(130.0, max(30.0, 0.08 * total)))
    phase2_end = max(phase1_end, min(phase2_end, deadline - 15.0))
    if time.perf_counter() < phase2_end:
        best_h, best_c = global_multistart(best_h, best_c, float(phase2_end), seed)

    high_end = deadline - max(45.0, 0.04 * total)
    high_start = time.perf_counter()

    cur_n = max(int(best_h.size), 120)
    stages = []
    for _ in range(4):
        if cur_n < 900:
            nxt = 900
        elif cur_n < 1700:
            nxt = 1800
        elif cur_n < 2500:
            nxt = 2700
        else:
            nxt = min(3600, int(cur_n * 1.5))
        if nxt <= cur_n:
            break
        stages.append(int(nxt))
        cur_n = int(nxt)

    if stages and high_start < high_end - 120.0:
        for idx, nxt in enumerate(stages):
            now = time.perf_counter()
            if now >= high_end - 60.0:
                break
            rem_stages = len(stages) - idx
            stage_end = min(high_end - 10.0, now + max(60.0, (high_end - now) / float(rem_stages)))
            if stage_end - now < 35.0:
                break

            base = best_h
            h_base = upsample_h(base, nxt)
            h_base, c_base = try_alternatives(h_base, _RNG)

            if parent_h is not None:
                h_par = upsample_h(parent_h, nxt)
                h_par, c_par = try_alternatives(h_par, _RNG)
                if c_par < c_base - 1e-12:
                    h_base = h_par

            h2, c2 = local_refine(h_base, min(stage_end - 5.0, deadline - 30.0), lite=False)
            if c2 < best_c - 1e-12:
                best_h = h2
                best_c = c2

    final_end = deadline - 5.0
    if best_h is not None and time.perf_counter() < final_end:
        best_h, best_c = local_refine(best_h, float(final_end), lite=False)

    if best_h is None:
        h = project_density(np.full(90, 0.5, dtype=float), 45.0)
        return h, erdos_exact(h), 90

    fin = project_density(best_h, 0.5 * float(best_h.size))
    fin_c = erdos_exact(fin)
    return fin, float(fin_c), int(fin.size)
