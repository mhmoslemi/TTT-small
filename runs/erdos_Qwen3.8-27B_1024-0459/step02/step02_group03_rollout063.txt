    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, alpha):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    if L == 0:
        grad = (1.0 - 2.0 * h)
    elif L > 0:
        grad[L:] += (1.0 - h[:n-L])
        grad[:n-L] -= h[L:]
    else:

        grad[:n-Lp] += (1.0 - h[Lp:])
        grad[Lp:] -= h[:n-Lp]
    
    h_new = h - alpha * grad
    return project_feasible(h_new, n)

    best_h = h.copy()
    best_c5 = compute_c5(h)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.maximum(h, 1.0 - h)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    h_b = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_b))
    n_want = n // 2
    if n_ones > n_want:
        idx1 = np.where(h_b == 1)[0]
        order = np.argsort(h[idx1])[:n_ones - n_want]
        h_b[idx1[order]] = 0.0
    elif n_ones < n_want:
        idx0 = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx0])[:n_want - n_ones]
        h_b[idx0[order]] = 1.0

def interp_n(h_src, n_dst):

def continuous_optimize(h0, n, time_limit, rng):

    h = h0.copy()

    best_c5 = np.max(corr) * (2.0 / n)

    alpha = 0.5
    start_t = time.time()
    
    step = 0
    while time.time() - start_t < time_limit:
        if step % 200 == 0:
            alpha = max(0.01, 0.5 / (step + 1) ** 0.3)
        
        # Get top shifts contributing to max
        top_k = np.argsort(corr)[-8:]
        
        grad_sum = np.zeros(n)
        for idx in top_k:
            L = idx - (n - 1)
            grad = np.zeros(n)
            if L == 0:
                grad[:] = 1.0 - 2.0 * h
            elif L > 0:
                grad[L:] += (1.0 - h[:n - L])
                grad[:n - L] -= h[L:]
            else:
                Lp = -L
                grad[:n - Lp] += (1.0 - h[Lp:])
                grad[Lp:] -= h[:n - Lp]

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        step += 1
    
    return best_h, best_c5
