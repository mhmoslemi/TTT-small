def smooth_grad(x, beta):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    value = (m + np.log(total) / beta) * dx
    
    weights = exps / total
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return value, dx * grad

def binary_sa_search(h0, n, rng, time_limit, t_start):
    """Simulated annealing on binary sequences."""
    h = h0.copy()
    # Make binary
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            ones = np.where(h == 1)[0]
            order = rng.permutation(len(ones))
            h[ones[order[:n_ones - target]]] = 0
        else:
            zeros = np.where(h == 0)[0]
            order = rng.permutation(len(zeros))
            h[zeros[order[:target - n_ones]]] = 1
    
    current_energy = np.max(np.correlate(h, 1.0 - h, mode='full'))
    best_h = h.copy()
    best_energy = current_energy
    
    T_start = 5.0
    T_end = 0.1
    n_steps_estimate = 50000
    cool_rate = (T_end / T_start) ** (1.0 / n_steps_estimate)
    
    step = 0
    while time.time() - t_start < time_limit:
        step += 1
        T = T_start * (cool_rate ** step)
        
        if T < T_end:
            T = T_end
        
        # Generate a move: flip two positions
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i1 = rng.randint(len(ones))
        i0 = rng.randint(len(zeros))
        p1 = ones[int(i1)]
        p0 = zeros[int(i0)]
        
        h[p1] = 0
        h[p0] = 1
        new_energy = np.max(np.correlate(h, 1.0 - h, mode='full'))
        
        delta = new_energy - current_energy
        if delta < 0 or (T > 0 and rng.random() < np.exp(-delta / T)):
            current_energy = new_energy
            if new_energy < best_energy:
                best_energy = new_energy
                best_h = h.copy()
        else:
            h[p1] = 1
            h[p0] = 0
        
        if step % 5000 == 0:
            pass  # Could print progress
    
    return best_h, best_energy * (2.0 / n)
