def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:

    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))

            h = np.clip(h, 0.0, 1.0)

        if c_t < best_c:
            best_h = h_t
            best_c = c_t

        # Run again with n_top=4 for balancing
        rem = deadline - time.perf_counter()
        if rem > 20.0:
            alloc2 = min(rem * 0.3, 200.0)
            h_t2, c_t2 = targeted_lag_search(best_h, time.perf_counter() + alloc2, max_iter=200000, n_top=5)
            if c_t2 < best_c:
                best_h = h_t2
                best_c = c_t2

    # Phase 3: Balanced gradient search
    rem = deadline - time.perf_counter()
    if rem > 20.0:
        alloc = min(rem * 0.25, 150.0)
        h_g, c_g = balanced_gradient_search(best_h, time.perf_counter() + alloc, max_iter=100000, top_count=6)
        if c_g < best_c:
            best_h = h_g
            best_c = c_g

    # Phase 4: Binary push and local search
    rem = deadline - time.perf_counter()
    if rem > 20.0:
        n_cur = best_h.size
        h_bin = best_h.copy()
        # Push towards binary
        for thresh in (0.15, 0.1, 0.05, 0.03):
            for i in range(n_cur):
                if h_bin[i] < thresh:
                    h_bin[i] = 0.0
                elif h_bin[i] > 1.0 - thresh:
                    h_bin[i] = 1.0
            h_bin = project_density(h_bin, 0.5 * n_cur)
            c_bin = evaluate_erdos(h_bin)
            if c_bin < best_c:
                best_c = c_bin
                best_h = h_bin.copy()

        # Binary local search
        rem = deadline - time.perf_counter()
        if rem > 15.0:
            alloc = min(rem * 0.15, 60.0)
            h_bl, c_bl = binary_local_search(h_bin, time.perf_counter() + alloc, max_iter=50000)
            if c_bl < best_c:
                best_h = h_bl
                best_c = c_bl

    # Phase 5: Try other resolutions
    rem = deadline - time.perf_counter()
    n_tried = {best_n, n_main}
    candidate_ns = [300, 360, 400, 450, 500, 540, 600]
    for cn in candidate_ns:
        if cn in n_tried or rem < 50.0:
            continue
        n_tried.add(cn)
        # Upsample current best
        h_up = upsample_h(best_h, cn)
        # SLSQP briefly
        h_s, c_s = slsqp_optimize(cn, h_up, time.perf_counter() + min(30.0, rem * 0.3))
        # Targeted search
        rem = deadline - time.perf_counter()
        if rem > 15.0:
            h_ts, c_ts = targeted_lag_search(h_s, time.perf_counter() + min(rem * 0.4, 80.0), max_iter=100000, n_top=3)
            if c_ts < c_s:
                h_s, c_s = h_ts, c_ts
        if c_s < best_c:
            best_h = h_s
            best_c = c_s
            best_n = cn

    # Phase 6: Final micro-optimization
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        n_final = best_h.size
        target = 0.5 * n_final
        h_fin = project_density(best_h, target)

        # Very fine targeted search
        alloc = min(rem * 0.4, 120.0)
        h_f1, c_f1 = targeted_lag_search(h_fin, time.perf_counter() + alloc, max_iter=200000, n_top=1)
        if c_f1 < best_c:
            best_h = h_f1
            best_c = c_f1

        # Balanced gradient
        rem = deadline - time.perf_counter()
        if rem > 10.0:
            alloc = min(rem * 0.3, 80.0)
            h_f2, c_f2 = balanced_gradient_search(best_h, time.perf_counter() + alloc, max_iter=50000, top_count=4)
            if c_f2 < best_c:
                best_h = h_f2
                best_c = c_f2

        # One more targeted pass
        rem = deadline - time.perf_counter()
        if rem > 8.0:
            alloc = min(rem * 0.5, 60.0)
            h_f3, c_f3 = targeted_lag_search(best_h, time.perf_counter() + alloc, max_iter=100000, n_top=3)
            if c_f3 < best_c:
                best_h = h_f3
                best_c = c_f3

    # Final projection and evaluation
    if best_h is None:
        fallback = project_density(np.full(450, 0.5), 225.0)
        return fallback, evaluate_erdos(fallback), 450

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
