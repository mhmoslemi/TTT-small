    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(tm, n)

def random_binary(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    target = n // 2
    if len(ones_idx) > target:
        h[rng.choice(ones_idx, len(ones_idx) - target, replace=False)] = 0.0
    elif len(ones_idx) < target:
        h[rng.choice(zeros_idx, target - len(ones_idx), replace=False)] = 1.0
    return h

def sa_search(h_init, n, rng, t_start=3.0, t_end=0.005, n_iter=500000, timeout=120.0, p_2swap=0.35):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    cur_c5 = compute_c5_raw(h)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    start_t = time.time()
    for it in range(n_iter):
        if time.time() - start_t > timeout:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        frac = float(it) / float(max(n_iter, 1))
        T = t_start * (t_end / t_start) ** frac
        if T < 1e-10:
            T = 1e-10
        use_2swap = (rng.rand() < p_2swap)
        if use_2swap:
            i1 = int(rng.choice(ones))
            i2 = int(rng.choice(ones))
            j1 = int(rng.choice(zeros))
            j2 = int(rng.choice(zeros))
            if len(set([i1, i2, j1, j2])) < 4:
                use_2swap = False
        if use_2swap:
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            new_c5 = compute_c5_raw(h)
            delta = new_c5 - cur_c5
            accept = (delta <= 0) or (rng.rand() < np.exp(-delta / T))
            if accept:
                cur_c5 = new_c5
                ones.remove(i1)
                ones.remove(i2)
                zeros.remove(j1)
                zeros.remove(j2)
                ones.append(j1)
                ones.append(j2)
                zeros.append(i1)
                zeros.append(i2)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
        else:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5_raw(h)
            delta = new_c5 - cur_c5
            accept = (delta <= 0) or (rng.rand() < np.exp(-delta / T))
            if accept:
                cur_c5 = new_c5
                ones.remove(i)
                zeros.remove(j)
                zeros.append(i)
                ones.append(j)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
    return best_h, best_c5

def targeted_greedy(h_init, n, rng, max_trials=200000, timeout=80.0, top_k=3, stall_limit=6000):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5_raw(h)
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    no_improve = 0
    start_t = time.time()
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        worst_idx = int(top_idx[int(rng.randint(0, top_k))])
        s = worst_idx - (n - 1)
        bad = []
        lo = max(0, -s)
        hi = min(n, n - s)
        for idx in range(lo, hi):
            ni = idx + s
            if 0 <= ni < n and h[idx] == 1.0 and h[ni] == 0.0:
                bad.append(idx)
        if bad:
            i = bad[int(rng.randint(0, len(bad)))]
        else:
            i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5_raw(h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h.copy()
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return best_h, best_c5

def sa_restart_local(h_init, n, rng, t_start=5.0, t_end=0.02, n_iter=100000, timeout=100.0, p_2swap=0.4):
    """SA that restarts with greedy local search when stagnating."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    cur_c5 = compute_c5_raw(h)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    start_t = time.time()
    stall = 0
    for it in range(n_iter):
        if time.time() - start_t > timeout:
            break
        if len(ones) < 2 or len(zeros) < 2:
            break
        frac = float(it) / float(max(n_iter, 1))
        T = t_start * (t_end / t_start) ** frac
        if T < 1e-10:
            T = 1e-10
        use_2swap = (rng.rand() < p_2swap)
        if use_2swap:
            i1 = int(rng.choice(ones))
            i2 = int(rng.choice(ones))
            j1 = int(rng.choice(zeros))
            j2 = int(rng.choice(zeros))
            if len(set([i1, i2, j1, j2])) < 4:
                use_2swap = False
        if use_2swap:
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            new_c5 = compute_c5_raw(h)
            delta = new_c5 - cur_c5
            accept = (delta <= 0) or (rng.rand() < np.exp(-delta / T))
            if accept:
                cur_c5 = new_c5
                ones.remove(i1)
                ones.remove(i2)
                zeros.remove(j1)
                zeros.remove(j2)
                ones.append(j1)
                ones.append(j2)
                zeros.append(i1)
                zeros.append(i2)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
                stall += 1
        else:
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5_raw(h)
            delta = new_c5 - cur_c5
            accept = (delta <= 0) or (rng.rand() < np.exp(-delta / T))
            if accept:
                cur_c5 = new_c5
                ones.remove(i)
                zeros.remove(j)
                zeros.append(i)
                ones.append(j)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                stall += 1
        if stall > 8000:
            h = best_h.copy()
            cur_c5 = best_c5
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
            stall = 0
    return best_h, best_c5

def slsqp_smooth_max(h, n, max_iter=2000):
    bounds = [(0.0, 1.0)] * n
    beta = 50.0
    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        cmax = np.max(corr)
        shifted = beta * (corr - cmax)
        shifted = np.clip(shifted, -50, 50)
        return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    con = {'type': 'eq', 'fun': con_fun, 'jac': con_jac}
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': max_iter, 'ftol': 1e-14})
        if hasattr(res, 'x') and res.x is not None:
            h_out = make_feasible(res.x, n)
            return h_out
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent solution
    h_parent = None
    n_parent = 0
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
            n_parent = len(h_parent_arr)
    except Exception:
        h_parent = None

    # Phase 1: Optimize at n=256 from parent
    n256 = 256
    if h_parent is not None:
        h_init_256 = make_feasible(interp_h(h_parent, n256), n256)
    else:
        h_init_256 = thue_morse_init(n256)

    best_h_256 = h_init_256.copy()
    best_c5_256 = compute_c5_raw(h_init_256)
    best_c5_256_scaled = best_c5_256 * (2.0 / n256)

    # SA pass 1 on parent (long)
    if remaining() > 200:
        h1, c5r1 = sa_restart_local(h_init_256, n256, rng,
                                     t_start=4.0, t_end=0.01,
                                     n_iter=400000, timeout=180.0, p_2swap=0.35)
        if c5r1 < best_c5_256:
            best_c5_256 = c5r1
            best_h_256 = h1.copy()
            best_c5_256_scaled = best_c5_256 * (2.0 / n256)

    # Targeted greedy from best
    if remaining() > 80:
        h2, c5r2 = targeted_greedy(best_h_256, n256, rng,
                                    max_trials=150000, timeout=70.0, top_k=3)
        if c5r2 < best_c5_256:
            best_c5_256 = c5r2
            best_h_256 = h2.copy()
            best_c5_256_scaled = best_c5_256 * (2.0 / n256)

    # SA from different inits
    inits_256 = [thue_morse_init(n256), random_binary(n256, rng), random_binary(n256, rng)]
    for h0 in inits_256:
        if remaining() < 60:
            break
        h3, c5r3 = sa_search(h0, n256, rng, t_start=3.0, t_end=0.01,
                              n_iter=200000, timeout=50.0, p_2swap=0.3)
        if c5r3 < best_c5_256:
            best_c5_256 = c5r3
            best_h_256 = h3.copy()
            best_c5_256_scaled = best_c5_256 * (2.0 / n256)

    # More targeted search
    if remaining() > 60:
        h4, c5r4 = targeted_greedy(best_h_256, n256, rng,
                                    max_trials=100000, timeout=remaining() - 30, top_k=5)
        if c5r4 < best_c5_256:
            best_c5_256 = c5r4
            best_h_256 = h4.copy()
            best_c5_256_scaled = best_c5_256 * (2.0 / n256)

    # SLSQP polish on n=256
    if remaining() > 40:
        h5 = slsqp_smooth_max(best_h_256, n256, max_iter=1500)
        c5r5 = compute_c5_raw(h5)
        if c5r5 < best_c5_256:
            best_c5_256 = c5r5
            best_h_256 = h5.copy()
            best_c5_256_scaled = best_c5_256 * (2.0 / n256)

    # Binary local search after SLSQP (round back)
    if remaining() > 40:
        h6 = np.round(best_h_256).astype(float)
        h6 = make_feasible(h6, n256)
        c5r6 = compute_c5_raw(h6)
        if c5r6 < best_c5_256:
            best_c5_256 = c5r6
            best_h_256 = h6.copy()
        h6b, c5r6b = targeted_greedy(h6, n256, rng,
                                       max_trials=80000, timeout=remaining() - 20, top_k=3)
        if c5r6b < best_c5_256:
            best_c5_256 = c5r6b
            best_h_256 = h6b.copy()
            best_c5_256_scaled = best_c5_256 * (2.0 / n256)

    # Phase 2: Try n=512 (upscale parent and optimize)
    n512 = 512
    best_h_512 = None
    best_c5_512 = float('inf')
    best_c5_512_scaled = float('inf')

    if remaining() > 300:
        if h_parent is not None:
            h_init_512 = make_feasible(interp_h(h_parent, n512), n512)
        else:
            h_init_512 = make_feasible(interp_h(best_h_256, n512), n512)

        h10, c5r10 = sa_restart_local(h_init_512, n512, rng,
                                       t_start=4.0, t_end=0.01,
                                       n_iter=150000, timeout=min(120, remaining() - 60),
                                       p_2swap=0.3)
        if c5r10 < best_c5_512:
            best_c5_512 = c5r10
            best_h_512 = h10.copy()
            best_c5_512_scaled = best_c5_512 * (2.0 / n512)

        if remaining() > 80 and best_h_512 is not None:
            h11, c5r11 = targeted_greedy(best_h_512, n512, rng,
                                          max_trials=80000, timeout=min(60, remaining() - 40),
                                          top_k=3)
            if c5r11 < best_c5_512:
                best_c5_512 = c5r11
                best_h_512 = h11.copy()
                best_c5_512_scaled = best_c5_512 * (2.0 / n512)

        # SA from upscaled best_256
        if remaining() > 100:
            h_up = make_feasible(interp_h(best_h_256, n512), n512)
            h12, c5r12 = sa_search(h_up, n512, rng, t_start=3.0, t_end=0.01,
                                    n_iter=100000, timeout=min(80, remaining() - 40),
                                    p_2swap=0.35)
            if c5r12 < best_c5_512:
                best_c5_512 = c5r12
                best_h_512 = h12.copy()
                best_c5_512_scaled = best_c5_512 * (2.0 / n512)

    # Phase 3: Final SLSQP on best overall
    overall_best_c5 = min(best_c5_256_scaled, best_c5_512_scaled)
    overall_best_h = best_h_256.copy()
    overall_n = n256

    if best_c5_512 < best_c5_256 * (2.0 / n256) * (n256 / n512):
        # Compare fairly
        pass

    # Determine best by scaled value
    if best_c5_512_scaled < best_c5_256_scaled and best_h_512 is not None:
        overall_best_h = best_h_512.copy()
        overall_n = n512
        overall_best_c5 = best_c5_512_scaled
    else:
        overall_best_h = best_h_256.copy()
        overall_n = n256
        overall_best_c5 = best_c5_256_scaled

    # Final polish
    if remaining() > 30:
        h_pol = slsqp_smooth_max(overall_best_h, overall_n, max_iter=1000)
        c5_pol = compute_c5(h_pol)
        if c5_pol < overall_best_c5:
            overall_best_h = h_pol.copy()
            overall_best_c5 = c5_pol

    # Final feasibility check
    overall_best_h = make_feasible(overall_best_h, overall_n)
    overall_best_c5 = compute_c5(overall_best_h)

    return (overall_best_h, overall_best_c5, overall_n)
