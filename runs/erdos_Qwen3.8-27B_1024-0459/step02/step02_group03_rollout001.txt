  target=n/2.0
  h=np.clip(h,0,1)

    s=h.sum(); diff=target-s
    if abs(diff)<1e-12: break
    if diff>0:

    else:
      mask=h>0.0
    cnt=mask.sum()
    if cnt==0:

import numpy as np
import time

try:
    from scipy.optimize import minimize, linprog
except Exception:
    from scipy.optimize import minimize
    linprog = None


_N = 0
_BETA = 1.0
_PEN = 0.0
_DEADLINE = 0.0


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    if m < 0.0 or not np.isfinite(m):
        m = 0.0
    return m * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    if h.size == 0:
        return h
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)

    for _ in range(10):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h = np.clip(h + diff / n, 0.0, 1.0)

    for _ in range(100):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            room = 1.0 - h[idx]
            if room > abs(diff):
                h[idx] += diff
                break
            h[idx] = 1.0
        else:
            idx = int(np.argmax(h))
            room = h[idx]
            if room > abs(diff):
                h[idx] += diff
                break
            h[idx] = 0.0

    h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    y_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(y_dst, n_dst)


def make_spectral_start(n, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5, dtype=float)
    for f in (1.0, 2.0, 3.0, 5.0):
        amp = 0.14 / max(0.6, 0.5 * f)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(2.0 * np.pi * f * x / 2.0 + ph)
    z = rng.normal(0.0, 0.05, n)
    if n >= 5:
        z = np.convolve(z, np.ones(5) / 5.0, mode="same")
    h += z
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)


def _smooth_value(corr, beta):
    if beta <= 0.0:
        return float(np.max(corr))
    m = float(np.max(corr))
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -745.0, 0.0)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + np.log(s) / beta)


def _smooth_weights(corr, beta):
    m = len(corr)
    if beta <= 0.0:
        return np.full(m, 1.0 / m)
    mx = float(np.max(corr))
    shifted = beta * (corr - mx)
    shifted = np.clip(shifted, -745.0, 0.0)
    e = np.exp(shifted)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(m, 1.0 / m)
    return e / s


def _lbfgs_obj(x):
    if _N <= 0:
        return 0.0
    dx = 2.0 / _N
    corr = np.correlate(x, 1.0 - x, mode="full")
    val = _smooth_value(corr, _BETA) * dx
    diff = float(np.sum(x)) - 0.5 * _N
    if _PEN != 0.0:
        val += _PEN * diff * diff
    return float(val)


def _lbfgs_grad(x):
    n = _N
    if n <= 0:
        return np.zeros(0)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    w = _smooth_weights(corr, _BETA)
    grad = np.zeros(n, dtype=float)

    for k in range(len(corr)):
        wk = float(w[k])
        if wk < 1e-12:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif L > 0:
            grad[L:n] += wk * (1.0 - x[:n - L])
            grad[:n - L] -= wk * x[L:n]
        else:
            Lp = -L
            grad[:n - Lp] += wk * (1.0 - x[Lp:n])
            grad[Lp:n] -= wk * x[:n - Lp]

    grad *= dx
    diff = float(np.sum(x)) - 0.5 * n
    if _PEN != 0.0:
        grad += (2.0 * _PEN * diff) * np.ones(n, dtype=float)
    return grad


def _lbfgs_callback(x, *args):
    return time.time() > _DEADLINE


def polish_lbfgs(h0, n, beta, maxiter, penalty, deadline):
    h0 = project_feasible(h0, n)
    c0 = compute_c5(h0)

    global _N, _BETA, _PEN, _DEADLINE
    _N = int(n)
    _BETA = float(beta)
    _PEN = float(penalty)
    _DEADLINE = max(0.0, float(deadline) - 5.0)

    bounds = [(0.0, 1.0)] * n
    options = {
        "maxiter": int(max(1, maxiter)),
        "ftol": 1e-15,
        "gtol": 1e-7,
        "maxls": 20,
        "iprint": 0,
    }
    try:
        res = minimize(
            _lbfgs_obj,
            h0.copy(),
            jac=_lbfgs_grad,
            method="L-BFGS-B",
            bounds=bounds,
            options=options,
            callback=_lbfgs_callback,
        )
        h1 = h0.copy()
        if res is not None and getattr(res, "x", None) is not None:
            h1 = project_feasible(res.x, n)
    except Exception:
        h1 = h0

    c1 = compute_c5(h1)
    if c1 < c0:
        return h1, float(c1)
    return h0, float(c0)


def polish_sequence(h0, n, deadline, rng):
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    if time.time() > deadline - 20.0:
        return best_h, float(best_c)

    starts = [best_h]
    if time.time() < deadline - 45.0:
        hp = project_feasible(best_h + rng.normal(0.0, 0.02, n), n)
        cp = compute_c5(hp)
        if cp < best_c + 0.02:
            starts.append(hp)

    specs = [
        (50.0, 40.0, 180),
        (200.0, 20.0, 220),
        (700.0, 8.0, 180),
    ]

    for st in starts:
        if time.time() > deadline - 15.0:
            break
        cur = project_feasible(st, n)
        cc = compute_c5(cur)
        for beta, pen, iters in specs:
            if time.time() > deadline - 10.0:
                break
            if n <= 256:
                mi = int(iters)
            else:
                mi = int(min(iters, 150))
            h2, c2 = polish_lbfgs(cur, n, beta, mi, pen, deadline)
            if c2 < cc - 1e-12:
                cur = h2
                cc = c2
            else:
                if beta < 200.0:
                    continue
                break

        if cc < best_c - 1e-12:
            best_h = cur.copy()
            best_c = float(cc)

    return best_h, float(best_c)


def append_lp_zero(rows, brun, h, n, count):
    row = np.zeros(n + 1, dtype=float)
    row[:n] = 1.0 - 2.0 * h
    row[n] = -1.0
    const = float(count) - float(np.dot(row[:n], h))
    rows.append(row)
    brun.append(-const + 1e-8)


def append_lp_pos(rows, brun, h, n, L, count):
    row = np.zeros(n + 1, dtype=float)
    row[:n - L] += 1.0 - h[L:n]
    row[L:n] -= h[:n - L]
    row[n] = -1.0
    const = float(count) - float(np.dot(row[:n], h))
    rows.append(row)
    brun.append(-const + 1e-8)


def append_lp_neg(rows, brun, h, n, L, count):
    row = np.zeros(n + 1, dtype=float)
    row[L:n] += 1.0 - h[:n - L]
    row[:n - L] -= h[L:n]
    row[n] = -1.0
    const = float(count) - float(np.dot(row[:n], h))
    rows.append(row)
    brun.append(-const + 1e-8)


def build_lp_data_active(h, n, active_tol):
    pos_vals = np.empty(n, dtype=float)
    neg_vals = np.empty(n, dtype=float)

    pos0 = float(np.dot(h, 1.0 - h))
    pos_vals[0] = pos0
    neg_vals[0] = pos0
    maxv = pos0

    for L in range(1, n):
        a = h[:n - L]
        b1 = 1.0 - h[L:n]
        p = float(np.dot(a, b1))
        pos_vals[L] = p

        b2 = 1.0 - h[:n - L]
        g = float(np.dot(h[L:n], b2))
        neg_vals[L] = g

        if p > maxv:
            maxv = p
        if g > maxv:
            maxv = g

    rows = []
    brun = []

    if pos0 >= maxv - active_tol:
        append_lp_zero(rows, brun, h, n, pos0)

    for L in range(1, n):
        if pos_vals[L] >= maxv - active_tol:
            append_lp_pos(rows, brun, h, n, L, pos_vals[L])
        if neg_vals[L] >= maxv - active_tol:
            append_lp_neg(rows, brun, h, n, L, neg_vals[L])

    if not rows:
        append_lp_zero(rows, brun, h, n, pos0)

    A = np.vstack(rows)
    b = np.asarray(brun, dtype=float)
    return A, b


def line_search(h0, h1, n):
    best_c = compute_c5(h0)
    best_a = 0.0
    best_h = h0

    for i in range(1, 25):
        a = float(i) / 24.0
        hh = (1.0 - a) * h0 + a * h1
        cc = compute_c5(hh)
        if cc < best_c:
            best_c = cc
            best_a = a
            best_h = hh

    if best_a > 0.0 and best_a < 1.0:
        lo = max(0.0, best_a - 0.08)
        hi = min(1.0, best_a + 0.08)
        for i in range(1, 25):
            a = lo + (hi - lo) * (float(i) / 24.0)
            hh = (1.0 - a) * h0 + a * h1
            cc = compute_c5(hh)
            if cc < best_c:
                best_c = cc
                best_a = a
                best_h = hh

    return float(best_c), float(best_a), best_h


def improve_linprog(h0, n, deadline, rng):
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    if linprog is None:
        return best_h, float(best_c)

    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n], dtype=float)
    c_obj = np.zeros(n + 1, dtype=float)
    c_obj[n] = 1.0
    bounds = [(0.0, 1.0)] * n + [(0.0, 1e12)]

    step = 0
    stall = 0
    h = best_h

    while time.time() < deadline - 5.0:
        step += 1

        if stall > 4:
            active_tol = 5.0
        elif step < 30:
            active_tol = 1.0
        elif step < 120:
            active_tol = 0.5
        else:
            active_tol = 0.2

        try:
            A, b = build_lp_data_active(h, n, active_tol)
        except Exception:
            break

        if A is None or A.size == 0:
            break

        tlim = max(0.2, min(3.0, max(0.0, deadline - time.time() - 6.0)))
        options = {"time_limit": tlim}

        h1 = None
        try:
            try:
                res = linprog(
                    c_obj,
                    A_ub=A,
                    b_ub=b,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="highs",
                    options=options,
                )
            except Exception:
                res = linprog(
                    c_obj,
                    A_ub=A,
                    b_ub=b,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="highs",
                )
            if res is not None and getattr(res, "x", None) is not None:
                h1 = np.asarray(res.x[:n], dtype=float)
        except Exception:
            break

        if h1 is None or not np.all(np.isfinite(h1)):
            stall += 1
        else:
            h1 = project_feasible(h1, n)
            d = float(np.linalg.norm(h1 - h, 1))
            if d < 1e-8:
                stall += 1
            else:
                c_new, alpha, h_new = line_search(h, h1, n)
                if c_new < best_c - 1e-12:
                    best_h = h_new
                    best_c = c_new
                    h = h_new
                    stall = 0
                else:
                    stall += 1
                    if stall > 8:
                        if time.time() < deadline - 25.0:
                            eps = 0.03 if n <= 320 else 0.02
                            hp = project_feasible(h + rng.normal(0.0, eps, n), n)
                            h = hp
                            stall = 0
                        else:
                            break

        if stall > 25:
            break

    cf = compute_c5(best_h)
    return best_h, float(cf)


def process_n(n, deadline, best_h, best_c, rng):
    if n == best_n:
        local_h = project_feasible(best_h, n)
    else:
        local_h = interp_h(best_h, n)
    local_c = compute_c5(local_h)

    rem = deadline - time.time()
    if rem < 30.0:
        return local_h, float(local_c)

    lp_deadline = min(deadline, time.time() + max(10.0, rem * 0.80))
    if time.time() < lp_deadline - 10.0:
        h1, c1 = improve_linprog(local_h, n, lp_deadline, rng)
        if c1 < local_c - 1e-12:
            local_h = h1
            local_c = c1

    if time.time() < deadline - 12.0:
        h2, c2 = polish_sequence(local_h, n, deadline, rng)
        if c2 < local_c - 1e-12:
            local_h = h2
            local_c = c2

    return local_h, float(local_c)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    budget = float(budget_s)
    if budget <= 0.0:
        budget = 1.0
    deadline = start_time + max(1.0, budget - 25.0)
    rng = np.random.RandomState(int(seed))

    parent = None
    try:
        parent = np.asarray(initial_h_values, dtype=float).ravel()
    except Exception:
        parent = None

    if parent is not None and parent.size > 0:
        pn = int(parent.size)
        parent = project_feasible(parent, pn)
        best_h = parent
        best_n = pn
        best_c = compute_c5(parent)
    else:
        best_n = 256
        best_h = make_spectral_start(best_n, rng)
        best_c = compute_c5(best_h)

    candidates = [best_n]
    try:
        extra_candidates = []
        vals = (
            int(round(best_n * 1.5)),
            int(round(best_n * 2.0)),
            256,
            320,
            best_n + 64,
        )
        for v in vals:
            v = int(v)
            if 64 <= v <= 800 and v not in candidates and v not in extra_candidates:
                extra_candidates.append(v)
        extra_candidates.sort()
        for v in extra_candidates:
            if len(candidates) >= 4:
                break
            candidates.append(v)
    except Exception:
        pass

    for idx, n in enumerate(candidates):
        rem = deadline - time.time()
        if rem < 50.0:
            break
        try:
            n = int(n)
        except Exception:
            continue
        if n < 32 or n > 800:
            continue

        if idx == 0:
            alloc = min(rem * 0.55, 800.0)
        elif idx == 1:
            alloc = min(rem * 0.30, 350.0)
        elif idx == 2:
            alloc = min(rem * 0.35, 250.0)
        else:
            alloc = min(rem * 0.60, 180.0)

        sub_deadline = time.time() + max(15.0, alloc)
        if sub_deadline > deadline:
            sub_deadline = deadline

        if idx > 0:
            trial_h = interp_h(best_h, n)
            try_c = compute_c5(trial_h)
            if try_c > best_c + 0.15:
                continue

        local_h, local_c = process_n(n, sub_deadline, best_h, best_c, rng)
        if local_c < best_c - 1e-12:
            best_h = local_h.copy()
            best_c = float(local_c)
            best_n = n

    if time.time() < deadline - 25.0:
        h3, c3 = polish_sequence(best_h, best_n, deadline, rng)
        if c3 < best_c - 1e-12:
            best_h = h3
            best_c = float(c3)

    final_h = project_feasible(best_h, best_n)
    final_c = compute_c5(final_h)

    try:
        eval_c = evaluate_erdos_solution(final_h)
        if eval_c is not None:
            eval_c = float(eval_c)
            if np.isfinite(eval_c) and eval_c < final_c:
                final_c = eval_c
    except Exception:
        pass

    return (final_h, float(final_c), int(best_n))
