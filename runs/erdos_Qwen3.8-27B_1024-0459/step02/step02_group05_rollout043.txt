def slsqp_polish(h_init, n, max_iter=5000):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))

    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argmax(c)
        # gradient of c[idx] w.r.t. x
        k = idx - (n - 1)
        ks = np.arange(n)
        grad = np.zeros(n)
        # d(c[idx])/d(x[j]) = (1-x[j-k])*I[0<=j-k<n] - x[j+k]*I[0<=j+k<n]

            val = 0.0
            if 0 <= j - k < n:
                val += (1.0 - x[j - k])
            if 0 <= j + k < n:
                val -= x[j + k]
            grad[j] = val
        return obj(x), grad

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,

        pass
    return None

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    h_parent = None
    try:
        if initial_h_values is not None and len(initial_h_values) > 0:
            h_parent = np.array(initial_h_values, dtype=float)
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B on parent resolution
    if remaining() > 150:
        tl = min(120, remaining() - 10)
        h_opt, c5_opt = multi_beta_lbfgsb(best_h, best_n, betas=(5, 20, 100, 500, 2000, 10000),
                                           max_iter_per_beta=2000, time_limit=tl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: SLSQP polish on parent
    if remaining() > 60:
        h_opt = slsqp_polish(best_h, best_n, max_iter=4000)
        if h_opt is not None:
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 3: Multi-resolution L-BFGS-B
    for n_try in [192, 224, 256, 288, 320, 384, 512]:
        if remaining() < 60:
            break
        if n_try == best_n:
            continue
        h_interp = make_feasible(interpolate_to_n(best_h, n_try), n_try)
        tl = min(90, (remaining() - 20) * 0.6)
        h_opt, c5_opt = multi_beta_lbfgsb(h_interp, n_try,
                                           betas=(5, 20, 100, 500, 2000),
                                           max_iter_per_beta=2500, time_limit=tl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try

    # Phase 4: Binary local search at best resolution and nearby
    for n_try in [best_n, best_n + 16, best_n - 16, 256, 320]:
        if remaining() < 40:
            break
        if n_try < 64 or n_try % 2 != 0:
            continue
        h_bin_init = (interpolate_to_n(best_h, n_try) > 0.5).astype(float)
        if abs(np.sum(h_bin_init) - n_try / 2.0) > n_try / 2.0 + 1:
            h_bin_init = random_binary_init(n_try, rng)
        # Ensure exactly n/2 ones
        if np.sum(h_bin_init) != n_try / 2.0:
            ones_idx = np.where(h_bin_init > 0.5)[0]
            zeros_idx = np.where(h_bin_init <= 0.5)[0]
            target_ones = n_try // 2
            cur_ones = len(ones_idx)
            if cur_ones > target_ones:
                for k in range(cur_ones - target_ones):
                    h_bin_init[ones_idx[k]] = 0.0
            elif cur_ones < target_ones:
                for k in range(target_ones - cur_ones):
                    h_bin_init[zeros_idx[k]] = 1.0

        tl = min(60, remaining() - 15)
        h_bs, c5_bs = binary_local_search(h_bin_init, n_try, max_rounds=200, time_limit=tl, rng=rng)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs
            best_n = n_try

    # Phase 5: Multiple random binary restarts
    for trial in range(8):
        if remaining() < 30:
            break
        n_try = best_n
        h_rand = random_binary_init(n_try, rng)
        tl = min(40, remaining() - 10)
        h_bs, c5_bs = binary_local_search(h_rand, n_try, max_rounds=150, time_limit=tl, rng=rng)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs
            best_n = n_try

    # Phase 6: Coordinate descent polish on continuous best
    if remaining() > 30:
        h_cd = best_h.copy()
        c5_cd = compute_c5(h_cd)
        for _ in range(50):
            if remaining() < 15:
                break
            improved = False
            for idx in range(best_n):
                for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                    h_try = h_cd.copy()
                    h_try[idx] = np.clip(h_try[idx] + delta, 0.0, 1.0)
                    # Adjust another variable to maintain sum
                    diff = best_n / 2.0 - np.sum(h_try)
                    if abs(diff) > 1e-12:
                        candidates = np.where((h_try > 0.01) & (h_try < 0.99) & (np.arange(best_n) != idx))[0]
                        if len(candidates) > 0:
                            adj = candidates[np.argmin(np.abs(h_try[candidates] - 0.5))]
                            h_try[adj] = np.clip(h_try[adj] + diff, 0.0, 1.0)
                            if abs(np.sum(h_try) - best_n / 2.0) > 1e-6:
                                continue
                        else:
                            continue
                    c5_try = compute_c5(h_try)
                    if c5_try < c5_cd - 1e-14:
                        h_cd = h_try
                        c5_cd = c5_try
                        improved = True
                        break
            if not improved:
                break
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 7: Final SLSQP polish
    if remaining() > 20:
        h_opt = slsqp_polish(best_h, best_n, max_iter=6000)
        if h_opt is not None:
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 8: Final L-BFGS-B with very high beta
    if remaining() > 20:
        new_h = lbfgsb_optimize(best_h, best_n, beta=20000, mu=100.0, max_iter=3000)
        if new_h is not None:
            new_h = make_feasible(new_h, best_n)
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
