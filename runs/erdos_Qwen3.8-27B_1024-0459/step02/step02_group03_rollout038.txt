from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_values(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    weights = exps / total
    value = m + np.log(total) / beta
    return value, weights

def project_feasible(h, n):
    """Project h onto {x: 0 <= x <= 1, sum(x) = n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Adjust non-saturated values
    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    if np.sum(mask) > 0:
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # If still not feasible, adjust any value
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Distribute among all
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Find the element closest to the boundary in the right direction
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, weights):
    """
    Compute gradient of smooth_max(corr(h)) w.r.t. h.
    weights: array of length 2n-1, the softmax weights for each correlation value.
    """
    n = len(h)
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:-L])
            grad[:n-L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n-Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n-Lp]
    
    return grad

def projected_gradient_step(h, beta, step_size, n):
    """One step of projected gradient descent."""
    corr = compute_corr(h)
    value, weights = smooth_max_values(corr, beta)
    grad = compute_gradient(h, weights)
    dx = 2.0 / n
    grad -= 0  # No constraint gradient needed since we project
    
    h_new = h - step_size * dx * grad
    h_new = project_feasible(h_new, n)
    return h_new, value * dx, weights

def pg_descent(h0, n, beta, max_iter, step_size=None, verbose=False):
    """Projected gradient descent with backtracking line search."""
    h = h0.copy()
    dx = 2.0 / n
    
    # Initial step size
    if step_size is None:
        step_size = n / (beta + 1)  # heuristic
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        corr = compute_corr(h)
        value, weights = smooth_max_values(corr, beta)
        obj = value * dx
        
        if obj < best_c5:
            best_c5 = obj
            best_h = h.copy()
        
        grad = compute_gradient(h, weights) * dx
        
        # Backtracking line search
        alpha = step_size
        for _ in range(30):
            h_try = project_feasible(h - alpha * grad, n)
            corr_try = compute_corr(h_try)
            val_try, _ = smooth_max_values(corr_try, beta)
            obj_try = val_try * dx
            
            if obj_try < obj - 1e-4 * alpha * np.sum(grad * (h_try - h)):
                break
            alpha *= 0.5
        
        h = h_try
        step_size = alpha * 1.1  # slight increase for next iteration
        
    return best_h, best_c5

    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    return m + np.log(total) / beta, exps / total

def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    mask = (h > 1e-12) & (h < 1.0 - 1e-12)
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n-Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n-Lp]
    return grad

def slsqp_optimize(h0, n, beta, max_iter=800):
    dx = 2.0 / n
    def obj(x):
        corr = compute_corr(x)
        val, _ = smooth_max(corr, beta)
        return val * dx
    def jac(x):
        corr = compute_corr(x)
        _, weights = smooth_max(corr, beta)
        return compute_grad(x, weights) * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def pg_optimize(h0, n, beta, max_iter=2000):
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.5 / (beta * dx + 1.0)
    for it in range(max_iter):
        corr = compute_corr(h)
        val, weights = smooth_max(corr, beta)
        obj = val * dx
        if obj < best_c5:
            best_c5 = obj
            best_h = h.copy()
        grad = compute_grad(h, weights) * dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        a = alpha
        for _ in range(40):
            h_new = project_feasible(h - a * grad, n)
            corr_new = compute_corr(h_new)
            val_new, _ = smooth_max(corr_new, beta)
            f_new = val_new * dx
            if f_new < obj - 1e-12:
                break
            a *= 0.5
        h = h_new
        alpha = min(a * 1.05, 10.0 / (beta * dx + 1.0))
    return best_h, best_c5

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def make_fourier_init(n, coeffs, phases):
    x = np.arange(n, dtype=float) / n
    h = np.full(n, 0.5)
    for m, (a, b, ph) in enumerate(zip(coeffs, coeffs, phases), start=1):
        h += a * np.cos(2 * np.pi * m * x + ph)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_random_init(n, rng):
    h = rng.uniform(0.2, 0.8, n)
    return project_feasible(h, n)
