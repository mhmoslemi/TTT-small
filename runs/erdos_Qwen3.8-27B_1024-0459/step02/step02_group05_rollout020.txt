        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def project_box_sum(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-13:
        mask = (proj < 1.0 - 1e-14) if diff > 0 else (proj > 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
    return np.clip(proj, 0.0, 1.0)

def grad_corr_at_lag(h, n, lag):
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        L = -lag
        g[:n - L] += (1.0 - h[L:])
        g[L:] -= h[:n - L]
    return g

def smooth_objective_and_grad(h, n, beta, top_k=20):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    w = exps / np.sum(exps)
    # Top-k
    idx = np.argsort(w)[::-1][:top_k]
    w_top = w[idx] / np.sum(w[idx])
    
    dx = 2.0 / n
    # f ≈ (1/beta) * log(sum exp(beta * corr)) = cmax + (1/beta)*log(sum exps)
    f = (cmax + np.log(np.sum(exps)) / beta) * dx
    
    g = np.zeros(n)
    for i, k in enumerate(idx):
        lag = k - (n - 1)
        g += w_top[i] * grad_corr_at_lag(h, n, lag)
    g *= dx
    return f, g

def pgd_smooth(h, n, beta, max_iter, lr_init=30.0, top_k=20):
    h = h.copy()
    target = n / 2.0
    best_h = h.copy()
    best_f = compute_c5(h)
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.85
    
    for it in range(max_iter):
        f, g = smooth_objective_and_grad(h, n, beta, top_k)
        
        # Momentum update
        momentum = mu * momentum - lr * g
        h_new = h + momentum
        h_new = project_box_sum(h_new, target)
        
        # Check if improved
        f_new = smooth_objective_and_grad(h_new, n, beta, top_k)[0]
        
        if f_new <= f:
            h = h_new
            f = f_new
            # Decay momentum influence
            momentum *= 0.9
        else:
            # Backtrack
            for _ in range(8):
                lr *= 0.5
                h_new = h + momentum * 0.5
                h_new = project_box_sum(h_new, target)
                f_new = smooth_objective_and_grad(h_new, n, beta, top_k)[0]
                if f_new <= f:
                    h = h_new
                    f = f_new
                    break
            else:
                lr = max(lr, 0.1)
        
        # Track best
        if it % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_f:
                best_f = c5
                best_h = h.copy()
        
        # Adapt lr
        if it % 500 == 499:
            lr *= 0.95
    
    c5_final = compute_c5(h)
    if c5_final < best_f:
        best_f = c5_final
        best_h = h.copy()
    return best_h, best_f

def slsqp_polish(n, h_init, max_iter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    def obj_fn(x):
        return compute_c5(x)
    def obj_grad(x):
        # Use subgradient of C5
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.argmax(corr)
        lag = m - (n - 1)
        dx = 2.0 / n
        return dx * grad_corr_at_lag(x, n, lag)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def block_local_search(h, n, n_rounds, rng, block_sizes=(2,3,4,5)):
    h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    for rnd in range(n_rounds):
        improved = False
        # Try shifting blocks
        for _ in range(n):  # n random block moves per round
            bs = rng.choice(block_sizes)
            if n < bs + 2:
                continue
            i = rng.randint(0, n - bs)
            j = rng.randint(0, n - bs)
            if abs(i - j) < bs:
                continue
            # Move block from i to j (swap)
            block_i = h[i:i+bs].copy()
            block_j = h[j:j+bs].copy()
            h[i:i+bs] = block_j
            h[j:j+bs] = block_i
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
            else:
                # Undo
                h[i:i+bs] = block_i
                h[j:j+bs] = block_j
        # Try gradient-based local perturb
        for _ in range(n // 2):
            idx = rng.choice(n, rng.randint(2, 8), replace=False)
            amt = rng.uniform(-0.05, 0.05)
            old_vals = h[idx].copy()
            h[idx] = np.clip(h[idx] + amt, 0.0, 1.0)
            h = project_box_sum(h, target)
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
            else:
                h[idx] = np.clip(old_vals, 0.0, 1.0)
                h = project_box_sum(h, target)
        if not improved and rnd > 5:
            break
    return h, best_c5

def random_perturb_improve(h, n, n_trials, base_step, rng):

def binary_search_sa(n, rng, n_steps=20000, t_start=0.5, t_end=0.01):

            idxs = rng.choice(n, ns, replace=False)
            amt = rng.uniform(-0.04, 0.04)
            old = h[idxs].copy()
            h[idxs] = np.clip(h[idxs] + amt, 0.0, 1.0)
            h = project_box_sum(h, target)
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                h[idxs] = old
                h = project_box_sum(h, target)
        if not improved and rnd >= 5:
            break
    return h, best_c5

def random_perturb_improve(h, n, n_trials, base_step, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    step = base_step
    for _ in range(n_trials):
        h_try = h.copy()
        n_mod = rng.randint(2, min(12, n // 4))
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = project_box_sum(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
        elif rng.random() < 0.04:
            step = max(step * 0.93, 0.005)
    return h, best_c5

def binary_sa(n, rng, n_steps=20000, t_start=0.6, t_end=0.005):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-15)):
            pass
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) < 40:
            h_parent = None
    except (NameError, Exception):
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent, best_n)
    else:
        best_n = 160
        best_h = make_feasible(np.ones(best_n) * 0.5, best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: Smooth PGD annealing (main workhorse - very fast per iteration)
    pgd_schedule = [
        (10, 60, 12000, 30.0),
        (40, 30, 10000, 25.0),
        (150, 15, 8000, 20.0),
        (600, 10, 6000, 15.0),
        (2000, 5, 4000, 10.0),
        (8000, 3, 3000, 8.0),
    ]
    for beta, topk, max_it, lr0 in pgd_schedule:
        if remaining() < 100:
            break
        h_new, c5_new = pgd_smooth(best_h, best_n, beta, max_it, lr_init=lr0, top_k=topk)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: SLSQP polish on current best
    if remaining() > 80:
        h_opt, c5_opt = slsqp_polish(best_n, best_h, max_iter=3000, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Block local search
    if remaining() > 50:
        n_rounds = min(150, int(remaining() / 0.25))
        best_h, best_c5 = block_local_search(best_h, best_n, n_rounds, rng)

    # Phase 4: Random perturbation improvement
    if remaining() > 30:
        n_trials = min(30000, int((remaining() - 10) / 0.004))
        best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.035, rng)

    # Phase 5: Multi-resolution exploration
    res_list = [80, 100, 128, 160, 192, 240, 256, 320]
    for n_try in res_list:
        if remaining() < 90:
            break
        if n_try == best_n:
            continue
        if n_try < 40:
            continue
        h_init = make_feasible(interpolate_to_n(best_h, n_try), n_try)
        for beta, topk, max_it in [(30, 20, 4000), (300, 8, 3000), (3000, 4, 2000)]:
            if remaining() < 30:
                break
            h_opt, c5_opt = pgd_smooth(h_init, n_try, beta, max_it, lr_init=18.0, top_k=topk)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try
                h_init = h_opt

    # Phase 6: Binary SA at select resolutions
    for n_try in [best_n, 160, 200, 256, 320]:
        if remaining() < 50:
            break
        if n_try < 40:
            continue
        if n_try % 2 != 0:
            n_try += 1
        h_bin, c5_bin = binary_sa(n_try, rng, n_steps=15000, t_start=0.6, t_end=0.005)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.astype(float)
            best_n = n_try

    # Phase 7: Multi-start PGD from perturbations
    for trial in range(4):
        if remaining() < 40:
            break
        h_try = best_h + rng.randn(best_n) * 0.03
        h_try = make_feasible(h_try, best_n)
        for beta in [100, 1000]:
            if remaining() < 20:
                break
            h_opt, c5_opt = pgd_smooth(h_try, best_n, beta, 3000, lr_init=15.0, top_k=8)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 8: Final SLSQP polish
    if remaining() > 40:
        h_opt, c5_opt = slsqp_polish(best_n, best_h, max_iter=4000, ftol=1e-16)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 9: Final local polish
    if remaining() > 20:
        n_rounds = min(80, int(remaining() / 0.2))
        best_h, best_c5 = block_local_search(best_h, best_n, n_rounds, rng)

    if remaining() > 10:
        n_trials = min(15000, int(max(remaining() - 5, 0) / 0.003))
        if n_trials > 200:
            best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.012, rng)

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8

    return (best_h, best_c5, best_n)
