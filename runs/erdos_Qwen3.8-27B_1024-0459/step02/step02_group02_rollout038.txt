        try:

def simulated_annealing(h_init, n, rng, t_start=1.0, t_end=0.01, max_iters=500000, timeout=120.0):
    """Simulated annealing for binary sequences with diverse moves."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    # Ensure exactly n/2 ones
    if np.sum(h) != n / 2:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        target = n // 2
        if len(ones) > target:
            to_flip = rng.choice(ones, len(ones) - target, replace=False)
            h[to_flip] = 0.0
        elif len(ones) < target:
            to_flip = rng.choice(zeros, target - len(ones), replace=False)
            h[to_flip] = 1.0
    
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    best_h = h.copy()
    
    start_t = time.time()
    n_iters_actual = 0
    
    for it in range(max_iters):
        if time.time() - start_t > timeout:
            break
        
        # Temperature (geometric cooling)
        T = t_start * (t_end / t_start) ** (it / max(max_iters - 1, 1))
        
        # Choose a random move type
        move_type = rng.randint(4)
        h_trial = h.copy()
        
        if move_type == 0:
            # Single pair swap: find a 1 and a 0, swap them
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_trial[i] = 0.0
            h_trial[j] = 1.0
        elif move_type == 1:
            # Double swap: swap two 1s with two 0s
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            h_trial[i1] = 0.0
            h_trial[i2] = 0.0
            h_trial[j1] = 1.0
            h_trial[j2] = 1.0
        elif move_type == 2:
            # Segment reversal: reverse a segment
            i = rng.randint(0, n - 2)
            j = rng.randint(i + 1, n - 1)
            h_trial[i:j+1] = h_trial[i:j+1][::-1]
        else:
            # Cyclic shift of a segment
            i = rng.randint(0, n - 1)
            j = rng.randint(i + 1, n - 1)
            shift = rng.randint(1, j - i)
            h_trial[i:j+1] = np.roll(h_trial[i:j+1], -shift)
        
        c5_trial = compute_c5(h_trial)
        delta = c5_trial - current_c5
        
        if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
            h = h_trial
            current_c5 = c5_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
        
        n_iters_actual += 1
    
    return best_h, best_c5

def targeted_search(h_init, n, rng, max_trials=100000, timeout=60.0):
    """Focus on reducing the worst shifts."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    if np.sum(h) != n / 2:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        target = n // 2
        if len(ones) > target:
            to_flip = rng.choice(ones, len(ones) - target, replace=False)
            h[to_flip] = 0.0
        elif len(ones) < target:
            to_flip = rng.choice(zeros, target - len(ones), replace=False)
            h[to_flip] = 1.0
    
    best_c5 = compute_c5(h)
    no_improve = 0
    start_t = time.time()
    stall_limit = 15000
    
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Focus on top 5 worst shifts
        top_k = 5
        worst_indices = np.argsort(corr)[-top_k:]
        worst_idx = worst_indices[rng.randint(top_k)]
        worst_shift = worst_idx - (n - 1)
        
        # Find positions contributing to this shift
        s = worst_shift
        if s >= 0:
            valid_range = range(0, n - s)
        else:
            valid_range = range(-s, n)
        
        bad_positions = []
        for i in valid_range:
            j = i + s if s >= 0 else i
            if s >= 0:
                if h[i] == 1.0 and h[i + s] == 0.0:
                    bad_positions.append(i)
            else:
                if h[i + s] == 1.0 and h[i] == 0.0:
                    bad_positions.append(i)
        
        if len(bad_positions) > 0:
            i = bad_positions[rng.randint(len(bad_positions))]
            zeros = np.where(h == 0.0)[0]
            if len(zeros) == 0:
                break
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                no_improve += 1
                continue
        else:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            no_improve = 0
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
            no_improve += 1
    
    return h, best_c5

def continuous_refine(h_init, n, rng, max_trials=50000, timeout=60.0):
    """Continuous local search with small perturbations."""
    h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    target = n / 2.0
    no_improve = 0
    stall_limit = 5000
    start_t = time.time()
    
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        
        h_try = h.copy()
        n_mod = rng.randint(1, 3)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-0.1, 0.1)
        h_try = np.clip(h_try, 0.0, 1.0)
        
        # Repair sum constraint
        diff = target - np.sum(h_try)
        if abs(diff) > 1e-12:
            mask = (h_try > 1e-6) & (h_try < 1.0 - 1e-6)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
            if abs(np.sum(h_try) - target) > 1e-8:
                continue
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
            no_improve = 0
        else:
            no_improve += 1
    
    return h, best_c5

def milp_refine(h_init, n, K=30, timeout=120.0):
    """Try to solve a MILP to find a better binary solution."""
    try:
        import cvxpy as cp
    except ImportError:
        return h_init, compute_c5(h_init)
    
    start_t = time.time()
    if time.time() - start_t > timeout:
        return h_init, compute_c5(h_init)
    
    # Variables
    h_var = cp.Variable(n, boolean=True)
    C_var = cp.Variable(nonneg=True)
    
    # z variables for each shift
    z_vars = {}
    for k in range(1, K + 1):
        z_vars[k] = cp.Variable(n - k, nonneg=True)
    
    # Objective: minimize C
    objective = cp.Minimize(C_var)
    
    constraints = []
    # Sum constraint
    constraints.append(cp.sum(h_var) == n / 2)
    
    # Correlation constraints
    for k in range(1, K + 1):
        constraints.append(cp.sum(z_vars[k]) <= C_var * n / 2)
        for i in range(n - k):
            # z[i] >= h[i] - h[i+k]
            constraints.append(z_vars[k][i] >= h_var[i] - h_var[i + k])
            # z[i] <= h[i]
            constraints.append(z_vars[k][i] <= h_var[i])
            # z[i] <= 1 - h[i+k]
            constraints.append(z_vars[k][i] <= 1 - h_var[i + k])
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCIP, time_limit=timeout - (time.time() - start_t))
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, time_limit=timeout - (time.time() - start_t))
        except Exception:
            return h_init, compute_c5(h_init)
    
    if h_var.value is not None:
        h_sol = np.round(h_var.value).astype(float)
        h_sol = make_feasible(h_sol, n)
        # Verify it has exactly n/2 ones
        if np.sum(h_sol) != n / 2:
            return h_init, compute_c5(h_init)
        c5_val = compute_c5(h_sol)
        return h_sol, c5_val
    
    return h_init, compute_c5(h_init)

_N_OPT = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary_feasible(h, n):
    h = np.round(h).astype(float)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    target = n // 2
    if len(ones) > target:
        excess = len(ones) - target
        to_flip = np.random.choice(ones, excess, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = np.random.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

    return np.sum(x) - _N_OPT / 2.0

def _sum_con_jac(x):
    return np.ones(_N_OPT)

def _smooth_max_obj(x, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted[shifted > 50] = 50
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val

def _direct_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

    best_c5 = compute_c5(current)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}

        return budget_s - (time.time() - start_time)
    
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None
    
    n = len(h_parent) if h_parent is not None else 256
    
    best_h = thue_morse_init(n)
    best_c5 = compute_c5(best_h)
    
    if h_parent is not None:
        h_p = make_feasible(h_parent.copy(), n)
        c5_p = compute_c5(h_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
    
    # Phase 1: Smooth max continuous optimization
    if remaining() > 30:
        inits = []
        if h_parent is not None:
            inits.append(h_parent.copy())
        inits.append(thue_morse_init(n))
        inits.append(random_binary_n(n, rng))
        
        for h0 in inits:
            if remaining() < 20:
                break
            t_alloc = min(remaining() * 0.25, 100.0)
            h_opt, c5_val = smooth_max_optimize(h0, n, beta_start=20.0, beta_end=600.0,
                                                n_steps=8, max_iter=1000, timeout=t_alloc)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
    
    # Phase 2: Simulated annealing (main improvement)
    if remaining() > 60:
        sa_init = np.round(best_h).astype(float)
        t_budget = min(remaining() * 0.3, 300.0)
        h_sa, c5_sa = simulated_annealing(sa_init, n, rng, t_start=1.0, t_end=0.001,
                                           max_iters=3000000, timeout=t_budget)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 3: Binary local search with diverse moves
    if remaining() > 40:
        h_bs, c5_bs = binary_local_search_diverse(best_h, n, rng, max_trials=500000,
                                                   stall_limit=30000, timeout=min(120, remaining() - 20))
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()
    
    # Phase 4: Targeted search on worst shifts
    if remaining() > 40:
        h_t, c5_t = targeted_search(best_h, n, rng, max_trials=300000,
                                     timeout=min(90, remaining() - 20))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
    
    # Phase 5: Second round of simulated annealing from current best
    if remaining() > 90:
        h_sa2, c5_sa2 = simulated_annealing(best_h, n, rng, t_start=0.5, t_end=0.001,
                                             max_iters=2000000, timeout=min(120, remaining() - 60))
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
    
    # Phase 6: More local search
    if remaining() > 40:
        h_bs2, c5_bs2 = binary_local_search_diverse(best_h, n, rng, max_trials=400000,
                                                     stall_limit=25000, timeout=min(80, remaining() - 20))
        if c5_bs2 < best_c5:
            best_c5 = c5_bs2
            best_h = h_bs2.copy()
    
    # Phase 7: Continuous refinement
    if remaining() > 30:
        h_c, c5_c = continuous_refine(best_h, n, rng, max_trials=30000,
                                       timeout=min(30, remaining() - 15))
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
    
    # Phase 8: SLSQP polish
    if remaining() > 20:
        h_f, c5_f = final_polish_slsqp(best_h, n, max_iter=5000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    # Phase 9: Final targeted search
    if remaining() > 30:
        h_t2, c5_t2 = targeted_search(best_h, n, rng, max_trials=200000,
                                       timeout=min(40, remaining() - 15))
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()
    
    # Phase 10: Final binary local search
    if remaining() > 25:
        h_b3, c5_b3 = binary_local_search_diverse(best_h, n, rng, max_trials=200000,
                                                   stall_limit=15000, timeout=min(40, remaining() - 10))
        if c5_b3 < best_c5:
            best_c5 = c5_b3
            best_h = h_b3.copy()
    
    # Final: also try at double resolution if time permits and n is small enough
    n2 = n * 2
    if n <= 256 and remaining() > 120:
        h_interp = interp_h(best_h, n2)
        h_interp = make_feasible(h_interp, n2)
        c5_interp = compute_c5(h_interp)
        
        if c5_interp < best_c5 + 0.01:
            h_opt2, c5_opt2 = smooth_max_optimize(h_interp, n2, beta_start=20.0,
                                                    beta_end=400.0, n_steps=5,
                                                    max_iter=500, timeout=40.0)
            if c5_opt2 < best_c5:
                h_bs2b, c5_bs2b = binary_local_search_diverse(h_opt2, n2, rng,
                                                               max_trials=200000,
                                                               stall_limit=20000,
                                                               timeout=min(50, remaining() - 30))
                if c5_bs2b < best_c5:
                    best_c5 = c5_bs2b
                    best_h = h_bs2b.copy()
                    n = n2
    
    # Ensure final feasibility
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n)
