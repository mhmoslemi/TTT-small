    return np.max(corr) * dx

def make_feasible_binary(h, n):
    """Make h binary with exactly n/2 ones."""
    h = np.round(h).astype(float)
    target = n // 2
    ones = np.sum(h == 1.0)
    if ones > target:
        excess = int(ones - target)
        one_positions = np.where(h == 1.0)[0]
        to_flip = np.random.choice(one_positions, excess, replace=False)
        h[to_flip] = 0.0
    elif ones < target:
        deficit = int(target - ones)
        zero_positions = np.where(h == 0.0)[0]
        to_flip = np.random.choice(zero_positions, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def upsample_binary(h_src, n_dst):
    """Upsample a binary sequence to a new length, preserving ~n/2 ones."""
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    # Nearest-neighbor upsampling
    indices = np.linspace(0, n_src, n_dst, endpoint=False).astype(int)
    indices = np.minimum(indices, n_src - 1)
    h_new = h_src[indices]
    # Fix to have exactly n_dst/2 ones
    target = n_dst // 2
    ones = np.sum(h_new == 1.0)
    if ones > target:
        excess = int(ones - target)
        one_pos = np.where(h_new == 1.0)[0]
        to_flip = np.random.choice(one_pos, excess, replace=False)
        h_new[to_flip] = 0.0
    elif ones < target:
        deficit = int(target - ones)
        zero_pos = np.where(h_new == 0.0)[0]
        to_flip = np.random.choice(zero_pos, deficit, replace=False)
        h_new[to_flip] = 1.0
    return h_new

def genetic_algorithm(h_init, n, rng, pop_size=50, timeout=200.0):
    """GA for binary sequences."""
    # Initialize population
    h0 = np.round(h_init).astype(float)
    target = n // 2
    # Ensure h0 has target ones

    zeros = np.where(h0 == 0.0)[0]

    elif len(ones) < target:
        h0[rng.choice(zeros, target-len(ones), replace=False)] = 1.0
    
    population = [h0.copy()]
    # Add random individuals
    for _ in range(pop_size - 1):
        h = (rng.rand(n) > 0.5).astype(float)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) > target:

        elif len(ones) < target:
            h[rng.choice(zeros, target-len(ones), replace=False)] = 1.0
        population.append(h)
    
    # Evaluate
    fitness = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = population[best_idx].copy()
    
    start_t = time.time()
    gen = 0
    no_improve = 0
    
    while time.time() - start_t < timeout:
        gen += 1
        # Sort by fitness
        order = np.argsort(fitness)
        population = [population[i].copy() for i in order]
        fitness = [fitness[i] for i in order]
        
        # Elitism: keep top 10%
        elite_size = max(2, pop_size // 10)
        new_pop = [population[i].copy() for i in range(elite_size)]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            if time.time() - start_t > timeout - 5:
                break
            # Tournament selection
            a = rng.integers(0, min(pop_size, 5))
            b = rng.integers(0, min(pop_size, 5))
            parent_a = population[a] if fitness[a] < fitness[b] else population[b]
            c = rng.integers(0, min(pop_size, 5))
            d = rng.integers(0, min(pop_size, 5))
            parent_b = population[c] if fitness[c] < fitness[d] else population[d]
            
            # Crossover: uniform
            mask = (rng.rand(n) > 0.5).astype(float)
            child = parent_a * mask + parent_b * (1 - mask)
            
            # Mutation: flip a pair
            if rng.random() < 0.8:
                ones = np.where(child == 1.0)[0]
                zeros = np.where(child == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    child[i] = 0.0
                    child[j] = 1.0
            
            # Repair: ensure exactly target ones
            child = np.round(child).astype(float)
            ones = np.where(child == 1.0)[0]
            zeros = np.where(child == 0.0)[0]
            if len(ones) > target:
                excess = len(ones) - target
                child[rng.choice(ones, excess, replace=False)] = 0.0
            elif len(ones) < target:
                deficit = target - len(ones)
                child[rng.choice(zeros, deficit, replace=False)] = 1.0
            
            new_pop.append(child)
        
        # Evaluate new population
        population = new_pop
        fitness = [compute_c5(h) for h in population]
        
        # Update best

            best_h = population[0].copy()
            no_improve = 0

            no_improve += 1

    return h

def thue_morse_binary(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return h

def simulate_anneal(h_init, n, rng, t_start=0.3, t_end=0.0005, timeout=300.0, div_prob=0.02):
    """Simulated annealing with pairwise and multi-pair swaps, plus diversification."""
    h = fix_binary(h_init.copy(), n, rng)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    ones_list = np.where(h == 1.0)[0]
    zeros_list = np.where(h == 0.0)[0]
    
    start_t = time.time()
    no_improve = 0
    
    while time.time() - start_t < timeout:
        elapsed = time.time() - start_t
        progress = min(elapsed / timeout, 1.0)
        T = t_start * (t_end / t_start) ** progress
        
        # Diversification: if stuck, perturb the current solution
        if no_improve > 8000 and T < 0.01:
            h = best_h.copy()
            n_flips = rng.randint(3, 8)
            ones_list = np.where(h == 1.0)[0]
            zeros_list = np.where(h == 0.0)[0]
            for _ in range(n_flips):
                if len(ones_list) == 0 or len(zeros_list) == 0:
                    break
                i = ones_list[rng.randint(len(ones_list))]
                j = zeros_list[rng.randint(len(zeros_list))]
                h[i] = 0.0
                h[j] = 1.0
                idx_i = np.where(ones_list == i)[0]
                if len(idx_i) > 0:
                    ones_list = np.delete(ones_list, idx_i[0])
                idx_j = np.where(zeros_list == j)[0]
                if len(idx_j) > 0:
                    zeros_list = np.delete(zeros_list, idx_j[0])
                if i not in zeros_list:
                    zeros_list = np.append(zeros_list, i)
                if j not in ones_list:
                    ones_list = np.append(ones_list, j)
            current_c5 = compute_c5(h)
            no_improve = 0
            continue
        
        r = rng.random()
        if r < 0.65:
            # Single pairwise swap
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
                current_c5 = new_c5
                idx_i = np.where(ones_list == i)[0]
                idx_j = np.where(zeros_list == j)[0]
                if len(idx_i) > 0:
                    ones_list = np.delete(ones_list, idx_i[0])
                if len(idx_j) > 0:
                    zeros_list = np.delete(zeros_list, idx_j[0])
                zeros_list = np.append(zeros_list, i)
                ones_list = np.append(ones_list, j)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        elif r < 0.85:
            # 2-pair swap
            if len(ones_list) < 2 or len(zeros_list) < 2:
                continue
            i1 = ones_list[rng.randint(len(ones_list))]
            i2 = ones_list[rng.randint(len(ones_list))]
            j1 = zeros_list[rng.randint(len(zeros_list))]
            j2 = zeros_list[rng.randint(len(zeros_list))]
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
                current_c5 = new_c5
                for idx in [i1, i2]:
                    pos = np.where(ones_list == idx)[0]
                    if len(pos) > 0:
                        ones_list = np.delete(ones_list, pos[0])
                    zeros_list = np.append(zeros_list, idx)
                for idx in [j1, j2]:
                    pos = np.where(zeros_list == idx)[0]
                    if len(pos) > 0:
                        zeros_list = np.delete(zeros_list, pos[0])
                    ones_list = np.append(ones_list, idx)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
                no_improve += 1
        else:
            # 3-pair swap
            if len(ones_list) < 3 or len(zeros_list) < 3:
                continue
            i1 = ones_list[rng.randint(len(ones_list))]
            i2 = ones_list[rng.randint(len(ones_list))]
            i3 = ones_list[rng.randint(len(ones_list))]
            j1 = zeros_list[rng.randint(len(zeros_list))]
            j2 = zeros_list[rng.randint(len(zeros_list))]
            j3 = zeros_list[rng.randint(len(zeros_list))]
            if len(set([i1, i2, i3, j1, j2, j3])) < 6:
                continue
            h[i1] = 0.0; h[i2] = 0.0; h[i3] = 0.0
            h[j1] = 1.0; h[j2] = 1.0; h[j3] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
                current_c5 = new_c5
                for idx in [i1, i2, i3]:
                    pos = np.where(ones_list == idx)[0]
                    if len(pos) > 0:
                        ones_list = np.delete(ones_list, pos[0])
                    zeros_list = np.append(zeros_list, idx)
                for idx in [j1, j2, j3]:
                    pos = np.where(zeros_list == idx)[0]
                    if len(pos) > 0:
                        zeros_list = np.delete(zeros_list, pos[0])
                    ones_list = np.append(ones_list, idx)
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i1] = 1.0; h[i2] = 1.0; h[i3] = 1.0
                h[j1] = 0.0; h[j2] = 0.0; h[j3] = 0.0
                no_improve += 1
    
    return best_h, best_c5

def genetic_algorithm(h_init, n, rng, pop_size=40, timeout=200.0):
    """Genetic algorithm for binary sequences."""
    target = n // 2
    h0 = fix_binary(h_init.copy(), n, rng)
    population = [h0.copy()]
    
    for _ in range(pop_size - 1):
        h = (rng.rand(n) > 0.5).astype(float)
        h = fix_binary(h, n, rng)
        population.append(h)
    
    fitness = [compute_c5(h) for h in population]
    best_c5 = min(fitness)
    best_h = population[np.argmin(fitness)].copy()
    
    start_t = time.time()
    gen = 0
    no_improve = 0
    
    while time.time() - start_t < timeout:
        gen += 1
        order = np.argsort(fitness)
        population = [population[i].copy() for i in order]
        fitness = [fitness[i] for i in order]
        
        elite_size = max(3, pop_size // 8)
        new_pop = [population[i].copy() for i in range(elite_size)]
        
        while len(new_pop) < pop_size:
            if time.time() - start_t > timeout - 3:
                break
            # Tournament selection (size 3)
            cands = rng.randint(0, min(10, pop_size), size=3)
            pa = population[cands[np.argmin([fitness[c] for c in cands])]]
            cands2 = rng.randint(0, min(10, pop_size), size=3)
            pb = population[cands2[np.argmin([fitness[c] for c in cands2])]]
            
            # Uniform crossover
            mask = (rng.rand(n) > 0.5).astype(float)
            child = pa * mask + pb * (1.0 - mask)
            
            # Mutation
            if rng.random() < 0.7:
                child = np.round(child).astype(float)
                ones = np.where(child == 1.0)[0]
                zeros = np.where(child == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    n_mut = rng.randint(1, 3)
                    for _ in range(n_mut):
                        if len(ones) == 0 or len(zeros) == 0:
                            break
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                        child[i] = 0.0
                        child[j] = 1.0
                        ones = np.delete(ones, np.where(ones == i)[0][0])
                        zeros = np.delete(zeros, np.where(zeros == j)[0][0])
                        ones = np.append(ones, j)
                        zeros = np.append(zeros, i)
            else:
                child = np.round(child).astype(float)
            
            child = fix_binary(child, n, rng)
            new_pop.append(child)
        
        population = new_pop
        fitness = [compute_c5(h) for h in population]
        
        if fitness[0] < best_c5 - 1e-15:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 150:
            break
    
    return best_h, best_c5

def greedy_local_search(h_init, n, rng, timeout=120.0, max_stall=20000):
    """Greedy pairwise swap local search."""
    h = fix_binary(h_init.copy(), n, rng)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    no_improve = 0
    start_t = time.time()
    
    while time.time() - start_t < timeout:
        if no_improve > max_stall:
            break
        if not ones or not zeros:
            break
        
        i = list(ones)[rng.randint(len(ones))]
        j = list(zeros)[rng.randint(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h.copy()
            ones.discard(i)
            zeros.add(i)
            zeros.discard(j)
            ones.add(j)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    
    return best_h, best_c5

def _smooth_obj(x, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.minimum(shifted, 50.0)
    shifted = np.maximum(shifted, -500.0)
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val * (2.0 / _N_GLOBAL)

def _sum_con_fun(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def _sum_con_jac(x):
    return np.ones(_N_GLOBAL)

def continuous_polish(h_init, n, timeout=80.0):
    """Continuous SLSQP optimization with smooth max."""
    global _N_GLOBAL
    _N_GLOBAL = n
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    start_t = time.time()
    
    betas = [30, 60, 120, 250, 500, 1000]
    for beta in betas:
        if time.time() - start_t > timeout - 10:
            break
        try:
            res = minimize(_smooth_obj, best_h, args=(beta,), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': 2000, 'ftol': 1e-18})
            if res.fun < 1e10:
                h_try = make_feasible(res.x, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
        except Exception:
            pass
    
    # One more pass with direct max (non-smooth)
    if time.time() - start_t < timeout - 5:
        try:
            def _direct_obj(x):
                h = np.clip(x, 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode='full')
                return np.max(corr) * (2.0 / _N_GLOBAL)
            def _direct_obj_grad(x):
                h = np.clip(x, 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode='full')
                k = np.argmax(corr)
                # Gradient at the max point
                n_loc = len(h)
                lag = k - (n_loc - 1)
                grad = np.zeros(n_loc)
                # d/dh[i] of sum_j h[j]*(1-h[j+lag])
                for i in range(n_loc):
                    # h[i] appears as first factor when j=i
                    if 0 <= i + lag < n_loc:
                        grad[i] -= (1.0 - h[i + lag]) / n_loc * n_loc
                    # h[i] appears as (1-h[i]) when j+lag=i, i.e., j=i-lag
                    if 0 <= i - lag < n_loc:
                        grad[i] += h[i - lag] / n_loc * n_loc
                return grad * (2.0 / n_loc)
            res = minimize(_smooth_obj, best_h, args=(2000,), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': 3000, 'ftol': 1e-18})
            if res.fun < 1e10:
                h_try = make_feasible(res.x, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try
        except Exception:
            pass
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None
    
    # Phase 1: n=256 search
    n1 = 256
    best_h_overall = None
    best_c5_overall = 1.0
    
    # Initial solutions for n=256
    inits_256 = []
    if h_parent is not None:
        h_p = make_feasible(h_parent.copy(), n1) if len(h_parent) == n1 else make_feasible(upsample_h(h_parent, n1), n1)
        inits_256.append(h_p)
    
    # Thue-Morse variants
    tm = thue_morse_binary(n1)
    inits_256.append(make_feasible(tm, n1))
    for shift in [1, 3, 5]:
        h_shift = np.zeros(n1)
        for i in range(n1):
            h_shift[i] = 1.0 if bin((i + shift) % n1).count('1') % 2 == 0 else 0.0
        inits_256.append(make_feasible(h_shift, n1))
    
    # Random inits
    for _ in range(3):
        h_r = (rng.rand(n1) > 0.5).astype(float)
        h_r = fix_binary(h_r, n1, rng)
        inits_256.append(h_r)
    
    # Evaluate all inits
    for h0 in inits_256:
        c5_val = compute_c5(h0)
        if c5_val < best_c5_overall:
            best_c5_overall = c5_val
            best_h_overall = h0.copy()
    
    # SA from best and a few other inits
    sa_timeout = min(250.0, remaining() * 0.35)
    for h0 in inits_256[:4]:
        if remaining() < 60:
            break
        h_sa, c5_sa = simulate_anneal(h0, n1, rng, t_start=0.2, t_end=0.0005,
                                      timeout=sa_timeout, div_prob=0.015)
        if c5_sa < best_c5_overall:
            best_c5_overall = c5_sa
            best_h_overall = h_sa.copy()
    
    # GA
    if remaining() > 120:
        ga_timeout = min(150.0, remaining() * 0.25)
        h_ga, c5_ga = genetic_algorithm(best_h_overall, n1, rng, pop_size=40, timeout=ga_timeout)
        if c5_ga < best_c5_overall:
            best_c5_overall = c5_ga
            best_h_overall = h_ga.copy()
        # GA from a different init
        if remaining() > 80:
            h_ga2, c5_ga2 = genetic_algorithm(inits_256[-1], n1, rng, pop_size=30, timeout=min(60.0, remaining() * 0.15))
            if c5_ga2 < best_c5_overall:
                best_c5_overall = c5_ga2
                best_h_overall = h_ga2.copy()
    
    # SA again from the best found
    if remaining() > 60:
        h_sa2, c5_sa2 = simulate_anneal(best_h_overall, n1, rng, t_start=0.1, t_end=0.0001,
                                        timeout=min(120.0, remaining() * 0.2), div_prob=0.01)
        if c5_sa2 < best_c5_overall:
            best_c5_overall = c5_sa2
            best_h_overall = h_sa2.copy()
    
    # Greedy local search
    if remaining() > 30:
        h_ls, c5_ls = greedy_local_search(best_h_overall, n1, rng, timeout=min(60.0, remaining() * 0.15), max_stall=15000)
        if c5_ls < best_c5_overall:
            best_c5_overall = c5_ls
            best_h_overall = h_ls.copy()
    
    # Continuous polish
    if remaining() > 30:
        h_cp, c5_cp = continuous_polish(best_h_overall, n1, timeout=min(80.0, remaining() * 0.3))
        if c5_cp < best_c5_overall:
            best_c5_overall = c5_cp
            best_h_overall = h_cp.copy()
    
    # Another round of SA from the continuous solution (round it)
    if remaining() > 30:
        h_rounded = fix_binary(best_h_overall, n1, rng)
        h_sa3, c5_sa3 = simulate_anneal(h_rounded, n1, rng, t_start=0.15, t_end=0.001,
                                        timeout=min(60.0, remaining() * 0.2), div_prob=0.01)
        if c5_sa3 < best_c5_overall:
            best_c5_overall = c5_sa3
            best_h_overall = h_sa3.copy()
        # Continuous polish again
        if remaining() > 20:
            h_cp2, c5_cp2 = continuous_polish(best_h_overall, n1, timeout=min(40.0, remaining() * 0.3))
            if c5_cp2 < best_c5_overall:
                best_c5_overall = c5_cp2
                best_h_overall = h_cp2.copy()
    
    # Phase 2: Try n=512 (if time allows)
    n2 = 512
    if remaining() > 200 and best_c5_overall > 0.38085:
        inits_512 = []
        if h_parent is not None:
            h_p512 = make_feasible(upsample_h(h_parent, n2), n2)
            inits_512.append(h_p512)
        # UpSample best from 256
        h_up = fix_binary(upsample_h(best_h_overall, n2), n2, rng)
        inits_512.append(h_up)
        # Thue-Morse at 512
        tm512 = thue_morse_binary(n2)
        inits_512.append(make_feasible(tm512, n2))
        # Random
        for _ in range(2):
            h_r512 = (rng.rand(n2) > 0.5).astype(float)
            inits_512.append(fix_binary(h_r512, n2, rng))
        
        best_h_512 = inits_512[0].copy()
        best_c5_512 = compute_c5(best_h_512)
        for h0 in inits_512:
            c5_v = compute_c5(h0)
            if c5_v < best_c5_512:
                best_c5_512 = c5_v
                best_h_512 = h0.copy()
        
        # SA at 512
        sa_timeout_512 = min(200.0, remaining() * 0.5)
        h_sa512, c5_sa512 = simulate_anneal(best_h_512, n2, rng, t_start=0.2, t_end=0.001,
                                           timeout=sa_timeout_512, div_prob=0.01)
        if c5_sa512 < best_c5_512:
            best_c5_512 = c5_sa512
            best_h_512 = h_sa512.copy()
        
        # Greedy at 512
        if remaining() > 30:
            h_ls512, c5_ls512 = greedy_local_search(best_h_512, n2, rng, timeout=min(60.0, remaining() * 0.15), max_stall=10000)
            if c5_ls512 < best_c5_512:
                best_c5_512 = c5_ls512
                best_h_512 = h_ls512.copy()
        
        if best_c5_512 < best_c5_overall:
            best_c5_overall = best_c5_512
            best_h_overall = best_h_512.copy()
    
    # Phase 3: Final continuous polish on the overall best
    n_final = len(best_h_overall)
    if remaining() > 15:
        h_final, c5_final = continuous_polish(best_h_overall, n_final, timeout=min(60.0, remaining() * 0.6))
        if c5_final < best_c5_overall:
            best_c5_overall = c5_final
            best_h_overall = h_final.copy()
    
    # Final verification
    best_h_overall = make_feasible(best_h_overall, len(best_h_overall))
    best_c5_overall = compute_c5(best_h_overall)
    
    return (best_h_overall, best_c5_overall, len(best_h_overall))
