from scipy.optimize import minimize
import time

def compute_c5_val(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            h = np.clip(h + diff / n, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def smooth_lse(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = corr * dx
    m = np.max(val)
    s = np.clip(beta * (val - m), -40.0, 40.0)
    return m + np.log(np.sum(np.exp(s))) / beta

def subgrad_topk(h, n, topk=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = corr * dx
    top_indices = np.argsort(val)[-topk:][::-1]
    weights = np.zeros(2 * n - 1)
    for idx in top_indices:
        weights[idx] = val[idx]
    wsum = weights.sum()
    if wsum < 1e-15:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    for k_idx in top_indices:
        offset = k_idx - (n - 1)
        w = weights[k_idx] * dx
        if offset >= 0:

            j2 = np.arange(abs_off, abs_off + npairs)
            grad[j2] += w * (1.0 - h[j2 - abs_off])

    n = len(h)
    val = smooth_lse(h, beta)
    penalty = mu * (h.sum() - n / 2.0) ** 2
    return val + penalty

    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_with_penalty, h0, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return project_feasible(res.x, n)

    h = h.copy()

                h_sg, c5_sg = subgrad_descent(h, n, iters=300, topk=topk, base_step=base_step)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()
                    h = h_sg.copy()
                    print("  Subgrad topk={}, step={}: C5={:.7f}".format(topk, base_step, best_c5))
    
    # L-BFGS-B intensive at best n
    if remaining_time(start_t, budget_s) > 60:
        print("\nPhase 2b: L-BFGS-B intensive at n={}".format(n))
        for beta_val in [100, 300, 1000, 3000]:
            if remaining_time(start_t, budget_s) < 30:
                break
            for mu_val in [50.0, 200.0, 1000.0]:
                if remaining_time(start_t, budget_s) < 15:
                    break
                h_opt = optimize_lbfgsb(h, n, beta_val, mu_val, maxiter=1000)
                c5_new = compute_c5_val(h_opt)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_opt.copy()
                    h = h_opt.copy()
                    print("  L-BFGS-B beta={}, mu={}: C5={:.7f}".format(beta_val, mu_val, best_c5))
    
    # Perturbation + re-optimization
    if remaining_time(start_t, budget_s) > 60:
        print("\nPhase 2c: Perturbation restarts at n={}".format(n))
        for attempt in range(6):
            if remaining_time(start_t, budget_s) < 30:
                break
            noise_scale = 0.15 * (1.0 - attempt / 8.0) + 0.02
            h_pert = h + rng.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, n)
            beta_val = 300
            mu_val = 500.0
            h_opt = optimize_lbfgsb(h_pert, n, beta_val, mu_val, maxiter=800)
            c5_new = compute_c5_val(h_opt)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_opt.copy()
                h = h_opt.copy()
                print("  Perturb {}: C5={:.7f}".format(attempt, best_c5)
    
    # Phase 3: Binary local search
    if remaining_time(start_t, budget_s) > 40:
        print("\nPhase 3: Binary local search at n={}".format(n))
        h_bin, c5_bin = binary_local_search(h, n, rng, max_sweeps=50)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Binary: C5={:.7f}".format(best_c5))
        # Also try Thue-Morse and Golden at this n
        for name, h_init in [("TM", make_thue_morse(n)), ("GR", make_golden(n))]:
            if remaining_time(start_t, budget_s) < 20:
                break
            h_init = project_feasible(h_init, n)
            h_b, c5_b = binary_local_search(h_init, n, rng, max_sweeps=30)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                print("  Binary {}: C5={:.7f}".format(name, best_c5))
    
    # Phase 4: Fine subgradient polishing
    if remaining_time(start_t, budget_s) > 30:
        print("\nPhase 4: Fine subgradient polish at n={}".format(n))
        h = best_h.copy()
        for it in range(2000):
            if remaining_time(start_t, budget_s) < 5:
                break
            dx = 2.0 / n
            corr = np.correlate(h, 1.0 - h, mode='full')
            val = corr * dx
            k_max = int(np.argmax(val))
            offset = k_max - (n - 1)
            grad = np.zeros(n)
            if offset >= 0:
                npairs = n - offset
                j1 = np.arange(npairs)
                grad[j1] += (1.0 - h[j1 + offset])
                j2 = np.arange(offset, offset + npairs)
                grad[j2] -= h[j2 - offset]
            else:
                abs_off = -offset
                npairs = n - abs_off
                j1 = np.arange(npairs)
                grad[j1] -= h[j1 + abs_off]
                j2 = np.arange(abs_off, abs_off + npairs)
                grad[j2] += (1.0 - h[j2 - abs_off])
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            improved = False
            for step in [0.01, 0.03, 0.005, 0.02, 0.05]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5_val(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            if not improved:
                # Try L-BFGS-B from current point
                if remaining_time(start_t, budget_s) > 20:
                    h_opt = optimize_lbfgsb(h, n, 500, 200.0, maxiter=300)
                    c5_new = compute_c5_val(h_opt)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_opt.copy()
                        h = h_opt.copy()
                    else:
                        break
    
    # Phase 5: Try slightly different n values around best
    if remaining_time(start_t, budget_s) > 60:
        print("\nPhase 5: Nearby n exploration")
        for dn in [-2, -1, 1, 2, 3, 4, 5, 6, 8]:
            if remaining_time(start_t, budget_s) < 30:
                break
            n_try = best_n + dn
            if n_try < 64 or n_try > 400:
                continue
            h_try = interp_to(best_h, best_n, n_try)
            h_try = project_feasible(h_try, n_try)
            for beta_val in [200, 800]:
                if remaining_time(start_t, budget_s) < 15:
                    break
                h_opt = optimize_lbfgsb(h_try, n_try, beta_val, 500.0, maxiter=500)
                c5_new = compute_c5_val(h_opt)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_opt.copy()
                    best_n = n_try
                    print("  n={}: C5={:.7f}".format(n_try, best_c5)
    
    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5_val(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    elapsed = time.time() - start_t
    print("\nFINAL: n={}, C5={:.7f}, time={:.1f}s".format(best_n, final_c5, elapsed))
    return (best_h, final_c5, best_n)
