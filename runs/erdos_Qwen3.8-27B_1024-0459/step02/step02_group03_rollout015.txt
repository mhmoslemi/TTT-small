def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def smooth_max_value(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return (m + np.log(total) / beta)

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def obj_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_value(corr, beta) * dx

def grad_func(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[L:] += w * (1.0 - x[:n - L])
            grad[:n - L] -= w * x[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - x[Lp:])
            grad[Lp:] -= w * x[:n - Lp]
    return dx * grad

            print("  Init {}: C5={:.6f}".format(idx, best_c5))
        # Simulated annealing from best
        if remaining() > 60:
            h_sa, c5_sa = simulated_annealing_binary(h_bs, rng, n_iter=5000)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_work
                print("  SA init {}: C5={:.6f}".format(idx, best_c5))

    # Phase 2: Continue binary search with more iterations from best
    print("\n=== Phase 2: Deep binary search ===")
    if remaining() > 120:
        h_bin = best_h.copy()
        if not np.all((h_bin == 0) | (h_bin == 1)):
            h_bin = (h_bin > 0.5).astype(float)
            n1 = int(np.sum(h_bin))
            target = best_n // 2
            if n1 > target:
                oi = np.where(h_bin == 1)[0]
                rm = rng.choice(oi, size=n1-target, replace=False)
                h_bin[rm] = 0
            elif n1 < target:
                zi = np.where(h_bin == 0)[0]
                ad = rng.choice(zi, size=target-n1, replace=False)
                h_bin[ad] = 1
        h_bs2, c5_bs2 = binary_local_search(h_bin, rng, max_iters=15000)
        if c5_bs2 < best_c5:
            best_c5 = c5_bs2
            best_h = h_bs2.copy()
            print("  Deep binary: C5={:.6f}".format(best_c5))
        if remaining() > 100:
            h_sa2, c5_sa2 = simulated_annealing_binary(h_bs2, rng, n_iter=8000)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                print("  Deep SA: C5={:.6f}".format(best_c5))

    # Phase 3: Try different resolutions
    print("\n=== Phase 3: Multi-resolution ===")
    n_trials = [n_work, n_work + 16, n_work - 16, 256, 320]
    n_trials = [nn for nn in set(n_trials) if 80 <= nn <= 400]
    for n_trial in n_trials:
        if n_trial == n_work or remaining() < 80:
            continue
        h_interp = interp_h(best_h, n_trial)
        c5_int = compute_c5(h_interp)
        h_bin_t = (h_interp > 0.5).astype(float)
        n1t = int(np.sum(h_bin_t))
        if n1t > n_trial // 2:
            oi = np.where(h_bin_t == 1)[0]
            rm = rng.choice(oi, size=n1t - n_trial//2, replace=False)
            h_bin_t[rm] = 0
        elif n1t < n_trial // 2:
            zi = np.where(h_bin_t == 0)[0]
            ad = rng.choice(zi, size=n_trial//2 - n1t, replace=False)
            h_bin_t[ad] = 1
        h_bs_t, c5_t = binary_local_search(h_bin_t, rng, max_iters=5000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_bs_t.copy()
            best_n = n_trial
            print("  n={}: C5={:.6f}".format(n_trial, best_c5))

    # Phase 4: Continuous refinement from best binary
    print("\n=== Phase 4: Continuous refinement ===")
    h_cont = best_h.copy().astype(float)
    h_cont = project_feasible(h_cont, best_n)
    
    # FISTA
    if remaining() > 60:
        h_f, c5_f = fista_refine(h_cont, best_n, max_iter=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
            print("  FISTA: C5={:.6f}".format(best_c5))

    # SLSQP with increasing beta
    cur = best_h.copy()
    for beta_val in [50, 150, 400, 1000, 2000]:
        if remaining() < 30:
            break
        try:
            h_new, c5_new = slsqp_refine(cur, best_n, beta_val, max_iter=400)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  SLSQP beta={}: C5={:.6f}".format(beta_val, best_c5))
            cur = h_new
        except Exception:
            break

    # Phase 5: Focused perturbation search
    print("\n=== Phase 5: Perturbation search ===")
    if remaining() > 60:
        for trial in range(8):
            if remaining() < 20:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(2, max(3, best_n // 10))
            pert_idx = rng.choice(best_n, size=n_pert, replace=False)
            h_pert[pert_idx] = np.clip(h_pert[pert_idx] + rng.uniform(-0.1, 0.1, n_pert), 0, 1)
            h_pert = project_feasible(h_pert, best_n)
            try:
                h_p, c5_p = slsqp_refine(h_pert, best_n, 300, max_iter=200)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()
                    print("  Perturb {}: C5={:.6f}".format(trial, best_c5))
            except Exception:
                pass

    # Phase 6: Final polish
    if remaining() > 30:
        try:
            h_pol, c5_pol = slsqp_refine(best_h, best_n, 3000, max_iter=300)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                print("  Final polish: C5={:.6f}".format(best_c5))
        except Exception:
            pass

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
