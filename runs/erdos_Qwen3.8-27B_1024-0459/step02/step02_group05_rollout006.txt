def binary_eval(bits,n,masks,limit=None):
    m=0
    for d in range(1,n):
       sh=bits >> d
       diff=(bits ^ sh) & masks[d-1]
       c10=(bits & diff).bit_count()
       if c10 >= limit:
          return c10
       tot=diff.bit_count()
       c01=tot-c10
       if c01 >= limit:
          return c01
       if c10 > m: m=c10
       if c01 > m: m=c01

```python
import numpy as np
import math
import time


def time_left(start, budget):
    return float(budget) - (time.time() - start)


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    target = float(target)
    n = len(v)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= n:
        return np.ones(n, dtype=float)

    vc = np.clip(v, 0.0, 1.0)
    s = float(vc.sum())
    if abs(s - target) < 1e-13:
        return vc

    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    t = 0.5 * (lo + hi)
    p = np.clip(v - t, 0.0, 1.0)

    diff = target - float(p.sum())
    if diff > 0.0:
        mask = p < 1.0
    elif diff < 0.0:
        mask = p > 0.0
    else:
        mask = None

    if mask is not None:
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            p[mask] = np.clip(p[mask] + diff / cnt, 0.0, 1.0)

    for _ in range(4):
        residual = target - float(p.sum())
        if abs(residual) < 1e-15:
            break
        if residual > 0.0:
            idx = int(np.argmax(1.0 - p))
            room = 1.0 - p[idx]
            add = min(residual, max(0.0, room))
            p[idx] += add
        else:
            idx = int(np.argmax(p))
            take = min(-residual, max(0.0, p[idx]))
            p[idx] -= take

    p = np.clip(p, 0.0, 1.0)
    return p


def make_feasible(h, n):
    return project_box_sum(np.asarray(h, dtype=float), n / 2.0)


def compute_c5_numpy(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def corr_fft(h):
    n = len(h)
    m = next_pow2(2 * n - 1)
    F = np.fft.rfft(h, m)
    ac = np.fft.irfft(F * np.conj(F), n=m)[:n]

    P = np.concatenate(([0.0], np.cumsum(h)))
    H = P[n]

    c = np.empty(2 * n - 1, dtype=float)
    c[n - 1::-1] = P[n:0:-1] - ac
    if n > 1:
        c[n:2 * n - 1] = H - P[1:n] - ac[1:n]
    return c


def softmax_weights_from_c(c, beta, dx):
    cmax = float(np.max(c))
    z = beta * dx * (c - cmax)
    mz = float(np.max(z))
    z = z - mz
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        p = np.zeros_like(c, dtype=float)
        p[int(np.argmax(c))] = 1.0
        return p
    return e / s


def smooth_value_and_cmax(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = corr_fft(h)
    cmax_c = float(np.max(c))
    cmax = cmax_c * dx

    z = beta * dx * (c - cmax_c)
    mz = float(np.max(z))
    z = z - mz
    s = float(np.sum(np.exp(z)))
    if s <= 0.0:
        val = cmax
    else:
        val = cmax + (mz + math.log(s)) / beta
    return val, cmax


def bilinear_grad_from_weights(h, w):
    n = len(h)
    R = np.zeros(n, dtype=float)

    if n > 1:
        nz = np.nonzero(w[1:n] > 1e-14)[0] + 1
        if len(nz) < max(4, n // 8):
            for d in nz:
                wd = float(w[d])
                R[d:] += wd * h[:-d]
                R[:-d] += wd * h[d:]
        else:
            K = np.zeros(2 * n - 1, dtype=float)
            seg = w[1:n]
            K[n:] = seg
            K[n - 2::-1] = seg
            m = next_pow2(3 * n - 2)
            Fh = np.fft.rfft(h, m)
            Fk = np.fft.rfft(K, m)
            conv = np.fft.irfft(Fh * Fk, n=m)
            R = conv[n - 1:n - 1 + n]

    return R + 2.0 * w[0] * h


def gradient_from_weights(h, p):
    n = len(h)
    dx = 2.0 / n

    w = np.zeros(n, dtype=float)
    if n == 1:
        w[0] = p[0]
    else:
        w[0] = p[n - 1]
        w[1:n] = p[n - 2::-1] + p[n:2 * n - 1]

    diff = np.zeros(n + 1, dtype=float)
    if n > 1:
        diff[1:n] += p[n:2 * n - 1]
        diff[0] += float(np.sum(p[:n - 1]))
        diff[1:n] -= p[:n - 1]
        diff[n] -= float(np.sum(p[n:2 * n - 1]))

    diff[0] += p[n - 1]
    diff[n] -= p[n - 1]
    linear = np.cumsum(diff, axis=0)[:-1]

    bilinear = bilinear_grad_from_weights(h, w)
    return (linear - bilinear) * dx


def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = corr_fft(h)
    cmax_c = float(np.max(c))
    cmax = cmax_c * dx

    z = beta * dx * (c - cmax_c)
    mz = float(np.max(z))
    z = z - mz
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        p = np.zeros_like(c, dtype=float)
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / s

    F = cmax + (mz + math.log(max(s, 1e-300))) / beta
    grad = gradient_from_weights(h, p)
    return F, grad, cmax


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    if len(h_src) == n_dst:
        return h_src.copy()
    if len(h_src) < 2:
        return np.ones(n_dst, dtype=float) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def quadratic_values(n, a, b, off):
    i = np.arange(n, dtype=np.int64)
    vals = ((a * i * (i + 1) // 2 + b * i + off) % n).astype(float)
    vals = vals / float(max(1, n))
    return project_box_sum(vals, n / 2.0)


def wiedemann_values(n):
    i = np.arange(n, dtype=np.int64)
    vals = ((i * (i + 1) // 2) % n).astype(float)
    vals = vals / float(max(1, n))
    return project_box_sum(vals, n / 2.0)


def pg_smooth_min(h, n, betas, deadline, rng, initial_eta=1.0):
    target = n / 2.0
    h_cur = project_box_sum(h, target)
    best_h = h_cur.copy()
    best_c = compute_c5_numpy(h_cur)

    max_eta = 8.0
    min_eta = 1e-8

    for beta in betas:
        if time.time() >= deadline - 2.0:
            break

        h_cur = project_box_sum(h_cur, target)
        F, grad, ccur = smooth_grad(h_cur, float(beta))
        if ccur < best_c:
            best_c = ccur
            best_h = h_cur.copy()

        eta = initial_eta
        stall = 0
        restarts = 0
        it = 0
        while time.time() < deadline - 2.0:
            it += 1
            if it > 30000:
                break

            F, grad, ccur = smooth_grad(h_cur, float(beta))
            if ccur < best_c:
                best_c = ccur
                best_h = h_cur.copy()

            g2 = float(np.dot(grad, grad))
            if g2 < 1e-24:
                break

            accepted = False
            step = eta
            for _ in range(10):
                y = project_box_sum(h_cur - step * grad, target)
                Fy, cy = smooth_value_and_cmax(y, float(beta))
                if cy < ccur - 1e-15 or Fy < F - 1e-5 * step * g2:
                    h_cur = y
                    F = Fy
                    ccur = cy
                    if ccur < best_c:
                        best_c = ccur
                        best_h = y.copy()
                    eta = min(eta * 1.3, max_eta)
                    accepted = True
                    break
                step *= 0.5

            if accepted:
                stall = 0
            else:
                stall += 1
                if stall >= 30:
                    stall = 0
                    restarts += 1
                    if restarts > 8:
                        break
                    pert = 0.015 + 0.008 * restarts
                    h_cur = project_box_sum(best_h + rng.randn(n) * pert, target)
                    F, grad, ccur = smooth_grad(h_cur, float(beta))
                    if ccur < best_c:
                        best_c = ccur
                        best_h = h_cur.copy()
                    eta = initial_eta
                else:
                    eta *= 0.6
                    if eta < min_eta:
                        break

    best_h = project_box_sum(best_h, target)
    bc = compute_c5_numpy(best_h)
    if bc < best_c:
        best_c = bc
    return best_h, best_c


def bits_from_values(vals, n):
    vals = np.asarray(vals, dtype=float)
    target = n // 2
    if target <= 0:
        return 0
    if target >= n:
        return (1 << n) - 1
    order = np.argsort(vals, kind="mergesort")[::-1][:target]
    bits = 0
    for idx in order:
        bits |= 1 << int(idx)
    return bits


def quadratic_bits(n, a, b, off, typ):
    i = np.arange(n, dtype=np.int64)
    if typ == 0:
        vals = ((a * i * (i + 1) // 2 + b * i + off) % n).astype(float)
    else:
        vals = ((a * i * i + b * i + off) % n).astype(float)
    return bits_from_values(vals, n)


def bits_to_h(bits, n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        if (bits >> i) & 1:
            h[i] = 1.0
    return h


def build_masks(n):
    return [(1 << (n - d)) - 1 for d in range(1, n)]


def binary_eval(bits, n, masks, limit):
    best = 0
    for d in range(1, n):
        sh = bits >> d
        low = masks[d - 1]
        diff = (bits ^ sh) & low
        c10 = (bits & diff).bit_count()
        if c10 >= limit:
            return c10
        tot = diff.bit_count()
        c01 = tot - c10
        if c01 >= limit:
            return c01
        if c10 > best:
            best = c10
        if c01 > best:
            best = c01
    return best


def random_swap_bits(bits, n, rng):
    while True:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        bi = (bits >> i) & 1
        bj = (bits >> j) & 1
        if bi != bj:
            return bits ^ (1 << i) ^ (1 << j)


def ensure_binary_count(bits, n):
    need = n // 2
    if bits.bit_count() == need:
        return bits
    vals = np.linspace(0.0, 1.0, n)
    return bits_from_values(vals, n)


def binary_greedy(bits, n, best_c, deadline, rng):
    bits = ensure_binary_count(bits, n)
    masks = build_masks(n)
    cur_max = binary_eval(bits, n, masks, n + 1)
    best_max = cur_max
    best_bits = bits
    cur_bits = bits

    if (2.0 * best_max) / n < best_c:
        best_c = (2.0 * best_max) / n

    jump_allow = max(2, n // 50)
    stall = 0

    while time.time() < deadline - 1.0:
        if cur_max <= 0:
            break

        nb = random_swap_bits(cur_bits, n, rng)
        m = binary_eval(nb, n, masks, cur_max)

        if m < cur_max:
            cur_bits = nb
            cur_max = m
            if m < best_max:
                best_max = m
                best_bits = nb
                cc = (2.0 * m) / n
                if cc < best_c:
                    best_c = cc
            stall = 0
        else:
            stall += 1
            if stall >= 2200:
                nb = best_bits
                for _ in range(rng.randint(2, 8)):
                    nb = random_swap_bits(nb, n, rng)
                m = binary_eval(nb, n, masks, n + 1)
                if m < best_max:
                    cur_bits = nb
                    cur_max = m
                    best_max = m
                    best_bits = nb
                    cc = (2.0 * m) / n
                    if cc < best_c:
                        best_c = cc
                elif m < cur_max + jump_allow and rng.random() < 0.25:
                    cur_bits = nb
                    cur_max = m
                stall = 0

    return best_bits, best_max, best_c


def generate_params(n, rng, max_params):
    params = []
    params.append((1, 0, 0, 0))
    if n > 2:
        params.append((1, 0, n // 4, 0))
        params.append((1, 0, (n // 4) + (n // 2), 0))

    for a in range(1, 12):
        for b in range(0, 12):
            if a == 1 and b == 0:
                continue
            params.append((a, b, 0, 0))

    while len(params) < max_params:
        a = rng.randint(1, 25)
        b = rng.randint(0, 49)
        off = rng.randint(n)
        typ = rng.randint(0, 2)
        params.append((a, b, off, typ))

    return params[:max_params]


def family_search(parent_h, global_best_c, deadline, rng):
    best_c = float(global_best_c)
    best_h = None
    best_n = 0

    nlist = [160, 128, 192, 224, 256, 288, 320, 352, 384,
             416, 448, 480, 512, 576, 640, 704]

    for idx, n in enumerate(nlist):
        if time.time() >= deadline - 2.0:
            break
        if n < 64 or n % 2 != 0:
            continue

        rem = max(1, len(nlist) - idx)
        share = max(2.0, (deadline - time.time()) / float(rem))
        sub_deadline = min(deadline - 2.0, time.time() + share)

        masks = build_masks(n)
        max_params = int(min(800, max(100, share * 50)))

        if parent_h is not None and time.time() < sub_deadline - 1.0:
            h_trial = interpolate_to_n(parent_h, n)
            bits = bits_from_values(h_trial, n)
            lim = int(math.floor(best_c * n / 2.0 + 1e-9)) + 1
            if lim < 1:
                lim = 1
            m = binary_eval(bits, n, masks, lim)
            if m < lim:
                cc = (2.0 * m) / n
                if cc < best_c:
                    best_c = cc
                    best_h = bits_to_h(bits, n)
                    best_n = n
                    if sub_deadline - time.time() > 25.0:
                        ndl = min(sub_deadline - 1.0, time.time() + 30.0)
                        b2, m2, bc2 = binary_greedy(bits, n, best_c, ndl, rng)
                        if bc2 < best_c:
                            best_c = bc2
                            best_h = bits_to_h(b2, n)
                            best_n = n
            elif m <= lim + 5 and sub_deadline - time.time() > 30.0:
                exact = binary_eval(bits, n, masks, n + 1)
                if (2.0 * exact) / n < best_c + 0.03:
                    ndl = min(sub_deadline - 1.0, time.time() + 25.0)
                    b2, m2, bc2 = binary_greedy(bits, n, best_c, ndl, rng)
                    if bc2 < best_c:
                        best_c = bc2
                        best_h = bits_to_h(b2, n)
                        best_n = n

        params = generate_params(n, rng, max_params)
        for a, b, off, typ in params:
            if time.time() > sub_deadline - 1.0:
                break
            bits = quadratic_bits(n, a, b, off, typ)
            lim = int(math.floor(best_c * n / 2.0 + 1e-9)) + 1
            if lim < 1:
                lim = 1
            m = binary_eval(bits, n, masks, lim)
            if m < lim:
                cc = (2.0 * m) / n
                if cc < best_c:
                    best_c = cc
                    best_h = bits_to_h(bits, n)
                    best_n = n
                    if sub_deadline - time.time() > 30.0:
                        ndl = min(sub_deadline - 1.0, time.time() + 35.0)
                        b2, m2, bc2 = binary_greedy(bits, n, best_c, ndl, rng)
                        if bc2 < best_c:
                            best_c = bc2
                            best_h = bits_to_h(b2, n)
                            best_n = n
            elif m <= lim + 4 and a == 1 and b == 0 and sub_deadline - time.time() > 35.0:
                exact = binary_eval(bits, n, masks, n + 1)
                if (2.0 * exact) / n < best_c + 0.04:
                    ndl = min(sub_deadline - 1.0, time.time() + 25.0)
                    b2, m2, bc2 = binary_greedy(bits, n, best_c, ndl, rng)
                    if bc2 < best_c:
                        best_c = bc2
                        best_h = bits_to_h(b2, n)
                        best_n = n

    return best_h, best_c


def continuous_phase(best_h, best_c, best_n, parent_h, deadline, rng):
    ordered = []
    if parent_h is not None:
        pn = len(parent_h)
        if pn % 2 == 0 and pn >= 64:
            ordered.append(pn)

    if best_n % 2 == 0 and best_n >= 64:
        ordered.append(best_n)

    for n in [192, 256, 320, 384, 448, 512]:
        ordered.append(n)

    seen = []
    for n in ordered:
        if n >= 64 and n % 2 == 0 and n not in seen:
            seen.append(n)

    betas = [20.0, 100.0, 500.0, 2000.0, 8000.0]

    for idx, n in enumerate(seen):
        if time.time() >= deadline - 15.0:
            break

        rem = max(1, len(seen) - idx)
        base = max(30.0, (deadline - time.time()) / float(rem))
        max_time = min(240.0, base + 30.0)

        if idx == 0 and parent_h is not None and n == len(parent_h):
            max_time = min(360.0, max(max_time, 200.0))
        if n > 500:
            max_time = min(max_time, 180.0)
        if max_time < 25.0:
            break

        if best_n == n:
            h0 = project_box_sum(best_h.copy(), n / 2.0)
        else:
            h0 = project_box_sum(interpolate_to_n(best_h, n), n / 2.0)

        alt_share = 0.0
        if max_time > 95.0:
            alt_share = min(60.0, max_time * 0.35)

        main_deadline = time.time() + max(5.0, max_time - alt_share)
        h_opt, c_opt = pg_smooth_min(h0, n, betas, main_deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = n

        if alt_share > 20.0 and time.time() < deadline - 20.0:
            a = int(rng.randint(1, 10))
            b = int(rng.randint(0, 20))
            off = int(rng.randint(n))
            h_alt = quadratic_values(n, a, b, off)
            alt_deadline = time.time() + alt_share
            h_alt_opt, c_alt = pg_smooth_min(h_alt, n, betas, alt_deadline, rng)
            if c_alt < best_c:
                best_c = c_alt
                best_h = h_alt_opt.copy()
                best_n = n

    best_h = project_box_sum(best_h, best_n / 2.0)
    bc = compute_c5_numpy(best_h)
    if bc < best_c:
        best_c = bc
    return best_h, best_c, best_n


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    budget = float(budget_s)
    rng = np.random.RandomState(seed)
    deadline = start + max(1.0, budget - 15.0)

    parent_h = None
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        if parent_h is None or parent_h.size == 0:
            parent_h = None
    except Exception:
        parent_h = None

    if parent_h is not None:
        best_n = len(parent_h)
        best_h = make_feasible(parent_h, best_n)
    else:
        best_n = 160
        best_h = wiedemann_values(best_n)

    best_c = compute_c5_numpy(best_h)

    if time.time() < deadline - 70.0:
        fam_deadline = min(deadline, start + min(220.0, budget * 0.18))
        fh, fc = family_search(parent_h, best_c, fam_deadline, rng)
        if fh is not None and fc < best_c:
            best_c = fc
            best_h = fh.copy()
            best_n = len(fh)

    if time.time() < deadline - 25.0:
        best_h, best_c, best_n = continuous_phase(
            best_h, best_c, best_n, parent_h, deadline, rng
        )

    best_h = project_box_sum(best_h, best_n / 2.0)

    for _ in range(6):
        residual = best_n / 2.0 - float(np.sum(best_h))
        if abs(residual) < 1e-15:
            break
        if residual > 0.0:
            idx = int(np.argmax(1.0 - best_h))
            add = min(residual, max(0.0, 1.0 - best_h[idx]))
            best_h[idx] += add
        else:
            idx = int(np.argmax(best_h))
            sub = min(-residual, max(0.0, best_h[idx]))
            best_h[idx] -= sub

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = compute_c5_numpy(best_h)

    return (best_h, best_c, best_n)
```
