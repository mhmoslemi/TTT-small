def project_feasible(x, n):
    """Project x onto {h: 0<=h<=1, sum(h)=n/2} using bisection on shift"""
    target = n / 2.0
    # We need to find t such that sum(clip(x+t, 0, 1)) = target
    lo, hi = -max(np.abs(x).max(), 1.0) - 1.0, max(np.abs(x).max(), 1.0) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x + mid, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + mid, 0.0, 1.0)

def lse_objective_and_grad(h, beta):
    """Log-sum-exp of correlation values, times dx, and its gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(corr - cmax, -700, 0)
    exp_vals = np.exp(beta * shifted)
    logsum = cmax + (1.0/beta) * np.log(exp_vals.sum() + 1e-300)
    
    # Softmax weights
    w = exp_vals / (exp_vals.sum() + 1e-300)
    
    # Gradient: dx * sum_k w_k * grad_c[k]
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-10:
            continue
        k = idx - (n - 1)  # shift value
        # c[k] = sum_i h[i+k]*(1-h[i]) for valid i
        # d c[k] / d h[j] = I(0<=j-k<n, 0<=j<n)*(1-h[j-k]) - I(0<=j<n, 0<=j+k<n)*h[j+k]
        if k == 0:
            grad += w[idx] * (1.0 - 2.0 * h)
        else:
            # First term: h[j] appears as h[i+k] when i=j-k
            # Need: 0 <= j-k < n and 0 <= j < n, and 0 <= (j-k) < n (i.e., i valid)
            # Also need i+k = j valid: 0 <= j < n (already) and 0 <= i = j-k < n
            j_vals = np.arange(n)
            valid1 = (j_vals - k >= 0) & (j_vals - k < n) & (j_vals >= 0) & (j_vals < n)
            # But also need i = j-k to be in valid range for the sum
            # The sum is over i from max(0, -k) to min(n-1, n-1-k)
            if k > 0:
                # i ranges from 0 to n-1-k
                # j = i+k ranges from k to n-1
                # First term: for j in [k, n-1], contribution (1-h[j-k])
                # Second term: for j in [0, n-1-k], contribution -h[j+k]
                idx1 = np.arange(k, n)
                grad[idx1] += w[idx] * (1.0 - h[idx1 - k])
                idx2 = np.arange(0, n - k)
                grad[idx2] -= w[idx] * h[idx2 + k]
            else:
                # k < 0, let k_neg = -k > 0
                # i ranges from -k to n-1 (i.e., k_neg to n-1)
                # j = i+k = i-k_neg ranges from 0 to n-1-k_neg
                # First term: for j in [0, n-1-k_neg], contribution (1-h[j-k]) = (1-h[j+k_neg])
                # Second term: for j in [k_neg, n-1], contribution -h[j+k] = -h[j-k_neg]... 
                # Wait, let me redo. k < 0. i+k must be in [0,n-1], so i >= k_neg = -k
                # i ranges from k_neg to n-1
                # j = i+k: j ranges from k_neg+k=0 to n-1+k
                # First term: h[j] appears as h[i+k] where i = j-k. Need 0<=j<n and k_neg<=j-k<n (i.e., k<=j<n)
                #   Valid j: max(0,k) <= j < n. Since k<0, this is 0<=j<n. But also need i=j-k in [k_neg, n-1]: j-k in [k_neg, n-1] → j in [k_neg+k, n-1+k] = [0, n-1+k]
                #   So valid j for first term: 0 <= j <= n-1+k (i.e., 0 to n-1-k_neg)
                k_neg = -k
                idx1 = np.arange(0, n - k_neg)
                grad[idx1] += w[idx] * (1.0 - h[idx1 - k])  # h[j-k] = h[j+k_neg]
                # Second term: h[j] appears as h[i] where i=j. Need 0<=j<n and 0<=j+k<n → j >= k_neg
                # Valid j for second term: k_neg <= j < n
                idx2 = np.arange(k_neg, n)
                grad[idx2] -= w[idx] * h[idx2 + k]  # h[j+k] = h[j-k_neg]
    
    return logsum * dx, dx * grad

def lbfgs_lse_solve(h_init, n, beta_start=20.0, beta_end=2000.0, n_betas=5, max_time=200.0):

    best_h = h_init.copy()

    h = h_init.copy()

            h_feas = project_feasible(x, n)
            val, _ = lse_objective_and_grad(h_feas, beta)

            val, _ = lse_objective_and_grad(h_feas, beta)

            gp = gp * mask
            return gp

        try:

                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

    for _ in range(1000):

        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def project_feasible(x, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    mid = 0.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(x + mid, 0.0, 1.0)
        s = np.sum(h)
        if abs(s - target) < 1e-13:
            break
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + mid, 0.0, 1.0)

def _lse_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.minimum(corr - cmax, 0.0)
    shifted = np.maximum(shifted, -700.0 / beta)
    exp_vals = np.exp(beta * shifted)
    esum = exp_vals.sum()
    if esum < 1e-300:
        esum = 1e-300
    lse = cmax + (1.0 / beta) * np.log(esum)
    w = exp_vals / esum
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-8:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif k > 0:
            idx1 = np.arange(k, n)
            grad[idx1] += wi * (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            grad[idx2] -= wi * h[idx2 + k]
        else:
            kneg = -k
            idx1 = np.arange(0, n - kneg)
            grad[idx1] += wi * (1.0 - h[idx1 + kneg])
            idx2 = np.arange(kneg, n)
            grad[idx2] -= wi * h[idx2 - kneg]
    return lse * dx, dx * grad

def _obj_func(x):
    h = np.clip(x, 0.0, 1.0)
    v, _ = _lse_val_grad(h, _global_beta)
    return v

def _grad_func(x):
    h = np.clip(x, 0.0, 1.0)
    _, g = _lse_val_grad(h, _global_beta)
    return g

def _sum_eq_fun(x):
    return np.sum(x) - _sum_target

def _sum_eq_jac(x):
    return np.ones(len(x))

def slsqp_anneal(h_init, n, betas, time_limit):
    global _global_beta, _global_n, _sum_target
    _global_n = n
    _global_n if False else None
    _sum_target = n / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    h = h_init.copy()
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        _global_beta = beta
        maxiter = 3000 if n <= 200 else 2000
        try:
            res = minimize(_obj_func, h, jac=_grad_func, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _sum_eq_fun, 'jac': _sum_eq_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
            h_new = make_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5

def lbfgs_anneal(h_init, n, betas, time_limit):
    global _global_beta, _sum_target
    _sum_target = n / 2.0
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    h = h_init.copy()
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        if time.time() - t0 > time_limit:
            break
        _global_beta = beta
        lam = 50.0
        def obj_pen(x):
            h_c = np.clip(x, 0.0, 1.0)
            s = np.sum(h_c)
            v, _ = _lse_val_grad(h_c, beta)
            return v + lam * (s - n / 2.0) ** 2
        def grad_pen(x):
            h_c = np.clip(x, 0.0, 1.0)
            s = np.sum(h_c)
            _, g = _lse_val_grad(h_c, beta)
            gp = g + 2.0 * lam * (s - n / 2.0)
            mask = (h_c > 1e-10) & (h_c < 1.0 - 1e-10)
            gp = np.where(mask, gp, 0.0)
            return gp
        try:
            res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 800, 'gtol': 1e-9, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5

def active_set_refine(h, n, n_rounds, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k_idx = np.argsort(corr)[-5:]
        improved = False
        for idx in top_k_idx:
            k = idx - (n - 1)
            c_k = corr[idx]
            if k == 0:
                # Reduce sum h[i]^2 by pushing values toward 0 or 1
                order = np.argsort(np.abs(h - 0.5))
                for i in order[:5]:
                    if h[i] > 0.5:
                        h_try = h.copy()
                        h_try[i] = min(1.0, h[i] + 0.02)
                        h_try = make_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            h = h_try
                            best_c5 = c5_try
                            improved = True
                    else:
                        h_try = h.copy()
                        h_try[i] = max(0.0, h[i] - 0.02)
                        h_try = make_feasible(h_try, n)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            h = h_try
                            best_c5 = c5_try
                            improved = True
            else:
                # For shift k, find positions where h[i+k]=1 and h[i]=0 (or vice versa)
                # and try to reduce the product
                if k > 0:
                    pairs = [(i, i + k) for i in range(n - k)]
                else:
                    pairs = [(i - k, i) for i in range(-k, n)]
                # Sort by contribution h[i+k]*(1-h[i])
                contribs = []
                for i, j in pairs:
                    a = h[j]
                    b = h[i]
                    contribs.append((a * (1.0 - b), i, j))
                contribs.sort(reverse=True)
                for _c, i, j in contribs[:8]:
                    if _c < 0.01:
                        break
                    # Try to reduce h[j] or increase h[i]
                    h_try = h.copy()
                    amt = 0.03
                    if h[j] > 0.3:
                        h_try[j] -= amt
                    if h[i] < 0.7:
                        h_try[i] += amt
                    h_try = make_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
        if not improved:
            break
    return h, best_c5

def block_move_search(h, n, n_trials, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        h_try = h.copy()
        # Pick a random contiguous block and shift it
        block_len = rng.randint(3, max(4, n // 15))
        start = rng.randint(0, n - block_len)
        block = h_try[start:start + block_len].copy()
        # Try moving values within the block
        if rng.random() < 0.5:
            # Bump up one end, down other
            h_try[start] += 0.05
            h_try[start + block_len - 1] -= 0.05
        else:
            # Rotate the block
            shift = rng.randint(1, block_len)
            h_try[start:start + block_len] = np.concatenate([block[shift:], block[:shift]])
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
    return h, best_c5

def random_perturb_improve(h, n, n_trials, base_step, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for _ in range(n_trials):
        h_try = h.copy()
        n_mod = rng.randint(2, 12)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
        else:
            step *= 0.97
    return h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def structured_init(n, rng):
    inits = []
    # Alternating blocks
    if n % 4 == 0:
        inits.append(np.tile([1.0, 1.0, 0.0, 0.0], n // 4).copy())
    # Fixed point free
    h_sf = np.zeros(n)
    for i in range(n):
        h_sf[i] = 1.0 if ((i * 7 + 3) % n) < n // 2 else 0.0
    inits.append(make_feasible(h_sf, n))
    # Sine-based
    h_sin = 0.5 + 0.5 * np.sign(np.sin(2.0 * np.pi * 3.0 * np.arange(n) / n))
    inits.append(make_feasible(h_sin, n))
    # Random
    h_r = (rng.rand(n) < 0.5).astype(float)
    inits.append(make_feasible(h_r, n))
    return inits

def binary_sa(n, rng, n_steps, t_start, t_end):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new <= best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-15)):
            pass
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _global_beta, _global_n, _sum_target
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent.copy(), best_n)
    else:
        best_n = 160
        best_h = make_feasible(np.array([1.0, 0.0] * (best_n // 2)), best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with analytic gradient, multi-beta at parent resolution
    if remaining() > 180:
        betas_p1 = [15, 50, 200, 800, 3000]
        h_opt, c5_opt = slsqp_anneal(best_h, best_n, betas_p1, min(170, remaining() - 20))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: L-BFGS-B with penalty at parent resolution
    if remaining() > 120:
        betas_p2 = [50, 200, 1000, 5000]
        h_opt, c5_opt = lbfgs_anneal(best_h, best_n, betas_p2, min(110, remaining() - 20))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Active-set refinement
    if remaining() > 40:
        n_rounds = min(30, int(remaining() / 2))
        best_h, best_c5 = active_set_refine(best_h, best_n, n_rounds, rng)

    # Phase 4: Resolution upgrade to n=240
    n_up = 240
    if remaining() > 200:
        h_up = make_feasible(interpolate_to_n(best_h, n_up), n_up)
        betas_up = [30, 100, 400, 1500]
        h_opt, c5_opt = slsqp_anneal(h_up, n_up, betas_up, min(180, remaining() - 30))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_up

    # Phase 5: L-BFGS-B at upgraded resolution
    if remaining() > 100:
        betas_up2 = [100, 400, 2000]
        h_opt, c5_opt = lbfgs_anneal(best_h, best_n, betas_up2, min(90, remaining() - 20))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Active-set + block-move at current resolution
    if remaining() > 30:
        n_rounds = min(20, int(remaining() / 4))
        best_h, best_c5 = active_set_refine(best_h, best_n, n_rounds, rng)

    if remaining() > 30:
        n_trials = min(3000, int(remaining() / 0.05))
        best_h, best_c5 = block_move_search(best_h, best_n, n_trials, rng)

    # Phase 7: Random perturbation improvement
    if remaining() > 30:
        n_trials = min(10000, int((remaining() - 10) / 0.02))
        best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.03, rng)

    # Phase 8: Try different resolutions with SLSQP
    for n_try in [200, 256, 320]:
        if remaining() < 60:
            break
        if n_try == best_n:
            continue
        h_try = make_feasible(interpolate_to_n(best_h, n_try), n_try)
        betas_try = [50, 300, 1500]
        h_opt, c5_opt = slsqp_anneal(h_try, n_try, betas_try, min(55, remaining() - 15))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try

    # Phase 9: Binary SA at a couple resolutions for exploration
    for n_sa in [best_n, 200]:
        if remaining() < 40:
            break
        if n_sa < 40 or n_sa % 2 != 0:
            continue
        h_bin, c5_bin = binary_sa(n_sa, rng, n_steps=20000, t_start=0.5, t_end=0.002)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.astype(float)
            best_n = n_sa

    # Phase 10: Multi-start SLSQP from random perturbations of best
    for trial in range(4):
        if remaining() < 30:
            break
        h_try = best_h + rng.randn(best_n) * 0.015
        h_try = make_feasible(h_try, best_n)
        betas_ms = [200, 1000, 5000]
        h_opt, c5_opt = slsqp_anneal(h_try, best_n, betas_ms, min(25, remaining() - 10))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 11: Final aggressive SLSQP polish
    if remaining() > 30:
        betas_final = [500, 2000, 8000]
        h_opt, c5_opt = slsqp_anneal(best_h, best_n, betas_final, min(25, remaining() - 10))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 12: Final active-set and perturbation polish
    if remaining() > 15:
        best_h, best_c5 = active_set_refine(best_h, best_n, 10, rng)

    if remaining() > 10:
        n_trials = min(5000, int((remaining() - 5) / 0.02))
        best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.01, rng)

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Safety: ensure constraints
    assert np.all(best_h >= -1e-10), "h has negative values"
    assert np.all(best_h <= 1.0 + 1e-10), "h exceeds 1"
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6, "sum constraint violated"

    return (best_h, best_c5, best_n)
