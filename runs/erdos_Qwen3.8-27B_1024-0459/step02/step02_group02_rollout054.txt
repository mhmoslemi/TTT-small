import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, n):
    """Round to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float)
    target = n // 2
    # Sort by value, set top target to 1, rest to 0
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    # Handle ties at the boundary
    if h[idx[target-1]] == h[idx[target]]:
        # Randomize among tied values
        tied = np.where(np.isclose(h, h[idx[target]], atol=1e-12))[0]
        n_to_one = target - np.sum(out)
        if n_to_one > 0 and len(tied) > n_to_one:
            rng = np.random.RandomState(0)
            extra = rng.choice(tied, n_to_one, replace=False)
            out[extra] = 1.0
    return make_feasible(out, n)  # Should already be feasible

def thue_morse_init(n):

    return np.interp(x_new, x_old, h_src)

    """Greedy local search with single swap moves."""
    h = np.round(h_init).astype(float)

    best_c5 = compute_c5(h)
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    no_improve = 0
    start_t = time.time()
    
    while no_improve < stall_limit:

        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = next(iter(rng.choice(list(ones))))
        j = next(iter(rng.choice(list(zeros))))

            ones.add(j)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    
    return h.copy(), best_c5

                if c5_val < best_c5:
                    best_c5 = c5_val
                    current = candidate
        except Exception:
            pass
    
    return current, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    n = 256
    
    # Get parent
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
            if len(h_parent) != n:
                h_parent = interp_h(h_parent, n)
    except Exception:
        h_parent = None
    
    # Generate initial solutions
    inits = []
    if h_parent is not None:
        inits.append(make_feasible(h_parent, n))
        inits.append(round_to_binary(h_parent, n, rng))
    inits.append(thue_morse_init(n))
    # Phase-shifted Thue-Morse
    for shift in [1, 3, 5]:
        tm = np.zeros(n)
        for i in range(n):
            tm[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
        inits.append(tm)
    # Random binary
    for s in range(8):
        inits.append(random_binary(n, np.random.RandomState(seed + 200 + s)))
    
    best_h = inits[0].copy()
    best_c5 = compute_c5(best_h)
    best_bin_h = round_to_binary(best_h, n, rng)
    best_bin_c5 = compute_c5(best_bin_h)
    
    # Phase 1: SLSQP from top inits
    for idx, h0 in enumerate(inits[:6]):
        if budget_s - (time.time() - start_time) < 50:
            break
        h_opt, c5_val = slsqp_optimize(h0, n, max_iter=1500)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
        h_b = round_to_binary(h_opt, n, rng)
        c5_b = compute_c5(h_b)
        if c5_b < best_bin_c5:
            best_bin_c5 = c5_b
            best_bin_h = h_b.copy()
    
    # Phase 2: ILS on binary (main search, allocate most time here)
    # Sort binary inits by quality
    bin_candidates = [(compute_c5(h), h) for h in [best_bin_h]]
    for h0 in inits:
        h_b = round_to_binary(h0, n, rng)
        c5_b = compute_c5(h_b)
        if c5_b < best_bin_c5 + 0.04:
            bin_candidates.append((c5_b, h_b))
    bin_candidates.sort(key=lambda x: x[0])
    
    for c5_b, h_b in bin_candidates[:5]:
        if budget_s - (time.time() - start_time) < 100:
            break
        t_budget = min(100, budget_s - (time.time() - start_time) - 80)
        if t_budget < 30:
            break
        h_ils, c5_ils = iterated_local_search(h_b, n, rng, timeout=t_budget,
                                               n_perturb_range=(2, 4),
                                               max_trials_lm=30000, stall_limit=7000)
        if c5_ils < best_bin_c5:
            best_bin_c5 = c5_ils
            best_bin_h = h_ils.copy()
        if c5_ils < best_c5:
            best_c5 = c5_ils
            best_h = h_ils.copy()
    
    # Phase 3: SA from best binary
    if budget_s - (time.time() - start_time) > 60:
        t_sa = min(80, budget_s - (time.time() - start_time) - 50)
        h_sa, c5_sa = simulated_annealing(best_bin_h, n, rng, timeout=t_sa,
                                          t_start=3.0, t_end=0.003, t_stride=0.9996)
        if c5_sa < best_bin_c5:
            best_bin_c5 = c5_sa
            best_bin_h = h_sa.copy()
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    # Phase 4: Another ILS round from best
    if budget_s - (time.time() - start_time) > 80:
        t_ils2 = min(80, budget_s - (time.time() - start_time) - 50)
        h_ils2, c5_ils2 = iterated_local_search(best_bin_h, n, rng, timeout=t_ils2,
                                                 n_perturb_range=(3, 6),
                                                 max_trials_lm=30000, stall_limit=6000)
        if c5_ils2 < best_bin_c5:
            best_bin_c5 = c5_ils2
            best_bin_h = h_ils2.copy()
        if c5_ils2 < best_c5:
            best_c5 = c5_ils2
            best_h = h_ils2.copy()
    
    # Phase 5: Smooth SLSQP from best binary (continuous refinement)
    if budget_s - (time.time() - start_time) > 30:
        h_smooth, c5_smooth = smooth_max_optimize(best_bin_h, n, beta_start=15.0,
                                                   beta_end=250.0, n_steps=4, max_iter=400)
        if c5_smooth < best_c5:
            best_c5 = c5_smooth
            best_h = h_smooth.copy()
    
    # Phase 6: Direct SLSQP polish
    if budget_s - (time.time() - start_time) > 20:
        h_pol, c5_pol = slsqp_optimize(best_h, n, max_iter=2500, use_smooth=False)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    # Phase 7: Binary local search on polished (rounded) + final SLSQP
    if budget_s - (time.time() - start_time) > 40:
        h_b7 = round_to_binary(best_h, n, rng)
        h_ls7, c5_ls7 = binary_local_search(h_b7, n, rng, max_trials=200000,
                                             stall_limit=8000, timeout=35, targeted=True)
        if c5_ls7 < best_c5:
            best_c5 = c5_ls7
            best_h = h_ls7.copy()
        if budget_s - (time.time() - start_time) > 20:
            h_f7, c5_f7 = slsqp_optimize(h_ls7, n, max_iter=2000, use_smooth=False)
            if c5_f7 < best_c5:
                best_c5 = c5_f7
                best_h = h_f7.copy()
    
    # Phase 8: One more targeted SA if time allows
    if budget_s - (time.time() - start_time) > 50:
        t_sa2 = min(40, budget_s - (time.time() - start_time) - 25)
        h_sa2, c5_sa2 = simulated_annealing(best_h, n, rng, timeout=t_sa2,
                                            t_start=2.0, t_end=0.01, t_stride=0.999)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
        if budget_s - (time.time() - start_time) > 15:
            h_f8, c5_f8 = slsqp_optimize(h_sa2, n, max_iter=1500)
            if c5_f8 < best_c5:
                best_c5 = c5_f8
                best_h = h_f8.copy()
    
    # Final feasibility check
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n)
