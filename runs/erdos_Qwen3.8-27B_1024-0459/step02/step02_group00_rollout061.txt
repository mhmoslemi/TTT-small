            dnorm = np.linalg.norm(d)
            if dnorm < 1e-14:
                break
        d /= dnorm
        improved = False
        for step in [0.01, 0.02, 0.05, 0.005, 0.03, 0.08, 0.015]:
            h_trial = h + step * d
            h_trial = project_feasible(h_trial, n)
            c_new = compute_c5(h_trial)
            if c_new < best_c5 - 1e-12:
                h = h_trial
                best_c5 = c_new
                improved = True
                break
        if not improved:
            break
    return h, best_c5


def binary_pair_search(h, n, max_iters=8000, time_limit=30, rng=None):
    """Aggressive binary pair-swap local search."""
    global _N
    _N = n
    if rng is None:
        rng = np.random.RandomState(42)
    h = (h > 0.5).astype(float).copy()
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones > n // 2:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - n // 2, replace=False)
        h[remove] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, n // 2 - n_ones, replace=False)
        h[add] = 1.0
    dx_val = 2.0 / n
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    best_c5 = cur_max * dx_val
    start = time.time()
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if remaining() < 5:
            break
        k_max = int(np.argmax(corr))
        offset = k_max - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        n_try = 250
        for _ in range(n_try):
            # Biased sampling toward worst shift contributors
            if offset >= 0:
                ci = np.array([i for i in ones if i + offset < n and h[i + offset] == 0])
                if len(ci) > 0 and rng.rand() < 0.7:
                    i = ci[rng.randint(len(ci))]
                else:
                    i = ones[rng.randint(len(ones))]
            elif offset < 0:
                ci = np.array([i for i in ones if i - offset < n and h[i - offset] == 0])
                if len(ci) > 0 and rng.rand() < 0.7:
                    i = ci[rng.randint(len(ci))]
                else:
                    i = ones[rng.randint(len(ones))]
            else:
                i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = float(np.max(new_corr))
            h[i] = 1.0
            h[j] = 0.0
            if new_max < cur_max - 1e-12:
                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                cur_max = new_max
                best_c5 = cur_max * dx_val
                found = True
                break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                # Perturbation
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
                stall = 0
    return h, best_c5


def make_initial_structural(n, seed_val):
    """Create diverse structural initializations."""
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    # Thue-Morse
    for i in range(n):
        c = 0
        v = i
        while v:
            c ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if c == 1 else 0.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _BETA, _TGT
    _START = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
    except Exception:
        parent_n = 160
        parent_h = np.zeros(parent_n)
        parent_h[:parent_n // 2] = 1.0
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n

    # Phase 1: Refine parent with gradient SLSQP at multiple betas
    n = best_n
    h = best_h.copy()
    for beta in [100, 300, 800, 2000, 5000]:
        if remaining() < 8:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
            h = h_new.copy()

    # Active-set equalize
    if remaining() > 10:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=300)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            best_n = n

    # Phase 2: Try different n values
    n_tries = sorted(set([
        best_n, best_n - 2, best_n + 2, best_n + 4,
        best_n + 6, best_n + 8, best_n + 10, best_n + 14,
        best_n + 16, best_n + 20, best_n + 24,
        int(best_n * 1.25), int(best_n * 1.5), int(best_n * 2)
    ]))
    n_tries = [nn for nn in n_tries if 100 <= nn <= 400 and nn % 2 == 0]

    for n_t in n_tries:
        if remaining() < 15:
            break
        if n_t == best_n:
            continue

        # Interpolate
        h_interp = interp_to(best_h, best_n, n_t)
        h_feas = project_feasible(h_interp, n_t)
        c5_interp = compute_c5(h_feas)

        # SLSQP on this n
        h_opt = h_feas
        for beta in [200, 700, 2000]:
            if remaining() < 8:
                break
            h_opt = slsqp_smooth(h_opt, n_t, beta, maxiter=300)
        c5_opt = compute_c5(h_opt)

        # Active set equalize
        h_opt, c5_opt = active_set_equalize(h_opt, n_t, max_iters=150)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_t

        # Binary search on this n
        if remaining() > 12:
            h_bin = (h_opt > 0.5).astype(float)
            # Ensure correct count
            n_ones = int(np.sum(h_bin))
            if n_ones > n_t // 2:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - n_t // 2, replace=False)]] = 0
            elif n_ones < n_t // 2:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[rng.choice(len(idx), n_t // 2 - n_ones, replace=False)]] = 1

            h_b, c5_b = binary_pair_search(h_bin, n_t, max_iters=4000,
                                           time_limit=min(20, max(5, remaining() - 10)), rng=rng)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n_t

    # Phase 3: Final intensive refinement on best solution
    n = best_n
    h = best_h.copy()

    # SLSQP with increasing beta
    for beta in [500, 1500, 5000, 10000]:
        if remaining() < 6:
            break
        h_new = slsqp_smooth(h, n, beta, maxiter=500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()

    # Active set
    if remaining() > 8:
        h_eq, c5_eq = active_set_equalize(h, n, max_iters=400)
        if c5_eq < best_c5:
            best_c5 = c5_eq
            best_h = h_eq.copy()
            h = h_eq.copy()

    # Binary fine-tune
    if remaining() > 10:
        h_bin = (h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones != n // 2:
            if n_ones > n // 2:
                idx = np.where(h_bin == 1)[0]
                h_bin[idx[rng.choice(len(idx), n_ones - n // 2, replace=False)]] = 0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[rng.choice(len(idx), n // 2 - n_ones, replace=False)]] = 1
        h_b, c5_b = binary_pair_search(h_bin, n, max_iters=10000,
                                       time_limit=min(25, max(5, remaining() - 5)), rng=rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
            best_n = n

    # Final SLSQP polish
    if remaining() > 8:
        n = best_n
        for beta in [3000, 8000, 15000]:
            if remaining() < 5:
                break
            h_new = slsqp_smooth(best_h, n, beta, maxiter=400)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5 + 1e-10:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
