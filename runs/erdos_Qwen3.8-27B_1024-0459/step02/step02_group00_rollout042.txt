        if remaining_time() > 30:
            h_p, c5_p = greedy_binary_polish(h_sa2, n, rng, max_sweeps=30)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
                print("  Phase2 polish: C5={:.7f}".format(c5_p))
    
    # Phase 3: Continuous optimization
    if remaining_time() > 30:
        n = best_n
        h = best_h.copy()
        
        # Multi-target subgradient
        h_sg, c5_sg = multi_target_subgradient(h, n, rng, max_iters=400)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            print("  Subgrad: C5={:.7f}".format(c5_sg))
        
        # SLSQP
        for beta_val in [50, 100, 300, 800]:
            if remaining_time() < 10:
                break
            h_new = slsqp_optimize(best_h, n, beta_val, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  SLSQP b={}: C5={:.7f}".format(beta_val, c5_new))
        
        # More subgradient
        if remaining_time() > 15:
            h_sg2, c5_sg2 = multi_target_subgradient(best_h, n, rng, max_iters=200)
            if c5_sg2 < best_c5:
                best_c5 = c5_sg2
                best_h = h_sg2.copy()
    
    # Phase 4: Final polish
    if remaining_time() > 10:
        n = best_n
        h_r = (best_h >= 0.5).astype(float)
        if h_r.sum() != n / 2:
            sorted_idx = np.argsort(best_h)[::-1]
            h_r = np.zeros(n)
            h_r[sorted_idx[:n // 2]] = 1.0
        h_r = project_feasible(h_r, n)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
        
        h_fp, c5_fp = greedy_binary_polish(best_h, n, rng, max_sweeps=15)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_fp.copy()
    
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
