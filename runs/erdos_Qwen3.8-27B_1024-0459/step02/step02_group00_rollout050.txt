def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        # Distribute among non-saturated coordinates
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
