        lo = np.min(h_raw) - 1

def compute_value_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Smooth max weights
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    p = exps / np.sum(exps)
    
    value = (m + np.log(np.sum(exps)) / beta) * dx
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        L = k - (n - 1)
        w = p[k]
        if L == 0:
            # c = sum h[i](1-h[i]), grad = (1-2h)
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            # c = sum_{j=0}^{n-1-L} h[j+L](1-h[j])
            # d/dh[i]: (1-h[i-L]) if i>=L (from h[i] term) - h[i+L] if i<n-L (from (1-h) term)
            if n - L > 0:
                # h[i+L] term for i = 0,...,n-1-L: contributes -h[i+L] to grad[i]
                # But wait: d/dh[i] of h[j+L](1-h[j]) is:
                #   if i = j+L (i.e., j = i-L): (1-h[i-L]) for i-L in [0, n-1-L]
                #   if i = j (i.e., j = i): -h[i+L] for i in [0, n-1-L]
                # So: grad[i] += (1-h[i-L]) for i in [L, n-1]
                #     grad[i] -= h[i+L] for i in [0, n-1-L]
                grad[L:] += w * (1.0 - h[:n-L])  # i from L to n-1, h[i-L] from h[0] to h[n-1-L]
                grad[:n-L] -= w * h[L:]  # i from 0 to n-1-L, h[i+L] from h[L] to h[n-1]
        else:
            Lp = -L  # positive
            # c = sum_{j=0}^{n-1-Lp} h[j](1-h[j+Lp])
            # d/dh[i]: (1-h[i+Lp]) if i in [0, n-1-Lp] - h[i-Lp] if i in [Lp, n-1]
            grad[:n-Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n-Lp]
    
    return value, dx * grad

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    lo = -1.0
    hi = 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:

        else:

def compute_value_and_grad(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

    s = np.sum(exps)
    p = exps / s

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def make_even_binary(n, rng):

def make_thue_morse(n):
    """Generate Thue-Morse based binary sequence."""
    h = np.zeros(n)
    t = 0
    for i in range(n):

        count = 0
        while x & 1:

            x >>= 1
        # Thue-Morse: t(i) = popcount(i) mod 2

def binary_local_search(h_bin, n, n_iterations, rng, time_limit):
    """Simulated annealing on binary sequence."""
    start = time.time()
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = 0.02  # initial temperature
    T_min = 0.001
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iterations):
        if time.time() - start > time_limit:
            break
        if it % len(ones) == 0:
            T = max(T_min, T * 0.999)
        
        # Random swap
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        if i1 == i0:
            continue
        
        h[i1] = 0
        h[i0] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        elif rng.random() < np.exp(-delta / T):
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i1] = 1
            h[i0] = 0
    
    return best_h, best_c5

def project_feasible(h, n):
    """Euclidean projection onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0

    lo = -1.0

        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    # Ensure exact sum (fix floating point)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        # Adjust one element
        if diff > 0:
            idx = np.argmin(1.0 - result)
        else:

        result[idx] += diff
        result = np.clip(result, 0.0, 1.0)

def compute_value_grad(h, beta):
    """Compute smooth-max value and gradient of C5 objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    p = exps / s
    value = (m + np.log(s) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = p[k]
        if w < 1e-14:

        L = k - (n - 1)
