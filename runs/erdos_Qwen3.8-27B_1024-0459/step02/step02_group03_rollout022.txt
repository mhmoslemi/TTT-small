```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / beta

def smooth_max_w(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(s)
    return e / np.sum(e)

def sm_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def sm_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_w(corr, beta)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += wk * (1.0 - h[L:])
            g[L:] -= wk * h[:n - L]
        else:
            Lp = -L
            g[Lp:] += wk * (1.0 - h[:n - Lp])
            g[:n - Lp] -= wk * h[Lp:]
    g *= dx
    return g

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = n / 2.0
    d = t - np.sum(h)
    if abs(d) < 1e-12:
        return h
    h += d / n
    h = np.clip(h, 0.0, 1.0)
    d = t - np.sum(h)
    if abs(d) > 1e-12:
        m = (h > 1e-9) & (h < 1.0 - 1e-9)
        c = np.sum(m)
        if c > 0:
            h[m] += d / c
            h = np.clip(h, 0.0, 1.0)
    d = t - np.sum(h)
    if abs(d) > 1e-14:
        i = int(np.argmin(np.minimum(h, 1.0 - h)))
        h[i] += d
        h = np.clip(h, 0.0, 1.0)
    return h

def interp(h_src, n_dst):
    ns = len(h_src)
    if ns == n_dst:
        return project(h_src.copy(), n_dst)
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    hd = np.interp(xd, xs, h_src)
    return project(hd, n_dst)

def soptsqolve(h0, n, beta, maxiter, ftol=1e-14):
    def fun(x):
        return sm_obj(x, beta)
    def jac(x):
        return sm_grad(x, beta)
    def con(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
    return project(res.x, n)

def fourier_init(n, rng, K=4):
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = rng.uniform(-0.15, 0.15)
        b = rng.uniform(-0.15, 0.15)
        h += a * np.cos(2 * np.pi * k * np.arange(n) / n)
        h += b * np.sin(2 * np.pi * k * np.arange(n) / n)
    return project(h, n)

def block_init(n, rng, nblocks):
    blocks = n // nblocks
    h = np.zeros(n)
    for b in range(nblocks):
        if b % 2 == 0:
            h[b*blocks:(b+1)*blocks] = 1.0
    # Randomize within blocks
    for b in range(nblocks):
        s = b * blocks
        e = s + blocks
        some = rng.choice(blocks, size=rng.randint(0, blocks//4), replace=False)
        h[s + some] = 1.0 - h[s + some]
    return project(h, n)

def binary_local_search(h, n, rng, max_rounds=5000, tol=1e-10):
    h = h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _sm_val(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(s))) / beta

def _sm_w(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(s)
    return e / np.sum(e)

def sm_objective(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _sm_val(corr, beta) * dx

def sm_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = _sm_w(corr, beta)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += wk * (1.0 - h[L:])
            g[L:] -= wk * h[:n - L]
        else:
            Lp = -L
            g[Lp:] += wk * (1.0 - h[:n - Lp])
            g[:n - Lp] -= wk * h[Lp:]
    g *= dx
    return g

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    d = target - np.sum(h)
    if abs(d) < 1e-13:
        return h
    h = h + d / n
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += d / cnt
            h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-15:
        idx = int(np.argmin(np.minimum(h, 1.0 - h)))
        h[idx] += d
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h_src, n_dst):
    ns = len(h_src)
    if ns == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    xs = (np.arange(ns) + 0.5) * (2.0 / ns)
    xd = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    hd = np.interp(xd, xs, h_src)
    return project_feasible(hd, n_dst)

def solve_slsqp(h0, n, beta, maxiter, ftol=1e-15):
    def fun(x):
        return sm_objective(x, beta)
    def jac(x):
        return sm_gradient(x, beta)
    def con_eq(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq}]
    res = minimize(fun, h0.copy(), jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
    return project_feasible(res.x, n)

def solve_lbfgs(h0, n, beta, maxiter, pen_lambda=50.0):
    def fun(x):
        base = sm_objective(x, beta)
        sd = np.sum(x) - n / 2.0
        return base + pen_lambda * sd * sd
    def jac(x):
        g = sm_gradient(x, beta)
        sd = np.sum(x) - n / 2.0
        g += 2.0 * pen_lambda * sd / n
        return g
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0.copy(), jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': 0})
    return project_feasible(res.x, n)

def fourier_init(n, rng, K=5):
    h = np.full(n, 0.5)
    t = np.arange(n) / n
    for k in range(1, K + 1):
        a = rng.uniform(-0.12, 0.12)
        b = rng.uniform(-0.12, 0.12)
        h += a * np.cos(2.0 * np.pi * k * t)
        h += b * np.sin(2.0 * np.pi * k * t)
    return project_feasible(h, n)

def block_init(n, rng, nblocks=6):
    bl = n // nblocks
    h = np.zeros(n)
    for b in range(nblocks):
        if b % 2 == 0:
            h[b * bl: min(n, (b + 1) * bl)] = 1.0
    for _ in range(n // 10):
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] != h[j]:
            h[i], h[j] = h[j], h[i]
    return project_feasible(h, n)

def golden_init(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    frac = 0.5 + rng.uniform(-0.05, 0.05)
    offset = rng.uniform(0, 1)

    for j in range(n):

def binary_local_search(h, n, rng, max_rounds=8000):

    h[h <= 0.5] = 0.0
    cnt = int(np.sum(h))

        order = np.argsort(h[ones])[:cnt - n // 2]

        order = np.argsort(-h[zeros])[:n // 2 - cnt]
        h[zeros[order]] = 1.0

    for _ in range(max_rounds):
        improved = False
        for _ in range(40):
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0.0
            h[i0] = 1.0
            nc = compute_c5(h)
            if nc < c5 - 1e-14:
                c5 = nc
                improved = True
            else:
                h[i1] = 1.0
                h[i0] = 0.0
        if not improved:
            break
    return h, c5

def anneal_binary(h, n, rng, steps=20000, T0=0.005, T1=0.0001):
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    cnt = int(np.sum(h))
    if cnt > n // 2:
        ones = np.where(h == 1.0)[0]
        order = np.argsort(h[ones])[:cnt - n // 2]
        h[ones[order]] = 0.0
    elif cnt < n // 2:
        zeros = np.where(h == 0.0)[0]
        order = np.argsort(-h[zeros])[:n // 2 - cnt]
        h[zeros[order]] = 1.0
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    for s in range(steps):
        T = T0 * (T1 / T0) ** (s / steps)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1] = 0.0
        h[i0] = 1.0
        nc = compute_c5(h)
        if nc < c5:
            c5 = nc
            if nc < best_c5:
                best_c5 = nc
                best_h = h.copy()
        else:
            delta = nc - c5
            if rng.uniform() > np.exp(-delta / max(T, 1e-10)):
                h[i1] = 1.0
                h[i0] = 0.0
            else:
                c5 = nc
    return best_h, best_c5

def multi_phase_optimize(h0, n, remaining_fn):
    cur = h0.copy()
    c5_cur = compute_c5(cur)
    beta_schedule = [(15, 200), (40, 200), (100, 300), (200, 400), (500, 500), (1200, 600), (3000, 400)]
    for beta, iters in beta_schedule:
        if remaining_fn() < 30:
            break
        use_iters = min(iters, max(50, int(remaining_fn() / 0.08)))
        try:
            h_new = solve_slsqp(cur, n, beta, use_iters)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
        try:
            h_new2 = solve_lbfgs(cur, n, beta, min(use_iters, 200))
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < c5_cur:
                cur = h_new2
                c5_cur = c5_new2
        except Exception:
            pass
    return cur, c5_cur

def targeted_polish(h, n, rng, remaining_fn):
    cur = h.copy()
    c5_cur = compute_c5(cur)
    for trial in range(6):
        if remaining_fn() < 40:
            break
        h_pert = cur.copy()
        n_pert = rng.randint(3, max(4, n // 10))
        pert_idx = rng.choice(n, size=n_pert, replace=False)
        h_pert[pert_idx] = np.clip(h_pert[pert_idx] + rng.uniform(-0.2, 0.2, n_pert), 0, 1)
        h_pert = project_feasible(h_pert, n)
        for beta in [200, 800]:
            if remaining_fn() < 20:
                break
            try:
                h_new = solve_slsqp(h_pert, n, beta, 300)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur:
                    cur = h_new
                    c5_cur = c5_new
            except Exception:
                break
    return cur, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def remaining():
        return max(0.0, budget_s - elapsed() - 10.0)

    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = golden_init(best_n, rng)
        best_c5 = compute_c5(best_h)
        print("Default init: n={}, C5={:.6f}".format(best_n, best_c5))

    n_list = sorted(set([best_n, 200, 256, 320, 384, 480]))
    n_list = [nn for nn in n_list if 128 <= nn <= 512]

    for n in n_list:
        if remaining() < 60:
            break
        print("=== n={} (remaining {:.0f}s) ===".format(n, remaining()))

        inits = []
        if parent_h is not None:
            inits.append(interp_to(parent_h, n))
        inits.append(interp_to(best_h, n))
        inits.append(fourier_init(n, rng, K=4))
        inits.append(fourier_init(n, rng, K=6))
        inits.append(block_init(n, rng, 6))
        inits.append(block_init(n, rng, 8))
        inits.append(golden_init(n, rng))

        for h0 in inits:
            if remaining() < 45:
                break
            h0 = project_feasible(h0, n)
            c5_init = compute_c5(h0)
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = h0.copy()
                best_n = n
                print("  Init: n={}, C5={:.6f}".format(n, best_c5))

            h_opt, c5_opt = multi_phase_optimize(h0, n, remaining)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
                print("  Opt: n={}, C5={:.6f}".format(n, best_c5))

    # Targeted polish at best resolution
    if remaining() > 80:
        print("=== Targeted polish at n={} ===".format(best_n))
        best_h, best_c5 = targeted_polish(best_h, best_n, rng, remaining)
        print("  Polished: C5={:.6f}".format(best_c5))

    # Try a couple more resolutions with the best solution transferred
    for n_try in [best_n + 64, best_n + 128]:
        if remaining() < 50:
            break
        if n_try > 512:
            continue
        h_transfer = interp_to(best_h, n_try)
        c5_tr = compute_c5(h_transfer)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_transfer.copy()
            best_n = n_try
        h_opt, c5_opt = multi_phase_optimize(h_transfer, n_try, remaining)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try
            print("  Transfer n={}: C5={:.6f}".format(n_try, best_c5))

    # Binary rounding + local search + annealing
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        cnt = int(np.sum(h_bin))
        n_want = best_n // 2
        if cnt > n_want:
            ones = np.where(h_bin == 1.0)[0]
            vals = best_h[ones]
            order = np.argsort(vals)[:cnt - n_want]
            h_bin[ones[order]] = 0.0
        elif cnt < n_want:
            zeros = np.where(h_bin == 0.0)[0]
            vals = best_h[zeros]
            order = np.argsort(-vals)[:n_want - cnt]
            h_bin[zeros[order]] = 1.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Binary: C5={:.6f}".format(best_c5))

        if remaining() > 40:
            h_ls, c5_ls = binary_local_search(best_h, best_n, rng, max_rounds=min(8000, int(remaining() * 2)))
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                print("  LocalSearch: C5={:.6f}".format(best_c5))

        if remaining() > 40:
            h_an, c5_an = anneal_binary(best_h, best_n, rng, steps=min(15000, int(remaining() * 3)), T0=0.004, T1=0.0001)
            if c5_an < best_c5:
                best_c5 = c5_an
                best_h = h_an.copy()
                print("  Anneal: C5={:.6f}".format(best_c5))

    # Final verification and correction
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
