import time
import numpy as np
from scipy.optimize import linprog, minimize

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def corr_grad_lag(h, k, n):
    """Gradient of corr_k at h, where k is the lag (index offset from center)"""
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:

        if k > 0:

            #           -h[j-lag] for j = lag..n-1 (from second factor)
            g[:n-lag] += (1.0 - h[lag:])  # (1-h[j+lag]) for j=0..n-1-lag
            g[lag:] -= h[:n-lag]  # -h[j-lag] for j=lag..n-1

            #           -h[j+lag] for j = 0..n-1-lag (from second factor)
            g[lag:] += (1.0 - h[:n-lag])  # (1-h[j-lag]) for j=lag..n-1
            g[:n-lag] -= h[lag:]  # -h[j+lag] for j=0..n-1-lag
    return g

def compute_corr_all(h):
    """Compute all correlation values"""
    return np.correlate(h, 1.0 - h, mode="full")

def lp_cutting_plane(h, deadline, n_active=None, max_iter=5000):

    n = h.size

    best_h = h.copy()

    center = n - 1
    n_lags = 2 * n - 1
    
    if n_active is None:

    for iteration in range(max_iter):
        if time.perf_counter() >= deadline:

        current_max = float(np.max(corr))
        current_c = current_max * (2.0 / n)
        
        if current_c < best_c:
            best_c = current_c
        
        # Select active lags (top n_active)
        if n_active < n_lags:
            active_idx = np.argsort(corr)[-n_active:]
        else:

        # Minimize t

        #     => grad[idx]^T h - t <= grad[idx]^T best_h - corr[idx]
        #   sum(h) = target
        #   0 <= h[i] <= 1
        #   t >= 0

        c_obj[-1] = 1.0

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    """Project x onto {sum = target, 0 <= x <= 1}"""
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

    return h

def smooth_lse(h, n, alpha):
    """Log-sum-exp smooth approximation of max(corr)"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return 1e100, np.zeros(n)
    val = (2.0 / n) * (m + np.log(z) / alpha)
    # Gradient
    w = e / z
    g = np.zeros(n, dtype=float)
    center = n - 1
    for t in range(len(corr)):
        wt = w[t]
        if wt < 1e-10:
            continue
        k = t - center
        if k == 0:
            g += wt * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += wt
                g[:n-lag] -= wt * h[lag:]
                g[lag:] -= wt * h[:n-lag]
            else:
                g[:n-lag] += wt
                g[lag:] -= wt * h[:n-lag]
                g[:n-lag] -= wt * h[lag:]
    g *= (2.0 / n)
    return val, g

def corr_grad_lag(h, k, n):
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:

        if k > 0:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
        else:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
    return g

def lp_iteration(h, target, n_active, rng):
    """One LP cutting plane iteration"""
    n = h.size
    center = n - 1
    n_lags = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Select active lags
    if n_active < n_lags:
        # Use top n_active with some randomness
        order = np.argsort(corr)
        top = order[-n_active:]
        # Add some random lags to avoid getting stuck
        n_rand = max(5, n_active // 10)
        rand_lags = rng.integers(0, n_lags, n_rand)
        active_idx = np.unique(np.concatenate([top, rand_lags]))
    else:
        active_idx = np.arange(n_lags)
    
    # Build LP
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    m_ineq = len(active_idx)
    A_ub = np.zeros((m_ineq, n + 1))
    b_ub = np.zeros(m_ineq)
    
    for ii in range(m_ineq):
        idx = int(active_idx[ii])
        k = idx - center
        grad = corr_grad_lag(h, k, n)
        A_ub[ii, :n] = grad
        A_ub[ii, n] = -1.0
        b_ub[ii] = float(grad @ h) - float(corr[idx])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True})
    except Exception:
        return None, h
    
    if res.status != 0:
        return None, h
    
    return res.x[:n], res.x[n]
