            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                alpha_adapt = min(2.0, alpha_adapt * 1.2)
                break
        if not improved:
            alpha_adapt = max(0.1, alpha_adapt * 0.7)
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0
    return project_density(best, target), evaluate_erdos(best)

def pair_swap_search(h, deadline, max_iter=3000, top_frac=0.06, min_k=6):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(12345)
    stall = 0
    n_pos = min(12, max(4, n // 50))
    n_neg = min(12, max(4, n // 50))
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        g, _, _, _ = active_set_gradient(best, int(np.ceil(n * top_frac)), min_k)
        neg_cand = np.argsort(g)[:n_neg]
        pos_cand = np.argsort(-g)[:n_pos]
        found = False
        for ii in neg_cand:
            ii = int(ii)
            for jj in pos_cand:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                for frac in (0.3, 0.6, 0.9, 1.0):
                    d = frac * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            for _r in range(20):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                d = float(rng.random()) * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
    return project_density(best, target), evaluate_erdos(best)

def binary_push_optimize(h, deadline, n_thresholds=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    for thresh in np.linspace(0.02, 0.20, n_thresholds):
        if time.perf_counter() >= deadline:
            break
        h2 = best.copy()
        for i in range(n):
            if h2[i] < thresh:
                h2[i] = 0.0
            elif h2[i] > 1.0 - thresh:
                h2[i] = 1.0
        h2 = project_density(h2, target)
        c2 = evaluate_erdos(h2)
        if c2 < best_c:
            best = h2
            best_c = c2
    return best, best_c

def coordinate_descent_binary(h, deadline, max_passes=50):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    for _ in range(max_passes):
        if time.perf_counter() >= deadline:
            break
        improved = False
        for i in range(n):
            if time.perf_counter() >= deadline:
                break
            old_val = best[i]
            for new_val in (0.0, 1.0):
                if abs(new_val - old_val) < 1e-10:
                    continue
                cand = best.copy()
                cand[i] = new_val
                cand = project_density(cand, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-14:
                    best = cand
                    best_c = c
                    improved = True
                    break
        if not improved:
            break
    return best, best_c

def make_random_start(n, rng, mode=0):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 7 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        nseg = int(rng.integers(6, 15))
        h = np.zeros(n)
        seg_len = n // nseg
        for i in range(0, n, seg_len):
            end = min(i + seg_len, n)
            h[i:end] = float(rng.uniform(0.1, 0.9))
    elif mode == 2:
        u = rng.random(n)
        h = np.where(u > 0.5, 0.8, 0.2).astype(float)
        h += rng.normal(0, 0.08, n)
    else:
        freq = int(rng.integers(2, 10))
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2 + rng.uniform(0, 6.28))
        h += 0.1 * np.sin(2 * np.pi * (2 * freq) * x / 2 + rng.uniform(0, 6.28))
    return project_density(h, 0.5 * n)

def optimize_at_resolution(n, h0, deadline_time):
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    rem = deadline_time - time.perf_counter()
    if rem < 10:
        return best_h, best_c

    t1 = time.perf_counter() + min(rem * 0.25, 120.0)
    alphas = (30.0, 100.0, 300.0) if n <= 500 else (30.0, 80.0)
    h, best_c = slsqp_phase(h, n, alphas, t1)
    if best_c < evaluate_erdos(best_h):
        best_h = h.copy()

    rem = deadline_time - time.perf_counter()
    if rem < 8:
        return project_density(best_h, target), evaluate_erdos(best_h)

    t2 = time.perf_counter() + rem * 0.30
    h2, c2 = gradient_descent_active(best_h, t2, max_iter=400, top_frac=0.06, min_k=6)
    if c2 < best_c:
        best_h, best_c = h2, c2

    rem = deadline_time - time.perf_counter()
    if rem < 8:
        return project_density(best_h, target), evaluate_erdos(best_h)

    t3 = time.perf_counter() + rem * 0.25
    h3, c3 = pair_swap_search(best_h, t3, max_iter=2000, top_frac=0.05, min_k=6)
    if c3 < best_c:
        best_h, best_c = h3, c3

    rem = deadline_time - time.perf_counter()
    if rem < 5:
        return project_density(best_h, target), evaluate_erdos(best_h)

    t4 = time.perf_counter() + rem * 0.20
    h4, c4 = binary_push_optimize(best_h, t4, n_thresholds=12)
    if c4 < best_c:
        best_h, best_c = h4, c4

    rem = deadline_time - time.perf_counter()
    if rem < 5:
        return project_density(best_h, target), evaluate_erdos(best_h)

    t5 = time.perf_counter() + rem * 0.25
    h5, c5 = gradient_descent_active(best_h, t5, max_iter=300, top_frac=0.05, min_k=6)
    if c5 < best_c:
        best_h, best_c = h5, c5

    rem = deadline_time - time.perf_counter()
    if rem < 5:
        return project_density(best_h, target), evaluate_erdos(best_h)

    t6 = time.perf_counter() + rem * 0.30
    h6, c6 = pair_swap_search(best_h, t6, max_iter=1500, top_frac=0.04, min_k=8)
    if c6 < best_c:
        best_h, best_c = h6, c6

    rem = deadline_time - time.perf_counter()
    if rem > 10 and n <= 500:
        t7 = time.perf_counter() + rem * 0.5
        h7, c7 = coordinate_descent_binary(project_density(best_h, target), t7, max_passes=30)
        if c7 < best_c:
            best_h, best_c = h7, c7
        rem = deadline_time - time.perf_counter()
        if rem > 8:
            t8 = time.perf_counter() + rem * 0.7
            h8, c8 = gradient_descent_active(best_h, t8, max_iter=200, top_frac=0.04, min_k=8)
            if c8 < best_c:
                best_h, best_c = h8, c8

    return project_density(best_h, target), evaluate_erdos(best_h)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 10.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    base_n = best_n

    resolutions = []
    if base_n >= 2:
        resolutions.append(base_n)
    for mult in [2, 3]:
        cand_n = base_n * mult
        if 2 <= cand_n <= 1000 and cand_n not in resolutions:
            resolutions.append(cand_n)
    resolutions = sorted(resolutions)

    total_time = deadline - time.perf_counter()
    weights = []
    for n in resolutions:
        if n == base_n:
            weights.append(1.0)
        elif n <= 2 * base_n:
            weights.append(1.5)
        else:
            weights.append(1.0)
    wsum = sum(weights)

    for idx, ni in enumerate(resolutions):
        rem = deadline - time.perf_counter()
        if rem < 30.0:
            break
        alloc = total_time * weights[idx] / wsum
        alloc = min(alloc, rem - 10.0)
        alloc = max(25.0, alloc)

        starts = []
        starts.append(upsample_h(best_h, ni))
        if parent is not None and ni > base_n:
            starts.append(upsample_h(parent, ni))
        n_rand = 2 if ni <= 500 else 1
        for mm in range(n_rand):
            starts.append(make_random_start(ni, rng, mm % 4))

        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-5):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        per_start = alloc / len(uniq)
        per_start = max(15.0, per_start)

        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 20.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 5.0)
            h_out, c_out = optimize_at_resolution(ni, s, sub_dl)
            h_out = project_density(h_out, 0.5 * ni)
            c_out = evaluate_erdos(h_out)
            if c_out < best_c:
                best_h = h_out.copy()
                best_c = c_out
                best_n = ni

    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)

        t_fin1 = time.perf_counter() + rem * 0.25
        h2, c2 = gradient_descent_active(h_fin, t_fin1, max_iter=500, top_frac=0.04, min_k=8)
        if c2 < best_c:
            h_fin, best_c = h2, c2

        rem = deadline - time.perf_counter()
        if rem > 10.0:
            t_fin2 = time.perf_counter() + rem * 0.30
            h3, c3 = pair_swap_search(h_fin, t_fin2, max_iter=3000, top_frac=0.03, min_k=10,
                                      )
            if c3 < best_c:
                h_fin, best_c = h3, c3

        rem = deadline - time.perf_counter()
        if rem > 8.0 and nbest <= 500:
            t_fin3 = time.perf_counter() + rem * 0.40
            h4, c4 = binary_push_optimize(h_fin, t_fin3, n_thresholds=15)
            if c4 < best_c:
                h_fin, best_c = h4, c2

        rem = deadline - time.perf_counter()
        if rem > 8.0:
            t_fin4 = time.perf_counter() + rem * 0.50
            h5, c5 = gradient_descent_active(h_fin, t_fin4, max_iter=400, top_frac=0.03, min_k=10)
            if c5 < best_c:
                h_fin, best_c = h5, c5

        rem = deadline - time.perf_counter()
        if rem > 6.0:
            t_fin5 = time.perf_counter() + rem * 0.70
            h6, c6 = pair_swap_search(h_fin, t_fin5, max_iter=2000, top_frac=0.03, min_k=10)
            if c6 < best_c:
                h_fin, best_c = h6, c6

        best_h = project_density(h_fin, target)
        best_c = evaluate_erdos(best_h)
        best_n = nbest

    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
