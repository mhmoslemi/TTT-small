  - Term at i=a-L (where h[a-L] multiplies (1-h[a])): was h[a-L]*(1-1)=0, now h[a-L]*(1-0)=h[a-L]. Change: +h[a-L]

  - Term at i=b-L: was h[b-L]*(1-0)=h[b-L], now h[b-L]*(1-1)=0. Change: -h[b-L]

Total change to corr at shift L:
= (-1 + h[a+L]) + h[a-L] + (1-h[b+L]) - h[b-L]
= h[a+L] + h[a-L] - h[b+L] - h[b-L]

This only depends on 4 values! So for a given flip (a,b), the change to ALL correlations can be computed in O(n) time (one for each L), not O(n²).

    def __init__(self, h, n):
        self.h = h.copy()
        self.n = n
        self.dx = 2.0 / n
        self.corr = np.correlate(h, 1.0 - h, mode='full')
        self.c5 = np.max(self.corr) * self.dx
    
    def flip(self, a, b):

        h = self.h

        # Term from a (1→0):
        #   i=a contributes: -(1-h[a+L]) if valid
        #   i=a-L contributes: +h[a-L] if valid (and h[a] was the (1-h) part)

        # L = k - (n-1)

        # As first factor (i=a): contributes h[a]*(1-h[a+L]) = 1*(1-h[a+L])
        #   Valid if 0≤a<n and 0≤a+L<n
        #   Change: lose this term → -(1-h[a+L])
        # As second factor (i+L=a, i=a-L): contributes h[a-L]*(1-h[a]) = h[a-L]*0 = 0
        #   After flip: h[a-L]*(1-0) = h[a-L]
        #   Valid if 0≤a-L<n and 0<a<n

        # For h[b] going 0→1:

        #   After: 1*(1-h[b+L])
        #   Valid if 0≤b<n and 0≤b+L<n
        #   Change: gain (1-h[b+L])
        # As second factor (i+L=b, i=b-L): contributes h[b-L]*(1-0) = h[b-L]
        #   After: h[b-L]*(1-1) = 0

        #   if 0≤a-L<n: delta += h[a-L]

def compute_delta(a, b, h, n):
    """Compute delta to correlation from flipping h[a]:1→0, h[b]:0→1"""
    L = np.arange(-(n-1), n)  # all possible shifts
    delta = np.zeros(len(L))
    
    # For a (1→0):
    # -(1-h[a+L]) where 0≤a+L<n
    valid_ap = (a + L >= 0) & (a + L < n)
    delta[valid_ap] -= (1 - h[a + L[valid_ap]])
    
    # +h[a-L] where 0≤a-L<n
    valid_am = (a - L >= 0) & (a - L < n)
    delta[valid_am] += h[a - L[valid_am]]
    
    # For b (0→1):
    # +(1-h[b+L]) where 0≤b+L<n
    valid_bp = (b + L >= 0) & (b + L < n)
    delta[valid_bp] += (1 - h[b + L[valid_bp]])
    
    # -h[b-L] where 0≤b-L<n
    valid_bm = (b - L >= 0) & (b - L < n)
    delta[valid_bm] -= h[b - L[valid_bm]]
    
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(shifted)
    return e / np.sum(e)

def obj_and_grad(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    val = smooth_max_val(corr, beta) * dx
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    n2m1 = 2 * n - 1
    for k in range(n2m1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[:n - L] += w * (1.0 - x[L:])
            grad[L:] -= w * x[:n - L]
        else:
            Lp = -L
            grad[Lp:] += w * (1.0 - x[:n - Lp])
            grad[:n - Lp] -= w * x[Lp:]
    return val, dx * grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def lbfgs_opt(h0, n, beta, max_iter):
    def fun(x):
        v, _ = obj_and_grad(x, beta)
        return v
    def jac(x):
        _, g = obj_and_grad(x, beta)
        return g
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'disp': False})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def compute_flip_delta(a, b, h, n):
    L = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    valid_ap = (a + L >= 0) & (a + L < n)
    delta[valid_ap] -= (1.0 - h[a + L[valid_ap]])
    valid_am = (a - L >= 0) & (a - L < n)
    delta[valid_am] += h[a - L[valid_am]]
    valid_bp = (b + L >= 0) & (b + L < n)
    delta[valid_bp] += (1.0 - h[b + L[valid_bp]])
    valid_bm = (b - L >= 0) & (b - L < n)
    delta[valid_bm] -= h[b - L[valid_bm]]
    return delta

def make_fourier_init(n, rng, n_modes=6):
    h = np.ones(n) * 0.5
    for m in range(1, n_modes + 1):
        amp = rng.uniform(0.05, 0.25) / (m ** 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * m * np.arange(n) / n + phase)
    h = np.clip(h, 0, 1)

def make_segmented_init(n, rng, n_seg=8):

    seg_size = n // n_seg
    for s in range(n_seg):
        val = rng.choice([0.0, 1.0])
        h[s * seg_size:(s + 1) * seg_size] = val
    h[-1 * seg_size:] = rng.choice([0.0, 1.0])
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B optimization at multiple resolutions
    # Focus on the parent resolution and slightly higher
    res_list = [best_n, best_n + 50, best_n + 100, best_n + 200]
    res_list = sorted(set(r for r in res_list if 128 <= r <= 600))

    for n in res_list:
        if remaining() < 50:
            break

        # Multiple inits
        inits = []
        if parent_h is not None:
            inits.append(interp_h(parent_h, n))
        inits.append(interp_h(best_h, n))
        inits.append(make_fourier_init(n, rng))
        inits.append(make_segmented_init(n, rng, n_seg=10))

        for h0 in inits:
            if remaining() < 30:
                break
            h0 = project_feasible(h0, n)
            cur = h0.copy()
            # Increasing beta curriculum
            for beta in [40, 100, 300, 800]:
                if remaining() < 15:
                    break
                iters = 600 if n <= 350 else 350
                try:
                    h_new, c5_new = lbfgs_opt(cur, n, beta, iters)
                    cur = h_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n
                except Exception:
                    break
            # Final high-beta polish
            if remaining() > 20 and best_n == n:
                try:
                    h_pol, c5_pol = lbfgs_opt(cur, n, 2000, 500)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol.copy()
                except Exception:
                    pass

    # Phase 2: Binary rounding + targeted local search
    # Try at best_n and a couple other sizes
    for n_try in [best_n, best_n + 50, best_n - 50]:
        if remaining() < 40:
            break
        if n_try < 100:
            continue

        h_cont = interp_h(best_h, n_try)
        h_bin = (h_cont >= 0.5).astype(float)
        n_ones = int(h_bin.sum())
        n_want = n_try // 2
        if n_ones > n_want:
            ones_idx = np.where(h_bin == 1)[0]
            scores = h_cont[ones_idx]
            order = np.argsort(scores)[:n_ones - n_want]
            h_bin[ones_idx[order]] = 0.0
        elif n_ones < n_want:
            zeros_idx = np.where(h_bin == 0)[0]
            scores = h_cont[zeros_idx]
            order = np.argsort(-scores)[:n_want - n_ones]
            h_bin[zeros_idx[order]] = 1.0

        h_bin, c5_bin = binary_targeted_search(h_bin, n_try, 2000, rng, start_time, budget_s)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n_try

    # Phase 3: Continue gradient optimization from best (possibly binary) solution
    for n in [best_n, best_n + 100, best_n + 200]:
        if remaining() < 40:
            break
        if n > 800:
            continue
        h0 = interp_h(best_h, n)
        # Binary-ish init
        h0_bin = (h0 > 0.5).astype(float)
        no = int(h0_bin.sum())
        nw = n // 2
        if no > nw:
            oi = np.where(h0_bin == 1)[0]
            h0_bin[oi[:no - nw]] = 0
        elif no < nw:
            zi = np.where(h0_bin == 0)[0]
            h0_bin[zi[:nw - no]] = 1
        h0 = project_feasible(h0_bin, n)

        for beta in [100, 500, 1500, 3000]:
            if remaining() < 15:
                break
            try:
                h_new, c5_new = lbfgs_opt(h0, n, beta, 500)
                h0 = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n
            except Exception:
                break

    # Phase 4: More binary local search rounds on best
    if remaining() > 30:
        h_work = (best_h >= 0.5).astype(float)
        no = int(h_work.sum())
        nw = best_n // 2
        if no > nw:
            oi = np.where(h_work == 1)[0]
            h_work[oi[:no - nw]] = 0
        elif no < nw:
            zi = np.where(h_work == 0)[0]
            h_work[zi[:nw - no]] = 1
        h_work, c5_work = binary_targeted_search(h_work, best_n, 5000, rng, start_time, budget_s)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    # Phase 5: Final fine-grained gradient polish
    if remaining() > 25:
        n_final = best_n
        h_start = best_h.copy()
        for beta in [3000, 8000]:
            if remaining() < 10:
                break
            try:
                h_new, c5_new = lbfgs_opt(h_start, n_final, beta, 800)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_start = h_new
            except Exception:
                break

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
