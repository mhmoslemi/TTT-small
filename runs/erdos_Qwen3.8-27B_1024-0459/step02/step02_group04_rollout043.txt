```python
def halton_binary(n, rng):
    """Create a binary sequence using Halton-like placement."""
    h = np.zeros(n)
    k = n // 2
    # Use a low-discrepancy sequence to choose positions
    # Simple approach: use the van der Corput sequence
    positions = []
    for i in range(1, k+1):
        # Van der Corput in base 2
        x = 0.0
        f = 0.5
        while i > 0:
            x += (i % 2) * f
            i //= 2
            f *= 0.5
        positions.append(int(x * n))
    positions = sorted(set(positions))
    # If we don't have enough unique positions, fill in randomly
    if len(positions) < k:
        remaining = [p for p in range(n) if p not in set(positions)]
        rng.shuffle(remaining)
        positions.extend(remaining[:k - len(positions)])
    h[positions[:k]] = 1.0
    return h
```

```python
def project_feasible(h, n=None):
```

```python
    h = x
```

    shifted = np.clip(shifted, -500.0, 0.0)
    s = float(np.sum(np.exp(shifted)))
    if s <= 0.0:
        return maxv
    return maxv + np.log(s) / beta

def smooth_grad(x, beta):
    h = x
    n = len(h)
    dx = 2.0 / float(n)
    vals = lag_values_full(h)
    maxv = float(np.max(vals))
    shifted = beta * (vals - maxv)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    total_w = float(np.sum(weights))
    if total_w <= 1e-30:
        return np.ones(n) / n
    g = np.zeros(n, dtype=float)
    center = n - 1
    for p in range(2 * n - 1):
        w = weights[p] / total_w
        if w < 1e-12:
            continue
        L = p - center
        if L == 0:
            g += w * dx * (1.0 - 2.0 * h)
        elif L > 0:
            m = n - L
            g[L:] += w * dx * (1.0 - h[:m])
            g[:m] += -w * dx * h[L:]
        else:
            Lp = -L
            m = n - Lp
            g[:m] += w * dx * (1.0 - h[Lp:])
            g[Lp:] += -w * dx * h[:m]
    return g
