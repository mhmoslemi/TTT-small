    g = np.zeros(n, dtype=float)

        k = int(t) - center
        g += w[i] * corr_grad_lag(h, k)
    return g

def corr_grad_lag(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g


def active_gradient(h, top_frac, min_k):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    top_idx = np.argsort(corr)[-k_count:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    denom = max(1e-10, dv.max() + 1e-10)
    w = np.exp(dv / denom)
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += w[i] * corr_grad_lag(h, k)
    return g

            h_cur, c_cur = swap_polish(h_cur, time.perf_counter() + rem * 0.95,
                                       max_iter=2000, top_frac=0.03, min_k=10,
                                       n_pos=8, n_neg=8, seed=int(seed % 10000) + 2)
        return project_density(h_cur, 0.5 * n_pts), eval_c5(project_density(h_cur, 0.5 * n_pts))

    # Resolutions to explore with DE
    n_list = [60, 90, 120, 150, 180, 240, 300, 360, 450]
    base_n = best_n if best_n > 1 else 90
    if base_n not in n_list:
        n_list.append(base_n)
    n_list = sorted(set(n_list))

    # Time weights: smaller resolutions get more relative time (faster per eval)
    total_avail = deadline - time.perf_counter()
    if total_avail < 60:
        # Very tight: just polish what we have
        h_fin, c_fin = _polish_at(best_h, best_n, total_avail - 5)
        if c_fin < best_c:
            best_h, best_c = h_fin, c_fin
        final_h = project_density(best_h, 0.5 * int(best_h.size))
        return final_h, float(eval_c5(final_h)), int(final_h.size)

    # Allocate: 60% to DE, 30% to polishing best, 10% to parent polishing
    de_time = total_avail * 0.55
    polish_time = total_avail * 0.30
    parent_time = total_avail * 0.15

    # DE phase
    de_deadline = start_time + (deadline - start_time) * 0.55
    for ni in n_list:
        rem = de_deadline - time.perf_counter()
        if rem < 15:
            break
        target = 0.5 * ni
        bounds = [(0.0, 1.0)] * ni

        if ni <= 90:
            popsize, maxiter, n_runs = 40, 250, 2
            mutation, recomb = (0.5, 1.0), 0.7
        elif ni <= 180:
            popsize, maxiter, n_runs = 30, 180, 2
            mutation, recomb = (0.5, 1.0), 0.7
        elif ni <= 300:
            popsize, maxiter, n_runs = 25, 120, 1
            mutation, recomb = (0.4, 0.9), 0.65
        else:
            popsize, maxiter, n_runs = 20, 100, 1
            mutation, recomb = (0.4, 0.8), 0.6

        per_run = min(rem / max(1, n_runs), de_deadline - time.perf_counter() - 5.0)

        for run_i in range(n_runs):
            rem2 = de_deadline - time.perf_counter()
            if rem2 < 10:
                break
            _DE_DEADLINE = time.perf_counter() + min(per_run, rem2 - 3.0)
            run_seed = int(((seed * 7 + ni * 13 + run_i) % 2147483647) + 1)

            try:
                init_pop = make_de_initial(ni, popsize, parent, rng)
                de_res = differential_evolution(
                    de_objective,
                    bounds,
                    args=(ni, target),
                    strategy='best1bin' if run_i == 0 else 'rand1bin',
                    maxiter=maxiter,
                    popsize=popsize,
                    tol=1e-7,
                    mutation=mutation,
                    recombination=recomb,
                    seed=run_seed,
                    init=init_pop,
                    polish=False,
                    updating='immediate',
                    workers=1,
                    callback=de_time_callback,
                )
                if de_res.x is not None and np.all(np.isfinite(de_res.x)):
                    cand = project_density(np.clip(de_res.x, 0, 1), target)
                    c = eval_c5(cand)
                    if c < best_c:
                        best_h = cand.copy()
                        best_c = c
                        best_n = ni
            except Exception:
                pass

    # Polish the overall best
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 20:
        h_fin, c_fin = _polish_at(best_h, int(best_h.size), min(polish_time, rem - 10))
        if c_fin < best_c:
            best_h = h_fin.copy()
            best_c = c_fin

    # Polish parent at its own resolution
    rem = deadline - time.perf_counter()
    if parent is not None and rem > 15:
        np_ = int(parent.size)
        h_p = project_density(parent, 0.5 * np_)
        c_p = eval_c5(h_p)
        if c_p < best_c + 0.002:
            h_pf, c_pf = _polish_at(h_p, np_, min(parent_time, rem - 5))
            if c_pf < best_c:
                best_h = h_pf.copy()
                best_c = c_pf
                best_n = np_

    # Final safety check and return
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, eval_c5(fallback), 90

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = float(eval_c5(final_h))
    return final_h, final_c, int(final_h.size)
