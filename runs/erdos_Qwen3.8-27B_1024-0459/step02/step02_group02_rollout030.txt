import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def make_binary_with_sum(h, n, rng):
    """Round to binary and adjust to have exactly n/2 ones."""
    h = np.round(np.array(h, dtype=float)).copy()
    h = np.clip(h, 0, 1)
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        excess = ones - target
        one_positions = np.where(h == 1)[0]
        to_flip = rng.choice(one_positions, excess, replace=False)
        h[to_flip] = 0
    elif ones < target:
        deficit = target - ones
        zero_positions = np.where(h == 0)[0]
        to_flip = rng.choice(zero_positions, deficit, replace=False)
        h[to_flip] = 1
    return h

def thue_morse_binary(n, shift=0):
    h = np.zeros(n)
    for i in range(n):
        idx = (i + shift) % n
        h[i] = 1.0 if bin(idx).count('1') % 2 == 0 else 0.0
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def two_swap_sa(h_init, n, rng, t_start=1.0, t_end=0.01, max_iter=1000000, 
                time_limit=60.0, anneal_steps=None):
    """2-swap simulated annealing for binary h with n/2 ones."""
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    one_idx = list(np.where(h == 1)[0])
    zero_idx = list(np.where(h == 0)[0])
    
    if len(one_idx) < 2 or len(zero_idx) < 2:
        return best_h, best_c5
    
    start_t = time.time()
    if anneal_steps is None:
        anneal_steps = max_iter
    
    no_improve = 0
    stall_limit = max(1000, max_iter // 20)
    
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        if no_improve > stall_limit:
            break
        
        # Temperature
        frac = it / max(anneal_steps - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        
        # Pick two 1s and two 0s
        i1 = one_idx[rng.randint(len(one_idx))]
        i2 = one_idx[rng.randint(len(one_idx))]
        if i1 == i2:
            continue
        j1 = zero_idx[rng.randint(len(zero_idx))]
        j2 = zero_idx[rng.randint(len(zero_idx))]
        if j1 == j2 or j1 in (i1, i2) or j2 in (i1, i2):
            continue
        
        # Do the swap
        h[i1] = 0; h[i2] = 0
        h[j1] = 1; h[j2] = 1
        
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            if delta < 0:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:

    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def _smooth_obj(x, beta):
    hx = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hx, 1.0 - hx, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500.0, 50.0)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def _sum_con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _sum_con_jac(x):
    return np.ones(_N_OPT)

def two_swap_sa(h_init, n, rng, time_limit=60.0, t_start=0.3, t_end=0.001,
                cooling=0.9998, stall_limit=80000, p_one_swap=0.25):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    one_idx = list(np.where(h == 1)[0])
    zero_idx = list(np.where(h == 0)[0])
    n1 = len(one_idx)
    n0 = len(zero_idx)
    if n1 < 1 or n0 < 1:
        return best_h, best_c5
    start_t = time.time()
    T = t_start
    no_improve = 0
    exp_clip = 50.0
    while time.time() - start_t < time_limit:
        if no_improve > stall_limit:
            break
        T = max(T * cooling, t_end)
        if p_one_swap > 0 and rng.random() < p_one_swap and n1 >= 1 and n0 >= 1:
            # 1-swap
            i = one_idx[rng.randint(n1)]
            j = zero_idx[rng.randint(n0)]
            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)
            delta = c5_new - best_c5
            accept = delta < 0 or (delta / max(T, 1e-15) < exp_clip and rng.random() < np.exp(-delta / max(T, 1e-15)))
            if accept:
                if delta < 0:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                one_idx.remove(i)
                zero_idx.remove(j)
                one_idx.append(j)
                zero_idx.append(i)
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        else:
            # 2-swap
            if n1 < 2 or n0 < 2:
                break
            i1 = one_idx[rng.randint(n1)]
            i2 = one_idx[rng.randint(n1)]
            tries = 0
            while i2 == i1 and tries < 20:
                i2 = one_idx[rng.randint(n1)]
                tries += 1
            if i2 == i1:
                continue
            j1 = zero_idx[rng.randint(n0)]
            j2 = zero_idx[rng.randint(n0)]
            tries = 0
            while j2 == j1 and tries < 20:
                j2 = zero_idx[rng.randint(n0)]
                tries += 1
            if j2 == j1:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            c5_new = compute_c5(h)
            delta = c5_new - best_c5
            accept = delta < 0 or (delta / max(T, 1e-15) < exp_clip and rng.random() < np.exp(-delta / max(T, 1e-15)))
            if accept:
                if delta < 0:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                one_idx.remove(i1)
                one_idx.remove(i2)
                zero_idx.remove(j1)
                zero_idx.remove(j2)
                one_idx.append(j1)
                one_idx.append(j2)
                zero_idx.append(i1)
                zero_idx.append(i2)
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
                no_improve += 1
    return best_h, best_c5

def targeted_two_swap(h_init, n, rng, time_limit=60.0):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    one_idx = list(np.where(h == 1)[0])
    zero_idx = list(np.where(h == 0)[0])
    n1 = n // 2
    n0 = n // 2
    start_t = time.time()
    no_improve = 0
    stall_limit = 40000
    while time.time() - start_t < time_limit:
        if no_improve > stall_limit:
            break
        corr = compute_corr(h)
        worst_k = int(np.argmax(corr))
        worst_shift = worst_k - (n - 1)
        # Find positions contributing to worst shift
        bad = []
        if worst_shift >= 0:
            s = worst_shift
            if s == 0:
                s = 1
            for i in range(n - s):
                if h[i] == 1.0 and h[i + s] == 0.0:
                    bad.append(i)
        else:
            s = -worst_shift
            if s == 0:
                s = 1
            for i in range(s, n):
                if h[i] == 1.0 and h[i - s] == 0.0:
                    bad.append(i)
        if len(bad) >= 2 and n0 >= 2:
            i1 = bad[rng.randint(len(bad))]
            i2 = bad[rng.randint(len(bad))]
            tr = 0
            while i2 == i1 and tr < 20:
                i2 = bad[rng.randint(len(bad))]
                tr += 1
            if i2 == i1:
                continue
            j1 = zero_idx[rng.randint(n0)]
            j2 = zero_idx[rng.randint(n0)]
            tr = 0
            while j2 == j1 and tr < 20:
                j2 = zero_idx[rng.randint(n0)]
                tr += 1
            if j2 == j1:
                continue
        elif n1 >= 2 and n0 >= 2:
            i1 = one_idx[rng.randint(n1)]
            i2 = one_idx[rng.randint(n1)]
            tr = 0
            while i2 == i1 and tr < 20:
                i2 = one_idx[rng.randint(n1)]
                tr += 1
            if i2 == i1:
                continue
            j1 = zero_idx[rng.randint(n0)]
            j2 = zero_idx[rng.randint(n0)]
            tr = 0
            while j2 == j1 and tr < 20:
                j2 = zero_idx[rng.randint(n0)]
                tr += 1
            if j2 == j1:
                continue
        else:
            break
        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
            idx11 = np.where(h == 1.0)[0].tolist()
            idx00 = np.where(h == 0.0)[0].tolist()
            one_idx = idx11
            zero_idx = idx00
            n1 = len(one_idx)
            n0 = len(zero_idx)
        else:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
            no_improve += 1
    return best_h, best_c5

def smooth_max_slsqp(h_init, n, time_limit=60.0):
    global _N_OPT
    _N_OPT = n
    h0 = np.array(h_init, dtype=float).copy()
    h0 = np.clip(h0, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h0)
    if abs(diff) > 1e-10:
        mask = (h0 > 1e-6) & (h0 < 1.0 - 1e-6)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h0[mask] += diff / cnt
        h0 = np.clip(h0, 0.0, 1.0)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    start_t = time.time()
    for beta in [15.0, 40.0, 100.0, 250.0]:
        if time.time() - start_t > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h0, args=(beta,), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': 300, 'ftol': 1e-13})
            if res.fun < 1e10:
                candidate = np.clip(res.x, 0.0, 1.0)
                diff2 = n / 2.0 - np.sum(candidate)
                if abs(diff2) > 1e-8:
                    mask = (candidate > 1e-6) & (candidate < 1.0 - 1e-6)
                    cnt = np.count_nonzero(mask)
                    if cnt > 0:
                        candidate[mask] += diff2 / cnt
                    candidate = np.clip(candidate, 0.0, 1.0)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = candidate.copy()
                h0 = candidate.copy()
        except Exception:
            pass
    return best_h, best_c5

def multi_start_sa(h_pool, n, rng, time_limit=120.0):
    """Run SA from multiple starting points, sharing time budget."""
    best_h = h_pool[0].copy()
    best_c5 = compute_c5(best_h)
    per_init = max(10.0, time_limit / len(h_pool))
    for h0 in h_pool:
        if time.time() > 0:  # placeholder
            pass
        h_sa, c5_sa = two_swap_sa(h0, n, rng, time_limit=per_init,
                                   t_start=0.2, t_end=0.0005, cooling=0.9998,
                                   stall_limit=50000, p_one_swap=0.2)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        per_init = max(10.0, (time_limit - (time.time() - _msa_start)) / max(len(h_pool) - _msa_idx - 1, 1))
    return best_h, best_c5

_msa_start = 0
_msa_idx = -1

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent solution
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None

    best_h = None
    best_c5 = float('inf')
    best_n = 256

    # Primary focus: n=256 (parent resolution)
    n = 256

    # Generate diverse initializations
    inits = []
    if h_parent is not None and len(h_parent) > 0:
        h_interp = interp_to_n(h_parent, n)
        inits.append(make_binary(h_interp, n, rng))
    inits.append(thue_morse(n, 0))
    inits.append(thue_morse(n, 1))
    inits.append(thue_morse(n, 3))
    # Random inits
    for _ in range(3):
        h_r = (rng.rand(n) > 0.5).astype(float)
        inits.append(make_binary(h_r, n, rng))
    # Antithetic-style (but NOT shift n/2 which is bad): use shift n/4
    h_as = np.zeros(n)
    h_as[:n//4] = thue_morse(n//4, 0)
    h_as[n//4:n//2] = 1.0 - h_as[:n//4]
    h_as[n//2:3*n//4] = h_as[:n//4]
    h_as[3*n//4:] = 1.0 - h_as[:n//4]
    inits.append(make_binary(h_as, n, rng))

    # Phase 1: SA from each init (main optimization)
    time_per_init = max(20.0, remaining() * 0.45 / len(inits))
    for i, h0 in enumerate(inits):
        if remaining() < 100:
            break
        h_sa, c5_sa = two_swap_sa(h0, n, rng, time_limit=time_per_init,
                                   t_start=0.4, t_end=0.001, cooling=0.9997,
                                   stall_limit=60000, p_one_swap=0.2)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n

    # Phase 2: Targeted 2-swap on best
    if remaining() > 80 and best_h is not None:
        h_t, c5_t = targeted_two_swap(best_h, n, rng, time_limit=min(60, remaining() * 0.12))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 3: SA with lower temperature (polishing)
    if remaining() > 100 and best_h is not None:
        h_p, c5_p = two_swap_sa(best_h, n, rng, time_limit=min(80, remaining() * 0.15),
                                 t_start=0.05, t_end=0.0001, cooling=0.9999,
                                 stall_limit=100000, p_one_swap=0.3)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()

    # Phase 4: SLSQP continuous optimization (might find better non-binary)
    if remaining() > 80 and best_h is not None:
        h_sl, c5_sl = smooth_max_slsqp(best_h, n, time_limit=min(50, remaining() * 0.12))
        # Check continuous solution
        h_sl_feas = np.clip(h_sl, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_sl_feas)
        if abs(diff) > 1e-8:
            mask = (h_sl_feas > 1e-6) & (h_sl_feas < 1.0 - 1e-6)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_sl_feas[mask] += diff / cnt
            h_sl_feas = np.clip(h_sl_feas, 0.0, 1.0)
        c5_sl_feas = compute_c5(h_sl_feas)
        if c5_sl_feas < best_c5:
            best_c5 = c5_sl_feas
            best_h = h_sl_feas.copy()
        # Also check rounded version
        h_sl_bin = make_binary(h_sl, n, rng)
        c5_sl_bin = compute_c5(h_sl_bin)
        if c5_sl_bin < best_c5:
            best_c5 = c5_sl_bin
            best_h = h_sl_bin.copy()

    # Phase 5: More targeted + SA rounds
    if remaining() > 60 and best_h is not None:
        h_t2, c5_t2 = targeted_two_swap(best_h, n, rng, time_limit=min(40, remaining() * 0.1))
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()

    if remaining() > 60 and best_h is not None:
        h_p2, c5_p2 = two_swap_sa(best_h, n, rng, time_limit=min(50, remaining() * 0.15),
                                   t_start=0.03, t_end=0.00005, cooling=0.99995,
                                   stall_limit=150000, p_one_swap=0.3)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2.copy()

    # Phase 6: Try other n values
    for n_alt in [320, 192, 384]:
        if remaining() < 80:
            break
        inits_alt = []
        if h_parent is not None and len(h_parent) > 0:
            inits_alt.append(make_binary(interp_to_n(h_parent, n_alt), n_alt, rng))
        inits_alt.append(thue_morse(n_alt, 0))
        inits_alt.append(make_binary((rng.rand(n_alt) > 0.5).astype(float), n_alt, rng))

        t_per = max(15.0, remaining() * 0.15 / len(inits_alt))
        for h0a in inits_alt:
            if remaining() < 80:
                break
            h_sa_a, c5_sa_a = two_swap_sa(h0a, n_alt, rng, time_limit=t_per,
                                           t_start=0.3, t_end=0.001, cooling=0.9997,
                                           stall_limit=40000, p_one_swap=0.2)
            if c5_sa_a < best_c5:
                best_c5 = c5_sa_a
                best_h = h_sa_a.copy()
                best_n = n_alt

    # Phase 7: Final intensive SA on the best
    if remaining() > 50 and best_h is not None:
        # Ensure binary for best search
        if np.any((best_h > 1e-10) & (best_h < 1.0 - 1e-10)):
            h_bin = make_binary(best_h, best_n, rng)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5 + 0.001:
                best_h = h_bin
                best_c5 = min(best_c5, c5_bin)

        h_final, c5_final = two_swap_sa(best_h, best_n, rng,
                                         time_limit=min(100, remaining() - 15),
                                         t_start=0.08, t_end=0.0002,
                                         cooling=0.9999, stall_limit=120000,
                                         p_one_swap=0.25)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final: one more targeted pass
    if remaining() > 30 and best_h is not None:
        h_tf, c5_tf = targeted_two_swap(best_h, best_n, rng, time_limit=min(25, remaining() - 10))
        if c5_tf < best_c5:
            best_c5 = c5_tf
            best_h = h_tf.copy()

    if best_h is None:
        best_h = thue_morse(256)
        best_c5 = compute_c5(best_h)
        best_n = 256

    # Final evaluation
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
