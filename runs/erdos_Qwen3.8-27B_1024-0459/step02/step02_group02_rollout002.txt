def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(20):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-13
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break

        else:
            mask = h > 1e-13
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    if h_src.size < 2:
        return np.full(n_dst, 0.5, dtype=float)
    x_old = np.linspace(0.0, 2.0, h_src.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))


def thue_morse_init(n):

def smooth_value(h, beta):
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = beta * (corr - cmax)
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    if s <= 0.0:
        return cmax, corr, cmax
    phi = cmax + math.log(s) / float(beta)
    return phi, corr, cmax


def smooth_grad_from_weights(h, weights):
    n = len(h)
    grad = np.zeros(n, dtype=float)
    caches = get_shift_arrays(n)
    for (ii, jj), w in zip(caches, weights):
        if w <= 1e-15:
            continue
        hii = h[ii]
        hjj = h[jj]
        grad[ii] += w * (1.0 - hjj)
        grad[jj] -= w * hii
    return grad


def continuous_descent_beta(h, n, beta, iters, deadline, base_step, rng):

    best_h = h.copy()
    best_c = compute_c5(h)
    best_phi = float("inf")
    no_improve = 0
    reject_jumps = 0
    step = float(base_step)

    for _ in range(int(iters)):
        if time.time() > deadline:
            break

        phi, corr, cmax = smooth_value(h, beta)
        c = cmax * (2.0 / n)
        if c < best_c - 1e-14:

            best_h = h.copy()
        if phi < best_phi - 1e-12:

        else:
            no_improve += 1
        if no_improve > 300:
            break

        shifted = beta * (corr - cmax)
        exps = np.exp(shifted)
        sw = float(np.sum(exps))
        if sw <= 0.0:
            break
        weights = exps / sw
        grad = smooth_grad_from_weights(h, weights)
        gnorm2 = float(np.dot(grad, grad))
        if gnorm2 < 1e-22:
            break
        normg = math.sqrt(gnorm2)

        if step <= 1e-12 or not math.isfinite(step):
            step = 0.05 / max(1.0, 0.05 * normg)

        attempts = []
        s0 = step
        if s0 < 1e-4:

        attempts.append(min(1.0, 2.0 * s0))
        attempts.append(s0)

            attempts.append(ss)

            break

def _polish_sum_fun(x):
    return float(np.sum(x) - 0.5 * _POLISH_N)


def _polish_sum_jac(x):

def _polish_obj(x):
    h = np.asarray(x, dtype=float)
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = _POLISH_BETA * (corr - cmax)
    s = float(np.sum(np.exp(shifted)))
    if s <= 0.0:
        return cmax
    return cmax + math.log(s) / _POLISH_BETA


def slsqp_polish(h, n, deadline, beta):
    global _POLISH_N, _POLISH_BETA
    if n > 600:
        return h, compute_c5(h)
    _POLISH_N = int(n)
    _POLISH_BETA = float(beta)
    x0 = make_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    con = {"type": "eq", "fun": _polish_sum_fun, "jac": _polish_sum_jac}
    try:
        max_it = max(100, int((deadline - time.time()) * 8))
        res = minimize(
            _polish_obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=con,
            options={"maxiter": max_it, "ftol": 1e-14},
        )
        if res.fun < 1e9:
            cand = make_feasible(res.x, n)
            return cand, compute_c5(cand)
    except Exception:
        pass
    return h, compute_c5(h)

            0.03,
        )
        if c_opt < best_c - 1e-15:
            best_c = c_opt
            best_h = h_opt.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(len(best_h)))
