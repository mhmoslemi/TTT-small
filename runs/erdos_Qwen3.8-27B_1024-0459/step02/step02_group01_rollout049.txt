import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:n_ones - n_want]:
            h_bin[ones[idx]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:n_want - n_ones]:
            h_bin[zeros[idx]] = 1.0
    return h_bin

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / beta
    return smax * (2.0 / n)

def raw_max_obj(x, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * (2.0 / n)

def topk_obj(x, n, k):
    corr = np.correlate(x, 1.0 - x, mode='full')
    top = np.sort(corr)[-min(k, len(corr)):]
    return np.mean(top) * (2.0 / n)

def eq_con(x, n):
    return np.sum(x) - n / 2.0

def binary_sa(h_init, n, t0, t1, steps, rng, time_limit, best_ref):
    h = project_binary(h_init.copy(), n)
    dx = 2.0 / n
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    best_h = h.copy()
    best_ref[0] = best_c5
    t_start = time.time()

    for step in range(steps):
        if time.time() - t_start > time_limit:
            break
        temp = t0 * (t1 / t0) ** (step / max(1, steps))
        h_save = h.copy()

        r = rng.rand()
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]

        if r < 0.65:
            i = ones_pos[rng.randint(len(ones_pos))]
            j = zeros_pos[rng.randint(len(zeros_pos))]
            h[i] = 0.0
            h[j] = 1.0
        elif r < 0.85:
            if len(ones_pos) >= 2 and len(zeros_pos) >= 2:
                idx1 = rng.randint(len(ones_pos))
                idx2 = rng.randint(len(ones_pos))
                if idx1 == idx2:
                    idx2 = (idx2 + 1) % len(ones_pos)
                idx3 = rng.randint(len(zeros_pos))
                idx4 = rng.randint(len(zeros_pos))
                if idx3 == idx4:
                    idx4 = (idx4 + 1) % len(zeros_pos)
                i1, i2 = ones_pos[idx1], ones_pos[idx2]
                j1, j2 = zeros_pos[idx3], zeros_pos[idx4]
                h[i1] = h[i2] = 0.0
                h[j1] = h[j2] = 1.0
            else:
                continue
        elif r < 0.95:
            # Targeted: find worst lag and try to fix it
            corr = compute_corr(h)
            worst_lag = np.argmax(corr)
            lag = worst_lag - (n - 1)
            if lag > 0:
                pairs = [(i, i + lag) for i in range(n - lag) if h[i] == 1 and h[i + lag] == 0]
                if len(pairs) > 0:
                    p = pairs[rng.randint(len(pairs))]
                    h[p[0]] = 0.0
                    # Find a zero to flip to one (away from this lag)
                    zcands = [z for z in zeros_pos if abs(z - p[0]) > lag and abs(z - p[1]) > lag]
                    if len(zcands) > 0:
                        h[zcands[rng.randint(len(zcands))]] = 1.0
                    else:
                        z = zeros_pos[rng.randint(len(zeros_pos))]
                        if z != p[0]:
                            h[z] = 1.0
            elif lag < 0:
                o = -lag
                pairs = [(i, i - o) for i in range(o, n) if h[i] == 1 and h[i - o] == 0]
                if len(pairs) > 0:
                    p = pairs[rng.randint(len(pairs))]
                    h[p[0]] = 0.0
                    zcands = [z for z in zeros_pos if abs(z - p[0]) > o and abs(z - p[1]) > o]
                    if len(zcands) > 0:
                        h[zcands[rng.randint(len(zcands))]] = 1.0
                    else:
                        z = zeros_pos[rng.randint(len(zeros_pos))]
                        if z != p[0]:
                            h[z] = 1.0
            else:
                continue
        else:
            # Block flip
            blen = rng.randint(2, max(3, n // 20))
            start = rng.randint(0, n - blen)
            h[start:start + blen] = 1.0 - h[start:start + blen]

        new_c5 = compute_c5(h)
        if new_c5 < curr_c5 or rng.rand() < np.exp(-(new_c5 - curr_c5) / max(temp, 1e-15)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                best_ref[0] = best_c5
        else:
            h = h_save

    return best_h, best_c5

def continuous_optimize(h_init, n, time_limit, beta_seq):
    h_cur = project_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()

    for beta in beta_seq:
        if time.time() - t_start > time_limit or time.time() - t_start > 0:
            remaining = time_limit - (time.time() - t_start)
            if remaining < 5:
                break
        else:
            remaining = time_limit

        def obj_wrapper(x):
            return soft_max_obj(x, n, beta)

        def con_wrapper(x):
            return eq_con(x, n)

        cons = [{'type': 'eq', 'fun': con_wrapper}]
        try:
            res = minimize(obj_wrapper, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass
        if time.time() - t_start > time_limit:
            break

    # Raw max polish
    remaining = time_limit - (time.time() - t_start)
    if remaining > 10:
        def obj_raw_w(x):
            return raw_max_obj(x, n)

        def con_w2(x):
            return eq_con(x, n)

        cons2 = [{'type': 'eq', 'fun': con_w2}]
        try:
            res = minimize(obj_raw_w, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons2,
                           options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass

    return h_cur

def greedy_binary_improve(h, n, rng, n_iter, time_limit):
    h = project_binary(h.copy(), n)
    dx = 2.0 / n
    curr_c5 = compute_c5(h)
    t_start = time.time()
    improved = True
    iteration = 0

    while improved and iteration < n_iter and time.time() - t_start < time_limit:
        improved = False
        iteration += 1
        corr = compute_corr(h)
        max_val = np.max(corr)
        active_lags = np.where(corr >= max_val - 1e-10)[0]

        best_delta = 1e-10
        best_move = None

        # Try all single swaps (for n=192, this is 96*96 = 9216 combinations)
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]

        # Sample a subset for speed if too many
        max_evals = 4096
        n_combos = len(ones_pos) * len(zeros_pos)
        if n_combos > max_evals:

            pairs = rng.choice(n_combos, max_evals, replace=False)

            pairs = np.arange(n_combos)

            h[j] = 1.0

            h[i] = 1.0

        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            curr_c5 -= best_delta
