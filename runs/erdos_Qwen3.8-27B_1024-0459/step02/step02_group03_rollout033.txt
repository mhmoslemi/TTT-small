import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_grad(h):
    """Compute C5 and its subgradient (at the max shift)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Subgradient: pick the max shift
    max_idx = np.argmax(corr)
    L = max_idx - (n - 1)
    grad = np.zeros(n)
    if L == 0:
        grad = (1.0 - 2.0 * h)
    elif L > 0:
        grad[L:] += (1.0 - h[:n-L])
        grad[:n-L] -= h[L:]
    else:
        Lp = -L
        grad[:n-Lp] += (1.0 - h[Lp:])
        grad[Lp:] -= h[:n-Lp]
    return max_val * dx, dx * grad

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    return weights

def c5_smooth_obj_and_grad(h, beta, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sm = smooth_max(corr, beta)
    weights = smooth_max_grad(corr, beta)
    
    # Gradient of smooth max w.r.t. h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n-Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n-Lp]
    
    # Add penalty gradient
    sum_diff = np.sum(h) - n / 2.0
    grad += 2.0 * mu * sum_diff
    
    obj = sm * dx + mu * sum_diff * sum_diff
    return obj, dx * grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def _smooth_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s < 1e-300:
        exps = np.zeros_like(corr)
        exps[np.argmax(corr)] = 1.0
        return exps
    return exps / s

def _c5_obj_grad_impl(x, beta, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    sm = _smooth_max(corr, beta)
    weights = _smooth_weights(corr, beta)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[:n-L] += w * (1.0 - x[L:])

    sum_diff = np.sum(x) - n / 2.0
    grad += 2.0 * mu * sum_diff
    
    obj = sm * dx + mu * sum_diff * sum_diff
    return obj, dx * grad

def _grad_func(x):

    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    # Iterative correction
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            # Distribute equally
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run_opt_lbfgs(h0, n, beta, mu, max_iter=500):
    _state['beta'] = beta
    _state['mu'] = mu
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_func, h0.copy(), jac=_grad_func, method='L-BFGS-B',
                   bounds=bounds,

                            'maxls': 100, 'maxcor': 20})
    h_out = project_feasible(res.x, n)
    return h_out

def subgradient_polish(h, n, n_steps=200):
    dx = 2.0 / n
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.02 * n  # initial step
    
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        L = max_idx - (n - 1)
        
        # Subgradient at the max shift
        sg = np.zeros(n)
        if L == 0:
            sg = (1.0 - 2.0 * h)
        elif L > 0:

            sg[L:] = -h[:n-L]
        else:
            Lp = -L
            sg[Lp:] = (1.0 - h[:n-Lp])
            sg[:n-Lp] = -h[Lp:]
        
        h_new = h - alpha * dx * sg
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        
        h = h_new
        alpha *= 0.995  # decreasing step
    
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def _smooth_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        return w
    return exps / s

def _c5_obj_grad_impl(x, beta, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    sm = _smooth_max(corr, beta)
    weights = _smooth_weights(corr, beta)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * x)
        elif L > 0:
            grad[:n - L] += w * (1.0 - x[L:])
            grad[L:] += -w * x[:n - L]
        else:
            Lp = -L
            grad[Lp:] += w * (1.0 - x[:n - Lp])
            grad[:n - Lp] += -w * x[Lp:]
    
    sum_diff = np.sum(x) - n / 2.0
    grad += 2.0 * mu * sum_diff
    
    obj = sm * dx + mu * sum_diff * sum_diff
    return obj, dx * grad

def _obj_func(x, beta, mu):
    o, _ = _c5_obj_grad_impl(x, beta, mu)
    return o

def _grad_func(x, beta, mu):
    _, g = _c5_obj_grad_impl(x, beta, mu)
    return g

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.maximum(h, 1.0 - h)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run_opt_lbfgs(h0, n, beta, mu, max_iter, time_limit=None):

    for n2 in n_list:
        if remaining() < 80:
            break
        print("  n={} (remaining {:.0f}s)".format(n2, remaining()))
        h2 = interp_h(best_h, n2)
        
        for beta, mu, max_iter in [(20, 1000.0, 200), (100, 500.0, 200), (500, 100.0, 200)]:
            if remaining() < 20:
                break
            h2 = run_opt_lbfgs(h2, n2, beta, mu, max_iter)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
                best_n = n2
                print("    beta={}, C5={:.6f}".format(beta, best_c5))
    
    # ===== Phase 3: Multi-restart at best n =====
    print("\n=== Phase 3: Multi-restart at n={} ===".format(best_n))
    for trial in range(8):
        if remaining() < 25:
            break
        # Different initializations
        styles = ['smooth', 'binary', 'multi', 'smooth', 'binary', 'multi', 'smooth', 'random']
        h3 = make_random_init(best_n, rng, style=styles[trial % len(styles)])
        # Blend with current best
        blend = 0.3 + 0.1 * trial
        h3 = (1 - blend) * best_h + blend * h3
        h3 = project_feasible(h3, best_n)
        
        for beta, mu in [(50, 500.0), (300, 100.0), (1500, 20.0)]:
            if remaining() < 15:
                break
            h3 = run_opt_lbfgs(h3, best_n, beta, mu, 200)
            c5 = compute_c5(h3)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3.copy()
                print("  Trial {}, beta={}: C5={:.6f}".format(trial, beta, best_c5))
    
    # ===== Phase 4: Very high beta polish =====
    print("\n=== Phase 4: High-beta polish ===")
    if remaining() > 30:
        for beta, mu, max_iter in [(5000, 10.0, 300), (20000, 5.0, 300), (100000, 1.0, 400)]:
            if remaining() < 15:
                break
            best_h = run_opt_lbfgs(best_h, best_n, beta, mu, max_iter)
            c5 = compute_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = best_h.copy()
            print("  beta={:6d}: C5={:.6f}".format(beta, c5))
    
    # ===== Phase 5: Final subgradient polish =====
    if remaining() > 15:
        print("\n=== Phase 5: Final subgradient ===")
        hp, cp = subgradient_polish(best_h, best_n, 500, 0.02 * best_n)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()
        
        hp, cp = top_k_subgradient_polish(best_h, best_n, 300, 0.01 * best_n, top_k=3)
        if cp < best_c5:
            best_c5 = cp
            best_h = hp.copy()
        print("  Final: C5={:.6f}".format(best_c5))
    
    # ===== Phase 6: Binary local search (if close to binary) =====
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_want = best_n // 2
        if n_ones > n_want:
            ones = np.where(h_bin == 1.0)[0]
            order = np.argsort(best_h[ones])[:n_ones - n_want]
            h_bin[ones[order]] = 0.0
        elif n_ones < n_want:
            zeros = np.where(h_bin == 0.0)[0]
            order = np.argsort(-best_h[zeros])[:n_want - n_ones]
            h_bin[zeros[order]] = 1.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Binary: C5={:.6f}".format(best_c5))
        
        # Flip-pair local search on binary
        if np.all((best_h == 0) | (best_h == 1)) and remaining() > 20:
            ones = np.where(best_h == 1)[0].copy()
            zeros = np.where(best_h == 0)[0].copy()
            for _ in range(500):
                if remaining() < 5:
                    break
                improved = False
                for _ in range(30):
                    i1 = rng.randint(len(ones))
                    i0 = rng.randint(len(zeros))
                    p1, p0 = ones[i1], zeros[i0]
                    best_h[p1] = 0.0
                    best_h[p0] = 1.0
                    new_c5 = compute_c5(best_h)
                    if new_c5 < best_c5 - 1e-12:
                        best_c5 = new_c5
                        ones = np.delete(ones, i1)
                        zeros = np.delete(zeros, i0)
                        ones = np.append(ones, p0)
                        zeros = np.append(zeros, p1)
                        improved = True
                    else:
                        best_h[p1] = 1.0
                        best_h[p0] = 0.0
                if not improved:
                    break
            print("  Binary local: C5={:.6f}".format(best_c5))
    
    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
