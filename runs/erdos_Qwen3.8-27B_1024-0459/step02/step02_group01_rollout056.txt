            D[p, v1] += g[idx1[v1]]
        idx2 = p - s
        v2 = (idx2 >= 0) & (idx2 < n)
        if np.any(v2):
            D[p, v2] -= h[idx2[v2]]
    return D

def binary_local_search(h, n, max_rounds=2000, time_limit=60.0):
    t0 = time.time()
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(corr))
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        D = _compute_d_matrix(h, n)
        
        best_new_max = current_max
        best_i = -1
        best_j = -1
        
        for i in ones:
            if time.time() - t0 > time_limit:
                return h
            base = corr - D[i, :]
            D_zeros = D[zeros, :]
            new_maxs = np.max(base[None, :] + D_zeros, axis=1)
            min_idx = np.argmin(new_maxs)
            if new_maxs[min_idx] < best_new_max - 1e-14:
                best_new_max = float(new_maxs[min_idx])
                best_i = i
                best_j = zeros[min_idx]
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_max = float(np.max(corr))
        else:
            break
    
    return h

def binary_local_search_multi(h, n, max_rounds=3000, time_limit=120.0):
    """More aggressive: also try 2-swap moves when stuck."""
    t0 = time.time()
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(corr))
    
    stuck_rounds = 0
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if stuck_rounds < 5:
            D = _compute_d_matrix(h, n)
            best_new_max = current_max
            best_i = -1
            best_j = -1
            for i in ones:
                if time.time() - t0 > time_limit:
                    return h
                base = corr - D[i, :]
                D_zeros = D[zeros, :]
                new_maxs = np.max(base[None, :] + D_zeros, axis=1)
                min_idx = np.argmin(new_maxs)
                if new_maxs[min_idx] < best_new_max - 1e-14:
                    best_new_max = float(new_maxs[min_idx])
                    best_i = i
                    best_j = zeros[min_idx]
            if best_i >= 0:
                h[best_i] = 0.0
                h[best_j] = 1.0
                corr = np.correlate(h, 1.0 - h, mode='full')
                current_max = float(np.max(corr))
                stuck_rounds = 0
            else:
                stuck_rounds += 1
        else:
            # 2-swap: move two 1s to two different 0 positions
            # Try random pairs to avoid O(n^4)
            n_trials = min(500, len(ones) * len(zeros))
            found = False
            for _ in range(n_trials):
                if time.time() - t0 > time_limit:
                    return h
                i1, i2 = rng_global.choice(len(ones), 2, replace=False)
                j1, j2 = rng_global.choice(len(zeros), 2, replace=False)
                i1, i2 = ones[i1], ones[i2]
                j1, j2 = zeros[j1], zeros[j2]
                h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
                new_c5 = float(np.max(np.correlate(h, 1.0-h, mode='full')))
                if new_c5 < current_max - 1e-14:
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    current_max = new_c5
                    stuck_rounds = 0
                    found = True
                    break
                else:
                    h[i1] = 1.0; h[i2] = 1.0; h[j1] = 0.0; h[j2] = 0.0
            if not found:
                stuck_rounds += 1
                if stuck_rounds > 15:
                    break
    
    return h

rng_global = np.random.RandomState(42)

def _smooth_max_obj(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    sh = beta * (c - m)
    sh = np.clip(sh, -500.0, 0.0)
    return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)

def projected_gradient_smooth(h, n, lr=0.015, steps=800, time_limit=40.0, beta=100.0):
    t0 = time.time()
    h = h.astype(float).copy()
    
    for step in range(steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        sh = beta * (corr - m)
        sh = np.clip(sh, -500.0, 0.0)
        w = np.exp(sh)
        w_sum = np.sum(w)
        if w_sum < 1e-15:
            break
        w /= w_sum
        
        s = (n - 1) - np.arange(2 * n - 1)
        grad = np.zeros(n)
        for L in range(2 * n - 1):
            wl = w[L]
            if wl < 1e-9:
                continue
            sl = s[L]
            # grad[i] += wl * (1 - h[i+sl]) for valid i+sl
            if sl >= 0:
                ii = np.arange(n - sl)
                grad[ii] += wl * (1.0 - h[ii + sl])
                jj = np.arange(sl, n)
                grad[jj] -= wl * h[jj - sl]
            else:
                oo = -sl
                ii = np.arange(oo, n)
                grad[ii] += wl * (1.0 - h[ii + sl])
                jj = np.arange(0, n - oo)
                grad[jj] -= wl * h[jj + oo]
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        if c5_new < c5_old + 1e-12:
            h = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.75
            if lr < 0.001:
                lr = 0.001
    
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global rng_global
    rng_global = np.random.RandomState(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def remaining():
        return max(0.0, budget_s - (time.time() - t_start) - 5.0)
    
    # Get parent solution
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 0:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 192
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary local search at parent resolution
    h_bin = round_to_binary(best_h, best_n)
    c5_bin = compute_c5(h_bin)
    
    h_ls = binary_local_search(h_bin, best_n, max_rounds=3000, time_limit=min(150, remaining() * 0.18))
    c5_ls = compute_c5(h_ls)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
    
    # Phase 2: Multi-resolution exploration
    n_list = sorted(set([best_n, 192, 224, 256, 288, 320, 384]))
    n_list = [n for n in n_list if n % 2 == 0 and 96 <= n <= 400]
    
    for n_trial in n_list:
        if remaining() < 50:
            break
        if n_trial == best_n:
            continue
        
        # Interpolate from best
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        xn = (np.arange(n_trial) + 0.5) * 2.0 / n_trial
        h_interp = project_feasible(np.interp(xn, xp, best_h), n_trial)
        
        # Projected gradient
        h_pg = projected_gradient_smooth(h_interp, n_trial, lr=0.01, steps=200,
                                          time_limit=min(25, remaining() * 0.1), beta=80.0)
        
        # Round and binary search
        h_rb = round_to_binary(h_pg, n_trial)
        h_bl = binary_local_search(h_rb, n_trial, max_rounds=1500,
                                    time_limit=min(45, remaining() * 0.2))
        c5_bl = compute_c5(h_bl)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            best_n = n_trial
    
    # Phase 3: Multi-start binary search at best resolution
    if remaining() > 100:
        n_s = best_n
        n_starts = min(12, int(remaining() / 25))
        
        for s in range(n_starts):
            if remaining() < 25:
                break
            
            if s == 0:
                h_s = round_to_binary(best_h, n_s)
            elif s == 1:
                h_s = np.zeros(n_s)
                h_s[rng.choice(n_s, n_s // 2, replace=False)] = 1.0
            elif s == 2:
                # Block pattern
                h_s = np.zeros(n_s)
                nb = 10 + s
                block = max(1, n_s // nb)
                for b in range(0, n_s, 2 * block):
                    h_s[b:min(b + block, n_s)] = 1.0
                h_s = round_to_binary(h_s, n_s)
            elif s == 3:
                # Spread pattern
                h_s = np.zeros(n_s)
                step = n_s // (n_s // 2)
                idx = list(range(0, n_s, step))
                if len(idx) > n_s // 2:
                    idx = idx[:n_s // 2]
                h_s[idx] = 1.0
                if np.sum(h_s) < n_s // 2:
                    rem = [i for i in range(n_s) if h_s[i] == 0]
                    h_s[rng.choice(rem, int(n_s // 2 - np.sum(h_s)), replace=False)] = 1.0
                h_s = round_to_binary(h_s, n_s)
            elif s == 4:
                h_pert = best_h + 0.08 * rng.randn(n_s)
                h_s = round_to_binary(h_pert, n_s)
            elif s == 5:
                h_pert = best_h + 0.15 * rng.randn(n_s)
                h_s = round_to_binary(h_pert, n_s)
            elif s == 6:
                # Reversed
                h_s = round_to_binary(best_h[::-1], n_s)
            elif s == 7:
                # Shift
                h_s = round_to_binary(np.roll(best_h, rng.randint(1, n_s // 2)), n_s)
            else:
                h_s = np.zeros(n_s)
                h_s[rng.choice(n_s, n_s // 2, replace=False)] = 1.0
                # Perturb with parent
                h_s = 0.7 * h_s + 0.3 * best_h
                h_s = round_to_binary(h_s, n_s)
            
            h_bl = binary_local_search(h_s, n_s, max_rounds=800,
                                        time_limit=min(30, remaining() * 0.25))
            c5_bl = compute_c5(h_bl)
            if c5_bl < best_c5:
                best_c5 = c5_bl
                best_h = h_bl.copy()
                best_n = n_s
    
    # Phase 4: Deeper local search on best
    if remaining() > 40:
        h_deep = binary_local_search_multi(best_h, best_n, max_rounds=5000,
                                            time_limit=min(120, remaining() * 0.3))
        c5_deep = compute_c5(h_deep)
        if c5_deep < best_c5:
            best_c5 = c5_deep
            best_h = h_deep.copy()
    
    # Phase 5: Continuous SLSQP polish
    if remaining() > 20:
        n_pol = best_n
        h_pol = project_feasible(best_h, n_pol)
        
        for beta in [500.0, 5000.0, 50000.0]:
            if remaining() < 10:
                break
            def obj(x, _b=beta):
                return _smooth_max_obj(x, n_pol, _b)
            def con(x):
                return np.sum(x) - n_pol / 2.0
            
            bounds = [(0.0, 1.0)] * n_pol
            cons = [{'type': 'eq', 'fun': con}]
            
            try:
                res = minimize(obj, h_pol, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
                h_pol = project_feasible(res.x, n_pol)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
            except Exception:
                pass
        
        # Binary search after continuous polish
        if remaining() > 15:
            h_rb = round_to_binary(h_pol, n_pol)
            h_bl = binary_local_search(h_rb, n_pol, max_rounds=500, time_limit=min(20, remaining() * 0.3))
            c5_bl = compute_c5(h_bl)
            if c5_bl < best_c5:
                best_c5 = c5_bl
                best_h = h_bl.copy()
    
    # Phase 6: Final binary polish
    if remaining() > 10:
        h_final = binary_local_search(best_h, best_n, max_rounds=2000,
                                       time_limit=min(30, remaining() * 0.5))
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
