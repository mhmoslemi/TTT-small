def evaluate_erdos(h):

def block_move_anneal(h, deadline, n_blocks=200):
    """Simulated annealing with block shifts"""
    n = h.size
    target = 0.5 * n
    dx = 2.0 / n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(12345)
    T = 0.005
    T_min = 0.0001
    
    for it in range(n_blocks):
        if time.perf_counter() >= deadline:
            break
        if T < T_min:
            break
        
        # Pick a block to shift
        blen = int(rng.integers(1, 6))
        start = int(rng.integers(0, n - blen))
        shift = int(rng.choice([-2, -1, 1, 2]))
        new_start = start + shift
        
        if new_start < 0 or new_start + blen > n:
            continue
        
        # Check no overlap
        block = best[start:start+blen].copy()
        # Simulate: remove block, place at new position
        cand = best.copy()
        removed_mass = block.copy()
        cand[start:start+blen] -= block
        cand[new_start:new_start+blen] += removed_mass
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_density(cand, target)
        
        c = evaluate_erdos(cand)
        if c < best_c or rng.random() < np.exp(-(c - best_c) / T):
            best = cand
            best_c = c
        
        T *= 0.995
    
    return best, best_c

def binary_push_and_refine(h, deadline):
    """Push toward binary, then refine"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    
    # Progressive binary push
    for thresh in (0.15, 0.10, 0.07, 0.05, 0.03, 0.02):
        if time.perf_counter() >= deadline:
            break
        h2 = best.copy()
        n_zero = np.sum(h2 < thresh)
        n_one = np.sum(h2 > 1.0 - thresh)
        mass_zero = np.sum(h2[h2 < thresh])
        mass_one = np.sum(1.0 - h2[h2 > 1.0 - thresh])
        
        # Can we push these without violating sum?
        # After push: sum changes by -mass_zero + mass_one
        # Need total sum to stay at target
        # Adjust by adding to free elements
        h2[h2 < thresh] = 0.0
        h2[h2 > 1.0 - thresh] = 1.0
        h2 = project_density(h2, target)
        
        # Check that projection didn't undo the push badly
        if np.sum(h2[h2 < 0.5] * h2) > mass_zero * 0.3:
            continue  # Projection undid too much
        
        c2 = evaluate_erdos(h2)
        if c2 < best_c:
            best = h2
            best_c = c2
    
    # Refine after binary push
    if time.perf_counter() < deadline - 30:
        best, best_c = single_lag_descent(best, min(deadline, time.perf_counter() + 60))
    
    return best, best_c

def exhaustive_pair_swap_binary(h, deadline, max_rounds=30):
    """For truly binary solutions, do exhaustive pair swap"""
    n = h.size
    target = 0.5 * n
    dx = 2.0 / n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    
    # Check if close to binary
    n_frac = np.sum((best > 0.01) & (best < 0.99))
    if n_frac > n * 0.1:
        return best, best_c  # Not binary enough
    
    # Snap to binary
    best[best > 0.5] = 1.0
    best[best <= 0.5] = 0.0
    best = project_density(best, target)
    best_c = evaluate_erdos(best)
    
    ones = list(np.where(best >= 1.0 - 1e-8)[0])
    zeros = list(np.where(best <= 1e-8)[0])
    
    best_corr = np.correlate(best, 1.0 - best, mode="full")
    
    for rnd in range(max_rounds):
        if time.perf_counter() >= deadline:
            break
        improved = False
        best_swap = None
        best_val = best_c
        
        for oi in ones:
            for zi in zeros:
                if time.perf_counter() >= deadline:
                    break
                best[oi] = 0.0
                best[zi] = 1.0
                c = float(np.max(np.correlate(best, 1.0 - best, mode="full")) * dx)
                if c < best_val - 1e-12:
                    best_val = c
                    best_swap = (oi, zi)
                best[oi] = 1.0
                best[zi] = 0.0
            if time.perf_counter() >= deadline:
                break

        best[ones] = 1.0
        best[zeros] = 0.0
        
        if best_swap is not None:
            oi, zi = best_swap
            best[oi] = 0.0
            best[zi] = 1.0
            best_c = best_val
            ones.remove(oi)
            zeros.remove(zi)
            ones.append(zi)
            zeros.append(oi)
            improved = True
        
        if not improved:
            break
    
    return best, best_c

def optimize_at_resolution(n, h0, deadline):
    """Full pipeline at a fixed resolution"""
    if n < 4:
        return project_density(h0, 0.5 * n), evaluate_erdos(project_density(h0, 0.5 * n))
    
    target = 0.5 * n
    h = project_density(h0, target)
    best_c = evaluate_erdos(h)
    best_h = h.copy()
    
    # Phase 1: Single-lag gradient descent
    t1 = min(deadline, time.perf_counter() + (deadline - time.perf_counter()) * 0.3)
    h, best_c = single_lag_descent(h, t1, max_iter=8000)
    best_h = h.copy()
    
    # Phase 2: Multi-lag gradient
    t2 = min(deadline, time.perf_counter() + (deadline - time.perf_counter()) * 0.5)
    h2, c2 = multi_lag_gradient(h, t2, top_k=3, max_iter=3000)
    if c2 < best_c:
        h, best_c = h2, c2
        best_h = h.copy()
    
    # Phase 3: Mass transfer
    t3 = min(deadline, time.perf_counter() + (deadline - time.perf_counter()) * 0.4)
    h3, c3 = mass_transfer_opt(h, t3, max_iter=2000)
    if c3 < best_c:
        h, best_c = h3, c3
        best_h = h.copy()
    
    # Phase 4: Fine coordinate descent
    t4 = min(deadline, time.perf_counter() + (deadline - time.perf_counter()) * 0.5)
    h4, c4 = coordinate_descent_fine(h, t4, max_iter=5000)
    if c4 < best_c:
        h, best_c = h4, c4
        best_h = h.copy()
    
    # Phase 5: Final single-lag polish
    rem = deadline - time.perf_counter()
    if rem > 10:
        h5, c5 = single_lag_descent(h, time.perf_counter() + rem * 0.8, max_iter=5000)
        if c5 < best_c:
            h, best_c = h5, c5
            best_h = h.copy()
    
    best_h = project_density(best_h, target)
    return best_h, evaluate_erdos(best_h)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float("inf")
    best_n = 0
    
    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 4:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        nparent = int(parent.size)
        pp = project_density(parent, 0.5 * nparent)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = nparent
    else:
        # Fallback initial construction
        n0 = 450
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0
    
    base_n = best_n
    
    # === MAIN OPTIMIZATION AT BASE RESOLUTION ===
    rem = deadline - time.perf_counter()
    if rem > 30:
        # Strategy: spend most time at base resolution with multiple passes
        alloc_base = rem * 0.55
        
        # Pass 1: Aggressive gradient descent
        dl1 = time.perf_counter() + alloc_base * 0.35
        h1, c1 = single_lag_descent(best_h, dl1, max_iter=15000)
        if c1 < best_c:
            best_h, best_c = h1, c1
        
        # Pass 2: Multi-lag
        dl2 = time.perf_counter() + alloc_base * 0.25
        h2, c2 = multi_lag_gradient(best_h, dl2, top_k=4, max_iter=5000)
        if c2 < best_c:
            best_h, best_c = h2, c2
        
        # Pass 3: Mass transfer
        dl3 = time.perf_counter() + alloc_base * 0.2
        h3, c3 = mass_transfer_opt(best_h, dl3, max_iter=3000)
        if c3 < best_c:
            best_h, best_c = h3, c2 if c2 < c3 else c3
        if c3 < best_c:
            best_h, best_c = h3, c3
        
        # Pass 4: Fine polish
        dl4 = time.perf_counter() + alloc_base * 0.2
        h4, c4 = coordinate_descent_fine(best_h, dl4, max_iter=8000)
        if c4 < best_c:
            best_h, best_c = h4, c4
        
        # Pass 5: Another single-lag round (often helps after other moves)
        rem5 = deadline - time.perf_counter()
        if rem5 > 15:
            h5, c5 = single_lag_descent(best_h, time.perf_counter() + rem5 * 0.7, max_iter=8000)
            if c5 < best_c:
                best_h, best_c = h5, c5
    
    # === TRY BINARY PATH ===
    rem = deadline - time.perf_counter()
    if rem > 50 and best_n <= 500:
        # Push toward binary and see if it helps
        h_bin, c_bin = binary_push_and_refine(best_h, time.perf_counter() + min(40, rem * 0.4))
        if c_bin < best_c:
            best_h, best_c = h_bin, c_bin
        
        # If close to binary, try exhaustive pair swap
        n_frac = np.sum((best_h > 0.02) & (best_h < 0.98))
        if n_frac < best_n * 0.05 and rem > 30:
            h_bs, c_bs = exhaustive_pair_swap_binary(best_h, time.perf_counter() + min(60, rem * 0.5), max_rounds=20)
            if c_bs < best_c:
                best_h, best_c = h_bs, c_bs
    
    # === TRY BLOCK MOVE ANNEALING ===
    rem = deadline - time.perf_counter()
    if rem > 30 and best_n <= 500:
        h_blk, c_blk = block_move_anneal(best_h, time.perf_counter() + min(40, rem * 0.5), n_blocks=500)
        if c_blk < best_c:
            best_h, best_c = h_blk, c_blk
        # Follow up with gradient if block move helped
        if c_blk < best_c + 0.001 and rem > 15:
            h_bd, c_bd = single_lag_descent(blk_h, time.perf_counter() + min(20, rem * 0.3), max_iter=3000)
    
    # === TRY ALTERNATIVE RESOLUTIONS ===
    rem = deadline - time.perf_counter()
    alt_ns = []
    for cand_n in [360, 300, 540, 600, 720]:
        if cand_n != best_n and rem > 40 and 4 <= cand_n <= 800:
            alt_ns.append(cand_n)
    
    for an in alt_ns[:3]:
        rem = deadline - time.perf_counter()
        if rem < 30:
            break
        alloc_alt = min(rem * 0.3, 120.0)
        h_ux = upsample_h(best_h, an)
        h_alt, c_alt = optimize_at_resolution(an, h_ux, time.perf_counter() + alloc_alt)
        if c_alt < best_c:
            best_h, best_c = h_alt, c_alt
            best_n = an
    
    # === FINAL INTENSIVE POLISH ===
    rem = deadline - time.perf_counter()
    if rem > 10 and best_h is not None:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        
        # Very tight single-lag optimization
        if rem > 15:
            hf1, cf1 = single_lag_descent(h_fin, time.perf_counter() + rem * 0.4, max_iter=10000)
            if cf1 < best_c:
                h_fin, best_c = hf1, cf1
        
        # Multi-lag with top_k=2 (more focused)
        rem2 = deadline - time.perf_counter()
        if rem2 > 10:
            hf2, cf2 = multi_lag_gradient(h_fin, time.perf_counter() + rem2 * 0.4, top_k=2, max_iter=3000)
            if cf2 < best_c:
                h_fin, best_c = hf2, cf2
        
        # Final mass transfer
        rem3 = deadline - time.perf_counter()
        if rem3 > 8:
            hf3, cf3 = mass_transfer_opt(h_fin, time.perf_counter() + rem3 * 0.5, max_iter=1500)
            if cf3 < best_c:
                h_fin, best_c = hf3, cf3
        
        # Ultra-fine coordinate descent
        rem4 = deadline - time.perf_counter()
        if rem4 > 5:
            hf4, cf4 = coordinate_descent_fine(h_fin, time.perf_counter() + rem4 * 0.7, max_iter=5000)
            if cf4 < best_c:
                h_fin, best_c = hf4, cf4
        
        best_h = h_fin
        best_n = nbest
    
    # Final validation
    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    
    # Safety: ensure constraints are met
    assert final_h.size >= 2
    assert np.all(final_h >= -1e-10) and np.all(final_h <= 1.0 + 1e-10)
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = project_density(final_h, 0.5 * final_h.size)
    final_c = evaluate_erdos(final_h)
    
    return final_h, float(final_c), int(final_h.size)
