import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def make_feasible(h, n):
    return project_feasible(h, n)

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def softmax_weights(corr, beta):
    shifted = beta * (corr - np.max(corr))
    p = np.exp(shifted)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros_like(corr)
        p[np.argmax(corr)] = 1.0
    else:
        p /= s
    return p

def compute_gradient_full(h, p, n):
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        pk = p[s + n - 1]
        if pk < 1e-14:
            continue
        if s == 0:
            grad += pk * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += pk * (1.0 - h[s:])
            grad[s:] += pk * (-h[:n-s])
        else:
            s2 = -s
            grad[s2:] += pk * (1.0 - h[:n-s2])
            grad[:n-s2] += pk * (-h[s2:])
    return grad

                    return h_best, c5_best
            corr = compute_corr(h_y)
            p = softmax_weights(corr, beta)
            g = compute_gradient_full(h_y, p, n)

def slsqp_run(n, h_init, max_iter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    def obj_func(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    try:
        res = minimize(obj_func, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def smooth_slsqp(n, h_init, betas, max_iter_per=1500):
    current = h_init.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    for beta in betas:
        def obj_s(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            return cmax + (1.0/beta) * np.log(np.sum(np.exp(beta * (corr - cmax) + 1e-10)) / len(corr))
        try:
            res = minimize(obj_s, current, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': max_iter_per, 'ftol': 1e-14, 'iprint': 0})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5

def pair_swap_local(h, n, n_rounds, rng, time_limit=None, start_time=None):
    h = h.copy()
    best_c5 = compute_c5(h)
    for r in range(n_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        for i in range(n - 1):
            if h[i] <= h[i+1]:
                continue
            for step in [0.01, 0.02, 0.05, 0.1]:
                h_try = h.copy()
                amt = min(step, (h[i] - h[i+1]) / 2.0)
                if amt < 1e-6:
                    continue
                h_try[i] -= amt
                h_try[i+1] += amt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

            n_try += 1
        tl = min(60, remaining() - 15)
        h_bin, c5_bin = binary_sa(n_try, rng, n_steps=20000, t_start=0.6, t_end=0.002,
                                   time_limit=tl, start_time=start_time)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.astype(float)
            best_n = n_try
    
    # Phase 8: PG from binary best at its resolution
    if remaining() > 50:
        tl = min(60, remaining() - 10)
        betas = [2, 10, 40, 150]
        h_opt, c5_opt = pg_solve(best_h, best_n, betas, iters_per_beta=300,
                                  lr_base=5.0, use_momentum=True,
                                  time_limit=tl, start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 9: Local refinement - pair swaps
    if remaining() > 30:
        n_rounds = max(20, min(100, int(remaining() / 1.0)))
        best_h, best_c5 = pair_swap_local(best_h, best_n, n_rounds, rng,
                                           time_limit=min(40, remaining()-10), start_time=start_time)
    
    # Phase 10: Random perturbation improvement
    if remaining() > 20:
        n_trials = max(1000, min(20000, int((remaining() - 5) / 0.015)))
        best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.03, rng,
                                                  time_limit=min(30, remaining()-5), start_time=start_time)
    
    # Phase 11: Final SLSQP polish
    if remaining() > 25:
        h_opt, c5_opt = slsqp_run(best_n, best_h, max_iter=3000, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 12: One more PG pass at finest resolution
    if remaining() > 20:
        n_fine = max(best_n, 320)
        h_interp = project_feasible(interpolate_to_n(best_h, n_fine), n_fine)
        tl = min(20, remaining() - 3)
        betas = [10, 50, 200]
        h_opt, c5_opt = pg_solve(h_interp, n_fine, betas, iters_per_beta=200,
                                  lr_base=8.0, use_momentum=True,
                                  time_limit=tl, start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_fine
    
    # Final feasibility check and cleanup
    best_h = make_feasible(best_h, best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    # Verify constraints
    assert np.all(best_h >= -1e-12), "h has negative values"
    assert np.all(best_h <= 1.0 + 1e-12), "h exceeds 1"
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6, f"sum constraint violated: {np.sum(best_h)} vs {best_n/2.0}"
    
    return (best_h, best_c5, best_n)
