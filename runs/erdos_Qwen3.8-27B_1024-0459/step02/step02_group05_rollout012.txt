def lse_gradient_descent(h, n, beta, lr, max_iter):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        # Compute overlaps
        omh = 1.0 - h
        corr = np.correlate(h, omh, mode='full')
        c5 = np.max(corr) * (2.0/n)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # LSE weights
        max_c = np.max(corr)
        exps = np.exp(beta * (corr - max_c))
        weights = exps / np.sum(exps)
        
        # Compute gradient
        grad = np.zeros(n)
        for j in range(2*n - 1):
            s = j - (n - 1)
            w = weights[j]
            if w < 1e-10:
                continue
            if s >= 0:
                # (1-h[m+s]) for m in [0, n-1-s]
                grad[:n-s] += w * (1.0 - h[s:])
                # -h[m-s] for m in [s, n-1]
                grad[s:] -= w * h[:n-s]
            else:
                s_neg = -s
                # (1-h[m-s_neg]) for m in [s_neg, n-1]
                grad[s_neg:] += w * (1.0 - h[:n-s_neg])
                # -h[m+s_neg] for m in [0, n-1-s_neg]
                grad[:n-s_neg] -= w * h[s_neg:]
        
        grad *= (2.0/n)  # dx
        
        # Projected gradient step
        h_new = h - lr * grad
        
        # Project onto [0,1] and sum = n/2
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum constraint
        target = n / 2.0
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-15:
            if diff > 0:
                mask = h_new < 1.0 - 1e-10
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h_new[mask] = np.minimum(h_new[mask] + diff/cnt, 1.0)
            else:
                mask = h_new > 1e-10
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h_new[mask] = np.maximum(h_new[mask] + diff/cnt, 0.0)
        
        h = np.clip(h_new, 0.0, 1.0)
        
        # Adaptive learning rate
        if c5 > best_c5 + 0.01:
            lr *= 0.8
    
    return best_h, best_c5

            ones = np.where(h_bin == 1)[0]
            extra = n_ones - n//2
            to_flip = rng.choice(ones, extra, replace=False)
            h_bin[to_flip] = 0
        else:
            zeros = np.where(h_bin == 0)[0]
            need = n//2 - n_ones
            to_flip = rng.choice(zeros, need, replace=False)
            h_bin[to_flip] = 1
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    t0 = time.time()
    
    n_iter = 0
    while time.time() - t0 < time_limit:
        n_iter += 1
        corr = compute_corr(best_h)
        worst_j = np.argmax(corr)
        worst_s = worst_j - (n - 1)
        
        # Find contributing positions for worst lag
        if worst_s >= 0:
            valid = np.arange(0, n - worst_s)
        else:
            valid = np.arange(-worst_s, n)
        
        contributors = valid[(best_h[valid] > 0.5) & (best_h[valid + worst_s] < 0.5)]
        
        if len(contributors) == 0:
            break
        
        improved = False
        n_attempts = min(50, len(contributors))
        for _ in range(n_attempts):
            i = contributors[rng.randint(len(contributors))]
            zeros = np.where(best_h < 0.5)[0]
            if len(zeros) == 0:
                break
            j = zeros[rng.randint(len(zeros))]
            if j == i:
                continue
            
            best_h[i] = 0.0
            best_h[j] = 1.0
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                improved = True
                break
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0
        
        if not improved:
            # Try random swap to escape
            ones = np.where(best_h > 0.5)[0]
            zeros = np.where(best_h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_h[i] = 0.0
            best_h[j] = 1.0
            c5_new = compute_c5(best_h)
            if c5_new < best_c5 + 0.005:  # Allow small increases
                best_c5 = c5_new
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0
    
    return best_h, best_c5

def slsqp_run(n, h_init, max_iter=3000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None
