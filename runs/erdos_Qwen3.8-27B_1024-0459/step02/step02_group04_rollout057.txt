def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:
        n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    target = 0.5 * float(n)
    if h.size < n:
        h = np.hstack([h, np.zeros(n - h.size)])
    else:
        h = h[:n].copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(100):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = out < 1.0 - 1e-13
        else:
            mask = out > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_lag_data(h):
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    nvals = 2 * n - 1
    vals = np.empty(nvals, dtype=float)
    lags = np.arange(-(n - 1), n, dtype=int)
    ac0 = float(np.dot(h, h))
    vals[n - 1] = (total - ac0) * dx
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals[n - 1 + L] = (pref[m] - ac) * dx
        vals[n - 1 - L] = (total - pref[L] - ac) * dx
    return vals, lags


def choose_active(vals, lags, max_count, margin):
    order = np.argsort(-vals)
    take = min(int(max_count), vals.size)
    idx = order[:take]
    top = float(vals[order[0]])
    near = np.where(vals >= top - margin)[0]
    idx = np.unique(np.concatenate((idx, near)))
    cap = max(int(max_count), 10)
    if idx.size > cap:
        idx = idx[np.argsort(-vals[idx])][:cap]
    return [int(lags[i]) for i in idx]


def lag_value(h, l):
    n = len(h)
    if l == 0:
        return float(np.sum(h) - np.dot(h, h))
    L = abs(int(l))
    if L >= n:
        return 0.0
    m = n - L
    ac = float(np.dot(h[:m], h[L:]))
    if l > 0:
        return float(np.sum(h[:m])) - ac
    else:
        return float(np.sum(h)) - float(np.sum(h[:L])) - ac


def lag_grad(h, l):
    n = len(h)
    g = np.zeros(n)
    if l == 0:
        return 1.0 - 2.0 * h
    L = abs(int(l))
    if L >= n:
        return g
    m = n - L
    if l > 0:
        g[:m] += 1.0 - h[L:]
        g[L:] += -h[:m]
    else:
        g[L:] += 1.0 - h[:m]
        g[:m] += -h[L:]
    return g


def obj_t(x):
    return float(x[-1])


def obj_t_jac(x):
    j = np.zeros(len(x))
    j[-1] = 1.0
    return j


def con_sum_eq(x):
    return float(np.sum(x[:-1]) - 0.5 * float(len(x) - 1))


def con_sum_eq_jac(x):
    j = np.zeros(len(x))
    j[:-1] = 1.0
    return j


def con_ineq_lags(x):
    n = len(x) - 1
    dx = 2.0 / float(n)
    m = len(AC_LAGS)
    out = np.empty(m)
    h = x[:-1]
    for i in range(m):
        out[i] = float(x[-1]) - lag_value(h, AC_LAGS[i]) * dx
    return out


def con_ineq_lags_jac(x):
    n = len(x) - 1
    dx = 2.0 / float(n)
    m = len(AC_LAGS)
    J = np.zeros((m, n + 1))
    h = x[:-1]
    for i in range(m):
        g = lag_grad(h, AC_LAGS[i])
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def solve_active(h0, active, t0, maxiter, ftol):
    global AC_LAGS
    n = h0.size
    AC_LAGS = [int(l) for l in active]
    x0 = np.empty(n + 1)
    x0[:n] = h0
    x0[n] = t0
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [{"type": "eq", "fun": con_sum_eq, "jac": con_sum_eq_jac}]
    if len(AC_LAGS) > 0:
        cons.append({"type": "ineq", "fun": con_ineq_lags, "jac": con_ineq_lags_jac})
    try:
        res = minimize(obj_t, x0, method="SLSQP", bounds=bounds,
                       jac=obj_t_jac, constraints=cons,
                       options={"maxiter": maxiter, "ftol": ftol, "disp": False})
        return res
    except Exception:
        return None


def smooth_obj(x):
    return _smooth_max_obj(x, SM_BETA)


def smooth_obj_jac(x):
    return _smooth_max_jac(x, SM_BETA)


def _smooth_max_obj(h, beta):
    n = h.size
    vals, _ = all_lag_data(h)
    vmax = float(np.max(vals))
    shifted = beta * (vals - vmax)
    shifted = np.clip(shifted, -700, 0)
    r = np.exp(shifted).sum()
    if r <= 0:
        return vmax
    return vmax + np.log(r) / beta


def _smooth_max_jac(h, beta):
    n = h.size
    dx = 2.0 / float(n)
    vals, lags = all_lag_data(h)
    vmax = float(np.max(vals))
    shifted = beta * (vals - vmax)
    shifted = np.clip(shifted, -700, 0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum <= 0:
        return np.zeros(n)
    w /= wsum
    g = np.zeros(n)
    thr = w.max() * 1e-9
    for i in range(len(vals)):
        if w[i] > thr:
            g += w[i] * dx * lag_grad(h, int(lags[i]))
    return g


def sum_eq(x):
    return float(np.sum(x) - 0.5 * float(len(x)))


def sum_eq_jac(x):
    return np.ones(len(x))


def smooth_slsqp(h0, beta, maxiter):
    global SM_BETA
    n = h0.size
    SM_BETA = max(1.0, float(beta))
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac}]
    try:
        res = minimize(smooth_obj, h, method="SLSQP",
                       jac=smooth_obj_jac, bounds=bounds,
                       constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-12, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def block_perturb(h0, tlimit, seed, max_trials, step):
    rng = np.random.RandomState(int(seed))
    n = h0.size
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.0008
    min_temp = 1e-10
    for _ in range(max_trials):
        if time.time() > tlimit:
            break
        max_len = min(5, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[st:st + L] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
        if c_try <= cur_c:
            cur_h = h_try
            cur_c = c_try
        elif c_try - cur_c < temp:
            if rng.rand() < np.exp(-(c_try - cur_c) / temp):
                cur_h = h_try
                cur_c = c_try
        temp = max(temp * 0.9985, min_temp)
    return best_h, float(best_c)


def interp_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, float(h_src[0]) if n_src == 1 else 0.5)
    xo = np.linspace(0, 2, n_src, endpoint=False)
    xn = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xn, xo, h_src)


def refine(h0, n, tlimit, ami):
    n = int(n)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    for beta in (8, 30, 100):
        if time.time() > tlimit - 50:
            break
        mx = 50 if n <= 170 else 35
        h2 = smooth_slsqp(h, beta, mx)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
        h = h2 if c2 <= best_c + 0.003 else best_h

    active_counts = [50, 35, 25, 18, 12, 10, 8]
    for ac in active_counts:
        if time.time() > tlimit - 15:
            break
        vals, lags = all_lag_data(h)
        ccur = float(np.max(vals))
        if ccur < best_c:
            best_c = ccur
            best_h = h.copy()
        active = choose_active(vals, lags, ac, 1e-6)
        t0 = ccur + 1e-7
        res = solve_active(h, active, t0, ami, 1e-13)
        if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x[:n], n)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            if c2 <= ccur + 1e-10:
                h = h2

    for _ in range(3):
        if time.time() > tlimit - 8:
            break
        vals, lags = all_lag_data(h)
        ccur = float(np.max(vals))
        if ccur < best_c:
            best_c = ccur
            best_h = h.copy()
        active = choose_active(vals, lags, 6, 1e-9)
        res = solve_active(h, active, ccur + 1e-9, 120, 1e-14)
        if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x[:n], n)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            if c2 <= ccur + 1e-11:
                h = h2

    return project_feasible(best_h, n), float(best_c)


def make_variants(h0, n, seed):
    rng = np.random.RandomState(int(seed))
    h0 = project_feasible(h0, n)
    out = [h0.copy()]
    flip = h0[::-1]
    out.append(project_feasible(0.5 * (h0 + flip), n))
    k = np.ones(5) / 5.0
    sm = np.convolve(h0, k, mode="same")
    out.append(project_feasible(sm, n))
    if n > 2:
        sm2 = np.convolve(h0, np.ones(9) / 9.0, mode="same")
        out.append(project_feasible(sm2, n))
    pert = h0 + rng.randn(n) * 0.04
    out.append(project_feasible(pert, n))
    return out


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    t0 = time.time()
    hard_end = t0 + max(60.0, budget_s - 25.0)
    rng = np.random.RandomState(seed)

    parent = get_parent_values()
    if parent is not None:
        pn = int(parent.size)
        best_h = project_feasible(parent, pn)
        best_c = compute_c5(best_h)
        best_n = pn
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Refine at parent resolution (most important)
    if time.time() < hard_end - 120:
        slt = min(280.0, hard_end - 120 - time.time())
        variants = make_variants(best_h, best_n, seed)
        best_c_phase1 = best_c
        for vi, v in enumerate(variants):
            if time.time() > hard_end - 120:
                break
            rem = hard_end - 120 - time.time()
            per = min(120.0, rem / max(1, len(variants) - vi))
            per = max(30.0, per)
            h2, c2 = refine(v, best_n, time.time() + per, 100)
            if c2 < best_c_phase1:
                best_c_phase1 = c2
                best_h = h2
        if best_c_phase1 < best_c:
            best_c = best_c_phase1

    # Phase 2: Try nearby resolutions
    cands = set()
    for d in (8, 16, 24, 32, 48):
        cands.add(best_n - d)
        cands.add(best_n + d)
    cands.update([128, 144, 160, 176, 192, 208, 224, 240])
    cands.discard(best_n)
    cands = sorted([c for c in cands if 40 <= c <= 300])

    for nn in cands:
        if time.time() > hard_end - 80:
            break
        rem = hard_end - 80 - time.time()
        if rem < 35:
            break
        slt = min(65.0, rem)
        init = interp_n(best_h, nn)
        h2, c2 = refine(init, nn, time.time() + slt, 80)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = nn

    # Phase 3: Final aggressive polish at best resolution
    if time.time() < hard_end - 20 and best_n >= 3:
        slt = min(80.0, hard_end - 12 - time.time())
        h2, c2 = refine(best_h, best_n, time.time() + slt, 130)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = int(best_n)

    # Phase 4: Block perturbation
    if time.time() < hard_end - 3 and 3 <= best_n <= 280:
        h2, c2 = block_perturb(best_h, hard_end - 1, seed + 7777, 2500, 0.012)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
