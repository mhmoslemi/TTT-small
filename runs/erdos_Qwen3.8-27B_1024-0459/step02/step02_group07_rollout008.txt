def slsqp_polish(h, n, max_iter=3000):
    def c5_obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    def sum_constraint(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    try:
        res = minimize(c5_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:

            return h_out, compute_c5(h_out)
