            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(40):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-13
        else:
            mask = out > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out

def smooth_max_val(h, beta):
    n = len(h)
    vals = all_lags(h)
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    return m + float(np.log(np.sum(np.exp(shifted)))) / beta

def smooth_max_grad(h, beta):
    n = len(h)
    vals = all_lags(h)
    m = float(np.max(vals))
    w = np.exp(beta * (vals - m))
    wsum = float(w.sum())
    if wsum <= 0:
        w = np.full(2*n-1, 1.0/float(2*n-1))
    else:
        w = w / wsum
    one_minus_h = 1.0 - h
    grad = np.zeros(n, dtype=float)
    for j in range(2*n-1):
        wk = float(w[j])
        if wk < 1e-15:
            continue
        offset = j - (n - 1)
        if offset == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif offset > 0:
            m2 = n - offset
            grad[offset:] += wk * (1.0 - h[:m2])
            grad[:m2] += wk * (-h[offset:])
        else:
            p = -offset
            m2 = n - p
            grad[p:] += wk * (1.0 - h[:m2])
            grad[:m2] += wk * (-h[p:])
    return grad

def sum_eq(x):
    return float(np.sum(x)) - 0.5 * float(len(x))

def sum_eq_jac(x):
    return np.ones(len(x), dtype=float)

def phase_smooth(h0, n, beta_list, time_limit, maxiter_list=None):
    best_h = h0.copy()
    best_c = compute_c5(h0)
    n = int(n)
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac}]
    if maxiter_list is None:
        maxiter_list = [50, 80, 120, 150]
    for i, beta in enumerate(beta_list):
        if time.time() > time_limit:
            break
        def obj(x, b=beta):
            return smooth_max_val(x, b)
        def grad(x, b=beta):
            return smooth_max_grad(x, b)
        try:
            res = minimize(
                obj, project_feasible(best_h, n), method="SLSQP", jac=grad,
                bounds=bounds, constraints=constraints,
                options={"maxiter": int(maxiter_list[min(i, len(maxiter_list)-1)]), "ftol": 1e-13, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h2 = project_feasible(res.x, n)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2
                elif c2 < best_c + 1e-6:
                    best_h = h2
        except Exception:
            pass
    return best_h, best_c

def lag_val(h, L):
    n = len(h)
    if L == 0:
        return float(np.sum(h) - np.dot(h, h))
    L = abs(int(L))
    m = n - L
    low = h[:m]
    high = h[L:]
    ac = float(np.dot(low, high))
    if int(L) > 0:
        return float(np.sum(low) - ac)
    return float(np.sum(high) - ac)

def lag_grad(h, L):
    n = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    Ls = abs(int(L))
    m = n - Ls
    low = h[:m]
    high = h[Ls:]
    g = np.zeros(n, dtype=float)
    if int(L) > 0:
        g[:m] += 1.0 - high
        g[Ls:] += -low
    else:
        g[Ls:] += 1.0 - low
        g[:m] += -high
    return g

def choose_top_lags(h, count, margin=1e-8):
    vals = all_lags(h)
    lags = np.arange(-(len(h)-1), len(h))
    order = np.argsort(-vals)
    top = float(vals[order[0]])
    near = np.where(vals >= top - margin)[0]
    idx = np.unique(np.concatenate((order[:count], near)))
    idx = idx[np.argsort(-vals[idx])]
    take = min(int(count) * 2, len(idx))
    return [int(lags[i]) for i in idx[:take]]

def t_obj(x):
    return float(x[-1])

def t_obj_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j

def active_sum_eq(x):
    return float(np.sum(x[:-1])) - 0.5 * float(len(x) - 1)

def active_sum_eq_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j

def active_ineq(x, lags, n):
    dx = 2.0 / float(n)
    out = np.empty(len(lags), dtype=float)
    h = x[:-1]
    for i, ll in enumerate(lags):
        out[i] = float(x[-1]) * dx - lag_val(h, int(ll))
    return out

def active_ineq_jac(x, lags, n):
    dx = 2.0 / float(n)
    m = len(lags)
    J = np.zeros((m, n + 1), dtype=float)
    h = x[:-1]
    for i, ll in enumerate(lags):
        g = lag_grad(h, int(ll))
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J

def phase_active(h0, n, time_limit, active_seq=(10, 20, 40, 64), maxiter=60):
    best_h = h0.copy()
    best_c = compute_c5(h0)
    n = int(n)
    bounds_h = [(0.0, 1.0) for _ in range(n)]
    for acnt in active_seq:
        if time.time() > time_limit:
            break
        for _ in range(2):
            if time.time() > time_limit:
                break
            lags = choose_top_lags(best_h, int(acnt), margin=1e-7)
            lags = lags[:min(acnt*2, 80)]
            t0 = best_c + 1e-5
            x0 = np.empty(n + 1, dtype=float)
            x0[:n] = best_h
            x0[n] = t0
            bounds = bounds_h + [(0.0, 2.0)]
            cons = [{"type": "eq", "fun": active_sum_eq, "jac": active_sum_eq_jac}]
            if len(lags) > 0:
                cons.append({"type": "ineq", "fun": lambda x: active_ineq(x, lags, n),
                             "jac": lambda x: active_ineq_jac(x, lags, n)})
            try:
                res = minimize(t_obj, x0, method="SLSQP", jac=t_obj_jac,
                               bounds=bounds, constraints=cons,
                               options={"maxiter": int(maxiter), "ftol": 1e-13, "disp": False})
                if res.x is not None and np.all(np.isfinite(res.x)) and res.x.size == n+1:
                    h2 = project_feasible(res.x[:n], n)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_h = h2
                        best_c = c2
                    elif c2 <= best_c + 1e-9:
                        best_h = h2
            except Exception:
                pass
    return best_h, best_c

def pairwise_swap_search(h0, n, time_limit, seed, max_trials=20000):
    rng = np.random.RandomState(int(seed))
    n = int(n)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    for trial in range(max_trials):
        if time.time() > time_limit:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        # Transfer delta from i to j (or j to i)
        delta = rng.choice([-1.0, 1.0]) * rng.uniform(0.01, 0.15)
        h_try = h.copy()
        h_try[i] += delta
        h_try[j] -= delta
        if h_try[i] < -1e-10 or h_try[i] > 1.0 + 1e-10 or h_try[j] < -1e-10 or h_try[j] > 1.0 + 1e-10:
            # Try a smaller delta
            delta = delta * 0.3
            h_try[i] += delta
            h_try[j] -= delta
            if h_try[i] < -1e-10 or h_try[i] > 1.0 + 1e-10 or h_try[j] < -1e-10 or h_try[j] > 1.0 + 1e-10:
                continue
        h_try = np.clip(h_try, 0.0, 1.0)
        # Re-project to maintain sum
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try
            best_c = c_try
            h = h_try
            cur_c = c_try
        elif c_try < cur_c:
            h = h_try
            cur_c = c_try
    return best_h, float(best_c)

def block_search(h0, n, time_limit, seed, max_trials=8000):
    rng = np.random.RandomState(int(seed) + 7919)
    n = int(n)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        L = int(rng.randint(2, min(12, n)))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * 0.04
        delta -= delta.mean()
        h_try = h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try
            best_c = c_try
            h = h_try
    return best_h, float(best_c)

def subgradient_descent(h0, n, time_limit, max_iter=5000):
    """Subgradient method for min max_lag(lag_k(h)) with sum constraint."""
    n = int(n)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    step = 0.01
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        vals = all_lags(h)
        k_max = int(np.argmax(vals))
        c_cur = float(vals[k_max])
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()
        L = int(k_max - (n - 1))
        g = lag_grad(h, L)
        # Projected subgradient step
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        # Adaptive step
        c_new = compute_c5(h_new)
        if c_new < c_cur:
            step = min(0.05, step * 1.05)
            h = h_new
        else:
            step = max(0.001, step * 0.9)
    return best_h, best_c

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, float(h_src[0]) if n_src == 1 else 0.5)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def structured_start(n, seed):
    rng = np.random.RandomState(seed)
    n = int(n)
    starts = []
    # Bipartite-like
    h = np.zeros(n)
    q = n // 2
    h[:q] = 1.0
    starts.append(h)
    # Alternating
    h = np.zeros(n)
    h[::2] = 1.0
    if n % 2 == 1:
        h[-1] = 0.0
    starts.append(h)
    # Four-block
    if n % 4 == 0:
        h = np.tile([1.0, 1.0, 0.0, 0.0], n//4)
        starts.append(h)
    # Random
    h = rng.rand(n) * 1.2
    starts.append(h)
    # Kosaratu-like (from van der Corput)
    h = np.zeros(n)
    for i in range(n):
        p = 0
        x = i
        while x > 0:
            p = 1 - p
            x = x >> 1
        h[i] = 1.0 if p == 0 else 0.0
    starts.append(h)
    out = [project_feasible(s, n) for s in starts]
    return out

def full_optimize(h0, n, time_limit, seed):
    """Full optimization pipeline for a fixed n."""
    n = int(n)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    # Phase 1: Smooth max with increasing beta
    t1 = time.time()
    remaining = time_limit - t1
    if remaining > 30:
        frac = min(0.45, remaining / 1300.0 + 0.2)
        t_end = time.time() + remaining * frac
        beta_list = [30, 80, 200, 500, 1000]
        maxit_list = [40, 60, 80, 100, 120]
        h2, c2 = phase_smooth(h, n, beta_list, t_end, maxiter_list)
        if c2 < best_c:
            best_h = h2
            best_c = c2
        h = best_h

    # Phase 2: Active set SLSQP
    remaining = time_limit - time.time()
    if remaining > 30:
        frac = min(0.30, remaining / 1300.0 + 0.1)
        t_end = time.time() + remaining * frac
        seq = (8, 16, 32, 48, 64)
        h2, c2 = phase_active(h, n, t_end, active_seq=seq, maxiter=55)
        if c2 < best_c:
            best_h = h2
            best_c = c2
        h = best_h

    # Phase 3: Subgradient for a bit
    remaining = time_limit - time.time()
    if remaining > 20 and n <= 256:
        t_end = time.time() + remaining * 0.15
        h2, c2 = subgradient_descent(h, n, t_end, max_iter=3000)
        if c2 < best_c:
            best_h = h2
            best_c = c2
        h = best_h

    # Phase 4: Active set again (finer)
    remaining = time_limit - time.time()
    if remaining > 20:
        t_end = time.time() + remaining * 0.2
        seq = (6, 12, 24, 40)
        h2, c2 = phase_active(h, n, t_end, active_seq=seq, maxiter=80)
        if c2 < best_c:
            best_h = h2
            best_c = c2
        h = best_h

    # Phase 5: Pairwise swaps
    remaining = time_limit - time.time()
    if remaining > 10:
        t_end = time.time() + remaining * 0.35
        h2, c2 = pairwise_swap_search(h, n, t_end, seed, max_trials=15000)
        if c2 < best_c:
            best_h = h2
            best_c = c2
        h = best_h

    # Phase 6: Block search
    remaining = time_limit - time.time()
    if remaining > 10:
        t_end = time.time() + remaining * 0.4
        h2, c2 = block_search(h, n, t_end, seed, max_trials=6000)
        if c2 < best_c:
            best_h = h2
            best_c = c2
        h = best_h

    # Phase 7: Final active set polish
    remaining = time_limit - time.time()
    if remaining > 8:
        t_end = time.time() + remaining * 0.4
        h2, c2 = phase_active(h, n, t_end, active_seq=(8, 16, 32), maxiter=70)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)

def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    start_time = time.time()
    hard_end = start_time + budget_s - 5.0

    # Get parent
    parent = None
    try:
        vals = initial_h_values
        if vals is not None:
            parent = np.asarray(vals, dtype=float).copy()
            if parent.size == 0:
                parent = None
    except Exception:
        parent = None

    if parent is not None:
        best_n = int(parent.size)
        best_h = project_feasible(parent, best_n)
        best_c = compute_c5(best_h)
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Strategy: focus on a few key n values
    # Primary: parent n; Secondary: nearby and structured n values
    n_candidates = [best_n]
    for nn in [128, 144, 160, 176, 192, 208, 224, 240, 256]:
        if nn != best_n and 80 <= nn <= 300:
            n_candidates.append(nn)
    # Sort by proximity to best_n
    n_candidates = list(dict.fromkeys(n_candidates))

    # If parent exists, start with its n; otherwise try 160 first
    if parent is not None:
        n_candidates.remove(best_n)
        n_candidates = [best_n] + n_candidates

    # Allocate time: ~50% to primary n, ~30% to secondary, ~20% to final polish
    primary_time = budget_s * 0.42
    secondary_time = budget_s * 0.33
    polish_time = budget_s * 0.18

    # Phase A: Optimize at primary n
    t_end_a = start_time + primary_time
    inits = [best_h.copy()]
    # Add a couple more inits
    if parent is not None:
        inits.append(project_feasible(parent[::-1], best_n))
        inits.append(project_feasible(0.5*(parent + parent[::-1]), best_n))
    st = structured_start(best_n, seed)
    inits.extend(st[:2])

    for j, h0 in enumerate(inits):
        if time.time() > t_end_a - 5:
            break
        per = (t_end_a - time.time()) / max(1, len(inits) - j)
        per = min(per, budget_s * 0.12)
        per = max(per, 20.0)
        h2, c2 = full_optimize(h0, best_n, time.time() + per, seed + 100*j)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase B: Try other n values
    t_end_b = start_time + primary_time + secondary_time
    for nn in n_candidates[1:6]:  # Try up to 5 other n values
        nn = int(nn)
        if time.time() > t_end_b - 10:
            break
        per = min(t_end_b - time.time(), budget_s * 0.10)
        per = max(per, 30.0)
        # Interpolate best solution to this n
        h_start = interpolate_to_n(best_h, nn)
        h2, c2 = full_optimize(h_start, nn, time.time() + per, seed + 5000 + nn)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = nn

    # Phase C: Final polish at best n
    t_end_c = hard_end - 3.0
    if time.time() < t_end_c - 10:
        # Another round of active set + swaps at the best n
        h2, c2 = phase_active(best_h, best_n, time.time() + (t_end_c - time.time()) * 0.4,
                              active_seq=(8, 16, 32, 64), maxiter=80)
        if c2 < best_c:
            best_h = h2
            best_c = c2
        if time.time() < t_end_c - 5:
            h2, c2 = pairwise_swap_search(best_h, best_n, t_end_c - 2, seed + 99999, max_trials=20000)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        if time.time() < t_end_c - 3:
            h2, c2 = block_search(best_h, best_n, t_end_c - 1, seed + 88888, max_trials=5000)
            if c2 < best_c:
                best_h = h2
                best_c = c2

    # Final projection
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
