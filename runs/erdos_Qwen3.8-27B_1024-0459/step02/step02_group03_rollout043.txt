```python
def swap_delta_all(h, p, q):
    """Compute the change in correlation at ALL lags when h[p]:1->0, h[q]:0->1"""
    n = len(h)
    delta = np.zeros(2*n-1)
    for L in range(-n+1, n):
        d = 0.0
        # i=p contributes: -(1-h[p+L]) if valid
        if 0 <= p + L < n:
            d -= (1.0 - h[p + L])
        # i=q contributes: +(1-h[q+L]) if valid
        if 0 <= q + L < n:
            d += (1.0 - h[q + L])
        # i=p-L contributes: -h[p-L] if valid
        if 0 <= p - L < n:
            d -= h[p - L]
        # i=q-L contributes: +h[q-L] if valid
        if 0 <= q - L < n:
            d += h[q - L]
        delta[L + n - 1] = d
    return delta
```

```python
def swap_delta_vec(h, p, q):
    n = len(h)
    # For each lag index k in [0, 2n-2], L = k-(n-1)
    k = np.arange(2*n-1)
    L = k - (n-1)
    
    delta = np.zeros(2*n-1)
    
    # p+L in [0, n): L in [-p, n-1-p]
    mask1 = (L >= -p) & (L <= n-1-p)
    delta[mask1] -= (1.0 - h[p + L[mask1]])
    
    # q+L in [0, n): L in [-q, n-1-q]
    mask2 = (L >= -q) & (L <= n-1-q)
    delta[mask2] += (1.0 - h[q + L[mask2]])
    
    # p-L in [0, n): L in [p-n+1, p]
    mask3 = (L >= p-n+1) & (L <= p)
    delta[mask3] -= h[p - L[mask3]]
    
    # q-L in [0, n): L in [q-n+1, q]
    mask4 = (L >= q-n+1) & (L <= q)
    delta[mask4] += h[q - L[mask4]]
    
    return delta
```

```python
    mask3 = (L >= p - n + 1) & (L <= p)
    delta[mask3] -= h[p - L[mask3]]
    mask4 = (L >= q - n + 1) & (L <= q)
    delta[mask4] += h[q - L[mask4]]
    return delta

def binary_local_search(h_init, n, max_rounds, rng):
    h = to_binary(h_init, n).copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    no_improve = 0
    stall_limit = 200
    
    for rnd in range(max_rounds):
        top_m = min(15, 2 * n - 1)
        top_idx = np.argsort(-corr)[:top_m]
        top_lags = top_idx - (n - 1)
        current_max = np.max(corr)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        best_delta_max = 0.0
        best_p = -1
        best_q = -1
        
        # Targeted swaps: focus on worst lag
        L_worst = top_lags[0]
        if L_worst > 0:
            for i in range(min(50, n - L_worst)):
                if h[i] == 1 and h[i + L_worst] == 0:
                    for trial in range(8):
                        j_idx = rng.randint(n_zeros)
                        q = zeros[j_idx]
                        if q == i or abs(int(q) - (i + L_worst)) <= L_worst:
                            continue
                        delta = swap_delta_vec(h, i, q)
                        new_max = np.max(corr + delta)
                        if new_max < current_max - 1e-12:
                            if best_delta_max == 0.0 or new_max < best_delta_max:
                                best_delta_max = new_max
                                best_p = i
                                best_q = q
        elif L_worst < 0:
            Lp = -L_worst
            for i in range(min(50, n - Lp)):
                if h[i] == 0 and h[i + Lp] == 1:
                    for trial in range(8):
                        j_idx = rng.randint(n_zeros)
                        q = zeros[j_idx]
                        if q == i or abs(int(q) - (i + Lp)) <= Lp:
                            continue
                        delta = swap_delta_vec(h, q, i)
                        new_max = np.max(corr + delta)
                        if new_max < current_max - 1e-12:
                            if best_delta_max == 0.0 or new_max < best_delta_max:
                                best_delta_max = new_max
                                best_p = q
                                best_q = i
        
        # Random swaps
        n_random = 40
        for _ in range(n_random):
            p = ones[rng.randint(n_ones)]
            q = zeros[rng.randint(n_zeros)]
            delta = swap_delta_vec(h, p, q)
            new_max = np.max(corr + delta)
            if new_max < current_max - 1e-12:
                if best_delta_max == 0.0 or new_max < best_delta_max:
                    best_delta_max = new_max
                    best_p = p
                    best_q = q
        
        # Try accepting slightly worse moves (simulated annealing)
        if best_p < 0 and rnd < max_rounds // 2:
            temperature = 0.02 * (1.0 - rnd / max_rounds)
            for _ in range(10):
                p = ones[rng.randint(n_ones)]
                q = zeros[rng.randint(n_zeros)]
                delta = swap_delta_vec(h, p, q)
                new_max = np.max(corr + delta)
                if new_max < current_max + temperature:
                    if best_delta_max == 0.0 or new_max < best_delta_max:
                        best_delta_max = new_max
                        best_p = p
                        best_q = q
        
        if best_p >= 0:
            h[best_p] = 0.0
            h[best_q] = 1.0
            corr = compute_corr(h)
            new_c5 = np.max(corr) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > stall_limit:
            break
    
    return best_h, best_c5

def make_fourier_init(n, freq_list, rng):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5
    for freq in freq_list:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.15, 0.35)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_greedy_init(n, rng):
    """Greedy: place ones to minimize max autocorrelation"""
    h = np.zeros(n)
    target = n // 2
    available = np.arange(n)
    for _ in range(target):
        best_pos = -1
        best_score = np.inf
        # Try a random subset for efficiency
        if n > 100:
            candidates = rng.choice(available, size=min(30, len(available)), replace=False)
        else:
            candidates = available
        for pos in candidates:
            h[pos] = 1.0
            c = compute_corr(h)
            score = np.max(c)
            h[pos] = 0.0
            if score < best_score:
                best_score = score
                best_pos = pos
        h[best_pos] = 1.0
        available = np.delete(available, np.where(available == best_pos)[0])
    return h

def continuous_polish(h0, n, max_iter, beta):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -300, 0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -300, 0)
        exps = np.exp(shifted)
        weights = exps / np.sum(exps)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-14:
                continue
            L = k - (n - 1)
            if L == 0:
                grad += w * (1.0 - 2.0 * x)
            elif L > 0:
                grad[L:] += w * (1.0 - x[:n - L])
                grad[:n - L] -= w * x[L:]
            else:
                Lp = -L
                grad[:n - Lp] += w * (1.0 - x[Lp:])
                grad[Lp:] -= w * x[:n - Lp]
        return dx * grad
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, budget_s - (time.time() - start_time) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 200
    
    # Try to get parent
    parent_h = None
    try:
        if 'initial_h_values' in dir() or True:
            parent_h = np.array(initial_h_values, dtype=float)
            parent_n = len(parent_h)
            if parent_n > 0:
                pc5 = compute_c5(parent_h)
                best_h = parent_h.copy()
                best_c5 = pc5
                best_n = parent_n
                print("Parent: n={}, C5={:.6f}".format(parent_n, pc5))
    except Exception as e:
        print("No parent: {}".format(e))
    
    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.6f}".format(best_n, best_c5))
    
    # Resolution candidates
    res_candidates = [128, 160, 200, 240, 288, 320, 384]
    if best_n in res_candidates:
        res_candidates.remove(best_n)
    res_candidates = [best_n] + sorted(res_candidates)
    res_candidates = [r for r in res_candidates if 80 <= r <= 400 and r % 2 == 0]
    
    # Phase 1: Binary local search at multiple resolutions
    for n in res_candidates:
        if remaining() < 60:
            break
        print("=== Binary search n={} (remain {:.0f}s) ===".format(n, remaining()))
        
        # Multiple initializations
        inits = []
        if parent_h is not None:
            inits.append(interp_h(parent_h, n))
        inits.append(interp_h(best_h, n))
        inits.append(make_fourier_init(n, [1, 2, 3], rng))
        inits.append(make_fourier_init(n, [2, 3, 5], rng))
        
        # Greedy init (slower but potentially better)
        if remaining() > 200 and n <= 200:
            inits.append(make_greedy_init(n, rng))
        
        # Random binary
        h_rand = np.zeros(n)
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand[idx] = 1.0
        inits.append(h_rand)
        
        for h_init in inits:
            if remaining() < 30:
                break
            h_bin = to_binary(h_init, n)
            max_rounds = 8000 if remaining() > 200 else 4000
            h_best, c5_best = binary_local_search(h_bin, n, max_rounds, rng)
            if c5_best < best_c5:
                best_c5 = c5_best
                best_h = h_best.copy()
                best_n = n
                print("  Binary n={}: C5={:.6f}".format(n, best_c5))
    
    # Phase 2: Continuous polish from best binary solution
    if remaining() > 80:
        print("=== Continuous polish n={} (remain {:.0f}s) ===".format(best_n, remaining()))
        h_start = best_h.copy()
        for beta_val in [200, 500, 1000]:
            if remaining() < 30:
                break
            iters = 500 if beta_val <= 500 else 300
            try:
                h_pol, c5_pol = continuous_polish(h_start, best_n, iters, beta_val)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    print("  Polish beta={}: C5={:.6f}".format(beta_val, best_c5))
                h_start = h_pol
            except Exception:
                pass
    
    # Phase 3: Iterated local search with perturbations
    if remaining() > 60:
        print("=== Iterated search (remain {:.0f}s) ===".format(remaining()))
        for trial in range(20):
            if remaining() < 30:
                break
            # Perturb the best solution
            h_pert = best_h.copy()
            n_pert = rng.randint(5, max(6, best_n // 10))
            pert_idx = rng.choice(best_n, size=n_pert, replace=False)
            h_pert[pert_idx] = np.clip(h_pert[pert_idx] + rng.uniform(-0.4, 0.4, n_pert), 0, 1)
            h_pert = project_feasible(h_pert, best_n)
            h_bin = to_binary(h_pert, best_n)
            h_try, c5_try = binary_local_search(h_bin, best_n, 3000, rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                print("  Iter trial {}: C5={:.6f}".format(trial, best_c5))
            # Also try continuous from perturbed
            if c5_try < best_c5 + 0.001 and remaining() > 40:
                try:
                    h_c, c5_c = continuous_polish(h_try, best_n, 300, 500)
                    if c5_c < best_c5:
                        best_c5 = c5_c
                        best_h = h_c.copy()
                        print("  Iter+cont trial {}: C5={:.6f}".format(trial, best_c5))
                except Exception:
                    pass
    
    # Phase 4: Try slightly different resolutions around the best
    if remaining() > 40:
        for dn in [-16, 16, -32, 32]:
            n_try = best_n + dn
            if n_try < 80 or n_try > 400 or n_try % 2 != 0:
                continue
            if remaining() < 25:
                break
            h_interp = interp_h(best_h, n_try)
            h_bin = to_binary(h_interp, n_try)
            h_try, c5_try = binary_local_search(h_bin, n_try, 4000, rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
                print("  Res n={}: C5={:.6f}".format(n_try, best_c5))
    
    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
```
