- Before: contribution of q to overlap(k) is h[q]*(1-h[q+k]) = 0
- After: contribution of q is 1*(1-h[q+k]) = 1-h[q+k]
- Change from q: +(1-h[q+k])

  - q as second: h[q-k]*(1-h[q]) = h[q-k]*0 = 0 (before), h[q-k]*1 = h[q-k] (after)

Before swap: h[p]=1, h[q]=0
After swap: h[p]=0, h[q]=1

C_new[k] = Σ_i h_new[i] * (1 - h_new[i+t])

The only terms that change are those involving h[p] or h[q] (either as first or second element).

Terms involving p:

- h[p-t]*(1-h[p]): before = h[p-t]*(1-1) = 0, after = h[p-t]*(1-0) = h[p-t]. Change: +h[p-t]

Terms involving q:

- h[q-t]*(1-h[q]): before = h[q-t]*(1-0) = h[q-t], after = h[q-t]*(1-1) = 0. Change: -h[q-t]

= h[p+t] - 1 + h[p-t] + 1 - h[q+t] - h[q-t]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
        else:
            order = np.argsort(h)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def smooth_obj_fn(x, n, beta):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -500.0, 0.0)
    sm = m + np.log(np.sum(np.exp(sh))) / beta
    return sm * dx
