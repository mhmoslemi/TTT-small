import numpy as np
from scipy.optimize import minimize
import time

_N = 0
_BETA = 1.0
_P = 20
_TGT = 0.0
_START_TIME = 0.0
_BUDGET_S = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def obj_smooth(x):
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0 / _BETA, 0.0)
    return m + np.log(np.sum(np.exp(_BETA * shifted))) / _BETA


def obj_pnorm(x):
    n = _N
    dx = 2.0 / n
    corr = np.clip(np.correlate(x, 1.0 - x, mode='full') * dx, 0.0, 1.0)
    p = _P
    return np.sum(corr ** p) ** (1.0 / p)


def con_sum(x):
    return np.sum(x) - _TGT


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                zm = h == 0.0
                if np.sum(zm) > 0:
                    h[zm] = diff / np.sum(zm)
                    h = np.clip(h, 0.0, 1.0)
                else:
                    h[:] += diff / n
                    h = np.clip(h, 0.0, 1.0)
            else:
                om = h == 1.0
                if np.sum(om) > 0:
                    h[om] = 1.0 + diff / np.sum(om)
                    h = np.clip(h, 0.0, 1.0)
                else:
                    h[:] += diff / n
                    h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 5.0)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        h[i] = 1.0 if cnt % 2 == 1 else 0.0
    return h


def make_decorated_tm(n, rng):
    h = make_thue_morse(n).copy()
    for _ in range(3):
        j = rng.randint(n)
        h[j] = 1.0 - h[j]
    return h


def make_mixed(n, rng):
    h_tm = make_thue_morse(n)
    idx = rng.permutation(n)
    h_rand = np.zeros(n)
    h_rand[idx[:n // 2]] = 1.0
    w = rng.uniform(0.4, 0.7)
    h = w * h_tm + (1.0 - w) * h_rand
    return project_feasible(h, n)


def make_sine_init(n, freq, phase):
    t_arr = np.arange(n) / float(n)
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * t_arr + phase)
    return np.clip(h, 0.0, 1.0)


def subgradient_topk(h, k_top):
    n = len(h)
    corr = compute_corr_vals(h)
    topk_idx = np.argsort(corr)[-k_top:][::-1]
    center = n - 1
    grad = np.zeros(n)
    for idx in topk_idx:
        offset = idx - center
        if offset >= 0:
            n_pairs = n - offset
            h_shift = h[offset:offset + n_pairs]
            h_base = 1.0 - h[:n_pairs]
            grad[offset:offset + n_pairs] += h_base
            grad[:n_pairs] -= h_shift
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            h_shift = h[:n_pairs]
            h_base = 1.0 - h[abs_off:abs_off + n_pairs]
            grad[:n_pairs] += h_base
            grad[abs_off:abs_off + n_pairs] -= h_shift
    return grad / float(k_top)


def optimize_smooth(h0, n, bounds, cons, beta_list, max_iter_each=250):
    _N_local = n
    _N_save = _N
    global _N
    _N = _N_local
    h_opt = project_feasible(h0, n)
    c5_cur = compute_c5(h_opt)
    for bv in beta_list:
        if remaining_time() < 10:
            break
        _BETA = bv
        try:
            res = minimize(obj_smooth, h_opt, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter_each, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h_opt = h_new
                c5_cur = c5_new
        except Exception:
            pass
    _N = _N_save
    return h_opt, c5_cur


def optimize_pnorm(h0, n, bounds, cons, p_list, max_iter_each=200):
    _N_save = _N
    global _N
    _N = n
    h_opt = project_feasible(h0, n)
    c5_cur = compute_c5(h_opt)
    for pv in p_list:

        _P = pv
        try:
            res = minimize(obj_pnorm, h_opt, method='SLSQP',
                           bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h_opt = h_new
                c5_cur = c5_new
        except Exception:
            pass
    _N = _N_save
    return h_opt, c5_cur
