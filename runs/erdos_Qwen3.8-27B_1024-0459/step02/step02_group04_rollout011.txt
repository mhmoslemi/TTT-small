    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:

    lo = h.min() - 1.0

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:
        n = int(n)
    if n <= 0:
        return np.empty(0)
    h = h[:n].copy()
    if h.size < n:
        h = np.concatenate((h, np.zeros(n - h.size)))
    target = 0.5 * float(n)
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(30):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-13
        else:
            mask = out > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def all_lag_values(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    nvals = 2 * n - 1
    vals = np.empty(nvals, dtype=float)
    lags = np.empty(nvals, dtype=int)
    vals[0] = (total - float(np.dot(h, h))) * dx
    lags[0] = 0
    idx = 1
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals[idx] = (pref[m] - ac) * dx
        lags[idx] = L
        idx += 1
        vals[idx] = (total - pref[L] - ac) * dx
        lags[idx] = -L
        idx += 1
    return vals, lags


def smooth_obj_and_grad(h, beta):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    vals, lags = all_lag_values(h)
    maxv = float(vals.max())
    shifted = beta * (vals - maxv)
    shifted = np.clip(shifted, -745.0, 0.0)
    exps = np.exp(shifted)
    total_exp = float(exps.sum())
    if total_exp <= 0:
        return maxv, np.zeros(n)
    lse = maxv + np.log(total_exp) / beta
    w = exps / total_exp
    g = np.zeros(n, dtype=float)
    w0 = float(w[0]) * dx
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    for i in range(1, len(lags)):
        wi = float(w[i]) * dx
        if wi < 1e-14:
            continue
        L = int(lags[i])
        m = n - abs(L)
        if L > 0:
            g[:m] += wi * (1.0 - h[L:])
            g[L:] += -wi * h[:m]
        else:
            g[-L:] += wi * (1.0 - h[:m])
            g[:m] += -wi * h[-L:]
    return lse, g


def _smooth_obj(h, beta):
    v, _ = smooth_obj_and_grad(h, beta)
    return v


def _smooth_grad(h, beta):
    _, g = smooth_obj_and_grad(h, beta)
    return g


def _sum_eq(h, n):
    return float(np.sum(h) - 0.5 * float(n))


def _sum_eq_jac(h, n):
    return np.ones(h.size, dtype=float)


def optimize_smooth_slqp(h0, n, beta, maxiter=80):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _sum_eq(x, n), 'jac': lambda x: _sum_eq_jac(x, n)}]
    try:
        res = minimize(_smooth_obj, h, args=(beta,),
                       jac=lambda x: _smooth_grad(x, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def lag_value(h, l):
    n = len(h)
    if l == 0:
        return float(np.sum(h) - float(np.dot(h, h)))
    L = abs(int(l))
    m = n - L
    low = h[:m]
    high = h[L:]
    ac = float(np.dot(low, high))
    if l > 0:
        return float(np.sum(low) - ac)
    return float(np.sum(high) - ac)


def lag_grad(h, l):
    n = len(h)
    if l == 0:
        return 1.0 - 2.0 * h
    L = abs(int(l))
    m = n - L
    low = h[:m]
    high = h[L:]
    g = np.zeros(n, dtype=float)
    if l > 0:
        g[:m] += 1.0 - high
        g[L:] += -low
    else:
        g[-L:] += 1.0 - low
        g[:m] += -high
    return g


def _active_t_obj(x):
    return float(x[-1])


def _active_t_obj_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j


def _active_sum_eq(x, n):
    return float(np.sum(x[:-1]) - 0.5 * float(n))


def _active_sum_eq_jac(x, n):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j


def _active_ineq(x, n, lags):
    dx = 2.0 / float(n)
    h = x[:-1]
    m = len(lags)
    out = np.empty(m, dtype=float)
    for i, ll in enumerate(lags):
        out[i] = float(x[-1]) - lag_value(h, int(ll)) * dx
    return out


def _active_ineq_jac(x, n, lags):
    dx = 2.0 / float(n)
    m = len(lags)
    J = np.zeros((m, n + 1), dtype=float)
    h = x[:-1]
    for i, ll in enumerate(lags):
        g = lag_grad(h, int(ll))
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def refine_active_lags(h0, n, active_lags, t0, maxiter=60):
    h = np.asarray(h0, dtype=float)
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = float(t0)
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [{'type': 'eq', 'fun': lambda x: _active_sum_eq(x, n),
             'jac': lambda x: _active_sum_eq_jac(x, n)}]
    if len(active_lags) > 0:
        cons.append({'type': 'ineq', 'fun': lambda x: _active_ineq(x, n, active_lags),
                     'jac': lambda x: _active_ineq_jac(x, n, active_lags)})
    try:
        res = minimize(_active_t_obj, x0, jac=_active_t_obj_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x[:-1], n)
    except Exception:
        pass
    return h


def choose_active(vals, lags, count, margin=5e-5):
    order = np.argsort(-vals)
    take = min(count, len(vals))
    idx = order[:take]
    if margin > 0:
        top = float(vals[order[0]])
        near = np.where(vals >= top - margin)[0]
        idx = np.unique(np.concatenate((idx, near)))
        idx = idx[np.argsort(-vals[idx])]
    return [int(lags[i]) for i in idx]


def active_refine_loop(h0, n, time_limit, seq=(16, 32, 64, 128), maxiter=50):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    for acnt in seq:
        if time.time() > time_limit - 3.0:
            break
        for _pass in range(2):
            if time.time() > time_limit - 3.0:
                break
            vals, lags = all_lag_values(h)
            cur_c = float(np.max(vals))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            active = choose_active(vals, lags, int(acnt), margin=5e-5)
            if not active:
                active = [int(lags[int(np.argmax(vals))])]
            t0 = cur_c + 1e-5
            h2 = refine_active_lags(h, n, active, t0, maxiter=maxiter)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            if c2 <= cur_c + 1e-9:
                h = h2
    return best_h, best_c


def local_search(h0, n, time_limit, seed, step_init=0.008):

    h = project_feasible(h0, n)
    c = compute_c5(h)

    cur_c = c
    temp = 0.0008
    min_temp = 1e-9
    step = step_init
    trials = 0
    while time.time() < time_limit:
        trials += 1
        r = rng.random()
        if r < 0.25:
            i = rng.randint(n)
            idx = np.array([i])
            delta = np.array([rng.randn() * step])
        elif r < 0.5:
            i = rng.randint(n - 1)
            idx = np.array([i, i + 1])
            delta = rng.randn(2) * step
            delta -= delta.mean()
        elif r < 0.8:
            L = rng.randint(2, 6)
            i = rng.randint(0, n - L)
            idx = np.arange(i, i + L)
            delta = rng.randn(L) * step * 0.8
            delta -= delta.mean()
        else:
            L = rng.randint(4, 12)
            i = rng.randint(0, n - L)

            delta = rng.randn(L) * step * 0.5
            delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try

            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.random() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.9995, min_temp)

    return best_h, best_c


def coord_refine(h0, n, time_limit):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    improved = True
    while improved and time.time() < time_limit:
        improved = False
        for i in range(n):
            if time.time() > time_limit:
                break
            for frac in (0.5, 0.25, -0.5, -0.25, 0.1, -0.1):
                h_try = h.copy()
                h_try[i] += frac * 0.01
                h_try = project_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-14:
                    best_h = h_try.copy()
                    best_c = c_try
                    h = h_try
                    improved = True
                    break
                elif c_try < c - 1e-14:
                    h = h_try
                    c = c_try
                    improved = True
                    break
    return best_h, best_c


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_src = h_src.size
    n_dst = int(n_dst)
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def gaussian_smooth(h, sigma=1.2):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 3:
        return h.copy()
    length = min(5, n)
    if length % 2 == 0:
        length -= 1
    length = max(3, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-0.5 * (k / sigma) ** 2)
    w /= w.sum()
    return np.convolve(h, w, mode="same")


def build_versus(h, n, rng, count=5):
