def pair_cd(h, n, n_passes, time_limit, rng):

            h[j] = 0.0
            h[l] = 0.0

            h[j] = 1.0
            h[l] = 0.0
            c1 = get_corr(h)
            h[j] = 0.0
            h[l] = 1.0
            c2 = get_corr(h)
            h[:] = hs
            
            v = c1 - c2
            w = c0 + c * (c2 - c0)
            
            t_lo = max(0.0, c - 1.0)
            t_hi = min(1.0, c)
            if t_lo >= t_hi - 1e-15:
                t_opt = 0.5 * c
            else:
                lo, hi = t_lo, t_hi
                for _ in range(60):

                    elif sg < -1e-12: lo = mid

        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:

            best_h = h.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
