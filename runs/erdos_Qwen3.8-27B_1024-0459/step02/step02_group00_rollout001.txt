def project_box_sum(h, target):
    x = np.array(h, dtype=float, copy=True)
    n = len(x)
    if n == 0:
        return x
    target = float(target)

    if (np.min(x) >= -1e-12) and (np.max(x) <= 1.0 + 1e-12):
        s = float(np.sum(x))
        if abs(s - target) < 1e-10:
            y = np.clip(x, 0.0, 1.0)
            diff = target - float(np.sum(y))
            if abs(diff) > 1e-14:
                mask = (y > 1e-12) & (y < 1.0 - 1e-12)
                if np.any(mask):
                    y[mask] += diff / float(np.count_nonzero(mask))
                    y = np.clip(y, 0.0, 1.0)

def make_thue_morse(n):

def _smooth_fg_calc(h, beta):
    h = np.asarray(h, dtype=float)
    n = len(h)
    dx = 2.0 / float(n)
    plus, minus = plus_minus(h)

    m = float(max(np.max(plus), np.max(minus)))
    if beta <= 1e-12:
        f = dx * m
        grad = np.zeros(n, dtype=float)
        return f, grad

    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        ep = np.exp(beta * (plus - m))
        em = np.exp(beta * (minus - m))
    em[0] = 0.0
    W = float(np.sum(ep) + np.sum(em))
    if (not np.isfinite(W)) or W <= 0.0:
        return dx * m, np.zeros(n, dtype=float)

    ep = ep / W
    em = em / W
    f = dx * (m + np.log(W) / beta)

    grad = np.zeros(n, dtype=float)
    thresh = 1e-12

    w0 = float(ep[0])
    if w0 > thresh:
        grad += w0 * (1.0 - 2.0 * h)

    idx = np.nonzero(ep[1:] > thresh)[0]

        d = int(ii) + 1
        w = float(ep[d])
        L = n - d

        L = n - d
        grad[:L] += w * (1.0 - h[d:d + L])

def _hinge_fg_calc(h, b, scale):
    h = np.asarray(h, dtype=float)
    n = len(h)
    plus, minus = plus_minus(h)

    total = 0.0
    grad = np.zeros(n, dtype=float)

    e = float(plus[0] - b)
    if e > 0.0:
        total += e * e
        w = 2.0 * scale * e
        grad += w * (1.0 - 2.0 * h)

    pe = plus[1:] - b
    act = pe > 0.0
    if np.any(act):
        idx = np.nonzero(act)[0]
        total += float(np.sum(pe[act] * pe[act]))
        for ii in idx:
            d = int(ii) + 1
            e = float(pe[ii])
            w = 2.0 * scale * e
            L = n - d
            grad[d:d + L] += w * (1.0 - h[:L])
            grad[:L] -= w * h[d:d + L]

    me = minus[1:] - b
    act = me > 0.0
    if np.any(act):
        idx = np.nonzero(act)[0]
        total += float(np.sum(me[act] * me[act]))
        for ii in idx:
            d = int(ii) + 1
            e = float(me[ii])
            w = 2.0 * scale * e
            L = n - d
            grad[:L] += w * (1.0 - h[d:d + L])
            grad[d:d + L] -= w * h[:L]

    return float(scale * total), grad

def projected_gradient_smooth(h0, n, beta, max_iter, step):
    global _N, _TGT, _BETA, _CACHE_X
    max_iter = int(max_iter)
    target = 0.5 * float(n)
    if max_iter <= 0 or remaining_time() < 25.0:
        h = project_box_sum(h0, n, target)
        return h, float(compute_c5(h))

    _N = int(n)
    _TGT = target
    _BETA = float(beta)
    _CACHE_X = None

    h = project_box_sum(h0, n, target)
    cur = float(compute_c5(h))
    scale = float(step)

    for _ in range(max_iter):
        if remaining_time() < 4.0:
            break
        _, grad = smooth_fg_cached(h)
        gnorm = float(np.linalg.norm(grad))
        if gnorm <= 1e-12:
            break

        d = -grad / gnorm
        d = d - float(np.mean(d))

        improved = False
        for s in (scale, 0.5 * scale, 2.0 * scale, 0.25 * scale):
            p = project_box_sum(h + s * d, n, target)
            c = float(compute_c5(p))
            if c < cur - 1e-14:
                h = p
                cur = c
                scale = s
                improved = True
                break

        if not improved:
            scale *= 0.5
            if scale < 1e-4:
                break

    return h, cur


def current_sym_limit(n, lim):
    rem = remaining_time()
    base = int(lim)
    if rem < 100.0:
        return max(16, base // 4)
    if rem < 200.0:
        return max(24, base // 2)
    if int(n) > 256 and base > 96:
        return 96
    return base


def improve_symmetries(h, n, limit_eval):
    target = 0.5 * float(n)
    n = int(n)
    best_h = project_box_sum(h, n, target)
    best_c = float(compute_c5(best_h))

    step = max(1, n // max(1, int(limit_eval)))

    for _ in range(2):
        old_c = best_c
        bases = [best_h]

        comp = np.clip(1.0 - best_h, 0.0, 1.0)
        bases.append(project_box_sum(comp, n, target))

        if n <= 288:
            rev = np.ascontiguousarray(best_h[::-1])
            bases.append(project_box_sum(rev, n, target))

        for base in bases:
            for s in range(0, n, step):
                if s == 0:
                    hp = base
                    c = float(compute_c5(hp))
                else:
                    hp = np.roll(base, s)
                    c = float(compute_c5(hp))

                if c < best_c - 1e-15:
                    best_c = c
                    best_h = project_box_sum(hp, n, target)

        if best_c >= old_c - 1e-15:
            break

    return best_h, best_c


def unique_desc(vals, eps=1e-6):
    if not vals:
        return []
    out = sorted(vals, reverse=True)
    res = []
    for v in out:
        v = float(v)
        if not res:
            res.append(v)
        elif abs(res[-1] - v) > float(eps):
            res.append(v)
    return res


def polish_at_n(h_init, n, rng, tier):
    n = int(n)
    target = 0.5 * float(n)
    h = project_box_sum(h_init, n, target)
    cur = float(compute_c5(h))

    if tier == 1:
        lim = n if n <= 192 else (128 if n <= 256 else 64)
        lb_betas = (20.0, 80.0)
        sl_betas = (30.0, 80.0, 200.0, 500.0, 1200.0)
        mi_smooth = 130 if n <= 192 else (90 if n <= 256 else 60)
        deltas = (0.003, 0.006, 0.010, 0.016, 0.024, 0.035, 0.050)
        base_targets = (0.38082, 0.38080, 0.38076, 0.38072, 0.38068)
        max_targets = 8
        gate = 70
    else:
        lim = n if n <= 160 else 64
        lb_betas = (60.0,)
        sl_betas = (80.0, 300.0)
        mi_smooth = 70 if n <= 224 else (50 if n <= 288 else 35)
        deltas = (0.005, 0.012, 0.025)
        base_targets = (0.38080, 0.38075)
        max_targets = 4
        gate = 40

    if remaining_time() > 30.0:
        h, cur = improve_symmetries(h, n, current_sym_limit(n, lim))

    for i in range(len(lb_betas)):
        if remaining_time() < 45.0:
            break
        h2, c2 = lbfgsb_smooth(h, n, lb_betas[i], 2.0, 80 if tier == 1 else 50)
        if c2 < cur - 1e-14:
            h, cur = h2, c2
            if remaining_time() > 35.0:
                h, cur = improve_symmetries(h, n, current_sym_limit(n, lim))

    for beta in sl_betas:
        if remaining_time() < float(gate):
            break
        h2, c2 = slqp_smooth(h, n, beta, mi_smooth)
        if c2 < cur - 1e-14:
            h, cur = h2, c2
            if remaining_time() > 35.0:
                h, cur = improve_symmetries(h, n, current_sym_limit(n, lim))

    curq = cur * float(n) * 0.5
    bvals = []
    for d in deltas:
        bvals.append(curq - d)
    for tc in base_targets:
        bvals.append(tc * float(n) * 0.5)
    bvals = unique_desc(bvals, 1e-6)

    attempts = 0
    for B in bvals:
        if attempts >= max_targets:
            break
        if remaining_time() < (60.0 if tier == 1 else 35.0):
            break
        curq = cur * float(n) * 0.5
        B = float(B)
        if B >= curq - 1e-4:
            continue
        attempts += 1

        diff = curq - B
        if diff < 0.015:
            scale = 3000.0
        elif diff < 0.04:
            scale = 1500.0
        else:
            scale = 700.0

        hinge_mi = mi_smooth if tier == 1 else (50 if n <= 224 else 35)
        c_before = cur
        h2, c2 = slqp_hinge(h, n, B, scale, hinge_mi)
        improved = False
        if c2 < c_before - 1e-14:
            h, cur = h2, c2
            improved = True

        if tier == 1 and remaining_time() > 70.0:
            h3, c3 = lbfgsb_hinge(h, n, B, 5.0, 60)
            if c3 < cur - 1e-14:
                h, cur = h3, c3
            elif c3 > cur - 1e-14 and c2 >= c_before - 1e-14:
                pass

        if improved and remaining_time() > 40.0:
            h, cur = improve_symmetries(h, n, current_sym_limit(n, lim))

    if tier == 1 and remaining_time() > 90.0:
        for beta in (1500.0, 3000.0):
            if remaining_time() < 60.0:
                break
            h2, c2 = slqp_smooth(h, n, beta, 60)
            if c2 < cur - 1e-14:
                h, cur = h2, c2
                if remaining_time() > 40.0:
                    h, cur = improve_symmetries(h, n, current_sym_limit(n, lim))

    if tier == 1 and remaining_time() > 40.0:
        h2, c2 = projected_gradient_smooth(h, n, 1000.0, 80, 0.05)
        if c2 < cur - 1e-14:
            h, cur = h2, c2
            if remaining_time() > 30.0:
                h, cur = improve_symmetries(h, n, current_sym_limit(n, lim))

    if remaining_time() > 20.0:
        h, cur = improve_symmetries(h, n, current_sym_limit(n, lim))

    return h, float(cur)


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = float(budget_s)
    rng = np.random.RandomState(int(seed))

    try:
        parent = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if parent.size < 8:
            raise ValueError("parent too small")
        first_n = int(parent.size)
        best_h = project_box_sum(parent, first_n, 0.5 * float(first_n))
    except Exception:
        first_n = 160
        best_h = make_thue_morse(first_n)
        best_h = project_box_sum(best_h, first_n, 0.5 * float(first_n))

    best_n = int(len(best_h))
    best_c5 = float(compute_c5(best_h))

    if remaining_time() > 20.0:
        lim = best_n if best_n <= 192 else (128 if best_n <= 256 else 64)
        best_h, best_c5 = improve_symmetries(best_h, best_n, lim)

    if remaining_time() > 120.0:
        h2, c2 = polish_at_n(best_h, best_n, rng, 1)
        if c2 < best_c5:
            best_h, best_n, best_c5 = h2, int(len(h2)), c2

    for scale in (0.10, 0.18):
        if remaining_time() < 110.0:
            break
        pert = best_h + rng.randn(best_n) * scale
        pert = project_box_sum(pert, best_n, 0.5 * float(best_n))
        if float(compute_c5(pert)) < best_c5 + 0.02:
            h2, c2 = polish_at_n(pert, best_n, rng, 0)
            if c2 < best_c5:
                best_h, best_n, best_c5 = h2, int(len(h2)), c2

    if remaining_time() > 120.0:
        h0 = make_thue_morse(best_n)
        h0 = project_box_sum(h0, best_n, 0.5 * float(best_n))
        if float(compute_c5(h0)) < best_c5 + 0.03:
            h2, c2 = polish_at_n(h0, best_n, rng, 0)
            if c2 < best_c5:
                best_h, best_n, best_c5 = h2, int(len(h2)), c2

    preferred = []
    for d in (32, 16, 48, 64, -16, -32, 80, -48, 96):
        nn = first_n + d
        if 96 <= nn <= 320 and nn not in preferred:
            preferred.append(nn)

    for nn in (144, 152, 160, 168, 176, 184, 192, 200, 208,
               216, 224, 232, 240, 248, 256, 272, 288, 304, 320):
        if 96 <= nn <= 320 and nn not in preferred:
            preferred.append(nn)

    processed = set()
    processed.add(first_n)

    for nn in preferred:
        nn = int(nn)
        if nn in processed:
            continue
        if remaining_time() < 150.0:
            break
        if nn == best_n:
            processed.add(nn)
            continue

        h0 = interp_to(best_h, best_n, nn)
        h0 = project_box_sum(h0, nn, 0.5 * float(nn))
        h2, c2 = polish_at_n(h0, nn, rng, 0)
        processed.add(nn)

        if c2 < best_c5:
            best_h = h2
            best_n = int(len(h2))
            best_c5 = float(c2)

            if remaining_time() > 130.0:
                h3, c3 = polish_at_n(best_h, best_n, rng, 1)
                processed.add(best_n)
                if c3 < best_c5:
                    best_h = h3
                    best_n = int(len(h3))
                    best_c5 = float(c3)

    if remaining_time() > 100.0:
        h2, c2 = polish_at_n(best_h, best_n, rng, 1)
        if c2 < best_c5:
            best_h, best_n, best_c5 = h2, int(len(h2)), c2

    if remaining_time() > 40.0:
        curq = best_c5 * float(best_n) * 0.5
        B = 0.38078 * float(best_n) * 0.5
        if B < curq - 1e-5:
            h2, c2 = slqp_hinge(best_h, best_n, B, 3000.0, 60)
            if c2 < best_c5:
                best_h = h2
                best_n = int(len(h2))
                best_c5 = float(c2)

    if remaining_time() > 15.0:
        lim = 160 if best_n <= 192 else (96 if best_n <= 256 else 48)
        best_h, best_c5 = improve_symmetries(best_h, best_n, lim)

    best_h = project_box_sum(best_h, best_n, 0.5 * float(best_n))
    final_c5 = float(compute_c5(best_h))

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if np.isfinite(eval_c5) and float(eval_c5) < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    return (best_h, final_c5, int(best_n))
