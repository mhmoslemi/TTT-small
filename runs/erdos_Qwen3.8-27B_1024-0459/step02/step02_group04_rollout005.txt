def active_lags_from_corr(c, n, max_count, margin=1e-6):
    c=np.asarray(c)
    m=float(np.max(c))
    order=np.argsort(-c)
    take=min(int(max_count), int(c.size))
    top=order[:take]
    near=np.where(c >= m - max(0.0, margin))[0]
    if near.size:
       idx=np.unique(np.concatenate((top,near)))
    else:
       idx=top
    cap=max(int(max_count)*2, 64)
    if idx.size > cap:
       idx=idx[np.argsort(-c[idx])][:cap]
    idx=idx[np.argsort(-c[idx])]
    lags=(idx - (n-1)).astype(int)
    return [int(v) for v in lags]

def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:
        n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)

    target = 0.5 * float(n)
    h = h[:n].copy()
    if h.size < n:
        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    elif h.size > n:
        h = h[:n]

    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)

    for _ in range(40):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-13
        else:
            mask = out > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)

    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_obj_grad_fast(h, beta):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n <= 0:
        return 0.0, np.empty(0, dtype=float)

    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr * dx
    m = float(np.max(v))
    b = max(1.0, float(beta))

    d = b * (v - m)
    d = np.clip(d, -745.0, 0.0)
    expd = np.exp(d)
    r = float(expd.sum())
    if r <= 0.0:
        return m, np.zeros(n, dtype=float)

    val = m + np.log(r) / b
    w = expd / r

    g = np.zeros(n, dtype=float)
    idx0 = n - 1

    c0 = float(w[idx0]) * dx
    if c0 != 0.0:
        g += c0 * (1.0 - 2.0 * h)

        wp = float(w[idx0 - L]) * dx
        wn = float(w[idx0 + L]) * dx
        s = wp + wn
        if s <= 1e-15:

        mlen = n - L
        low = h[:mlen]
        high = h[L:]
        g[:mlen] += wn - s * high
        g[L:] += wp - s * low

    return float(val), g

def lag_value(h, l):
    n = len(h)
    L = abs(int(l))
    if L == 0:
        return float(np.sum(h) - float(np.dot(h, h)))
    m = n - L
    if m <= 0:
        return 0.0
    low = h[:m]
    high = h[L:]
    ac = float(np.dot(low, high))
    if l > 0:
        return float(np.sum(low) - ac)
    return float(np.sum(high) - ac)


def lag_gradient_only(h, l):
    n = len(h)
    if l == 0:
        return 1.0 - 2.0 * h
    L = abs(int(l))
    m = n - L
    g = np.zeros(n, dtype=float)

    low = h[:m]
    high = h[L:]
    if l > 0:
        g[:m] += 1.0 - high
        g[L:] += -low
    else:
        g[L:] += 1.0 - low
        g[:m] += -high
    return g

def optimize_active_t(h0, active_lags, t0, maxiter=50, ftol=1e-12):

def active_lags_from_corr(c, n, max_count, margin=1e-6):

    top = order[:take]
    near = np.where(c >= m - max(0.0, float(margin)))[0]
    if near.size > 0:
        idx = np.unique(np.concatenate((top, near)))
    else:
        idx = top
    cap = max(int(max_count) * 2, 64)
    if idx.size > cap:
        idx = idx[np.argsort(-c[idx])][:cap]
    idx = idx[np.argsort(-c[idx])]
    lags = (idx - (n - 1)).astype(int)
    return [int(v) for v in lags]


def refine_active(h0, n, time_limit, active_seq=(16, 32, 64), maxiter=50):
    n = int(n)
    cur_h = project_feasible(h0, n)
    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)

    for acnt in active_seq:
        if time.time() > time_limit:
            break
        passes = 2 if int(acnt) <= 32 else 1
        for _ in range(passes):
            if time.time() > time_limit:
                break
            cvals = np.correlate(cur_h, 1.0 - cur_h, mode="full") * (2.0 / float(n))
            current = float(np.max(cvals))
            if current < best_c:
                best_c = current
                best_h = cur_h.copy()

            margin = max(1e-6, 3e-4 * max(1.0, current))
            active = active_lags_from_corr(cvals, n, int(acnt), margin=margin)
            if len(active) == 0:
                break

            t0 = current + 1e-8
            res = optimize_active_t(cur_h, active, t0, maxiter=int(maxiter), ftol=1e-12)
            accepted = False
            if res is not None and getattr(res, "x", None) is not None:
                x = res.x
                if x is not None and x.size == n + 1 and np.all(np.isfinite(x)):
                    h2 = project_feasible(x[:-1], n)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2.copy()
                        accepted = True
                    if c2 <= current + 1e-8:
                        cur_h = h2
                        accepted = True
            if not accepted:
                break

    return best_h, float(best_c)


def interpolate_to_n(h_src, n_dst):
    n_dst = int(n_dst)
    h_src = np.asarray(h_src, dtype=float).reshape(-1)
    n_src = h_src.size
    if n_dst <= 0:
        return np.empty(0, dtype=float)
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def gaussian_smooth(h, sigma=1.5):
    h = np.asarray(h, dtype=float).reshape(-1)
    n = h.size
    if n <= 1:
        return h.copy()
    length = min(7, n)
    if length % 2 == 0:
        length -= 1
    length = max(1, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * float(sigma) * float(sigma)))
    w /= w.sum()
    return np.convolve(h, w, mode="same")


def analytic_starts(n):
    n = int(n)
    if n <= 0:
        return []
    out = []
    x = np.linspace(0.0, 2.0, n, endpoint=False)

    out.append(project_feasible(np.full(n, 0.5), n))
    out.append(project_feasible(0.5 + 0.5 * np.cos(np.pi * x), n))
    out.append(project_feasible(0.5 - 0.5 * np.cos(np.pi * x), n))
    out.append(project_feasible(1.0 - np.abs(x - 1.0), n))

    h = np.zeros(n, dtype=float)
    q = max(1, n // 4)
    h[:q] = 1.0
    h[n // 2:n // 2 + q] = 1.0
    out.append(project_feasible(h, n))

    h = np.zeros(n, dtype=float)
    if n % 4 == 0:
        pat = np.tile(np.asarray([1.0, 1.0, 0.0, 0.0]), n // 4)
        out.append(project_feasible(pat, n))

    h = np.zeros(n, dtype=float)
    if n > 0:
        parity = np.zeros(n, dtype=int)
        for i in range(1, n):
            parity[i] = 1 - parity[i >> 1]
        h[parity == 0] = 1.0
        out.append(project_feasible(h, n))

    return out


def build_inits_for_n(n, parent, best_h, best_n, rng, count=4):
    n = int(n)
    inits = []
    if parent is not None:
        if parent.size == n:
            inits.append(np.asarray(parent, dtype=float).copy())
        else:
            inits.append(interpolate_to_n(parent, n))

    base = inits[0] if len(inits) > 0 else project_feasible(np.full(n, 0.5), n)

    if best_h is not None:
        if best_h.size == n:
            inits.append(np.asarray(best_h, dtype=float).copy())
        else:
            inits.append(interpolate_to_n(best_h, n))

    out = []
    out.append(project_feasible(base, n))

    if n > 1:
        flip = base[::-1]
        out.append(project_feasible(0.5 * (base + flip), n))

    out.append(project_feasible(gaussian_smooth(base, 1.3), n))
    out.append(project_feasible(gaussian_smooth(base, 0.7), n))

    pert = base + rng.randn(n) * 0.03
    pert = pert - float(np.mean(pert))
    out.append(project_feasible(pert, n))

    an = analytic_starts(n)
    if len(an) > 0:
        out.append(an[0])

    out = [project_feasible(hh, n) for hh in out]
    if count is not None and len(out) > int(count):
        out = out[:int(count)]
    return out


def smooth_betas_for_n(n):
    n = int(n)
    if n <= 96:
        return (6, 20, 60, 160, 320)
    if n <= 160:
        return (8, 25, 80, 200, 400)
    if n <= 220:
        return (6, 20, 60, 150)
    return (5, 16, 50, 120)


def active_seq_for_n(n):
    n = int(n)
    if n <= 96:
        return (16, 32, 64)
    if n <= 160:
        return (12, 24, 48, 64)
    if n <= 220:
        return (12, 24, 48)
    return (8, 16, 32)


def continuous_refine(h0, n, time_limit, seed):
    n = int(n)
    cur_h = project_feasible(h0, n)
    best_h = cur_h.copy()
    best_c = compute_c5(cur_h)

    if time.time() > time_limit:
        return best_h, float(best_c)

    betas = smooth_betas_for_n(n)
    avail = float(time_limit - time.time())

    if avail < 40.0:
        use_betas = (betas[0],)
        maxit = 20
    elif avail < 90.0:
        use_betas = betas[:2]
        maxit = 40
    else:
        use_betas = betas
        maxit = 60 if n <= 160 else 30

    for beta in use_betas:
        if time.time() > time_limit - 30.0:
            break
        avail2 = float(time_limit - time.time())
        if avail2 < 25.0:
            break
        mi = int(maxit)
        if avail2 < 35.0:
            mi = min(mi, 12)
        elif avail2 < 60.0:
            mi = min(mi, 25)
        h2, c2 = optimize_smooth_fast(cur_h, n, beta, maxiter=mi, ftol=1e-11)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
        if c2 <= best_c + 2e-4:
            cur_h = h2
        else:
            cur_h = best_h

    if time.time() < time_limit - 20.0:
        rem = float(time_limit - time.time())
        seq = active_seq_for_n(n)
        if rem > 55.0:
            act_end = time.time() + min(rem - 5.0, 90.0)
            maxit_act = 50 if n <= 160 else 25
        elif rem > 25.0:
            act_end = time.time() + max(5.0, rem - 5.0)
            maxit_act = 20 if n <= 160 else 10
        else:
            maxit_act = 0

        if maxit_act > 0:
            h2, c2 = refine_active(cur_h, n, act_end, active_seq=seq, maxiter=maxit_act)
            if c2 < best_c:
                best_h = h2
                best_c = c2

    return best_h, float(best_c)


def process_initial_set(inits, n, slot_end, seed):
    best_h = None
    best_c = np.inf
    m = len(inits)

    for j, h0 in enumerate(inits):
        if time.time() >= slot_end:
            break
        rem = float(slot_end - time.time())
        if rem <= 0.0:
            break
        per = rem / float(max(1, m - j))
        per = min(90.0, per)
        per = max(10.0, per)
        per = min(per, rem)
        hh, cc = continuous_refine(h0, n, time.time() + per, seed + 7919 * j)
        if hh is not None and cc < best_c:
            best_h = hh
            best_c = cc

    if best_h is None:
        if len(inits) > 0:
            best_h = project_feasible(inits[0], n)
            best_c = compute_c5(best_h)
        else:
            best_h = project_feasible(np.full(int(n), 0.5), n)
            best_c = compute_c5(best_h)

    return best_h, float(best_c)


def local_block_perturb(h0, n, time_limit, seed, step=0.02):
    n = int(n)
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    cur_h = h.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    temp = 2e-4
    min_temp = 1e-8

    max_trials = 5000
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(8, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * float(step)
        delta -= delta.mean()

        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(2e-3, max(temp * 1.15, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                p = np.exp(-d / temp)
                if rng.rand() < p:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.995, min_temp)

    return best_h, float(best_c)


def binary_threshold(vals, n):
    n = int(n)
    target = n // 2
    v = np.asarray(vals, dtype=float).reshape(-1)
    if v.size < n:
        pad = np.zeros(n - v.size, dtype=float)
        v = np.concatenate((v, pad))
    elif v.size > n:
        v = v[:n]
    v = np.where(np.isfinite(v), v, 0.0)

    if target <= 0:
        return np.zeros(n, dtype=np.int16)
    if target >= n:
        return np.ones(n, dtype=np.int16)

    order = np.argsort(-v)[:target]
    a = np.zeros(n, dtype=np.int16)
    a[order] = 1
    return a


def binary_score(c):
    M = int(np.max(c))
    cc = c.astype(np.int64)
    s = int(np.dot(cc, cc))
    return M * 100000000 + s


def binary_local_search(n, start_a, time_limit, seed):
    n = int(n)
    if n < 2:
        return None, 0

    rng = np.random.RandomState(int(seed))
    a = np.asarray(start_a, dtype=np.int16).reshape(-1).copy()
    if a.size < n:
        a = np.concatenate((a, np.zeros(n - a.size, dtype=np.int16)))
    elif a.size > n:
        a = a[:n]

    target = n // 2
    if int(np.sum(a)) != target:
        a = binary_threshold(a.astype(float), n)

    cur_c = np.correlate(a, 1 - a, mode="full")
    best_a = a.copy()
    best_score = int(binary_score(cur_c))
    best_count = int(np.max(cur_c))
    cur_score = best_score
    no_improve = 0

    while time.time() < time_limit:
        Mcur = int(np.max(cur_c))
        accepted = False

        hot = np.where(cur_c >= Mcur - 1)[0]
        if hot.size > 0:
            rng.shuffle(hot)
            evaluated = 0
            for idx in hot:
                if evaluated >= 16:
                    break
                idx = int(idx)
                if idx == n - 1:
                    continue
                if idx < n - 1:
                    L = n - 1 - idx
                    m = n - L
                    if m <= 0:
                        continue
                    pairs = np.nonzero((a[:m] == 0) & (a[L:] == 1))[0]
                else:
                    L = idx - (n - 1)
                    m = n - L
                    if m <= 0:
                        continue
                    pairs = np.nonzero((a[:m] == 1) & (a[L:] == 0))[0]

                if pairs.size == 0:
                    continue
                if pairs.size <= 2:
                    sel = pairs
                else:
                    sel = rng.choice(pairs, size=2, replace=False)

                for i in sel:
                    evaluated += 1
                    i = int(i)
                    L = int(L)
                    p1 = i
                    p2 = i + L
                    if p1 < 0 or p2 >= n:
                        continue
                    old1 = int(a[p1])
                    old2 = int(a[p2])
                    if old1 == old2:
                        continue

                    a[p1] = old2
                    a[p2] = old1
                    c2 = np.correlate(a, 1 - a, mode="full")
                    s2 = int(binary_score(c2))
                    a[p1] = old1
                    a[p2] = old2

                    if s2 < cur_score:
                        a[p1] = old2
                        a[p2] = old1
                        cur_c = c2
                        cur_score = s2
                        accepted = True
                        M2 = int(np.max(c2))
                        if s2 < best_score:
                            best_score = s2
                            best_a = a.copy()
                            best_count = M2
                        no_improve = 0
                        break
                if accepted:
                    break

        if not accepted:
            ones = np.nonzero(a == 1)[0]
            zeros = np.nonzero(a == 0)[0]
            if ones.size > 0 and zeros.size > 0:
                for _ in range(10):
                    p1 = int(ones[rng.randint(0, ones.size)])
                    p2 = int(zeros[rng.randint(0, zeros.size)])
                    if p1 == p2:
                        continue
                    old1 = int(a[p1])
                    old2 = int(a[p2])
                    if old1 == old2:
                        continue

                    a[p1] = old2
                    a[p2] = old1
                    c2 = np.correlate(a, 1 - a, mode="full")
                    s2 = int(binary_score(c2))
                    a[p1] = old1
                    a[p2] = old2

                    if s2 <= cur_score:
                        a[p1] = old2
                        a[p2] = old1
                        cur_c = c2
                        cur_score = s2
                        accepted = True
                        M2 = int(np.max(c2))
                        if s2 < best_score:
                            best_score = s2
                            best_a = a.copy()
                            best_count = M2
                        no_improve = 0
                        break

            if not accepted:
                no_improve += 1
                if no_improve >= 80:
                    a = best_a.copy()
                    cur_c = np.correlate(a, 1 - a, mode="full")
                    cur_score = int(binary_score(cur_c))
                    for _ in range(5):
                        ones = np.nonzero(a == 1)[0]
                        zeros = np.nonzero(a == 0)[0]
                        if ones.size == 0 or zeros.size == 0:
                            break
                        p1 = int(ones[rng.randint(0, ones.size)])
                        p2 = int(zeros[rng.randint(0, zeros.size)])
                        if p1 != p2:
                            tmp = int(a[p1])
                            a[p1] = int(a[p2])
                            a[p2] = tmp
                    cur_c = np.correlate(a, 1 - a, mode="full")
                    cur_score = int(binary_score(cur_c))
                    no_improve = 0

    return best_a, best_count


def binary_search_all(parent, best_h, best_n, best_c, time_limit, seed):
    rng = np.random.RandomState(int(seed))
    best_c_local = float(best_c)
    best_a = None
    best_n_local = 0

    parent_n = int(parent.size) if parent is not None else 0
    n_set = [32, 48, 64, 96, 128]
    if parent_n > 0 and 32 <= parent_n <= 256 and parent_n % 2 == 0:
        n_set.append(parent_n)
    if best_n > 0 and 32 <= best_n <= 256 and int(best_n) % 2 == 0:
        n_set.append(int(best_n))

    n_list = []
    seen = set()
    for nn in n_set:
        nn = int(nn)
        if nn % 2 != 0:
            continue
        if nn not in seen:
            seen.add(nn)
            n_list.append(nn)
    n_list.sort()

    rem0 = float(time_limit - time.time())
    if rem0 < 80.0 and len(n_list) > 2:
        n_list = n_list[:2]
    elif rem0 < 120.0 and len(n_list) > 3:
        n_list = n_list[:3]
    else:
        n_list = n_list[:5]

    for pos, nn in enumerate(n_list):
        if time.time() > time_limit - 10.0:
            break
        rem = float(time_limit - time.time())
        left = max(1, len(n_list) - pos)
        per_total = rem / float(left)
        if nn <= 64:
            slot = min(45.0, max(8.0, per_total * 1.5))
        else:
            slot = min(25.0, max(6.0, per_total))
        slot_end = min(float(time_limit), time.time() + slot)

        starts = []
        if parent is not None:
            starts.append(binary_threshold(interpolate_to_n(parent, nn), nn))
        if best_h is not None:
            starts.append(binary_threshold(interpolate_to_n(best_h, nn), nn))

        h = np.zeros(nn, dtype=float)
        if nn % 4 == 0:
            h[np.arange(0, nn, 4)] = 1.0
            h[np.arange(1, nn, 4)] = 1.0
        starts.append(binary_threshold(h, nn))

        starts.append(binary_threshold(rng.rand(nn), nn))
        starts.append(binary_threshold(rng.rand(nn), nn))

        m = len(starts)
        for j, start_a in enumerate(starts[:3]):
            if time.time() >= slot_end:
                break
            per = float(slot_end - time.time())
            per = min(15.0, max(3.0, per / float(max(1, m - j))))
            aa, bbc = binary_local_search(nn, start_a, time.time() + per, seed + 101 * nn + j)
            if aa is not None and bbc is not None:
                cc = 2.0 * int(bbc) / float(nn)
                if cc < best_c_local:
                    best_c_local = cc
                    best_a = aa.astype(float).copy()
                    best_n_local = int(nn)

    return best_a, float(best_c_local), int(best_n_local)


def build_candidate_order(base_n):
    base_n = int(base_n)
    cand = [base_n]
    deltas = (16, -16, 32, -32, 48, -48, 8, -8, 24, -24, 40, -40)
    for d in deltas:
        cand.append(base_n + d)
    presets = (64, 80, 96, 112, 128, 144, 160, 176, 192, 208, 224, 240, 256, 288)
    for p in presets:
        cand.append(p)

    items = []
    seen = set()
    for nn in cand:
        nn = int(nn)
        if nn < 32 or nn > 320:
            continue
        if nn in seen:
            continue
        seen.add(nn)
        primary = 0 if nn == base_n else 1
        distance = abs(nn - base_n)
        tie = -nn
        items.append((primary, distance, tie, nn))

    items.sort()
    return [x[3] for x in items]


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)

    try:
        parent = get_parent_values()
    except Exception:
        parent = None

    best_h = None
    best_c = np.inf
    best_n = 160
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_end = t0 + max(120.0, budget_s - 90.0)

    try:
        if parent is not None and parent.size > 0:
            parent_n = int(parent.size)
            best_h = project_feasible(parent, parent_n)
            best_c = compute_c5(best_h)
            best_n = parent_n
        else:
            best_n = 160
            best_h = project_feasible(np.full(best_n, 0.5), best_n)
            best_c = compute_c5(best_h)

        # Cheap integer search. If it succeeds, it can produce a clearly better bound.
        if time.time() < hard_end - 350.0:
            bin_budget = min(100.0, max(30.0, (hard_end - time.time()) * 0.08))
            if bin_budget > 15.0:
                bin_h, bin_c, bin_n = binary_search_all(
                    parent, best_h, best_n, best_c, time.time() + bin_budget, seed + 12345
                )
                if bin_h is not None and bin_c < best_c:
                    best_h = bin_h
                    best_c = float(bin_c)
                    best_n = int(bin_n)

        # Base polish at the current best resolution.
        base_budget = 300.0 if budget_s > 1000 else (120.0 if budget_s > 500 else 45.0)
        if time.time() < hard_end - max(base_budget, 60.0) - 60.0:
            inits = build_inits_for_n(best_n, parent, best_h, best_n, rng, count=4)
            local_end = min(hard_end - 120.0, time.time() + base_budget)
            if local_end > time.time() + 20.0:
                hh, cc = process_initial_set(inits, best_n, local_end, seed + 1)
                if hh is not None and cc < best_c:
                    best_h = hh
                    best_c = float(cc)
                    best_n = int(hh.size)

        # Explore neighboring resolutions.
        cand = build_candidate_order(best_n)
        processed = {int(best_n)}
        main_end = hard_end - 100.0
        for nn in cand:
            nn = int(nn)
            if time.time() > main_end:
                break
            if nn in processed:
                continue
            if nn < 32 or nn > 320:
                continue
            processed.add(nn)

            rem = float(main_end - time.time())
            if rem < 45.0:
                break

            dist = abs(nn - best_n)
            if dist <= 16:
                slot = min(120.0, rem)
            elif dist <= 48:
                slot = min(80.0, rem)
            else:
                slot = min(50.0, rem)

            inits = build_inits_for_n(nn, parent, best_h, best_n, rng, count=3)
            hh, cc = process_initial_set(inits, nn, time.time() + slot, seed + 5000 + 13 * nn)
            if hh is not None and cc < best_c:
                best_h = hh
                best_c = float(cc)
                best_n = int(hh.size)

        # Final polish.
        if time.time() < hard_end - 20.0 and best_n >= 4:
            fin_end = hard_end - 10.0
            if best_n <= 260:
                for beta in (120, 320):
                    if time.time() > fin_end - 30.0:
                        break
                    h2, c2 = optimize_smooth_fast(
                        best_h,
                        best_n,
                        beta,
                        maxiter=35 if best_n <= 176 else 20,
                        ftol=1e-12,
                    )
                    if c2 < best_c:
                        best_h = h2
                        best_c = c2

                if time.time() < fin_end - 25.0:
                    if best_n <= 176:
                        fin_seq = (32, 64)
                        fin_maxit = 45
                    else:
                        fin_seq = (16, 32)
                        fin_maxit = 25
                    h2, c2 = refine_active(
                        best_h, best_n, fin_end - 5, active_seq=fin_seq, maxiter=fin_maxit
                    )
                    if c2 < best_c:
                        best_h = h2
                        best_c = c2

                if time.time() < fin_end - 10.0 and best_n <= 240:
                    h2, c2 = local_block_perturb(best_h, best_n, fin_end, seed + 99991, step=0.015)
                    if c2 < best_c:
                        best_h = h2
                        best_c = c2

        best_h = project_feasible(best_h, best_n)
        best_c = compute_c5(best_h)
        return (best_h, float(best_c), int(best_h.size))

    except Exception:
        if best_h is None:
            if parent is not None and parent.size > 0:
                best_h = project_feasible(parent, parent.size)
                best_n = int(parent.size)
            else:
                best_n = 160
                best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_h = project_feasible(best_h, best_n)
        best_c = compute_c5(best_h)
        return (best_h, float(best_c), int(best_h.size))
