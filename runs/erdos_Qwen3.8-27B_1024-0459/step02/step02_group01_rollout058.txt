    if rng is None:
        rng = np.random.RandomState(0)

    # Get initial active lags (top k)
    active_idx = np.argsort(corr)[-k_active:]
    
    for sweep in range(n_sweeps):
        # Update active set
        corr = compute_corr(h)
        active_idx = np.argsort(corr)[-k_active:]
        
        # Coordinate descent
        order = list(range(n))
        rng.shuffle(order)
        
        for j in order:
            old_val = h[j]
            # Compute slope and intercept for each active lag
            slopes = np.zeros(k_active)
            intercepts = np.zeros(k_active)
            
            for ai, m in enumerate(active_idx):
                k = m - (n - 1)
                slope = 0.0
                contrib = 0.0
                # h[j] as first factor: h[j] * (1-h[j-k])
                if 0 <= j - k < n:
                    slope += (1.0 - h[j - k])
                    contrib += old_val * (1.0 - h[j - k])
                # h[j] as second factor: h[j+k] * (1-h[j])
                if 0 <= j + k < n:
                    slope -= h[j + k]
                    contrib += h[j + k] * (1.0 - old_val)
                slopes[ai] = slope
                intercepts[ai] = corr[m] - contrib
            
            # Minimize max(slopes* x + intercepts) over x in [0,1]
            # This is a 1D problem
            # Candidates: 0, 1, and intersection points
            best_x = old_val
            best_obj = np.max(slopes * old_val + intercepts)
            
            for x_c in [0.0, 1.0]:
                obj = np.max(slopes * x_c + intercepts)
                if obj < best_obj:
                    best_obj = obj
                    best_x = x_c
            
            # Check intersection points (only adjacent in sorted order for efficiency)
            # For small k_active, just check all pairs
            for a in range(k_active):
                for b in range(a + 1, k_active):
                    ds = slopes[a] - slopes[b]
                    if abs(ds) > 1e-14:
                        x_int = (intercepts[b] - intercepts[a]) / ds
                        if -0.5 <= x_int <= 1.5:  # slight margin then clamp
                            x_int = max(0.0, min(1.0, x_int))
                            obj = np.max(slopes * x_int + intercepts)
                            if obj < best_obj:
                                best_obj = obj
                                best_x = x_int
            
            h[j] = best_x
        
        # Periodic projection
        if sweep % 3 == 2:
            h = project_feasible(h, n)
    
    return h

Let me also think about what "binary local search" means more concretely:
- We have a binary h with n/2 ones and n/2 zeros
- A "move" is: pick a 1-position a and a 0-position b, swap them
- The change in the correlation vector is: Δc[m] = -slope[a,m] + slope[b,m] where slope[p,m] is the influence of h[p] on c[m]
- We want to find moves that reduce max(c)
- Greedy: for the current max lag m*, find the move that most reduces c[m*] (while not increasing other lags too much)

For efficiency, I'll do:
- Identify the top K lags (K=5)
- For each 1-position a: compute the vector -slope[a,:] (change if we flip a to 0)
- For each 0-position b: compute the vector slope[b,:] (change if we flip b to 1)
- The best pair (a,b) minimizes max of (c - slope[a] + slope[b]) over the top K lags
- This is O((n/2)² * K) per iteration, which for n=512, K=5 is about 1.6M ops. Fast enough for a few iterations.

Actually, let's be smarter. For a fixed a (the 1-position we're removing), the best b (the 0-position to add) is the one that minimizes max_k (c[k] - slope[a,k] + slope[b,k]) for k in active set. This is a 1D search over b, which is O(n/2 * K) for a fixed a. Total: O((n/2)² * K) per full pass.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_beta = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                if add > 1e-15:
                    h[idx] += add
                    diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                rem = min(-diff, h[idx])
                if rem > 1e-15:
                    h[idx] -= rem
                    diff += rem
                if abs(diff) < 1e-12:
                    break
    return h

def _obj_smooth(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    sh = _g_beta * (c - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / _g_beta
    return smax * (2.0 / n)

def _eq_con(x):
    return np.sum(x) - _g_n / 2.0

def _upsample(h, n_old, n_new):
    xp = (np.arange(n_old) + 0.5) * 2.0 / n_old
    xn = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_interp = np.interp(xn, xp, h)
    return project_feasible(h_interp, n_new)

def coord_descent(h, n, n_sweeps=30, k_active=30, rng=None, t_start=None, time_limit=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    for sweep in range(n_sweeps):
        if t_start is not None and time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        corr = compute_corr(h)
        active_idx = np.argsort(corr)[-k_active:]
        order = np.arange(n)
        rng.shuffle(order)
        for j in order:
            old_val = h[j]
            slopes = np.zeros(k_active)
            intercepts = np.zeros(k_active)
            for ai in range(k_active):
                m = active_idx[ai]
                k = m - (n - 1)
                slope = 0.0
                contrib = 0.0
                if 0 <= j - k < n:
                    slope += (1.0 - h[j - k])
                    contrib += old_val * (1.0 - h[j - k])
                if 0 <= j + k < n:
                    slope -= h[j + k]
                    contrib += h[j + k] * (1.0 - old_val)
                slopes[ai] = slope
                intercepts[ai] = corr[m] - contrib
            best_x = old_val
            best_obj = np.max(slopes * best_x + intercepts)
            for x_c in (0.0, 1.0):
                obj = np.max(slopes * x_c + intercepts)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_x = x_c
            for a in range(k_active):
                for b in range(a + 1, k_active):
                    ds = slopes[a] - slopes[b]
                    if abs(ds) > 1e-14:
                        x_int = (intercepts[b] - intercepts[a]) / ds
                        if 0.0 <= x_int <= 1.0:
                            obj = np.max(slopes * x_int + intercepts)
                            if obj < best_obj - 1e-14:
                                best_obj = obj
                                best_x = x_c if False else x_int
            h[j] = best_x
        if sweep % 5 == 4:
            h = project_feasible(h, n)
    return h

def binary_local_search(h, n, max_passes=50, k_active=5, t_start=None, time_limit=None):
    h_bin = (h > 0.5).astype(float)
    target_ones = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones > target_ones:
        ones = np.where(h_bin == 1)[0]
        to_rm = ones[np.argsort(h[ones])[:n_ones - target_ones]]
        h_bin[to_rm] = 0.0
    elif n_ones < target_ones:
        zeros = np.where(h_bin == 0)[0]
        to_add = zeros[np.argsort(-h[zeros])[:target_ones - n_ones]]
        h_bin[to_add] = 1.0
    for _pass in range(max_passes):
        if t_start is not None and time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        corr = compute_corr(h_bin)
        active_idx = np.argsort(corr)[-k_active:]
        max_val = corr[active_idx].max()
        infl = np.zeros((n, k_active))
        for ai in range(k_active):
            m = active_idx[ai]
            k = m - (n - 1)
            for p in range(n):
                s = 0.0
                if 0 <= p - k < n:
                    s += (1.0 - h_bin[p - k])
                if 0 <= p + k < n:
                    s -= h_bin[p + k]
                infl[p, ai] = s
        ones_pos = np.where(h_bin == 1)[0]
        zeros_pos = np.where(h_bin == 0)[0]
        best_gain = 1e-10
        best_pair = None
        for a in ones_pos:
            new_vals = corr[active_idx][np.newaxis, :] - infl[a][np.newaxis, :] + infl[zeros_pos, :]
            new_max_per_b = np.max(new_vals, axis=1)
            gains = max_val - new_max_per_b
            bi = np.argmax(gains)
            if gains[bi] > best_gain:
                best_gain = gains[bi]
                best_pair = (a, zeros_pos[bi])
        if best_pair is not None:
            h_bin[best_pair[0]] = 0.0
            h_bin[best_pair[1]] = 1.0
        else:
            break
    return h_bin

        return best_h, best_c5
    # Phase 3: More coordinate descent with larger active set
    cd_time2 = remaining * 0.4
