    n = len(h)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)


def interpolate_to_n(h_src, n_dst):
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def lse_objective(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(beta * (corr - cmax), -50, 50)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    return (cmax + np.log(sum_exp)) / beta


def lse_gradient(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = np.clip(beta * (corr - cmax), -50, 50)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        off = k - (n - 1)
        lo1 = max(0, -off)
        hi1 = min(n - 1, n - 1 - off)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wk * (1.0 - x[lo1 + off:hi1 + off + 1])
        lo2 = max(0, off)
        hi2 = min(n - 1, n - 1 + off)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += wk * (-x[lo2 - off:hi2 - off + 1])
    return grad
