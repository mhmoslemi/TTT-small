def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 0) & (h < 1)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)

            h[idx] = min(1.0, h[idx] + diff)

        starts.append(h_rb)

        # 4. Periodic patterns
        for period in [6, 8, 10, 12, 14, 16, 20, 24]:
            if n % period == 0:
                h_p = np.zeros(n)
                for p in range(0, n, period):
                    h_p[p:p + period // 2] = 1.0
                starts.append(h_p)

        # 5. Blended with noise
        for w in [0.3, 0.5, 0.7]:
            h_m = w * h_int + (1.0 - w) * rng.rand(n)
            starts.append(project_feasible(h_m, n))

        # 6. Sine patterns
        t_arr = np.arange(n) / n
        for freq in [3, 5, 7, 9, 11, 13]:
            h_s = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * t_arr + rng.rand() * 2.0 * np.pi)
            starts.append(project_feasible(h_s, n))

        # 7. Block-amplitude patterns
        for nblocks in [3, 5, 7]:
            if n % nblocks == 0:
                blk = n // nblocks
                h_blk = np.zeros(n)
                for b in range(0, n, 2 * blk):
                    h_blk[b:b + blk] = 1.0
                starts.append(project_feasible(h_blk, n))

        # Score and sort
        scored = [(compute_c5(hs), hs) for hs in starts]
        scored.sort(key=lambda x: x[0])

        for c, hs in scored:
            if c < best_c5:
                best_c5 = c
                best_h = hs.copy()
                best_n = n

        top_k = min(5, len(scored))
        top_starts = [scored[i][1] for i in range(top_k)]

        # Allocate time per optimization
        time_per_opt = max(8.0, min(120.0, _tl() * 0.65 / max(1, top_k)))

        for h0 in top_starts:
            if _tl() < 12:
                break

            # L-BFGS-B optimization
            h_opt = _run_lbfgs(h0, n, t0, budget_s)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                best_n = n

            # Subgradient refinement
            if _tl() > 15:
                h_sg = _subgradient(h_opt, n, 400, t0, budget_s)
                c_sg = compute_c5(h_sg)
                if c_sg < best_c5:
                    best_c5 = c_sg
                    best_h = h_sg.copy()
                    best_n = n

        # Extra passes on best at this resolution
        if _tl() > 25 and best_n == n:
            h2 = _run_lbfgs(best_h, n, t0, budget_s, beta=500.0)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

            if _tl() > 15:
                h_sg2 = _subgradient(best_h, n, 500, t0, budget_s)
                c_sg2 = compute_c5(h_sg2)
                if c_sg2 < best_c5:
                    best_c5 = c_sg2
                    best_h = h_sg2.copy()

        # Binary round and bit-flip
        if _tl() > 10:
            h_br = _binary_round(best_h, best_n)
            c_br = compute_c5(h_br)
            if c_br < best_c5:
                best_c5 = c_br
                best_h = h_br.copy()

            if _tl() > 20 and best_n == n:
                h_bf, c_bf = _bitflip_local(best_h, n, t0, budget_s, max_iter=1500)
                if c_bf < best_c5:
                    best_c5 = c_bf
                    best_h = h_bf.copy()

    # Final polish
    if _tl() > 30:
        h3 = _run_lbfgs(best_h, best_n, t0, budget_s, beta=1000.0)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

        if _tl() > 15:
            h_sg3 = _subgradient(best_h, best_n, 600, t0, budget_s)
            c_sg3 = compute_c5(h_sg3)
            if c_sg3 < best_c5:
                best_c5 = c_sg3
                best_h = h_sg3.copy()

    # Binary round final
    if _tl() > 5:
        h_br = _binary_round(best_h, best_n)
        c_br = compute_c5(h_br)
        if c_br < best_c5:
            best_c5 = c_br
            best_h = h_br.copy()

    # Bit-flip on final
    if _tl() > 15:
        h_bf, c_bf = _bitflip_local(best_h, best_n, t0, budget_s, max_iter=2000)
        if c_bf < best_c5:
            best_c5 = c_bf
            best_h = h_bf.copy()

    # Try small shifts
    for shift in [1, -1, 2, -2, 3]:
        h_sh = np.roll(best_h, shift)
        h_sh = project_feasible(h_sh, best_n)
        c_sh = compute_c5(h_sh)
        if c_sh < best_c5:
            best_c5 = c_sh
            best_h = h_sh.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
