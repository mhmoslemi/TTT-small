import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_value(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return m + np.log(total) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection to fix the sum while maintaining [0,1] bounds
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Add uniform offset, then clip and iterate
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Adjust the element closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        adj = diff
        if h[idx] + adj > 1.0 or h[idx] + adj < 0.0:
            # Find another element
            for i in range(n):
                if h[i] + adj >= 0.0 and h[i] + adj <= 1.0:
                    h[i] += adj
                    break
            else:
                # Spread among all elements
                h += diff / n
                h = np.clip(h, 0, 1)
        else:
            h[idx] += adj
    return h

def obj_func(x, n, beta, penalty):
    dx = 2.0 / n
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max_value(corr, beta) * dx
    s = np.sum(h) - n / 2.0
    val += penalty * s * s
    return val

def grad_func(x, n, beta, penalty):
    dx = 2.0 / n
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    grad *= dx
    s = np.sum(h) - n / 2.0
    grad += 2.0 * penalty * s * np.ones(n)
    return grad

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run_lbfgsb(h0, n, beta, penalty, max_iter=500):
    bounds = [(0.0, 1.0)] * n

    total = np.sum(np.exp(shifted))
    return m + np.log(total) / beta

    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def project_feasible(h, n):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        idx = np.argmin(np.maximum(h, 1.0 - h))
        adj = min(max(diff, h[idx]), 1.0 - h[idx])
        h[idx] += adj
        diff -= adj

                a = min(max(diff, 0.0 - h[i]), 1.0 - h[i])
                if abs(a) > 1e-15:
                    h[i] += a
                    diff -= a
                    if abs(diff) < 1e-14:
                        break
    return h

    grad *= dx
    grad += 2.0 * penalty * s

    return project_feasible(h_dst, n_dst)

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def run_lbfgsb(h0, n, beta, penalty, max_iter, max_fun):
    def f(x):
        val, _ = obj_and_grad(x, n, beta, penalty)
        return val
    def g(x):
        _, gr = obj_and_grad(x, n, beta, penalty)
        return gr
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h0, jac=g, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-12,
                            'maxls': 40, 'maxfun': max_fun})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def make_smooth_random(n, rng):
    xf = np.arange(n)
    h = np.full(n, 0.5)
    nterms = rng.randint(2, 5)
    for _ in range(nterms):
        f = rng.uniform(1.0, 8.0)
        p = rng.uniform(0, 2 * np.pi)
        a = rng.uniform(0.05, 0.25)
        h += a * np.sin(2 * np.pi * f * xf / n + p)
    h = np.clip(h, 0.01, 0.99)
    return project_feasible(h, n)

def make_block_random(n, rng):
    h = np.zeros(n)
    target = n // 2
    pos = 0
    while pos < n and target > 0:
        blk = rng.randint(1, max(2, n // 10))
        blk = min(blk, n - pos, target)
        h[pos:pos+blk] = 1.0
        target -= blk
        pos += blk + rng.randint(1, max(2, n // 12))
    if target > 0:
        remaining = np.where(h == 0)[0]
        idx = rng.choice(remaining, size=min(target, len(remaining)), replace=False)
        h[idx] = 1.0
    return project_feasible(h, n)

def binary_local_search(h, n, rng, max_rounds, time_limit):
    t0 = time.time()
    c5 = compute_c5(h)
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_tries = min(200, max(20, int(time_limit // 5)))
        for _ in range(n_tries):
            if time.time() - t0 > time_limit:
                break
            pos1 = ones[rng.randint(len(ones))]
            pos0 = zeros[rng.randint(len(zeros))]
            h[pos1] = 0.0
            h[pos0] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-15:
                c5 = new_c5
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[pos1] = 1.0
                h[pos0] = 0.0
    return h, c5

def simulated_anneal_binary(h, n, rng, time_limit):
    t0 = time.time()
    c5 = compute_c5(h)
    T = 0.05
    T_min = 0.001
    cooldown = 0.9995
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    while time.time() - t0 < time_limit and T > T_min:
        if len(ones) == 0 or len(zeros) == 0:
            break
        pos1 = ones[rng.randint(len(ones))]
        pos0 = zeros[rng.randint(len(zeros))]
        h[pos1] = 0.0
        h[pos0] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - c5
        if delta < -1e-15 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[pos1] = 1.0
            h[pos0] = 0.0
        T *= cooldown
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time) - 8.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = make_smooth_random(best_n, rng)
        best_c5 = compute_c5(best_h)
        print("Random init: n={}, C5={:.6f}".format(best_n, best_c5))

    # Main optimization across resolutions
    n_list = sorted(set([best_n, 192, 200, 208, 224, 240, 256, 288, 320]))
    n_list = [nn for nn in n_list if 64 <= nn <= 384]

    for n in n_list:
        if remaining() < 50:
            break
        print("\n=== n={} (remaining {:.0f}s) ===".format(n, remaining()))

        starts = []
        starts.append(interp_h(best_h, n))
        starts.append(make_smooth_random(n, rng))
        if remaining() > 120:
            starts.append(make_block_random(n, rng))

        for si, h0 in enumerate(starts):
            if remaining() < 25:
                break
            h0 = project_feasible(h0, n)
            cur = h0.copy()

            stages = [
                (5, 10, 500, 4000),
                (15, 50, 700, 5000),
                (50, 200, 800, 6000),
                (100, 1000, 1000, 7000),
                (300, 5000, 1000, 7000),
            ]

            for beta, pen, miter, mfun in stages:
                if remaining() < 15:
                    break
                emit = min(miter, max(50, int(remaining() * 3)))
                emfun = min(mfun, max(300, int(remaining() * 6)))
                try:
                    h_new, c5_new = run_lbfgsb(cur, n, beta, pen, emit, emfun)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n
                        print("  start{}, beta={}, C5={:.6f}".format(si, beta, best_c5))
                    cur = h_new
                except Exception as e:
                    print("  err: {}".format(e))
                    break

    # Extra polish on best solution
    if best_h is not None and remaining() > 25:
        n = best_n
        print("\n=== Polish n={} (remaining {:.0f}s) ===".format(n, remaining()))
        for beta in [500, 1500, 5000]:
            if remaining() < 12:
                break
            try:
                h_new, c5_new = run_lbfgsb(best_h, n, beta, 20000, 600, 6000)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  polish beta={}: C5={:.6f}".format(beta, best_c5))
            except Exception:
                pass

    # Binary rounding and search
    if remaining() > 15:
        h_round = (best_h > 0.5).astype(float)
        n_want = best_n // 2
        n_ones = int(np.round(np.sum(h_round)))
        if n_ones > n_want:
            ones = np.where(h_round == 1.0)[0]
            order = np.argsort(best_h[ones])[:n_ones - n_want]
            h_round[ones[order]] = 0.0
        elif n_ones < n_want:
            zeros = np.where(h_round == 0.0)[0]
            order = np.argsort(-best_h[zeros])[:n_want - n_ones]
            h_round[zeros[order]] = 1.0
        c5_round = compute_c5(h_round)
        if c5_round < best_c5:
            best_c5 = c5_round
            best_h = h_round.copy()
            print("Binary round: C5={:.6f}".format(best_c5))

    # Binary local search if we have a binary solution
    if remaining() > 30 and np.all((best_h == 0) | (best_h == 1)):
        alloc = min(remaining() - 10, 80)
        h_bs, c5_bs = binary_local_search(best_h.copy(), best_n, rng, 1, alloc)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs
            print("Binary local: C5={:.6f}".format(best_c5))

    # Simulated annealing from best binary
    if remaining() > 40:
        h_sa = (best_h > 0.5).astype(float)
        n_want = best_n // 2
        n_ones = int(np.round(np.sum(h_sa)))
        if n_ones > n_want:
            ones = np.where(h_sa == 1.0)[0]
            order = np.argsort(best_h[ones])[:n_ones - n_want]
            h_sa[ones[order]] = 0.0
        elif n_ones < n_want:
            zeros = np.where(h_sa == 0.0)[0]
            order = np.argsort(-best_h[zeros])[:n_want - n_ones]
            h_sa[zeros[order]] = 1.0
        alloc = min(remaining() - 10, 60)
        h_sa, c5_sa = simulated_anneal_binary(h_sa, best_n, rng, alloc)
        h_sa, c5_sa = binary_local_search(h_sa, best_n, rng, 1, min(remaining() - 10, 20))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            print("Simulated anneal: C5={:.6f}".format(best_c5))

    # Final: try one more L-BFGS-B from the best (possibly binary) solution
    if remaining() > 20 and np.all((best_h == 0) | (best_h == 1)):
        n = best_n
        h_cont = np.where(best_h > 0.5, 0.8, 0.2).astype(float)
        h_cont = project_feasible(h_cont, n)
        try:
            h_new, c5_new = run_lbfgsb(h_cont, n, 200, 2000, 800, 6000)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("Cont from binary: C5={:.6f}".format(best_c5))
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
