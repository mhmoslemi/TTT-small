**Approach: Alternating minimization / block coordinate descent with QP subproblems**

At each step, we have a current max value T = max_k dx*corr[k]. We want to reduce T.

The subproblem: minimize t subject to:
- dx * sum_j h[j]*(1-h[j+r]) ≤ t for all r in active set
- 0 ≤ h[j] ≤ 1
- sum h[j] = n/2

This is a QP (quadratic objective is t, quadratic constraints). But solving a QP with O(n²) constraints is expensive.

Better: use a cutting-plane approach. At each iteration:
1. Compute current corr and find the max lag r*
2. Add constraint dx*corr[r*] ≤ t (linearized at current h)
3. Solve the resulting LP
4. Repeat

This is linear in h! So we can use LP.

    elif r < 0:

        grad[:-s] += (1.0 - h[s:])
        grad[s:] += -h[:-s]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad


def c5_func(h):
    return np.max(np.correlate(h, 1.0 - h, mode='full'))


def c5_grad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        r = k - (n - 1)
        grad += grad_lag(h, r)
    grad /= len(active)
    return grad


def sum_con_fun(x, n):
    return np.sum(x) - n / 2.0


def sum_con_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h_init, n, max_iter=10000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(c5_func, h_init, jac=c5_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            c5 = compute_c5(h_out)
            return h_out, c5
    except Exception:
        pass
    return None, None


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def projected_gradient_step(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-10)[0]
    grad = np.zeros(n)
    for k in active:
        r = k - (n - 1)
        grad += grad_lag(h, r)
    grad /= len(active)
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)
    return h_new


def coord_descent_fast(h, n, n_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    ells = np.arange(-(n - 1), n)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            idx_plus = j + ells
            idx_minus = j - ells
            valid_plus = (idx_plus >= 0) & (idx_plus < n)
            valid_minus = (idx_minus >= 0) & (idx_minus < n)
            a = np.zeros(2 * n - 1)
            if np.any(valid_plus):
                a[valid_plus] += -h[idx_plus[valid_plus]]
            if np.any(valid_minus):
                a[valid_minus] += (1.0 - h[idx_minus[valid_minus]])
            b = corr - h[j] * a
            lo = 0.0
            hi = 1.0
            for _ in range(50):
                mid = 0.5 * (lo + hi)
                vals = a * mid + b
                mx = np.max(vals)
                active_mask = vals >= mx - 1e-14
                sg = np.mean(a[active_mask])
                if sg > 1e-13:
                    hi = mid
                elif sg < -1e-13:
                    lo = mid
                else:
                    break
            new_val = 0.5 * (lo + hi)
            delta = new_val - h[j]
            h[j] = new_val
            if abs(delta) > 1e-15:
                for k in range(2 * n - 1):
                    r = k - (n - 1)
                    d = 0.0
                    if 0 <= j + r < n:
                        d += -h[j + r] * (-1) if False else 0
                corr = np.correlate(h, 1.0 - h, mode='full')
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def local_perturb_search(h, n, rng, n_trials=200, time_limit=30):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for _ in range(n_trials):
        if time.time() - start > time_limit:
            break
        h_try = best_h.copy()
        n_pert = rng.randint(1, 4)
        idxs = rng.choice(n, n_pert, replace=False)
        deltas = rng.uniform(-0.1, 0.1, n_pert)
        h_try[idxs] += deltas
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if len(h_parent) == 0:
            h_parent = None
    except (NameError, Exception):
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent.copy(), best_n)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at parent resolution with high iteration count
    if remaining() > 150:
        h_sol, c5_sol = slsqp_optimize(best_h, best_n, max_iter=15000, ftol=1e-17)
        if h_sol is not None and c5_sol < best_c5:
            best_c5 = c5_sol
            best_h = h_sol.copy()

    # Phase 2: Interpolate to higher resolutions and optimize
    res_list = []
    if best_n >= 160:
        res_list = [320, 480, 640]
    else:
        res_list = [256, 320, 480, 640]
    for n_new in res_list:
        if remaining() < 120:
            break
        h_interp = interpolate_to_n(best_h, n_new)
        h_sol, c5_sol = slsqp_optimize(h_interp, n_new, max_iter=12000, ftol=1e-17)
        if h_sol is not None:
            c5_sol = compute_c5(h_sol)
            if c5_sol < best_c5 - 1e-13:
                best_c5 = c5_sol
                best_h = h_sol.copy()
                best_n = n_new

    # Phase 3: Coordinate descent polish at best resolution
    if remaining() > 80:
        h_cd, c5_cd = coord_descent_fast(best_h, best_n, n_passes=200,
                                         time_limit=min(80, remaining() - 40))
        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Random restarts (perturb + SLSQP)
    n_restarts = 12
    for trial in range(n_restarts):
        if remaining() < 100:
            break
        scale = 0.03 + 0.02 * trial
        h_pert = best_h + rng.randn(best_n) * scale
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = make_feasible(h_pert, best_n)
        h_sol, c5_sol = slsqp_optimize(h_pert, best_n, max_iter=8000, ftol=1e-16)
        if h_sol is not None:
            c5_sol = compute_c5(h_sol)
            if c5_sol < best_c5 - 1e-13:
                best_c5 = c5_sol
                best_h = h_sol.copy()

    # Phase 5: Projected gradient descent + SLSQP
    if remaining() > 120:
        h_curr = best_h.copy()
        alpha = 0.05
        for step in range(300):
            if remaining() < 80:
                break
            h_curr = projected_gradient_step(h_curr, best_n, alpha)
            alpha *= 0.995
            if step % 30 == 29:
                h_sol, c5_sol = slsqp_optimize(h_curr, best_n, max_iter=5000, ftol=1e-16)
                if h_sol is not None:
                    c5_sol = compute_c5(h_sol)
                    if c5_sol < best_c5 - 1e-13:
                        best_c5 = c5_sol
                        best_h = h_sol.copy()
                        h_curr = best_h.copy()

    # Phase 6: Local perturbation search
    if remaining() > 40:
        h_loc, c5_loc = local_perturb_search(best_h, best_n, rng,
                                             n_trials=500,
                                             time_limit=min(40, remaining() - 20))
        if c5_loc < best_c5 - 1e-13:
            best_c5 = c5_loc
            best_h = h_loc.copy()

    # Phase 7: Final SLSQP polish
    if remaining() > 60:
        h_sol, c5_sol = slsqp_optimize(best_h, best_n, max_iter=15000, ftol=1e-17)
        if h_sol is not None and c5_sol < best_c5:
            best_c5 = c5_sol
            best_h = h_sol.copy()

    # Phase 8: Try one more resolution jump
    if remaining() > 100:
        n_try = best_n + (best_n // 4)
        if n_try <= 1024 and n_try != best_n:
            h_interp = interpolate_to_n(best_h, n_try)
            h_sol, c5_sol = slsqp_optimize(h_interp, n_try, max_iter=8000, ftol=1e-16)
            if h_sol is not None:
                c5_sol = compute_c5(h_sol)
                if c5_sol < best_c5 - 1e-13:
                    best_c5 = c5_sol
                    best_h = h_sol.copy()
                    best_n = n_try

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
