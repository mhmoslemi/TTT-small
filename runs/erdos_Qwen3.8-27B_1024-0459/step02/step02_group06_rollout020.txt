def project_density(x, target):
    """Project x to [0,1]^n with sum = target using bisection."""
    x = np.asarray(x, dtype=float).copy()

    x = np.clip(x, 0.0, 1.0)

    res = target - float(np.sum(h))

def corr_gradient_single(h, k):
    """Gradient of correlation at lag k with respect to h."""
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            # corr[k] = sum_i h[i+k]*(1-h[i]) for i=0..n-1-lag
            # d/dh[j]: if j >= lag: appears in h[j]*(1-h[j-k]) -> (1-h[j-k])
            #           if j < n-lag: appears in h[j+k]*(1-h[j]) -> -h[j+k]
            for i in range(n - lag):
                # term: h[i+lag]*(1-h[i])
                # d/dh[i]: -h[i+lag]
                # d/dh[i+lag]: (1-h[i])
                g[i] -= h[i + lag]
                g[i + lag] += (1.0 - h[i])
        else:
            lag2 = -k
            for i in range(n - lag2):
                # term: h[i]*(1-h[i+lag2])
                g[i] += (1.0 - h[i + lag2])
                g[i + lag2] -= h[i]
    return g

def weighted_lag_gradient(h, top_idx, corr, top_k=10, temperature=0.01):
    """Compute weighted gradient from top lags."""
    n = h.size
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    scale = max(dv.max() + 1e-15, temperature)
    w = np.exp(dv / scale)
    w /= w.sum()

    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += w[i] * corr_gradient_single(h, k)
    return g

def upsample_h(parent, n):

def eval_c5(h):
    n = h.size

        return float('inf')

def corr_grad_lag(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    return g


def active_grad(h, top_k=10, temp=0.02):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_idx = np.argsort(corr)[-top_k:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    scale = max(dv.max() + 1e-15, temp)
    w = np.exp(dv / scale)
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for i in range(len(top_idx)):
        k = int(top_idx[i]) - center
        g += w[i] * corr_grad_lag(h, k)
    return g, corr, center, top_idx

                h_start = diverse_init(nt, rng, int(rng.integers(4)))
            else:
                h_start = upsample_h(best_h, nt)
            t_dl = time.perf_counter() + min(per_alloc, rem2 - 5)
            h_try, c_try = local_search(h_start, t_dl, seed=seed + nt + 500, top_k=10)
            if c_try < best_c:
                best_h, best_c, best_n = h_try, c_try, int(h_try.size)

    # Phase 5: Final intensive polish (remaining time)
    rem = deadline - time.perf_counter()
    if rem > 10 and best_h is not None:
        t5 = time.perf_counter() + rem * 0.6
        h5, c5 = local_search(best_h, t5, seed=seed + 700, top_k=12)
        if c5 < best_c:
            best_h, best_c, best_n = h5, c5, int(h5.size)

    # Phase 6: One more anneal if time remains
    rem = deadline - time.perf_counter()
    if rem > 15:
        t6 = time.perf_counter() + rem * 0.85
        h6, c6 = annealed_search(best_h, t6, seed=seed + 900)
        if c6 < best_c:
            best_h, best_c, best_n = h6, c6, int(h6.size)

    # Final local search to squeeze out last improvements
    rem = deadline - time.perf_counter()
    if rem > 8 and best_h is not None:
        t7 = time.perf_counter() + rem * 0.9
        h7, c7 = local_search(best_h, t7, seed=seed + 1100, top_k=12)
        if c7 < best_c:
            best_h, best_c, best_n = h7, c7, int(h7.size)

    # Final projection
    if best_h is None:
        fallback = project_density(np.full(450, 0.5), 225.0)
        return fallback, eval_c5(fallback), 450

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = eval_c5(final_h)
    return final_h, float(final_c), int(final_h.size)
