    # Numerically stable log-sum-exp

    exps = np.exp(beta * shifted)
    s = np.sum(exps)
    obj = (cmax + np.log(s) / beta) * dx
    
    # Softmax weights
    p = exps / s  # length 2n-1

def project_feasible(h, n):

    target = n / 2.0

def two_bit_search(h, n, rng, max_iters=500):

Wait, I just realized a critical issue: **L-BFGS-B does NOT handle equality constraints** (like sum(h) = n/2). It only handles bound constraints (which we have: 0 ≤ h ≤ 1). The sum constraint needs to be handled differently.

Options:
1. Use SLSQP (which handles equality constraints) - but SLSQP is what the parent uses
2. Use a penalty method: add λ*(sum(h) - n/2)² to the objective
3. Use a reparametrization: express h in terms of n-1 free variables, with the last determined by the constraint
4. Project after each iteration

I think option 4 is cleanest: run L-BFGS-B with only bound constraints, then project to satisfy the sum constraint. The projection is cheap and the L-BFGS-B iterations should keep the solution near-feasible.

Actually, a better approach: use the penalty method. Add a large penalty for violating the sum constraint:
f_total = f_smooth(h) + μ * (sum(h) - n/2)²

with μ large enough. Then L-BFGS-B can handle this with just bounds.

Or even better: reparametrize. Let h[i] = sigmoid(u[i]) for free parameters u, and choose u such that sum(sigmoid(u)) = n/2. This is harder to implement.

def lse_obj(x, beta, mu):

    shifted = corr - cmax

    s = np.sum(exps)
    obj = (cmax + np.log(s) / beta) * dx
    diff = np.sum(x) - n / 2.0

def subgradient_polish(h, n, max_iters=500):
    """Subgradient descent on the max correlation term."""
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    for it in range(max_iters):
        if remaining() < 2.0:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        m = k_max - (n - 1)

        grad = np.zeros(n)
        for j in range(n):
            s1 = 0.0
            s2 = 0.0
            # First factor: h[j] * (1-h[j+m]), valid if 0<=j+m<n
            if 0 <= j + m < n:
                s1 = 1.0 - h[j + m]
            # Second factor: h[j-m] * (1-h[j]), valid if 0<=j-m<n
            if 0 <= j - m < n:
                s2 = -h[j - m]
            grad[j] = (s1 + s2) * dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm

        improved = False
        for step in [0.05, 0.02, 0.1, 0.03, 0.08]:
            h_trial = h - step * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c5_trial
                improved = True
                break

                if c5_trial < best_c5 - 1e-14:
                    h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break

                break

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h

    for a in range(1, p):

    for i in range(n):
        h[i] = 1.0 if (i % p) in residues else 0.0
    return h
