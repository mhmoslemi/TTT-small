    # Compute all I_L for L = -(n-1), ..., n-1

def projected_gradient_step(h, grad, step_size, n):
    h_new = h - step_size * grad
    # Project to {0≤h≤1, sum(h)=n/2}
    h_new = np.clip(h_new, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return m + np.log(total) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def obj_fn(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max_val(corr, beta) * dx

def grad_fn(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return dx * grad

def sum_fun(x):
    return np.sum(x) - len(x) / 2.0

def sum_jac(x):
    return np.ones(len(x))

def project_feasible(h, n):

    n_src = len(h_src)

    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def make_sinusoidal(n, freq, amp, phase):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_block_pattern(n, n_blocks, rng):
    widths = rng.dirichlet(np.ones(n_blocks) * 0.8)
    h = np.zeros(n)
    pos = 0
    for i in range(n_blocks):
        w = max(1, int(round(widths[i] * n)))
        w = min(w, n - pos)
        val = 1.0 if i % 2 == 0 else 0.0
        h[pos:pos + w] = val
        pos += w
    if pos < n:
        h[pos:] = 1.0 if (n_blocks % 2 == 0) else 0.0
    return project_feasible(h, n)

def run_slsqp(h0, n, beta, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    res = minimize(obj_fn, h0, jac=lambda x: grad_fn(x, beta),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)
