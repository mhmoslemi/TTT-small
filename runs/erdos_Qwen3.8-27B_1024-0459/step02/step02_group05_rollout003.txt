def make_feasible(h,n):
    h=np.array(h,dtype=float).copy()
    h=np.clip(h,0,1)
    target=0.5*n
    diff=target-np.sum(h)
    if diff>0:
      idx=np.argsort(1.0-h)
      rem=diff
      for ii in idx:
        if rem <= 1e-14: break
        add = 1.0 - h[ii]
        if add > rem: add=rem
        h[ii] += add
        rem -= add
    elif diff<0:
      idx=np.argsort(h)
      rem=-diff
      for ii in idx:
        if rem <=1e-14: break
        add=h[ii]
        if add > rem: add=rem
        h[ii] -= add
        rem -= add
    h=np.clip(h,0,1)
    # if residual due fp, one more adjust on nonbounds
    diff=target-np.sum(h)
    if abs(diff)>1e-12:
       if diff>0: idx=np.where(h<1-1e-12)[0]
       else: idx=np.where(h>1e-12)[0]
       if len(idx): h[idx] += diff/len(idx); h=np.clip(h,0,1)
    return h

def make_feasible(h,n):
    h = np.array(h, dtype=float).copy()
    if len(h) != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - float(np.sum(h))
    if diff > 1e-13:
        order = np.argsort(1.0 - h)
        rem = diff
        for ii in order:
            if rem <= 1e-13:
                break
            add = 1.0 - h[ii]
            if add > rem:
                add = rem
            h[ii] += add
            rem -= add
    elif diff < -1e-13:
        order = np.argsort(h)
        rem = -diff
        for ii in order:
            if rem <= 1e-13:
                break
            add = h[ii]
            if add > rem:
                add = rem
            h[ii] -= add
            rem -= add
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size > 0:
            h[idx] += diff / idx.size
            h = np.clip(h, 0.0, 1.0)
    return h

def structured_mask_wiedemann(n):

    target=n//2
    a=np.arange(n)
    vals=(a*(a+1)//2) % n
    arr=(vals < target).astype(int)

def binary_lag_order(n,S):
    if n <=1: return []
    mask=get_full_mask(n); Z=mask^S
    bc=int.bit_count
    items=[]
    for d in range(1,n):
       p=bc(S & (Z >> d))
       q=bc((S >> d) & Z)
       cmax = p if p>q else q
       items.append((cmax,d))
    items.sort(reverse=True)
    return [d for cmax,d in items]

def get_full_mask(n):
    m = _FULL_MASK_CACHE.get(n)
    if m is None:
        m = (1 << int(n)) - 1
        _FULL_MASK_CACHE[int(n)] = m
    return m


def get_bits(n):
    n = int(n)
    b = _BIT_CACHE.get(n)
    if b is None:
        b = [1 << i for i in range(n)]
        _BIT_CACHE[n] = b
    return b


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    if len(h) != int(n):
        h = np.resize(h, int(n))
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    diff = target - float(np.sum(h))
    if diff > 1e-13:
        order = np.argsort(1.0 - h)
        rem = diff
        for ii in order:
            if rem <= 1e-13:
                break
            add = 1.0 - h[ii]
            if add > rem:
                add = rem
            h[ii] += add
            rem -= add
    elif diff < -1e-13:
        order = np.argsort(h)
        rem = -diff
        for ii in order:
            if rem <= 1e-13:
                break
            add = h[ii]
            if add > rem:
                add = rem
            h[ii] -= add
            rem -= add
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size > 0:
            h[idx] += diff / float(idx.size)
            h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    if h_src.size == n_dst:
        return h_src.copy()
    ns = h_src.size
    if ns == 0:
        return np.zeros(n_dst, dtype=float)
    x_old = np.linspace(0.0, 2.0, ns, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def project_sum(h, n):

        else:

def smooth_gradient_c(h, n, beta):
    h = np.asarray(h, dtype=float)
    n = int(n)
    if not np.all(np.isfinite(h)):
        return float("inf"), np.zeros(n, dtype=float)
    g = 1.0 - h
    c = np.correlate(h, g, mode="full")
    dx = 2.0 / float(n)
    cC = c * dx
    maxC = float(np.max(cC))
    if not np.isfinite(maxC):
        return float("inf"), np.zeros(n, dtype=float)
    beta = float(beta)
    if beta <= 1e-12:
        w = np.full(len(cC), 1.0 / len(cC), dtype=float)
    else:
        sh = cC - maxC
        ex = np.exp(beta * sh)
        s = float(np.sum(ex))
        if s <= 0.0 or not np.isfinite(s):
            w = np.full(len(cC), 1.0 / len(cC), dtype=float)
        else:
            w = ex / s
    grad = np.zeros(n, dtype=float)
    for idx in range(len(cC)):
        wi = float(w[idx])
        if wi <= 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            d = k
            seg = n - d
            if seg > 0:
                grad[:seg] += wi * g[d:]
                grad[d:] += -wi * h[:seg]
        else:
            d = -k
            seg = n - d
            if seg > 0:
                grad[d:] += wi * g[:seg]
                grad[:seg] += -wi * h[d:]
    return maxC, dx * grad


def pg_descent(n, h, beta, deadline, lr_start=1.0):
    n = int(n)
    cur = make_feasible(h, n)
    best_c = compute_c5(cur)
    lr = float(lr_start)
    lr_min = 1e-7
    no_improve = 0
    resets = 0

    while time.time() < float(deadline):
        maxC, grad = smooth_gradient_c(cur, n, beta)
        if not np.all(np.isfinite(grad)):
            break
        d = grad - float(np.mean(grad))
        norm = float(np.linalg.norm(d))
        if norm < 1e-14:
            break
        h_try = cur - lr * d
        h_try = project_sum(h_try, n)
        c_try = compute_c5(h_try)
        if np.isfinite(c_try) and c_try < best_c - 1e-14:
            cur = h_try
            best_c = c_try
            no_improve = 0
            lr = min(5.0, lr * 1.05)
        else:
            lr *= 0.6
            no_improve += 1
            if no_improve > 60:
                if resets > 3:
                    break
                resets += 1
                lr = max(1e-5, float(lr_start) * 0.5 / float(resets + 1))
                no_improve = 0
        if lr < lr_min:
            if resets > 3:
                break
            resets += 1
            lr = max(1e-5, float(lr_start) * 0.25 / float(resets + 1))
            lr_min = 1e-8
    return cur, float(best_c)


def random_perturb(n, h, rng, deadline, base_step=0.02):
    n = int(n)
    cur = make_feasible(h, n)
    best_c = compute_c5(cur)
    step = float(base_step)
    target = 0.5 * float(n)
    cap = max(3000, int(3000000 / max(1, n)))

    for _ in range(cap):
        if time.time() >= float(deadline):
            break
        h_try = cur.copy()
        m = int(rng.randint(1, n + 1))
        idx = rng.choice(n, m, replace=False)
        h_try[idx] += rng.uniform(-step, step, m)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = target - float(np.sum(h_try))
        if abs(diff) > 1e-13:
            mask = (h_try > 1e-12) & (h_try < 1.0 - 1e-12)
            cnt = int(np.count_nonzero(mask))
            if cnt > 0:
                h_try[mask] += diff / float(cnt)
                h_try = np.clip(h_try, 0.0, 1.0)
            if abs(target - float(np.sum(h_try))) > 1e-11:
                h_try = make_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if np.isfinite(c_try) and c_try < best_c - 1e-14:
            cur = make_feasible(h_try, n)
            best_c = compute_c5(cur)
            step = min(0.06, step * 1.03)
        else:
            step *= 0.98
            if step < 1e-5:
                step = max(1e-5, float(base_step) * 0.25)
    return cur, float(best_c)


def shift_blend_search(n, h, deadline):
    n = int(n)
    cur = make_feasible(h, n)
    best_c = compute_c5(cur)
    max_shift = max(1, min(n // 6, 24))
    weights = (0.25, 0.5, 0.75)
    shifts = range(1, max_shift + 1)

    for sgn in (1, -1):
        for shift in shifts:
            if time.time() >= float(deadline):
                break
            sh = sgn * shift
            for w in weights:
                h_try = w * cur + (1.0 - w) * np.roll(cur, sh)
                h_try = make_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-14:
                    cur = h_try
                    best_c = c_try
    return cur, float(best_c)


def smooth_obj(x):
    global _SLSQP_LAST_X, _SLSQP_STOP
    x = np.asarray(x, dtype=float)
    _SLSQP_LAST_X = x.copy()
    if _SLSQP_STOP or time.time() > _SLSQP_DEADLINE:
        _SLSQP_STOP = True
        raise RuntimeError("SLSQP deadline exceeded")
    n = int(_SLSQP_N)
    if n <= 0:
        return 0.0
    if not np.all(np.isfinite(x)):
        return 1e100
    c = np.correlate(x, 1.0 - x, mode="full")
    dx = 2.0 / float(n)
    cC = c * dx
    maxC = float(np.max(cC))
    if not np.isfinite(maxC):
        return 1e100
    beta = float(_SLSQP_BETA)
    if beta <= 1e-12:
        return maxC
    sh = cC - maxC
    ex = np.exp(beta * sh)
    s = float(np.sum(ex))
    if s <= 0.0 or not np.isfinite(s):
        return maxC
    return float(maxC + (1.0 / beta) * np.log(s))


def smooth_jac(x):
    global _SLSQP_STOP
    if _SLSQP_STOP or time.time() > _SLSQP_DEADLINE:
        _SLSQP_STOP = True
        raise RuntimeError("SLSQP deadline exceeded")
    x = np.asarray(x, dtype=float)
    n = int(_SLSQP_N)
    if n <= 0:
        return np.zeros(0, dtype=float)
    if not np.all(np.isfinite(x)):
        return np.zeros(n, dtype=float)
    h = x
    g = 1.0 - h
    c = np.correlate(h, g, mode="full")
    dx = 2.0 / float(n)
    cC = c * dx
    maxC = float(np.max(cC))
    if not np.isfinite(maxC):
        return np.zeros(n, dtype=float)
    beta = float(_SLSQP_BETA)
    if beta <= 1e-12:
        w = np.full(len(cC), 1.0 / len(cC), dtype=float)
    else:
        sh = cC - maxC
        ex = np.exp(beta * sh)
        s = float(np.sum(ex))
        if s <= 0.0 or not np.isfinite(s):
            w = np.full(len(cC), 1.0 / len(cC), dtype=float)
        else:
            w = ex / s
    grad = np.zeros(n, dtype=float)
    for idx in range(len(cC)):
        wi = float(w[idx])
        if wi <= 1e-15:
            continue
        k = idx - (n - 1)
        if k >= 0:
            d = k
            seg = n - d
            if seg > 0:
                grad[:seg] += wi * g[d:]
                grad[d:] += -wi * h[:seg]
        else:
            d = -k
            seg = n - d
            if seg > 0:
                grad[d:] += wi * g[:seg]
                grad[:seg] += -wi * h[d:]
    return dx * grad


def sum_eq(x):

    return np.ones(int(_SLSQP_N), dtype=float)

def structured_mask_wiedemann(n):
    n = int(n)
    if n <= 0:
        return 0
    target = n // 2
    a = np.arange(n, dtype=np.int64)
    vals = ((a * (a + 1)) // 2) % n
    arr = (vals < target).astype(int)
    cnt = int(np.sum(arr))
    ones = np.where(arr == 1)[0]
    zeros = np.where(arr == 0)[0]
    if cnt < target:
        for ii in zeros[:target - cnt]:
            arr[int(ii)] = 1
    elif cnt > target:
        for ii in ones[:cnt - target]:
            arr[int(ii)] = 0
    bits = get_bits(n)
    S = 0
    for ii in np.where(arr == 1)[0]:
        S |= bits[int(ii)]
    return S


def structured_mask_modular(n, rng):

def reverse_mask(n, S):
    n = int(n)
    R = 0
    bits = get_bits(n)
    for i in range(n):
        if S & bits[i]:
            R |= bits[n - 1 - i]
    return R


def mask_to_values(n, S):

    bits = get_bits(n)

def binary_lag_order(n, S):
    n = int(n)
    if n <= 1:
        return []
    mask = get_full_mask(n)
    Z = mask ^ S
    bc = int.bit_count
    items = []
    for d in range(1, n):
        p = bc(S & (Z >> d))
        q = bc((S >> d) & Z)
        cmax = p if p > q else q
        items.append((cmax, d))
    items.sort(reverse=True)
    return [d for cmax, d in items]


def binary_count_ordered(n, S, order, limit=None):
    n = int(n)
    if not order:
        return 0
    mask = get_full_mask(n)
    Z = mask ^ S
    bc = int.bit_count
    cmax = 0
    for d in order:
        p = bc(S & (Z >> d))
        if p > cmax:
            cmax = p
            if limit is not None and cmax > limit:
                return cmax
        q = bc((S >> d) & Z)
        if q > cmax:
            cmax = q
            if limit is not None and cmax > limit:
                return cmax
    return cmax


def binary_rotation_search(n, S, deadline):
    n = int(n)
    if n <= 1:
        return S, 0.0
    dx = 2.0 / float(n)
    mask = get_full_mask(n)
    order = binary_lag_order(n, S)
    cur_count = binary_count_ordered(n, S, order, None)
    best_S = S
    best_count = cur_count

    for shift in range(1, n):
        if time.time() >= float(deadline):
            break
        S2 = ((S << shift) | (S >> (n - shift))) & mask
        if S2 == S:
            continue
        target_count = best_count - 1
        if target_count < 0:
            break
        c = binary_count_ordered(n, S2, order, target_count)
        if c <= target_count:
            best_S = S2
            best_count = c
            order = binary_lag_order(n, S2)
    return best_S, float(best_count) * dx


def binary_hill(n, S, deadline, rng=None):
    n = int(n)
    if n <= 1:
        return S, 0.0
    dx = 2.0 / float(n)
    order = binary_lag_order(n, S)
    cur_count = binary_count_ordered(n, S, order, None)
    best_S = S
    best_count = cur_count
    bits = get_bits(n)

    while time.time() < float(deadline):
        if best_count <= 0:
            break
        target_count = best_count - 1

        ones = []
        zeros = []
        for i in range(n):
            if S & bits[i]:
                ones.append(i)
            else:
                zeros.append(i)
        if not ones or not zeros:
            break

        improved = False
        use_sample = (rng is not None) and (len(ones) * len(zeros) > 30000)
        if use_sample:
            num = min(50000, len(ones) * len(zeros))
            for _ in range(num):
                if time.time() >= float(deadline):
                    break
                p = ones[int(rng.randint(len(ones)))]
                q = zeros[int(rng.randint(len(zeros)))]
                S2 = S ^ bits[p] ^ bits[q]
                c = binary_count_ordered(n, S2, order, target_count)
                if c <= target_count:
                    S = S2
                    cur_count = c
                    best_S = S
                    best_count = c
                    order = binary_lag_order(n, S)
                    improved = True
                    break
        else:
            for p in ones:
                if time.time() >= float(deadline):
                    break
                bp = bits[p]
                for q in zeros:
                    S2 = S ^ bp ^ bits[q]
                    c = binary_count_ordered(n, S2, order, target_count)
                    if c <= target_count:
                        S = S2
                        cur_count = c
                        best_S = S
                        best_count = c
                        order = binary_lag_order(n, S)
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best_S, float(best_count) * dx


def binary_sa(n, S, deadline, rng, t_start=10.0, t_end=0.15):
    n = int(n)
    if n <= 1:
        return S, 0.0
    dx = 2.0 / float(n)
    order = binary_lag_order(n, S)
    cur_count = binary_count_ordered(n, S, order, None)
    cur_c = float(cur_count) * dx
    best_S = S
    best_count = cur_count
    best_c = cur_c
    bits = get_bits(n)

    start = time.time()
    remaining = max(0.0, float(deadline) - start)
    if remaining <= 0.0:
        return S, cur_c

    if n <= 256:
        steps_per_sec = 12000
    elif n <= 384:
        steps_per_sec = 5000
    else:
        steps_per_sec = 2000
    max_steps = int(max(20000, min(1000000, remaining * float(steps_per_sec))))

    last_improve = 0
    for step in range(max_steps):
        if step % 1000 == 0 and time.time() >= float(deadline):
            break

        while True:
            p = int(rng.randint(n))
            if S & bits[p]:
                break
        while True:
            q = int(rng.randint(n))
            if not (S & bits[q]):
                break
        if p == q:
            continue

        progress = float(step) / float(max(1, max_steps))
        T = float(t_start) * (float(t_end) / float(t_start)) ** progress
        if T < 1e-6:
            T = 1e-6

        worst = cur_count + int(round(T)) + 1
        if worst > n - 1:
            worst = n - 1
        S2 = S ^ bits[p] ^ bits[q]
        c = binary_count_ordered(n, S2, order, worst)
        if c <= worst:
            delta = c - cur_count
            if delta < 0 or (delta >= 0 and rng.random() < math.exp(-float(delta) / max(T, 1e-9))):
                S = S2
                cur_count = c
                cur_c = float(c) * dx
                if c < best_count:
                    best_S = S
                    best_count = c
                    best_c = cur_c
                    order = binary_lag_order(n, S)
                    last_improve = step
                if step - last_improve > 30000:
                    S = best_S
                    cur_count = best_count
                    cur_c = best_c
                    order = binary_lag_order(n, S)
                    last_improve = step
    return best_S, float(best_c)


def binary_refine(n, S, deadline, rng):
    n = int(n)
    if n <= 1:
        return S, 0.0
    if n % 2 != 0:
        n -= 1
    dx = 2.0 / float(n)
    order = binary_lag_order(n, S)
    init_count = binary_count_ordered(n, S, order, None)
    best_S = S
    best_c = float(init_count) * dx

    if time.time() >= float(deadline):
        return best_S, best_c

    rem = float(deadline) - time.time()
    d_rot = time.time() + min(rem * 0.15, 20.0)
    if d_rot > float(deadline):
        d_rot = float(deadline)
    S_r, c_r = binary_rotation_search(n, S, d_rot)
    if c_r < best_c:
        best_S, best_c = S_r, c_r
        S = S_r

    rem = float(deadline) - time.time()
    d_hill = time.time() + min(rem * 0.3, 45.0)
    if d_hill > float(deadline):
        d_hill = float(deadline)
    S1, c1 = binary_hill(n, S, d_hill, rng)
    if c1 < best_c:
        best_S, best_c = S1, c1
        S = S1

    rem = float(deadline) - time.time()
    if rem > 15.0:
        d_sa = time.time() + min(rem * 0.6, 140.0)
        if d_sa > float(deadline):
            d_sa = float(deadline)
        S2, c2 = binary_sa(n, S, d_sa, rng)
        if c2 < best_c:
            best_S, best_c = S2, c2
            S = S2

    rem = float(deadline) - time.time()
    if rem > 8.0:
        d_hill2 = time.time() + min(rem * 0.6, 60.0)
        if d_hill2 > float(deadline):
            d_hill2 = float(deadline)
        S3, c3 = binary_hill(n, S, d_hill2, rng)
        if c3 < best_c:
            best_S, best_c = S3, c3

    return best_S, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(int(seed))
    t0 = time.time()
    total = max(30.0, float(budget_s))
    end = t0 + total - 5.0

    parent = None
    try:
        raw = initial_h_values
        if raw is not None:
            arr = np.asarray(raw, dtype=float).ravel()
            if arr.size > 0:
                parent = arr
    except Exception:
        parent = None

    if parent is not None:
        n = int(parent.size)
        h = make_feasible(parent, n)
    else:
        n = 160
        h = structured_initial(n)

    best_h = h
    best_n = n
    best_c = compute_c5(best_h)

    # Parent-space continuous polish.
    phase_deadline = min(end - 420.0, t0 + 260.0)
    if phase_deadline > time.time() + 25:
        if best_n <= 200:
            sls_words = [(40, 200, 70), (150, 160, 50), (500, 120, 35), (1500, 90, 25)]
        elif best_n <= 280:
            sls_words = [(150, 80, 40), (500, 60, 30)]
        else:
            sls_words = []
        for beta, mit, dur in sls_words:
            sub = min(phase_deadline, time.time() + float(dur))
            if sub <= time.time() + 10:
                break
            h1, c1 = run_slsqp_smooth(best_n, best_h, beta, mit, sub)
            if c1 < best_c - 1e-12:
                best_h, best_c = h1, c1

        if best_n <= 400:
            sub = min(phase_deadline, time.time() + 70.0)
            if sub > time.time() + 15:
                h2, c2 = pg_descent(best_n, best_h, 700, sub, 1.0)
                if c2 < best_c - 1e-12:
                    best_h, best_c = h2, c2

        if best_n <= 400:
            sub = min(phase_deadline, time.time() + 30.0)
            if sub > time.time() + 10:
                h3, c3 = shift_blend_search(best_n, best_h, sub)
                if c3 < best_c - 1e-12:
                    best_h, best_c = h3, c3

    # Binary search at the parent resolution.
    phase_deadline = min(end - 360.0, time.time() + 220.0)
    if phase_deadline > time.time() + 35:
        n_bin = best_n if (best_n % 2 == 0) else (best_n - 1)
        if n_bin < 8:
            n_bin = 160
        if n_bin != best_n:
            base_h = make_feasible(interpolate_to_n(best_h, n_bin), n_bin)
        else:
            base_h = best_h

        S0 = mask_from_h(base_h, n_bin)
        masks = [
            S0,
            reverse_mask(n_bin, S0),
            structured_mask_wiedemann(n_bin),
            structured_mask_modular(n_bin, rng),
            random_mask(n_bin, rng),
        ]

        local_best_c = best_c
        local_best_S = S0
        for Smask in masks:
            if time.time() >= phase_deadline - 12:
                break
            per = max(15.0, (phase_deadline - time.time()) / max(1, len(masks)))
            sub = min(phase_deadline, time.time() + min(per, 60.0))
            if sub <= time.time() + 8:
                continue
            Sopt, cbin = binary_refine(n_bin, Smask, sub, rng)
            if cbin < local_best_c - 1e-12:
                local_best_S = Sopt
                local_best_c = cbin
                hbin = make_feasible(mask_to_values(n_bin, Sopt), n_bin)
                cexact = compute_c5(hbin)
                if cexact < best_c - 1e-12:
                    best_n = n_bin
                    best_h = hbin
                    best_c = cexact

        if best_n == n_bin:
            sub = min(phase_deadline, time.time() + 45.0)
            if sub > time.time() + 10:
                h2, c2 = continuous_refine(n_bin, best_h, sub, rng)
                if c2 < best_c - 1e-12:
                    best_n = n_bin
                    best_h = h2
                    best_c = c2

    # A few nearby / stronger resolutions.
    if time.time() < end - 260:
        base_n = best_n if (best_n % 2 == 0) else (best_n - 1)
        cand = []
        for m in (base_n + 30, 256, 320):
            if m % 2 != 0:
                m += 1
            if 120 <= m <= 480 and m != best_n and m not in cand:
                cand.append(m)
        cand.sort()
        extra_deadline = min(end - 160.0, time.time() + 320.0)

        for n2 in cand[:2]:
            if time.time() >= extra_deadline - 90:
                break
            h2 = make_feasible(interpolate_to_n(best_h, n2), n2)
            c2 = compute_c5(h2)

            sub_cont = min(extra_deadline - 70, time.time() + (85.0 if n2 <= 220 else 40.0))
            if sub_cont > time.time() + 15:
                h3, c3 = continuous_refine(n2, h2, sub_cont, rng)
                if c3 < c2 - 1e-12:
                    h2, c2 = h3, c3

            if c2 < best_c - 1e-12:
                best_n = n2
                best_h = h2
                best_c = c2

            n_b = n2 if (n2 % 2 == 0) else (n2 - 1)
            if n_b != n2:
                h_b = make_feasible(interpolate_to_n(h2, n_b), n_b)
            else:
                h_b = h2
            S2 = mask_from_h(h_b, n_b)

            sub_bin = min(extra_deadline - 45, time.time() + 70.0)
            if sub_bin > time.time() + 15:
                Sopt, cbin = binary_refine(n_b, S2, sub_bin, rng)
                if cbin < best_c - 1e-12:
                    hbin = make_feasible(mask_to_values(n_b, Sopt), n_b)
                    cexact = compute_c5(hbin)
                    if cexact < best_c - 1e-12:
                        best_n = n_b
                        best_h = hbin
                        best_c = cexact

    # Final polish.
    if time.time() < end - 12:
        sub = min(end - 5.0, time.time() + 70.0)
        if sub > time.time() + 8:
            h2, c2 = pg_descent(best_n, best_h, 1200, sub, 0.45)
            if c2 < best_c - 1e-12:
                best_h, best_c = h2, c2

    if time.time() < end - 6:
        sub = min(end - 2.0, time.time() + 25.0)
        if sub > time.time() + 4:
            h3, c3 = random_perturb(best_n, best_h, rng, sub, 0.012)
            if c3 < best_c - 1e-12:
                best_h, best_c = h3, c3

    best_h = make_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
