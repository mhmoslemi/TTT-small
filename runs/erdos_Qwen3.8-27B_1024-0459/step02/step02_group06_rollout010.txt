# Global variables for the smooth objective
_NC = 1
_ALPHA = 1.0
_TARGET = 0.5

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    # Project onto [0,1]^n with sum = target
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-10:
                free2 = np.where((h > 0.0) & (h < 1.0))[0]
                if free2.size > 0:
                    h[free2[0]] += res2
                    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)

def smooth_objective(x):
    n = _NC
    alpha = _ALPHA
    if n <= 1:
        return 0.0
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / float(n)) * (m + np.log(z) / alpha))

def smooth_jacobian(x):
    n = _NC
    alpha = _ALPHA
    if n <= 1:
        return np.zeros(max(1, n), dtype=float)
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return np.zeros(n, dtype=float)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n, dtype=float)
    w = e / z
    # Only use significant weights
    idx = np.where(w > 1e-8)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(w))], dtype=int)
    g = np.zeros(n, dtype=float)
    center = n - 1
    for t in idx:
        t = int(t)
        p = float(w[t])
        if p <= 0.0:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
            else:
                g[:n - lag] += p
            g[:n - lag] -= p * h[lag:]
            g[lag:] -= p * h[:n - lag]
    return (2.0 / float(n)) * g

def sum_constraint(x):
    return float(np.sum(x) - _TARGET)

def sum_constraint_jac(x):

def corr_grad_lag(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

def active_set_gradient(h, top_frac, min_k, temp_scale=1.0):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    top_idx = np.argsort(corr)[-k_count:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    w = np.exp(dv / (temp_scale * (dv.max() + 1e-10)))
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += w[i] * corr_grad_lag(h, k)
    return g

    h = np.asarray(h, dtype=float)
