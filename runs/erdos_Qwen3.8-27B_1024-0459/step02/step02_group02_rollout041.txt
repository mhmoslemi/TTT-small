        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None

    best_h = None
    best_c5 = np.inf
    best_n = 0

    # Primary resolution: 512
    n = 512

    # Generate initial points
    inits = []

    # From parent
    if h_parent is not None:
        h_p = make_feasible(interp_h(h_parent, n), n)
        inits.append(h_p)

    # Thue-Morse
    inits.append(thue_morse_init(n))

    # Random inits
    for _ in range(3):
        h_r = (rng.rand(n) > 0.5).astype(float)
        ones = np.where(h_r == 1.0)[0]
        zeros = np.where(h_r == 0.0)[0]
        target = n // 2
        if len(ones) > target:
            h_r[rng.choice(ones, len(ones) - target, replace=False)] = 0.0
        elif len(ones) < target:
            h_r[rng.choice(zeros, target - len(ones), replace=False)] = 1.0
        inits.append(h_r)

    # Phase 1: Continuous optimization from each init
    for h0 in inits:
        if remaining() < 90:
            break
        h_opt, c5_val = continuous_optimize(h0, n,
            beta_list=[15.0, 80.0, 250.0, 500.0], max_iter=150)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n

    # Phase 2: Local search from best
    if remaining() > 60 and best_h is not None:
        h_ls, c5_ls = local_search(best_h, n, rng,
            max_trials=40000, timeout=min(40.0, remaining() * 0.15), stall_limit=8000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n

    # Phase 3: Simulated annealing
    if remaining() > 60 and best_h is not None:
        h_sa, c5_sa = simulated_annealing(best_h, n, rng,
            t_start=0.2, t_end=0.001, n_iter=120000,
            timeout=min(80.0, remaining() * 0.2))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n

    # Phase 4: Targeted search
    if remaining() > 40 and best_h is not None:
        h_t, c5_t = targeted_local_search(best_h, n, rng,
            max_trials=20000, timeout=min(30.0, remaining() * 0.15), stall_limit=4000)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
            best_n = n

    # Phase 5: Another round of local search + SA
    if remaining() > 50 and best_h is not None:
        h_ls2, c5_ls2 = local_search(best_h, n, rng,
            max_trials=30000, timeout=min(25.0, remaining() * 0.15), stall_limit=6000)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
            best_n = n

    if remaining() > 40 and best_h is not None:
        h_sa2, c5_sa2 = simulated_annealing(best_h, n, rng,
            t_start=0.1, t_end=0.0005, n_iter=100000,
            timeout=min(50.0, remaining() * 0.25))
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()
            best_n = n

    # Phase 6: Try n=1024 if time permits
    if remaining() > 200:
        n2 = 1024
        inits2 = []
        if h_parent is not None:
            inits2.append(make_feasible(interp_h(h_parent, n2), n2))
        if best_h is not None:
            inits2.append(make_feasible(interp_h(best_h, n2), n2))
        inits2.append(thue_morse_init(n2))

        for h0 in inits2[:3]:
            if remaining() < 120:
                break
            h_opt2, c5_val2 = continuous_optimize(h0, n2,
                beta_list=[20.0, 100.0, 300.0], max_iter=100)
            if c5_val2 < best_c5:
                best_c5 = c5_val2
                best_h = h_opt2.copy()
                best_n = n2

        if remaining() > 60 and best_n == n2:
            h_ls3, c5_ls3 = local_search(best_h, n2, rng,
                max_trials=20000, timeout=min(30.0, remaining() * 0.2), stall_limit=5000)
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()
                best_n = n2

    # Phase 7: Final polish on best solution
    if remaining() > 20 and best_h is not None:
        h_final, c5_final = local_search(best_h, best_n, rng,
            max_trials=15000, timeout=min(15.0, remaining() * 0.3), stall_limit=3000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility check
    if best_h is not None:
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    else:
        # Fallback
        best_h = thue_morse_init(256)
        best_c5 = compute_c5(best_h)
        best_n = 256

    return (best_h, best_c5, best_n)
