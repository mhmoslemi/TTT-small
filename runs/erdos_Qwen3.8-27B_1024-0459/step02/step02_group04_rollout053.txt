def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:
        n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    target = 0.5 * float(n)
    h = h[:n].copy()
    if h.size < n:
        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    elif h.size > n:
        h = h[:n]
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(50):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-14
        else:
            mask = out > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_all_lags(h):
    n = h.size
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def lag_grad(h, L):
    n = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    LA = abs(int(L))
    m = n - LA
    g = np.zeros(n, dtype=float)
    if L > 0:
        g[:m] += 1.0 - h[LA:]
        g[LA:] += -h[:m]
    else:
        g[LA:] += 1.0 - h[:m]
        g[:m] += -h[LA:]
    return g


def sum_lag_grad(h, lags):
    n = len(h)
    g = np.zeros(n, dtype=float)
    for L in lags:
        g += lag_grad(h, int(L))
    return g

def smooth_obj(x, beta):
    h = np.asarray(x, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    lv = compute_all_lags(h)
    m = float(np.max(lv))
    shifted = beta * (lv - m)
    shifted[shifted > 0] = 0
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    if s <= 0:
        return m
    return m + np.log(s) / beta


def smooth_obj_jac(x, beta):
    h = np.asarray(x, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    lv = compute_all_lags(h)
    m = float(np.max(lv))
    shifted = beta * (lv - m)
    shifted[shifted > 0] = 0
    exps = np.exp(shifted)
    total = float(np.sum(exps))
    if total <= 0:
        return np.zeros(n)
    w = exps / total
    g = np.zeros(n, dtype=float)
    for i in range(len(lv)):
        lag_idx = i - (n - 1)
        if w[i] > 1e-15:
            g += w[i] * lag_grad(h, lag_idx)
    return g

def active_constraint_obj(x, lags):
    h = np.asarray(x[:-1], dtype=float)
    t = float(x[-1])
    n = h.size
    dx = 2.0 / float(n)
    penalty = 0.0
    for L in lags:
        L = int(L)
        if L == 0:
            acc = float(np.sum(h) - np.dot(h, h))
        elif L > 0:
            m = n - L
            acc = float(np.sum(h[:m]) - np.dot(h[:m], h[L:]))
        else:
            m = n + L
            acc = float(np.sum(h[-L:]) - np.dot(h[:m], h[-L:]))
        v = acc * dx
        if v > t:
            penalty += (v - t) ** 2
    return t + 1000.0 * penalty


def active_obj_jac(x, lags):

    t = float(x[-1])
    n = h.size
    dx = 2.0 / float(n)
    g = np.zeros(n + 1, dtype=float)
    g[-1] = 1.0
    for L in lags:
        L = int(L)
        Lg = lag_grad(h, L)
        v = 0.0
        if L == 0:
            v = float(np.sum(h) - np.dot(h, h)) * dx

            v = (float(np.sum(h[:m])) - float(np.dot(h[:m], h[L:]))) * dx

def run_slsqp_active(h0, n, lags, maxiter):

def projected_grad_descent(h0, n, time_limit, start_step=0.02, decay=0.995):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    step = start_step
    lv = compute_all_lags(h)
    m = lv.size
    K = min(8, m)

    trial = 0
    max_trials = 3000
    while time.time() < time_limit and trial < max_trials:
        lv = compute_all_lags(h)
        cur_c = float(np.max(lv))
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

        order = np.argsort(-lv)
        K = min(int(len(lv)), max(4, trial // 50 + 4))
        top_k = order[:K]
        top_vals = -lv[top_k]
        sort_top = np.argsort(-top_vals)
        active_lags = top_k[sort_top[:K]]

        g = np.zeros(n)
        for L in active_lags:
            lag = int(L - (n - 1))
            g += lag_grad(h, lag)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break

        for attempt in range(8):
            h_new = project_feasible(h - step * g / g_norm, n)
            c_new = compute_c5(h_new)
            if c_new <= cur_c + 0.5 * step * g_norm:
                break
            step *= 0.5
        h = h_new
        step = max(1e-5, step * decay)
        trial += 1

    return best_h, float(best_c)


def local_block_perturb(h0, n, time_limit, seed, step_init=0.025):

def variant_starts(base, n, rng):
    base = project_feasible(base, n)
    out = [base.copy()]
    if n > 1:
        flip = base[::-1]
        out.append(project_feasible(0.5 * (base + flip), n))
    sigma = max(0.5, n * 0.01)
    half = int(3 * sigma) + 1
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * sigma * sigma))
    w /= w.sum()
    out.append(project_feasible(np.convolve(base, w, mode="same"), n))
    pert = base + rng.randn(n) * 0.04
    out.append(project_feasible(pert, n))
    return out[:4]

            best_n = int(best_n)

    # Phase B: Try nearby resolutions
    resolutions = sorted(set([best_n, best_n - 16, best_n + 16, best_n - 32, best_n + 32,
                               128, 192, 224, 256, 288]))
    resolutions = [r for r in resolutions if 64 <= r <= 320]

    processed = {best_n}
    phase_b_end = min(hard_end - 120.0, t_start + 500.0)

    for nn in resolutions:
        nn = int(nn)
        if time.time() > phase_b_end:
            break
        if nn in processed:
            continue
        processed.add(nn)
        rem = phase_b_end - time.time()
        if rem < 30:
            break
        slot = min(90.0, rem)

        h_interp = interp_to_n(best_h, nn)
        if parent is not None and parent.size != nn:
            h_interp_parent = interp_to_n(parent, nn)
        else:
            h_interp_parent = h_interp

        cands = [h_interp, h_interp_parent]
        cands.extend(variant_starts(h_interp, nn, rng))
        if len(cands) < 3:
            cands.extend(structured_inits(nn, rng))
        cands = cands[:4]

        for kj, cand in enumerate(cands):
            if time.time() > t_start + slot:
                break
            per_time = t_start + slot
            h3, c3 = refine_loop(cand, nn, per_time)
            if c3 < best_c:
                best_h = h3
                best_c = c3
                best_n = nn

    # Phase C: Final intensive refinement at best resolution
    if time.time() < hard_end - 40 and 2 <= best_n <= 320:
        h4, c4 = refine_loop(best_h, best_n, hard_end - 30)
        if c4 < best_c:
            best_h = h4
            best_c = c4
            best_n = int(best_n)

    # Phase D: Local block perturb (annealing)
    if time.time() < hard_end - 10 and 2 <= best_n <= 280:
        h5, c5 = local_block_perturb(best_h, best_n, hard_end - 3, seed + 7777, step_init=0.02)
        if c5 < best_c:
            best_h = h5
            best_c = c5
            best_n = int(best_n)

    # Phase E: One more SLSQP polish
    if time.time() < hard_end - 5:
        h6, c6 = run_slsqp_smooth(best_h, best_n, 600, maxiter=30)
        if c6 < best_c:
            best_h = h6
            best_c = c6
            best_n = int(best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    # Verify constraints
    assert len(best_h) == best_n
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(np.sum(best_h) - 0.5 * best_n) < 1e-8
    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h, float(best_c), int(best_n))
