from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_opt_j(h, j, n, corr, ells):
    """Find optimal value for h[j] given all other coordinates fixed."""
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        sg = np.mean(a[active])
        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, n_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    ells = np.arange(-(n - 1), n)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr, ells)
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def fix_sum_binary(h, n):
    """Fix a binary vector to have exactly n/2 ones."""
    target = n // 2
    s = int(np.sum(h))
    if s > target:
        # Need to flip (s - target) ones to zeros
        ones = np.where(h == 1)[0]
        if len(ones) >= s - target:
            # Flip the "least important" ones (random)
            to_flip = np.random.choice(ones, s - target, replace=False)
            h[to_flip] = 0
    elif s < target:
        # Need to flip (target - s) zeros to ones
        zeros = np.where(h == 0)[0]
        if len(zeros) >= target - s:
            to_flip = np.random.choice(zeros, target - s, replace=False)
            h[to_flip] = 1
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_opt_j(h, j, n, corr):
    ells = np.arange(-(n - 1), n)
    idx1 = j - ells
    idx2 = j + ells
    a = np.zeros(2 * n - 1)
    mask1 = (idx1 >= 0) & (idx1 < n)
    if np.any(mask1):
        a[mask1] += 1.0 - h[idx1[mask1]]
    mask2 = (idx2 >= 0) & (idx2 < n)
    if np.any(mask2):
        a[mask2] += -h[idx2[mask2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        sg = np.mean(a[active])
        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def coord_descent(h, n, n_passes, time_limit, rng=None):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        for j in order:
            t_opt = coord_opt_j(h, j, n, corr)
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def to_binary_smart(h, n, rng):
    h_bin = (h > 0.5).astype(float)
    s = int(np.sum(h_bin))
    target = n // 2
    if s > target:
        ones = np.where(h_bin == 1)[0]
        idx_to_flip = rng.choice(ones, s - target, replace=False)
        h_bin[idx_to_flip] = 0.0
    elif s < target:
        zeros = np.where(h_bin == 0)[0]
        idx_to_flip = rng.choice(zeros, target - s, replace=False)
        h_bin[idx_to_flip] = 1.0
    return h_bin


def binary_sa(h_init, n, time_limit, rng, t0=1.0, cooling=0.9995, swap_size=1):
    h = h_init.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = t0
    start = time.time()
    no_improve = 0
    max_no_improve = 5000

    while time.time() - start < time_limit and T > 1e-5:
        if no_improve > max_no_improve:
            T = max(T * 0.95, 0.05)
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3)

        nf = rng.randint(1, swap_size + 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < nf or len(zeros) < nf:
            break

        flip_to_zero = rng.choice(ones, nf, replace=False)
        flip_to_one = rng.choice(zeros, nf, replace=False)

        h_try = h.copy()
        h_try[flip_to_zero] = 0.0
        h_try[flip_to_one] = 1.0

        c5_try = compute_c5(h_try)
        delta = c5_try - c5_cur

        if delta < -1e-14:
            h = h_try
            c5_cur = c5_try
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                no_improve = 0
            else:
                no_improve += 1
        elif rng.random() < np.exp(-delta / T):
            h = h_try
            c5_cur = c5_try
            no_improve += 1
        else:
            no_improve += 1

        T *= cooling

    return best_h, best_c5


def continuous_sa(h_init, n, time_limit, rng, t0=0.05):
    h = h_init.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = t0
    start = time.time()
    step_scale = 0.3

    while time.time() - start < time_limit and T > 1e-6:
        j = rng.randint(n)
        delta = rng.uniform(-step_scale, step_scale)
        h_try = h.copy()
        h_try[j] = np.clip(h[j] + delta, 0.0, 1.0)
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        diff = c5_try - c5_cur
        if diff < -1e-14:
            h = h_try
            c5_cur = c5_try
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
        elif rng.random() < np.exp(-diff / T):
            h = h_try
            c5_cur = c5_try
        T *= 0.9997

    return best_h, best_c5


def slsqp_polish(h, n, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}]
    
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def structured_inits(n, rng):
    inits = []
    # Block pattern
    h1 = np.zeros(n)
    h1[:n//4] = 1.0
    h1[n//2 : n//2 + n//4] = 1.0
    inits.append(h1.copy())
    # Alternating pairs
    if n % 4 == 0:
        h2 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h2.copy())
    # Alternating singles
    if n % 2 == 0:
        h3 = np.tile([1.0, 0.0], n // 2)
        inits.append(h3.copy())
    # Thue-Morse
    h4 = np.zeros(n)
    for i in range(n):
        h4[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(h4, n))
    # Random binary
    for _ in range(3):
        hr = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        hr[idx] = 1.0
        inits.append(hr)
    return inits
