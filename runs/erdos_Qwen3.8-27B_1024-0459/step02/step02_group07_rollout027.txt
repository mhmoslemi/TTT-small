import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_subgradient_vec(h, corr, tol):
    n = len(h)
    max_c = np.max(corr)
    active = corr >= max_c - tol
    n_active = np.sum(active)
    if n_active == 0:
        active = np.zeros(len(corr), dtype=bool)
        active[np.argmax(corr)] = True
        n_active = 1
    grad = np.zeros(n)
    ells = np.arange(-(n - 1), n)
    for k in range(len(corr)):
        if not active[k]:
            continue
        m = ells[k]
        if m >= 0:
            n_terms = n - m
            if n_terms > 0:
                grad[0:n_terms] += (1.0 - h[m:])
            if m > 0:
                grad[m:n] += (-h[0:n - m])
        else:
            m2 = -m
            n_terms = n - m2
            if n_terms > 0:
                grad[m2:] += (1.0 - h[0:n_terms])
                grad[0:n_terms] += (-h[m2:])
    grad /= n_active
    return grad


def fw_linear_min(grad, n):
    target = n // 2
    idx = np.argsort(grad)
    h_fw = np.zeros(n)
    h_fw[idx[:target]] = 1.0
    return h_fw


def fw_optimize(h, n, max_iter=100000, time_limit=120, tol_init=0.15, tol_min=0.001):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    tol = tol_init
    stagnant = 0
    last_improve = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h.copy()
            stagnant = 0
            last_improve = t
        else:
            stagnant = t - last_improve
        if stagnant > 5000:
            tol = max(tol * 0.7, tol_min)
            stagnant = 0
        grad = compute_subgradient_vec(h, corr, tol)
        h_fw = fw_linear_min(grad, n)
        alpha = 2.0 / (t + 2)
        h_new = (1.0 - alpha) * h + alpha * h_fw
        h = make_feasible(h_new, n)
    return best_h, best_c5


def coord_opt_j(h, j, n, corr):
    ells = np.arange(-(n - 1), n)
    a = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        m = ells[k]
        if m >= 0:
            if 0 <= j <= n - 1 - m:
                a[k] += (1.0 - h[j + m])
            if m > 0 and m <= j <= n - 1:
                a[k] += (-h[j - m])
        else:
            m2 = -m
            if m2 <= j <= n - 1:
                a[k] += (1.0 - h[j - m2])
            if 0 <= j <= n - 1 - m2:
                a[k] += (-h[j + m2])
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(30):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        sg = np.mean(a[active])
        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            hi = mid
            lo = mid
            break
    return 0.5 * (lo + hi)


def coord_descent(h, n, n_passes=100, time_limit=60):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr)
            if abs(t_opt - h[j]) > 1e-10:
                h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def binary_sa(h_init, n, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    h_bin = (h_init > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = n / 2.0 - s
    if deficit > 0.5:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_init[zeros])[:int(round(deficit))]
        h_bin[zeros[order]] = 1.0
    elif deficit < -0.5:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_init[ones])[:int(round(-deficit))]
        h_bin[ones[order]] = 0.0
    # Force exact sum
    s = np.sum(h_bin)
    if s > n / 2.0:
        ones = np.where(h_bin == 1)[0]
        for _ in range(int(s - n / 2.0)):
            h_bin[ones[rng.randint(len(ones))]] = 0.0
    elif s < n / 2.0:
        zeros = np.where(h_bin == 0)[0]
        for _ in range(int(n / 2.0 - s)):
            h_bin[zeros[rng.randint(len(zeros))]] = 1.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_corr = get_corr(best_h).copy()
    start = time.time()
    T = 0.1
    T_end = 0.001
    n_iters = 0
    while time.time() - start < time_limit:
        n_iters += 1
        T = max(T_end, T * 0.9998)
        n_flip = rng.randint(1, 5)
        idxs = rng.choice(n, n_flip, replace=False)
        h_try = best_h.copy()
        for idx in idxs:
            h_try[idx] = 1.0 - h_try[idx]
        if abs(np.sum(h_try) - n / 2.0) > 1e-10:
            continue
        c5_try = compute_c5(h_try)
        delta = c5_try - best_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-8)):
            best_h = h_try
            best_c5 = c5_try
            best_corr = get_corr(best_h)
    return best_h, best_c5


def m_sequence_init(n, seed=42):
    rng = np.random.RandomState(seed)
    # Generate a random binary sequence with n/2 ones
    h = np.zeros(n)
    positions = rng.choice(n, n // 2, replace=False)
    h[positions] = 1.0
    return h


def slsqp_polish(h, n, max_iter=3000):
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))

    def grad_func(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        max_c = np.max(corr)
        active = corr >= max_c - 0.01
        n_active = max(np.sum(active), 1)
        grad = np.zeros(n)
        ells = np.arange(-(n - 1), n)
        for k in range(len(corr)):
            if not active[k]:
                continue
            m = ells[k]
            if m >= 0:
                n_terms = n - m
                if n_terms > 0:
                    grad[0:n_terms] += (1.0 - x[m:])
                if m > 0:
                    grad[m:n] += (-x[0:n - m])
            else:
                m2 = -m
                n_terms = n - m2
                if n_terms > 0:
                    grad[m2:] += (1.0 - x[0:n_terms])
                    grad[0:n_terms] += (-x[m2:])
        return (2.0 / n) * grad / n_active

    def con_func(x):
        return np.sum(x) - n / 2.0

    try:
        res = minimize(obj_func, h, jac=grad_func, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_func}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def multi_start_fw(h_init, n, n_starts=3, time_limit=60, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    tl_per = time_limit / (n_starts + 1)
    # First: optimize the init itself
    h_opt, c5_opt = fw_optimize(h_init, n, max_iter=50000, time_limit=tl_per, tol_init=0.1)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()
    for s in range(n_starts):
        if time.time() - (time.time() - time_limit) > time_limit:
            break
        h_rand = make_feasible(rng.rand(n), n)
        # Blend with best
        h_blend = 0.5 * best_h + 0.5 * h_rand
        h_blend = make_feasible(h_blend, n)
        remaining_tl = time_limit - (s + 1) * tl_per
        if remaining_tl < 5:
            break
        h_opt2, c5_opt2 = fw_optimize(h_blend, n, max_iter=30000, time_limit=remaining_tl, tol_init=0.15)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()
    return best_h, best_c5
