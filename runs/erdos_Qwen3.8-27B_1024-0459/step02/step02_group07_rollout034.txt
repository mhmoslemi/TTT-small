def fast_coord_descent(h, n, n_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # shape (2n-1,)
    shifts = np.arange(-(n-1), n)  # shift for each correlation index
    start = time.time()
    
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            # Compute a_k for this j
            s = shifts  # shape (2n-1,)
            valid1 = (j + s >= 0) & (j + s < n)
            valid2 = (j - s >= 0) & (j - s < n)
            a = np.zeros(2*n-1)
            a[valid1] += (1.0 - h[j + s[valid1]])
            a[valid2] += (-h[j - s[valid2]])
            
            # b_k = corr[k] - h[j] * a[k]
            b = corr - h[j] * a
            
            # Find optimal t in [0,1] to minimize max_k (a[k]*t + b[k])
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = a * mid + b
                mx = np.max(vals)
                active = vals >= mx - 1e-14
                sg = np.mean(a[active])
                if sg > 1e-13:
                    hi = mid
                elif sg < -1e-13:
                    lo = mid
                else:
                    break
            
            new_val = 0.5 * (lo + hi)
            delta = new_val - h[j]
            if abs(delta) > 1e-15:
                # Update corr
                corr += delta * a
                h[j] = new_val
        
        h = make_feasible(h, n)
        # Recompute corr after feasibility fix
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    f_max = np.max(f)
    Tf = T * (f - f_max)
    Tf = np.clip(Tf, -700.0, 700.0)
    exp_Tf = np.exp(Tf)
    Z = np.sum(exp_Tf)
    weights = exp_Tf / Z
    obj = f_max + np.log(Z) / T
    sum_h = np.sum(h)
    diff_sum = sum_h - n / 2.0
    obj += penalty * diff_sum * diff_sum
    J = np.arange(n)[:, None]
    S = np.arange(-(n - 1), n)[None, :]
    valid1 = (J + S >= 0) & (J + S < n)
    valid2 = (J - S >= 0) & (J - S < n)
    idx1 = np.clip(J + S, 0, n - 1)
    idx2 = np.clip(J - S, 0, n - 1)
    A = np.sum(h[idx1] * weights[None, :] * valid1, axis=1)
    B = np.sum(h[idx2] * weights[None, :] * valid2, axis=1)
    grad = dx * (1.0 - A - B)
    grad += 2.0 * penalty * diff_sum
    return obj, grad


def smooth_max_optimize(h0, n, T_values, penalty, time_limit):
    h = make_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    for T in T_values:
        if time.time() - start > time_limit:
            break
        T_use = float(T)

        def obj_fn(x):
            v, _ = smooth_max_obj_grad(x, T_use, penalty)
            return v

        def grad_fn(x):
            _, g = smooth_max_obj_grad(x, T_use, penalty)
            return g

        try:
            res = minimize(obj_fn, h, method='L-BFGS-B', jac=grad_fn,
                           bounds=bounds,
                           options={'maxiter': 3000, 'ftol': 1e-16,
                                    'gtol': 1e-10, 'maxls': 80})
            h_new = make_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
        h = make_feasible(h, n)
    return best_h, best_c5

def generate_structured_init(n):
    inits = []
    if n % 2 == 0:
        # Block: first half 1, second half 0
        h3 = np.zeros(n)
        h3[:n // 2] = 1.0
        inits.append(h3.copy())
        # Interleaved
        h2 = np.zeros(n)
        h2[::2] = 1.0
        inits.append(h2.copy())
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h1.copy())
    # Thue-Morse based
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm.copy(), n))
    # Golden ratio based
    gr = np.zeros(n)
    for i in range(n):
        gr[i] = 1.0 if (i * 0.6180339887 % 1.0) < 0.5 else 0.0
    inits.append(make_feasible(gr.copy(), n))
    return inits


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except NameError:
        h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent.copy(), best_n)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: Fast coordinate descent at parent resolution
    if remaining() > 60:
        tl = min(120, remaining() - 40)
        h_cd, c5_cd = fast_coord_descent(best_h, best_n, n_passes=5000, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: Smooth max optimization at parent resolution
    if remaining() > 60:
        tl = min(100, remaining() - 50)
        T_vals = [10, 25, 50, 100, 200, 400]
        h_sm, c5_sm = smooth_max_optimize(best_h, best_n, T_vals, penalty=1000.0, time_limit=tl)
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()

    # Phase 3: More coordinate descent after smooth max
    if remaining() > 60:
        tl = min(100, remaining() - 50)
        h_cd, c5_cd = fast_coord_descent(best_h, best_n, n_passes=5000, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Try higher resolutions
    resolutions = sorted(set([best_n + 60, best_n + 100, best_n + 160, best_n + 240]))
    for n_new in resolutions:
        if remaining() < 80:
            break
        # Only try if n_new is even (for balanced binary)
        if n_new % 2 != 0:
            n_new += 1
        h_init = interpolate_to_n(best_h, n_new)
        h_cd, c5_cd = fast_coord_descent(h_init, n_new, n_passes=3000, time_limit=min(60, remaining() - 60))
        if remaining() > 60:
            T_vals2 = [20, 50, 100, 200]
            h_sm, c5_sm = smooth_max_optimize(h_cd, n_new, T_vals2, penalty=1000.0, time_limit=min(50, remaining() - 50))
            if c5_sm < c5_cd:
                h_cd = h_sm
                c5_cd = c5_sm
        if remaining() > 40:
            h_cd2, c5_cd2 = fast_coord_descent(h_cd, n_new, n_passes=2000, time_limit=min(30, remaining() - 30))
            if c5_cd2 < c5_cd:
                h_cd = h_cd2
                c5_cd = c5_cd2
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_new

    # Phase 5: Binary refinement
    if remaining() > 30:
        tl = min(40, remaining() - 20)
        h_bin, c5_bin = binary_refine(best_h, best_n, time_limit=tl)
        if c5_bin < best_c5 - 1e-12:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 6: Final coordinate descent polish
    if remaining() > 20:
        tl = max(5, remaining() - 10)
        h_cd, c5_cd = fast_coord_descent(best_h, best_n, n_passes=5000, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: One more smooth max with high T
    if remaining() > 30:
        T_final = [100, 300, 500]
        h_sm, c5_sm = smooth_max_optimize(best_h, best_n, T_final, penalty=5000.0, time_limit=min(25, remaining() - 10))
        if c5_sm < best_c5:
            best_c5 = c5_sm
            best_h = h_sm.copy()
        if remaining() > 15:
            h_cd, c5_cd = fast_coord_descent(best_h, best_n, n_passes=2000, time_limit=max(5, remaining() - 5))
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 8: Try structured inits at best_n if time permits
    if remaining() > 60:
        for h_init in generate_structured_init(best_n):
            if remaining() < 30:
                break
            h_cd, c5_cd = fast_coord_descent(h_init, best_n, n_passes=2000, time_limit=min(20, remaining() - 20))
            if c5_cd < best_c5 - 1e-12:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
