        # j ranges over [0,n), need j+s in [0,n)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_lo < j_hi:
            # grad[j] += wm * (1-h[j+s]) for j in [j_lo, j_hi)
            grad[j_lo:j_hi] += wm * (1.0 - h[j_lo+s:j_hi+s])
        # Contribution from h[j] in (1-h) term: i+s=j, i=j-s
        # j ranges over [0,n), need j-s in [0,n)
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_lo2 < j_hi2:
            # grad[j] -= wm * h[j-s] for j in [j_lo2, j_hi2)
            grad[j_lo2:j_hi2] -= wm * h[j_lo2-s:j_hi2-s]
    
    # Add penalty gradient
    grad += 2e6 * (np.sum(h) - n / 2.0)
    
    return f, grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, beta, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    exp_bc = np.exp(beta * (c - c_max))
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp
    f = c_max + np.log(sum_exp) / beta
    sviol = np.sum(h) - n / 2.0
    f += 1e5 * sviol * sviol
    
    grad = np.zeros(n)

        s = m - (n - 1)
        wm = w[m] * dx

            continue

        j_hi = min(n, n - s)

        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= wm * h[j_lo2 - s:j_hi2 - s]
    grad += 2e5 * sviol

    return np.interp(x_new, xp, src_h)

        grad = np.zeros(n)
        if offset >= 0:
            n_pairs = n - offset
            grad[offset:offset + n_pairs] += (1.0 - h[:n_pairs])

        else:
