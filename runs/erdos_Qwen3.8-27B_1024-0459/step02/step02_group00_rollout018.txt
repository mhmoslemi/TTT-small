def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(50):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        c = int(m.sum())
        if c > 0:
            h[m] += d / c
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def project_binary(h, n):
    k = n // 2
    idx = np.argsort(-h, kind='stable')
    hb = np.zeros(n)
    hb[idx[:k]] = 1.0
    return hb

def smooth_obj(x):
    n = _N
    dx = 2.0 / n
    c = get_corr(x)
    m = float(c.max())
    b = _BETA
    if b < 1.0:
        return m * dx
    sh = (c - m) * b
    sh[sh < -50.0] = -50.0
    val = (m + np.log(np.exp(sh).sum()) / b) * dx
    return float(val)

def raw_obj(x):
    n = _N
    dx = 2.0 / n
    c = get_corr(x)
    return float(c.max() * dx)

def sum_con(x):
    return float(x.sum() - _TGT)

    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) / sn
    xn = (np.arange(dn) + 0.5) / dn
    return np.interp(xn, xp, src)

def tm_seq(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        p = 0
        while v:
            p ^= v & 1
            v >>= 1
        h[i] = p
    return h.astype(float)

    _N = n
    _TGT = n / 2.0

                     bounds=bnd, constraints=cns,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(r.x, n)
    except Exception:
        return project_feasible(h0, n)

def swap_local_search(h, n, tlim=60, seed=None):
    """Greedy pairwise-swap local search for binary h."""
    h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    rng_ls = np.random.RandomState(seed if seed is not None else 0)
    
    while time.time() - t0 < tlim:
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        rng_ls.shuffle(ones)
        rng_ls.shuffle(zeros)
        for p in ones:
            for q in zeros:
                if time.time() - t0 > tlim:
                    return h, best
                h[p] = 0.0
                h[q] = 1.0
                c = compute_c5(h)
                if c < best - 1e-15:
                    best = c
                    improved = True
                    # restart inner loop
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    rng_ls.shuffle(ones)
                    rng_ls.shuffle(zeros)
                    break
                else:
                    h[p] = 1.0
                    h[q] = 0.0
        if not improved:
            break
    return h, best

def simulated_annealing(h, n, tmax=60, seed=0):
    """SA with swap moves on binary h."""
    rng_sa = np.random.RandomState(seed)
    h = h.copy()
    best = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    T = 0.3
    cooling = 0.995

    n_reject = 0

                n_accept += 1

            h[q] = 0.0
            n_reject += 1

    return best_h, best

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(50):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(m.sum())
        if cnt > 0:
            h[m] += d / cnt
        else:
            h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h


def project_binary(h, n):
    k = n // 2
    idx = np.argsort(-h, kind='stable')
    hb = np.zeros(n)
    hb[idx[:k]] = 1.0
    return hb


def smooth_obj(x):
    n = _N
    dx = 2.0 / n
    c = get_corr(x)
    m = float(c.max())
    b = _BETA
    if b < 1.0:
        return m * dx
    sh = (c - m) * b
    sh = np.where(sh < -50.0, -50.0, sh)
    return float((m + np.log(np.exp(sh).sum()) / b) * dx)


def raw_obj(x):
    n = _N
    dx = 2.0 / n
    c = get_corr(x)
    return float(c.max() * dx)


def sum_con(x):
    return float(x.sum() - _TGT)


def reschedule(src, sn, dn):
    if sn == dn:
        return src.copy()
    xp = (np.arange(sn) + 0.5) / sn
    xn = (np.arange(dn) + 0.5) / dn
    return np.interp(xn, xp, src)


def tm_seq(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        p = 0
        while v:
            p ^= v & 1
            v >>= 1
        h[i] = float(p)
    return h


def rand_bin(n, rng):
    h = np.zeros(n)
    sel = rng.choice(n, size=n // 2, replace=False)
    h[sel] = 1.0
    return h


def slsqp_opt(h0, n, beta, maxiter=400):
    global _N, _TGT, _BETA
    _N = n
    _TGT = n / 2.0
    _BETA = beta
    bnd = [(0.0, 1.0)] * n
    cns = [{'type': 'eq', 'fun': sum_con}]
    try:
        r = minimize(smooth_obj, h0.astype(float), method='SLSQP',
                     bounds=bnd, constraints=cns,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_feasible(r.x, n)
    except Exception:
        return project_feasible(h0, n)


def swap_ls(h, n, tlim=60.0, sfmt=0):
    h = h.copy()
    best = compute_c5(h)
    t0 = time.time()
    rl = np.random.RandomState(sfmt)
    while time.time() - t0 < tlim:
        imp = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        rl.shuffle(ones)
        rl.shuffle(zeros)
        for p in ones:
            for q in zeros:
                if time.time() - t0 > tlim:
                    return h, best
                h[p] = 0.0
                h[q] = 1.0
                cv = compute_c5(h)
                if cv < best - 1e-15:
                    best = cv
                    imp = True
                    ones = np.where(h > 0.5)[0]
                    zeros = np.where(h < 0.5)[0]
                    rl.shuffle(ones)
                    rl.shuffle(zeros)
                    break
                else:
                    h[p] = 1.0
                    h[q] = 0.0
        if not imp:
            break
    return h, best


def sa_search(h, n, tmax=60.0, sfmt=0):
    rs = np.random.RandomState(sfmt)
    h = h.copy()
    cur = compute_c5(h)
    best = cur
    bh = h.copy()
    t0 = time.time()
    T = 0.3
    cool = 0.998
    while (time.time() - t0) < tmax and T > 0.005:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        p = ones[rs.randint(len(ones))]
        q = zeros[rs.randint(len(zeros))]
        h[p] = 0.0
        h[q] = 1.0
        cn = compute_c5(h)
        dlt = cn - cur
        if dlt < 0 or rs.rand() < np.exp(-dlt / T):
            cur = cn
            if cn < best:
                best = cn
                bh = h.copy()
        else:
            h[p] = 1.0
            h[q] = 0.0
        T *= cool
    return bh, best

f(h) = (m + (1/β) ln Σ exp(β(c_k - m))) × dx

As β→∞, f(h) → m × dx (the exact max).

∂f/∂h_i = dx × [w_max + Σ_k w_k × ∂c_k/∂h_i]
