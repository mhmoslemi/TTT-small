        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(tm, n)

def grad_descent_anneal(h_init, n, timeout, beta_start=3.0, beta_end=300.0,
                        n_phases=10, iters_per_phase=4000, lr0=0.04,
                        use_adaptive=True):
    h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    start_t = time.time()
    lr = lr0
    
    for phase in range(n_phases):
        if time.time() - start_t > timeout:
            break
        t = phase / max(n_phases - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr0 / (1.0 + 0.8 * phase)
        
        stall = 0
        max_stall = 500
        for it in range(iters_per_phase):
            if time.time() - start_t > timeout:
                break
            val, grad, corr = smooth_max_and_grad(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            h_new = h - lr * grad
            h_new = make_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                h = h_new
                stall = 0
            else:
                stall += 1
                if stall > max_stall:
                    # Small perturbation to escape
                    h_pert = h + 0.01 * np.random.randn(n)
                    h_pert = make_feasible(h_pert, n)
                    if compute_c5(h_pert) <= best_c5:
                        h = h_pert
                    lr *= 0.5
                    stall = 0
                    if lr < 1e-6:
                        break
    return h, best_c5

_SLSQP_N = 0
_SLSQP_BETA = 50.0

def _slsqp_obj(x):
    h = np.clip(x, -0.5, 1.5)
    val, _, _ = smooth_max_and_grad(h, _SLSQP_BETA)
    return val

def _slsqp_grad(x):
    h = np.clip(x, -0.5, 1.5)
    _, grad, _ = smooth_max_and_grad(h, _SLSQP_BETA)
    return grad

def _slsqp_sum_fun(x):
    return np.sum(x) - _SLSQP_N / 2.0

def _slsqp_sum_jac(x):
    return np.ones(_SLSQP_N)

def slsqp_polish(h_init, n, beta=50.0, max_iter=5000):
    global _SLSQP_N, _SLSQP_BETA
    _SLSQP_N = n
    _SLSQP_BETA = beta
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _slsqp_sum_fun, 'jac': _slsqp_sum_jac}
    try:
        res = minimize(_slsqp_obj, h0, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=con,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h0, compute_c5(h0)

def binary_local_search(h_init, n, rng, max_trials=200000, timeout=120.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    stall_limit = 25000
    start_t = time.time()
    
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, best_c5

def targeted_search(h_init, n, rng, max_trials=150000, timeout=90.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    stall_limit = 20000
    start_t = time.time()
    
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        corr = compute_corr(h)
        # Find top 3 worst shifts
        top_shifts = np.argsort(corr)[-3:]
        worst_idx = top_shifts[-1]
        s = worst_idx - (n - 1)
        
        # Find positions contributing to worst shift
        bad_ones = []
        if s >= 0:
            for i in range(n - s):
                if h[i] == 1.0 and h[i + s] == 0.0:
                    bad_ones.append(i)
        else:
            for i in range(-s, n):
                if h[i] == 1.0 and h[i + s] == 0.0:
                    bad_ones.append(i)
        
        if len(bad_ones) > 0:
            i = bad_ones[rng.randint(len(bad_ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None
    
    # === n=256 optimization ===
    n256 = 256
    if h_parent is not None and len(h_parent) == n256:
        h256 = h_parent.copy()
    elif h_parent is not None:
        h256 = make_feasible(interp_h(h_parent, n256), n256)
    else:
        h256 = thue_morse_init(n256)
    
    best_h256 = h256.copy()
    best_c5_256 = compute_c5(best_h256)
    
    # GD round 1: low beta, many iterations
    if remaining() > 200:
        h_g, c5_g = grad_descent_anneal(h256, n256, timeout=min(250, remaining() - 50),
                                         beta_start=2.0, beta_end=150.0,
                                         n_phases=12, iters_per_phase=3000, lr0=0.03)
        if c5_g < best_c5_256:
            best_c5_256 = c5_g
            best_h256 = h_g.copy()
    
    # SLSQP polish with analytical gradient
    if remaining() > 120:
        h_s, c5_s = slsqp_polish(best_h256, n256, beta=30.0, max_iter=4000)
        if c5_s < best_c5_256:
            best_c5_256 = c5_s
            best_h256 = h_s.copy()
    
    # Binary local search
    if remaining() > 80:
        h_b, c5_b = binary_local_search(best_h256, n256, rng, max_trials=300000,
                                         timeout=min(80, remaining() - 30))
        if c5_b < best_c5_256:
            best_c5_256 = c5_b
            best_h256 = make_feasible(np.round(h_b), n256)
    
    # Targeted search
    if remaining() > 60:
        h_t, c5_t = targeted_search(best_h256, n256, rng, max_trials=200000,
                                     timeout=min(50, remaining() - 25))
        if c5_t < best_c5_256:
            best_c5_256 = c5_t
            best_h256 = make_feasible(np.round(h_t), n256)
    
    # GD round 2: high beta for final polish
    if remaining() > 80:
        h_g2, c5_g2 = grad_descent_anneal(best_h256, n256, timeout=min(100, remaining() - 30),
                                           beta_start=20.0, beta_end=400.0,
                                           n_phases=8, iters_per_phase=3000, lr0=0.015)
        if c5_g2 < best_c5_256:
            best_c5_256 = c5_g2
            best_h256 = h_g2.copy()
    
    # SLSQP with higher beta
    if remaining() > 60:
        h_s2, c5_s2 = slsqp_polish(best_h256, n256, beta=80.0, max_iter=3000)
        if c5_s2 < best_c5_256:
            best_c5_256 = c5_s2
            best_h256 = h_s2.copy()
    
    # More binary search from best
    if remaining() > 50:
        h_b2, c5_b2 = binary_local_search(best_h256, n256, rng, max_trials=200000,
                                           timeout=min(40, remaining() - 20))
        if c5_b2 < best_c5_256:
            best_c5_256 = c5_b2
            best_h256 = make_feasible(np.round(h_b2), n256)
    
    # Final SLSQP
    if remaining() > 40:
        h_s3, c5_s3 = slsqp_polish(best_h256, n256, beta=100.0, max_iter=3000)
        if c5_s3 < best_c5_256:
            best_c5_256 = c5_s3
            best_h256 = h_s3.copy()
    
    # === n=512 optimization ===
    n512 = 512
    h512 = make_feasible(interp_h(best_h256, n512), n512)
    best_h512 = h512.copy()
    best_c5_512 = compute_c5(best_h512)
    
    if remaining() > 150:
        h_g5, c5_g5 = grad_descent_anneal(h512, n512, timeout=min(200, remaining() - 50),
                                           beta_start=2.0, beta_end=200.0,
                                           n_phases=10, iters_per_phase=2500, lr0=0.02)
        if c5_g5 < best_c5_512:
            best_c5_512 = c5_g5
            best_h512 = h_g5.copy()
    
    if remaining() > 80:
        h_b5, c5_b5 = binary_local_search(best_h512, n512, rng, max_trials=200000,
                                           timeout=min(70, remaining() - 30))
        if c5_b5 < best_c5_512:
            best_c5_512 = c5_b5
            best_h512 = make_feasible(np.round(h_b5), n512)
    
    if remaining() > 60:
        h_g5b, c5_g5b = grad_descent_anneal(best_h512, n512, timeout=min(80, remaining() - 25),
                                             beta_start=15.0, beta_end=350.0,
                                             n_phases=7, iters_per_phase=2000, lr0=0.01)
        if c5_g5b < best_c5_512:
            best_c5_512 = c5_g5b
            best_h512 = h_g5b.copy()
    
    if remaining() > 40:
        h_s5, c5_s5 = slsqp_polish(best_h512, n512, beta=60.0, max_iter=2000)
        if c5_s5 < best_c5_512:
            best_c5_512 = c5_s5
            best_h512 = h_s5.copy()
    
    # Also try from Thue-Morse at n=512
    if remaining() > 60:
        h_tm512 = thue_morse_init(n512)
        h_gtm, c5_gtm = grad_descent_anneal(h_tm512, n512, timeout=min(50, remaining() - 20),
                                             beta_start=3.0, beta_end=150.0,
                                             n_phases=6, iters_per_phase=1500, lr0=0.02)
        if c5_gtm < best_c5_512:
            best_c5_512 = c5_gtm
            best_h512 = h_gtm.copy()
    
    # === Choose best ===
    best_h256 = make_feasible(best_h256, n256)
    best_c5_256 = compute_c5(best_h256)
    best_h512 = make_feasible(best_h512, n512)
    best_c5_512 = compute_c5(best_h512)
    
    if best_c5_256 <= best_c5_512:
        return (best_h256, best_c5_256, n256)
    else:
        return (best_h512, best_c5_512, n512)
