def pair_swap(h, deadline, max_iter=2000, top_frac=0.05, min_k=8, n_pos=10, n_neg=10):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    rng = np.random.default_rng(54321)
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        center = n - 1
        k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        rng_val = dv.max() + 1e-10
        w = np.exp(dv / rng_val)
        w /= w.sum()
        g = np.zeros(n, dtype=float)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            g += w[i] * corr_grad_lag(best, k)
        neg_cand = np.argsort(g)[:n_neg]
        pos_cand = np.argsort(-g)[:n_pos]
        found = False
        for ii in neg_cand:
            ii = int(ii)
            for jj in pos_cand:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                for frac in (0.25, 0.5, 0.75, 1.0, 0.9):
                    d = frac * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    if cand[ii] < -1e-12 or cand[ii] > 1 + 1e-12:
                        continue
                    if cand[jj] < -1e-12 or cand[jj] > 1 + 1e-12:
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            for _r in range(25):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * 0.8 * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
    return project_density(best, target), evaluate_erdos(best)


def triple_swap(h, deadline, max_iter=400, top_frac=0.05, min_k=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 3:
        return best, best_c
    rng = np.random.default_rng(777)
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        center = n - 1
        k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        rng_val = dv.max() + 1e-10
        w = np.exp(dv / rng_val)
        w /= w.sum()
        g = np.zeros(n, dtype=float)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            g += w[i] * corr_grad_lag(best, k)
        idx_sorted = np.argsort(g)
        neg_pool = idx_sorted[:5]
        pos_pool = idx_sorted[-5:]
        found = False
        for ii in neg_pool:
            ii = int(ii)
            for jj in pos_pool:
                jj = int(jj)
                for kk in pos_pool:
                    kk = int(kk)
                    if ii == jj or ii == kk or jj == kk:
                        continue
                    cap = min(best[ii], 1.0 - best[jj], 1.0 - best[kk])
                    if cap < 1e-10:
                        continue
                    d = 0.5 * cap
                    cand = best.copy()
                    cand[ii] -= d
                    cand[jj] += 0.6 * d
                    cand[kk] += 0.4 * d
                    if np.any(cand < -1e-12) or np.any(cand > 1 + 1e-12):
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            for _r in range(8):
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                i3 = int(rng.integers(n))
                if i1 == i2 or i1 == i3 or i2 == i3:
                    continue
                d = float(rng.random()) * 0.05
                cand = best.copy()
                cand[i1] -= d
                cand[i2] += 0.6 * d
                cand[i3] += 0.4 * d
                if np.any(cand < -1e-12) or np.any(cand > 1 + 1e-12):
                    continue
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if not found:
            stall += 1
            if stall > 15:
                break
        else:
            stall = 0
    return project_density(best, target), evaluate_erdos(best)


def make_diverse_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 5 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.empty(n, dtype=float)
        seg = max(1, int(n // 10))
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = float(rng.uniform(0.1, 0.9))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.75, 0.25).astype(float)
        h += rng.normal(0, 0.04, n)
    elif mode == 3:
        freq = int(rng.integers(3, 9))
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2 + rng.uniform(0, 6.28))
    else:
        h = 0.5 + rng.normal(0, 0.15, n)
    return project_density(h, 0.5 * n)


def optimize_at_resolution(n, h0, deadline):
    global _G_N, _G_ALPHA, _G_TARGET
    _G_N = n
    _G_TARGET = 0.5 * n
    h = project_density(h0, _G_TARGET)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": g_sum_fun, "jac": g_sum_jac},)

    if n <= 500:
        alphas = (15.0, 50.0, 150.0, 400.0)
        maxiter = 200
    elif n <= 1000:
        alphas = (15.0, 50.0, 150.0)
        maxiter = 120
    else:
        alphas = (15.0, 50.0)
        maxiter = 80

    for alpha in alphas:
        if time.perf_counter() >= deadline - 8.0:
            break
        _G_ALPHA = alpha
        try:
            res = minimize(
                smooth_objective, h,
                method="SLSQP", jac=smooth_jacobian,
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-11, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, _G_TARGET)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                h = cand if c < best_c + 0.02 else best_h
        except Exception:
            pass

    rem = deadline - time.perf_counter()
    if rem > 6.0:
        dl2 = time.perf_counter() + rem * 0.35
        h2, c2 = pair_swap(best_h, dl2, max_iter=1000, top_frac=0.05, min_k=8)
        if c2 < best_c:
            best_h, best_c = h2, c2

    rem = deadline - time.perf_counter()
    if rem > 5.0:
        dl3 = time.perf_counter() + rem * 0.55
        h3, c3 = polish_gradient(best_h, dl3, max_iter=500, top_frac=0.04, min_k=10)
        if c3 < best_c:
            best_h, best_c = h3, c3

    rem = deadline - time.perf_counter()
    if rem > 4.0 and n <= 600:
        dl4 = time.perf_counter() + rem * 0.6
        h4, c4 = triple_swap(best_h, dl4, max_iter=300, top_frac=0.04, min_k=10)
        if c4 < best_c:
            best_h, best_c = h4, c4

    rem = deadline - time.perf_counter()
    if rem > 3.0:
        dl5 = time.perf_counter() + rem * 0.8
        h5, c5 = polish_gradient(best_h, dl5, max_iter=300, top_frac=0.03, min_k=12)
        if c5 < best_c:
            best_h, best_c = h5, c5

    return project_density(best_h, _G_TARGET), evaluate_erdos(best_h)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 20.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_

    if best_h is None:
        n0 = 100
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    base_n = best_n if best_n > 1 else 100

    resolutions = set()
    resolutions.add(base_n)
    for mult in (2, 4):
        target_n = base_n * mult
        if 2 <= target_n <= 2000:
            resolutions.add(target_n)
    for extra in (90, 120, 180, 240, 360, 450, 600, 900, 1200, 1800):
        if 2 <= extra <= 2000:
            resolutions.add(extra)
    n_list = sorted(resolutions)

    # Allocate time: prioritize resolutions close to and above base_n
    total_avail = deadline - time.perf_counter()
    n_list_filtered = []
    for n in n_list:
        if n <= 2500:
            n_list_filtered.append(n)
    n_list_filtered.sort()

    weights = []
    for n in n_list_filtered:
        if n <= base_n:
            weights.append(1.0)
        elif n <= 2 * base_n:
            weights.append(2.0)
        elif n <= 4 * base_n:
            weights.append(2.5)
        else:
            weights.append(1.5)
    wsum = sum(weights)
    time_alloc = {}
    for n, w in zip(n_list_filtered, weights):
        time_alloc[n] = max(20.0, total_avail * w / wsum)

    # Sort by priority: base_n first, then nearby, then others
    def priority(n):
        if n == base_n:
            return 0
        if n == 2 * base_n:
            return 1
        if n == 4 * base_n:
            return 2
        dist = abs(n - base_n)
        return dist
    n_list_sorted = sorted(n_list_filtered, key=priority)

    for ni in n_list_sorted:
        rem = deadline - time.perf_counter()
        if rem < 20.0:
            break
        alloc = time_alloc.get(ni, max(15.0, rem / max(1, len(n_list_sorted))))
        alloc = min(alloc, rem - 5.0)

        # Build starting points
        starts = []
        starts.append(upsample_h(best_h, ni))
        if parent is not None and parent.size != ni:
            starts.append(upsample_h(parent, ni))
        elif parent is not None:
            starts.append(parent.copy())

        n_rand = 2 if ni <= 300 else (1 if ni <= 900 else 0)
        for mm in range(n_rand):
            starts.append(make_diverse_start(ni, rng, mm % 4))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-5):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        per_start = alloc / max(1, len(uniq))
        per_start = max(8.0, per_start)

        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 10.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 3.0)
            try:
                h_out, c_out = optimize_at_resolution(ni, s, sub_dl)
                if h_out is not None:
                    h_out = project_density(h_out, 0.5 * ni)
                    c_out = evaluate_erdos(h_out)
                    if c_out < best_c:
                        best_h = h_out.copy()
                        best_c = c_out
                        best_n = ni
            except Exception:
                pass

    # Final intensive polish
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 10.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)

        # Extended pair swap
        if rem > 15.0:
            dl = time.perf_counter() + rem * 0.3
            h2, c2 = pair_swap(h_fin, dl, max_iter=5000, top_frac=0.03, min_k=12, n_pos=15, n_neg=15)
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h_fin.copy()

        # Gradient polish
        rem2 = deadline - time.perf_counter()
        if rem2 > 8.0:
            dl = time.perf_counter() + rem2 * 0.5
            h3, c3 = polish_gradient(h_fin, dl, max_iter=2000, top_frac=0.03, min_k=15)
            if c3 < best_c:
                h_fin = h3
                best_c = c3
                best_h = h_fin.copy()

        # Another pair swap round
        rem3 = deadline - time.perf_counter()
        if rem3 > 6.0:
            dl = time.perf_counter() + rem3 * 0.6
            h4, c4 = pair_swap(h_fin, dl, max_iter=3000, top_frac=0.02, min_k=15, n_pos=8, n_neg=8)
            if c4 < best_c:
                h_fin = h4
                best_c = c4
                best_h = h_fin.copy()

        # Final gradient
        rem4 = deadline - time.perf_counter()
        if rem4 > 4.0:
            dl = time.perf_counter() + rem4 * 0.8
            h5, c5 = polish_gradient(h_fin, dl, max_iter=1000, top_frac=0.02, min_k=20)
            if c5 < best_c:
                h_fin = h5
                best_c = c5
                best_h = h_fin.copy()

    if best_h is None:
        fallback = project_density(np.full(100, 0.5), 50.0)
        return fallback, evaluate_erdos(fallback), 100

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
