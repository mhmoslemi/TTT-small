Let me now think about the LP formulation more carefully.

We want to solve:
minimize t
subject to:
  For each active lag (type, L): f_type(L; h) ≤ t   (these are the "real" constraints)
  0 ≤ h[i] ≤ 1 for all i
  sum(h) = n/2

The problem is that f_type(L; h) is NOT linear in h (it has h[i]*h[j] terms). So we approximate:
f_type(L; h) ≈ f_type(L; h*) + ∇f_type(L; h*)·(h - h*)

This gives: f_type(L; h*) + ∇f_type(L; h*)·h - ∇f_type(L; h*)·h* ≤ t
i.e.: ∇f_type(L; h*)·h - t ≤ ∇f_type(L; h*)·h* - f_type(L; h*)

In matrix form for linprog:
minimize c^T x where x = [h_0, ..., h_{n-1}, t]
subject to A_ub x ≤ b_ub, A_eq x = b_eq, bounds

c = [0, ..., 0, 1] (minimize t)

For each active lag constraint:
row of A_ub: [∇f_1(h*), ..., ∇f_n(h*), -1]
b_ub entry: ∇f(h*)·h* - f(h*)

A_eq: [[1, 1, ..., 1, 0]] (sum constraint)
b_eq: [n/2]

Bounds: 0 ≤ h[i] ≤ 1, 0 ≤ t ≤ 1 (or some upper bound)

    """

    b_ub_list = []

        grad = lag_gradient(h_star, idx, n)  # length n

        g[Lp:] -= h[:m]
        return g


def solve_lp_step(h_star, active_indices, n, relax=1e-6):
    c = np.concatenate([np.zeros(n), [1.0]])
    A_rows = []
    b_list = []
    vals_star = all_lag_values(h_star)
    for idx in active_indices:
        idx = int(idx)
        if idx < 0 or idx >= len(vals_star):
            continue
        val = float(vals_star[idx])
        grad = lag_gradient_vec(h_star, idx, n)
        row = np.concatenate([grad, [-1.0]])
        rhs = float(np.dot(grad, h_star)) - val + relax
        A_rows.append(row)
        b_list.append(rhs)
    if not A_rows:
        return None
    A_ub = np.array(A_rows, dtype=float)
    b_ub = np.array(b_list, dtype=float)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n], dtype=float)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None
    if res is not None and res.success:
        return res.x[:n].copy(), float(res.x[n])
    return None


def lp_optimize(h0, n, time_limit, max_iter=300, seed=42):
    rng = np.random.RandomState(seed)
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / float(n)
    no_progress = 0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        vals = all_lag_values(h)
        c5_cur = float(np.max(vals)) * dx
        if c5_cur < best_c - 1e-13:
            best_c = c5_cur
            best_h = h.copy()
        no_progress = 0 if c5_cur < best_c + 1e-12 else no_progress + 1
        if no_progress > 30:
            h_pert = h + rng.randn(n) * 0.008
            h = project_feasible(h_pert, n)
            no_progress = 0
            continue
        order = np.argsort(-vals)
        top_k = min(40, 2 * n - 1)
        active = order[:top_k].tolist()
        if it % 7 == 3:
            n_extra = min(10, len(vals) - top_k)
            if n_extra > 0:
                extra = rng.choice(len(vals), size=n_extra, replace=False)
                active.extend(int(x) for x in extra)
        result = solve_lp_step(h, active, n, relax=1e-7)
        if result is None:
            no_progress += 1
            if no_progress > 20:
                h_pert = h + rng.randn(n) * 0.005
                h = project_feasible(h_pert, n)
                no_progress = 0
            continue
        h_lp, t_lp = result
        h_lp = project_feasible(h_lp, n)
        c5_lp = compute_c5(h_lp)
        accepted = False
        if c5_lp < c5_cur - 1e-12:
            h = h_lp
            accepted = True
        else:
            for alpha in (0.5, 0.3, 0.15, 0.08):
                h_try = project_feasible((1.0 - alpha) * h + alpha * h_lp, n)
                c5_try = compute_c5(h_try)
                if c5_try < c5_cur - 1e-13:
                    h = h_try
                    accepted = True
                    break
        if not accepted:
            no_progress += 1
    final_c = compute_c5(h)
    if final_c < best_c:
        best_h = h.copy()
        best_c = final_c
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def local_block_search(h0, n, time_limit, seed=42, step=0.025):
    rng = np.random.RandomState(seed)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h.copy()
    cur_c = c
    temp = 0.002
    min_temp = 1e-7
    max_trials = 8000
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(10, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.03, max(temp * 1.2, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def smooth_obj_func(x, n, beta):
    h = np.asarray(x, dtype=float)
    dx = 2.0 / float(n)
    vals = all_lag_values(h)
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -745.0, 0.0)
    r = float(np.sum(np.exp(shifted)))
    return (m + np.log(r) / beta) * dx


def smooth_sum_eq(x):
    return float(np.sum(x)) - 0.5 * len(x)


def smooth_sum_eq_jac(x):
    return np.ones(len(x), dtype=float)


def polish_smooth(h0, n, beta, maxiter=60, ftol=1e-12):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": smooth_sum_eq, "jac": smooth_sum_eq_jac}]
    try:
        res = minimize(smooth_obj_func, h, args=(n, beta), method="SLSQP",
                       bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": ftol, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    return h, compute_c5(h)


def structured_start(n, rng):
    h = np.zeros(n, dtype=float)
    q = max(1, n // 4)
    h[:q] = 1.0
    h[n // 2:n // 2 + q] = 1.0
    return project_feasible(h, n)


def oscillation_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5, dtype=float)
    for freq in (1, 2, 3, 5, 7):
        amp = rng.uniform(0.05, 0.2)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(freq * np.pi * x + phase)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        best_h = project_feasible(parent, parent_n)
        best_c = compute_c5(best_h)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: LP optimization at parent resolution
    if time.time() < hard_end - 80.0:
        t1_end = min(hard_end - 80.0, start_time + budget_s * 0.45)
        hh, cc = lp_optimize(best_h, best_n, t1_end, max_iter=400, seed=seed)
        if cc < best_c:
            best_h = hh
            best_c = cc

    # Phase 2: Local block search at best resolution
    if time.time() < hard_end - 60.0:
        t2_end = min(hard_end - 60.0, start_time + budget_s * 0.65)
        hh, cc = local_block_search(best_h, best_n, t2_end, seed=seed + 1000, step=0.02)
        if cc < best_c:
            best_h = hh
            best_c = cc

    # Phase 3: Multi-resolution exploration
    if time.time() < hard_end - 50.0:
        resolutions = [n for n in (128, 160, 192, 224, 256, 288, 320) if n != best_n]
        resolutions.sort(key=lambda x: abs(x - best_n))
        for nn in resolutions[:4]:
            if time.time() > hard_end - 50.0:
                break
            rem = hard_end - 50.0 - time.time()
            if rem < 20.0:
                break
            init = interpolate_to_n(best_h, nn)
            slot = min(rem * 0.5, 80.0)
            slot = max(10.0, slot)
            hh, cc = lp_optimize(init, nn, time.time() + slot, max_iter=150, seed=seed + 5000 + nn)
            if cc < best_c:
                best_h = hh
                best_c = cc
                best_n = nn

    # Phase 4: Second LP pass on best
    if time.time() < hard_end - 40.0 and best_n <= 320:
        t4_end = min(hard_end - 40.0, start_time + budget_s * 0.82)
        hh, cc = lp_optimize(best_h, best_n, t4_end, max_iter=200, seed=seed + 9000)
        if cc < best_c:
            best_h = hh
            best_c = cc

    # Phase 5: Smooth polish
    if time.time() < hard_end - 25.0 and 4 <= best_n <= 320:
        for beta in (50, 150):
            if time.time() > hard_end - 25.0:
                break
            hh, cc = polish_smooth(best_h, best_n, beta, maxiter=50, ftol=1e-12)
            if cc < best_c:
                best_h = hh
                best_c = cc

    # Phase 6: Final local search
    if time.time() < hard_end - 5.0 and best_n <= 320:
        hh, cc = local_block_search(best_h, best_n, hard_end - 3.0,
                                    seed=seed + 77777, step=0.01)
        if cc < best_c:
            best_h = hh
            best_c = cc

    # Final LP micro-step
    if time.time() < hard_end - 2.0 and best_n <= 256:
        hh, cc = lp_optimize(best_h, best_n, hard_end - 1.0, max_iter=30, seed=seed + 1)
        if cc < best_c:
            best_h = hh
            best_c = cc

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
