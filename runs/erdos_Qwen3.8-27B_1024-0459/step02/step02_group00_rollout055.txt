        h_interp = _proj(h_interp, n)
        c5_interp = _c5(h_interp)
        
        inits = [h_interp.copy()]
        inits.append(_proj(_thue_morse(n), n))
        inits.append(_proj(_golden_ratio(n), n))
        
        for _ in range(2):
            hb = np.zeros(n)
            perm = rng.permutation(n)
            hb[perm[:n // 2]] = 1.0
            inits.append(_proj(hb, n))
        
        # Perturbed parent
        hp = best_h + rng.randn(best_n) * 0.05
        inits.append(_proj(_interp(hp, best_n, n), n))
        
        for h0 in inits:
            if _time_left() < 15:
                break
            c5_0 = _c5(h0)
            for beta in [50, 200, 500, 1000]:
                if _time_left() < 10:
                    break
                h_new = _run_slsqp(h0, n, 'smo', beta, 0, 350)
                c5_new = _c5(h_new)
                if c5_new < c5_0:
                    h0 = h_new
                    c5_0 = c5_new
            for k in [2, 3]:
                if _time_left() < 8:
                    break
                h_new = _run_slsqp(h0, n, 'topk', k, 0.5, 250)
                c5_new = _c5(h_new)
                if c5_new < c5_0:
                    h0 = h_new
                    c5_0 = c5_new
            
            if c5_0 < best_c5:
                best_c5 = c5_0
                best_h = h0.copy()
                best_n = n
                print("  Improved n={}, C5={:.8f}".format(n, best_c5))
    
    # Phase 5: Binary swap local search on best
    if _time_left() > 25:
        n = best_n
        h_bin = np.zeros(n)
        h_sorted = np.sort(best_h)[::-1]
        threshold = h_sorted[n // 2]
        h_bin[best_h >= threshold] = 1.0
        # Ensure exactly n/2 ones
        if np.sum(h_bin) > n // 2:
            # Flip largest zeros to... no, flip smallest ones
            one_idx = np.where(h_bin == 1.0)[0]
            to_flip = one_idx[np.argsort(best_h[one_idx])][:int(np.sum(h_bin) - n // 2)]
            h_bin[to_flip] = 0.0
        elif np.sum(h_bin) < n // 2:
            zero_idx = np.where(h_bin == 0.0)[0]
            to_flip = zero_idx[np.argsort(-best_h[zero_idx])][:int(n // 2 - np.sum(h_bin))]
            h_bin[to_flip] = 1.0
        
        c5_bin = _c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        
        # Local search: try single swaps
        ones = list(np.where(h_bin == 1.0)[0])
        zeros = list(np.where(h_bin == 0.0)[0])
        improved_any = True
        while improved_any and _time_left() > 8:
            improved_any = False
            rng.shuffle(ones)
            rng.shuffle(zeros)
            for i in ones:
                if _time_left() < 4:
                    break
                for j in zeros:
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    c5_s = _c5(h_bin)
                    if c5_s < c5_bin - 1e-12:
                        c5_bin = c5_s
                        ones.remove(i)
                        ones.append(j)
                        zeros.remove(j)
                        ones.remove(j)
                        ones.append(i)
                        zeros.remove(i)
                        # Rebuild
                        ones = list(np.where(h_bin == 1.0)[0])
                        zeros = list(np.where(h_bin == 0.0)[0])
                        improved_any = True
                        if c5_bin < best_c5:
                            best_c5 = c5_bin
                            best_h = h_bin.copy()
                        break
                    else:
                        h_bin[i] = 1.0
                        h_bin[j] = 0.0
        
        # Refine: SLSQP from binary to allow fractional
        if _time_left() > 10:
            h_new = _run_slsqp(h_bin.copy(), n, 'smo', 200, 0, 300)
            c5_new = _c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
    
    # Phase 6: Final polish
    if _time_left() > 8:
        for beta in [1000, 3000, 5000]:
            if _time_left() < 4:
                break
            h_new = _run_slsqp(best_h, best_n, 'smo', beta, 0, 500)
            c5_new = _c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        # Top-k final
        if _time_left() > 6:
            h_new = _run_slsqp(best_h, best_n, 'topk', 3, 0.3, 400)
            c5_new = _c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    best_h = _proj(best_h, best_n)
    final_c5 = _c5(best_h)
    
    try:
        ec = evaluate_erdos_solution(best_h)
        if ec < final_c5:
            final_c5 = ec
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
