from scipy.optimize import linprog

def block_cd_lp(h, n, block_size=4, n_passes=100):
    """Block coordinate descent using LP subproblems."""
    h = h.copy()
    corr = get_corr(h)
    lags = np.arange(-(n-1), n)  # length 2n-1
    best_c5 = np.max(corr) * (2.0/n)
    best_h = h.copy()
    
    for p in range(n_passes):
        # Precompute coefficient vectors for each coordinate
        # coeff[j, m] = coefficient of h[j] in corr[m]
        # This is expensive to compute for all j, so let's compute on the fly
        
        improved = False
        for start in range(0, n, block_size):
            block = list(range(start, min(start + block_size, n)))
            B = len(block)
            if B == 0:
                break
            
            # Compute the coefficient for each coordinate in the block
            # coeff[j, m] for j in block, m in 0..2n-2
            A = np.zeros((B, 2*n-1))
            for b_idx, j in enumerate(block):
                lag = lags
                # h[j] as first factor
                idx1 = j + lag
                mask1 = (idx1 >= 0) & (idx1 < n)
                A[b_idx, mask1] += (1.0 - h[idx1[mask1]])
                # h[j] as second factor
                idx2 = j - lag
                mask2 = (idx2 >= 0) & (idx2 < n)
                A[b_idx, mask2] += (-h[idx2[mask2]])
            
            # Current corr = h[j]*A[j,m] + b[m] for each m
            # When we change h[j] to h[j] + delta_j for j in block:
            # new_corr[m] = corr[m] + sum_j delta_j * A[j,m]
            # where delta_j = new_h[j] - h[j]
            
            # LP: min t
            # s.t. corr[m] + sum_j A[j,m] * x[j] ≤ t  for all m (where x[j] is new h[j])
            #      0 ≤ x[j] ≤ 1 for all j in block
            # Variables: x[0..B-1], t (total B+1 variables)
            # Constraints: (2n-1) inequality constraints + bounds
            
            # c (objective): minimize t, so c = [0,0,...,0,1] (B+1 entries)
            c_obj = np.zeros(B + 1)
            c_obj[-1] = 1.0  # minimize t
            
            # Inequality constraints: A_ub @ vars ≤ b_ub
            # corr[m] + sum_j A[j,m]*x[j] - t ≤ 0 for all m
            # => sum_j A[j,m]*x[j] - t ≤ -corr[m]
            A_ub = np.zeros((2*n-1, B+1))
            A_ub[:, :B] = A.T  # coefficients of x[j]
            A_ub[:, B] = -1.0  # coefficient of t
            b_ub = -corr
            
            # Bounds: 0 ≤ x[j] ≤ 1, t ≤ large
            bounds = [(0, 1)] * B + [(0, None)]  # t ≥ 0
            
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            
            if res.status == 0:
                h[block] = res.x[:B]
                improved = True
        
        # Handle sum constraint
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0/n)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def cd_pass(h, n, corr, lags):
    """One pass of coordinate descent (Jacobi style - use same corr for all updates)."""
    h_new = h.copy()
    for j in range(n):
        # Compute coefficient of h[j] in each corr[m]
        k = lags  # array of length 2n-1, lags[m] = m - (n-1)
        
        # h[j] as first factor: contribution (1 - h[j+k[m]])
        idx1 = j + k
        mask1 = (idx1 >= 0) & (idx1 < n)
        coeff1 = np.zeros(2*n-1)
        idx1_valid = idx1[mask1]
        coeff1[mask1] = (1.0 - h[idx1_valid])
        
        # h[j] as second factor: contribution -h[j-k[m]]
        idx2 = j - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        idx2_valid = idx2[mask2]
        coeff1[mask2] += (-h[idx2_valid])
        
        # corr[m] = h[j]*coeff1[m] + b[m]
        b = corr - h[j] * coeff1
        
        # Binary search for optimal h[j]
        lo = 0.0
        hi = 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            vals = coeff1 * mid + b
            mx = np.max(vals)
            active = vals >= mx - 1e-12
            sg = np.mean(coeff1[active])
            if sg > 1e-12:
                hi = mid
            elif sg < -1e-12:
                lo = mid
            else:
                break
        h_new[j] = 0.5 * (lo + hi)
    return h_new
