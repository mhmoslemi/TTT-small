def compute_delta_c(h, p, q, n):
    """Compute change in correlation array when h[p]: 1→0 and h[q]: 0→1."""
    delta = np.zeros(2 * n - 1)
    # p: 1→0
    # Term 1: -(1-h[p-shift]) for valid indices
    # shift = k-(n-1), valid when 0 ≤ p-shift ≤ n-1
    # k ∈ [p, p+n-1], h index: p-k+n-1
    ks = max(0, p)
    ke = min(2*n-2, p+n-1)
    if ks <= ke:
        hidx = p - np.arange(ks, ke+1) + n - 1
        delta[ks:ke+1] -= (1.0 - h[hidx])
    # Term 2: +h[p+shift] for valid indices
    # valid when 0 ≤ p+shift ≤ n-1
    # k ∈ [n-1-p, 2n-2-p], h index: p+k-n+1
    ks = max(0, n-1-p)
    ke = min(2*n-2, 2*n-2-p)
    if ks <= ke:
        hidx = p + np.arange(ks, ke+1) - n + 1
        delta[ks:ke+1] += h[hidx]
    
    # q: 0→1
    # Term 3: +(1-h[q-shift]) for valid indices
    ks = max(0, q)
    ke = min(2*n-2, q+n-1)
    if ks <= ke:
        hidx = q - np.arange(ks, ke+1) + n - 1
        delta[ks:ke+1] += (1.0 - h[hidx])
    # Term 4: -h[q+shift] for valid indices
    ks = max(0, n-1-q)
    ke = min(2*n-2, 2*n-2-q)
    if ks <= ke:
        hidx = q + np.arange(ks, ke+1) - n + 1
        delta[ks:ke+1] -= h[hidx]
    
    return delta
