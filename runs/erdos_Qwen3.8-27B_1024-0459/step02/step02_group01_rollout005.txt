def remaining_until(t0, budget_s, deadline, reserve_s):
    now = time.time()
    end_total = t0 + float(budget_s) - float(reserve_s)
    if deadline is not None:
        end = min(end_total, float(deadline))
    else:
        end = end_total
    if end < now:
        return 0.0
    return end - now


def compute_c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size != n:
        if h.size == 0:
            h = np.full(n, 0.5, dtype=float)
        else:
            xn = (np.arange(n) + 0.5) * (2.0 / float(n))
            xo = (np.arange(h.size) + 0.5) * (2.0 / float(h.size))
            h = np.interp(xn, xo, h)

    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * float(n)

    for _ in range(50):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h = h + diff / float(n)
        h = np.clip(h, 0.0, 1.0)

    s = float(np.sum(h))
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0.0:
            order = np.argsort(1.0 - h)
            rem = diff
            for idx in order:
                if rem <= 0.0:
                    break
                idx = int(idx)
                room = 1.0 - h[idx]
                if room <= 0.0:
                    continue
                add = rem if rem < room else room
                h[idx] += add
                rem -= add
        else:
            order = np.argsort(h)
            rem = diff
            for idx in order:
                if rem >= 0.0:

                room = -h[idx]
                if room >= 0.0:
                    continue
                add = rem if rem > room else room

        if diff == 0.0:

def gradient_from_lags(h, idxs, weights):
    n = int(len(h))
    dx = 2.0 / float(n)
    grad = np.zeros(n, dtype=float)
    base = n - 1

    for k in range(len(idxs)):
        w = float(weights[k])
        if w == 0.0:
            continue
        idx = int(idxs[k])
        if idx < 0 or idx >= 2 * n - 1:
            continue
        o = idx - base

        if o >= 0:
            i1 = n - o
            if i1 > 0:
                grad[:i1] -= w * h[o:n]
                grad[o:n] += w * (1.0 - h[:i1])
        else:
            p = -o
            i1 = n - p
            if i1 > 0:
                grad[:i1] += w * (1.0 - h[p:n])
                grad[p:n] -= w * h[:i1]

    grad *= dx
    return grad

def _smooth_obj(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode="full")

    if _g_active is None:
        vals = c
    else:

    m = float(np.max(vals))
    z = _g_beta * (vals - m)

    s = float(e.sum())

    if s <= 1e-300:
        val = m
    else:

    return float(val) * (2.0 / float(n))


def _smooth_jac(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode="full")

    if _g_active is None:
        idxs = np.arange(len(c), dtype=np.int64)
        vals = c
    else:
        idxs = np.asarray(_g_active, dtype=np.int64)
        vals = c[idxs]

    if vals.size == 0:
        return np.zeros(n, dtype=float)

    m = float(np.max(vals))
    z = _g_beta * (vals - m)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())

    if s <= 1e-300:
        return np.zeros(n, dtype=float)

    w = e / s
    mask = w > 1e-14
    if not bool(np.all(mask)):
        idxs = idxs[mask]
        w = w[mask]
        sw = float(w.sum())
        if sw <= 0.0:
            return np.zeros(n, dtype=float)
        w = w / sw

    global _g_n, _g_beta, _g_active, _g_min_deadline

    _g_n = int(n)
    h = project_feasible(h0, _g_n)
    best = h.copy()
    best_c = compute_c5(h)

    bounds = [(0.0, 1.0)] * _g_n
    cons = [{"type": "eq", "fun": _eq_fun, "jac": _eq_jac}]

    stages = [
        (2, 30),
        (5, 50),
        (10, 60),
        (20, 80),
        (50, 90),
        (100, 100),
        (300, 120),
        (800, 130),
        (2000, 140),
        (5000, 180),
        (10000, 220),
    ]

    total_end = t0 + float(budget_s) - float(reserve_s)

    for beta, maxit in stages:
        if deadline is None:
            stop = total_end
        else:
            stop = min(float(deadline), total_end)

        if remaining_until(t0, budget_s, deadline, reserve_s) < 20.0:
            break

        _g_beta = float(beta)
        _g_active = None
        _g_min_deadline = stop

        try:
            res = minimize(
                _smooth_obj,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(maxit), "ftol": 1e-11, "iprint": 0},
                callback=_minimize_callback,
            )
            h2 = project_feasible(res.x, _g_n)
        except _OptTimeout:
            h2 = h
        except Exception:
            h2 = h

        c2 = compute_c5(h2)
        if np.isfinite(c2):
            if c2 < best_c - 1e-12:
                best = h2.copy()
                best_c = float(c2)
                h = h2
            elif c2 < best_c + 5e-4:
                h = h2
            else:
                h = best

    _g_active = None
    return best, best_c


def active_refine_one(h, n, t0, budget_s, deadline, reserve_s):
    global _g_n, _g_beta, _g_active, _g_min_deadline

    _g_n = int(n)
    best = project_feasible(h, _g_n)
    best_c = compute_c5(best)

    bounds = [(0.0, 1.0)] * _g_n
    cons = [{"type": "eq", "fun": _eq_fun, "jac": _eq_jac}]
    total_end = t0 + float(budget_s) - float(reserve_s)

    for _outer in range(5):
        if remaining_until(t0, budget_s, deadline, reserve_s) < 25.0:
            break

        c = np.correlate(best, 1.0 - best, mode="full")
        idx = np.argsort(c)[-10:]

        for beta, maxit in [(20, 70), (80, 90), (300, 110), (1000, 140)]:
            if remaining_until(t0, budget_s, deadline, reserve_s) < 20.0:
                break

            _g_beta = float(beta)
            _g_active = idx.astype(np.int64)

            stop = total_end
            if deadline is not None:
                stop = min(stop, float(deadline))
            _g_min_deadline = stop

            try:
                res = minimize(
                    _smooth_obj,
                    best,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=cons,
                    options={"maxiter": int(maxit), "ftol": 1e-11, "iprint": 0},
                    callback=_minimize_callback,
                )
                h2 = project_feasible(res.x, _g_n)
            except _OptTimeout:
                h2 = best
            except Exception:
                h2 = best

            c2 = compute_c5(h2)
            if np.isfinite(c2) and c2 < best_c - 1e-12:
                best = h2.copy()
                best_c = float(c2)
                c = np.correlate(best, 1.0 - best, mode="full")
                idx = np.argsort(c)[-10:]

    if remaining_until(t0, budget_s, deadline, reserve_s) > 30.0:
        _g_active = None
        _g_beta = 30000.0
        stop = total_end
        if deadline is not None:
            stop = min(stop, float(deadline))
        _g_min_deadline = stop

        try:
            res = minimize(
                _smooth_obj,
                best,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 200, "ftol": 1e-12, "iprint": 0},
                callback=_minimize_callback,
            )
            h2 = project_feasible(res.x, _g_n)
        except _OptTimeout:
            h2 = best
        except Exception:
            h2 = best

        c2 = compute_c5(h2)
        if np.isfinite(c2) and c2 < best_c - 1e-12:
            best = h2.copy()
            best_c = float(c2)

    _g_active = None
    return best, best_c


def subgradient_refine(h, n, t0, budget_s, deadline, reserve_s):
    best = project_feasible(h, n)
    best_c = compute_c5(best)
    cur_c = best_c
    base_step = 0.10
    no_decrease = 0

    while remaining_until(t0, budget_s, deadline, reserve_s) > 1.0:
        c = np.correlate(best, 1.0 - best, mode="full")
        top = np.argsort(c)[-4:]
        w = np.full(len(top), 1.0 / float(len(top)), dtype=float)
        grad = gradient_from_lags(best, top, w)

        if not np.any(grad):
            break

        improved = False
        for s in (base_step, 0.5 * base_step, 0.25 * base_step, 0.1 * base_step):
            cand = project_feasible(best - s * grad, n)
            cand_c = compute_c5(cand)
            if np.isfinite(cand_c) and cand_c <= cur_c + 1e-14:
                best = cand
                cur_c = float(cand_c)
                if cur_c < best_c:
                    best_c = cur_c
                improved = True
                break

        if improved:
            base_step = min(0.5, base_step * 1.08)
            no_decrease = 0
        else:
            base_step = max(0.005, base_step * 0.5)
            no_decrease += 1
            if no_decrease > 70:
                break

    return best, best_c


def max_count_correlate(x):
    if len(x) == 0:
        return 0
    if len(x) == 1:
        v = float(x[0])
        return int(round(v * (1.0 - v)))
    corr = np.correlate(x, 1 - x, mode="full")
    v = float(np.max(corr))
    if v < 0.0 and v > -1e-8:
        v = 0.0
    return int(round(v))


def binary_c5_from_count(m, n):
    return float(m) * (2.0 / float(n))


def round_to_binary(h, n, threshold=0.5):
    x = (h > threshold).astype(np.int8)
    want = n // 2
    cnt = int(np.sum(x))

    if cnt > want:
        ones = np.flatnonzero(x == 1)
        order = np.argsort(h[ones])
        x[ones[order][: cnt - want]] = 0
    elif cnt < want:
        zeros = np.flatnonzero(x == 0)
        order = np.argsort(-h[zeros])
        x[zeros[order][: want - cnt]] = 1

    return x


def balanced_random_binary(n, rng):
    x = np.zeros(n, dtype=np.int8)
    if n >= 2:
        idx = rng.choice(n, n // 2, replace=False)
        x[idx] = 1
    return x


def make_binary_starts(h, n, rng):
    if n % 2 != 0:
        return []

    starts = []
    for q in (0.35, 0.45, 0.55, 0.70):
        starts.append(round_to_binary(h, n, q))
    starts.append(balanced_random_binary(n, rng))
    return starts


def binary_best_swap_search(x0, n, t0, budget_s, deadline, reserve_s, rng, target_count=None, max_steps=800):
    x = np.array(x0, dtype=np.int8).copy()
    cur_m = max_count_correlate(x)
    best_x = x.copy()
    best_m = cur_m

    steps = 0
    perturbations = 0

    while remaining_until(t0, budget_s, deadline, reserve_s) > 2.0 and steps < max_steps:
        if target_count is not None and best_m <= target_count:
            break

        if cur_m > best_m + 8:
            x = best_x.copy()
            cur_m = best_m

        ones = np.flatnonzero(x == 1)
        zeros = np.flatnonzero(x == 0)
        if len(ones) == 0 or len(zeros) == 0:
            break

        found = False
        for ii in range(len(ones)):
            if ii % 256 == 0 and remaining_until(t0, budget_s, deadline, reserve_s) < 1.0:
                break
            i = int(ones[ii])
            for jj in range(len(zeros)):
                j = int(zeros[jj])
                if x[i] != 1 or x[j] != 0:
                    continue

                x[i] = 0
                x[j] = 1
                m = max_count_correlate(x)

                if m < cur_m:
                    cur_m = m
                    if m < best_m:
                        best_m = m
                        best_x = x.copy()
                    found = True
                    steps += 1
                    break

                x[i] = 1
                x[j] = 0

            if found:
                break

        if found:
            perturbations = 0
            continue

        if remaining_until(t0, budget_s, deadline, reserve_s) < 5.0:
            break

        changed = False
        for _ in range(30):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j and x[i] != x[j]:
                x[i], x[j] = x[j], x[i]
                changed = True
                break

        if not changed:
            break

        cur_m = max_count_correlate(x)
        if cur_m < best_m:
            best_m = cur_m
            best_x = x.copy()

        perturbations += 1
        steps += 1
        if perturbations > 14:
            break

    return best_x, best_m


def binary_anneal(x0, n, t0, budget_s, deadline, reserve_s, rng, target_count=None, temp0=2.0):
    x = np.array(x0, dtype=np.int8).copy()
    cur_m = max_count_correlate(x)
    best_x = x.copy()
    best_m = cur_m

    temp = float(temp0)
    steps = 0

    while remaining_until(t0, budget_s, deadline, reserve_s) > 0.5:
        if target_count is not None and best_m <= target_count:
            break

        found = False
        i = -1
        j = -1
        for _ in range(30):
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i != j and x[i] != x[j]:
                found = True
                break
        if not found:
            break

        old_i = int(x[i])
        old_j = int(x[j])
        x[i] = old_j
        x[j] = old_i
        new_m = max_count_correlate(x)
        steps += 1

        delta = new_m - cur_m
        accept = False
        if delta <= 0:
            accept = True
        elif temp > 1e-6:
            prob = float(np.exp(-float(delta) / temp))
            if prob > 0.0 and float(rng.rand()) < prob:
                accept = True

        if accept:
            cur_m = new_m
            if new_m < best_m:
                best_m = new_m
                best_x = x.copy()
        else:
            x[i] = old_i
            x[j] = old_j

        temp *= 0.9995
        if temp < 1e-4:
            temp = 0.0

    return best_x, best_m


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_beta, _g_active, _g_min_deadline

    t0 = time.time()
    budget_s = float(budget_s)
    reserve_s = 10.0
    budget_eff = max(30.0, budget_s - reserve_s)
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c = float("inf")
    best_n = 0

    try:
        ph = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if ph.size > 0:
            pn = int(ph.size)
            ph = project_feasible(ph, pn)
            best_h = ph
            best_n = pn
            best_c = compute_c5(ph)
    except Exception:
        best_h = None

    if best_h is None:
        best_n = 192
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = compute_c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec is not None and np.isfinite(ec):
            best_c = float(ec)
    except Exception:
        pass

    n_list = []
    if 64 <= best_n <= 288:
        n_list.append(best_n)
    for nn in (256, 224, 192, 160, 128, 96):
        if nn not in n_list and 64 <= nn <= 288:
            n_list.append(nn)
    if not n_list:
        n_list = [192]

    primary_n = n_list[0]
    primary_cont_end = t0 + 0.55 * budget_eff
    primary_polish_end = t0 + 0.72 * budget_eff
    primary_binary_end = t0 + 0.88 * budget_eff
    secondary_end = t0 + 0.97 * budget_eff
    total_end = t0 + budget_eff

    starts = []
    starts.append(prepare_start(best_h, best_n, primary_n, rng, 0.0))
    starts.append(prepare_start(best_h, best_n, primary_n, rng, 0.04))
    starts.append(prepare_start(best_h, best_n, primary_n, rng, 0.10))
    tt = np.arange(primary_n) / float(primary_n)
    hc = 0.5 + 0.35 * np.cos(2.0 * np.pi * tt + np.pi / 4.0)
    starts.append(project_feasible(hc, primary_n))

    first_end = t0 + (0.34 * budget_eff if len(starts) > 1 else primary_cont_end)
    if first_end <= time.time() + 10.0:
        first_end = primary_cont_end

    for si in range(min(4, len(starts))):
        if remaining_until(t0, budget_s, primary_cont_end, reserve_s) < 30.0:
            break
        if si == 0:
            d = min(first_end, primary_cont_end)
        else:
            d = primary_cont_end
        if remaining_until(t0, budget_s, d, reserve_s) < 15.0:
            continue
        hopt, cop = optimize_continuous_one(starts[si], primary_n, t0, budget_s, d, reserve_s)
        if np.isfinite(cop) and cop < best_c - 1e-12:
            best_h = hopt
            best_c = float(cop)
            best_n = len(hopt)

    polish_n = best_n
    if not (64 <= polish_n <= 288):
        polish_n = primary_n

    polish_now = time.time()
    polish_end = min(primary_polish_end, total_end)
    if polish_end > polish_now + 25.0:
        active_end = polish_now + 0.70 * (polish_end - polish_now)
        hopt, cop = active_refine_one(best_h, polish_n, t0, budget_s, active_end, reserve_s)
        if np.isfinite(cop) and cop < best_c - 1e-12:
            best_h = hopt
            best_c = float(cop)
            best_n = len(hopt)

        if remaining_until(t0, budget_s, polish_end, reserve_s) > 15.0:
            hopt, cop = subgradient_refine(best_h, best_n, t0, budget_s, polish_end, reserve_s)
            if np.isfinite(cop) and cop < best_c - 1e-12:
                best_h = hopt
                best_c = float(cop)
                best_n = len(hopt)

    bin_n = best_n
    if bin_n % 2 != 0:
        for nn in (primary_n, best_n, 192):
            if nn % 2 == 0 and 64 <= nn <= 288:
                bin_n = nn
                break

    if bin_n % 2 == 0 and remaining_until(t0, budget_s, primary_binary_end, reserve_s) > 20.0:
        bin_end = min(primary_binary_end, total_end)
        if bin_end > time.time() + 20.0:
            h_for_bin = prepare_start(best_h, best_n, bin_n, rng, 0.0)
            bstarts = make_binary_starts(h_for_bin, bin_n, rng)

            best_bin_x = None
            best_bin_m = 10 ** 9
            target_count = int(np.floor(best_c * bin_n / 2.0 - 1e-9))
            if target_count < 0:
                target_count = 0

            window = bin_end - time.time()
            if bstarts and window > 20.0:
                fr = [0.40, 0.25, 0.20, 0.15]
                for bi, bx in enumerate(bstarts):
                    if remaining_until(t0, budget_s, bin_end, reserve_s) < 15.0:
                        break
                    f = fr[bi] if bi < len(fr) else 0.10
                    sub_end = time.time() + f * window
                    if sub_end > bin_end:
                        sub_end = bin_end
                    if sub_end <= time.time() + 5.0:
                        sub_end = bin_end
                    rx, rm = binary_best_swap_search(
                        bx, bin_n, t0, budget_s, sub_end, reserve_s, rng, target_count
                    )
                    if rm < best_bin_m:
                        best_bin_m = rm
                        best_bin_x = rx
                    cbin = binary_c5_from_count(rm, bin_n)
                    if np.isfinite(cbin) and cbin < best_c - 1e-12:
                        best_h = rx.astype(float)
                        best_c = float(cbin)
                        best_n = bin_n
                        target_count = int(np.floor(best_c * bin_n / 2.0 - 1e-9))
                        if target_count < 0:
                            target_count = 0

            if remaining_until(t0, budget_s, bin_end, reserve_s) > 40.0 and best_bin_x is not None:
                rx, rm = binary_anneal(
                    best_bin_x, bin_n, t0, budget_s, bin_end, reserve_s, rng, target_count
                )
                cbin = binary_c5_from_count(rm, bin_n)
                if np.isfinite(cbin) and cbin < best_c - 1e-12:
                    best_h = rx.astype(float)
                    best_c = float(cbin)
                    best_n = bin_n

    if remaining_until(t0, budget_s, secondary_end, reserve_s) > 70.0:
        for nn in n_list:
            if remaining_until(t0, budget_s, secondary_end, reserve_s) < 60.0:
                break
            if nn == best_n and nn == primary_n:
                continue
            h0 = prepare_start(best_h, best_n, nn, rng, 0.0)
            now = time.time()
            rem = secondary_end - now
            sub_end = now + min(rem, max(60.0, 0.35 * rem))
            if sub_end > secondary_end:
                sub_end = secondary_end
            hopt, cop = optimize_continuous_one(h0, nn, t0, budget_s, sub_end, reserve_s)
            if np.isfinite(cop) and cop < best_c - 1e-12:
                best_h = hopt
                best_c = float(cop)
                best_n = len(hopt)

    if remaining_until(t0, budget_s, total_end, reserve_s) > 25.0:
        final_n = best_n
        if not (64 <= final_n <= 288):
            final_n = primary_n
        hopt, cop = active_refine_one(best_h, final_n, t0, budget_s, total_end, reserve_s)
        if np.isfinite(cop) and cop < best_c - 1e-12:
            best_h = hopt
            best_c = float(cop)
            best_n = len(hopt)

        if remaining_until(t0, budget_s, total_end, reserve_s) > 10.0:
            hopt, cop = subgradient_refine(best_h, best_n, t0, budget_s, total_end, reserve_s)
            if np.isfinite(cop) and cop < best_c - 1e-12:
                best_h = hopt
                best_c = float(cop)
                best_n = len(hopt)

    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec is not None and np.isfinite(ec):
            final_c5 = float(ec)
    except Exception:
        try:
            ec = evaluate_erdos_solution(best_h, best_n)
            if ec is not None and np.isfinite(ec):
                final_c5 = float(ec)
        except Exception:
            pass

    return (best_h, float(final_c5), int(best_n))
