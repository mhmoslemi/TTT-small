```python
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = cmax + (1.0 / beta) * np.log(sum_exp)
```

```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def softmax_obj_grad(x, beta, penalty=1e6):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    # Numerical stability
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum = np.log(sum_exp)
    
    # Add penalty for sum constraint
    s = np.sum(x)
    target = n / 2.0
    penalty_val = penalty * (s - target) ** 2
    obj = cmax + (1.0 / beta) * log_sum + penalty_val
    
    p = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # First term: (1-h[t-lag]) for t in [lag, n-1]
            grad[lag:n] += pk * (1.0 - x[:n-lag])
            # Second term: -h[t+lag] for t in [0, n-1-lag]
            grad[0:n-lag] -= pk * x[lag:n]
        else:
            # First term: (1-h[t-lag]) for t in [0, n-1+lag]
            grad[0:n+lag] += pk * (1.0 - x[-lag:])
            # Second term: -h[t+lag] for t in [-lag, n-1]
            grad[-lag:n] -= pk * x[:n+lag]
    
    # Add penalty gradient
    grad += 2.0 * penalty * (s - target)
    
    return obj, grad

def lbfgs_b_solve(h_init, n, betas, penalty=1e6, max_iter=3000):
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        def obj(x):
            return softmax_obj_grad(x, beta, penalty)[0]
        def grad(x):
            return softmax_obj_grad(x, beta, penalty)[1]
        try:
            res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
            h_new = make_feasible(res.x, n)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c5

def project_h(h, n):
    """Project h to satisfy constraints."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = n/2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            h[mask] = np.minimum(h[mask] + diff/np.sum(mask), 1.0)
        else:
            mask = h > 0.0
            h[mask] = np.maximum(h[mask] + diff/np.sum(mask), 0.0)
    return np.clip(h, 0.0, 1.0)

def interpolate(h_src, n_dst):
    if len(h_src) < 2:
        return np.full(n_dst, 0.5)
    x_old = np.linspace(0, 2, len(h_src), endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    h_new = np.interp(x_new, x_old, h_src)
    return make_feasible(h_new, n_dst)

def binary_sa(n, rng, n_steps, t_start, t_end):
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for s in range(n_steps):
        T = t_start * (t_end/t_start) ** (s / max(n_steps-1, 1))
        i, j = rng.randint(n), rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elif rng.random() > np.exp(-(c5-best_c5)/max(T, 1e-10)):
            h[i], h[j] = h[j], h[i]
    return best_h.astype(float), best_c5

def projected_grad_descent(h_init, n, beta, lr_init, n_iters, penalty=1e6):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        obj, grad = softmax_obj_grad(h, beta, penalty)
        lr = lr_init * (0.999 ** it)
        h_new = h - lr * grad
        h_new = project_h(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
    return best_h, best_c5
```
