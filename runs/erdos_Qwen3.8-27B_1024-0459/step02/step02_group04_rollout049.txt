    for i in range(n):
        x = 0.0
        d = 1.0
        ii = i
        while ii > 0:
            d /= 2.0
            x += (ii & 1) * d
            ii >>= 1
        positions.append((x, i))
    positions.sort()
    rng = np.random.RandomState(seed)
    target = n // 2
    if n % 2 == 0:
        for j in range(target):
            h[positions[j][1]] = 1.0
    else:
        for j in range(target):
            h[positions[j][1]] = 1.0
    return h


def random_smooth_init(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for freq in (1, 2, 3, 5, 7, 11):
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(freq * np.pi * x + phase)
    return project_feasible(h, n)


def periodic_init(n, period, ones_in_period, seed=0):
    n = int(n)
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < ones_in_period:
            h[i] = 1.0
    return project_feasible(h, n)


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, float(h_src[0]) if n_src == 1 else 0.5)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def overlap_at_lag(h, L):
    n = len(h)
    L = abs(int(L))
    if L >= n:
        return 0.0
    m = n - L
    return float(np.sum(h[:m]) - np.dot(h[:m], h[L:]))


def overlap_grad_at_lag(h, L):
    n = len(h)
    L = abs(int(L))
    g = np.zeros(n)
    if L == 0:
        return 1.0 - 2.0 * h
    m = n - L
    low = h[:m]
    high = h[L:]
    if L > 0:
        g[:m] += 1.0 - high
        g[L:] += -low
    else:
        g[L:] += 1.0 - low
        g[:m] += -high
    return g


def soft_max_obj(h, beta):
    n = len(h)
    dx = 2.0 / float(n)
    h = h.copy()
    vals = []
    vals.append(overlap_at_lag(h, 0))
    for L in range(1, n):
        vals.append(overlap_at_lag(h, L))
        vals.append(overlap_at_lag(h, -L))
    vals = np.array(vals) * dx
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -700, 0)
    logsum = m + np.log(np.sum(np.exp(shifted))) / beta
    return float(logsum)


def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / float(n)
    vals_raw = []
    lags = []
    vals_raw.append(overlap_at_lag(h, 0))
    lags.append(0)
    for L in range(1, n):
        vals_raw.append(overlap_at_lag(h, L))
        lags.append(L)
        vals_raw.append(overlap_at_lag(h, -L))
        lags.append(-L)
    vals = np.array(vals_raw) * dx
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -700, 0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    g = np.zeros(n)
    for i in range(len(vals)):
        if weights[i] > 1e-12:
            g += weights[i] * dx * overlap_grad_at_lag(h, lags[i])
    return g


def sum_eq_fun(x):
    return float(np.sum(x) - 0.5 * len(x))


def sum_eq_jac(x):
    return np.ones(len(x))


def slsqp_smooth(h0, n, beta, maxiter=60):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_eq_fun, "jac": sum_eq_jac}]
    try:
        res = minimize(
            soft_max_obj, h, args=(beta,),
            jac=lambda x: soft_max_grad(x, beta),
            method="SLSQP", bounds=bounds, constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h


def active_set_slsqp(h0, n, active_lags, maxiter=80):
    n = int(n)
    h = project_feasible(h0, n)
    c0 = compute_c5(h)
    m = len(active_lags)
    if m == 0:
        return h, c0

    def obj(x):
        return float(x[n])

    def obj_jac(x):
        j = np.zeros(n + 1)
        j[n] = 1.0
        return j

    def sum_c(x):
        return float(np.sum(x[:n]) - 0.5 * n)

    def sum_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    def ineq(x):
        out = np.empty(m)
        hh = x[:n]
        dx = 2.0 / float(n)
        for i, l in enumerate(active_lags):
            out[i] = float(x[n]) - overlap_at_lag(hh, int(l)) * dx
        return out

    def ineq_jac(x):
        hh = x[:n]
        dx = 2.0 / float(n)
        J = np.zeros((m, n + 1))
        for i, l in enumerate(active_lags):
            J[i, :n] = -dx * overlap_grad_at_lag(hh, int(l))
            J[i, n] = 1.0
        return J

    x0 = np.empty(n + 1)
    x0[:n] = h
    x0[n] = c0 + 1e-4

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [{"type": "eq", "fun": sum_c, "jac": sum_jac}]
    if m > 0:
        cons.append({"type": "ineq", "fun": ineq, "jac": ineq_jac})

    try:
        res = minimize(
            obj, x0, jac=obj_jac, method="SLSQP",
            bounds=bounds, constraints=cons,
            options={"maxiter": maxiter, "ftol": 1e-13, "disp": False}
        )
        if res.x is not None and res.x.size == n + 1 and np.all(np.isfinite(res.x)):
            h_out = project_feasible(res.x[:n], n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h, c0


def gradient_local_search(h0, n, time_limit, seed, max_iters=3000):
    rng = np.random.RandomState(seed)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h.copy()
    cur_c = c
    temp = 0.003
    min_temp = 1e-7
    dx = 2.0 / float(n)

    for it in range(max_iters):
        if time.time() > time_limit:
            break
        move_type = rng.randint(0, 5)
        h_try = cur_h.copy()

        if move_type == 0:
            # Gradient step on top active lags
            vals, lags = all_overlap_values(cur_h)
            top_k = min(5, len(vals))
            top_idx = np.argsort(-vals)[:top_k]
            g = np.zeros(n)
            for idx in top_idx:
                g += overlap_grad_at_lag(cur_h, int(lags[idx]))
            g /= top_k
            step = 0.15 / (n ** 0.5)
            h_try[:] = cur_h - step * g

        elif move_type == 1:
            # Block smooth perturbation
            L = int(rng.randint(3, min(15, n)))
            st = int(rng.randint(0, n - L + 1))
            idx = np.arange(st, st + L)
            delta = rng.randn(L) * 0.06
            delta -= delta.mean()
            h_try[idx] += delta

        elif move_type == 2:
            # Pair exchange
            i = int(rng.randint(0, n))
            j = int(rng.randint(0, n))
            if i != j:
                hi_ = cur_h[i]
                hj_ = cur_h[j]
                t_ = rng.uniform(0.0, 1.0)
                h_try[i] = (1 - t_) * hi_ + t_ * hj_
                h_try[j] = t_ * hi_ + (1 - t_) * hj_

        elif move_type == 3:
            # Sinusoidal perturbation
            freq = rng.uniform(0.5, 8.0)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            amp = rng.uniform(0.02, 0.08)
            x = np.arange(n) * 2.0 / n
            h_try[:] = cur_h + amp * np.sin(freq * np.pi * x + phase)

        else:
            # Single site nudge
            i = int(rng.randint(0, n))
            h_try[i] += rng.uniform(-0.3, 0.3)

        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try.copy()
            cur_c = c_try
            temp = min(0.015, temp * 1.1 + 1e-6)
        elif c_try < cur_c:
            cur_h = h_try.copy()
            cur_c = c_try
        else:
            d = c_try - cur_c
            if d < temp:
                prob = np.exp(-d / max(temp, 1e-10))
                if rng.rand() < prob:
                    cur_h = h_try.copy()
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)

    return best_h, float(best_c)


def choose_active_lags(vals, lags, max_count=20, margin=0.005):
    if vals.size == 0:
        return []
    top = float(np.max(vals))
    near = np.where(vals >= top - margin)[0]
    order = np.argsort(-vals)
    take = min(int(max_count), int(vals.size))
    idx = set(order[:take].tolist())
    idx.update(near.tolist())
    idx = sorted(idx, key=lambda i: -vals[i])
    cap = max(int(max_count) * 2, int(vals.size))
    if len(idx) > cap:
        idx = idx[:cap]
    return [int(lags[i]) for i in idx]


def multi_phase_optimize(h0, n, time_limit, seed):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    # Phase 1: Soft-max SLSQP with increasing beta
    if n <= 256:
        for beta in (8, 30, 100, 300):
            if time.time() > time_limit - 30:
                break
            rem = time_limit - time.time()
            if rem < 10:
                break
            h2 = slsqp_smooth(h, n, beta, maxiter=min(60, int(rem * 8)))
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
            h = h2 if c2 <= best_c + 1e-6 else best_h

    # Phase 2: Active set SLSQP
    if n <= 256 and time.time() < time_limit - 15:
        vals, lags = all_overlap_values(h)
        active = choose_active_lags(vals, lags, max_count=24, margin=0.004)
        if len(active) > 0:
            h2, c2 = active_set_slsqp(h, n, active, maxiter=70)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
            h = h2 if c2 <= best_c + 1e-8 else best_h
        # Second active set pass with updated lags
        if time.time() < time_limit - 10:
            vals, lags = all_overlap_values(h)
            active = choose_active_lags(vals, lags, max_count=16, margin=0.002)
            if len(active) > 0:
                h2, c2 = active_set_slsqp(h, n, active, maxiter=80)
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2

    # Phase 3: Local search
    if n <= 320 and time.time() < time_limit - 5:
        h2, c2 = gradient_local_search(best_h, n, time_limit - 1, seed, max_iters=5000)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def get_parent_values():
    try:
        vals = initial_h_values
        if vals is None:
            return None
        arr = np.asarray(vals, dtype=float).copy()
        if arr.size == 0:
            return None
        return arr
    except Exception:
        return None


def build_initializations(n, parent, best_h, best_n, rng, seed):
    inits = []
    # Parent interpolation
    if parent is not None:
        if parent.size == n:
            inits.append(parent.copy())
        else:
            inits.append(interpolate_to_n(parent, n))
    if best_h is not None and best_n != n:
        inits.append(interpolate_to_n(best_h, n))
    elif best_h is not None:
        inits.append(best_h.copy())

    # Thue-Morse
    inits.append(project_feasible(thue_morse_init(n, seed), n))
    # Van der Corput
    inits.append(project_feasible(vdc_init(n, seed + 1), n))
    # Random smooth
    inits.append(random_smooth_init(n, seed + 2))
    # Periodic 1100
    inits.append(project_feasible(periodic_init(n, 4, 2, seed), n))
    # Periodic 1010
    inits.append(project_feasible(periodic_init(n, 2, 1, seed + 3), n))

    # Structured: quarter blocks
    h = np.zeros(n)
    q = n // 4
    h[:q] = 1.0
    h[n // 2:n // 2 + q] = 1.0
    inits.append(project_feasible(h, n))

    return [project_feasible(ii, n) for ii in inits[:6]]


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        proj = project_feasible(parent, parent_n)
        best_h = proj.copy()
        best_c = compute_c5(proj)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Try multiple resolutions
    n_candidates = [best_n]
    # Add nearby and structured sizes
    for nn in (128, 160, 192, 200, 224, 256, 144, 176, 208, 240, 288, 320):
        if nn not in n_candidates and 64 <= nn <= 320:
            n_candidates.append(nn)
    n_candidates.sort(key=lambda x: abs(x - best_n))
    n_candidates = n_candidates[:8]

    allocated = budget_s - 60.0
    per_n = allocated / len(n_candidates)
    per_n = min(per_n, 180.0)
    per_n = max(per_n, 40.0)

    processed_ns = set()
    for nn in n_candidates:
        nn = int(nn)
        if time.time() > hard_end - 20:
            break
        if nn in processed_ns:
            continue
        processed_ns.add(nn)

        slot_end = min(hard_end - 10, time.time() + per_n)
        rem = slot_end - time.time()
        if rem < 15:
            continue

        inits = build_initializations(nn, parent, best_h, best_n, rng, seed + 100 * nn)
        n_inits = len(inits)
        per_init = max(12.0, rem / n_inits * 0.85)

        for j, h0 in enumerate(inits):
            if time.time() >= slot_end:
                break
            slot = min(per_init, slot_end - time.time() - 2.0)
            if slot < 8.0:
                continue
            hh, cc = multi_phase_optimize(
                h0, nn, time.time() + slot,
                seed + 1000 * nn + j
            )
            if cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = nn

    # Final polish on best
    if time.time() < hard_end - 15 and 2 <= best_n <= 320:
        # One more round of active set
        vals, lags = all_overlap_values(best_h)
        active = choose_active_lags(vals, lags, max_count=20, margin=0.003)
        if len(active) > 0 and time.time() < hard_end - 10:
            h2, c2 = active_set_slsqp(best_h, best_n, active, maxiter=100)
            if c2 < best_c:
                best_h = h2
                best_c = c2

        # Final local search
        if time.time() < hard_end - 5:
            h2, c2 = gradient_local_search(
                best_h, best_n, hard_end - 1, seed + 99991, max_iters=8000
            )
            if c2 < best_c:
                best_h = h2
                best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
