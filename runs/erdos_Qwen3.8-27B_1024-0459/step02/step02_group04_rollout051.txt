def _compute_top_lags(h, top_k):
    n = len(h)

def gaussian_smooth(h, sigma=1.5):
    n = h.size
    if n <= 1:
        return h.copy()
    length = min(7, n)
    if length % 2 == 0:
        length -= 1
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k ** 2) / (2.0 * sigma * sigma))
    w /= w.sum()
    return np.convolve(h, w, mode="same")


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    hard_end = t_start + max(60.0, budget_s - 30.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        best_h = project_feasible(parent, parent_n)
        best_c = compute_c5(best_h)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    resolutions = []
    base = best_n
    resolutions.append(base)
    for d in (32, 48, 64, 80):
        for nn in (base + d, base - d):
            if 80 <= nn <= 400 and nn not in resolutions:
                resolutions.append(nn)
    resolutions.sort()

    # Phase 1: Projected gradient at parent resolution
    if time.time() < hard_end - 200:
        t1_end = min(hard_end - 200, time.time() + 180)
        inits = [best_h.copy()]
        if parent is not None:
            inits.append(project_feasible(gaussian_smooth(best_h, 1.2), best_n))
            inits.append(project_feasible(gaussian_smooth(best_h, 2.5), best_n))
        inits.extend(structured_starts(best_n, rng, count=2))
        for h0 in inits:
            if time.time() > t1_end:
                break
            slot = t1_end - time.time()
            if slot < 10:
                break
            for beta in (30, 150):
                if time.time() > t1_end:
                    break
                lr = 3.0 / max(1.0, beta * 0.005)
                h2, c2 = projected_gradient_optimize(
                    h0, best_n, beta, maxiter=8000,
                    lr=lr, momentum=0.92,
                    time_limit=t1_end
                )
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()

    # Phase 2: Multi-resolution projected gradient
    for nn in resolutions:
        if time.time() > hard_end - 120:
            break
        if nn < 64 or nn > 380:
            continue
        slot = min(140.0, hard_end - 120 - time.time())
        if slot < 30:
            break
        h_interp = interpolate_to_n(best_h, nn)
        c_interp = compute_c5(h_interp)
        inits_r = [h_interp]
        inits_r.append(project_feasible(gaussian_smooth(h_interp, 1.5), nn))
        inits_r.append(project_feasible(gaussian_smooth(h_interp, 3.0), nn))
        st = structured_starts(nn, rng, count=2)
        inits_r.extend(st)
        for h0 in inits_r[:4]:
            if time.time() > hard_end - 120:
                break
            rem = hard_end - 120 - time.time()
            if rem < 15:
                break
            for beta in (50, 200):
                if time.time() > hard_end - 120:
                    break
                lr = 2.0 / max(1.0, beta * 0.003)
                h2, c2 = projected_gradient_optimize(
                    h0, nn, beta, maxiter=10000,
                    lr=lr, momentum=0.93,
                    time_limit=hard_end - 120
                )
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()
                    best_n = nn

    # Phase 3: SLSQP smooth max polish
    if time.time() < hard_end - 60 and 2 <= best_n <= 400:
        t3_end = min(hard_end - 60, time.time() + 120)
        h2, c2 = slsqp_polish(best_h, best_n, t3_end, maxiter=60)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 4: Active-set SLSQP
    if time.time() < hard_end - 50 and 2 <= best_n <= 400:
        t4_end = min(hard_end - 50, time.time() + 100)
        top_lags = _compute_top_lags(best_h, 50)
        h2, c2 = active_slsqp(best_h, top_lags, t4_end, maxiter=70)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 5: Another round of projected gradient with higher beta
    if time.time() < hard_end - 30 and 2 <= best_n <= 400:
        t5_end = min(hard_end - 30, time.time() + 60)
        for beta in (300, 600):
            if time.time() > t5_end:
                break
            lr = 1.0 / max(1.0, beta * 0.002)
            h2, c2 = projected_gradient_optimize(
                best_h, best_n, beta, maxiter=6000,
                lr=lr, momentum=0.95,
                time_limit=t5_end
            )
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 6: Active set again
    if time.time() < hard_end - 25 and 2 <= best_n <= 400:
        t6_end = min(hard_end - 25, time.time() + 50)
        top_lags = _compute_top_lags(best_h, 40)
        h2, c2 = active_slsqp(best_h, top_lags, t6_end, maxiter=50)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 7: Local perturbation
    if time.time() < hard_end - 10 and 2 <= best_n <= 400:
        t7_end = min(hard_end - 5, time.time() + 40)
        h2, c2 = local_perturb(best_h, best_n, t7_end, seed + 99991, step=0.02)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Final: quick SLSQP if time remains
    if time.time() < hard_end - 3 and 2 <= best_n <= 300:
        t_end = hard_end - 1.0
        top_lags = _compute_top_lags(best_h, 30)
        h2, c2 = active_slsqp(best_h, top_lags, t_end, maxiter=30)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
