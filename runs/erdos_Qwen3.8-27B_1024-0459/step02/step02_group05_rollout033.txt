import numpy as np
from scipy.optimize import minimize
import time

def compute_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, n, rng):
    """Round h to binary with exactly n//2 ones, preferring higher values"""
    h = np.array(h, dtype=float).copy()
    # Add tiny noise to break ties
    h_noisy = h + rng.rand(n) * 1e-10
    sorted_idx = np.argsort(h_noisy)[::-1]
    result = np.zeros(n)
    result[sorted_idx[:n // 2]] = 1.0
    return result

def steepest_descent_binary(h, n, max_steps=200, rng=None):
    """Try all 1-0 swaps, pick the best improving one per step"""
    h = h.copy()
    best_val = compute_max_corr(h)
    dx = 2.0 / n
    
    for step in range(max_steps):
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        best_new_val = best_val
        best_i, best_j = -1, -1
        
        for i in one_idx:
            for j in zero_idx:
                h[i], h[j] = 0.0, 1.0
                val = compute_max_corr(h)
                if val < best_new_val - 1e-12:
                    best_new_val = val
                    best_i, best_j = i, j
                h[i], h[j] = 1.0, 0.0
        
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            best_val = best_new_val
        else:
            break
    
    return h, best_val * dx

def first_improvement_ls(h, n, max_steps=20000, rng=None):
    """First-improvement local search with random 1-0 swaps"""
    h = h.copy()
    best_val = compute_max_corr(h)
    best_h = h.copy()
    dx = 2.0 / n
    
    for step in range(max_steps):
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        improved = False
        
        for _ in range(500):
            i = one_idx[rng.randint(len(one_idx))]
            j = zero_idx[rng.randint(len(zero_idx))]
            h[i], h[j] = 0.0, 1.0
            val = compute_max_corr(h)
            if val < best_val - 1e-12:
                best_val = val
                best_h = h.copy()
                improved = True
                break
            elif val < compute_max_corr(best_h):
                # Accept (it's improving over the local state even if not global best)
                improved = True
                break
            h[i], h[j] = 1.0, 0.0
        
        if not improved:
            break
    
    return best_h, best_val * dx

def ilc_search(h, n, budget_s, rng):
    """Iterated Local Search combining steepest descent, first improvement, and perturbation"""
    dx = 2.0 / n
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_val = compute_max_corr(h)
    
    consec_no_improve = 0
    perturb_size = 3
    
    while time.time() - start < budget_s:
        # Phase A: Steepest descent (limited steps)
        h, val = steepest_descent_binary(h, n, max_steps=30, rng=rng)
        if val < best_val * dx:
            best_val = val / dx
            best_h = h.copy()
            consec_no_improve = 0
        else:
            consec_no_improve += 1
        
        # Phase B: First improvement
        h, val = first_improvement_ls(h, n, max_steps=3000, rng=rng)
        if val < best_val * dx:
            best_val = val / dx
            best_h = h.copy()
            consec_no_improve = 0
        else:
            consec_no_improve += 1
        
        # Perturbation
        if consec_no_improve >= 2 or rng.random() < 0.1:
            h = best_h.copy()
            n_pert = rng.randint(2, 2 + perturb_size)
            for _ in range(n_pert):
                one_idx = np.where(h == 1)[0]
                zero_idx = np.where(h == 0)[0]
                i = one_idx[rng.randint(len(one_idx))]
                j = zero_idx[rng.randint(len(zero_idx))]
                h[i], h[j] = 0.0, 1.0
            perturb_size = min(8, perturb_size + 1)
            consec_no_improve = 0
        elif rng.random() < 0.3:
            perturb_size = max(2, perturb_size - 1)
    
    return best_h, best_val * dx

def two_for_two_ls(h, n, max_steps=5000, rng=None):
    """Local search using 2-for-2 swaps"""
    h = h.copy()
    best_val = compute_max_corr(h)
    best_h = h.copy()
    dx = 2.0 / n
    
    for step in range(max_steps):
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]

            i2 = one_idx[rng.randint(len(one_idx))]

                continue

                best_h = h.copy()

                break

            h[j1], h[j2] = 0.0, 0.0

    return best_h, best_val * dx

def simulated_annealing_binary(h, n, budget_s, rng, t_start=2.0, t_end=0.01):
    """Simulated annealing on binary solutions"""
    dx = 2.0 / n
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_val = compute_max_corr(h)
    current_val = best_val
    step = 0
    
    while time.time() - start < budget_s:
        T = t_start * (t_end / t_start) ** (step / max(1, int((budget_s) * 500)))
        step += 1
        
        one_idx = np.where(h == 1)[0]
        zero_idx = np.where(h == 0)[0]
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]
        if i == j:
            continue
        
        h[i], h[j] = 0.0, 1.0
        new_val = compute_max_corr(h)
        delta = new_val - current_val
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_val * dx

def slsqp_continuous(h_init, n, max_iter=3000):
    """SLSQP optimization on continuous relaxation"""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr)
    
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def softmax_slsqp(h_init, n, betas=(50, 200, 1000, 5000)):
    """Smooth max approximation with SLSQP"""
    current = h_init.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    for beta in betas:
        def obj_s(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))
        
        try:
            res = minimize(obj_s, current, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-14, 'iprint': 0})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def structured_init(n, rng):
    """Generate diverse initial binary solutions"""
    inits = []
    
    # Random
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    inits.append(h)
    
    # Arithmetic progression with golden ratio spacing
    h = np.zeros(n)
    phi = (1 + 5**0.5) / 2
    positions = [(int((i * phi) % n)) for i in range(n // 2)]
    positions = list(set(positions))
    if len(positions) < n // 2:
        remaining = [p for p in range(n) if p not in positions]
        positions.extend(rng.choice(remaining, n//2 - len(positions), replace=False).tolist())
    h[positions[:n//2]] = 1.0
    inits.append(h)
    
    # Quadratic residues
    h = np.zeros(n)
    positions = set()
    for i in range(n):
        positions.add((i * i) % n)
    positions = list(positions)
    if len(positions) > n // 2:
        positions = list(rng.choice(positions, n//2, replace=False))
    elif len(positions) < n // 2:
        remaining = [p for p in range(n) if p not in positions]
        positions.extend(rng.choice(remaining, n//2 - len(positions), replace=False).tolist())
    h[positions] = 1.0
    inits.append(h)
    
    # Low-discrepancy (van der Corput-like)
    h = np.zeros(n)
    positions = []
    for i in range(n):
        pos = 0
        denom = 1
        num = i
        while num > 0:
            pos += (num % 2) / (denom * 2)
            num //= 2
            denom *= 2
        positions.append(int(pos * n) % n)
    positions = list(set(positions))
    if len(positions) > n // 2:
        positions = list(rng.choice(positions, n//2, replace=False))
    elif len(positions) < n // 2:
        remaining = [p for p in range(n) if p not in positions]
        positions.extend(rng.choice(remaining, n//2 - len(positions), replace=False).tolist())
    h[positions[:n//2]] = 1.0
    inits.append(h)
    
    # Random with local structure
    h = np.zeros(n)
    block = n // 10
    for b in range(0, n, block):
        end = min(b + block, n)
        fill = int((end - b) / 2)
        onset = rng.choice(end - b - fill + 1) if end - b > fill else 0
        h[b+onset:b+onset+fill] = 1.0
    h = round_to_binary(h, n, rng)
    inits.append(h)
    
    return inits

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)
