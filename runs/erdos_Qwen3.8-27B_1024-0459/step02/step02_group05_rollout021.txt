For fixed j: A[j] = Σ_{lag=0}^{n-1-j} w[n-1+lag] * (1-h[j+lag])

For lag < 0 (m = n-1+lag, lag = -(n-1) to -1, let l = -lag = 1 to n-1):
Contribution to g[j]:
- (1-h[j-l]) for j ≥ l: (1-h[j-l]) * 1(j ≥ l)
- -h[j+l] for j ≤ n-1-l: -h[j+l] * 1(j ≤ n-1-l)

For fixed j: A_j2 = Σ_{l=1}^{j} w[n-1-l] * (1-h[j-l])

And B gets: -Σ_{l=1}^{n-1} w[n-1-l] * h[j+l] * 1(j ≤ n-1-l)
For fixed j: B_j2 = -Σ_{l=1}^{n-1-j} w[n-1-l] * h[j+l]

         + Σ_{l=1}^{j} w[n-1-l] * (1-h[j-l])
         - Σ_{l=1}^{n-1-j} w[n-1-l] * h[j+l]

This can be computed as:
- Term 1: convolution of w[n-1:n] (weights for non-negative lags) with (1-h), truncated
- Term 2: correlation of w[n-1:n] with h
- Term 3: correlation of w[0:n] (reversed) with (1-h)
- Term 4: convolution of w[0:n] (reversed) with h

def compute_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    weights = np.exp(shifted - np.max(shifted))  # for numerical stability
    weights /= weights.sum()
    
    value = cmax + np.log(np.exp(np.max(shifted)) * weights.sum()) / beta
    # Actually: smooth_max = cmax + np.log(sum(exp(beta*(corr-cmax))))/beta
    # = max(corr) + np.log(sum(exp(beta*(corr-max(corr))))) / beta
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        w = weights[m]
        if w < 1e-16:
            continue
        lag = m - n + 1
        if lag >= 0:
            grad[:n-lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
        else:
            l = -lag
            grad[l:] += w * (1.0 - h[:n-l])
            grad[:n-l] -= w * h[l:]
    
    return grad, value

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum
    for _ in range(50):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def projected_gradient_step(h, n, beta, lr, max_iter=10000):
    current = h.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()
    
    for it in range(max_iter):
        grad, val = compute_grad(current, beta)
        # Backtracking line search
        alpha = lr
        for _ in range(30):
            candidate = project(current - alpha * grad, n)
            c5_cand = compute_c5(candidate)
            if c5_cand < best_c5 or c5_cand < val:
                current = candidate
                break
            alpha *= 0.5
        else:
            break
        
        c5_now = compute_c5(current)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = current.copy()
    
    return best_h, best_c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def compute_grad_smooth_max(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    logsum = np.max(shifted) + np.log(exps.sum())
    value = cmax + logsum / beta
    grad = np.zeros(n)

def slsqp_optimize(n, h_init, max_iter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr)
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

    best_h = current.copy()

    lr_schedule = [0.08, 0.06, 0.04, 0.03, 0.02, 0.01, 0.005]

    # Phase 6: Targeted local search
    if remaining() > 30:
        h_new, c5_new = targeted_local(best_h, best_n, rng, n_iterations=3000,
                                       time_limit=min(20, remaining() - 5), start_time=start_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 7: Pair swap
    if remaining() > 15:
        n_rounds = min(30, int(remaining() / 1.0))
        best_h, best_c5 = pair_swap_local(best_h, best_n, n_rounds, rng,
                                          time_limit=min(10, remaining() - 3), start_time=start_time)

    # Phase 8: Random perturbation
    if remaining() > 10:
        n_trials = min(8000, int((remaining() - 2) / 0.008))
        best_h, best_c5 = random_perturb(best_h, best_n, n_trials, 0.02, rng,
                                         time_limit=min(8, remaining() - 2), start_time=start_time)

    # Phase 9: Try a different resolution if time allows
    if remaining() > 60:
        for n_alt in [200, 280, 360]:
            if remaining() < 40:
                break
            if n_alt == best_n:
                continue
            if n_alt % 2 != 0:
                n_alt += 1
            h_alt = make_feasible(interpolate_to_n(best_h, n_alt), n_alt)
            betas_alt = [10, 50, 200, 800]
            h_new, c5_new = smooth_anneal_pg(n_alt, h_alt, betas_alt, max_iter_per_beta=1500,
                                             time_limit=min(30, remaining() - 10), start_time=start_time)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_alt

    # Phase 10: Final polish
    if remaining() > 20:
        h_opt, c5_opt = slsqp_optimize(best_n, best_h, max_iter=3000, ftol=1e-16)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 11: Final projected gradient
    if remaining() > 10:
        h_new, c5_new = projected_gradient_opt(best_n, best_h, beta=5000, max_iter=2000,
                                               lr_init=0.003, time_limit=min(8, remaining() - 2),
                                               start_time=start_time)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8

    return (best_h, best_c5, best_n)
