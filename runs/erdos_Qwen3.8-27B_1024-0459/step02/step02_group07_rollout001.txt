import time
import math
import numpy as np
from scipy.optimize import minimize

OPT_N = 0
OPT_DX = 1.0
OPT_TOP = 20
OPT_TAU = 0.1
OPT_SMOOTH = True
OPT_NFFT = 0
OPT_VALID_IDX = np.array([], dtype=int)
OPT_LAG_VALS = np.array([], dtype=int)
OPT_BASIS = np.empty((0, 0), dtype=float)


def sanitize_array(x, limit=100.0):
    x = np.array(x, dtype=float)
    x = np.where(np.isfinite(x), x, 0.0)
    return np.clip(x, -limit, limit)


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)

    for _ in range(8):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            add = diff / cnt
            if add > 0.0:
                h[mask] = np.minimum(1.0, h[mask] + add)
        else:
            mask = h > 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            add = diff / cnt
            if add < 0.0:
                h[mask] = np.maximum(0.0, h[mask] + add)

    diff = target - float(np.sum(h))
    if abs(diff) > 1e-13:
        dist = np.minimum(h, 1.0 - h)
        order = np.argsort(-dist)
        for idx in order:
            if abs(diff) <= 1e-13:
                break
            if diff > 0.0:
                room = 1.0 - h[idx]
                step = min(diff, room)
            else:
                step = max(diff, -h[idx])
            if step == 0.0:
                continue
            h[idx] += step
            diff -= step

    if abs(diff) > 1e-13 and h.size > 0:
        idx = int(np.argmax(np.minimum(h, 1.0 - h)))
        if diff > 0.0:
            h[idx] = min(1.0, h[idx] + diff)
        else:
            h[idx] = max(0.0, h[idx] + diff)

    return h


def exact_c5(h):
    h = np.asarray(h, dtype=float)
    if h.size < 2 or not np.all(np.isfinite(h)):
        return np.inf
    dx = 2.0 / float(h.size)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def correlate_fft(h):
    N = OPT_NFFT
    b = 1.0 - h
    fh = np.fft.rfft(h, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fh * np.conj(fb), N)


def prepare_lag_data(n):
    target = 2 * n - 1
    N = 1
    while N < target:
        N <<= 1
    lags = np.arange(N, dtype=int)
    half = N // 2
    lags[half:] -= N
    valid = np.where((lags >= -(n - 1)) & (lags <= n - 1))[0]
    lag_vals = lags[valid].astype(int)
    return N, valid, lag_vals


def top_weights_function(f, top_k, tau):
    f = np.asarray(f, dtype=float)
    maxf = float(np.max(f))
    nf = f.size
    if top_k <= 0:
        return maxf, np.array([], dtype=int), np.array([], dtype=float)

    if tau <= 1e-12:
        tol = 1e-10 + 1e-12 * max(1.0, abs(maxf))
        active = np.where(f >= maxf - tol)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(f))])
        if active.size > top_k:
            fa = f[active]
            pos = np.argpartition(-fa, top_k - 1)[:top_k]
            active = active[pos]
        w = np.full(active.size, 1.0 / float(active.size), dtype=float)
        return maxf, active, w

    if top_k >= nf:
        idx = np.arange(nf)
    else:
        idx = np.argpartition(f, -top_k)[-top_k:]

    d = maxf - f[idx]
    d = np.maximum(d, 0.0)
    e = np.exp(-d / tau)
    s = float(np.sum(e))

    if not math.isfinite(s) or s <= 0.0:
        j = int(np.argmax(f[idx]))
        idx = idx[j:j + 1]
        e = np.array([1.0], dtype=float)
        s = 1.0
        val = maxf
    else:
        val = maxf + tau * math.log(s)

    w = e / s
    return val, idx, w


def weighted_grad_h(h, lags, weights, n):
    G = np.zeros(n, dtype=float)
    if lags.size == 0:
        return G
    y = 1.0 - h
    for ii in range(lags.size):
        w = float(weights[ii])
        if w <= 1e-16:
            continue
        lb = int(lags[ii])
        if lb >= 0:
            end = n - lb
            if end > 0:
                G[lb:lb + end] += w * y[:end]
                G[:end] -= w * h[lb:lb + end]
        else:
            d = -lb
            if d >= n:
                continue
            end = n - d
            G[:end] += w * y[d:d + end]
            G[d:n] -= w * h[:end]
    return G


def center_z(z):
    z = np.asarray(z, dtype=float)
    if z.size == 0:
        return z
    return z - float(np.mean(z))


def z_to_h(z):
    z = sanitize_array(z, 100.0)
    n = z.size
    if n == 0:
        return z.copy(), 0.0, z.copy()

    c = -float(np.mean(z))
    t = np.tanh(z + c)
    F = float(np.sum(t))

    for _ in range(12):
        Q = float(np.sum(1.0 - t * t))
        if Q < 1e-12:
            break
        step = F / Q
        if step > 5.0:
            step = 5.0
        elif step < -5.0:
            step = -5.0
        c -= step
        t = np.tanh(z + c)
        F = float(np.sum(t))
        if abs(F) < 1e-12:
            break

    if abs(F) > 1e-8:
        span = float(np.max(z) - np.min(z))
        if span < 10.0:
            span = 10.0
        lo = c - span - 10.0
        hi = c + span + 10.0

        for _ in range(5):
            Fl = float(np.sum(np.tanh(z + lo)))
            if Fl < 0.0:
                break
            lo -= 10.0

        for _ in range(5):
            Fh = float(np.sum(np.tanh(z + hi)))
            if Fh > 0.0:
                break
            hi += 10.0

        for _ in range(40):
            mid = 0.5 * (lo + hi)
            Fm = float(np.sum(np.tanh(z + mid)))
            if Fm > 0.0:
                hi = mid
            else:
                lo = mid
        c = 0.5 * (lo + hi)
        t = np.tanh(z + c)

    h = 0.5 + 0.5 * t
    q = 0.25 * (1.0 - t * t)
    return h, c, q


def h_from_z(z):
    z = sanitize_array(z, 100.0)
    h, _, _ = z_to_h(z)
    return h


def h_from_theta(theta, basis):
    theta = sanitize_array(theta, 100.0)
    z = basis @ theta
    return h_from_z(z)


def z_from_h(h, eps=1e-10):
    h = np.asarray(h, dtype=float)
    x = np.clip(h, eps, 1.0 - eps)
    z = np.arctanh(2.0 * x - 1.0)
    return center_z(z)


def make_trig_basis(n, m):
    if m <= 0 or n <= 0:
        return np.empty((n, 0), dtype=float)
    j = np.arange(n, dtype=float)
    norm = math.sqrt(0.5 * float(n))
    cols = []
    for r in range(1, m + 1):
        ang = 2.0 * math.pi * float(r) * j / float(n)
        c = np.cos(ang)
        s = np.sin(ang)
        cols.append(c / norm)
        cols.append(s / norm)
    return np.column_stack(cols)


def project_z_to_basis(z, basis):
    if basis.size == 0:
        return np.zeros(0, dtype=float)
    return basis.T @ z


def random_theta(basis, rng, amp=1.2):
    d = basis.shape[1]
    if d == 0:
        return np.zeros(0, dtype=float)
    th = amp * rng.randn(d) / math.sqrt(float(d))
    return sanitize_array(th, 50.0)


def interp_h(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    if h_src.size == n_dst:
        return project_feasible(h_src, n_dst)
    if h_src.size < 2:
        return project_feasible(np.full(n_dst, 0.5), n_dst)
    x_old = np.linspace(0.0, 2.0, h_src.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_new = np.interp(x_new, x_old, h_src)
    return project_feasible(h_new, n_dst)


def full_eval(z):
    z = sanitize_array(z, 100.0)
    if z.size != OPT_N:
        if z.size > OPT_N:
            z = z[:OPT_N]
        else:
            pad = OPT_N - z.size
            z = np.concatenate([z, np.zeros(pad, dtype=float)])

    h, _, q = z_to_h(z)
    r = correlate_fft(h)
    f = r[OPT_VALID_IDX]
    maxf = float(np.max(f))

    val_f, idx, w = top_weights_function(f, OPT_TOP, OPT_TAU)
    if not OPT_SMOOTH:
        val_f = maxf

    lags = OPT_LAG_VALS[idx]
    G = weighted_grad_h(h, lags, w, OPT_N)
    G *= OPT_DX

    Q = float(np.sum(q))
    if Q > 1e-15:
        wmean = float(np.dot(G, q) / Q)
        grad = q * (G - wmean)
    else:
        grad = np.zeros_like(z)

    grad = sanitize_array(grad, 1e6)
    return float(val_f * OPT_DX), grad


def low_eval(theta):
    theta = sanitize_array(theta, 100.0)
    z = OPT_BASIS @ theta
    h, _, q = z_to_h(z)

    r = correlate_fft(h)
    f = r[OPT_VALID_IDX]
    maxf = float(np.max(f))

    val_f, idx, w = top_weights_function(f, OPT_TOP, OPT_TAU)
    if not OPT_SMOOTH:
        val_f = maxf

    lags = OPT_LAG_VALS[idx]
    G = weighted_grad_h(h, lags, w, OPT_N)
    G *= OPT_DX

    Q = float(np.sum(q))
    if Q > 1e-15:
        wmean = float(np.dot(G, q) / Q)
        grad_z = q * (G - wmean)
    else:
        grad_z = np.zeros_like(z)

    grad = OPT_BASIS.T @ grad_z
    grad = sanitize_array(grad, 1e6)
    return float(val_f * OPT_DX), grad


def try_lbfgs_full(z0, top_k, tau, maxiter, smooth):
    global OPT_N, OPT_DX, OPT_TOP, OPT_TAU, OPT_SMOOTH, OPT_NFFT, OPT_VALID_IDX, OPT_LAG_VALS
    z0 = sanitize_array(z0, 100.0)
    n = z0.size
    if n < 2:
        return z0

    OPT_N = n
    OPT_DX = 2.0 / float(n)
    OPT_TOP = top_k
    OPT_TAU = tau
    OPT_SMOOTH = bool(smooth)
    N, valid, lag = prepare_lag_data(n)
    OPT_NFFT = N
    OPT_VALID_IDX = valid
    OPT_LAG_VALS = lag

    x0 = center_z(z0)
    options = {"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-8, "maxls": 20}
    try:
        res = minimize(full_eval, x0, method="L-BFGS-B", jac=True, options=options)
        if isinstance(res, dict):
            x1 = res.get("x")
        else:
            x1 = getattr(res, "x", None)
        if x1 is not None and np.all(np.isfinite(x1)):
            return sanitize_array(x1, 100.0)
    except Exception:
        pass
    return x0


def try_lbfgs_low(theta0, basis, top_k, tau, maxiter, smooth):
    global OPT_N, OPT_DX, OPT_TOP, OPT_TAU, OPT_SMOOTH, OPT_NFFT, OPT_VALID_IDX, OPT_LAG_VALS, OPT_BASIS
    theta0 = sanitize_array(theta0, 100.0)
    if basis.size == 0 or basis.shape[1] == 0:
        return theta0

    n = basis.shape[0]
    OPT_N = n
    OPT_DX = 2.0 / float(n)
    OPT_TOP = top_k
    OPT_TAU = tau
    OPT_SMOOTH = bool(smooth)
    N, valid, lag = prepare_lag_data(n)
    OPT_NFFT = N
    OPT_VALID_IDX = valid
    OPT_LAG_VALS = lag
    OPT_BASIS = basis

    x0 = theta0
    options = {"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-8, "maxls": 20}
    try:
        res = minimize(low_eval, x0, method="L-BFGS-B", jac=True, options=options)
        if isinstance(res, dict):
            x1 = res.get("x")
        else:
            x1 = getattr(res, "x", None)
        if x1 is not None and np.all(np.isfinite(x1)):
            return sanitize_array(x1, 100.0)
    except Exception:
        pass
    return x0


def fallback_initial(n):
    if n <= 0:
        n = 256
    x = np.arange(n, dtype=float)
    h = (
        0.5
        + 0.30 * np.sin(2.0 * math.pi * 3.0 * x / n)
        + 0.18 * np.sin(2.0 * math.pi * 5.0 * (x + 0.27 * n) / n)
        + 0.10 * np.cos(2.0 * math.pi * 7.0 * x / n)
    )
    return project_feasible(h, n)


def random_hill_z(z0, max_trials, time_limit, rng, amp=0.6):
    z0 = sanitize_array(z0, 100.0)
    n = z0.size
    if n < 2:
        h = project_feasible(np.full(n, 0.5), n)
        return h, exact_c5(h)

    best_z = center_z(z0)
    best_h = project_feasible(h_from_z(best_z), n)
    best_c = exact_c5(best_h)
    if not math.isfinite(best_c):
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = exact_c5(best_h)

    start = time.time()
    for _ in range(int(max_trials)):
        if time.time() - start > time_limit:
            break
        delta = amp * rng.randn(n)
        delta -= float(np.mean(delta))
        z_try = best_z + delta
        h_try = project_feasible(h_from_z(z_try), n)
        c_try = exact_c5(h_try)
        if math.isfinite(c_try) and c_try < best_c - 1e-14:
            best_h = h_try
            best_c = c_try
            best_z = center_z(z_from_h(best_h))

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    global OPT_N, OPT_DX, OPT_TOP, OPT_TAU, OPT_SMOOTH, OPT_NFFT, OPT_VALID_IDX, OPT_LAG_VALS, OPT_BASIS

    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_parent = None
    try:
        h_parent = np.asarray(initial_h_values, dtype=float).ravel()
        if h_parent.size == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is None:
        h_parent = fallback_initial(256)

    h_best = project_feasible(h_parent, h_parent.size)
    c_best = exact_c5(h_best)
    if not math.isfinite(c_best):
        h_best = fallback_initial(max(160, h_parent.size))
        c_best = exact_c5(h_best)

    z_cur = center_z(z_from_h(h_best))

    # 1. Polish the parent resolution directly.
    if h_best.size >= 2 and (budget_s - (time.time() - start_time)) > 260:
        for tau in (0.005, 0.02, 0.06, 0.15):
            if time.time() - start_time > budget_s - 220:
                break
            z_try = try_lbfgs_full(z_cur, 50, tau, 60, True)
            h_try = project_feasible(h_from_z(z_try), h_best.size)
            c_try = exact_c5(h_try)
            if math.isfinite(c_try):
                if c_try < c_best - 1e-14:
                    c_best = c_try
                    h_best = h_try
                    z_cur = center_z(z_try)
                elif c_try < c_best + 1e-6:
                    z_cur = center_z(z_try)

    # 2. Low-frequency polish at the current/fine resolution.
    if h_best.size <= 512 and (budget_s - (time.time() - start_time)) > 200:
        mp = max(4, min(12, h_best.size // 4))
        basis_p = make_trig_basis(h_best.size, mp)
        theta_cur = project_z_to_basis(z_cur, basis_p)
        for tau in (0.02, 0.08, 0.20):
            if time.time() - start_time > budget_s - 160:
                break
            theta_try = try_lbfgs_low(theta_cur, basis_p, 50, tau, 50, True)
            h_try = project_feasible(h_from_theta(theta_try, basis_p), h_best.size)
            c_try = exact_c5(h_try)
            if math.isfinite(c_try):
                if c_try < c_best - 1e-14:
                    c_best = c_try
                    h_best = h_try
                    theta_cur = theta_try
                    z_cur = center_z(basis_p @ theta_try)
                elif c_try < c_best + 1e-6:
                    theta_cur = theta_try
                    z_cur = center_z(basis_p @ theta_try)

    # 3. Move to a finer resolution (512) and optimize both low-dimensional and full z.
    target_n = 512
    if h_best.size > target_n:
        target_n = h_best.size

    if target_n >= 320 and (budget_s - (time.time() - start_time)) > 620:
        h_start = interp_h(h_best, target_n) if h_best.size != target_n else h_best.copy()
        z_start = center_z(z_from_h(h_start))

        m_low = 10 if target_n <= 384 else (14 if target_n <= 640 else 10)
        basis_t = make_trig_basis(target_n, m_low)
        theta_cur = project_z_to_basis(z_start, basis_t)

        if time.time() - start_time < budget_s - 420:
            for tau in (0.02, 0.08):
                if time.time() - start_time > budget_s - 320:
                    break
                theta_try = try_lbfgs_low(theta_cur, basis_t, 60, tau, 70, True)
                h_try = project_feasible(h_from_theta(theta_try, basis_t), target_n)
                c_try = exact_c5(h_try)
                if math.isfinite(c_try):
                    if c_try < c_best - 1e-14:
                        c_best = c_try
                        h_best = h_try
                        theta_cur = theta_try
                        z_start = center_z(basis_t @ theta_try)
                    elif c_try < c_best + 1e-6:
                        theta_cur = theta_try
                        z_start = center_z(basis_t @ theta_try)

            for rr in range(2):
                if time.time() - start_time > budget_s - 260:
                    break
                theta_r = random_theta(basis_t, rng, amp=1.15)
                theta_try = try_lbfgs_low(theta_r, basis_t, 60, 0.08, 40, True)
                h_try = project_feasible(h_from_theta(theta_try, basis_t), target_n)
                c_try = exact_c5(h_try)
                if math.isfinite(c_try) and c_try < c_best - 1e-14:
                    c_best = c_try
                    h_best = h_try
                    theta_cur = theta_try
                    z_start = center_z(basis_t @ theta_try)

        if (budget_s - (time.time() - start_time)) > 320:
            maxit_full = 70 if target_n <= 512 else 50
            for tau in (0.005, 0.02, 0.08, 0.20):
                if time.time() - start_time > budget_s - 220:
                    break
                z_try = try_lbfgs_full(z_start, 80, tau, maxit_full, True)
                h_try = project_feasible(h_from_z(z_try), target_n)
                c_try = exact_c5(h_try)
                if math.isfinite(c_try):
                    if c_try < c_best - 1e-14:
                        c_best = c_try
                        h_best = h_try
                        z_start = center_z(z_try)
                    elif c_try < c_best + 1e-6:
                        z_start = center_z(z_try)

    # 4. Higher-resolution finalization at 1024 if time permits.
    remaining = budget_s - (time.time() - start_time)
    if remaining > 360:
        high_n = 1024 if h_best.size <= 1024 else h_best.size
        h_start = interp_h(h_best, high_n)
        z_start = center_z(z_from_h(h_start))

        for tau in (0.02, 0.08):
            if time.time() - start_time > budget_s - 160:
                break
            z_try = try_lbfgs_full(z_start, 80, tau, 35, True)
            h_try = project_feasible(h_from_z(z_try), high_n)
            c_try = exact_c5(h_try)
            if math.isfinite(c_try):
                if c_try < c_best - 1e-14:
                    c_best = c_try
                    h_best = h_try
                    z_start = center_z(z_try)
                elif c_try < c_best + 2e-6:
                    z_start = center_z(z_try)

    # 5. Very small active-set polish at the best final resolution.
    remaining = budget_s - (time.time() - start_time)
    if remaining > 90 and h_best.size >= 2:
        z_fin = center_z(z_from_h(h_best))
        z_try = try_lbfgs_full(z_fin, 30, 1e-12, 40, False)
        h_try = project_feasible(h_from_z(z_try), h_best.size)
        c_try = exact_c5(h_try)
        if math.isfinite(c_try) and c_try < c_best - 1e-14:
            c_best = c_try
            h_best = h_try

    # 6. Short randomized local search in the parameter space.
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80 and h_best.size >= 2 and h_best.size <= 1024:
        z_start = center_z(z_from_h(h_best))
        if h_best.size <= 256:
            max_tries = 1500
            amp = 0.7
        elif h_best.size <= 512:
            max_tries = 800
            amp = 0.6
        else:
            max_tries = 350
            amp = 0.45
        h_try, c_try = random_hill_z(z_start, max_tries, max(5.0, remaining - 35.0), rng, amp=amp)
        if math.isfinite(c_try) and c_try < c_best - 1e-14:
            c_best = c_try
            h_best = h_try

    # Final strict feasibility and exact evaluation.
    h_best = project_feasible(h_best, h_best.size)
    c_final = exact_c5(h_best)
    if not math.isfinite(c_final):
        h_best = fallback_initial(h_best.size)
        c_final = exact_c5(h_best)

    return (h_best, float(c_final), int(h_best.size))
