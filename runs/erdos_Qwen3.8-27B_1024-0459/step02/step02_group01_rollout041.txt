                    h_work[a] = 1.0
                    h_work[b] = 0.0
                    
                    if np_max < best_delta if best_pair is not None else True:
                        pass  # placeholder
                    
                    delta = np_max - cur_max
                    if delta < (best_new_max - cur_max) if best_pair is not None else (0.0 if best_new_max >= cur_max else 0.0):
                        pass
                    # Simplified: just track the best
                    if best_pair is None:
                        if np_max < cur_max:
                            best_new_max = np_max
                            best_pair = (a, b)
                            best_new_cpos = np_new_cpos.copy()
                            best_new_cneg = np_new_cneg.copy()
                    elif np_max < best_new_max:
                        best_new_max = np_max
                        best_pair = (a, b)
                        best_new_cpos = np_new_cpos.copy()
                        best_new_cneg = np_new_cneg.copy()
                    
                    if time.time() > t_limit:
                        break
                if time.time() > t_limit:
                    break
            
            total_scans += 1
            
            if best_pair is not None:
                a, b = best_pair
                h[a] = 0.0
                h[b] = 1.0
                cpos, cneg = best_new_cpos, best_new_cneg
                cur_max = best_new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                    cooldown = 0
                pos1 = np.where(h == 1)[0]
                pos0 = np.where(h == 0)[0]
                T = max(0.01, T * 0.5)
            else:
                cooldown += 1
                if cooldown > 3:
                    # Perturb to escape
                    n_perturb = rng.randint(3, 8)
                    for _ in range(n_perturb):
                        a = pos1[rng.randint(len(pos1))]
                        b = pos0[rng.randint(len(pos0))]
                        h[a], h[b] = h[b], h[a]
                    cpos, cneg = get_corrs(h, n)
                    cur_max = max_corr_val(cpos, cneg)
                    pos1 = np.where(h == 1)[0]
                    pos0 = np.where(h == 0)[0]
                    cooldown = 0
                    T = 0.1
                else:
                    T = max(0.005, T * 0.5)
        else:
            # Random SA move
            a = pos1[rng.randint(len(pos1))]
            b = pos0[rng.randint(len(pos0))]
            h[a], h[b] = h[b], h[a]
            ncpos, ncneg = get_corrs(h, n)
            new_max = max_corr_val(ncpos, ncneg)
            
            if new_max <= cur_max:
                cur_max = new_max
                cpos, cneg = ncpos, ncneg
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                    cooldown = 0
                pos1 = np.where(h == 1)[0]

            else:

                if T > 0.001 and rng.random() < np.exp(-delta / T):
                    cur_max = new_max
                    cpos, cneg = ncpos, ncneg
                    pos1 = np.where(h == 1)[0]
                    pos0 = np.where(h == 0)[0]
                else:
                    h[a], h[b] = h[b], h[a]
            T = max(0.001, T * 0.999)
    
    return best_h, best_max * (2.0 / n)

def continuous_refine(h, n, t_limit):
    """SLSQP refinement on continuous relaxation."""
    h = h.astype(np.float64).copy()
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    h += (target_sum - h.sum()) / n
    h = np.clip(h, 0.0, 1.0)
    
    bounds = [(0.0, 1.0)] * n
    
    def obj_smooth(x, beta):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = beta * (c - m)
        sh = np.clip(sh, -500.0, 0.0)
        return (m + np.log(np.sum(np.exp(sh))) / beta) * (2.0 / n)
    
    def obj_raw(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return c.max() * (2.0 / n)
    
    def obj_topk(x, k=3):
        c = np.correlate(x, 1.0 - x, mode='full')
        top = np.sort(c)[-k:]
        return top.mean() * (2.0 / n)
    
    def eq_con(x):
        return x.sum() - target_sum
    
    cons = [{'type': 'eq', 'fun': eq_con}]
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta in [10.0, 50.0, 200.0, 1000.0, 5000.0]:
        if time.time() > t_limit:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(beta,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
            rh = np.clip(res.x, 0.0, 1.0)
            rh += (target_sum - rh.sum()) / n
            rh = np.clip(rh, 0.0, 1.0)
            c5_val = compute_c5(rh)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = rh
        except Exception:
            pass
    
    # Raw max refinement
    if time.time() < t_limit:
        try:
            res = minimize(obj_raw, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
            rh = np.clip(res.x, 0.0, 1.0)
            rh += (target_sum - rh.sum()) / n
            rh = np.clip(rh, 0.0, 1.0)
            c5_val = compute_c5(rh)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = rh
        except Exception:
            pass
    
    # Top-k refinement
    if time.time() < t_limit:
        try:
            res = minimize(obj_topk, best_h, args=(3,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
            rh = np.clip(res.x, 0.0, 1.0)
            rh += (target_sum - rh.sum()) / n
            rh = np.clip(rh, 0.0, 1.0)
            c5_val = compute_c5(rh)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = rh
        except Exception:
            pass
    
    return best_h, best_c5

def make_initial_binary(n, rng, pattern='random'):
    h = np.zeros(n)
    if pattern == 'alternating':
        h[::2] = 1.0
    elif pattern == 'block':
        h[:n//2] = 1.0
    elif pattern == 'quadratic':
        # Use quadratic residues mod p (for p = n if prime)
        for i in range(n):
            if (i * i) % (n + 1) < (n + 1) // 2:
                h[i] = 1.0
        h = fix_binary_sum(h, n)
    elif pattern == 'golden':
        # Golden ratio based
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for i in range(n):
            if (i * phi) % 1.0 < 0.5:
                h[i] = 1.0
        h = fix_binary_sum(h, n)
    else:
        # Random
        idx = rng.choice(n, n // 2, replace=False)
        h[idx] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Try to use parent
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 0 and pn % 2 == 0:
            # Use as continuous starting point
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 192
        best_h = make_initial_binary(best_n, rng, 'alternating')
        best_c5 = compute_c5(best_h)
    
    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)
    
    # Phase 1: Binary optimization at best_n
    if time_left() > 30:
        n1 = best_n
        h_bin = (best_h > 0.5).astype(float)
        h_bin = fix_binary_sum(h_bin, n1)
        
        # Try multiple strategies
        time_for_binary = min(time_left() * 0.5, 600.0)
        t_binary_start = time.time()
        
        # Strategy 1: Local search from rounded parent
        h_opt, c5_opt = binary_local_search(h_bin, n1, t_binary_start, 
                                            t_binary_start + time_for_binary * 0.5, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n1
        
        # Strategy 2: Local search from different init
        if time_left() > 60:
            h_init2 = make_initial_binary(n1, rng, 'random')
            t2_start = time.time()
            t2_limit = min(t2_start + time_left() * 0.2, t0 + budget_s - 60)
            h_opt2, c5_opt2 = binary_local_search(h_init2, n1, t2_start, t2_limit, rng)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n1
        
        # Strategy 3: Try other n values
        for n_try in [128, 160, 224, 256]:
            if time_left() < 60:
                break
            if n_try % 2 != 0:
                continue
            # Interpolate
            xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
            xn = (np.arange(n_try) + 0.5) * 2.0 / n_try
            h_interp = np.interp(xn, xp, best_h)
            h_bin_try = (h_interp > 0.5).astype(float)
            h_bin_try = fix_binary_sum(h_bin_try, n_try)
            
            t3_start = time.time()
            t3_limit = min(t3_start + min(120.0, time_left() * 0.15), t0 + budget_s - 60)
            h_opt3, c5_opt3 = binary_local_search(h_bin_try, n_try, t3_start, t3_limit, rng)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()
                best_n = n_try
    
    # Phase 2: Continuous refinement (SLSQP)
    if time_left() > 15:
        n2 = best_n
        h_start = best_h.copy() if len(best_h) == n2 else np.interp(
            (np.arange(n2) + 0.5) * 2.0 / n2,
            (np.arange(len(best_h)) + 0.5) * 2.0 / len(best_h),
            best_h
        ).astype(float)
        
        t_cont_start = time.time()
        t_cont_limit = min(t_cont_start + time_left() * 0.6, t0 + budget_s - 10)
        h_ref, c5_ref = continuous_refine(h_start, n2, t_cont_limit)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            best_n = n2
    
    # Phase 3: Binary round + re-refine
    if time_left() > 10:
        h_bin_final = (best_h > 0.5).astype(float)
        h_bin_final = fix_binary_sum(h_bin_final, best_n)
        c5_bin = compute_c5(h_bin_final)
        
        # Quick local search from this binary
        if c5_bin < best_c5 + 0.01:
            t4_start = time.time()
            t4_limit = min(t4_start + time_left() * 0.3, t0 + budget_s - 8)
            h_refined, c5_refined = binary_local_search(h_bin_final, best_n, t4_start, t4_limit, rng)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
        
        # Final SLSQP from best binary
        if time_left() > 5:
            h_final_start = best_h.astype(float)
            if (h_final_start == 0).sum() > 0 and (h_final_start == 1).sum() > 0:
                t5_start = time.time()
                t5_limit = min(t5_start + time_left() - 2, t0 + budget_s - 3)
                h_final, c5_final = continuous_refine(h_final_start, best_n, t5_limit)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
    
    # Final feasibility check
    best_h = np.clip(best_h, 0.0, 1.0)
    target = best_n / 2.0
    if abs(best_h.sum() - target) > 1e-10:
        best_h += (target - best_h.sum()) / best_n
        best_h = np.clip(best_h, 0.0, 1.0)
    
    final_c5 = compute_c5(best_h)
    
    # Try evaluate_erdos_solution
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
