def active_subgradient_polish(h, deadline):
    n=len(h); target=n/2.0
    h=project_box_sum(h,target); best=compute_c5(h); cur=best; best_h=h.copy(); alpha=0.02; iterations=0
    while time.time()<deadline and alpha>1e-6:
        c=np.correlate(h,1.0-h,mode='full')
        m=c.max()
        idx=np.argsort(c)[-8:]
        vals=c[idx]
        # weights by (vals - vals.min())? If all top at m, uniform. Use softmax beta 10 on vals.
        ex=np.exp(10*(vals-vals.max())); w=ex/ex.sum()
        g=np.zeros(n)
        for ii,ww in zip(idx,w):
           off=ii-(n-1)
           if off>=0:
              o=off
              g[o:] += ww*(1.0 - h[:n-o])
              g[:n-o] -= ww*h[o:]
           else:
              o=-off
              g[:n-o] += ww*(1.0 - h[o:])
              g[o:] -= ww*h[:n-o]
        gn=np.linalg.norm(g)
        if gn<1e-12: break
        g/=gn
        improved=False
        for f in (1.0,0.5,0.25,0.125,2.0):
            cand=project_box_sum(h - f*alpha*g, target)
            c2=np.correlate(cand,1.0-cand,mode='full')
            val=float(np.max(c2))* (2.0/n)
            if val < cur - 1e-14:
                h=cand; cur=val; improved=True
                break
        if improved:
            if cur<best:
                best=cur; best_h=h.copy()
            alpha=min(0.1, alpha*1.05)
        else:
            alpha*=0.7
        iterations +=1
        if iterations>2000: break
    return best_h,best

    n = h.size

import numpy as np
from scipy.optimize import minimize
import time
import math

TARGET_C = 0.38085
TARGET_NUM = 7617
TARGET_DEN = 20000

_G_N = 1
_G_BETA = 10.0

_MASK_CACHE = {}
_PAIR_CACHE = {}


def compute_c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / n)


def project_box_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = (1.0 - h) > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                h.fill(1.0)
                break
            add = diff / float(cnt)
            h[mask] = np.clip(h[mask] + add, 0.0, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                h.fill(0.0)
                break
            sub = -diff / float(cnt)
            h[mask] = np.clip(h[mask] - sub, 0.0, 1.0)

    s = float(np.sum(h))
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h


def target_M_for_n(n):
    return (TARGET_NUM * n) // (2 * TARGET_DEN)


def get_bit_masks(n):
    v = _MASK_CACHE.get(n)
    if v is None:
        lst = []
        for t in range(1, n):
            lst.append((1 << (n - t)) - 1)
        _MASK_CACHE[n] = lst
        v = lst
    return v


def get_pair_masks(n):
    v = _PAIR_CACHE.get(n)
    if v is None:
        m = n // 2
        lst = []
        for i in range(m):
            lst.append((1 << i) | (1 << (n - 1 - i)))
        _PAIR_CACHE[n] = lst
        v = lst
    return v


def eval_sym_limit(B, n, masks, limit):
    best = 0
    plat = 0
    for t, msk in enumerate(masks, 1):
        val = (B & msk & ~(B >> t)).bit_count()
        if val > best:
            best = val
            plat = 1
            if limit is not None and best > limit:
                return best, 0
        elif val == best:
            plat += 1
    return best, plat


def eval_asym_limit(B, n, masks, limit):
    best = 0
    plat = 0
    for t, msk in enumerate(masks, 1):
        shifted = B >> t
        pos = (B & msk & ~shifted).bit_count()
        neg = (shifted & msk & ~B).bit_count()
        val = pos if pos >= neg else neg
        if val > best:
            best = val
            plat = 1
            if limit is not None and best > limit:
                return best, 0
        elif val == best:
            plat += 1
    return best, plat


def build_B_sym(half, pair_masks):
    B = 0
    for i, val in enumerate(half):
        if val:
            B |= pair_masks[i]
    return B


def build_B_asym(bits):
    B = 0
    for i, val in enumerate(bits):
        if val:
            B |= (1 << i)
    return B


def repair_weight(bits, want, rng):
    bits = np.array(bits, dtype=int).copy()
    bits = (bits != 0).astype(int)
    cur = int(np.sum(bits))
    if cur > want:
        idx = np.where(bits == 1)[0]
        if idx.size >= cur - want:
            chosen = idx[rng.permutation(idx.size)[:cur - want]]
            bits[chosen] = 0
    elif cur < want:
        idx = np.where(bits == 0)[0]
        if idx.size >= want - cur:
            chosen = idx[rng.permutation(idx.size)[:want - cur]]
            bits[chosen] = 1
    return bits


def top_count_bits(scores, count, rng):
    s = np.asarray(scores, dtype=float).reshape(-1)
    n = s.size
    if count <= 0:
        return np.zeros(n, dtype=int)
    if count >= n:
        return np.ones(n, dtype=int)
    if rng is not None:
        s = s + 1e-9 * rng.random(n)
    order = np.argsort(s)
    bits = np.zeros(n, dtype=int)
    bits[order[n - count:]] = 1
    return bits


def interp_scores(n, ph, pn, rng, noise):
    if ph is not None and pn is not None and pn > 1:
        xp = (np.arange(pn) + 0.5) * (2.0 / float(pn))
        xn = (np.arange(n) + 0.5) * (2.0 / float(n))
        s = np.interp(xn, xp, ph)
    else:
        s = np.full(n, 0.5, dtype=float)
    if noise is not None and noise > 0.0:
        s = s + rng.standard_normal(n) * float(noise)
    return s

    hw = m // 2
    starts = []

    if ph is not None and pn is not None and pn > 1:
        for noise in (0.0, 0.02, 0.05, 0.10):
            s = interp_scores(n, ph, pn, rng, noise)
            pair = 0.5 * (s[:m] + s[n - 1:0:-1])
            starts.append(top_count_bits(pair, hw, rng))

    r = np.zeros(m, dtype=int)
    if hw > 0 and hw < m:
        idx = rng.choice(m, hw, replace=False)
    elif hw > 0:
        idx = np.arange(m)
    else:
        idx = np.array([], dtype=int)
    r[idx] = 1
    starts.append(r)

    for f in (1, 2, 3):
        x = np.arange(m) / float(max(1, m))
        phase = rng.random() * 2.0 * np.pi
        sc = 0.5 + 0.5 * np.cos(2.0 * np.pi * f * x + phase)
        starts.append(top_count_bits(sc, hw, rng))

    return starts[:max_starts]


def make_asym_bits_starts(n, ph, pn, rng, max_starts=5):
    want = n // 2
    starts = []

    if ph is not None and pn is not None and pn > 1:
        for noise in (0.0, 0.02, 0.05, 0.10):
            s = interp_scores(n, ph, pn, rng, noise)
            starts.append(top_count_bits(s, want, rng))

    r = np.zeros(n, dtype=int)
    if 0 < want < n:
        idx = rng.choice(n, want, replace=False)
    elif want == n:
        idx = np.arange(n)
    else:
        idx = np.array([], dtype=int)
    r[idx] = 1
    starts.append(r)

    for f in (1, 2, 3):
        x = np.arange(n) / float(max(1, n))
        phase = rng.random() * 2.0 * np.pi
        sc = 0.5 + 0.5 * np.cos(2.0 * np.pi * f * x + phase)
        starts.append(top_count_bits(sc, want, rng))

    return starts[:max_starts]


def sa_sym(n, half_init, deadline, seed):
    m = n // 2
    hw = m // 2
    if hw <= 0 or m % 2 != 0:
        h = np.full(n, 0.5, dtype=float)
        return h, compute_c5(h)

    rng = np.random.RandomState(seed)
    if half_init is None:
        half = np.zeros(m, dtype=int)
        if hw > 0 and hw < m:
            idx = rng.choice(m, hw, replace=False)
        elif hw > 0:
            idx = np.arange(m)
        else:
            idx = np.array([], dtype=int)
        half[idx] = 1
    else:
        half = repair_weight(np.asarray(half_init, dtype=int), hw, rng)

    pair_masks = get_pair_masks(n)
    masks = get_bit_masks(n)
    targetM = target_M_for_n(n)

    B = build_B_sym(half, pair_masks)
    bestM, bestPlat = eval_sym_limit(B, n, masks, None)
    bestB = B
    bestHalf = half.copy()

    curM, curPlat = bestM, bestPlat
    ones = np.where(half == 1)[0].tolist()
    zeros = np.where(half == 0)[0].tolist()

    delta = bestM - targetM
    if delta > 0:
        T = max(1.0, min(5.0, 0.25 * float(delta) + 1.0))
    else:
        T = 1.0

    move = 0
    noimp = 0

    while time.time() < deadline:
        if not ones or not zeros:
            break
        ri = int(rng.randint(len(ones)))
        rj = int(rng.randint(len(zeros)))
        i = ones[ri]
        j = zeros[rj]

        Bnew = (B & ~pair_masks[i]) | pair_masks[j]
        limit = curM + int(math.ceil(T)) + 1
        Mnew, platnew = eval_sym_limit(Bnew, n, masks, limit)

        accept = False
        if Mnew < curM or (Mnew == curM and platnew < curPlat):
            accept = True
        elif Mnew > curM and T > 1e-4:
            dm = float(Mnew - curM)
            if dm < 25.0:
                accept = (rng.random_sample() < math.exp(-dm / T))
        elif Mnew == curM and T > 0.10:
            accept = (rng.random_sample() < 0.05)

        if accept:
            half[i] = 0
            half[j] = 1
            B = Bnew
            curM, curPlat = Mnew, platnew

            ones[ri] = ones[-1]
            ones.pop()
            zeros[rj] = zeros[-1]
            zeros.pop()
            ones.append(j)
            zeros.append(i)

            if Mnew < bestM:
                bestM = Mnew
                bestPlat = platnew
                bestB = B
                bestHalf = half.copy()
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1

        move += 1
        if move % 500 == 0:
            T = max(0.05, T * 0.97)

        if noimp > 3000:
            T = max(T, 1.25)
            noimp = 0
            if rng.random_sample() < 0.35:
                half = bestHalf.copy()
                B = bestB
                curM, curPlat = bestM, bestPlat
                ones = np.where(half == 1)[0].tolist()
                zeros = np.where(half == 0)[0].tolist()

    h = np.zeros(n, dtype=float)
    for i, val in enumerate(bestHalf):
        if val:
            h[i] = 1.0
            h[n - 1 - i] = 1.0
    return h, compute_c5(h)


def sa_asym(n, bits_init, deadline, seed):

    rng = np.random.RandomState(seed)

        bits = np.zeros(n, dtype=int)

        elif want == n:
            idx = np.arange(n)
        else:
            idx = np.array([], dtype=int)
        bits[idx] = 1
    else:
        bits = repair_weight(np.asarray(bits_init, dtype=int), want, rng)

    masks = get_bit_masks(n)
    targetM = target_M_for_n(n)

    B = build_B_asym(bits)
    bestM, bestPlat = eval_asym_limit(B, n, masks, None)
    bestB = B
    bestBits = bits.copy()

    curM, curPlat = bestM, bestPlat

    zeros = np.where(bits == 0)[0].tolist()

    delta = bestM - targetM
    if delta > 0:
        T = max(1.0, min(5.0, 0.25 * float(delta) + 1.0))
    else:
        T = 1.0

    move = 0
    noimp = 0

    while time.time() < deadline:
        if not ones or not zeros:
            break
        ri = int(rng.randint(len(ones)))
        rj = int(rng.randint(len(zeros)))
        i = ones[ri]
        j = zeros[rj]

        Bnew = (B & ~(1 << i)) | (1 << j)
        limit = curM + int(math.ceil(T)) + 1
        Mnew, platnew = eval_asym_limit(Bnew, n, masks, limit)

        accept = False
        if Mnew < curM or (Mnew == curM and platnew < curPlat):
            accept = True
        elif Mnew > curM and T > 1e-4:
            dm = float(Mnew - curM)
            if dm < 25.0:
                accept = (rng.random_sample() < math.exp(-dm / T))
        elif Mnew == curM and T > 0.10:
            accept = (rng.random_sample() < 0.05)

        if accept:
            bits[i] = 0
            bits[j] = 1
            B = Bnew
            curM, curPlat = Mnew, platnew

            ones[ri] = ones[-1]
            ones.pop()
            zeros[rj] = zeros[-1]
            zeros.pop()
            ones.append(j)
            zeros.append(i)

            if Mnew < bestM:
                bestM = Mnew
                bestPlat = platnew
                bestB = B
                bestBits = bits.copy()
                noimp = 0
            else:
                noimp += 1
        else:
            noimp += 1

        move += 1
        if move % 500 == 0:
            T = max(0.05, T * 0.97)

        if noimp > 3000:
            T = max(T, 1.25)
            noimp = 0
            if rng.random_sample() < 0.35:
                bits = bestBits.copy()
                B = bestB
                curM, curPlat = bestM, bestPlat
                ones = np.where(bits == 1)[0].tolist()
                zeros = np.where(bits == 0)[0].tolist()

    h = np.zeros(n, dtype=float)
    if len(bestBits) == n:
        h[bestBits == 1] = 1.0
    return h, compute_c5(h)


def _smooth_value(x):
    n = _G_N
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    val = m + math.log(float(np.sum(np.exp(_G_BETA * (c - m))))) / _G_BETA
    return float(val) * (2.0 / float(n))


def _smooth_grad(x):
    n = _G_N
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    ex = np.exp(_G_BETA * (c - m))
    s = float(np.sum(ex))
    if s <= 0.0:
        return np.zeros(n, dtype=float)
    w = ex / s
    g = np.zeros(n, dtype=float)
    for idx, wi in enumerate(w):
        if wi < 1e-12:
            continue
        off = idx - (n - 1)
        if off >= 0:
            o = off
            if o < n:
                g[o:] += wi * (1.0 - x[:n - o])
                g[:n - o] -= wi * x[o:]
        else:
            o = -off
            if o < n:
                g[:n - o] += wi * (1.0 - x[o:])
                g[o:] -= wi * x[:n - o]
    return g


def _eq_con(x):
    return float(np.sum(x)) - (_G_N / 2.0)


def continuous_slsqp_polish(h, deadline):
    global _G_N, _G_BETA
    n = len(h)
    if n <= 1:
        return h, compute_c5(h)

    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h)

    if time.time() >= deadline:
        return best_h, best_c

    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{"type": "eq", "fun": _eq_con}]
    stages = [(10.0, 70), (50.0, 90), (200.0, 90), (1000.0, 70)]

    for beta, mi in stages:
        if time.time() >= deadline - 5.0:
            break
        rem = max(3.0, deadline - time.time())
        mi_u = int(min(mi, max(12, rem / 2.0)))
        _G_N = n
        _G_BETA = float(beta)
        try:
            res = minimize(
                _smooth_value,
                h,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                jac=_smooth_grad,
                options={"maxiter": mi_u, "ftol": 1e-12, "iprint": 0},
            )
            if hasattr(res, "x") and np.all(np.isfinite(res.x)):
                h = project_box_sum(res.x, n / 2.0)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        except Exception:
            pass

    return best_h, best_c


def active_subgradient_polish(h, deadline):
    n = len(h)
    if n <= 1 or time.time() >= deadline:
        h = project_box_sum(h, n / 2.0)
        return h, compute_c5(h)

    h = project_box_sum(h, n / 2.0)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    alpha = 0.02

    for _ in range(350):
        if time.time() >= deadline or alpha < 1e-7:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argsort(c)[-6:]
        vals = c[idx]
        ex = np.exp(20.0 * (vals - vals.max()))
        w = ex / float(np.sum(ex))
        g = np.zeros(n, dtype=float)

        for ii, ww in zip(idx, w):
            off = ii - (n - 1)
            if off >= 0:
                o = off
                if o < n:
                    g[o:] += ww * (1.0 - h[:n - o])
                    g[:n - o] -= ww * h[o:]
            else:
                o = -off
                if o < n:
                    g[:n - o] += ww * (1.0 - h[o:])
                    g[o:] -= ww * h[:n - o]

        gn = float(np.linalg.norm(g))
        if gn < 1e-13:
            break
        g /= gn

        improved = False
        for f in (1.0, 0.5, 0.25, 0.125, 2.0):
            cand = project_box_sum(h - f * alpha * g, n / 2.0)
            c2 = np.correlate(cand, 1.0 - cand, mode="full")
            val = float(np.max(c2)) * (2.0 / float(n))
            if val < cur_c - 1e-14:
                h = cand
                cur_c = val
                improved = True
                break

        if improved:
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            alpha = min(0.1, alpha * 1.05)
        else:
            alpha *= 0.75

    return best_h, best_c


def local_smoothing_candidates(h):
    n = len(h)
    if n <= 1:
        return []
    target = n / 2.0
    base = np.asarray(h, dtype=float)
    g = np.clip(2.0 * base - 1.0, -1.0, 1.0)
    cands = []

    for r in (1, 2, 3):
        kernel = np.ones(2 * r + 1, dtype=float)
        kernel /= float(2 * r + 1)
        gs = np.convolve(g, kernel, mode="same")
        hs = project_box_sum(0.5 + 0.5 * gs, target)
        cands.append(hs)
        for a in (0.1, 0.3):
            hb = project_box_sum((1.0 - a) * base + a * hs, target)
            cands.append(hb)

    return cands


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + max(1.0, float(budget_s)) - 5.0
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    ph = None
    pn = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn == 0:
            ph = None
    except Exception:
        ph = None
        pn = 0

    if ph is not None and pn > 0:
        h0 = project_box_sum(ph, pn / 2.0)
        best_h = h0
        best_c = compute_c5(h0)
        best_n = pn
    else:
        best_n = 192
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = compute_c5(best_h)

    if time.time() < hard_deadline - 10.0:
        for h_cand in local_smoothing_candidates(best_h):
            c_cand = compute_c5(h_cand)
            if c_cand < best_c:
                best_c = c_cand
                best_h = h_cand
                best_n = len(h_cand)

    if time.time() < hard_deadline - 160.0:
        cont_budget = min(220.0, max(60.0, 0.16 * float(budget_s)))
        cont_deadline = min(hard_deadline - 160.0, t0 + cont_budget)
        if cont_deadline > time.time() + 20.0:
            h_pol, c_pol = continuous_slsqp_polish(best_h, cont_deadline)
            if c_pol < best_c:
                best_c = c_pol
                best_h = h_pol
                best_n = len(h_pol)

    if time.time() < hard_deadline - 160.0:
        sg_deadline = min(hard_deadline - 160.0, time.time() + 80.0)
        if sg_deadline > time.time() + 10.0:
            h_sg, c_sg = active_subgradient_polish(best_h, sg_deadline)
            if c_sg < best_c:
                best_c = c_sg
                best_h = h_sg
                best_n = len(h_sg)

    src_h = best_h
    src_n = best_n

    if time.time() < hard_deadline - 150.0:
        sym_ns = [192, 196, 188, 204, 220]
        sym_ns = [nn for nn in sym_ns if nn % 4 == 0]
        seed_add = 11

        for n in sym_ns:
            if time.time() >= hard_deadline - 60.0:
                break
            starts = make_sym_half_starts(n, src_h, src_n, rng, 3)
            for start in starts:
                if time.time() >= hard_deadline - 60.0:
                    break
                deadline_n = min(hard_deadline - 60.0, time.time() + 95.0)
                if deadline_n - time.time() < 25.0:
                    break
                h_b, c_b = sa_sym(n, start, deadline_n, seed + seed_add + n)
                if c_b < best_c:
                    best_c = c_b
                    best_h = h_b
                    best_n = n
                    src_h = h_b
                    src_n = n
                seed_add += 7
            if best_c <= 0.37 and time.time() >= hard_deadline - 250.0:
                break

    if time.time() < hard_deadline - 120.0:
        asym_ns = [190, 194, 192, 198, 202]
        seed_add = 101

        for n in asym_ns:
            if time.time() >= hard_deadline - 60.0:
                break
            starts = make_asym_bits_starts(n, src_h, src_n, rng, 2)
            for start in starts:
                if time.time() >= hard_deadline - 60.0:
                    break
                deadline_n = min(hard_deadline - 60.0, time.time() + 85.0)
                if deadline_n - time.time() < 25.0:
                    break
                h_b, c_b = sa_asym(n, start, deadline_n, seed + seed_add + n)
                if c_b < best_c:
                    best_c = c_b
                    best_h = h_b
                    best_n = n
                    src_h = h_b
                    src_n = n
                seed_add += 11
            if best_c <= 0.37 and time.time() >= hard_deadline - 200.0:
                break

    if time.time() < hard_deadline - 40.0 and best_h is not None:
        bh = np.asarray(best_h, dtype=float)
        bn = len(bh)
        is_binary = bool(np.all((bh == 0.0) | (bh == 1.0)))
        if is_binary and bn >= 4:
            bits = (bh == 1.0).astype(int)
            if best_c <= TARGET_C:
                deadline_n = hard_deadline - 10.0
            else:
                deadline_n = hard_deadline - 40.0
            if bn % 4 == 0:
                m = bn // 2
                hw = m // 2
                half = repair_weight(bits[:m], hw, rng)
                h_b, c_b = sa_sym(bn, half, deadline_n, seed + 9973)
            else:
                h_b, c_b = sa_asym(bn, bits, deadline_n, seed + 9973)
            if c_b < best_c:
                best_c = c_b
                best_h = h_b
                best_n = len(h_b)

    best_h = project_box_sum(best_h, best_n / 2.0)
    final_c = compute_c5(best_h)

    try:
        ec = evaluate_erdos_solution(best_h)
        if ec is not None:
            ec = float(ec)
            if ec < final_c:
                final_c = ec
    except Exception:
        pass

    return (best_h, float(final_c), int(best_n))
