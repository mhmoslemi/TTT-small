For P(L) = sum_{i=0}^{n-1-L} h[i+L] * (1-h[i]):
- dP/dh[j] for j from 0 to n-1:
  - If j appears as h[i+L] (i.e., j = i+L, i = j-L): valid for L ≤ j ≤ n-1
    Contribution: (1-h[j-L])
  - If j appears as h[i] (i.e., j = i): valid for 0 ≤ j ≤ n-1-L
    Contribution: -h[j+L]

So:
- j in [0, L-1]: only second term → -h[j+L]
- j in [L, min(n-1-L, n-1)]: both terms → (1-h[j-L]) - h[j+L]
  - Note: n-1-L ≥ L when L ≤ (n-1)/2
- j in [n-L, n-1]: only first term → (1-h[j-L])

For N(L) = sum_{i=0}^{n-1-L} h[i] * (1-h[i+L]):
- dN/dh[j]:
  - If j appears as h[i] (j = i): valid for 0 ≤ j ≤ n-1-L
    Contribution: (1-h[j+L])
  - If j appears as h[i+L] (j = i+L, i = j-L): valid for L ≤ j ≤ n-1
    Contribution: -h[j-L]

So:
- j in [0, L-1]: only first term → (1-h[j+L])
- j in [L, n-1-L]: both → (1-h[j+L]) - h[j-L]
- j in [n-L, n-1]: only second → -h[j-L]

def lag_gradient(h, L, n, dx):
    """Compute gradient of lag L objective w.r.t. h, scaled by dx."""
    g = np.zeros(n, dtype=float)
    if L >= 0:
        m = n - L
        if m <= 0:
            return g
        # P(L) = sum_{i=0}^{m-1} h[i+L](1-h[i])
        g[:m] += -h[L:] * dx
        g[L:] += (1.0 - h[:m]) * dx
    else:
        AL = -L
        m = n - AL
        if m <= 0:
            return g
        # N(L) = sum_{i=0}^{m-1} h[i](1-h[i+L])
        g[:m] += (1.0 - h[AL:]) * dx
        g[AL:] += -h[:m] * dx
    return g

def subgrad_steps(h, n, time_limit, num_active=10, lr_init=0.01, max_iters=3000):

        cur_c = float(np.max(vals))

            best_h = h.copy()

        # Find active lags (within 1e-8 of max)
        thresh = cur_c - 1e-9
        active_mask = vals >= thresh
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = [int(np.argmax(vals))]

        g = np.zeros(n, dtype=float)

        # Negative gradient step
        h_try = h - lr * g
        h_try = project_feasible(h_try, n)

            h = h_try
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-6:

def block_perturb(h0, n, time_limit, seed, max_trials=5000, step_init=0.02):

    rng = np.random.RandomState(int(seed) % (2**31 - 1))
    h = project_feasible(h0, n)

    min_temp = 1e-8

        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))

        delta = rng.randn(L) * step
        h_try = cur_h.copy()
        h_try[idx] += delta

            cur_c = c_try

            cur_c = c_try
            step = min(0.03, step * 1.05)
        else:

                    cur_h = h_try

def gaussian_smooth(h, sigma=1.5):
    h = np.asarray(h, dtype=float)

    w = np.exp(-(k * k) / (2.0 * sigma * sigma))

    return np.convolve(h, w, mode="same")

        h2, c2 = block_perturb(best_h, best_n, twist_end, seed + 7777,
                               max_trials=4000, step_init=0.015)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 7: Final polish
    if time.time() < hard_end - 5.0:
        polish_end = hard_end - 2.0
        h2, c2 = lp_refine(best_h, best_n, polish_end, num_active=15, max_iters=1500)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
