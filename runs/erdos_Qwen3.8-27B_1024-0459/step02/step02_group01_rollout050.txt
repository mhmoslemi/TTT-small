_g_n = 192
_g_beta = 100.0

def smooth_obj(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = _g_beta * (c - m)
    shifted = np.clip(shifted, -500, 0)
    smax = m + np.log(np.sum(np.exp(shifted))) / _g_beta
    return smax * (2.0 / n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14: break
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            j = np.argmax(1.0 - h)
            h[j] = min(1.0, h[j] + diff)
        else:
            j = np.argmax(h)
            h[j] = max(0.0, h[j] + diff)
    return h

def to_binary(h):
    n = len(h)
    n_want = n // 2
    h_b = (h >= 0.5).astype(np.float64)
    n_ones = int(h_b.sum())
    if n_ones > n_want:
        idx = np.where(h_b == 1)[0]
        order = np.argsort(h[idx])
        h_b[idx[order[:n_ones - n_want]]] = 0
    elif n_ones < n_want:
        idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[idx])
        h_b[idx[order[:n_want - n_ones]]] = 1
    return h_b

    """Simulated annealing on binary sequences with exactly n/2 ones."""

    if n_iter < 2: n_iter = 2

    for it in range(n_iter):
        if t_end is not None and time.time() > t_end:
            break
        T = T0 * (cool ** it) if T0 > 0 else 0.0

        q = zeros[qi]
