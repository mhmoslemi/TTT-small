        return np.max(corr)
    
    try:

    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
        # Structured init
        if best_n % 4 == 0:
            pat = np.tile([1.0, 1.0, 0.0, 0.0], best_n // 4)
            best_h = make_feasible(pat, best_n)
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Nesterov subgradient at parent resolution
    if remaining() > 180:
        t0 = time.time()
        h_sg, c5_sg = nesterov_subgradient(best_h, best_n, 8000, seed)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
        t1 = time.time()
        # SLSQP polish
        if remaining() > 60:
            h_opt, c5_opt = slsqp_solve(best_h, best_n, max_iter=10000, ftol=1e-16)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Phase 2: Smooth gradient descent (log-sum-exp)
    if remaining() > 120:
        h_smooth, c5_smooth = gradient_descent_smooth(best_h, best_n, 5000, 100.0, 0.02, seed+1)
        if c5_smooth < best_c5:
            best_c5 = c5_smooth
            best_h = h_smooth
        if remaining() > 40:
            h_opt, c5_opt = slsqp_solve(best_h, best_n, max_iter=6000, ftol=1e-16)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Phase 3: Structure-aware local search
    if remaining() > 60:
        n_rounds = min(300, int(remaining() / 0.2))
        best_h, best_c5 = structure_aware_local(best_h, best_n, n_rounds, rng)
    
    # Phase 4: Trust-constr solver
    if remaining() > 80:
        h_tc, c5_tc = trust_constr_solve(best_h, best_n, max_iter=2000)
        if h_tc is not None and c5_tc < best_c5:
            best_c5 = c5_tc
            best_h = h_tc
        if remaining() > 30:
            h_opt, c5_opt = slsqp_solve(best_h, best_n, max_iter=8000, ftol=1e-17)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Phase 5: Multi-resolution sweep
    res_trials = []
    for delta in [20, 40, 60, 80, -20, -40]:
        nn = best_n + delta
        if nn >= 40 and nn <= 500:
            res_trials.append(nn)
    # Also try specific values
    for nn in [200, 240, 280, 320, 128, 192]:
        if nn not in res_trials:
            res_trials.append(nn)
    
    for n_try in res_trials:
        if remaining() < 50:
            break
        h_interp = make_feasible(interpolate_h(best_h, n_try), n_try)
        
        # Quick subgradient
        h_sg, c5_sg = nesterov_subgradient(h_interp, n_try, 3000, seed + n_try)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
            best_n = n_try
        
        # SLSQP at this resolution
        if remaining() > 35:
            h_opt, c5_opt = slsqp_solve(best_h if best_n == n_try else h_interp,
                                         best_n if best_n == n_try else n_try,
                                         max_iter=5000, ftol=1e-15)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try
    
    # Phase 6: Differential evolution at reduced dimension (if time)
    if remaining() > 120:
        de_n = min(best_n, 80)  # DE is expensive, use lower dim
        h_de_init = make_feasible(interpolate_h(best_h, de_n), de_n)
        h_de, c5_de = de_solve(de_n, h_de_init, max_iter=100, popsize=15, seed=seed+100)
        if h_de is not None:
            # Interpolate back to full resolution and polish
            h_full = make_feasible(interpolate_h(h_de, best_n), best_n)
            h_opt, c5_opt = slsqp_solve(h_full, best_n, max_iter=5000, ftol=1e-15)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Phase 7: Multi-start SLSQP with perturbations
    for trial in range(6):
        if remaining() < 25:
            break
        noise = rng.randn(best_n) * (0.01 + 0.01 * trial)
        h_try = best_h + noise
        h_try = make_feasible(h_try, best_n)
        h_opt, c5_opt = slsqp_solve(h_try, best_n, max_iter=4000, ftol=1e-16)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 8: Final aggressive polish
    if remaining() > 40:
        h_opt, c5_opt = slsqp_solve(best_h, best_n, max_iter=15000, ftol=1e-17)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 9: Final local search
    if remaining() > 20:
        n_rounds = min(200, int(remaining() / 0.15))
        best_h, best_c5 = structure_aware_local(best_h, best_n, n_rounds, rng)
    
    # Phase 10: One more SLSQP at very tight tolerance
    if remaining() > 10:
        h_opt, c5_opt = slsqp_solve(best_h, best_n, max_iter=20000, ftol=1e-18)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Final feasibility enforcement
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Validate
    if not (np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
