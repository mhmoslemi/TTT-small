def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)

    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.0, 1.0

        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

def corr_grad_at_lag(h, k, n):

    center = n - 1

    else:
        lag = abs(kk)

            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

        return _project(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return _project(h, 0.5 * n)


def _grad_lag(h, k, n):
    center = n - 1
    kk = k - center
    g = np.zeros(n, dtype=float)
    if kk == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        m = abs(kk)
        if kk > 0:
            g[m:] += 1.0
            g[m:] -= h[:n - m]
            g[:n - m] -= h[m:]
        else:
            g[:n - m] += 1.0
            g[:n - m] -= h[m:]
            g[m:] -= h[:n - m]
    return g


def _active_grad(h, corr, n, top_k):
    center = n - 1
    top_idx = np.argsort(corr)[-top_k:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    scale = max(1e-10, dv.max())
    w = np.exp(dv / scale)
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for i in range(top_idx.size):
        t = int(top_idx[i])
        k = t - center
        g += w[i] * _grad_lag(h, t, n)
    return g


def _grad_descent(h, target, n, deadline, top_k=10, max_iter=50000, momentum=0.0, kick_every=80):
    best = _project(h, target)
    best_c = _eval_erdos(best)
    best_corr = np.correlate(best, 1.0 - best, mode="full")
    vel = np.zeros(n, dtype=float)
    rng = np.random.default_rng(777)
    stall = 0
    it = 0

    while it < max_iter:
        if time.perf_counter() >= deadline:
            break
        it += 1

        g = _active_grad(best, best_corr, n, top_k)
        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-15:
            break
        d /= norm

        if momentum > 0:
            vel = momentum * vel + d
            d = vel
            d -= np.mean(d)
            vn = np.linalg.norm(d)
            if vn < 1e-15:
                break
            d /= vn

        improved = False
        step = 0.03
        for _ in range(16):
            cand = _project(best + step * d, target)
            c = _eval_erdos(cand)
            if c < best_c - 1e-14:
                best = cand
                best_c = c
                best_corr = np.correlate(best, 1.0 - best, mode="full")
                improved = True
                break
            step *= 0.6

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > kick_every and stall % 10 == 0:
                kick = rng.normal(0, 0.008, n)
                kick -= kick.mean()
                cand = _project(best + kick, target)
                c = _eval_erdos(cand)
                if c < best_c + 0.0003:
                    best = cand
                    best_c = c
                    best_corr = np.correlate(best, 1.0 - best, mode="full")
            if stall > 300:
                break

    return _project(best, target), _eval_erdos(_project(best, target))


def _pair_search(h, target, n, deadline, top_k=12, max_iter=10000):
    best = _project(h, target)
    best_c = _eval_erdos(best)
    best_corr = np.correlate(best, 1.0 - best, mode="full")
    rng = np.random.default_rng(42)
    stall = 0

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break

        g = np.zeros(n, dtype=float)
        top_idx = np.argsort(best_corr)[-top_k:]
        for t in top_idx:
            g += _grad_lag(best, int(t), n)
        d = -g
        d -= np.mean(d)

        neg_pool = np.argsort(g)[:10]
        pos_pool = np.argsort(-g)[:10]

        found = False
        for ii in neg_pool:
            ii = int(ii)
            for jj in pos_pool:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[jj], best[ii])
                if cap < 1e-11:
                    continue
                for frac in (0.25, 0.5, 0.8, 1.0):
                    dmove = frac * cap
                    cand = best.copy()
                    cand[ii] -= dmove
                    cand[jj] += dmove
                    if cand[jj] > 1.0 + 1e-12 or cand[ii] < -1e-12:
                        continue
                    c = _eval_erdos(cand)
                    if c < best_c - 1e-14:
                        best = cand
                        best_c = c
                        best_corr = np.correlate(best, 1.0 - best, mode="full")
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:
            for _r in range(4):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[jj], best[ii])
                if cap < 1e-11:
                    continue
                dmove = float(rng.random()) * 0.4 * cap
                if dmove < 1e-12:
                    continue
                cand = best.copy()
                cand[ii] -= dmove
                cand[jj] += dmove
                c = _eval_erdos(cand)
                if c < best_c:
                    best = cand
                    best_c = c
                    best_corr = np.correlate(best, 1.0 - best, mode="full")
                    found = True
                    break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break

    return _project(best, target), _eval_erdos(_project(best, target))


def _swap_search(h, target, n, deadline, max_iter=5000):
    best = _project(h, target)
    best_c = _eval_erdos(best)
    best_corr = np.correlate(best, 1.0 - best, mode="full")
    rng = np.random.default_rng(314)
    stall = 0

    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        # Try swapping two values (maintains sum automatically)
        for _r in range(8):
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii == jj:
                continue
            cand = best.copy()
            tmp = float(cand[ii])
            cand[ii] = cand[jj]
            cand[jj] = tmp
            c = _eval_erdos(cand)
            if c < best_c - 1e-14:
                best = cand
                best_c = c
                best_corr = np.correlate(best, 1.0 - best, mode="full")
                stall = 0
                break
        else:
            stall += 1
        if stall > 40:
            break

    return _project(best, target), _eval_erdos(_project(best, target))


def _optimize_at_resolution(h0, n, deadline):
    target = 0.5 * n
    h = _project(h0, target)
    best_c = _eval_erdos(h)

    # Phase 1: Broad gradient descent
    rem = deadline - time.perf_counter()
    if rem > 10:
        dl = time.perf_counter() + min(rem * 0.25, 120.0)
        h1, c1 = _grad_descent(h, target, n, dl, top_k=15, max_iter=30000, momentum=0.3)
        if c1 < best_c:
            h, best_c = h1, c1

    # Phase 2: Medium focus
    rem = deadline - time.perf_counter()
    if rem > 8:
        dl = time.perf_counter() + min(rem * 0.25, 100.0)
        h2, c2 = _grad_descent(h, target, n, dl, top_k=8, max_iter=25000, momentum=0.2)
        if c2 < best_c:
            h, best_c = h2, c2

    # Phase 3: Pair search
    rem = deadline - time.perf_counter()
    if rem > 8:
        dl = time.perf_counter() + min(rem * 0.2, 80.0)
        h3, c3 = _pair_search(h, target, n, dl, top_k=10, max_iter=8000)
        if c3 < best_c:
            h, best_c = h3, c3

    # Phase 4: Narrow focus
    rem = deadline - time.perf_counter()
    if rem > 8:
        dl = time.perf_counter() + min(rem * 0.25, 80.0)
        h4, c4 = _grad_descent(h, target, n, dl, top_k=4, max_iter=20000, momentum=0.1)
        if c4 < best_c:
            h, best_c = h4, c2

    # Phase 5: Very narrow
    rem = deadline - time.perf_counter()
    if rem > 8:
        dl = time.perf_counter() + min(rem * 0.3, 60.0)
        h5, c5 = _grad_descent(h, target, n, dl, top_k=2, max_iter=15000, momentum=0.05)
        if c5 < best_c:
            h, best_c = h5, c5

    return _project(h, target), _eval_erdos(_project(h, target))


def _slsqp_refine(h, target, n, deadline, maxiter=100):
    """Quick SLSQP polish with smooth max"""
    if n > 1000:
        return h, _eval_erdos(h)

    def obj(x):
        alpha = 80.0
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(np.max(corr))
        y = alpha * (corr - m)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0 or not np.isfinite(z):
            return 1e100
        return (2.0 / n) * (m + np.log(z) / alpha)

    def jac(x):
        alpha = 80.0
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = float(np.max(corr))
        y = alpha * (corr - m)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0 or not np.isfinite(z):
            return np.zeros(n)
        w = e / z
        center = n - 1
        g = np.zeros(n)
        idx = np.where(w > 1e-8)[0]
        for t in idx:
            t = int(t)
            g += w[t] * _grad_lag(x, t, n)
        return (2.0 / n) * g

    def cfun(x):
        return float(np.sum(x) - target)

    def cjac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": cfun, "jac": cjac},)

    try:
        res = minimize(obj, h, method="SLSQP", jac=jac,
                       bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": 1e-12, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            cand = _project(res.x, target)
            c = _eval_erdos(cand)
            return cand, c
    except Exception:
        pass
    return h, _eval_erdos(h)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = t_start + max(1.0, budget - 12.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Load parent
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            best_h = np.clip(raw, 0.0, 1.0).copy()
    except Exception:
        best_h = None

    if best_h is not None:
        best_n = int(best_h.size)
        best_h = _project(best_h, 0.5 * best_n)
        best_c = _eval_erdos(best_h)
    else:
        best_n = 90
        x = (np.arange(best_n) + 0.5) * 2.0 / best_n
        best_h = _project(0.5 + 0.3 * np.sin(3 * np.pi * x) + 0.15 * np.sin(5 * np.pi * x + 1.0), 0.5 * best_n)
        best_c = _eval_erdos(best_h)

    n0 = best_n
    target0 = 0.5 * n0

    # === PHASE A: Optimize at parent resolution ===
    rem = deadline - time.perf_counter()
    phaseA_time = min(rem * 0.30, 350.0)
    if rem > 30:
        h_a, c_a = _optimize_at_resolution(best_h, n0, t_start + max(rem, 10) * 0.30)
        if c_a < best_c:
            best_h, best_c, best_n = h_a, c_a, n0

    # SLSQP quick polish at parent res
    rem = deadline - time.perf_counter()
    if rem > 20 and n0 <= 600:
        h_s, c_s = _slsqp_refine(best_h, 0.5 * n0, n0, time.perf_counter() + min(rem * 0.15, 60.0), maxiter=80)
        if c_s < best_c:
            best_h, best_c = h_s, c_s

    # === PHASE B: Try 2x resolution ===
    n2 = 2 * n0
    if n2 <= 1500:
        rem = deadline - time.perf_counter()
        if rem > 60:
            h_hi = _upsample(best_h, n2)
            # Also try a slightly perturbed version
            h_hi2 = _project(h_hi + rng.normal(0, 0.02, n2), 0.5 * n2)

            phaseB_time = min(rem * 0.30, 300.0)
            dl_b = time.perf_counter() + phaseB_time

            h_b, c_b = _optimize_at_resolution(h_hi, n2, dl_b)
            if c_b < best_c:
                best_h, best_c, best_n = h_b, c_b, n2

            # Try perturbed start
            rem = deadline - time.perf_counter()
            if rem > 30 and not (best_n == n2 and c_b < best_c):
                dl_b2 = time.perf_counter() + min(rem * 0.2, 100.0)
                h_b2, c_b2 = _optimize_at_resolution(h_hi2, n2, dl_b2)
                if c_b2 < best_c:
                    best_h, best_c, best_n = h_b2, c_b2, n2

    # === PHASE C: Try downscaled resolution (might find different structure) ===
    n_down = max(90, n0 // 2)
    if n_down != n0 and n_down <= 300:
        rem = deadline - time.perf_counter()
        if rem > 30:
            h_dn = _upsample(best_h, n_down)
            dl_c = time.perf_counter() + min(rem * 0.15, 60.0)
            h_c, c_c = _optimize_at_resolution(h_dn, n_down, dl_c)
            if c_c < best_c:
                best_h, best_c, best_n = h_c, c_c, n_down
                # Upsample back to higher res
                n_up = 2 * n_down
                if n_up <= 1200:
                    rem = deadline - time.perf_counter()
                    if rem > 30:
                        h_up = _upsample(best_h, n_up)
                        dl_c2 = time.perf_counter() + min(rem * 0.3, 80.0)
                        h_c2, c_c2 = _optimize_at_resolution(h_up, n_up, dl_c2)
                        if c_c2 < best_c:
                            best_h, best_c, best_n = h_c2, c_c2, n_up

    # === PHASE D: Intensive final polish at best resolution ===
    rem = deadline - time.perf_counter()
    if rem > 15 and best_h is not None:
        n_best = int(best_h.size)
        target_best = 0.5 * n_best
        h_fin = _project(best_h, target_best)

        # Multiple focus levels
        for tk in [5, 3, 7, 3, 2]:
            rem = deadline - time.perf_counter()
            if rem < 8:
                break
            dl = time.perf_counter() + min(rem * 0.20, 60.0)
            h_f1, c_f1 = _grad_descent(h_fin, target_best, n_best, dl, top_k=tk, max_iter=20000, momentum=0.1)
            if c_f1 < best_c:
                h_fin, best_c = h_f1, c_f1
                best_h = h_fin.copy()
                best_n = n_best

        # Pair search
        rem = deadline - time.perf_counter()
        if rem > 10:
            dl = time.perf_counter() + min(rem * 0.25, 50.0)
            h_f2, c_f2 = _pair_search(h_fin, target_best, n_best, dl, top_k=8, max_iter=5000)
            if c_f2 < best_c:
                h_fin, best_c = h_f2, c_f2
                best_h = h_fin.copy()

        # Swap search
        rem = deadline - time.perf_counter()
        if rem > 8 and n_best <= 600:
            dl = time.perf_counter() + min(rem * 0.15, 20.0)
            h_f3, c_f3 = _swap_search(h_fin, target_best, n_best, dl, max_iter=3000)
            if c_f3 < best_c:
                h_fin, best_c = h_f3, c_f3
                best_h = h_fin.copy()

        # Final ultra-narrow gradient
        rem = deadline - time.perf_counter()
        if rem > 8:
            dl = time.perf_counter() + min(rem * 0.35, 40.0)
            h_f4, c_f4 = _grad_descent(h_fin, target_best, n_best, dl, top_k=2, max_iter=10000, momentum=0.0)
            if c_f4 < best_c:
                h_fin, best_c = h_f4, c_f4
                best_h = h_fin.copy()

        # One more pair pass
        rem = deadline - time.perf_counter()
        if rem > 6:
            dl = time.perf_counter() + min(rem * 0.5, 20.0)
            h_f5, c_f5 = _pair_search(h_fin, target_best, n_best, dl, top_k=5, max_iter=3000)
            if c_f5 < best_c:
                h_fin, best_c = h_f5, c_f5
                best_h = h_fin.copy()

    # Final projection and evaluation
    if best_h is None:
        best_h = _project(np.full(90, 0.5), 45.0)
        best_n = 90

    n_final = int(best_h.size)
    best_h = _project(best_h, 0.5 * n_final)
    best_c = _eval_erdos(best_h)

    return best_h, float(best_c), n_final
