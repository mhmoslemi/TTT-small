def compute_subgrad_active(h, n, frac=0.98):
    """Compute subgradient for active (near-max) correlations."""
    c = np.correlate(h, 1.0 - h, mode='full')
    threshold = frac * np.max(c)
    active = np.where(c >= threshold)[0]
    grad = np.zeros(n)
    for ak in active:
        offset = ak - (n - 1)
        if offset >= 0:
            npairs = n - offset
            # corr = sum_{i=0}^{npairs-1} h[i+offset]*(1-h[i])
            # d/dh[j] for j in [0, npairs): contributes -h[j+offset]
            # d/dh[j] for j in [offset, offset+npairs): contributes (1-h[j-offset])
            grad[:npairs] -= h[offset:offset + npairs]
            grad[offset:offset + npairs] += (1.0 - h[:npairs])
        else:
            aoff = -offset
            npairs = n - aoff
            # corr = sum_{i=0}^{npairs-1} h[i]*(1-h[i+aoff])
            # d/dh[j] for j in [0, npairs): (1-h[j+aoff])
            # d/dh[j] for j in [aoff, aoff+npairs): -h[j-aoff]
            grad[:npairs] += (1.0 - h[aoff:aoff + npairs])
            grad[aoff:aoff + npairs] -= h[:npairs]
    return grad

        _BETA = beta
        try:
            res = minimize(obj_smooth, h_cont, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 1500, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feas(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_cont = h_new
        except Exception:
            pass

    # === Phase 4: Subgradient with momentum ===
    if remaining() > 100:
        h_sg = best_h.copy()
        mom = np.zeros(n)
        mu = 0.85
        lr = 0.03
        stall = 0
        for it in range(8000):
            if remaining() < 5:
                break
            grad = subgrad_active(h_sg, n, frac=0.92)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            grad /= gn
            mom = mu * mom + (1.0 - mu) * grad
            h_trial = project_feas(h_sg - lr * mom, n)
            c5_tr = compute_c5(h_trial)
            if c5_tr < best_c5 - 1e-12:
                best_c5 = c5_tr
                best_h = h_trial.copy()
                h_sg = h_trial.copy()
                stall = 0
                lr = min(0.08, lr * 1.02)
            else:
                stall += 1
                lr = max(0.002, lr * 0.95)
            if stall > 600:
                break

    # === Phase 5: Targeted coordinate refinement ===
    if remaining() > 40:
        h_cd = best_h.copy()
        c_cor

r = np.correlate(h_cd, 1.0 - h_cd, mode='full')
        max_c = float(np.max(c_corr))
        # Identify positions contributing to top correlations
        top_k = np.argsort(c_corr)[-min(10, len(c_corr)):]
        hot_positions = set()
        for ak in top_k:
            offset = int(ak) - (n - 1)
            if offset >= 0:
                hot_positions.update(range(max(0, offset), n))
                hot_positions.update(range(max(0, n - offset)))
            else:
                aoff = -offset
                hot_positions.update(range(min(n, n - aoff)))
                hot_positions.update(range(aoff, n))

        max_sweeps = 20
        for sw in range(max_sweeps):
            if remaining() < 8:
                break
            improved = False
            order = list(hot_positions)
            rng.shuffle(order)
            for idx in order:
                if remaining() < 3:
                    break
                old = h_cd[idx]
                candidates = [0.0, 1.0, 0.25, 0.5, 0.75]
                best_nv = old
                best_cv = compute_c5(h_cd)
                for nv in candidates:
                    if abs(nv - old) < 1e-9:
                        continue
                    h_cd[idx] = nv
                    h_cd = project_feas(h_cd, n)
                    cv = compute_c5(h_cd)
                    if cv < best_cv:
                        best_cv = cv
                        best_nv = nv
                if best_nv != old:
                    h_cd[idx] = best_nv
                    h_cd = project_feas(h_cd, n)
                    if best_cv < best_c5:
                        best_c5 = best_cv
                        best_h = h_cd.copy()
                    improved = True
                else:
                    h_cd[idx] = old
                    h_cd = project_feas(h_cd, n)
            if not improved:
                break

    # === Phase 6: Try alternative resolutions ===
    alt_ns = sorted(set([n - 16, n + 16, n - 32, n + 32, n + 48]))
    alt_ns = [an for an in alt_ns if 80 <= an <= 512 and an != n]
    for an in alt_ns:
        if remaining() < 50:
            break
        h_alt = interp_resample(best_h, best_n, an)
        h_alt = project_feas(h_alt, an)
        _N = an
        _TGT = an / 2.0
        bounds_a = [(0.0, 1.0)] * an
        cons_a = [{'type': 'eq', 'fun': con_sum}]
        for beta in [100, 500, 1500]:
            if remaining() < 15:
                break
            _BETA = beta
            try:
                res = minimize(obj_smooth, h_alt, method='SLSQP', bounds=bounds_a,
                               constraints=cons_a,
                               options={'maxiter': 600, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feas(res.x, an)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    best_n = an
            except Exception:
                pass

    # === Phase 7: Final high-precision polish ===
    _N = best_n
    _TGT = best_n / 2.0
    bounds_f = [(0.0, 1.0)] * best_n
    cons_f = [{'type': 'eq', 'fun': con_sum}]
    for beta in [5000, 10000, 50000]:
        if remaining() < 8:
            break
        _BETA = beta
        try:
            res = minimize(obj_smooth, best_h, method='SLSQP', bounds=bounds_f,
                           constraints=cons_f,
                           options={'maxiter': 3000, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_feas(res.x, best_n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Final raw objective as tiebreaker
    if remaining() > 5:
        try:
            res = minimize(obj_raw, best_h, method='SLSQP', bounds=bounds_f,
                           constraints=cons_f,
                           options={'maxiter': 2000, 'ftol': 1e-18, 'iprint': 0})
            h_new = project_feas(res.x, best_n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    best_h = project_feas(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev < final_c5:
            final_c5 = ev
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
