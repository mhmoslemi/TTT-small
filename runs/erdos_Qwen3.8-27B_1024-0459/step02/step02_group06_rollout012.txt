            stall += 1
            if stall > 60:
                break
    best = project_density(best, target)
    return best, evaluate_erdos(best)


def small_step_hill_climb(h, n, deadline, max_iter=50000):
    """Very small random pair adjustments - high acceptance rate"""
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(777)
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        ii = int(rng.integers(n))
        jj = int(rng.integers(n))
        if ii == jj:
            continue
        d = 0.01 + 0.04 * float(rng.random())
        direction = 1.0 if float(rng.random()) < 0.5 else -1.0
        best[ii] += direction * d
        best[jj] -= direction * d
        if best[ii] < -1e-12 or best[ii] > 1 + 1e-12 or best[jj] < -1e-12 or best[jj] > 1 + 1e-12:
            best[ii] -= direction * d
            best[jj] += direction * d
            continue
        c = evaluate_erdos(best)
        if c >= best_c - 1e-14:
            best[ii] -= direction * d
            best[jj] += direction * d
            # Occasionally allow tiny uphill moves for diversification
            if it % 200 == 0 and c < best_c + 0.001:
                if float(rng.random()) < 0.05:
                    best_c = c
    best = project_density(best, target)
    return best, evaluate_erdos(best)


def binarize_push(h, target, thresh=0.15):
    h2 = h.copy()
    for i in range(h2.size):
        if h2[i] < thresh:
            h2[i] = 0.0
        elif h2[i] > 1.0 - thresh:
            h2[i] = 1.0
    return project_density(h2, target)


def make_random_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * rng.integers(2, 8) * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * rng.integers(3, 12) * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.empty(n)
        nseg = int(rng.integers(4, 12))
        seg = n // nseg + 1
        for i in range(0, n, seg):
            h[i:min(i + seg, n)] = float(rng.uniform(0.05, 0.95))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.7, 0.3).astype(float)
        h += rng.normal(0, 0.08, n)
    else:
        h = rng.uniform(0.1, 0.9, n)
    return project_density(h, 0.5 * n)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 20.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Phase 1: Optimize at parent resolution with more iteration
    rem = deadline - time.perf_counter()
    if rem > 60 and best_n <= 600:
        h1, c1 = slsqp_optimize(best_h, best_n, time.perf_counter() + rem * 0.3, alphas=(30, 100, 300))
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, best_n
        rem2 = deadline - time.perf_counter()
        if rem2 > 30:
            h2, c2 = pair_swap_optimize(best_h, best_n, time.perf_counter() + rem2 * 0.6, max_iter=100000, top_frac=0.04, min_k=6)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, best_n

    # Phase 2: Upsample to 2x and optimize
    n_up = min(900, 2 * best_n)
    if n_up >= 300 and n_up <= 1200:
        rem = deadline - time.perf_counter()
        if rem > 100:
            h_up = upsample_h(best_h, n_up)
            # SLSQP
            dl_slsqp = time.perf_counter() + min(80.0, rem * 0.25)
            h3, c3 = slsqp_optimize(h_up, n_up, dl_slsqp, alphas=(20, 60, 200))
            if c3 < best_c:
                best_h, best_c, best_n = h3, c3, n_up
            # Projected gradient
            rem2 = deadline - time.perf_counter()
            if rem2 > 40:
                dl_pg = time.perf_counter() + min(rem2 * 0.35, 180.0)
                h4, c4 = proj_grad_optimize(best_h, n_up, dl_pg, alpha=400, max_iter=15000)
                if c4 < best_c:
                    best_h, best_c, best_n = h4, c4, n_up
            # Pair swap
            rem3 = deadline - time.perf_counter()
            if rem3 > 30:
                dl_ps = time.perf_counter() + min(rem3 * 0.5, 400.0)
                h5, c5 = pair_swap_optimize(best_h, n_up, dl_ps, max_iter=200000, top_frac=0.04, min_k=10)
                if c5 < best_c:
                    best_h, best_c, best_n = h5, c5, n_up
            # Small step hill climb
            rem4 = deadline - time.perf_counter()
            if rem4 > 20:
                dl_hc = time.perf_counter() + min(rem4 * 0.4, 150.0)
                h6, c6 = small_step_hill_climb(best_h, n_up, dl_hc, max_iter=50000)
                if c6 < best_c:
                    best_h, best_c, best_n = h6, c6, n_up

    # Phase 3: Try n=450 specifically if not already done
    if best_n != 450:
        rem = deadline - time.perf_counter()
        if rem > 80:
            h_450 = upsample_h(best_h, 450)
            dl = time.perf_counter() + min(40.0, rem * 0.2)
            h7, c7 = slsqp_optimize(h_450, 450, dl, alphas=(40, 150, 400))
            if c7 < best_c:
                best_h, best_c, best_n = h7, c7, 450
            rem2 = deadline - time.perf_counter()
            if rem2 > 30:
                dl2 = time.perf_counter() + min(rem2 * 0.4, 120.0)
                h8, c8 = pair_swap_optimize(best_h, min(best_n, 450), dl2, max_iter=80000, top_frac=0.04, min_k=8)
                if c8 < best_c:
                    best_h, best_c, best_n = h8, c8, min(best_n, 450)

    # Phase 4: Random perturbations + re-optimization
    rem = deadline - time.perf_counter()
    if rem > 60 and best_n > 0:
        n_try = best_n
        for restart in range(5):
            rem2 = deadline - time.perf_counter()
            if rem2 < 15:
                break
            pert_size = 0.05 + 0.03 * restart
            perturbed = best_h + rng.normal(0, pert_size, n_try)
            perturbed = project_density(perturbed, 0.5 * n_try)
            # Quick gradient polish
            dl = time.perf_counter() + min(rem2 * 0.4, 40.0)
            hp, cp = proj_grad_optimize(perturbed, n_try, dl, alpha=300, max_iter=3000)
            # Quick pair swap
            rem3 = deadline - time.perf_counter()
            if rem3 < 10:
                break
            dl2 = time.perf_counter() + min(rem3 * 0.4, 60.0)
            hp2, cp2 = pair_swap_optimize(hp, n_try, dl2, max_iter=30000, top_frac=0.05, min_k=8)
            if cp2 < best_c:
                best_h, best_c, best_n = hp2, cp2, n_try

    # Phase 5: Binarize + re-optimize
    rem = deadline - time.perf_counter()
    if rem > 30 and best_n > 0:
        n_b = best_n
        h_bin = binarize_push(best_h, 0.5 * n_b, thresh=0.2)
        c_bin = evaluate_erdos(h_bin)
        if c_bin < best_c:
            best_h, best_c = h_bin.copy(), c_bin
        # Re-optimize from binarized
        if rem > 40:
            dl = time.perf_counter() + min(rem * 0.4, 60.0)
            hb, cb = pair_swap_optimize(h_bin, n_b, dl, max_iter=50000, top_frac=0.04, min_k=10)
            if cb < best_c:
                best_h, best_c, best_n = hb, cb, n_b

    # Phase 6: Final intensive polish
    rem = deadline - time.perf_counter()
    if rem > 15 and best_h is not None:
        n_f = best_n
        # Final pair swap
        if rem > 20:
            dl = time.perf_counter() + min(rem * 0.5, 200.0)
            hf, cf = pair_swap_optimize(best_h, n_f, dl, max_iter=100000, top_frac=0.03, min_k=12)
            if cf < best_c:
                best_h, best_c, best_n = hf, cf, n_f
        # Final small-step
        rem2 = deadline - time.perf_counter()
        if rem2 > 10:
            dl = time.perf_counter() + min(rem2 * 0.5, 80.0)
            hf2, cf2 = small_step_hill_climb(best_h, best_n, dl, max_iter=30000)
            if cf2 < best_c:
                best_h, best_c, best_n = hf2, cf2, best_n

    # Final projection and evaluation
    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
