def subgradient_direction(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(corr))
    offset = k_max - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        h_shift = h[offset:offset + n_pairs]
        h_base = 1.0 - h[:n_pairs]
        grad[offset:offset + n_pairs] += h_base
        grad[:n_pairs] -= h_shift
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        h_shift = h[:n_pairs]
        h_base = 1.0 - h[abs_off:abs_off + n_pairs]
        grad[:n_pairs] += h_base
        grad[abs_off:abs_off + n_pairs] -= h_shift
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return None
    return grad / gnorm

        n = best_n
        _N = n
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        h = best_h.copy()
        print("\n--- Phase 4: Subgradient + polish, n={} ---".format(n))

        for it in range(400):
            if remaining_time() < 3:
                break
            grad = subgradient_direction(h, n)
            if grad is None:
                break
            improved = False
            for step in [0.03, 0.08, 0.02, 0.12, 0.05]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial
                    improved = True
                    break
            if not improved and it % 30 == 29 and remaining_time() > 15:
                try:
                    res = minimize(
                        lambda x, b=500: smooth_obj(x, b, n, dx),
                        h, method='SLSQP', bounds=bounds,
                        constraints=[{'type': 'eq', 'fun': lambda x: sum_constraint(x, n)}],
                        options={'maxiter': 150, 'ftol': 1e-14, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new
                except Exception:
                    pass

    # Phase 5: Very fine SLSQP polish
    if remaining_time() > 12:
        n = best_n
        dx = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        print("\n--- Phase 5: Final polish, n={} ---".format(n))

        for beta_val in [1000, 3000, 10000]:
            if remaining_time() < 5:
                break
            try:
                res = minimize(
                    lambda x, b=beta_val: smooth_obj(x, b, n, dx),
                    best_h, method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': lambda x: sum_constraint(x, n)}],
                    options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  Final beta={}: C5={:.7f}".format(beta_val, c5_new))
            except Exception:
                pass

    # Final: one more binary 2-flip pass if the solution is close to binary
    if remaining_time() > 8:
        n = best_n
        h_check = best_h.copy()
        frac_count = np.sum((h_check > 0.1) & (h_check < 0.9))
        if frac_count < n * 0.1:
            h_bin = (h_check > 0.5).astype(float)
            if h_bin.sum() == n // 2:
                h_final, c5_final = binary_2flip_search(h_bin, n, max_sweeps=3, rng=rng)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    if abs(np.sum(best_h) - best_n / 2.0) > 1e-8:
        print("WARNING: sum = {}, expected {}".format(np.sum(best_h), best_n / 2.0))

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
