                if cap < 1e-10:
                    continue
                for d in (0.2 * cap, 0.5 * cap, 0.8 * cap, cap, 0.99 * cap):
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    if cand[ii] < -1e-12 or cand[ii] > 1 + 1e-12 or cand[jj] < -1e-12 or cand[jj] > 1 + 1e-12:
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            for _r in range(20):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * 0.7 * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    best = project_density(best, target)
    return best, evaluate_erdos(best)


def de_run(h_init, n, deadline, pop_size, F, CR, max_gen, seed_val):
    """Simple differential evolution with projection"""
    rng = np.random.default_rng(seed_val)
    target = 0.5 * n
    h_init = project_density(h_init, target)
    c_init = evaluate_erdos(h_init)
    
    # Build population
    pop = [h_init.copy()]
    cvals = [c_init]
    
    x_grid = (np.arange(n) + 0.5) * (2.0 / float(n))
    for _ in range(pop_size - 1):
        mode = _ % 5
        if mode == 0:
            h_new = 0.5 + 0.35 * np.sin(2 * np.pi * rng.integers(2, 8) * x_grid / 2 + rng.uniform(0, 6.28))
        elif mode == 1:
            seg = max(1, n // rng.integers(4, 12))
            h_new = np.empty(n)
            for i in range(0, n, seg):
                h_new[i:min(i+seg, n)] = rng.uniform(0.05, 0.95)
        elif mode == 2:
            h_new = h_init.copy()
            mask = rng.random(n) < 0.3
            h_new[mask] += rng.normal(0, 0.25, mask.sum())
        elif mode == 3:
            h_new = 0.5 + 0.4 * np.sin(2 * np.pi * rng.uniform(1.5, 6.0) * x_grid / 2)
            h_new += 0.1 * np.sin(2 * np.pi * rng.uniform(2, 10) * x_grid / 2 + rng.uniform(0, 6.28))
        else:
            u = rng.random(n)
            med = float(np.median(u))
            h_new = np.where(u > med, rng.uniform(0.6, 0.9), rng.uniform(0.1, 0.4)).astype(float)
        h_new = project_density(h_new, target)
        pop.append(h_new)
        cvals.append(evaluate_erdos(h_new))
    
    best_idx = int(np.argmin(cvals))
    best_h = pop[best_idx].copy()
    best_c = cvals[best_idx]
    
    no_improve = 0
    for gen in range(max_gen):
        if time.perf_counter() >= deadline:
            break
        for i in range(pop_size):
            if time.perf_counter() >= deadline:
                break
            j1, j2, j3 = list(rng.choice(pop_size, size=3, replace=False))
            if i in (j1, j2, j3):
                continue
            mutant = pop[j3].copy() + F * (pop[j1] - pop[j2])
            mutant = project_density(mutant, target)
            
            # Exponential crossover
            trial = pop[i].copy()
            start = int(rng.integers(n))
            while True:
                if rng.random() < CR:
                    trial[start] = mutant[start]
                else:
                    break
                start = (start + 1) % n
            trial = project_density(trial, target)
            c_trial = evaluate_erdos(trial)
            if c_trial < cvals[i]:
                pop[i] = trial
                cvals[i] = c_trial
                if c_trial < best_c:
                    best_h = trial.copy()
                    best_c = c_trial
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        if no_improve > pop_size * 3:
            break
    
    return best_h, best_c


def make_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.12 * np.sin(2 * np.pi * 7 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        seg = int(n // rng.integers(6, 14)) + 1
        h = np.empty(n)
        for i in range(0, n, seg):
            h[i:min(i+seg, n)] = rng.uniform(0.1, 0.9)
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.72, 0.28).astype(float)
        h += rng.normal(0, 0.06, n)
    elif mode == 3:
        # Periodic with period 1
        h = 0.5 + 0.45 * np.sin(2 * np.pi * x + rng.uniform(0, 6.28))
        h += 0.08 * np.sin(4 * np.pi * x + rng.uniform(0, 6.28))
    elif mode == 4:
        # Two-scale
        h = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * x / 2) + 0.2 * np.sin(2 * np.pi * 9 * x / 2)
        h += rng.normal(0, 0.03, n)
    else:
        h = 0.5 + 0.38 * np.sin(2 * np.pi * int(rng.integers(2, 12)) * x / 2 + rng.uniform(0, 6.28))
    return project_density(h, 0.5 * n)


def binarize_push(h, target, thresh):
    n = h.size
    h2 = h.copy()
    changed = False
    for i in range(n):
        if h2[i] < thresh:
            target -= h2[i]
            h2[i] = 0.0
            changed = True
        elif h2[i] > 1.0 - thresh:
            target -= (1.0 - h2[i])
            h2[i] = 1.0
            changed = True
    if not changed:
        return h
    h2 = project_density(h2, 0.5 * n)
    return h2


def slsqp_opt(h, n, deadline, alphas, max_iters):
    global _NC, _ALPHA, _TARGET
    target = 0.5 * n
    _NC = n
    _TARGET = target
    best_h = project_density(h, target)
    best_c = evaluate_erdos(best_h)
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac},)
    
    for alpha in alphas:
        if time.perf_counter() >= deadline - 5.0:
            break
        _ALPHA = alpha
        try:
            res = minimize(
                smooth_objective, best_h,
                method="SLSQP", jac=smooth_jacobian,
                bounds=bounds, constraints=constraints,
                options={"maxiter": max_iters, "ftol": 1e-12, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
        except Exception:
            pass
    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 25.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 7 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    base_n = best_n if best_n > 1 else 450

    # Phase 1: DE global search at multiple resolutions
    # Prioritize resolutions near the parent and a few others
    n_candidates = []
    if base_n > 1:
        n_candidates.append(base_n)
    for tn in [360, 420, 450, 480, 540, 600, 720]:
        if tn not in n_candidates and 20 <= tn <= 800:
            n_candidates.append(tn)
    n_candidates = sorted(set(n_candidates))
    
    # Time allocation for DE phase
    de_phase_end = start + budget * 0.45
    de_time_per = max(20.0, (de_phase_end - start - 10.0) / len(n_candidates))

    for ni in n_candidates:
        if time.perf_counter() >= de_phase_end - 10.0:
            break
        sub_deadline = time.perf_counter() + min(de_time_per, de_phase_end - time.perf_counter() - 5.0)
        
        # Build initial population
        h0 = upsample_h(best_h, ni)
        c0 = evaluate_erdos(h0)
        if c0 < best_c:
            best_h = h0.copy()
            best_c = c0
            best_n = ni
        
        # Run DE with different parameters
        if ni <= 500:
            pop_sz = 25
            max_gen = 200
            F_vals = [0.5, 0.8]
            CR_vals = [0.7, 0.9]
        else:
            pop_sz = 18
            max_gen = 150
            F_vals = [0.6, 0.9]
            CR_vals = [0.8, 1.0]
        
        de_sub_dl = sub_deadline
        n_de_runs = len(F_vals) * len(CR_vals)
        per_de = max(5.0, (de_sub_dl - time.perf_counter()) / n_de_runs)
        
        for fi, F in enumerate(F_vals):
            for ci, CR in enumerate(CR_vals):
                dl_de = time.perf_counter() + min(per_de, de_sub_dl - time.perf_counter())
                sv = seed + fi * 100 + ci * 17 + ni
                try:
                    h_de, c_de = de_run(h0, ni, dl_de, pop_sz, F, CR, max_gen, sv)
                    if h_de is not None:
                        h_de = project_density(h_de, 0.5 * ni)
                        c_de = evaluate_erdos(h_de)
                        if c_de < best_c:
                            best_h = h_de.copy()
                            best_c = c_de
                            best_n = ni
                except Exception:
                    pass
    
    # Phase 2: SLSQP refinement on best solution
    phase2_end = start + budget * 0.70
    ns = best_n
    target_s = 0.5 * ns
    h_work = project_density(best_h, target_s)
    
    alphas = (30.0, 80.0, 200.0, 500.0)
    if ns <= 300:
        alphas = (20.0, 60.0, 150.0, 400.0)
        max_it = 200
    elif ns <= 600:
        alphas = (25.0, 70.0, 200.0, 500.0)
        max_it = 120
    else:
        alphas = (30.0, 100.0, 300.0)
        max_it = 80
    
    try:
        h2, c2 = slsqp_opt(h_work, ns, phase2_end, alphas, max_it)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
            best_n = ns
    except Exception:
        pass

    # Phase 3: Intensive local optimization
    phase3_end = start + budget * 0.88
    n3 = best_n
    h3_in = project_density(best_h, 0.5 * n3)
    
    # Gradient polish
    rem3 = phase3_end - time.perf_counter()
    if rem3 > 15.0:
        try:
            h_g, c_g = polish_gradient(h3_in, time.perf_counter() + rem3 * 0.3, 500, 0.04, 8,
                                       [0.002, 0.005, 0.01, 0.02, 0.05, 0.1])
            if c_g < best_c:
                best_h = h_g.copy()
                best_c = c_g
        except Exception:
            pass
    
    # Pair swap
    rem3 = phase3_end - time.perf_counter()
    if rem3 > 15.0 and n3 <= 600:
        try:
            h_ps, c_ps = pair_swap(best_h, time.perf_counter() + rem3 * 0.4, 4000, 0.05, 6, 10, 10)
            if c_ps < best_c:
                best_h = h_ps.copy()
                best_c = c_ps
        except Exception:
            pass
    
    # More gradient
    rem3 = phase3_end - time.perf_counter()
    if rem3 > 10.0:
        try:
            h_g2, c_g2 = polish_gradient(best_h, time.perf_counter() + rem3 * 0.5, 600, 0.03, 10,
                                         [0.001, 0.003, 0.008, 0.02, 0.04])
            if c_g2 < best_c:
                best_h = h_g2.copy()
                best_c = c_g2
        except Exception:
            pass
    
    # More pair swap
    rem3 = phase3_end - time.perf_counter()
    if rem3 > 10.0 and n3 <= 500:
        try:
            h_ps2, c_ps2 = pair_swap(best_h, time.perf_counter() + rem3 * 0.5, 3000, 0.04, 8, 8, 8)
            if c_ps2 < best_c:
                best_h = h_ps2.copy()
                best_c = c_ps2
        except Exception:
            pass

    # Phase 4: Binarize push and re-optimize
    phase4_end = deadline
    nb = best_n
    h4_in = project_density(best_h, 0.5 * nb)
    
    for thresh in [0.15, 0.10, 0.08, 0.05]:
        if time.perf_counter() >= phase4_end - 10.0:
            break
        try:
            h_bin = binarize_push(h4_in, 0.5 * nb, thresh)
            c_bin = evaluate_erdos(h_bin)
            if c_bin < best_c:
                best_h = h_bin.copy()
                best_c = c_bin
                h4_in = h_bin.copy()
        except Exception:
            pass
    
    # Post-binarize optimization
    rem4 = phase4_end - time.perf_counter()
    if rem4 > 10.0:
        try:
            h_gf, c_gf = polish_gradient(best_h, time.perf_counter() + rem4 * 0.4, 800, 0.03, 8,
                                         [0.001, 0.003, 0.01, 0.02, 0.05])
            if c_gf < best_c:
                best_h = h_gf.copy()
                best_c = c_gf
        except Exception:
            pass
    
    rem4 = phase4_end - time.perf_counter()
    if rem4 > 10.0 and nb <= 500:
        try:
            h_pf, c_pf = pair_swap(best_h, time.perf_counter() + rem4 * 0.5, 5000, 0.03, 8, 12, 12)
            if c_pf < best_c:
                best_h = h_pf.copy()
                best_c = c_pf
        except Exception:
            pass
    
    # Final SLSQP squeeze
    rem4 = phase4_end - time.perf_counter()
    if rem4 > 15.0 and nb <= 600:
        try:
            h_sf, c_sf = slsqp_opt(best_h, nb, time.perf_counter() + rem4 * 0.6, (100.0, 300.0, 800.0), 100)
            if c_sf < best_c:
                best_h = h_sf.copy()
                best_c = c_sf
        except Exception:
            pass

    # Phase 5: Try nearby resolutions
    rem5 = deadline - time.perf_counter()
    if rem5 > 30.0:
        for dn in [-10, -5, 5, 10, 15, -15, 20, -20]:
            rem5 = deadline - time.perf_counter()
            if rem5 < 20.0:
                break
            nn = best_n + dn
            if nn < 20 or nn > 800:
                continue
            try:
                h_up = upsample_h(best_h, nn)
                c_up = evaluate_erdos(h_up)
                if c_up < best_c - 0.0005:
                    # Worth trying to optimize at this resolution
                    h_opt, c_opt = polish_gradient(h_up, time.perf_counter() + min(rem5 * 0.4, 30.0),
                                                   300, 0.04, 6, [0.002, 0.01, 0.03, 0.08])
                    if c_opt < best_c:
                        best_h = h_opt.copy()
                        best_c = c_opt
                        best_n = nn
            except Exception:
                pass

    # Final validation and return
    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    if final_c < best_c:
        best_c = final_c
        best_h = final_h
    
    return best_h, float(best_c), int(best_h.size)
