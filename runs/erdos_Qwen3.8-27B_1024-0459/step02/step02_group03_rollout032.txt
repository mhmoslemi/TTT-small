def slsqp_obj(x, beta, top_k):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, beta) * dx

def slsqp_grad(x, beta, top_k):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    weights = smooth_max_weights(corr, beta, top_k)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for k in active:
        w = weights[k]
        L = k - (n - 1)
        g = grad_lag(x, L)
        grad += w * g
    return dx * grad

def run_slsqp(h0, n, beta, top_k, max_iter):
    def obj(x):
        return slsqp_obj(x, beta, top_k)
    def jac(x):
        return slsqp_grad(x, beta, top_k)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def projected_subgradient(h, n, max_iters, alpha0, alpha_decay, top_k=3,
                          rng=None, temp=0.0):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_order = np.argsort(corr)[::-1][:top_k]
        max_val = corr[top_order[0]] * dx
        
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
        grad = np.zeros(n)
        for k in top_order:
            L = k - (n - 1)
            grad += grad_lag(h, L)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        
        alpha = alpha0 / (1.0 + alpha_decay * it)
        h_new = h - alpha * grad / max(1.0, grad_norm)
        
        if temp > 0:
            h_new += temp * rng.randn(n) * alpha / max(1.0, grad_norm)
        
        h_new = project_feasible(h_new, n)
        
        new_c5 = compute_c5(h_new)
        if new_c5 <= best_c5 + 1e-10:
            h = h_new
        else:
            h = h_new  # still accept for exploration
        
        if (it + 1) % 500 == 0:
            pass
    
    # Ensure best is returned
    final_c5 = compute_c5(best_h)
    return best_h, final_c5

def binary_local_search(h_bin, n, max_rounds, rng):
    h = h_bin.copy()
    c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for rnd in range(max_rounds):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        top_lags = np.argsort(corr)[::-1][:5]
        
        n_trials = min(300, n * 3)
        for t in range(n_trials):
            # Pick a 1 to flip and a 0 to flip
            # Bias towards ones that are in high-overlap positions
            i1 = rng.randint(len(ones))
            i0 = rng.randint(len(zeros))
            p1 = ones[i1]
            p0 = zeros[i0]
            
            h[p1] = 0.0
            h[p0] = 1.0
            new_c5 = compute_c5(h)
            
            if new_c5 < c5 - 1e-15:
                c5 = new_c5
                improved = True
                # Update the ones/zeros sets
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[p1] = 1.0
                h[p0] = 0.0
        
        if not improved:
            # Try targeted flips for the max lag
            max_k = int(np.argmax(corr))
            L = max_k - (n - 1)
            if L > 0:
                # Find 1s at positions i where h[i+L]=0 (they contribute)
                contrib_ones = [i for i in ones if i + L < n and h[i + L] == 0.0]
                contrib_zeros = [j for j in zeros if j - L >= 0 and h[j - L] == 1.0]
                
                if len(contrib_ones) > 0 and len(contrib_zeros) > 0:
                    for _ in range(50):
                        p1 = contrib_ones[rng.randint(len(contrib_ones))]
                        p0 = contrib_zeros[rng.randint(len(contrib_zeros))]
                        h[p1] = 0.0
                        h[p0] = 1.0
                        new_c5 = compute_c5(h)
                        if new_c5 < c5 - 1e-15:
                            c5 = new_c5
                            improved = True
                            ones = np.where(h == 1.0)[0]
                            zeros = np.where(h == 0.0)[0]
                        else:
                            h[p1] = 1.0
                            h[p0] = 0.0
            elif L < 0:
                Lp = -L
                contrib_ones = [i for i in ones if i - Lp >= 0 and h[i - Lp] == 0.0]
                contrib_zeros = [j for j in zeros if j + Lp < n and h[j + Lp] == 1.0]
                
                if len(contrib_ones) > 0 and len(contrib_zeros) > 0:
                    for _ in range(50):
                        p1 = contrib_ones[rng.randint(len(contrib_ones))]
                        p0 = contrib_zeros[rng.randint(len(contrib_zeros))]
                        h[p1] = 0.0
                        h[p0] = 1.0
                        new_c5 = compute_c5(h)
                        if new_c5 < c5 - 1e-15:
                            c5 = new_c5
                            improved = True
                            ones = np.where(h == 1.0)[0]
                            zeros = np.where(h == 0.0)[0]
                        else:
                            h[p1] = 1.0
                            h[p0] = 0.0
            
            if not improved:
                break
    
    return h, c5

def make_fourier_init(n, rng, n_modes=6):
    h = np.zeros(n)
    for m in range(1, n_modes + 1):
        amp = rng.uniform(0.1, 0.4) / m
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * m * np.arange(n) / n + phase)
    h = 0.5 + h
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)

def make_block_init(n, rng):
    h = np.zeros(n)
    n_blocks = rng.randint(3, 10)
    sizes = np.sort(rng.randint(5, n // n_blocks + 5, n_blocks))
    sizes = sizes[:n // 2 // max(1, n_blocks // 2)]
    pos = 0
    for sz in sizes:
        if pos + sz > n // 2:
            break
        h[pos:pos + sz] = 1.0
        pos += sz + rng.randint(1, 3)
    return project_feasible(h, n)

def to_binary(h, n):
    h_round = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_round))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_round == 1.0)[0]
        vals = h[ones]
        order = np.argsort(vals)[:n_ones - n_want]
        h_round[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_round == 0.0)[0]
        vals = h[zeros]
        order = np.argsort(-vals)[:n_want - n_ones]
        h_round[zeros[order]] = 1.0
    return h_round

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - start_time
    
    def remaining():
        return max(0.0, budget_s - elapsed() - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0 and np.all(parent_h >= -0.01) and np.all(parent_h <= 1.01):
            parent_h = np.clip(parent_h, 0, 1)
            parent_h = project_feasible(parent_h, parent_n)
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        pass
    
    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default binary: n={}, C5={:.8f}".format(best_n, best_c5))
    
    # Choose working resolution
    n_work = 300
    # Ensure n_work is even
    if n_work % 2 != 0:
        n_work += 1
    
    # Phase 1: Interpolate and do SLSQP at working resolution
    if remaining() > 200:
        print("\n=== Phase 1: SLSQP at n={} ===".format(n_work))
        h_work = interp_h(best_h, n_work)
        
        # SLSQP with increasing beta and shrinking top_k
        for beta_val in [50, 150, 400]:
            if remaining() < 60:
                break
            top_k = max(1, int(2.0 / n_work * 500 / beta_val * 10))
            top_k = min(top_k, 50)
            iters = 300
            try:
                h_new, c5_new = run_slsqp(h_work, n_work, beta_val, top_k, iters)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_work
                    print("  SLSQP beta={} top_k={}: C5={:.8f}".format(beta_val, top_k, c5_new))
                h_work = h_new
            except Exception as e:
                print("  SLSQP error: {}".format(str(e)[:50]))
                break
    
    # Phase 2: Projected subgradient with active set
    if remaining() > 200:
        print("\n=== Phase 2: Projected subgradient ===")
        h_sg = interp_h(best_h, n_work) if len(best_h) != n_work else best_h.copy()
        
        # Multiple passes with different top_k and alpha
        for (tk, a0, aD, mi, temp) in [
            (5, 0.3, 0.005, 2000, 0.0),
            (3, 0.2, 0.005, 2000, 0.0),
            (2, 0.1, 0.003, 2000, 0.0),
            (1, 0.08, 0.003, 2000, 0.0),
            (3, 0.1, 0.002, 1500, 0.01),
            (1, 0.05, 0.002, 1500, 0.0),
            (5, 0.15, 0.004, 1000, 0.02),
        ]:
            if remaining() < 40:
                break
            h_new, c5_new = projected_subgradient(
                h_sg, n_work, mi, a0, aD, tk, rng, temp)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_work
                print("  SG tk={} a0={}: C5={:.8f}".format(tk, a0, c5_new))
            else:
                # Still update h_sg to the result for continuity
                h_sg = h_new
                       
    
    # Phase 3: Try at a different resolution
    n_work2 = 256
    if remaining() > 150 and n_work2 != best_n:
        print("\n=== Phase 3: n={} ===".format(n_work2))
        h_256 = interp_h(best_h, n_work2)
        for beta_val in [100, 300]:
            if remaining() < 50:
                break
            try:
                h_new, c5_new = run_slsqp(h_256, n_work2, beta_val, 30, 200)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_work2
                    print("  SLSQP n={} beta={}: C5={:.8f}".format(n_work2, beta_val, c5_new))
                h_256 = h_new
            except Exception:
                break
    
    # Phase 4: Binary rounding and local search
    if remaining() > 150:
        print("\n=== Phase 4: Binary local search ===")
        # Try binary at best_n
        h_bin = to_binary(best_h, best_n)
        c5_bin = compute_c5(h_bin)
        print("  Binary at n={}: C5={:.8f}".format(best_n, c5_bin))
        
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        
        # Binary local search
        if remaining() > 100:
            h_bs, c5_bs = binary_local_search(h_bin, best_n, 2000, rng)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()
                print("  Binary LS n={}: C5={:.8f}".format(best_n, c5_bs))
        
        # Also try binary at n_work
        if remaining() > 80 and n_work != best_n:
            h_bin2 = to_binary(best_h, n_work)
            c5_bin2 = compute_c5(h_bin2)
            if c5_bin2 < best_c5:
                best_c5 = c5_bin2
                best_h = h_bin2.copy()
                print("  Binary n={}: C5={:.8f}".format(n_work, c5_bin2))
            
            if remaining() > 60:
                h_bs2, c5_bs2 = binary_local_search(h_bin2, n_work, 1500, rng)
                if c5_bs2 < best_c5:
                    best_c5 = c5_bs2
                    best_h = h_bs2.copy()
                    print("  Binary LS n={}: C5={:.8f}".format(n_work, c5_bs2))
        
        # Multiple restart binary local search with perturbations
        if remaining() > 100:
            for restart in range(5):
                if remaining() < 30:
                    break
                h_pert = to_binary(best_h, best_n)
                n_flips = rng.randint(5, max(6, best_n // 20))
                ones_p = np.where(h_pert == 1.0)[0]
                zeros_p = np.where(h_pert == 0.0)[0]
                for _ in range(n_flips):
                    p1 = ones_p[rng.randint(len(ones_p))]
                    p0 = zeros_p[rng.randint(len(zeros_p))]
                    h_pert[p1] = 0.0
                    h_pert[p0] = 1.0
                h_p = np.where(h_pert == 1.0)[0]
                h_z = np.where(h_pert == 0.0)[0]
                
                h_r, c5_r = binary_local_search(h_pert, best_n, 1000, rng)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    print("  Restart {}: C5={:.8f}".format(restart, c5_r))
    
    # Phase 5: Final continuous polish
    if remaining() > 40:
        print("\n=== Phase 5: Final polish ===")
        n_final = best_n
        h_fin = best_h.copy() if len(best_h) == n_final else interp_h(best_h, n_final)
        
        # If binary, add small perturbation for continuous optimization
        if np.all((h_fin == 0) | (h_fin == 1)):
            h_fin = h_fin + rng.uniform(-0.05, 0.05, n_final)
            h_fin = project_feasible(h_fin, n_final)
        
        for beta_val in [500, 1000]:
            if remaining() < 20:
                break
            try:
                h_new, c5_new = run_slsqp(h_fin, n_final, beta_val, 20, 200)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_final
                    print("  Polish beta={}: C5={:.8f}".format(beta_val, c5_new))
                h_fin = h_new
            except Exception:
                break
        
        # Also try binary polish
        h_b = to_binary(best_h, best_n)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
    
    # Final evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
