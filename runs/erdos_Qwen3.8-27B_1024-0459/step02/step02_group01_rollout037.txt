import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(corr, m, beta):
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def make_smooth_obj(n, beta):
    def obj(h):
        corr = get_correlation(h)
        m = np.max(corr)
        val = smooth_max_value(corr, m, beta)
        return val * (2.0 / n)
    return obj

def make_topk_obj(n, k):
    def obj(h):
        corr = get_correlation(h)
        top = np.sort(corr)[-k:]
        return np.max(top) * (2.0 / n)
    return obj

def make_eq_con(n):
    def con(h):
        return np.sum(h) - n / 2.0
    return con

def slsqp_optimize(h, n, beta, maxiter=500):
    """Run SLSQP with smooth max objective"""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': make_eq_con(n)}]
    obj = make_smooth_obj(n, beta)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except:
        return h

def binary_local_search(h, n, max_rounds=200, time_limit=None, t0=None):
    """Exhaustive 2-swap local search on binary vector"""

    """Compute change in correlation vector when h[i]:1→0 and h[j]:0→1"""
    delta = np.zeros(2*n-1)
    
    # Term 1: -(1-h[i+τ]) for valid τ where 0 ≤ i+τ ≤ n-1
    # τ ranges from -i to n-1-i
    tau_min1 = max(0 - n + 1, -i)  # min valid τ overall is -(n-1)
    tau_max1 = min(n - 1, n - 1 - i)  # max valid τ where i+τ ≤ n-1
    # Actually τ range: -i ≤ τ ≤ n-1-i, intersected with [-(n-1), n-1]
    t1_start = max(-(n-1), -i)
    t1_end = min(n-1, n-1-i)
    if t1_start <= t1_end:
        taus = np.arange(t1_start, t1_end + 1)
        idx = np.arange(i + t1_start, i + t1_end + 1)
        delta[n-1+taus] -= (1.0 - h[idx])
    
    # Term 2: +h[i-τ] for valid τ where 0 ≤ i-τ ≤ n-1
    # τ ranges from i-(n-1) to i
    t2_start = max(-(n-1), i-(n-1))
    t2_end = min(n-1, i)
    if t2_start <= t2_end:
        taus = np.arange(t2_start, t2_end + 1)
        idx = np.arange(i + t2_start, i + t2_end + 1)  # i - τ for τ from t2_start to t2_end
        # i - t2_start, i - (t2_start+1), ..., i - t2_end
        idx = i - taus
        delta[n-1+taus] += h[idx]
    
    # Term 3: +(1-h[j+τ]) for valid τ where 0 ≤ j+τ ≤ n-1
    t3_start = max(-(n-1), -j)
    t3_end = min(n-1, n-1-j)
    if t3_start <= t3_end:
        taus = np.arange(t3_start, t3_end + 1)
        idx = np.arange(j + t3_start, j + t3_end + 1)
        delta[n-1+taus] += (1.0 - h[idx])
    
    # Term 4: -h[j-τ] for valid τ where 0 ≤ j-τ ≤ n-1
    t4_start = max(-(n-1), j-(n-1))
    t4_end = min(n-1, j)
