        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def optimize_slsqp(h0, n, beta, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj_smooth, h0, jac=lambda x: grad_smooth(x, beta),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def optimize_lbfgs(h0, n, beta, mu, max_iter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_pen, h0, jac=lambda x: grad_pen(x, beta, mu),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                            'maxls': 40, 'maxfun': 5000, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def make_fourier_init(n, rng, n_comp=4):
    x = np.arange(n) / n
    h = np.full(n, 0.5)
    for k in range(1, n_comp + 1):
        amp = rng.uniform(0.03, 0.2) / (0.5 + k * 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(2 * np.pi * k * x + phase)
        phase2 = rng.uniform(0, 2 * np.pi)
        h += amp * 0.7 * np.cos(2 * np.pi * k * x + phase2)
    return project_feasible(np.clip(h, 0, 1), n)

def make_block_init(n, rng, n_blocks=None):
    if n_blocks is None:
        n_blocks = rng.randint(4, 10)
    h = np.zeros(n)
    boundaries = np.sort(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    edges = np.concatenate([[0], boundaries, [n]])
    for i in range(len(edges) - 1):
        if rng.random() < 0.5:
            h[edges[i]:edges[i + 1]] = 1.0
    return project_feasible(h, n)

def subgradient_polish(h, n, n_steps, time_limit):
    h_cur = h.copy()
    best_local_c5 = compute_c5(h_cur)
    t0 = time.time()
    for step in range(n_steps):
        if time.time() - t0 > time_limit or (time.time() - t0 > time_limit and step > 10):
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        active = np.where(corr > np.max(corr) - 1e-10)[0]
        # Use top few active lags for stability
        top_k = np.argsort(corr)[-min(5, len(active)):]
        grad = np.zeros(n)
        for k in top_k:
            grad += grad_correlation(h_cur, k, n)
        grad /= len(top_k)
        alpha = 2.0 / (5.0 + step * 0.3)
        h_try = h_cur - alpha * grad
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_local_c5:
            h_cur = h_try
            best_local_c5 = c5_try
        else:
            # Try smaller step
            h_try2 = h_cur - 0.3 * alpha * grad
            h_try2 = project_feasible(h_try2, n)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < best_local_c5:
                h_cur = h_try2
                best_local_c5 = c5_try2
    return h_cur, best_local_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time) - 12.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = make_fourier_init(best_n, rng, 3)
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 1: Polish at parent resolution
    print("\n--- Phase 1: Polish n={} ---".format(best_n))
    cur = best_h.copy()
    for beta in [30, 80, 200, 500, 1000]:
        if remaining() < 40:
            break
        try:
            h_new, c5_new = optimize_slsqp(cur, best_n, beta, max_iter=400)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  beta={}, C5={:.6f}".format(beta, best_c5))
            cur = h_new
        except Exception:
            break

    # Phase 2: Coarse-to-fine optimization
    n_targets = [256, 320, 400, 500, 640, 800, 1000, 1200]
    n_targets = [nn for nn in n_targets if nn >= 128]
    if best_n not in n_targets:
        n_targets.insert(0, best_n)
    n_targets = sorted(set(n_targets))

    for n in n_targets:
        if remaining() < 70:
            break
        print("\n--- Phase 2: n={} (rem {:.0f}s) ---".format(n, remaining()))

        # Build initializations
        inits = [interp_h(best_h, n)]
        inits.append(make_fourier_init(n, rng, n_comp=3))
        inits.append(make_block_init(n, rng))

        for init_idx, h0 in enumerate(inits):
            if remaining() < 45:
                break
            h0 = project_feasible(h0, n)
            cur = h0.copy()

            if n <= 500:
                betas = [25, 60, 150, 400]
                max_iters = [300, 300, 250, 200]
                for i, (beta, mit) in enumerate(zip(betas, max_iters)):
                    if remaining() < 35:
                        break
                    try:
                        h_new, c5_new = optimize_slsqp(cur, n, beta, max_iter=mit)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            best_n = n
                            print("  init{}, beta={}, C5={:.6f}".format(init_idx, beta, best_c5))
                        cur = h_new
                    except Exception:
                        break
            else:
                # Use L-BFGS-B with penalty for larger n
                betas = [30, 80, 200, 500]
                mus = [50, 200, 1000, 5000]
                max_iters = [200, 200, 150, 150]
                for i, (beta, mu, mit) in enumerate(zip(betas, mus, max_iters)):
                    if remaining() < 30:
                        break
                    try:
                        h_new, c5_new = optimize_lbfgs(cur, n, beta, mu, max_iter=mit)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            best_n = n
                            print("  init{}, beta={}, C5={:.6f}".format(init_idx, beta, best_c5))
                        cur = h_new
                    except Exception:
                        break

        # Polish best at this n
        if remaining() > 30 and best_n == n:
            try:
                if n <= 500:
                    h_pol, c5_pol = optimize_slsqp(best_h, n, 1000, max_iter=400)
                else:
                    h_pol, c5_pol = optimize_lbfgs(best_h, n, 1000, 5000, max_iter=300)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
                    print("  Polished n={}: C5={:.6f}".format(n, best_c5))
            except Exception:
                pass

    # Phase 3: Multi-scale refinement
    if remaining() > 100:
        print("\n--- Phase 3: Multi-scale ---")
        # Go up one resolution from best
        n_up = best_n * 2 if best_n * 2 <= 1600 else best_n + 200
        if n_up != best_n:
            h_up = interp_h(best_h, n_up)
            if n_up <= 600:
                for beta in [100, 300, 800]:
                    if remaining() < 35:
                        break
                    try:
                        h_new, c5_new = optimize_slsqp(h_up, n_up, beta, max_iter=300)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            best_n = n_up
                            print("  Upscale n={}, beta={}, C5={:.6f}".format(n_up, beta, best_c5))
                        h_up = h_new
                    except Exception:
                        break
            else:
                for beta in [100, 400]:
                    if remaining() < 30:
                        break
                    try:
                        h_new, c5_new = optimize_lbfgs(h_up, n_up, beta, 3000, max_iter=250)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            best_n = n_up
                            print("  Upscale n={}, beta={}, C5={:.6f}".format(n_up, beta, best_c5))
                        h_up = h_new
                    except Exception:
                        break

    # Phase 4: Perturbation search
    if remaining() > 50:
        print("\n--- Phase 4: Perturbation at n={} ---".format(best_n))
        for trial in range(10):
            if remaining() < 30:
                break
            h_pert = best_h.copy()
            n_pert = max(5, best_n // 8)
            pert_idx = rng.choice(best_n, size=n_pert, replace=False)
            scale = rng.uniform(0.1, 0.35)
            h_pert[pert_idx] = np.clip(h_pert[pert_idx] + rng.uniform(-scale, scale, n_pert), 0, 1)
            h_pert = project_feasible(h_pert, best_n)

            if best_n <= 500:
                try:
                    h_new, c5_new = optimize_slsqp(h_pert, best_n, 300, max_iter=250)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  Perturb {}: C5={:.6f}".format(trial, best_c5))
                except Exception:
                    pass
            else:
                try:
                    h_new, c5_new = optimize_lbfgs(h_pert, best_n, 300, 2000, max_iter=200)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  Perturb {}: C5={:.6f}".format(trial, best_c5))
                except Exception:
                    pass

    # Phase 5: Subgradient polish
    if remaining() > 25:
        print("\n--- Phase 5: Subgradient polish ---")
        sg_time = min(remaining() - 15, 60)
        h_sg, c5_sg = subgradient_polish(best_h, best_n, n_steps=500, time_limit=sg_time)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            print("  Subgrad: C5={:.6f}".format(best_c5))

    # Phase 6: Final SLSQP/SLSQP polish
    if remaining() > 20:
        try:
            if best_n <= 500:
                h_fin, c5_fin = optimize_slsqp(best_h, best_n, 2000, max_iter=300)
            else:
                h_fin, c5_fin = optimize_lbfgs(best_h, best_n, 2000, 8000, max_iter=300)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
                print("  Final polish: C5={:.6f}".format(best_c5))
        except Exception:
            pass

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
