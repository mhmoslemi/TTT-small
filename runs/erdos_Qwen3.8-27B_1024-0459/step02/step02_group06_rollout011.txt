```python
def solve_lp(h_prime, n):
    """Solve the LP: min t s.t. sum_i (1-h'[i+k])*h[i] <= t for all k, 0<=h<=1, sum(h)=n/2"""

    # Objective: minimize t → c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag k (0 to 2n-2, actual lag = k - (n-1)):
    #   sum_i (1 - h'[i+actual_k]) * h[i] - t <= 0
    # (where h'[i+actual_k] is treated as 0 if out of range)

    b_ub = np.zeros(2*n-1)

            if 0 <= j < n:
                A_ub[k, i] = 1.0 - h_prime[j]
            # else: h'[j] is 0 (out of domain), so coefficient is 1
            else:
                A_ub[k, i] = 1.0
        A_ub[k, n] = -1.0  # coefficient of t

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    return result
```

```python
import time
import numpy as np
from scipy.optimize import linprog

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return project_density(p.copy(), 0.5 * n)
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def solve_lp_subproblem(h_prime, n):
    """Solve: min t s.t. sum_i h[i]*(1-h'[i+k]) <= t for all k, sum(h)=n/2, 0<=h<=1"""
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Build A_ub: (2n-1) x (n+1)
    # For each Lag (indexed 0 to 2n-2, actual lag = idx - (n-1)):
    #   sum_i coeff[i] * h[i] - t <= 0
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    
    center = n - 1
    for idx in range(n_lags):
        lag = idx - center
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                A_ub[idx, i] = 1.0 - h_prime[j]
        A_ub[idx, n] = -1.0
    
    b_ub = np.zeros(n_lags)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None
    return res.x[:n].copy()

def lp_optimize(h0, n, deadline, max_iter=1000):
    """LP-based alternating optimization"""
    h = project_density(h0.copy(), 0.5 * n)
    best_h = h.copy()
    best_c = evaluate_erdos(h)
    
    alpha = 1.0  # relaxation parameter
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        h_lp = solve_lp_subproblem(h, n)
        if h_lp is None:
            break
        
        h_lp = project_density(h_lp, 0.5 * n)
        c_lp = evaluate_erdos(h_lp)
        
        if c_lp < best_c - 1e-12:
            h = h_lp if alpha >= 1.0 else project_density((1-alpha)*h + alpha*h_lp, 0.5*n)
            c_new = evaluate_erdos(h)
            if c_new < best_c:
                best_h = h.copy()
                best_c = c_new
            stall = 0
        elif c_lp < best_c + 0.01:
            # Slightly worse or about same - try blending
            if alpha < 1.0:
                h_blend = project_density((1-alpha)*h + alpha*h_lp, 0.5*n)
                c_blend = evaluate_erdos(h_blend)
                if c_blend < best_c:
                    h = h_blend
                    best_h = h.copy()
                    best_c = c_blend
                    stall = 0
                    continue
            # Try smaller alpha
            alpha = max(0.1, alpha * 0.5)
            stall += 1
        else:
            # Much worse - decrease alpha significantly
            alpha = max(0.1, alpha * 0.3)
            stall += 2
        
        if stall > 30:
            # Try restart from best
            h = best_h.copy()
            alpha = 1.0
            stall = 0
            # Slight random perturbation
            pert = np.random.default_rng(it).normal(0, 0.02, n)
            h = project_density(h + pert, 0.5 * n)
    
    return best_h, best_c

def pair_swap_polish(h, deadline, max_iter=5000):
    """Greedy pair swap optimization"""
    n = h.size
    target = 0.5 * n
    best = project_density(h.copy(), target)
    best_c = evaluate_erdos(best)
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        # Compute correlations
        corr = np.correlate(best, 1.0 - best, mode='full')
        center = n - 1
        
        # Find active lags
        top_m = max(5, n // 30)
        top_idx = np.argsort(corr)[-top_m:]
        
        # Compute gradient from top lags
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - center
            if lag == 0:
                g += 1.0 - 2.0 * best
            elif lag > 0:
                g[:n-lag] += (1.0 - best[lag:])
                g[lag:] -= best[:n-lag]
            else:
                l = -lag
                g[l:] += (1.0 - best[:n-l])
                g[:n-l] -= best[l:]
        
        # Try moves: decrease at high-g, increase at low-g
        sorted_idx = np.argsort(g)
        neg_candidates = sorted_idx[:15]  # want to decrease h here
        pos_candidates = sorted_idx[-15:]  # want to increase h here
        
        found = False
        for ii in neg_candidates:
            ii = int(ii)
            for jj in pos_candidates:
                jj = int(jj)
                if ii == jj:
                    continue
                # Transfer mass from jj to ii (decrease jj, increase ii)
                cap = min(best[jj], 1.0 - best[ii])
                if cap < 1e-10:
                    continue
                for d in (0.1*cap, 0.3*cap, 0.5*cap, 0.7*cap, 0.9*cap, cap):
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    if cand[ii] < -1e-12 or cand[jj] < -1e-12:
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-12:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        
        if not found:
            # Random moves
            rng = np.random.default_rng(it * 31 + 7)
            for _ in range(10):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(best[jj], 1.0 - best[ii])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                if cand[ii] < -1e-12 or cand[jj] < -1e-12:
                    continue
                c = evaluate_erdos(cand)
                if c < best_c - 1e-12:
                    best = cand
                    best_c = c
                    found = True
                    break
            if not found:
                break  # stuck
        
        if found and it % 100 == 0:
            best = project_density(best, target)
    
    return project_density(best, target), evaluate_erdos(best)

def gradient_polish(h, deadline, max_iter=200):
    """Projected gradient with line search"""
    n = h.size
    target = 0.5 * n
    best = project_density(h.copy(), target)
    best_c = evaluate_erdos(best)
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr = np.correlate(best, 1.0 - best, mode='full')
        center = n - 1
        top_m = max(5, n // 40)
        top_idx = np.argsort(corr)[-top_m:]
        
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - center
            if lag == 0:
                g += 1.0 - 2.0 * best
            elif lag > 0:
                g[:n-lag] += (1.0 - best[lag:])
                g[lag:] -= best[:n-lag]
            else:
                l = -lag
                g[l:] += (1.0 - best[:n-l])
                g[:n-l] -= best[l:]
        
        # Projected direction: subtract mean to preserve sum
        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            break
        d /= norm
        
        improved = False
        for step in (0.001, 0.003, 0.01, 0.03, 0.1, 0.3):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        
        if not improved:
            break
    
    return project_density(best, target), evaluate_erdos(best)
```

But wait - I should also consider that the LP might not improve much beyond the first few iterations. The method converges when h_LP ≈ h. So in practice, it might converge in 50-200 iterations.

Let me also add a check: if the LP solution is very close to the current h (||h_LP - h|| < 1e-6), stop.

Another concern: the LP gives a solution that minimizes the *linearized* max correlation. But the actual max correlation of the LP solution might be higher. This is because the linearization is only exact at h'. As h moves away from h', the linearization becomes less accurate.

To mitigate this, I can:
1. Use a smaller step (blend with current h)
2. After each LP step, verify the improvement
3. If no improvement, try a smaller blend

def lp_optimize(h0, n, deadline, max_iter=1000):
    h = project_density(h0.copy(), 0.5 * n)
    best_h = h.copy()
    best_c = evaluate_erdos(h)
    
    alpha = 1.0
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        h_lp = solve_lp_subproblem(h, n)
        if h_lp is None:
            break
        
        h_lp = project_density(h_lp, 0.5 * n)
        
        # Check if LP solution is close to current
        if np.max(np.abs(h_lp - h)) < 1e-8:
            break
        
        # Try accepting LP solution
        c_lp = evaluate_erdos(h_lp)
        
        if c_lp < best_c:
            # Accept
            h = h_lp
            best_h = h.copy()
            best_c = c_lp
            alpha = 1.0  # keep full step
            stall = 0
        else:
            # Try blending
            h_blend = project_density((1 - alpha) * h + alpha * h_lp, 0.5 * n)
            c_blend = evaluate_erdos(h_blend)
            if c_blend < best_c:
                h = h_blend
                best_h = h.copy()
