    shifted = corr / mu
    shifted -= np.max(shifted)  # for numerical stability

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return make_feasible(np.interp(x_new, x_old, h_src), n_dst)

def coord_descent_fast(h, n, n_passes, start_time, time_limit):
    """Fast coordinate descent with O(n) per coordinate update."""
    h = h.copy()
    corr = get_corr(h)
    best_c5 = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    
    h_rev = h[::-1]
    
    for p in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        max_change = 0.0
        for j in range(n):
            # Compute coefficient vector
            a = np.zeros(2 * n - 1)
            a[n - 1 - j : 2 * n - 1 - j] += (1.0 - h[:n - j])
            a[j : j + n] += (-h[::-1])
            
            b = corr - h[j] * a
            
            # Binary search for optimal t
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = a * mid + b
                mx = np.max(vals)
                active = vals >= mx - 1e-14
                if np.any(active):
                    sg = np.mean(a[active])
                else:
                    sg = 0.0
                if sg > 1e-14:
                    hi = mid
                elif sg < -1e-14:
                    lo = mid
                else:
                    break
            t_opt = 0.5 * (lo + hi)
            
            delta = t_opt - h[j]
            if abs(delta) > 1e-14:
                h[j] = t_opt
                corr += delta * a
                max_change = max(max_change, abs(delta))
        
        if max_change < 1e-12:
            h = make_feasible(h, n)
            corr = get_corr(h)
            c5 = np.max(corr) * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            break
        
        # Periodic feasibility check
        if p % 10 == 0:
            h = make_feasible(h, n)
            corr = get_corr(h)
            c5 = np.max(corr) * (2.0 / n)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
    
    h = make_feasible(h, n)
    c5 = np.max(get_corr(h)) * (2.0 / n)
    if c5 < best_c5:

        best_h = h.copy()
    
    return best_h, best_c5

def slsqp_polish(h, n, max_iter=3000):
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        k_max = np.argmax(corr)
        e = k_max - (n - 1)
        grad = np.zeros(n)
        # dc_e/dx[j] = (1-x[j+e]) if valid - x[j-e] if valid
        jp = max(0, -e)
        jep = min(n, n - e)
        if jp < jep:
            grad[jp:jep] += (1.0 - x[jp + e:jep + e])
        jm = max(0, e)
        jem = min(n, n + e)
        if jm < jem:
            grad[jm:jem] += (-x[jm - e:jem - e])
        return grad * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.success or res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

    return best_h, best_c5

    h = np.zeros(n)
    for i in range(n):

        f = 1.0

Actually, one more important consideration: the `corr += delta * a` update in coordinate descent. After updating h[j] by delta, the correlation at lag m changes by:
δcorr[m] = a[m] * delta

This is correct because corr[m] is linear in h[j] (when other h's are fixed), and a[m] = ∂corr[m]/∂h[j].

But wait - is corr[m] really linear in h[j]? Let's check:
corr[m] = sum_i h[i] * (1 - h[i+e_m])

The terms involving h[j] are:
- When i = j: h[j] * (1 - h[j+e_m])
- When i+e_m = j (i.e., i = j-e_m): h[j-e_m] * (1 - h[j])

So corr[m] = h[j]*(1-h[j+e_m]) + h[j-e_m]*(1-h[j]) + (other terms not involving h[j])
= h[j] - h[j]*h[j+e_m] + h[j-e_m] - h[j-e_m]*h[j] + other
= h[j]*(1 - h[j+e_m] - h[j-e_m]) + (h[j-e_m] + other)

This IS linear in h[j] (the coefficient is (1 - h[j+e_m] - h[j-e_m]) which doesn't depend on h[j]). ✓

So the update corr += delta * a is exact. 

Now, one subtlety: after updating h[j], the coefficients for OTHER coordinates change (because they depend on h[j]). But we don't precompute all coefficients; we compute them on the fly for each coordinate. So this is fine - each coordinate update uses the current h to compute its coefficients.
