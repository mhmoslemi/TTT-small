def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def slsqp_polish(h, n, max_iter=3000):
    def objective(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argmax(corr)
        k = idx - (n - 1)
        grad = np.zeros(n)
        js1 = max(0, k)
        je1 = min(n, n + k)
        if js1 < je1:
            grad[js1:je1] += (1.0 - x[js1 - k:je1 - k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            grad[js2:je2] += -x[js2 + k:je2 + k]
        return grad
    def sum_fn(x):
        return np.sum(x) - n / 2.0
    def sum_grad(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_grad}]
    try:
        res = minimize(objective, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            curr_c5 = c5_try
            ones_pos = np.where(h_bin == 1)[0]
            zeros_pos = np.where(h_bin == 0)[0]
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
        T *= cooling
    return best_h, best_c5

def generate_structured_inits(n, rng):
    inits = []
    # Periodic patterns
    for period in [4, 6, 8, 10, 12, 16, 20, 24, 32]:
        if n % period == 0:
            freq = (n // period) / 2.0
            if freq >= 1 and freq < period:
                block = np.zeros(period)
                block[:int(freq)] = 1.0
                h = np.tile(block, n // period)
                inits.append(h.copy())
    # Thue-Morse
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    inits.append(make_feasible(tm, n))
    # Quadratic residue-like
    for pp in [7, 11, 13, 17, 19, 23, 29, 31]:
        if n >= pp:
            qr = np.zeros(n)
            for i in range(n):
                if (i * i) % pp < pp // 2:
                    qr[i] = 1.0
            inits.append(make_feasible(qr, n))
    # Random
    for _ in range(3):
        inits.append(make_feasible(rng.rand(n) * 2.5, n))
    # Walsh-like
    wl = np.zeros(n)
    for i in range(n):
        bits = bin(i).count('1')
        wl[i] = 1.0 if (bits * 3) % 5 < 2 else 0.0
    inits.append(make_feasible(wl, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except NameError:
        h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = make_feasible(h_parent.copy(), len(h_parent))
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: CD at parent resolution
    if remaining() > 60:
        tl = min(150, remaining() - 40)
        h_cd, c5_cd = coord_descent(best_h, best_n, max_passes=500, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: Try multiple resolutions with structured inits
    all_resolutions = sorted(set([best_n, 160, 200, 240, 256, 320, 400]))
    for n in all_resolutions:
        if remaining() < 80:
            break
        if n == best_n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_to_n(best_h, n)

        tl_cd = min(100, remaining() - 100)
        if tl_cd < 20:
            continue

        h_cd, c5_cd = coord_descent(h_init, n, max_passes=400, time_limit=tl_cd)

        if remaining() > 80:
            h_sa, c5_sa = simulated_annealing(h_cd, n, max_iters=200000,
                                              time_limit=min(40, remaining() - 60), rng=rng)
            if c5_sa < c5_cd:
                h_cd = h_sa
                c5_cd = c5_sa
                # Re-run CD after SA
                if remaining() > 60:
                    h_cd2, c5_cd2 = coord_descent(h_cd, n, max_passes=200,
                                                   time_limit=min(50, remaining() - 50))
                    if c5_cd2 < c5_cd:
                        h_cd = h_cd2
                        c5_cd = c5_cd2

        if remaining() > 50:
            h_slq, c5_slq = slsqp_polish(h_cd, n, max_iter=2000)
            if h_slq is not None and c5_slq < c5_cd:
                h_cd = h_slq
                c5_cd = c5_slq

        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n

    # Phase 3: Try structured inits at best resolution and nearby
    target_resolutions = [best_n]
    for extra in [best_n - 40, best_n + 40, best_n + 80]:
        if extra % 2 == 0 and extra >= 40:
            target_resolutions.append(extra)
    target_resolutions = sorted(set(target_resolutions))

    for n in target_resolutions:
        if remaining() < 60:
            break
        inits = generate_structured_inits(n, rng)
        for h_init in inits:
            if remaining() < 30:
                break
            tl = min(40, remaining() - 30)
            if tl < 10:
                break
            h_cd, c5_cd = coord_descent(h_init, n, max_passes=200, time_limit=tl)
            if remaining() > 30:
                h_slq, c5_slq = slsqp_polish(h_cd, n, max_iter=1000)
                if h_slq is not None and c5_slq < c5_cd:
                    h_cd = h_slq
                    c5_cd = c5_slq
            if c5_cd < best_c5 - 1e-12:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

    # Phase 4: Binary local search at best resolution
    if remaining() > 30:
        h_bl, c5_bl = binary_local_search(best_h, best_n, max_iters=300000,
                                          time_limit=min(50, remaining() - 20), rng=rng)
        if c5_bl < best_c5 - 1e-12:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            # CD after binary search
            if remaining() > 30:
                h_cd, c5_cd = coord_descent(best_h, best_n, max_passes=200,
                                            time_limit=min(40, remaining() - 20))
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

    # Phase 5: Final SLSQP polish
    if remaining() > 40:
        h_fin, c5_fin = slsqp_polish(best_h, best_n, max_iter=5000)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 6: One more CD round
    if remaining() > 20:
        tl = max(5, remaining() - 10)
        h_cd, c5_cd = coord_descent(best_h, best_n, max_passes=300, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Final SLSQP if time
    if remaining() > 30:
        h_fin, c5_fin = slsqp_polish(best_h, best_n, max_iter=3000)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
