    """Compute all lag values using FFT. O(n log n) instead of O(n^2)."""
    n = len(h)
    # We need: for each lag L, v_L = sum_{i} h[i] * (1-h[i+L]) (with appropriate bounds)
    # This is equivalent to: sum(h) * ones - autocorrelation of h
    # Actually: v_L = sum_{i=0}^{n-1-L} h[i] - sum_{i=0}^{n-1-L} h[i]*h[i+L] for L >= 1
    # v_0 = sum(h) - sum(h^2)
    
    # The first term (sums) can be computed with prefix sums
    # The second term (products) can be computed with FFT (autocorrelation)

    # Using FFT: ac = ifft(fft(h, 2n) * conj(fft(h, 2n)))[:n]

def compute_all_lags(h):
    """Compute all 2n-1 lag values using FFT. O(n log n)."""
    n = len(h)
    if n <= 0:
        return np.array([0.0])
    
    N = 2 * n
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N)[:n]
    # ac[k] = sum_{i=0}^{n-1-k} h[i] * h[i+k] for k=0,...,n-1
    
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    
    # 2n-1 lags
    vals = np.empty(2 * n - 1)
    # lag 0 (index n-1 in correlate output, but let's just index 0 to 2n-2)
    vals[0] = total - ac[0]
    for k in range(1, n):
        vals[k] = pref[n-k] - ac[k]      # positive lag k: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        vals[2*n-1-k] = total - pref[k] - ac[k]  # negative lag k: sum_{i=k}^{n-1} h[i]*(1-h[i-k])
    
    return vals

    vals = compute_all_lags(h)

def smooth_max_and_weights(lags, beta):
    """Compute smooth max and softmax weights of lags."""
    m = float(np.max(lags))
    shifted = beta * (lags - m)
    shifted = np.clip(shifted, -700, 0)
    weights = np.exp(shifted)
    total_w = float(np.sum(weights))
    if total_w < 1e-300:
        # Hard max
        idx = np.where(lags == m)[0]
        weights = np.zeros_like(lags)
        for i in idx:
            weights[i] = 1.0 / len(idx)
        total_w = 1.0
    else:
        weights /= total_w
    smooth_val = m + np.log(total_w) / beta
    return smooth_val, weights

def gradient_smooth_max(h, beta):
    """Compute gradient of smooth max of lag values w.r.t. h."""
    n = len(h)
    dx = 2.0 / float(n)
    
    vals = compute_all_lags(h)
    _, weights = smooth_max_and_weights(vals, beta)
    
    # Gradient computation
    # val[0] = total - ac[0] (lag 0)
    # val[k] for k=1..n-1: positive lag k
    # val[2n-1-k] for k=1..n-1: negative lag k
    
    g = np.zeros(n)
    
    # Lag 0: d/dh_i (total - sum h^2) = 1 - 2*h_i
    g += weights[0] * dx * (1.0 - 2.0 * h)
    
    # For positive lag k (val[k]): v_k = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i]*h[i+k]
    # d/dh_j: (1-h[j+k]) for j in [0, n-1-k], -h[j-k] for j in [k, n-1]
    
    # For negative lag k (val[2n-1-k]): v_{-k} = sum_{i=k}^{n-1} h[i] - sum_{i=0}^{n-1-k} h[i]*h[i+k]
    # d/dh_j: (1-h[j-k]) for j in [k, n-1], -h[j+k] for j in [0, n-1-k]
    
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    
    for k in range(1, n):
        wk = weights[k] * dx
        wm = weights[2*n-1-k] * dx
        
        if wk < 1e-15 and wm < 1e-15:
            continue
        
        m = n - k
        # Positive lag k contribution
        if wk > 0:
            g[:m] += wk * (1.0 - h[k:])  # (1-h[j+k]) for j in [0, m-1]
            g[k:] += -wk * h[:m]           # -h[j-k] for j in [k, n-1]
        
        # Negative lag k contribution
        if wm > 0:
            g[k:] += wm * (1.0 - h[:m])   # (1-h[j-k]) for j in [k, n-1]
            g[:m] += -wm * h[k:]           # -h[j+k] for j in [0, m-1]
    
    return g

def top_k_lags(vals, k):
    """Return indices of top-k lag values."""
    order = np.argsort(-vals)
    return order[:min(k, len(vals))]

def smooth_max_topk(h, beta, topk):
    """Smooth max with focus on top-k lags (higher weight)."""
    n = len(h)
    dx = 2.0 / float(n)
    vals = compute_all_lags(h)
    m = float(np.max(vals))

    shifted = np.clip(shifted, -700, 0)

    if topk < len(vals):
        top_idx = np.argsort(-vals)[:topk]
        weights[top_idx] *= 2.0
    
    total_w = float(np.sum(weights))
    if total_w < 1e-300:
        return m
    return m + np.log(total_w) / beta

def gradient_smooth_max_topk(h, beta, topk):
    """Gradient of smooth max with top-k boost."""
    n = len(h)
    dx = 2.0 / float(n)
    vals = compute_all_lags(h)
    m = float(np.max(vals))
    
    shifted = beta * (vals - m)

    weights = np.exp(shifted)

        top_idx = np.argsort(-vals)[:topk]

    if total_w < 1e-300:
        best_idx = np.argmax(vals)
        weights = np.zeros_like(vals)
        weights[best_idx] = 1.0
        total_w = 1.0
    else:
        weights /= total_w

    g += weights[0] * dx * (1.0 - 2.0 * h)

    pref[0] = 0.0

        wm = weights[2*n-1-k] * dx
        
        if wk < 1e-15 and wm < 1e-15:
            continue

        if wk > 0:

            g[k:] += -wk * h[:m_]
        
        if wm > 0:
            g[k:] += wm * (1.0 - h[:m_])
            g[:m_] += -wm * h[k:]
    
    return g

    L = abs(int(l))
    if L == 0:
        return float(h.sum()) - float(np.dot(h, h))
    m = n - L
    if m <= 0:
        return 0.0
    ac = float(np.dot(h[:m], h[L:]))
    if l >= 0:
        return float(h[:m].sum()) - ac
    return float((h[L:]).sum()) - ac

def lag_grad(h, l):
    n = len(h)
    L = abs(int(l))
    g = np.zeros(n, dtype=float)
    if L == 0:
        return 1.0 - 2.0 * h
    m = n - L
    if m <= 0:
        return g
    if l >= 0:
        g[:m] += 1.0 - h[L:]
        g[L:] += -h[:m]
    else:
        g[L:] += 1.0 - h[:m]
        g[:m] += -h[L:]
    return g

def active_obj(x):
    return float(x[-1])

def active_obj_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j

def active_sum(x):
    return float(x[:-1].sum()) - 0.5 * (len(x) - 1)

def active_sum_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j

def active_ineq(x):
    N = len(x) - 1
    dx = 2.0 / float(N)
    h = x[:-1]
    out = np.empty(len(AC_LAGS), dtype=float)
    for i, ll in enumerate(AC_LAGS):
        out[i] = float(x[-1]) - lag_val(h, ll) * dx
    return out

def active_ineq_jac(x):
    N = len(x) - 1
    dx = 2.0 / float(N)
    m = len(AC_LAGS)
    J = np.zeros((m, N + 1), dtype=float)
    h = x[:-1]
    for i, ll in enumerate(AC_LAGS):
        g = lag_grad(h, ll)
        J[i, :N] = -dx * g
        J[i, N] = 1.0
    return J

def optimize_active(h0, lags, t0, maxiter=80, ftol=1e-13):
    global AC_LAGS
    n = len(h0)
    AC_LAGS = [int(l) for l in lags]
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h0
    x0[n] = float(t0)
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [{"type": "eq", "fun": active_sum, "jac": active_sum_jac}]
    if len(AC_LAGS) > 0:
        cons.append({"type": "ineq", "fun": active_ineq, "jac": active_ineq_jac})
    try:
        res = minimize(
            active_obj, x0, jac=active_obj_jac, method="SLSQP",
            bounds=bounds, constraints=cons,
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False}
        )
        return res
    except Exception:
        return None

def refine_active(h0, n, time_limit, seq=(16, 32, 64, 128), maxit=60):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = c5_val(h)
    for acnt in seq:
        if time.time() > time_limit - 3.0:
            break
        for _ in range(2):
            if time.time() > time_limit - 3.0:
                break
            vals = all_lags_fft(h)
            cur_c = float(vals.max()) * (2.0 / n)
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            order = np.argsort(-vals)
            top_k = min(int(acnt), len(vals))
            idx = order[:top_k]
            lags = []
            for i in idx:
                ii = int(i)
                if ii == 0:
                    lags.append(0)
                elif ii < n:
                    lags.append(ii)
                else:
                    lags.append(ii - (2 * n - 1))
            t0 = cur_c + 1e-5
            res = optimize_active(h, lags, t0, maxiter=maxit, ftol=1e-13)
            if res is not None and res.x is not None and res.x.size == n + 1:
                h2 = project_feasible(res.x[:n], n)
                c2 = c5_val(h2)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()
                if c2 <= cur_c + 1e-8:
                    h = h2
                else:
                    h = best_h
    return best_h, best_c

def interp_to(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return project_feasible(h_src, n_dst)
    if n_src <= 1:
        return project_feasible(np.full(n_dst, h_src[0] if n_src == 1 else 0.5), n_dst)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_src), n_dst)

def make_starts(n, parent, best_h, best_n, rng, count=5):
    n = int(n)
    starts = []
    if parent is not None:
        starts.append(interp_to(parent, n))
    if best_h is not None:
        starts.append(interp_to(best_h, n))
    base = starts[0] if starts else np.full(n, 0.5)

    h = np.full(n, 0.5)
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in range(1, 8):
        amp = rng.uniform(0.05, 0.3)
        ph = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(freq * np.pi * x + ph)
    starts.append(project_feasible(h, n))

    h = np.zeros(n)
    q = n // 4
    h[:q] = 1.0
    h[n // 2:n // 2 + q] = 1.0
    starts.append(project_feasible(h, n))

    if n % 4 == 0:
        pat = np.array([1, 1, 0, 0], dtype=float)
        starts.append(project_feasible(np.tile(pat, n // 4), n))

    h = np.zeros(n)
    parity = np.zeros(n, dtype=int)
    for i in range(1, n):
        parity[i] = 1 - parity[i >> 1]
    h[parity == 0] = 1.0
    starts.append(project_feasible(h, n))

    h = base + rng.randn(n) * 0.08
    starts.append(project_feasible(h, n))

    out = []
    for s in starts:
        out.append(project_feasible(s, n))
    return out[:count]

def local_perturb(h0, n, time_limit, seed, step=0.025):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = c5_val(h)
    cur_h = h.copy()
    cur_c = best_c
    temp = 0.002
    min_temp = 1e-7
    max_trials = 6000
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        L = int(rng.randint(1, min(10, n) + 1))
        st = int(rng.randint(0, max(1, n - L)))
        idx = np.arange(st, min(st + L, n))
        delta = rng.randn(len(idx)) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = c5_val(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.02, max(temp * 1.12, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp * 3:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_end = t0 + max(60.0, budget_s - 35.0)

    parent = get_parent_values()
    if parent is not None:
        pn = int(parent.size)
        ph = project_feasible(parent, pn)
        pc = c5_val(ph)
        raw_c = None
        if (parent.size > 0 and parent.min() >= -1e-12 and parent.max() <= 1.0 + 1e-12
                and abs(parent.sum() - 0.5 * parent.size) <= 1e-8):
            raw_c = c5_val(parent)
        if raw_c is not None and raw_c <= pc + 1e-12:
            best_h = parent.copy()
            best_c = raw_c
        else:
            best_h = ph
            best_c = pc
        best_n = pn
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = c5_val(best_h)

    # Phase 1: Optimize at parent resolution
    if 2 <= best_n <= 200 and time.time() < hard_end - 200:
        for beta in (8, 30, 80, 180):
            if time.time() > hard_end - 200:
                break
            h2, c2 = optimize_smooth(best_h, best_n, beta, 48, maxiter=80, ftol=1e-13)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        if time.time() < hard_end - 180:
            h2, c2 = refine_active(best_h, best_n, hard_end - 170, seq=(16, 32, 64), maxit=50)
            if c2 < best_c:
                best_h = h2
                best_c = c2

    # Phase 2: Multi-resolution search
    target_ns = [192, 224, 256, 288, 320]
    # Also include parent n if not already
    if best_n not in target_ns and 2 <= best_n <= 400:
        target_ns.insert(0, best_n)
    # Sort by proximity to current best (prefer nearby resolutions)
    target_ns = sorted(set(target_ns), key=lambda x: abs(x - best_n))

    for tn in target_ns:
        if time.time() > hard_end - 80:
            break
        if not (32 <= tn <= 400):
            continue
        rem = hard_end - 80 - time.time()
        if rem < 100:
            break
        slot = min(rem, 320.0)
        slot_end = time.time() + slot

        starts = make_starts(tn, parent, best_h, best_n, rng, count=4)
        # Add interpolated best
        starts = [project_feasible(s, tn) for s in starts]

        cur_best_h_tri = best_h if best_n == tn else interp_to(best_h, tn)
        cur_best_c_tri = best_c if best_n == tn else c5_val(cur_best_h_tri)

        for j, h0 in enumerate(starts):
            if time.time() > slot_end - 5:
                break
            sub_rem = slot_end - time.time()
            per = max(20.0, min(120.0, sub_rem / max(1, len(starts) - j)))
            sub_end = time.time() + per

            # Smooth optimization with increasing beta
            for beta in (10, 50, 150):
                if time.time() > sub_end - 3:
                    break
                topk = min(48, 2 * tn - 1)
                h2, c2 = optimize_smooth(h0, tn, beta, topk, maxiter=70, ftol=1e-13)
                if c2 < cur_best_c_tri:
                    cur_best_c_tri = c2
                    cur_best_h_tri = h2
                h0 = h2 if c2 <= cur_best_c_tri + 1e-4 else cur_best_h_tri

            # L-BFGS-style refinement
            if time.time() < sub_end - 5 and 2 <= tn <= 320:
                h2, c2 = optimize_smooth_lbfgs(cur_best_h_tri, tn, 100, 48, maxiter=200)
                if c2 < cur_best_c_tri:
                    cur_best_c_tri = c2
                    cur_best_h_tri = h2

            if cur_best_c_tri < best_c:
                best_c = cur_best_c_tri
                best_h = cur_best_h_tri.copy()
                best_n = tn

    # Phase 3: Active set refinement at best resolution
    if 2 <= best_n <= 350 and time.time() < hard_end - 60:
        seq = (24, 48, 96, 160) if best_n <= 200 else (16, 32, 64, 128)
        h2, c2 = refine_active(best_h, best_n, hard_end - 40, seq=seq, maxit=80)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 4: Final smooth polish
    if time.time() < hard_end - 20 and 2 <= best_n <= 320:
        for beta in (200, 400):
            if time.time() > hard_end - 22:
                break
            h2, c2 = optimize_smooth(best_h, best_n, beta, 32, maxiter=100, ftol=1e-14)
            if c2 < best_c:
                best_h = h2
                best_c = c2

    # Phase 5: Local perturbation
    if time.time() < hard_end - 5 and 2 <= best_n <= 320:
        h2, c2 = local_perturb(best_h, best_n, hard_end - 1.0, seed + 7777, step=0.02)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Final projection and check
    best_h = project_feasible(best_h, best_n)
    best_c = c5_val(best_h)
    return (best_h, float(best_c), int(best_n))
