def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):

        diff = target - s
        if abs(diff) < 1e-14:

        if diff > 0:

            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:

            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def lse_corr_grad(h, n, beta):
    """
    Compute LSE of correlation values and gradient.
    Returns: lse_value (before *dx), gradient
    """
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    exp_corr = np.exp(beta * (corr - cmax))
    sum_exp = np.sum(exp_corr)
    p = exp_corr / sum_exp
    lse_val = cmax + np.log(sum_exp) / beta
    
    grad = np.zeros(n)
    # For each output index j, lag d = j - (n-1)
    # d(c[j])/dh[m] = (1-h[m-d])*1{0<=m-d<n} - h[m+d]*1{0<=m+d<n}
    
    for j in range(2 * n - 1):
        pj = p[j]
        if pj < 1e-14:
            continue
        d = j - (n - 1)
        # Term 1: +pj * (1-h[m-d]) for m in [max(0,d), min(n, d+n))
        lo1 = max(0, d)
        hi1 = min(n, d + n)
        if lo1 < hi1:
            grad[lo1:hi1] += pj * (1.0 - h[lo1-d:hi1-d])
        # Term 2: -pj * h[m+d] for m in [max(0,-d), min(n, n-d))
        lo2 = max(0, -d)
        hi2 = min(n, n - d)
        if lo2 < hi2:
            grad[lo2:hi2] -= pj * h[lo2+d:hi2+d]

                best_h = h.copy()

        h_opt, c5_opt = adam_lse_optimize(best_h, best_n, betas, base_lr=0.003,
                                          max_iters=8000, check_every=30,
                                          time_limit=t_limit, start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: SLSQP polish after Adam
    if time_left() > 150:
        h_opt, c5_opt = slsqq_polish(best_h, best_n, max_iter=5000, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Multi-resolution Adam LSE
    n_candidates = [128, 144, 160, 176, 192, 224, 256, 320]
    for n_try in n_candidates:
        if time_left() < 80:
            break
        if n_try % 2 != 0:
            n_try += 1
        if n_try < 40:
            continue
        h_init = make_feasible(interpolate_to_n(best_h, n_try), n_try)
        t_limit = min(time_left() - 30, 120)
        h_opt, c5_opt = adam_lse_optimize(h_init, n_try, [20, 100, 500, 2000],
                                          base_lr=0.003, max_iters=5000,
                                          check_every=30, time_limit=t_limit,
                                          start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try
            # Quick SLSQP polish at new resolution
            if time_left() > 60:
                h_p, c5_p = slsqq_polish(best_h, best_n, max_iter=3000, ftol=1e-15)
                if h_p is not None and c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p

    # Phase 5: Multi-start at best resolution
    for trial in range(4):
        if time_left() < 60:
            break
        h_try = best_h + rng.randn(best_n) * 0.03
        h_try = make_feasible(h_try, best_n)
        t_limit = min(time_left() - 20, 80)
        h_opt, c5_opt = adam_lse_optimize(h_try, best_n, [50, 200, 1000],
                                          base_lr=0.003, max_iters=4000,
                                          check_every=30, time_limit=t_limit,
                                          start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 6: Structured initializations at various n
    for n_try in [160, 200, 256]:
        if time_left() < 50:
            break
        inits = structured_init(n_try, rng)
        for h_init in inits[:2]:
            if time_left() < 30:
                break
            t_limit = min(time_left() - 15, 50)
            h_opt, c5_opt = adam_lse_optimize(h_init, n_try, [30, 150, 600],
                                              base_lr=0.004, max_iters=3000,
                                              check_every=30, time_limit=t_limit,
                                              start_time=start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 7: Final SLSQP polish
    if time_left() > 40:
        h_opt, c5_opt = slsqq_polish(best_h, best_n, max_iter=8000, ftol=1e-16)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 8: Pair swap local search
    if time_left() > 20:
        n_rounds = min(400, int((time_left() - 5) / 0.1))
        best_h, best_c5 = pair_swap_polish(best_h, best_n, max_rounds=n_rounds)

    # Phase 9: Random perturbation
    if time_left() > 15:
        n_trials = min(10000, int((time_left() - 5) / 0.01))
        best_h, best_c5 = random_perturb_polish(best_h, best_n, n_trials, 0.02, rng)

    # Final: one more Adam pass with high beta
    if time_left() > 30:
        t_limit = min(time_left() - 5, 100)
        h_opt, c5_opt = adam_lse_optimize(best_h, best_n, [5000, 20000, 50000],
                                          base_lr=0.001, max_iters=5000,
                                          check_every=20, time_limit=t_limit,
                                          start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final feasibility and evaluation
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8

    return (best_h, best_c5, best_n)

def np_tile_init(n):
    if n % 4 == 0:
        return np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
    elif n % 2 == 0:
        return np.tile([1.0, 0.0], n // 2)
    else:
        h = np.zeros(n)
        h[::2] = 1.0
        return make_feasible(h, n)
