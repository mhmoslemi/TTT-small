        elapsed = time.time() - start_time
        if elapsed > time_budget:
            break
        if elapsed > time_budget - 0.5:
            break

        T = T_start * (T_end / T_start) ** (elapsed / time_budget)

        # 30% directed moves targeting argmax lag
        if rng.rand() < 0.35:
            corr = np.correlate(arr, 1.0 - arr, mode='full')
            k_idx = int(np.argmax(corr))
            offset = k_idx - (n - 1)

            improved = False
            if offset >= 0:
                # Find positions where arr[i+offset]=1, arr[i]=0
                mask1 = (arr[offset:] == 1) & (arr[:n-offset] == 0)
                candidates = np.where(mask1)[0]
                if len(candidates) > 0:
                    i = int(candidates[rng.randint(len(candidates))])
                    j = i + offset
                    arr[i] = 1.0
                    arr[j] = 0.0
                    new_c5 = compute_c5(arr)
                    delta = new_c5 - curr_c5
                    if delta < 0 or (delta > 1e-15 and rng.rand() < np.exp(-delta / T)):
                        curr_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_arr = arr.copy()
                            iterations_since_improve = 0
                    else:
                        arr[i] = 0.0
                        arr[j] = 1.0
                    improved = True
            else:
                a = -offset
                mask1 = (arr[:n-a] == 1) & (arr[a:] == 0)
                candidates = np.where(mask1)[0]
                if len(candidates) > 0:
                    i = int(candidates[rng.randint(len(candidates))])
                    j = i + a
                    arr[i] = 0.0
                    arr[j] = 1.0
                    new_c5 = compute_c5(arr)
                    delta = new_c5 - curr_c5
                    if delta < 0 or (delta > 1e-15 and rng.rand() < np.exp(-delta / T)):
                        curr_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_arr = arr.copy()
                            iterations_since_improve = 0
                    else:
                        arr[i] = 1.0
                        arr[j] = 0.0
                    improved = True

            if not improved:
                pass  # fall through to random

        # Random swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or arr[i] == arr[j]:
            # Try a few more times
            for _ in range(5):
                j = rng.randint(n)
                if i != j and arr[i] != arr[j]:
                    break
            else:
                continue

        arr[i], arr[j] = arr[j], arr[i]
        new_c5 = compute_c5(arr)
        delta = new_c5 - curr_c5
        if delta < 0 or (delta > 1e-15 and rng.rand() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_arr = arr.copy()
                iterations_since_improve = 0
        else:
            arr[i], arr[j] = arr[j], arr[i]

        iterations_since_improve += 1

        # Kick if stuck
        if iterations_since_improve > N_NO_IMPROVE_KICK:
            iterations_since_improve = 0
            n_kicks = rng.randint(3, 8)
            for _ in range(n_kicks):
                ki = rng.randint(n)
                kj = rng.randint(n)
                if ki != kj and arr[ki] != arr[kj]:
                    arr[ki], arr[kj] = arr[kj], arr[ki]
            curr_c5 = compute_c5(arr)

    return best_arr, best_c5

def subgradient_polish(h, n, time_budget):
    dx = 2.0 / n
    arr = h.copy()
    best_arr = arr.copy()
    best_c5 = compute_c5(arr)
    start_time = time.time()
    step_sizes = [0.02, 0.05, 0.01, 0.1, 0.03, 0.005, 0.08]

    while True:
        elapsed = time.time() - start_time
        if elapsed > time_budget - 1.0:
            break

        corr = np.correlate(arr, 1.0 - arr, mode='full')
        k_idx = int(np.argmax(corr))
        offset = k_idx - (n - 1)

        # Compute gradient of c_offset
        grad = np.zeros(n)
        if offset >= 0:
            nk = n - offset
            grad[offset:] += (1.0 - arr[:nk])
            grad[:nk] -= arr[offset:]
        else:
            a = -offset
            na = n - a
            grad[:na] += (1.0 - arr[a:])
            grad[a:] -= arr[:na]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm

        improved = False
        for step in step_sizes:
            h_new = arr - step * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_arr = h_new.copy()
                arr = h_new.copy()
                improved = True
                break

        if not improved:
            # Try smaller steps
            for step in [0.002, 0.001, 0.003]:
                h_new = arr - step * grad
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_arr = h_new.copy()
                    arr = h_new.copy()
                    improved = True
                    break

    return best_arr, best_c5

def slsqp_refine(h, n, time_budget, beta=500):
    dx = 2.0 / n
    N = n
    TGT = n / 2.0

    def obj_smooth(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = corr - m
        exps = np.exp(beta * shifted)
        return (m + np.log(np.sum(exps)) / beta) * dx

    def obj_raw(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * dx

    def con_sum(x):
        return np.sum(x) - TGT

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]

    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()

    for b in [beta, beta * 4]:
        if time.time() - start_time > time_budget - 2.0:
            break
        try:
            res = minimize(obj_smooth, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 800, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    if time.time() - start_time < time_budget - 2.0:
        try:
            res = minimize(obj_raw, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    _START_TIME = time.time()
    _BUDGET = budget_s
    rng = np.random.RandomState(seed)

    def remaining():
        return max(0.0, _BUDGET - (time.time() - _START_TIME) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get initial solutions
    inits = []
    init_ns = []

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        inits.append(parent_h)
        init_ns.append(parent_n)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    # Thue-Morse at n=160
    n_tm = 160
    h_tm = make_thue_morse(n_tm)
    c5_tm = compute_c5(h_tm)
    inits.append(h_tm)
    init_ns.append(n_tm)
    if c5_tm < best_c5:
        best_c5 = c5_tm
        best_h = h_tm.copy()
        best_n = n_tm
    print("Thue-Morse n=160: C5={:.6f}".format(c5_tm))

    # Random binary at n=160
    h_rand = make_random_binary(160, rng)
    c5_rand = compute_c5(h_rand)
    inits.append(h_rand)
    init_ns.append(160)
    if c5_rand < best_c5:
        best_c5 = c5_rand
        best_h = h_rand.copy()
        best_n = 160
    print("Random n=160: C5={:.6f}".format(c5_rand))

    # Main phase: Binary SA from best initial
    # Allocate ~500s for SA
    sa_budget = min(500.0, remaining() * 0.45)
    if sa_budget > 20:
        print("\n--- Binary SA (budget {:.0f}s) ---".format(sa_budget))
        h_sa, c5_sa = binary_simulated_annealing(best_h, best_n, sa_budget, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = len(h_sa)
        print("After SA: n={}, C5={:.6f}".format(best_n, best_c5))

    # Continuous subgradient polish
    if remaining() > 30:
        sg_budget = min(150.0, remaining() * 0.25)
        if sg_budget > 10:
            print("\n--- Subgradient polish (budget {:.0f}s) ---".format(sg_budget))
            h_sg, c5_sg = subgradient_polish(best_h, best_n, sg_budget)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
            print("After SG: C5={:.6f}".format(best_c5))

    # Round to binary and do another SA round
    if remaining() > 60:
        h_bin = (best_h > 0.5).astype(float)
        # Fix sum
        n_ones = int(np.sum(h_bin))
        target = best_n // 2
        if n_ones < target:
            zeros = np.where(h_bin == 0)[0]
            for i in range(target - n_ones):
                h_bin[zeros[i]] = 1.0
        elif n_ones > target:
            ones = np.where(h_bin == 1)[0]
            for i in range(n_ones - target):
                h_bin[ones[i]] = 0.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        sa_budget2 = min(200.0, remaining() * 0.3)
        if sa_budget2 > 15:
            print("\n--- Binary SA round 2 (budget {:.0f}s) ---".format(sa_budget2))
            h_sa2, c5_sa2 = binary_simulated_annealing(h_bin, best_n, sa_budget2, rng)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
            print("After SA2: C5={:.6f}".format(best_c5))

    # Second subgradient polish
    if remaining() > 20:
        sg_budget2 = min(80.0, remaining() * 0.2)
        if sg_budget2 > 5:
            print("\n--- Subgradient polish 2 ---")
            h_sg2, c5_sg2 = subgradient_polish(best_h, best_n, sg_budget2)
            if c5_sg2 < best_c5:
                best_c5 = c5_sg2
                best_h = h_sg2.copy()
            print("After SG2: C5={:.6f}".format(best_c5))

    # SLSQP final
    if remaining() > 15:
        slsqp_budget = min(60.0, remaining() * 0.3)
        if slsqp_budget > 5:
            print("\n--- SLSQP ---")
            h_sl, c5_sl = slsqp_refine(best_h, best_n, slsqp_budget)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
            print("After SLSQP: C5={:.6f}".format(best_c5))

    # Try a second SA from a perturbed version of best
    if remaining() > 40:
        h_pert = best_h.copy()
        for _ in range(10):
            p = rng.randint(best_n)
            q = rng.randint(best_n)
            if p != q and h_pert[p] != h_pert[q]:
                h_pert[p], h_pert[q] = h_pert[q], h_pert[p]
        h_pert = (h_pert > 0.5).astype(float)
        n_ones = int(np.sum(h_pert))
        target = best_n // 2
        if n_ones < target:
            zeros = np.where(h_pert == 0)[0]
            for i in range(target - n_ones):
                h_pert[zeros[i]] = 1.0
        elif n_ones > target:
            ones = np.where(h_pert == 1)[0]
            for i in range(n_ones - target):
                h_pert[ones[i]] = 0.0

        sa_budget3 = min(60.0, remaining() * 0.4)
        if sa_budget3 > 10:
            print("\n--- Binary SA round 3 (perturbed) ---")
            h_sa3, c5_sa3 = binary_simulated_annealing(h_pert, best_n, sa_budget3, rng)
            if c5_sa3 < best_c5:
                best_c5 = c5_sa3
                best_h = h_sa3.copy()
            print("After SA3: C5={:.6f}".format(best_c5))

    # Final subgradient if time
    if remaining() > 10:
        h_f, c5_f = subgradient_polish(best_h, best_n, min(30.0, remaining() - 2.0))
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        print("After final SG: C5={:.6f}".format(best_c5))

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
