def interpolate_to_n(h_src, n_dst):
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def block_perturb(h, n, rng, block_size=4, time_limit=10):
    """Try random block flips/perturbations and keep improvements."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    n_trials = 0
    while time.time() - start < time_limit and n_trials < 50000:
        n_trials += 1
        # Pick a random block
        start_idx = rng.randint(0, n - block_size)
        idxs = np.arange(start_idx, start_idx + block_size)
        h_try = best_h.copy()
        # Randomly perturb the block
        for idx in idxs:
            h_try[idx] = np.clip(h_try[idx] + rng.uniform(-0.3, 0.3), 0, 1)
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_opt_j(h, j, n, corr):
    ells = np.arange(-(n - 1), n)
    idx1 = j + ells
    idx2 = j - ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        a[valid2] += (-h[idx2[valid2]])
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        sg = np.mean(a[active])
        if sg > 1e-13:
            hi = mid
        elif sg < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, n_passes, time_limit, order='forward'):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        if order == 'forward':
            indices = range(n)
        elif order == 'backward':
            indices = range(n - 1, -1, -1)
        else:
            indices = np.random.permutation(n)
        for j in indices:
            t_opt = coord_opt_j(h, j, n, corr)
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-13:

            best_h = h.copy()

    h_interp = np.interp(x_new, x_old, h_src)

    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10 and res.x is not None:
            h_out = make_feasible(res.x, n)
            c5_out = compute_c5(h_out)
            return h_out, c5_out
    except Exception:
        pass
    return None, None

def perturb_and_optimize(h, n, rng, n_pert_frac, cd_passes, cd_time):
    """Perturb a fraction of coordinates, then re-optimize."""
    n_pert = max(1, int(n * n_pert_frac))
    h_pert = h.copy()
    idxs = rng.choice(n, n_pert, replace=False)
    h_pert[idxs] = np.clip(h_pert[idxs] + rng.uniform(-0.25, 0.25, n_pert), 0, 1)
    h_pert = make_feasible(h_pert, n)
    h_cd, c5_cd = coord_descent(h_pert, n, n_passes=cd_passes, time_limit=cd_time, order='forward')
    return h_cd, c5_cd
