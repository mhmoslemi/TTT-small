def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_val_grad(h, tau):

    corr = np.correlate(h, 1.0 - h, mode='full')

def slsqp_smooth(h0, n, tau, max_iter=500):
    def obj(x):
        v, _ = smooth_val_grad(x, tau)
        return v

    def grad_obj(x):
        _, g = smooth_val_grad(x, tau)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad_obj, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.fun < 1e10:

            return h_out, compute_c5(h_out)

    return None, None

def projected_grad_descent(h0, n, tau_seq, lr0, steps_per_tau, time_limit, seed=0):

        for s in range(steps_per_tau):

                return best_h, best_c5

            h_new = np.clip(h_new, 0.0, 1.0)

            h = h_new
            if s % 200 == 199:

    if c5 < best_c5:

        best_h = h.copy()

def lbbf_smooth(h0, n, tau, max_iter=200):
    def penalty_obj(x, rho):
        corr = np.correlate(x, 1.0 - x, mode='full')
        g = corr * (2.0 / len(x))
        g_max = np.max(g)
        shifted = np.clip(tau * (g - g_max), -500, 500)
        exp_s = np.exp(shifted)
        s_exp = np.sum(exp_s)
        val = g_max + np.log(s_exp) / tau
        s = np.sum(x) - len(x) / 2.0
        val += rho * s * s
        return val

    def penalty_grad(x, rho):
        _, grad = smooth_val_grad(x, tau)
        s = np.sum(x) - len(x) / 2.0
        grad += 2.0 * rho * s * np.ones(len(x))
        return grad

    rho = 1.0
    h = h0.copy()
    for _ in range(15):
        res = minimize(penalty_obj, h, jac=lambda x: penalty_grad(x, rho),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max(10, max_iter // 15), 'ftol': 1e-16})
        h = make_feasible(res.x, n)
        rho *= 10.0
        if rho > 1e8:
            break
    h = make_feasible(h, n)
    return h, compute_c5(h)

    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

                if c5_cd < c5_try:

                    c5_try = c5_cd

            # SLSQP raw
            if remaining() > 40:
                h_sl, c5_sl = slsqp_raw(h_try, n, max_iter=2000)
                if h_sl is not None and c5_sl < c5_try:
                    h_try = h_sl.copy()
                    c5_try = c5_sl

            # Binary SA
            if remaining() > 50:
                tl = min(40, remaining() - 30)
                h_bin, c5_bin = binary_sa(h_try, n, max_steps=30000,
                                          time_limit=tl, seed=seed + n * 10 + init_idx)
                if c5_bin < c5_try:
                    h_try = h_bin.copy()
                    c5_try = c5_bin
                # Continuous refinement from binary
                if c5_bin < c5_try + 0.01 and remaining() > 30:
                    tl2 = min(30, remaining() - 20)
                    h_ref, c5_ref = multi_tau_smooth_optimize(h_bin, n, tl2, seed=seed + 9000 + n)
                    if c5_ref < c5_try:
                        h_try = h_ref.copy()
                        c5_try = c5_ref

            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n

    # Final polish on best
    if remaining() > 30:
        tl = max(10, remaining() - 20)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=50, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if remaining() > 30:
        h_sl, c5_sl = slsqp_smooth(best_h, best_n, 50, max_iter=500)
        if h_sl is not None and c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    if remaining() > 25:
        h_sl, c5_sl = slsqp_raw(best_h, best_n, max_iter=3000)
        if h_sl is not None and c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    if remaining() > 20:
        tl = max(5, remaining() - 10)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=30, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
