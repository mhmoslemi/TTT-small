def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size

    target = 0.5 * float(n)
    h = h[:n].copy()

        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    elif h.size > n:
        h = h[:n]
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0

        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(50):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-14
        else:
            mask = out > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def lag_value(h, l):
    n = len(h)
    if l == 0:
        return float(np.sum(h) - np.dot(h, h))
    L = abs(int(l))
    m = n - L
    ac = float(np.dot(h[:m], h[L:]))
    if l > 0:
        return float(np.sum(h[:m]) - ac)
    return float(np.sum(h[L:]) - ac)


def all_lag_values(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    vals = np.empty(2 * n - 1, dtype=float)
    ac0 = float(np.dot(h, h))
    vals[0] = (total - ac0) * dx
    idx = 1
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals[idx] = (pref[m] - ac) * dx
        idx += 1
        vals[idx] = (total - pref[L] - ac) * dx
        idx += 1
    lags = list(range(n - 1, -1, -1)) + list(range(n - 1, 0, -1))
    return vals, lags

def lag_linear_coeffs(h, l, n):
    coef = np.zeros(n, dtype=float)

        return None
    L = abs(int(l))
    m = n - L

        coef[L:] += -h[:m]
    else:

        coef[:m] += -h[L:]
    return coef

def smooth_max_obj(x, h_aux=None):

    n = h.size

    dx = 2.0 / float(n)

    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    ac0 = float(np.dot(h, h))
    v0 = (total - ac0) * dx
    m = v0
    r = 1.0
    for L in range(1, n):
        mm = n - L
        ac = float(np.dot(h[:mm], h[L:]))
        vp = (pref[mm] - ac) * dx
        vn = (total - pref[L] - ac) * dx
        if vp > m:
            delta = beta * (m - vp)
            r = r * np.exp(delta) if delta > -745.0 else 0.0
            r += 1.0
            m = vp
        else:
            delta = beta * (vp - m)
            r += np.exp(delta) if delta > -745.0 else 0.0
        if vn > m:
            delta = beta * (m - vn)
            r = r * np.exp(delta) if delta > -745.0 else 0.0
            r += 1.0
            m = vn
        else:
            delta = beta * (vn - m)
            r += np.exp(delta) if delta > -745.0 else 0.0

        return float(m)

def smooth_max_jac(x):
    h = np.asarray(x, dtype=float)
    n = h.size
    beta = max(1.0, float(SM_BETA))
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    ac0 = float(np.dot(h, h))
    v0 = (total - ac0) * dx
    posv = np.zeros(n, dtype=float)
    negv = np.zeros(n, dtype=float)
    posv[0] = v0
    negv[0] = v0
    maxv = v0
    for L in range(1, n):
        mm = n - L
        ac = float(np.dot(h[:mm], h[L:]))
        vp = (pref[mm] - ac) * dx
        vn = (total - pref[L] - ac) * dx
        posv[L] = vp
        negv[L] = vn
        if vp > maxv:
            maxv = vp
        if vn > maxv:
            maxv = vn
    wpos = np.zeros(n, dtype=float)
    wneg = np.zeros(n, dtype=float)
    for i in range(n):
        d = beta * (posv[i] - maxv)
        if d > -745.0:
            wpos[i] = np.exp(d)
    for i in range(1, n):
        d = beta * (negv[i] - maxv)
        if d > -745.0:
            wneg[i] = np.exp(d)
    total_w = float(wpos.sum() + wneg[1:].sum())
    g = np.zeros(n, dtype=float)
    if total_w <= 0.0:
        return g
    inv = 1.0 / total_w
    g[:n - 1] += inv * dx * wpos[0:n - 1]
    g[1:] += inv * dx * wneg[1:]
    g[:] -= 2.0 * h * (wpos[0] * inv * dx)
    for L in range(1, n):
        cp = wpos[L] * inv * dx
        if cp > 0.0:
            g[:n - L] += cp * (1.0 - h[L:])
            g[L:] += -cp * h[:n - L]
        cn = wneg[L] * inv * dx
        if cn > 0.0:
            g[L:] += cn * (1.0 - h[:n - L])
            g[:n - L] += -cn * h[L:]
    return g

    J = np.zeros((m, n + 1), dtype=float)
    h = x[:n]
    for i, ll in enumerate(AC_LAGS):
        g = lag_linear_coeffs(h, int(ll), n)
        if g is not None:
            J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def optimize_active_t(h0, active_lags, t0, maxiter=100):
    global AC_LAGS
    h = np.asarray(h0, dtype=float)
    n = h.size
    AC_LAGS = [int(l) for l in active_lags]
    if len(AC_LAGS) == 0:
        return None
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = float(t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    cons = [{"type": "eq", "fun": active_sum_fun, "jac": active_sum_jac},
            {"type": "ineq", "fun": active_ineq_fun, "jac": active_ineq_jac}]
    try:
        res = minimize(active_t_fun, x0, jac=active_t_jac, method="SLSQP",
                       bounds=bounds, constraints=cons,
                       options={"maxiter": maxiter, "ftol": 1e-14, "disp": False})
        return res
    except Exception:
        return None


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def coordinate_refine(h, n, time_limit, max_sweeps=8):
    h = project_feasible(h, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    target_sum = 0.5 * float(n)

    for sweep in range(max_sweeps):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n - 1):
            if time.time() > time_limit + 30:
                break
            for j in range(i + 1, n):
                if time.time() > time_limit:
                    break
                s = h[i] + h[j]
                if s < 1e-12 or s > 1.0 + 1e-12:
                    continue
                lo_t = max(0.0, s - 1.0)
                hi_t = min(1.0, s)
                if hi_t - lo_t < 1e-10:
                    continue
                best_t = h[i]
                best_cv = c
                n_eval = 40
                for k in range(n_eval + 1):
                    t_val = lo_t + (hi_t - lo_t) * k / float(n_eval)
                    h_try = h.copy()
                    h_try[i] = t_val
                    h_try[j] = s - t_val
                    h_try = np.clip(h_try, 0.0, 1.0)
                    s_diff = target_sum - float(np.sum(h_try))
                    if abs(s_diff) > 1e-10:
                        continue
                    cv = compute_c5(h_try)
                    if cv < best_cv - 1e-13:
                        best_cv = cv
                        best_t = t_val
                if best_t != h[i] and best_cv < c - 1e-13:
                    h = h.copy()
                    h[i] = best_t
                    h[j] = s - best_t
                    h = np.clip(h, 0.0, 1.0)
                    h = project_feasible(h, n)
                    c = compute_c5(h)
                    improved = True
                    if c < best_c:
                        best_c = c
                        best_h = h.copy()
        if not improved:
            break
    return best_h, best_c


def make_variants(h, n, rng, count=5):
    base = project_feasible(h, n)
    out = [base.copy()]
    if n > 2:
        rev = base[::-1]
        out.append(project_feasible(0.5 * (base + rev), n))
        out.append(project_feasible(0.5 + 0.3 * (base - rev), n))
    sigma = 2.0 * 2.0 / n
    klen = min(9, n)
    if klen % 2 == 0:
        klen -= 1
    if klen >= 3:
        half = klen // 2
        kk = np.arange(-half, half + 1, dtype=float)
        w = np.exp(-(kk * kk) / (2.0 * sigma * sigma))
        w /= w.sum()
        out.append(project_feasible(np.convolve(base, w, mode="same"), n))
    pert = base + rng.randn(n) * (0.8 * 2.0 / n)
    out.append(project_feasible(pert, n))
    pert2 = base + rng.randn(n) * (2.0 * 2.0 / n)
    out.append(project_feasible(pert2, n))
    if count is not None and len(out) > count:
        out = out[:count]
    return out


def improve_one(h0, n, time_limit, use_lp=True):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    if use_lp and 8 <= n <= 400 and time.time() < time_limit - 5.0:
        active = pick_active_set(h, max_count=100, margin=0.01)
        for lp_round in range(6):
            if time.time() > time_limit - 5.0:
                break
            h2, c2, active = solve_active_lp(h, n, active, max_iter_lp=150)
            h2 = project_feasible(h2, n)
            c2 = compute_c5(h2)
            if c2 < best_c - 1e-13:
                best_c = c2
                best_h = h2.copy()
            h = h2
            active = pick_active_set(h, max_count=120, margin=0.008)

    if time.time() < time_limit - 20.0 and 4 <= n <= 300:
        for beta in (20, 60, 150):
            if time.time() > time_limit - 15.0:
                break
            h2, c2 = optimize_smooth(h, n, beta, maxiter=50)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            h = h2 if c2 <= best_c + 1e-4 else best_h

    if time.time() < time_limit - 10.0 and 4 <= n <= 300:
        active = pick_active_set(best_h, max_count=60, margin=0.005)
        t0 = best_c + 1e-6
        res = optimize_active_t(best_h, active, t0, maxiter=80)
        if res is not None and res.x is not None:
            h2 = project_feasible(res.x[:n], n)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()

    if time.time() < time_limit - 3.0 and 4 <= n <= 200:
        h2, c2 = coordinate_refine(best_h, n, time_limit - 1.0, max_sweeps=4)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 40.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        proj = project_feasible(parent, parent_n)
        raw_c = compute_c5(parent) if (np.min(parent) >= -1e-12 and np.max(parent) <= 1.0 + 1e-12) else np.inf
        proj_c = compute_c5(proj)
        if raw_c <= proj_c + 1e-12:
            best_h = parent.astype(float).copy()
            best_c = raw_c
        else:
            best_h = proj
            best_c = proj_c
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    phase1_end = min(hard_end - 120.0, start_time + 200.0)
    variants0 = make_variants(best_h, best_n, rng, count=5)
    for j, v in enumerate(variants0):
        if time.time() > phase1_end:
            break
        slot = min(45.0, phase1_end - time.time())
        if slot < 10.0:
            break
        hh, cc = improve_one(v, best_n, time.time() + slot, use_lp=True)
        if cc < best_c:
            best_c = cc
            best_h = hh
            best_n = len(hh)

    n_candidates = []
    for nn in (best_n - 16, best_n - 8, best_n + 8, best_n + 16, best_n + 32, 128, 192, 200, 224, 240, 256, 320):
        if 32 <= nn <= 400 and nn != best_n:
            n_candidates.append(nn)
    n_candidates = sorted(set(n_candidates))[:8]

    phase2_end = hard_end - 100.0
    for nn in n_candidates:
        if time.time() > phase2_end:
            break
        slot = min(80.0, phase2_end - time.time())
        if slot < 25.0:
            break
        h_init = interpolate_to_n(best_h, nn)
        hh, cc = improve_one(h_init, nn, time.time() + slot, use_lp=True)
        if cc < best_c:
            best_c = cc
            best_h = hh
            best_n = nn

    if time.time() < hard_end - 60.0:
        for beta in (80, 200, 500):
            if time.time() > hard_end - 55.0:
                break
            h2, c2 = optimize_smooth(best_h, best_n, beta, maxiter=60)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()

    if time.time() < hard_end - 40.0:
        active = pick_active_set(best_h, max_count=80, margin=0.006)
        h2, c2, _ = solve_active_lp(best_h, best_n, active, max_iter_lp=200)
        h2 = project_feasible(h2, best_n)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    if time.time() < hard_end - 35.0:
        active = pick_active_set(best_h, max_count=100, margin=0.004)
        t0 = best_c + 1e-6
        res = optimize_active_t(best_h, active, t0, maxiter=120)
        if res is not None and res.x is not None:
            h2 = project_feasible(res.x[:best_n], best_n)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    if time.time() < hard_end - 15.0 and best_n <= 256:
        h2, c2 = coordinate_refine(best_h, best_n, hard_end - 3.0, max_sweeps=10)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    if time.time() < hard_end - 8.0:
        for _ in range(3):
            if time.time() > hard_end - 6.0:
                break
            pert_r = rng.randn(best_n) * (0.3 * 2.0 / best_n)
            h_try = project_feasible(best_h + pert_r, best_n)
            h2, c2 = improve_one(h_try, best_n, hard_end - 4.0, use_lp=True)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
