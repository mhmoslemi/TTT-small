**Different approach: SPSA (Simultaneous Perturbation Stochastic Approximation)**

SPSA is very efficient for high-dimensional problems because it estimates the gradient with only 2 function evaluations per iteration. With n=450, this is much more efficient than finite differences.

SPSA update:
- Generate random direction Δ (±1 for each coordinate)
- Evaluate f(x + c*Δ) and f(x - c*Δ)  
- Gradient estimate: g = (f(x+cΔ) - f(x-cΔ)) / (2c) * Δ
- Update: x_{new} = project(x - a * g)
- c and a are decaying step sizes

The constraint (sum = n/2) is handled by projection after each step.

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = 2.0 * (target / n - 1.0), 2.0 * (target / n)
    # Binary search for the shift
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-9:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def downsample_h(h, n):
    m = h.size
    if m == n:
        return h.copy()
    if m < n:
        return upsample_h(h, n)
    # Average blocks
    h2 = h[: (m // n) * n].reshape(n, m // n).mean(axis=1)
    return project_density(h2, 0.5 * n)

    center = n - 1

    weights = np.ones(top_k) / top_k  # Equal weight for now

        else:

            if k > 0:
                g[lag:] += w

            else:
                g[:n-lag] += w
                g[:n-lag] -= w * h[lag:]
                g[lag:] -= w * h[:n-lag]

                    g[lag:] -= wi * h[:n-lag]

def project_density(x, target):
    """Project x to [0,1]^n with sum = target"""
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    lo, hi = target / n - 1.0, target / n
    lo = max(lo, -1.5)
    hi = min(hi, 1.5)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-9:
        free = np.where((h > 1e-13) & (h < 1.0 - 1e-13))[0]
        if free.size > 0:

    """Upsample parent to n points via linear interpolation"""
    p = np.asarray(parent, dtype=float)

    vals = corr[idx]
    dv = vals - vals.min()
    w = np.exp(dv / (temp * (dv.max() + 1e-15)))
    w /= w.sum()
    g = np.zeros(n)
    for i, t in enumerate(idx):
        k = int(t) - center
        g += w[i] * corr_gradient(h, k)
    return g, float(np.max(corr)), idx

def spsa_run(h0, deadline, seed=42, c0=0.015, a0=0.004, alpha_p=0.6, gamma_p=0.6, sparsity=0.0, max_iter=20000):
    n = h0.size
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = eval_c5(best_h)
    rng = np.random.default_rng(seed)
    n_pert = max(3, int(n * sparsity)) if sparsity > 0 else n
    
    for k in range(1, max_iter + 1):
        if time.perf_counter() >= deadline:
            break
        c = c0 * (c0 / float(k)) ** alpha_p
        a = a0 / (float(k) + 20.0) ** gamma_p
        
        if n_pert < n:
            delta = np.zeros(n)
            idx = rng.choice(n, size=n_pert, replace=False)
            delta[idx] = rng.choice([-1.0, 1.0], size=n_pert)
        else:
            delta = rng.choice([-1.0, 1.0], size=n)
        
        hp = project_density(h + c * delta, target)
        hm = project_density(h - c * delta, target)
        fp = eval_c5(hp)
        fm = eval_c5(hm)
        
        g = (fp - fm) / (2.0 * c) * delta
        h_new = project_density(h - a * g, target)
        cn = eval_c5(h_new)
        
        if cn < best_c:
            best_h = h_new.copy()
            best_c = cn
        h = h_new
    
    return best_h, best_c

def grad_polish(h0, deadline, max_iter=5000, top_k=10, temp=1.0, min_step=0.0005, max_step=0.3):
    n = h0.size
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = eval_c5(best_h)
    rng = np.random.default_rng(99)
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        g, m_corr, _ = active_grad(h, top_k, temp)
        d = -g
        d -= np.mean(d)
        nr = np.linalg.norm(d)
        if nr < 1e-15:
            break
        d /= nr
        
        improved = False
        lo_s, hi_s = min_step, max_step
        for _ls in range(20):
            step = 0.5 * (lo_s + hi_s)
            cand = project_density(h + step * d, target)
            c = eval_c5(cand)
            if c < best_c - 1e-14:
                h = cand
                best_h = cand.copy()
                best_c = c
                improved = True
                hi_s = step
            else:
                lo_s = step
            if hi_s - lo_s < 1e-6:
                break
        
        if not improved:
            stall += 1
            if stall > 50:
                # Perturb to escape
                pert = rng.normal(0, 0.005, n)
                pert -= np.mean(pert)
                h = project_density(h + pert, target)
            else:
                break
        else:
            stall = 0
            # Adapt temp: increase focus on top lags over time
            temp = max(0.3, temp * 0.999)
    
    return best_h, eval_c5(best_h)

                        h[jj] += d
                    if found:

            for _ in range(20):

                if ii == jj:
                    continue

        else:
            stall += 1
            if stall > 200:
                break
