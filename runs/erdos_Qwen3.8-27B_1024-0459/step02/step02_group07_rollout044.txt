    if d2_lo <= d2_hi:
        d_arr = np.arange(d2_lo, d2_hi+1)
        b[d_arr + (n-1)] -= h[j - d_arr] * (1.0 - h[j])
    
    # Minimize max_m (a[m]*t + b[m]) over t in [0,1]
    lo, hi_t = 0.0, 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi_t)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-13
        sg = np.mean(a[active])
        if sg > 1e-12:
            hi_t = mid
        elif sg < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi_t)

def coord_descent(h, n, n_passes, time_limit, tol=1e-14):
    """Coordinate descent to minimize C5."""
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    start = time.time()
    
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr)
            if abs(t_opt - h[j]) > tol:
                h[j] = t_opt
                improved = True
        # Update corr
        h = np.clip(h, 0.0, 1.0)
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0/n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)
