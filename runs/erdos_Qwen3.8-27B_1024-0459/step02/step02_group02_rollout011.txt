import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(tm, n)

def antisymmetric_tm_init(n):
    half = n // 2
    h = np.zeros(n)
    for i in range(half):
        h[i] = 1.0 if bin(i * 5 + 3).count('1') % 2 == 0 else 0.0
        h[i + half] = 1.0 - h[i]
    return make_feasible(h, n)

def random_binary(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    cnt_ones = len(ones)
    if cnt_ones > target:
        excess = cnt_ones - target
        to_flip = rng.choice(ones, excess, replace=False)
        h[to_flip] = 0.0
    elif cnt_ones < target:
        deficit = target - cnt_ones
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def simulate_annealing(h_init, n, rng, t_start=0.4, t_end=0.0005, n_iter=200000,
                       timeout=90.0, multi_swap_prob=0.08, reheat_every=30000):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    ones_idx = np.where(h == 1.0)[0].copy()
    zeros_idx = np.where(h == 0.0)[0].copy()
    no_improve = 0
    max_no_improve = 25000
    start_t = time.time()

    for it in range(n_iter):
        if time.time() - start_t > timeout:
            break
        if no_improve > max_no_improve:
            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        # Reheating
        if it > 0 and it % reheat_every == 0 and no_improve > reheat_every * 0.5:
            t_start = t_start * 0.6
            no_improve = 0

        frac = min(1.0, it / n_iter)
        temp = max(t_end, t_start * (t_end / t_start) ** frac)

        do_multi = (rng.rand() < multi_swap_prob and len(ones_idx) >= 2 and len(zeros_idx) >= 2)

        if not do_multi:
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or (temp > 1e-14 and rng.rand() < np.exp(-delta / temp)):
                current_c5 = new_c5
                ones_idx = np.where(h == 1.0)[0]
                zeros_idx = np.where(h == 0.0)[0]
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        else:
            i1, i2 = rng.choice(ones_idx, 2, replace=False)
            j1, j2 = rng.choice(zeros_idx, 2, replace=False)
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or (temp > 1e-14 and rng.rand() < np.exp(-delta / temp)):
                current_c5 = new_c5
                ones_idx = np.where(h == 1.0)[0]
                zeros_idx = np.where(h == 0.0)[0]
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
                no_improve += 1

    return best_h, best_c5

def targeted_sa(h_init, n, rng, t_start=0.2, t_end=0.001, n_iter=100000, timeout=60.0):

    zeros_idx = np.where(h == 0.0)[0].copy()

    max_no_improve = 15000

    for it in range(n_iter):

            break

            break
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        frac = min(1.0, it / n_iter)
        temp = max(t_end, t_start * (t_end / t_start) ** frac)

        # 40% chance to target worst shift

            corr = np.correlate(h, 1.0 - h, mode='full')
            worst_k = np.argmax(corr)
            s = worst_k - (n - 1)
            # Find i where h[i]=1 and h[i+s]=0 (contributes to worst)

            for ii in range(max(0, -s), min(n, n - s)):

                i = valid_i[rng.randint(len(valid_i))]

            else:
                i = ones_idx[rng.randint(len(ones_idx))]
                j = zeros_idx[rng.randint(len(zeros_idx))]

            current_c5 = new_c5
            ones_idx = np.where(h == 1.0)[0]

def continuous_polish(h_init, n, max_iter=5000):
    """Polish using smooth max with increasing beta."""
    h0 = make_feasible(h_init.copy(), n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    bounds = [(0.0, 1.0)] * n
    target = n / 2.0

    betas = [50.0, 100.0, 200.0, 400.0, 800.0]
    current = h0.copy()

    for beta in betas:
        def obj(x):
            hx = np.clip(x, 0.0, 1.0)
            corr = np.correlate(hx, 1.0 - hx, mode='full')
            cmax = np.max(corr)
            shifted = beta * (corr - cmax)
            shifted[shifted > 100] = 100
            return (cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))) * (2.0 / n)

        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                        'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, current, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': max_iter // len(betas), 'ftol': 1e-14})
            if res.fun < 1e10:
                candidate = make_feasible(res.x, n)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = candidate.copy()
                    current = candidate.copy()
        except Exception:
            pass

    # Final direct max optimization
    def direct_obj(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        return np.max(corr) * (2.0 / n)

    try:
        res = minimize(direct_obj, current, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': 3000, 'ftol': 1e-15})
        if res.fun < 1e10:
            candidate = make_feasible(res.x, n)
            c5_val = compute_c5(candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = candidate.copy()
    except Exception:
        pass

    return best_h, best_c5

def l_bfgs_polish(h_init, n):
    """Polish using L-BFGS-B with smooth approximation."""
    h0 = make_feasible(h_init.copy(), n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    target = n / 2.0
    dx = 2.0 / n

    for beta in [100.0, 300.0, 1000.0]:
        def obj_grad(x):
            hx = np.clip(x, 0.0, 1.0)
            corr = np.correlate(hx, 1.0 - hx, mode='full')
            cmax = np.max(corr)
            shifted = beta * (corr - cmax)
            shifted[shifted > 80] = 80
            fval = (cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))) * dx
            # Gradient (approximate via the weights)
            weights = np.exp(shifted - np.log(np.sum(np.exp(shifted))))
            # gradient of corr[k] wrt h[j]
            grad = np.zeros(n)
            # For efficiency, just return approximate gradient
            # corr[k] = sum_i h[i]*(1-h[i+k])
            # d(corr[k])/d(h[j]) = (1-h[j+k]) - h[j-k] for valid indices
            for k in range(len(corr)):
                s = k - (n - 1)
                wk = weights[k]
                if abs(wk) < 1e-10:
                    continue
                for ii in range(max(0, -s), min(n, n - s)):
                    jj = ii + s
                    d = 0.0
                    if jj < n:
                        d += (1.0 - hx[jj])
                    if ii - s >= 0:
                        d -= hx[ii - s]
                    grad[ii] += wk * d * dx
            return fval, grad

        try:
            res = minimize(obj_grad, h0.copy(), method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10})
            if res.fun < 1e10:
                candidate = make_feasible(res.x, n)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = candidate.copy()
        except Exception:
            pass

    return best_h, best_c5

def small_perturbation_search(h_init, n, rng, n_trials=50000, timeout=40.0):
    """Try small continuous perturbations around a binary solution."""
    h = make_feasible(h_init.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    start_t = time.time()
    no_improve = 0

    for trial in range(n_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > 8000:
            break

        h_try = h.copy()
        n_mod = rng.randint(1, 5)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-0.3, 0.3)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = target - np.sum(h_try)
        mask = (h_try > 1e-6) & (h_try < 1.0 - 1e-6)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
            diff2 = target - np.sum(h_try)
            if abs(diff2) > 1e-6:
                make_feasible(h_try, n)
                continue
            else:
                h_try = make_feasible(h_try, n)

        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try.copy()
            best_c5 = c5_try
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

                if c5_2cp < best_c5:
                    best_c5 = c5_2cp
                    best_h = h2_cp.copy()
            n = n2

    # Final feasibility check
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
