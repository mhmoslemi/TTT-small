    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

            break

            if not np.any(mask):

            h[mask] = np.minimum(h[mask], 1.0)

                break
            h[mask] += diff / np.count_nonzero(mask)
            h[mask] = np.maximum(h[mask], 0.0)
    return np.clip(h, 0.0, 1.0)

def subgradient_of_max_corr(h, active_idx):
    n = len(h)
    g = np.zeros(n)
    for idx in active_idx:
        m = abs(idx - (n - 1))
        if m == 0:
            g += 1.0 - 2.0 * h
        else:
            g[m:] += 1.0 - h[:n - m]
            g[:n - m] -= h[m:n]
    g /= len(active_idx)
    return g

    h_new = np.clip(h_new, 0.0, 1.0)
