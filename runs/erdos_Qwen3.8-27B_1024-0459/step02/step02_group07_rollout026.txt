def smooth_gradient_descent(h, n, budget_s, start_time):
    """Optimize using smooth max approximation + gradient descent with momentum."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Start with low temperature, increase over time
    tau = 5.0
    tau_max = 200.0
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    
    elapsed = time.time() - start_time
    while elapsed < budget_s:
        # Compute gradient
        g = smooth_max_gradient(h, tau)
        
        # Nesterov momentum
        v = momentum * v + lr * g
        h_new = h - v
        
        # Project to feasible set
        h_new = make_feasible(h_new, n)
        
        # Evaluate
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            # If it got worse, try smaller step
            v *= 0.5
            h = make_feasible(h - 0.5 * v, n)
        
        # Anneal temperature
        tau = min(tau * 1.01, tau_max)
        # Decay learning rate slowly
        lr = max(0.001, lr * 0.999)
        
        elapsed = time.time() - start_time
    
    return best_h, best_c5

            # g(k) is linear in h[j] (for fixed other h values).
            # g(k) = alpha_k * h[j] + beta_k
            # where alpha_k = dx * [(1-h[j+s_k]) - h[j-s_k]] (with appropriate boundary handling)
            # and beta_k = g(k) - alpha_k * h[j] (current value minus the h[j] contribution)
            
            # We want to minimize max_k g(k) = max_k (alpha_k * h[j] + beta_k)
            # This is a 1D minimax problem!
            # The optimal h[j] is where the maximum is minimized.
            
            # For h[j] = t: g(k) = alpha_k * t + beta_k
            # We want min_t max_k (alpha_k * t + beta_k)
            # subject to 0 ≤ t ≤ 1
            
            # This is a 1D convex optimization (min of max of linear functions).
            # Can be solved by bisection on t.
            
            # Compute alpha_k and beta_k
            s_k = ells  # shift for each k
            # alpha_k = dx * [(1-h[j+s_k]) - h[j-s_k]] with boundary handling
            idx_plus = j + s_k  # index j+s_k
            idx_minus = j - s_k  # index j-s_k
            
            valid_plus = (idx_plus >= 0) & (idx_plus < n)
            valid_minus = (idx_minus >= 0) & (idx_minus < n)
            
            alpha = np.zeros(2*n-1)
            alpha[valid_plus] += dx * (1.0 - h[idx_plus[valid_plus]])
            alpha[valid_minus] += -dx * h[idx_minus[valid_minus]]
            
            # beta_k = g(k) - alpha_k * h[j]
            g_curr = get_g(h)
            beta = g_curr - alpha * h[j]
            
            # Binary search for optimal t
            lo, hi = 0.0, 1.0
            for _ in range(60):
                mid = 0.5 * (lo + hi)
                vals = alpha * mid + beta
                mx = np.max(vals)
                active = vals >= mx - 1e-15
                sg = np.mean(alpha[active])
                if sg > 1e-14:
                    hi = mid
                elif sg < -1e-14:
                    lo = mid
                else:
                    break
            h[j] = 0.5 * (lo + hi)
        
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_g(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    g = get_g(h)
    g_max = np.max(g)
    exp_g = np.exp(tau * (g - g_max))
    p = exp_g / np.sum(exp_g)
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    for s in range(-(n - 1), n):
        w = p[s + (n - 1)]
        if w < 1e-16:
            continue
        js1 = max(0, -s)
        je1 = min(n, n - s)
        if js1 < je1:
            term1[js1:je1] += w * h[js1 + s:je1 + s]
        js2 = max(0, s)
        je2 = min(n, n + s)
        if js2 < je2:
            term2[js2:je2] += w * h[js2 - s:je2 - s]
    return dx * ((1.0 - term1) - term2)

def coord_descent(h, n, n_passes, time_limit, start_time):

    best_c5 = compute_c5(h)
    best_h = h.copy()

            alpha = np.zeros(2 * n - 1)
            if np.any(valid_plus):
                alpha[valid_plus] += dx * (1.0 - h[idx_plus[valid_plus]])
            if np.any(valid_minus):

                elif sg < -1e-14:
                    lo = mid
                else:

            h[j] = 0.5 * (lo + hi)
        h = make_feasible(h, n)

def smooth_fd(h, n, time_limit, start_time):

    best_h = h.copy()

    tau = 8.0
    lr = 0.03

def slsqp_polish(h, n, max_iter=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(corr)
        active = corr >= mx - 1e-12
        grad = np.zeros_like(x)
        nn = len(x)
        ells = np.arange(-(nn - 1), nn)
        for k in np.where(active)[0]:
            s = ells[k]
            idx_p = np.arange(nn) + s
            valid_p = (idx_p >= 0) & (idx_p < nn)
            idx_m = np.arange(nn) - s
            valid_m = (idx_m >= 0) & (idx_m < nn)
            grad[valid_p] += (1.0 - x[np.arange(nn)[valid_p] + s])
            grad[valid_m] += -x[np.arange(nn)[valid_m] - s]
        grad /= np.sum(active)
        return grad
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None
