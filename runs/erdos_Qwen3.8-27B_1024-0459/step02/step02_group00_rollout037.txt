import time

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = np.sum(mask)

            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def remaining_time(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time) - 3.0)

def binary_local_search(h, n, start_time, budget_s, max_iters=5000, rng=None):
    """2-flip local search with simulated annealing"""
    if rng is None:
        rng = np.random.RandomState(42)
    
    h = np.array(h, dtype=float).copy()
    # Round to binary, maintaining sum constraint
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    if n_ones > target_ones:
        # Need to flip some 1s to 0s
        ones_idx = np.where(h_bin == 1.0)[0]
        # Flip the ones with lowest "importance" (farthest from needing to be 1)
        rng.shuffle(ones_idx)
        for idx in ones_idx[:n_ones - target_ones]:
            h_bin[idx] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0.0)[0]
        rng.shuffle(zeros_idx)
        for idx in zeros_idx[:target_ones - n_ones]:
            h_bin[idx] = 1.0
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    current_c5 = best_c5
    current_h = h_bin.copy()
    
    # Get list of 1-positions and 0-positions
    ones_set = set(np.where(h_bin == 1.0)[0].tolist())
    zeros_set = set(np.where(h_bin == 0.0)[0].tolist())
    
    temp = 0.001  # initial temperature
    cool_rate = 0.999
    
    for it in range(max_iters):
        if remaining_time(start_time, budget_s) < 2:
            break
        
        # Pick a random 1-position and 0-position to swap
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        
        # Try the swap
        old_h = current_h.copy()
        current_h[i] = 0.0
        current_h[j] = 1.0
        ones_set.remove(i)
        ones_set.add(j)
        zeros_set.remove(j)
        zeros_set.add(i)
        
        new_c5 = compute_c5(current_h)
        
        if new_c5 <= current_c5 or rng.random() < np.exp(-(new_c5 - current_c5) / max(temp, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:

            current_h = old_h
            ones_set.remove(j)
            ones_set.add(i)
            zeros_set.remove(i)
            zeros_set.add(j)
        
        temp *= cool_rate
    
    return best_h, best_c5
