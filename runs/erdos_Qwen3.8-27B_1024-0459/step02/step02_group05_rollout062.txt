import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def lse_objective(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - cmax))))

def slsqp_optimize(n, h_init, beta_list, max_iter=5000, ftol=1e-15):
    """Run SLSQP with log-sum-exp objective at various betas"""
    current = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    bounds = [(0.0, 1.0)] * n
    
    for beta in beta_list:
        def obj(x):
            return lse_objective(x, beta)
        try:
            res = minimize(obj, current, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                                       'jac': lambda x: np.ones(n)}],
                          options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
            if res.x is not None:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5

def project_feasible(h, n):
    """Project h onto {0≤x≤1, sum(x)=n/2}"""
    target = n / 2.0
    # Start with clipping
    x = np.clip(h, 0.0, 1.0)
    # Now fix the sum via bisection on offset
    # want: sum(clamp(x + offset, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def grad_lse(h, beta):
    """Gradient of log-sum-exp of correlations"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    w = np.exp(beta * (corr - cmax))
    w /= np.sum(w)
    
    # Gradient computation
    # For lag m (corr index = n-1+m):
    # c(m) = sum_{i: 0≤i<n, 0≤i+m<n} h[i] * (1-h[i+m])
    # d/dh[j] = (1-h[j+m]) * [0≤j<n, 0≤j+m<n] - h[j-m] * [0≤j-m<n, 0≤j<n, 0≤j<n, (j-m)+m=j≥0]
    # Simplified: (1-h[j+m]) * [j+m < n] - h[j-m] * [j-m ≥ 0, j-m < n]
    
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        w_m = w[n-1+m]
        if w_m < 1e-15:
            continue
        if m >= 0:
            # c(m) = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            # d/dh[j]: 
            #   h[j] term: (1-h[j+m]) for j in [0, n-1-m]
            #   -h[j] term (from -h[i]*h[i+m] where i+m=j): -h[j-m] for j-m in [0, n-1-m], i.e., j in [m, n-1]
            for j in range(0, n - m):
                grad[j] += w_m * (1.0 - h[j + m])
            for j in range(m, n):
                grad[j] += w_m * (-h[j - m])
        else:
            # m < 0, let m' = -m
            m_prime = -m
            # c(-m') = sum_{i=m'}^{n-1} h[i]*(1-h[i-m'])
            # = sum_{i=0}^{n-1-m'} h[i+m']*(1-h[i])
            # d/dh[j]:
            #   h[j] as h[i+m']: (1-h[j-m']) for j-m' in [0, n-1-m'], i.e., j in [m', n-1]
            #   -h[j] as -h[i]: -h[j+m'] for i=j, need j in [0, n-1-m']
            for j in range(m_prime, n):
                grad[j] += w_m * (1.0 - h[j - m_prime])
            for j in range(0, n - m_prime):
                grad[j] += w_m * (-h[j + m_prime])
    
    return grad

Let me think again. For m ≥ 0:
c(m) = ∑_{i=0}^{n-1-m} h[i] · (1-h[i+m])

∂c(m)/∂h[j] = 
  (1-h[j+m]) if 0 ≤ j ≤ n-1-m  [j appears as the first factor]

So in vector form for m ≥ 0:

For m < 0, let p = -m > 0:

def grad_lse_fast(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    w = np.exp(beta * (corr - cmax))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # m goes from -(n-1) to n-1, corresponding to corr indices 0 to 2n-2
    for idx in range(2*n - 1):
        w_m = w[idx]
        if w_m < 1e-12:
            continue
        m = idx - (n - 1)
        if m >= 0:
            # (1-h[m:]) for j=0..n-1-m
            grad[:n-m] += w_m * (1.0 - h[m:])
            # -h[:n-m] for j=m..n-1
            grad[m:] += w_m * (-h[:n-m])
        else:
            p = -m
            # -h[:n-p] for j=0..n-1-p
            grad[:n-p] += w_m * (-h[p:])
            # (1-h[:n-p]) for j=p..n-1
            grad[p:] += w_m * (1.0 - h[:n-p])
    
    return grad

So minimizing C₅ ≈ maximizing min_k ∑ h[i]h[i+k] (the minimum autocorrelation).

This is related to "good" sequences in signal processing (low peak-to-average ratio of autocorrelation). 

For binary sequences (h[i] ∈ {0,1}), this is exactly the problem of finding a binary sequence with good autocorrelation properties. The Welch bound gives a lower bound on the maximum autocorrelation.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def lse_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    # Handle overflow
    if np.max(shifted) > 500:
        shifted = shifted - np.max(shifted)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def project_sum(h, n):
    """Project h (already in [0,1]) so that sum = n/2, using bisection on offset"""
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    # Bisection: find offset such that sum(clip(h+offset, 0, 1)) = target
    lo = -max(np.max(1.0-h), 1e-15)
    hi = -min(np.min(h), -1e-15) if np.min(h) > 0 else max(1.0 - np.max(h), 1e-15)
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_lse(h, beta):
    """Compute gradient of log-sum-exp of correlations"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    if np.max(shifted) > 500:
        shifted = shifted - np.max(shifted)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w_m = w[idx]
        if w_m < 1e-14:
            continue
        m = idx - (n - 1)
        if m >= 0:
            # For j in [0, n-1-m]: grad[j] += w_m * (1-h[j+m])
            # For j in [m, n-1]:   grad[j] += w_m * (-h[j-m])
            if n - m > 0:
                grad[:n-m] += w_m * (1.0 - h[m:])
            if n - m > 0:
                grad[m:] += w_m * (-h[:n-m])
        else:
            p = -m
            if n - p > 0:
                grad[:n-p] += w_m * (-h[p:])
            if n - p > 0:
                grad[p:] += w_m * (1.0 - h[:n-p])
    
    return grad

def gd_step(h, n, beta, lr):
    """One gradient descent step with projection"""
    g = grad_lse(h, beta)
    # Project gradient onto sum=0 subspace
    g = g - np.mean(g)
    h_new = h + lr * (-g)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    vals = np.interp(x_new, x_old, h_src)
    return np.clip(vals, 0.0, 1.0)

def lse_obj(x, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    max_shifted = np.max(shifted)
    if max_shifted > 400:
        shifted = shifted - max_shifted
    return cmax + (1.0 / beta) * (max_shifted + np.log(np.sum(np.exp(shifted))))

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def grad_lse(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    ms = np.max(shifted)
    if ms > 400:
        shifted = shifted - ms
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w_m = w[idx]
        if w_m < 1e-15:
            continue
        m = idx - (n - 1)
        if m >= 0:
            k = n - m
            if k > 0:
                grad[:k] += w_m * (1.0 - h[m:])
                grad[m:] += w_m * (-h[:k])
        else:
            p = -m
            k = n - p
            if k > 0:
                grad[:k] += w_m * (-h[p:])
                grad[p:] += w_m * (1.0 - h[:k])
    return grad

def gd_optimize(h0, n, beta, lr_init, n_iter):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    lr = lr_init
    for _ in range(n_iter):
        g = grad_lse(h, beta)
        g = g - np.mean(g)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_sum(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-18:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.9
        if lr < 1e-12:
            break
    return best_h, best_c5

def slsqp_optimize(n, h_init, beta_list, max_iter=3000, ftol=1e-15):
    current = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    for beta in beta_list:
        def obj(x):
            return lse_obj(x, beta)
        try:
            res = minimize(obj, current, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
            if res.x is not None and np.all(np.isfinite(res.x)):
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5

def coordinate_polish(h, n, n_sweeps=8):
    h = h.copy()
    best_c5 = compute_c5(h)
    target = n / 2.0
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for _ in range(n_sweeps):
        improved = False
        order = np.random.RandomState(int(time.time() * 1e6) % 100000).permutation(n)
        for i in order:
            for st in steps:
                for d in [-1, 1]:
                    h_try = h.copy()
                    h_try[i] += d * st
                    h_try = np.clip(h_try, 0.0, 1.0)
                    diff = target - np.sum(h_try)
                    if abs(diff) > 1e-14:
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask &= (h_try > 0.01) & (h_try < 0.99)
                        cnt = np.count_nonzero(mask)
                        if cnt > 0:
                            h_try[mask] += diff / cnt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-16:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c5
