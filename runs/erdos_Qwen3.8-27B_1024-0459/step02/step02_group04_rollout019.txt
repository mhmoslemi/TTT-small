import numpy as np
import time
from scipy.optimize import minimize, linprog

_MOMENTUM = 0.85
_LP_ITERS = 6
_PG_ITERS = 200


def get_parent_values():
    try:
        vals = initial_h_values
        if vals is None:
            return None
        arr = np.asarray(vals, dtype=float).copy()
        if arr.size == 0:

        return arr
    except Exception:

def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    h = h[:n].copy()
    if h.size < n:
        h = np.concatenate((h, np.zeros(n - h.size)))
    target = 0.5 * float(n)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(30):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-13
        else:
            mask = out > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_lag_values(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    vals = np.empty(2 * n - 1, dtype=float)
    lags = np.empty(2 * n - 1, dtype=int)
    vals[0] = total - float(np.dot(h, h))
    lags[0] = 0
    idx = 1
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals[idx] = pref[m] - ac
        lags[idx] = L
        idx += 1
        vals[idx] = total - pref[L] - ac
        lags[idx] = -L
        idx += 1
    return vals, lags


def lag_gradient(h, L):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        m = n - L
        g[:m] = 1.0 - h[L:]
        g[L:] -= h[:m]
    else:
        La = -L
        m = n - La
        g[La:] = 1.0 - h[:m]
        g[:m] -= h[La:]
    return g


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def smooth_max_beta(h, beta=30.0):
    """Compute smooth max of all lag values scaled by dx."""
    n = h.size
    dx = 2.0 / float(n)
    vals, _ = all_lag_values(h)
    scaled = vals * dx
    m = float(np.max(scaled))
    if m < -700.0 / beta:
        return m
    shifted = scaled - m
    np.clip(shifted * beta, -700.0, 700.0, out=shifted)
    r = float(np.sum(np.exp(shifted * beta)))
    return m + np.log(r) / beta


def smooth_max_grad(h, beta=30.0):
    n = h.size
    dx = 2.0 / float(n)
    vals, lags = all_lag_values(h)
    scaled = vals * dx
    m = float(np.max(scaled))
    shifted = (scaled - m) * beta
    np.clip(shifted, -700.0, 700.0, out=shifted)
    w = np.exp(shifted)
    w = w / float(np.sum(w))
    g = np.zeros(n, dtype=float)
    for i in range(len(lags)):
        if w[i] > 1e-12:
            g += w[i] * lag_gradient(h, int(lags[i])) * dx
    return g


def smooth_obj(x):
    return smooth_max_beta(x, 30.0)


def smooth_obj_grad(x):
    g = smooth_max_grad(x, 30.0)
    return g


def sum_eq(x):
    return float(np.sum(x)) - 0.5 * float(len(x))


def sum_eq_jac(x):
    return np.ones(len(x))


def optimize_smooth_min(h0, n, beta_list=(15, 40, 100, 200), maxiter=50):
    global _MOMENTUM
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac}]

    for beta in beta_list:
        def obj_b(x):
            return smooth_max_beta(x, float(beta))

        def jac_b(x):
            return smooth_max_grad(x, float(beta))

        try:
            res = minimize(
                obj_b, h, jac=jac_b, method="SLSQP",
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-13, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h2 = project_feasible(res.x, n)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2
                if c2 <= best_c + 2e-4:
                    h = h2
                else:
                    h = best_h
        except Exception:
            pass
    return best_h, best_c


def lp_active_step(h, active_lags, t_target, n):
    """Solve LP: min T s.t. linearized lag constraints, bounds, sum."""
    dx = 2.0 / float(n)
    m = len(active_lags)
    if m == 0:
        return h.copy()

    # Build A_ub, b_ub for: g_k · h' - T ≤ g_k · h - v_k (for each active lag)
    # Variables: [h'(n), T(1)]
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, L in enumerate(active_lags):
        g = lag_gradient(h, int(L))
        v, _ = _single_lag_value(h, int(L))
        A_ub[i, :n] = g * dx
        A_ub[i, n] = -1.0
        b_ub[i] = float(np.dot(g * dx, h)) - v * dx

    # Equality: sum(h') = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize T

    res = linprog(
        c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
        bounds=[(0.0, 1.0)] * n + [(0.0, 2.0)],
        method="highs"
    )
    if res.success and res.x is not None:
        h_new = project_feasible(res.x[:n], n)
        return h_new
    return h.copy()


def _single_lag_value(h, L):
    n = len(h)
    if L == 0:
        v = float(np.sum(h) - np.dot(h, h))
    elif L > 0:
        m = n - L
        v = float(np.sum(h[:m]) - np.dot(h[:m], h[L:]))
    else:
        La = -L
        m = n - La
        v = float(np.sum(h[La:]) - np.dot(h[:m], h[La:]))
    return v, L


def lp_active_loop(h, n, time_limit, max_iter=_LP_ITERS, max_active=40):
    """Iterate: identify active lags, solve LP, check improvement."""
    h = project_feasible(h, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    dx = 2.0 / float(n)

    for it in range(max_iter * 3):
        if time.time() > time_limit:
            break
        vals, lags = all_lag_values(h)
        curr_c = float(np.max(vals)) * dx
        if curr_c < best_c:
            best_c = curr_c
            best_h = h.copy()

        # Select active lags (top K by value, plus margin)
        order = np.argsort(-vals)
        top_val = float(vals[order[0]])
        margin = 0.15 * dx  # include lags within 15% of max
        active_idx = order[vals >= top_val - margin]
        if len(active_idx) > max_active:
            active_idx = order[:max_active]
        elif len(active_idx) < 3:
            active_idx = order[:min(3, len(order))]
        active_lags = [int(lags[i]) for i in active_idx]

        # Solve LP
        t_target = curr_c + 0.005
        h_new = lp_active_step(h, active_lags, t_target, n)
        if h_new is None or not np.all(np.isfinite(h_new)):
            break

        c_new = compute_c5(h_new)
        if c_new < curr_c - 1e-12:
            h = h_new
        elif np.allclose(h_new, h, atol=1e-8):
            break
        else:
            # Try smaller step: blend
            blend = project_feasible(0.5 * (h + h_new), n)
            c_blend = compute_c5(blend)
            if c_blend < curr_c:
                h = blend
            else:
                if it > max_iter:
                    break

    return best_h, best_c


def projected_gradient_loop(h, n, time_limit, n_iters=_PG_ITERS, step_init=0.01):
    """Momentum projected gradient on smooth max."""
    h = project_feasible(h, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    vel = np.zeros(n, dtype=float)
    step = step_init
    beta = 40.0

    for it in range(n_iters):
        if time.time() > time_limit:
            break
        g = smooth_max_grad(h, beta)
        vel = _MOMENTUM * vel + (1.0 - _MOMENTUM) * g
        h_try = project_feasible(h - step * vel, n)
        c_try = compute_c5(h_try)
        if c_try < c:
            h = h_try
            c = c_try
            step = min(0.05, step * 1.05)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            step *= 0.7
            if step < 1e-6:
                break
        # Gradually increase beta
        if it % 50 == 49:
            beta *= 1.5
    return best_h, best_c


def gaussian_blur(h, sigma=1.0):
    n = len(h)
    if n <= 3:
        return h.copy()
    half = 3
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * sigma * sigma))
    w /= w.sum()
    out = np.convolve(h, w, mode="same")
    return out


def block_perturb(h, n, time_limit, seed, n_trials=3000, step=0.02):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h.copy()
    cur_c = c
    temp = 0.002
    min_temp = 1e-7

    for _ in range(n_trials):
        if time.time() > time_limit:
            break
        L = int(rng.randint(1, min(6, n) + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.03, max(temp * 1.2, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, best_c


def build_inits(n, parent, best_h, best_n, rng, n_inits=4):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            inits.append(interpolate_to_n(parent, n))
    if best_h is not None:
        if len(best_h) == n:
            inits.append(best_h.copy())
        else:
            inits.append(interpolate_to_n(best_h, n))
    src = inits[0] if inits else np.full(n, 0.5)
    inits.append(project_feasible(gaussian_blur(src, 1.2), n))
    inits.append(project_feasible(0.7 * src + 0.3 * src[::-1], n))
    inits.append(project_feasible(src + rng.randn(n) * 0.03, n))
    out = [project_feasible(h, n) for h in inits[:n_inits]]
    return out


def process_resolution(n, parent, best_h, best_n, rng, seed, time_limit):
    """Run full pipeline for one resolution."""
    inits = build_inits(n, parent, best_h, best_n, rng, n_inits=3)
    best_local_h = None
    best_local_c = 1.0

    per_init = max(10.0, (time_limit - 5.0) / len(inits))
    for j, h0 in enumerate(inits):
        t_end = time.time() + per_init
        if time.time() >= t_end:
            break
        # Phase 1: smooth max
        h1, c1 = optimize_smooth_min(h0, n, beta_list=(15, 50, 150), maxiter=35)
        # Phase 2: LP active
        h2, c2 = lp_active_loop(h1, n, min(t_end - 8.0, time.time() + 40.0))
        # Phase 3: projected gradient
        h3, c3 = projected_gradient_loop(h2, n, min(t_end - 3.0, time.time() + 20.0), n_iters=100)
        # Pick best
        candidates = [(h1, c1), (h2, c2), (h3, c3)]
        best_h_loc = min(candidates, key=lambda x: x[1])
        if best_h_loc[1] < best_local_c:
            best_local_c = best_h_loc[1]
            best_local_h = best_h_loc[0].copy()

    if best_local_h is None:
        best_local_h = project_feasible(inits[0], n)
        best_local_c = compute_c5(best_local_h)
    return best_local_h, best_local_c

        h4, c4 = projected_gradient_loop(best_h, best_n, t3, n_iters=300, step_init=0.005)

        h5, c5 = block_perturb(best_h, best_n, t4, seed + 7777, n_trials=5000, step=0.015)
        if c5 < best_c:
            best_h = h5

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
