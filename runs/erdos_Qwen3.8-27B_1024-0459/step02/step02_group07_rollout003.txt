def project_box_sum(x, target):
    x = np.array(x, dtype=float).flatten()
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if target < 0.0:
        target = 0.0
    if target > n:
        target = float(n)

    lo = float(np.min(x) - 1.0)
    hi = float(np.max(x) + 1.0)
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        p = np.clip(x - mid, 0.0, 1.0)
        s = float(p.sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    p = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(12):
        s = float(p.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - p
            mask = room > 1e-15
            if not np.any(mask):
                break
            idx = int(np.argmax(np.where(mask, room, -1.0)))
            add = min(diff, room[idx])
            p[idx] += add
        else:
            room = p
            mask = room > 1e-15
            if not np.any(mask):
                break
            idx = int(np.argmax(np.where(mask, room, -1.0)))
            rem = min(-diff, room[idx])
            p[idx] -= rem

    if n > 0:
        idx = int(np.argmax(np.minimum(p, 1.0 - p)))
        others = p.copy()
        others[idx] = 0.0
        val = target - float(others.sum())
        if val < 0.0:
            val = 0.0
        elif val > 1.0:
            val = 1.0
        p[idx] = val
    return p

def resample_step(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_src = h_src.size
    n_dst = int(n_dst)
    if n_src == n_dst:

        idx = np.floor(centers / width_old).astype(np.int64)

def binary_c5(x):
    n = len(x)
    if n == 0:
        return 0.0
    nfft = 1
    need = 2 * n - 1
    while nfft < need:
        nfft <<= 1

    fx = np.fft.rfft(x, nfft)
    ac = np.fft.irfft(np.abs(fx) ** 2, nfft)[:n]
    ac = np.rint(ac)
    ac = np.where(ac < 0.0, 0.0, ac)
    ac = np.where(ac > n, float(n), ac)

    P = np.empty(n + 1, dtype=float)
    P[0] = 0.0
    P[1:] = np.cumsum(x)
    e = np.arange(n)
    A = P[n - e] - ac
    B = P[n] - P[e] - ac
    m = float(np.maximum(A.max(), B.max()))
    return (2.0 / float(n)) * m


def binary_best_scan(x, c, ones, zeros, time_limit):

    best_d = 1e9
    bi = -1
    bj = -1

    for i in ones:
        x[i] = 0.0

            cn = binary_c5(x)
            d = cn - c
            if d < best_d:
                best_d = d
                bi = i
                bj = j
            x[j] = 0.0

            if time.time() - t0 > time_limit:
                x[i] = 1.0

                    x[bj] = 1.0
                    c = binary_c5(x)
                    ones.remove(bi)
                    zeros.remove(bj)

        x[bj] = 1.0

def softmax(a):
    a = np.asarray(a, dtype=float)
    b = a - float(np.max(a))
    e = np.exp(b)
    s = float(e.sum())
    if s <= 0.0:
        return np.full(a.shape, 1.0 / max(1, len(a)))
    return e / s


def interval_endpoints(theta, R):
    R = int(R)
    lraw = np.asarray(theta[:R], dtype=float)
    grow = np.asarray(theta[R:], dtype=float)
    l = softmax(lraw)
    g = softmax(grow)

    s = np.zeros(R, dtype=float)
    e = np.zeros(R, dtype=float)
    pos = 0.0
    for i in range(R):
        pos += float(g[i])
        s[i] = pos
        pos += float(l[i])
        e[i] = pos
    return s, e


def interval_q_at_k(s, e, k):
    n = s.size
    if n == 0:
        return 0.0
    if k >= 0.0:
        lo = 0.0
        hi = 2.0 - k
    else:
        lo = -k
        hi = 2.0
    if hi <= lo:
        return 0.0

    base = np.maximum(0.0, np.minimum(e, hi) - np.maximum(s, lo))
    ss = s - k
    ee = e - k
    total = 0.0
    for i in range(n):
        bi = float(base[i])
        if bi <= 0.0:
            continue
        ov = np.sum(np.maximum(0.0, np.minimum(e[i], ee) - np.maximum(s[i], ss)))
        total += bi - float(ov)
    if total < 0.0:
        total = 0.0
    return float(total)


def interval_c5(theta, R):
    s, e = interval_endpoints(theta, R)
    pts = np.concatenate(([0.0, 2.0], s, e))
    diffs = pts[:, None] - pts[None, :]
    mask = (diffs >= -2.0 - 1e-10) & (diffs <= 2.0 + 1e-10)
    kvals = np.unique(np.round(diffs[mask].ravel(), 10))
    kvals = np.concatenate((kvals, [-2.0, -1.0, 0.0, 1.0, 2.0]))
    kvals = np.unique(np.round(kvals, 10))

    best = 0.0
    for kv in kvals:
        q = interval_q_at_k(s, e, float(kv))
        if q > best:
            best = q
    return float(best)


def interval_obj(theta):

def interval_to_binary(s, e, n, rng):

        d1 = np.abs(centers - b)
        d = np.minimum(d0, d1)
        d = np.where(in_now, 0.0, d)
        dist_out = np.minimum(dist_out, d)

        db = np.minimum(centers - a, b - centers)
        db = np.where(in_now, db, np.inf)
        dist_bound = np.minimum(dist_bound, db)

def smooth_c5_lse(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    c5 = corr * _SL_DX
    m = float(np.max(c5))
    shifted = _SL_BETA * (c5 - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    s = float(np.exp(shifted).sum())
    if s <= 0.0:
        return float(m)
    return float(m + np.log(s) / _SL_BETA)


def smooth_gradient(x):
    n = x.size
    corr = np.correlate(x, 1.0 - x, mode="full")
    c5 = corr * _SL_DX
    m = float(np.max(c5))
    shifted = _SL_BETA * (c5 - m)
    shifted = np.clip(shifted, -50.0, 50.0)
    exps = np.exp(shifted)
    z = float(exps.sum())
    if z <= 0.0:
        return np.zeros(n, dtype=float)

    p = exps / z
    L = len(p)
    topk = min(L, int(_SL_TOPK))
    if topk < L:
        idx = np.argpartition(p, -topk)[-topk:]
    else:
        idx = np.nonzero(p > 1e-18)[0]

    grad = np.zeros(n, dtype=float)
    for t in idx:
        wt = float(p[t])
        if wt <= 1e-14:
            continue
        k = int(t) - (n - 1)
        if k > 0:
            rem = n - k
            grad[:rem] += wt * (1.0 - x[k:])
            grad[k:] -= wt * x[:rem]
        elif k < 0:
            ell = -k
            rem = n - ell
            grad[ell:] += wt * (1.0 - x[:rem])
            grad[:rem] -= wt * x[ell:]
        else:
            grad[:] += wt * (1.0 - 2.0 * x)
    return grad * _SL_DX


def mean_eq_func(x):
    return float(np.mean(x) - 0.5)


def smooth_slsqp(h, n, beta, maxiter, topk=200):
    h = project_box_sum(h, 0.5 * n)
    global _SL_N, _SL_DX, _SL_BETA, _SL_TOPK
    _SL_N = int(n)
    _SL_DX = 2.0 / float(n)
    _SL_BETA = float(beta)
    _SL_TOPK = int(topk)

    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": mean_eq_func}]

    try:
        res = minimize(
            smooth_c5_lse,
            h,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            jac=smooth_gradient,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False},
        )
        x = np.asarray(res.x, dtype=float)
        x = project_box_sum(x, 0.5 * n)
        return x, eval_c5(x)
    except Exception:
        return h, eval_c5(h)


def dual_cd(h, n, time_limit, mu=0.0, max_passes=5000, seed=12345):
    n = int(n)
    target = 0.5 * n
    h = project_box_sum(h, target).copy()
    dx = 2.0 / float(n)
    L = 2 * n - 1
    ell = np.arange(-(n - 1), n)

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    best_h = h.copy()
    best_c = float(np.max(corr))

    if time_limit <= 0.0:
        return best_h, eval_c5(best_h)

    rng = np.random.RandomState(int(seed))
    t0 = time.time()
    sum_h = float(h.sum())
    stagnant = 0
    passed = 0

    while time.time() - t0 < time_limit and passed < max_passes:
        order = rng.permutation(n)
        for count, j in enumerate(order):
            if (count & 63) == 63 and time.time() - t0 > time_limit:
                break

            idx1 = j + ell
            valid1 = (idx1 >= 0) & (idx1 < n)
            a = np.zeros(L, dtype=float)
            if np.any(valid1):
                a[valid1] += 1.0 - h[idx1[valid1]]

            idx2 = j - ell
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid2):
                a[valid2] -= h[idx2[valid2]]

            a *= dx
            old = float(h[j])
            b = corr - old * a

            lo = 0.0
            hi = 1.0
            for _ in range(24):
                mid = 0.5 * (lo + hi)
                vals = b + mid * a
                mx = float(np.max(vals))
                active = vals >= (mx - 1e-13)
                slope = mu + float(np.mean(a[active]))
                if slope > 1e-12:
                    hi = mid
                elif slope < -1e-12:
                    lo = mid
                else:
                    break
            tval = 0.5 * (lo + hi)

            if abs(tval - old) > 1e-12:
                h[j] = tval
                sum_h += tval - old
                corr += (tval - old) * a

            drift = (sum_h - target) / float(n)
            mu += 0.5 * drift
            if mu > 1.0:
                mu = 1.0
            elif mu < -1.0:
                mu = -1.0

        if abs(sum_h - target) > 1e-8 * max(1, n):
            h = project_box_sum(h, target)
            mu = 0.0

        sum_h = float(h.sum())
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        c = float(np.max(corr))

        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        passed += 1
        if stagnant > 12 and time.time() - t0 > 20.0:
            break

    best_h = project_box_sum(best_h, target)
    return best_h, eval_c5(best_h)


def polish_candidate(h, n, time_limit, seed=42, try_slqp=True):
    n = int(n)
    h = project_box_sum(h, 0.5 * n)
    bh = h.copy()
    bc = eval_c5(bh)

    if time_limit <= 2.0:
        return bh, bc

    t0 = time.time()

    cd1 = min(45.0, 0.40 * time_limit)
    if cd1 > 5.0:
        h1, c1 = dual_cd(bh, n, cd1, mu=0.0, seed=seed)
        if c1 < bc:
            bh = h1
            bc = c1

    elapsed = time.time() - t0
    left = time_limit - elapsed
    if left > 20.0:
        cd2 = min(30.0, 0.25 * time_limit, left - 5.0)
        if cd2 > 5.0:
            h2, c2 = dual_cd(bh, n, cd2, mu=0.05, seed=seed + 1)
            if c2 < bc:
                bh = h2
                bc = c2

    elapsed = time.time() - t0
    left = time_limit - elapsed
    if try_slqp and n <= 220 and left > 60.0:
        maxiter = 220 if n <= 180 else 110
        h3, c3 = smooth_slsqp(bh, n, beta=80.0, maxiter=maxiter, topk=200)
        if c3 < bc:
            bh = h3
            bc = c3

    elapsed = time.time() - t0
    left = time_limit - elapsed
    if left > 12.0:
        cd3 = min(120.0, left - 2.0)
        if cd3 > 5.0:
            h4, c4 = dual_cd(bh, n, cd3, mu=0.0, seed=seed + 2)
            if c4 < bc:
                bh = h4
                bc = c4

    bh = project_box_sum(bh, 0.5 * n)
    bc = eval_c5(bh)
    return bh, bc


def run(seed=42, budget_s=1300, **kwargs):
    global _SL_N, _SL_DX, _SL_BETA, _SL_TOPK, _INT_R

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    budget_s = float(budget_s)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = int(h_parent.size)
        best_h = project_box_sum(h_parent, 0.5 * best_n)
    else:
        best_n = 160
        best_h = build_fallback_binary(best_n, seed + 7)
        best_h = project_box_sum(best_h, 0.5 * best_n)

    best_c = eval_c5(best_h)
    starts = {}
    starts[best_n] = (best_h.copy(), best_c)

    # Initial local refinement of the parent/current best.
    rem = budget_s - (time.time() - start_time)
    if rem > 150.0:
        tlim = min(90.0, rem - 120.0)
        if tlim > 20.0:
            hp, cp = polish_candidate(best_h, best_n, tlim, seed=1000, try_slqp=True)
            starts[best_n] = (hp.copy(), cp)
            if cp < best_c - 1e-15:
                best_h = hp.copy()
                best_c = float(cp)
                best_n = int(len(hp))
                starts[best_n] = (best_h.copy(), best_c)

    # Interval-based constructions, used as diverse low-dimensional starts.
    rem = budget_s - (time.time() - start_time)
    if rem > 700.0:
        n_int = 320
        if best_n > 240:
            n_int = min(448, max(288, int(round(best_n / 32.0) * 32)))
        if n_int % 2 == 1:
            n_int += 1
        n_int = max(256, min(512, int(n_int)))

        int_best_h = None
        int_best_c = 1.0

        for R in (4, 5, 6):
            rem = budget_s - (time.time() - start_time)
            if rem < 560.0:
                break
            _INT_R = int(R)
            dim = 2 * int(R) + 1
            try:
                bounds = [(-8.0, 8.0)] * dim
                res = differential_evolution(
                    interval_obj,
                    bounds=bounds,
                    maxiter=55,
                    popsize=15,
                    seed=int(seed + 100 * R),
                    tol=1e-5,
                    init="latinhypercube",
                    workers=1,
                    polish=False,
                )
                theta = np.asarray(res.x, dtype=float)
                s, e = interval_endpoints(theta, R)
                x = interval_to_binary(s, e, n_int, rng)
                cb = binary_c5(x)
                if cb < int_best_c:
                    int_best_c = cb
                    int_best_h = x
            except Exception:
                pass

        if int_best_h is not None:
            starts[n_int] = (int_best_h.copy(), float(int_best_c))
            if int_best_c < best_c - 1e-15:
                best_h = int_best_h.copy()
                best_c = float(int_best_c)
                best_n = n_int
                starts[best_n] = (best_h.copy(), best_c)

            rem = budget_s - (time.time() - start_time)
            if rem > 520.0:
                tlim = min(90.0, rem - 460.0)
                if tlim > 20.0:
                    hp, cp = polish_candidate(int_best_h, n_int, tlim, seed=5000, try_slqp=(n_int <= 220))
                    if cp < float(starts.get(n_int, (None, 1.0))[1]):
                        starts[n_int] = (hp.copy(), cp)
                    if cp < best_c - 1e-15:
                        best_h = hp.copy()
                        best_c = float(cp)
                        best_n = n_int
                        starts[best_n] = (best_h.copy(), best_c)

    # Binary search to create a potentially different basin.
    rem = budget_s - (time.time() - start_time)
    if rem > 650.0:
        n_bin = best_n
        if n_bin < 256:
            n_bin = 256
        elif n_bin > 384:
            n_bin = 384
        if n_bin % 2 == 1:
            n_bin += 1

        h_bin0 = continuous_to_binary(resample_step(best_h, n_bin))
        tlim = min(110.0, rem - 560.0)
        if tlim > 20.0:
            xb, cb = binary_local_search(h_bin0, n_bin, tlim, rng)
            if cb < float(starts.get(n_bin, (None, 1.0))[1]):
                starts[n_bin] = (xb.copy(), cb)
            if cb < best_c - 1e-15:
                best_h = xb.copy()
                best_c = float(cb)
                best_n = n_bin
                starts[best_n] = (best_h.copy(), best_c)

            rem = budget_s - (time.time() - start_time)
            if rem > 520.0 and cb < best_c + 0.02:
                tlim = min(80.0, rem - 460.0)
                if tlim > 20.0:
                    hp, cp = polish_candidate(xb, n_bin, tlim, seed=7000, try_slqp=(n_bin <= 220))
                    if cp < float(starts.get(n_bin, (None, 1.0))[1]):
                        starts[n_bin] = (hp.copy(), cp)
                    if cp < best_c - 1e-15:
                        best_h = hp.copy()
                        best_c = float(cp)
                        best_n = n_bin
                        starts[best_n] = (best_h.copy(), best_c)

    # Higher-resolution improvements, initialized from the best known and from special starts.
    rem = budget_s - (time.time() - start_time)
    if rem > 260.0:
        cand_set = set()
        for val in (best_n + 16, best_n + 48, best_n + 96, 256, 320, 448, 512):
            m = int(val)
            if m % 2 == 1:
                m += 1
            if 64 <= m <= 640 and m > best_n:
                cand_set.add(m)
        for nn in list(starts.keys()):
            if int(nn) > best_n and 64 <= int(nn) <= 640:
                cand_set.add(int(nn))

        cand_ns = sorted(cand_set)
        if len(cand_ns) > 4:
            cand_ns = sorted(set(cand_ns[:2] + cand_ns[-2:]))

        done = set()
        for m in cand_ns:
            rem = budget_s - (time.time() - start_time)
            if rem < 220.0:
                break

            rem_c = 0
            for mm in cand_ns:
                if mm >= m and mm not in done:
                    rem_c += 1
            tlim = min(150.0, (rem - 160.0) / max(1, rem_c))
            if tlim < 25.0:
                break

            h_init = resample_step(best_h, m)
            c_init = eval_c5(h_init)
            if m in starts:
                hs, cs = starts[m]
                if float(cs) < c_init:
                    h_init = np.asarray(hs, dtype=float)
                    c_init = float(cs)

            hpol, cpol = polish_candidate(h_init, m, tlim, seed=10000 + int(m), try_slqp=(m <= 220))
            if cpol < float(starts.get(m, (None, 1.0))[1]):
                starts[m] = (hpol.copy(), cpol)
            if cpol < best_c - 1e-15:
                best_h = hpol.copy()
                best_c = float(cpol)
                best_n = m
                starts[best_n] = (best_h.copy(), best_c)
            done.add(m)

    # Final polish.
    rem = budget_s - (time.time() - start_time)
    if rem > 25.0:
        tlim = max(10.0, rem - 8.0)
        hpol, cpol = polish_candidate(best_h, best_n, tlim, seed=99999, try_slqp=(best_n <= 180 and rem > 80.0))
        if cpol < best_c - 1e-15:
            best_h = hpol.copy()
            best_c = float(cpol)
            best_n = int(len(hpol))

    best_h = project_box_sum(best_h, 0.5 * int(best_n))
    best_c = eval_c5(best_h)
    best_n = int(len(best_h))

    return best_h, float(best_c), best_n
