def _evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def _project_density(x, target):

def _grad_corr_lag(h, k, n):
    g = np.zeros(n)

        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:n] += (1.0 - h[0:n - k])
        g[0:n - k] -= h[k:n]
    else:
        L = -k
        g[0:n - L] += (1.0 - h[L:n])
        g[L:n] -= h[0:n - L]
    return g


def _active_gradient(h, top_frac, min_k):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    top_idx = np.argsort(corr)[-k_count:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    scale = max(1e-10, dv.max() + 1e-10)
    w = np.exp(dv / scale)
    w /= w.sum()
    g = np.zeros(n)
    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += w[i] * _grad_corr_lag(h, k, n)
    return g


def _smooth_max_obj(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return m * (2.0 / n)
    return (2.0 / n) * (m + np.log(s) / alpha)


def _smooth_max_grad(h, alpha):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)

    s = float(np.sum(e))

        return np.zeros(n)
    w = e / s
    center = n - 1
    g = np.zeros(n)
    for idx in range(len(corr)):

        if wi < 1e-14:

        k = idx - center

        return _smooth_max_grad(x, alpha)

        return np.ones(n)

                continue
            d = 0.02 * float(rng.random())
            cand = best.copy()
            cand[i1] -= d
            cand[i2] += 0.6 * d
            cand[i3] += 0.4 * d
            if np.any(cand < -1e-12) or np.any(cand > 1.0 + 1e-12):
                continue
            c = _evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                found = True
                break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
    best = _project_density(best, target)
    return best, _evaluate_erdos(best)


def _optimize_resolution(n, h0, deadline):
    target = 0.5 * n
    h = _project_density(h0, target)
    best_h = h.copy()
    best_c = _evaluate_erdos(best_h)

    # Phase 1: SLSQP with progressive alpha
    if n <= 500:
        alphas = [20.0, 60.0, 150.0, 400.0]
        maxiters = [200, 150, 120, 80]
    else:
        alphas = [20.0, 60.0, 120.0]
        maxiters = [120, 100, 70]

    for alpha, mi in zip(alphas, maxiters):
        rem = deadline - time.perf_counter()
        if rem < 10.0:
            break
        cand = _slsqp_optimize(best_h, n, alpha, deadline, maxiter=mi)
        cand = _project_density(cand, target)
        c = _evaluate_erdos(cand)
        if c < best_c:
            best_h = cand
            best_c = c

    # Phase 2: Active gradient descent
    rem = deadline - time.perf_counter()
    if rem > 8.0:
        h2, c2 = _gradient_descent_active(best_h, time.perf_counter() + rem * 0.4, max_iter=400, top_frac=0.04, min_k=10)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 3: Pair swap search
    rem = deadline - time.perf_counter()
    if rem > 8.0:
        h3, c3 = _pair_swap_search(best_h, time.perf_counter() + rem * 0.4, max_iter=2000, top_frac=0.04, min_k=10, n_pos=12, n_neg=12)
        if c3 < best_c:
            best_h = h3
            best_c = c3

    # Phase 4: Binary push + re-optimize
    rem = deadline - time.perf_counter()
    if rem > 6.0 and n <= 600:
        h_bin = _binary_push(best_h, target, thresh=0.08)
        c_bin = _evaluate_erdos(h_bin)
        if c_bin < best_c:
            best_h = h_bin
            best_c = c_bin
        # Re-refine after binary push
        h4, c4 = _pair_swap_search(best_h, time.perf_counter() + min(rem * 0.3, 60.0), max_iter=800, top_frac=0.04, min_k=10, n_pos=8, n_neg=8)
        if c4 < best_c:
            best_h = h4
            best_c = c4

    # Phase 5: Final gradient polish
    rem = deadline - time.perf_counter()
    if rem > 5.0:
        h5, c5 = _gradient_descent_active(best_h, time.perf_counter() + rem * 0.7, max_iter=300, top_frac=0.03, min_k=12)
        if c5 < best_c:
            best_h = h5
            best_c = c5

    best_h = _project_density(best_h, target)
    return best_h, _evaluate_erdos(best_h)


def _make_diverse_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 5 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.empty(n, dtype=float)
        seg = max(1, int(n // 10))
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = float(rng.uniform(0.05, 0.95))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.8, 0.2).astype(float)
        h += rng.normal(0, 0.04, n)
    elif mode == 3:
        freq = int(rng.integers(3, 9))
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * x / 2 + rng.uniform(0, 6.28))
    else:
        # Block structure
        h = np.zeros(n)
        blocks = int(rng.integers(2, 6))
        for b in range(blocks):
            start_idx = int(b * n / blocks)
            end_idx = int((b + 1) * n / blocks)
            h[start_idx:end_idx] = 1.0 if b % 2 == 0 else 0.0
        h += rng.normal(0, 0.03, n)
    return _project_density(h, 0.5 * n)


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = t_start + max(5.0, budget - 12.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        pn = parent.size
        best_h = _project_density(parent, 0.5 * pn)
        best_c = _evaluate_erdos(best_h)
        best_n = pn
    else:
        n0 = 450
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.12 * np.sin(2 * np.pi * 7 * x / 2 + 0.7)
        best_h = _project_density(h_init, 0.5 * n0)
        best_c = _evaluate_erdos(best_h)
        best_n = n0

    base_n = best_n if best_n >= 2 else 450

    # Define resolutions to try
    n_candidates = set()
    n_candidates.add(base_n)
    for extra in [base_n, base_n + 30, base_n + 60, base_n + 90,
                  480, 500, 540, 570, 600, 630, 660, 700]:
        if 40 <= extra <= 800:
            n_candidates.add(extra)

    # Sort: prioritize resolutions close to parent, then increasing
    n_list = sorted(n_candidates, key=lambda x: (abs(x - base_n), x))

    # Time allocation: more time for the best resolution
    total_time = deadline - time.perf_counter()
    # First resolution (parent) gets less, larger ones get more
    weights = []
    for n in n_list:
        if n == base_n:
            weights.append(1.5)
        elif n <= base_n + 60:
            weights.append(2.0)
        elif n <= base_n + 150:
            weights.append(2.5)
        else:
            weights.append(2.0)
    wsum = sum(weights)
    time_alloc = {n_list[i]: max(20.0, total_time * weights[i] / wsum) for i in range(len(n_list))}

    for ni in n_list:
        rem = deadline - time.perf_counter()
        if rem < 20.0:
            break

        alloc = min(time_alloc.get(ni, 60.0), rem - 5.0)
        alloc = max(15.0, alloc)

        # Build starting points
        starts = []
        if best_h.size == ni:
            starts.append(best_h.copy())
        else:
            starts.append(_upsample_h(best_h, ni))

        if parent is not None:
            if parent.size == ni:
                starts.append(parent.copy())
            else:
                starts.append(_upsample_h(parent, ni))

        # Add diverse starts for exploration
        n_rand = 2 if ni <= 500 else 1
        for mm in range(n_rand):
            starts.append(_make_diverse_start(ni, rng, mm % 5))

        # Deduplicate
        uniq = []
        for s in starts:
            s = _project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.linalg.norm(np.asarray(s) - np.asarray(u)) < 1e-5:
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        per_start = alloc / max(1, len(uniq))
        per_start = max(12.0, per_start)

        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 12.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 3.0)
            try:
                h_out, c_out = _optimize_resolution(ni, s, sub_dl)
                if h_out is not None:
                    h_out = _project_density(h_out, 0.5 * ni)
                    c_out = _evaluate_erdos(h_out)
                    if c_out < best_c:
                        best_h = h_out.copy()
                        best_c = c_out
                        best_n = ni
            except Exception:
                pass

    # Final intensive polish on the best solution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = _project_density(best_h, target)

        # Extended pair swap
        if rem > 20.0:
            h2, c2 = _pair_swap_search(h_fin, time.perf_counter() + rem * 0.3,
                                       max_iter=5000, top_frac=0.03, min_k=12, n_pos=15, n_neg=15)
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h2.copy()

        # Triple swap
        rem2 = deadline - time.perf_counter()
        if rem2 > 10.0 and nbest <= 700:
            h3, c3 = _triple_swap(h_fin, time.perf_counter() + rem2 * 0.4, max_iter=500)
            if c3 < best_c:
                h_fin = h3
                best_c = c3
                best_h = h3.copy()

        # Gradient polish
        rem3 = deadline - time.perf_counter()
        if rem3 > 8.0:
            h4, c4 = _gradient_descent_active(h_fin, time.perf_counter() + rem3 * 0.6,
                                              max_iter=600, top_frac=0.03, min_k=14)
            if c4 < best_c:
                h_fin = h4
                best_c = c4
                best_h = h4.copy()

        # Binary push final
        rem4 = deadline - time.perf_counter()
        if rem4 > 8.0 and nbest <= 700:
            h_bin = _binary_push(h_fin, target, thresh=0.06)
            c_bin = _evaluate_erdos(h_bin)
            if c_bin < best_c:
                best_c = c_bin
                best_h = h_bin.copy()
                h_fin = h_bin
            # Quick re-refine
            rem5 = deadline - time.perf_counter()
            if rem5 > 5.0:
                h5, c5 = _pair_swap_search(h_fin, time.perf_counter() + rem5 * 0.7,
                                           max_iter=2000, top_frac=0.03, min_k=12, n_pos=10, n_neg=10)
                if c5 < best_c:
                    best_c = c5
                    best_h = h5.copy()

    # Final SLSQP squeeze on best solution
    rem_final = deadline - time.perf_counter()
    if best_h is not None and rem_final > 15.0 and best_n <= 650:
        nfin = best_h.size
        target = 0.5 * nfin
        h_fin = _project_density(best_h, target)
        for alpha in [200.0, 500.0]:
            rem_f2 = deadline - time.perf_counter()
            if rem_f2 < 8.0:
                break
            cand = _slsqp_optimize(h_fin, nfin, alpha, deadline, maxiter=50)
            cand = _project_density(cand, target)
            c = _evaluate_erdos(cand)
            if c < best_c:
                best_h = cand
                best_c = c
                h_fin = cand

    if best_h is None:
        fallback = _project_density(np.full(45, 0.5), 22.5)
        return fallback, _evaluate_erdos(fallback), 45

    final_h = _project_density(best_h, 0.5 * int(best_h.size))
    final_c = _evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
