Actually, for a given shift s, the gradient is:
grad[m] = (1-h[m+s]) * [0≤m+s<n] - h[m-s] * [0≤m-s<n]

This can be computed as:
grad = (1-h) shifted by -s (with zero padding) - h shifted by s (with zero padding)

```python
def corr_grad_single(h, s, n):
    grad = np.zeros(n)
    i_min = max(0, -s)
    i_max = min(n, n - s)
    valid = np.arange(i_min, i_max)
    grad[valid] += (1.0 - h[valid + s])
    grad[valid + s] -= h[valid]
    return grad
```

    hi = np.max(v) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        if np.sum(h) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta, top_k=40):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    mx = np.max(bc)
    exp_bc = np.exp(bc - mx)
    tot = np.sum(exp_bc)
    w = exp_bc / tot
    val = (mx + np.log(tot)) / beta * dx
    
    top_idx = np.argsort(w)[::-1][:top_k]
    grad = np.zeros(n)
    for ki in top_idx:
        wk = w[ki]
        if wk < 1e-12:
            break
        s = ki - (n - 1)
        i_min = max(0, -s)
        i_max = min(n, n - s)
        valid = np.arange(i_min, i_max)
        grad[valid] += wk * (1.0 - h[valid + s])
        grad[valid + s] -= wk * h[valid]
    grad *= dx
    return grad, val

def gd_smooth(h_init, n, n_iter, beta, lr0, lr_decay):
    h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr0
    for it in range(n_iter):
        g, _ = smooth_max_and_grad(h, beta)
        h = project_feasible(h - lr * g, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= lr_decay
    return best_h, best_c5

def greedy_bin(h_init, n, rng, max_trials, timeout):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    stall = 0
    stall_limit = max(5000, max_trials // 10)
    t0 = time.time()
    for _ in range(max_trials):
        if time.time() - t0 > timeout:
            break
        if stall > stall_limit:
            break
        if not ones or not zeros:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            stall = 0
        else:
            h[i], h[j] = 1.0, 0.0
            stall += 1
    return h, best_c5

def anneal_bin(h_init, n, rng, n_iter, t0_temp, t1_temp, timeout):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h < 0.5)[0])
    t_start = time.time()
    for it in range(n_iter):
        if time.time() - t_start > timeout:
            break
        if not ones or not zeros:
            break
        frac = it / max(n_iter - 1, 1)
        T = t0_temp * (t1_temp / t0_temp) ** frac
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def tm_binary(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.full(n_dst, 0.5)
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)
