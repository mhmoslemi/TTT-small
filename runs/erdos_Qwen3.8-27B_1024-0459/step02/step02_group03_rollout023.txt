def round_to_binary(h, n):
    """Round h to binary with exactly n/2 ones"""
    h_round = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_round))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_round == 1.0)[0]
        vals = h[ones]
        order = np.argsort(vals)
        for idx in order[:n_ones - n_want]:
            h_round[ones[idx]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_round == 0.0)[0]
        vals = h[zeros]
        order = np.argsort(-vals)
        for idx in order[:n_want - n_ones]:
            h_round[zeros[idx]] = 1.0
    return h_round

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(3):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Adjust free variables
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            # All at bounds, adjust the one closest to the needed direction
            if diff > 0:
                idx = np.argmin(h)
                h[idx] += diff
            else:
                idx = np.argmax(h)
                h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):
    """Returns (smooth_max_value, gradient) of the smooth max of correlation * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    val = (m + np.log(total) / beta) * dx
    weights = exps / total
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return val, dx * grad

def slsqp_optimize(h0, n, beta, max_iter, time_limit):
    """SLSQP optimization with smooth max objective"""
    def obj(x):
        v, _ = smooth_max_and_grad(x, beta)
        return v
    
    def jac(x):
        _, g = smooth_max_and_grad(x, beta)
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    
    x = project_feasible(h0, n)
    res = minimize(obj, x, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    x = project_feasible(res.x, n)
    return x, compute_c5(x)

def projected_gradient(h0, n, beta, n_steps, lr, time_limit):
    """Projected gradient descent"""
    h = project_feasible(h0, n)
    for step in range(n_steps):
        if time.time() > time_limit:
            break
        val, grad = smooth_max_and_grad(h, beta)
        h_new = h - lr * grad
        h = project_feasible(h_new, n)
    return h, compute_c5(h)

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)
