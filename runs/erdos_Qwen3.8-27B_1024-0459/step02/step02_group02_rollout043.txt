def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

Let me also think about the **effectiveness of different move types**:
- Single pair swap: (i, j) where h[i]=1, h[j]=0. This is the basic move.
- The "impact" of a swap on the worst shift depends on whether i and j are in the "bad" region for that shift.

For a pair swap (i: 1→0, j: 0→1):
- For shift s where i and i+s are both valid: the term h[i]*(1-h[i+s]) = 1*(1-h[i+s]) changes to 0. If h[i+s]=0, the change is -1 (good for this shift).
- For shift s where j and j+s are both valid: the term h[j]*(1-h[j+s]) = 0 changes to 1*(1-h[j+s]). If h[j+s]=0, the change is +1 (bad).
- Similar for the "reverse" shifts.

So the net effect on shift s is: -(1-h[i+s]) + (1-h[j+s]) + h[i-s] - h[j-s] (with bounds checking).

For the worst shift s*, we want this to be negative (reducing the worst shift).

This gives us a way to **score potential moves** before actually making them, which could speed up the search significantly.

Let me implement this:

```python
def score_swap(h, i, j, n, corr, worst_s):
    """Compute the change in corr at the worst shift if we swap i (1→0) and j (0→1)."""
    delta = 0.0
    # Effect on shift worst_s
    s = worst_s
    # h[i] as first factor: -(1-h[i+s]) if 0 <= i+s < n
    if 0 <= i + s < n:
        delta -= (1.0 - h[i + s])
    # h[i] as second factor: +h[i-s] if 0 <= i-s < n
    if 0 <= i - s < n:
        delta += h[i - s]
    # h[j] as first factor: +(1-h[j+s]) if 0 <= j+s < n
    if 0 <= j + s < n:
        delta += (1.0 - h[j + s])
    # h[j] as second factor: -h[j-s] if 0 <= j-s < n
    if 0 <= j - s < n:
        delta -= h[j - s]
    return delta
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(tm, n)

def random_binary_init(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    target = int(n / 2)
    if len(ones) > target:
        to_flip = rng.choice(ones, len(ones) - target, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        to_flip = rng.choice(zeros, target - len(ones), replace=False)
        h[to_flip] = 1.0
    return h

def greedy_local_search(h_init, n, rng, max_trials=100000, timeout=60.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    stall_limit = max(5000, int(max_trials * 0.08))
    start_t = time.time()
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, best_c5

def targeted_local_search(h_init, n, rng, max_trials=100000, timeout=90.0, pool_size=15):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    no_improve = 0
    stall_limit = max(5000, int(max_trials * 0.1))
    start_t = time.time()
    worst_shifts = np.array([0])
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        if trial % 50 == 0:
            corr = compute_corr(h)
            top_idx = np.argsort(corr)[-3:][::-1]
            worst_shifts = top_idx - (n - 1)
        best_delta = 0.0
        best_i, best_j = -1, -1
        for _ in range(pool_size):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = 0.0
            for s in worst_shifts:
                if 0 <= i + s < n:
                    delta -= (1.0 - h[i + s])
                if 0 <= i - s < n:
                    delta += h[i - s]
                if 0 <= j + s < n:
                    delta += (1.0 - h[j + s])
                if 0 <= j - s < n:
                    delta -= h[j - s]
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
        if best_i < 0:
            best_i = ones[rng.randint(len(ones))]
            best_j = zeros[rng.randint(len(zeros))]
        h[best_i] = 0.0
        h[best_j] = 1.0
        c5_try = compute_c5(h)

            best_c5 = c5_try
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
        else:
            h[best_i] = 1.0
            h[best_j] = 0.0
            no_improve += 1
    return h, best_c5

def simulated_annealing(h_init, n, rng, t_start=0.4, t_end=0.001, max_trials=200000, timeout=120.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    start_t = time.time()
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        T = t_start * (t_end / t_start) ** (trial / max_trials)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        delta = c5_try - current_c5
        if delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            current_c5 = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def iterated_local_search(h_init, n, rng, max_rounds=8, ls_trials=30000, shake_size=4, timeout=120.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_t = time.time()

        elapsed = time.time() - start_t

            break

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        h = h_cur.copy()
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(shake_size):
            if len(ones) == 0 or len(zeros) == 0:

            h[i] = 0.0
            h[j] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
    return best_h, best_c5

def _N_OPT = 0

def _sum_con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _sum_con_jac(x):
    return np.ones(_N_OPT)

def _smooth_max_obj(x, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted[shifted > 50] = 50
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val

def _direct_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def slsqp_smooth_optimize(h_init, n, beta_start=20.0, beta_end=300.0, n_steps=5, max_iter=500, timeout=60.0):
    global _N_OPT
    _N_OPT = n
    current = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(current)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    start_t = time.time()
    for idx in range(n_steps):
        if time.time() - start_t > timeout:
            break
        t = idx / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        try:
            res = minimize(_smooth_max_obj, current, args=(beta,), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.fun < 1e10:
                candidate = make_feasible(res.x, n)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    current = candidate
        except Exception:
            pass
    return current, best_c5

def slsqp_direct_polish(h_init, n, max_iter=2000, timeout=30.0):
    global _N_OPT
    _N_OPT = n
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    start_t = time.time()
    try:
        res = minimize(_direct_obj, h0, method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        if time.time() - start_t < timeout and res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            c5_val = compute_c5(h_out)
            if c5_val < compute_c5(h0):
                return h_out, c5_val
    except Exception:
        pass
    return h_init, compute_c5(h_init)
