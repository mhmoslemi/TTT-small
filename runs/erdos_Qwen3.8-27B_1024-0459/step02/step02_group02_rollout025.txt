    shifted = beta * (corr - cmax)
    shifted[shifted > 50] = 50
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val

def continuous_refine(h_init, n, time_budget=60.0):
    h0 = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h0)
    start_t = time.time()
    bounds = [(0.0, 1.0)] * n

    for beta in [20, 50, 100, 200, 400]:
        if time.time() - start_t > time_budget - 10:
            break
        try:
            res = minimize(smooth_max_obj, h0, args=(beta,), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-16})
            if res.fun < 1e10:
                h_try = make_feasible(np.clip(res.x, 0, 1), n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    h0 = h_try
        except Exception:
            pass

    return h0, best_c5

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    n = 256
    best_c5 = 0.5
    best_h = thue_morse_n(n)
    best_c5 = compute_c5(best_h)

    # Get parent if available
    h_parent = None
    try:
        h_p_arr = np.array(initial_h_values, dtype=float)
        if h_p_arr is not None and len(h_p_arr) > 0:
            h_parent = h_p_arr.copy()
            h_p_interp = make_feasible(interp_h(h_parent, n), n)
            c5_p = compute_c5(h_p_interp)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p_interp.copy()
    except Exception:
        h_parent = None

    # Generate diverse initials
    inits = [best_h.copy(), thue_morse_n(n)]
    for s in [1, 2, 3, 5, 7]:
        inits.append(thue_morse_shifted(n, s))
    for _ in range(3):
        inits.append(random_binary_n(n, rng))
    if h_parent is not None:
        inits.append(make_feasible(interp_h(h_parent, n), n))

    # Evaluate all initials
    for h0 in inits:
        c5_v = compute_c5(h0)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h0.copy()

    # Phase 1: Simulated annealing from best initial (large budget)
    if remaining() > 200:
        sa_budget = min(250.0, remaining() * 0.35)
        h_sa, mc_sa = sa_binary(best_h, n, rng, time_budget=sa_budget, T_start=0.04, T_end=0.0003)
        c5_sa = mc_sa * 2.0 / n
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 2: SA from other initials
    for h0 in inits[1:5]:
        if remaining() < 80:
            break
        sa_b = min(80.0, remaining() * 0.15)
        h_sa2, mc_sa2 = sa_binary(h0, n, rng, time_budget=sa_b, T_start=0.03, T_end=0.0005)
        c5_sa2 = mc_sa2 * 2.0 / n
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    # Phase 3: Targeted SA from best
    if remaining() > 80:
        tsa_b = min(100.0, remaining() * 0.2)
        h_tsa, mc_tsa = targeted_sa_binary(best_h, n, rng, time_budget=tsa_b)
        c5_tsa = mc_tsa * 2.0 / n
        if c5_tsa < best_c5:
            best_c5 = c5_tsa
            best_h = h_tsa.copy()

    # Phase 4: Greedy 2-swap
    if remaining() > 60:
        g2_b = min(80.0, remaining() * 0.15)
        h_g2, mc_g2 = greedy_2swap(best_h, n, rng, time_budget=g2_b)
        c5_g2 = mc_g2 * 2.0 / n
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    # Phase 5: More SA rounds with restarts from perturbed best
    if remaining() > 100:
        for restart in range(4):
            if remaining() < 40:
                break
            h_pert = best_h.copy()
            k = rng.randint(3, 8)
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert < 0.5)[0]
            f0 = rng.choice(ones, min(k, len(ones)), replace=False)
            f1 = rng.choice(zeros, min(k, len(zeros)), replace=False)
            h_pert[f0] = 0.0
            h_pert[f1] = 1.0

            sa_b = min(60.0, remaining() * 0.15)
            h_r, mc_r = sa_binary(h_pert, n, rng, time_budget=sa_b, T_start=0.05, T_end=0.001)
            c5_r = mc_r * 2.0 / n
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Inpainting refinement
    if remaining() > 60:
        inp_b = min(60.0, remaining() * 0.2)
        h_inp, mc_inp = inpainting_refine(best_h, n, rng, time_budget=inp_b)
        c5_inp = mc_inp * 2.0 / n
        if c5_inp < best_c5:
            best_c5 = c5_inp
            best_h = h_inp.copy()

    # Phase 7: SA one more from best (lower temperature for fine tuning)
    if remaining() > 50:
        sa_b = min(60.0, remaining() * 0.4)
        h_fin, mc_fin = sa_binary(best_h, n, rng, time_budget=sa_b, T_start=0.01, T_end=0.0001)
        c5_fin = mc_fin * 2.0 / n
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 8: Continuous refinement (SLSQP / L-BFGS-B with smooth max)
    if remaining() > 40:
        cont_b = min(50.0, remaining() * 0.5)
        h_cont, c5_cont = continuous_refine(best_h, n, time_budget=cont_b)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 9: Final targeted SA on the best
    if remaining() > 30:
        tsa_b = min(40.0, remaining() * 0.7)
        h_final_sa, mc_final = targeted_sa_binary(best_h, n, rng, time_budget=tsa_b)
        c5_final = mc_final * 2.0 / n
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_sa.copy()

    # Phase 10: Tiny continuous polish
    if remaining() > 15:
        cont_b2 = min(12.0, remaining() * 0.7)
        h_polish, c5_pol = continuous_refine(best_h, n, time_budget=cont_b2)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_polish.copy()

    # Final: ensure feasibility
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
