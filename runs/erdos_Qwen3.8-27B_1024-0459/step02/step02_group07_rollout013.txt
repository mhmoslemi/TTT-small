        best_h = make_feasible(h_parent.copy(), len(h_parent))
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive coordinate descent at parent resolution
    if remaining() > 80:
        tl = min(180, remaining() - 70)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=2000, time_limit=tl, check_interval=1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 2: SLSQP polish at parent resolution
    if remaining() > 60:
        h_slq, c5_slq = slsqp_smooth(best_h, best_n, tau=1000.0, max_iter=8000)
        if h_slq is not None and c5_slq < best_c5 - 1e-14:
            best_c5 = c5_slq
            best_h = h_slq
        # Follow up with CD
        if remaining() > 40:
            h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=500, time_limit=min(40, remaining() - 30), check_interval=1)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Phase 3: Multi-resolution optimization
    resolutions = sorted(set([best_n, 128, 160, 200, 240, 280, 320, 400]))
    for n in resolutions:
        if remaining() < 60:
            break
        if n == best_n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_to_n(best_h, n)
        if len(h_init) != n:
            continue
        
        tl = min(60, remaining() - 50)
        if tl < 20:
            continue
        h_cd, c5_cd = coord_descent(h_init, n, n_passes=800, time_limit=tl, check_interval=1)
        
        if remaining() > 50:
            h_slq, c5_slq = slsqp_smooth(h_cd, n, tau=500.0, max_iter=3000)
            if h_slq is not None and c5_slq < c5_cd - 1e-14:
                h_cd = h_slq
                c5_cd = c5_slq
        
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd
            best_n = n

    # Phase 4: Final CD at best resolution
    if remaining() > 40:
        tl = min(80, remaining() - 30)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=3000, time_limit=tl, check_interval=1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP smooth polish
    if remaining() > 40:
        h_slq, c5_slq = slsqp_smooth(best_h, best_n, tau=2000.0, max_iter=10000)
        if h_slq is not None and c5_slq < best_c5 - 1e-14:
            best_c5 = c5_slq
            best_h = h_slq
        # Quick CD after
        if remaining() > 20:
            h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=500, time_limit=min(15, remaining() - 10), check_interval=1)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Phase 6: Perturbation search
    if remaining() > 30:
        tl = min(40, remaining() - 20)
        h_pert, c5_pert = perturbation_search(best_h, best_n, n_trials=5000, time_limit=tl, rng=rng)
        if c5_pert < best_c5 - 1e-14:
            best_c5 = c5_pert
            best_h = h_pert
            # Polish after perturbation
            if remaining() > 20:
                h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=1000, time_limit=min(15, remaining() - 10), check_interval=1)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd

    # Phase 7: Final SLSQP
    if remaining() > 30:
        h_slq, c5_slq = slsqp_smooth(best_h, best_n, tau=5000.0, max_iter=8000)
        if h_slq is not None and c5_slq < best_c5 - 1e-14:
            best_c5 = c5_slq
            best_h = h_slq

    # Phase 8: One last CD sweep
    if remaining() > 15:
        tl = max(5, remaining() - 8)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=2000, time_limit=tl, check_interval=1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
