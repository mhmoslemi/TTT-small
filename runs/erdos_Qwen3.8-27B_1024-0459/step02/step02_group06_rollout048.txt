import time
import numpy as np

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def corr_grad_lag(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        k = -k
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    return g

def compute_subgradient(h, top_k=3, temperature=0.0):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    if top_k >= len(corr):
        top_idx = np.arange(len(corr))
        w = np.ones(len(corr)) / len(corr)
    else:
        top_idx = np.argsort(corr)[-top_k:]
        top_vals = corr[top_idx]
        if temperature > 0:
            w = np.exp((top_vals - top_vals.max()) / temperature)
        else:
            dv = top_vals - top_vals.min()
            sc = dv.max() if dv.max() > 1e-12 else 1.0
            w = np.exp(dv / sc)
        w = w / w.sum()
    g = np.zeros(n, dtype=float)
    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += w[i] * corr_grad_lag(h, k)
    return g, corr

def local_search_sga(h_init, deadline, max_iter=50000, top_k=3, sa_temp=0.0):

    n = h_init.size
    target = 0.5 * n
    h = project_density(h_init, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    rng = np.random.default_rng(12345)
    stall = 0
    step_base = 0.1

                cap = min(1.0 - h[ii], h[jj])

                    cand[ii] += d_swap
                    cand[jj] -= d_swap
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-14:
                        best_h = cand
                        best_c = c

                        break
                if found:

            if found:

            cap = min(1.0 - h[ii], h[jj])

                continue
            d_swap = float(rng.random()) * cap
            cand = h.copy()
            cand[ii] += d_swap
            cand[jj] -= d_swap
            c = evaluate_erdos(cand)
            if c < best_c - 1e-14:
                best_h = cand
                best_c = c
                h = cand
                found = True
                break
            elif sa_temp > 0 and c - current_c < 0.005 and rng.random() < np.exp(-(c - current_c) / sa_temp):
                h = cand
                sa_accepted += 1
                stall = 0
                break

            stall = 0
        else:
            stall += 1
            if stall % 50 == 0:

                    step_base = 0.1
                    sa_temp *= 0.9
        if stall > 300:
            break

def neighborhood_perturb(h, rng, scale=0.01):
    """Apply a random perturbation that preserves the sum constraint"""
    n = h.size
    target = 0.5 * n
    pert = rng.normal(0, scale, n)
    pert -= np.mean(pert)
    return project_density(h + pert, target)

def multi_start_local_search(h_init, deadline, n_starts=8, max_iter_per=8000):
    """Try multiple perturbed starting points"""
    n = h_init.size
    target = 0.5 * n
    best_h = project_density(h_init, target)
    best_c = evaluate_erdos(best_h)
    rng = np.random.default_rng(2024)

    per_start_time = max(2.0, (deadline - time.perf_counter()) / max(1, n_starts))

    for s in range(n_starts):
        rem = deadline - time.perf_counter()
        if rem < 3.0:
            break
        if s == 0:
            start_h = best_h.copy()
        else:
            scale = 0.02 + 0.03 * s
            start_h = neighborhood_perturb(best_h, rng, scale=scale)

        sub_dl = time.perf_counter() + min(per_start_time, rem - 2.0)
        h_out, c_out = local_search_sga(start_h, sub_dl, max_iter=max_iter_per, top_k=3, sa_temp=0.002)
        if c_out < best_c:
            best_h = h_out
            best_c = c_out

    return best_h, best_c

def binarize_and_optimize(h, deadline):
    """Push towards binary, then optimize transition positions"""
    n = h.size

    h2 = h.copy()
    # Sort by value, assign top n/2 to 1, rest to 0
    sorted_idx = np.argsort(h2)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[n - int(target):]] = 1.0
    # Slight relaxation: allow small values near the boundary
    boundary_lo = sorted_idx[n - int(target) - 2:n - int(target)]
    boundary_hi = sorted_idx[n - int(target):n - int(target) + 2]
    for i in range(n - int(target) - 2, min(n - int(target) + 2, n)):
        if sorted_idx[i] in boundary_lo:
            h_bin[sorted_idx[i]] = 0.15
        elif sorted_idx[i] in boundary_hi:
            h_bin[sorted_idx[i]] = 0.85

    c2 = evaluate_erdos(h2)
    if c2 >= evaluate_erdos(h):

    # Polish the binarized version
    h_pol, c_pol = local_search_fine(h2, min(deadline, time.perf_counter() + 30), max_iter=5000)
