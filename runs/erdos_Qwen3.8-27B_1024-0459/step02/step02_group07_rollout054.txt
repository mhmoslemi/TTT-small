    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def walsh_init(n, rng):
    h = np.zeros(n)
    for i in range(n):
        v = i
        c = 0
        while v:
            c += v & 1
            v >>= 1
        if c % 2 == 0:
            h[i] = 1.0
    h = make_feasible(h, n)
    return h


def fourier_init(n, rng):
    h = np.full(n, 0.5)
    n_basis = 4
    for _ in range(n_basis):

        amp = rng.uniform(0.15, 0.45)

def binary_structured(n, rng):

    if remainder > 0:

    return make_feasible(h, n)

    shifted = corr * tau - np.max(corr) * tau
    exp_vals = np.exp(shifted - np.max(shifted))
    weights = exp_vals / (np.sum(exp_vals) + 1e-30)
    m = 2 * n - 1
    shifts_all = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for k_idx in range(m):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        shift = shifts_all[k_idx]
        js1 = max(0, shift)
        je1 = min(n, n + shift)
        if js1 < je1:
            grad[js1:je1] += w * (1.0 - h[js1 - shift:je1 - shift])
        js2 = max(0, -shift)
        je2 = min(n, n - shift)
        if js2 < je2:
            grad[js2:je2] += w * (-h[js2 + shift:je2 + shift])
    return grad / len(corr)


def projected_smooth_grad_descent(h, n, time_limit=30.0, tau_init=20.0):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    tau = tau_init
    alpha = 0.05
    it = 0
    while time.time() - start < time_limit:
        grad = smooth_max_grad(h, n, tau)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
            alpha *= 1.01
        else:
            alpha *= 0.85
        tau *= 1.02
        it += 1
        if it % 200 == 0:
            h = make_feasible(h, n)
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except (NameError, Exception):
        h_parent = None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    main_n = 256

    # Build diverse initializations at main_n
    inits = []
    if h_parent is not None:
        inits.append(interpolate_to_n(h_parent, main_n))
    inits.append(walsh_init(main_n, rng))
    inits.append(fourier_init(main_n, rng))
    inits.append(binary_structured(main_n, rng))
    for _ in range(4):
        inits.append(make_feasible(rng.rand(main_n), main_n))
    # Another structured: two-block
    h_two = np.zeros(main_n)
    gap = main_n // 8
    h_two[:main_n // 4] = 1.0
    h_two[main_n // 2:main_n // 2 + main_n // 4] = 1.0
    inits.append(make_feasible(h_two, main_n))
    # Three-block
    h_three = np.zeros(main_n)
    blk = main_n // 6
    h_three[:blk] = 1.0
    h_three[main_n // 3:2 * main_n // 3] = 1.0
    h_three[5 * main_n // 6:] = 1.0
    inits.append(make_feasible(h_three, main_n))

    # Phase 1: Coordinate descent from each init at main_n
    for h_init in inits:
        if remaining() < 50:
            break
        tl = min(55, remaining() - 30)
        if tl < 8:
            continue
        h_opt, c5_opt = coord_descent(h_init, main_n, max_passes=400, time_limit=tl)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = main_n

    # Phase 2: Further coordinate descent from best
    if remaining() > 80 and best_h is not None:
        tl = min(120, remaining() - 60)
        h_cd, c5_cd = coord_descent(best_h, best_n, max_passes=800, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Smooth gradient descent from best
    if remaining() > 50 and best_h is not None:
        tl = min(40, remaining() - 30)
        h_sg, c5_sg = projected_smooth_grad_descent(best_h, best_n, time_limit=tl, tau_init=15.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        # Follow up with coordinate descent
        if remaining() > 40:
            tl2 = min(50, remaining() - 25)
            h_cd2, c5_cd2 = coord_descent(best_h, best_n, max_passes=400, time_limit=tl2)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    # Phase 4: Try other resolutions
    other_ns = [200, 320, 360]
    for n_alt in other_ns:
        if remaining() < 50:
            break
        h_alt = interpolate_to_n(best_h, n_alt)
        tl = min(45, remaining() - 30)
        if tl < 10:
            continue
        h_opt, c5_opt = coord_descent(h_alt, n_alt, max_passes=300, time_limit=tl)
        if c5_opt < best_c5 - 1e-13:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_alt

    # Phase 5: Perturbation restarts from best
    for restart in range(4):
        if remaining() < 50:
            break
        noise_scale = 0.15 / (restart + 1)
        h_pert = best_h + rng.randn(best_n) * noise_scale
        h_pert = make_feasible(np.clip(h_pert, 0, 1), best_n)
        tl = min(40, remaining() - 30)
        if tl < 10:
            break
        h_opt, c5_opt = coord_descent(h_pert, best_n, max_passes=350, time_limit=tl)
        if c5_opt < best_c5 - 1e-13:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: SLSQP polish
    if remaining() > 60 and best_h is not None:
        tl_slsqp = min(80, remaining() - 40)
        max_it = min(5000, int(tl_slsqp * 20))
        h_slq, c5_slq = slsqp_polish(best_h, best_n, max_iter=max_it)
        if h_slq is not None and c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()

    # Phase 7: Final coordinate descent polish
    if remaining() > 30 and best_h is not None:
        tl = max(10, remaining() - 15)
        h_cd, c5_cd = coord_descent(best_h, best_n, max_passes=1000, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: Another SLSQP if time
    if remaining() > 40 and best_h is not None:
        h_slq2, c5_slq2 = slsqp_polish(best_h, best_n, max_iter=3000)
        if h_slq2 is not None and c5_slq2 < best_c5:
            best_c5 = c5_slq2
            best_h = h_slq2.copy()

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
