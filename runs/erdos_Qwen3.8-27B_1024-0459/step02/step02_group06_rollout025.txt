def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def corr_grad_lag(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

def get_top_lags(corr, top_frac, min_k):
    n_corr = len(corr)
    k_count = max(min_k, int(n_corr * top_frac))
    if k_count >= n_corr:
        return np.arange(n_corr)
    idx = np.argpartition(corr, -k_count)
    top = idx[-k_count:]
    return top[np.argsort(corr[top])[::-1]]

def smooth_max_val(corr, alpha, top_frac, min_k):
    top_idx = get_top_lags(corr, top_frac, min_k)
    top_vals = corr[top_idx]
    m = float(top_vals.max())
    y = np.clip(alpha * (top_vals - m), -500.0, 0.0)
    e = np.exp(y)
    z = float(e.sum())
    if z <= 0 or not np.isfinite(z):
        return m
    return m + np.log(z) / alpha

def active_gradient(h, corr, alpha, top_frac, min_k):
    n = h.size
    center = n - 1
    top_idx = get_top_lags(corr, top_frac, min_k)
    top_vals = corr[top_idx]
    m = float(top_vals.max())
    y = np.clip(alpha * (top_vals - m), -500.0, 0.0)
    e = np.exp(y)
    w = e / e.sum()
    g = np.zeros(n, dtype=float)
    for i in range(len(top_idx)):
        t = int(top_idx[i])
        k = t - center
        g += w[i] * corr_grad_lag(h, k)
    return g

def projected_grad_descent(h, target, deadline, alpha_schedule=(50, 150, 400),
                           top_frac=0.08, min_k=5, max_iter_per_alpha=200):
    n = h.size
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    
    for alpha in alpha_schedule:
        if time.perf_counter() >= deadline:
            break
        stall = 0
        for it in range(max_iter_per_alpha):
            if time.perf_counter() >= deadline:
                break
            corr = np.correlate(best, 1.0 - best, mode="full")
            grad = active_gradient(best, corr, alpha, top_frac, min_k)
            grad -= np.mean(grad)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            d = -grad / gnorm
            improved = False
            for step in (0.008, 0.02, 0.04, 0.08, 0.15, 0.3):
                cand = project_density(best + step * d, target)
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    improved = True
                    break
            if not improved:
                for step in (0.002, 0.005, 0.012):
                    cand = project_density(best + step * d, target)
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        improved = True
                        break
            if improved:
                stall = 0
            else:
                stall += 1
                if stall > 30:
                    break
    return best, evaluate_erdos(best)

def pair_swap_search(h, target, deadline, top_frac=0.06, min_k=6,
                     n_pos=10, n_neg=10, max_iter=2000):
    n = h.size
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(12345)
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        center = n - 1
        k_count = max(min_k, int(len(corr) * top_frac))
        top_idx = get_top_lags(corr, top_frac, min_k)
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / max(1e-10, dv.max() + 1e-10))
        w /= w.sum()
        g = np.zeros(n, dtype=float)
        for i in range(len(top_idx)):
            t = int(top_idx[i])
            k = t - center
            g += w[i] * corr_grad_lag(best, k)
        neg_cand = np.argsort(g)[:n_neg]
        pos_cand = np.argsort(-g)[:n_pos]
        found = False
        for ii in neg_cand:
            ii = int(ii)
            for jj in pos_cand:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                for d in (0.1 * cap, 0.3 * cap, 0.6 * cap, 0.9 * cap, cap):
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    if cand[ii] < -1e-12 or cand[ii] > 1 + 1e-12 or cand[jj] < -1e-12 or cand[jj] > 1 + 1e-12:
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            for _r in range(20):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                d = float(rng.random()) * 0.7 * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return best, evaluate_erdos(best)

def fine_tune(h, target, deadline, n_steps=500):
    """Very fine local adjustments on individual elements"""
    n = h.size
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(777)
    stall = 0
    for _ in range(n_steps):
        if time.perf_counter() >= deadline:
            break
        # Pick 2-3 random indices and try small adjustments
        idxs = rng.integers(0, n, 3)
        idxs = [int(x) for x in idxs]
        if len(set(idxs)) < 3:
            continue
        i1, i2 = idxs[0], idxs[1]
        cap = min(1.0 - best[i1], best[i2])
        if cap < 1e-10:
            continue
        for d in (0.005, 0.01, 0.02, 0.05):
            if d > cap:
                continue
            cand = best.copy()
            cand[i1] += d
            cand[i2] -= d
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                stall = 0
                break
            # Try reverse
            cand2 = best.copy()
            cand2[i1] -= d
            cand2[i2] += d
            c2 = evaluate_erdos(cand2)
            if c2 < best_c - 1e-13:
                best = cand2
                best_c = c2
                stall = 0
                break
        else:
            stall += 1
        if stall > 100:
            break
    return best, evaluate_erdos(best)

def make_structured_start(n, rng, n_segments=6):
    """Create a step function with n_segments equal segments, random values"""
    h = np.zeros(n)
    seg_size = n // n_segments
    for i in range(n_segments):
        start = i * seg_size
        end = start + seg_size if i < n_segments - 1 else n
        val = float(rng.uniform(0.1, 0.9))
        h[start:end] = val
    return project_density(h, 0.5 * n)

def make_sinusoidal_start(n, rng, n_harm=3):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.zeros(n)
    for k in range(1, n_harm + 1):
        freq = int(rng.integers(2, 8))
        amp = float(rng.uniform(0.1, 0.3)) / (k + 1)
        phase = float(rng.uniform(0, 2 * np.pi))
        h += amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h += 0.5
    return project_density(h, 0.5 * n)

def optimize_at_resolution(n, h0, deadline, seed=42):
    """Full optimization pipeline at fixed resolution n"""
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    
    # Phase 1: Projected gradient with alpha sweep
    rem = deadline - time.perf_counter()
    if rem > 30:
        grad_deadline = time.perf_counter() + rem * 0.4
        h1, c1 = projected_grad_descent(h, target, grad_deadline,
                                         alpha_schedule=(50, 150, 400),
                                         top_frac=0.08, min_k=5,
                                         max_iter_per_alpha=150)
        if c1 < best_c:
            best_h, best_c = h1, c1
    
    # Phase 2: Pair swap
    rem = deadline - time.perf_counter()
    if rem > 20:
        swap_deadline = time.perf_counter() + rem * 0.35
        h2, c2 = pair_swap_search(best_h, target, swap_deadline,
                                   top_frac=0.05, min_k=6,
                                   n_pos=12, n_neg=12, max_iter=1500)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    # Phase 3: Another gradient pass with tighter focus
    rem = deadline - time.perf_counter()
    if rem > 15:
        grad_dl2 = time.perf_counter() + rem * 0.35
        h3, c3 = projected_grad_descent(best_h, target, grad_dl2,
                                         alpha_schedule=(200, 500),
                                         top_frac=0.04, min_k=8,
                                         max_iter_per_alpha=100)
        if c3 < best_c:
            best_h, best_c = h3, c3
    
    # Phase 4: Fine tuning
    rem = deadline - time.perf_counter()
    if rem > 10:
        ft_dl = time.perf_counter() + rem * 0.6
        h4, c4 = fine_tune(best_h, target, ft_dl, n_steps=800)
        if c4 < best_c:
            best_h, best_c = h4, c4
    
    # Phase 5: Final pair swap
    rem = deadline - time.perf_counter()
    if rem > 8:
        ps_dl = time.perf_counter() + rem * 0.8
        h5, c5 = pair_swap_search(best_h, target, ps_dl,
                                   top_frac=0.04, min_k=8,
                                   n_pos=8, n_neg=8, max_iter=1000)
        if c5 < best_c:
            best_h, best_c = h5, c5
    
    return project_density(best_h, target), evaluate_erdos(best_h)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 20.0)
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float("inf")
    best_n = 0
    
    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 10:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        n_p = int(parent.size)
        p_proj = project_density(parent, 0.5 * n_p)
        p_c = evaluate_erdos(p_proj)
        best_h = p_proj.copy()
        best_c = p_c
        best_n = n_p
    else:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0
    
    # Resolution plan: focus on higher resolutions
    base_n = best_n
    resolutions = []
    # Always try 2x and 3x the base
    for mult in (2, 3):
        rn = base_n * mult
        if 100 <= rn <= 2000:
            resolutions.append(rn)
    # Add some standard resolutions
    for rn in [450, 600, 900, 1200, 1350, 1800]:
        if rn not in resolutions and 100 <= rn <= 2000:
            resolutions.append(rn)
    resolutions = sorted(set(resolutions))
    
    # Time allocation: weight higher resolution more
    total_rem = deadline - time.perf_counter()
    # Split: 50% to the 2x resolution, 30% to 3x, 20% to extras
    n_main = min(2 * base_n, 1800) if 2 * base_n >= 100 else 900
    n_high = min(3 * base_n, 1800) if 3 * base_n >= 200 else 1350
    
    # Define priority order
    priority = []
    if n_main in resolutions:
        priority.append(n_main)
        resolutions.remove(n_main)
    if n_high in resolutions:
        priority.append(n_high)
        resolutions.remove(n_high)
    # Add remaining, up to 2 more
    for r in sorted(resolutions)[:-1]:
        if len(priority) < 4:
            priority.append(r)
    
    # Time per resolution
    time_allocs = []
    rem = deadline - time.perf_counter()
    n_res = max(1, len(priority))
    if n_res == 1:
        time_allocs = [rem - 15]
    elif n_res == 2:
        time_allocs = [rem * 0.55 - 10, rem * 0.45 - 15]
    elif n_res == 3:
        time_allocs = [rem * 0.45 - 10, rem * 0.35 - 10, rem * 0.2 - 10]
    else:
        time_allocs = [rem * 0.4 - 10, rem * 0.3 - 10, rem * 0.2 - 8, rem * 0.1 - 8]
    
    for ri, nr in enumerate(priority[:4]):
        rem = deadline - time.perf_counter()
        if rem < 30:
            break
        alloc = min(time_allocs[ri] if ri < len(time_allocs) else rem * 0.3, rem - 10)
        if alloc < 25:
            break
        
        # Build starting point from current best
        h_start = upsample_h(best_h, nr)
        sub_dl = time.perf_counter() + alloc
        
        h_out, c_out = optimize_at_resolution(nr, h_start, sub_dl, seed=seed)
        if c_out < best_c:
            best_h = h_out.copy()
            best_c = c_out
            best_n = nr
    
    # Try structured starts at a mid resolution
    rem = deadline - time.perf_counter()
    if rem > 60:
        n_struct = 900 if best_n < 900 else min(1350, best_n + 450)
        n_struct = int(n_struct)
        alloc_struct = rem * 0.25
        if alloc_struct > 40:
            for seg in (4, 6, 8):
                rem2 = deadline - time.perf_counter()
                if rem2 < 25:
                    break
                h_s = make_structured_start(n_struct, rng, n_segments=seg)
                sub_dl = time.perf_counter() + min(alloc_struct / 3, rem2 - 5)
                h_out, c_out = optimize_at_resolution(n_struct, h_s, sub_dl, seed=seed + seg)
                if c_out < best_c:
                    best_h = h_out.copy()
                    best_c = c_out
                    best_n = n_struct
    
    # Final intensive polish on the best solution found
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 20:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        
        # Extended pair swap
        if rem > 30:
            h2, c2 = pair_swap_search(h_fin, target, time.perf_counter() + rem * 0.35,
                                       top_frac=0.04, min_k=8,
                                       n_pos=14, n_neg=14, max_iter=3000)
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h2.copy()
        
        # Fine tune
        rem2 = deadline - time.perf_counter()
        if rem2 > 15:
            h3, c3 = fine_tune(h_fin, target, time.perf_counter() + rem2 * 0.5, n_steps=2000)
            if c3 < best_c:
                h_fin = h3
                best_c = c3
                best_h = h3.copy()
        
        # One more gradient pass
        rem3 = deadline - time.perf_counter()
        if rem3 > 12:
            h4, c4 = projected_grad_descent(h_fin, target, time.perf_counter() + rem3 * 0.6,
                                             alpha_schedule=(300, 800),
                                             top_frac=0.03, min_k=10,
                                             max_iter_per_alpha=80)
            if c4 < best_c:
                h_fin = h4
                best_c = c4
                best_h = h4.copy()
        
        # Final pair swap
        rem4 = deadline - time.perf_counter()
        if rem4 > 8:
            h5, c5 = pair_swap_search(h_fin, target, time.perf_counter() + rem4 * 0.8,
                                       top_frac=0.03, min_k=10,
                                       n_pos=8, n_neg=8, max_iter=2000)
            if c5 < best_c:
                h_fin = h5
                best_c = c5
                best_h = h5.copy()
    
    if best_h is None:
        fallback = project_density(np.full(90, 0.5), 45.0)
        return fallback, evaluate_erdos(fallback), 90
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
