import time
import numpy as np

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_binary(h, target_sum):
    """Project to binary (0/1) with exact sum"""
    n = len(h)
    h = np.asarray(h, dtype=float)
    # Sort and threshold
    sorted_idx = np.argsort(h)[::-1]
    out = np.zeros(n, dtype=float)
    out[sorted_idx[:int(target_sum)]] = 1.0
    return out

def compute_corr(h):
    """Compute full correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode="full")

def binary_local_search(h, deadline, max_rounds=100):
    """Combinatorial local search on binary pattern"""
    n = len(h)
    target = n // 2
    assert abs(np.sum(h) - target) < 1e-10
    h = h.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    current_max = float(np.max(corr) * dx)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for round in range(max_rounds):
        if time.perf_counter() > deadline:
            break
        
        best_gain = 0
        best_swap = None
        best_new_corr = None
        
        # Try all swaps: move 1 from position a to position b
        for a in ones:
            for b in zeros:
                if a == b:
                    continue
                # Compute delta at all lags
                # delta[t] for t in [0, 2n-2], corresponding to lag k = t - (n-1)
                # delta = -(1-h)[a+k] + h[a-k] + (1-h)[b+k] - h[b-k] (where k = t-(n-1))
                
                # Using numpy: for lag k, the terms are at indices:
                # (1-h)[a+k] valid when 0 <= a+k < n
                # h[a-k] valid when 0 <= a-k < n  
                # (1-h)[b+k] valid when 0 <= b+k < n
                # h[b-k] valid when 0 <= b-k < n
                
                # k ranges from -(n-1) to (n-1)
                # t = n-1+k, so t ranges from 0 to 2n-2
                
                # Build delta array
                delta = np.zeros(2*n-1)
                
                # Term 1: -(1-h)[a+k] for valid a+k
                lo1 = max(0, -(n-1))  # k >= -(n-1)
                hi1 = min(n-1-a, n-1)  # a+k <= n-1 → k <= n-1-a
                # k from max(lo1, -(n-1)) to min(hi1, n-1)
                k_lo = max(-(n-1), -(a))  # a+k >= 0
                k_hi = min(n-1, n-1-a)  # a+k <= n-1
                k_lo = max(k_lo, -(n-1))
                k_hi = min(k_hi, n-1)
                if k_lo <= k_hi:
                    ks = np.arange(k_lo, k_hi+1)
                    idx = a + ks
                    delta[n-1+ks] -= (1.0 - h[idx])
                
                # Term 2: +h[a-k] for valid a-k
                k_lo2 = max(-(n-1), -(a))  # a-k >= 0 → k <= a
                k_hi2 = min(n-1, a)  # a-k >= 0, and a-k <= n-1 → k >= a-(n-1)
                k_lo2 = max(k_lo2, a-(n-1))
                k_hi2 = min(k_hi2, n-1)
                k_lo2 = max(k_lo2, -(n-1))
                k_hi2 = min(k_hi2, n-1)
                if k_lo2 <= k_hi2:
                    ks = np.arange(k_lo2, k_hi2+1)
                    idx = a - ks
                    delta[n-1+ks] += h[idx]
                
                # Term 3: +(1-h)[b+k] for valid b+k
                k_lo3 = max(-(n-1), -b)  # b+k >= 0
                k_hi3 = min(n-1, n-1-b)  # b+k <= n-1
                if k_lo3 <= k_hi3:
                    ks = np.arange(k_lo3, k_hi3+1)
                    idx = b + ks
                    delta[n-1+ks] += (1.0 - h[idx])
                
                # Term 4: -h[b-k] for valid b-k
                k_lo4 = max(-(n-1), -(b))  # b-k >= 0 → k <= b
                k_hi4 = min(n-1, n-1)  # b-k <= n-1 → k >= b-(n-1)
                k_lo4 = max(k_lo4, b-(n-1))
                k_hi4 = min(k_hi4, b)
                if k_lo4 <= k_hi4:
                    ks = np.arange(k_lo4, k_hi4+1)
                    idx = b - ks
                    delta[n-1+ks] -= h[idx]
                
                new_corr = corr + delta
                new_max = float(np.max(new_corr) * dx)
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_swap = (a, b)
                    best_new_corr = new_corr
        
        if best_swap is None or best_gain < 1e-12:
            break
        
        a, b = best_swap
        h[a] = 0.0
        h[b] = 1.0
        corr = best_new_corr
        current_max = current_max - best_gain
        
        # Update ones and zeros
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

def fast_swap_filter(h, a, b, k_star, n):
    """Check if swap (a,b) reduces overlap at lag k_star"""
    d = 0
    if 0 <= a + k_star < n:
        d -= (1.0 - h[a + k_star])
    if 0 <= a - k_star < n:
        d += h[a - k_star]
    if 0 <= b + k_star < n:
        d += (1.0 - h[b + k_star])
    if 0 <= b - k_star < n:
        d -= h[b - k_star]
    return d

def evaluate_erdos(h):

    n = h.size

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

def swap_delta_at_lag(h, a, b, k, n):
    d = 0.0
    ai = a + k

        alloc = time_alloc.get(ni, max(30.0, rem / max(1, len(n_list))))
        alloc = min(alloc, rem - 10.0)

        # Build starting points
        starts = []
        starts.append(upsample_h(best_h, ni))
        if parent is not None and parent.size != ni:
            starts.append(upsample_h(parent, ni))
        elif parent is not None:
            starts.append(np.asarray(parent, dtype=float).copy())
        for mm in range(2):
            starts.append(make_structural_start(ni, mm, rng))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = any(np.allclose(s, u, atol=0.05) for u in uniq)
            if not dup:
                uniq.append(s)

        per_start = max(10.0, alloc / len(uniq))

        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 10.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 5.0)
            target = 0.5 * ni

            # Phase 1: Push to binary and do combinatorial search
            h_bin = to_binary(s, ni // 2)
            c0 = evaluate_erdos(h_bin)

            # Phase 2: Binary local search
            t2 = time.perf_counter() + (sub_dl - time.perf_counter()) * 0.55
            h_bi, c_bi = binary_local_search(h_bin, t2, batch_size=5000, max_rounds=50000, top_count=10, seed=seed + ni)
            if c_bi < best_c:
                best_h = h_bi.copy()
                best_c = c_bi
                best_n = ni

            # Phase 3: Double swap
            rem3 = sub_dl - time.perf_counter()
            if rem3 > 5.0:
                h_ds, c_ds = double_swap_search(h_bi, time.perf_counter() + rem3 * 0.4, batch_size=1000, max_rounds=2000, seed=seed + 7 + ni)
                if c_ds < best_c:
                    best_h = h_ds.copy()
                    best_c = c_ds
                    best_n = ni
                    h_bi = h_ds

            # Phase 4: Continuous polish from binary
            rem4 = sub_dl - time.perf_counter()
            if rem4 > 5.0:
                h_cp, c_cp = continuous_polish(h_bi, time.perf_counter() + rem4 * 0.8, ni)
                if c_cp < best_c:
                    best_h = h_cp.copy()
                    best_c = c_cp
                    best_n = ni

            # Also try continuous polish from the original continuous start
            rem5 = sub_dl - time.perf_counter()
            if rem5 > 5.0:
                h_cp2, c_cp2 = continuous_polish(s, time.perf_counter() + rem5 * 0.7, ni)
                if c_cp2 < best_c:
                    best_h = h_cp2.copy()
                    best_c = c_cp2
                    best_n = ni

    # Final intensive polish on the global best
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest

        # Binary push + local search
        h_bf = to_binary(best_h, nbest // 2)
        c_bf = evaluate_erdos(h_bf)
        if c_bf < best_c:
            best_h = h_bf.copy()
            best_c = c_bf

        rem2 = deadline - time.perf_counter()
        if rem2 > 10.0 and rem > 50.0:
            h_fn, c_fn = binary_local_search(h_bf, time.perf_counter() + rem2 * 0.4,
                                             batch_size=8000, max_rounds=100000, top_count=12, seed=seed + 100)
            if c_fn < best_c:
                best_h = h_fn.copy()
                best_c = c_fn

        rem3 = deadline - time.perf_counter()
        if rem3 > 8.0:
            h_d2, c_d2 = double_swap_search(best_h, time.perf_counter() + rem3 * 0.3,
                                            batch_size=2000, max_rounds=5000, seed=seed + 200)
            if c_d2 < best_c:
                best_h = h_d2.copy()
                best_c = c_d2

        rem4 = deadline - time.perf_counter()
        if rem4 > 5.0:
            h_final, c_final = continuous_polish(best_h, time.perf_counter() + rem4 * 0.85, nbest)
            if c_final < best_c:
                best_h = h_final.copy()
                best_c = c_final

    if best_h is None:
        fallback = project_density(np.full(450, 0.5), 225.0)
        return fallback, evaluate_erdos(fallback), 450

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
