import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if 0.25 <= frac < 0.75 else 0.0
    return h


def compute_grad_lag(h, offset, n):
    grad = np.zeros(n)
    if offset >= 0:
        n_pairs = n - offset
        if n_pairs > 0:
            grad[:n_pairs] += (1.0 - h[offset:offset + n_pairs])
            grad[offset:offset + n_pairs] += -h[:n_pairs]
    else:
        abs_off = -offset
        n_pairs = n - abs_off
        if n_pairs > 0:
            grad[abs_off:] += (1.0 - h[:n_pairs])
            grad[:n_pairs] += -h[abs_off:]
    return grad


def smooth_objective_factory(n, beta, mu):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(corr - m, -60.0 / beta, 0.0)
        smax = m + np.log(np.sum(np.exp(beta * shifted))) / beta
        pen = mu * (np.sum(x) - n / 2.0) ** 2
        return smax * dx + pen

    return obj
