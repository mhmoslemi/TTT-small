from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)

def obj_logsumexp(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    # For numerical stability
    exps = np.exp(beta * shifted)
    val = (m + np.log(np.sum(exps)) / beta) * dx
    return val

def obj_topk(h, n, topk):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -topk)[::-1][:topk]
    return np.sum(corr[idx]) * dx

def coordinate_descent(h, n, max_sweeps=200, try_continuous=True):
    """Fast coordinate descent trying binary and optionally continuous values."""
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

            break
        improved = False
        for i in range(n):
            old = h[i]
            candidates = [0.0, 1.0]
            if try_continuous and 0.0 < old < 1.0:
                candidates = [0.0, 1.0, old + 0.3, old - 0.3]
                candidates = [c for c in candidates if 0.0 <= c <= 1.0]
            
            for new_val in candidates:
                if abs(new_val - old) < 1e-8:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h = best_h.copy()  # revert
            # if no improvement at this coordinate, h is already at best_h
            h = best_h.copy()
        
        if not improved:
            break
    
    return best_h, best_c5

def active_set_slsqp(h, n, topk=5, maxiter=300):
    """SLSQP minimizing sum of top-k correlations."""
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    
    def obj(x):
        return obj_topk(x, n, topk)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except:
        return h.copy(), compute_c5(h)

        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except:

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():

def obj_logsumexp(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -500, 500))
    return (m + np.log(np.sum(exps)) / beta) * dx


def obj_topk_sum(h, n, topk):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -topk)[::-1][:topk]
    return np.sum(corr[idx]) * dx


def slsqp_optimize(h, n, beta, topk, maxiter):
