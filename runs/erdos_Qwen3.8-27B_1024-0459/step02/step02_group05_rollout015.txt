import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2}."""
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fine adjustment
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def subgradient(h, n, beta):
    """Compute soft subgradient of max_k corr[k]."""
    corr = compute_corr(h)
    max_c = np.max(corr)
    # Softmax weights
    exps = np.exp(beta * (corr - max_c))
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)
    # For each lag, add weighted gradient
    # lag = k - (n-1) for k = 0..2n-2
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            L = lag
            if L < n:
                grad[L:] += wk * (1.0 - h[:n-L])
                grad[:n-L] -= wk * h[L:]
        else:
            L = -lag
            if L < n:
                grad[:n-L] += wk * (1.0 - h[L:])
                grad[L:] -= wk * h[:n-L]
    
    return grad, max_c, corr

def line_search(h, n, grad, alpha):
    """Backtracking line search."""
    best_c = np.max(compute_corr(h))
    best_h = h.copy()
    a = alpha
    for _ in range(30):
        h_try = project_feasible(h - a * grad, n)
        c_try = np.max(compute_corr(h_try))
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
        a *= 0.5
        if a < 1e-10:
            break
    return best_h, best_c

def subgradient_descent(h, n, n_iters, beta_start, beta_end, alpha_start, alpha_decay):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        t = it / max(n_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        alpha = alpha_start * (alpha_decay ** it)
        
        grad, max_c, corr = subgradient(h, n, beta)
        
        h_new, c_new = line_search(h, n, grad, alpha)
        
        if c_new * (2.0/n) < best_c5:
            best_c5 = c_new * (2.0/n)
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def local_search_fine(h, n, n_trials, rng, step=0.01):
    """Fine local search with small perturbations."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        h_try = h.copy()
        idx = rng.randint(n)
        h_try[idx] += rng.uniform(-step, step)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
        elif rng.random() < 0.05:
            step *= 0.95
        if step < 1e-5:
            step = 0.01
    return h, best_c5

    start_time = time.time()

        h_interp = np.interp(x_new, x_old, best_h)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p


def subgradient(h, n, beta):
    corr = compute_corr(h)
    max_c = np.max(corr)
    exps = np.exp(beta * (corr - max_c))
    s = np.sum(exps)
    if s < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    else:
        w = exps / s

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        wk = w[k]
        lag = k - (n - 1)
        if lag >= 0:
            L = lag
            if L < n:
                grad[L:] += wk * (1.0 - h[:n - L])
                grad[:n - L] -= wk * h[L:]
        else:
            L = -lag
            if L < n:
                grad[:n - L] += wk * (1.0 - h[L:])
                grad[L:] -= wk * h[:n - L]
    return grad, max_c


def line_search(h, n, grad, alpha_init):
    best_c = np.max(compute_corr(h))
    best_h = h.copy()
    a = alpha_init
    for _ in range(25):
        h_try = project_feasible(h - a * grad, n)
        c_try = np.max(compute_corr(h_try))
        if c_try < best_c - 1e-15:
            best_c = c_try
            best_h = h_try
            break
        a *= 0.5
        if a < 1e-12:
            break
    return best_h, best_c


def subgrad_descent(h, n, n_iters, beta_start, beta_end, alpha_start, alpha_decay):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        t = it / max(n_iters - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        alpha = alpha_start * (alpha_decay ** it)
        grad, _ = subgradient(h, n, beta)
        h_new, _ = line_search(h, n, grad, alpha)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def softmax_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 0)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))


def slsqp_smooth(h, n, betas=(100, 500, 2000, 8000, 20000), max_iter=3000):
    current = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    for beta in betas:
        def obj_s(x):
            return softmax_obj(x, n, beta)
        try:
            res = minimize(obj_s, current, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
            if res.fun < 1e10:
                current = project_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5


def pair_local_search(h, n, n_rounds, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        pair_count = 0
        while pair_count < n:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            pair_count += 1
            amt = min(0.02, abs(h[i] - h[j]) * 0.5)
            if amt < 1e-6:
                continue
            direction = 1.0 if h[i] > h[j] else -1.0
            h_try = h.copy()
            h_try[i] += direction * amt
            h_try[j] -= direction * amt
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
        if not improved:
            break
    return h, best_c5


def fine_search(h, n, n_trials, rng, base_step=0.008):
    h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for _ in range(n_trials):
        h_try = h.copy()
        k = rng.randint(1, 4)
        idxs = rng.choice(n, k, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
            step = max(step * 1.05, 0.001)
        else:
            if rng.random() < 0.02:
                step *= 0.85
                step = max(step, 0.0005)
    return h, best_c5


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.full(n_dst, 0.5)
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)
