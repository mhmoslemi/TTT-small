    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

        cnt = np.sum(mask)
        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def binary_project(h, n):
    """Project to binary with exactly n/2 ones."""
    h = h.copy()
    # Sort by value, threshold
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def smooth_max_grad_c5(h, beta, n):
    """Gradient of smooth_max(corr) * dx with respect to h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)  # actual lag
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            # c(L) = sum_{j=0}^{n-1-L} h[j+L]*(1-h[j])
            # d/dh[i] = (1-h[i-L]) for i in [L, n-1] - h[i+L] for i in [0, n-1-L]
            grad[L:] += w * (1.0 - h[:n-L])  # (1-h[i-L]) for i=L..n-1
            grad[:n-L] -= w * h[L:]           # -h[i+L] for i=0..n-1-L
        else:
            Lp = -L
            # c(-Lp) = sum_{j=Lp}^{n-1} h[j]*(1-h[j-Lp])
            # d/dh[i] = (1-h[i+Lp]) for i in [0, n-1-Lp] - h[i-Lp] for i in [Lp, n-1]
            grad[:n-Lp] += w * (1.0 - h[Lp:])  # (1-h[i+Lp]) for i=0..n-1-Lp
            grad[Lp:] -= w * h[:n-Lp]          # -h[i-Lp] for i=Lp..n-1
    
    return dx * grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.maximum(h, 1.0 - h)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def to_binary(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0

def compute_delta_vectors(h, n):

    rem = np.zeros((n, 2 * n - 1))
    add = np.zeros((n, 2 * n - 1))
    offset = n - 1

    for a in range(n):
        if h[a] < 0.5:
            continue
        for L in range(1, n):
            k_pos = offset + L
            k_neg = offset - L
            delta_pos = 0.0
            if L <= a:
                delta_pos -= (1.0 - h[a - L])
            if a <= n - 1 - L:
                delta_pos += h[a + L]
            delta_neg = 0.0
            if L <= a:
                delta_neg -= (1.0 - h[a - L])
            if a <= n - 1 - L:
                delta_neg += h[a + L]
            rem[a, k_pos] = delta_pos
            rem[a, k_neg] = delta_neg

    for b in range(n):

            k_neg = offset - L
            delta_pos = 0.0

            if b <= n - 1 - L:
                delta_neg -= h[b + L]

def greedy_one_round(h, n, rem, add):
    """One round of greedy best-improvement. Returns (improved, h_new, c5_new)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    current_max = np.max(corr) * dx

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    n_ones = len(ones)
    n_zeros = len(zeros)
    if n_ones == 0 or n_zeros == 0:
        return False, h, current_max

    best_val = current_max
    best_a = -1
    best_b = -1

    for i in range(n_ones):
        a = ones[i]
        rem_a = rem[a]
        for j in range(n_zeros):
            b = zeros[j]
            new_corr = corr + rem_a + add[b]
            val = np.max(new_corr) * dx
            if val < best_val - 1e-15:
                best_val = val
                best_a = a
                best_b = b

    if best_a < 0:
        return False, h, current_max

    h_new = h.copy()
    h_new[best_a] = 0.0
    h_new[best_b] = 1.0
    return True, h_new, best_val

def greedy_binary_search(h, n, max_rounds=5000, time_budget=60.0):
    """Run greedy best-improvement binary local search."""
    start = time.time()
    h_cur = h.copy()
    rem, add = compute_delta_vectors(h_cur, n)
    c5_cur = compute_c5(h_cur)

    for rnd in range(max_rounds):
        if time.time() - start > time_budget:
            break
        improved, h_new, c5_new = greedy_one_round(h_cur, n, rem, add)
        if not improved:
            break
        h_cur = h_new
        c5_cur = c5_new
        rem, add = compute_delta_vectors(h_cur, n)

    return h_cur, c5_cur

def simulated_annealing_binary(h, n, time_budget, rng, T0=0.15, T1=0.005, cooldown=0.9999):
    """Simulated annealing on binary strings with swap moves."""
    start = time.time()
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    T = T0
    n_ones = n // 2

    while time.time() - start < time_budget:
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h_cur[a] = 0.0
        h_cur[b] = 1.0
        c5_new = compute_c5(h_cur)
        delta = c5_new - c5_cur

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
        else:
            h_cur[a] = 1.0
            h_cur[b] = 0.0

        T = max(T * cooldown, T1)

    return h_cur, c5_cur

def continuous_optimize(h0, n, beta, max_iter=200):
    """L-BFGS-B continuous optimization with sum penalty."""

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        sm = m + np.log(np.sum(np.exp(shifted))) / beta
        penalty = 100.0 * (np.sum(x) - n / 2.0) ** 2
        return (2.0 / n) * sm + penalty

    def grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        exps = np.exp(shifted)
        weights = exps / np.sum(exps)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-14:
                continue
            L = k - (n - 1)
            if L == 0:
                g += w * (1.0 - 2.0 * x)
            elif L > 0:
                g[L:] += w * (1.0 - x[:n - L])
                g[:n - L] -= w * x[L:]
            else:
                Lp = -L
                g[:n - Lp] += w * (1.0 - x[Lp:])
                g[Lp:] -= w * x[:n - Lp]
        pen_grad = 200.0 * (np.sum(x) - n / 2.0) * np.ones(n)
        return dx * g + pen_grad

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            best_n = parent_n
            best_h = project_feasible(parent_h, parent_n)
            best_c5 = compute_c5(best_h)
            print("Parent: n={}, C5={:.6f}".format(parent_n, best_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 1: Continuous optimization at parent resolution
    if remaining() > 100:
        print("Phase 1: Continuous optimization at n={}".format(best_n))
        for beta_val in [30, 80, 200]:
            if remaining() < 30:
                break
            try:
                h_cont, c5_cont = continuous_optimize(best_h.copy(), best_n, beta_val, 150)
                if c5_cont < best_c5:
                    best_c5 = c5_cont
                    best_h = h_cont.copy()
                    print("  Continuous beta={}: C5={:.6f}".format(beta_val, best_c5))
            except Exception:
                pass

    # Phase 2: Binary projection and greedy search at parent n
    if remaining() > 80:
        print("Phase 2: Binary greedy search at n={}".format(best_n))
        h_bin = to_binary(best_h, best_n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        t_budget = min(remaining() - 20, 250)
        h_g, c5_g = greedy_binary_search(h_bin, best_n, max_rounds=3000, time_budget=t_budget)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        print("  After greedy: C5={:.6f}".format(best_c5))

    # Phase 3: Simulated annealing
    if remaining() > 100:
        print("Phase 3: Simulated annealing at n={}".format(best_n))
        t_budget = min(remaining() - 30, 350)
        h_sa, c5_sa = simulated_annealing_binary(best_h, best_n, t_budget, rng,
                                                 T0=0.12, T1=0.003, cooldown=0.99995)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        print("  After SA: C5={:.6f}".format(best_c5))

    # Phase 4: Greedy + SA at different resolutions
    for n_try in [best_n + 100, best_n - 50, best_n + 200]:
        if remaining() < 120:
            break
        if n_try < 80 or n_try > 500:
            continue
        if n_try % 2 != 0:
            n_try += 1
        print("Phase 4: Trying n={}".format(n_try))
        h_try = interp_h(best_h, n_try)
        h_bin_try = to_binary(h_try, n_try)
        c5_try = compute_c5(h_bin_try)

        t_budget = min(remaining() - 20, 150)
        h_g2, c5_g2 = greedy_binary_search(h_bin_try, n_try, max_rounds=2000, time_budget=t_budget)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()
            best_n = n_try
            print("  n={}: C5={:.6f}".format(n_try, best_c5))

        if remaining() > 80 and c5_g2 < best_c5 + 0.005:
            t_sa = min(remaining() - 20, 100)
            h_sa2, c5_sa2 = simulated_annealing_binary(h_g2, n_try, t_sa, rng,
                                                       T0=0.10, T1=0.003, cooldown=0.99995)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                best_n = n_try
                print("  n={} SA: C5={:.6f}".format(n_try, best_c5))

    # Phase 5: Final greedy polish
    if remaining() > 30:
        print("Phase 5: Final greedy polish at n={}".format(best_n))
        is_binary = np.all((best_h == 0) | (best_h == 1))
        if is_binary:
            h_fin, c5_fin = greedy_binary_search(best_h, best_n, max_rounds=1000,
                                                 time_budget=min(remaining() - 10, 60))
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        print("  Final: C5={:.6f}".format(best_c5))

    # Phase 6: Try a few continuous refinements
    if remaining() > 40 and not np.all((best_h == 0) | (best_h == 1)):
        try:
            h_ref, c5_ref = continuous_optimize(best_h.copy(), best_n, 500, 200)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
