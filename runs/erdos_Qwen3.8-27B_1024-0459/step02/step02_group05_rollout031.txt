def compute_c5(h):
    n = len(h)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)

def compute_corr(h):

    """Smooth max over selected indices"""
    vals = corr[mask]
    cmax = np.max(vals)
    return cmax + (1.0/beta) * np.log(np.sum(np.exp(beta * (vals - cmax))))

def active_set_solve(h, n, mask, beta, max_iter=2000, ftol=1e-14):
    """SLSQP with objective = smooth max over active lags"""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr[mask]
        cmax = np.max(vals)
        return cmax + (1.0/beta) * np.log(np.sum(np.exp(beta * (vals - cmax))))
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:

                if 0 <= i - L < n:
                    g[i] += w * h[i - L]
        g_centered = g - np.mean(g)
        h_try = h - step_size * g_centered
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
        else:
            step_size *= 0.8
            if step_size < 1e-8:
                break
    return h, best_c5

def binary_sa(n, rng, n_steps, t_start, t_end, deadline):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-10)):
            pass
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def random_perturb(h, n, rng, n_trials, base_step, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for _ in range(n_trials):
        if time.time() > deadline:
            break
        h_try = h.copy()
        n_mod = rng.randint(1, 7)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.005) & (h_try < 0.995)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
        else:
            continue
        c5 = compute_c5(h_try)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            h = h_try
        elif rng.random() < 0.15:
            step *= 0.92
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.94

    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None and h_parent.size > 0:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent, best_n)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: Aggressive pair golden descent
    n_iter = min(50000, int((deadline - time.time()) / 0.001))
    best_h, best_c5 = pair_golden_descent(best_h, best_n, n_iter, rng, deadline)

    # Phase 2: Subgradient steps (fast O(n) per step)
    if time.time() < deadline - 60:
        best_h, best_c5 = subgradient_steps(best_h, best_n, 50000, 0.01, deadline, top_M=5)

    # Phase 3: Active-set SLSQP with progressive focus
    if time.time() < deadline - 80:
        best_h, best_c5 = iterative_active_set(best_h, best_n, deadline)

    # Phase 4: More pair golden descent
    if time.time() < deadline - 40:
        n_iter = min(40000, int((deadline - time.time()) / 0.001))
        best_h, best_c5 = pair_golden_descent(best_h, best_n, n_iter, rng, deadline)

    # Phase 5: Binary round-trip for exploration
    if time.time() < deadline - 200:
        h_bin = (best_h > 0.5).astype(float)
        n_bin = len(h_bin)
        h_bin = make_feasible(h_bin, n_bin)
        h_bin, c5_bin = binary_sa(n_bin, rng, 30000, 0.6, 0.005, deadline)
        # Relax binary
        h_relaxed, c5_relaxed = pair_golden_descent(h_bin, n_bin, 30000, rng, deadline)
        if c5_relaxed < best_c5:
            best_h = h_relaxed
            best_c5 = c5_relaxed
            best_n = n_bin
        if time.time() < deadline - 40:
            best_h, best_c5 = iterative_active_set(best_h, best_n, deadline)
            best_h, best_c5 = pair_golden_descent(best_h, best_n, 20000, rng, deadline)

    # Phase 6: Multi-resolution exploration
    for n_try in [best_n + 40, best_n + 80, best_n + 160]:
        if time.time() > deadline - 120:
            break
        if n_try % 2 != 0:
            n_try += 1
        if n_try > 400:
            continue
        h_interp = make_feasible(interpolate_to_n(best_h, n_try), n_try)
        h_new, c5_new = pair_golden_descent(h_interp, n_try, 25000, rng, deadline)
        if time.time() < deadline - 40:
            h_new, c5_new = iterative_active_set(h_new, n_try, deadline)
            h_new, c5_new = pair_golden_descent(h_new, n_try, 15000, rng, deadline)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_n = n_try

    # Phase 7: Subgradient polish at final resolution
    if time.time() < deadline - 30:
        best_h, best_c5 = subgradient_steps(best_h, best_n, 30000, 0.005, deadline, top_M=3)

    # Phase 8: Final pair descent polish
    if time.time() < deadline - 15:
        n_iter = min(30000, int((deadline - time.time()) / 0.001))
        best_h, best_c5 = pair_golden_descent(best_h, best_n, n_iter, rng, deadline)

    # Phase 9: Random perturbation for last-mile improvement
    if time.time() < deadline - 10:
        n_trials = min(10000, int((deadline - time.time()) / 0.005))
        best_h, best_c5 = random_perturb(best_h, best_n, rng, n_trials, 0.02, deadline)
        if time.time() < deadline - 5:
            best_h, best_c5 = pair_golden_descent(best_h, best_n, 10000, rng, deadline)

    # Final feasibility enforcement
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-6

    return (best_h, best_c5, best_n)
