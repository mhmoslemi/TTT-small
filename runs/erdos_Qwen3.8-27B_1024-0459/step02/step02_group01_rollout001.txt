def _compute_f(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return np.empty(0, dtype=float)

    ac = np.correlate(h, h, mode="full")
    o = np.asarray(ac[n - 1:], dtype=float)

    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])

    f = np.empty(2 * n - 1, dtype=float)
    ks = np.arange(n)
    L = n - ks
    f[n - 1:n - 1 + n] = pref[L] - o

    if n > 1:
        k = np.arange(1, n)
        f[n - 1 - k] = total - pref[k] - o[k]

    f = np.maximum(f, 0.0)
    return f

        return 0.0
    f = _compute_f(h)
    return float(np.max(f) * (2.0 / float(n)))


def _compute_c5_eval(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return 0.0
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def _uniform_weights(n):
    return np.full(n, 1.0 / float(n), dtype=float)


def _smooth_weights(f, beta, topk):
    n = len(f)
    if n == 0:
        return np.empty(0, dtype=float)

    if topk is not None and int(topk) > 0:
        k = min(int(topk), n)
        if k <= 0:
            return np.zeros(n, dtype=float)
        if k >= n:
            return _uniform_weights(n)
        idx = np.argpartition(f, -k)[-k:]
        w = np.zeros(n, dtype=float)
        w[idx] = 1.0 / float(k)
        return w

    beta = float(beta)
    if beta <= 0.0 or not np.isfinite(beta):
        return _uniform_weights(n)

    m = float(np.max(f))
    diff = beta * (f - m)
    diff = np.clip(diff, -50.0, 0.0)
    e = np.exp(diff)
    s = float(np.sum(e))
    if s <= 1e-300:
        w = np.zeros(n, dtype=float)
        w[int(np.argmax(f))] = 1.0
        return w
    return e / s


def _topk_weights(f, k):
    n = len(f)
    w = np.zeros(n, dtype=float)
    if n == 0:
        return w
    center = n // 2
    k = int(k)
    if k <= 0:
        return w
    if k >= n - 1:
        if n > 1:
            w[:] = 1.0 / float(n - 1)
            w[center] = 0.0
        else:
            w[0] = 1.0
        return w
    f2 = f.copy()
    f2[center] = -1e30
    idx = np.argpartition(f2, -k)[-k:]
    w[idx] = 1.0 / float(k)
    if w[center] != 0.0:
        w[center] = 0.0
        s = float(np.sum(w))
        if s > 0.0:
            w *= 1.0 / s
    return w


def _weighted_gradient(h, w):
    h = np.asarray(h, dtype=float)
    n = len(h)
    grad = np.zeros(n, dtype=float)
    if n == 0:
        return grad

    eps = 1e-14
    c0 = float(w[n - 1])
    if abs(c0) > eps:
        grad += c0 * (1.0 - 2.0 * h)

    for k in range(1, n):
        L = n - k
        wp = float(w[n - 1 + k])
        if wp > eps:
            grad[:L] += wp * (1.0 - h[k:])
            grad[k:] += -wp * h[:L]
        wm = float(w[n - 1 - k])
        if wm > eps:
            grad[k:] += wm * (1.0 - h[:L])
            grad[:L] += -wm * h[k:]

    return grad


def _obj_jac(x):
    x = np.asarray(x, dtype=float)
    f = _compute_f(x)
    w = _smooth_weights(f, _BETA_OPT, _TOPK_OPT)
    obj = float(np.dot(f, w) * _DX_OPT)
    grad = _weighted_gradient(x, w) * _DX_OPT
    return obj, grad


def _eq(x):
    x = np.asarray(x, dtype=float)
    return float(np.sum(x) - 0.5 * float(len(x)))


def _eq_jac(x):
    return np.ones(np.shape(x), dtype=float)

def _slsqp_once(n, h0, beta, topk, maxiter):
    global _N_OPT, _BETA_OPT, _TOPK_OPT, _DX_OPT

    h0 = _project_feasible(h0)
    _N_OPT = int(n)
    _BETA_OPT = float(beta)
    _TOPK_OPT = int(topk)
    _DX_OPT = 2.0 / float(n)

    bounds = [(0.0, 1.0)] * int(n)
    cons = [{"type": "eq", "fun": _eq, "jac": _eq_jac}]
    opts = {"maxiter": int(maxiter), "ftol": 1e-12, "disp": 0}

    try:
        res = minimize(
            _obj_jac,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options=opts,
        )
    except Exception:
        return h0

    x = getattr(res, "x", None)
    if x is None:
        return h0
    return _project_feasible(x)


def _contin_opt(n, h0, deadline):
    h = _project_feasible(h0)
    best_h = h.copy()
    best_c = _compute_c5_fast(h)

    for beta, topk, maxiter in _stages_for(n):
        if time.time() >= deadline - 18.0:
            break
        hnew = _slsqp_once(n, h, beta, topk, maxiter)
        cnew = _compute_c5_fast(hnew)
        if cnew < best_c - 1e-13:
            h = hnew
            best_c = cnew
            best_h = hnew.copy()

    return best_h, best_c


def _nearest_binary(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = n // 2 if n % 2 == 0 else int(round(0.5 * float(n)))

    bits = (h >= 0.5).astype(float)
    cnt = int(np.sum(bits))

    if cnt > target:
        ones = np.where(bits > 0.5)[0]
        order = np.argsort(h[ones])
        for i in range(cnt - target):
            bits[ones[order[i]]] = 0.0
    elif cnt < target:
        zeros = np.where(bits <= 0.5)[0]
        order = np.argsort(-h[zeros])
        for i in range(target - cnt):
            bits[zeros[order[i]]] = 1.0

    return bits


def _balance_bits(b, n):
    b = np.asarray(b, dtype=float).copy()
    target = n // 2 if n % 2 == 0 else int(round(0.5 * float(n)))
    cnt = int(np.round(np.sum(b)))

    if cnt > target:
        ones = np.where(b > 0.5)[0]
        for idx in ones[cnt - target:]:
            b[int(idx)] = 0.0
    elif cnt < target:
        zeros = np.where(b <= 0.5)[0]
        for idx in zeros[:target - cnt]:
            b[int(idx)] = 1.0

    return b


def _thue_bits(n):
    arr = np.zeros(int(n), dtype=np.int8)
    for i in range(1, int(n)):
        arr[i] = int(arr[i >> 1] ^ (i & 1))
    return _balance_bits(arr.astype(float), n)


def _random_bits(rng, n):
    if rng is None or int(n) % 2 != 0:
        return None
    b = np.zeros(int(n), dtype=float)
    idx = rng.choice(int(n), int(n) // 2, replace=False)
    b[idx] = 1.0
    return b


def _pattern_bits(n, period):
    period = max(2, int(period))
    half = period // 2
    mods = np.arange(int(n)) % period
    b = (mods >= half).astype(float)
    return _balance_bits(b, n)


def _binary_hill(n, bits, deadline):
    if int(n) % 2 != 0:
        return bits.astype(float), float("inf")

    h = np.zeros(int(n), dtype=float)
    h[bits > 0.5] = 1.0
    target = n // 2
    cnt = int(np.sum(h))
    if cnt > target:
        ones = np.where(h > 0.5)[0]
        for idx in ones[cnt - target:]:
            h[int(idx)] = 0.0
    elif cnt < target:
        zeros = np.where(h <= 0.5)[0]
        for idx in zeros[:target - cnt]:
            h[int(idx)] = 1.0

    best_h = h.copy()
    best_c = _compute_c5_fast(h)
    f = _compute_f(h)

    klist = [64, 32, 16, 8, 4, 2, 1]
    no_improve = 0
    it = 0

    while time.time() < deadline - 5.0:
        K = klist[it % len(klist)]
        w = _topk_weights(f, K)
        if not np.any(w):
            break

        g = _weighted_gradient(h, w)
        gains = np.where(h > 0.5, -g, g)

        min_gain = float(np.min(gains))
        if min_gain > -1e-10:
            no_improve += 1
            if no_improve > 60:
                break
            order = np.argsort(gains)
            pick = order[it % min(10, len(order))]
            p = int(pick)
        else:
            m = min(30, len(gains))
            order = np.argsort(gains)[:m]
            p = int(order[it % m])

        if h[p] > 0.5:
            newv = 0.0
        else:
            newv = 1.0

        htry = h.copy()
        htry[p] = newv
        ctry = _compute_c5_fast(htry)

        if ctry < best_c - 1e-12:
            h = htry
            best_c = ctry
            best_h = htry.copy()
            f = _compute_f(h)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break

        it += 1

    return best_h, best_c


def _scored_key(item):
    return float(item[0])


def _try_evaluator(h):
    val = None
    try:
        val = evaluate_erdos_solution(h)
    except Exception:
        try:
            val = evaluate_erdos_solution()
        except Exception:
            val = None

    if val is None:
        return None
    try:
        val = float(val)
    except Exception:
        return None
    if not np.isfinite(val):
        return None
    return val


def run(seed=42, budget_s=1300, **kwargs):
    global _RNG, _N_OPT, _BETA_OPT, _TOPK_OPT, _DX_OPT

    t0 = time.time()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = t0 + budget
    _RNG = np.random.RandomState(int(seed))

    parent = _safe_parent()
    best_h = None
    best_c5 = float("inf")
    best_n = 0

    if parent is not None and len(parent) > 0:
        best_h = _project_feasible(parent)
        best_n = len(best_h)
    if best_h is None:
        best_n = 192
        best_h = np.ones(best_n, dtype=float) * 0.5

    best_c5 = _compute_c5_eval(best_h)

    reserve = 8.0

    n_list = []
    for n in [best_n, 192, 256, 320, 384, 160, 448]:
        nn = int(n)
        if 64 <= nn <= 512 and nn not in n_list:
            n_list.append(nn)

    for n in n_list:
        if time.time() > deadline - 70.0:
            break

        remaining = deadline - time.time() - reserve
        if remaining < 50.0:
            break

        h0 = _resample(best_h, n)
        inits = [h0]

        inits.append(_project_feasible(h0 + 0.08 * _RNG.randn(n)))
        inits.append(_project_feasible(h0 + 0.20 * _RNG.randn(n)))

        inits.append(_nearest_binary(h0))

        if n % 2 == 0:
            inits.append(_random_bits(_RNG, n))

        t = np.arange(n, dtype=float) / float(n)
        freq = 2 + int(_RNG.randint(0, 7))
        phase = 2.0 * np.pi * float(_RNG.rand())
        hf = 0.75 * h0 + 0.25 * (0.5 + 0.35 * np.cos(2.0 * np.pi * float(freq) * t + phase))
        inits.append(_project_feasible(hf))

        scored = []
        for hh in inits:
            hh = _project_feasible(hh)
            cc = _compute_c5_fast(hh)
            scored.append((cc, hh))
        scored.sort(key=_scored_key)

        if n in (best_n, 192):
            n_budget = min(360.0, max(60.0, remaining * 0.45))
        else:
            n_budget = min(180.0, max(45.0, remaining * 0.25))

        allocs = [0.50, 0.25, 0.15, 0.10]
        for idx in range(min(4, len(scored))):
            if time.time() > deadline - 40.0:
                break
            tl = n_budget * allocs[idx] if idx < len(allocs) else 0.0
            if tl < 20.0:
                tl = min(20.0, max(0.0, deadline - time.time() - reserve))
            if tl <= 0.0:
                break

            stage_deadline = min(time.time() + tl, deadline - reserve)
            hopt, c_opt = _contin_opt(n, scored[idx][1], stage_deadline)

            if c_opt < best_c5 - 1e-12:
                best_c5 = c_opt
                best_h = hopt.copy()
                best_n = n

            if n % 2 == 0:
                hb = _nearest_binary(hopt)
                cb = _compute_c5_fast(hb)
                if cb < best_c5 - 1e-12:
                    best_c5 = cb
                    best_h = hb.copy()
                    best_n = n

    if time.time() < deadline - 60.0:
        bin_deadline = time.time() + min(400.0, max(60.0, (deadline - time.time()) * 0.45))

        bin_ns = []
        for n in [best_n, 256, 384, 192, 320]:
            nn = int(n)
            if nn % 2 == 0 and 128 <= nn <= 448 and nn not in bin_ns:
                bin_ns.append(nn)

        for n in bin_ns:
            if time.time() > deadline - 30.0:
                break

            best_at_n = _resample(best_h, n)
            bit_inits = [
                _nearest_binary(best_at_n),
                _thue_bits(n),
                _random_bits(_RNG, n),
                _pattern_bits(n, 4),
                _pattern_bits(n, 8),
                _pattern_bits(n, 16),
            ]

            for j, bits in enumerate(bit_inits):
                if bits is None:
                    continue
                if time.time() > deadline - 20.0:
                    break

                if j == 0:
                    limit = min(150.0, max(25.0, bin_deadline - time.time()))
                else:
                    limit = min(60.0, max(15.0, bin_deadline - time.time()))

                hb, cb = _binary_hill(n, bits, time.time() + limit)
                if cb < best_c5 - 1e-12:
                    best_c5 = cb
                    best_h = hb.copy()
                    best_n = n

    if time.time() < deadline - 30.0:
        final_cont_deadline = deadline - 8.0
        hpol, cpol = _contin_opt(best_n, best_h, final_cont_deadline)
        if cpol < best_c5 - 1e-12:
            best_c5 = cpol
            best_h = hpol.copy()

    if best_n % 2 == 0 and time.time() < deadline - 15.0:
        hb = _nearest_binary(best_h)
        cb = _compute_c5_fast(hb)
        if cb < best_c5 - 1e-12:
            best_c5 = cb
            best_h = hb.copy()

        if time.time() < deadline - 10.0:
            hb2, cb2 = _binary_hill(best_n, hb, deadline - 4.0)
            if cb2 < best_c5 - 1e-12:
                best_c5 = cb2
                best_h = hb2.copy()

    best_h = _project_feasible(best_h)

    final_c5 = min(_compute_c5_eval(best_h), _compute_c5_fast(best_h))
    ev = _try_evaluator(best_h)
    if ev is not None and ev < final_c5:
        final_c5 = ev

    if final_c5 > best_c5:
        final_c5 = best_c5

    return (best_h, float(final_c5), int(best_n))
