def compute_c5(h):

def all_lag_values(h):
    """Return (values, lags) for all lag values."""
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    
    # Use np.correlate for efficiency
    corr = np.correlate(h, 1.0 - h, mode="full")
    # corr has 2n-1 elements, from lag -(n-1) to +(n-1)
    # corr[j] corresponds to lag j - (n-1)
    lags = np.arange(-(n-1), n)
    vals = corr * dx
    return vals, lags

def lag_value(h, k):
    """Compute overlap at a single lag k (can be negative)."""
    n = len(h)
    dx = 2.0 / float(n)
    if k == 0:
        return float(np.sum(h - h*h)) * dx
    if k > 0:
        m = n - k
        return float(np.sum(h[:m] - h[:m]*h[k:])) * dx
    else:
        k = -k
        return float(np.sum(h[k:] - h[k:]*h[:-k])) * dx

def lag_grad(h, k):
    """Gradient of lag_value(h, k) w.r.t. h."""
    n = len(h)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = (1.0 - 2.0*h) * dx
    elif k > 0:
        m = n - k
        g[:m] += (1.0 - h[k:]) * dx
        g[k:] += -h[:m] * dx
    else:
        k = -k
        g[k:] += (1.0 - h[:-k]) * dx
        g[:-k] += -h[k:] * dx
    return g

        return j

        j[:-1] = 1.0

def thue_morse_n(n):
    """Generate Thue-Morse-like sequence of length n with sum n/2."""
    # Thue-Morse: t(i) = popcount(i) % 2
    tm = np.zeros(n)
    for i in range(n):
        # Count bits
        x = i
        cnt = 0
        while x:
            cnt += x & 1
            x >>= 1
        tm[i] = 1.0 if cnt % 2 == 0 else 0.0
    # This gives sum = n/2 for n = 2^k
    # For other n, we need to adjust
    return project_feasible(tm, n)

            best_c = ct
            cur_h = ht
            cur_c = ct
            temp = min(0.01, temp * 1.1)
        elif ct < cur_c:
            cur_h = ht
            cur_c = ct
        else:
            d = ct - cur_c
            if 0.0 < d < temp * 5:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = ht
                    cur_c = ct
        temp = max(temp * 0.997, min_temp)
        # Occasionally try a larger move
        if it % 200 == 0:
            L2 = int(rng.randint(max(1, n // 10), max(2, n // 3)))
            st2 = int(rng.randint(0, max(1, n - L2)))
            idx2 = np.arange(st2, min(n, st2 + L2))
            delta2 = rng.randn(len(idx2)) * step_size * 3
            delta2 -= delta2.mean()
            ht2 = cur_h.copy()
            ht2[idx2] += delta2
            ht2 = project_feasible(ht2, n)
            ct2 = compute_c5(ht2)
            if ct2 < cur_c:
                cur_h = ht2
                cur_c = ct2
                if ct2 < best_c:
                    best_h = ht2.copy()
                    best_c = ct2
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(120.0, budget_s - 45.0)

    parent = get_parent_values()
    if parent is not None and len(parent) > 0:
        best_h = parent.copy().astype(float)
        best_n = int(parent.size)
        best_c = compute_c5(best_h)
    else:
        best_n = 160
        best_h = np.full(best_n, 0.5)
        best_h = project_feasible(best_h, best_n)
        best_c = compute_c5(best_h)

    # Determine resolutions to try
    res_list = [best_n]
    for d in [8, 16, 32, 48, 64, 96]:
        for nn in [best_n + d, best_n - d]:
            if 48 <= nn <= 400 and nn not in res_list:
                res_list.append(nn)
    for nn in [128, 192, 256, 320, 384]:
        if nn not in res_list and 48 <= nn <= 400:
            res_list.append(nn)
    res_list.sort(key=lambda x: abs(x - best_n))
    res_list = res_list[:7]

    # Phase 1: Multi-resolution exploration
    phase1_end = min(hard_end - 250.0, start_time + budget_s * 0.35)
    for nn in res_list:
        if time.time() > phase1_end:
            break
        rem = phase1_end - time.time()
        if rem < 40:
            break
        n_starts = 3
        per_start = min(rem / float(n_starts + 1), 55.0)
        per_start = max(15.0, per_start)
        starts = make_starts(nn, best_h, rng)
        for si, s in enumerate(starts[:n_starts]):
            if time.time() > phase1_end - 5:
                break
            t_dl = min(time.time() + per_start, phase1_end - 5)
            hh, cc = optimize_one_start(s, nn, t_dl)
            if cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = nn

    # Phase 2: Focused exploitation on best
    phase2_end = min(hard_end - 120.0, start_time + budget_s * 0.7)
    # Multiple passes on the best resolution
    for pass_i in range(3):
        if time.time() > phase2_end:
            break
        rem = phase2_end - time.time()
        if rem < 30:
            break
        # Interpolate best to same n (already there), make slight variants
        n = best_n
        variants = [best_h.copy()]
        if n > 1:
            variants.append(project_feasible(best_h + (best_h[::-1] - best_h) * 0.1, n))
            variants.append(project_feasible(np.roll(best_h, 1), n))
        per_v = rem / float(len(variants) + 0.5)
        per_v = min(per_v, 60.0)
        per_v = max(20.0, per_v)
        for vi, v in enumerate(variants):
            if time.time() > phase2_end - 5:
                break
            t_dl = min(time.time() + per_v, phase2_end - 5)
            hh, cc = optimize_one_start(v, n, t_dl)
            if cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = n

    # Phase 3: Final intensive refinement
    n = best_n
    if time.time() < hard_end - 50 and 2 <= n <= 400:
        # High-alpha gradient
        for alpha in [200, 400, 800]:
            if time.time() > hard_end - 30:
                break
            t_dl = min(time.time() + 30, hard_end - 15)
            hh, cc = projected_grad_optimize(best_h, n, alpha, t_dl, max_iter=200)
            if cc < best_c:
                best_h = hh
                best_c = cc

    # Final active set with many lags
    if time.time() < hard_end - 20 and 2 <= n <= 400:
        vals, lags = all_lag_vals_and_lags(best_h)
        idx = np.argsort(-vals)
        top_val = float(vals[idx[0]])
        active = [int(lags[i]) for i in idx if vals[i] > top_val - 0.001][:40]
        if len(active) < 6:
            active = [int(lags[i]) for i in idx[:16]]
        t0 = best_c + 1e-6
        t_dl = min(time.time() + 30, hard_end - 5)
        hh, cc = active_slsqp(best_h, n, active, t0, t_dl, max_iter=100)
        if cc < best_c:
            best_h = hh
            best_c = cc

    # Very fine local perturbation
    if time.time() < hard_end - 10 and 2 <= n <= 300:
        t_dl = min(time.time() + 40, hard_end - 2)
        hh, cc = local_perturb_search(best_h, n, t_dl, seed + 7777, step_size=0.008)
        if cc < best_c:
            best_h = hh
            best_c = cc

    # Ultra-fine polish
    if time.time() < hard_end - 3:
        t_dl = hard_end - 1
        hh, cc = projected_grad_optimize(best_h, best_n, 500, t_dl, max_iter=100)
        if cc < best_c:
            best_h = hh
            best_c = cc

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
