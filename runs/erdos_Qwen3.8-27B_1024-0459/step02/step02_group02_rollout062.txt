def projected_subgrad(h, n, n_iter=10000, lr_init=0.01, lr_decay=0.999, timeout=30.0):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_t = time.time()
    dx = 2.0 / n
    target_sum = n / 2.0
    
    lr = lr_init
    for it in range(n_iter):
        if time.time() - start_t > timeout:
            break
        
        r = np.correlate(h, 1.0 - h, mode='full')
        r_max = np.max(r)
        # Find active shifts (within 1% of max)
        active = r > r_max - 0.01 * max(r_max, 1e-10)
        
        # Compute subgradient: average gradient over active shifts
        g = np.zeros(n)
        for s_idx in np.where(active)[0]:
            s = s_idx - (n - 1)  # actual shift
            # ∂r_s/∂h[a] = (1-h[a+s])*[valid] - h[a-s]*[valid]
            if s >= 0:
                # a ranges from 0 to n-1-s
                # First factor: (1-h[a+s]) for a in [0, n-1-s]
                g[:n-s] += (1.0 - h[s:])  # (1-h[a+s]) for a=0..n-s-1
                # Second factor: -h[a-s] for a in [s, n-1]
                g[s:] -= h[:n-s]  # h[a-s] for a=s..n-1
            else:
                s_abs = -s
                # a ranges from s_abs to n-1
                # First factor: (1-h[a+s]) = (1-h[a-s_abs]) for a in [s_abs, n-1]
                g[s_abs:] += (1.0 - h[:n-s_abs])  # (1-h[a-s_abs]) for a=s_abs..n-1
                # Second factor: -h[a-s] = -h[a+s_abs] for a in [0, n-1-s_abs]
                g[:n-s_abs] -= h[s_abs:]  # h[a+s_abs] for a=0..n-s_abs-1
        
        g /= np.sum(active)  # average over active shifts
        
        # Gradient descent step
        h = h - lr * g
        h = np.clip(h, 0.0, 1.0)
        
        # Fix sum
        diff = target_sum - np.sum(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = int(np.sum(mask))
            if cnt == 0:
                break

        h = np.clip(h, 0.0, 1.0)
    return h

        h[i] = 1.0 if bin(idx).count('1') % 2 == 1 else 0.0

    idx = rng.permutation(n)

    return h

    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    
    t_start_time = time.time()
    log_ratio = np.log(t_end / t_start) if t_start > 0 else -20
    
    for it in range(n_iter):
        if time.time() - t_start_time > timeout:
            break
        if not ones or not zeros:
            break
        
        temp = t_start * np.exp(log_ratio * it / max(n_iter, 1))
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - cur_c5
        accept = delta < 0 or (temp > 1e-18 and rng.rand() < np.exp(-delta / temp))
        
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def continuous_refine(h_init, n, rng, timeout=120):
    h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    target = n / 2.0
    t0 = time.time()
    
    # Strategy 1: Fractional adjustments at worst-shift positions
    for round_num in range(20):
        if time.time() - t0 > timeout - 5:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        # Find active shifts
        active_shifts = np.where(corr > max_val - 0.1)[0]
        
        # For each active shift, find contributing positions
        # and compute gradient
        grad = np.zeros(n)
        for si in active_shifts:
            s = si - (n - 1)
            if s >= 0:
                # r_s = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
                # dr_s/dh[i] = (1-h[i+s]) for i in [0, n-1-s]
                #           + (-h[i-s]) for i in [s, n-1]
                valid_i = np.arange(0, n - s)
                grad[valid_i] += (1.0 - h[valid_i + s])
                valid_i2 = np.arange(s, n)
                grad[valid_i2] += (-h[valid_i2 - s])
            else:
                t = -s
                valid_i = np.arange(t, n)
                grad[valid_i] += (1.0 - h[valid_i - t])
                valid_i2 = np.arange(0, n - t)
                grad[valid_i2] += (-h[valid_i2 + t])
        
        grad /= len(active_shifts)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Line search
        for lr in [0.001, 0.005, 0.01, 0.02, 0.05]:
            h_try = h - lr * grad
            h_try = np.clip(h_try, 0.0, 1.0)
            diff = target - np.sum(h_try)
            if abs(diff) > 1e-12:
                mask = (h_try > 1e-6) & (h_try < 1.0 - 1e-6)
                cnt = int(np.sum(mask))
                if cnt > 0:
                    h_try[mask] += diff / cnt
                    h_try = np.clip(h_try, 0.0, 1.0)
            
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                break
    
    # Strategy 2: Random fractional perturbations
    for _ in range(50000):
        if time.time() - t0 > timeout - 2:
            break
        
        h_try = h.copy()
        n_mod = rng.randint(1, 5)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-0.1, 0.1)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = target - np.sum(h_try)
        mask = (h_try > 1e-4) & (h_try < 1.0 - 1e-4)
        cnt = int(np.sum(mask))

        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
    
    return best_h, best_c5
