        y = project_feasible(y, n)

        # Update best

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_logsumexp(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return m + np.log(total) / beta

def softmax_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s < 1e-300:
        return np.ones(len(corr)) / len(corr)
    return exps / s

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    weights = softmax_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return dx * grad

            c5_init = compute_c5(h0)
            cur = h0.copy()

            for beta in [20, 60, 150, 400]:
                if remaining() < 20:
                    break
                tl = min(35, remaining() * 0.25)
                mi = min(4000, int(tl * 80))
                try:
                    h_new, c5_new = fista_optimize(cur, n, beta, mi, tl)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = n
                        print("  FISTA n={}, beta={}, init={}, C5={:.6f}".format(n, beta, ii, best_c5))
                    if c5_new < best_c5_n:
                        best_c5_n = c5_new
                        best_h_n = h_new.copy()
                    cur = h_new
                except Exception:
                    break

            if remaining() > 20 and best_h_n is not None:
                h_bin = (best_h_n > 0.5).astype(float)
                n_ones = int(np.sum(h_bin))
                n_want = n // 2
                if n_ones > n_want:
                    oi = np.where(h_bin == 1)[0]
                    order = np.argsort(best_h_n[oi])[:n_ones - n_want]
                    h_bin[oi[order]] = 0.0
                elif n_ones < n_want:
                    zi = np.where(h_bin == 0)[0]
                    order = np.argsort(-best_h_n[zi])[:n_want - n_ones]
                    h_bin[zi[order]] = 1.0
                bin_c5 = compute_c5(h_bin)
                if bin_c5 < best_c5:
                    best_c5 = bin_c5
                    best_h = h_bin.copy()
                    best_n = n
                    print("  Bin n={}, C5={:.6f}".format(n, best_c5))
                if remaining() > 15:
                    tl_ls = min(20, remaining() * 0.2)
                    h_ls, c5_ls = binary_local_search_greedy(h_bin, n, tl_ls, rng)
                    if c5_ls < best_c5:
                        best_c5 = c5_ls
                        best_h = h_ls.copy()
                        best_n = n
                        print("  BinLS n={}, C5={:.6f}".format(n, c5_ls))

        if remaining() > 40 and best_n == n and np.all((best_h == 0) | (best_h == 1)):
            tl_sa = min(35, remaining() * 0.3)
            h_sa, c5_sa = binary_local_search_sa(best_h, n, tl_sa, rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                print("  SA n={}, C5={:.6f}".format(n, c5_sa))

    if remaining() > 30:
        for high_n in [500, 600]:
            if remaining() < 25:
                break
            h_hi = interp_h(best_h, high_n)
            for beta in [80, 250]:
                if remaining() < 15:
                    break
                try:
                    h_new, c5_new = fista_optimize(h_hi, high_n, beta, 3000, min(25, remaining() * 0.3))
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n = high_n
                        print("  HighRes n={}, C5={:.6f}".format(high_n, c5_new))
                except Exception:
                    pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
