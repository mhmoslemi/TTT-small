def compute_all_lag_values(h):

        lags[idx] = L
        idx += 1

        lags[idx] = -L
        idx += 1
    return vals, lags

def lag_gradient(h, l):
    n = len(h)
    if l == 0:
        return 1.0 - 2.0 * h
    L = abs(int(l))
    m = n - L
    low = h[:m]
    high = h[L:]
    g = np.zeros(n, dtype=float)
    if l > 0:
        g[:m] += 1.0 - high
        g[L:] += -low
    else:
        g[L:] += 1.0 - low
        g[:m] += -high
    return g

    v0 = (total - ac0) * dx
    
    m = v0
    r = 1.0
    
    for L in range(1, n):

        ac = float(np.dot(h[:mm], h[L:]))
        vp = (pref[mm] - ac) * dx
        vn = (total - pref[L] - ac) * dx

            delta = beta * (m - vp)
            if delta > -700.0:
                r = r * np.exp(delta)
            r += 1.0
            m = vp
        else:
            delta = beta * (vp - m)
            if delta > -700.0:
                r += np.exp(delta)
        
        if vn > m:

            m = vn

def local_block_perturb(h0, n, time_limit, seed, max_step=0.03):

    cur_c = c
    temp = 0.002
    min_temp = 1e-7
    
    max_trials = 5000
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * max_step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)

            cur_h = h_try
            cur_c = c_try
        else:

            if 0.0 < d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try

def lag_gradient(h, l):
    n = len(h)

    L = abs(int(l))

    high = h[L:]

    if l > 0:
        g[:m] += 1.0 - high
        g[L:] += -low
    else:
        g[L:] += 1.0 - low
        g[:m] += -high
    return g

    order = np.argsort(-vals)
    take = min(int(top_k), len(vals))
    near_mask = vals >= (top_val - margin)
    near = np.where(near_mask)[0]
    idx = np.unique(np.concatenate((order[:take], near)))
    idx = idx[np.argsort(-vals[idx])]
    idx = idx[:min(int(top_k) * 3, len(idx))]
    active = [int(lags[i]) for i in idx]
    return active, float(top_val)

        h2, c2 = subgradient_polish(best_h, best_n, max_iter=4000, lr0=0.0012)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            print("After subgrad: C5=%.6f t=%.1f" % (best_c, time.time() - start_time))

    # Phase 4: Second smooth + active round
    phase4_end = min(hard_end - 100.0, start_time + 700.0)
    if time.time() < phase4_end:
        for beta in (200, 400, 600):
            if time.time() > phase4_end - 30.0:
                break
            h2, c2 = optimize_smooth(best_h, best_n, beta, maxiter=35, ftol=1e-12)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        if time.time() < phase4_end - 10.0:
            best_h, best_c = refine_active_round(best_h, best_n, phase4_end - 5.0,
                                                 top_k_seq=(30, 60, 90), maxiter=50)
        print("After round 2: C5=%.6f t=%.1f" % (best_c, time.time() - start_time))

    # Phase 5: Third round focusing on very high beta
    phase5_end = min(hard_end - 80.0, start_time + 900.0)
    if time.time() < phase5_end:
        h2, c2 = optimize_smooth(best_h, best_n, 800, maxiter=30, ftol=1e-13)
        if c2 < best_c:
            best_h = h2
            best_c = c2
        if time.time() < phase5_end - 10.0:
            best_h, best_c = refine_active_round(best_h, best_n, phase5_end - 5.0,
                                                 top_k_seq=(40, 80, 120), maxiter=60)
        print("After round 3: C5=%.6f t=%.1f" % (best_c, time.time() - start_time))

    # Phase 6: Local search (block perturb)
    phase6_end = min(hard_end - 50.0, start_time + 1100.0)
    if time.time() < phase6_end:
        slot = min(80.0, phase6_end - time.time())
        h2, c2 = local_block_perturb(best_h, best_n, time.time() + slot, seed + 7)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            print("After block perturb: C5=%.6f t=%.1f" % (best_c, time.time() - start_time))

    # Phase 7: Pair swap
    phase7_end = min(hard_end - 30.0, start_time + 1180.0)
    if time.time() < phase7_end:
        slot = min(50.0, phase7_end - time.time())
        h2, c2 = pair_swap_perturb(best_h, best_n, time.time() + slot, seed + 13)
        if c2 < best_c:
            best_h = h2
            best_c = c2
            print("After pair swap: C5=%.6f t=%.1f" % (best_c, time.time() - start_time))

    # Phase 8: Try nearby resolutions
    phase8_end = min(hard_end - 50.0, time.time() + 200.0)
    if time.time() < phase8_end:
        for dn in (-8, 8, -16, 16):
            if time.time() > phase8_end - 20.0:
                break
            tn = best_n + dn
            if tn < 48 or tn > 260:
                continue
            h_init = interpolate_to_n(best_h, tn)
            h_init = project_feasible(h_init, tn)
            c_init = compute_c5(h_init)
            for beta in (100, 300):
                if time.time() > phase8_end - 15.0:
                    break
                h2, c2 = optimize_smooth(h_init, tn, beta, maxiter=30, ftol=1e-12)
                if c2 < best_c:
                    if c2 < best_c - 0.0001:
                        best_h = h2
                        best_c = c2
                        best_n = tn
                        print("  Res n=%d beta=%d: C5=%.6f" % (tn, beta, c2))
                    h_init = h2
            if time.time() < phase8_end - 10.0 and best_n == tn:
                best_h, best_c = refine_active_round(best_h, best_n, time.time() + 30.0,
                                                     top_k_seq=(30, 60), maxiter=40)

    # Phase 9: Final subgradient + local polish
    if time.time() < hard_end - 15.0:
        h2, c2 = subgradient_polish(best_h, best_n, max_iter=3000, lr0=0.0006)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    if time.time() < hard_end - 5.0:
        h2, c2 = local_block_perturb(best_h, best_n, hard_end - 3.0, seed + 99, max_step=0.015)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    print("FINAL: n=%d C5=%.6f t=%.1f" % (best_n, best_c, time.time() - start_time))
    return (best_h, float(best_c), int(best_n))
