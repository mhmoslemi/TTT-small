The optimal g should have high autocorrelation at all lags. This is related to "binary sequences with good autocorrelation properties" - Barker sequences, m-sequences, etc.

∂f/∂h[j] = (dx/β) * (sum_k exp(β*c_k) * ∂c_k/∂h[j]) / (sum_k exp(β*c_k))

where w_k = exp(β*c_k) / sum_k exp(β*c_k) = softmax(β*c_k)_k

And ∂c_k/∂h[j]:
Let's denote the full correlation output as C[m] for m = 0, 1, ..., 2n-2.
C[n-1] = sum_i h[i]*(1-h[i])  [lag 0]
C[n-1+m] = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])  [right, lag m]
C[n-1-m] = sum_{i=m}^{n-1} h[i]*(1-h[i-m])  [left, lag m]

∂C[n-1]/∂h[j] = 1 - 2h[j]

∂C[n-1+m]/∂h[j] for m >= 1:
- h[j] appears in h[i+m] when i = j-m (if 0 <= j-m <= n-1-m, i.e., m <= j <= n-m): contribution (1-h[j-m])
- h[j] appears in (1-h[i]) when i = j (if 0 <= j <= n-1-m, i.e., m <= n-1-j): contribution -h[j+m]
So: ∂C[n-1+m]/∂h[j] = (1-h[j-m])*[m<=j] - h[j+m]*[m<=n-1-j]

∂C[n-1-m]/∂h[j] for m >= 1:
- h[j] appears in h[i] when i = j (if m <= j <= n-1, i.e., m <= j): contribution (1-h[j-m])
- h[j] appears in (1-h[i-m]) when i-m = j, i = j+m (if j+m <= n-1, i.e., m <= n-1-j): contribution -h[j+m]
So: ∂C[n-1-m]/∂h[j] = (1-h[j-m])*[m<=j] - h[j+m]*[m<=n-1-j]

Same formula for left and right! Good.

            inits.append(project_feasible(hp, n))

        # Structured oscillation
        t_arr = np.arange(n) / n
        for freq in [2, 3, 5, 7]:
            phase = rng.rand() * 2.0 * np.pi
            hf = 0.5 + 0.35 * np.cos(2.0 * np.pi * freq * t_arr + phase)
            inits.append(project_feasible(hf, n))

        # Block pattern
        if n % 4 == 0:
            block = n // 4
            hb2 = np.zeros(n)
            for bi in range(n // block):
                if bi % 2 == 0:
                    hb2[bi*block:(bi+1)*block] = 1.0
            inits.append(project_feasible(hb2 + 0.1*rng.randn(n), n))

        # Random binary
        if n % 2 == 0:
            hrb = np.zeros(n)
            hrb[rng.choice(n, n//2, replace=False)] = 1.0
            hrb = hrb + 0.05 * rng.randn(n)
            inits.append(project_feasible(hrb, n))

        bounds_l = [(0.0, 1.0)] * n
        cons_l = [{'type': 'eq', 'fun': lambda x, nn=n: np.sum(x) - nn/2.0}]

        # Determine time per initialization
        n_inits = len(inits)
        time_per_init = max(15.0, min(90.0, tl() / max(1, n_inits) * 0.7))

        for ii, h0 in enumerate(inits):
            if tl() < 15:
                break
            h_cur = project_feasible(h0, n)
            alloc = min(time_per_init, tl() - 10)
            if alloc < 10:
                break
            h_opt = slsqp_optimize(h_cur, n, alloc, bounds_l, cons_l)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

        # Coordinate descent polish on best_h at this n
        if tl() > 20:
            h_cd, c5_cd = coordinate_descent(best_h, best_n, max_passes=20)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Final polishing at best_n
    if tl() > 15:
        h_pol = project_feasible(best_h, best_n)
        bounds_l = [(0.0, 1.0)] * best_n
        cons_l = [{'type': 'eq', 'fun': lambda x, nn=best_n: np.sum(x) - nn/2.0}]

        # Very high beta SLSQP
        global _g_n, _g_beta, _g_dx
        _g_n = best_n
        _g_dx = 2.0 / best_n
        for beta in [20000.0, 100000.0, 500000.0]:
            if tl() < 8:
                break
            _g_beta = beta
            try:
                res = minimize(_smooth_obj_grad, h_pol, jac=True, method='SLSQP',
                               bounds=bounds_l, constraints=cons_l,
                               options={'maxiter': 1500, 'ftol': 1e-17, 'iprint': 0})
                h_pol = project_feasible(res.x, best_n)
                c5_p = compute_c5(h_pol)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pol.copy()
                    h_pol = best_h.copy()
            except Exception:
                pass

    # Coordinate descent final
    if tl() > 10:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_passes=40)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Binary local search
    if tl() > 5:
        h_bl, c5_bl = binary_local_search(best_h, best_n, max_rounds=5)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            # Polish the binary solution
            h_pol2 = project_feasible(best_h, best_n)
            _g_n = best_n
            _g_dx = 2.0 / best_n
            _g_beta = 50000.0
            bounds_l2 = [(0.0, 1.0)] * best_n
            cons_l2 = [{'type': 'eq', 'fun': lambda x, nn=best_n: np.sum(x) - nn/2.0}]
            try:
                res2 = minimize(_smooth_obj_grad, h_pol2, jac=True, method='SLSQP',
                                bounds=bounds_l2, constraints=cons_l2,
                                options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0})
                h_pol2 = project_feasible(res2.x, best_n)
                c5_p2 = compute_c5(h_pol2)
                if c5_p2 < best_c5:
                    best_c5 = c5_p2
                    best_h = h_pol2.copy()
            except Exception:
                pass

    # Final coordinate descent
    if tl() > 5:
        h_cd2, c5_cd2 = coordinate_descent(best_h, best_n, max_passes=30)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
