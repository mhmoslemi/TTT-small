    for k in range(1, n):
        d = 0.0
        if i + k < n:
            d += h[i+k] - 1.0  # = -(1-h[i+k])
        if k <= i < n:
            d += h[i-k]
        if j + k < n:
            d += 1.0 - h[j+k]
        if k <= j < n:
            d += -h[j-k]
        delta[n - 1 - k] = d
    
    # For lag 0 (output index n-1): Δc_0 = 0 (already derived)
    return delta
```

def swap_delta_vectorized(h, i, j, n):
    """Vectorized swap delta computation for binary h."""
    k = np.arange(1, n)  # lags 1 to n-1
    
    # For positive lag k: Δc_k
    # Term 1: -(1-h[i-k]) for k <= i, else 0
    t1_pos = np.zeros(n-1)
    valid1 = k <= i
    t1_pos[valid1] = h[i - k[valid1]] - 1.0
    
    # Term 2: h[i+k] for i+k < n, else 0
    t2_pos = np.zeros(n-1)
    valid2 = i + k < n
    t2_pos[valid2] = h[i + k[valid2]]
    
    # Term 3: (1-h[j-k]) for k <= j, else 0
    t3_pos = np.zeros(n-1)
    valid3 = k <= j
    t3_pos[valid3] = 1.0 - h[j - k[valid3]]
    
    # Term 4: -h[j+k] for j+k < n, else 0
    t4_pos = np.zeros(n-1)
    valid4 = j + k < n
    t4_pos[valid4] = -h[j + k[valid4]]
    
    delta_pos = t1_pos + t2_pos + t3_pos + t4_pos
    
    # For negative lag k: Δc_{-k}
    # Term 1: -(1-h[i+k]) for i+k < n
    t1_neg = np.zeros(n-1)
    valid1 = i + k < n
    t1_neg[valid1] = h[i + k[valid1]] - 1.0
    
    # Term 2: h[i-k] for k <= i
    t2_neg = np.zeros(n-1)
    valid2 = k <= i
    t2_neg[valid2] = h[i - k[valid2]]
    
    # Term 3: (1-h[j+k]) for j+k < n
    t3_neg = np.zeros(n-1)
    valid3 = j + k < n
    t3_neg[valid3] = 1.0 - h[j + k[valid3]]
    
    # Term 4: -h[j-k] for k <= j
    t4_neg = np.zeros(n-1)
    valid4 = k <= j
    t4_neg[valid4] = -h[j - k[valid4]]
    
    delta_neg = t1_neg + t2_neg + t3_neg + t4_neg
    
    # Assemble full delta (length 2n-1)
    delta = np.zeros(2*n - 1)
    delta[n:] = delta_pos  # indices n to 2n-2 correspond to k=1 to n-1
    delta[:n-1] = delta_neg  # indices 0 to n-2 correspond to k'=-1 to -(n-1)
    # delta[n-1] = 0 (lag 0)
    
    return delta
```

def subgradient_optimize(h_init, n, max_iter=10000, lr=0.1, momentum=0.9, top_k=3):
    h = h_init.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        corr = compute_corr(h)
        cmax = np.max(corr)
        
        # Pick top-k active lags
        top_idx = np.where(corr >= cmax - 0.01)[0]
        if len(top_idx) > top_k:
            top_idx = top_idx[np.random.choice(len(top_idx), top_k, replace=False)]
        
        # Compute gradient as average of active lags' gradients
        grad = np.zeros(n)
        for k_abs in top_idx:
            k = k_abs - (n-1)  # convert to lag
            if k >= 0:
                # c_k = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
                # grad[j] = (1-h[j-k])*1[k<=j<n] - h[j+k]*1[j<n-k]
                for j in range(k, n):
                    grad[j] += (1 - h[j-k])
                for j in range(n-k):
                    grad[j] -= h[j+k]
            else:
                kp = -k
                # c_{-kp} = sum_{n=0}^{n-1-kp} h[n]*(1-h[n+kp])
                # grad[j] = (1-h[j+kp])*1[j<n-kp] - h[j-kp]*1[kp<=j<n]
                for j in range(n-kp):
                    grad[j] += (1 - h[j+kp])
                for j in range(kp, n):
                    grad[j] -= h[j-kp]
        
        grad /= len(top_idx)
        
        # Update with momentum
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_sum_box(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        # Adaptive learning rate
        if it % 1000 == 0 and it > 0:
            lr *= 0.8
    
    return best_h, best_c5
