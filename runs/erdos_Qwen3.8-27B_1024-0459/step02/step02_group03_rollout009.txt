        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n_on - n_want, replace=False)] = 0
    elif n_on < n_want:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, n_want - n_on, replace=False)] = 1
    
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return max(0.0, budget_s - (time.time() - start_time) - 5.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass
    
    if best_h is None:
        best_n = 200
        best_h = make_structured_init(best_n, rng)
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.6f}".format(best_n, best_c5))
    
    # Phase 1: Continuous optimization at parent resolution
    if remaining() > 200:
        for n_cont in [best_n, int(best_n * 1.25)]:
            if remaining() < 80:
                break
            if n_cont % 2 != 0:
                n_cont += 1
            
            h0 = interp_h(best_h, n_cont) if len(best_h) != n_cont else best_h.copy()
            h_opt, c5_opt = continuous_optimize(h0, n_cont, start_time, budget_s, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_cont
                print("  Continuous n={}, C5={:.6f}".format(n_cont, best_c5))
            
            # Also try from structured init
            if remaining() > 60:
                h_struct = make_structured_init(n_cont, rng)
                h_opt2, c5_opt2 = continuous_optimize(h_struct, n_cont, start_time, budget_s, rng)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2
                    best_n = n_cont
                    print("  Structured+Cont n={}, C5={:.6f}".format(n_cont, best_c5))
    
    # Phase 2: Binary local search at multiple resolutions
    n_binary_list = []
    for nn in [best_n, best_n - 10, best_n + 10, best_n + 20, best_n + 40, 200, 220, 240, 256, 280, 300]:
        if nn >= 80 and nn <= 400 and nn % 2 == 0:
            if nn not in n_binary_list:
                n_binary_list.append(nn)
    n_binary_list.sort()
    
    for n in n_binary_list:
        if remaining() < 60:
            break
        
        # Multiple binary inits for this n
        h_round_base = interp_h(best_h, n)
        h_bin = round_to_binary(h_round_base, n, rng)
        c5_start = compute_c5(h_bin)
        
        # Run multi-start local search
        n_starts = 3 if remaining() > 200 else 2
        iters = max(5000, int(remaining() * 30 / max(1, n_starts)))
        iters = min(iters, 15000)
        
        h_best_n, c5_best_n = multi_start_binary_search(
            h_bin, n, n_starts, iters, start_time, budget_s, rng
        )
        
        if c5_best_n < best_c5:
            best_c5 = c5_best_n
            best_h = h_best_n
            best_n = n
            print("  Binary n={}, C5={:.6f}".format(n, c5_best_n))
        
        # Also try structured init at this n
        if remaining() > 60:
            h_struct = make_structured_init(n, rng)
            h_ls, c5_ls = targeted_binary_search(h_struct, n, 8000, start_time, budget_s, rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls
                best_n = n
                print("  Structured n={}, C5={:.6f}".format(n, c5_ls))
    
    # Phase 3: Extended local search on best solution
    if remaining() > 60:
        h_ext, c5_ext = multi_start_binary_search(
            best_h, best_n, 5, 8000, start_time, budget_s, rng
        )
        if c5_ext < best_c5:
            best_c5 = c5_ext
            best_h = h_ext
            print("  Extended: C5={:.6f}".format(best_c5))
    
    # Phase 4: Simulated annealing
    if remaining() > 40:
        h_sa, c5_sa = simulated_annealing_binary(
            best_h, best_n, 0.005, 0.0001,
            int(min(15000, remaining() * 20)),
            start_time, budget_s, rng
        )
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
            print("  SA: C5={:.6f}".format(c5_sa))
    
    # Phase 5: Final local search polish
    if remaining() > 20:
        h_final, c5_final = targeted_binary_search(
            best_h, best_n, int(min(10000, remaining() * 25)),
            start_time, budget_s, rng
        )
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            print("  Final LS: C5={:.6f}".format(best_c5))
    
    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass
    
    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
