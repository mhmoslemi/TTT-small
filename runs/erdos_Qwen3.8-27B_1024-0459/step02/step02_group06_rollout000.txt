def pair_candidate_ts(T, lo, hi, c_zero_max, pos_inv_b, pos_c_over_b, neg_inv_b, neg_c_over_b, b1, c1, b2, c2, b0, c0):
    tol=1e-12
    L=lo; R=hi
    if c_zero_max > T + 1e-10:
        return ()
    if pos_inv_b.size:
        # max of T*inv - c_over_b
        # if no, skip
        up = float(np.max(T * pos_inv_b - pos_c_over_b))
        if up < R: R = up
        if L > R + 1e-11: return ()
    if neg_inv_b.size:
        low = float(np.max(T * neg_inv_b - neg_c_over_b))
        if low > L: L = low
        if L > R + 1e-11: return ()
    # convex 1
    for bb, cc in ((b1,c1),(b2,c2)):
       disc = bb*bb - 4.0*(cc - T)
       if disc < -1e-10: return ()
       if disc < 0: disc=0
       s=np.sqrt(disc)
       r1=(-bb-s)*0.5; r2=(-bb+s)*0.5
       if r1 > L: L=r1
       if r2 < R: R=r2
       if L > R + 1e-11: return ()
    disc0 = b0*b0 - 8.0*(T - c0)
    if disc0 >= 0:
       if disc0 < 0: disc0=0
       s=np.sqrt(disc0)
       rlo=(b0 - s)*0.25
       rhi=(b0 + s)*0.25
       cands=[]
       endA = R if R < rlo else rlo
       if L <= endA + 1e-11:
           cands.append((L + endA)*0.5)
       startB = L if L > rhi else rhi
       if startB <= R + 1e-11:

       # If rlo > R and rhi > R: A [L,R] valid endA=R. If rhi < L: B [L,R].
       if not cands:
           # as fallback if numerical, choose midpoint
           # But if no candidate, invalid
           return ()

def corr_counts(h):

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()

            else:

        )
    elif mode == 1:
        h = np.empty(n, dtype=float)
        seg = max(2, int(n) // 10)
        for i in range(0, n, seg):
            e = min(i + seg, n)
            h[i:e] = float(rng.uniform(0.0, 1.0))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.75, 0.25).astype(float)
        h += rng.normal(0.0, 0.05, n)
    else:
        f = int(rng.integers(2, 8))
        h = 0.5 + 0.42 * np.sin(2.0 * np.pi * f * x / 2.0 + float(rng.uniform(0.0, 2.0 * np.pi)))
    return project_density(h, 0.5 * n)


def gradient_active(h, c, top_frac=0.06, min_lags=8, tau=5.0):
    n = h.size
    m = c.size
    center = n - 1
    tmax = float(np.max(c))
    k_count = max(min_lags, int(np.ceil(m * top_frac)))
    k_count = min(k_count, m)
    top = np.argsort(c)[-k_count:]
    w = np.exp((c[top] - tmax) / max(1e-12, float(tau)))
    ws = float(np.sum(w))
    if ws <= 0.0:
        w = np.ones_like(w, dtype=float)
        ws = float(w.size)
    w = w / ws

    g = np.zeros(n, dtype=float)
    for ii in range(top.size):
        tt = int(top[ii])
        ww = float(w[ii])
        if ww <= 1e-14:
            continue
        k = tt - center
        if k >= 0:
            if k == 0:
                g += ww * (1.0 - 2.0 * h)
            else:
                nn = n - k
                if nn > 0:
                    g[k:] += ww * (1.0 - h[:nn])
                    g[:nn] -= ww * h[k:]
        else:
            ell = -k
            nn = n - ell
            if nn > 0:
                g[:nn] += ww * (1.0 - h[ell:])
                g[ell:] -= ww * h[:nn]
    g -= float(np.mean(g))
    return g


def add_pair_to_list(ordered, seen, n, a, b):
    a = int(a)
    b = int(b)
    if a == b:
        return
    if a < 0 or b < 0 or a >= n or b >= n:
        return
    if a > b:
        a, b = b, a
    code = a * n + b
    if code not in seen:
        seen.add(code)
        ordered.append(code)


def make_pair_candidates(h, c, top_d=6, grad_p=12, term_q=10, rand_p=100, rng=None, max_pairs=3200):
    n = int(h.size)
    m = int(c.size)
    center = n - 1
    ordered = []
    seen = set()

    top_count = max(10, m // 20)
    top_count = min(top_count, m)
    top_lags = np.argsort(c)[-top_count:]

    score = {}
    for tt in top_lags:
        tt = int(tt)
        k = tt - center
        if k != 0:
            d = abs(k)
            val = float(c[tt])
            old = score.get(d)
            if old is None or val > old:
                score[d] = val

    if score:
        d_items = []
        for d in score:
            d_items.append((score[d], d))
        d_items.sort(reverse=True)
        dists = []
        for _, d in d_items[:top_d]:
            dists.append(d)
        for d in dists:
            if len(ordered) >= max_pairs:
                break
            if 1 <= d < n:
                for i in range(n - d):
                    add_pair_to_list(ordered, seen, n, i, i + d)
                    if len(ordered) >= max_pairs:
                        break

    if len(ordered) < max_pairs:
        g = gradient_active(h, c, top_frac=0.06, min_lags=8, tau=5.0)
        low = np.argsort(g)[:grad_p]
        high = np.argsort(-g)[:grad_p]
        for aa in low:
            for bb in high:
                if len(ordered) >= max_pairs:
                    break
                add_pair_to_list(ordered, seen, n, int(aa), int(bb))

    if len(ordered) < max_pairs:
        best_lags = top_lags[-min(8, top_count):]
        for tt in best_lags:
            tt = int(tt)
            k = tt - center
            if k > 0:
                length = n - k
                if length <= 0:
                    continue
                terms = h[k:] * (1.0 - h[:length])
                if terms.size == 0:
                    continue
                qq = min(term_q, terms.size)
                order = np.argsort(terms)[-qq:]
                for pos in order:
                    if len(ordered) >= max_pairs:
                        break
                    add_pair_to_list(ordered, seen, n, int(pos), int(pos) + k)
            elif k < 0:
                ell = -k
                length = n - ell
                if length <= 0:
                    continue
                terms = h[:length] * (1.0 - h[ell:])
                if terms.size == 0:
                    continue
                qq = min(term_q, terms.size)
                order = np.argsort(terms)[-qq:]
                for pos in order:
                    if len(ordered) >= max_pairs:
                        break
                    add_pair_to_list(ordered, seen, n, int(pos) + ell, int(pos))

    if rng is not None and rand_p > 0 and len(ordered) < max_pairs:
        hsort = np.argsort(h)
        half = max(1, n // 2)
        low_idx = hsort[:half]
        high_idx = hsort[half:]
        for _ in range(rand_p):
            if len(ordered) >= max_pairs:
                break
            if low_idx.size and high_idx.size:
                a = int(low_idx[int(rng.integers(low_idx.size))])
                b = int(high_idx[int(rng.integers(high_idx.size))])
                add_pair_to_list(ordered, seen, n, a, b)
        for _ in range(max(0, rand_p // 2)):
            if len(ordered) >= max_pairs:
                break
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            add_pair_to_list(ordered, seen, n, a, b)

    return ordered


def pair_b(h, i, j):
    n = int(h.size)
    m = 2 * n - 1
    b = np.zeros(m, dtype=float)
    rev = 1.0 - h[::-1]

    i = int(i)
    j = int(j)

    b[i:i + n] += rev
    b[n - 1 - i:2 * n - 1 - i] -= h

    b[j:j + n] -= rev
    b[n - 1 - j:2 * n - 1 - j] += h

    return b


def pair_F(c, b, t, center, idx1, idx2):
    v = c + float(t) * b
    t2 = t * t
    v[center] -= 2.0 * t2
    v[idx1] += t2
    v[idx2] += t2
    return float(np.max(v))


def pair_candidate_ts(
    T, lo, hi,
    c_zero_max,
    pos_inv_b, pos_c_over_b,
    neg_inv_b, neg_c_over_b,
    b1, c1, b2, c2,
    b0, c0
):
    tol = 1e-11
    L = float(lo)
    R = float(hi)

    if c_zero_max is not None and c_zero_max > T + 1e-10:
        return ()

    if pos_inv_b.size:
        up = float(np.max(T * pos_inv_b - pos_c_over_b))
        if up < R:
            R = up
        if L > R + tol:
            return ()

    if neg_inv_b.size:
        low = float(np.max(T * neg_inv_b - neg_c_over_b))
        if low > L:
            L = low
        if L > R + tol:
            return ()

    disc = b1 * b1 - 4.0 * (c1 - T)
    if disc < -1e-9:
        return ()
    if disc < 0.0:
        disc = 0.0
    s = float(np.sqrt(disc))
    r1 = (-b1 - s) * 0.5
    r2 = (-b1 + s) * 0.5
    if r1 > L:
        L = r1
    if r2 < R:
        R = r2
    if L > R + tol:
        return ()

    disc = b2 * b2 - 4.0 * (c2 - T)
    if disc < -1e-9:
        return ()
    if disc < 0.0:
        disc = 0.0
    s = float(np.sqrt(disc))
    r1 = (-b2 - s) * 0.5
    r2 = (-b2 + s) * 0.5
    if r1 > L:
        L = r1
    if r2 < R:
        R = r2
    if L > R + tol:
        return ()

    disc0 = b0 * b0 - 8.0 * (T - c0)
    if disc0 >= -1e-12:
        if disc0 < 0.0:
            disc0 = 0.0
        s = float(np.sqrt(disc0))
        rlo = (b0 - s) * 0.25
        rhi = (b0 + s) * 0.25
        cands = []

        endA = R if R < rlo else rlo
        if L <= endA + tol:
            cands.append((L + endA) * 0.5)

        startB = L if L > rhi else rhi
        if startB <= R + tol:
            cands.append((startB + R) * 0.5)

        if not cands:
            return ()
        return tuple(cands)

    return ((L + R) * 0.5,)


def line_search_pair(h, c, i, j, max_iter=42):
    n = int(h.size)
    m = int(c.size)
    center = n - 1
    cmax = float(np.max(c))

    i = int(i)
    j = int(j)
    if i == j:
        return 0.0, cmax, None, center, center

    lo = max(-float(h[i]), float(h[j]) - 1.0)
    hi = min(1.0 - float(h[i]), float(h[j]))
    if hi - lo <= 1e-13:
        return 0.0, cmax, None, center, center

    b = pair_b(h, i, j)
    idx1 = center + i - j
    idx2 = center + j - i
    if idx1 < 0 or idx1 >= m:
        idx1 = 0
    if idx2 < 0 or idx2 >= m:
        idx2 = m - 1

    if center == 0:
        lin_idx = np.arange(0, m)
    else:
        lin_idx = np.concatenate((np.arange(0, center), np.arange(center + 1, m)))

    c_lin = c[lin_idx]
    b_lin = b[lin_idx]

    tol_b = 1e-13
    pos = b_lin > tol_b
    neg = b_lin < -tol_b
    zero = ~(pos | neg)

    if zero.any():
        c_zero_max = float(np.max(c_lin[zero]))
    else:
        c_zero_max = -1e100

    if pos.any():
        pos_b = b_lin[pos]
        pos_inv_b = 1.0 / pos_b
        pos_c_over_b = pos_c_over_b = pos_inv_b * c_lin[pos]
    else:
        pos_inv_b = np.empty(0, dtype=float)
        pos_c_over_b = np.empty(0, dtype=float)

    if neg.any():
        neg_mb = -b_lin[neg]
        neg_inv_b = 1.0 / neg_mb
        neg_c_over_b = neg_inv_b * c_lin[neg]
    else:
        neg_inv_b = np.empty(0, dtype=float)
        neg_c_over_b = np.empty(0, dtype=float)

    b1 = float(b[idx1]); c1 = float(c[idx1])
    b2 = float(b[idx2]); c2 = float(c[idx2])
    b0 = float(b[center]); c0 = float(c[center])

    f0 = cmax
    f_lo = pair_F(c, b, lo, center, idx1, idx2)
    f_hi = pair_F(c, b, hi, center, idx1, idx2)
    hi_T = min(f0, f_lo, f_hi)
    lo_T = 0.0
    if hi_T < lo_T:
        hi_T = lo_T

    for _ in range(max(1, int(max_iter))):
        mid = 0.5 * (lo_T + hi_T)
        cands = pair_candidate_ts(
            mid, lo, hi,
            c_zero_max,
            pos_inv_b, pos_c_over_b,
            neg_inv_b, neg_c_over_b,
            b1, c1, b2, c2,
            b0, c0
        )
        if cands is None or len(cands) == 0:
            lo_T = mid
        else:
            hi_T = mid
        if hi_T - lo_T <= 1e-12:
            break

    T = hi_T

    tests = []
    for tt in (lo, hi, 0.0):
        if lo - 1e-12 <= tt <= hi + 1e-12:
            tests.append(float(tt))

    for tt in (-0.5 * b1, -0.5 * b2, -0.25 * b0):
        if lo - 1e-12 <= tt <= hi + 1e-12:
            tests.append(float(tt))

    for off in (0.0, 1e-9, 1e-7, 1e-5, 1e-3, 1e-2, 1e-1):
        TT = T + off
        if TT < 0.0:
            continue
        cands = pair_candidate_ts(
            TT, lo, hi,
            c_zero_max,
            pos_inv_b, pos_c_over_b,
            neg_inv_b, neg_c_over_b,
            b1, c1, b2, c2,
            b0, c0
        )
        if cands is not None:
            for cc in cands:
                if lo - 1e-12 <= cc <= hi + 1e-12:
                    tests.append(float(cc))

    uniq = []
    seen_t = set()
    for tt in tests:
        if tt < lo:
            tt = lo
        elif tt > hi:
            tt = hi
        key = round(float(tt), 12)
        if key not in seen_t:
            seen_t.add(key)
            uniq.append(float(tt))

    best_t = 0.0
    best_f = min(f0, f_lo, f_hi)
    for tt in uniq:
        f = pair_F(c, b, tt, center, idx1, idx2)
        if f < best_f:
            best_f = f
            best_t = float(tt)

    return best_t, best_f, b, idx1, idx2


def apply_pair_update(h, c, i, j, t, b, center, idx1, idx2):
    i = int(i)
    j = int(j)
    h[i] += float(t)
    h[j] -= float(t)

    if h[i] < 0.0:
        h[i] = 0.0
    elif h[i] > 1.0:
        h[i] = 1.0
    if h[j] < 0.0:
        h[j] = 0.0
    elif h[j] > 1.0:
        h[j] = 1.0

    c += float(t) * b
    t2 = float(t) * float(t)
    c[center] -= 2.0 * t2
    c[idx1] += t2
    c[idx2] += t2
    if c.size:
        c[c < 0.0] = 0.0


def random_perturb(h, target, rng, n_moves=30, scale=0.03, tol_count=None):
    n = int(h.size)
    h = np.asarray(h, dtype=float).copy()
    c = corr_counts(h)
    cur = float(np.max(c))
    if tol_count is None:
        tol_count = max(0.05, 0.002 * cur)

    for _ in range(int(n_moves)):
        a = int(rng.integers(n))
        b = int(rng.integers(n))
        if a == b:
            continue
        t = float(rng.uniform(-scale, scale))
        if h[a] + t < -1e-12 or h[a] + t > 1.0 + 1e-12:
            continue
        if h[b] - t < -1e-12 or h[b] - t > 1.0 + 1e-12:
            continue
        old_a = h[a]
        old_b = h[b]
        h[a] += t
        h[b] -= t
        c2 = corr_counts(h)
        m2 = float(np.max(c2))
        if m2 <= cur + tol_count:
            c = c2
            cur = m2
        else:
            h[a] = old_a
            h[b] = old_b

    return project_density(h, target)


def pair_local_search(h, deadline, max_pairs=3200, line_iters=42, seed=1):
    n = int(h.size)
    target = 0.5 * n
    h = project_density(np.asarray(h, dtype=float), target)
    c = corr_counts(h)
    cur_max = float(np.max(c))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(int(seed) * 1000003 + n)
    center = n - 1
    no_improve = 0

    while True:
        rem = deadline - time.perf_counter()
        if rem < 25.0:
            break

        if rem > 260.0:
            td = 6
            mp = max_pairs
            rand_p = 250
        elif rem > 90.0:
            td = 4
            mp = int(max_pairs * 0.75)
            rand_p = 100
        else:
            td = 2
            mp = int(max_pairs * 0.40)
            rand_p = 30

        pairs = make_pair_candidates(
            h, c,
            top_d=td,
            grad_p=12,
            term_q=10,
            rand_p=rand_p,
            rng=rng,
            max_pairs=max(1, mp)
        )

        if not pairs:
            dummy = []
            seen = set()
            for _ in range(200):
                a = int(rng.integers(n))
                b = int(rng.integers(n))
                add_pair_to_list(dummy, seen, n, a, b)
            pairs = dummy
            if not pairs:
                break

        updated = 0
        for code in pairs:
            if time.perf_counter() > deadline - 8.0:
                break
            a = int(code // n)
            b = int(code % n)
            if a == b:
                continue

            t, f, bvec, idx1, idx2 = line_search_pair(h, c, a, b, max_iter=line_iters)
            if bvec is None:
                continue
            if f < cur_max - 1e-12:
                apply_pair_update(h, c, a, b, t, bvec, center, idx1, idx2)
                cur_max = f
                updated += 1
                if cur_max < best_max - 1e-15:
                    best_max = cur_max
                    best_h = h.copy()

                if updated % 40 == 0:
                    h = project_density(h, target)
                    c = corr_counts(h)
                    cur_max = float(np.max(c))
                    if cur_max < best_max - 1e-15:
                        best_max = cur_max
                        best_h = h.copy()
                    if cur_max > best_max + 0.05:
                        h = project_density(best_h, target)
                        c = corr_counts(h)
                        cur_max = float(np.max(c))

        h = project_density(h, target)
        c = corr_counts(h)
        cur_max = float(np.max(c))
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_h = h.copy()

        if updated == 0:
            no_improve += 1
            if no_improve == 1:
                rem = deadline - time.perf_counter()
                if rem > 140.0:
                    h_p = random_perturb(h, target, rng, 35, 0.03, max(0.05, 0.002 * cur_max))
                    c_p = corr_counts(h_p)
                    m_p = float(np.max(c_p))
                    if m_p <= cur_max + max(0.05, 0.002 * cur_max):
                        h = h_p
                        c = c_p
                        cur_max = m_p
                        no_improve = 0
            if no_improve >= 2:
                break
        else:
            no_improve = 0

    best_h = project_density(best_h, target)
    bc = corr_counts(best_h)
    best_c5 = float(np.max(bc)) * (2.0 / float(n))
    return best_h, best_c5, n


_SMOOTH_N = 0
_SMOOTH_ALPHA = 50.0
_SMOOTH_TARGET = 0.0


def smooth_objective(x):
    n = int(_SMOOTH_N)
    alpha = float(_SMOOTH_ALPHA)
    if n <= 1:
        return 0.0
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return 1e100
    return float((2.0 / float(n)) * (m + np.log(z) / alpha))


def smooth_gradient(x):
    n = int(_SMOOTH_N)
    alpha = float(_SMOOTH_ALPHA)
    if n <= 1:
        return np.zeros(max(1, n), dtype=float)
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0 or not np.all(np.isfinite(corr)):
        return np.zeros(n, dtype=float)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -745.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0 or not np.isfinite(z):
        return np.zeros(n, dtype=float)
    w = e / z
    g = np.zeros(n, dtype=float)
    center = n - 1
    for tt in range(w.size):
        pw = float(w[tt])
        if pw <= 1e-12:
            continue
        k = tt - center
        if k >= 0:
            if k == 0:
                g += pw * (1.0 - 2.0 * h)
            else:
                nn = n - k
                if nn > 0:
                    g[k:] += pw * (1.0 - h[:nn])
                    g[:nn] -= pw * h[k:]
        else:
            ell = -k
            nn = n - ell
            if nn > 0:
                g[:nn] += pw * (1.0 - h[ell:])
                g[ell:] -= pw * h[:nn]
    return (2.0 / float(n)) * g


def smooth_sum_constraint(x):
    return float(np.sum(x) - _SMOOTH_TARGET)


def smooth_sum_jac(x):
    return np.ones_like(np.asarray(x, dtype=float), dtype=float)


def slsqp_polish(h, deadline):
    if _scipy_minimize is None:
        return None, float("inf")
    n = int(h.size)
    if n < 3 or n > 650:
        return None, float("inf")
    target = 0.5 * n
    best = project_density(np.asarray(h, dtype=float), target)
    best_c5 = float(np.max(corr_counts(best))) * (2.0 / float(n))

    global _SMOOTH_N, _SMOOTH_ALPHA, _SMOOTH_TARGET
    _SMOOTH_N = n
    _SMOOTH_TARGET = target

    bounds = []
    for _ in range(n):
        bounds.append((0.0, 1.0))
    constraints = ({"type": "eq", "fun": smooth_sum_constraint, "jac": smooth_sum_jac},)

    alphas = (30.0, 80.0, 150.0)
    for alpha in alphas:
        rem = deadline - time.perf_counter()
        if rem < 40.0:
            break
        _SMOOTH_ALPHA = alpha
        maxit = 28 if n <= 350 else 18
        try:
            res = _scipy_minimize(
                smooth_objective,
                best,
                method="SLSQP",
                jac=smooth_gradient,
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxit, "ftol": 1e-12, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                cval = float(np.max(corr_counts(cand))) * (2.0 / float(n))
                if cval < best_c5:
                    best_c5 = cval
                    best = cand
        except Exception:
            pass

    best = project_density(best, target)
    bc = corr_counts(best)
    return best, float(np.max(bc)) * (2.0 / float(n))


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1300.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 1300.0
    deadline = start + max(10.0, budget - 20.0)
    target_c5 = 0.38085
    rng = np.random.default_rng(int(seed))

    best_h = None
    best_c5 = float("inf")
    best_n = 0

    parent = None
    try:
        raw = initial_h_values
        if raw is not None:
            arr = np.asarray(raw, dtype=float)
            if arr.ndim == 1 and arr.size >= 2:
                arr = np.nan_to_num(arr, nan=0.5, posinf=1.0, neginf=0.0)
                arr = np.clip(arr, 0.0, 1.0).copy()
                parent = arr
    except Exception:
        parent = None

    if parent is not None:
        p = project_density(parent, 0.5 * parent.size)
        best_h = p.copy()
        best_c5 = safe_evaluate(p)
        best_n = int(p.size)
    else:
        n0 = 450
        x = (np.arange(n0) + 0.5) * (2.0 / float(n0))
        h0 = (
            0.5
            + 0.30 * np.sin(2.0 * np.pi * 3.0 * x / 2.0)
            + 0.15 * np.sin(2.0 * np.pi * 5.0 * x / 2.0 + 1.0)
        )
        best_h = project_density(h0, 0.5 * n0)
        best_c5 = safe_evaluate(best_h)
        best_n = n0

    if best_h is None or not np.all(np.isfinite(best_h)):
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / float(n0))
        best_h = project_density(
            0.5 + 0.3 * np.sin(2.0 * np.pi * 3.0 * x / 2.0),
            0.5 * n0
        )
        best_c5 = safe_evaluate(best_h)
        best_n = n0

    def _try_update(h, c5, n):
        nonlocal best_h, best_c5, best_n
        try:
            h = np.asarray(h, dtype=float)
            c5 = float(c5)
            n = int(n)
        except Exception:
            return
        if h.size != n:
            n = int(h.size)
        if n < 2 or not np.isfinite(c5):
            return
        if c5 < best_c5:
            best_h = project_density(h, 0.5 * n)
            best_c5 = c5
            best_n = n

    rem_total = max(30.0, deadline - time.perf_counter())
    primary_deadline = start + rem_total * 0.60
    primary_deadline = min(primary_deadline, deadline - 5.0)

    try:
        h1, c1, n1 = pair_local_search(
            best_h,
            primary_deadline,
            max_pairs=3400,
            line_iters=42,
            seed=int(seed) + 1
        )
        _try_update(h1, c1, n1)
    except Exception:
        pass

    if time.perf_counter() < deadline - 70.0 and best_n >= 3:
        ncur = best_n
        target_sum = 0.5 * ncur
        if abs(target_sum - round(target_sum)) < 1e-9:
            hb = integerize(best_h, target_sum)
            try:
                h2, c2, n2 = pair_local_search(
                    hb,
                    time.perf_counter() + min(120.0, max(20.0, (deadline - time.perf_counter()) * 0.18)),
                    max_pairs=2800,
                    line_iters=40,
                    seed=int(seed) + 2
                )
                _try_update(h2, c2, n2)
            except Exception:
                pass

    extra_starts = []
    for mode in (0, 2):
        extra_starts.append((best_n, mode))
    for mult in (0.80, 1.20):
        aa = int(round(best_n * mult))
        if aa % 2 == 1:
            aa += 1
        if 80 <= aa <= 800 and aa != best_n:
            extra_starts.append((aa, None))

    seen_n = set()
    for nn, mode in extra_starts:
        if time.perf_counter() > deadline - 90.0:
            break
        nn = int(nn)
        if nn in seen_n:
            continue
        seen_n.add(nn)
        try:
            if mode is None:
                hstart = upsample_h(best_h, nn)
            else:
                hstart = make_diverse_start(nn, rng, int(mode))
            hnext, cnext, nnext = pair_local_search(
                hstart,
                time.perf_counter() + min(100.0, max(20.0, (deadline - time.perf_counter()) * 0.16)),
                max_pairs=2200,
                line_iters=38,
                seed=int(seed) + len(extra_starts) + 3
            )
            _try_update(hnext, cnext, nnext)
        except Exception:
            pass

    if time.perf_counter() < deadline - 130.0 and 3 <= best_n <= 650:
        try:
            hs, cs = slsqp_polish(best_h, deadline - 10.0)
            if hs is not None and np.isfinite(cs):
                _try_update(hs, cs, int(hs.size))
        except Exception:
            pass

        rem = deadline - time.perf_counter()
        if rem > 50.0:
            try:
                hf, cf, nf = pair_local_search(
                    best_h,
                    deadline - 5.0,
                    max_pairs=1800,
                    line_iters=36,
                    seed=int(seed) + 90
                )
                _try_update(hf, cf, nf)
            except Exception:
                pass

    if best_h is None:
        nn = 90
        best_h = project_density(np.full(nn, 0.5), 0.5 * nn)
        best_c5 = safe_evaluate(best_h)
        best_n = nn

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c5 = safe_evaluate(final_h)
    if not np.isfinite(final_c5):
        cc = corr_counts(final_h)
        final_c5 = float(np.max(cc)) * (2.0 / float(final_h.size))

    return final_h, float(final_c5), int(final_h.size)
