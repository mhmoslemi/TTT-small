def pair_swapPhase(h, target, deadline, max_iter=3000, top_frac=0.04, min_k=10):
    n = h.size
    best = project_density(h, target).copy()
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(12345)
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr, center, top_idx = compute_corr_and_top(best, top_frac, min_k)
        # Get weighted gradient
        top_vals = corr[top_idx]
        spread = max(1e-10, top_vals.max() - top_vals.min())
        w = np.exp((top_vals - top_vals.min()) / spread)
        w /= w.sum()
        
        g = np.zeros(n, dtype=float)
        for i, t in enumerate(top_idx):
            t = int(t)
            k = t - center
            p = w[i]
            if p < 1e-12:
                continue
            if k == 0:
                g += p * (1.0 - 2.0 * best)
            elif k > 0:
                lag = k
                g[lag:] += p * (1.0 - best[:n-lag])
                g[:n-lag] -= p * best[lag:]
            else:
                lag = -k
                g[:n-lag] += p * (1.0 - best[lag:])
                g[lag:] -= p * best[:n-lag]
        
        # Find best pair swap
        neg_pool = np.argsort(g)[:15]
        pos_pool = np.argsort(-g)[:15]
        
        found = False
        for ii in neg_pool:
            ii = int(ii)
            for jj in pos_pool:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                for frac in (0.5, 0.8, 1.0, 0.3):
                    dd = frac * cap
                    cand = best.copy()
                    cand[ii] += dd
                    cand[jj] -= dd
                    if cand[ii] < -1e-12 or cand[ii] > 1+1e-12 or cand[jj] < -1e-12 or cand[jj] > 1+1e-12:
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-14:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        
        if not found:
            # Random pair
            for _r in range(10):
                ii, jj = int(rng.integers(n)), int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                dd = float(rng.random()) * 0.6 * cap
                cand = best.copy()
                cand[ii] += dd
                cand[jj] -= dd
                c = evaluate_erdos(cand)
                if c < best_c - 1e-14:
                    best = cand
                    best_c = c
                    found = True
                    break
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
    
    return project_density(best, target), evaluate_erdos(project_density(best, target))

def binarize_h(h, target, thresh=0.10):

import time
import numpy as np


def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-13:
        return x
    lo = float(np.min(0.0 - x) - 0.1)
    hi = float(np.max(1.0 - x) + 0.1)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def get_top_lags(h, top_frac, min_k):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    k_count = min(k_count, len(corr))
    top_idx = np.argpartition(corr, -k_count)[-k_count:]
    top_vals = corr[top_idx]
    spread = max(1e-10, float(top_vals.max() - top_vals.min()))
    # Use temperature-based weighting
    # Higher spread → more peaked
    return corr, center, top_idx, top_vals, spread


def compute_active_gradient(h, center, top_idx, weights):
    n = h.size
    g = np.zeros(n, dtype=float)
    for i in range(len(top_idx)):
        t = int(top_idx[i])
        p = float(weights[i])
        if p < 1e-12:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        elif k > 0:
            lag = k
            g[lag:] += p * (1.0 - h[:n - lag])
            g[:n - lag] -= p * h[lag:]
        else:
            lag = -k
            g[:n - lag] += p * (1.0 - h[lag:])
            g[lag:] -= p * h[:n - lag]
    return g


def make_weights(top_vals, spread, temperature):
    if len(top_vals) == 0:
        return np.array([])
    scaled = (top_vals - top_vals.max()) / max(1e-10, spread * temperature)
    scaled = np.clip(scaled, -700.0, 0.0)
    w = np.exp(scaled)
    s = float(np.sum(w))
    if s < 1e-15:
        w = np.ones_like(top_vals) / len(top_vals)
    else:
        w /= s
    return w


def phase_gradient(h, target, deadline, max_iter=10000, top_frac=0.06, min_k=12,
                   init_temp=8.0, min_temp=0.15, step_sizes=None,
                   kick_every=0, kick_mag=0.015, momentum=0.0):
    n = h.size
    if step_sizes is None:
        step_sizes = [0.04, 0.025, 0.015, 0.01, 0.006, 0.004, 0.002, 0.001]
    
    curr = project_density(h, target).copy()
    curr_c = evaluate_erdos(curr)
    best = curr.copy()
    best_c = curr_c
    vel = np.zeros(n, dtype=float)
    
    stall = 0
    kick_remaining = kick_mag
    rng = np.random.default_rng(int(time.perf_counter() * 1e6) % 100000)
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        # Adaptive temperature (exponential decay)
        decay_rate = max(0.5, 500.0 / max(1, max_iter))
        temp = max(min_temp, init_temp * np.exp(-it / decay_rate))
        
        corr, center, top_idx, top_vals, spread = get_top_lags(curr, top_frac, min_k)
        weights = make_weights(top_vals, spread, temp)
        g = compute_active_gradient(curr, center, top_idx, weights)
        
        # Remove mean to preserve sum
        g = g - np.mean(g)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        d = -g / gn
        
        # Apply momentum
        if momentum > 0.0:
            d = momentum * vel + d
            vel = d.copy()
        
        improved = False
        for step in step_sizes:
            cand = project_density(curr + step * d, target)
            c = evaluate_erdos(cand)
            if c < curr_c - 1e-14:
                curr = cand
                curr_c = c
                if c < best_c:
                    best = cand.copy()
                    best_c = c
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if kick_every > 0 and stall >= kick_every and kick_remaining > 0.001:
                kick = rng.normal(0, kick_remaining, n)
                kick -= np.mean(kick)
                cand = project_density(curr + kick, target)
                c = evaluate_erdos(cand)
                if c < curr_c:
                    curr = cand
                    curr_c = c
                    stall = 0
                kick_remaining *= 0.7
            elif stall > 80:
                break
    
    final = project_density(best, target)
    return final, evaluate_erdos(final)


def phase_swap(h, target, deadline, max_iter=5000, top_frac=0.04, min_k=12,
               pool_size=12, n_random=8):
    n = h.size
    best = project_density(h, target).copy()
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(54321)
    stall = 0
    
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        
        corr, center, top_idx, top_vals, spread = get_top_lags(best, top_frac, min_k)
        weights = make_weights(top_vals, spread, 0.8)
        g = compute_active_gradient(best, center, top_idx, weights)
        
        neg_pool = np.argsort(g)[:pool_size]
        pos_pool = np.argsort(-g)[:pool_size]
        
        found = False
        for ii in neg_pool:
            ii = int(ii)
            for jj in pos_pool:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                for frac in (0.5, 1.0, 0.75, 0.25, 0.9):
                    dd = frac * cap
                    cand = best.copy()
                    cand[ii] += dd
                    cand[jj] -= dd
                    if cand[ii] < -1e-12 or cand[ii] > 1.0 + 1e-12:
                        continue
                    if cand[jj] < -1e-12 or cand[jj] > 1.0 + 1e-12:
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-14:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        
        if not found:
            for _r in range(n_random):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                dd = float(rng.random()) * 0.7 * cap
                cand = best.copy()
                cand[ii] += dd
                cand[jj] -= dd
                c = evaluate_erdos(cand)
                if c < best_c - 1e-14:
                    best = cand
                    best_c = c
                    found = True
                    break
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 150:
                break
    
    final = project_density(best, target)
    return final, evaluate_erdos(final)


def try_binarize(h, target, thresh=0.10):
    n = h.size
    h2 = h.copy()
    changed = 0
    for i in range(n):
        if h2[i] < thresh:
            h2[i] = 0.0
            changed += 1
        elif h2[i] > 1.0 - thresh:
            h2[i] = 1.0
            changed += 1
    if changed == 0:
        return h
    return project_density(h2, target)

import time
import numpy as np


def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-13:
        return x
    lo = float(np.min(0.0 - x)) - 0.5
    hi = float(np.max(1.0 - x)) + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

    top_vals = corr[top_idx]
    spread = max(1e-10, float(np.max(top_vals) - np.min(top_vals)))
    return corr, center, top_idx, top_vals, spread


def compute_active_gradient(h, center, top_idx, weights):
    n = h.size
    g = np.zeros(n, dtype=float)
    for i in range(len(top_idx)):
        t = int(top_idx[i])
        w = float(weights[i])
        if w < 1e-12:
            continue
        k = t - center
        if k == 0:
            g += w * (1.0 - 2.0 * h)
