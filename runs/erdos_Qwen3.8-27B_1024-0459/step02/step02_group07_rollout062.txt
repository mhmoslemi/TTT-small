    h = np.clip(x, 0.0, 1.0)
    s = np.sum(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_opt_j(h, j, n, corr, ells):
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(35):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-13
        sg = np.mean(a[active])
        if sg > 1e-12:
            hi = mid
        elif sg < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def coord_descent(h, n, n_passes, time_limit, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    ells = np.arange(-(n - 1), n)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for j in order:
            t_opt = coord_opt_j(h, j, n, corr, ells)
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def lse_objective(x, n, tau):
    h = np.clip(x, 0.0, 1.0)
    s = np.sum(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_corr = float(2 * n - 1)
    scaled = tau * corr / m_corr
    mx = np.max(scaled)
    exps = np.exp(scaled - mx)
    lse = (mx + np.log(np.sum(exps))) / m_corr
    pen = 1e8 * (s - n / 2.0) ** 2
    return lse + pen


def lse_gradient(x, n, tau):
    h = np.clip(x, 0.0, 1.0)
    s = np.sum(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_corr = 2 * n - 1
    scaled = tau * corr / float(m_corr)
    mx = np.max(scaled)
    exps = np.exp(scaled - mx)
    w = exps / np.sum(exps)
    ells = np.arange(-(n - 1), n)
    pen_coeff = 2e8 * (s - n / 2.0)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - ells
        idx2 = j + ells
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        a = np.zeros(m_corr)
        if np.any(valid1):
            a[valid1] = 1.0 - h[idx1[valid1]]
        if np.any(valid2):
            a[valid2] += -h[idx2[valid2]]
        grad[j] = np.dot(w, a) * tau / m_corr + pen_coeff
    return grad


def lbfgsb_smoothed(h0, n, time_limit):
    h = h0.copy()
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    taus = [20, 60, 150, 300, 600, 1000, 2000]
    for tau in taus:
        if time.time() - start > time_limit - 15:
            break
        max_iter = max(50, int((15 - (time.time() - start)) * 2))
        try:
            res = minimize(lse_objective, h, method='L-BFGS-B',
                           jac=lambda x: lse_gradient(x, n, tau),
                           bounds=bounds,
                           options={'maxiter': max_iter, 'ftol': 1e-14,
                                    'gtol': 1e-8, 'maxfun': max_iter * 3})
            h_new = make_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new < c5_old - 1e-8:
                h = h_new
        except Exception:
            pass
    return h, compute_c5(h)


def slsqp_polish(h, n, max_iter=5000, ftol=1e-16):
    def obj(x):
        return float(np.max(np.correlate(x, 1.0 - x, mode='full')))
    def sum_con(x):
        return float(np.sum(x) - n / 2.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return make_feasible(np.ones(n_dst) * 0.5, n_dst)
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def binary_hill_climb(h_init, n, time_limit, rng):
    h_bin = (h_init > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = n / 2.0 - s
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_init[zeros])[:int(deficit + 0.5)]
        if len(order) > 0:
            h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_init[ones])[:int(-deficit + 0.5)]
        if len(order) > 0:
            h_bin[ones[order]] = 0.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    trial = 0
    while time.time() - start < time_limit:
        trial += 1
        if np.sum(h_bin) != n / 2.0:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if h_bin[i] == h_bin[j]:
            continue
        h_try = h_bin.copy()
        h_try[i] = 1.0 - h_try[i]
        h_try[j] = 1.0 - h_try[j]
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_bin = h_try.copy()
    return best_h, best_c5


def perturb_reoptimize(h, n, rng, n_trials, time_per):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        h_pert = h + rng.randn(n) * 0.005
        h_pert = make_feasible(h_pert, n)
        h_new, c5_new = coord_descent(h_pert, n, n_passes=80, time_limit=time_per, rng=rng)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5
