        grad += corr_gradient(h, lag)
    return grad

Let me also think about a key improvement: **the sum constraint interaction**. When we take a gradient step, we might violate the sum constraint. The projection handles this, but it could distort the direction. A better approach: subtract the mean of the gradient (to make it orthogonal to the constraint direction) before the step.

grad_projected = grad - mean(grad) * ones  (this makes sum(grad_projected) = 0, preserving the sum constraint to first order)

grad_perp = grad - (1/n) * sum(grad) * ones_n

This ensures the step doesn't change the sum (to first order), so the projection is a no-op (or very small correction).

- A "swap" exchanges h[i] and h[j] (maintains sum constraint exactly if h[i]+h[j] is preserved, which it is since we're swapping)

from scipy.optimize import minimize
import time

_TGT = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(free))
        if cnt > 0:
            h[free] += diff / cnt
        else:
            order = np.argsort(h)
            if diff > 0:
                for idx in order[::-1]:
                    room = 1.0 - h[idx]
                    adj = min(diff, room)
                    h[idx] += adj
                    diff -= adj
                    if abs(diff) < 1e-14:
                        break
            else:
                for idx in order:
                    room = h[idx]
                    adj = max(diff, -room)
                    h[idx] += adj
                    diff -= adj
                    if abs(diff) < 1e-14:
                        break
        h = np.clip(h, 0.0, 1.0)
    return h

        mask[idx] = True
    exps = np.exp(_BETA * (corr[mask] - cmax))

def corr_gradient(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        n_pairs = n - lag
        if n_pairs > 0:
            grad[lag:lag+n_pairs] += (1.0 - h[:n_pairs])
            grad[:n_pairs] -= h[lag:lag+n_pairs]
    else:
        a = -lag
        n_pairs = n - a
        if n_pairs > 0:
            grad[:n_pairs] += (1.0 - h[a:a+n_pairs])
            grad[a:a+n_pairs] -= h[:n_pairs]
    return grad

def topk_subgradient(h, k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = idx - (n - 1)
        grad += corr_gradient(h, lag)
    return grad
