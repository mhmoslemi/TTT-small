For the second term sum_i h[i]*h[i+s]: gradient is h[i+s] for j=i (if i+s is valid) plus h[i] for j=i+s (if i is valid).

2. Compute p_{i,j} = h[i]*h[j]
3. Solve the LP to get new h
4. Repeat until convergence or time runs out

import numpy as np

def solve_lp(h_current, n):
    """Solve the linearized LP for the ERD problem."""
    h = cp.Variable(n)
    t = cp.Variable()
    
    dx = 2.0 / n
    # Compute p_{i,j} = h[i]*h[j]
    p = np.outer(h_current, h_current)
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2.0]
    
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            valid_range = slice(0, n-s)
            sum_h = cp.sum(h[valid_range])
            # sum p[i, i+s] for i in 0..n-1-s
            const = np.sum(p[0:n-s, range(s, n)])
            # Actually p[i, i+s] where i ranges 0 to n-1-s
            const = np.sum([p[i, i+s] for i in range(n-s)])

            sum_h = cp.sum(h[valid_range])
            # sum p[i, i+s] for i in abs_s..n-1, i.e., p[i, i-abs_s]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
