```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.count_nonzero(mask)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-15:
        # Add/subtract to the coordinate with most slack
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def coord_opt_j(h, j, n, corr, ells):
    """Find optimal value of h[j] given all other coordinates."""
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]
    b = corr - h[j] * a
    
    # Minimize max_k (a[k]*t + b[k]) over t in [0,1]
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-13
        sg = np.mean(a[active])
        if sg > 1e-12:
            hi = mid
        elif sg < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, n_passes, t_start, t_limit):
    """Run coordinate descent."""
    h = h.copy()
    dx = 2.0 / n
    ells = np.arange(-(n - 1), n)
    corr = get_corr(h)
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    for p in range(n_passes):
        if time.time() > t_limit:
            break
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr, ells)
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def perturb(h, n, rng, scale=0.05):
    """Add random perturbation maintaining feasibility."""
    h_new = h.copy() + rng.randn(n) * scale
    return make_feasible(h_new, n)

def slsqp_polish(h, n, max_iter=5000):
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def generate_init(n, rng):
    """Generate a diverse initial point."""
    # Start with random binary-ish pattern
    h = rng.rand(n)
    h = (h > 0.5).astype(float)  # Binary
    # Fix to exactly n/2 ones
    s = np.sum(h)
    if s > n/2:
        ones = np.where(h == 1)[0]
        remove = rng.choice(ones, int(s - n/2), replace=False)
        h[remove] = 0
    elif s < n/2:
        zeros = np.where(h == 0)[0]
        add = rng.choice(zeros, int(n/2 - s), replace=False)
        h[add] = 1
    return make_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    t_end = t_start + budget_s - 5  # 5s safety margin
    
    def remaining():
        return max(0, t_end - time.time())
    
    # Get parent
    global initial_h_values
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except (NameError, Exception):
        h_parent = None
    
    if h_parent is not None:
        best_h = make_feasible(h_parent.copy(), len(h_parent))
        best_n = len(best_h)
    else:
        best_n = 200
        best_h = generate_init(best_n, rng)
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent at parent resolution
    if remaining() > 100:
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=1000, 
                                     t_start=t_start, t_limit=t_start + 200)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 2: Try multiple resolutions
    for n_try in [best_n, 192, 256, 320, 384, 512]:
        if remaining() < 80:
            break
        if n_try == best_n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_to_n(best_h, n_try)
        
        # Coordinate descent
        tl = min(150, remaining() - 60)
        if tl < 30:
            continue
        h_cd, c5_cd = coord_descent(h_init, n_try, n_passes=500,
                                     t_start=t_start, t_limit=time.time() + tl)
        
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_try
    
    # Phase 3: Perturbation + CD
    perturbations = 20
    for i in range(perturbations):
        if remaining() < 60:
            break
        scale = 0.02 + 0.03 * (i % 3)
        h_pert = perturb(best_h, best_n, rng, scale)
        tl = min(50, remaining() - 40)
        if tl < 15:
            break
        h_cd, c5_cd = coord_descent(h_pert, best_n, n_passes=100,
                                     t_start=t_start, t_limit=time.time() + tl)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 4: SLSQP polish
    if remaining() > 60:
        h_fin, c5_fin = slsqp_polish(best_h, best_n, max_iter=3000)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    # Phase 5: Final CD
    if remaining() > 30:
        tl = remaining() - 10
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=500,
                                     t_start=t_start, t_limit=time.time() + tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Final
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```

Actually, for coordinate j, we need corr to be up-to-date. If we update corr incrementally after each coordinate change, we get Gauss-Seidel behavior. The update is:
corr[m] += (h[j]_new - h[j]_old) * a_j[m]

where a_j is the coefficient vector for coordinate j. This is O(n) per coordinate update, same as computing a_j.

```python
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    for p in range(n_passes):
        if time.time() > t_limit:
            break
        for j in range(n):
            old_hj = h[j]
            # Compute coefficient vector for coordinate j
            idx1 = j - ells
            idx2 = j + ells
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
```

```python
def compute_c5(h):

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff > 1e-14:
                mask = h < 1.0 - 1e-14
                h[mask] += diff / np.count_nonzero(mask)
        else:
            idx = np.argmax(h)
            sub = min(-diff, h[idx])
            h[idx] -= sub

def slsqp_polish(h, n, max_iter=5000):
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Binary quantization + re-optimize
    if remaining() > 80:
        h_bin = binary_quantize(best_h, best_n, rng)
        c5_bin = compute_c5(h_bin)
        tl = min(100, t_start + remaining())
        h_cd, c5_cd = gauss_seidel_cd(h_bin, best_n, n_passes=800, t_limit=tl)
        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Try diverse initializations at best resolution
    for m in range(5):
        if remaining() < 60:
            break
        h_div = generate_diverse_init(best_n, rng, method=m)
        tl = min(60, t_start + remaining())
        if time.time() > tl - 10:
            break
        h_cd, c5_cd = gauss_seidel_cd(h_div, best_n, n_passes=400, t_limit=tl)
        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: SLSQP polish
    if remaining() > 80:
        h_fin, c5_fin = slsqp_polish(best_h, best_n, max_iter=5000)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if remaining() > 40:
            h_cd, c5_cd = gauss_seidel_cd(best_h, best_n, n_passes=500,
                                           t_limit=t_start + remaining())
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 7: Final CD + SLSQP
    if remaining() > 40:
        tl = t_start + remaining() - 10
        h_cd, c5_cd = gauss_seidel_cd(best_h, best_n, n_passes=1000, t_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    if remaining() > 30:
        h_fin, c5_fin = slsqp_polish(best_h, best_n, max_iter=3000)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    if remaining() > 15:
        h_cd, c5_cd = gauss_seidel_cd(best_h, best_n, n_passes=300,
                                       t_limit=t_start + remaining() - 2)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
