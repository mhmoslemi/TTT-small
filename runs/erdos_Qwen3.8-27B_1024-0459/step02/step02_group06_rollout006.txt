def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)


def to_binary(h, n_ones):
    """Project h to binary with exactly n_ones ones"""
    n = h.size
    h2 = np.zeros(n)
    if n_ones >= n:
        h2[:] = 1.0
        return h2
    if n_ones <= 0:
        return h2
    idx = np.argsort(h)[::-1][:n_ones]
    h2[idx] = 1.0
    return h2

    accept_count = 0
    stall_count = 0
    temp = t_start
    
    while it < n_iter:

            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        
        # Cooling schedule
        progress = it / max(1, n_iter)
        temp = max(t_end, t_start * (t_end / t_start) ** progress)

                i_idx = rng.integers(len(ones_list))

                else:

                    bad_ones = [t - lag for t in range(lag, n) if h[t] == 0 and h[t - lag] == 1]
                    good_zeros = [t for t in range(lag, n) if h[t] == 0 and h[t - lag] == 1]

                    i = bad_ones[rng.integers(len(bad_ones))]

                    j = good_zeros[rng.integers(len(good_zeros))]

        h[j] = 1.0

def directed_binary_search(h0, deadline, seed=42, max_iter=50000):

        improved = False

            if k == 0:
                continue

            if lag >= n:
                continue
            
            # Find contributing positions
            if k > 0:

                t_range = range(0, n - lag)
            else:
                # h[t]=1, h[t-lag]=0 means at position t-lag it's 1, at t it's 0
                # overlap term: h[t-lag]*(1-h[t]) for t in [lag, n)
                t_range = range(lag, n)

            candidate_swaps = []

                if k > 0:
                    if h[i] == 1 and h[i + lag] == 0:
                        candidate_swaps.append((i, i + lag))
                        bad_count += 1
                else:
                    if h[i - lag] == 1 and h[i] == 0:
                        candidate_swaps.append((i - lag, i))
                        bad_count += 1

def make_structured_init(n, mode, rng):
    """Create structured initializations"""
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    target = 0.5 * n
    
    if mode == 0:
        # Two equal intervals
        seg = n // 4
        h = np.zeros(n)
        h[0:seg] = 1.0
        h[n//2:n//2+seg] = 1.0
    elif mode == 1:
        # Three intervals
        seg = n // 6
        h = np.zeros(n)
        h[0:seg] = 1.0
        h[n//3:2*n//3] = 1.0
        h[n-2*seg:n-seg] = 1.0
    elif mode == 2:
        # Sinusoidal
        freq = int(rng.integers(2, 8))
        phase = rng.uniform(0, 2*np.pi)
        h = 0.5 + 0.4 * np.sin(2*np.pi*freq*x/2 + phase)
    elif mode == 3:
        # Random binary
        h = np.zeros(n)
        idx = rng.choice(n, size=n//2, replace=False)
        h[idx] = 1.0
    elif mode == 4:
        # Alternating blocks of varying size
        h = np.zeros(n)
        pos = 0
        val = 1.0
        while pos < n:
            blen = int(rng.integers(n//30, n//8))
            h[pos:pos+blen] = val
            pos += blen
            val = 1.0 - val
    elif mode == 5:
        # Golden ratio based
        phi = (1 + np.sqrt(5)) / 2
        h = np.zeros(n)
        for i in range(n):
            if (i / n * phi) % 1.0 < 0.5:
                h[i] = 1.0
    else:
        h = np.full(n, 0.5)
    
    return project_density(h, target)


def smooth_max_objective(x, n, alpha, target):
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return 1e100
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return 1e100
    val = (2.0 / n) * (m + np.log(z) / alpha)
    # Penalty for constraint violation
    s = float(np.sum(x))
    val += 10.0 * (s - target) ** 2
    return float(val)


def smooth_max_jacobian(x, n, alpha, target):
    h = np.asarray(x, dtype=float)
    g = np.zeros(n, dtype=float)
    if not np.all(np.isfinite(h)):
        return g
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return g
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return g
    w = e / z
    center = n - 1
    for t in range(len(corr)):
        p = w[t]
        if p < 1e-10:
            continue
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
            else:
                g[:n - lag] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
    dx_ = 2.0 / n
    g *= dx_
    # Constraint gradient
    g += 20.0 * (float(np.sum(x)) - target)
    return g

def pair_polish(h0, deadline, n):

                best_c = c_bin
                best_h = h_bin.copy()
                best_n = n
            
            # Phase 2: Directed binary search
            rem3 = deadline - time.perf_counter()
            if rem3 > 10.0:
                sub_dl2 = time.perf_counter() + min(per_init * 0.25, rem3 - 5.0)
                h_dir, c_dir = directed_binary_search(h_bin, sub_dl2, seed=seed + 1,
                                                      max_iter=20000)
                if c_dir < best_c:
                    best_c = c_dir
                    best_h = h_dir.copy()
                    best_n = n
            
            # Phase 3: Non-binary polish (small)
            rem4 = deadline - time.perf_counter()
            if rem4 > 10.0 and n <= 900:
                h_nb = project_density(h_dir if c_dir < c_bin else h_bin, float(target))
                sub_dl3 = time.perf_counter() + min(per_init * 0.2, rem4 - 3.0)
                h_pol, c_pol = nonbinary_polish(h_nb, sub_dl3, n, float(target))
                if c_pol < best_c:
                    best_c = c_pol
                    best_h = h_pol.copy()
                    best_n = n
    
    # Final intensive polish on the best solution found
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        best_c = evaluate_erdos(h_fin)
        
        if nbest % 2 == 0:
            # Try binary refinement at this resolution
            if rem > 20.0:
                h_b = to_binary(h_fin, nbest // 2)
                c_b = evaluate_erdos(h_b)
                sub_dl = time.perf_counter() + min(rem * 0.3, deadline - time.perf_counter() - 5.0)
                h_b2, c_b2 = binary_sa(h_b, sub_dl, seed=seed + 100,
                                       max_iter=100000, t_start=0.04, t_end=0.002)
                if c_b2 < best_c:
                    best_c = c_b2
                    h_fin = h_b2
                    best_h = h_b2.copy()
            elif rem > 10.0:
                h_b = to_binary(h_fin, nbest // 2)
                c_b = evaluate_erdos(h_b)
                if c_b < best_c:
                    best_c = c_b
                    h_fin = h_b
                    best_h = h_b.copy()
        
        # Pair polish
        rem2 = deadline - time.perf_counter()
        if rem2 > 8.0 and nbest <= 1000:
            sub_dl = time.perf_counter() + min(rem2 * 0.4, deadline - time.perf_counter() - 3.0)
            h_pp, c_pp = pair_polish(h_fin, sub_dl, nbest)
            if c_pp < best_c:
                best_c = c_pp
                h_fin = h_pp
                best_h = h_pp.copy()
        
        # Final small SA refinement
        rem3 = deadline - time.perf_counter()
        if rem3 > 8.0 and nbest % 2 == 0:
            h_b = to_binary(h_fin, nbest // 2)
            sub_dl = time.perf_counter() + min(rem3 * 0.5, deadline - time.perf_counter() - 2.0)
            h_b3, c_b3 = binary_sa(h_b, sub_dl, seed=seed + 200,
                                   max_iter=50000, t_start=0.02, t_end=0.001)
            if c_b3 < best_c:
                best_c = c_b3
                h_fin = h_b3
                best_h = h_b3.copy()
    
    # Ensure feasibility
    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45
    
    n_final = int(best_h.size)
    final_h = project_density(best_h, 0.5 * n_final)
    final_c = evaluate_erdos(final_h)
    
    return final_h, float(final_c), n_final
