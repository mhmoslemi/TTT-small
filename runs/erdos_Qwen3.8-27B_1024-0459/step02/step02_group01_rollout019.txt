def compute_c5(h):
    n = len(h)

    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute evenly among non-saturated entries
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) == 0:
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

    grad = np.zeros(n)

    # For s >= 0: c_s = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])

        s = k_idx - (n - 1)  # shift
        if s >= 0:

            # ∂/∂h[m]: 
            #   m appears in h[i+s] when i = m-s, valid if 0<=m-s<=n-1-s → s<=m<=n-1
            #   m appears in h[i] when i = m, valid if 0<=m<=n-1-s
            if s <= n - 1:
                # First term: (1-h[m-s]) for m in [s, n-1]
                valid_first = np.arange(s, n)
                grad[valid_first] += w * (1.0 - h[valid_first - s])
                # Second term: -h[m+s] for m in [0, n-1-s]
                valid_second = np.arange(0, n - s)
                grad[valid_second] += w * (-h[valid_second + s])
        else:
            r = -s  # r > 0
            # c_{-r} = sum_{i=0}^{n-1-r} h[i]*(1-h[i+r])

            valid_first = np.arange(0, n - r)

            valid_second = np.arange(r, n)
            grad[valid_second] += w * (-h[valid_second - r])
    
    return grad
