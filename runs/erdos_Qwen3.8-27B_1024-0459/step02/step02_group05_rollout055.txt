import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            inc = diff / cnt
            h[mask] = np.minimum(h[mask] + inc, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            inc = diff / cnt
            h[mask] = np.maximum(h[mask] + inc, 0.0)
    h = np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    bc_max = np.max(bc)
    exp_bc = np.exp(bc - bc_max)
    sum_exp = np.sum(exp_bc)
    p = exp_bc / sum_exp
    
    # Objective: (1/beta) * (log(sum_exp) + bc_max) = (1/beta)*(log(sum_exp)) + max(corr)
    obj = (1.0 / beta) * np.log(sum_exp) + bc_max / beta
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(len(p)):
        if p[idx] < 1e-13:
            continue
        shift = idx - (n - 1)
        w = p[idx]
        # d(corr[idx])/dh[j]:
        # For j where 0<=j<n and 0<=j+shift<n: contributes (1-h[j+shift])
        # For j where 0<=j<n and 0<=j-shift<n: contributes -h[j-shift]
        
        # term1: j+shift in [0,n-1], j in [0,n-1]
        # j in [max(0,-shift), min(n-1, n-1-shift)]
        j1s = max(0, -shift)
        j1e = min(n - 1, n - 1 - shift)
        if j1s <= j1e:
            js = j1s + shift  # j1s+shift, ..., j1e+shift (all in [0,n-1])
            grad[j1s:j1e+1] += w * (1.0 - h[js])
        
        # term2: j-shift in [0,n-1], j in [0,n-1]
        # j in [max(0,shift), min(n-1, n-1+shift)]
        j2s = max(0, shift)
        j2e = min(n - 1, n - 1 + shift)
        if j2s <= j2e:
            jm = j2s - shift  # j2s-shift, ..., j2e-shift (all in [0,n-1])
            grad[j2s:j2e+1] -= w * h[jm]
    
    # Project onto sum(h)=n/2 hyperplane: subtract mean
    grad -= np.mean(grad)
    return obj, grad

def lse_lbfgs(h_init, beta, max_iter=200):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        obj, _ = lse_objective_and_grad(x, beta)
        return obj
    
    def jac(x):
        _, g = lse_objective_and_grad(x, beta)
        return g
    
    try:
        res = minimize(fun, h_init, jac=jac, method='L-BFGS-B', 
                      bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_init, n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def lse_val_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr - beta * np.max(corr)
    exp_bc = np.exp(bc)
    logsum = np.log(np.sum(exp_bc)) + beta * np.max(corr)
    obj = logsum / beta
    p = exp_bc / np.sum(exp_bc)
    
    # Gradient: for each shift, compute contribution
    # numpy correlate: result[k_full] = sum_n h[n+shift]*(1-h[n]), shift = k_full-(n-1)
    # d/dh[j] result[k_full]:
    #   Term from h[n+shift]: (1-h[n]) when n=j-shift, valid if 0<=j-shift<n and 0<=j<n
    #   Term from (1-h[n]): -h[n+shift] when n=j, valid if 0<=j<n and 0<=j+shift<n
    grad = np.zeros(n)
    for k_full in range(2 * n - 1):
        if p[k_full] < 1e-12:
            continue
        shift = k_full - (n - 1)
        w = p[k_full]
        
        # Term 1: (1-h[j-shift]) for valid j where 0<=j-shift<n and 0<=j<n
        # j-shift in [0,n-1] => j in [shift, n-1+shift]
        # j in [0,n-1]
        # So j in [max(0,shift), min(n-1, n-1+shift)]
        # AND we need n=j-shift to be in valid sum range: max(0,-shift) <= j-shift <= min(n-1, n-1-shift)
        # j-shift >= max(0,-shift) => j >= max(shift, 0) (already covered)
        # j-shift <= min(n-1, n-1-shift) => j <= min(n-1+shift, n-1) (already covered)
        # So the valid range for j in term 1 is [max(0,shift), min(n-1, n-1+shift)]
        
        j1s = max(0, shift)
        j1e = min(n - 1, n - 1 + shift)
        if j1s <= j1e:
            # j ranges from j1s to j1e, j-shift ranges from j1s-shift to j1e-shift
            ns = j1s - shift
            ne = j1e - shift
            grad[j1s:j1e+1] += w * (1.0 - h[ns:ne+1])
        
        # Term 2: -h[j+shift] for valid j where 0<=j<n and 0<=j+shift<n
        # j+shift in [0,n-1] => j in [-shift, n-1-shift]
        # j in [0,n-1]
        # So j in [max(0,-shift), min(n-1, n-1-shift)]
        j2s = max(0, -shift)
        j2e = min(n - 1, n - 1 - shift)
        if j2s <= j2e:
            js = j2s + shift
            je = j2e + shift
            grad[j2s:j2e+1] -= w * h[js:je+1]
    
    # Project gradient to sum=0
    grad -= np.mean(grad)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def lse_val_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    bc = beta * corr
    bc_max = np.max(bc)
    exp_bc = np.exp(bc - bc_max)
    sum_exp = np.sum(exp_bc)
    obj = (np.log(sum_exp) + bc_max) / beta
    p = exp_bc / sum_exp
    grad = np.zeros(n)
    for k_full in range(2 * n - 1):
        if p[k_full] < 1e-11:
            continue
        shift = k_full - (n - 1)
        w = p[k_full]
        # Term 1: (1-h[j-shift]) for j where 0<=j-shift<n and 0<=j<n
        j1s = max(0, shift)
        j1e = min(n - 1, n - 1 + shift)
        if j1s <= j1e:
            ns = j1s - shift
            ne = j1e - shift
            grad[j1s:j1e + 1] += w * (1.0 - h[ns:ne + 1])
        # Term 2: -h[j+shift] for j where 0<=j<n and 0<=j+shift<n
        j2s = max(0, -shift)
        j2e = min(n - 1, n - 1 - shift)
        if j2s <= j2e:
            js = j2s + shift
            je = j2e + shift
            grad[j2s:j2e + 1] -= w * h[js:je + 1]
    grad -= np.mean(grad)
    return obj, grad

def optimize_lse(h_init, betas, max_iter_each=300):
    n = len(h_init)
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        def fun(x):
            v, _ = lse_val_and_grad(x, beta)
            return v
        def jac(x):
            _, g = lse_val_and_grad(x, beta)
            return g
        try:
            res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_iter_each, 'ftol': 1e-18, 'gtol': 1e-14})
            h = make_feasible(res.x, n)
        except Exception:
            pass
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def pair_adjust_search(h, n, budget_s, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step_sizes = [0.02, 0.05, 0.1, 0.15, 0.2]
    no_improve = 0
    while time.time() - start < budget_s:
        improved = False
        for _ in range(300):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.choice(step_sizes) * (1.0 if rng.random() < 0.5 else -1.0)
            if h[i] + delta < -1e-9 or h[i] + delta > 1.0 + 1e-9:
                continue
            if h[j] - delta < -1e-9 or h[j] - delta > 1.0 + 1e-9:
                continue
            h[i] += delta
            h[j] -= delta
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
                no_improve = 0
            else:
                h[i] -= delta
                h[j] += delta
        if improved:
            continue
        no_improve += 1
        if no_improve >= 4:
            break
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def binary_sa(n, rng, n_steps, t_start, t_end):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-15)):
            pass
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def slsqp_direct(h_init, n, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def multi_start_optimize(h_ref, n, budget_s, rng):
    best_h = h_ref.copy()
    best_c5 = compute_c5(h_ref)
    remaining_check = lambda: budget_s - (time.time() - _start_time)
    
    # Generate diverse starts
    starts = [h_ref.copy()]
    # Perturbed copies

    # Random
    for _ in range(2):

        starts.append(make_feasible(h_rand, n))
    # Binary from SA
    h_bin, _ = binary_sa(n, rng, n_steps=8000, t_start=0.5, t_end=0.01)
    starts.append(h_bin.astype(float))
    
    for h_start in starts:

        h_opt = optimize_lse(h_start, [50, 200, 800], max_iter_each=150)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    return best_h, best_c5
