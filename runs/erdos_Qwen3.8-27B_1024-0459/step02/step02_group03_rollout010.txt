    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Find element with most room
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        idxs = np.argsort(-room)
        for idx in idxs:
            step = min(diff, room[idx]) if diff > 0 else max(diff, -h[idx])
            h[idx] += step
            diff -= step
            if abs(diff) < 1e-14:
                break
    return h

def lag_gradient(h, L, n):
    """Gradient of overlap at lag L w.r.t. h."""
    grad = np.zeros(n)
    if L == 0:
        grad = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:n - L])
        grad[:n - L] -= h[L:]
    else:
        Lp = -L
        grad[:n - Lp] += (1.0 - h[Lp:])
        grad[Lp:] -= h[:n - Lp]
    return grad

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def run_slsqp(h0, n, beta, max_iter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, beta) * dx
    
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        weights = smooth_max_grad(corr, beta)
        grad = np.zeros(n)
        for k in range(len(weights)):
            w = weights[k]
            if w < 1e-12:
                continue
            L = k - (n - 1)
            grad += w * lag_gradient(x, L, n)
        return dx * grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def subgradient_method(h0, n, max_iter, top_m, lr_init=0.2, decay=0.999):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    
    for it in range(max_iter):
        overlaps = compute_overlaps(h)
        # Focus on top m
        sorted_idx = np.argsort(overlaps)
        top_idx = sorted_idx[-top_m:]
        
        # Softmax weights over top m
        top_vals = overlaps[top_idx]
        tv_max = np.max(top_vals)
        exps = np.exp(100 * (top_vals - tv_max))
        weights = np.zeros(len(overlaps))
        weights[top_idx] = exps / np.sum(exps)
        
        # Combined gradient
        grad = np.zeros(n)
        for k in top_idx:
            w = weights[k]
            if w > 1e-10:
                grad += w * lag_gradient(h, k - (n-1), n)
        grad *= 2.0 / n
        
        # Step
        h = h - lr * grad
        h = project_feasible(h, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        lr *= decay
    
    return best_h, best_c5

                    h[f1] = 1.0
                    h[f0] = 0.0
        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 200:
                break

    return h, best_c5

def sa_bin_search(h, n, rng, max_rounds, t_init, t_final, t_decay):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    T = t_init

    for rnd in range(max_rounds):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
        T = max(T * t_decay, t_final)

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - t0) - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        pn = len(parent_h)
        if pn > 0:
            parent_h = project_feasible(parent_h, pn)
            best_h = parent_h.copy()
            best_c5 = compute_c5(parent_h)
            best_n = pn
            print("Parent: n={}, C5={:.8f}".format(pn, best_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)

    # Multi-resolution exploration
    n_candidates = sorted(set([best_n, 180, 200, 220, 250, 280, 320, 360]))
    n_candidates = [nn for nn in n_candidates if 140 <= nn <= 400]

    for n in n_candidates:
        if rem() < 45:
            break
        print("n={} rem={:.0f}s".format(n, rem()))
        h_cur = interp_to_n(best_h, n)

        # Subgradient phase - multiple settings
        for (tm, lr_i, dec, bs) in [(4, 0.25, 0.9995, 80), (6, 0.15, 0.9997, 100), (8, 0.10, 0.9998, 120)]:
            if rem() < 15:
                break
            h_sg, c5_sg = subgrad_opt(h_cur, n, 400, tm, lr_i, dec, bs)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                best_n = n
                h_cur = h_sg.copy()
                print("  SG tm={} lr={}: {:.8f}".format(tm, lr_i, best_c5))
            else:
                h_cur = h_sg  # use as next starting point anyway

        # SLSQP phase - increasing beta
        for beta in [40, 100, 250, 500, 800]:
            if rem() < 18:
                break
            try:
                mx = 600 if beta <= 100 else 350
                h_s, c5_s = run_slsqp(h_cur, n, beta, mx)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()
                    best_n = n
                    print("  SLQ b={}: {:.8f}".format(beta, best_c5))
                h_cur = h_s
            except Exception:
                break

        # Extra subgradient on best at this n
        if best_n == n and rem() > 25:
            h_s2, c5_s2 = subgrad_opt(best_h, n, 500, 5, 0.08, 0.9999, 150)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2.copy()
                print("  SG2: {:.8f}".format(best_c5))

    # High-resolution refinement
    if rem() > 70:
        nr = best_n
        hr = best_h.copy()
        for beta in [1200, 2000, 4000]:
            if rem() < 25:
                break
            try:
                h_r, c5_r = run_slsqp(hr, nr, beta, 400)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    hr = h_r.copy()
                    print("  Refine b={}: {:.8f}".format(beta, best_c5))
                else:
                    hr = h_r
            except Exception:
                break

    # Binary rounding + local search
    if rem() > 25:
        h_bin = binary_round(best_h, best_n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Bin: {:.8f}".format(best_c5))

    if rem() > 40:
        h_ls, c5_ls = local_bin_search(best_h, best_n, rng, max_rounds=5000, n_tries=40, n_flip_max=4)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            print("  LBS: {:.8f}".format(best_c5))

    # Simulated annealing on binary
    if rem() > 50 and np.all((best_h == 0) | (best_h == 1)):
        h_sa, c5_sa = sa_bin_search(best_h, best_n, rng, max_rounds=8000,
                                     t_init=0.001, t_final=1e-7, t_decay=0.9998)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            print("  SA: {:.8f}".format(best_c5))

    # More local search after SA
    if rem() > 30:
        h_ls2, c5_ls2 = local_bin_search(best_h, best_n, rng, max_rounds=5000, n_tries=50, n_flip_max=5)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
            print("  LBS2: {:.8f}".format(best_c5))

    # Try a different resolution for binary
    if rem() > 60:
        for n2 in [best_n + 1, best_n - 1, best_n + 2, best_n - 2]:
            if n2 < 100 or n2 > 500 or rem() < 20:
                continue
            h_i = interp_to_n(best_h, n2)
            h_b2 = binary_round(h_i, n2)
            c5_b2 = compute_c5(h_b2)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
                best_n = n2
                print("  Bin n={}: {:.8f}".format(n2, best_c5))
            if c5_b2 < best_c5 + 0.001 and rem() > 30:
                h_ls3, c5_ls3 = local_bin_search(h_b2, n2, rng, max_rounds=3000, n_tries=40, n_flip_max=4)
                if c5_ls3 < best_c5:
                    best_c5 = c5_ls3
                    best_h = h_ls3.copy()
                    best_n = n2
                    print("  LB n={}: {:.8f}".format(n2, best_c5))

    # Final SLSQP polish if non-binary
    if rem() > 25 and not np.all((best_h == 0) | (best_h == 1)):
        try:
            h_f, c5_f = run_slsqp(best_h, best_n, 3000, 500)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
                print("  Final SLQ: {:.8f}".format(best_c5))
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
