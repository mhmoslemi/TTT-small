    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:

        if diff > 0:

        else:

            if cnt == 0:

def slsqp_polish(h_init, n, max_iter=2000):
    """Final SLSQP polish on the continuous relaxation."""
    h0 = make_feasible(h_init.copy(), n)
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr)
    
    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        # Gradient of corr[k_max] w.r.t. h
        # corr[k] = sum_i h[i] * (1-h[i+k])
        # d corr[k] / d h[j] = (1-h[j+k]) - h[j-k] (with boundary handling)
        n_local = n
        grad = np.zeros(n_local)
        s = k_max - (n_local - 1)
        for j in range(n_local):
            # Contribution from h[j] as first factor
            if 0 <= j + s < n_local:
                grad[j] += (1.0 - h[j + s])
            # Contribution from h[j] as second factor (when j = i+s, i = j-s)
            if 0 <= j - s < n_local:
                grad[j] += -h[j - s]
        return grad
    
    bounds = [(0.0, 1.0)] * n
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    con = {'type': 'eq', 'fun': con_fun, 'jac': con_jac}
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h_init, compute_c5(h_init)

For binary sequences, this is exactly the problem of finding binary sequences with low autocorrelation. The Thue-Morse sequence has good but not optimal autocorrelation properties.

Actually, let me reconsider. The correlation np.correlate(h, 1-h, mode='full') is NOT the autocorrelation of h. It's the cross-correlation of h and (1-h). For binary h, 1-h is the complement. So we're looking at how much h overlaps with the complement of h shifted.

For binary h: h*(1-h_shifted) counts positions where h=1 and h_shifted=0.
The total: sum of h*(1-h_shifted) over all shifts = sum_i h[i] * (number of zeros in h) = (n/2) * (n/2) = n²/4.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    ones = np.where(tm == 1.0)[0]
    zeros = np.where(tm == 0.0)[0]
    target = n // 2
    if len(ones) > target:
        tm[ones[len(ones)-target:]] = 0.0
    elif len(ones) < target:
        tm[zeros[:(target-len(ones))]] = 1.0
    return tm


def antisymmetric_init(n, seed_val=7):
    h = np.zeros(n)
    half = n // 2
    for i in range(half):
        h[i] = 1.0 if bin(i * seed_val + 3).count('1') % 2 == 0 else 0.0
        h[i + half] = 1.0 - h[i]
    return make_feasible(h, n)


def random_binary(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    target = n // 2
    if len(ones) > target:
        excess = len(ones) - target
        to_flip = rng.choice(ones, excess, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    return h


def simulated_annealing(h_init, n, rng, t_start=0.5, t_end=0.003,
                        n_iterations=100000, timeout=100.0, double_swap_prob=0.3):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    best_h = h.copy()

    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()

    start_t = time.time()

    for it in range(n_iterations):
        if time.time() - start_t > timeout:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iterations - 1, 1))
        t = max(t, 1e-8)

        if rng.random() < double_swap_prob and len(ones_list) >= 2 and len(zeros_list) >= 2:
            i1 = ones_list[rng.randint(len(ones_list))]
            i2 = ones_list[rng.randint(len(ones_list))]
            if i1 == i2:
                continue
            j1 = zeros_list[rng.randint(len(zeros_list))]
            j2 = zeros_list[rng.randint(len(zeros_list))]
            if j1 == j2:
                continue
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            c5_try = compute_c5(h)
            if c5_try < current_c5 or rng.random() < np.exp(-(c5_try - current_c5) / t):
                current_c5 = c5_try
                ones_list.remove(i1)
                ones_list.remove(i2)
                ones_list.extend([j1, j2])
                zeros_list.remove(j1)
                zeros_list.remove(j2)
                zeros_list.extend([i1, i2])
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
        else:
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            c5_try = compute_c5(h)
            if c5_try < current_c5 or rng.random() < np.exp(-(c5_try - current_c5) / t):
                current_c5 = c5_try
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0

    return best_h, best_c5


def targeted_sa(h_init, n, rng, timeout=60.0, n_iterations=60000):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    best_h = h.copy()

    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()

    start_t = time.time()
    corr_cache = np.correlate(h, 1.0 - h, mode='full')

    for it in range(n_iterations):
        if time.time() - start_t > timeout:
            break
        t = max(0.02, 0.3 * (1.0 - it / n_iterations) ** 1.5)

        if it % 50 == 0:
            corr_cache = np.correlate(h, 1.0 - h, mode='full')

        top_k = np.argsort(corr_cache)[-4:]
        worst_idx = top_k[rng.randint(len(top_k))]
        s = worst_idx - (n - 1)

        bad_ones = []
        for i in ones_list:
            j = i + s
            if 0 <= j < n and h[j] == 0.0:
                bad_ones.append(i)

        if len(bad_ones) > 0 and rng.random() < 0.7:
            i = bad_ones[rng.randint(len(bad_ones))]
        else:
            i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < current_c5 or rng.random() < np.exp(-(c5_try - current_c5) / (t * 0.01)):
            current_c5 = c5_try
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5


def slsqp_polish(h_init, n, max_iter=3000):
    h0 = make_feasible(h_init.copy(), n)

    def obj_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        val = corr[k_max]
        grad = np.zeros(n)
        s = k_max - (n - 1)
        for j in range(n):
            if 0 <= j + s < n:
                grad[j] += (1.0 - h[j + s])
            if 0 <= j - s < n:
                grad[j] += (-h[j - s])
        return val, grad

    bounds = [(0.0, 1.0)] * n

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    con = {'type': 'eq', 'fun': con_fun, 'jac': con_jac}

    best_val, _ = obj_and_grad(h0)
    best_x = h0.copy()
    try:
        res = minimize(obj_and_grad, h0, jac=True, method='SLSQP',
                       bounds=bounds, constraints=con,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            val, _ = obj_and_grad(res.x)
            if val < best_val:
                best_val = val
                best_x = res.x.copy()
    except Exception:
        pass
    h_out = make_feasible(best_x, n)
    return h_out, compute_c5(h_out)


def smooth_max_slsqp(h_init, n, beta_list, max_iter=500):
    h_cur = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    bounds = [(0.0, 1.0)] * n

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    con = {'type': 'eq', 'fun': con_fun, 'jac': con_jac}

    for beta in beta_list:
        def obj(x):
            h = np.clip(x, 0.0, 1.0)
            corr = np.correlate(h, 1.0 - h, mode='full')
            cmax = np.max(corr)
            shifted = beta * (corr - cmax)
            shifted[shifted > 50] = 50
            val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
            return val

        try:
            res = minimize(obj, h_cur, method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': max_iter, 'ftol': 1e-13})
            if res.fun < 1e10:
                candidate = make_feasible(res.x, n)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = candidate.copy()
                    h_cur = candidate.copy()
        except Exception:
            pass
    return best_h, best_c5


def generate_inits(n, rng, h_parent=None):
    inits = []
    if h_parent is not None and len(h_parent) > 0:
        h_p = make_feasible(interp_h(h_parent, n), n)
        inits.append(h_p)
    inits.append(thue_morse_init(n))
    inits.append(antisymmetric_init(n, 7))
    inits.append(antisymmetric_init(n, 11))
    for _ in range(4):
        inits.append(random_binary(n, rng))
    for shift in [1, 3, 5]:
        tm = np.zeros(n)
        for i in range(n):
            tm[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
        ones = np.where(tm == 1.0)[0]
        zeros = np.where(tm == 0.0)[0]
        target = n // 2
        if len(ones) > target:
            tm[ones[len(ones) - target:]] = 0.0
        elif len(ones) < target:
            tm[zeros[:(target - len(ones))]] = 1.0
        inits.append(tm)
    return inits

    best_c5_global = compute_c5(best_h_global)

    return (best_h_global, best_c5_global, len(best_h_global))
