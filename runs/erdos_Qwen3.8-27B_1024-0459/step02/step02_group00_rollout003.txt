def project_fast(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    for _ in range(12):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            if diff > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + diff)
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
            s = diff
            for idx in order:
                if s <= 0: break
                room = 1.0 - h[idx]
                add = room if room < s else s
                h[idx] += add
                s -= add
        else:
            order = np.argsort(h)[::-1]
            s = -diff
            for idx in order:
                if s <= 0: break
                avail = h[idx]
                sub = avail if avail < s else s
                h[idx] -= sub
                s -= sub
    return h

    h = np.asarray(h, dtype=float)
    n = len(h)
    if n == 0:
        return float('inf')
    q = compute_q(h)
    m = float(np.max(q))
    if m < 0.0:
        m = 0.0
    return m * (2.0 / float(n))


def project_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0

    for _ in range(12):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = h < (1.0 - 1e-13)
        else:
            mask = h > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt > 0:
            h[mask] += diff / float(cnt)
        else:
            if diff > 0.0:
                idx = int(np.argmax(1.0 - h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + diff)
        np.clip(h, 0.0, 1.0, out=h)

    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
            s = diff
            for idx in order:
                if s <= 0.0:
                    break
                room = 1.0 - h[idx]
                add = room if room < s else s
                h[idx] += add
                s -= add
        else:
            order = np.argsort(h)[::-1]
            s = -diff
            for idx in order:
                if s <= 0.0:
                    break
                avail = h[idx]
                sub = avail if avail < s else s
                h[idx] -= sub
                s -= sub
    return h


def force_exact_sum(h, n):
    h = np.array(h, dtype=float, copy=True)
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - h
            idx = int(np.argmax(room))
            add = min(diff, room[idx])
            h[idx] += add
        else:
            avail = h
            idx = int(np.argmax(avail))
            sub = min(-diff, avail[idx])
            h[idx] -= sub
    return h


def add_lag_grad(grad, h, off, w):
    n = len(h)
    if w == 0.0:
        return
    if off >= 0:
        m = n - off
        if m <= 0:
            return
        grad[off:off + m] += w * (1.0 - h[:m])
        grad[:m] -= w * h[off:off + m]
    else:
        a = -off
        m = n - a
        if m <= 0:
            return
        grad[:m] += w * (1.0 - h[a:a + m])
        grad[a:] -= w * h[:m]


def smooth_value_eval(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    beta = float(_OPT_BETA)
    if beta <= 0.0:
        beta = 1.0
    K = int(_OPT_K)
    if K <= 0:
        K = 1

    q = compute_q(h)
    m = float(np.max(q))
    if not np.isfinite(m):
        return 1e12, q, np.arange(q.size), np.ones(q.size) / max(1, q.size)

    if K > q.size:
        K = q.size
    if K < q.size:
        idx0 = np.argpartition(q, -K)[-K:]
    else:
        idx0 = np.arange(q.size)

    vals = q[idx0]
    a = beta * (vals - m)
    a = np.clip(a, -700.0, 0.0)
    ea = np.exp(a)
    s = float(ea.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.ones(idx0.size) / idx0.size
        F = m
    else:
        w = ea / s
        F = m + math.log(float(s)) / beta
    return float(F), q, idx0, w


def opt_fun(x):
    try:
        F, _, _, _ = smooth_value_eval(x)
        return float(F)
    except Exception:
        x = np.asarray(x, dtype=float)
        return 1e10 + 1e-6 * float(np.sum(x * x))


def opt_jac(x):
    try:
        n = int(_OPT_N)
        F, q, idx0, w = smooth_value_eval(x)
        grad = np.zeros(n, dtype=float)
        for t in range(idx0.size):
            wgt = float(w[t])
            if wgt > 1e-12:
                off = int(idx0[t]) - (n - 1)
                add_lag_grad(grad, x, off, wgt)
        return grad
    except Exception:
        return np.zeros(len(x), dtype=float)


def con_sum(x):
    return float(np.sum(x) - _OPT_TGT)


def con_sum_jac(x):
    return np.ones(len(x), dtype=float)


def attempt_slsqp(h0, beta, K, maxiter, ftol):
    global _OPT_N, _OPT_BETA, _OPT_K, _OPT_TGT

    n = int(len(h0))
    h0 = project_feasible(h0, n)
    c0 = compute_c5(h0)

    _OPT_N = n
    _OPT_BETA = float(beta)
    _OPT_K = int(K)
    _OPT_TGT = n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]

    try:
        res = minimize(opt_fun,
                       h0,
                       jac=opt_jac,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': int(maxiter), 'ftol': float(ftol), 'disp': False})
        h = project_feasible(res.x, n)
        c = compute_c5(h)
        if not np.isfinite(c):
            return h0, c0
        return h, float(c)
    except Exception:
        return h0, float(c0)


def multibeta_slsqp(h0, plans, K, ftol):
    h = project_feasible(h0, len(h0)).copy()
    c = compute_c5(h)
    for plan in plans:
        if remaining_time() < 25:
            break
        beta = float(plan[0])
        maxiter = int(plan[1])
        try:
            h2, c2 = attempt_slsqp(h, beta, K, maxiter, ftol)
            if np.isfinite(c2) and c2 < c - 1e-14:
                h = h2
                c = c2
        except Exception:
            pass
    return h, float(c)


def pg_smooth(h0, beta, K, max_iters=300, alpha_init=0.25):
    global _OPT_BETA, _OPT_K

    n = int(len(h0))
    _OPT_BETA = float(beta)
    _OPT_K = int(K)

    h = project_feasible(h0, n).copy()
    best = compute_c5(h)
    F, _, _, _ = smooth_value_eval(h)
    alpha = float(alpha_init)

    for it in range(int(max_iters)):
        if remaining_time() < 3:
            break
        F, q, idx0, w = smooth_value_eval(h)
        if not np.isfinite(F):
            break
        grad = np.zeros(n, dtype=float)
        for t in range(idx0.size):
            wgt = float(w[t])
            if wgt > 1e-12:
                off = int(idx0[t]) - (n - 1)
                add_lag_grad(grad, h, off, wgt)

        grad -= grad.mean()
        gn = float(np.linalg.norm(grad))
        if gn < 1e-11:
            break

        improved = False
        a = alpha
        for _ in range(14):
            h_try = project_feasible(h - a * grad, n)
            F_try, q_try, _, _ = smooth_value_eval(h_try)
            c_try = float(np.max(q_try))
            if c_try < 0.0:
                c_try = 0.0
            c_try *= (2.0 / float(n))

            if np.isfinite(F_try) and F_try < F - 1e-13 and c_try <= best + 1e-8:
                h = h_try
                F = F_try
                if c_try < best:
                    best = c_try
                alpha = a
                improved = True
                break
            a *= 0.5

        if not improved:
            alpha *= 0.5
            if alpha < 1e-6:
                break

    h = project_feasible(h, n)
    c = compute_c5(h)
    if c < best:
        best = c
    return h, float(best)


def active_subgrad_polish(h0, max_iters=400, K=4):
    n = int(len(h0))
    h = project_feasible(h0, n).copy()
    best = compute_c5(h)
    alpha = 0.05
    no_improve = 0
    K = int(K)
    if K <= 0:
        K = 1

    for it in range(int(max_iters)):
        if remaining_time() < 3:
            break
        q = compute_q(h)
        m = float(np.max(q))
        if not np.isfinite(m):
            break

        if K < q.size:
            idx = np.argpartition(q, -K)[-K:]
        else:
            idx = np.arange(q.size)

        vals = q[idx]
        a = 80.0 * (vals - m)
        a = np.clip(a, -700.0, 0.0)
        ea = np.exp(a)
        s = float(ea.sum())
        if s <= 0.0 or not np.isfinite(s):
            w = np.ones(idx.size) / idx0.size if False else np.ones(idx.size) / idx.size
        else:
            w = ea / s

        grad = np.zeros(n, dtype=float)
        for t in range(idx.size):
            wgt = float(w[t])
            if wgt > 1e-12:
                off = int(idx[t]) - (n - 1)
                add_lag_grad(grad, h, off, wgt)

        grad -= grad.mean()
        gscale = float(np.max(np.abs(grad)))
        if gscale < 1e-12:
            break
        g = grad / gscale

        improved = False
        a = alpha
        for _ in range(12):
            h_try = project_feasible(h - a * g, n)
            c_try = compute_c5(h_try)
            if c_try < best - 1e-13:
                h = h_try
                best = c_try
                alpha = a
                improved = True
                break
            a *= 0.5

        if not improved:
            for a_small in (1e-4, 5e-5, 2e-5, 1e-5, 5e-6):
                h_try = project_feasible(h - a_small * g, n)
                c_try = compute_c5(h_try)
                if c_try < best - 1e-13:
                    h = h_try
                    best = c_try
                    alpha = a_small
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            alpha = max(1e-6, alpha * 0.6)
            if no_improve > 40:
                break

    h = project_feasible(h, n)
    c = compute_c5(h)
    if c < best:
        best = c
    return h, float(best)


def q_update_inplace(h, q, i, new_val):
    old = float(h[i])
    new = float(new_val)
    if old == new:
        return
    delta = new - old
    if delta == 0.0:
        return

    n = len(h)

    # h_i appears as the shifted factor h_{n+k}:
    # affected lags k in [i-n+1, i]
    q[i:i + n] += delta * (1.0 - h[::-1])

    # h_i appears as the base factor 1-h_{n}:
    # affected lags k in [-i, n-1-i]
    q[n - 1 - i:2 * n - 1 - i] -= delta * h

    # same-lag quadratic correction
    q[n - 1] -= delta * delta

    h[i] = new


def make_binary_from_float(h, n):
    h = np.asarray(h, dtype=float)
    h = np.nan_to_num(h, nan=0.0, posinf=1.0, neginf=0.0)
    if h.size != n:
        tmp = np.zeros(n, dtype=float)
        m = min(n, h.size)
        tmp[:m] = h[:m]
        h = tmp

    target = n // 2
    hb = np.zeros(n, dtype=float)
    if target > 0:
        order = np.argsort(h)[::-1]
        hb[order[:target]] = 1.0
    return hb


def random_binary(n, rng):
    hb = np.zeros(n, dtype=float)
    target = n // 2
    if target > 0:
        idx = rng.permutation(n)
        hb[idx[:target]] = 1.0
    return hb


def make_tm(n):
    h = np.zeros(n, dtype=float)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        if cnt:
            h[i] = 1.0
    return h


def make_golden_score(n, phase=0.0):
    gr = (1.0 + math.sqrt(5.0)) / 2.0
    h = np.zeros(n, dtype=float)
    for i in range(n):
        frac = (i * gr + phase) % 1.0
        h[i] = 0.5 - abs(frac - 0.5)
    return h


def make_sinusoid(n, freq, phase):
    t = np.arange(n, dtype=float) / float(n)
    h = 0.5 + 0.45 * np.sin(2.0 * math.pi * freq * t + phase)
    return np.clip(h, 0.0, 1.0)


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return np.array(src_h, dtype=float, copy=True)
    if src_n <= 0:
        return 0.5 * np.ones(dst_n, dtype=float)
    src_h = np.asarray(src_h, dtype=float)
    xp = (np.arange(src_n) + 0.5) * (2.0 / float(src_n))
    xn = (np.arange(dst_n) + 0.5) * (2.0 / float(dst_n))
    return np.interp(xn, xp, src_h)


def greedy_binary(n, rng):
    h = np.zeros(n, dtype=float)
    q = np.zeros(2 * n - 1, dtype=float)
    target = n // 2

    for s in range(target):
        if remaining_time() < 3:
            break
        best_j = -1
        best_M = 1e30

        if n <= 256:
            candidates = np.arange(n)
        else:
            candidates = rng.permutation(n)

        cand_count = 0
        for jj in range(candidates.size):
            j = int(candidates[jj])
            if h[j] > 0.5:
                continue
            cand_count += 1
            q_update_inplace(h, q, j, 1.0)
            Mt = float(np.max(q))
            if Mt < best_M - 1e-12 or (abs(Mt - best_M) < 1e-12 and rng.random() < 0.5):
                best_M = Mt
                best_j = j
            q_update_inplace(h, q, j, 0.0)
            if cand_count % 64 == 0:
                q = compute_q(h)

        if best_j < 0:
            break
        q_update_inplace(h, q, best_j, 1.0)
        q = compute_q(h)

    h = make_binary_from_float(h, n)
    c = compute_c5(h)
    return h, float(c)


def hill_search_binary(h0, n, rng, max_trials, plateau_prob=0.15):
    h = make_binary_from_float(h0, n)
    q = compute_q(h)
    M = float(np.max(q))
    best_h = h.copy()
    best_c = M * (2.0 / float(n))

    ones = np.flatnonzero(h > 0.5).astype(int)
    zeros = np.flatnonzero(h < 0.5).astype(int)
    stall = 0
    stall_limit = max(2500, min(12000, max_trials // 4))

    for t in range(int(max_trials)):
        if remaining_time() < 2:
            break
        if ones.size == 0 or zeros.size == 0:
            break

        ii = int(rng.randint(ones.size))
        jj = int(rng.randint(zeros.size))
        i = int(ones[ii])
        j = int(zeros[jj])

        if h[i] <= 0.5 + 1e-12 or h[j] >= 0.5 - 1e-12:
            ones = np.flatnonzero(h > 0.5).astype(int)
            zeros = np.flatnonzero(h < 0.5).astype(int)
            continue

        q_update_inplace(h, q, i, 0.0)
        q_update_inplace(h, q, j, 1.0)
        Mt = float(np.max(q))

        accept = False
        if Mt < M - 1e-12:
            accept = True
        elif abs(Mt - M) <= 1e-12:
            if rng.random() < float(plateau_prob):
                accept = True

        if accept:
            M = Mt
            ones[ii] = j
            zeros[jj] = i
            c = Mt * (2.0 / float(n))
            if c < best_c - 1e-13:
                best_c = c
                best_h = h.copy()
                stall = 0
            else:
                stall = max(0, stall - 1)
        else:
            q_update_inplace(h, q, j, 0.0)
            q_update_inplace(h, q, i, 1.0)
            stall += 1

        if (t + 1) % 1000 == 0:
            q = compute_q(h)
            M = float(np.max(q))
            c = M * (2.0 / float(n))
            if c < best_c - 1e-13:
                best_c = c
                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5).astype(int)
            zeros = np.flatnonzero(h < 0.5).astype(int)

        if stall > stall_limit:
            break

    q = compute_q(h)
    c = float(np.max(q)) * (2.0 / float(n))
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = float(budget_s)
    rng = np.random.RandomState(seed)

    best_h = None
    best_n = 0
    best_c5 = float('inf')

    # Use the supplied parent construction directly if present.
    try:
        parent_h = np.asarray(initial_h_values, dtype=float)
        parent_n = int(parent_h.size)
        if parent_n > 0:
            parent_h = project_feasible(parent_h, parent_n)
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h
            best_n = parent_n
            best_c5 = float(parent_c5)
    except Exception:
        best_h = None

    if best_h is None:
        best_n = 160
        try:
            best_h = project_feasible(make_sinusoid(best_n, 4, 0.17), best_n)
            best_c5 = compute_c5(best_h)
            hg, cg = greedy_binary(best_n, rng)
            if cg < best_c5:
                best_h = hg
                best_c5 = float(cg)
        except Exception:
            best_h = project_feasible(make_sinusoid(best_n, 4, 0.17), best_n)
            best_c5 = compute_c5(best_h)
            best_n = best_n

    # ------------------------------------------------------------------
    # Stage 1: local smooth nonconvex optimization around the parent grid
    # and neighboring even resolutions.
    # ------------------------------------------------------------------
    near = []
    if best_n > 0:
        cands = [
            best_n,
            best_n - 16, best_n + 16,
            best_n - 32, best_n + 32,
            128, 144, 160, 176, 192, 208, 224, 256
        ]
        for nn in cands:
            if nn % 2 == 0 and 64 <= nn <= 240:
                if nn not in near:
                    near.append(nn)
        near = near[:4]

    for n in near:
        if remaining_time() < 140:
            break
        if n % 2 != 0 or n < 64 or n > 240:
            continue
        try:
            if best_n > 0:
                h0 = interp_to(best_h, best_n, n)
            else:
                h0 = make_sinusoid(n, 4, 0.25)
            h0 = project_feasible(h0, n)

            K = 8 if n <= 200 else 6
            plans = [(5.0, 60), (20.0, 80), (80.0, 110)]
            h2, c2 = multibeta_slsqp(h0, plans, K, 1e-12)
            if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                best_h = h2
                best_n = n
                best_c5 = float(c2)

            if remaining_time() > 220:
                hnoise = project_feasible(h0 + 0.02 * (rng.rand(n) - 0.5), n)
                plans2 = [(20.0, 60), (100.0, 90)]
                h3, c3 = multibeta_slsqp(hnoise, plans2, K, 1e-12)
                if np.isfinite(c3) and c3 < best_c5 - 1e-13:
                    best_h = h3
                    best_n = n
                    best_c5 = float(c3)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 2: binary constructions and exact binary local search.
    # Binary solutions at larger n can give a cleaner lower bound; then
    # fractional polishing is applied.
    # ------------------------------------------------------------------
    bin_ns = []
    bin_cands = [best_n, 160, 192, 200, 224, 256, 288, 320]
    for nn in bin_cands:
        if nn % 2 == 0 and 96 <= nn <= 360:
            if nn not in bin_ns:
                bin_ns.append(nn)
    bin_ns = bin_ns[:4]

    for n in bin_ns:
        if remaining_time() < 90:
            break

        local_h = None
        local_c = float('inf')
        starts = []

        try:
            if best_n > 0:
                starts.append(make_binary_from_float(interp_to(best_h, best_n, n), n))
            starts.append(make_binary_from_float(make_tm(n), n))
            starts.append(make_binary_from_float(make_golden_score(n, 0.0), n))
            starts.append(make_binary_from_float(make_golden_score(n, 0.31), n))
            starts.append(random_binary(n, rng))
            starts.append(random_binary(n, rng))

            if remaining_time() > 35:
                hg, cg = greedy_binary(n, rng)
                starts.append(hg)

            for hb in starts:
                hb = project_feasible(hb, n)
                c = compute_c5(hb)
                if c < local_c:
                    local_c = float(c)
                    local_h = hb
                if np.isfinite(c) and c < best_c5 - 1e-13:
                    best_h = hb
                    best_n = n
                    best_c5 = float(c)

            if local_h is not None and remaining_time() > 40:
                max_trials = 16000 if n <= 200 else 8000
                h2, c2 = hill_search_binary(local_h, n, rng, max_trials, plateau_prob=0.15)
                if np.isfinite(c2) and c2 < local_c - 1e-13:
                    local_h = h2
                    local_c = float(c2)
                if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                    best_h = h2
                    best_n = n
                    best_c5 = float(c2)

                # Fractional improvement from the best binary object.
                if remaining_time() > 60 and local_c < 0.395:
                    if n <= 220:
                        Kb = 8 if n <= 200 else 6
                        plans_b = [(10.0, 60), (50.0, 80), (200.0, 100)]
                        h3, c3 = multibeta_slsqp(local_h, plans_b, Kb, 1e-12)
                        if np.isfinite(c3) and c3 < best_c5 - 1e-13:
                            best_h = h3
                            best_n = n
                            best_c5 = float(c3)
                    else:
                        h3, c3 = active_subgrad_polish(local_h, max_iters=120, K=4)
                        if np.isfinite(c3) and c3 < best_c5 - 1e-13:
                            best_h = h3
                            best_n = n
                            best_c5 = float(c3)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 3: exact active-subgradient polishing and smooth projected
    # gradient on the current global best.
    # ------------------------------------------------------------------
    if best_h is not None and remaining_time() > 90:
        try:
            it3 = 350 if best_n <= 200 else 140
            h2, c2 = active_subgrad_polish(best_h, max_iters=it3, K=4)
            if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                best_h = h2
                best_c5 = float(c2)
        except Exception:
            pass

    if best_h is not None and remaining_time() > 100:
        try:
            h2, c2 = pg_smooth(best_h, beta=10.0, K=8, max_iters=260, alpha_init=0.22)
            if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                best_h = h2
                best_c5 = float(c2)
        except Exception:
            pass

    if best_h is not None and remaining_time() > 110:
        try:
            h2, c2 = pg_smooth(best_h, beta=60.0, K=6, max_iters=220, alpha_init=0.12)
            if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                best_h = h2
                best_c5 = float(c2)
        except Exception:
            pass

    # One more binary refinement from the fractional global best.
    if best_h is not None and remaining_time() > 90:
        try:
            n = best_n
            hb = make_binary_from_float(best_h, n)
            h2, c2 = hill_search_binary(hb, n, rng, 8000 if n > 200 else 12000, plateau_prob=0.12)
            if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                best_h = h2
                best_n = n
                best_c5 = float(c2)
            if remaining_time() > 50:
                h3, c3 = active_subgrad_polish(h2, max_iters=100 if n > 200 else 200, K=4)
                if np.isfinite(c3) and c3 < best_c5 - 1e-13:
                    best_h = h3
                    best_n = n
                    best_c5 = float(c3)
        except Exception:
            pass

    # Final high-beta SLSQP polish on the best known object.
    if best_h is not None and remaining_time() > 90:
        try:
            n = best_n
            K = 8 if n <= 200 else 6
            plans_final = [(40.0, 70), (150.0, 110)]
            h2, c2 = multibeta_slsqp(best_h, plans_final, K, 1e-12)
            if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                best_h = h2
                best_n = n
                best_c5 = float(c2)
        except Exception:
            pass

    # A couple of stochastic last-chance restarts.
    for rnd in range(3):
        if remaining_time() < 100:
            break
        try:
            if rnd == 0:
                n2 = best_n
            elif rnd == 1:
                n2 = 192 if best_n != 192 else 224
            else:
                n2 = 256 if best_n != 256 else 224

            if n2 % 2 != 0 or n2 < 64 or n2 > 320:
                n2 = 160

            if best_n > 0:
                h0 = interp_to(best_h, best_n, n2)
            else:
                h0 = make_sinusoid(n2, 5, 0.35)
            h0 = project_feasible(h0 + 0.045 * (rng.rand(n2) - 0.5), n2)

            K = 8 if n2 <= 200 else 6
            plans_r = [(10.0, 50), (70.0, 70)]
            h2, c2 = multibeta_slsqp(h0, plans_r, K, 1e-12)
            if np.isfinite(c2) and c2 < best_c5 - 1e-13:
                best_h = h2
                best_n = n2
                best_c5 = float(c2)

            if remaining_time() > 40:
                h3, c3 = active_subgrad_polish(h2, max_iters=90, K=4)
                if np.isfinite(c3) and c3 < best_c5 - 1e-13:
                    best_h = h3
                    best_n = n2
                    best_c5 = float(c3)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Final feasibility enforcement and exact evaluation.
    # ------------------------------------------------------------------
    try:
        best_h = force_exact_sum(project_feasible(best_h, best_n), best_n)
        final_c5 = compute_c5(best_h)
        if np.isfinite(final_c5) and final_c5 < best_c5:
            best_c5 = float(final_c5)

        try:
            eval_c5 = evaluate_erdos_solution(best_h)
            if np.isfinite(eval_c5) and float(eval_c5) < best_c5:
                best_c5 = float(eval_c5)
        except Exception:
            pass
    except Exception:
        if best_h is None:
            best_n = 160
            best_h = project_feasible(make_sinusoid(best_n, 4, 0.17), best_n)
        else:
            best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        best_n = 160
        best_h = make_sinusoid(best_n, 4, 0.17)
        best_c5 = compute_c5(best_h)

    best_h = np.asarray(best_h, dtype=float)
    return (best_h, float(best_c5), int(best_n))
