def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta)

def obj_lbfgs(x, n, beta, lam):
    dx = 2.0 / n
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max(corr, beta) * dx
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return val + penalty

def grad_lbfgs(x, n, beta, lam):

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_func(x, n, beta, lam):
    dx = 2.0 / n
    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max_val(corr, beta) * dx
    val += lam * (np.sum(h) - n / 2.0) ** 2
    return val

def grad_func(x, n, beta, lam):

    h = np.clip(x, 0, 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)

        L = k - (n - 1)

    cur_score = best_score
    for it in range(max_iters):
        T = T0 * (Tend / T0) ** (it / max_iters)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1] = 0.0
        h[i0] = 1.0
        new_score = score_binary(h)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return best_h, best_score

def targeted_flip(h, n, rng, max_rounds, budget_fn=None):
    best_h = h.copy()
    best_score = score_binary(best_h)
    h = h.copy()
    for r in range(max_rounds):
        if budget_fn is not None and budget_fn() < 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        max_lags = np.where(corr >= max_val - 1e-10)[0]
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for _ in range(80):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            if i1 == i0:
                continue
            h[i1] = 0.0
            h[i0] = 1.0
            new_score = score_binary(h)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                improved = True
            else:
                h[i1] = 1.0
                h[i0] = 0.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        if not improved:
            break
    return best_h, best_score

def continuous_optimize(h0, n, betas, budget_fn):
    cur = h0.copy()
    for beta in betas:
        if budget_fn() < 25:
            break
        lam = 500.0
        try:
            res = minimize(
                fun=lambda x: obj_func(x, n, beta, lam),
                x0=cur,
                jac=lambda x: grad_func(x, n, beta, lam),
                method='L-BFGS-B',
                bounds=[(0, 1)] * n,
                options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            cur = project_feasible(res.x, n)
        except Exception:
            break
    return cur

def round_to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1)[0]
        vals = h[ones]
        order = np.argsort(vals)[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0)[0]
        vals = h[zeros]
        order = np.argsort(-vals)[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = make_sturmian(best_n, 0.6180339887, rng)
        best_c5 = compute_c5(best_h)
        print("Generated: n={}, C5={:.6f}".format(best_n, best_c5))

    n_candidates = sorted(set([best_n, 200, 250, 300, 400]))
    n_candidates = [nn for nn in n_candidates if 32 <= nn <= 500]

    for n in n_candidates:
        if remaining() < 50:
            break
        print("=== n={} (remaining {:.0f}s) ===".format(n, remaining()))

        # Multiple initializations
        inits = []
        h_interp = interp_h(best_h, n)
        inits.append(h_interp)

        # Sturmian initializations
        alphas = [0.6180339887, 0.5 + 0.1 * np.sqrt(2), 0.7, 0.5 + 0.3 / 2.0]
        for alpha in alphas:
            inits.append(make_sturmian(n, alpha, rng))

        # Random binary
        h_rand = np.zeros(n)
        h_rand[rng.choice(n, size=n // 2, replace=False)] = 1.0
        inits.append(h_rand)

        for h0 in inits:
            if remaining() < 30:
                break
            h0 = project_feasible(h0.copy(), n)

            # Continuous optimization
            if remaining() > 40:
                h_cont = continuous_optimize(h0, n, [30, 100, 300], remaining)
                c5_cont = compute_c5(h_cont)
                if c5_cont < best_c5:
                    best_c5 = c5_cont
                    best_h = h_cont.copy()
                    best_n = n
                    print("  Continuous: C5={:.6f}".format(best_c5))

                # Binary from continuous
                h_bin = round_to_binary(h_cont, n)
                c5_bin = score_binary(h_bin)
            else:
                h_bin = h0.copy()
                if not np.all((h_bin == 0) | (h_bin == 1)):
                    h_bin = round_to_binary(h_bin, n)
                c5_bin = score_binary(h_bin)

            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n
                print("  Binary init: C5={:.6f}".format(best_c5))

            # Greedy binary search
            if remaining() > 20:
                h_g, c5_g = greedy_binary_search(h_bin, n, rng, 200)
                if c5_g < best_c5:
                    best_c5 = c5_g
                    best_h = h_g.copy()
                    best_n = n
                    print("  Greedy: C5={:.6f}".format(best_c5))
                h_bin = h_g

            # Simulated annealing
            if remaining() > 20:
                n_sa = int(min(8000, remaining() * 50))
                h_sa, c5_sa = simulated_annealing_binary(h_bin, n, rng, n_sa)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n
                    print("  SA: C5={:.6f}".format(best_c5))
                h_bin = h_sa

            # Targeted search on best
            if remaining() > 15:
                h_t, c5_t = targeted_flip(best_h, n, rng, 300, remaining)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h_t.copy()
                    best_n = n
                    print("  Targeted: C5={:.6f}".format(best_c5))

    # Final intensive search on the best solution
    if remaining() > 30:
        n = best_n
        if not np.all((best_h == 0) | (best_h == 1)):
            h_bin = round_to_binary(best_h, n)
            best_h = h_bin
            best_c5 = score_binary(best_h)

        print("Final intensive: n={}, C5={:.6f}".format(n, best_c5))

        # Multiple SA rounds with perturbation
        for sa_round in range(5):
            if remaining() < 15:
                break
            # Perturb best solution
            h_pert = best_h.copy()
            n_flips = rng.randint(3, max(4, n // 10))
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_flips):
                i1 = ones[rng.randint(len(ones))]
                i0 = zeros[rng.randint(len(zeros))]
                h_pert[i1] = 0.0
                h_pert[i0] = 1.0
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            n_iters = int(min(10000, remaining() * 40))
            h_sa, c5_sa = simulated_annealing_binary(h_pert, n, rng, n_iters, T0=0.05, Tend=0.0002)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                print("  SA round {}: C5={:.6f}".format(sa_round, best_c5))

            # Greedy after SA
            if remaining() > 15:
                h_g, c5_g = greedy_binary_search(h_sa, n, rng, 300)
                if c5_g < best_c5:
                    best_c5 = c5_g
                    best_h = h_g.copy()
                    print("  Greedy round {}: C5={:.6f}".format(sa_round, best_c5))

        # Final targeted
        if remaining() > 15:
            h_t, c5_t = targeted_flip(best_h, n, rng, 500, remaining)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                print("  Final targeted: C5={:.6f}".format(best_c5))

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
