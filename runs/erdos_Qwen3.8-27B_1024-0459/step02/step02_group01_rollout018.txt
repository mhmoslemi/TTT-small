    h_bin = (h > 0.5).astype(float)

    n_want = n // 2

        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0

    T_min = 1e-7
    cooling = 0.9995

        zeros_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zeros_idx])[:n_want - n_ones]
        h_b[zeros_idx[order]] = 1.0
    return h_b

def binary_simulated_annealing(h_init, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    curr = round_to_binary(h_init, n).copy()
    curr_c5 = compute_c5(curr)
    best = curr.copy()
    best_c5 = curr_c5
    
    T = 0.003
    T_min = 1e-7
    cooling = 0.99992
    t_start = time.time()
    n_iter = 0
    
    while time.time() - t_start < time_limit:
        ones = np.where(curr == 1)[0]
        zeros = np.where(curr == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        curr[a], curr[b] = curr[b], curr[a]
        new_c5 = compute_c5(curr)
        delta = new_c5 - curr_c5
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, T_min)):
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best = curr.copy()
        else:
            curr[a], curr[b] = curr[b], curr[a]
        T *= cooling
        n_iter += 1
        if n_iter % 50000 == 0 and T < T_min:
            break
    
    return best, best_c5, n_iter

def directed_local_search(h_bin, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    curr = h_bin.copy()
    curr_c5 = compute_c5(curr)
    best = curr.copy()
    best_c5 = curr_c5
    t_start = time.time()
    max_rounds = 50
    
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(curr, 1.0 - curr, mode='full')
        k_star = np.argmax(corr)
        offset = k_star - (n - 1)
        improved = True
        while improved and time.time() - t_start < time_limit:
            improved = False
            ones = np.where(curr == 1)[0]
            zeros = np.where(curr == 0)[0]
            rng.shuffle(ones)
            rng.shuffle(zeros)
            for a in ones[:200]:
                if time.time() - t_start > time_limit:
                    break
                for b in zeros[:200]:
                    if a == b:
                        continue
                    curr[a], curr[b] = curr[b], curr[a]
                    new_c5 = compute_c5(curr)
                    if new_c5 < curr_c5:
                        curr_c5 = new_c5
                        if curr_c5 < best_c5:
                            best_c5 = curr_c5
                            best = curr.copy()
                        improved = True
                    else:
                        curr[a], curr[b] = curr[b], curr[a]
                    if improved:
                        break
                if improved:
                    break
    return best, best_c5

def _smooth_obj_func(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / beta
    return smax * (2.0 / n)

def _eq_con_func(x, n):
    return np.sum(x) - n / 2.0

def continuous_refine(h_init, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    h_cur = project_feasible(h_init, n)
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    
    beta_schedule = [20, 50, 100, 300, 800, 2000, 5000, 10000, 30000, 80000]
    
    for beta in beta_schedule:
        if time.time() - t_start > time_limit - 5:
            break
        def obj(x):
            return _smooth_obj_func(x, n, beta)
        def eqc(x):
            return _eq_con_func(x, n)
        
        cons = [{'type': 'eq', 'fun': eqc}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-18, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            try:
                res = minimize(obj, h_cur, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 2000, 'ftol': 1e-20, 'gtol': 1e-16})
                h_cur = project_feasible(res.x, n)
            except Exception:
                pass
    
    return h_cur

def interpolate_h(h_src, n_src, n_dst):
    x_src = (np.arange(n_src) + 0.5) * 2.0 / n_src
    x_dst = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng_main = np.random.RandomState(seed)
    
    def tl():
        return max(0.0, budget_s - (time.time() - t0) - 3.0)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    n_parent = 0
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
        best_h = h_parent.copy()
        best_c5 = compute_c5(h_parent)
        best_n = n_parent
    except Exception:
        pass
    
    if best_h is None:
        best_n = 192
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary SA at parent resolution (35% of time)
    t_phase1 = 0.25 * budget_s
    n_sa = best_n
    h_bin_init = round_to_binary(best_h, n_sa)
    
    sa_best_h = h_bin_init.copy()
    sa_best_c5 = compute_c5(h_bin_init)
    
    n_runs_sa = 6
    time_per_sa = t_phase1 / n_runs_sa
    
    for i in range(n_runs_sa):
        if tl() < 10:
            break
        s = seed + 100 + i * 7
        h_res, c5_res, _ = binary_simulated_annealing(h_bin_init, n_sa, time_per_sa, s)
        if c5_res < sa_best_c5:
            sa_best_c5 = c5_res
            sa_best_h = h_res.copy()
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            best_n = n_sa
    
    # Phase 2: Directed local search (15% of time)
    t_phase2 = 0.15 * budget_s
    h_ls, c5_ls = directed_local_search(sa_best_h, n_sa, t_phase2, seed + 500)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
        best_n = n_sa
        sa_best_h = h_ls.copy()
        sa_best_c5 = c5_ls
    
    # Phase 3: Try other resolutions for binary SA (15% of time)
    t_phase3 = 0.15 * budget_s
    other_ns = [256, 128]
    for n_try in other_ns:
        if tl() < 30:
            break
        h_interp = interpolate_h(sa_best_h, n_sa, n_try)
        h_bin_try = round_to_binary(h_interp, n_try)
        h_res, c5_res, _ = binary_simulated_annealing(h_bin_try, n_try, t_phase3 / len(other_ns), seed + 900 + n_try)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            best_n = n_try
            sa_best_h = h_res.copy()
            sa_best_c5 = c5_res
            n_sa = n_try
    
    # Phase 4: Continuous refinement from best binary (30% of time)
    t_phase4 = 0.30 * budget_s
    n_cont = best_n
    
    # Try at a few resolutions
    cont_resolutions = [best_n]
    if best_n <= 200:
        cont_resolutions.append(224)
    if best_n <= 256:
        cont_resolutions.append(256)
    cont_resolutions.append(best_n + 64)
    cont_resolutions = sorted(set([nn for nn in cont_resolutions if 64 <= nn <= 400]))
    
    time_per_cont = t_phase4 / len(cont_resolutions)
    
    for n_c in cont_resolutions:
        if tl() < 15:
            break
        if n_c == best_n:
            h_start = project_feasible(sa_best_h, n_c)
        else:
            h_start = interpolate_h(sa_best_h, best_n, n_c)
            h_start = project_feasible(h_start, n_c)
        
        h_cont = continuous_refine(h_start, n_c, time_per_cont, seed + 2000 + n_c)
        c5_cont = compute_c5(h_cont)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()
            best_n = n_c
    
    # Phase 5: Final polish with very high beta (15% of time)
    t_phase5 = 0.15 * budget_s
    if tl() > 10:
        n_pol = best_n
        h_pol = project_feasible(best_h, n_pol)
        bounds = [(0.0, 1.0)] * n_pol
        
        for beta in [50000, 100000, 200000, 500000]:
            if tl() < 5:
                break
            def obj5(x):
                return _smooth_obj_func(x, n_pol, beta)
            def eqc5(x):
                return _eq_con_func(x, n_pol)
            cons = [{'type': 'eq', 'fun': eqc5}]
            try:
                res = minimize(obj5, h_pol, method='SLSQP', bounds=bounds,
                              constraints=cons,
                              options={'maxiter': 1000, 'ftol': 1e-20, 'iprint': 0})
                h_pol = project_feasible(res.x, n_pol)
                c5_p = compute_c5(h_pol)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pol.copy()
            except Exception:
                pass
    
    # Phase 6: Try binary rounding of continuous solution
    if tl() > 3:
        h_bin_final = round_to_binary(best_h, best_n)
        c5_bin = compute_c5(h_bin_final)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin_final.copy()
    
    # Final: quick local search on the best solution
    if tl() > 5 and (best_h == best_h).all() and np.isin(best_h, [0.0, 1.0]).all():
        h_ls2, c5_ls2 = directed_local_search(best_h, best_n, min(tl() - 3, 5.0), seed + 5000)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()
    
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Verify with evaluator
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 is not None and ec5 < best_c5:
            best_c5 = ec5
    except Exception:
        pass
    
    return (best_h, best_c5, best_n)
