            idx_a = np.where(ones == a)[0]
            idx_b = np.where(zeros == b)[0]
            if len(idx_a) > 0:
                ones = np.delete(ones, idx_a[0])
            if len(idx_b) > 0:
                zeros = np.delete(zeros, idx_b[0])
            ones = np.append(ones, b)
            zeros = np.append(zeros, a)
        else:
            h[a] = 1.0
            h[b] = 0.0
            stall += 1
    
    return h, cur, best_h, best
