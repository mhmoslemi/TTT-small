def make_fourier_random(n,rng,kmax=5,amp=0.35):
    t=(np.arange(n)+0.5)/(1.0*n)
    comp=np.zeros(n)
    scale=0.0

    w0=w[n-1]

def lp_linearize(h0,n,rng,random_force=1e-4):
   h0=project_box_sum(h0,n)
   m=len(h0); mrows=2*m-1
   A=np.zeros((mrows,m+1)); b=np.empty(mrows)
   corr=compute_corr(h0)
   for r in range(mrows):
      L=r-(m-1)
      g=np.zeros(m); f0=corr[r]
      if L==0: g=1.0-2.0*h0
      elif L>0:
         g[L:]=1.0-h0[:m-L]
         g[:m-L]-=h0[L:]
      else:
         Lp=-L
         g[:m-Lp]=1.0-h0[Lp:]
         g[Lp:]-=h0[:m-Lp]
      A[r,:m]=g; A[r,m]=-1.0; b[r]=float(np.dot(g,h0)-f0)
   A_eq=np.zeros((1,m+1)); A_eq[0,:m]=1.0; A_eq[0,m]=0.0
   b_eq=np.array([float(m)/2.0])
   c=np.empty(m+1); c[:m]=rng.uniform(-random_force,random_force,m); c[m]=1.0
   bounds=[(0.0,1.0)]*m + [(0.0,1.0)]
   try:
      res=linprog(c,A_ub=A,b_ub=b,A_eq=A_eq,b_ub? wait b_ub parameter, b_eq, bounds=bounds, method='highs', options={'disp':False,'presolve':True})
   except Exception:
      try: res=linprog(c,A_ub=A,b_ub=b,A_eq=A_eq,b_eq=b_eq,bounds=bounds)
      except: return None
   if res.success:
      return project_box_sum(res.x[:m],m)
   return None

import numpy as np
import time
from scipy.optimize import minimize, linprog


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    return float(np.max(compute_corr(h)) * dx)


def smooth_value_weights(corr, beta):
    corr = np.asarray(corr, dtype=float)
    m = float(np.max(corr))
    if beta <= 1e-6:
        k = len(corr)
        w = np.full(k, 1.0 / float(k))
        return m, w
    sh = beta * (corr - m)
    sh = np.clip(sh, -700.0, 0.0)
    e = np.exp(sh)
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.zeros(len(corr), dtype=float)
        w[int(np.argmax(corr))] = 1.0
        return m, w
    w = e / s
    return m + float(np.log(s)) / float(beta), w


def grad_from_weights(h, corr, w):
    n = len(h)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    w0 = w[n - 1]
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    tol = 1e-15
    for L in range(1, n):
        wp = w[n - 1 + L]
        if wp > tol:
            g[L:] += wp * (1.0 - h[:n - L])
            g[:n - L] -= wp * h[L:]
        wn = w[n - 1 - L]
        if wn > tol:
            g[:n - L] += wn * (1.0 - h[L:])
            g[L:] -= wn * h[:n - L]
    return dx * g


def smooth_obj(x, beta):
    x = np.asarray(x, dtype=float)
    corr = compute_corr(x)
    val, _ = smooth_value_weights(corr, beta)
    return float(val) * (2.0 / float(len(x)))


def smooth_jac(x, beta):
    x = np.asarray(x, dtype=float)
    corr = compute_corr(x)
    _, w = smooth_value_weights(corr, beta)
    return grad_from_weights(x, corr, w)


def sum_eq(x):
    return float(np.sum(x)) - float(len(x)) / 2.0


def sum_jac(x):
    return np.ones_like(x, dtype=float)


def adjust_sum_feasible(p, n):
    p = np.asarray(p, dtype=float).copy()
    target = float(n) / 2.0
    diff = target - float(np.sum(p))
    if abs(diff) < 1e-13:
        return p
    if diff > 0.0:
        mask = p < 1.0 - 1e-10
        if np.any(mask):
            p[mask] += diff / float(np.sum(mask))
        else:
            idx = int(np.argmin(1.0 - p))
            p[idx] = min(1.0, p[idx] + diff)
    else:
        mask = p > 1e-10
        if np.any(mask):
            p[mask] += diff / float(np.sum(mask))
        else:
            idx = int(np.argmin(p))
            p[idx] = max(0.0, p[idx] + diff)
    p = np.clip(p, 0.0, 1.0)
    diff = target - float(np.sum(p))
    if abs(diff) > 1e-12:
        if diff > 0.0:
            mask = p < 1.0 - 1e-12
        else:
            mask = p > 1e-12
        if np.any(mask):
            p[mask] += diff / float(np.sum(mask))
    return np.clip(p, 0.0, 1.0)


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float)
    target = float(n) / 2.0
    fast = float(np.min(y)) >= -1e-9 and float(np.max(y)) <= 1.0 + 1e-9
    if fast:
        p = np.clip(y, 0.0, 1.0)
        if abs(float(np.sum(p)) - target) < 1e-7:
            return adjust_sum_feasible(p, n)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        p = np.clip(y - mid, 0.0, 1.0)
        s = float(np.sum(p))
        if s > target:
            hi = mid
        else:
            lo = mid
    p = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    return adjust_sum_feasible(p, n)


def interp_h(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_sum(h_src, n_dst)
    x_src = (np.arange(n_src, dtype=float) + 0.5) * (2.0 / float(n_src))
    x_dst = (np.arange(int(n_dst), dtype=float) + 0.5) * (2.0 / float(n_dst))
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_box_sum(h_dst, n_dst)


def lowpass_h(h, kmax):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if kmax <= 0:
        return project_box_sum(h, n)
    F = np.fft.rfft(h)
    L = F.size
    keep = min(kmax + 1, L)
    F2 = F.copy()
    if keep < L:
        F2[keep:] = 0.0
    h2 = np.fft.irfft(F2, n)
    return project_box_sum(h2, n)


def anti_full_from_z(z):
    z = np.asarray(z, dtype=float)
    m = len(z)
    h = np.empty(2 * m, dtype=float)
    h[:m] = z
    h[m:] = 1.0 - z[::-1]
    return h


def anti_z_from_h(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n % 2 != 0:
        h = interp_h(h, n + 1)
        n = len(h)
    m = n // 2
    return np.clip(0.5 * (h[:m] + 1.0 - h[m:][::-1]), 0.0, 1.0)


def project_anti_full(h):
    return anti_full_from_z(anti_z_from_h(h))


def anti_obj(z, m, beta):
    z = np.asarray(z, dtype=float)
    z = np.clip(z, 0.0, 1.0)
    h = anti_full_from_z(z)
    corr = compute_corr(h)
    val, _ = smooth_value_weights(corr, beta)
    return float(val) / float(m)


def anti_jac(z, m, beta):
    z = np.asarray(z, dtype=float)
    z = np.clip(z, 0.0, 1.0)
    h = anti_full_from_z(z)
    corr = compute_corr(h)
    _, w = smooth_value_weights(corr, beta)
    gh = grad_from_weights(h, corr, w)
    return gh[:m] + gh[m:][::-1]


def deadline_ok(deadline, reserve=0.0):
    return time.time() <= float(deadline) - float(reserve)


def phase_remaining(deadline):
    return max(0.0, float(deadline) - time.time())


def anti_lbb_chain(z0, m, betas, maxiters, start_time, budget_s, reserve=30.0):
    z = np.clip(np.asarray(z0, dtype=float).copy(), 0.0, 1.0)
    if z.shape[0] != m:
        z = interp_h(z, m)
        z = np.clip(z, 0.0, 1.0)
    best_h = anti_full_from_z(z)
    best_z = z.copy()
    best_c = compute_c5(best_h)
    for idx, beta in enumerate(betas):
        if not deadline_ok(start_time + float(budget_s), reserve):
            break
        if isinstance(maxiters, (int, float)):
            mi = int(maxiters)
        else:
            mi = int(maxiters[min(idx, len(maxiters) - 1)])
        try:
            res = minimize(
                anti_obj,
                z,
                method="L-BFGS-B",
                jac=anti_jac,
                args=(m, float(beta)),
                bounds=[(0.0, 1.0)] * m,
                options={"maxiter": mi, "ftol": 1e-14, "disp": 0, "maxls": 50},
            )
            z2 = np.clip(np.asarray(res.x, dtype=float), 0.0, 1.0)
            h2 = anti_full_from_z(z2)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2
                best_z = z2.copy()
            if c2 <= best_c + 1e-7:
                z = z2.copy()
            else:
                z = best_z.copy()
        except Exception:
            z = best_z.copy()
    return best_h, best_z, best_c


def slsqp_chain(h0, n, betas, maxiters, start_time, budget_s, reserve=30.0):
    h = project_box_sum(h0, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for idx, beta in enumerate(betas):
        if not deadline_ok(start_time + float(budget_s), reserve):

            mi = int(maxiters)
        else:
            mi = int(maxiters[min(idx, len(maxiters) - 1)])

            cons = [{"type": "eq", "fun": sum_eq, "jac": sum_jac}]
            res = minimize(
                smooth_obj,
                h,
                method="SLSQP",
                jac=smooth_jac,
                args=(float(beta),),
                bounds=[(0.0, 1.0)] * n,
                constraints=cons,
                options={"maxiter": mi, "ftol": 1e-13, "iprint": 0},
            )
            h2 = project_box_sum(np.asarray(res.x, dtype=float), n)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            if c2 <= best_c + 1e-7:
                h = h2.copy()
            else:
                h = best_h.copy()
        except Exception:
            h = best_h.copy()
    return best_h, best_c


def make_fourier_random(n, rng, kmax=5, amp=0.35):
    n = int(n)
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    comp = np.zeros(n, dtype=float)
    norm = 0.0
    for k in range(1, int(kmax) + 1):
        a = rng.uniform(-1.0, 1.0)
        b = rng.uniform(-1.0, 1.0)
        w = rng.uniform(0.2, 1.0) / float(k)
        comp += w * (a * np.cos(2.0 * np.pi * k * t) + b * np.sin(2.0 * np.pi * k * t))
        norm += abs(w)
    if norm > 1e-12:
        comp = amp * comp / norm
    return project_box_sum(0.5 + comp, n)


def make_perturb_parent(h, scale, rng):
    h = project_box_sum(h, len(h)).copy()
    if scale <= 0.0:
        return h
    return project_box_sum(h + rng.normal(0.0, scale, len(h)), len(h))


def binary_from_frac(h, n):
    h = project_box_sum(h, n)
    order = np.argsort(h)
    b = np.zeros(n, dtype=float)
    b[order[n // 2:]] = 1.0
    return b


def adam_polish(h0, n, betas, lrs, start_time, budget_s, rng, reserve=30.0):
    h = project_box_sum(h0, n).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    m = np.zeros(n, dtype=float)
    v = np.zeros(n, dtype=float)
    t = 0
    eval_int = 20 if n <= 256 else 40
    for idx, beta in enumerate(betas):
        if not deadline_ok(start_time + float(budget_s), reserve):
            break
        lr = float(lrs[min(idx, len(lrs) - 1)])
        steps = int(min(2500, max(300, 500000.0 / max(1, n))))
        noimpr = 0
        for step in range(steps):
            if not deadline_ok(start_time + float(budget_s), reserve + 5.0):
                break
            corr = compute_corr(h)
            _, w = smooth_value_weights(corr, beta)
            g = grad_from_weights(h, corr, w)
            t += 1
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * (g * g)
            mhat = m / (1.0 - 0.9 ** t)
            vhat = v / (1.0 - 0.999 ** t)
            h = project_box_sum(h - lr * mhat / (np.sqrt(vhat) + 1e-8), n)
            if step % eval_int == 0:
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = h.copy()
                    noimpr = 0
                else:
                    noimpr += eval_int
                if noimpr > 5000:
                    h = project_box_sum(best_h + rng.normal(0.0, 0.02, n), n)
                    noimpr = 0
        h = best_h.copy()
    return best_h, best_c


def lp_linearize(h0, n, rng, random_force=1e-4):
    h0 = project_box_sum(h0, n)
    npts = n
    mrows = 2 * npts - 1
    A = np.zeros((mrows, npts + 1), dtype=float)
    b = np.empty(mrows, dtype=float)
    corr = compute_corr(h0)
    for r in range(mrows):
        L = r - (npts - 1)
        g = np.zeros(npts, dtype=float)
        f0 = float(corr[r])
        if L == 0:
            g = 1.0 - 2.0 * h0
        elif L > 0:
            g[L:] = 1.0 - h0[:npts - L]
            g[:npts - L] -= h0[L:]
        else:
            Lp = -L
            g[:npts - Lp] = 1.0 - h0[Lp:]
            g[Lp:] -= h0[:npts - Lp]
        A[r, :npts] = g
        A[r, npts] = -1.0
        b[r] = float(np.dot(g, h0) - f0)
    Aeq = np.zeros((1, npts + 1), dtype=float)
    Aeq[0, :npts] = 1.0
    Aeq[0, npts] = 0.0
    beq = np.array([float(npts) / 2.0])
    c = np.empty(npts + 1, dtype=float)
    c[:npts] = rng.uniform(-random_force, random_force, npts)
    c[npts] = 1.0
    bounds = [(0.0, 1.0)] * npts + [(0.0, 1.0)]
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            A_eq=Aeq,
            b_eq=beq,
            bounds=bounds,
            method="highs",
            options={"disp": False},
        )
    except Exception:
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                A_eq=Aeq,
                b_eq=beq,
                bounds=bounds,
            )
        except Exception:
            return None
    if getattr(res, "success", False):
        return project_box_sum(np.asarray(res.x[:npts], dtype=float), npts)
    return None


def binary_local_search(h0, n, rng, start_time, budget_s, reserve=30.0):
    h = binary_from_frac(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    ones = np.nonzero(h == 1.0)[0]
    zeros = np.nonzero(h == 0.0)[0]
    stuck = 0
    while deadline_ok(start_time + float(budget_s), reserve):
        improved = False
        for _ in range(220):
            if not deadline_ok(start_time + float(budget_s), reserve):
                break
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            if h[i] != 1.0 or h[j] != 0.0:
                continue
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
                ones = np.nonzero(h == 1.0)[0]
                zeros = np.nonzero(h == 0.0)[0]
                improved = True
                stuck = 0
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            stuck += 1
            if stuck >= 3:
                break
    return best_h, best_c

        nn = 200
        best_h = np.full(nn, 0.5, dtype=float)
        best_c5 = compute_c5(best_h)
        best_n = nn

    if best_n % 2 != 0:
        best_h = interp_h(best_h, best_n + 1)
        best_n = len(best_h)
        best_c5 = compute_c5(best_h)

    # Quick anti-palindromic symmetrization, usually effective.
    if deadline_ok(floor_time, 5.0):
        h_ant = project_anti_full(best_h)
        c_ant = compute_c5(h_ant)
        if c_ant < best_c5:
            best_c5 = c_ant
            best_h = h_ant
            best_n = len(h_ant)

    anchor_n = best_n

    # Phase A: high-quality optimization inside the anti-palindromic subspace.
    tA = start_time + 0.55 * budget_s
    if anchor_n % 2 == 0 and anchor_n >= 64:
        m0 = anchor_n // 2
        zbest = anti_z_from_h(best_h)

        anti_starts = [zbest.copy()]
        if deadline_ok(tA, 160.0):
            h_rand = make_fourier_random(2 * m0, rng, kmax=5, amp=0.45)
            anti_starts.append(np.clip(h_rand[:m0], 0.0, 1.0))
        if deadline_ok(tA, 160.0):
            anti_starts.append(rng.uniform(0.0, 1.0, m0))
        if deadline_ok(tA, 160.0):
            anti_starts.append(anti_z_from_h(binary_from_frac(best_h, anchor_n)))
        if deadline_ok(tA, 160.0):
            anti_starts.append(anti_z_from_h(lowpass_h(best_h, 4)))

        for idx_A, zs in enumerate(anti_starts):
            if not deadline_ok(tA, 90.0):
                break
            if idx_A == 0:
                h1, z1, c1 = anti_lbb_chain(
                    zs,
                    m0,
                    betas=[5.0, 20.0, 75.0, 250.0, 800.0, 2500.0, 8000.0],
                    maxiters=[160, 260, 420, 620, 820, 920, 920],
                    start_time=time.time(),
                    budget_s=phase_remaining(tA),
                    reserve=90.0,
                )
            else:
                h1, z1, c1 = anti_lbb_chain(
                    zs,
                    m0,
                    betas=[10.0, 50.0, 200.0, 700.0, 2200.0],
                    maxiters=[180, 280, 400, 520, 620],
                    start_time=time.time(),
                    budget_s=phase_remaining(tA),
                    reserve=90.0,
                )
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()
                best_n = anchor_n
            if c1 < best_c5 + 1e-9 and deadline_ok(tA, 80.0):
                h2, z2, c2 = anti_lbb_chain(
                    z1,
                    m0,
                    betas=[2500.0, 8000.0, 20000.0],
                    maxiters=[420, 520, 620],
                    start_time=time.time(),
                    budget_s=phase_remaining(tA),
                    reserve=80.0,
                )
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()
                    best_n = anchor_n

        # Periodic perturbation polish in the symmetric subspace.
        for _ in range(6):
            if not deadline_ok(tA, 70.0):
                break
            znoise = np.clip(anti_z_from_h(best_h) + rng.normal(0.0, 0.012, anchor_n // 2), 0.0, 1.0)
            h1, z1, c1 = anti_lbb_chain(
                znoise,
                anchor_n // 2,
                betas=[500.0, 2000.0, 8000.0],
                maxiters=[320, 420, 520],
                start_time=time.time(),
                budget_s=phase_remaining(tA),
                reserve=70.0,
            )
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()
                best_n = anchor_n

    # Phase B: full-dimensional polish from several starts.
    tB = start_time + 0.75 * budget_s
    if best_n % 2 == 0 and best_n >= 64:
        full_inits = [best_h.copy()]
        if deadline_ok(tB, 120.0):
            full_inits.append(project_anti_full(best_h))
        if deadline_ok(tB, 120.0):
            full_inits.append(lowpass_h(best_h, 5))
        if deadline_ok(tB, 120.0):
            full_inits.append(make_perturb_parent(best_h, 0.03, rng))

        for h_init in full_inits:
            if not deadline_ok(tB, 80.0):
                break
            h1, c1 = slsqp_chain(
                h_init,
                best_n,
                betas=[15.0, 60.0, 220.0, 800.0, 2500.0],
                maxiters=[90, 160, 260, 360, 430],
                start_time=time.time(),
                budget_s=phase_remaining(tB),
                reserve=80.0,
            )
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()
                best_n = len(h1)
            if c1 < best_c5 + 1e-9 and deadline_ok(tB, 70.0):
                h_ant = project_anti_full(best_h)
                c_ant = compute_c5(h_ant)
                if c_ant < best_c5:
                    best_c5 = c_ant
                    best_h = h_ant.copy()
                    best_n = len(h_ant)
                if best_n % 2 == 0:
                    h2, z2, c2 = anti_lbb_chain(
                        anti_z_from_h(best_h),
                        best_n // 2,
                        betas=[700.0, 2500.0, 8000.0],
                        maxiters=[320, 420, 520],
                        start_time=time.time(),
                        budget_s=phase_remaining(tB),
                        reserve=70.0,
                    )
                    if c2 < best_c5:
                        best_c5 = c2
                        best_h = h2.copy()
                        best_n = len(h2)

        if deadline_ok(tB, 90.0) and best_n <= 320:
            h1, c1 = adam_polish(
                best_h,
                best_n,
                betas=[20.0, 90.0, 350.0, 1200.0, 4000.0],
                lrs=[0.05, 0.02, 0.008, 0.003, 0.0012],
                start_time=time.time(),
                budget_s=phase_remaining(tB),
                rng=rng,
                reserve=90.0,
            )
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()
                best_n = len(h1)

        if deadline_ok(tB, 60.0) and best_n <= 320:
            h_cur = best_h.copy()
            for _ in range(3):
                if not deadline_ok(tB, 50.0):
                    break
                hlp = lp_linearize(h_cur, best_n, rng, random_force=8e-5)
                if hlp is None:
                    break
                clp = compute_c5(hlp)
                improved = clp < best_c5
                if improved:
                    best_c5 = clp
                    best_h = hlp.copy()
                    best_n = len(hlp)
                if not improved:
                    break
                h_cur = best_h.copy()
                if deadline_ok(tB, 70.0):
                    hsl, csl = slsqp_chain(
                        h_cur,
                        best_n,
                        betas=[80.0, 350.0, 1200.0],
                        maxiters=[120, 220, 300],
                        start_time=time.time(),
                        budget_s=phase_remaining(tB),
                        reserve=70.0,
                    )
                    if csl < best_c5:
                        best_c5 = csl
                        best_h = hsl.copy()
                        best_n = len(hsl)

    # Phase C: try neighboring resolutions.
    tC = start_time + 0.93 * budget_s
    if deadline_ok(tC, 100.0) and best_n % 2 == 0:
        cands = []
        seen = {int(best_n)}
        for nn_raw in (
            best_n,
            best_n + 32,
            best_n - 32,
            best_n + 64,
            best_n - 64,
            128,
            160,
            200,
            240,
            280,
            320,
            384,
            512,
        ):
            nn = int(nn_raw)
            if nn % 2 != 0:
                nn += 1
            if 96 <= nn <= 512 and nn not in seen:
                seen.add(nn)
                cands.append(nn)
        cands.sort(key=abs)
        cands = cands[:5]

        for nn in cands:
            if not deadline_ok(tC, 80.0):
                break
            if int(nn) == int(best_n):
                continue
            h_init = interp_h(best_h, nn)
            h_ant = project_anti_full(h_init)
            c_ant = compute_c5(h_ant)
            if c_ant < best_c5:
                best_c5 = c_ant
                best_h = h_ant.copy()
                best_n = nn
            if best_n % 2 != 0:
                continue
            h1, z1, c1 = anti_lbb_chain(
                anti_z_from_h(h_ant),
                nn // 2,
                betas=[10.0, 50.0, 200.0, 800.0, 2500.0],
                maxiters=[140, 240, 360, 480, 580],
                start_time=time.time(),
                budget_s=phase_remaining(tC),
                reserve=80.0,
            )
            if c1 < best_c5:
                best_c5 = c1
                best_h = h1.copy()
                best_n = nn

            if nn <= 280 and deadline_ok(tC, 60.0):
                h2, c2 = slsqp_chain(
                    h1,
                    nn,
                    betas=[50.0, 220.0, 800.0],
                    maxiters=[100, 180, 240],
                    start_time=time.time(),
                    budget_s=phase_remaining(tC),
                    reserve=60.0,
                )
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()
                    best_n = nn

            if c1 < best_c5 + 1e-9 and deadline_ok(tC, 60.0):
                h3, z3, c3 = anti_lbb_chain(
                    z1,
                    nn // 2,
                    betas=[2500.0, 8000.0, 20000.0],
                    maxiters=[340, 440, 540],
                    start_time=time.time(),
                    budget_s=phase_remaining(tC),
                    reserve=60.0,
                )
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = h3.copy()
                    best_n = nn

    # Binary local search only on modest sizes; sometimes gives a useful jump.
    if deadline_ok(tC, 50.0) and best_n <= 240:
        hb, cb = binary_local_search(
            best_h,
            best_n,
            rng,
            start_time=time.time(),
            budget_s=phase_remaining(tC),
            reserve=50.0,
        )
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
            best_n = len(hb)

    # Final symmetrization and feasibility repair.
    if best_n % 2 == 0:
        h_ant = project_anti_full(best_h)
        c_ant = compute_c5(h_ant)
        if c_ant < best_c5:
            best_c5 = c_ant
            best_h = h_ant.copy()
            best_n = len(h_ant)

    best_h = np.asarray(best_h, dtype=float).copy()
    final_h = project_box_sum(best_h, best_n)
    final_c5 = compute_c5(final_h)

    try:
        eval_c5 = float(evaluate_erdos_solution(final_h))
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    if abs(final_c5 - best_c5) <= 1e-8:
        final_c5 = min(final_c5, best_c5)

    return final_h, float(final_c5), int(final_h.size)
