                if i == j:
                    continue
                cap = min(1.0 - best[i], best[j])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * 0.6 * cap
                cand = best.copy()
                cand[i] += d
                cand[j] -= d
                if cand[i] > 1 + 1e-12 or cand[j] < -1e-12:
                    continue
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break
    
    best = project_density(best, target)
    return best, evaluate_erdos(best)

def de_objective(x, n):
    """Objective for differential evolution"""
    h = project_density(x, 0.5 * n)
    return evaluate_erdos(h)

def make_initial_pop(n, rng, n_pop=15):
    """Generate diverse initial population"""
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    pop = []
    for i in range(n_pop):
        mode = i % 5
        if mode == 0:
            h = 0.5 + 0.3 * np.sin(2*np.pi*3*x/2 + rng.uniform(0, 6.28)) + 0.1*np.sin(2*np.pi*7*x/2)
        elif mode == 1:
            h = np.where(rng.random(n) > 0.5, 0.8, 0.2) + rng.normal(0, 0.05, n)
        elif mode == 2:
            freq = int(rng.integers(2, 10))
            h = 0.5 + 0.4 * np.sin(2*np.pi*freq*x/2 + rng.uniform(0, 6.28))
        elif mode == 3:
            # Block structure
            block = max(1, n // int(rng.integers(4, 12)))
            h = np.tile([0.2, 0.8][:int(rng.integers(1,3))] + [0.5], n//block + 1)[:n]
        else:
            h = 0.5 + rng.normal(0, 0.2, n)
        h = project_density(h, 0.5 * n)
        pop.append(h)
    return pop

def polish_with_nelder_mead(h, deadline, max_iter=500):
    """Use Nelder-Mead on a subset of free variables"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    
    # Find "free" variables (not at 0 or 1)
    free_mask = (best > 0.01) & (best < 0.99)
    free_idx = np.where(free_mask)[0]
    
    if len(free_idx) < 2:
        return best, best_c
    
    # Limit to at most 20 free variables for NM
    if len(free_idx) > 20:
        # Pick the 20 with values closest to 0.5
        order = np.argsort(np.abs(best[free_idx] - 0.5))
        free_idx = free_idx[order[:20]]
    
    def nm_obj(x_free):
        x_full = best.copy()
        x_full[free_idx] = x_free
        x_full = project_density(x_full, target)
        return evaluate_erdos(x_full)
    
    x0 = best[free_idx]
    try:
        res = minimize(nm_obj, x0, method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10})
        if res.x is not None:
            x_full = best.copy()
            x_full[free_idx] = res.x
            cand = project_density(x_full, target)
            c = evaluate_erdos(cand)
            if c < best_c:
                best = cand
                best_c = c
    except Exception:
        pass
    
    return best, evaluate_erdos(best)
