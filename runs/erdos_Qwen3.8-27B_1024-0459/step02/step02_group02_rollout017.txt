```python
                candidates.append((m, 1))
        
        if not candidates:
            continue
            
        # Pick a random candidate (or the best one)
        m, sc = candidates[rng.randint(len(candidates))]
        
        # Evaluate the full effect
        h[nz] = 1.0
        h[m] = 0.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(new_corr) * dx)
        
        if new_c5 < best_score:
            best_score = new_c5
            best_swap = (nz, m, new_c5)
        
        h[nz] = 0.0
        h[m] = 1.0
        
        if best_score < best_c5 - 1e-14:
            break
    
    if best_swap is not None and best_score < best_c5:
        nz, m, new_c5 = best_swap
        h[nz] = 1.0
        h[m] = 0.0
        return h, new_c5
    
    return h, best_c5
```

Hmm, this is O(n²) per call in the worst case. For n=256, that's 65K operations per call, times the number of bad_zeros (could be up to 128). So ~8M ops per call. In Python, that's maybe 1-2 seconds per call. Too slow.

Let me optimize: instead of checking all m for each nz, I can precompute which m's are "good" for a given shift s.

For shift s, a position m is "good" (h[m]=1, h[m+s]=0, h[m-s]=0) if:
- h[m] = 1
- h[m+s] = 0 (if valid)
- h[m-s] = 0 (if valid)

I can compute this set of "good m" once per shift, then for each bad_zero, just pick randomly from the good set.

Actually, let me simplify the whole approach. Instead of this complex targeting, let me use a simpler but still effective method:

**Improved binary local search with momentum:**
1. Maintain a list of the top-5 worst shifts
2. For each trial, randomly pick one of the top-5 shifts
3. Find a random position (a,b) swap that reduces that particular shift
4. Apply it if it also doesn't increase the overall max too much

Let me also try the **gradient-free approach on intervals**:

Represent h as k intervals. The parameters are the 2k endpoints (start, end of each interval). The constraint is sum of lengths = n/2. Use Nelder-Mead on these parameters.

For k=8 intervals: 16 parameters - 1 constraint = 15 DOF.
For k=12 intervals: 24 parameters - 1 constraint = 23 DOF.

Let me try this with k=10 (19 DOF). This is low enough for Nelder-Mead to work well.

```python
def intervals_to_binary(params, n):
    """params: [s1, e1, s2, e2, ..., sk, ek] sorted, all in [0, n]
    Returns binary h of length n."""
    h = np.zeros(n)
    for i in range(0, len(params), 2):
        s = int(round(params[i]))
        e = int(round(params[i+1]))
        s = max(0, min(n, s))
        e = max(0, min(n, e))
        if e > s:
            h[s:e] = 1.0
    return h

def interval_obj(params, n):
    h = intervals_to_binary(params, n)
    # Check constraint
    if abs(h.sum() - n/2) > 0.5:
        return 1e6
    return compute_c5(h)
```

But this is discrete (rounding to integers). For continuous optimization, I'd need to work with the actual values.

Actually, for the discrete problem, the "interval" representation with integer endpoints is exact. There are C(n, n/2) possible binary strings, but only O(n^k) strings can be represented as k intervals. For k=10 and n=256, that's about 256^10 ≈ 10^23, which is still huge but much less than C(256, 128) ≈ 10^76.

The issue is that Nelder-Mead on 19 discrete parameters is not straightforward. Let me think of another way.

Actually, I think the most promising approach for beating 0.380956 is:

1. **Longer binary local search** with a better move selection strategy
2. **Try different n values** (maybe n=192 or n=320 gives a better answer)
3. **Simulated annealing** to escape local optima
4. **2-swap moves** (swap two pairs)

Let me also consider: maybe the answer is better at a smaller n. For n=128, the "granularity" is coarser, so the optimal binary pattern might achieve a lower C5/dx ratio.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:

    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)

def greedy_binary_search(h_init, n, rng, time_budget, multi_shift=3):
    """Greedy binary local search targeting top worst shifts."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    dx = 2.0 / n
    start_time = time.time()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    no_improve = 0
    patience = 20000
    
    while time.time() - start_time < time_budget:
        if no_improve > patience:
            break
        
        # Find top-worst shifts
        shift_vals = corr.copy()
        worst_shifts = np.argsort(shift_vals)[-multi_shift:]

        s = target_shift_idx - (n - 1)

        if s >= 0:
            rng_start, rng_end = 0, n - s
        else:
            rng_start, rng_end = -s, n

        found = False
        ones = np.where(h == 1.0)[0]

            b = ones[rng.randint(len(ones))]

            h[b] = 0.0

            new_c5 = float(np.max(new_corr) * dx)

                best_h = h.copy()

                found = True
