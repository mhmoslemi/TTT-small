def continuous_shift_values(h, n):

        chunk = min(100.0, max(20.0, cont_deadline - time.time() - 10.0))
        hc, cc = continuous_top_descent(h0, n_cont, time.time() + chunk, rng, K=6, max_iter=6000)
        hc = make_feasible(hc, n_cont)
        cc = compute_c5(hc, n_cont)
        if cc < best_c5:
            best_h = hc
            best_n = n_cont
            best_c5 = cc

    if time.time() < cont_deadline - 35.0:
        n2 = 128
        if parent is not None:
            h2 = parent if len(parent) == n2 else interp_h(parent, n2)
        else:
            h2 = int_to_array(thue_morse_int(n2, 0), n2)
        h2 = make_feasible(h2, n2)
        chunk2 = min(70.0, max(10.0, cont_deadline - time.time() - 5.0))
        hc2, cc2 = continuous_top_descent(h2, n2, time.time() + chunk2, rng, K=6, max_iter=4000)
        hc2 = make_feasible(hc2, n2)
        cc2 = compute_c5(hc2, n2)
        if cc2 < best_c5:
            best_h = hc2
            best_n = n2
            best_c5 = cc2

    if best_h is None:
        n = 256
        best_n = n
        best_h = int_to_array(thue_morse_int(n, 0), n)
        best_c5 = compute_c5(best_h, n)

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, float(best_c5), int(best_n))
