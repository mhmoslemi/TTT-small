def lag_value(h, l):
    """Compute the overlap for a specific lag."""
    n = len(h)
    L = abs(int(l))
    if L >= n:
        return 0.0
    if l >= 0:
        m = n - L
        return float(np.sum(h[:m]) - np.dot(h[:m], h[L:]))
    else:
        m = n - L
        return float(np.sum(h[L:]) - np.dot(h[L:], h[:m]))

def lag_gradient(h, l):
    """Gradient of lag_value(h, l) with respect to h."""
    n = len(h)
    L = abs(int(l))
    if L >= n:
        return np.zeros(n)
    if l >= 0:
        m = n - L
        g = np.zeros(n)
        g[:m] = 1.0 - h[L:]
        g[L:] -= h[:m]
        return g
    else:
        m = n - L
        g = np.zeros(n)
        g[L:] = 1.0 - h[:m]
        g[:m] -= h[L:]
        return g

def subgradient_step(h, n, step_size, time_limit):

    corr = np.correlate(h, 1.0 - h, mode="full") * dx

    max_val = float(np.max(corr))
    mask = corr >= max_val - 1e-10
    active_idx = np.where(mask)[0]
    
    # Average gradient over active lags

        g += lag_gradient(h, lag)
    g /= len(active_idx)
    g *= dx
    
    # Project gradient to satisfy sum constraint (remove mean)
    g -= np.mean(g)
    
    # Take step

    h_new = project_feasible(h_new, n)
    return h_new, max_val

import numpy as np
import time
from scipy.optimize import minimize


def get_parent_values():
    try:
        vals = initial_h_values
        if vals is None:
            return None
        arr = np.asarray(vals, dtype=float).copy()
        if arr.size == 0:

    except Exception:
        return None


def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:
        n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    target = 0.5 * float(n)
    h = h[:n].copy()
    if h.size < n:
        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    elif h.size > n:
        h = h[:n]
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(20):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-13
        else:
            mask = out > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] = np.clip(out[mask] + diff / float(cnt), 0.0, 1.0)
    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_c5_detailed(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    lags = np.arange(-n + 1, n)
    return float(np.max(corr)), corr, lags


def lag_value(h, l):
    n = len(h)
    L = abs(int(l))
    if L >= n:
        return 0.0
    if l >= 0:
        m = n - L
        return float(np.sum(h[:m]) - np.dot(h[:m], h[L:]))
    else:
        m = n - L
        return float(np.sum(h[L:]) - np.dot(h[L:], h[:m]))


def lag_gradient(h, l):
    n = len(h)
    L = abs(int(l))
    if L >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if l >= 0:
        m = n - L
        g[:m] = 1.0 - h[L:]
        g[L:] -= h[:m]
    else:
        m = n - L
        g[L:] = 1.0 - h[:m]
        g[:m] -= h[L:]
    return g


def subgradient_with_momentum(h0, n, time_limit, max_iter=10000, init_step=0.03, momentum=0.7):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    step = init_step
    velocity = np.zeros(n)
    dx = 2.0 / float(n)
    no_improve = 0
    stall_limit = 800

    for it in range(max_iter):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        max_val = float(np.max(corr))
        mask = corr >= max_val - 1e-10
        active_idx = np.where(mask)[0]
        g = np.zeros(n)
        for idx in active_idx:
            lag = int(idx - (n - 1))
            g += lag_gradient(h, lag)
        g /= float(len(active_idx))
        g *= dx
        g -= np.mean(g)
        velocity = momentum * velocity + (1.0 - momentum) * g
        g_norm = float(np.sqrt(np.sum(velocity * velocity)))
        if g_norm < 1e-12:
            break
        h_trial = h - step * velocity / g_norm
        h_trial = project_feasible(h_trial, n)
        c_trial = compute_c5(h_trial)
        if c_trial < best_c - 1e-12:
            best_h = h_trial.copy()
            best_c = c_trial
            no_improve = 0
            step = min(step * 1.03, 0.12)
        else:
            no_improve += 1
            if no_improve > stall_limit:
                h_trial = project_feasible(best_h + (h_trial - best_h) * 0.5, n)
                c_trial = compute_c5(h_trial)
                if c_trial >= best_c:
                    h_trial = best_h.copy()
                    c_trial = best_c
                no_improve = 0
                step = max(step * 0.6, 0.002)
                velocity *= 0.5
        h = h_trial
        c = c_trial
    return best_h, best_c


def active_slsqp(h0, n, time_limit, maxiter=50):
    h = project_feasible(h0, n)
    dx = 2.0 / float(n)
    _, corr, lags_arr = compute_c5_detailed(h)
    max_val = float(np.max(corr))
    order = np.argsort(-corr)
    top_k = min(12, len(order))
    active_lags = []
    for i in range(top_k):
        if corr[order[i]] > max_val - 5e-5:
            active_lags.append(int(lags_arr[order[i]]))
    if not active_lags:
        active_lags = [int(lags_arr[order[0]])]

    def obj(x):
        return float(np.max([lag_value(x, ll) for ll in active_lags])) * dx

    def obj_jac(x):
        vals = np.array([lag_value(x, ll) for ll in active_lags]) * dx
        max_v = float(np.max(vals))
        mask = vals >= max_v - 1e-9
        g = np.zeros(n)
        for i in np.where(mask)[0]:
            g += lag_gradient(x, active_lags[i])
        g /= float(mask.sum())
        return g * dx

    def sum_eq(x):
        return float(np.sum(x) - 0.5 * n)

    def sum_eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac}]
    try:
        res = minimize(obj, h, jac=obj_jac, method="SLSQP", bounds=bounds,
                       constraints=constraints, options={"maxiter": maxiter, "ftol": 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    return h, compute_c5(h)


def t_constrained_slsqp(h0, n, t_target, time_limit, maxiter=60):
    h = project_feasible(h0, n)
    dx = 2.0 / float(n)
    _, corr, lags_arr = compute_c5_detailed(h)
    threshold = t_target * dx - 0.01
    active = [int(lags_arr[i]) for i in range(len(lags_arr)) if corr[i] > threshold]
    if not active:
        active = [int(lags_arr[int(np.argmax(corr))])]
    if len(active) > 16:
        order = np.argsort(-np.array([corr[ll + n - 1] for ll in active]))
        active = [active[i] for i in order[:16]]

    def obj(x):
        return float(x[-1])

    def obj_jac(x):
        j = np.zeros(len(x))
        j[-1] = 1.0
        return j

    def sum_eq(x):
        return float(np.sum(x[:-1]) - 0.5 * n)

    def sum_eq_jac(x):
        j = np.zeros(len(x))
        j[:-1] = 1.0
        return j

    def ineq(x):
        hx = x[:-1]
        t = float(x[-1])
        out = np.empty(len(active))
        for i, ll in enumerate(active):
            out[i] = t - lag_value(hx, ll) * dx
        return out

    def ineq_jac(x):
        hx = x[:-1]
        J = np.zeros((len(active), n + 1))
        for i, ll in enumerate(active):
            J[i, :n] = -dx * lag_gradient(hx, ll)
            J[i, n] = 1.0
        return J

    x0 = np.empty(n + 1)
    x0[:n] = h
    x0[n] = t_target

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac},
                   {"type": "ineq", "fun": ineq, "jac": ineq_jac}]
    try:
        res = minimize(obj, x0, jac=obj_jac, method="SLSQP", bounds=bounds,
                       constraints=constraints, options={"maxiter": maxiter, "ftol": 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x[:n], n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    return h, compute_c5(h)


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, float(h_src[0]) if n_src == 1 else 0.5)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def local_search(h0, n, time_limit, seed, block_sizes=(3, 5, 7), step_init=0.025):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    temp = 0.002
    bsize_list = list(block_sizes)
    for _ in range(20000):
        if time.time() > time_limit:
            break
        L = int(rng.choice(bsize_list))
        L = min(L, n)
        st = int(rng.randint(0, max(1, n - L + 1)))
        idx = np.arange(st, min(n, st + L))
        delta = rng.randn(len(idx)) * step_init
        delta -= delta.mean()
        h_try = h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            temp = min(temp * 1.2, 0.015)
        if c_try < c or (c_try < c + temp and rng.rand() < np.exp(-(c_try - c) / max(temp, 1e-12))):
            h = h_try
            c = c_try
        temp = max(temp * 0.997, 1e-7)
    return best_h, best_c


def structured_init(n, rng):

    if n % 4 == 0:

    inits.append(project_feasible(h, n))
    tm = np.zeros(n)
    for i in range(n):
        v = bin(i).count("1")

    x = np.linspace(0, 2, n, endpoint=False)

def smooth_max_objective(h, n, beta, n_top=8):
    _, corr, lags_arr = compute_c5_detailed(h)
    order = np.argsort(-corr)
    top = order[:n_top]
    top_vals = corr[top]
    mx = float(np.max(top_vals))
    s = 0.0
    for v in top_vals:
        d = beta * (v - mx)
        if d > -700:
            s += np.exp(d)
    return mx + np.log(s) / beta


def smooth_max_grad(h, n, beta, n_top=8):
    _, corr, lags_arr = compute_c5_detailed(h)
    order = np.argsort(-corr)
    top = order[:n_top]
    top_vals = corr[top]
    mx = float(np.max(top_vals))
    weights = np.zeros(len(top))
    for i, v in enumerate(top_vals):
        d = beta * (v - mx)
        if d > -700:
            weights[i] = np.exp(d)
    total_w = float(np.sum(weights))
    if total_w < 1e-30:
        return np.zeros(n), mx
    g = np.zeros(n)
    for i in range(len(top)):
        if weights[i] > 1e-30:
            ll = int(lags_arr[top[i]])
            w = weights[i] / total_w
            g += w * lag_gradient(h, ll)
    return g * (2.0 / float(n)), float(smooth_max_objective(h, n, beta, n_top))


def smooth_slsqp(h0, n, beta, maxiter=40):
    h = project_feasible(h0, n)

    def obj(x):
        return smooth_max_objective(x, n, beta)

    def obj_jac(x):
        g, _ = smooth_max_grad(x, n, beta)
        return g

    def sum_eq(x):
        return float(np.sum(x) - 0.5 * n)

    def sum_eq_jac(x):
        return np.ones(n)

    return (best_h, float(best_c), int(best_n))
