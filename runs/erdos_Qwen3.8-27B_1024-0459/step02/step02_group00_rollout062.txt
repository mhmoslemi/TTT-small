def corr_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / np.sum(exps)  # weights for each lag
    w = w / beta  # divide by beta for the soft max derivative
    
    # Gradient of corr[k] w.r.t. h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            # corr[k] = sum_{i=0}^{n-1-shift} h[i+shift]*(1-h[i])
            # d/dh[j]: 
            #   from h[i+shift]: j = i+shift, i = j-shift, contrib = (1-h[j-shift])
            #   from (1-h[i]): j = i, contrib = -h[j+shift]
            npairs = n - shift
            # Contribution to grad[j] for j in [shift, shift+npairs-1]: (1-h[j-shift])
            grad[shift:shift+npairs] += w[k] * (1.0 - h[:npairs])
            # Contribution to grad[j] for j in [0, npairs-1]: -h[j+shift]
            grad[:npairs] += w[k] * (-h[shift:shift+npairs])
        else:
            s = -shift

    obj = (m + np.log(np.sum(exps)) / beta) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_corr(h)) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    # Fix to have exactly n/2 ones
    n_ones = int(np.sum(h))
    if n_ones < n // 2:
        # Flip some 0s to 1s
        zeros = np.where(h == 0)[0]
        for i in range(n // 2 - n_ones):
            h[zeros[i]] = 1.0
    elif n_ones > n // 2:
        ones = np.where(h == 1)[0]
        for i in range(n_ones - n // 2):
            h[ones[i]] = 0.0
    return h

def soft_max_obj(h, beta):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    shifted = corr - m
    val = (m + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    return val

def soft_max_grad(h, beta):
    corr = compute_corr(h)
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)  # softmax weights
    
    # Compute gradient: dx * sum_k w_k * g_k
    # where g_k = d(corr_k)/d(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        wk = w[k]
        shift = k - (n - 1)
        if shift >= 0:
            npairs = n - shift
            # corr[k] = sum_{i=0}^{npairs-1} h[i+shift]*(1-h[i])
            # d/dh[j]: from h[i+shift] (j=i+shift): (1-h[j-shift]) for j in [shift, shift+npairs-1]
            #          from (1-h[i])  (j=i):      -h[j+shift]    for j in [0, npairs-1]
            grad[shift:shift+npairs] += wk * (1.0 - h[:npairs])
            grad[:npairs] += wk * (-h[shift:shift+npairs])
        else:
            s = -shift
            npairs = n - s
            # corr[k] = sum_{i=0}^{npairs-1} h[i]*(1-h[i+s])
            # d/dh[j]: from h[i]     (j=i):   (1-h[j+s])  for j in [0, npairs-1]
            #          from (1-h[i+s])(j=i+s): -h[j-s]     for j in [s, s+npairs-1]
            grad[:npairs] += wk * (1.0 - h[s:s+npairs])
            grad[s:s+npairs] += wk * (-h[:npairs])
    
    grad *= dx
    obj = (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx
    return obj, grad

def adam_optimize(h0, n, beta, max_steps, lr, time_check):
    """Adam optimizer for the soft-max objective"""
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    h = h0.copy()
    for step in range(max_steps):
        if time_check():
            break
        obj, grad = soft_max_grad(h, beta)
        t += 1
        m_acc = b1 * m_acc + (1 - b1) * grad
        v_acc = b2 * v_acc + (1 - b2) * grad**2
        m_hat = m_acc / (1 - b1**t)
        v_hat = v_acc / (1 - b2**t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
    
    return h

def bit_flip_search(h, n, max_sweeps, time_check):
    """Try flipping individual elements to 0 or 1 to reduce C5"""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time_check():
            break
        improved = False
        order = np.arange(n)
        np.random.shuffle(order)
        for i in order:
            if time_check():
                break
            for new_val in [0.0, 1.0]:
                if abs(h[i] - new_val) < 1e-8:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()
        if not improved:
            break
    
    return best_h, best_c5

        best_h = h
        best_c5 = compute_c5(best_h)
        print("No parent, using block. C5={:.8f}".format(best_c5))
    
    # Try multiple resolutions
    n_candidates = sorted(set([best_n, 128, 160, 192, 224, 256, 320, 384]))
    n_candidates = [nn for nn in n_candidates if 64 <= nn <= 400]

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)

def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        for i in range(target - n_ones):
            h[zeros[i]] = 1.0
    elif n_ones > target:
        ones = np.where(h == 1)[0]
        for i in range(n_ones - target):
            h[ones[i]] = 0.0
    return h

def make_golden(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones < target:
        zeros = np.where(h == 0)[0]
        for i in range(target - n_ones):
            h[zeros[i]] = 1.0
    elif n_ones > target:
        ones = np.where(h == 1)[0]
        for i in range(n_ones - target):
            h[ones[i]] = 0.0
    return h

def soft_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_shifted = np.max(shifted)
    exps = np.exp(shifted - max_shifted)
    s = np.sum(exps)
    w = exps / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        wk = w[k]
        shift = k - (n - 1)
        if shift >= 0:
            npairs = n - shift
            grad[shift:shift + npairs] += wk * (1.0 - h[:npairs])
            grad[:npairs] += wk * (-h[shift:shift + npairs])
        else:
            sp = -shift
            npairs = n - sp
            grad[:npairs] += wk * (1.0 - h[sp:sp + npairs])
            grad[sp:sp + npairs] += wk * (-h[:npairs])
    grad *= dx
    obj = (m + (max_shifted + np.log(s)) / beta) * dx
    return obj, grad

def adam_step(h, beta, m_acc, v_acc, t, lr):
    obj, grad = soft_max_grad(h, beta)
    b1, b2, eps = 0.85, 0.999, 1e-8
    m_acc = b1 * m_acc + (1.0 - b1) * grad
    v_acc = b2 * v_acc + (1.0 - b2) * grad * grad
    m_hat = m_acc / (1.0 - b1**t)
    v_hat = v_acc / (1.0 - b2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    return obj, h_new, m_acc, v_acc, t

def run_adam(h0, n, beta, max_steps, lr, time_fn, early_stop_patience=500):
    n = int(n)
    h = h0.copy()
    m_acc = np.zeros(n)
    v_acc = np.zeros(n)
    t = 0
    best_obj = soft_max_grad(h, beta)[0]
    stall = 0
    for step in range(max_steps):
        if time_fn() < 3.0:
            break
        obj, h, m_acc, v_acc, t = adam_step(h, beta, m_acc, v_acc, t, lr)
        h = project_feasible(h, n)
        if obj < best_obj:
            best_obj = obj
            stall = 0
        else:
            stall += 1
        if stall > early_stop_patience:
            break
    return h

def bit_flip_search(h, n, max_sweeps, time_fn):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time_fn() < 2.0:
            break
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time_fn() < 1.0:
                break
            for new_val in (0.0, 1.0):
                if abs(h[i] - new_val) < 1e-10:
                    continue
                h[i] = new_val
                h = project_feasible(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                    improved = True
                else:
                    h = best_h.copy()
        if not improved:
            break
    return best_h, best_c5

def simulated_annealing(h, n, time_fn, T_start=0.05, T_end=0.001, max_iters=50000):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_h = h.copy()
    cur_c5 = best_c5
    total_steps = max_iters
    for it in range(total_steps):
        if time_fn() < 2.0:
            break
        frac = it / float(total_steps)
        T = T_start * (T_end / T_start) ** frac
        i = np.random.randint(n)
        old_val = cur_h[i]
        if old_val < 0.5:
            cur_h[i] = 1.0
        else:
            cur_h[i] = 0.0
        cur_h = project_feasible(cur_h, n)
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h = best_h.copy()
            cur_c5 = best_c5
    return best_h, best_c5

def slow_sum_constraint(x, n):
    return np.sum(x) - n / 2.0
