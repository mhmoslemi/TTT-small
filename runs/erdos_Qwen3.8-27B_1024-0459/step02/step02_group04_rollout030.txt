def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")

def all_lag_values_fast(h):
    """Compute all lag values using FFT for autocorrelation."""
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    
    # Autocorrelation via FFT
    N = 1
    while N < 2 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N)[:n]
    
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = (total - ac[0]) * dx  # lag 0
    for L in range(1, n):
        vals[2*L - 1] = (pref[n - L] - ac[L]) * dx  # lag +L
        vals[2*L] = (total - pref[L] - ac[L]) * dx  # lag -L
    return vals

def lag_gradient(h, k):
    """Gradient of lag k value (before dx multiplication) wrt h."""
    n = h.size
    if k == 0:
        return 1.0 - 2.0 * h
    L = abs(int(k))
    m = n - L
    g = np.zeros(n, dtype=float)
    if k > 0:
        g[:m] += 1.0 - h[L:]
        g[L:] += -h[:m]
    else:
        g[L:] += 1.0 - h[:m]
        g[:m] += -h[L:]
    return g

def idx_to_lag(idx, n):
    """Convert array index to lag value. vals[0]=0, vals[1]=+1, vals[2]=-1, vals[3]=+2, vals[4]=-2, ..."""
    if idx == 0:
        return 0
    if idx % 2 == 1:
        return (idx + 1) // 2
    else:
        return -(idx // 2)

def solve_qp(h, n, active_lags, active_vals, active_grads, dx, rho):
    """Solve the SQP subproblem using cvxpy."""
    K = len(active_lags)
    delta = cp.Variable(n)
    t = cp.Variable()
    
    obj = cp.Minimize(t + rho * cp.sum_squares(delta))
    
    cons = [
        cp.sum(delta) == 0.0,  # maintain sum constraint (h_new sum = h_cur sum = n/2)
        h + delta >= 0,
        h + delta <= 1,
    ]
    
    for i in range(K):
        v_k = active_vals[i]
        g_k = active_grads[i]
        # Linearized constraint: v_k + dx * g_k · delta <= t
        cons.append(v_k + dx * g_k @ delta <= t)
    
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None
    
    if delta.value is None:
        return None
    return np.asarray(delta.value, dtype=float).flatten()

        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(20):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-12:
            break
        if diff > 0:0
            mask = out < 1.0 - 1e-12
        else:
            mask = out > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def all_lag_values_fast(h):
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    N = 1
    while N < 2 * n:
        N *= 2
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N)[:n]
    vals = np.empty(2 * n - 1, dtype=float)
    vals[0] = (total - ac[0]) * dx
    for L in range(1, n):
        vals[2 * L - 1] = (pref[n - L] - ac[L]) * dx
        vals[2 * L] = (total - pref[L] - ac[L]) * dx
    return vals

def lag_gradient(h, k):
    n = h.size
    if k == 0:
        return 1.0 - 2.0 * h
    L = abs(int(k))
    m = n - L
    g = np.zeros(n, dtype=float)
    if k > 0:
        g[:m] += 1.0 - h[L:]
        g[L:] += -h[:m]
    else:
        g[L:] += 1.0 - h[:m]
        g[:m] += -h[L:]
    return g

def idx_to_lag(idx):
    if idx == 0:
        return 0
    if idx % 2 == 1:
        return (idx + 1) // 2
    else:
        return -(idx // 2)

def solve_sqp_subproblem(h, n, active_indices, vals, dx, rho):
    K = len(active_indices)
    if K == 0:
        return None
    delta_v = cp.Variable(n)
    t_v = cp.Variable()
    obj = cp.Minimize(t_v + 0.5 * rho * cp.sum_squares(delta_v))
    cons = [cp.sum(delta_v) == 0.0, h + delta_v >= 0, h + delta_v <= 1]
    for idx in active_indices:
        k = idx_to_lag(int(idx))
        g_k = lag_gradient(h, k)
        v_k = float(vals[int(idx)])
        cons.append(v_k + dx * (g_k @ delta_v) <= t_v)
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iter=5000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
        except Exception:
            return None
    if delta_v.value is None:
        return None
    return np.asarray(delta_v.value, dtype=float).flatten()

def sqp_optimize(h0, n, time_limit, max_iter=300, K_init=60, K_max=100):
    h = project_feasible(h0, n)
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_c = compute_c5(h)
    rho = 1.0
    K = K_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() > time_limit:
            break
        vals = all_lag_values_fast(h)
        c_cur = float(np.max(vals))
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if c_cur < 0.375:
            break
        if no_improve > 80:
            break
        v_max = float(vals[0])
        order = np.argsort(-vals)
        K_eff = min(K, len(vals))
        top_idx = order[:K_eff]
        margin = 0.03 * v_max
        near_mask = (vals >= v_max - margin) & (vals > 0.01)
        near_idx = np.where(near_mask)[0]
        all_idx = np.unique(np.concatenate([top_idx, near_idx]))
        cap = K_max
        if all_idx.size > cap:
            all_idx = all_idx[np.argsort(-vals[all_idx])][:cap]
        delta = solve_sqp_subproblem(h, n, all_idx, vals, dx, rho)
        if delta is None:
            rho *= 2.0
            if rho > 200.0:
                g_total = np.zeros(n, dtype=float)
                for idx in all_idx[:20]:
                    g_total += lag_gradient(h, idx_to_lag(int(idx)))
                gn = np.linalg.norm(g_total)
                if gn > 1e-10:
                    h_try = project_feasible(h - 0.005 * dx * g_total / gn, n)
                    h = h_try
                rho = 5.0
            continue
        step_norm = np.linalg.norm(delta)
        if step_norm < 1e-12:
            break
        h_new = project_feasible(h + delta, n)
        c_new = compute_c5(h_new)
        if c_new <= c_cur + 1e-10:
            h = h_new
            rho = max(0.1, rho * 0.92)
            if c_new < best_c - 1e-9:
                K = min(K + 5, K_max)
        else:
            improved = False
            for alpha in (0.5, 0.25, 0.1, 0.05):
                h_try = project_feasible(h + alpha * delta, n)
                c_try = compute_c5(h_try)
                if c_try < c_cur:
                    h = h_try
                    improved = True
                    if c_try < best_c - 1e-9:
                        best_c = c_try
                        best_h = h.copy()
                        no_improve = 0
                    break
            if not improved:
                rho *= 1.5
                if rho > 100.0:
                    g_total = np.zeros(n, dtype=float)
                    for idx in all_idx[:15]:
                        g_total += lag_gradient(h, idx_to_lag(int(idx)))
                    gn = np.linalg.norm(g_total)
                    if gn > 1e-10:
                        h = project_feasible(h - 0.003 * dx * g_total / gn, n)
                    rho = 5.0
                    K = max(20, K - 10)
        if it % 30 == 29 and K > 40:
            K = max(40, K - 5)
    return best_h, float(best_c)

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def local_refine(h0, n, time_limit, seed, max_trials=3000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.0015
    min_temp = 1e-7
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * 0.02
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, max(temp * 1.1, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)

def subgradient_refine(h0, n, time_limit, steps=200):
    h = project_feasible(h0, n)
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = 0.01
    for _ in range(steps):
        if time.time() > time_limit:
            break
        vals = all_lag_values_fast(h)
        c_cur = float(np.max(vals))
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
        v_max = vals[0]
        active = np.where(vals >= v_max - 0.02 * v_max)[0]
        g_total = np.zeros(n, dtype=float)
        for idx in active:
            g_total += lag_gradient(h, idx_to_lag(int(idx)))
        gn = np.linalg.norm(g_total)
        if gn < 1e-12:
            break
        alpha = max(0.0005, alpha * 0.98)
        h_try = project_feasible(h - alpha * dx * g_total / gn, n)
        c_try = compute_c5(h_try)
        if c_try >= c_cur:
            alpha *= 0.5
            h_try = project_feasible(h - alpha * dx * g_total / gn, n)
            c_try = compute_c5(h_try)
        h = h_try
    return best_h, float(best_c)

def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    start_time = time.time()
    hard_end = start_time + budget_s - 30.0

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        best_h = project_feasible(parent, parent_n)
        best_c = compute_c5(best_h)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    resolutions = [best_n]
    for r in sorted(set([128, 160, 192, 200, 240, 288, 320])):
        if r not in resolutions and 60 <= r <= 400:
            resolutions.append(r)
    resolutions = sorted(set(resolutions), key=lambda r: (abs(r - best_n), r))[:5]

    for n in resolutions:
        n = int(n)
        if time.time() > hard_end - 80:
            break
        if n == best_n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_to_n(best_h, n)
        h_init = project_feasible(h_init, n)

        slot_time = hard_end - 100.0
        rem = slot_time - time.time()
        if rem < 30:
            break
        per_res = rem / max(1, len(resolutions) - resolutions.index(n))
        per_res = min(280.0, per_res)
        per_res = max(40.0, per_res)
        slot_end = time.time() + per_res

        h_opt, c_opt = sqp_optimize(h_init, n, slot_end, max_iter=250, K_init=50, K_max=100)
        if c_opt < best_c - 1e-10:
            best_h = h_opt
            best_c = c_opt
            best_n = n

        if time.time() < hard_end - 60:
            h_sub, c_sub = subgradient_refine(best_h, best_n, hard_end - 55, steps=100)
            if c_sub < best_c - 1e-10:
                best_h = h_sub
                best_c = c_sub

    if time.time() < hard_end - 40 and 2 <= best_n <= 400:
        slot = min(80.0, hard_end - time.time() - 10.0)
        h2, c2 = local_refine(best_h, best_n, time.time() + slot, seed + 7777, max_trials=2000)
        if c2 < best_c - 1e-12:
            best_h = h2
            best_c = c2

    if time.time() < hard_end - 15 and 2 <= best_n <= 320:
        h3, c3 = subgradient_refine(best_h, best_n, hard_end - 5, steps=100)
        if c3 < best_c - 1e-12:
            best_h = h3
            best_c = c3

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
