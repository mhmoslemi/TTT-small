import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_corr(h):
    """Returns the full correlation for fast max lookup."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr

def make_feasible_binary(h, n):
    """Ensure h is binary with exactly n/2 ones."""
    h = np.round(h).astype(float)
    ones = np.sum(h)
    target = n // 2
    if ones > target:
        indices = np.where(h == 1.0)[0]
        rng = np.random.default_rng(12345)
        to_flip = rng.choice(indices, ones - target, replace=False)
        h[to_flip] = 0.0
    elif ones < target:
        indices = np.where(h == 0.0)[0]
        to_flip = rng.choice(indices, target - ones, replace=False)
        h[to_flip] = 1.0
    return h

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible_binary(tm, n)

def antisymmetric_tm(n):
    h = np.zeros(n)
    half = n // 2
    for i in range(half):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
        h[i + half] = 1.0 - h[i]
    return h

def random_binary(n, seed):
    rng = np.random.default_rng(seed)
    h = (rng.rand(n) > 0.5).astype(float)
    h = make_feasible_binary(h, n)
    return h

def binary_swap_search(h, n, rng, max_trials, timeout, stall_limit=20000):
    """Local search: swap a random 1 and a random 0."""
    h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_c5_corr(h)
    best_corr_max = np.max(corr)
    no_improve = 0
    start_t = time.time()
    
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        corr = compute_c5_corr(h)
        cmax = np.max(corr)
        
        if cmax < best_corr_max - 1e-10:
            best_corr_max = cmax
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    
    h_best_c5 = best_corr_max * 2.0 / n
    return h, h_best_c5

                h[idx] = 0.0
