        offset = int(k_idx) - (n - 1)

            h_shift = h[offset:offset + n_pairs]
            h_base = 1.0 - h[:n_pairs]

    if gnorm < 1e-16:

                    h_cur = h_r.copy()
                    c5_cur = c5_r
                break

        # Multi-stage beta optimization
        beta_schedule = [50, 100, 200, 400, 800, 1500, 3000, 5000]
        for beta_val in beta_schedule:
            if budget_remain() < 10.0:
                break
            _G_BETA = float(beta_val)
            try:
                maxiter = 500 if beta_val < 500 else 300
                res = minimize(obj_smooth, h_cur, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    c5_cur = c5_new
                    h_cur = h_new
            except Exception:
                pass

        # Final raw objective polish
        if budget_remain() > 10.0:
            _G_BETA = 10.0
            try:
                res = minimize(obj_raw, h_cur, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    c5_cur = c5_new
                    h_cur = h_new
            except Exception:
                pass

        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
            print("  Phase2 improved: n={}, C5={:.8f}".format(n, best_c5))

    # Phase 3: Subgradient refinement at best resolution
    if budget_remain() > 30.0:
        print("\n--- Phase 3: Subgradient ---")
        best_h, best_c5 = do_subgradient_phase(best_h, best_n, budget_remain, rng)
        best_n = len(best_h)
        print("  After subgradient: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 4: Coordinate descent
    if budget_remain() > 15.0:
        print("\n--- Phase 4: Coordinate Descent ---")
        best_h, best_c5 = do_coordinate_descent(best_h, best_n, budget_remain, rng)
        best_n = len(best_h)
        print("  After coord descent: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 5: Final SLSQP polish with very high beta
    if budget_remain() > 10.0:
        print("\n--- Phase 5: Final Polish ---")
        n = best_n
        _G_N = n
        _G_TGT = n / 2.0
        _G_DX = 2.0 / n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum}]

        for beta_val in [5000, 10000]:
            if budget_remain() < 5.0:
                break
            _G_BETA = float(beta_val)
            try:
                res = minimize(obj_smooth, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 600, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 6: Extra coordinate descent rounds
    if budget_remain() > 10.0:
        best_h, best_c5 = do_coordinate_descent(best_h, best_n, budget_remain, rng)
        best_n = len(best_h)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
