    vals, lags = all_lag_data(h)
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    tw = float(np.sum(w))
    if tw <= 0.0:
        g = lag_grad(h, int(lags[int(np.argmax(vals))]))
        return g * dx
    g = np.zeros(n, dtype=float)
    top_k = min(40, len(lags))
    top_idx = np.argsort(-w)[:top_k]
    for ii in top_idx:
        ci = (float(w[ii]) / tw) * dx
        if ci > 1e-15:
            g += ci * lag_grad(h, int(lags[ii]))
    return g


def sum_eq_fun(h):
    return float(np.sum(h) - 0.5 * float(len(h)))


def sum_eq_jac(h):
    return np.ones(len(h), dtype=float)


def smooth_slsqp(h0, n, beta, maxiter=80, ftol=1e-12):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{"type": "eq", "fun": sum_eq_fun, "jac": sum_eq_jac}]
    try:
        res = minimize(
            _smooth_obj, h, method="SLSQP", bounds=bounds,
            constraints=constraints, args=(beta,),
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    return h, compute_c5(h)


def _active_t_obj(x):
    return float(x[-1])


def _active_t_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j


def _active_sum_eq(x):
    return float(np.sum(x[:-1]) - 0.5 * float(len(x) - 1))


def _active_sum_eq_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j


def _active_ineq(x, lags_list):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    out = np.empty(len(lags_list), dtype=float)
    for i, ll in enumerate(lags_list):
        out[i] = float(x[-1]) - float(np.dot(np.where(ll > 0, h[:n-abs(int(ll))] * (1 - h[abs(int(ll)):]), h[abs(int(ll)):] * (1 - h[:n-abs(int(ll))])) if ll != 0 else h * (1 - h)) * dx)
    return out


def _active_ineq_simple(x, lags_list):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    out = np.empty(len(lags_list), dtype=float)
    for i, ll in enumerate(lags_list):
        L = abs(int(ll))
        if L == 0:
            v = float(np.dot(h, 1.0 - h)) * dx
        else:
            m = n - L
            if ll > 0:
                v = float(np.dot(h[:m], 1.0 - h[L:])) * dx
            else:
                v = float(np.dot(h[L:], 1.0 - h[:m])) * dx
        out[i] = float(x[-1]) - v
    return out


def _active_ineq_jac(x, lags_list):
    n = len(x) - 1
    dx = 2.0 / float(n)
    m = len(lags_list)
    J = np.zeros((m, n + 1), dtype=float)
    h = x[:-1]
    for i, ll in enumerate(lags_list):
        g = lag_grad(h, int(ll))
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def active_set_opt(h0, active_lags, t0, maxiter=60, ftol=1e-12):
    h = np.asarray(h0, dtype=float)
    n = h.size
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = float(t0)
    bounds = [(0.0, 1.0) for _ in range(n)] + [(0.0, 1.0)]
    constraints = [{"type": "eq", "fun": _active_sum_eq, "jac": _active_sum_eq_jac}]
    if len(active_lags) > 0:
        constraints.append({"type": "ineq", "fun": _active_ineq_simple, "jac": _active_ineq_jac, "args": (active_lags,)})
    try:
        res = minimize(
            _active_t_obj, x0, method="SLSQP", bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res
    except Exception:
        pass
    return None


def refine_active_set(h0, n, time_limit, active_counts=(16, 32, 48, 64), maxiter=50):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    for acnt in active_counts:
        if time.time() > time_limit:
            break
        for _ in range(2):
            if time.time() > time_limit:
                break
            vals, lags = all_lag_data(h)
            cur_c = float(np.max(vals))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            top_k = min(int(acnt), len(lags))
            top_idx = np.argsort(-vals)[:top_k]
            active_list = [int(lags[i]) for i in top_idx]
            t0 = cur_c + 1e-4
            res = active_set_opt(h, active_list, t0, maxiter=maxiter)
            if res is not None:
                x = res.x
                h2 = project_feasible(x[:-1], n)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2.copy()
                if c2 <= cur_c + 1e-8:
                    h = h2
    return best_h, best_c


def nesterov_subgrad(h0, n, time_limit, max_iter=8000):
    h = project_feasible(h0, n)
    dx = 2.0 / float(n)
    best_h = h.copy()
    best_c = compute_c5(h)
    y = h.copy()
    t_nest = 1.0
    step = 0.02
    patience = 0
    patience_limit = 800

    for it in range(max_iter):
        if time.time() > time_limit:
            break
        beta_n = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        y = h + (beta_n - 1.0) / beta_n * (h - y)
        y = project_feasible(y, n)

        vals, lags = all_lag_data(y)
        max_v = float(np.max(vals))
        top_k = min(16, len(lags))
        top_idx = np.argsort(-vals)[:top_k]
        tv = vals[top_idx]
        w = np.exp(80.0 * (tv - tv[0]))
        tw = float(np.sum(w))
        g = np.zeros(n, dtype=float)
        for ii in range(top_k):
            if w[ii] / tw > 1e-6:
                g += (w[ii] / tw) * lag_grad(y, int(lags[top_idx[ii]]))
        g *= dx

        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)

        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
            step = min(0.04, step * 1.05)
        else:
            patience += 1
            if patience > patience_limit:
                break
            if c_new < best_c:
                step = min(0.04, step * 1.02)
            else:
                step *= 0.97

        h = h_new
        t_nest += 1.0

    return best_h, best_c


def block_perturb(h0, n, time_limit, seed, step=0.025):
    n = int(n)
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h
    cur_c = best_c
    temp = 0.003
    min_temp = 1e-7
    max_trials = 6000

    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(10, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, max(1, n - L + 1)))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.02, temp * 1.1)
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)

    return best_h, float(best_c)


def coord_optimize(h0, n, time_limit, n_passes=5):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    vals, lags = all_lag_data(h)
    cur_c = float(np.max(vals))

    for _pass in range(n_passes):
        if time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() > time_limit:
                break
            old_val = h[i]
            lo_b = max(0.0, old_val - 0.15)
            hi_b = min(1.0, old_val + 0.15)
            if lo_b >= hi_b:
                continue
            grid = np.linspace(lo_b, hi_b, 15)
            best_v = cur_c
            best_gv = old_val
            for gv in grid:
                h[i] = gv
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_v:
                    best_v = c
                    best_gv = gv
                h[i] = old_val
            if best_v < cur_c - 1e-12:
                h[i] = best_gv
                h = project_feasible(h, n)
                cur_c = best_v
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c


def build_starts(h_base, n, rng, count=5):
    n = int(n)
    starts = []
    if h_base is not None:
        starts.append(project_feasible(interpolate_to_n(h_base, n), n))
        fb = project_feasible(interpolate_to_n(h_base, n), n)
        starts.append(project_feasible(fb[::-1], n))
        starts.append(project_feasible(0.5 * (fb + fb[::-1]), n))
    else:
        starts.append(project_feasible(np.full(n, 0.5, dtype=float), n))
    h4 = np.zeros(n, dtype=float)
    q = n // 4
    if q > 0:
        h4[:q] = 1.0
        h4[n // 2:n // 2 + q] = 1.0
    starts.append(project_feasible(h4, n))
    hs = np.zeros(n, dtype=float)
    for i in range(n):
        if i % 2 == 0:
            hs[i] = 1.0
    starts.append(project_feasible(hs, n))
    hr = project_feasible(np.clip(np.full(n, 0.5, dtype=float) + rng.randn(n) * 0.15, 0, 1), n)
    starts.append(hr)
    out = [project_feasible(s, n) for s in starts]
    if count is not None:
        out = out[:count]
    return out


def improve_one(h0, n, time_limit, seed):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)

    if time.time() < time_limit - 30.0 and 2 <= n <= 320:
        for beta in (30, 100, 300):
            if time.time() > time_limit - 10.0:
                break
            h2, c2 = smooth_slsqp(h, n, beta, maxiter=60, ftol=1e-12)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
            h = h2 if c2 <= best_c + 1e-6 else best_h

    if time.time() < time_limit - 20.0 and 2 <= n <= 320:
        h2, c2 = refine_active_set(best_h, n, time_limit - 5.0,
                                    active_counts=(16, 32, 64), maxiter=50)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < time_limit - 8.0:
        h2, c2 = nesterov_subgrad(best_h, n, time_limit - 5.0)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < time_limit - 3.0:
        h2, c2 = block_perturb(best_h, n, time_limit - 1.0, seed, step=0.02)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        best_h = project_feasible(parent, parent_n)
        best_c = compute_c5(best_h)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    resolutions = []
    pn = int(best_n)
    for nn in (pn, pn - 16, pn + 32, 128, 192, 240, 288, 320):
        if 48 <= nn <= 360 and nn not in resolutions:
            resolutions.append(nn)
    resolutions.sort(key=lambda x: abs(x - pn))

    slot = min(200.0, (hard_end - start_time) / max(1, len(resolutions)))
    slot = max(slot, 40.0)

    for ri, nn in enumerate(resolutions):
        if time.time() > hard_end - 30.0:
            break
        nn = int(nn)
        if nn < 48 or nn > 360:
            continue
        rem = hard_end - time.time()
        if rem < 35.0:
            break
        s_end = min(time.time() + slot, hard_end - 25.0)
        starts = build_starts(parent if parent is not None else best_h, nn, rng, count=4)
        for si, s0 in enumerate(starts):
            if time.time() > s_end:
                break
            per = min(s_end - time.time(), max(15.0, (s_end - time.time()) / max(1, len(starts) - si)))
            hh, cc = improve_one(s0, nn, time.time() + per, seed + 1000 * ri + si)
            if cc < best_c:
                best_h = hh
                best_c = cc
                best_n = nn

    if time.time() < hard_end - 40.0 and 2 <= best_n <= 320:
        for beta in (200, 500):
            if time.time() > hard_end - 15.0:
                break
            h2, c2 = smooth_slsqp(best_h, best_n, beta, maxiter=80, ftol=1e-13)
            if c2 < best_c:
                best_h = h2
                best_c = c2

    if time.time() < hard_end - 25.0 and 2 <= best_n <= 300:
        h2, c2 = refine_active_set(best_h, best_n, hard_end - 10.0,
                                    active_counts=(32, 64, 96), maxiter=80)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < hard_end - 15.0 and 2 <= best_n <= 300:
        h2, c2 = nesterov_subgrad(best_h, best_n, hard_end - 8.0)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < hard_end - 8.0:
        h2, c2 = block_perturb(best_h, best_n, hard_end - 2.0, seed + 77777, step=0.015)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
