    m = np.max(c)
    sh = _g_beta * (c - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / _g_beta
    penalty = _g_pen * (np.sum(x) - n / 2.0) ** 2
    return smax * (2.0 / n) + penalty

def _obj_topk_pen(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode='full')
    s = np.sort(c)
    top = s[-_g_k:]
    penalty = _g_pen * (np.sum(x) - n / 2.0) ** 2
    return np.mean(top) * (2.0 / n) + penalty

def _obj_raw_pen(x):
    n = _g_n
    c = np.correlate(x, 1.0 - x, mode='full')
    penalty = _g_pen * (np.sum(x) - n / 2.0) ** 2
    return np.max(c) * (2.0 / n) + penalty

def _eq_con(x):
    return np.sum(x) - _g_n / 2.0

def subgradient_step_vec(h, n, step0, n_steps, t0, time_limit):
    dx = 2.0 / n
    step = step0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for s in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        active = corr >= max_val - 1e-6
        n_active = max(1, int(np.sum(active)))
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for lag_idx in active_idx:
            offset = lag_idx - (n - 1)
            if offset >= 0:
                i0, i1 = 0, n - offset
                grad[i0:i1] += (1.0 - h[i0+offset:i1+offset])
                grad[i0+offset:i1+offset] += (-h[i0:i1])
            else:
                o = -offset
                i0, i1 = o, n
                grad[i0:i1] += (1.0 - h[i0-o:i1-o])
                grad[i0-o:i1-o] += (-h[i0:i1])
        grad /= n_active
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        step *= 0.93
    return best_h, best_c5

def binary_local_search_sys(h, n, t0, time_limit):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(12345)
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        ones = list(np.where(best_h == 1)[0])
        zeros = list(np.where(best_h == 0)[0])
        n_tries = min(len(ones) * len(zeros), 5000)
        for _ in range(n_tries):
            if time.time() - t0 > time_limit:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            best_h[i] = 0.0
            best_h[j] = 1.0
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                improved = True
                ones = list(np.where(best_h == 1)[0])
                zeros = list(np.where(best_h == 0)[0])
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0
    return best_h, best_c5

def structured_init(n, rng, pattern_type):
    t = np.arange(n) / n
    if pattern_type == 0:
        freq = rng.randint(2, 12)
        phase = rng.rand() * 2 * np.pi
        h = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t + phase)
    elif pattern_type == 1:
        period = max(4, n // rng.randint(4, 20))
        half = period // 2
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % period) < half else 0.0
    elif pattern_type == 2:
        n_blocks = rng.randint(3, 15)
        block_size = max(1, n // n_blocks)
        h = np.zeros(n)
        for b in range(n_blocks):
            if rng.rand() < 0.5:
                h[b*block_size:min((b+1)*block_size, n)] = 1.0
    else:
        h = 0.5 + 0.3 * rng.randn(n)
    return project_feasible(h, n)

def optimize_slsqp_at_n(h_init, n, t0, time_limit):
    global _g_n, _g_beta, _g_k, _g_pen
    _g_n = n
    h_cur = project_feasible(h_init, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]
    t_start = time.time()
    for beta in [10.0, 30.0, 100.0, 300.0, 1000.0]:
        if time.time() - t0 > time_limit - 5:
            break
        _g_beta = beta
        _g_pen = 0.0
        try:
            res = minimize(_obj_smooth_pen, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass
    # Top-k phase
    if time.time() - t0 < time_limit - 5:
        _g_beta = 1000.0
        _g_k = max(3, (2 * n - 1) // 60)
        _g_pen = 0.0
        try:
            res = minimize(_obj_topk_pen, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass
    # Final raw max
    if time.time() - t0 < time_limit - 5:
        _g_beta = 10000.0
        _g_pen = 0.0
        try:
            res = minimize(_obj_raw_pen, h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass
    # Subgradient polish
    if time.time() - t0 < time_limit - 3:
        h_cur, _ = subgradient_step_vec(h_cur, n, 0.005, 40, t0, time_limit - 3)
    return h_cur

def optimize_lbfgs_at_n(h_init, n, t0, time_limit):
    global _g_n, _g_beta, _g_k, _g_pen
    _g_n = n
    h_cur = project_feasible(h_init, n)
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    for beta, pen in [(20.0, 100.0), (100.0, 500.0), (500.0, 2000.0)]:
        if time.time() - t0 > time_limit - 5:
            break
        _g_beta = beta
        _g_pen = pen
        try:
            res = minimize(_obj_smooth_pen, h_cur, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass
    if time.time() - t0 < time_limit - 3:
        h_cur, _ = subgradient_step_vec(h_cur, n, 0.005, 50, t0, time_limit - 3)
    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 0:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass

    if best_h is None:
        best_n = 192
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    def remaining():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)

    # Phase 1: Optimize at parent resolution with multiple starts
    n_parent = best_n
    time_phase1 = min(300.0, budget_s * 0.25)
    t1_start = time.time()

    inits_p = [best_h.copy()]
    for scale in [0.03, 0.08, 0.15, 0.3]:
        inits_p.append(project_feasible(best_h + scale * rng.randn(n_parent), n_parent))
    for pt in range(4):
        inits_p.append(structured_init(n_parent, rng, pt))

    time_per_init_p = max(8.0, time_phase1 / max(1, len(inits_p)))
    for h0 in inits_p:
        if time.time() - t1_start > time_phase1 or remaining() < 50:
            break
        h_opt = optimize_slsqp_at_n(h0, n_parent, t0, time.time() - t1_start + time_phase1 - 10)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n_parent

    # Phase 2: Try nearby resolutions
    n_list = [n_parent]
    for nn in [n_parent - 4, n_parent - 8, n_parent + 4, n_parent + 8,
               n_parent + 16, n_parent + 32, n_parent + 64, n_parent + 128,
               128, 256, 384, 512]:
        if 48 <= nn <= 768 and nn not in n_list:
            n_list.append(nn)
    n_list.sort()

    time_phase2 = min(budget_s * 0.35, remaining())
    t2_start = time.time()
    n_per = max(1, len(n_list))

    for n in n_list:
        if n == n_parent:
            continue
        if time.time() - t2_start > time_phase2 or remaining() < 40:
            break
        xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
        xn = (np.arange(n) + 0.5) * 2.0 / n
        h_start = project_feasible(np.interp(xn, xp, best_h), n)
        inits_n = [h_start.copy()]
        for pt in range(2):
            inits_n.append(structured_init(n, rng, pt))
        inits_n.append(project_feasible(h_start + 0.1 * rng.randn(n), n))

        time_per_n = max(10.0, time_phase2 / max(1, n_per))
        t_n_start = time.time()
        for h0 in inits_n:
            if time.time() - t_n_start > time_per_n or remaining() < 30:
                break
            if n <= 320:
                h_opt = optimize_slsqp_at_n(h0, n, t0, time.time() - t_n_start + time_per_n - 5)
            else:
                h_opt = optimize_lbfgs_at_n(h0, n, t0, time.time() - t_n_start + time_per_n - 5)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n

    # Phase 3: Binary local search on best
    if remaining() > 15:
        h_bin, c5_bin = binary_local_search_sys(best_h, best_n, t0, remaining() - 10)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 4: Fine SLSQP polish at best resolution
    if remaining() > 15:
        n = best_n
        global _g_n, _g_beta, _g_k, _g_pen
        _g_n = n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_con}]
        h_pol = project_feasible(best_h, n)
        for beta in [2000.0, 10000.0, 50000.0]:
            if remaining() < 10:
                break
            _g_beta = beta
            _g_pen = 0.0
            try:
                res = minimize(_obj_smooth_pen, h_pol, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 800, 'ftol': 1e-16, 'iprint': 0})
                h_pol = project_feasible(res.x, n)
                c5_p = compute_c5(h_pol)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pol.copy()
            except Exception:
                pass

    # Phase 5: Subgradient refinement
    if remaining() > 10:
        h_sg, c5_sg = subgradient_step_vec(best_h, best_n, 0.003, 60, t0, remaining() - 5)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 6: Final binary search
    if remaining() > 8:
        h_bin, c5_bin = binary_local_search_sys(best_h, best_n, t0, remaining() - 3)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 7: One more SLSQP after binary
    if remaining() > 8:
        n = best_n
        global _g_n, _g_beta, _g_pen
        _g_n = n
        _g_beta = 20000.0
        _g_pen = 0.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_con}]
        try:
            res = minimize(_obj_smooth_pen, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 600, 'ftol': 1e-16, 'iprint': 0})
            h_pol = project_feasible(res.x, n)
            c5_p = compute_c5(h_pol)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pol.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
