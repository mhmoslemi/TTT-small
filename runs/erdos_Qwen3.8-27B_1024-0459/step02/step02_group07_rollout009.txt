Where k_a and k_b depend on m. Let me just compute it directly.

For position a, its contribution to corr[m] (where m = k + n - 1):
- As first factor: h[a]*(1-h[a+k]) where k = m-(n-1), valid when 0≤a+k<n
  → contributes (1-h[a+k]) to corr[m]
- As second factor: h[a-k]*(1-h[a]) where k = m-(n-1), valid when 0≤a-k<n
  → contributes -h[a-k] to corr[m]

So the influence vector for position a (length 2n-1) is:
For m = k + n - 1:
  if 0 ≤ a+k < n: influence[m] += (1-h[a+k])
  if 0 ≤ a-k < n: influence[m] += -h[a-k]

This is O(n) to compute per position. For a swap of a and b:
Δcorr = -influence[a] + influence[b] (since h[a] goes 1→0, δ=-1; h[b] goes 0→1, δ=+1)

New max = max(corr + Δcorr)

But computing influence vectors for all a is O(n²) total, and we only need 2 per swap. So per swap: O(n) for the two influence vectors + O(n) for the new max. Total O(n) per swap.

For 200,000 swaps at n=160: 200,000 * 320 ≈ 64 million. Very fast!

This makes simulated annealing very feasible.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_desc(h, n, n_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    nc = 2 * n - 1
    start = time.time()
    
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        rng = np.random.RandomState(p + 42)
        order = rng.permutation(n)
        for j in order:
            old_j = h[j]
            # Compute change in corr if h[j] changes
            # corr[m] new = corr[m] + delta * inf_j[m]
            # where delta = new_h[j] - old_j
            # inf_j[m]: coefficient of h[j] in corr[m]
            
            # Compute influence of position j
            # As first factor: h[j]*(1-h[j+k]) → coeff = (1-h[j+k])
            #   valid when 0 <= j+k < n, k = m-(n-1)
            # As second factor: h[j-k]*(1-h[j]) → coeff = -h[j-k]
            #   valid when 0 <= j-k < n, k = m-(n-1)
            
            # First factor: k in [-j, n-1-j], m in [n-1-j, 2n-2-j]
            k_lo1 = max(-(n-1), -j)
            k_hi1 = min(n-1, n-1-j)
            # Second factor: k in [j-n+1, j], m in [j, 2n-1-j]... wait
            # k in [j-n+1, j], m = k+n-1 in [j, 2n-1-j]... 
            # Hmm let me be more careful.
            # m = k + n - 1
            # First factor valid: 0 <= j+k < n → -j <= k < n-j → k in [max(-(n-1),-j), min(n-1, n-1-j)]
            #   m in [max(0, n-1-j), min(2n-2, 2n-2-j)]
            # Second factor valid: 0 <= j-k < n → j-n+1 <= k <= j → k in [max(-(n-1),j-n+1), min(n-1, j)]
            #   m in [max(0, j), min(2n-2, 2n-2-j)]... 
            #   m = k+n-1, so m in [max(-(n-1),j-n+1)+n-1, min(n-1,j)+n-1]
            #   = [max(0, j), min(2n-2, n-1+j)]
            
            # Let me just compute it directly for clarity:
            inf = np.zeros(nc)
            if k_lo1 <= k_hi1:
                ks = np.arange(k_lo1, k_hi1+1)
                m_idx = ks + n - 1
                inf[m_idx] += (1.0 - h[j + ks])
            k_lo2 = max(-(n-1), j-n+1)
            k_hi2 = min(n-1, j)
            if k_lo2 <= k_hi2:
                ks2 = np.arange(k_lo2, k_hi2+1)
                m_idx2 = ks2 + n - 1
                inf[m_idx2] += -h[j - ks2]
            
            # Binary search for optimal h[j]
            lo, hi = -old_j, 1.0 - old_j
            for _ in range(30):
                mid = 0.5 * (lo + hi)
                new_corr = corr + mid * inf
                mx = np.max(new_corr)
                active = new_corr >= mx - 1e-13
                sg = np.mean(inf[active])
                if sg > 1e-12:
                    lo = mid
                elif sg < -1e-12:
                    hi = mid
                else:
                    break
            new_val = np.clip(old_j + 0.5*(lo+hi), 0.0, 1.0)
            
            if abs(new_val - old_j) > 1e-10:
                h[j] = new_val
                corr += (new_val - old_j) * inf
                improved = True
        
        # Enforce constraint
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def simulated_anneal_binary(h_bin, n, n_steps, time_limit, rng, T0=1.0):
    h = h_bin.copy().astype(float)
    corr = get_corr(h)
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    start = time.time()
    T = T0
    cooling = (1e-5 / T0) ** (1.0 / max(n_steps, 1))
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        # Pick a swap: one 1 and one 0
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        
        if a == b:
            continue
        
        # Compute new correlation
        # h[a]: 1→0, h[b]: 0→1
        # Δcorr[m] = -(contribution of a as 1) + (contribution of b as 1)
        # Contribution of position p as h[p]=1 to corr[m]:
        #   As first factor: (1-h[p+k]) for valid k
        #   As second factor: -h[p-k] for valid k
        # But when we change h[a] from 1 to 0, the contributions change:
        #   First factor terms: were h[a]*(1-h[a+k]) = 1*(1-h[a+k]), now 0 → change = -(1-h[a+k])
        #   Second factor terms: were h[a-k]*(1-h[a]) = h[a-k]*0 = 0, now h[a-k]*1 → change = +h[a-k]
        # Similarly for b going 0→1:
        #   First factor: 0→1, change = +(1-h[b+k])
        #   Second factor: h[b-k]*1 → h[b-k]*0, change = -h[b-k]
        
        # So Δcorr = -[(1-h[a+k]) terms] + [h[a-k] terms] + [(1-h[b+k]) terms] - [h[b-k] terms]
        
        # Let me compute this more carefully:
        # Old corr at m = k+n-1:
        #   sum_j h[j]*(1-h[j+k]) for valid j
        # After swap (h[a]:1→0, h[b]:0→1):
        #   For terms where j=a: old = 1*(1-h[a+k]), new = 0 → Δ = -(1-h[a+k])
        #   For terms where j+k = a, i.e., j = a-k: old = h[a-k]*(1-1) = 0, new = h[a-k]*(1-0) = h[a-k] → Δ = +h[a-k]
        #   For terms where j=b: old = 0, new = 1*(1-h[b+k]) → Δ = +(1-h[b+k])
        #   For terms where j+k = b, i.e., j = b-k: old = h[b-k]*(1-0) = h[b-k], new = h[b-k]*(1-1) = 0 → Δ = -h[b-k]
        
        # So:
        # Δcorr[m] for m = k+n-1:
        #   -(1-h[a+k]) [if 0<=a+k<n]
        #   + h[a-k] [if 0<=a-k<n]
        #   +(1-h[b+k]) [if 0<=b+k<n]
        #   - h[b-k] [if 0<=b-k<n]
        
        delta = np.zeros(2*n-1)
        # a first factor: k in [-a, n-1-a]
        k_lo = max(-(n-1), -a)
        k_hi = min(n-1, n-1-a)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi+1)
            delta[ks + n - 1] += -(1.0 - h[a + ks])
        # a second factor: k in [a-n+1, a]
        k_lo2 = max(-(n-1), a-n+1)
        k_hi2 = min(n-1, a)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2+1)
            delta[ks2 + n - 1] += h[a - ks2]
        # b first factor: k in [-b, n-1-b]
        k_lo3 = max(-(n-1), -b)
        k_hi3 = min(n-1, n-1-b)
        if k_lo3 <= k_hi3:
            ks3 = np.arange(k_lo3, k_hi3+1)
            delta[ks3 + n - 1] += (1.0 - h[b + ks3])
        # b second factor: k in [b-n+1, b]
        k_lo4 = max(-(n-1), b-n+1)
        k_hi4 = min(n-1, b)
        if k_lo4 <= k_hi4:
            ks4 = np.arange(k_lo4, k_hi4+1)
            delta[ks4 + n - 1] += -h[b - ks4]
        
        new_corr = corr + delta
        old_max = np.max(corr)
        new_max = np.max(new_corr)
        
        if new_max <= old_max or rng.random() < np.exp((old_max - new_max) / T):
            # Accept
            h[a] = 0.0
            h[b] = 1.0
            ones[ia] = b  # was a, now b
            zeros[ib] = a  # was b, now a
            corr = new_corr.copy()
            if new_max * dx < best_c5 - 1e-14:
                best_c5 = new_max * dx
                best_h = h.copy()
        
        T *= cooling
    
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def slsqp_polish(h, n, max_iter=3000):
    def sum_constraint(x, n_):
        return np.sum(x) - n_ / 2.0
    
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def generate_structured_inits(n, rng):
    inits = []
    # Block patterns
    if n % 4 == 0:
        inits.append(np.tile([1.0, 1.0, 0.0, 0.0], n//4).copy())
        inits.append(np.tile([1.0, 0.0, 0.0, 1.0], n//4).copy())
    # Thue-Morse
    tm = np.array([1.0 if bin(i).count('1') % 2 == 0 else 0.0 for i in range(n)])
    inits.append(make_feasible(tm, n))
    # Two blocks
    for frac in [0.3, 0.35, 0.4, 0.45]:
        h = np.zeros(n)
        blk = int(n * frac)
        h[:blk] = 1.0
        mid = n//2
        h[mid:mid+blk] = 1.0
        inits.append(make_feasible(h, n))
    # Random
    for _ in range(3):
        h = (rng.rand(n) > 0.5).astype(float)
        inits.append(make_feasible(h, n))
    # Golden ratio based
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * 0.618033988749895) % 1.0 > 0.5 else 0.0
    inits.append(make_feasible(h, n))
    return inits
```
