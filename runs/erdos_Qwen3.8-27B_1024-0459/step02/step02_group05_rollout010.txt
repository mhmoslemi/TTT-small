    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    p = exps / exps.sum()  # length 2n-1

    c2r = np.correlate(h, p_rev, 'full')    # length 3n-2
    
    grad = (c1 - c2r)[3*n-3 : 2*n-3 : -1]  # length n
    return grad, cmax

def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)

def compute_smooth_grad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, 'full')
    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    p = exps / exps.sum()
    p_rev = p[::-1]
    c1 = np.correlate(1.0 - h, p, 'full')
    c2r = np.correlate(h, p_rev, 'full')
    grad = (c1 - c2r)[3*n-3 : 2*n-3 : -1].copy()
    return grad, cmax

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            return h
        h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
    else:

        cnt = np.count_nonzero(mask)
        if cnt == 0:
            return h
        h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    return np.clip(h, 0.0, 1.0)

def pgd_solve(h_init, n, n_iter=10000, beta_start=50, beta_end=5000, 
              alpha_start=0.05, time_limit=100):
    """Projected gradient descent with adaptive beta and step size."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_t = time.time()
    
    alpha = alpha_start
    for it in range(n_iter):
        if time.time() - start_t > time_limit:
            break
        
        # Anneal beta
        frac = it / max(n_iter - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        grad, cmax = compute_smooth_grad(h, n, beta)
        c5 = cmax  # without dx multiplier for gradient (dx is constant)
        
        # Remove mean component of gradient (for sum constraint)
        grad = grad - np.mean(grad)
        
        # Backtracking line search
        alpha_try = alpha
        for _ in range(20):
            h_new = project_feasible(h - alpha_try * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                alpha = min(alpha_try * 1.5, 0.5)
                break
            elif c5_new < compute_c5(h) - 1e-15:
                h = h_new
                alpha = alpha_try
                break
            else:
                alpha_try *= 0.5
        else:
            alpha *= 0.7
    
    return best_h, best_c5
