def _topk_ineq(x):

def _topk_ineq_jac(x):
    n = _g_n
    h = x[:n]

    sm = 1.0 - 2.0 * h
    k_list = _g_lags
    k_count = len(k_list)

    n = len(h)

    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            cnt = len(free)
            if cnt == 0:
                h += diff / n
            else:
                h[free] += diff / cnt
        else:
            free = np.where(h > 1e-12)[0]
            cnt = len(free)
            if cnt == 0:
                h += diff / n
            else:
                h[free] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            h[int(np.argmax(1.0 - h))] += diff
        else:
            h[int(np.argmax(h))] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_fun(h, beta):

    s = np.sum(np.exp(sh))
    if s < 1e-300:
        return cmax * dx
    return dx * (cmax + np.log(s) / beta)
