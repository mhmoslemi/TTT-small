import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    # Simple projection: adjust all values
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # If still off, adjust only free variables
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Find the most "free" variable
        idx = np.argmin(np.minimum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    
    # L = 0

    if w0 > 1e-15:

        w = weights[n-1+L] + weights[n-1-L]
        if w < 1e-15:

        grad[:n-L] += w * (1.0 - h[L:])
        grad[L:] -= w * h[:n-L]

def slsqp_optimize(h0, n, beta, max_iter=300):
    """SLSQP optimization with smooth max"""
    def fun(x):
        val, _ = compute_gradient_smoothmat(x, n, beta)
        return val
    
    def jac(x):
        g, _ = compute_gradient_smoothmat(x, n, beta)
        return g
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    
    res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
    
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

                count_active += 1
                w = 1
                if active[n-1+L] and active[n-1-L]:
                    w = 2  # both +L and -L are active... 
                    # Actually they contribute the same gradient direction
                # Gradient for |L|:
                grad[:n-L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n-L]
        
        if count_active == 0:
            count_active = 1
            grad += (1.0 - 2.0 * h)
        
        grad = dx * grad / count_active
        
        # Gradient step
        h = h - lr * grad
        h = project_feasible(h, n)
        
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
        
        lr *= lr_decay
    
    return best_h, best_val

def binary_local_search(h, n, max_rounds=20000, rng=None):
    """Local search on binary: flip pairs"""
    if rng is None:
        rng = np.random.RandomState(0)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h_bin == 1.0)[0]
        h_bin[ones[n_ones-target_ones:]] = 0.0
    elif n_ones < target_ones:
        zeros = np.where(h_bin == 0.0)[0]
        h_bin[zeros[:target_ones-n_ones]] = 1.0
    
    best_c5 = compute_c5(h_bin)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    
    for rnd in range(max_rounds):
        corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        max_k = np.argmax(corr)
        max_L = max_k - (n - 1)
        
        # Try flips involving positions that contribute to the worst lag
        improved = False
        for attempt in range(10):
            # Pick a position to flip
            if len(ones) > 0 and len(zeros) > 0:
                i1 = rng.choice(ones)
                i0 = rng.choice(zeros)
                h_bin[i1] = 0.0
                h_bin[i0] = 1.0
                new_c5 = compute_c5(h_bin)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    ones = np.where(h_bin == 1.0)[0]
                    zeros = np.where(h_bin == 0.0)[0]
                    improved = True
                    break
                else:
                    h_bin[i1] = 1.0
                    h_bin[i0] = 0.0
        if not improved:
            # Try a few more random flips
            for attempt in range(5):
                if len(ones) > 2 and len(zeros) > 2:
                    i1 = rng.choice(ones)
                    i0 = rng.choice(zeros)
                    h_bin[i1] = 0.0
                    h_bin[i0] = 1.0
                    new_c5 = compute_c5(h_bin)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        ones = np.where(h_bin == 1.0)[0]
                        zeros = np.where(h_bin == 0.0)[0]
                        improved = True
                        break
                    else:
                        h_bin[i1] = 1.0
                        h_bin[i0] = 0.0
            if not improved:
                # Check if we've stagnated
                if rnd > 100:
                    break
    
    return h_bin, best_c5
