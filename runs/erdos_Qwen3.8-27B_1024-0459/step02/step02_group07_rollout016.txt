```python
def solve_1d_minimax(b, a, lo=0.0, hi=1.0):
    """Solve min_{t in [lo,hi]} max_k (b[k] + a[k]*t)"""
    n_lines = len(b)
    # Evaluate g at boundaries
    candidates = [lo, hi]
    
    # Find crossing points of adjacent lines (in t)
    # A crossing of lines i and j is at t = (b[j]-b[i])/(a[i]-a[j]) if a[i]≠a[j]
    # But we only care about crossings in [lo,hi]
    # To be efficient, only check top lines at a few sample points
    
    # Sample at several points to find active sets
    n_samples = 20
    sample_ts = np.linspace(lo, hi, n_samples)
    for t_s in sample_ts:
        vals = b + t_s * a
        mx = np.max(vals)
        active = np.where(vals >= mx - 1e-10)[0]
        # For each active line, find where it crosses with the (active+1)th line
        sorted_idx = np.argsort(b + t_s * a)[::-1]
        for i_idx in range(min(5, len(sorted_idx))):
            for j_idx in range(i_idx+1, min(10, len(sorted_idx))):
                i, j = sorted_idx[i_idx], sorted_idx[j_idx]
                if abs(a[i] - a[j]) > 1e-15:
                    t_cross = (b[j] - b[i]) / (a[i] - a[j])
                    if lo <= t_cross <= hi and t_cross not in candidates:
                        # Check if at t_cross, these two lines are actually on the max
                        candidates.append(t_cross)
    
    # Evaluate g at all candidates
    best_t = lo
    best_val = np.inf
    for t_c in candidates:
        t_c = np.clip(t_c, lo, hi)
        vals = b + t_c * a
        g_val = np.max(vals)
        if g_val < best_val - 1e-15:
            best_val = g_val
            best_t = t_c
    
    return best_t
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_stable(x, T):
    m = np.max(x)
    shifted = T * (x - m)
    log_sum = m + np.log(np.sum(np.exp(shifted))) / T
    return log_sum


def smooth_obj_and_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = lse_stable(corr, T) * dx
    m = np.max(corr)
    exp_shifted = np.exp(T * (corr - m))
    w = exp_shifted / np.sum(exp_shifted)
    nw = 2 * n - 1
    grad = np.zeros(n)
    for j in range(n):
        s_lo1 = 0
        s_hi1 = min(n - 1 - j, n - 1)
        if s_lo1 <= s_hi1:
            indices = np.arange(s_lo1, s_hi1 + 1)
            grad[j] += np.dot(w[n - 1 + indices], 1.0 - h[j + indices])
        s_lo2 = max(-(n - 1), -(n - 1 - j))
        s_hi2 = j
        if s_lo2 <= s_hi2:
            indices2 = np.arange(s_lo2, s_hi2 + 1)
            grad[j] -= np.dot(w[n - 1 - indices2], h[j - indices2])

    return f, grad


def solve_1d_minimax(b, a, lo=0.0, hi=1.0):

    n_samples = 30
    sample_ts = np.linspace(lo, hi, n_samples)

    for t_s in sample_ts:

def coord_descent(h, n, n_passes, time_limit):

    best_h = h.copy()
    dx = 2.0 / n
    nw = 2 * n - 1
    start = time.time()
    stagnant = 0
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_improve = 0.0
        for j in range(n):
            old_val = h[j]
            a = np.zeros(nw)
            m_lo1 = max(0, 0)
            m_hi1 = min(n - 1 - j, n - 1)
            if m_lo1 <= m_hi1:
                idx = np.arange(m_lo1, m_hi1 + 1)
                a[n - 1 + idx] += (1.0 - h[j + idx])
            m_lo2 = max(-(n - 1), -(n - 1 - j))
            m_hi2 = min(n - 1, j)
            if m_lo2 <= m_hi2:
                idx2 = np.arange(m_lo2, m_hi2 + 1)
                a[n - 1 + idx2] += -h[j - idx2]
            b = corr - old_val * a

def slsqp_smooth(h, n, T=30, max_iter=3000):

        f, _ = smooth_obj_and_grad(h_curr, n, T)
        return f

    def obj_grad(x):

        return f, g

    h_curr = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

        res = minimize(lambda x: smooth_obj_and_grad(x, n, T)[0],
                       h_curr,

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)

        pass
    return None, None


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def generate_periodic_h(n, period, pattern_values, rng):
    h = np.zeros(n)
    for i in range(n):
        h[i] = pattern_values[i % period]
    return make_feasible(h, n)
```
