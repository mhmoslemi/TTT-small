def precompute_deltas(h):
    """Precompute delta matrices for binary local search.
    Returns A_pos, B_pos, A_neg, B_neg: shape (n, n-1) each.
    A_*[a, L-1] = effect of flipping h[a] from 1 to 0 on overlap at lag L
    B_*[b, L-1] = effect of flipping h[b] from 0 to 1 on overlap at lag L
    """
    n = len(h)
    L = np.arange(1, n)  # L = 1, 2, ..., n-1, length n-1
    
    # For A_pos[a, L]: -[L<=n-1-a]*(1-h[a+L]) + [L<=a]*h[a-L]
    # a is 0..n-1, L is 1..n-1
    a_idx = np.arange(n)[:, None]  # (n, 1)
    L_idx = L[None, :]  # (1, n-1)
    
    # Condition L <= n-1-a: a+L <= n-1
    cond1 = (a_idx + L_idx) <= (n - 1)  # (n, n-1)
    # h[a+L] where valid, 0 otherwise
    # We need h[a+L] for a+L in [0, n-1]
    aL = a_idx + L_idx  # (n, n-1)
    # For valid positions (a+L <= n-1), h[a+L]
    h_al = np.where(cond1, h[np.minimum(aL, n-1)], 0)  # clip to avoid index error
    
    # Condition L <= a: a-L >= 0
    cond2 = a_idx >= L_idx  # (n, n-1)
    aLm = a_idx - L_idx  # (n, n-1), valid when >= 0
    h_aLm = np.where(cond2, h[np.maximum(aLm, 0)], 0)
    
    A_pos = -cond1.astype(float) * (1 - h_al) + cond2.astype(float) * h_aLm
    
    # For B_pos[b, L]: [L<=n-1-b]*(1-h[b+L]) - [L<=b]*h[b-L]
    # b is 0..n-1
    b_idx = np.arange(n)[:, None]
    
    cond1_b = (b_idx + L_idx) <= (n - 1)
    bL = b_idx + L_idx
    h_bL = np.where(cond1_b, h[np.minimum(bL, n-1)], 0)
    
    cond2_b = b_idx >= L_idx
    bLm = b_idx - L_idx
    h_bLm = np.where(cond2_b, h[np.maximum(bLm, 0)], 0)
    
    B_pos = cond1_b.astype(float) * (1 - h_bL) - cond2_b.astype(float) * h_bLm
    
    # For A_neg[a, L]: -[L<=a]*(1-h[a-L]) + [L<=n-1-a]*h[a+L]
    A_neg = cond2.astype(float) * (-(1 - h_aLm)) + cond1.astype(float) * h_al
    # = -cond2*(1-h[a-L]) + cond1*h[a+L]
    
    # For B_neg[b, L]: [L<=b]*(1-h[b-L]) - [L<=n-1-b]*h[b+L]
    B_neg = cond2_b.astype(float) * (1 - h_bLm) - cond1_b.astype(float) * h_bLm
    
    return A_pos, B_pos, A_neg, B_neg


def binary_local_search(h, max_iters=10000, time_limit=100):
    """Intensive binary local search using best-improvement."""
    start = time.time()
    n = len(h)
    h = h.copy()
    
    def elapsed():
        return time.time() - start
    
    pos, neg, c5 = get_overlaps(h)
    best_c5 = c5
    stuck_count = 0

            break
        if remaining_budget < 30:  # global check
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Precompute deltas
        A_pos, B_pos, A_neg, B_neg = precompute_deltas(h)
        
        # For best-improvement: find (a in ones, b in zeros) minimizing new_max
        # Vectorize over b for each a
        best_new_max = best_c5 / (2.0/n) - 1  # current max overlap value (integer)
        best_pair = None
        best_a = None
        best_b = None
        
        ones_A_pos = A_pos[ones]  # (n_ones, n-1)
        ones_A_neg = A_neg[ones]
        zeros_B_pos = B_pos[zeros]  # (n_zeros, n-1)
        zeros_B_neg = B_neg[zeros]
        
        # For each a, compute base: pos + A_pos[a], neg + A_neg[a]
        # Then for each b, new_max = max(max(base_pos + B_pos[b]), max(base_neg + B_neg[b]))
        
        # This is O(n_ones * n_zeros * n) which for n=200 is 100*100*199 ≈ 2M
        # Let's do it in chunks to manage memory
        
        chunk_size = 20
        for a_start in range(0, len(ones), chunk_size):
            a_chunk = ones[a_start:a_start+chunk_size]
            
            base_pos = pos[None, :] + A_pos[a_chunk, :]  # (chunk, n-1)
            base_neg = neg[None, :] + A_neg[a_chunk, :]
            
            # For all b:
            new_pos_all = base_pos[:, None, :] + zeros_B_pos[None, :, :]  # (chunk, n_zeros, n-1)
            new_neg_all = base_neg[:, None, :] + zeros_B_neg[None, :, :]
            
            max_pos = new_pos_all.reshape(chunk * len(zeros), -1).max(axis=1).reshape(chunk, -1)
            max_neg = new_neg_all.reshape(chunk * len(zeros), -1).max(axis=1).reshape(chunk, -1)
            
            new_max = np.maximum(max_pos, max_neg)  # (chunk, n_zeros)
            
            flat_max = np.max(new_max)
            if flat_max < best_new_max:
                idx = np.argmin(new_max)
                local_a = idx // len(zeros)
                local_b = idx % len(zeros)
                best_new_max = new_max[local_a, local_b]
                best_a = a_chunk[local_a]
                best_b = zeros[local_b]
        
        if best_a is not None and best_new_max < np.max(np.maximum(pos, neg)):
            h[best_a] = 0
            h[best_b] = 1
            pos, neg, c5 = get_overlaps(h)
            best_c5 = c5
            stuck_count = 0
        else:
            stuck_count += 1
            # If stuck for a while, do a random kick
            if stuck_count > 20:
                # Randomly flip 2-3 pairs
                n_kick = rng.randint(2, 5)
                for _ in range(n_kick):
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    h[a] = 0
                    h[b] = 1
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                pos, neg, c5 = get_overlaps(h)
                stuck_count = 0
    
    return h, best_c5

- j=a: h[a+L]*(1-h[a]) changes from h[a+L]*0 = 0 to h[a+L]*1 = h[a+L]. Change: +h[a+L]. Valid when a ≤ n-1-L.

Effect of h[b] 0→1:
- j+L=b, j=b-L: h[b]*(1-h[b-L]) changes from 0 to (1-h[b-L]). Change: +(1-h[b-L]). Valid when b ≥ L.
- j=b: h[b+L]*(1-h[b]) changes from h[b+L]*1 = h[b+L] to h[b+L]*0 = 0. Change: -h[b+L]. Valid when b ≤ n-1-L.
