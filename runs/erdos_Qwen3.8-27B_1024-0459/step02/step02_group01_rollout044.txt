        if c5_cur <= 0.38085:
            break
        i = rng.randint(n)
        h_try = h_cur.copy()
        h_try[i] = np.clip(h_try[i] + rng.randn() * 0.1, 0.0, 1.0)
        h_try += (n / 2.0 - np.sum(h_try)) / n
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < c5_cur:
            h_cur = h_try
            c5_cur = c5_try
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
    return best_h, best_c5

def top_lag_refinement(h, n, n_iter=2000):
    h_cur = project_feasible(h, n)
    c5_cur = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = c5_cur
    for it in range(n_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top5_idx = np.argsort(corr)[-5:]
        max_val = np.max(corr)
        grad = np.zeros(n)
        for lag in top5_idx:
            offset = int(lag) - (n - 1)
            if offset >= 0:
                for i in range(n - offset):
                    grad[i] += (1.0 - h_cur[i + offset])
                    grad[i + offset] += (-h_cur[i])
            else:
                o = -offset
                for i in range(o, n):
                    grad[i] += (1.0 - h_cur[i - o])
                    grad[i - o] += (-h_cur[i])
        grad /= 5.0
        h_new = h_cur - 0.01 * grad
        h_new += (n / 2.0 - np.sum(h_new)) / n
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
        else:
            h_cur = h_new
    return best_h, best_c5

def interpolate_to_n(h, n_from, n_to):
    xp = (np.arange(n_from) + 0.5) * 2.0 / n_from
    xn = (np.arange(n_to) + 0.5) * 2.0 / n_to
    h_new = np.interp(xn, xp, h)
    return project_feasible(h_new, n_to)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_beta, _g_total_budget, _g_t0
    _g_t0 = time.time()
    _g_total_budget = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 0:
            pph = project_feasible(ph, pn)
            best_h = pph.copy()
            best_c5 = compute_c5(pph)
            best_n = pn
    except Exception:
        pass

    if best_h is None:
        best_n = 192
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    n_candidates = sorted(set([best_n, 192, 224, 256, 288]))
    n_candidates = [nn for nn in n_candidates if 96 <= nn <= 320]

    for n in n_candidates:
        if _time_left() < 60:
            break

        if best_n == n:
            h_start = best_h.copy()
        else:
            h_start = interpolate_to_n(best_h, best_n, n)

        # Phase 1: SLSQP staged optimization
        t_stage_start = time.time()
        h_opt = project_feasible(h_start, n)
        stages = [(5.0, 100), (20.0, 200), (80.0, 300), (300.0, 400), (1500.0, 600)]
        for beta, maxit in stages:
            if _time_left() < 20:
                break
            h_opt = slsqp_stage(h_opt, n, beta, maxit)
            c5_tmp = compute_c5(h_opt)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h_opt.copy()
                best_n = n

        # Phase 2: Top-k SLSQP
        if _time_left() > 20:
            h_opt = slsqp_topk(h_opt, n, 300)
            c5_tmp = compute_c5(h_opt)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h_opt.copy()
                best_n = n

        # Phase 3: Subgradient refinement
        if _time_left() > 15:
            h_sg, c5_sg = subgradient_refine(h_opt, n, n_iter=400, lr0=0.015)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                best_n = n

        # Phase 4: Top-lag refinement
        if _time_left() > 10:
            h_tl, c5_tl = top_lag_refinement(best_h, best_n, n_iter=1500)
            if c5_tl < best_c5:
                best_c5 = c5_tl
                best_h = h_tl.copy()

        # Phase 5: Binary + local search
        if _time_left() > 30:
            h_bin = binary_round_with_repair(best_h, best_n)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
            # Binary local search
            if _time_left() > 25:
                h_bls, c5_bls = local_search_binary(best_h, best_n, n_flips=15000, seed=seed + n)
                if c5_bls < best_c5:
                    best_c5 = c5_bls
                    best_h = h_bls.copy()
                    best_n = best_n

        # Phase 6: Coordinate descent on continuous
        if _time_left() > 15:
            h_cd, c5_cd = coordinate_descent(best_h, best_n, n_iter=3000, seed=seed + 100)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = best_n

    # Final high-beta SLSQP polish
    if _time_left() > 10:
        for beta in [5000.0, 50000.0, 200000.0]:
            if _time_left() < 8:
                break
            h_pol = slsqp_stage(best_h, best_n, beta, 800)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = best_n

    # Raw max SLSQP
    if _time_left() > 10:
        h_raw = slsqp_raw(best_h, best_n, 600)
        c5_raw = compute_c5(h_raw)
        if c5_raw < best_c5:
            best_c5 = c5_raw
            best_h = h_raw.copy()

    # Another top-lag refinement
    if _time_left() > 8:
        h_tl2, c5_tl2 = top_lag_refinement(best_h, best_n, n_iter=2000)
        if c5_tl2 < best_c5:
            best_c5 = c5_tl2
            best_h = h_tl2.copy()

    # Binary rounding check
    if _time_left() > 5:
        h_bin2 = binary_round_with_repair(best_h, best_n)
        c5_bin2 = compute_c5(h_bin2)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Final subgradient
    if _time_left() > 8:
        h_sg2, c5_sg2 = subgradient_refine(best_h, best_n, n_iter=300, lr0=0.005)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < best_c5:
            best_c5 = ec5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
