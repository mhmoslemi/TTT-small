        if remaining_time() < 10 or (time.time() - t_start) > time_limit * 0.5:
            break
        try:
            h_new = lbfgsb_optimize(best_h, n, beta, maxiter=150)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Nesterov PGD
    if remaining_time() > 15:
        h_new, c5_new = nesterov_pgd(best_h, n, 500, lr_init=0.05, beta=200)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Pair-flip SA
    if remaining_time() > 10:
        h_new, c5_new = pair_flip_sa(best_h, n, 3000, rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _START_TIME, _BUDGET_S, _N_GLOB
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_h = project_feasible(parent_h, parent_n)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_golden(best_n)
        best_h = project_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. Golden: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 1: Multi-resolution exploration
    n_candidates = [160, 192, 256, 320, 384, 448, 512]
    n_candidates = [n for n in n_candidates if 64 <= n <= 512]

    for n in n_candidates:
        if remaining_time() < 40:
            break
        print("\n--- Exploring n={} ---".format(n))

        inits = []
        h_interp = interp_to(best_h, best_n, n)
        inits.append(project_feasible(h_interp, n))
        for _ in range(2):
            inits.append(project_feasible(make_random_binary(n, rng) if 'make_random_binary' in dir() else make_golden(n), n))
        inits.append(project_feasible(make_thue_morse(n), n))
        inits.append(project_feasible(make_golden(n), n))
        inits.append(project_feasible(make_lfsr_binary(n, rng), n))
        inits.append(make_fourier(n, rng))

        for h0 in inits:
            if remaining_time() < 15:
                break
            h0 = project_feasible(h0, n)
            c5_cur = compute_c5(h0)

            # Quick optimization: L-BFGS-B then pair-flip
            try:
                _N_GLOB = n
                _BETA_GLOB = 100
                h_opt = lbfgsb_optimize(h0, n, 100, maxiter=80)
                c5_opt = compute_c5(h_opt)
                if c5_opt < c5_cur:
                    h0 = h_opt
                    c5_cur = c5_opt
            except Exception:
                pass

            if remaining_time() > 8:
                h_pf, c5_pf = pair_flip_sa(h0, n, 2000, rng)
                if c5_pf < c5_cur:
                    h0 = h_pf
                    c5_cur = c5_pf

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n
                print("  n={} C5={:.6f}".format(n, c5_cur))

    print("\nPhase 1 best: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 2: Intensive refinement on best candidate
    if remaining_time() > 30:
        n = best_n
        h = best_h.copy()
        print("\n--- Phase 2: Intensive refinement n={} ---".format(n))

        # Multiple rounds of L-BFGS-B with increasing beta
        for beta in [100, 300, 500, 800, 1200]:
            if remaining_time() < 15:
                break
            try:
                h_new = lbfgsb_optimize(h, n, beta, maxiter=300)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  L-BFGS beta={}: C5={:.6f}".format(beta, c5_new))
            except Exception:
                pass

        # Nesterov PGD with multiple restarts
        if remaining_time() > 20:
            for lr_try in [0.03, 0.08, 0.12]:
                if remaining_time() < 10:
                    break
                h_retry = h + rng.randn(n) * 0.02
                h_retry = project_feasible(h_retry, n)
                h_new, c5_new = nesterov_pgd(h_retry, n, 800, lr_init=lr_try, beta=300)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  Nesterov lr={}: C5={:.6f}".format(lr_try, c5_new))

        # Long pair-flip SA
        if remaining_time() > 15:
            h_pf, c5_pf = pair_flip_sa(h, n, 20000, rng)
            if c5_pf < best_c5:
                best_c5 = c5_pf
                best_h = h_pf.copy()
                h = h_pf.copy()
                print("  Pair-flip SA: C5={:.6f}".format(c5_pf))

    # Phase 3: Try alternating continuous/discrete refinement
    if remaining_time() > 30:
        n = best_n
        h = best_h.copy()
        print("\n--- Phase 3: Alternating refinement ---".format(n))

        for round_ix in range(5):
            if remaining_time() < 12:
                break

            # Continuous: L-BFGS-B
            try:
                h_new = lbfgsb_optimize(h, n, 500, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass

            # Discrete: pair-flip
            if remaining_time() > 8:
                h_pf, c5_pf = pair_flip_sa(h, n, 5000, rng)
                if c5_pf < best_c5:
                    best_c5 = c5_pf
                    best_h = h_pf.copy()
                    h = h_pf.copy()

            # Small perturbation restart
            if remaining_time() > 10:
                h_pert = h + rng.randn(n) * 0.01
                h_pert = project_feasible(h_pert, n)
                h_pf2, c5_pf2 = pair_flip_sa(h_pert, n, 3000, rng)
                if c5_pf2 < best_c5:
                    best_c5 = c5_pf2
                    best_h = h_pf2.copy()
                    h = h_pf2.copy()

        print("  Phase 3 best: C5={:.6f}".format(best_c5))

    # Phase 4: Final high-precision polish
    if remaining_time() > 10:
        n = best_n
        h = best_h.copy()
        print("\n--- Phase 4: Final polish ---".format(n))

        for beta in [1000, 2000, 3000]:
            if remaining_time() < 5:
                break
            try:
                h_new = lbfgsb_optimize(h, n, beta, maxiter=200)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
            except Exception:
                pass

        # Final pair-flip for binary tightening
        if remaining_time() > 5:
            h_pf, c5_pf = pair_flip_sa(h, n, 5000, rng)
            if c5_pf < best_c5:
                best_c5 = c5_pf
                best_h = h_pf.copy()

    # Final SLSQP polish with very high beta
    if remaining_time() > 8:
        n = best_n
        _N_GLOB = n
        _BETA_GLOB = 2000
        try:
            def final_obj(x):
                val, _ = lse_obj_grad(x, 2000, n)
                return val
            def final_grad(x):
                _, g = lse_obj_grad(x, 2000, n)
                return g
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
            bounds = [(0.0, 1.0)] * n
            res = minimize(final_obj, best_h, jac=final_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Try evaluation function
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < best_c5:
            best_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, best_c5))
    return (best_h, best_c5, best_n)
