    return np.sum(x) - _TGT


def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float).copy()
    target = n // 2
    h = np.where(h >= 0.5, 1.0, 0.0)
    s = int(h.sum())
    if s > target:
        idx = np.where(h == 1)[0]
        h[idx[:s - target]] = 0.0
    elif s < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - s]] = 1.0
    return h


def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src_h, src_n, dst_n):
    if src_n == dst_n:
        return src_h.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    x_new = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(x_new, xp, src_h)


def remaining_time():
    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 4.0)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if 0.25 <= frac < 0.75 else 0.0
    return h


def make_sine_pattern(n, freq, phase):
    t = np.arange(n) / float(n)
    h = 0.5 + 0.48 * np.sin(2.0 * np.pi * freq * t + phase)
    return np.where(h >= 0.5, 1.0, 0.0)


def targeted_local_search(h, n, rng, max_iters=20000):
    """Aggressive targeted local search on binary sequences."""
    target = n // 2
    h = h.copy()
    best_c5 = compute_c5(h)
    best_corr = int(round(np.max(np.correlate(h, 1.0 - h, mode='full'))))
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    if len(ones) == 0 or len(zeros) == 0:
        return h, best_c5
    
    no_improve = 0
    stall_limit = 300
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        
        # Get current max lags
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        
        if no_improve > stall_limit:
            # Periodic random kicks
            for _ in range(5):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
                c5_k = compute_c5(h)
                if c5_k < best_c5:
                    break
                else:
                    h[i], h[j] = h[j], h[i]
            no_improve = 0
            stall_limit = min(stall_limit + 100, 5000)
        
        # Targeted: find contributing positions at max lags
        top_lags = np.argsort(corr)[-3:]
        
        # Build candidate flip pairs from top lags
        candidates = []
        for lag_idx in top_lags:
            offset = int(lag_idx) - (n - 1)
            if abs(offset) == 0:
                continue
            if offset > 0:
                n_pairs = n - offset
                for k in range(offset, n):
                    if h[k] == 1 and h[k - offset] == 0:
                        candidates.append((k, k - offset))
            else:
                aoff = -offset
                n_pairs = n - aoff
                for k in range(n_pairs):
                    if h[k] == 1 and h[k + aoff] == 0:
                        candidates.append((k, k + aoff))
        
        # Also add some random candidates
        for _ in range(min(20, len(ones) * len(zeros))):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            candidates.append((i, j))
        
        improved = False
        rng.shuffle(candidates)
        
        for (i, j) in candidates[:40]:
            if h[i] != 1 or h[j] != 0:
                continue
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_corr = int(round(np.max(np.correlate(h, 1.0 - h, mode='full'))))
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                no_improve = 0
                improved = True
                break
            else:
                h[i], h[j] = h[j], h[i]
        
        if not improved:
            no_improve += 1
            # Simpler random flips when targeted fails
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            c5_r = compute_c5(h)
            if c5_r < best_c5:
                best_c5 = c5_r
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                no_improve = 0
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
    
    return h, best_c5


def multi_lag_local_search(h, n, rng, max_iters=10000):
    """Try to reduce multiple near-max lags simultaneously."""
    target = n // 2
    h = h.copy()
    best_c5 = compute_c5(h)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if remaining_time() < 3:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        # Find all lags within 2 of max
        active_lags = np.where(corr >= max_val - 2)[0]
        
        # For each active lag, find a pair that reduces it
        found = False
        for lag_idx in active_lags:
            offset = int(lag_idx) - (n - 1)
            if offset == 0:
                continue
            if offset > 0:
                n_pairs = n - offset
                # Positions contributing: h[k]=1, h[k-offset]=0
                contrib_ones = [k for k in range(offset, n) if h[k] == 1 and h[k - offset] == 0]
                contrib_zeros = [k - offset for k in range(offset, n) if h[k] == 1 and h[k - offset] == 0]
            else:
                aoff = -offset
                n_pairs = n - aoff
                contrib_ones = [k for k in range(n_pairs) if h[k] == 1 and h[k + aoff] == 0]
                contrib_zeros = [k + aoff for k in range(n_pairs) if h[k] == 1 and h[k + aoff] == 0]
            
            if not contrib_ones:
                continue
            
            i =contrib_ones[rng.randint(len(contrib_ones))]
            j = contrib_zeros[rng.randint(len(contrib_zeros))]
            
            if h[i] == 1 and h[j] == 0:
                h[i], h[j] = h[j], h[i]
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    found = True
                    break
                else:
                    h[i], h[j] = h[j], h[i]
        
        if not found and it % 50 == 0:
            # Random pair flip
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i], h[j] = h[j], h[i]
    
    return h, best_c5


def slsqp_optimize(h0, n, beta=100, maxiter=200):
    global _N, _BETA, _TGT
    _N = n
    _BETA = beta
    _TGT = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum}]
    try:
        res = minimize(obj_smooth, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return h0


def run(seed=42, budget_s=1300, **kwargs):
    global _N, _BETA, _TGT, _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = make_thue_morse(best_n)
        best_h = project_binary(best_h, best_n)
        best_c5 = compute_c5(best_h)
        print("No parent. TM n={}, C5={:.6f}".format(best_n, best_c5))
    
    # Phase 1: Try diverse binary constructions at multiple resolutions
    n_list = [128, 160, 192, 256]
    
    print("\n=== Phase 1: Binary constructions ===")
    for n in n_list:
        if remaining_time() < 30:
            break
        
        inits = []
        
        # Thue-Morse
        h_tm = make_thue_morse(n)
        inits.append(project_binary(h_tm, n))
        
        # Golden ratio
        h_gr = make_golden_ratio(n)
        inits.append(project_binary(h_gr, n))
        
        # Interpolated parent (binarized)
        if best_n != n:
            h_ip = interp_to(best_h, best_n, n)
        else:
            h_ip = best_h.copy()
        h_ip_b = project_binary(h_ip, n)
        inits.append(h_ip_b)
        
        # Sine patterns
        for freq in [2, 3, 4, 5]:
            h_sin = make_sine_pattern(n, freq, rng.rand() * 2 * np.pi)
            inits.append(project_binary(h_sin, n))
        
        # Random
        h_rand = np.zeros(n)
        idx = rng.permutation(n)
        h_rand[idx[:n // 2]] = 1.0
        inits.append(h_rand)
        
        for h_init in inits:
            if remaining_time() < 10:
                break
            c5_init = compute_c5(h_init)
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = h_init.copy()
                best_n = n
                print("  Init improved: n={}, C5={:.6f}".format(n, best_c5))
            
            # Quick local search on this init
            h_ls, c5_ls = targeted_local_search(h_init, n, rng, max_iters=3000)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls.copy()
                best_n = n
                print("  LS improved: n={}, C5={:.6f}".format(n, c5_ls))
    
    # Phase 2: Deep local search on best solution
    print("\n=== Phase 2: Deep local search n={} ===".format(best_n))
    n = best_n
    h = best_h.copy()
    h = project_binary(h, n)
    
    # Multiple passes of targeted local search
    for passage in range(3):
        if remaining_time() < 20:
            break
        h, c5_ls = targeted_local_search(h, n, rng, max_iters=15000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h.copy()
            print("  Passage {}: C5={:.6f}".format(passage, c5_ls))
        else:
            break
    
    # Phase 3: Multi-lag search
    if remaining_time() > 20:
        print("\n=== Phase 3: Multi-lag search ===")
        h = project_binary(best_h, best_n)
        h, c5_ml = multi_lag_local_search(h, best_n, rng, max_iters=10000)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h.copy()
            print("  Multi-lag: C5={:.6f}".format(c5_ml))
    
    # Phase 4: Try other resolutions with the best pattern
    if remaining_time() > 30:
        print("\n=== Phase 4: Resolution sweep ===")
        for n2 in [96, 112, 128, 144, 160, 176, 192, 208, 224, 256, 288, 320]:
            if remaining_time() < 15:
                break
            if n2 == best_n:
                continue
            
            h2 = interp_to(best_h, best_n, n2)
            h2 = project_binary(h2, n2)
            c5_2 = compute_c5(h2)
            
            if c5_2 < best_c5 + 0.01:
                h2, c5_2 = targeted_local_search(h2, n2, rng, max_iters=5000)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h2.copy()
                    best_n = n2
                    print("  n={} improved: C5={:.6f}".format(n2, c5_2))
    
    # Phase 5: SLSQP continuous optimization (exploit non-binary)
    if remaining_time() > 25:
        print("\n=== Phase 5: SLSQP polish ===")
        n = best_n
        h_cont = best_h.copy()
        for beta_val in [50, 200, 800]:
            if remaining_time() < 10:
                break
            h_new = slsqp_optimize(h_cont, n, beta=beta_val, maxiter=300)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n
                print("  SLSQP beta={}: C5={:.6f}".format(beta_val, c5_new))
                h_cont = h_new.copy()
        
        # Also try binarizing the SLSQP result and doing local search
        h_bin = project_binary(best_h, best_n)
        h_bin, c5_bin = targeted_local_search(h_bin, best_n, rng, max_iters=5000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Binarized LS: C5={:.6f}".format(c5_bin))
    
    # Phase 6: Final intensive local search
    if remaining_time() > 15:
        print("\n=== Phase 6: Final local search ===")
        n = best_n
        h = project_binary(best_h, n)
        
        # Multiple short targeted searches with different randomization
        for _ in range(5):
            if remaining_time() < 5:
                break
            h_tmp, c5_tmp = targeted_local_search(h, n, rng, max_iters=5000)
            if c5_tmp < best_c5:
                best_c5 = c5_tmp
                best_h = h_tmp.copy()
                h = h_tmp.copy()
            else:
                # Try a perturbed version
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = h[j], h[i]
                h_tmp, c5_tmp = targeted_local_search(h, n, rng, max_iters=3000)
                if c5_tmp < best_c5:
                    best_c5 = c5_tmp
                    best_h = h_tmp.copy()
                    h = h_tmp.copy()
    
    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    
    # Verify sum constraint (may need slight adjustment for non-binary)
    s = np.sum(best_h)
    if abs(s - best_n / 2.0) > 1e-6:
        best_h = project_feasible(best_h, best_n)
        final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
