    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def gaussian_smooth(h, sigma=1.2):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 3:
        return h.copy()
    length = min(5, n)
    if length % 2 == 0:
        length -= 1
    length = max(3, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * float(sigma) * float(sigma)))
    w /= w.sum()
    return np.convolve(h, w, mode="same")


def local_block_perturb(h0, n, time_limit, seed, step=0.02, max_trials=6000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h.copy()
    cur_c = c
    temp = 0.0015
    min_temp = 1e-7
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, max(temp * 1.12, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)


def diverse_starts(n, rng, parent=None):
    starts = []
    if parent is not None:
        if len(parent) == n:
            starts.append(parent.copy())
        else:
            starts.append(interpolate_to_n(parent, n))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    for freq_set in [(1, 2, 3), (2, 3, 5), (1, 3, 4)]:
        h = np.full(n, 0.5, dtype=float)
        for f in freq_set:
            amp = rng.uniform(0.05, 0.3)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            h += amp * np.cos(f * np.pi * x + ph)
        starts.append(project_feasible(h, n))
    if n % 4 == 0:
        h = np.tile(np.array([1.0, 1.0, 0.0, 0.0]), n // 4)
        starts.append(project_feasible(h, n))
    if n > 0:
        h = np.full(n, 0.5, dtype=float)
        h[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(project_feasible(h, n))
    out = [project_feasible(s, n) for s in starts]
    return out[:5]


def improve_at_n(start_h, n, time_limit, seed, smooth_betas=None, active_counts=None):
    n = int(n)
    h = project_feasible(start_h, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    if smooth_betas is None:
        smooth_betas = (5, 15, 50, 120, 250)
    for beta in smooth_betas:
        if time.time() > time_limit - 15.0:
            break
        h2, c2 = opt_smooth(h, n, beta, maxiter=40)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
            h = h2
        else:
            h = best_h

    if active_counts is None:
        active_counts = (8, 15, 25, 40, 65, 100)
    if time.time() < time_limit - 10.0:
        h2, c2 = refine_active(best_h, n, time_limit - 10.0,
                               lag_counts=active_counts, maxiter=45)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    if time.time() < time_limit - 5.0 and 2 <= n <= 350:
        h2, c2 = local_block_perturb(best_h, n, time_limit - 2.0, seed, step=0.018)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(120.0, budget_s - 35.0)

    parent = get_parent_values()
    if parent is not None:
        pn = int(parent.size)
        raw_c = None
        if (parent.size > 0
                and float(np.min(parent)) >= -1e-12
                and float(np.max(parent)) <= 1.0 + 1e-12
                and abs(float(np.sum(parent)) - 0.5 * parent.size) <= 1e-8):
            raw_h = parent.copy()
            raw_c = compute_c5(raw_h)
        proj = project_feasible(parent, pn)
        proj_c = compute_c5(proj)
        if raw_c is not None and raw_c <= proj_c + 1e-13:
            best_h = raw_h.astype(float).copy()
            best_c = raw_c
        else:
            best_h = proj
            best_c = proj_c
        best_n = pn
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    resolutions = []
    if best_n in (128, 160):
        resolutions = [best_n, 192, 224, 256, 288, 320]
    elif best_n <= 200:
        resolutions = [best_n, 224, 256, 288, 320]
    else:
        resolutions = [best_n, min(best_n + 32, 320), 256, 320]
    resolutions = sorted(set(resolutions))
    resolutions = [r for r in resolutions if 32 <= r <= 360]
    if best_n not in resolutions:
        resolutions.insert(0, best_n)

    total_opt_time = hard_end - start_time - 60.0
    total_opt_time = max(100.0, total_opt_time)
    n_res = len(resolutions)
    weights = []
    for i, r in enumerate(resolutions):
        if r == best_n:
            weights.append(4.0)
        elif abs(r - best_n) <= 48:
            weights.append(3.0)
        else:
            weights.append(1.5)
    wsum = sum(weights)
    slots = []
    rem = total_opt_time
    for i in range(n_res):
        if i == n_res - 1:
            sl = rem
        else:
            sl = total_opt_time * (weights[i] / wsum)
            sl = max(40.0, min(sl, rem))
        slots.append(sl)
        rem -= sl

    for i, (res_n, slot) in enumerate(zip(resolutions, slots)):
        res_n = int(res_n)
        slot_end = start_time + total_opt_time - sum(slots[i + 1:]) - 5.0
        if slot_end <= time.time() + 10.0:
            break
        n_inits = 3 if res_n == best_n else 2
        inits = diverse_starts(res_n, rng, parent=best_h if best_h.size != res_n else None)
        if res_n == best_n and parent is not None:
            inits.insert(0, project_feasible(
                parent if parent.size == res_n else interpolate_to_n(parent, res_n), res_n))
        inits = inits[:n_inits]
        per_time = (slot_end - time.time()) / float(max(1, len(inits)))
        per_time = min(per_time, 120.0)
        per_time = max(25.0, per_time)

        smooth_betas = (5, 15, 50, 120) if res_n <= 200 else (5, 15, 50)
        active_counts = (8, 15, 25, 40, 65, 100) if res_n <= 200 else (8, 15, 25, 40, 65)

        for j, init_h in enumerate(inits):
            if time.time() >= slot_end:
                break
            t_end = min(slot_end, time.time() + per_time)
            hh, cc = improve_at_n(init_h, res_n, t_end, seed + 1000 * res_n + j,
                                  smooth_betas=smooth_betas, active_counts=active_counts)
            if cc < best_c - 1e-12:
                best_h = hh.copy()
                best_c = cc
                best_n = res_n

    if time.time() < hard_end - 20.0 and 2 <= best_n <= 350:
        fin_betas = (100, 200, 400)
        for beta in fin_betas:
            if time.time() > hard_end - 22.0:
                break
            h2, c2 = opt_smooth(best_h, best_n, beta, maxiter=50)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2

    if time.time() < hard_end - 30.0 and 2 <= best_n <= 300:
        fin_counts = (20, 40, 80, 150)
        if best_n <= 180:
            fin_counts = (20, 40, 80, 120, 180)
        h2, c2 = refine_active(best_h, best_n, hard_end - 25.0,
                               lag_counts=fin_counts, maxiter=70)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    if time.time() < hard_end - 8.0 and 2 <= best_n <= 350:
        h2, c2 = local_block_perturb(best_h, best_n, hard_end - 3.0, seed + 77777, step=0.012, max_trials=8000)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    if time.time() < hard_end - 5.0 and 2 <= best_n <= 250:
        for beta in (200, 500):
            if time.time() > hard_end - 6.0:
                break
            h2, c2 = opt_smooth(best_h, best_n, beta, maxiter=30)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
