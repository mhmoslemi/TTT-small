def compute_f(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def exact_c5(h):
    f = compute_f(h)
    return float(np.max(f))


def lse_value_from_f(f, beta):
    cmax = float(np.max(f))
    z = beta * (f - cmax)
    z = np.minimum(z, 0.0)
    z = np.maximum(z, -745.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        return cmax
    return cmax + np.log(s) / float(beta)


def make_softmax_V(f, beta):
    n = (len(f) - 1) // 2
    dx = 2.0 / float(n)
    cmax = float(np.max(f))
    z = beta * (f - cmax)
    z = np.minimum(z, 0.0)
    z = np.maximum(z, -745.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.zeros(len(f), dtype=float)
        w[int(np.argmax(f))] = 1.0
    else:
        w = e / s
    return w * dx


def grad_from_V(V, h):
    h = np.asarray(h, dtype=float)
    n = h.size
    off = n - 1
    c1 = np.convolve(V, 1.0 - h, mode='full')
    c2 = np.convolve(V[::-1], h, mode='full')
    return c1[off:off + n] - c2[off:off + n]


def lse_value_grad(h, beta):
    h = np.asarray(h, dtype=float)
    f = compute_f(h)
    cmax = float(np.max(f))
    z = beta * (f - cmax)
    z = np.minimum(z, 0.0)
    z = np.maximum(z, -745.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0:
        w = np.zeros(len(f), dtype=float)
        w[int(np.argmax(f))] = 1.0
        val = cmax
    else:
        w = e / s
        val = cmax + np.log(s) / float(beta)
    n = h.size
    dx = 2.0 / float(n)
    V = w * dx
    g = grad_from_V(V, h)
    return val, g


def active_gradient(h, frac=0.10):
    f = compute_f(h)
    n = h.size
    dx = 2.0 / float(n)
    cmax = float(np.max(f))
    tol = max(1e-10, frac * cmax)
    mask = f >= (cmax - tol)
    cnt = int(np.count_nonzero(mask))
    maxc = max(1, len(f) // 8)
    if cnt > maxc:
        idx = np.argsort(f)[-maxc:]
        mask = np.zeros(len(f), dtype=bool)
        for k in idx:
            mask[int(k)] = True
        cnt = maxc
    if cnt <= 0:
        mask = np.zeros(len(f), dtype=bool)
        mask[int(np.argmax(f))] = True
        cnt = 1
    V = np.zeros(len(f), dtype=float)
    V[mask] = dx / float(cnt)
    g = grad_from_V(V, h)
    return g, f, V

def project_sum_box(v, target):

    n = v.size
    target = float(target)
    if n == 0:
        return v.copy()
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = float(np.sum(h))
        if s > target:
            lo = mid
        else:
            hi = mid

    bounds = bounds_for(m)
    cons = [{'type': 'eq', 'fun': sym_eq, 'jac': sym_eq_jac}]

    for beta in betas:
        if time.time() > max(deadline - 10.0, 0.0):
            break
        G_N = n_full
        G_BETA = float(beta)
        G_PEN = 0.0
        _reset_cache()

        mi = int(max(20, int(maxiter) * (130.0 / float(n_full))))
        try:
            res = minimize(
                sym_obj_noargs,
                x,
                jac=sym_grad_noargs,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': mi, 'ftol': 1e-15, 'iprint': 0}
            )
        except Exception:
            res = None

        if res is not None and hasattr(res, 'x') and res.x is not None:
            rx = np.asarray(res.x, dtype=float)
            if rx.size == m and np.all(np.isfinite(rx)):
                x2 = project_sum_box(rx, 0.25 * float(n_full))
                G_N = n_full
                h2 = expand_sym(x2)
                c2 = exact_c5(h2)
                if c2 < best_c - 1e-15:
                    best_h = h2.copy()
                    best_c = c2
                status = int(getattr(res, 'status', 2))
                if status in (0, 1):
                    x = x2
                    current_c = c2
                elif c2 < current_c + 1e-14:
                    x = x2
                    current_c = c2
    return best_h, best_c


def pair_search(h, n, deadline, cap, candidates, rng):
    n = int(n)
    if n < 2 or time.time() > deadline:
        hp = project_sum_box(h, 0.5 * float(n))
        return hp, exact_c5(hp)

    h = project_sum_box(h, 0.5 * float(n))
    best_h = h.copy()
    best_c = exact_c5(h)

    betas = (None, 80.0, 300.0)
    for beta in betas:
        if time.time() > deadline:
            break
        improved = True
        sweeps = 0
        max_sweeps = 10

        while improved and sweeps < max_sweeps and time.time() < deadline:
            improved = False
            if beta is None:
                grad = active_gradient(h, frac=0.10)[0]
            else:
                grad = softmax_gradient(h, beta)[0]

            if not np.all(np.isfinite(grad)):
                break
            if float(np.max(np.abs(grad))) < 1e-13:
                break

            order = np.argsort(grad)
            K = max(8, n // 4)
            high = order[-K:]
            low = order[:K]

            high_elig = []
            for idx in high:
                if h[int(idx)] > 1e-4:
                    high_elig.append(int(idx))
            low_elig = []
            for idx in low:
                if 1.0 - h[int(idx)] > 1e-4:
                    low_elig.append(int(idx))

            if len(high_elig) == 0:
                for idx in high:
                    high_elig.append(int(idx))
            if len(low_elig) == 0:
                for idx in low:
                    low_elig.append(int(idx))

            P = int(candidates)
            pairs = []
            det = min(len(high_elig), len(low_elig), max(8, P // 4))
            for r in range(det):
                i = high_elig[-1 - r]
                j = low_elig[r]
                if i != j:
                    pairs.append((i, j))

            random_count = P - det
            for _ in range(random_count):
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if int(rng.randint(2)) == 0 and len(high_elig) > 0:
                    i = int(high_elig[int(rng.randint(len(high_elig)))])
                if int(rng.randint(2)) == 0 and len(low_elig) > 0:
                    j = int(low_elig[int(rng.randint(len(low_elig)))])
                if i != j:
                    pairs.append((i, j))

            if len(pairs) > 1:
                perm = rng.permutation(len(pairs))
                newpairs = []
                for pidx in perm:
                    newpairs.append(pairs[int(pidx)])
                pairs = newpairs

            for idx in range(len(pairs)):
                if time.time() > deadline:
                    break
                i, j = pairs[idx]
                if grad[i] > grad[j]:
                    a = i
                    b = j
                else:
                    a = j
                    b = i

                big = min(float(h[a]), 1.0 - float(h[b]), float(cap))
                big = big * (1.0 - 1e-12) - 1e-12
                if big <= 1e-8:
                    continue

                old_a = float(h[a])
                old_b = float(h[b])
                best_val = best_c
                best_m = 0.0

                mvals = [0.25 * big, 0.5 * big, 0.75 * big, big,
                         0.125 * big, 0.0625 * big]
                for m in mvals:
                    if m <= 0.0:
                        continue
                    h[a] = old_a - m
                    h[b] = old_b + m
                    if h[a] < -1e-12 or h[b] > 1.0 + 1e-12:
                        h[a] = old_a
                        h[b] = old_b
                        continue
                    if h[a] < 0.0:
                        h[a] = 0.0
                    if h[b] > 1.0:
                        h[b] = 1.0
                    cval = exact_c5(h)
                    h[a] = old_a
                    h[b] = old_b
                    if cval < best_val - 1e-14:
                        best_val = cval
                        best_m = m

                if best_m > 0.0:
                    old_a = float(h[a])
                    old_b = float(h[b])
                    for m in [0.5 * best_m, 0.75 * best_m, best_m,
                              1.1 * best_m, 1.25 * best_m, 1.5 * best_m,
                              2.0 * best_m]:
                        if m <= 0.0 or m > big + 1e-10:
                            continue
                        h[a] = old_a - m
                        h[b] = old_b + m
                        if h[a] < -1e-12 or h[b] > 1.0 + 1e-12:
                            h[a] = old_a
                            h[b] = old_b
                            continue
                        if h[a] < 0.0:
                            h[a] = 0.0
                        if h[b] > 1.0:
                            h[b] = 1.0
                        cval = exact_c5(h)
                        h[a] = old_a
                        h[b] = old_b
                        if cval < best_val - 1e-14:
                            best_val = cval
                            best_m = m

                    h[a] = old_a - best_m
                    h[b] = old_b + best_m
                    if h[a] < 0.0:
                        h[a] = 0.0
                    if h[b] > 1.0:
                        h[b] = 1.0

                    best_c = best_val
                    best_h = h.copy()
                    improved = True
                    break

            sweeps += 1
            if not improved:
                break

        hp = project_sum_box(h, 0.5 * float(n))
        cp = exact_c5(hp)
        if cp < best_c - 1e-15:
            best_c = cp
            best_h = hp.copy()
        if cp <= best_c + 1e-9:
            h = hp
        else:
            h = best_h.copy()

    bp = project_sum_box(best_h, 0.5 * float(n))
    cp = exact_c5(bp)
    return bp, cp


def make_thue_morse(n):
    h = np.zeros(int(n), dtype=float)
    for i in range(int(n)):
        x = i
        p = 0
        while x:
            p ^= 1
            x &= (x - 1)
        h[i] = float(p)
    return h


def make_wiedemann(n):
    n = int(n)
    x = np.arange(n, dtype=int)
    tri = (x * (x + 1) // 2) % n
    h = np.zeros(n, dtype=float)
    h[tri < (n // 2)] = 1.0
    return h


def make_sine(n, freq, phase):
    n = int(n)
    x = np.arange(n, dtype=float)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * float(freq) * x / float(n) + float(phase))
    return np.clip(h, 0.0, 1.0)


def generate_start(n, style, rng):
    n = int(n)
    if style == 0:
        h = rng.rand(n)
    elif style == 1:
        h = np.zeros(n, dtype=float)
        if n >= 2:
            k = n // 2
            idx = rng.choice(n, k, replace=False)
            for q in idx:
                h[int(q)] = 1.0
    elif style == 2:
        h = make_thue_morse(n)
    elif style == 3:
        h = 1.0 - make_thue_morse(n)
    elif style == 4:
        h = make_wiedemann(n)
    elif style == 5:
        h = make_sine(n, 5.0, 0.0)
    elif style == 6:
        h = make_sine(n, 7.0, 1.1)
    elif style == 7:
        h = make_sine(n, 11.0, 2.3)
    else:
        h = rng.rand(n)
    return project_sum_box(h, 0.5 * float(n))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + float(budget_s)
    rng = np.random.RandomState(seed)

    parent = None
    try:
        parent = initial_h_values
    except Exception:
        parent = None

    base_h = None
    if parent is not None:
        try:
            hpar = np.asarray(parent, dtype=float).reshape(-1)
            if hpar.size > 0:
                base_h = project_sum_box(hpar, 0.5 * float(hpar.size))
        except Exception:
            base_h = None

    if base_h is None:
        base_h = generate_start(160, 2, rng)

    best_h = base_h.copy()
    best_n = int(best_h.size)
    best_c = exact_c5(best_h)
    if not np.isfinite(best_c):
        best_c = 1.0

    # Phase 1: refine the supplied construction.
    parent_deadline = min(deadline, start_time + 0.22 * float(budget_s))

    if time.time() < max(parent_deadline - 5.0, 0.0):
        d = min(parent_deadline, start_time + 95.0)
        hh, cc = lbfgs_refine(best_h, best_n, d,
                              (10.0, 50.0, 200.0, 800.0), 120, 30.0)
        if hh is not None and cc < best_c:
            best_h = hh
            best_n = int(hh.size)
            best_c = cc

    if time.time() < max(parent_deadline - 8.0, 0.0):
        d = min(parent_deadline, start_time + 180.0)
        hh, cc = slsqp_refine(best_h, best_n, d,
                              (200.0, 800.0), 170)
        if hh is not None and cc < best_c:
            best_h = hh
            best_n = int(hh.size)
            best_c = cc

    if time.time() < max(parent_deadline - 3.0, 0.0):
        hh, cc = pair_search(best_h, best_n, parent_deadline, 0.05, 320, rng)
        if cc < best_c:
            best_h = hh
            best_n = int(best_h.size)
            best_c = cc

    # Phase 2: diverse global starts at a practical resolution.
    global_deadline = min(deadline, start_time + 0.48 * float(budget_s))
    target_n = best_n if 60 <= best_n <= 288 else 160
    if target_n % 2 != 0:
        target_n += 1
    if target_n > 320:
        target_n = 288

    if time.time() < max(global_deadline - 30.0, 0.0):
        styles = (2, 4, 1, 0, 5, 6, 3, 7)
        for style in styles:
            if time.time() + 25.0 > global_deadline:
                break
            d = min(global_deadline, time.time() + 42.0)
            hstart = generate_start(target_n, style, rng)
            hh, cc = lbfgs_refine(hstart, target_n, d,
                                  (20.0, 100.0, 400.0), 80, 25.0)
            if hh is not None and cc < best_c:
                best_h = hh
                best_n = int(hh.size)
                best_c = cc
                if time.time() + 20.0 < global_deadline:
                    hh2, cc2 = slsqp_refine(best_h, best_n,
                                            min(global_deadline, time.time() + 25.0),
                                            (300.0,), 80)
                    if hh2 is not None and cc2 < best_c:
                        best_h = hh2
                        best_n = int(hh2.size)
                        best_c = cc2

    # Phase 3: multiresolution refinement from the current best.
    multi_deadline = min(deadline, start_time + 0.72 * float(budget_s))
    desired = [best_n, 128, 144, 160, 176, 192, 208, 224, 256, 288, 320]
    ns = []
    for nn in desired:
        nn = int(nn)
        if 60 <= nn <= 320 and (nn % 2) == 0:
            duplicate = False
            for ee in ns:
                if ee == nn:
                    duplicate = True
                    break
            if not duplicate:
                ns.append(nn)

    keyed = []
    for nn in ns:
        keyed.append((abs(nn - best_n), nn))
    keyed.sort()
    ns = []
    for kk in keyed:
        ns.append(kk[1])

    for nn in ns:
        if time.time() + 45.0 > multi_deadline:
            break
        d = min(multi_deadline, time.time() + 75.0)
        h0 = interpolate_to_n(best_h, nn)
        h0 = project_sum_box(h0, 0.5 * float(nn))
        hh, cc = lbfgs_refine(h0, nn, d,
                              (20.0, 100.0, 400.0), 70, 25.0)
        if hh is not None and cc < best_c:
            best_h = hh
            best_n = int(hh.size)
            best_c = cc
            if nn <= 272 and time.time() + 20.0 < d:
                hh2, cc2 = slsqp_refine(best_h, best_n,
                                        min(d, time.time() + 25.0),
                                        (300.0,), 70)
                if hh2 is not None and cc2 < best_c:
                    best_h = hh2
                    best_n = int(hh2.size)
                    best_c = cc2
            if time.time() + 18.0 < d:
                hh3, cc3 = pair_search(best_h, best_n, min(d, time.time() + 25.0),
                                       0.02, 160, rng)
                if cc3 < best_c:
                    best_h = hh3
                    best_n = int(hh3.size)
                    best_c = cc3

    # Optional symmetric polish at the current resolution.
    if best_n % 2 == 0 and best_n >= 64 and time.time() < deadline - 60.0:
        xhalf = mirror_project_full(best_h)
        d = min(deadline, time.time() + 110.0)
        hh, cc = lbfgs_sym_refine(xhalf, best_n, d,
                                  (20.0, 100.0, 400.0), 100, 25.0)
        if hh is not None and cc < best_c:
            best_h = hh
            best_n = int(hh.size)
            best_c = cc
            if time.time() < deadline - 50.0:
                x2 = mirror_project_full(best_h)
                hh2, cc2 = slsqp_sym_refine(x2, best_n, min(deadline, time.time() + 50.0),
                                            (400.0,), 80)
                if hh2 is not None and cc2 < best_c:
                    best_h = hh2
                    best_n = int(hh2.size)
                    best_c = cc2

    # Phase 4: final high-resolution polishing.
    if time.time() < max(deadline - 25.0, 0.0):
        maxit = 220 if best_n <= 200 else 100
        d = min(deadline - 35.0, time.time() + 170.0)
        hh, cc = slsqp_refine(best_h, best_n, d,
                              (500.0, 2000.0), maxit)
        if hh is not None and cc < best_c:
            best_h = hh
            best_n = int(hh.size)
            best_c = cc

    if time.time() < max(deadline - 30.0, 0.0):
        d = min(deadline - 20.0, time.time() + 45.0)
        hh, cc = lbfgs_refine(best_h, best_n, d,
                              (800.0, 3000.0), 60, 5.0)
        if hh is not None and cc < best_c:
            best_h = hh
            best_n = int(hh.size)
            best_c = cc

    if time.time() < max(deadline - 5.0, 0.0):
        hh, cc = pair_search(best_h, best_n,
                              min(deadline - 2.0, time.time() + 160.0),
                              0.02, 520, rng)
        if cc < best_c:
            best_h = hh
            best_n = int(hh.size)
            best_c = cc

    best_h = project_sum_box(np.asarray(best_h, dtype=float),
                             0.5 * float(best_n))
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c = float(exact_c5(best_h))
    return best_h, best_c, int(best_n)
