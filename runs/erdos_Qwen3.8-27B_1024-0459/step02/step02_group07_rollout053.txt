def smooth_max_obj(x, delta=0.005):
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr / delta
    max_scaled = np.max(scaled)
    shifted = scaled - max_scaled
    logsum = max_scaled + np.log(np.sum(np.exp(shifted)))
    return logsum * delta


def smooth_max_grad(x, delta=0.005):

    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr / delta
    max_scaled = np.max(scaled)
    shifted = scaled - max_scaled
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    active_idx = np.where(weights > 1e-10)[0]
    ells = (active_idx - (n - 1)).astype(int)
    g = np.zeros(n)
    for e in ells:

        js2 = max(0, -e)
        je2 = min(n, n - e)

        return smooth_max_obj(x, delta)
    def obj_grad(x):
        return smooth_max_grad(x, delta)

        return np.sum(x) - len(x) / 2.0
    def con_jac(x):
        return np.ones(len(x))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

def subgradient_step(h, n, alpha, n_active=10, rng=None):

    thresh = np.sort(corr)[-n_active][0]
    active = np.where(corr >= thresh - 1e-10)[0]
    ells = (active - (n - 1)).astype(int)
    grad = np.zeros(n)
    for e in ells:
        js1 = max(0, e)
        je1 = min(n, n + e)
        if js1 < je1:
            grad[js1:je1] += (1.0 - h[js1 - e:je1 - e])
        js2 = max(0, -e)
        je2 = min(n, n - e)
        if js2 < je2:
            grad[js2:je2] += (-h[js2 + e:je2 + e])
    grad /= max(len(ells), 1)
    noise = rng.randn(n) * alpha * 0.3 if rng is not None else 0
    h_new = h - alpha * grad + noise
    h_new = np.clip(h_new, 0.0, 1.0)
    return make_feasible(h_new, n)

            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n

    # Phase 3: Subgradient search with restarts
    if remaining() > 80:
        h_curr = best_h.copy()
        n_curr = best_n
        alpha = 0.008
        sg_start = time.time()
        best_seen = best_c5
        gap_count = 0
        while remaining() > 50 and time.time() - sg_start < remaining() - 60:
            h_curr = subgradient_step(h_curr, n_curr, alpha, n_active=8, rng=rng)
            alpha *= 0.997
            step_num = int((time.time() - sg_start) * 2)
            if step_num % 25 < 2 and remaining() > 60:
                h_cd, c5_cd = coord_descent(h_curr, n_curr, max_passes=20, time_limit=6)
                if c5_cd < best_c5 - 1e-12:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n_curr
                    h_curr = best_h.copy()
                    gap_count = 0
                    alpha = 0.008
                else:
                    gap_count += 1
                    if gap_count > 8:
                        h_curr = smooth_perturb(best_h, best_n, rng, scale=0.04)
                        gap_count = 0
                        alpha = 0.012

    # Phase 4: Multi-start at best resolution
    if remaining() > 60:
        n_try = best_n
        for attempt in range(3):
            if remaining() < 50:
                break
            if attempt == 0:
                h_init = best_h.copy()
            elif attempt == 1:
                h_init = smooth_perturb(best_h, n_try, rng, scale=0.03)
            else:
                h_init = random_init(n_try, rng)

            tl_cd = min(50, remaining() - 70)
            if tl_cd < 10:
                continue
            h_cd, c5_cd = coord_descent(h_init, n_try, max_passes=100, time_limit=tl_cd)

            if remaining() > 60:
                h_slq, c5_slq = slsqp_polish(h_cd, n_try, max_iter=1500, delta=0.003)
                if h_slq is not None and c5_slq < c5_cd:
                    h_cd = h_slq
                    c5_cd = c5_slq

            if c5_cd < best_c5 - 1e-12:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_try

    # Phase 5: Final polish at multiple deltas
    if remaining() > 40:
        for d in [0.003, 0.002, 0.001]:
            if remaining() < 35:
                break
            h_fin, c5_fin = slsqp_polish(best_h, best_n, max_iter=3000, delta=d)
            if h_fin is not None and c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Phase 6: Final exact SLSQP
    if remaining() > 35:
        h_fin, c5_fin = slsqp_polish_exact(best_h, best_n, max_iter=5000)
        if h_fin is not None and c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 7: Last coordinate descent
    if remaining() > 20:
        tl = max(5, remaining() - 10)
        h_cd, c5_cd = coord_descent(best_h, best_n, max_passes=100, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
