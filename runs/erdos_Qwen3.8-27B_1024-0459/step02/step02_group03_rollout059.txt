    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Bisection on lambda for: sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.exp(shifted).sum()
    val = m + np.log(total) / beta
    return val, shifted / beta  # return value and weight log

def project_feasible(h, n):
    target = n / 2.0
    # Project onto {0 <= h <= 1, sum(h) = target}
    # Bisection on lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.maximum(result, 1.0 - result))
        result[idx] += diff
        result = np.clip(result, 0.0, 1.0)
    return result

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def compute_grad_smooth_max(h, corr, beta):
    n = len(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-13:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < L < n:
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        elif -n < L < 0:
            Lp = -L
            grad[:n-Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n-Lp]
    dx = 2.0 / n
    return dx * grad

def projected_gradient_step(h, n, beta, alpha):

    m = np.max(corr)
    f_smooth = (m + np.log(np.exp(np.clip(beta*(corr-m), -500, 0)).sum()) / beta) * (2.0/n)
    grad = compute_grad_smooth_max(h, corr, beta)
    
    # Backtracking line search
    for _ in range(50):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, n)
        corr_new = get_corr(h_new)
        m_new = np.max(corr_new)
        f_new = (m_new + np.log(np.exp(np.clip(beta*(corr_new-m_new), -500, 0)).sum()) / beta) * (2.0/n)
        if f_new <= f_smooth + 1e-16:

def binary_round(h, n):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    n_want = n // 2
    if n_ones > n_want:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - n_want):
            h_bin[ones_idx[order[i]]] = 0
    elif n_ones < n_want:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(n_want - n_ones):
            h_bin[zeros_idx[order[i]]] = 1
    return h_bin

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_obj(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    logsum = np.log(np.exp(shifted).sum())
    return (m + logsum / beta)

def smooth_max_grad(h, corr, beta, n):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)
    exps = np.exp(shifted)
    wsum = exps.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exps / wsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < L < n:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        elif -n < L < 0:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return grad

def project_feasible(h, n):
    target = n / 2.0
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_out)
    if abs(diff) > 1e-14:
        # Find the element closest to boundary
        margins = np.minimum(h_out, 1.0 - h_out)
        idx = np.argmax(margins)
        h_out[idx] += diff
        h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def pg_optimize(h0, n, beta_schedule, iters_per_beta, base_lr, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for beta, n_iters in beta_schedule:
        alpha = base_lr
        for it in range(n_iters):
            if time.time() > time_limit:
                break
            corr = get_corr(h)
            f = smooth_max_obj(corr, beta) * (2.0 / n)
            grad = smooth_max_grad(h, corr, beta, n) * (2.0 / n)
            
            # Backtracking line search
            acceptable = False
            for _ in range(40):
                h_new = h - alpha * grad
                h_new = project_feasible(h_new, n)
                if np.allclose(h_new, h, atol=1e-15):
                    break
                corr_new = get_corr(h_new)
                f_new = smooth_max_obj(corr_new, beta) * (2.0 / n)
                if f_new <= f + 1e-16:
                    h = h_new
                    c5_val = compute_c5(h)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h.copy()
                    acceptable = True
                    break
                alpha *= 0.5
            if not acceptable:
                # Stagnation - try smaller alpha or break
                if alpha < 1e-10:
                    break
            # Adaptive alpha
            if np.linalg.norm(grad) < 1e-10:
                break
    return best_h, best_c5

def binary_round(h, n):
    h_bin = np.zeros(n)
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[n // 2 - 1] if n // 2 > 0 else 0.5
    h_bin = (h >= threshold).astype(float)
    # Fix count
    n_ones = int(h_bin.sum())
    n_want = n // 2
    if n_ones != n_want:
        h_bin = (h > 0.5).astype(float)
        n_ones = int(h_bin.sum())
        if n_ones > n_want:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(h[ones])
            for i in range(n_ones - n_want):
                h_bin[ones[order[i]]] = 0
        elif n_ones < n_want:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros])
            for i in range(n_want - n_ones):
                h_bin[zeros[order[i]]] = 1
    return h_bin

def binary_local_search(h_bin, n, rng, max_rounds, time_limit):
    best_c5 = compute_c5(h_bin)
    h = h_bin.copy()
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = min(30, len(ones) * len(zeros))
        for _ in range(n_tries):
            i1 = rng.randint(len(ones))
            i0 = rng.randint(len(zeros))
            p1 = ones[i1]
            p0 = zeros[i0]
            h[p1] = 0
            h[p0] = 1
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                improved = True
            else:
                h[p1] = 1
                h[p0] = 0
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 8:
                break
    return h, best_c5

def make_fourier_init(n, rng, parent=None):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, 8):
        amp = rng.uniform(0.02, 0.2) / np.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * k * x + phase)
    h = np.clip(h, 0, 1)
    return project_feasible(h, n)
