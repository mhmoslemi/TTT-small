                          options={'maxiter': maxit, 'ftol': 1e-14, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass

    # Top-k refinement
    if (time.time() - t0) < budget_s - 10:
        _g_beta = 5000.0
        try:
            res = minimize(_obj_topk, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
            h_cur = project_feasible(res.x, n)
        except Exception:
            pass

    return h_cur

def optimize_lbfgsb(n, h_init, time_limit, t0, budget_s):
    global _g_n, _g_beta, _g_mu
    _g_n = n
    h_cur = project_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()

    for mu in [100.0, 1000.0, 10000.0]:
        for beta in [30.0, 150.0, 800.0, 3000.0]:
            if time.time() - t_start > time_limit:
                break
            if (time.time() - t0) > budget_s - 15:
                break
            _g_beta = beta
            _g_mu = mu
            try:
                res = minimize(_obj_lbfgs, h_cur, method='L-BFGS-B',
                              bounds=bounds,
                              options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-12})
                h_cur = project_feasible(res.x, n)
            except Exception:
                pass

    return h_cur

def binary_local_search(h_bin, n, t0, budget_s):
    global _g_rng
    t_start = time.time()
    time_limit = max(5.0, min(180.0, (budget_s - (time.time() - t0)) * 0.5 - 10.0))
    corr = compute_corr(h_bin)
    max_val = np.max(corr)
    dx = 2.0 / n
    no_improve = 0

    for it in range(1000000):
        if time.time() - t_start > time_limit:
            break
        if no_improve > 8000:
            # Random restart perturbation
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            if len(ones) < 2 or len(zeros) < 2:
                break
            improved = False
            for _ in range(20):
                i = int(ones[_g_rng.randint(len(ones))])
                j = int(zeros[_g_rng.randint(len(zeros))])
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
                nc = compute_corr(h_bin)
                nm = np.max(nc)
                if nm < max_val + 3.0:
                    corr = nc
                    max_val = nm
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            if not improved:
                break
            no_improve = 0
            continue

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = int(ones[_g_rng.randint(len(ones))])
        j = int(zeros[_g_rng.randint(len(zeros))])

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        corr_new = compute_corr(h_bin)
        max_new = np.max(corr_new)

        if max_new < max_val - 1e-12:
            corr = corr_new
            max_val = max_new
            no_improve = 0
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            no_improve += 1

    final_c5 = compute_c5(h_bin)
    return h_bin, final_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_beta, _g_mu, _g_rng
    t0 = time.time()
    _g_rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return max(0.0, budget_s - elapsed() - 3.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 4:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        best_n = 192
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    Phase_1_time = min(500.0, budget_s * 0.4)
    t_phase1_start = time.time()

    res_list = [best_n]
    for delta in [-8, -4, 0, 4, 8, 12, 16, 24, 32, 64]:
        r = best_n + delta
        if 80 <= r <= 512 and r % 2 == 0:
            res_list.append(r)
    res_list = sorted(set(res_list))

    for n in res_list:
        if (time.time() - t_phase1_start) > Phase_1_time:
            break
        if remaining() < 80:
            break

        if best_n == n:
            h_base = project_feasible(best_h.copy(), n)
        else:
            xp = (np.arange(best_n) + 0.5) * 2.0 / best_n
            xn = (np.arange(n) + 0.5) * 2.0 / n
            h_base = project_feasible(np.interp(xn, xp, best_h), n)

        inits = [h_base.copy()]
        for scale in [0.015, 0.04, 0.09]:
            inits.append(project_feasible(h_base + scale * _g_rng.randn(n), n))

        # Structured: cosine modulation
        t_arr = np.arange(n) / n
        for freq in [3, 7]:
            phase = _g_rng.rand() * 2.0 * np.pi
            hf = 0.65 * h_base + 0.35 * (0.5 + 0.4 * np.cos(2.0 * np.pi * freq * t_arr + phase))
            inits.append(project_feasible(hf, n))

        # Binary-ish init
        h_bin_rand = np.zeros(n)
        h_bin_rand[_g_rng.choice(n, n // 2, replace=False)] = 1.0
        inits.append(project_feasible(h_bin_rand, n))

        # Time allocation per init
        time_left_phase = Phase_1_time - (time.time() - t_phase1_start)
        if time_left_phase < 30:
            break
        time_per_init = max(15.0, min(70.0, time_left_phase / max(1, len(inits) - 1)))

        for idx, h0 in enumerate(inits):
            if (time.time() - t_phase1_start) > Phase_1_time:
                break
            if remaining() < 20:
                break

            h_opt = optimize_slsqp(n, h0, time_per_init, t0, budget_s)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n = n

            # Also try L-BFGS-B from this result
            if (time.time() - t_phase1_start) < Phase_1_time - 20 and remaining() > 40:
                h_lb = optimize_lbfgsb(n, h_opt, min(30.0, remaining() * 0.3), t0, budget_s)
                c5_lb = compute_c5(h_lb)
                if c5_lb < best_c5:
                    best_c5 = c5_lb
                    best_h = h_lb.copy()
                    best_n = n

    # Phase 2: More aggressive optimization at best resolution
    if remaining() > 40:
        n = best_n
        _g_n = n
        h_cur = project_feasible(best_h, n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_con}]

        # SLSQP with very high beta
        for beta in [50000.0, 200000.0, 1000000.0]:
            if remaining() < 8:
                break
            _g_beta = beta
            try:
                res = minimize(_obj_slsqp, h_cur, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-16, 'iprint': 0})
                h_cur = project_feasible(res.x, n)
                c5_p = compute_c5(h_cur)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_cur.copy()
            except Exception:
                pass

        # Perturbation restarts
        for _ in range(5):
            if remaining() < 15:
                break
            n_pert = _g_rng.randint(5, 15)
            h_pert = best_h.copy()
            idxs = _g_rng.choice(n, n_pert, replace=False)
            h_pert[idxs] = _g_rng.uniform(0.0, 1.0, n_pert)
            h_pert = project_feasible(h_pert, n)

            _g_beta = 20000.0
            try:
                res = minimize(_obj_slsqp, h_pert, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 2000, 'ftol': 1e-15, 'iprint': 0})
                h_pert = project_feasible(res.x, n)
                c5_p = compute_c5(h_pert)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pert.copy()
            except Exception:
                pass

    # Phase 3: Binary local search
    if remaining() > 20:
        n = best_n
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_want = n // 2
        if n_ones > n_want:
            ones_idx = np.where(h_bin == 1)[0]
            order = np.argsort(best_h[ones_idx])[:n_ones - n_want]
            h_bin[ones_idx[order]] = 0.0
        elif n_ones < n_want:
            zeros_idx = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros_idx])[:n_want - n_ones]
            h_bin[zeros_idx[order]] = 1.0

        h_bin, c5_bin = binary_local_search(h_bin, n, t0, budget_s)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 4: Final polish
    if remaining() > 8:
        n = best_n
        _g_n = n
        h_pol = project_feasible(best_h, n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_con}]
        _g_beta = 1000000.0
        try:
            res = minimize(_obj_slsqp, h_pol, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-16, 'iprint': 0})
            h_pol = project_feasible(res.x, n)
            c5_f = compute_c5(h_pol)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_pol.copy()
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
