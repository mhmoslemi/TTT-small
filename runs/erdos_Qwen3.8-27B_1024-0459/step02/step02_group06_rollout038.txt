            g[:n-L] -= w * h[L:]         # -h[i+L] for i <= n-1-L
        else:
            L = -L
            # Negative lag
            # ∂/∂h[i] = (1-h[i+L]) * 1[i<=n-1-L] - h[i-L] * 1[i>=L]
            g[:n-L] += w * (1 - h[L:])   # (1-h[i+L]) for i <= n-1-L
            g[L:] -= w * h[:n-L]         # -h[i-L] for i >= L
    return g

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x

    if abs(s - target) < 1e-14:
        return x
    lo, hi = -1.0, 1.0

            lo = mid
        else:

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float("inf")
    best_n = 0
    
    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        best_h = pp.copy()
        best_c, _ = eval_c5(pp)
        best_n = np_
    else:
        n0 = 450
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 7 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c, _ = eval_c5(best_h)
        best_n = n0
    
    # Resolution strategy: try parent resolution, 2x, and 0.5x
    res_list = []
    bn = best_n
    for rn in [bn, 2 * bn, 450, 900, 300, 600, 180, 120]:
        if 40 <= rn <= 1800 and rn not in res_list:
            res_list.append(rn)
    res_list = sorted(res_list)
    
    # Time allocation: more time to higher resolutions near current best
    total_time = deadline - time.perf_counter()
    
    # Phase 1: Quick exploration across resolutions
    time_per_res = total_time * 0.35 / len(res_list)
    for rn in res_list:
        rem = deadline - time.perf_counter()
        if rem < 10.0:
            break
        alloc = min(time_per_res, rem - 5.0)
        if alloc < 8.0:
            break
        
        # Starting points
        starts = [upsample(best_h, rn)]
        if parent is not None and parent.size != rn:
            starts.append(upsample(parent, rn))
        # One random start
        starts.append(make_random_start(rn, rng, mode=int(rng.integers(4))))
        
        per_start = max(5.0, alloc / len(starts))
        for s in starts:
            rem2 = deadline - time.perf_counter()
            if rem2 < 8.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 3.0)
            h_out, c_out = optimize_at_resolution(rn, s, sub_dl, seed)
            c_out = eval_c5(h_out)[0]
            if c_out < best_c:
                best_h = h_out.copy()
                best_c = c_out
                best_n = rn
    
    # Phase 2: Intensive optimization at the best resolution and nearby
    rem = deadline - time.perf_counter()
    if rem > 20.0:
        # Focus on best_n and 2*best_n
        intensive_res = set()
        for rn in [best_n, 2 * best_n, best_n // 2 if best_n >= 80 else best_n, 450, 900]:
            if 40 <= rn <= 1800:
                intensive_res.add(rn)
        
        time_per_intensive = rem * 0.7 / max(1, len(intensive_res))
        for rn in sorted(intensive_res):
            rem2 = deadline - time.perf_counter()
            if rem2 < 15.0:
                break
            alloc = min(time_per_intensive, rem2 - 5.0)
            if alloc < 10.0:
                break
            
            h_start = upsample(best_h, rn)
            h_out, c_out = optimize_at_resolution(rn, h_start, time.perf_counter() + alloc, seed)
            c_out = eval_c5(h_out)[0]
            if c_out < best_c:
                best_h = h_out.copy()
                best_c = c_out
                best_n = rn
    
    # Phase 3: Final intensive polish on the absolute best
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        
        # Multiple rounds of swap + subgrad
        for round_idx in range(5):
            rem2 = deadline - time.perf_counter()
            if rem2 < 5.0:
                break
            alloc = rem2 * 0.18
            hf, cf = swap_search(h_fin, time.perf_counter() + alloc, top_frac=0.03, min_k=12, max_iter=3000, n_try=30)
            if cf < best_c:
                h_fin, best_c = hf, cf
                best_h = hf.copy()
                best_n = nbest
        
        # SA for the final push
        rem3 = deadline - time.perf_counter()
        if rem3 > 10.0:
            hf, cf = simulated_annealing(h_fin, time.perf_counter() + rem3 * 0.4, max_iter=15000, T0=0.002, T_end=0.00003)
            if cf < best_c:
                h_fin, best_c = hf, cf
                best_h = hf.copy()
                best_n = nbest
        
        # Final subgrad
        rem4 = deadline - time.perf_counter()
        if rem4 > 5.0:
            hf, cf = subgrad_search(h_fin, time.perf_counter() + rem4 * 0.7, top_frac=0.03, min_k=15, max_iter=1000)
            if cf < best_c:
                h_fin, best_c = hf, cf
                best_h = hf.copy()
                best_n = nbest
        
        # Final swap
        rem5 = deadline - time.perf_counter()
        if rem5 > 5.0:
            hf, cf = swap_search(h_fin, time.perf_counter() + rem5 * 0.8, top_frac=0.02, min_k=20, max_iter=2000, n_try=40)
            if cf < best_c:
                best_h = hf.copy()
                best_c = cf
                best_n = nbest
    
    # Ensure valid output
    if best_h is None:
        fallback = project_density(np.full(450, 0.5), 225.0)
        return fallback, eval_c5(fallback)[0], 450
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c, _ = eval_c5(final_h)
    return final_h, float(final_c), int(final_h.size)
