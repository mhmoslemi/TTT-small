def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    n = x.size
    if n == 0:
        return x.copy()
    target = float(target)
    x = np.clip(x, 0.0, 1.0).copy()
    s = float(np.sum(x))
    if abs(s - target) <= 1e-13:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
            res = target - float(np.sum(h))
        if abs(res) > 1e-12:
            if res > 0.0:
                idx = int(np.argmax(1.0 - h))
                room = 1.0 - h[idx]
                if room > 0.0:
                    h[idx] += min(res, room)
            else:
                idx = int(np.argmin(h))
                room = h[idx]
                if room > 0.0:
                    h[idx] += max(res, -room)
            h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

_FRACTIONS = (0.02, 0.08, 0.20, 0.40, 0.60, 0.85, 1.0)


def project_density(x, target):
    x = np.asarray(x, dtype=float)
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    n = x.size
    if n == 0:
        return x.copy()
    target = float(target)
    x = np.clip(x, 0.0, 1.0).copy()

    s = float(np.sum(x))
    if abs(s - target) <= 1e-13:
        return x

    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))

    if abs(res) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / float(free.size)
            h = np.clip(h, 0.0, 1.0)
            res = target - float(np.sum(h))

        if abs(res) > 1e-12:
            if res > 0.0:
                idx = int(np.argmax(1.0 - h))
                room = 1.0 - h[idx]
                if room > 0.0:
                    h[idx] += min(res, room)
            else:
                idx = int(np.argmin(h))
                room = h[idx]
                if room > 0.0:
                    h[idx] += max(res, -room)
            h = np.clip(h, 0.0, 1.0)

    return h


def exact_c(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)


def compute_corr_arrays(h):
    n = int(h.size)
    pos = np.empty(n, dtype=float)
    neg = np.empty(n, dtype=float)
    if n == 0:
        return pos, neg

    one_h = 1.0 - h
    l0 = float(np.dot(h, one_h))
    pos[0] = l0
    neg[0] = l0

    for k in range(1, n):
        N = n - k
        pos[k] = float(np.dot(h[:N], one_h[k:k + N]))
        neg[k] = float(np.dot(h[k:k + N], one_h[:N]))
    return pos, neg


def corr_max_from_h(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        return float("inf")
    h = project_density(h, 0.5 * float(n))
    pos, neg = compute_corr_arrays(h)
    return float(max(np.max(pos), np.max(neg)))


def state_from_h(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n == 0:
        h = np.zeros(2, dtype=float)
        n = 2
    target = 0.5 * float(n)
    h = project_density(h, target)
    pos, neg = compute_corr_arrays(h)

    hR = np.zeros(2 * n, dtype=float)
    hL = np.zeros(2 * n, dtype=float)
    if n > 0:
        hR[:n] = h
        hL[n:] = h

    m = float(max(np.max(pos), np.max(neg))) if n > 0 else 0.0
    return h, pos, neg, hR, hL, m


def resample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    n = int(n)
    if n <= 0:
        return np.array([], dtype=float)
    if n0 == n:
        return project_density(p, 0.5 * float(n))
    if n0 <= 1:
        val = float(p[0]) if n0 == 1 else 0.5
        return project_density(np.full(n, val, dtype=float), 0.5 * float(n))

    x0 = (np.arange(n0, dtype=float) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n, dtype=float) + 0.5) * (2.0 / float(n))
    out = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(out, 0.5 * float(n))


def make_masks(n):
    n = int(n)
    if n <= 1:
        return np.zeros((1, 0), dtype=float), np.zeros((1, 0), dtype=float)

    first = np.zeros((n, n - 1), dtype=float)
    second = np.zeros((n, n - 1), dtype=float)

    for col in range(n - 1):
        lag = col + 1
        cnt_first = n - lag
        if cnt_first > 0:
            first[:cnt_first, col] = 1.0
        if lag < n:
            second[lag:, col] = 1.0

    return first, second


def prepare_pair_coeff(hR, hL, first, second, a, b, n, coeff_pos, coeff_neg):
    right_a = hR[a + 1:a + n]
    right_b = hR[b + 1:b + n]
    left_a = hL[a + 1:n + a][::-1]
    left_b = hL[b + 1:n + b][::-1]

    fa = first[a]
    fb = first[b]
    sa = second[a]
    sb = second[b]

    prod = fa * right_a
    prod -= fb * right_b
    prod += sa * left_a
    prod -= sb * left_b

    coeff_pos[:] = (fa - fb) - prod
    coeff_neg[:] = (sa - sb) - prod

    dist = a - b
    if dist < 0:
        dist = -dist
    return dist


def eval_pair_d(coeff_pos, coeff_neg, dist, pos, neg, h, a, b, d, temp_pos, temp_neg):
    if d <= 1e-15:
        if coeff_pos.size == 0:
            return float(max(float(pos[0]), float(neg[0])))
        return float(max(float(pos[0]), float(neg[0]),
                         float(np.max(pos[1:])), float(np.max(neg[1:]))))

    l0 = 2.0 * d * (float(h[b]) - float(h[a])) - 2.0 * d * d
    vp = float(pos[0]) + l0
    vn = float(neg[0]) + l0

    if coeff_pos.size > 0:
        temp_pos[0] = vp
        np.multiply(coeff_pos, d, out=temp_pos[1:])
        temp_pos[1:] += pos[1:]
        if dist > 0:
            temp_pos[dist] += d * d
        mp = vp
        tm = float(np.max(temp_pos[1:]))
        if tm > mp:
            mp = tm
    else:
        temp_pos[0] = vp
        mp = vp

    temp_neg[0] = vn
    np.multiply(coeff_neg, d, out=temp_neg[1:])
    temp_neg[1:] += neg[1:]
    if dist > 0:
        temp_neg[dist] += d * d

    mn = vn
    tm = float(np.max(temp_neg[1:]))
    if tm > mn:
        mn = tm

    return mp if mp >= mn else mn


def apply_pair_move(h, pos, neg, hR, hL, coeff_pos, coeff_neg, dist, a, b, d, n):
    if d <= 1e-15:
        return

    l0 = 2.0 * d * (float(h[b]) - float(h[a])) - 2.0 * d * d
    pos[0] += l0
    neg[0] += l0

    if coeff_pos.size > 0:
        pos[1:] += coeff_pos * d
        neg[1:] += coeff_neg * d
        if dist > 0:
            pos[dist] += d * d
            neg[dist] += d * d

    h[a] += d
    h[b] -= d

    hR[a] += d
    hR[b] -= d
    hL[n + a] += d
    hL[n + b] -= d


def golden_pair_search(coeff_pos, coeff_neg, dist, pos, neg, h, a, b, cap,
                       d0, m0, temp_pos, temp_neg):
    if cap <= 1e-15:
        return d0, m0

    lo = max(0.0, d0 - 0.25 * cap)
    hi = min(cap, d0 + 0.25 * cap)
    if hi - lo <= max(1e-8, 1e-5 * cap):
        return d0, m0

    gr = 0.5 * (np.sqrt(5.0) - 1.0)
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)

    f1 = eval_pair_d(coeff_pos, coeff_neg, dist, pos, neg, h, a, b, x1, temp_pos, temp_neg)
    f2 = eval_pair_d(coeff_pos, coeff_neg, dist, pos, neg, h, a, b, x2, temp_pos, temp_neg)

    if f1 <= f2:
        best_d, best_m = x1, f1
    else:
        best_d, best_m = x2, f2

    for _ in range(8):
        if hi - lo < max(1e-8, 1e-5 * cap):
            break

        if f1 <= f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = eval_pair_d(coeff_pos, coeff_neg, dist, pos, neg, h, a, b, x1, temp_pos, temp_neg)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = eval_pair_d(coeff_pos, coeff_neg, dist, pos, neg, h, a, b, x2, temp_pos, temp_neg)

        if f1 < best_m:
            best_m = f1
            best_d = x1
        if f2 < best_m:
            best_m = f2
            best_d = x2

    return best_d, best_m


def lag_kind(t, n):
    if t <= 0:
        return 0, 0
    if t < n:
        return 1, t
    return 2, t - n + 1


def derivative_vec(h, kind, k):
    n = int(h.size)
    d = np.zeros(n, dtype=float)
    if kind == 0:
        d[:] = 1.0 - 2.0 * h
    elif kind == 1:
        if k <= 0 or k >= n:
            return d
        N = n - k
        if N > 0:
            d[:N] += 1.0 - h[k:]
            d[k:] -= h[:N]
    else:
        if k <= 0 or k >= n:
            return d
        N = n - k
        if N > 0:
            d[k:] += 1.0 - h[:N]
            d[:N] -= h[k:]
    return d


def active_gradient(h, pos, neg, topK):
    n = int(h.size)
    size = 2 * n - 1
    vals = np.empty(size, dtype=float)
    vals[0] = pos[0]
    if n > 1:
        vals[1:n] = pos[1:]
        vals[n:] = neg[1:]

    if topK >= size:
        idx = np.arange(size)
    else:
        topK = int(max(1, min(size, topK)))
        idx = np.argpartition(vals, -topK)[-topK:]

    tv = vals[idx]
    ord_t = np.argsort(tv)[::-1]
    idx = idx[ord_t]
    tv = tv[ord_t]

    span = float(tv.max() - tv.min())
    if span > 1e-12:
        w = np.exp((tv - tv.min()) / span)
    else:
        w = np.ones(idx.size, dtype=float) / float(max(1, idx.size))

    s = float(np.sum(w))
    if s <= 1e-300 or not np.isfinite(s):
        w = np.ones(idx.size, dtype=float) / float(max(1, idx.size))
    else:
        w = w / s

    g = np.zeros(n, dtype=float)
    for ii in range(idx.size):
        wt = float(w[ii])
        if wt <= 1e-15:
            continue
        t = int(idx[ii])

        if t == 0:
            g += wt * (1.0 - 2.0 * h)
        elif t < n:
            k = t
            N = n - k
            if N > 0:
                g[:N] += wt * (1.0 - h[k:])
                g[k:] -= wt * h[:N]
        else:
            k = t - n + 1
            N = n - k
            if N > 0:
                g[k:] += wt * (1.0 - h[:N])
                g[:N] -= wt * h[k:]

    g -= g.mean()
    return g, idx


def add_candidate(pa, pb, pc, ps, seen, a, b, cap, score):
    if cap <= 1e-10:
        return False
    if score >= 1e-7:
        return False
    a = int(a)
    b = int(b)
    key = (a, b)
    if key in seen:
        return False
    seen.add(key)
    pa.append(a)
    pb.append(b)
    pc.append(float(cap))
    ps.append(float(score))
    return True


def add_candidate_force(pa, pb, pc, ps, seen, a, b, cap, score):
    if cap <= 1e-10:
        return False
    a = int(a)
    b = int(b)
    key = (a, b)
    if key in seen:
        return False
    seen.add(key)
    pa.append(a)
    pb.append(b)
    pc.append(float(cap))
    ps.append(float(score))
    return True


def gradient_step_eval(h, g, target, current_m, steps=(0.002, 0.005, 0.01, 0.02, 0.05)):
    n = int(h.size)
    if n == 0:
        return None

    scale = float(np.max(np.abs(g)))
    if scale < 1e-14:
        return None

    gu = g / scale
    best = None

    for st in steps:
        cand = project_density(h - st * gu, target)
        pos, neg = compute_corr_arrays(cand)
        m = float(max(np.max(pos), np.max(neg)))
        if best is None or m < best[1]:
            best = (cand, m)

    if best is not None and best[1] < current_m - 1e-8:
        return best
    return None


def coarse_perturb(h, rng, scale):
    n = int(h.size)
    m = max(30, n // 6)
    hc = resample_h(h, m)
    t = np.arange(m, dtype=float)
    base = hc.copy()
    base = base + scale * rng.normal(0.0, 1.0, m)

    for _ in range(4):
        freq = float(rng.uniform(1.0, 8.0))
        phase = float(rng.uniform(0.0, 6.283185307179586))
        amp = 0.5 * scale * float(rng.uniform(0.5, 1.5))
        base = base + amp * np.sin((2.0 * np.pi * freq / float(m)) * t + phase)

    base = project_density(base, 0.5 * float(m))
    return resample_h(base, n)


def binarize_push(h, thresh):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    h2 = h.copy()
    h2 = np.where(h2 < thresh, 0.0, h2)
    h2 = np.where(h2 > 1.0 - thresh, 1.0, h2)
    return project_density(h2, 0.5 * float(n))


def local_search(h0, deadline, seed=42):
    h0 = np.asarray(h0, dtype=float)
    if h0.size < 2:
        h0 = project_density(np.full(max(2, int(h0.size)), 0.5, dtype=float), 0.5 * max(2, int(h0.size)))
        deadline = max(deadline, time.perf_counter() + 1.0)

    if deadline <= time.perf_counter():
        hh = project_density(h0, 0.5 * float(h0.size))
        return hh, corr_max_from_h(hh)

    h, pos, neg, hR, hL, m = state_from_h(h0)
    n = int(h.size)
    target = 0.5 * float(n)

    best_h = h.copy()
    best_m = m

    first, second = make_masks(n)
    coeff_pos = np.empty(max(0, n - 1), dtype=float)
    coeff_neg = np.empty(max(0, n - 1), dtype=float)
    temp_pos = np.empty(n, dtype=float)
    temp_neg = np.empty(n, dtype=float)

    rng = np.random.default_rng(seed)

    topK = int(min(120, max(24, int(n * 0.25))))
    P = int(min(14, max(6, n // 35)))
    Q = int(max(3, min(7, P // 2)))

    improve_tol = 1e-9
    stagnation = 0
    move_count = 0
    iter_count = 0
    anneal_tol_m = max(1e-3, 0.0003 * float(n) / 2.0)

    while time.perf_counter() < deadline:
        if time.perf_counter() >= deadline - 0.5:
            break

        iter_count += 1

        if m < best_m:
            best_m = m
            best_h = h.copy()

        if (move_count > 0 and move_count % 40 == 0) or (iter_count % 200 == 0):
            h, pos, neg, hR, hL, m = state_from_h(h)
            if m < best_m:
                best_m = m
                best_h = h.copy()
            if move_count > 0 and move_count % 40 == 0:
                move_count = 0

        g, top_t = active_gradient(h, pos, neg, topK)

        pa = []
        pb = []
        pc = []
        ps = []
        seen = set()

        low = np.argsort(g)[:P]
        high = np.argsort(-g)[:P]

        for ia in low:
            a = int(ia)
            for ib in high:
                b = int(ib)
                if a == b:
                    continue
                cap = min(float(h[b]), float(1.0 - h[a]))
                sc = float(g[a] - g[b])
                add_candidate(pa, pb, pc, ps, seen, a, b, cap, sc)

        used_lags = 0
        for tt in top_t:
            if used_lags >= 4:
                break
            t = int(tt)
            kind, k = lag_kind(t, n)
            dvec = derivative_vec(h, kind, k)
            lowd = np.argsort(dvec)[:Q]
            highd = np.argsort(-dvec)[:Q]
            for ia in lowd:
                a = int(ia)
                for ib in highd:
                    b = int(ib)
                    if a == b:
                        continue
                    cap = min(float(h[b]), float(1.0 - h[a]))
                    sc = 0.5 * float(dvec[a] - dvec[b]) + 0.5 * float(g[a] - g[b])
                    add_candidate(pa, pb, pc, ps, seen, a, b, cap, sc)
            used_lags += 1

        if len(ps) == 0:
            for ia in low:
                a = int(ia)
                for ib in high:
                    b = int(ib)
                    if a == b:
                        continue
                    cap = min(float(h[b]), float(1.0 - h[a]))
                    sc = float(g[a] - g[b])
                    if add_candidate_force(pa, pb, pc, ps, seen, a, b, cap, sc):
                        if len(ps) >= 8:
                            break
                if len(ps) >= 8:
                    break

        found_m = m
        found_d = 0.0
        found_a = -1
        found_b = -1
        found_dist = -1
        best_coeff_pos = None
        best_coeff_neg = None

        if len(ps) > 0:
            ps_arr = np.asarray(ps, dtype=float)
            order = np.argsort(ps_arr)
            R = min(len(order), max(3, min(8, n // 60)))

            for oi in order[:R]:
                oi = int(oi)
                a = pa[oi]
                b = pb[oi]
                cap = pc[oi]
                cap = min(float(h[b]), float(1.0 - h[a]))
                if cap <= 1e-10:
                    continue

                dist = prepare_pair_coeff(hR, hL, first, second, a, b, n, coeff_pos, coeff_neg)
                local_m = m
                local_d = 0.0

                for f in _FRACTIONS:
                    d = float(f) * cap
                    val = eval_pair_d(coeff_pos, coeff_neg, dist, pos, neg, h, a, b, d, temp_pos, temp_neg)
                    if val < local_m - improve_tol:
                        local_m = val
                        local_d = d

                if local_m < m - improve_tol and local_d > 0.02 * cap and local_d < 0.98 * cap:
                    gd, gm = golden_pair_search(coeff_pos, coeff_neg, dist, pos, neg, h, a, b,
                                                cap, local_d, local_m, temp_pos, temp_neg)
                    if gm < local_m - 1e-12:
                        local_m = gm
                        local_d = gd

                if local_m < found_m - improve_tol:
                    found_m = local_m
                    found_d = local_d
                    found_a = a
                    found_b = b
                    found_dist = dist
                    best_coeff_pos = coeff_pos.copy()
                    best_coeff_neg = coeff_neg.copy()

        if found_a >= 0:
            apply_pair_move(h, pos, neg, hR, hL, best_coeff_pos, best_coeff_neg,
                            found_dist, found_a, found_b, found_d, n)
            if coeff_pos.size > 0:
                m = float(max(float(np.max(pos)), float(np.max(neg))))
            else:
                m = float(max(float(pos[0]), float(neg[0])))
            move_count += 1
            stagnation = 0
            if m < best_m:
                best_m = m
                best_h = h.copy()
            continue

        stagnation += 1

        if stagnation in (2, 4, 8, 16, 32, 64) or (stagnation % 10 == 0 and stagnation <= 100):
            cand = gradient_step_eval(h, g, target, m)
            if cand is not None:
                h, pos, neg, hR, hL, m = state_from_h(cand[0])
                if m < best_m:
                    best_m = m
                    best_h = h.copy()
                stagnation = 0
                continue

        if stagnation > 60:
            scale = 0.02 + min(0.06, 0.005 * float(stagnation - 60))
            h_pert = coarse_perturb(h, rng, scale)
            m_pert = corr_max_from_h(h_pert)
            if m_pert < m + anneal_tol_m:
                h, pos, neg, hR, hL, m = state_from_h(h_pert)
            else:
                h, pos, neg, hR, hL, m = state_from_h(best_h)

            if m < best_m:
                best_m = m
                best_h = h.copy()

            stagnation = 0
            anneal_tol_m = max(1e-3, anneal_tol_m * 0.65)

    h_final = project_density(best_h, 0.5 * float(best_h.size))
    return h_final, best_m


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    if budget < 30.0:
        budget = 30.0
    reserve = min(25.0, 0.04 * budget)
    hard_deadline = t0 + max(1.0, budget - reserve)
    total = max(1.0, budget - reserve)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 8:
            parent = np.array(raw, dtype=float, copy=True)
            parent = np.nan_to_num(parent, nan=0.5, posinf=1.0, neginf=0.0)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        p = project_density(parent, 0.5 * float(parent.size))
        best_h = p
        best_c = exact_c(p)
        best_n = int(p.size)
    else:
        n0 = 450
        x = (np.arange(n0) + 0.5) * (2.0 / float(n0))
        h0 = (0.5
              + 0.3 * np.sin(2.0 * np.pi * 3.0 * x / 2.0)
              + 0.15 * np.sin(2.0 * np.pi * 5.0 * x / 2.0 + 1.0))
        best_h = project_density(h0, 0.5 * float(n0))
        best_c = exact_c(best_h)
        best_n = n0

    if best_n > 0 and best_n <= 900:
        main_n = int(best_n)
    elif best_n > 900:
        main_n = 900
    else:
        main_n = 450
    if main_n < 2:
        main_n = 450
    if main_n % 2 == 1:
        main_n += 1
    main_n = min(900, main_n)

    # Coarse exploration, useful only if there is enough time.
    if total > 500.0:
        rem = hard_deadline - time.perf_counter()
        if rem > 250.0:
            alloc = min(0.12 * total, rem * 0.35, rem - 30.0)
            if alloc > 30.0:
                coarse_n = max(30, main_n // 5)
                if coarse_n % 2 == 1:
                    coarse_n += 1
                dl = time.perf_counter() + alloc
                hc, _ = local_search(resample_h(best_h, coarse_n), dl, seed=int(seed) + 101)
                up = resample_h(hc, main_n)
                c = exact_c(up)
                if c < best_c:
                    best_c = c
                    best_h = up
                    best_n = main_n

    # Main refinement.
    rem = hard_deadline - time.perf_counter()
    if rem > 15.0:
        alloc = min(rem * 0.65, rem - 8.0)
        if alloc > 10.0:
            h_main = resample_h(best_h, main_n)
            cm0 = exact_c(h_main)
            if cm0 < best_c:
                best_c = cm0
                best_h = h_main
                best_n = main_n

            dl = time.perf_counter() + alloc
            hm, _ = local_search(h_main, dl, seed=int(seed) + 11)
            cm = exact_c(hm)
            if cm < best_c:
                best_c = cm
                best_h = hm
                best_n = main_n

    # Higher-resolution refinement.
    hi_n = None
    if best_n < 900:
        hi_n = min(900, max(600, 2 * main_n))
        if hi_n % 2 == 1:
            hi_n += 1
            hi_n = min(900, hi_n)
        if hi_n <= best_n:
            hi_n = min(900, best_n + 300)
            if hi_n % 2 == 1:
                hi_n += 1
                hi_n = min(900, hi_n)

    if hi_n is not None and hi_n > best_n:
        rem = hard_deadline - time.perf_counter()
        if rem > 150.0:
            alloc = min(rem * 0.75, rem - 30.0)
            if alloc > 30.0:
                h_hi = resample_h(best_h, hi_n)
                dl = time.perf_counter() + alloc
                hmm, _ = local_search(h_hi, dl, seed=int(seed) + 21)
                cm = exact_c(hmm)
                if cm < best_c:
                    best_c = cm
                    best_h = hmm
                    best_n = int(hi_n)

    # Final short polish and quick binarization probes.
    rem = hard_deadline - time.perf_counter()
    if best_h is not None and rem > 25.0:
        nbest = int(best_h.size)
        for th in (0.05, 0.10, 0.15, 0.25):
            rem = hard_deadline - time.perf_counter()
            if rem < 5.0:
                break
            cand = binarize_push(best_h, th)
            cc = exact_c(cand)
            if cc < best_c:
                best_c = cc
                best_h = cand
                best_n = nbest

        rem = hard_deadline - time.perf_counter()
        if rem > 30.0:
            alloc = min(rem * 0.6, rem - 10.0)
            if alloc > 10.0:
                dl = time.perf_counter() + alloc
                hf, _ = local_search(best_h, dl, seed=int(seed) + 31)
                cf = exact_c(hf)
                if cf < best_c:
                    best_c = cf
                    best_h = hf
                    best_n = int(hf.size)

    if best_h is None:
        final_h = project_density(np.full(450, 0.5, dtype=float), 225.0)
        final_c = exact_c(final_h)
        return final_h, float(final_c), 450

    final_h = project_density(best_h, 0.5 * float(best_h.size))
    final_c = exact_c(final_h)

    if final_c > best_c:
        cand_c = exact_c(best_h)
        if cand_c < final_c:
            final_h = best_h
            final_c = cand_c

    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = project_density(final_h, 0.5 * float(final_h.size))
    final_c = exact_c(final_h)

    try:
        pc = float(evaluate_erdos_solution(final_h))
        if np.isfinite(pc) and pc < final_c:
            final_c = pc
    except Exception:
        pass

    return final_h, float(final_c), int(final_h.size)
