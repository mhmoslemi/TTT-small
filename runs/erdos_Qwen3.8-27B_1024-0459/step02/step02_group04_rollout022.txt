c_{-l} = sum_{i=l}^{n-1} h[i](1-h[i-l]) = sum_{j=0}^{m-1} h[j+l](1-h[j]) (sub j=i-l)

So c_{+l} = c_{-l}. The correlate IS symmetric here.

So C₅ = max(0, 1, ..., n-1) of c_l * dx where c_l = sum_{i=0}^{n-1-l} h[i+l](1-h[i]).

And the gradient for lag l (>0) is:
d c_l / d h[j]:
- h[j] as h[i+l]: i=j-l, valid if 0<=j-l<m → l<=j<l+m=n
- h[j] in (1-h[i]): i=j, valid if 0<=j<m

g[j] = (1-h[j-l]) if l<=j<n  [from h[i+l] term]
     + (-h[j+l]) if 0<=j<m  [from (1-h[i]) term]

import numpy as np
import time
from scipy.optimize import minimize

AC_LAGS = []
CURRENT_BETA = 1.0
CURRENT_N = 160


def get_parent_values():
    try:
        vals = initial_h_values
        if vals is None:
            return None
        arr = np.asarray(vals, dtype=float).copy()
        if arr.size == 0:
            return None
        return arr
    except Exception:
        return None


def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    h = h[:n]
    if h.size < n:
        h = np.concatenate((h, np.zeros(n - h.size, dtype=float)))
    target = 0.5 * float(n)
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(40):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-12
        else:
            mask = out > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        np.clip(out, 0.0, 1.0, out=out)
    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_src = h_src.size
    n_dst = int(n_dst)
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def all_lag_vals(h):
    n = h.size
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx


def lag_value_single(h, L):
    n = len(h)
    m = n - abs(L)
    if m <= 0:
        return 0.0
    dx = 2.0 / float(n)
    if L == 0:
        return float(np.sum(h) - np.dot(h, h)) * dx
    ac = float(np.dot(h[:m], h[L if L > 0 else :m]))
    if L > 0:
        s = float(np.sum(h[:m]))
    else:
        s = float(np.sum(h[-m:]))
    return (s - ac) * dx


def lag_grad_single(h, L):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
        return g
    La = abs(L)
    m = n - La
    if m <= 0:
        return g
    # For both positive and negative L, the gradient formula is the same:
    # c_L = sum_{i=0}^{m-1} h[i+L](1-h[i])
    g[La:La+m] += (1.0 - h[:m])
    g[:m] += -h[La:La+m]
    return g


def smooth_obj(x):
    global CURRENT_BETA, CURRENT_N
    beta = CURRENT_BETA
    n = CURRENT_N
    h = x
    vals = all_lag_vals(h)
    # Due to symmetry, only need first n values (lags 0 to n-1)
    # Actually correlate is symmetric for h vs 1-h, but let's be safe and use all
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    e = np.exp(np.clip(shifted, -700.0, 700.0))
    s = float(np.sum(e))
    if s <= 0:
        return m
    return m + float(np.log(s)) / beta


def smooth_grad(x):
    global CURRENT_BETA, CURRENT_N
    beta = CURRENT_BETA
    n = CURRENT_N
    h = x
    vals = all_lag_vals(h)
    m = float(np.max(vals))
    shifted = beta * (vals - m)
    e = np.exp(np.clip(shifted, -700.0, 700.0))
    w = e / float(np.sum(e))
    dx = 2.0 / float(n)
    lags = np.arange(-(n - 1), n)
    g = np.zeros(n, dtype=float)
    for i in range(len(w)):
        if w[i] < 1e-12:
            continue
        L = int(lags[i])
        g += (w[i] * dx) * lag_grad_single(h, L)
    return g


def sum_eq(x):
    return float(np.sum(x) - 0.5 * len(x))


def sum_eq_jac(x):
    return np.ones(len(x), dtype=float)


def optimize_smooth_slsqp(h0, n, beta, maxiter=40):
    global CURRENT_BETA, CURRENT_N
    CURRENT_BETA = float(beta)
    CURRENT_N = int(n)
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": sum_eq, "jac": sum_eq_jac}]
    try:
        res = minimize(
            smooth_obj, h, method="SLSQP", bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": 1e-12, "disp": False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    return h, compute_c5(h)


def active_obj(x):
    return float(x[-1])


def active_obj_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j


def active_sum_eq(x):
    return float(np.sum(x[:-1]) - 0.5 * float(len(x) - 1))


def active_sum_eq_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j


def active_ineq(x):
    global AC_LAGS
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    out = np.zeros(len(AC_LAGS), dtype=float)
    for i, L in enumerate(AC_LAGS):
        out[i] = float(x[-1]) - lag_value_single(h, int(L))
    return out


def active_ineq_jac(x):
    global AC_LAGS
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    m = len(AC_LAGS)
    J = np.zeros((m, n + 1), dtype=float)
    for i, L in enumerate(AC_LAGS):
        L = int(L)
        g = lag_grad_single(h, L)
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def optimize_active_slsqp(h0, active_lags, t0, maxiter=60):
    global AC_LAGS
    n = len(h0)
    AC_LAGS = [int(l) for l in active_lags]
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h0
    x0[n] = float(t0)
    bounds = [(0.0, 1.0)] * (n + 1)
    constraints = [{"type": "eq", "fun": active_sum_eq, "jac": active_sum_eq_jac}]
    if len(AC_LAGS) > 0:
        constraints.append({"type": "ineq", "fun": active_ineq, "jac": active_ineq_jac})
    try:
        res = minimize(
            active_obj, x0, method="SLSQP", bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": 1e-13, "disp": False}
        )
        return res
    except Exception:
        return None


def refine_active(h0, n, time_limit, active_counts=(8, 16, 32, 64), maxiter=50):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    for acnt in active_counts:
        if time.time() > time_limit - 3.0:
            break
        for _pass in range(2 if acnt <= 16 else 1):
            if time.time() > time_limit - 3.0:
                break
            vals = all_lag_vals(h)
            c_now = float(np.max(vals))
            if c_now < best_c:
                best_c = c_now
                best_h = h.copy()
            lags = np.arange(-(n - 1), n)
            top_idx = np.argsort(-vals)[:acnt]
            active = [int(lags[i]) for i in top_idx]
            t0 = c_now + 1e-6
            res = optimize_active_slsqp(h, active, t0, maxiter=maxiter)
            if res is not None and res.x is not None and res.x.size == n + 1:
                if np.all(np.isfinite(res.x)):
                    h2 = project_feasible(res.x[:-1], n)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2.copy()
                    h = h2
    return best_h, best_c


def local_perturb(h0, n, time_limit, seed, step=0.02, max_trials=3000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.002
    min_temp = 1e-7
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, max(temp * 1.2, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)


def gauss_smooth(h, sigma=1.2):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 3:
        return h.copy()
    length = min(7, n)
    if length % 2 == 0:
        length -= 1
    length = max(3, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * sigma * sigma))
    w /= w.sum()
    return np.convolve(h, w, mode="same")


def make_diverse_starts(h_ref, n, rng, count=4):
    starts = []
    base = project_feasible(h_ref, n)
    starts.append(base.copy())
    # Smoothed
    starts.append(project_feasible(gauss_smooth(base, 1.5), n))
    # Perturbed
    p = base + rng.randn(n) * 0.06
    starts.append(project_feasible(p, n))
    # Flip symmetry
    starts.append(project_feasible(0.5 * (base + base[::-1]), n))
    # Cosine perturbation
    x = np.linspace(0, 2, n, endpoint=False)
    p2 = base + 0.04 * np.cos(3 * np.pi * x + rng.uniform(0, 2 * np.pi))
    starts.append(project_feasible(p2, n))
    if count:
        starts = starts[:count]
    return starts


def improve_full(h0, n, time_limit, seed, do_smooth=True, do_active=True, do_perturb=True):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    if do_smooth and 2 <= n <= 300:
        for beta in [5, 25, 100, 400]:
            if time.time() > time_limit - 10.0:
                break
            mx = 35 if n > 200 else 50
            h2, c2 = optimize_smooth_slsqp(h, n, beta, maxiter=mx)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
            h = h2 if c2 <= best_c + 1e-4 else best_h

    if do_active and 2 <= n <= 320 and time.time() < time_limit - 8.0:
        if n <= 160:
            seq = (12, 24, 48, 96)
        elif n <= 240:
            seq = (10, 24, 48)
        else:
            seq = (8, 20, 40)
        mx = 40 if n > 200 else 55
        h2, c2 = refine_active(best_h, n, time.time(), active_counts=seq, maxiter=mx)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if do_perturb and 2 <= n <= 260 and time.time() < time_limit - 5.0:
        sub_limit = min(time_limit, time.time() + max(5.0, (time_limit - time.time()) * 0.3))
        h2, c2 = local_perturb(best_h, n, sub_limit, seed + 7777, step=0.015, max_trials=2000)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 35.0)

    # Get parent
    parent = get_parent_values()
    if parent is not None and parent.size > 0:
        pn = int(parent.size)
        ph = project_feasible(parent, pn)
        pc = compute_c5(ph)
        best_h = ph
        best_c = pc
        best_n = pn
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    # Resolution list: parent first, then diverse
    res_list = []
    for r in [best_n, 192, 200, 224, 240, 256, 160, 128]:
        r = int(r)
        if 48 <= r <= 320 and r not in res_list:
            res_list.append(r)

    # Process resolutions
    processed = set()
    for n in res_list:
        if time.time() > hard_end - 50.0:
            break
        if n in processed:
            continue
        processed.add(n)

        rem = hard_end - time.time()
        if rem < 40.0:
            break

        # Allocate time: first resolution gets more
        slot = min(rem * 0.45, 280.0) if n == best_n else min(rem * 0.3, 180.0)
        slot = max(30.0, slot)

        # Build starts
        if n == best_n:
            h_ref = best_h
        else:
            h_ref = interpolate_to_n(best_h, n)
        starts = make_diverse_starts(h_ref, n, rng, count=3)

        # Optimize each start
        per_start = slot / float(len(starts))
        for j, hs in enumerate(starts):
            if time.time() > hard_end - 50.0:
                break
            remaining = hard_end - 50.0 - time.time()
            if remaining < 10.0:
                break
            per = min(per_start, remaining)
            do_sm = (2 <= n <= 280)
            do_act = (2 <= n <= 300)
            do_pt = (2 <= n <= 240)
            hh, cc = improve_full(hs, n, time.time() + per, seed + 100 * j + n,
                                   do_smooth=do_sm, do_active=do_act, do_perturb=do_pt)
            if cc < best_c:
                best_h = hh.copy()
                best_c = cc
                best_n = n

    # Final intensive polish on best solution
    if time.time() < hard_end - 30.0 and 2 <= best_n <= 300:
        # More aggressive smooth
        for beta in [200, 500, 1000]:
            if time.time() > hard_end - 35.0:
                break
            h2, c2 = optimize_smooth_slsqp(best_h, best_n, beta, maxiter=40)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
                best_n = best_n

    if time.time() < hard_end - 20.0 and 2 <= best_n <= 280:
        # More active set passes
        if best_n <= 160:
            fin_seq = (16, 32, 64, 128)
        else:
            fin_seq = (12, 24, 48)
        h2, c2 = refine_active(best_h, best_n, hard_end - 15.0,
                                active_counts=fin_seq, maxiter=70)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    if time.time() < hard_end - 10.0 and 2 <= best_n <= 256:
        # Final local perturbation
        limit = hard_end - 3.0
        h2, c2 = local_perturb(best_h, best_n, limit, seed + 4242, step=0.01, max_trials=5000)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
