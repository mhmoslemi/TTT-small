```python
def project_feasible(h, n):
    target = n / 2.0
    # Binary search for tau
    lo, hi = h.min() - 2, h.max() + 2
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target - projected.sum()
    if diff != 0:
        mask = (projected > 0) & (projected < 1)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0, 1)
    return projected
```

```python
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):

        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:

        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.clip(h - tau, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()

        cnt = mask.sum()

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)

def compute_gradient(h, p):
    """Compute gradient of softmax objective w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    A = 1.0 - h
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        # term1: grad[j] += pk * (1-h[j+k-n+1]) for valid j
        # j+k-n+1 in [0, n-1] => j in [n-1-k, 2n-2-k] intersect [0, n-1]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            j_lo = i_lo - k + n - 1
            length = i_hi - i_lo + 1
            grad[j_lo:j_lo + length] += pk * A[i_lo:i_hi + 1]
        
        # term2: grad[j] -= pk * h[j-k+n-1] for valid j
        # j-k+n-1 in [0, n-1] => j in [k-n+1, k] intersect [0, n-1]
        i_lo2 = max(0, n - 1 - k)
        i_hi2 = min(n - 1, 2 * n - 2 - k)
        if i_lo2 <= i_hi2:
            j_lo2 = i_lo2 + k - n + 1
            length2 = i_hi2 - i_lo2 + 1
            grad[j_lo2:j_lo2 + length2] -= pk * h[i_lo2:i_hi2 + 1]
    
    return grad * (2.0 / n)

def adam_optimize(h_init, n, n_steps, lr_start=0.1, lr_end=0.001,
                  beta_start=30, beta_end=20000, start_time=None, budget_s=1300):
    h = h_init.astype(float).copy()
    h = project_feasible(h, n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # ADAM parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    for step in range(n_steps):
        if start_time is not None:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 60:
                break
        
        # Schedule
        frac = step / max(n_steps - 1, 1)
        lr = lr_start * (lr_end / lr_start) ** frac
        beta = beta_start * (beta_end / beta_start) ** frac
        
        # Compute softmax weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        shifted = corr - cmax
        exp_corr = np.exp(beta * shifted)
        p = exp_corr / np.sum(exp_corr)
        
        # Compute gradient
        grad = compute_gradient(h, p)
        
        # ADAM update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def slsqp_run(n, h_init, max_iter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        return cmax * (2.0 / n)
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def coordinate_descent(h, n, n_rounds=50):
    """Coordinate descent with line search on the max-correlation objective."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for rd in range(n_rounds):
        improved = False
        for i in range(n):
            # Try moving h[i] while adjusting another coordinate to maintain sum
            # Actually, let's try: change h[i] by delta, and adjust h[j] by -delta
            # for the j that gives the best improvement
            for j in [(i+1)%n, (i-1+n)%n, (i+n//2)%n]:
                if i == j:
                    continue
                # Try small moves
                for delta in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]:
                    if h[i] + delta < 0 or h[i] + delta > 1:
                        continue
                    if h[j] - delta < 0 or h[j] - delta > 1:
                        continue
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.clip(h - tau, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def compute_gradient(h, p):
    n = len(h)
    grad = np.zeros(n)
    A = 1.0 - h
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            j_lo = i_lo - k + n - 1
            length = i_hi - i_lo + 1
            grad[j_lo:j_lo + length] += pk * A[i_lo:i_hi + 1]
        i_lo2 = max(0, n - 1 - k)
        i_hi2 = min(n - 1, 2 * n - 2 - k)
        if i_lo2 <= i_hi2:
            j_lo2 = i_lo2 + k - n + 1
            length2 = i_hi2 - i_lo2 + 1
            grad[j_lo2:j_lo2 + length2] -= pk * h[i_lo2:i_hi2 + 1]
    return grad * (2.0 / n)

def adam_optimize(h_init, n, n_steps, lr_start=0.1, lr_end=0.001,
                  beta_start=20.0, beta_end=30000.0,
                  start_time=None, budget_s=1300, margin=60):
    h = project_feasible(h_init.astype(float).copy(), n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    for step in range(n_steps):
        if start_time is not None:
            if time.time() - start_time > budget_s - margin:
                break
        frac = step / max(n_steps - 1, 1)
        lr = lr_start * (lr_end / lr_start) ** frac
        beta = beta_start * (beta_end / beta_start) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = np.max(corr)
        shifted = corr - cmax
        exp_corr = np.exp(beta * shifted)
        p = exp_corr / np.sum(exp_corr)
        grad = compute_gradient(h, p)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_run(n, h_init, max_iter=3000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * (2.0 / n)
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def coordinate_descent(h, n, n_rounds=30):
    h = h.copy()
    best_c5 = compute_c5(h)
    for rd in range(n_rounds):
        improved = False
        for i in range(n):
            partners = [(i + 1) % n, (i - 1 + n) % n, (i + n // 2) % n]
            for j in partners:
                if i == j:
                    continue
                for delta in [0.05, -0.05, 0.03, -0.03, 0.01, -0.01]:
                    if h[i] + delta < 0 or h[i] + delta > 1:
                        continue
                    if h[j] - delta < 0 or h[j] - delta > 1:
                        continue
                    h_try = h.copy()
                    h_try[i] += delta
                    h_try[j] -= delta
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c5

def smooth_anneal_solve(h_init, n, betas=(50, 200, 1000, 5000, 20000),
                        start_time=None, budget_s=1300, margin=60):
    current = project_feasible(h_init.astype(float).copy(), n)
    best_c5 = compute_c5(current)
    best_h = current.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    for beta in betas:
        if start_time is not None:
            if time.time() - start_time > budget_s - margin:
                break
        def obj_s(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr) * (2.0 / n)
            return cmax + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr * (2.0/n) - cmax))))
        try:
            res = minimize(obj_s, current, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 2000, 'ftol': 1e-13, 'iprint': 0})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass
    return best_h, best_c5

def random_perturb_improve(h, n, n_trials, base_step, rng):

        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 11: Final coordinate descent + perturbation
    if remaining() > 30:
        best_h, best_c5 = coordinate_descent(best_h, best_n, n_rounds=10)
    if remaining() > 20:
        n_trials = min(10000, int((remaining() - 10) / 0.012))
        best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.015, rng)

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
```
