```
def project_feasible(h):
    n = h.size
    target = 0.5*n
    h = np.array(h, dtype=float).copy()
    h = np.clip(h,0,1)
    for _ in range(10):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            # no slack; distribute over all
            mask = np.ones(n, dtype=bool)
            cnt=n
        h[mask] += diff / cnt
        h = np.clip(h,0,1)
    s = float(np.sum(h)); diff=target-s
    if abs(diff)>1e-15:
       # choose element with max room
       room = np.where(diff>0, 1.0-h, h)
       idx = int(np.argmax(room))
       if room[idx] >= abs(diff):
          h[idx] += diff
       else:
          # distribute over top rooms
          # fallback sum all
          h += diff/n
          h=np.clip(h,0,1)
    return h
```

```python
import time
import numpy as np
from scipy.optimize import minimize

_ACTIVE_N = 0
_ACTIVE_LAGS = []
_SMOOTH_N = 0
_SMOOTH_BETA = 1.0


def time_remaining(start_time, budget_s):
    return max(0.0, float(budget_s) - (time.time() - start_time))


def compute_c5(h):
    n = int(h.size)
    if n <= 0:
        return 1.0
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    if m < 0.0:
        m = 0.0
    return m * dx


def project_feasible(h):
    h = np.array(h, dtype=float).copy()
    n = h.size
    if n == 0:
        return h
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(24):
        s = float(np.sum(h))
        diff = target - s
        if abs(diff) <= 1e-12:
            break
        if diff > 0.0:
            mask = h < (1.0 - 1e-12)
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            mask = np.ones(n, dtype=bool)
            cnt = n
        h[mask] += diff / float(cnt)
        h = np.clip(h, 0.0, 1.0)

    s = float(np.sum(h))
    diff = target - s
    if abs(diff) > 1e-14:
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) <= 1e-15:
                break
            idx = int(idx)
            if diff > 0.0:
                room = 1.0 - h[idx]
                if room <= 0.0:
                    continue
                add = min(room, diff)
            else:
                room = -h[idx]
                if room <= 0.0:
                    continue
                add = max(room, diff)
            h[idx] += add
            diff -= add
    return h


def interp_h(src, n):
    n = int(n)
    n0 = int(src.size)
    if n <= 0:
        return project_feasible(src)
    if n0 == n:
        return project_feasible(src)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    y = np.interp(x1, x0, src, left=float(src[0]), right=float(src[-1]))
    return project_feasible(y)


def default_binary(n):
    n = int(n)
    if n % 2:
        n -= 1
    h = np.zeros(n, dtype=float)
    if n > 0:
        h[: n // 2] = 1.0
    return h


def make_binary_random(n, rng):
    n = int(n)
    if n % 2:
        n -= 1
    h = np.zeros(n, dtype=float)
    if n > 0:
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0
    return h


def make_fourier_start(n, rng, kmax, amp):
    n = int(n)
    if n % 2:
        n -= 1
    if n <= 0:
        return np.array([], dtype=float)
    kmax = min(max(1, int(kmax)), max(1, n // 4))
    amp = float(amp)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 * np.ones(n, dtype=float)
    for f in range(1, kmax + 1):
        a = rng.uniform(-1.0, 1.0) * amp / np.sqrt(float(f))
        b = rng.uniform(-1.0, 1.0) * amp / np.sqrt(float(f))
        h += a * np.cos(np.pi * f * x) + b * np.sin(np.pi * f * x)
    return project_feasible(np.clip(h, 0.0, 1.0))


def round_binary(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    want = n // 2
    h2 = (h >= 0.5).astype(float)
    ones = np.where(h2 == 1.0)[0]
    zeros = np.where(h2 == 0.0)[0]
    if len(ones) > want:
        drop = len(ones) - want
        vals = h[ones]
        ord_idx = np.argsort(vals)[:drop]
        h2[ones[ord_idx]] = 0.0
    elif len(ones) < want:
        add = want - len(ones)
        vals = h[zeros]
        ord_idx = np.argsort(-vals)[:add]
        h2[zeros[ord_idx]] = 1.0
    return h2


def add_candidate(lst, nn, minn=128, cap=800):
    if nn is None:
        return
    nn = int(nn)
    if nn % 2:
        nn -= 1
    if nn < minn or nn > cap:
        return
    if nn not in lst:
        lst.append(nn)

        if nn in out:

    rest = [nn for nn in out if nn not in pr]
    rest.sort()
    pr.extend(rest)

def smooth_max_value(corr, beta):
    corr = np.asarray(corr, dtype=float)
    m = float(np.max(corr))
    shifted = np.clip(float(beta) * (corr - m), -700.0, 0.0)
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    if s <= 0.0:
        return m
    return m + np.log(s) / float(beta)


def smooth_max_weights(corr, beta):
    corr = np.asarray(corr, dtype=float)
    m = float(np.max(corr))
    shifted = np.clip(float(beta) * (corr - m), -700.0, 0.0)
    exps = np.exp(shifted)
    s = float(np.sum(exps))
    if s <= 0.0:
        return np.full(corr.shape, 1.0 / max(1, corr.size))
    return exps / s


def smooth_grad_from_weights(h, weights):
    n = len(h)
    grad = np.zeros(n, dtype=float)
    for k in range(len(weights)):
        w = float(weights[k])
        if w <= 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            r = L
            grad[: n - r] -= w * h[r:]
            grad[r:] += w * (1.0 - h[: n - r])
        else:
            r = -L
            grad[: n - r] += w * (1.0 - h[r:])
            grad[r:] -= w * h[: n - r]
    return grad


def smooth_obj_global(x):
    n = int(_SMOOTH_N)
    if n <= 0:
        return 0.0
    corr = np.correlate(x, 1.0 - x, mode="full")
    return float(smooth_max_value(corr, _SMOOTH_BETA) * (2.0 / n))


def smooth_jac_global(x):
    n = int(_SMOOTH_N)
    if n <= 0:
        return np.zeros(0, dtype=float)
    corr = np.correlate(x, 1.0 - x, mode="full")
    w = smooth_max_weights(corr, _SMOOTH_BETA)
    grad = smooth_grad_from_weights(x, w)
    return grad * (2.0 / n)


def smooth_sum_fun(x):
    return float(np.sum(x) - 0.5 * _SMOOTH_N)


def smooth_sum_jac(x):
    return np.ones(int(_SMOOTH_N), dtype=float)


def run_smooth_slsqp(h0, n, beta, maxiter):
    global _SMOOTH_N, _SMOOTH_BETA
    n = int(n)
    h = project_feasible(h0)
    if n <= 0:
        return h, compute_c5(h)
    _SMOOTH_N = n
    _SMOOTH_BETA = float(beta)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": smooth_sum_fun, "jac": smooth_sum_jac}]
    hnew = h
    try:
        res = minimize(
            smooth_obj_global,
            h,
            jac=smooth_jac_global,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-9, "iprint": 0},
        )
        if hasattr(res, "x") and res.x is not None:
            hnew = project_feasible(res.x)
    except Exception:
        hnew = h
    return hnew, compute_c5(hnew)


def smooth_refine(h, n, rng, start_time, budget_s, beta_list, iters):
    h = project_feasible(h)
    C = compute_c5(h)
    if not beta_list:
        return h, C
    for bi, beta in enumerate(beta_list):
        rem = time_remaining(start_time, budget_s)
        if rem < 50:
            break
        it = int(iters[bi]) if bi < len(iters) else int(iters[-1])
        if n > 320:
            it = min(it, 150)
        htry, ctry = run_smooth_slsqp(h, n, beta, it)
        if ctry < C - 1e-10:
            h, C = htry, ctry
        else:
            rem = time_remaining(start_time, budget_s)
            if rem > 90:
                scale = 0.02 * np.sqrt(200.0 / float(max(1, n)))
                h2 = project_feasible(h + scale * rng.normal(size=n))
                htry, ctry = run_smooth_slsqp(h2, n, beta, it)
                if ctry < C - 1e-10:
                    h, C = htry, ctry
    return h, C


def active_obj(x):
    return float(x[-1])


def active_obj_jac(x):
    g = np.zeros(int(x.size), dtype=float)
    g[-1] = 1.0
    return g


def active_sum_fun(x):
    return float(np.sum(x[:-1]) - 0.5 * _ACTIVE_N)


def active_sum_jac(x):
    g = np.zeros(int(x.size), dtype=float)
    g[:-1] = 1.0
    return g


def active_constraint_fun(x):
    n = int(_ACTIVE_N)
    h = x[:-1]
    t = float(x[-1])
    dx = 2.0 / float(n)
    m = len(_ACTIVE_LAGS)
    out = np.empty(m, dtype=float)
    for i, p in enumerate(_ACTIVE_LAGS):
        r = abs(int(p))
        if p == 0:
            val = float(np.sum(h - h * h))
        elif p > 0:
            val = float(np.dot(1.0 - h[: n - r], h[r:]))
        else:
            val = float(np.dot(h[: n - r], 1.0 - h[r:]))
        out[i] = t - val * dx
    return out


def active_constraint_jac(x):
    n = int(_ACTIVE_N)
    h = x[:-1]
    m = len(_ACTIVE_LAGS)
    J = np.zeros((m, n + 1), dtype=float)
    dx = 2.0 / float(n)
    for i, p in enumerate(_ACTIVE_LAGS):
        r = abs(int(p))
        g = np.zeros(n, dtype=float)
        if p == 0:
            g = 1.0 - 2.0 * h
        elif p > 0:
            g[: n - r] -= h[r:]
            g[r:] += 1.0 - h[: n - r]
        else:
            g[: n - r] += 1.0 - h[r:]
            g[r:] -= h[: n - r]
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def select_active_lags(c, n, max_active, count_tol=0.5):
    n = int(n)
    max_active = int(max_active)
    idx = np.argsort(c)[::-1]
    cmax = float(c[idx[0]])
    lags = []
    for k in idx:
        if len(lags) >= 2 and (float(c[k]) + 1e-12 < cmax - float(count_tol)):
            break
        lags.append(int(k - (n - 1)))
        if len(lags) >= max_active:
            break
    if not lags:
        lags = [0]
    return lags


def run_active_slsqp_once(h0, n, max_active, maxiter):
    global _ACTIVE_N, _ACTIVE_LAGS
    n = int(n)
    h = project_feasible(h0)
    if n <= 0:
        return h, compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(np.max(corr))
    C = cmax * (2.0 / n)
    tol = 0.75 if n <= 320 else (0.55 if n <= 512 else 0.40)
    _ACTIVE_LAGS = select_active_lags(corr, n, max_active, tol)
    _ACTIVE_N = n
    x0 = np.empty(n + 1, dtype=float)
    x0[:-1] = h
    x0[-1] = min(1.0, C + 1e-7)
    bounds = [(0.0, 1.0)] * (n + 1)
    cons = [
        {"type": "eq", "fun": active_sum_fun, "jac": active_sum_jac},
        {"type": "ineq", "fun": active_constraint_fun, "jac": active_constraint_jac},
    ]
    hnew = h
    try:
        res = minimize(
            active_obj,
            x0,
            jac=active_obj_jac,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-9, "iprint": 0},
        )
        if hasattr(res, "x") and res.x is not None:
            hnew = project_feasible(res.x[:-1])
    except Exception:
        hnew = h
    return hnew, compute_c5(hnew)


def active_refine(h, n, rng, start_time, budget_s, max_rounds, maxiter):
    n = int(n)
    h = project_feasible(h)
    C = compute_c5(h)
    max_rounds = max(0, int(max_rounds))
    for r in range(max_rounds):
        rem = time_remaining(start_time, budget_s)
        if rem < 50:
            break
        if n <= 320:
            max_active = min(4 + r, 10)
        elif n <= 512:
            max_active = min(4 + r, 7)
        else:
            max_active = min(4 + r, 5)
        it = int(maxiter) + 15 * r
        if n > 640:
            it = min(it, 30)
        elif n > 512:
            it = min(it, 45)
        elif n > 320:
            it = min(it, 90)

        htry, ctry = run_active_slsqp_once(h, n, max_active, it)
        if ctry < C - 1e-10:
            h, C = htry, ctry
        else:
            rem = time_remaining(start_time, budget_s)
            if rem > 90:
                scale = 0.02 * np.sqrt(200.0 / float(max(1, n)))
                h2 = project_feasible(h + scale * rng.normal(size=n))
                htry, ctry = run_active_slsqp_once(h2, n, max_active, it)
                if ctry < C - 1e-10:
                    h, C = htry, ctry
    return h, C


def exact_top_gradient(h, c, n, max_active):
    n = int(n)
    lags = select_active_lags(c, n, max_active, 0.55)
    g = np.zeros(n, dtype=float)
    for p in lags:
        r = abs(int(p))
        if p == 0:
            g += 1.0 - 2.0 * h
        elif p > 0:
            g[: n - r] -= h[r:]
            g[r:] += 1.0 - h[: n - r]
        else:
            g[: n - r] += 1.0 - h[r:]
            g[r:] -= h[: n - r]
    return g / float(max(1, len(lags)))


def exact_line_search(h, n, rng, start_time, budget_s, max_evals):
    n = int(n)
    h = project_feasible(h)
    C = compute_c5(h)
    evals = 0
    no_improve = 0
    mode = 0
    while True:
        rem = time_remaining(start_time, budget_s)
        if rem < 5 or evals >= int(max_evals) or no_improve > 55:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        cmax = float(np.max(corr))
        C = cmax * (2.0 / n)

        if mode % 3 == 0:
            v = -exact_top_gradient(h, corr, n, 5)
        else:
            v = rng.normal(size=n)
        v -= float(np.mean(v))
        norm = float(np.linalg.norm(v))
        if norm < 1e-12:
            break
        v = v / norm

        local_h = h
        local_C = C
        steps = [0.02, 0.05, 0.10, 0.20, 0.40]
        for s in steps:
            if evals >= int(max_evals) or rem < 3:
                break
            htry = project_feasible(h + s * v)
            ctry = compute_c5(htry)
            evals += 1
            if ctry < local_C - 1e-12:
                local_C = ctry
                local_h = htry

            if evals >= int(max_evals):
                break
            if s < 0.06:
                htry = project_feasible(h - 0.5 * s * v)
                ctry = compute_c5(htry)
                evals += 1
                if ctry < local_C - 1e-12:
                    local_C = ctry
                    local_h = htry

        if local_C < C - 1e-10:
            h, C, no_improve = local_h, local_C, 0
        else:
            no_improve += 1
            if no_improve > 8 and evals + 2 < int(max_evals) and rem > 2:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                while j == i:
                    j = int(rng.randint(n))
                a = float(h[i])
                b = float(h[j])
                if a > b:
                    room = min(a, 1.0 - b)
                else:
                    room = min(b, 1.0 - a)
                if room > 1e-8:
                    delta = 0.10 * rng.uniform(0.25, 1.0) * room
                    if delta > 0.0:
                        htry = h.copy()
                        if a > b:
                            htry[i] -= delta
                            htry[j] += delta
                        else:
                            htry[j] -= delta
                            htry[i] += delta
                        htry = project_feasible(htry)
                        ctry = compute_c5(htry)
                        evals += 1
                        if ctry < C - 1e-10:
                            h, C, no_improve = htry, ctry, 0
        mode += 1
    return h, C


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(int(seed))

    best_h = None
    best_c = float("inf")
    best_n = 0

    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent is not None and parent.size > 0:
            parent = project_feasible(parent)
            parent = parent.copy()
            cpar = compute_c5(parent)
            best_h = parent
            best_c = cpar
            best_n = int(parent.size)
    except Exception:
        best_h = None

    if best_h is None or best_h.size == 0:
        n0 = 200
        if n0 % 2:
            n0 -= 1
        best_h = default_binary(n0)
        best_c = compute_c5(best_h)
        best_n = int(best_h.size)

    if best_n % 2:
        best_n = 2 * (int(best_n) // 2)
        best_h = interp_h(best_h, best_n)
        best_c = compute_c5(best_h)

    n_base = int(best_n)

    cur = best_h.copy()
    if time_remaining(start_time, budget_s) > 60:
        cur, ccur = active_refine(cur, n_base, rng, start_time, budget_s, 4, 200)
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = int(cur.size)

    rem = time_remaining(start_time, budget_s)
    if rem > 260 and n_base <= 320:
        cur, ccur = smooth_refine(
            cur, n_base, rng, start_time, budget_s,
            [60.0, 200.0, 600.0], [300, 350, 500]
        )
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = int(cur.size)

    rem = time_remaining(start_time, budget_s)
    if rem > 130:
        max_evals = 2500 if n_base <= 200 else 1500
        cur, ccur = exact_line_search(cur, n_base, rng, start_time, budget_s, max_evals)
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = int(cur.size)

    ns = candidate_n_list(n_base)
    for n in ns:
        rem = time_remaining(start_time, budget_s)
        if rem < 130:
            break
        n = int(n)
        if n == n_base:
            continue
        if n > 640 and rem < 600:
            continue
        if n > 512 and rem < 450:
            continue
        cur = interp_h(best_h, n)
        ccur = compute_c5(cur)
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = n

        if n <= 320:
            max_rounds = 4
            maxiter = 200
        elif n <= 512:
            max_rounds = 2
            maxiter = 120
        else:
            max_rounds = 1
            maxiter = 35

        cur, ccur = active_refine(cur, n, rng, start_time, budget_s, max_rounds, maxiter)
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = int(cur.size)

        rem = time_remaining(start_time, budget_s)
        if rem > 160 and n <= 320:
            cur, ccur = smooth_refine(
                cur, n, rng, start_time, budget_s,
                [100.0, 300.0], [260, 320]
            )
            if ccur < best_c:
                best_c = ccur
                best_h = cur.copy()
                best_n = int(cur.size)

        rem = time_remaining(start_time, budget_s)
        if rem > 100:
            max_evals = 1300 if n <= 320 else (700 if n <= 512 else 300)
            cur, ccur = exact_line_search(cur, n, rng, start_time, budget_s, max_evals)
            if ccur < best_c:
                best_c = ccur
                best_h = cur.copy()
                best_n = int(cur.size)

    rem = time_remaining(start_time, budget_s)
    if rem > 480:
        expl_n = n_base if n_base <= 400 else 400
        if expl_n < 160:
            expl_n = 160

        local_h = interp_h(best_h, expl_n)
        local_c = compute_c5(local_h)
        if local_c < best_c:
            best_c = local_c
            best_h = local_h.copy()
            best_n = int(local_h.size)

        n_starts = int(min(8, max(2, (rem - 280) // 160)))
        for i in range(n_starts):
            rem = time_remaining(start_time, budget_s)
            if rem < 170:
                break
            if i == 0:
                h0 = interp_h(best_h, expl_n)
            elif i % 3 == 1:
                h0 = make_binary_random(expl_n, rng)
            else:
                kmax = 8 if expl_n <= 256 else 12
                h0 = make_fourier_start(expl_n, rng, kmax, 0.40)

            h0, c0 = active_refine(h0, expl_n, rng, start_time, budget_s, 2, 90)
            if c0 < local_c:
                local_h = h0.copy()
                local_c = c0
                if c0 < best_c:
                    best_c = c0
                    best_h = h0.copy()
                    best_n = int(h0.size)

            rem = time_remaining(start_time, budget_s)
            if rem > 100:
                max_evals = 900 if expl_n <= 256 else 550
                h0, c0 = exact_line_search(h0, expl_n, rng, start_time, budget_s, max_evals)
                if c0 < local_c:
                    local_h = h0.copy()
                    local_c = c0
                    if c0 < best_c:
                        best_c = c0
                        best_h = h0.copy()
                        best_n = int(h0.size)

            rem = time_remaining(start_time, budget_s)
            if c0 < best_c + 1e-5 and rem > 150 and expl_n <= 320:
                h0, c0 = smooth_refine(
                    h0, expl_n, rng, start_time, budget_s,
                    [220.0], [220]
                )
                if c0 < best_c:
                    best_c = c0
                    best_h = h0.copy()
                    best_n = int(h0.size)

    rem = time_remaining(start_time, budget_s)
    if rem > 270:
        n_pol = int(best_n)
        if n_pol < 400:
            n_pol = 400 if rem > 430 else n_pol
        if n_pol > 800:
            n_pol = 800
        if n_pol < best_n:
            n_pol = best_n

        cur = interp_h(best_h, n_pol)
        ccur = compute_c5(cur)
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = int(cur.size)

        if n_pol <= 400:
            max_rounds = 3
            maxiter = 140
        else:
            max_rounds = 1
            maxiter = 35

        cur, ccur = active_refine(cur, n_pol, rng, start_time, budget_s, max_rounds, maxiter)
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = int(cur.size)

        rem = time_remaining(start_time, budget_s)
        if rem > 120 and n_pol <= 400:
            cur, ccur = smooth_refine(cur, n_pol, rng, start_time, budget_s, [320.0], [220])
            if ccur < best_c:
                best_c = ccur
                best_h = cur.copy()
                best_n = int(cur.size)

        rem = time_remaining(start_time, budget_s)
        if rem > 90:
            max_evals = 850 if n_pol <= 400 else 350
            cur, ccur = exact_line_search(cur, n_pol, rng, start_time, budget_s, max_evals)
            if ccur < best_c:
                best_c = ccur
                best_h = cur.copy()
                best_n = int(cur.size)

    rem = time_remaining(start_time, budget_s)
    if rem > 320 and best_n < 700:
        n2 = min(800, max(int(best_n) * 2, 400))
        cur = interp_h(best_h, n2)
        ccur = compute_c5(cur)
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = int(cur.size)
        if n2 <= 512:
            max_rounds = 2
            maxiter = 120
        else:
            max_rounds = 1
            maxiter = 35
        cur, ccur = active_refine(cur, n2, rng, start_time, budget_s, max_rounds, maxiter)
        if ccur < best_c:
            best_c = ccur
            best_h = cur.copy()
            best_n = int(cur.size)

    rem = time_remaining(start_time, budget_s)
    if rem > 18:
        hr = round_binary(best_h)
        cr = compute_c5(hr)
        if cr < best_c:
            best_c = cr
            best_h = hr.copy()
            best_n = int(hr.size)

    best_h = project_feasible(best_h)
    final_c5 = compute_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        if ev is not None:
            ev = float(ev)
            if np.isfinite(ev):
                final_c5 = min(final_c5, ev)
    except Exception:
        pass

    if final_c5 < 0.0:
        final_c5 = 0.0
    if final_c5 > 1.0 and final_c5 < 1.0 + 1e-8:
        final_c5 = 1.0

    return (best_h, float(final_c5), int(best_h.size))
```
