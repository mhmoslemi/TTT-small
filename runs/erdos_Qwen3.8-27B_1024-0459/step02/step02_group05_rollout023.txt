    best_h = h.copy()

        c5 = np.max(c) * (2.0/n)

            if h[i] <= 1e-6:
                continue

                j = order_j[jj]
                if j == i or h[j] >= 1.0 - 1e-6:
                    continue
                D = G[j, :] - G[i, :]
                for delta in steps:
                    if h[i] - delta < 0 or h[j] + delta > 1.0:
                        continue
                    new_c = c + delta * D
                    new_max = np.max(new_c)
                    if new_max < np.max(c) - 1e-14:
                        # Accept
                        h[i] -= delta
                        h[j] += delta
                        c = new_c
                        delta_changes = np.zeros(n)
                        delta_changes[i] = -delta
                        delta_changes[j] = delta
                        update_G(G, delta_changes, n)
                        best_c5 = np.max(c) * dx
                        improved = True
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not improved:
            break
    return h, best_c5

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):

def compute_G(h, n):

    for i in range(n):

        m_vals = np.arange(i, i+n)
        G[i, m_vals] += 1.0 - h[j_vals]
        m_vals2 = np.arange(n-1-i, 2*n-1-i)
        j_vals2 = np.arange(n)
        G[i, m_vals2] -= h[j_vals2]
    return G

    p = np.arange(n)
    m1 = p + n - 1 - idx
    G[p, m1] -= delta
    m2 = idx - p + n - 1
    G[p, m2] -= delta
