def simulated_annealing(h_init, n, rng, start_temp=1.0, end_temp=0.01, n_iter=100000, timeout=60.0):
    h = np.round(h_init).astype(float)
    h = make_binary_feasible(h, n)
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    start_t = time.time()
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    target = n // 2
    
    best_h = h.copy()
    
    for it in range(n_iter):
        if time.time() - start_t > timeout:
            break
        
        t_frac = it / max(n_iter - 1, 1)
        temp = start_temp * (end_temp / start_temp) ** t_frac
        
        # Choose move: single flip or double flip
        if rng.random() < 0.7:
            # Single swap
            i = rng.choice(list(ones))
            j = rng.choice(list(zeros))
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / (temp * 0.01)):
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                current_c5 = new_c5
                ones.discard(i); zeros.add(i)
                zeros.discard(j); ones.add(j)
            else:
                h[i], h[j] = 1.0, 0.0
        else:
            # Double swap: flip two 1s to 0 and two 0s to 1
            if len(ones) < 2 or len(zeros) < 2:
                continue
            pair1 = rng.choice(list(ones), 2, replace=False)
            pair2 = rng.choice(list(zeros), 2, replace=False)
            h[pair1[0]] = 0.0; h[pair1[1]] = 0.0
            h[pair2[0]] = 1.0; h[pair2[1]] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / (temp * 0.02)):
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                current_c5 = new_c5
                for i in pair1:
                    ones.discard(i); zeros.add(i)
                for j in pair2:
                    zeros.discard(j); ones.add(j)
            else:
                h[pair1[0]] = 1.0; h[pair1[1]] = 1.0
                h[pair2[0]] = 0.0; h[pair2[1]] = 0.0
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_binary_feasible(h, rng):
    n = len(h)
    h = np.round(h).astype(float)
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        idx = rng.choice(ones, excess, replace=False)
        h[idx] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        idx = rng.choice(zeros, deficit, replace=False)
        h[idx] = 1.0
    return h

def make_feasible(h, n):

def simulated_annealing(h_init, n, rng, timeout=60.0, kick_interval=5000, kick_strength=5):
    h = np.round(h_init).astype(float)
    h = make_binary_feasible(h, rng)

    current_c5 = best_c5
    best_h = h.copy()

    zeros_list = list(np.where(h == 0.0)[0])

        # Periodic kick: large random perturbation
        if kick_interval > 0 and it % kick_interval == 0:
            kick = min(kick_strength, n // 10)
            if len(ones_list) >= kick and len(zeros_list) >= kick:
                for _ in range(kick):
                    if len(ones_list) == 0 or len(zeros_list) == 0:
                        break

                    zi = rng.randint(0, len(zeros_list))

                    zi_val = zeros_list[zi]
                    h[oi_val] = 0.0
                    h[zi_val] = 1.0
                    ones_list[oi] = zeros_list.pop(zi)
                    zeros_list[oi] = ones_list.pop(oi)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            continue
        
        # Choose move: single swap (70%) or double swap (30%)
        r = rng.random()
        if r < 0.70:
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            oi = rng.randint(0, len(ones_list))
            zi = rng.randint(0, len(zeros_list))
            pi = ones_list[oi]
            pj = zeros_list[zi]
            h[pi] = 0.0
            h[pj] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < -1e-15 or rng.random() < np.exp(-delta / max(temp, 1e-8)):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                ones_list[oi] = zeros_list.pop(zi)
                zeros_list[oi] = ones_list.pop(oi)
            else:
                h[pi] = 1.0
                h[pj] = 0.0
        else:
            if len(ones_list) < 2 or len(zeros_list) < 2:
                break
            # Double swap
            o1 = rng.randint(0, len(ones_list))
            o2 = rng.randint(0, len(ones_list) - 1)
            if o2 >= o1:
                o2 += 1
            z1 = rng.randint(0, len(zeros_list))
            z2 = rng.randint(0, len(zeros_list) - 1)
            if z2 >= z1:
                z2 += 1
            p1, p2 = ones_list[o1], ones_list[o2]
            q1, q2 = zeros_list[z1], zeros_list[z2]
            if len(set([p1, p2, q1, q2])) < 4:
                continue
            h[p1] = 0.0; h[p2] = 0.0
            h[q1] = 1.0; h[q2] = 1.0
            new_c5 = compute_c5(h)

                ones_list[o1] = zeros_list.pop(z1)

                    ones_list[o2] = zeros_list.pop(z2 - 1)
                else:
                    ones_list[o1] = zeros_list.pop(z1 - (1 if z1 > o1 else 0))
                    ones_list[o2] = zeros_list.pop(z2 - (1 if z2 > o2 else 0))

                h[p1] = 1.0; h[p2] = 1.0

def slsqp_polish(h_init, n, max_iter=2000):
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr)
    
    def obj_jac(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Gradient of max correlation element
        max_idx = np.argmax(corr)
        # corr[m] = sum_i h[i]*(1-h[i+m-(n-1)])
        # dcorr[m]/dh[j] depends on m
        m = max_idx
        k = m - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            # corr = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            for i in range(n - k):
                grad[i] += (1.0 - h[i + k])
                grad[i + k] += (-h[i])
        else:
            # corr = sum_{i=-k}^{n-1} h[i]*(1-h[i+k])
            for i in range(-k, n):
                grad[i] += (1.0 - h[i + k])
                grad[i + k] += (-h[i])
        return grad
    
    cons = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
         'jac': lambda x: np.ones(n)}
    ]
    
    try:
        res = minimize(obj, h0, jac=obj_jac, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h_init, compute_c5(h_init)

        h_p = make_binary_feasible(np.round(interp_h(h_parent, n)), rng)
        inits.append(h_p)
    inits.append(thue_morse_init(n, rng))
    inits.append(antisymmetric_init(n, rng))
    for s in [1, 2, 3, 5, 7]:
        inits.append(phased_tm_init(n, s, rng))
    for _ in range(3):
        inits.append(random_binary(n, rng))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None
    
    best_c5_overall = 1.0
    best_h_overall = None
    best_n_overall = 256
    
    # Process multiple resolutions
    for n in [128, 256]:
        if remaining() < 100:
            break
        
        # Add budget allocation based on n
        if n == 128:
            sa_budget = min(200, remaining() - 60)
            target_budget = min(100, remaining() - 60)
            seg_budget = min(80, remaining() - 40)
            polish_budget = 60
        else:
            sa_budget = min(300, remaining() - 60)
            target_budget = min(150, remaining() - 60)
            seg_budget = min(100, remaining() - 40)
            polish_budget = 80
        
        inits = generate_inits(n, rng, h_parent)
        
        best_h_n = None
        best_c5_n = 1.0
        
        # Score initial solutions
        for h0 in inits:
            c5_val = compute_c5(h0)
            if c5_val < best_c5_n:
                best_c5_n = c5_val
                best_h_n = h0.copy()
        
        # Phase 1: Standard SA with kicks (multiple inits)
        sa_per_init = max(30, int(sa_budget / min(len(inits), 4)))
        for h0 in inits[:4]:
            if remaining() < 50:
                break
            h_sa, c5_sa = simulated_annealing(h0, n, rng, timeout=sa_per_init,
                                               kick_interval=3000, kick_strength=n // 32)
            if c5_sa < best_c5_n:
                best_c5_n = c5_sa
                best_h_n = h_sa.copy()
        
        # Phase 2: SA from best found so far
        if remaining() > 60:
            h_sa2, c5_sa2 = simulated_annealing(best_h_n, n, rng,
                                                 timeout=min(sa_budget // 2, remaining() - 40),
                                                 kick_interval=4000, kick_strength=n // 24)
            if c5_sa2 < best_c5_n:
                best_c5_n = c5_sa2
                best_h_n = h_sa2.copy()
        
        # Phase 3: Targeted SA (bias toward worst shifts)
        if remaining() > 80:
            h_tar, c5_tar = targeted_sa(best_h_n, n, rng,
                                         timeout=min(target_budget, remaining() - 40))
            if c5_tar < best_c5_n:
                best_c5_n = c5_tar
                best_h_n = h_tar.copy()
            # Also from a different init
            if remaining() > 60 and len(inits) > 1:
                h_tar2, c5_tar2 = targeted_sa(inits[1] if len(inits) > 1 else inits[0],
                                               n, rng, timeout=min(target_budget // 2, remaining() - 40))
                if c5_tar2 < best_c5_n:
                    best_c5_n = c5_tar2
                    best_h_n = h_tar2.copy()
        
        # Phase 4: Segment swap SA
        if remaining() > 60:
            h_seg, c5_seg = segment_swap_sa(best_h_n, n, rng,
                                             timeout=min(seg_budget, remaining() - 40))
            if c5_seg < best_c5_n:
                best_c5_n = c5_seg
                best_h_n = h_seg.copy()
        
        # Phase 5: More SA from best (fine-tuning)
        if remaining() > 50:
            h_sa3, c5_sa3 = simulated_annealing(best_h_n, n, rng,
                                                 timeout=min(60, remaining() - 30),
                                                 kick_interval=10000, kick_strength=3)
            if c5_sa3 < best_c5_n:
                best_c5_n = c5_sa3
                best_h_n = h_sa3.copy()
        
        # Phase 6: SLSQP polish (non-binary)
        if remaining() > 30:
            h_pol, c5_pol = slsqp_polish(best_h_n, n, max_iter=2000)
            if c5_pol < best_c5_n:
                best_c5_n = c5_pol
                best_h_n = h_pol.copy()
        
        # Phase 7: Quick binary SA after polish
        if remaining() > 20:
            h_bin = make_binary_feasible(np.round(best_h_n), rng)
            h_sa4, c5_sa4 = simulated_annealing(h_bin, n, rng,
                                                 timeout=min(30, remaining() - 10),
                                                 kick_interval=5000, kick_strength=n // 32)
            if c5_sa4 < best_c5_n:
                best_c5_n = c5_sa4
                best_h_n = h_sa4.copy()
        
        # Update overall best
        if best_h_n is not None and best_c5_n < best_c5_overall:
            best_c5_overall = best_c5_n
            best_h_overall = best_h_n.copy()
            best_n_overall = n
    
    # If we still have time, do a final SLSQP on the best
    if remaining() > 20 and best_h_overall is not None:
        h_final, c5_final = slsqp_polish(best_h_overall, best_n_overall, max_iter=3000)
        if c5_final < best_c5_overall:
            best_c5_overall = c5_final
            best_h_overall = h_final.copy()
    
    # One final binary SA
    if remaining() > 15 and best_h_overall is not None:
        h_bin_final = make_binary_feasible(np.round(best_h_overall), rng)
        c5_bin = compute_c5(h_bin_final)
        if c5_bin < best_c5_overall:
            best_c5_overall = c5_bin
            best_h_overall = h_bin_final
    
    # Ensure feasibility
    if best_h_overall is not None:
        best_h_overall = make_feasible(best_h_overall, best_n_overall)
        best_c5_overall = compute_c5(best_h_overall)
    else:
        # Fallback: Thue-Morse at n=256
        best_h_overall = thue_morse_init(256, rng)
        best_c5_overall = compute_c5(best_h_overall)
        best_n_overall = 256
    
    return (best_h_overall, best_c5_overall, best_n_overall)
