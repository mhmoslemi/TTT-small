    """Project h onto {x: 0 <= x[i] <= 1, sum(x) = target}."""
    n = len(h)
    # This is a convex optimization problem: minimize ||x - h||^2 subject to constraints
    # Use bisection on the Lagrange multiplier
    lo, hi = -1e6, 1e6
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Correct any numerical error
    diff = target - np.sum(x)
    # Distribute among non-saturated

def project_feasible(h, n):

    """Simulated annealing for binary solutions."""
    # Initialize from h_init (round to binary)
    h = h_init.copy()

    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h == 1)[0]
        order = np.argsort(h_init[ones])[:n_ones - n_want]
        h[ones[order]] = 0
    elif n_ones < n_want:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-h_init[zeros])[:n_want - n_ones]

        return np.max(corr) * (2.0 / n)

    best_c5 = current_c5
    best_h = h.copy()

        ones = np.where(h == 1)[0]

        j = rng.choice(len(zeros))

        h[old_j] = 1

This is O(n^2) per step (try all n/2 ones × n/2 zeros = n^2/4 swaps, each O(n) to evaluate). For n=192, that's about 9216 × 192 ≈ 1.8M operations per step. If each step is fast (in C), we can do many steps.

Actually, we can be smarter: for each swap (a, b), the change in C5 is:
ΔC5 = max_k (corr[k] + ΔC[k]) - max_k(corr[k])) * dx

where ΔC[k] = -(1-h[a+off_k]) + h[a-off_k] + (1-h[b+off_k]) - h[b-off_k] (with boundary handling)

We can precompute the "contribution" of each position to each lag, and then the effect of a swap is just lookup + combine.

But this is getting complex. Let me just use the direct approach (recompute correlate after each trial swap) and see if it's fast enough.

For n=192, np.correlate takes about 5-10 μs (in C). Trying all n^2/4 = 9216 swaps means about 50-100 μs per step. In 60 seconds, we can do about 600k-1.2M steps. That's plenty for hill climbing.

Let me implement hill climbing for the binary case:

```python
def binary_hill_climb(n, h_init, budget_s, rng):
    """Hill climbing for binary solutions."""
    h = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h == 1)[0]
        order = np.argsort(h_init[ones])[:n_ones - n_want]
        h[ones[order]] = 0
    elif n_ones < n_want:
        zeros = np.where(h == 0)[0]
        order = np.argsort(-h_init[zeros])[:n_want - n_ones]
        h[zeros[order]] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)
    current_c5 = current_max * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    t_start = time.time()
    no_improve = 0
    
    while time.time() - t_start < budget_s:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_delta = 0
        best_swap = None
        
        for a in ones:
            h[a] = 0
            for b in zeros:
                h[b] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = np.max(new_corr)
                delta = (current_max - new_max) * dx
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (a, b)
                    if best_delta > 0.001:  # Early exit if good improvement found
                        break
                h[b] = 0
            if best_delta > 0.001:
                break
            h[a] = 1
        
        if best_swap is not None and best_delta > 1e-10:
            a, b = best_swap
            h[a] = 0
            h[b] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            current_max = np.max(corr)
            current_c5 = current_max * dx
            no_improve = 0
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 3:
                break
    
    return best_h, best_c5
```

In 60 seconds, we can do about 8600 steps. Each step is a "full sweep" over all swaps. That might be enough for hill climbing.
