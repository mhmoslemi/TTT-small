import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.full(n_dst, 0.5)
    x_old = np.arange(len(h_src), dtype=float)
    x_new = np.linspace(0, len(h_src) - 1, n_dst)
    return np.interp(x_new, x_old, h_src)

def coordinate_descent(h, n, n_sweeps=100, grid_n=80, time_limit_s=60, rng=None):
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    m_vals = np.arange(2 * n - 1, dtype=int) - (n - 1)
    k0 = n - 1
    
    grid = np.linspace(0.0, 1.0, grid_n)
    start = time.time()
    
    for sweep in range(n_sweeps):
        if (time.time() - start) > time_limit_s:
            break
        order = np.arange(n)
        if rng is not None and sweep > 0:
            rng.shuffle(order)
        
        any_improve = False
        for i in order:
            old = h[i]
            
            # Sensitivity: how corr[k] changes per unit change in h[i]
            m = m_vals
            valid1 = (i + m >= 0) & (i + m < n)
            valid2 = (i - m >= 0) & (i - m < n)
            m_nz = (m != 0)
            
            sens = np.zeros(2 * n - 1)
            msk1 = valid1 & m_nz
            msk2 = valid2 & m_nz
            if np.any(msk1):
                sens[msk1] += (1.0 - h[i + m[msk1]])
            if np.any(msk2):
                sens[msk2] -= h[i - m[msk2]]

            base_with_sens[k0] = corr[k0]  # will be overridden

            # shape: (grid_n, 2n-1)
            new_corrs = corr[np.newaxis, :] + sens[np.newaxis, :] * deltas[:, np.newaxis]
            # Fix k0 column
            new_corrs[:, k0] = corr[k0] + grid * (1.0 - grid) - old * (1.0 - old)
            maxes = np.max(new_corrs, axis=1)

            if not any_improve:
                # Try a few more sweeps, then break
                pass
    
    return h, best_c5

def binary_sa(h_init, n, n_steps=50000, t_start=1.0, t_end=0.001, rng=None, time_limit_s=30):
    h = h_init.copy()
    # Threshold to binary
    h = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) != n // 2:
        # Fix
        if len(ones) > n // 2:
            excess = len(ones) - n // 2
            to_flip = rng.choice(ones, excess, replace=False)
            h[to_flip] = 0
        else:
            deficit = n // 2 - len(ones)
            to_flip = rng.choice(zeros, deficit, replace=False)
            h[to_flip] = 1
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for step in range(n_steps):
        if (time.time() - start) > time_limit_s:
            break
        T = t_start * (t_end / t_start) ** (step / n_steps)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        
        if c5_new <= best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-10)):
            pass  # accept
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.full(n_dst, 0.5)
    x_old = np.arange(len(h_src), dtype=float)
    x_new = np.linspace(0, len(h_src) - 1, n_dst)
    return np.interp(x_new, x_old, h_src)


def coordinate_descent(h, n, n_sweeps=100, grid_n=100, time_limit_s=60, rng=None):
    h = h.astype(float).copy()
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    m_vals = np.arange(2 * n - 1, dtype=int) - (n - 1)
    k0 = n - 1
    grid = np.linspace(0.0, 1.0, grid_n)
    start = time.time()

    for sweep in range(n_sweeps):
        if (time.time() - start) > time_limit_s:
            break
        order = np.arange(n)
        if rng is not None and sweep > 0:
            rng.shuffle(order)

        any_improve = False
        for i in order:
            old = h[i]
            m = m_vals
            valid1 = (i + m >= 0) & (i + m < n)
            valid2 = (i - m >= 0) & (i - m < n)
            m_nz = (m != 0)

            sens = np.zeros(2 * n - 1)
            msk1 = valid1 & m_nz
            msk2 = valid2 & m_nz
            if np.any(msk1):
                sens[msk1] += (1.0 - h[i + m[msk1]])
            if np.any(msk2):
                sens[msk2] -= h[i - m[msk2]]

            current_max = np.max(corr)
            best_t = old
            best_obj = current_max

            deltas = grid - old
            new_corrs = corr[np.newaxis, :] + sens[np.newaxis, :] * deltas[:, np.newaxis]
            new_corrs[:, k0] = corr[k0] + grid * (1.0 - grid) - old * (1.0 - old)
            maxes = np.max(new_corrs, axis=1)

            idx_best = np.argmin(maxes)
            if maxes[idx_best] < best_obj - 1e-14:
                best_t = grid[idx_best]
                best_obj = maxes[idx_best]
                any_improve = True

            if best_t != old:
                delta = best_t - old
                corr += sens * delta
                corr[k0] += best_t * (1.0 - best_t) - old * (1.0 - old)
                h[i] = best_t

        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5

    return h, best_c5

def smooth_gradient_solve(h_init, n, time_limit_s=40, betas=(50, 200, 800, 3000)):
    current = h_init.astype(float).copy()
    current = make_feasible(current, n)
    best_c5 = compute_c5(current)
    best_h = current.copy()
    start = time.time()

    for beta in betas:
        if (time.time() - start) > time_limit_s:
            break
        bounds = [(0.0, 1.0)] * n
        cons_list = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0,
                      'jac': lambda x, n=n: np.ones(n)}]

        def obj_smooth(x, beta=beta, n=n):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            shifted = beta * (corr - cmax)
            return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

        try:
            res = minimize(obj_smooth, current, method='SLSQP', bounds=bounds,
                           constraints=cons_list,
                           options={'maxiter': 3000, 'ftol': 1e-14, 'iprint': 0})
            if res.fun < 1e10:
                current = make_feasible(res.x, n)
                c5 = compute_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
        except Exception:
            pass

    return best_h, best_c5

    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()

    for _ in range(n_rounds):
        if (time.time() - start) > time_limit_s:
            break
        improved = False
        for i in range(n - 1):
            if abs(h[i] - h[i + 1]) < 1e-8:
                continue
            for step in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                amt = min(step, abs(h[i] - h[i + 1]) / 2.0)

                    continue

                    h_try[i] -= amt
                    h_try[i + 1] += amt

                    h_try = h.copy()
                    h_try[i] += amt
                    h_try[i + 1] -= amt
                h_try = np.clip(h_try, 0.0, 1.0)

                if c5_try < best_c5 - 1e-14:

                    best_c5 = c5_try
