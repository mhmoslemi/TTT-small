def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")

def project_density(x, target):
    # Binary search for threshold to make sum = target
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-10:
                free2 = np.where((h > 0.0) & (h < 1.0))[0]
                if free2.size > 0:
                    h[free2[0]] += res2
                    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def corr_grad_lag(h, k):
    """Gradient of corr at lag k w.r.t. h"""
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

def get_active_lags(corr, top_frac=0.08, min_k=5):
    """Get indices of top lags in correlation"""
    center = len(corr) // 2
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    top_idx = np.argsort(corr)[-k_count:]
    return top_idx

def active_gradient(h, top_frac=0.08, min_k=5):
    """Compute gradient targeting active (top) lags"""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    center = n - 1
    top_idx = get_active_lags(corr, top_frac, min_k)
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    scale = max(1e-10, dv.max() + 1e-10)
    w = np.exp(dv / scale)
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for i, t in enumerate(top_idx):
        k = int(t) - center
        g += w[i] * corr_grad_lag(h, k)
    return g, corr

            cand[int(i)] += float(rng.uniform(-0.04, 0.04))
        cand = project_density(cand, target)
        c = evaluate_erdos(cand)
        delta = c - current_c
        if delta < 0 or float(rng.random()) < np.exp(-delta / max(T, 1e-10)):
            current = cand
            current_c = c
            if c < best_c - 1e-14:
                best = cand.copy()
                best_c = c
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 300:
            break
    best = project_density(best, target)
    return best, evaluate_erdos(best)


def binary_push(h, target, thresh):
    n = h.size
    h2 = h.copy()
    changed = False
    for i in range(n):
        if h2[i] < thresh:
            h2[i] = 0.0
            changed = True
        elif h2[i] > 1.0 - thresh:
            h2[i] = 1.0
            changed = True
    if not changed:
        return h
    return project_density(h2, target)


def coordinate_descent(h, target, deadline, max_iter, top_frac, min_k, rng):
    n = h.size
    best = h.copy()
    best_c = evaluate_erdos(best)
    stall = 0
    order = rng.permutation(n)
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        idx = int(order[_ % n])
        best_c_local = best_c
        best_val = best[idx]
        for dv in (-0.05, -0.02, -0.01, 0.01, 0.02, 0.05):
            newVal = best_val + dv
            if newVal < -1e-12 or newVal > 1 + 1e-12:
                continue
            cand = best.copy()
            cand[idx] = newVal
            cand = project_density(cand, target)
            c = evaluate_erdos(cand)
            if c < best_c_local - 1e-13:
                best = cand
                best_c_local = c
                best_val = cand[idx]
        if best_c_local < best_c - 1e-14:
            best_c = best_c_local
            stall = 0
        else:
            stall += 1
            if stall > 3 * n:
                break
    return best, best_c


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = t_start + max(1.0, budget - 25.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_
    else:
        n0 = 200
        x = (np.arange(n0) + 0.5) * (2.0 / float(n0))
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.1 * np.sin(2 * np.pi * 7 * x / 2 + 0.7)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    n_work = best_n
    target_w = 0.5 * n_work
    h_cur = best_h.copy()

    # Phase 1: Gradient descent at parent resolution
    rem = deadline - time.perf_counter()
    if rem > 30:
        t_sub = time.perf_counter() + min(80.0, rem * 0.12)
        h_cur, c = gradient_descent(h_cur, target_w, t_sub, max_iter=600, top_frac=0.07, min_k=8)
        if c < best_c:
            best_h, best_c = h_cur.copy(), c

    # Phase 2: Intensive pair swaps
    rem = deadline - time.perf_counter()
    if rem > 40:
        t_sub = time.perf_counter() + min(180.0, rem * 0.22)
        h_cur, c = pair_swap_search(h_cur, target_w, t_sub, max_iter=8000,
                                     top_frac=0.05, min_k=8, n_pos=18, n_neg=18, rng=rng)
        if c < best_c:
            best_h, best_c = h_cur.copy(), c

    # Phase 3: Binary push + re-optimize
    rem = deadline - time.perf_counter()
    if rem > 30 and n_work <= 600:
        for thresh in (0.15, 0.10, 0.08):
            h_bin = binary_push(h_cur, target_w, thresh)
            c_bin = evaluate_erdos(h_bin)
            if c_bin < best_c - 1e-6:
                h_cur = h_bin
            t_sub = time.perf_counter() + min(40.0, rem * 0.1)
            h_cur, c = pair_swap_search(h_cur, target_w, t_sub, max_iter=2000,
                                         top_frac=0.05, min_k=8, n_pos=15, n_neg=15, rng=rng)
            if c < best_c:
                best_h, best_c = h_cur.copy(), c
            rem = deadline - time.perf_counter()
            if rem < 20:
                break

    # Phase 4: More gradient + pairs
    rem = deadline - time.perf_counter()
    if rem > 25:
        t_sub = time.perf_counter() + min(60.0, rem * 0.15)
        h_cur, c = gradient_descent(h_cur, target_w, t_sub, max_iter=400, top_frac=0.04, min_k=10)
        if c < best_c:
            best_h, best_c = h_cur.copy(), c

    rem = deadline - time.perf_counter()
    if rem > 40:
        t_sub = time.perf_counter() + min(150.0, rem * 0.25)
        h_cur, c = pair_swap_search(h_cur, target_w, t_sub, max_iter=10000,
                                     top_frac=0.04, min_k=10, n_pos=22, n_neg=22, rng=rng)
        if c < best_c:
            best_h, best_c = h_cur.copy(), c

    # Phase 5: Two-pair swaps
    rem = deadline - time.perf_counter()
    if rem > 30 and n_work <= 500:
        t_sub = time.perf_counter() + min(60.0, rem * 0.2)
        h_cur, c = two_pair_swap(h_cur, target_w, t_sub, max_iter=1500,
                                  top_frac=0.05, min_k=8, rng=rng)
        if c < best_c:
            best_h, best_c = h_cur.copy(), c

    # Phase 6: Annealing to escape local optima
    rem = deadline - time.perf_counter()
    if rem > 40:
        t_sub = time.perf_counter() + min(80.0, rem * 0.2)
        h_cur, c = annealing_phase(h_cur, target_w, t_sub, rng, max_iter=4000)
        if c < best_c:
            best_h, best_c = h_cur.copy(), c
        # Re-polish after annealing
        rem = deadline - time.perf_counter()
        if rem > 30:
            t_sub = time.perf_counter() + min(50.0, rem * 0.2)
            h_cur, c = gradient_descent(h_cur, target_w, t_sub, max_iter=300, top_frac=0.04, min_k=10)
            if c < best_c:
                best_h, best_c = h_cur.copy(), c
            rem = deadline - time.perf_counter()
            if rem > 30:
                t_sub = time.perf_counter() + min(60.0, rem * 0.3)
                h_cur, c = pair_swap_search(h_cur, target_w, t_sub, max_iter=4000,
                                             top_frac=0.03, min_k=10, n_pos=20, n_neg=20, rng=rng)
                if c < best_c:
                    best_h, best_c = h_cur.copy(), c

    # Phase 7: Coordinate descent pass
    rem = deadline - time.perf_counter()
    if rem > 30 and n_work <= 500:
        t_sub = time.perf_counter() + min(50.0, rem * 0.15)
        h_cur, c = coordinate_descent(h_cur, target_w, t_sub, max_iter=n_work * 3,
                                       top_frac=0.04, min_k=8, rng=rng)
        if c < best_c:
            best_h, best_c = h_cur.copy(), c

    # Phase 8: Try higher resolution
    rem = deadline - time.perf_counter()
    if rem > 80:
        n_hi = min(n_work * 2, 900)
        if n_hi > n_work:
            target_hi = 0.5 * n_hi
            h_hi = upsample_h(best_h, n_hi)
            # Quick gradient
            t_sub = time.perf_counter() + min(50.0, rem * 0.18)
            h_hi, c_hi = gradient_descent(h_hi, target_hi, t_sub, max_iter=300, top_frac=0.06, min_k=10)
            # Pair swaps
            rem = deadline - time.perf_counter()
            if rem > 30:
                t_sub = time.perf_counter() + min(100.0, rem * 0.35)
                h_hi, c_hi = pair_swap_search(h_hi, target_hi, t_sub, max_iter=3000,
                                               top_frac=0.05, min_k=10, n_pos=15, n_neg=15, rng=rng)
            # Final gradient
            rem = deadline - time.perf_counter()
            if rem > 20:
                t_sub = time.perf_counter() + min(30.0, rem * 0.2)
                h_hi, c_hi = gradient_descent(h_hi, target_hi, t_sub, max_iter=200, top_frac=0.04, min_k=12)
            if c_hi < best_c:
                best_h, best_c, best_n = h_hi.copy(), c_hi, n_hi

    # Phase 9: Try slightly different resolution
    rem = deadline - time.perf_counter()
    if rem > 60 and best_n >= 300:
        n_alt = best_n + 50
        if n_alt != best_n:
            target_alt = 0.5 * n_alt
            h_alt = upsample_h(best_h, n_alt)
            t_sub = time.perf_counter() + min(40.0, rem * 0.25)
            h_alt, c_alt = gradient_descent(h_alt, target_alt, t_sub, max_iter=200, top_frac=0.05, min_k=8)
            rem = deadline - time.perf_counter()
            if rem > 20:
                t_sub = time.perf_counter() + min(40.0, rem * 0.4)
                h_alt, c_alt = pair_swap_search(h_alt, target_alt, t_sub, max_iter=2000,
                                                 top_frac=0.04, min_k=10, n_pos=15, n_neg=15, rng=rng)
            if c_alt < best_c:
                best_h, best_c, best_n = h_alt.copy(), c_alt, n_alt

    # Phase 10: Final intensive polish loop
    rem = deadline - time.perf_counter()
    if rem > 10 and best_h is not None:
        n_fin = int(best_h.size)
        target_fin = 0.5 * n_fin
        h_fin = project_density(best_h, target_fin)
        for _phase in range(6):
            rem = deadline - time.perf_counter()
            if rem < 5:
                break
            t_sub = time.perf_counter() + min(20.0, rem * 0.12)
            h_fin, c = gradient_descent(h_fin, target_fin, t_sub, max_iter=150,
                                         top_frac=0.03, min_k=12)
            if c < best_c:
                best_c = c
            rem = deadline - time.perf_counter()
            if rem < 5:
                break
            t_sub = time.perf_counter() + min(30.0, rem * 0.2)
            h_fin, c = pair_swap_search(h_fin, target_fin, t_sub, max_iter=3000,
                                         top_frac=0.03, min_k=12, n_pos=25, n_neg=25, rng=rng)
            if c < best_c:
                best_h, best_c, best_n = h_fin.copy(), c, n_fin

    # Final fixup
    if best_h is None:
        fallback = project_density(np.full(100, 0.5), 50.0)
        return fallback, evaluate_erdos(fallback), 100

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
