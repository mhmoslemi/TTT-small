def get_corr(h):
    h = np.asarray(h, dtype=np.float64)
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64, copy=False)


def compute_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return 0.0
    c = get_corr(h)
    return float(np.max(c) * (2.0 / n))


def fix_sum(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(40):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def project_feasible(z, n):
    z = np.asarray(z, dtype=np.float64).copy()
    if n == 0:
        return z
    target = 0.5 * n
    h = np.clip(z, 0.0, 1.0)
    s0 = float(np.sum(h))
    if abs(s0 - target) < 1e-12:
        return h

    lo = -float(np.max(z)) - 1e-12
    hi = 1.0 - float(np.min(z)) + 1e-12

        lo, hi = -1.0, 1.0

        else:
            hi = mid

def project_subset(h, idxs, n):

                mask = lags != 0
                if np.any(mask):
                    A[lags[mask] + idx0] += 1.0 - h[idx[mask]]

                lags2 = idx - j
                mask2 = lags2 != 0
                if np.any(mask2):
                    A[lags2[mask2] + idx0] -= h[idx[mask2]]

                A[idx0] = 0.0
                q = corr[idx0] - old * (1.0 - old)
                B = corr - old * A
                B[idx0] = -np.inf

                t_star = solve_coord(A, B, q, old, sec_iters, root_iters)
                if not np.isfinite(t_star):
                    t_star = old
                t_star = float(np.clip(t_star, 0.0, 1.0))

                if abs(t_star - old) > 1e-13:
                    f_old = max(q + old - old * old, float(np.max(A * old + B)))
                    f_star = max(q + t_star - t_star * t_star, float(np.max(A * t_star + B)))
                    if f_star > f_old + 1e-10:
                        t_mid = 0.5 * (old + t_star)
                        f_mid = max(q + t_mid - t_mid * t_mid, float(np.max(A * t_mid + B)))
                        if f_mid < f_old - 1e-12:
                            t_star = t_mid
                        else:
                            t_star = old

                if abs(t_star - old) > 1e-13:
                    delta = t_star - old
                    h[j] = t_star
                    corr += A * delta
                    corr[idx0] = q + t_star - t_star * t_star

            h = project_subset(h, block_idx, n)
            if abs(float(np.sum(h)) - 0.5 * n) > 1e-9:
                h = project_feasible(h, n)

            corr = get_corr(h)
            c = float(np.max(corr) * dx)

            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

            if c > best_c + 0.004:
                h = best_h.copy()
                corr = get_corr(h)
                no_improve = 0

            if no_improve == 30 and block > 16:
                block = max(16, block // 2)
                no_improve = 0
            if no_improve > 85:
                pass

            st = en

        pass_no += 1
        if pass_no > 5000:
            break

    return best_h, best_c


def solve_b_for_mean(h, n, a):
    target = 0.5 * n
    a = float(a)
    lo = -a - 1.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(a * h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    return 0.5 * (lo + hi)


def amplitude_search(h, n):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)

    for a in CONTRAST_A:
        b = solve_b_for_mean(h, n, a)
        ht = np.clip(float(a) * h + b, 0.0, 1.0)
        ht = fix_sum(ht, n)
        c = compute_c5(ht)
        if c < best_c - 1e-12:
            best_c = c
            best_h = ht.copy()

    for p in POWER_P:
        ht = np.power(best_h, float(p))
        ht = project_feasible(ht, n)
        c = compute_c5(ht)
        if c < best_c - 1e-12:
            best_c = c
            best_h = ht.copy()

    return best_h, best_c


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=np.float64)
    if len(h_src) < 2:
        return project_feasible(np.full(n_dst, 0.5, dtype=np.float64), n_dst)
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    hh = np.interp(x_new, x_old, h_src)
    return project_feasible(hh, n_dst)


def initial_generation(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    starts = []

    if n % 4 == 0:
        starts.append(np.tile(np.array([1.0, 1.0, 0.0, 0.0], dtype=np.float64), n // 4))
        starts.append(np.tile(np.array([1.0, 0.0, 1.0, 0.0], dtype=np.float64), n // 4))

    if n % 2 == 0:
        hb = np.zeros(n, dtype=np.float64)
        hb[:n // 4] = 1.0
        hb[n // 2:n // 2 + n // 4] = 1.0
        starts.append(hb)

    orig = np.arange(n, dtype=np.uint64)
    parity = orig.copy()
    for s in (1, 2, 4, 8, 16, 32, 64, 128, 256):
        parity ^= (orig >> np.uint64(s))
    starts.append((parity & np.uint64(1)).astype(np.float64))

    if n > 1:
        rr = np.zeros(n, dtype=np.float64)
        if n % 2 == 0:
            pos = rng.choice(n, n // 2, replace=False)
        else:
            pos = rng.choice(n, (n - 1) // 2, replace=False)
        rr[pos] = 1.0
        starts.append(rr)

    starts.append(0.5 + 0.35 * np.sin(2.0 * np.pi * 3.0 * x) + 0.15 * np.sin(
        2.0 * np.pi * 7.0 * x + (seed % 100) * 0.11))
    starts.append(0.5 + 0.40 * np.sin(2.0 * np.pi * 5.0 * x + 0.4) *
                 np.sin(2.0 * np.pi * 11.0 * x + 1.1))

    best_h = project_feasible(np.full(n, 0.5, dtype=np.float64), n)
    best_c = compute_c5(best_h)
    for s in starts:
        hh = project_feasible(s, n)
        c = compute_c5(hh)
        if c < best_c - 1e-12:
            best_c = c
            best_h = hh
    return best_h


def binary_from_h(h, n):
    h = np.asarray(h, dtype=np.float64)
    hb = (h > 0.5).astype(np.float64)
    if n % 2 == 1:
        return project_feasible(hb, n)
    target = n // 2
    cur = int(round(np.sum(hb)))
    deficit = target - cur
    if deficit > 0:
        zeros = np.where(hb == 0.0)[0]
        if len(zeros) > 0:
            order = np.argsort(-h[zeros])
            take = min(deficit, len(zeros))
            hb[zeros[order[:take]]] = 1.0
    elif deficit < 0:
        ones = np.where(hb == 1.0)[0]
        if len(ones) > 0:
            order = np.argsort(h[ones])
            take = min(-deficit, len(ones))
            hb[ones[order[:take]]] = 0.0
    return fix_sum(hb, n)


def run(seed=42, budget_s=1300, **kwargs):
    start_all = time.time()

    h_par = None
    try:
        h_par = np.array(initial_h_values, dtype=np.float64)
        if h_par is None or h_par.size == 0:
            h_par = None
    except Exception:
        h_par = None

    if h_par is None:
        n0 = 160
        h0 = initial_generation(n0, seed)
    else:
        n0 = int(len(h_par))
        h0 = project_feasible(h_par, n0)

    best_h = h0
    best_n = n0
    best_c = compute_c5(best_h)

    rem = budget_s - (time.time() - start_all)
    if rem > 80:
        ha, ca = amplitude_search(best_h, best_n)
        if ca < best_c - 1e-12:
            best_h, best_c = ha, ca

    rem = budget_s - (time.time() - start_all)
    if rem > 120:
        if rem > 800:
            t_main = min(520.0, rem - 280.0)
        elif rem > 400:
            t_main = min(300.0, rem - 140.0)
        else:
            t_main = max(20.0, rem - 50.0)
        h1, c1 = coordinate_descent(best_h, best_n, t_main, rng_seed=seed + 1)
        if c1 < best_c - 1e-12:
            best_h, best_c = h1, c1

        rem = budget_s - (time.time() - start_all)
        if rem > 60:
            ha, ca = amplitude_search(best_h, best_n)
            if ca < best_c - 1e-12:
                best_h, best_c = ha, ca

    tried = set()
    tried.add(best_n)
    candidate_ns = [best_n, 112, 128, 144, 160, 176, 192, 208, 224, 256, 288, 320, 352, 384]

    for n in candidate_ns:
        rem = budget_s - (time.time() - start_all)
        if rem < 140:
            break
        if n in tried:
            continue
        if n > 384:
            continue
        if n > 256 and rem < 220:
            continue

        if n == best_n:
            h_start = best_h.copy()
        else:
            h_start = interpolate_to_n(best_h, n)

        rem = budget_s - (time.time() - start_all)
        if rem > 60:
            h_start, _ = amplitude_search(h_start, n)

        rem = budget_s - (time.time() - start_all)
        if rem < 100:
            continue

        if n <= 220:
            t_n = min(120.0, rem - 100.0)
        elif n <= 320:
            t_n = min(90.0, rem - 100.0)
        else:
            t_n = min(60.0, rem - 100.0)
        if t_n > 5:
            hn, cn = coordinate_descent(h_start, n, t_n, rng_seed=seed + 17 + n)
            if cn < best_c - 1e-12:
                best_h, best_c, best_n = hn, cn, n

        rem = budget_s - (time.time() - start_all)
        if rem > 40:
            hcn, ccn = amplitude_search(best_h if n == best_n else interpolate_to_n(best_h, n), n)
            if n == best_n:
                if ccn < best_c - 1e-12:
                    best_h, best_c = hcn, ccn
            else:
                if ccn < best_c - 1e-12:
                    best_h, best_c, best_n = hcn, ccn, n

        tried.add(n)

    rem = budget_s - (time.time() - start_all)
    if rem > 160:
        rng = np.random.RandomState(seed + 999)
        for amp in (0.05, 0.12, 0.25):
            rem = budget_s - (time.time() - start_all)
            if rem < 110:
                break
            noise = amp * rng.standard_normal(best_n).astype(np.float64)
            hp = project_feasible(best_h + noise, best_n)
            hp, _ = amplitude_search(hp, best_n)
            t_p = min(90.0, rem - 90.0)
            if t_p > 5:
                hpo, cpo = coordinate_descent(hp, best_n, t_p, rng_seed=seed + int(amp * 10000) + 5)
                if cpo < best_c - 1e-12:
                    best_h, best_c = hpo, cpo
                rem = budget_s - (time.time() - start_all)
                if rem > 50:
                    hca, cca = amplitude_search(best_h, best_n)
                    if cca < best_c - 1e-12:
                        best_h, best_c = hca, cca

    rem = budget_s - (time.time() - start_all)
    if rem > 130:
        hb = binary_from_h(best_h, best_n)
        hb, _ = amplitude_search(hb, best_n)
        t_b = min(70.0, rem - 90.0)
        if t_b > 5:
            hbo, cbo = coordinate_descent(hb, best_n, t_b, rng_seed=seed + 424242)
            if cbo < best_c - 1e-12:
                best_h, best_c = hbo, cbo

    rem = budget_s - (time.time() - start_all)
    if rem > 70:
        t_fin = max(15.0, rem - 25.0)
        hfin, cfin = coordinate_descent(best_h, best_n, t_fin, rng_seed=seed + 8080,
                                        block_size=max(16, best_n // 4))
        if cfin < best_c - 1e-12:
            best_h, best_c = hfin, cfin

    rem = budget_s - (time.time() - start_all)
    if rem > 30:
        hfin, cfin = amplitude_search(best_h, best_n)
        if cfin < best_c - 1e-12:
            best_h, best_c = hfin, cfin

    h_final = project_feasible(best_h, best_n)
    c_final = compute_c5(h_final)
    if c_final <= best_c + 1e-10:
        best_h, best_c = h_final, c_final

    return best_h, best_c, best_n
