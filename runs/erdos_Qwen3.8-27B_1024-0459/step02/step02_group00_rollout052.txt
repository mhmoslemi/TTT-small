```python
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            step = diff / cnt
            # Don't overshoot
            max_up = 1.0 - 1e-10
            max_down = 1e-10
            h[mask] += step
            h = np.clip(h, 0.0, 1.0)
        else:
            h[:] += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n, step_size):
    """One subgradient descent step for min max-k corr[k]"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find all correlations at (or near) the maximum
    max_val = np.max(corr)
    threshold = max_val - 1e-6  # include near-max correlations
    
    mask = corr >= threshold
    n_active = np.sum(mask)
    if n_active == 0:
        return h.copy()
    
    # Subgradient: average of gradients of active correlations
    grad = np.zeros(n)
    active_indices = np.where(mask)[0]
    
    for k in active_indices:
        offset = k - (n - 1)
        if offset >= 0:
            n_pairs = n - offset
            # corr[k] = sum_{i=0}^{n-1-offset} h[i+offset] * (1-h[i])
            # d/dh[j] for j in [offset, n): (1-h[j-offset])
            # d/dh[j] for j in [0, n-offset): -h[j+offset]
            grad[offset:] += 1.0 - h[:n_pairs]
            grad[:n_pairs] -= h[offset:]
        else:
            abs_off = -offset
            n_pairs = n - abs_off
            # corr[k] = sum_{i=abs_off}^{n-1} h[i] * (1-h[i-abs_off])
            # d/dh[j] for j in [abs_off, n): (1-h[j-abs_off])

            grad[abs_off:] += 1.0 - h[:n_pairs]
            grad[:n_pairs] -= h[abs_off:]
    
    grad /= n_active
    
    # Step

    h_new = project_feasible(h_new, n)
```

```python
        # First index (h[i]): for i in [m, n-1], ∂/∂h[i] = (1-h[i-m])
        grad[m:n] += (1.0 - h[0:n_pairs])
        # Second index (h[i-m]): for i in [m, n-1], ∂/∂h[i-m] = -h[i]
        grad[0:n_pairs] -= h[m:n]
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

_START = 0.0
_BUDGET = 1300.0

def time_left():
    return max(0.0, _BUDGET - (time.time() - _START) - 2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_single(h, k_idx, n):
    """Gradient of corr[k] w.r.t. h, where k_idx is index into full correlation output."""
    k = k_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        n_pairs = n - k
        grad[0:n_pairs] += (1.0 - h[k:k+n_pairs])
        grad[k:k+n_pairs] -= h[0:n_pairs]
    else:
        m = -k
        n_pairs = n - m
        grad[m:n] += (1.0 - h[0:n_pairs])
        grad[0:n_pairs] -= h[m:n]
    return grad

def subgradient_descent(h0, n, max_iters=5000, p_top=3):
    """Subgradient descent for min max-k corr[k]*dx"""
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Learning rate schedule
    lr_init = 2.0 / n  # scaled by dx
    lr_min = 1e-6 / n
    decay = 0.9997
    
    for it in range(max_iters):
        if time_left() < 5:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        c5_cur = max_val * dx
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Find top-p correlations
        sorted_idx = np.argsort(-corr)[:p_top]
        
        # Average gradient of top-p
        grad = np.zeros(n)
        for idx in sorted_idx:
            grad += corr_grad_single(h, idx, n)
        grad /= p_top
        
        # Step
        lr = max(lr_min, lr_init * (decay ** it))
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Accept only if improvement (or with some probability for exploration)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 or (np.random.random() < 0.01 and c5_new < best_c5 * 1.005):
            h = h_new
        else:
            # Try smaller step
            for scale in [0.5, 0.1]:
                h_trial = h - lr * scale * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    h = h_trial
                    break
            else:
                # No improvement, reduce lr
                lr_init *= 0.95
    
    return best_h, best_c5

def smooth_objective(h, beta):
    """Smooth approximation of max(corr) * dx using log-sum-exp"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    val = np.log(np.sum(np.exp(beta * shifted))) / beta
    return (m + val) * dx

def smooth_grad(h, beta):
    """Gradient of smooth objective"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k, w in enumerate(weights):
        if w > 1e-10:
            grad += w * corr_grad_single(h, k, n)
    return grad * dx

def slsqp_optimize(h0, n, beta, maxiter=300):
    """SLSQP with smooth objective"""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return smooth_objective(x, beta)
    
    def obj_grad(x):
        return smooth_grad(x, beta)
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h0, compute_c5(h0)
