        return np.ones(n_dst) * 0.5
    x_old = np.arange(len(h_src))
    x_new = np.linspace(0, len(h_src) - 1, n_dst)
    return np.interp(x_new, x_old, h_src)

def grad_max_corr(h, n, top_k=5):
    """Compute gradient of max(corr) w.r.t. h, using top_k active correlations."""
    corr = compute_corr(h)
    max_val = np.max(corr)
    # Get top_k indices
    idx = np.argsort(corr)[-top_k:]
    # Weight by proximity to max
    weights = np.exp(10 * (corr[idx] - max_val))

        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def lbfgs_smooth(h_init, n, beta, max_iter=300, pw=5000.0):
    target = n / 2.0
    h0 = h_init.copy()
    def fpen(x):
        return smooth_obj(x, n, beta) + pw * (np.sum(x) - target) ** 2
    def gpen(x):
        return grad_smooth_corr(x, n, beta) + 2.0 * pw * (np.sum(x) - target) * np.ones(n)
    try:
        res = minimize(fpen, h0, jac=gpen, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10, 'iprint': 0})
        h_out = make_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return None, None

def projected_grad_descent(h_init, n, beta, lr=0.01, max_iter=500):
    h = h_init.copy()
    f_best = compute_c5(h)
    h_best = h.copy()
    alpha = lr
    for it in range(max_iter):
        g = grad_smooth_corr(h, n, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        f_curr = smooth_obj(h, n, beta)
        step_alpha = alpha
        h_new = None
        for _ in range(30):
            candidate = project_constrained(h - step_alpha * g, n)
            f_cand = smooth_obj(candidate, n, beta)
            if f_cand <= f_curr - 1e-4 * step_alpha * g_norm * g_norm:
                h_new = candidate
                break
            step_alpha *= 0.5
        if h_new is None:
            alpha *= 0.5
            if alpha < 1e-10:
                break
            continue
        h = h_new
        c5 = compute_c5(h)
        if c5 < f_best:
            f_best = c5
            h_best = h.copy()
    return h_best, f_best

def directed_local_search(h, n, n_rounds=30, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        corr = compute_corr(h)
        worst_m = int(np.argmax(corr))
        k_worst = worst_m - (n - 1)
        for _ in range(50):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            step = rng.choice([0.005, 0.01, 0.02, 0.03])
            h_try = h.copy()
            amt = min(step, abs(h[i] - h[j]) * 0.5)
            if h[i] > h[j]:
                h_try[i] -= amt
                h_try[j] += amt
            else:
                h_try[i] += amt
                h_try[j] -= amt
            h_try = np.clip(h_try, 0.0, 1.0)
            diff = n / 2.0 - np.sum(h_try)
            if abs(diff) > 1e-13:
                mask = (h_try > 1e-8) & (h_try < 1.0 - 1e-8)
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h_try[mask] += diff / cnt
                    h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
        if not improved:
            break
    return h, best_c5

def pair_swap_improve(h, n, n_rounds=20, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for _ in range(200):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_try = h.copy()
            h_try[i], h_try[j] = h_try[j], h_try[i]
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                improved = True
        if not improved:
            break
    return h, best_c5

def binary_sa(h_init, n, n_steps=20000, t_start=0.5, t_end=0.01, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    if h_init is not None:
        h_rounded = np.round(h_init)
        h_rounded = np.clip(h_rounded, 0, 1)
        n_ones = int(np.sum(h_rounded))
        if n_ones != n // 2:
            if n_ones < n // 2:
                extra = rng.choice(np.flatnonzero(h_rounded < 1), n // 2 - n_ones, replace=False)
                h_rounded[extra] = 1.0
            else:
                extra = rng.choice(np.flatnonzero(h_rounded > 0), n_ones - n // 2, replace=False)
                h_rounded[extra] = 0.0
        h = h_rounded
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(min(0, -(c5_new - best_c5) / max(T, 1e-15))):
            pass
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

def multi_beta_lbfgs(h_init, n, betas=(20, 60, 200, 600, 2000, 5000), start_time=None, time_limit=None):
    current = h_init.copy()
    best_c5 = compute_c5(current)
    best_h = current.copy()
    for beta in betas:
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        h_opt, c5_opt = lbfgs_smooth(current, n, beta, max_iter=200)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            current = h_opt.copy()
    return best_h, best_c5

def structured_init(n, rng):
    inits = []
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h1.copy())
    if n % 2 == 0:
        h2 = np.tile([1.0, 0.0], n // 2)
        inits.append(h2.copy())
    h3 = np.zeros(n)
    for i in range(n):
        h3[i] = 1.0 if ((i * (i + 1) // 2) % n) < n // 2 else 0.0
    inits.append(make_feasible(h3, n))
    # Thue-Morse based
    h4 = np.zeros(n)
    tm = 0
    for i in range(n):
        tm = bin(i).count('1') % 2
        h4[i] = 1.0 if tm == 0 else 0.0
    h4 = make_feasible(h4, n)
    inits.append(h4)
    h5 = rng.rand(n)
    inits.append(make_feasible(h5, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def elapsed():
        return time.time() - start_time

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent, best_n)
    else:
        best_n = 160
        best_h = make_feasible(np.full(best_n, 0.5), best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-beta L-BFGS at parent resolution
    if elapsed() < budget_s - 100:
        t_lim = min(budget_s - 200, elapsed() + 250)
        h_opt, c5_opt = multi_beta_lbfgs(best_h, best_n,
                                          betas=(20, 60, 200, 600, 2000, 5000),
                                          start_time=start_time, time_limit=t_lim)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Projected gradient polish
    if elapsed() < budget_s - 80:
        h_opt, c5_opt = projected_grad_descent(best_h, best_n, beta=2000, lr=0.005, max_iter=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Directed local search
    if elapsed() < budget_s - 60:
        h_opt, c5_opt = directed_local_search(best_h, best_n, n_rounds=40, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Pair swap
    if elapsed() < budget_s - 40:
        h_opt, c5_opt = pair_swap_improve(best_h, best_n, n_rounds=15, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Try higher resolutions
    higher_ns = [int(best_n * 1.25), int(best_n * 1.5), int(best_n * 1.75)]
    for n_try in higher_ns:
        if elapsed() > budget_s - 80:
            break
        if n_try < 20:
            continue
        if n_try % 2 != 0:
            n_try += 1
        h_interp = make_feasible(interp_to(best_h, n_try), n_try)
        t_lim = min(budget_s - 60, elapsed() + 120)
        h_opt, c5_opt = multi_beta_lbfgs(h_interp, n_try,
                                          betas=(30, 100, 400, 1500, 5000),
                                          start_time=start_time, time_limit=t_lim)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_try
        h_interp = make_feasible(interp_to(best_h, n_try), n_try)
        h_opt2, c5_opt2 = projected_grad_descent(h_interp, n_try, beta=3000, lr=0.003, max_iter=200)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2
            best_n = n_try

    # Phase 6: Binary SA at current and nearby resolutions
    if elapsed() < budget_s - 60:
        for n_try in [best_n, max(40, best_n - 20), best_n + 20]:
            if elapsed() > budget_s - 40:
                break
            if n_try < 40 or n_try % 2 != 0:
                continue
            h_init_bin = best_h if n_try == best_n else interp_to(best_h, n_try)
            h_sa, c5_sa = binary_sa(h_init_bin, n_try, n_steps=15000, t_start=0.4, t_end=0.005, rng=rng)
            if c5_sa < best_c5:
                best_h = h_sa.astype(float)
                best_c5 = c5_sa
                best_n = n_try

    # Phase 7: Multi-start L-BFGS at best resolution
    if elapsed() < budget_s - 50:
        for trial in range(4):
            if elapsed() > budget_s - 30:
                break
            noise_scale = 0.03 / (trial + 1)
            h_try = best_h + rng.randn(best_n) * noise_scale
            h_try = make_feasible(h_try, best_n)
            h_opt, c5_opt = lbfgs_smooth(h_try, best_n, beta=1000, max_iter=300)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 8: Final aggressive polish with L-BFGS
    if elapsed() < budget_s - 25:
        h_opt, c5_opt = lbfgs_smooth(best_h, best_n, beta=5000, max_iter=500, pw=10000.0)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 9: Final local search
    if elapsed() < budget_s - 15:
        h_opt, c5_opt = directed_local_search(best_h, best_n, n_rounds=30, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if elapsed() < budget_s - 10:
        h_opt, c5_opt = pair_swap_improve(best_h, best_n, n_rounds=10, rng=rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 10: Projected gradient final
    if elapsed() < budget_s - 8:
        h_opt, c5_opt = projected_grad_descent(best_h, best_n, beta=5000, lr=0.002, max_iter=200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8

    return (best_h, best_c5, best_n)
