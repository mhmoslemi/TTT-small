def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def gaussian_smooth(h, sigma=1.5):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        return h.copy()
    length = min(7, n)
    if length % 2 == 0:
        length -= 1
    length = max(1, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * float(sigma) * float(sigma)))
    w /= w.sum()
    return np.convolve(h, w, mode="same")

def variant_starts(h, n, rng, count=4):
    base = project_feasible(h, n)
    out = [base.copy()]
    if n > 1:
        flip = base[::-1]
        out.append(project_feasible(0.5 * (base + flip), n))
        out.append(project_feasible(0.5 + 0.5 * (base - flip), n))
    out.append(project_feasible(gaussian_smooth(base, 1.6), n))
    out.append(project_feasible(gaussian_smooth(base, 0.8), n))
    if n > 0:
        pert = base + rng.randn(n) * 0.05
        pert = pert - float(np.mean(pert))
        out.append(project_feasible(pert, n))
    if count is not None and len(out) > count:
        out = out[:count]
    return out

def process_initial_set(inits, n, slot_end, seed, smooth=True):
    best_h = None
    best_c = np.inf
    m = len(inits)

    for j, h0 in enumerate(inits):
        if time.time() >= slot_end:
            break
        rem = slot_end - time.time()
        if rem <= 0.0:
            break
        per = rem / float(max(1, m - j))
        per = min(cap, per)
        per = max(8.0, per)
        per = min(per, rem)

            best_h = project_feasible(np.full(int(n), 0.5, dtype=float), n)

def local_block_perturb(h0, n, time_limit, seed, step=0.03):
    n = int(n)
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.0025
    min_temp = 1e-6

    max_trials = 4000
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(8, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.02, max(temp * 1.15, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.995, min_temp)

    return best_h, float(best_c)
