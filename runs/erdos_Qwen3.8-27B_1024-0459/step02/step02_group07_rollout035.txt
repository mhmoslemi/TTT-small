def compute_c5(h):

def make_feasible(h, n):

def slsqp_optimize(h, n, max_iter=5000):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return c.max()
    bounds = [(0.0, 1.0)] * n
    cons_list = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                   'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons_list,
                       options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
