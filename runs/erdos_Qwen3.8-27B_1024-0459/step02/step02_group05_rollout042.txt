def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    """Project h to be in [0,1] with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float)
    indices = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[indices[:n//2]] = 1.0
    return result

                    improved = True
                    break
                else:
