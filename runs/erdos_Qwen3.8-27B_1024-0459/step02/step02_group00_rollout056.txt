def simulated_annealing_bin(h_init, n, rng, max_iters=20000):
    h = np.zeros(n)
    idx_desc = np.argsort(-h_init)
    h[idx_desc[:n // 2]] = 1.0
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    T_start = 0.04
    T_end = 0.001
    T_cur = T_start

    for it in range(max_iters):
        if remaining_time() < 5:
            break
        T_cur = T_start * (T_end / T_start) ** (it / max_iters)

        # Random pair swap
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T_cur, 1e-8)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

    return best_h, best_c5

    _TGT = n / 2.0
    _BETA = beta

    def fun(x):
        return smooth_obj_and_grad(x, beta)[0]

    def fun_and_grad(x):
        return smooth_obj_and_grad(x, beta)

    penalty_lambda = 100.0

    def penalized_fun(x):
        v, g = smooth_obj_and_grad(x, beta)
        # Add penalty for sum constraint
        s = np.sum(x) - n / 2.0
        v += penalty_lambda * s * s
        g += 2.0 * penalty_lambda * s * np.ones(n)
        return v, g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalized_fun, h0, jac=True, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter,
                                               'ftol': 1e-16,
                                               'gtol': 1e-10,
                                               'iprint': 0})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0, compute_c5(h0)


def run(seed=42, budget_s=1300, **kwargs):
    global _N, _TGT, _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.7f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("No parent. Block: n={}, C5={:.7f}".format(best_n, best_c5))

    # N values to try (focused around parent)
    n_candidates = sorted(set([best_n, best_n - 8, best_n + 8,
                               best_n - 16, best_n + 16,
                               144, 152, 168, 176, 192]))
    n_candidates = [nn for nn in n_candidates if 100 <= nn <= 256 and nn % 2 == 0]

    # Phase 1: L-BFGS-B with multiple betas from best_n
    for n in n_candidates:
        if remaining_time() < 30:
            break
        _N = n
        _TGT = n / 2.0

        if n == best_n:
            h0 = best_h.copy()
        else:
            h0 = interp_to(best_h, best_n, n)
        h0 = project_feasible(h0, n)
        c5_cur = compute_c5(h0)

        print("\n--- n={}, start C5={:.7f} ---".format(n, c5_cur))

        # Multi-beta L-BFGS-B
        for beta in [20, 50, 150, 400, 1000, 3000]:
            if remaining_time() < 20:
                break
            h_new, c5_new = lbfgb_optimize(h0, n, beta, max_iter=300)
            if c5_new < c5_cur:
                h0 = h_new
                c5_cur = c5_new
                print("  beta={}: C5={:.7f}".format(beta, c5_cur))

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h0.copy()
            best_n = n
            print("  New best: n={}, C5={:.7f}".format(n, best_c5))

    # Phase 2: Subgradient descent on best solution
    if remaining_time() > 40:
        n = best_n
        _N = n
        h = best_h.copy()
        h, c5_sg = exact_subgrad_descent(h, n, max_iters=500)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h.copy()
            print("After subgrad: C5={:.7f}".format(best_c5))

    # Phase 3: L-BFGS-B again from improved point
    if remaining_time() > 40:
        n = best_n
        for beta in [500, 1500, 5000]:
            if remaining_time() < 15:
                break
            h_new, c5_new = lbfgb_optimize(best_h, n, beta, max_iter=400)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Polish beta={}: C5={:.7f}".format(beta, c5_new))

    # Phase 4: Binary local search with pair swaps
    if remaining_time() > 30:
        n = best_n
        h_bl, c5_bl = binary_local_search(best_h, n, max_sweeps=50, rng=rng)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            print("After binary LS: C5={:.7f}".format(best_c5))

    # Phase 5: Simulated annealing on binary
    if remaining_time() > 60:
        n = best_n
        h_sa, c5_sa = simulated_annealing_bin(best_h, n, rng, max_iters=50000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            print("After SA: C5={:.7f}".format(best_c5))

    # Phase 6: Continuous polish from best (L-BFGS-B with very high beta)
    if remaining_time() > 25:
        n = best_n
        for beta in [2000, 8000, 20000]:
            if remaining_time() < 12:
                break
            h_new, c5_new = lbfgb_optimize(best_h, n, beta, max_iter=300)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Final polish beta={}: C5={:.7f}".format(beta, c5_new))

    # Phase 7: Final binary local search after all continuous optimization
    if remaining_time() > 20:
        n = best_n
        h_bl, c5_bl = binary_local_search(best_h, n, max_sweeps=80, rng=rng)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
            print("Final binary LS: C5={:.7f}".format(best_c5))

    # Phase 8: very short subgradient from final
    if remaining_time() > 15:
        n = best_n
        h, c5_sg = exact_subgrad_descent(best_h, n, max_iters=100)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if isinstance(eval_c5, (int, float)) and eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
