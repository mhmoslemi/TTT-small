        return dx*(m + np.log(np.sum(np.exp(shifted)))/beta)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

    h_dst = np.interp(x_dst, x_src, h_src)

def corr_and_smoothmax(h, beta, pen_coeff=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    smooth_val = m + np.log(total) / beta
    penalty = pen_coeff * (np.sum(h) - n / 2.0) ** 2
    return smooth_val * dx + penalty, corr, exps / total

def grad_smoothmax(h, beta, pen_coeff=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    w = exps / np.sum(exps)
    
    g = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-15:
            continue
        L = t - (n - 1)
        if L == 0:
            g += wt * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += wt * (1.0 - h[L:])
            g[L:] -= wt * h[:n - L]
        else:
            Lp = -L
            g[Lp:] += wt * (1.0 - h[:n - Lp])
            g[:n - Lp] -= wt * h[Lp:]
    
    if pen_coeff > 0:
        g += 2.0 * pen_coeff * (np.sum(h) - n / 2.0)
    
    return dx * g

def optimize_lbfgsb(h0, n, beta, max_iter=300, pen_coeff=500.0):
    h0 = project_feasible(h0.copy(), n)
    
    def obj(x):
        val, _, _ = corr_and_smoothmax(x, beta, pen_coeff)
        return val
    
    def grad(x):
        return grad_smoothmax(x, beta, pen_coeff)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10,
                            'maxls': 50, 'maxfun': max_iter * 3})
    
    h_out = project_feasible(res.x, n)
    c5 = compute_c5(h_out)
    return h_out, c5

def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1.0)[0]
        vals = h[ones]
        order = np.argsort(vals)[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0.0)[0]
        vals = h[zeros]
        order = np.argsort(-vals)[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    return h_bin

def make_structured_init(n, rng):
    """Create a structured initialization based on knight's tour-like pattern."""
    h = np.zeros(n)
    # Try a "spread out" pattern
    step = n // 2
    positions = set()
    for i in range(n // 2):
        pos = (i * step) % n
        while pos in positions:
            pos = (pos + 1) % n
        positions.add(pos)
    for p in positions:
        h[p] = 1.0

def make_fourier_init(n, rng):
    """Signed Fourier series initialization."""
    h = np.zeros(n)

- The average overlap over all lags is... let me compute.

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.maximum(h, 1.0 - h)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)


def smooth_max_value(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return m + np.log(total) / beta


def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)


def obj_func(h, beta, pen_coeff):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max_value(corr, beta) * dx
    if pen_coeff > 0:
        val += pen_coeff * (np.sum(h) - n / 2.0) ** 2
    return val


def grad_func(h, beta, pen_coeff):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(corr, beta)
    g = np.zeros(n)
    for t in range(2 * n - 1):
        wt = w[t]
        if wt < 1e-15:
            continue
        L = t - (n - 1)
        if L == 0:
            g += wt * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - L] += wt * (1.0 - h[L:])
            g[L:] -= wt * h[:n - L]
        else:
            Lp = -L
            g[Lp:] += wt * (1.0 - h[:n - Lp])
            g[:n - Lp] -= wt * h[Lp:]
    if pen_coeff > 0:
        g += 2.0 * pen_coeff * (np.sum(h) - n / 2.0)
    return dx * g


def optimize_lbfgsb(h0, n, beta, max_iter, pen_coeff=200.0):
    h0 = project_feasible(h0.copy(), n)

    def obj(x):
        return obj_func(x, beta, pen_coeff)

    def grad(x):
        return grad_func(x, beta, pen_coeff)

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-9,
                            'maxls': 40, 'maxfun': max_iter * 4})
    h_out = project_feasible(np.asarray(res.x, dtype=float), n)
    c5 = compute_c5(h_out)
    return h_out, c5


def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1.0)[0]
        vals = h[ones]
        order = np.argsort(vals)[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0.0)[0]
        vals = h[zeros]
        order = np.argsort(-vals)[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    return h_bin


def binary_local_search(h, n, rng, max_rounds=8000, time_limit=50):

    best_c5 = compute_c5(h)
    start = time.time()
    no_improve = 0

    for _ in range(max_rounds):

        n_attempts = 40

            pos1 = ones[i1]
            pos0 = zeros[i0]
            h[pos1] = 0.0
            h[pos0] = 1.0
            new_c5 = compute_c5(h)

                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[pos1] = 1.0
                h[pos0] = 0.0

            no_improve = 0
        else:
            no_improve += 1

    return h, best_c5

def make_spread_binary(n, rng):
    """Create a binary pattern with spread-out 1s."""
    h = np.zeros(n)
    positions = set()
    step = n // 2
    for i in range(n // 2):
        pos = (i * step) % n
        tries = 0
        while pos in positions and tries < n:
            pos = (pos + 1) % n
            tries += 1
        if pos not in positions:
            positions.add(pos)
            h[pos] = 1.0
    if int(np.sum(h)) < n // 2:
        remaining = np.where(h == 0)[0]
        extra = rng.choice(remaining, size=n // 2 - int(np.sum(h)), replace=False)
        h[extra] = 1.0
    elif int(np.sum(h)) > n // 2:
        ones = np.where(h == 1)[0]
        extras = rng.choice(ones, size=int(np.sum(h)) - n // 2, replace=False)
        h[extras] = 0.0
    return h


def make_fourier_h(n, rng, n_modes=4):
    """Random Fourier-like smooth function."""
    x = np.arange(n, dtype=float) / n
    h = np.zeros(n)
    for k in range(n_modes):
        freq = rng.randint(1, max(2, n // 6))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.25)
        h += amp * np.sin(2 * np.pi * freq * x + phase)
    h = 0.5 + 0.35 * h
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)
