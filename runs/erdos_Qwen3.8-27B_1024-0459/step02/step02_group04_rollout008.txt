    x0[n] = max(0.0, float(t0))
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = [{"type": "eq", "fun": lambda x: sum_eq(x, n), "jac": lambda x: sum_eq_jac(x, n)}]
    if len(active_lags) > 0:
        constraints.append({"type": "ineq", "fun": lambda x: ineq_val(x, active_lags, n),
                            "jac": lambda x: ineq_jac(x, active_lags, n)})
    try:
        res = minimize(
            lambda x: x[n],
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": maxiter, "ftol": ftol, "disp": False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return x0

def soft_max_obj(x, beta, n):
    h = x[:n]
    vals, lags = all_lag_values(h)
    v_max = vals.max()
    shifted = np.exp(beta * (vals - v_max))
    return v_max + np.log(shifted.sum()) / beta

def soft_max_jac(x, beta, n):
    h = x[:n]
    vals, lags = all_lag_values(h)
    v_max = vals.max()
    shifted = np.exp(beta * (vals - v_max))
    total = shifted.sum()
    weights = shifted / total
    g = np.zeros(n)
    dx = 2.0 / n
    for i, L in enumerate(lags):
        if weights[i] > 1e-10:
            g += weights[i] * dx * lag_grad_raw(h, int(L))
    return g

def smooth_optimize(h0, n, beta, maxiter=80, ftol=1e-13):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": lambda x: sum_eq(x, n),
                    "jac": lambda x: sum_eq_jac(x, n)}]
    try:
        res = minimize(
            lambda x: soft_max_obj(x, beta, n),
            h,
            jac=lambda x: np.concatenate([soft_max_jac(x, beta, n), [0.0]]),
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": maxiter, "ftol": ftol, "disp": False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    h2 = project_feasible(h, n)
    return h2, compute_c5(h2)

def interpolate(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, 0.5)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def ref_perturb(h0, n, time_limit, seed, max_trials=5000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h.copy()
    cur_c = c
    temp = 0.003
    min_temp = 1e-7
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(10, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * 0.04
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = max(temp * 0.9, min_temp)
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)

def pick_active(vals, lags, count, margin=5e-5):
    order = np.argsort(-vals)
    take = min(count, len(vals))
    idx = order[:take]
    top = float(vals[order[0]])
    near = np.where(vals >= top - margin)[0]
    idx = np.unique(np.concatenate((idx, near)))
    idx = idx[np.argsort(-vals[idx])]
    cap = max(count * 3, len(vals))
    if idx.size > cap:
        idx = idx[:cap]
    return [int(lags[i]) for i in idx]

def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    t0 = time.time()
    hard_end = t0 + max(60.0, budget_s - 30.0)
    rng = np.random.RandomState(seed)

    parent = get_parent()
    if parent is not None:
        pn = int(parent.size)
        raw_h = parent.copy()
        raw_c = compute_c5(raw_h)
        proj_h = project_feasible(raw_h, pn)
        proj_c = compute_c5(proj_h)
        if raw_c <= proj_c + 1e-12:
            best_h = raw_h.copy()
            best_c = raw_c
        else:
            best_h = proj_h
            best_c = proj_c
        best_n = pn
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Multi-resolution quick sweep (60s)
    sweep_end = min(hard_end, t0 + 60.0)
    sweep_ns = [best_n - 16, best_n - 8, best_n, best_n + 8, best_n + 16]
    sweep_ns = [nn for nn in sweep_ns if 8 <= nn <= 280]
    seen = set()
    unique_ns = []
    for nn in sweep_ns:
        nn = int(nn)
        if nn not in seen:
            seen.add(nn)
            unique_ns.append(nn)

    for nn in unique_ns:
        if time.time() > sweep_end:
            break
        h0 = interpolate(best_h, nn)
        h0 = project_feasible(h0, nn)
        # Quick smooth pass
        for beta in (30, 100):
            if time.time() > sweep_end - 2:
                break
            h2, c2 = smooth_optimize(h0, nn, beta, maxiter=40, ftol=1e-12)
            if c2 < compute_c5(h0):
                h0 = h2
        c_final = compute_c5(h0)
        if c_final < best_c - 1e-10:
            best_h = h0.copy()
            best_c = c_final
            best_n = nn

    # Phase 2: Main optimization on best_n (majority of time)
    n = best_n
    h_cur = best_h.copy()
    c_cur = best_c

    # 2a: Smooth optimization with multiple betas
    main_end = hard_end - 120.0
    betas = [10, 30, 80, 200, 500]
    for beta in betas:
        if time.time() > main_end - 60:
            break
        h2, c2 = smooth_optimize(h_cur, n, beta, maxiter=100, ftol=1e-14)
        if c2 < c_cur - 1e-12:
            h_cur = h2
            c_cur = c2
        elif c2 < c_cur + 1e-4:
            h_cur = 0.5 * (h_cur + h2)
            h_cur = project_feasible(h_cur, n)

    # 2b: Active set optimization with growing sets
    for acnt in [2, 3, 4, 6, 8, 12, 16, 24, 32, 48, 64]:
        if time.time() > main_end - 30:
            break
        vals, lags = all_lag_values(h_cur)
        c_now = float(vals.max())
        if c_now < c_cur - 1e-12:
            c_cur = c_now
            best_h = h_cur.copy()
            best_c = c_cur

        active = pick_active(vals, lags, acnt, margin=2e-5)
        t0_val = c_now + 1e-6
        maxit = 300 if acnt <= 8 else 200
        x_sol = solve_active_set(h_cur, active, t0_val, n, maxiter=maxit, ftol=1e-15)
        if x_sol is not None:
            h2 = project_feasible(x_sol[:n], n)
            c2 = compute_c5(h2)
            if c2 < c_cur - 1e-14:
                h_cur = h2
                c_cur = c2
                best_h = h_cur.copy()
                best_c = c_cur
            elif c2 <= c_cur + 1e-10:
                h_cur = h2

    # 2c: Second round of active set with tighter margins
    if time.time() < main_end:
        for acnt in [4, 8, 16, 32, 48]:
            if time.time() > main_end - 10:
                break
            vals, lags = all_lag_values(h_cur)
            c_now = float(vals.max())
            if c_now < c_cur - 1e-12:
                c_cur = c_now
                best_h = h_cur.copy()
                best_c = c_cur
            active = pick_active(vals, lags, acnt, margin=1e-6)
            t0_val = c_now + 1e-7
            x_sol = solve_active_set(h_cur, active, t0_val, n, maxiter=400, ftol=1e-16)
            if x_sol is not None:
                h2 = project_feasible(x_sol[:n], n)
                c2 = compute_c5(h2)
                if c2 < c_cur - 1e-14:
                    h_cur = h2
                    c_cur = c2
                    best_h = h_cur.copy()
                    best_c = c_cur

    # Phase 3: Final refinement (last ~120s)
    if time.time() < hard_end - 15:
        # More smooth passes
        for beta in [200, 500, 1000]:
            if time.time() > hard_end - 30:
                break
            h2, c2 = smooth_optimize(h_cur, n, beta, maxiter=60, ftol=1e-15)
            if c2 < c_cur - 1e-13:
                h_cur = h2
                c_cur = c2
                best_h = h_cur.copy()
                best_c = c_cur

        # Final active set with all near-max lags
        if time.time() < hard_end - 40:
            vals, lags = all_lag_values(h_cur)
            c_now = float(vals.max())
            active = pick_active(vals, lags, 64, margin=1e-7)
            x_sol = solve_active_set(h_cur, active, c_now + 1e-8, n, maxiter=500, ftol=1e-16)
            if x_sol is not None:
                h2 = project_feasible(x_sol[:n], n)
                c2 = compute_c5(h2)
                if c2 < c_cur - 1e-14:
                    h_cur = h2
                    c_cur = c2
                    best_h = h_cur.copy()
                    best_c = c_cur

    # Phase 4: Local perturbation
    if time.time() < hard_end - 5:
        pert_end = hard_end - 3.0
        h2, c2 = ref_perturb(h_cur, n, pert_end, seed + 7777, max_trials=8000)
        if c2 < best_c - 1e-14:
            best_h = h2
            best_c = c2

    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
