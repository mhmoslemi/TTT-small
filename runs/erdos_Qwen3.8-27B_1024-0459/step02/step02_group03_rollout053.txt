    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(s)
    return e / np.sum(e)

    """Weights for top-k smooth approximation"""
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 0)
    e = np.exp(s)
    # Focus on top k by thresholding
    threshold = np.sort(e)[::-1][k-1] if k < len(e) else e.min()
    e[e < threshold * 0.1] *= 0.1  # Soften non-top-k
    return e / np.sum(e)

    """Compute gradient of c(L) w.r.t. h"""
    n = len(h)

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    weights = smooth_max_weights(corr, beta)
    return dx * full_gradient(x, weights)

    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint}]
    res = minimize(obj_func, h0, jac=grad_func, method='SLSQP',

                   options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

    """Take a subgradient step using top-k lags"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)

    top_idx = np.argsort(corr)[::-1][:top_k]

    grad = full_gradient(h, weights)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

from scipy.optimize import linprog

def solve_lp_direct(M, n):
    # Variables: [h_0, ..., h_{n-1}, t]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # M @ h - t <= 0  =>  [M, -1] @ [h, t] <= 0
    A1 = np.hstack([M, -np.ones((2*n-1, 1))])
    b1 = np.zeros(2*n-1)
    
    # -h <= 0  =>  [-I, 0] @ [h, t] <= 0
    A2 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b2 = np.zeros(n)
    
    # h <= 1  =>  [I, 0] @ [h, t] <= 1
    A3 = np.hstack([np.eye(n), np.zeros((n, 1))])
    b3 = np.ones(n)
    
    A_ub = np.vstack([A1, A2, A3])
    b_ub = np.concatenate([b1, b2, b3])
    
    # Equality constraint: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    bounds = [(0, 1)] * n + [(None, None)]  # h in [0,1], t free
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success and res.x is not None:
        return res.x[:n]
    return None
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_val(corr, beta):
    m = float(np.max(corr))
    s = np.clip(beta * (corr - m), -500.0, 0.0)
    return m + float(np.log(np.sum(np.exp(s)))) / beta

def smooth_max_weights(corr, beta):
    m = float(np.max(corr))
    s = np.clip(beta * (corr - m), -500.0, 0.0)
    e = np.exp(s)
    return e / np.sum(e)

def grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < L < n:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    elif -n < L < 0:
        Lp = -L
        g[:n-Lp] += 1.0 - h[Lp:]
        g[Lp:] -= h[:n-Lp]
    return g
