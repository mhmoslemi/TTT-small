def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:

            cnt = np.count_nonzero(mask)
            if cnt == 0: break

        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0: break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Fix any remaining issue
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Force fix
        h = np.clip(h, 0, 1)

```python
def compute_delta(h, n, a, b):
    # h[a]=1, h[b]=0 before swap
    delta = np.zeros(2*n-1)
    # s ranges from -(n-1) to (n-1), index k = s + n - 1
    
    # Term 1: -(1-h[a+s]) for valid a+s
    s_vals = np.arange(-(n-1), n)
    a_plus_s = a + s_vals
    mask1 = (a_plus_s >= 0) & (a_plus_s < n)
    delta[mask1] -= (1 - h[a_plus_s[mask1]])
    
    # Term 2: +h[a-s] for valid a-s
    a_minus_s = a - s_vals
    mask2 = (a_minus_s >= 0) & (a_minus_s < n)
    delta[mask2] += h[a_minus_s[mask2]]
    
    # Term 3: +(1-h[b+s]) for valid b+s
    b_plus_s = b + s_vals
    mask3 = (b_plus_s >= 0) & (b_plus_s < n)
    delta[mask3] += (1 - h[b_plus_s[mask3]])
    
    # Term 4: -h[b-s] for valid b-s
    b_minus_s = b - s_vals
    mask4 = (b_minus_s >= 0) & (b_minus_s < n)
    delta[mask4] -= h[b_minus_s[mask4]]
    
    return delta
```
