import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    res = minimize(_obj_cb, h0, jac=_jac_cb, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return project_feasible(res.x, n)

def local_search(h, n, rng, n_iter, step_size=0.1):
    """Hill climbing: try random coordinate perturbations targeting the max lag."""
    best_c5 = compute_c5(h)
    h = h.copy()
    for it in range(n_iter):
        # Identify the max lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        L = max_idx - (n - 1)
        
        # Generate perturbation indices
        n_pert = rng.randint(2, max(4, n // 10))
        if L > 0:
            # Max lag involves h[i] and h[i+L], try perturbing in those ranges
            candidates1 = range(max(0, L - 5), min(n, n - L + 5))
            candidates0 = range(max(0, -5), min(n, n - L + 5))
            perm = list(range(n))
            rng.shuffle(perm)
            pert_idx = []
            for i in perm:
                if i in candidates1 or i in candidates0:
                    pert_idx.append(i)
                    if len(pert_idx) >= n_pert:
                        break
            if len(pert_idx) < n_pert:
                for i in perm:
                    if i not in pert_idx:
                        pert_idx.append(i)
                        if len(pert_idx) >= n_pert:
                            break
        else:
            perm = list(range(n))
            rng.shuffle(perm)
            pert_idx = perm[:n_pert]
        
        h_pert = h.copy()
        deltas = rng.uniform(-step_size, step_size, len(pert_idx))
        h_pert[pert_idx] = np.clip(h_pert[pert_idx] + deltas, 0, 1)
        h_pert = project_feasible(h_pert, n)
        
        c5_new = compute_c5(h_pert)
        if c5_new < best_c5 - 1e-15:
            h = h_pert
            best_c5 = c5_new
    
    return h, best_c5

def binary_round(h, n):
    h_b = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_b))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_b == 1.0)[0]
        vals = h[ones]
        order = np.argsort(vals)[:n_ones - n_want]
        h_b[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_b == 0.0)[0]
        vals = h[zeros]
        order = np.argsort(-vals)[:n_want - n_ones]
        h_b[zeros[order]] = 1.0
    return h_b

def binary_local_search(h, n, rng, n_rounds=200):
    """Flip pairs to reduce C₅."""
    best_c5 = compute_c5(h)
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for rnd in range(n_rounds):
        improved = False
        for _ in range(30):
            i1 = rng.randint(len(ones))
            i0 = rng.randint(len(zeros))
            p1 = int(ones[i1])
            p0 = int(zeros[i0])
            if p1 == p0:
                continue
            h[p1] = 0.0
            h[p0] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                # Update ones and zeros
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                improved = True
            else:
                h[p1] = 1.0
                h[p0] = 0.0
        if not improved:
            break
    return h, best_c5
