def compute_c5(h):

    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if np.count_nonzero(mask) == 0:
                break
            h[mask] += diff / np.count_nonzero(mask)
        else:
            mask = h > 1e-14
            if np.count_nonzero(mask) == 0:
                break
            h[mask] += diff / np.count_nonzero(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_bin, n, time_limit, rng):
    """Pair-flip local search on binary sequence."""
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    start = time.time()
    n_improvements = 0

        improved = False
        # Random pair flips
        for _ in range(200):
            if len(ones) < 1 or len(zeros) < 1:

            i = rng.choice(ones)

                ones = np.where(h == 1)[0]

                improved = True

def precompute_V(h, n):
    """Precompute V[k][ell_index] = h[k-ell] - h[k+ell] for all k and ell."""
    ells = np.arange(-(n-1), n)  # length 2n-1

        km = k - ells  # k - ell for each ell
        kp = k + ells
        V[k, (km >= 0) & (km < n)] = h[km[(km >= 0) & (km < n)]]
        V[k, (kp >= 0) & (kp < n)] -= h[kp[(kp >= 0) & (kp < n)]]

def subgradient_step(h, n, alpha):
    corr = get_corr(h)
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-10)[0]
    ells = np.arange(-(n-1), n)[active]
    
    # Gradient of C(ell) w.r.t. h:
    # dC(ell)/dh[j] = (1-h[j+ell]) - h[j-ell] (with bounds)
    grad = np.zeros(n)
    for ell in ells:
        e = int(ell)
        # Terms where j+ell is valid: (1-h[j+ell])
        # j ranges from max(0, -e) to min(n-1, n-1-e)
        jm = max(0, -e)
        jp = min(n, n - e)
        if jm < jp:
            grad[jm:jp] += (1.0 - h[jm+e:jp+e])
        # Terms where j-ell is valid: -h[j-ell]
        # j ranges from max(0, e) to min(n-1, n-1+e)
        jm = max(0, e)
        jp = min(n, n + e)
        if jm < jp:
            grad[jm:jp] += -h[jm-e:jp-e]
    
    grad /= len(ells)
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    # Project onto sum constraint
    h_new = make_feasible(h_new, n)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0: break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0: break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_greedy_search(h_bin, n, time_limit):
    """Greedy pair-flip search on binary sequence."""
    h = h_bin.copy()
    corr = get_corr(h)
    best_c5 = np.max(corr) * (2.0/n)
    start = time.time()
    
    ells = np.arange(-(n-1), n)  # length 2n-1
    
    # Precompute V: V[k] is the delta vector for flipping h[k]
    # V[k][ell_idx] = h[k-ell] - h[k+ell]
    def compute_V(h):
        V = np.zeros((n, 2*n-1))
        for k in range(n):
            km = k - ells
            kp = k + ells
            vm = (km >= 0) & (km < n)
            vp = (kp >= 0) & (kp < n)
            V[k, vm] = h[km[vm]]
            V[k, vp] -= h[kp[vp]]
        return V
    
    V = compute_V(h)
    n_accepted = 0

        zeros = np.where(h == 0)[0]

- Term j+k=i (j=i-k): h[i-k](1-h[i]) changes from 0 to h[i-k](1-t). Change: h[i-k](1-t)

So the total change in C(k) from changing h[i]: 1→t is:
delta_C(k) = (t-1)(1-h[i+k]) + h[i-k](1-t) = (t-1)[(1-h[i+k]) - h[i-k]]
= (t-1)[1 - h[i+k] - h[i-k]]

where h[i+k] and h[i-k] are the original values (in the binary solution).

So C_new(k) = C_old(k) + (t-1) * D_i(k)

where D_i(k) = 1 - h[i+k] - h[i-k] (for binary h, this is in {-1, 0, 1}).

We want to minimize max_k C_new(k) over t ∈ [0,1].

This is a 1D optimization problem: minimize max_k [C_old(k) + (t-1) * D_i(k)] over t ∈ [0,1].

Since this is the max of linear functions of t, the minimum occurs at a point where two (or more) of the linear functions cross, or at t=0 or t=1.

I can solve this by:
1. For each active lag k (where C_old(k) is near the max), compute the linear function f_k(t) = C_old(k) + (t-1) * D_i(k)
2. Find the t that minimizes max_k f_k(t)

This is a 1D convex optimization problem (min of max of linear functions is convex). I can solve it with bisection or by checking all pairwise intersection points.

This is a very efficient approach! For each of the n positions, I can find the optimal t in O(n log n) time (sort the active lags and find the minimax point).

    """Given a binary h, optimize one fractional coordinate."""
    corr = get_corr(h)
    best_c5 = np.max(corr) * (2.0/n)
    best_h = h.copy()
    ells = np.arange(-(n-1), n)
    
    for i in range(n):
        # D_i(k) = 1 - h[i+k] - h[i-k] for each lag k
        # h[i+k] is valid when 0 <= i+k < n
        # h[i-k] is valid when 0 <= i-k < n
        a = np.zeros(2*n-1)  # -h[i+k] for each ell
        b = np.zeros(2*n-1)  # -h[i-k] for each ell
        ik = i + ells
        ik_valid = (ik >= 0) & (ik < n)
        a[ik_valid] = -h[ik[ik_valid]]
        ik2 = i - ells
        ik2_valid = (ik2 >= 0) & (ik2 < n)
        b[ik2_valid] = -h[ik2[ik2_valid]]
        D = 1 + a + b  # 1 - h[i+k] - h[i-k]
        
        # C_new(k) = corr[k] + (t-1) * D[k]
        # We want to minimize max_k [corr[k] + (t-1) * D[k]] over t in [0,1]
        # Let u = t-1, u in [-1, 0]
        # minimize max_k [corr[k] + u * D[k]] over u in [-1, 0]
        
        # The objective as a function of u is: f(u) = max_k (corr[k] + u*D[k])
        # This is a convex piecewise-linear function of u.
        # Its minimum is where the subgradient contains 0.
        
        # At u, the active set is {k : corr[k] + u*D[k] = max}
        # The subgradient is the convex hull of {D[k] : k active}
        # We want 0 in this convex hull, i.e., min(D[active]) <= 0 <= max(D[active])
        
        # Bisection on u:
        lo = -1.0
        hi = 0.0
        for _ in range(100):
            mid = 0.5 * (lo + hi)
            vals = corr + mid * D
            mx = np.max(vals)
            active = vals >= mx - 1e-14
            min_d = np.min(D[active])
            max_d = np.max(D[active])
            if min_d <= 0 <= max_d:
                break
            elif min_d > 0:
                # f is increasing at mid, go left
                hi = mid
            else:
                # f is decreasing at mid, go right
                lo = mid
        
        u_opt = 0.5 * (lo + hi)
        t_opt = u_opt + 1
        t_opt = np.clip(t_opt, 0, 1)
        
        # Evaluate
        new_corr = corr + u_opt * D
        new_c5 = np.max(new_corr) * (2.0/n)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            best_h[i] = t_opt
    
    return best_h, best_c5

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()

    target = n / 2.0
    for _ in range(2000):

        V[k, vp] -= h[kp[vp]]
    return V

def structured_inits(n, rng):

        h2 = np.tile([1.0, 0.0, 1.0, 0.0], n // 4)

            idx1 = j - ells
            idx2 = j + ells
            a_vec = np.zeros(2 * n - 1)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            a_vec[v1] += 1.0 - h[idx1[v1]]
            a_vec[v2] += -h[idx2[v2]]
            b_vec = corr - h[j] * a_vec
            
            lo = 0.0
            hi = 1.0
            for _ in range(40):
                mid = 0.5 * (lo + hi)
                vals = a_vec * mid + b_vec
                mx = np.max(vals)
                active = vals >= mx - 1e-14
                sg = np.mean(a_vec[active])
                if sg > 1e-13:
                    hi = mid
                elif sg < -1e-13:
                    lo = mid
                else:
                    break
            h[j] = 0.5 * (lo + hi)
        
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except NameError:
        h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent.copy(), best_n)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: Binary greedy search at parent resolution
    h_bin = binarize_h(best_h, best_n)
    if remaining() > 100:
        tl = min(150, remaining() - 80)
        h_g, c5_g = binary_greedy_search(h_bin, best_n, tl, rng)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 2: Fractional optimization of best binary
    if remaining() > 60:
        h_frac, c5_frac = optimize_fractional(best_h, best_n)
        if c5_frac < best_c5:
            best_c5 = c5_frac
            best_h = h_frac.copy()

    # Phase 3: Multi-fractional coordinate descent
    if remaining() > 80:
        h_mf, c5_mf = multi_fractional_optimize(best_h, best_n, max_iter=50)
        if c5_mf < best_c5:
            best_c5 = c5_mf
            best_h = h_mf.copy()

    # Phase 4: Try other resolutions with binary + fractional
    resolutions = []
    for rn in [128, 160, 192, 200, 240, 256, 320]:
        if rn != best_n:
            resolutions.append(rn)
    resolutions = sorted(resolutions)
    
    for n in resolutions:
        if remaining() < 60:
            break
        h_init = interpolate_to_n(best_h, n)
        h_b = binarize_h(h_init, n)
        
        tl_bin = min(60, remaining() - 70)
        if tl_bin < 10:
            continue
        h_g2, c5_g2 = binary_greedy_search(h_b, n, tl_bin, rng)
        
        if remaining() > 40:
            h_f2, c5_f2 = optimize_fractional(h_g2, n)
            if c5_f2 < c5_g2:
                h_g2 = h_f2
                c5_g2 = c5_f2
        
        if c5_g2 < best_c5 - 1e-12:
            best_c5 = c5_g2
            best_h = h_g2.copy()
            best_n = n

    # Phase 5: Restart with structured inits at best resolution
    if remaining() > 120:
        inits = structured_inits(best_n, rng)
        if h_parent is not None:
            inits.insert(0, binarize_h(best_h, best_n))
        for h_init in inits:
            if remaining() < 50:
                break
            tl_bin = min(50, remaining() - 60)
            if tl_bin < 10:
                continue
            h_g3, c5_g3 = binary_greedy_search(h_init, best_n, tl_bin, rng)
            if remaining() > 30:
                h_f3, c5_f3 = optimize_fractional(h_g3, best_n)
                if c5_f3 < c5_g3:
                    h_g3 = h_f3
                    c5_g3 = c5_f3
            if c5_g3 < best_c5 - 1e-12:
                best_c5 = c5_g3
                best_h = h_g3.copy()

    # Phase 6: Simulated annealing on best binary
    if remaining() > 40:
        h_b6 = binarize_h(best_h, best_n)
        sa_tl = min(50, remaining() - 30)
        if sa_tl > 10:
            h_sa, c5_sa = binary_sa_search(h_b6, best_n, sa_tl, rng)
            if remaining() > 20:
                h_f6, c5_f6 = optimize_fractional(h_sa, best_n)
                if c5_f6 < c5_sa:
                    h_sa = h_f6
                    c5_sa = c5_f6
            if c5_sa < best_c5 - 1e-12:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 7: Final coordinate descent polish
    if remaining() > 30:
        tl_cd = max(5, remaining() - 15)
        h_cd, c5_cd = coord_descent_fast(best_h, best_n, n_passes=100, time_limit=tl_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: One more fractional pass
    if remaining() > 20:
        h_f8, c5_f8 = optimize_fractional(best_h, best_n)
        if c5_f8 < best_c5:
            best_c5 = c5_f8
            best_h = h_f8.copy()

    # Phase 9: Final coordinate descent
    if remaining() > 15:
        tl_cd2 = max(3, remaining() - 5)
        h_cd2, c5_cd2 = coord_descent_fast(best_h, best_n, n_passes=50, time_limit=tl_cd2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
