```
def project_box_sum(v, n):
    target = n/2.0
    v = np.asarray(v, dtype=float).ravel()
    if v.size != n: v = np.resize(v, n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    # solve sum(clip(v-tau,0,1)) = target
    for _ in range(60):
       mid=(lo+hi)/2
       s=np.clip(v-mid,0,1).sum()
       if s > target: lo=mid else hi=mid
    h=np.clip(v-(lo+hi)/2,0,1)
    diff=target-h.sum()
    # adjust residual
    for _ in range(20):
       if abs(diff) < 1e-14: break
       slack = 1.0 - h
       # if diff positive, add to entry with max slack
       idx = int(np.argmax(slack if diff>0 else h))
       adj = diff
       if diff > 0:
          adj = min(adj, slack[idx])
       else:
          adj = max(adj, -h[idx])
       h[idx] += adj
       h = np.clip(h,0,1)
       diff = target - h.sum()
    # final exact adjust: set one variable to target-sum others if within [0,1]
    if abs(h.sum()-target)>1e-12:
       # find idx with room
       if target > h.sum():
          order=np.argsort(1-h);
       else:
          order=np.argsort(h);
       for idx in order:
          room = target - (h.sum()-h[idx])
          if -1e-12 <= room <= 1+1e-12:
             h[idx]=np.clip(room,0,1); break
       h=np.clip(h,0,1)
    return h
```

```
import numpy as np

from scipy.optimize import minimize

_CUR_MODE = 'full'

_CUR_BETA = 100.0
_CUR_TOPK = 0
```

```python
def compute_counts(h):
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return np.array([0.0], dtype=float)
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    c = compute_counts(h)
    return float(np.max(c) * dx)


def project_box_sum(v, n):
    n = int(n)
    v = np.asarray(v, dtype=float).ravel()
    if v.size != n:
        v = np.resize(v, n)

    target = 0.5 * float(n)
    if not np.all(np.isfinite(v)):
        v = np.nan_to_num(v, nan=0.5, posinf=0.5, neginf=0.5)

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(30):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                if diff <= 1e-15:
                    break
                add = min(diff, 1.0 - h[idx])
                if add > 0.0:
                    h[idx] += add
                    diff -= add
        else:
            order = np.argsort(h)
            for idx in order:
                if diff >= -1e-15:
                    break
                sub = min(-diff, h[idx])
                if sub > 0.0:
                    h[idx] -= sub
                    diff += sub

    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        if diff > 0.0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-13:
                break
            room = diff
            if room > 0.0:
                room = min(room, 1.0 - h[idx])
            else:
                room = max(room, -h[idx])
            h[idx] += room
            diff = target - float(h.sum())

    h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h, n):
    h = np.asarray(h, dtype=float).ravel()
    old = h.size
    n = int(n)
    if old == n:
        return project_box_sum(h, n)
    xp = (np.arange(old) + 0.5) * (2.0 / float(old))
    xn = (np.arange(n) + 0.5) * (2.0 / float(n))
    y = np.interp(xn, xp, h, left=h[0], right=h[-1])
    return project_box_sum(y, n)


def sym_variant(h, n):
    h = np.asarray(h, dtype=float).ravel()
    if h.size != n:
        h = np.resize(h, n)
    return project_box_sum(0.5 * (h + h[::-1]), n)


def anti_variant(h, n):
    h = np.asarray(h, dtype=float).ravel()
    if h.size != n:
        h = np.resize(h, n)
    return project_box_sum(0.5 * (h + 1.0 - h[::-1]), n)


def noise_variant(h, n, rng, sigma):
    h = np.asarray(h, dtype=float).ravel()
    if h.size != n:
        h = np.resize(h, n)
    return project_box_sum(h + float(sigma) * rng.normal(size=n), n)


def thue_morse_binary(n):
    n = int(n)
    h = np.zeros(n, dtype=float)
    for i in range(n):
        b = i
        p = 0
        while b:
            p ^= (b & 1)
            b >>= 1
        if p:
            h[i] = 1.0
    return project_box_sum(h, n)


def sym_to_full(z, n):
    n = int(n)
    m = (n + 1) // 2
    z = np.asarray(z, dtype=float).ravel()
    if z.size < m:
        z = np.pad(z, (0, m - z.size), mode='edge')
    z = z[:m]

    h = np.empty(n, dtype=float)
    c = n // 2
    if n % 2 == 0:
        h[:c] = z
        h[c:] = z[::-1]
    else:
        h[:c] = z[:c]
        h[c] = z[c]
        h[c + 1:] = z[:c][::-1]
    return h


def full_to_sym_vars(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).ravel()
    m = (n + 1) // 2
    z = np.zeros(m, dtype=float)
    c = n // 2
    if n % 2 == 0:
        z[:] = 0.5 * (h[:c] + h[n - 1:c - 1:-1])
    else:
        if c > 0:
            z[:c] = 0.5 * (h[:c] + h[n - 1:c - 1:-1])
        z[c] = h[c]
    return np.clip(z, 0.0, 1.0)


def sym_grad(g, n):
    n = int(n)
    g = np.asarray(g, dtype=float).ravel()
    m = (n + 1) // 2
    z = np.zeros(m, dtype=float)
    c = n // 2
    if n % 2 == 0:
        z[:] = g[:c] + g[n - 1:c - 1:-1]
    else:
        if c > 0:
            z[:c] = g[:c] + g[n - 1:c - 1:-1]
        z[c] = g[c]
    return z


def weights_obj_from_counts(c):
    c = np.asarray(c, dtype=float).ravel()
    v = c * _CUR_DX

    if _CUR_TOPK > 0:
        k = min(int(_CUR_TOPK), len(v))
        if k <= 0:
            return 0.0, np.zeros(len(v), dtype=float)
        order = np.argsort(v)[::-1][:k]
        w = np.zeros(len(v), dtype=float)
        w[order] = 1.0 / float(k)
        obj = float(np.dot(w, v))
        return obj, w

    m = float(np.max(v))
    sh = float(_CUR_BETA) * (v - m)
    sh = np.clip(sh, -50.0, 0.0)
    e = np.exp(sh)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(len(v), dtype=float)
        w[np.argmax(v)] = 1.0
        return m, w
    w = e / s
    obj = m + float(np.log(s)) / float(_CUR_BETA)
    return obj, w


def weighted_gradient_counts(h, W):
    h = np.asarray(h, dtype=float).ravel()
    n = h.size
    W = np.asarray(W, dtype=float).ravel()
    if n == 0 or W.size != 2 * n - 1:
        return np.zeros(n, dtype=float)

    if np.sum(np.abs(W)) == 0.0:
        return np.zeros(n, dtype=float)

    Wt = W.copy()
    if int(np.count_nonzero(Wt)) > 12:
        thresh = max(1e-14, float(np.max(np.abs(Wt))) * 1e-10)
        Wt[np.abs(Wt) <= thresh] = 0.0
        s = float(np.sum(Wt))
        if s > 0.0:
            Wt = Wt / s
        elif np.sum(np.abs(Wt)) > 0.0:
            Wt = Wt / float(np.sum(np.abs(Wt)))

    nnz = int(np.count_nonzero(Wt))
    if nnz == 0:
        return np.zeros(n, dtype=float)

    off = n - 1

    if nnz <= 12:
        grad = np.zeros(n, dtype=float)
        idxs = np.nonzero(Wt)[0]
        for idx in idxs:
            w = float(Wt[idx])
            if w == 0.0:
                continue
            d = int(idx) - off
            if d >= 0:
                L = n - d
                if L <= 0:
                    continue
                grad[:L] += w * (1.0 - h[d:d + L])
                grad[d:] -= w * h[:L]
            else:
                e = -d
                L = n - e
                if L <= 0:
                    continue
                grad[:L] -= w * h[e:e + L]
                grad[e:] += w * (1.0 - h[:L])
        return grad

    prefix = np.empty(Wt.size + 1, dtype=float)
    prefix[0] = 0.0
    prefix[1:] = np.cumsum(Wt)

    i = np.arange(n, dtype=float)
    starts = (off - i).astype(int)
    ends = (Wt.size - 1 - (n - 1 - i)).astype(int) + 1
    R = prefix[ends] - prefix[starts]

    c1 = np.convolve(h, Wt, mode='full')
    c2 = np.convolve(h, Wt[::-1], mode='full')
    g = R - c1[off:off + n] - c2[off:off + n]
    return g


def prepare_weights(x):
    global _CACHE_OK, _CACHE_X_COPY, _CACHE_H, _CACHE_W

    n = int(_CUR_N)
    if n <= 0:
        return 0.0, np.zeros(0, dtype=float), np.zeros(0, dtype=float), np.zeros(0, dtype=float)

    if _CUR_MODE == 'full':
        h = np.asarray(x, dtype=float).ravel()
    else:
        h = sym_to_full(x, n)

    c = compute_counts(h)
    obj, w = weights_obj_from_counts(c)

    _CACHE_X_COPY = np.asarray(x, dtype=float).ravel().copy()
    _CACHE_H = h.copy()
    _CACHE_W = w.copy()
    _CACHE_OK = True
    return obj, w, h, c


def obj_func(x):
    n = int(_CUR_N)
    if n <= 0:
        return 0.0
    obj, w, h, c = prepare_weights(x)
    return float(obj)


def grad_func(x):
    n = int(_CUR_N)
    if n <= 0:
        return np.zeros(0, dtype=float)

    use_cache = False
    if _CACHE_OK and _CACHE_X_COPY is not None:
        xx = np.asarray(x, dtype=float).ravel()
        if xx.size == _CACHE_X_COPY.size and np.array_equal(xx, _CACHE_X_COPY):
            use_cache = True

    if use_cache:
        h = _CACHE_H
        w = _CACHE_W
    else:
        obj, w, h, c = prepare_weights(x)

    g = weighted_gradient_counts(h, w) * _CUR_DX
    if _CUR_MODE == 'sym':
        g = sym_grad(g, n)
    return g


def eq_constraint(x):
    n = int(_CUR_N)
    x = np.asarray(x, dtype=float).ravel()
    if _CUR_MODE == 'full':
        val = float(np.sum(x)) - 0.5 * float(n)
    else:
        c = n // 2
        if n % 2 == 0:
            val = 2.0 * float(np.sum(x)) - 0.5 * float(n)
        else:
            val = 2.0 * float(np.sum(x[:c])) + float(x[c]) - 0.5 * float(n)
    return np.array([val], dtype=float)


def eq_jac_constraint(x):
    n = int(_CUR_N)
    x = np.asarray(x, dtype=float).ravel()
    if _CUR_MODE == 'full':
        arr = np.ones((1, n), dtype=float)
    else:
        c = n // 2
        if n % 2 == 0:
            arr = 2.0 * np.ones((1, c), dtype=float)
        else:
            arr = np.empty((1, c + 1), dtype=float)
            arr[0, :c] = 2.0
            arr[0, c] = 1.0
    return arr


def polish_start(var, n, mode, time_limit, hard_deadline):
    global _CUR_MODE, _CUR_N, _CUR_DX, _CUR_BETA, _CUR_TOPK, _CACHE_OK

    n = int(n)
    if n <= 0:
        return np.zeros(0, dtype=float), float('inf')

    t_start = time.time()
    limit = min(float(hard_deadline), t_start + max(5.0, float(time_limit)))

    if mode == 'full':
        h = project_box_sum(var, n)
        x = h.copy()
        m = n
    else:
        h = project_box_sum(var, n)
        x = full_to_sym_vars(h, n)
        m = x.size

    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_c = best_c

    bounds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac_constraint}]

    if n <= 224:
        base_iter = 260
    elif n <= 320:
        base_iter = 200
    elif n <= 448:
        base_iter = 150
    else:
        base_iter = 100

    if float(time_limit) < 25.0:
        soft_stages = [(100, base_iter), (5000, base_iter)]
    elif float(time_limit) < 70.0:
        soft_stages = [(50, base_iter), (300, base_iter), (2000, base_iter)]
    else:
        soft_stages = [(20, base_iter), (80, base_iter), (300, base_iter),
                       (1000, base_iter), (5000, base_iter)]

    def local_update_from_result(res, x):
        global best_c
        return None  # placeholder only to satisfy structural expectation

    for beta, maxit in soft_stages:
        if time.time() >= limit:
            break

        _CUR_MODE = mode
        _CUR_N = n
        _CUR_DX = 2.0 / float(n)
        _CUR_BETA = float(beta)
        _CUR_TOPK = 0
        _CACHE_OK = False

        try:
            res = minimize(obj_func, x, jac=grad_func, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': int(maxit), 'ftol': 1e-12, 'iprint': 0})
            x_new = res.x if hasattr(res, 'x') else x
            if x_new is None:
                continue
            x_new = np.asarray(x_new, dtype=float).ravel()
            if not np.all(np.isfinite(x_new)):
                continue

            if mode == 'full':
                h_new = project_box_sum(x_new, n)
            else:
                h_full = sym_to_full(x_new, n)
                h_sym = 0.5 * (h_full + h_full[::-1])
                h_new = project_box_sum(h_sym, n)

            c_new = compute_c5(h_new)
            if np.isfinite(c_new):
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
                if c_new <= cur_c + 1e-10:
                    cur_c = c_new
                    if mode == 'full':
                        x = h_new.copy()
                    else:
                        x = full_to_sym_vars(h_new, n)
        except Exception:
            pass

    if time.time() < limit - 3.0:
        topk_stages = [(3, max(60, int(0.7 * base_iter))), (1, base_iter)]
        for k, maxit in topk_stages:
            if time.time() >= limit:
                break

            _CUR_MODE = mode
            _CUR_N = n
            _CUR_DX = 2.0 / float(n)
            _CUR_BETA = 100.0
            _CUR_TOPK = int(k)
            _CACHE_OK = False

            try:
                res = minimize(obj_func, x, jac=grad_func, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': int(maxit), 'ftol': 1e-12, 'iprint': 0})
                x_new = res.x if hasattr(res, 'x') else x
                if x_new is None:
                    continue
                x_new = np.asarray(x_new, dtype=float).ravel()
                if not np.all(np.isfinite(x_new)):
                    continue

                if mode == 'full':
                    h_new = project_box_sum(x_new, n)
                else:
                    h_full = sym_to_full(x_new, n)
                    h_sym = 0.5 * (h_full + h_full[::-1])
                    h_new = project_box_sum(h_sym, n)

                c_new = compute_c5(h_new)
                if np.isfinite(c_new):
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h_new.copy()
                    if c_new <= cur_c + 1e-10:
                        cur_c = c_new
                        if mode == 'full':
                            x = h_new.copy()
                        else:
                            x = full_to_sym_vars(h_new, n)
            except Exception:
                pass

    best_h = project_box_sum(best_h, n)
    if mode == 'sym':
        best_h = project_box_sum(0.5 * (best_h + best_h[::-1]), n)
    return best_h, compute_c5(best_h)


def binary_round_preserve(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).ravel()
    if h.size != n:
        h = np.resize(h, n)
    target = 0.5 * float(n)
    k = int(target)
    order = np.argsort(-h)
    h_bin = np.zeros(n, dtype=float)
    if k > 0:
        h_bin[order[:k]] = 1.0
    if target != float(k) and k < n:
        h_bin[order[k]] = target - float(k)
    return project_box_sum(h_bin, n)


def active_descent(h, n, hard_deadline):
    n = int(n)
    h = project_box_sum(h, n)
    cur = compute_c5(h)
    dx = 2.0 / float(n)
    alpha = 0.012
    stagnant = 0

    for it in range(500):
        if time.time() >= float(hard_deadline):
            break

        c = compute_counts(h)
        v = c * dx
        k = 1 if (it % 3 == 0) else 3
        order = np.argsort(v)[::-1][:k]
        W = np.zeros(len(v), dtype=float)
        W[order] = 1.0 / float(k)

        g = weighted_gradient_counts(h, W) * dx
        gn = float(np.sqrt(np.dot(g, g)))
        if gn < 1e-12:
            break

        accepted = False
        lr = alpha
        for _ in range(8):
            h2 = project_box_sum(h - lr * g, n)
            f2 = compute_c5(h2)
            if np.isfinite(f2) and f2 < cur - 1e-14:
                h = h2
                cur = f2
                alpha = min(0.05, lr * 1.15)
                accepted = True
                stagnant = 0
                break
            lr *= 0.5

        if not accepted:
            stagnant += 1
            alpha *= 0.5
            if alpha < 1e-8 or stagnant > 35:
                break

    return h, compute_c5(h)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    hard_deadline = t0 + max(60.0, float(budget_s)) - 35.0
    rng = np.random.RandomState(int(seed))

    parent = None
    try:
        p = np.asarray(initial_h_values, dtype=float).ravel()
        if p.size >= 8 and np.all(np.isfinite(p)):
            parent = p
    except Exception:
        parent = None

    if parent is None:
        n_par = 192
        parent = thue_morse_binary(n_par)
    else:
        n_par = int(parent.size)

    best_h = project_box_sum(parent, n_par)
    best_n = n_par
    best_c = compute_c5(best_h)

    for hq in (sym_variant(best_h, n_par), anti_variant(best_h, n_par)):
        cq = compute_c5(hq)
        if np.isfinite(cq) and cq < best_c:
            best_c = cq
            best_h = hq.copy()
            best_n = n_par

    time_left = hard_deadline - time.time()
    if time_left > 90.0:
        starts = [
            best_h,
            sym_variant(best_h, n_par),
            noise_variant(best_h, n_par, rng, 0.035)
        ]
        per = min(45.0, max(8.0, (time_left - 50.0) / float(len(starts))))
        for h0 in starts:
            if hard_deadline - time.time() < 35.0:
                break
            hopt, cval = polish_start(h0, n_par, 'full', max(8.0, per), hard_deadline)
            if np.isfinite(cval) and cval < best_c:
                best_c = cval
                best_h = hopt.copy()
                best_n = n_par

    res_list_raw = set()
    for n in (int(round(n_par * 1.25)), int(round(n_par * 1.5)),
              int(round(n_par * 1.75)), int(round(n_par * 2.0)),
              256, 320, 384, 512):
        n = int(n)
        if n % 2 != 0:
            n += 1
        if n_par < n <= 512:
            res_list_raw.add(n)
    res_list = sorted(res_list_raw)
    if len(res_list) > 4:
        res_list = res_list[-4:]

    for n in res_list:
        rem = 0
        for r in res_list:
            if r >= n:
                rem += 1

        time_left = hard_deadline - time.time()
        if time_left < 90.0:
            break

        if n <= 260:
            base = 120.0
        elif n <= 330:
            base = 170.0
        elif n <= 400:
            base = 220.0
        else:
            base = 170.0

        alloc = min(base, max(45.0, time_left / float(rem + 1) * 1.55))

        if best_n == n:
            h0 = project_box_sum(best_h, n)
        else:
            h0 = interpolate_to(best_h, n)

        starts = [
            h0,
            sym_variant(h0, n)
        ]
        if time_left > 380.0 and n <= 384:
            starts.append(noise_variant(h0, n, rng, 0.05))

        per = alloc / float(max(1, len(starts)))
        for hstart in starts:
            if hard_deadline - time.time() < 35.0:
                break
            alloc_this = min(per, max(10.0, (hard_deadline - time.time()) - 45.0))
            if alloc_this < 10.0:
                continue
            hopt, cval = polish_start(hstart, n, 'full', alloc_this, hard_deadline)
            if np.isfinite(cval) and cval < best_c:
                best_c = cval
                best_h = hopt.copy()
                best_n = n

        if hard_deadline - time.time() > 85.0:
            hstart = sym_variant(h0, n)
            alloc_sym = min(95.0, max(25.0, (hard_deadline - time.time()) - 50.0))
            hopt, cval = polish_start(hstart, n, 'sym', alloc_sym, hard_deadline)
            if np.isfinite(cval) and cval < best_c:
                best_c = cval
                best_h = hopt.copy()
                best_n = n

    if (hard_deadline - time.time()) > 360.0 and best_n < 768:
        h768 = interpolate_to(best_h, 768)
        alloc768 = min(220.0, max(50.0, (hard_deadline - time.time()) - 60.0))
        hopt, cval = polish_start(h768, 768, 'full', alloc768, hard_deadline)
        if np.isfinite(cval) and cval < best_c:
            best_c = cval
            best_h = hopt.copy()
            best_n = 768

    time_left = hard_deadline - time.time()
    if time_left > 70.0:
        hopt, cval = polish_start(best_h, best_n, 'full',
                                  min(130.0, time_left - 35.0), hard_deadline)
        if np.isfinite(cval) and cval < best_c:
            best_c = cval
            best_h = hopt.copy()
            best_n = best_n

    if hard_deadline - time.time() > 35.0:
        hopt, cval = active_descent(best_h, best_n, hard_deadline - 8.0)
        if np.isfinite(cval) and cval < best_c:
            best_c = cval
            best_h = hopt.copy()
            best_n = best_n

    h_bin = binary_round_preserve(best_h, best_n)
    c_bin = compute_c5(h_bin)
    if np.isfinite(c_bin) and c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()

    best_h = project_box_sum(best_h, best_n)
    final_c = compute_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        try:
            evf = float(ev)
            if np.isfinite(evf) and evf < final_c:
                final_c = evf
        except Exception:
            pass
    except Exception:
        pass

    if not (0.0 <= final_c < 2.0):
        final_c = compute_c5(best_h)

    return (best_h, float(final_c), int(best_n))
```
