def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible_binary(h, n, rng):
    h = np.round(np.array(h, dtype=float)).astype(float)
    h = np.clip(h, 0.0, 1.0)
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        idx = rng.choice(ones, excess, replace=False)
        h[idx] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        idx = rng.choice(zeros, deficit, replace=False)
        h[idx] = 1.0
    return h

def make_feasible_cont(h, n):

        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_polish(h_init, n, max_iter=2000):
    h0 = make_feasible_cont(h_init.copy(), n)

    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        return np.max(corr)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': con_fun, 'jac': con_jac}
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=con,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        h_out = make_feasible_cont(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        return h0, compute_c5(h0)

def smooth_max_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -500, 50)
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val * (2.0 / n)
