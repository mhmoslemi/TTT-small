def compute_d_max(h):
    """Compute max disagreement count for all shifts."""
    n = len(h)
    max_d = 0
    for m in range(1, n):
        d = np.sum(h[:-m] != h[m:])
        if d > max_d:
            max_d = d
    return int(max_d)

def run_sa_symmetric(n, k_pairs, t0_temp, t_end_temp, n_iters, h_init=None):
    n_pairs = n // 2
    assert n % 2 == 0
    
    if h_init is not None:
        pairs = h_init[:n_pairs].copy().astype(int)
    else:
        pairs = np.zeros(n_pairs, dtype=int)
        pairs[np.random.choice(n_pairs, k_pairs, replace=False)] = 1
    
    def build(pairs):
        h = np.zeros(n, dtype=float)
        h[:n_pairs] = pairs
        h[n_pairs:] = pairs[::-1]
        return h
    
    current_h = build(pairs)
    current_d = compute_d_max(current_h)
    best_h = current_h.copy()
    best_d = current_d
    
    ones_idx = np.where(pairs == 1)[0]
    zeros_idx = np.where(pairs == 0)[0]
    
    for it in range(n_iters):
        temp = t0_temp * (t_end_temp / t0_temp) ** (it / n_iters)
        
        # Random swap: pick one "on" pair and one "off" pair, swap them
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        
        pairs[i] = 0
        pairs[j] = 1
        new_h = build(pairs)
        new_d = compute_d_max(new_h)
        
        delta = new_d - current_d
        if delta <= 0 or np.random.random() < np.exp(-delta / temp):
            current_h = new_h
            current_d = new_d
            ones_idx = np.where(pairs == 1)[0]
            zeros_idx = np.where(pairs == 0)[0]
            if new_d < best_d:
                best_d = new_d
                best_h = new_h.copy()
        else:
            # Revert
            pairs[i] = 1
            pairs[j] = 0
    
    return best_h, best_d
