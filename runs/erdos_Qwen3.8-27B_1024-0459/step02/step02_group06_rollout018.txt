```python
def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return float("inf")
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if corr.size == 0:
        return float("inf")
    return float(np.max(corr) * dx)

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:

    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))

def compute_contrib(h, p):
    """Compute ∂c[k]/∂h[p] for all lags k (indices 0 to 2n-2 in the full correlate output)"""
    n = h.size
    contrib = np.zeros(2*n - 1)
    # For lag d >= 0 (k = n-1+d): c[k] = sum_{m=0}^{n-1-d} h[m+d]*(1-h[m])
    # ∂/∂h[p]:
    #   h[p] in first factor: m+d=p → m=p-d, valid: 0<=p-d<=n-1-d → d<=p<=n-1
    #     contribution: (1-h[m]) = (1-h[p-d])
    #   h[p] in second factor: m=p, valid: 0<=p<=n-1-d → d<=n-1-p
    #     contribution: -h[m+d] = -h[p+d]
    for d in range(0, n):
        k = n - 1 + d
        # First factor: h[p] = h[m+d] where m = p-d
        if d <= p <= n - 1:
            m = p - d
            contrib[k] += (1.0 - h[m])
        # Second factor: h[p] = h[m] where m = p
        if 0 <= p <= n - 1 - d:
            m = p
            contrib[k] -= h[m + d]
    
    # For lag d < 0 (let d' = -d > 0, k = n-1-d'): c[k] = sum_{m=d'}^{n-1} h[m]*(1-h[m-d'])
    # ∂/∂h[p]:
    #   h[p] in first factor: m=p, valid: d'<=p<=n-1
    #     contribution: (1-h[p-d'])
    #   h[p] in second factor: m-d'=p → m=p+d', valid: d'<=p+d'<=n-1 → 0<=p<=n-1-d'
    #     contribution: -h[p+d']
    for dp in range(1, n):
        k = n - 1 - dp
        # First factor
        if dp <= p <= n - 1:
            contrib[k] += (1.0 - h[p - dp])
        # Second factor
        if 0 <= p <= n - 1 - dp:
            contrib[k] -= h[p + dp]
    
    return contrib
```

```python
def compute_contrib_fast(h, p):
    n = h.size
    center = n - 1
    contrib = np.zeros(2 * n - 1)
    
    # Positive and zero lags: k = 0, 1, ..., n-1, output index = center + k
    # First term: valid when k <= p (i.e., k from 0 to min(p, n-1))
    #   contrib[center+k] += (1-h[p-k])
    max_k1 = min(p, n - 1)
    if max_k1 >= 0:
        ks = np.arange(0, max_k1 + 1)
        contrib[center + ks] += 1.0 - h[p - ks]
    
    # Second term: valid when k <= n-1-p (i.e., k from 0 to min(n-1-p, n-1))
    #   contrib[center+k] -= h[p+k]
    max_k2 = min(n - 1 - p, n - 1)
    if max_k2 >= 0:
        ks = np.arange(0, max_k2 + 1)
        contrib[center + ks] -= h[p + ks]
    
    # Negative lags: m = 1, 2, ..., n-1, output index = center - m
    # First term: valid when p <= n-1-m (i.e., m <= n-1-p, m from 1 to min(n-1-p, n-1))
    #   contrib[center-m] += (1-h[p+m])
    max_m1 = min(n - 1 - p, n - 1)
    if max_m1 >= 1:
        ms = np.arange(1, max_m1 + 1)
        contrib[center - ms] += 1.0 - h[p + ms]
    
    # Second term: valid when m <= p (i.e., m from 1 to min(p, n-1))
    #   contrib[center-m] -= h[p-m]
    max_m2 = min(p, n - 1)
    if max_m2 >= 1:
        ms = np.arange(1, max_m2 + 1)
        contrib[center - ms] -= h[p - ms]
    
    return contrib
```
