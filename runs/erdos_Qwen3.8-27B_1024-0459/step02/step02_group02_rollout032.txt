def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)

            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt

    return h

def thue_morse(n):

        return h_src.copy()
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

    h = np.clip(x, 0.0, 1.0)

    return cmax + (1.0/beta) * np.log(np.sum(np.exp(shifted)))

    bounds = [(0.0, 1.0)] * n

            return smooth_max_obj(x, beta, n)

            return np.sum(x) - n / 2.0

            return np.ones(n)

        try:

                if c5_val < best_c5:
                    best_c5 = c5_val
                    current = candidate.copy()
        except Exception:
            pass

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_seq(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return h

def interp_to_n(h_src, n_dst):
    if len(h_src) == n_dst:
        return h_src.copy()
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def smooth_max_obj(x, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -60, 60)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def get_sum_con(n):
    def _f(x):
        return np.sum(x) - n / 2.0
    def _j(x):
        return np.ones(n)
    return _f, _j

def slsqp_smooth_max(h_init, n, n_steps=8, max_iter=2000, beta_start=15.0, beta_end=600.0):
    h0 = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h0)
    current = h0.copy()
    bounds = [(0.0, 1.0)] * n
    _cf, _cj = get_sum_con(n)
    con = {'type': 'eq', 'fun': _cf, 'jac': _cj}

    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** frac

        def obj(x):
            return smooth_max_obj(x, beta)

        try:
            res = minimize(obj, current, method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.fun < 1e10:
                candidate = make_feasible(res.x, n)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    current = candidate.copy()
        except Exception:
            pass
    return current, best_c5

def simulated_annealing(h_init, n, rng, n_sweeps=300, start_temp=0.08, end_temp=0.0005, timeout=300.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    start_t = time.time()

    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()

    for sweep in range(n_sweeps):
        if time.time() - start_t > timeout:
            break
        frac = sweep / n_sweeps
        temp = start_temp * (end_temp / start_temp) ** frac

        for _ in range(n):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue

            h[i] = 0.0
            h[j] = 1.0
            c5_new = compute_c5(h)

            if c5_new < current_c5:
                current_c5 = c5_new
                ones.remove(i)
                zeros.append(i)
                zeros.remove(j)
                ones.append(j)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                delta = c5_new - current_c5
                if delta > 0 and rng.rand() < np.exp(-delta / max(temp, 1e-12)):
                    current_c5 = c5_new
                    ones.remove(i)
                    zeros.append(i)
                    zeros.remove(j)
                    ones.append(j)
                else:
                    h[i] = 1.0
                    h[j] = 0.0

    return best_h, best_c5

def greedy_binary(h_init, n, rng, max_trials=500000, stall_limit=30000, timeout=120.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    no_improve = 0
    start_t = time.time()

    for _ in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

    return h, best_c5

def targeted_greedy(h_init, n, rng, max_trials=200000, stall_limit=15000, timeout=100.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    no_improve = 0
    start_t = time.time()

    for _ in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_idx = np.argmax(corr)
        worst_shift = worst_idx - (n - 1)

        bad_positions = []
        s = worst_shift
        if s > 0:
            for i in range(n - s):
                if h[i] == 1.0 and h[i + s] == 0.0:
                    bad_positions.append(i)
        elif s < 0:
            for i in range(-s, n):
                if h[i + s] == 1.0 and h[i] == 0.0:
                    bad_positions.append(i)

        if len(bad_positions) > 0 and len(zeros) > 0:
            i = bad_positions[rng.randint(len(bad_positions))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue

        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            ones.remove(i)
            zeros.append(i)
            zeros.remove(j)
            ones.append(j)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1

    return h, best_c5

def double_swap_sa(h_init, n, rng, n_sweeps=200, start_temp=0.06, end_temp=0.001, timeout=200.0):
    """Simulated annealing with 2-for-2 swaps."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    start_t = time.time()

    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()

    for sweep in range(n_sweeps):
        if time.time() - start_t > timeout:
            break
        frac = sweep / n_sweeps
        temp = start_temp * (end_temp / start_temp) ** frac

        for _ in range(n // 2):
            if len(ones) < 2 or len(zeros) < 2:
                break
            i = ones[rng.randint(len(ones))]
            k = ones[rng.randint(len(ones))]
            if i == k:
                continue
            j = zeros[rng.randint(len(zeros))]
            l = zeros[rng.randint(len(zeros))]
            if j == l:
                continue
            if len({i, k, j, l}) < 4:
                continue

            h[i] = 0.0
            h[k] = 0.0
            h[j] = 1.0
            h[l] = 1.0
            c5_new = compute_c5(h)

            if c5_new < current_c5:
                current_c5 = c5_new
                ones.remove(i)
                ones.remove(k)
                zeros.append(i)
                zeros.append(k)
                zeros.remove(j)
                zeros.remove(l)
                ones.append(j)
                ones.append(l)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                delta = c5_new - current_c5
                if delta > 0 and rng.rand() < np.exp(-delta / max(temp, 1e-12)):
                    current_c5 = c5_new
                    ones.remove(i)
                    ones.remove(k)
                    zeros.append(i)
                    zeros.append(k)
                    zeros.remove(j)
                    zeros.remove(l)
                    ones.append(j)
                    ones.append(l)
                else:
                    h[i] = 1.0
                    h[k] = 1.0
                    h[j] = 0.0
                    h[l] = 0.0

    return best_h, best_c5
