    dx = 2.0 / n

def smooth_grad(h, n, tau):
    """Gradient of smoothed C5"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exps = np.exp(tau * (corr - m))
    w = exps / np.sum(exps)  # softmax weights over lags
    
    grad = np.zeros(n)
    # For each lag m (from -(n-1) to n-1):
    # d(c_m)/d(h[i]) = (1-h[i-m])*[i-m valid] - h[i+m]*[i+m valid] (for m>=0)
    # For m=0: grad[i] += w_0 * dx * (1-h[i]) - w_0 * dx * h[i] = w_0*dx*(1-2h[i])
    # But c_0 = sum h[i](1-h[i]) which is not necessarily 0 for non-binary h!
    
    for idx_m in range(2*n-1):
        wm = w[idx_m]
        if wm < 1e-16:
            continue
        wm *= dx
        m = idx_m - (n-1)
        if m == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif m > 0:
            # For i in [m, n-1]: grad[i] += wm*(1-h[i-m])
            grad[m:] += wm * (1.0 - h[:n-m])
            # For i in [0, n-1-m]: grad[i] -= wm*h[i+m]
            grad[:n-m] -= wm * h[m:]
        else:
            ss = -m
            # For i in [0, n-1-ss]: grad[i] += wm*(1-h[i+ss])

            # For i in [ss, n-1]: grad[i] -= wm*h[i-ss]

        else:

            g[ss:] -= x[:n-ss]
        return g

    return None, None

The key advantage of binary solutions is that they have c[0] = 0 (no self-overlap), which is the best possible for that lag. The trade-off is that other lags might be higher.

Let me include a more sophisticated binary search: instead of random flips, try all possible 2-flips (swap a 1 with a 0) and pick the one that most reduces the maximum correlation.

For n=160, the number of 2-flip moves is C(80,1)*C(80,1) = 6400. Evaluating each takes O(n) time for the correlation update (if we update incrementally). Total: 6400*160 = 1M operations per "round". Each round maybe 10ms. So 100 rounds in 1 second. Very feasible.

Actually, updating the correlation incrementally after a flip:
If we flip position i from 0 to 1 (and position j from 1 to 0 to maintain sum):
- For each lag m, corr[m] changes by:
  - The contribution of h[i] changes from 0 to 1
  - The contribution of h[j] changes from 1 to 0
  
For corr[m] = Σ_{n valid} (1-h[n])*h[n+m]:
- Flipping h[i] from 0 to 1:
  - If i is used as n: (1-h[i]) changes from 1 to 0, so corr[m] decreases by h[i+m] (for m where i+m is valid)
  - If i is used as n+m (i.e., n=i-m): h[n+m]=h[i] changes from 0 to 1, so corr[m] increases by (1-h[i-m]) (for m where i-m is valid)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h_proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-12
        else:
            mask = h_proj > 1e-12
        if np.any(mask):
            h_proj[mask] += diff / np.count_nonzero(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return project_feasible(h_interp, n_dst)

def coord_opt_j(h, j, n, corr):
    a = np.zeros(2 * n - 1)
    for idx_m in range(2 * n - 1):
        m = idx_m - (n - 1)
        im = j - m
        ip = j + m
        if 0 <= im < n:
            a[idx_m] += 1.0 - h[im]
        if 0 <= ip < n:
            a[idx_m] -= h[ip]
    b = corr - h[j] * a
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-14
        sg = np.mean(a[active])
        if sg > 1e-12:
            hi = mid
        elif sg < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent(h, n, max_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    start = time.time()
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            h[j] = coord_opt_j(h, j, n, corr)
        h = project_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def smooth_obj(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = tau * (corr - m)
    shifted -= np.max(shifted)
    return m + (1.0 / tau) * np.log(np.sum(np.exp(shifted)))

def smooth_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exps = np.exp(tau * (corr - m))
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx_m in range(2 * n - 1):
        wm = w[idx_m]
        if wm < 1e-16:
            continue
        wm *= dx
        m_val = idx_m - (n - 1)
        if m_val == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif m_val > 0:
            grad[m_val:] += wm * (1.0 - h[:n - m_val])
            grad[:n - m_val] -= wm * h[m_val:]
        else:
            ss = -m_val
            grad[:n - ss] += wm * (1.0 - h[ss:])
            grad[ss:] -= wm * h[:n - ss]
    return grad

def smooth_optimize(h, n, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    tau = 15.0
    lr = 0.5
    momentum = 0.85
    v = np.zeros(n)
    patience = 0
    max_patience = 300
    it = 0
    while time.time() - start < time_limit:
        g = smooth_grad(h, n, tau)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        g = g / gn
        v = momentum * v + lr * g
        h_new = h - v
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h = h_new
        it += 1
        if patience > max_patience:
            rng_local = np.random.RandomState(it * 31)
            h = h + 0.05 * rng_local.randn(n)
            h = project_feasible(h, n)
            v = np.zeros(n)
            tau = max(10.0, tau * 0.5)
            lr = min(0.5, lr * 1.5)
            patience = 0
        tau *= 1.03
        lr *= 0.997
        if tau > 400.0:
            tau = 400.0
        if lr < 0.01:
            lr = 0.01
    return best_h, best_c5

def binary_local_search(h, n, max_trials, time_limit):
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = int(round(n / 2.0 - s))
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        if len(zeros) >= deficit:
            order = np.argsort(-h[zeros])[:deficit]
            h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        if len(ones) >= -deficit:
            order = np.argsort(h[ones])[:(-deficit)]
            h_bin[ones[order]] = 0.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    start = time.time()
    for _ in range(max_trials):
        if time.time() - start > time_limit:
            break
        n_flip = rng.randint(1, 4)
        idxs = rng.choice(n, n_flip, replace=False)
        h_try = best_h.copy()
        for idx in idxs:
            h_try[idx] = 1.0 - h_try[idx]
        if abs(np.sum(h_try) - n / 2.0) < 1e-10:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5

_slsq_n = 0

def _slsqp_obj(x):
    return np.max(np.correlate(x, 1.0 - x, mode='full'))

def _slsqp_obj_grad(x):
    n = _slsqp_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    kmax = np.argmax(corr)
    m = kmax - (n - 1)
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * x
    elif m > 0:
        g[m:] += (1.0 - x[:n - m])
        g[:n - m] -= x[m:]
    else:
        ss = -m
        g[:n - ss] += (1.0 - x[ss:])
        g[ss:] -= x[:n - ss]
    return g

def _slsqp_cons(x):
    return np.sum(x) - _slsqp_n / 2.0

def _slsqp_cons_jac(x):
    return np.ones(_slsqp_n)

def slsqp_polish(h, n, max_iter=5000):
    global _slsq_n
    _slsq_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_cons_jac}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=_slsqp_obj_grad,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = project_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None
