def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)

    if n == 0:
        return x
    target = float(target)
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    lo, hi = -2.0, 2.0

        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)

def upsample_h(parent, n):

def corr_grad_lag(h, k):
    n = h.size
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

def active_gradient(h, top_frac=0.08, min_k=10, temperature=0.01):
    n = h.size
    corr = compute_correlation(h)
    center = n - 1
    k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
    top_idx = np.argsort(corr)[-k_count:]
    top_vals = corr[top_idx]
    dv = top_vals - top_vals.min()
    scale = dv.max() + 1e-10

    w /= w.sum()
    g = np.zeros(n, dtype=float)
    for i, t in enumerate(top_idx):

        g += w[i] * corr_grad_lag(h, k)
    return g

            return 1e100
        corr = np.correlate(h2, 1.0 - h2, mode="full")
        m = float(np.max(corr))
        alpha = 50.0
        y = alpha * (corr - m)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0 or not np.isfinite(z):
            return 1e100
        return (2.0 / n) * (m + np.log(z) / alpha)
    
    def smooth_jac(x):
        h2 = np.asarray(x, dtype=float)
        if not np.all(np.isfinite(h2)):
            return np.zeros(n)
        corr = np.correlate(h2, 1.0 - h2, mode="full")
        m = float(np.max(corr))
        alpha = 50.0
        y = alpha * (corr - m)
        y = np.clip(y, -700.0, 0.0)
        e = np.exp(y)
        z = float(np.sum(e))
        if z <= 0 or not np.isfinite(z):
            return np.zeros(n)
        w = e / z
        center = n - 1
        g = np.zeros(n)
        idx = np.where(w > 1e-12)[0]
        for t in idx:
            t = int(t)
            p = float(w[t])
            k = t - center
            gl = corr_grad_lag(h2, k)
            g += p * gl
        return (2.0 / n) * g
    
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)},)
    
    try:
        res = minimize(smooth_obj, best, method='SLSQP', jac=smooth_jac,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            cand = project_density(res.x, target)
            c = evaluate_erdos(cand)
            if c < best_c:
                best = cand
                best_c = c
    except:
        pass
    
    return project_density(best, target), evaluate_erdos(project_density(best, target))

def full_optimize(n, h0, deadline):
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    
    # Phase 1: SLSQP
    rem = deadline - time.perf_counter()
    if rem > 5 and n <= 500:
        h, c = slsqp_optimize(h, time.perf_counter() + min(rem * 0.25, 60))
        if c < best_c:
            best_h = h
            best_c = c
    
    # Phase 2: Gradient descent
    rem = deadline - time.perf_counter()
    if rem > 5:
        h, c = gradient_descent_step(best_h, time.perf_counter() + rem * 0.2,
                                      max_iter=8000, top_frac=0.08, min_k=10,
                                      base_lr=0.015, lr_decay=0.9998, patience=300)
        if c < best_c:
            best_h = h
            best_c = c
    
    # Phase 3: Simulated annealing
    rem = deadline - time.perf_counter()
    if rem > 8:
        h, c = simulated_annealing(best_h, time.perf_counter() + rem * 0.25,
                                    max_iter=30000, T_init=0.04, T_min=0.002,
                                    top_frac=0.06, min_k=8, move_size=0.06)
        if c < best_c:
            best_h = h
            best_c = c
    
    # Phase 4: Block moves
    rem = deadline - time.perf_counter()
    if rem > 8:
        h, c = block_move_search(best_h, time.perf_counter() + rem * 0.25,
                                  max_iter=5000, top_frac=0.06, min_k=8)
        if c < best_c:
            best_h = h
            best_c = c
    
    # Phase 5: Pair swap intensive
    rem = deadline - time.perf_counter()
    if rem > 5:
        h, c = pair_swap_search(best_h, time.perf_counter() + rem * 0.35,
                                 max_iter=8000, top_frac=0.04, min_k=10, n_try=10)
        if c < best_c:
            best_h = h
            best_c = c
    
    # Phase 6: Final gradient polish
    rem = deadline - time.perf_counter()
    if rem > 5:
        h, c = gradient_descent_step(best_h, time.perf_counter() + rem * 0.5,
                                      max_iter=10000, top_frac=0.04, min_k=12,
                                      base_lr=0.005, lr_decay=0.9999, patience=200)
        if c < best_c:
            best_h = h
            best_c = c
    
    # Phase 7: Final pair swap
    rem = deadline - time.perf_counter()
    if rem > 3:
        h, c = pair_swap_search(best_h, time.perf_counter() + rem * 0.8,
                                 max_iter=5000, top_frac=0.03, min_k=12, n_try=6)
        if c < best_c:
            best_h = h
            best_c = c
    
    return project_density(best_h, target), evaluate_erdos(project_density(best_h, target))

def make_diverse_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 5 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.empty(n, dtype=float)
        seg = max(4, n // 10)
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = rng.uniform(0.1, 0.9)
    elif mode == 2:
        u = rng.random(n)
        med = np.median(u)
        h = np.where(u > med, 0.7, 0.3).astype(float)
        h += rng.normal(0, 0.04, n)
    elif mode == 3:
        freq = rng.integers(2, 8)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2 + rng.uniform(0, 6.28))
        h += 0.1 * np.sin(2 * np.pi * (freq + 2) * x / 2 + rng.uniform(0, 6.28))
    else:
        # Blocky
        nblocks = rng.integers(4, 12)
        h = np.empty(n)
        block = n // nblocks
        for i in range(0, n, block):
            end = min(i + block, n)
            h[i:end] = rng.uniform(0, 1)
    return project_density(h, 0.5 * n)

def de_search(n, popsize, max_iter, deadline, seed=42):
    target = 0.5 * n
    rng = np.random.default_rng(seed)
    
    def de_obj(x):
        h = np.clip(x, 0.0, 1.0)
        h = project_density(h, target)
        return evaluate_erdos(h)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = differential_evolution(de_obj, bounds, maxiter=max_iter,
                                      popmax=popsize, seed=seed, tol=1e-8,
                                      polish=False, updateinterval=0,
                                      seed=seed)
        if res.x is not None:
            h = project_density(res.x, target)
            return h, evaluate_erdos(h)
    except:
        pass
    return None, float('inf')

def run(seed=42, budget_s=1300, **kwargs):
    start_t = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    hard_deadline = start_t + max(10.0, budget - 10.0)
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float('inf')
    best_n = 0
    
    # Get parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0).copy()
    except:
        pass
    
    # Initialize from parent or random
    if parent is not None:
        pn = parent.size
        pt = 0.5 * pn
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = pn
    else:
        n0 = 90
        x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x0 / 2) + 0.1 * np.sin(2 * np.pi * 5 * x0 / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0
    
    # Phase A: DE at low resolution for diversity (if time allows)
    rem = hard_deadline - time.perf_counter()
    de_starts = []
    if rem > 300:
        de_n = 90
        de_deadline = time.perf_counter() + min(60, rem * 0.1)
        h_de, c_de = de_search(de_n, popsize=25, max_iter=40, deadline=de_deadline, seed=seed)
        if h_de is not None and c_de < best_c + 0.05:
            de_starts.append(h_de)
            if c_de < best_c:
                best_h = h_de.copy()
                best_c = c_de
                best_n = de_n
        
        # More diverse starts at low res
        for mode in range(4):
            h_d = make_diverse_start(de_n, rng, mode)
            c_d = evaluate_erdos(h_d)
            if c_d > best_c - 0.05:  # not too bad
                de_starts.append(h_d)
    
    # Phase B: Optimize at parent resolution (n=450 or similar)
    rem = hard_deadline - time.perf_counter()
    main_n = best_n if best_n >= 100 else 180
    main_n = min(main_n, 450)  # cap for SLSQP feasibility
    
    # Build starting points for main optimization
    starts_main = []
    starts_main.append(upsample_h(best_h, main_n))
    if parent is not None and parent.size != main_n:
        starts_main.append(upsample_h(parent, main_n))
    for d_s in de_starts:
        starts_main.append(upsample_h(d_s, main_n))
    # Add a few diverse starts
    for mode in range(3):
        starts_main.append(make_diverse_start(main_n, rng, mode + 1))
    
    # Deduplicate
    uniq_starts = []
    for s in starts_main:
        s = project_density(s, 0.5 * main_n)
        dup = False
        for u in uniq_starts:
            if np.allclose(s, u, atol=5e-4):
                dup = True
                break
        if not dup:
            uniq_starts.append(s)
    
    # Allocate time for main optimization
    n_starts = len(uniq_starts)
    rem = hard_deadline - time.perf_counter()
    time_per_start = min(rem * 0.45 / max(1, n_starts), 80.0)
    time_per_start = max(15.0, time_per_start)
    
    for si, s in enumerate(uniq_starts):
        rem = hard_deadline - time.perf_counter()
        if rem < 15:
            break
        sub_dl = time.perf_counter() + min(time_per_start, rem - 5)
        h_out, c_out = full_optimize(main_n, s, sub_dl)
        if h_out is not None:
            c_out = evaluate_erdos(h_out)
            if c_out < best_c:
                best_h = h_out.copy()
                best_c = c_out
                best_n = main_n
    
    # Phase C: Higher resolution try
    rem = hard_deadline - time.perf_counter()
    if rem > 120 and best_n <= 500:
        hi_n = min(int(best_n * 2), 900)
        if hi_n > best_n + 50:
            hi_start = upsample_h(best_h, hi_n)
            hi_deadline = time.perf_counter() + min(rem * 0.3, 200)
            h_hi, c_hi = full_optimize(hi_n, hi_start, hi_deadline)
            if h_hi is not None:
                c_hi = evaluate_erdos(h_hi)
                if c_hi < best_c:
                    best_h = h_hi.copy()
                    best_c = c_hi
                    best_n = hi_n
    
    # Phase D: Final intensive polish on the best
    rem = hard_deadline - time.perf_counter()
    if best_h is not None and rem > 20:
        n_best = best_h.size
        target = 0.5 * n_best
        
        # Gradient descent with very fine lr
        h, c = gradient_descent_step(best_h, time.perf_counter() + rem * 0.2,
                                      max_iter=20000, top_frac=0.05, min_k=15,
                                      base_lr=0.003, lr_decay=0.9999, patience=300)
        if c < best_c:
            best_h = h
            best_c = c
        
        rem = hard_deadline - time.perf_counter()
        # Pair swap very intensive
        if rem > 15:
            h, c = pair_swap_search(best_h, time.perf_counter() + rem * 0.3,
                                     max_iter=20000, top_frac=0.03, min_k=15, n_try=12)
            if c < best_c:
                best_h = h
                best_c = c
        
        rem = hard_deadline - time.perf_counter()
        # Simulated annealing final
        if rem > 15:
            h, c = simulated_annealing(best_h, time.perf_counter() + rem * 0.3,
                                        max_iter=50000, T_init=0.02, T_min=0.001,
                                        top_frac=0.04, min_k=12, move_size=0.04)
            if c < best_c:
                best_h = h
                best_c = c
        
        rem = hard_deadline - time.perf_counter()
        # Final gradient
        if rem > 10:
            h, c = gradient_descent_step(best_h, time.perf_counter() + rem * 0.4,
                                          max_iter=15000, top_frac=0.03, min_k=15,
                                          base_lr=0.002, lr_decay=0.9999, patience=200)
            if c < best_c:
                best_h = h
                best_c = c
        
        rem = hard_deadline - time.perf_counter()
        # Last pair swap
        if rem > 5:
            h, c = pair_swap_search(best_h, time.perf_counter() + rem * 0.7,
                                     max_iter=10000, top_frac=0.02, min_k=15, n_try=8)
            if c < best_c:
                best_h = h
                best_c = c
    
    # Final validation and cleanup
    final_h = project_density(best_h, 0.5 * best_h.size)
    final_c = evaluate_erdos(final_h)
    
    # Ensure sum constraint is met exactly
    if abs(float(np.sum(final_h)) - 0.5 * final_h.size) > 1e-10:
        final_h = project_density(final_h, 0.5 * final_h.size)
        final_c = evaluate_erdos(final_h)
    
    return final_h, float(final_c), int(final_h.size)
