def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size

        return float("inf")

def corr_grad(h, k):
    """Gradient of correlation at lag k with respect to h"""
    n = h.size

        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

    best_c = float("inf")
    best_n = 0
    
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0).copy()
    except Exception:
        parent = None
    
    if parent is not None:
        np_ = parent.size
        pp = project_density(parent, 0.5 * np_)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_
    
    if best_h is None:
        n0 = 120
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2*np.pi*3*x/2) + 0.1 * np.sin(2*np.pi*7*x/2)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0
    
    remaining = deadline - time.perf_counter()
    
    # Phase 1: DE global exploration at moderate resolution
    if remaining > 60:
        de_n = 160
        de_time = min(40.0, remaining * 0.2)
        de_deadline = time.perf_counter() + de_time
        de_h, de_c = run_de_search(de_n, de_deadline, seed, popsize=14, maxiter=50)
        if de_h is not None and de_c < best_c:
            best_h = de_h.copy()
            best_c = de_c
            best_n = de_n
        
        # Second DE with different params
        remaining2 = deadline - time.perf_counter()
        if remaining2 > 30:
            de_n2 = 200
            de_time2 = min(30.0, remaining2 * 0.2)
            de_deadline2 = time.perf_counter() + de_time2
            de_h2, de_c2 = run_de_search(de_n2, de_deadline2, seed + 1, popsize=12, maxiter=40)
            if de_h2 is not None and de_c2 < best_c:
                best_h = de_h2.copy()
                best_c = de_c2
                best_n = de_n2
    
    # Phase 2: Multi-resolution local refinement
    base_n = best_n
    res_list = sorted(set([
        max(100, base_n),
        int(base_n * 1.25),
        int(base_n * 1.5),
        int(base_n * 2),
    ]))
    res_list = [r for r in res_list if 80 <= r <= 800]
    
    remaining = deadline - time.perf_counter()
    n_res = max(1, len(res_list))
    time_budget = remaining / (n_res + 1.5)
    
    for ni in res_list:
        rem = deadline - time.perf_counter()
        if rem < 25:
            break
        alloc = min(time_budget * 1.5, rem - 12)
        target = 0.5 * ni
        
        # Starting point: upsample best
        h_start = upsample_h(best_h, ni)
        h_start = project_density(h_start, target)
        
        # Sub-gradient refinement
        sg_deadline = time.perf_counter() + alloc * 0.3
        h1, c1 = subgrad_refine(h_start, sg_deadline, target, max_iter=2000, top_frac=0.05, min_k=5)
        if c1 < best_c:
            best_h = h1.copy()
            best_c = c1
            best_n = ni
        
        # Pair swap refinement
        rem2 = deadline - time.perf_counter()
        if rem2 > 15:
            ps_deadline = time.perf_counter() + min(alloc * 0.3, rem2 * 0.4)
            h2, c2 = pair_swap_refine(h1, ps_deadline, target, max_iter=5000, top_frac=0.03, min_k=6, n_pool=10)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2
                best_n = ni
        
        # Block swap
        rem3 = deadline - time.perf_counter()
        if rem3 > 15 and ni <= 500:
            bs_deadline = time.perf_counter() + min(alloc * 0.2, rem3 * 0.3)
            h3, c3 = block_swap_refine(h2, bs_deadline, target, max_iter=1500, min_k=6, top_frac=0.04)
            if c3 < best_c:
                best_h = h3.copy()
                best_c = c3
                best_n = ni
        
        # Second round of subgradient + pair swap
        rem4 = deadline - time.perf_counter()
        if rem4 > 15:
            h4, c4 = subgrad_refine(h2, time.perf_counter() + min(alloc * 0.15, rem4 * 0.3), target, max_iter=1000, top_frac=0.03, min_k=8)
            if c4 < best_c:
                best_h = h4.copy()
                best_c = c4
                best_n = ni
            rem5 = deadline - time.perf_counter()
            if rem5 > 10:
                h5, c5 = pair_swap_refine(h4, time.perf_counter() + min(alloc * 0.15, rem5 * 0.4), target, max_iter=3000, top_frac=0.03, min_k=8, n_pool=12)
                if c5 < best_c:
                    best_h = h5.copy()
                    best_c = c5
                    best_n = ni
    
    # Phase 3: Intensive final polish
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 20:
        nbest = best_h.size
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        c_fin = evaluate_erdos(h_fin)
        
        # Binarize push
        h_bin = binarize_push(h_fin, target, thresh=0.08)
        c_bin = evaluate_erdos(h_bin)
        if c_bin < c_fin:
            h_fin = h_bin
            c_fin = c_bin
        
        # Multiple refinement rounds
        for rnd in range(5):
            rem2 = deadline - time.perf_counter()
            if rem2 < 8:
                break
            t_round = rem2 / (5 - rnd)
            
            # Subgradient
            h_a, c_a = subgrad_refine(h_fin, time.perf_counter() + t_round * 0.35, target,
                                       max_iter=2000, top_frac=0.03, min_k=8)
            if c_a < c_fin:
                h_fin = h_a
                c_fin = c_a
            
            # Pair swap
            rem3 = deadline - time.perf_counter()
            if rem3 > 8:
                h_b, c_b = pair_swap_refine(h_fin, time.perf_counter() + t_round * 0.35, target,
                                             max_iter=5000, top_frac=0.025, min_k=10, n_pool=14)
                if c_b < c_fin:
                    h_fin = h_b
                    c_fin = c_b
            
            # Block swap
            rem4 = deadline - time.perf_counter()
            if rem4 > 8 and nbest <= 500:
                h_c, c_c = block_swap_refine(h_fin, time.perf_counter() + t_round * 0.2, target,
                                              max_iter=800, min_k=8, top_frac=0.03)
                if c_c < c_fin:
                    h_fin = h_c
                    c_fin = c_c
        
        if c_fin < best_c:
            best_h = h_fin
            best_c = c_fin
            best_n = nbest
    
    # Final validation and return
    if best_h is None:
        fallback = project_density(np.full(100, 0.5), 50.0)
        return fallback, evaluate_erdos(fallback), 100
    
    final_h = project_density(best_h, 0.5 * best_h.size)
    final_c = evaluate_erdos(final_h)
    
    return final_h, float(final_c), int(final_h.size)
