def logsumexp_objective(h_params, n, gamma, dx):
    """
    Minimize log-sum-exp of correlations as smooth approximation to max.
    h_params is the first n-1 components; last is determined by sum constraint.
    """
    h = np.empty(n)
    h[:n-1] = h_params
    h[n-1] = n/2.0 - np.sum(h_params)
    if h[n-1] < -1e-10 or h[n-1] > 1.0 + 1e-10:
        return 1e10
    h = np.clip(h, 0.0, 1.0)  # soft clip for numerical safety
    # Actually, we should not clip here as it makes the gradient wrong.
    # Let's just check feasibility.
    corr = get_corr(h)
    vals = gamma * corr * dx
    # Numerically stable log-sum-exp
    vals_shifted = vals - np.max(vals)
    lse = np.max(vals) / gamma + np.log(np.sum(np.exp(vals_shifted))) / gamma
    return lse

def logsumexp_gradient(h_params, n, gamma, dx):

    h[n-1] = n/2.0 - np.sum(h_params)

    vals = gamma * corr * dx
    vals_shifted = vals - np.max(vals)
    exp_vals = np.exp(vals_shifted)
    weights = exp_vals / np.sum(exp_vals)  # softmax weights
    # grad w.r.t. h[j] for j = 0..n-2
    # d/dh[j] of (1/gamma)*log(sum exp(gamma*corr(ell)*dx))
    # = dx * sum_ell weights[ell] * d corr(ell)/d h[j]
    # d corr(ell)/d h[j] = (1-h[j+ell])*1[valid] - h[j-ell]*1[valid]
    ells = np.arange(-(n-1), n)

def binary_local_search(h, n, max_trials, time_limit):
