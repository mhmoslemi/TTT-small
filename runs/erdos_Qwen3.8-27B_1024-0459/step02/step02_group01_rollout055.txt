                h[a] = 1.0

        if not improved:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            no_improve = 0

    return best_h, best_max * 2.0 / n

def smooth_max_obj(x, n, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    sh = beta * (corr - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / beta
    return smax * (2.0 / n)

def eq_constraint(x, n):
    return np.sum(x) - n / 2.0

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    parent_h = None
    parent_n = 0
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            best_h = parent_h.copy()
            best_c5 = compute_c5(parent_h)
            best_n = parent_n
    except Exception:
        pass

    if parent_h is None or parent_n == 0:
        parent_n = 192
        parent_h = np.ones(parent_n) * 0.5
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = parent_n

    def remaining():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)

    # Phase 1: Binary swap search at multiple resolutions
    # Resolutions to try (must be even)
    n_candidates = [parent_n]
    for nn in [256, 320, 384, 448, 512, 576, 640, 768, 896, 1024]:
        if nn != parent_n and nn % 2 == 0:
            n_candidates.append(nn)
    n_candidates = sorted(set([nn for nn in n_candidates if nn >= 64 and nn <= 1024]))

    # Allocate time: more time for mid-to-high resolutions
    total_swap_time = remaining() * 0.55
    n_res = len(n_candidates)
    
    for idx_n, n in enumerate(n_candidates):
        if remaining() < 20:
            break

        # Time allocation: weight higher resolutions more
        weight = 1.0 + 0.3 * idx_n
        time_this_res = min(80.0, total_swap_time * weight / max(1.0, sum(1.0 + 0.3 * i for i in range(idx_n, n_res))))
        if time_this_res < 5.0:
            continue

        # Create multiple starts
        starts = []

        # Start 1: Interpolated parent -> binary
        h_interp = interp_h(parent_h, parent_n, n)
        starts.append(project_binary(h_interp, n))

        # Start 2: Random binary
        idx_rand = rng.permutation(n)
        h_rand = np.zeros(n)
        h_rand[idx_rand[:n // 2]] = 1.0
        starts.append(h_rand)

        # Start 3: Best so far interpolated
        if best_n != n:
            h_best_interp = interp_h(best_h, best_n, n)
            starts.append(project_binary(h_best_interp, n))

        # Start 4: Structured (golden-ratio based)
        phi = 0.618033988749895
        h_struct = np.zeros(n)
        positions = set()
        for i in range(n // 2):
            pos = int((i * phi * n) % n)
            h_struct[pos] = 1.0
        # Fix count
        n_ones_now = int(np.sum(h_struct))
        if n_ones_now < n // 2:
            remaining_pos = [i for i in range(n) if h_struct[i] == 0]
            for p in remaining_pos[:n // 2 - n_ones_now]:
                h_struct[p] = 1.0
        elif n_ones_now > n // 2:
            ones_pos = np.where(h_struct == 1)[0]
            for p in ones_pos[:n_ones_now - n // 2]:
                h_struct[p] = 0.0
        starts.append(h_struct)

        # Limit starts based on time
        time_per_start = time_this_res / max(1, len(starts))
        time_per_start = max(5.0, min(40.0, time_per_start))

        for h_start in starts:
            if remaining() < 10:
                break

            # First targeted swap, then SA
            h_t, c5_t = targeted_swap(h_start, n, min(time_per_start * 0.4, 15.0), rng)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n

            if remaining() > 10:
                h_sa, c5_sa = swap_sa_search(h_t, n, min(time_per_start * 0.6, remaining() - 5), rng)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n

    # Phase 2: Continuous SLSQP refinement on best binary solution
    if remaining() > 20:
        n_ref = best_n
        h_bin_best = project_binary(best_h, n_ref)

        # Relax: allow small continuous perturbations
        for pert_scale in [0.05, 0.15, 0.3]:
            if remaining() < 8:
                break
            h_relaxed = h_bin_best + pert_scale * rng.randn(n_ref)
            h_relaxed = project_feasible_cont(h_relaxed, n_ref)
            bounds = [(0.0, 1.0)] * n_ref
            cons = [{'type': 'eq', 'fun': lambda x, n=n_ref: np.sum(x) - n / 2.0}]

            for beta in [500.0, 5000.0, 50000.0]:
                if remaining() < 5:
                    break
                try:
                    res = minimize(lambda x: smooth_max_obj(x, n_ref, beta), h_relaxed,
                                   method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 150, 'ftol': 1e-15, 'iprint': 0})
                    h_res = project_feasible_cont(res.x, n_ref)
                    c5_res = compute_c5(h_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = h_res.copy()
                        best_n = n_ref
                    # Also try rounding
                    h_rounded = project_binary(h_res, n_ref)
                    c5_r = compute_c5(h_rounded)
                    if c5_r < best_c5:
                        best_c5 = c5_r
                        best_h = h_rounded.copy()
                        best_n = n_ref
                    h_relaxed = h_res.copy()
                except Exception:
                    pass

    # Phase 3: Final targeted swap on the absolute best
    if remaining() > 5:
        h_final_bin = project_binary(best_h, best_n)
        c5_check = compute_c5(h_final_bin)
        h_targeted, c5_final_sw = targeted_swap(h_final_bin, best_n, min(40.0, remaining() - 2), rng)
        if c5_final_sw < best_c5:
            best_c5 = c5_final_sw
            best_h = h_targeted.copy()

        # Also try SA on the best
        if remaining() > 10:
            h_sa_final, c5_sa_final = swap_sa_search(best_h, best_n, min(30.0, remaining() - 2), rng,
                                                      t_init=0.02, t_final=0.0002)
            if c5_sa_final < best_c5:
                best_c5 = c5_sa_final
                best_h = h_sa_final.copy()

    # Phase 4: Validate and finalize
    best_h = np.array(best_h, dtype=float)
    best_n = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)

    # Ensure sum constraint
    if np.allclose(best_h, best_h.round(), atol=1e-10):
        n_ones = int(np.round(best_h).sum())
        target = best_n // 2
        if n_ones != target:
            best_h = project_binary(best_h, best_n)
    else:
        diff = best_n / 2.0 - np.sum(best_h)
        best_h += diff / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < best_c5:
            best_c5 = ec5
    except Exception:
        pass

    return (best_h, best_c5, best_n)
