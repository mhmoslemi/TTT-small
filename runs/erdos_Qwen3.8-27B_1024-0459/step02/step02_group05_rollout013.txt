import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 0.0
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} via bisection."""
    h = np.array(h, dtype=float)
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = result < 1.0 - 1e-10
        else:
            mask = result > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_obj_grad(h, beta):
    """Compute LSE smooth max of correlation and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    # For numerical stability
    c_shifted = corr - cmax
    log_sum = np.log(np.sum(np.exp(beta * c_shifted)))
    f = (cmax + log_sum) / beta
    
    # Softmax weights
    w = np.exp(beta * c_shifted)
    w /= np.sum(w)
    
    # Gradient: sum_k w[k] * dcorr_k/dh
    # dcorr_k/dh[j] = (1-h[j-lag_k])*I(0<=j-lag_k<n) - h[j+lag_k]*I(0<=j+lag_k<n)
    # where lag_k = k - (n-1)
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    for j in range(n):
        # (1-h[j-lag]) for valid j-lag in [0,n)
        j_minus_lag = j - lags  # shape (2n-1,)
        valid1 = (j_minus_lag >= 0) & (j_minus_lag < n)
        # h[j+lag] for valid j+lag in [0,n)
        j_plus_lag = j + lags
        valid2 = (j_plus_lag >= 0) & (j_plus_lag < n)
        
        term1 = np.where(valid1, 1.0 - h[j_minus_lag], 0.0)
        term2 = np.where(valid2, h[j_plus_lag], 0.0)
        
        grad[j] = np.dot(w, term1 - term2)
    
    return f, grad

def projected_gradient_solve(h, n, beta, lr, n_iter, verbose=False):
    """Projected gradient method with LSE smooth max."""
    h = project_feasible(h.copy(), n)
    best_f = np.max(np.correlate(h, 1.0 - h, mode='full'))
    best_h = h.copy()
    
    for it in range(n_iter):
        f, grad = lse_obj_grad(h, beta)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        # Backtracking line search
        t = 1.0
        while t > 1e-6:
            h_try = h - t * lr * grad
            h_try = project_feasible(h_try, n)
            c_try = np.max(np.correlate(h_try, 1.0 - h_try, mode='full'))
            if c_try < best_f or c_try < f:
                break
            t *= 0.5
        h = h_try
        c_now = np.max(np.correlate(h, 1.0 - h, mode='full'))
        if c_now < best_f:
            best_f = c_now
            best_h = h.copy()
        
        if verbose and it % 100 == 0:
            print(f"  PG iter {it}: f={f:.6f}, best={best_f:.6f}")
    
    return best_h, best_f

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def lbfgsb_solve(h_init, n, beta, n_outer=5):
    """L-BFGS-B with penalty for sum constraint."""
    h = h_init.copy()
    target = n / 2.0
    mu = 1.0
    
    for outer in range(n_outer):
        def obj(x):
            c = np.max(np.correlate(x, 1.0 - x, mode='full'))
            cshift = np.correlate(x, 1.0 - x, mode='full')
            cshift = cshift - c
            lse = c + np.log(np.sum(np.exp(beta * cshift))) / beta
            pen = mu * (np.sum(x) - target) ** 2
            return lse + pen
        
        def obj_grad(x):
            f, g = lse_obj_grad(x, beta)
            g = g + 2.0 * mu * (np.sum(x) - target)
            return f + mu * (np.sum(x) - target) ** 2, g
        
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 10000})
            h = project_feasible(res.x, n)
        except Exception:
            pass
        mu *= 10.0
    
    return h, np.max(np.correlate(h, 1.0 - h, mode='full'))

def pair_swap_local(h, n, n_rounds, rng):
    """Local search via pair swaps (maintains sum automatically)."""
    h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for i in range(n - 1):
            for j in range(i + 1, min(i + 20, n)):  # Local swaps
                if abs(h[i] - h[j]) < 1e-8:
                    continue
                for step in [0.01, 0.02, 0.05, 0.1, 0.2]:
                    amt = min(step, abs(h[i] - h[j]))
                    if h[i] > h[j]:
                        h_try = h.copy()
                        h_try[i] -= amt
                        h_try[j] += amt
                    else:
                        h_try = h.copy()
                        h_try[i] += amt
                        h_try[j] -= amt
                    h_try = np.clip(h_try, 0.0, 1.0)
                    if abs(np.sum(h_try) - n/2.0) < 1e-10:
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            h = h_try
                            best_c5 = c5_try
                            improved = True
                            break
        if not improved:
            break
    return h, best_c5

    return h, best_c5
