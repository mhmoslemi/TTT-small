def _single_delta(h, p, s):
    n = len(h)
    c_len = 2*n - 1
    center = n - 1
    d = np.zeros(c_len, dtype=float)
    if s == 0.0: return d
    L = p
    if L > 0:
        # positive lags +M for M=1..L, partners p-1,...,0
        d[center+1:center+1+L] += s * (1.0 - h[p-1::-1])
        # negative lags -M for M=L..1, partners 0..L-1

import numpy as np
from scipy.optimize import minimize


_SOL_N = 1
_SOL_BETA = 1.0
_SOL_TARGET = 0.0
_PEN_MU = 0.0


def compute_corr(h):
    h = np.asarray(h, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    return float(np.max(compute_corr(h)) * (2.0 / float(len(h))))


def project_box_sum(y, n):
    y = np.asarray(y, dtype=float)
    if y.size != n:
        y = np.resize(y, n)
    y = np.where(np.isfinite(y), y, 0.5)
    target = 0.5 * float(n)

    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0

    for _ in range(80):
        lam = 0.5 * (lo + hi)
        s = float(np.clip(y - lam, 0.0, 1.0).sum())
        if s > target:
            lo = lam
        else:
            hi = lam

    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(100):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            idxs = np.nonzero(x < 1.0 - 1e-14)[0]
            if idxs.size == 0:
                break
            cap = 1.0 - x[idxs]
            total_cap = float(cap.sum())
            if total_cap <= 1e-15:
                break
            frac = min(1.0, diff / total_cap)
            x[idxs] += cap * frac
        else:
            idxs = np.nonzero(x > 1e-14)[0]
            if idxs.size == 0:
                break
            cap = x[idxs]
            total_cap = float(cap.sum())
            if total_cap <= 1e-15:
                break
            frac = min(1.0, -diff / total_cap)
            x[idxs] -= cap * frac

    x = np.clip(x, 0.0, 1.0)
    return x


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    if len(h_src) < 2:
        return project_box_sum(np.full(n_dst, 0.5), n_dst)
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)

    z = np.interp(x_new, x_old, h_src)

def set_solver_state(n, beta):
    global _SOL_N, _SOL_BETA, _SOL_TARGET
    _SOL_N = int(n)
    _SOL_BETA = float(beta)
    _SOL_TARGET = 0.5 * float(n)


def _grad_from_weights(h, p):
    h = np.asarray(h, dtype=float)
    p = np.asarray(p, dtype=float)
    n = len(h)
    center = n - 1
    if p.size != 2 * n - 1 or not np.all(np.isfinite(p)):
        p = np.zeros(2 * n - 1, dtype=float)
        p[center] = 1.0
    s = float(p.sum())
    if s <= 0.0:
        p = np.zeros(2 * n - 1, dtype=float)
        p[center] = 1.0
        s = 1.0
    p = p / s

    grad = p[center] * (1.0 - 2.0 * h)
    if n > 1:
        wP = p[center + 1:]
        wN = p[center - 1::-1]
        for M in range(1, n):
            WP = float(wP[M - 1])
            WN = float(wN[M - 1])
            W = WP + WN
            if W != 0.0:
                grad[M:] -= W * h[:-M]
                grad[:-M] -= W * h[M:]
            if WP != 0.0:
                grad[M:] += WP
            if WN != 0.0:
                grad[:-M] += WN
    return grad


def _raw_smooth_value(x):
    c = compute_corr(x)
    m = float(np.max(c))
    beta = _SOL_BETA
    if beta <= 1e-8:
        return m
    e = np.exp(beta * (c - m))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return m
    return float(m + np.log(s) / beta)


def _raw_smooth_grad(x):
    c = compute_corr(x)
    m = float(np.max(c))
    beta = _SOL_BETA
    if beta <= 1e-8:
        p = (c >= m - 1e-9).astype(float)
    else:
        p = np.exp(beta * (c - m))
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[len(c) // 2] = 1.0
    return _grad_from_weights(x, p)


def smooth_gradient_explicit(h, beta):

def _sol_grad(x):

def _sol_eq_fun(x):

def slsqp_smooth_solve(n, h, beta, max_iter, ftol):
    try:
        set_solver_state(n, beta)

            _sol_fun,

            constraints=cons,

        )
        if res.fun < 1e100 and np.all(np.isfinite(res.x)):
            x = project_box_sum(res.x, n)
            val = compute_c5(x)
            if np.isfinite(val):

    except Exception:
        pass
    return None, None


def _pen_fun(x):
    val = _raw_smooth_value(x)
    mu = _PEN_MU
    if mu != 0.0:
        d = float(np.sum(x) - _SOL_TARGET)
        val += mu * d * d
    return val


def _pen_grad(x):

    mu = _PEN_MU
    if mu != 0.0:
        d = float(np.sum(x) - _SOL_TARGET)
        if d != 0.0:
            g = g + 2.0 * mu * d
    return g


def lbfgs_smooth_solve(n, h, beta, mu, max_iter):

        _PEN_MU = float(mu)

    except Exception:

    return None, None


def _single_delta(h, p, s):
    h = np.asarray(h, dtype=float)
    n = len(h)
    c_len = 2 * n - 1
    center = n - 1
    d = np.zeros(c_len, dtype=float)
    if s == 0.0:
        return d

    L = int(p)
    if L > 0:
        d[center + 1:center + 1 + L] += s * (1.0 - h[p - 1::-1])
        d[center - L:center] += -s * h[0:L]

    R = n - 1 - int(p)
    if R > 0:
        d[center + 1:center + 1 + R] += -s * h[p + 1:]
        d[center - R:center] += s * (1.0 - h[p + 1:][::-1])

    return d


def pair_delta(h, i, j, amt):
    h = np.asarray(h, dtype=float)
    n = len(h)
    d = _single_delta(h, i, amt)
    if j != i:
        d += _single_delta(h, j, -amt)
        D = abs(int(i) - int(j))
        if D > 0:
            center = n - 1
            qa = amt * amt
            d[center - D] += qa
            d[center + D] += qa
    return d


def make_pair_linear(h, i, j):
    h = np.asarray(h, dtype=float)
    n = len(h)
    lin = _single_delta(h, i, 1.0)
    if j != i:
        lin += _single_delta(h, j, -1.0)
    center = n - 1
    D = abs(int(i) - int(j))
    if D == 0:
        return lin, -1, -1
    return lin, center - D, center + D

def balanced_binary(h, n):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = np.resize(h, n)
    h = np.clip(h, 0.0, 1.0)
    h = np.where(h >= 0.5, 1.0, 0.0)
    target = int(round(0.5 * float(n)))

    s = int(np.sum(h))
    if s > target:
        diff = s - target
        if diff > 0:
            ones = np.nonzero(h)[0]
            if ones.size >= diff:
                h[ones[-diff:]] = 0.0
    elif s < target:
        diff = target - s
        if diff > 0:
            zeros = np.nonzero(1.0 - h)[0]
            if zeros.size >= diff:
                h[zeros[:diff]] = 1.0
    return h


def _popcount(x):
    c = 0
    while x:
        x &= x - 1
        c += 1
    return c


def wiped_binary(n):
    i = np.arange(n, dtype=np.int64)
    score = (i * (i + 1) // 2) % n
    h = np.zeros(n, dtype=float)
    order = np.argsort(score, kind="mergesort")
    h[order[: n // 2]] = 1.0
    return balanced_binary(h, n)


def thue_binary(n):
    h = np.zeros(n, dtype=float)
    for k in range(n):
        if _popcount(k) & 1:
            h[k] = 1.0
    return balanced_binary(h, n)


def random_binary(n, rng):

def binary_sa_fast(n, rng, time_limit, h0=None):
    if n < 4 or (n % 2) != 0:
        h = project_box_sum(np.asarray(h0, dtype=float), n) if h0 is not None else np.full(n, 0.5)
        return h, compute_c5(h)

    if h0 is None:
        h = random_binary(n, rng)
    else:
        h = balanced_binary(h0, n)

    h = h.astype(float)
    c = compute_corr(h)
    cur = float(np.max(c) * (2.0 / float(n)))
    best = cur
    best_h = h.copy()

    t_start = 0.05
    t_end = 1e-6
    start = time.time()
    step = 0
    expected = max(float(time_limit), 1e-6)

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / expected)
        T = t_end + (t_start - t_end) * (1.0 - progress) * (1.0 - progress)

        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        bi = h[i]
        bj = h[j]
        if abs(bi - bj) < 1e-8:
            continue

        amt = 1.0 if bi < bj else -1.0
        d = pair_delta(h, i, j, amt)
        new_c = c + d
        new_cur = float(np.max(new_c) * (2.0 / float(n)))

        accept = False
        if new_cur < cur:
            accept = True
        elif new_cur > cur and T > 1e-12:
            accept = rng.random() < np.exp(-(new_cur - cur) / max(T, 1e-12))

        if accept:
            h[i] += amt
            h[j] -= amt
            c = new_c
            cur = new_cur
            if cur < best:
                best = cur
                best_h = h.copy()

        step += 1
        if step % 20000 == 0:
            c = compute_corr(h)
            cur = float(np.max(c) * (2.0 / float(n)))

    best_h = balanced_binary(best_h, n)
    best_val = compute_c5(best_h)
    return best_h, best_val


def continuous_starts(n, rng):

    return out


def run(seed=42, budget_s=1300, **kwargs):
    if isinstance(seed, (float, np.floating)):
        seed = int(seed)
    seed = int(seed) % (2**31 - 1)
    budget_s = float(budget_s)
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    h_parent = None
    try:
        h_parent = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h_parent = None

    if h_parent is None or h_parent.size == 0:
        n0 = 160
        h_parent = np.full(n0, 0.5, dtype=float)
    else:
        h_parent = np.ravel(h_parent)
        if h_parent.size < 4:
            h_parent = np.full(160, 0.5, dtype=float)

    n0 = int(h_parent.size)
    best_h = project_box_sum(np.clip(h_parent, 0.0, 1.0), n0)
    best_n = n0
    best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - start_time)
    if rem < 60 and not np.all(np.isfinite(best_h)):
        best_h = np.full(n0, 0.5)
        best_h = project_box_sum(best_h, n0)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - start_time)
    if rem > 80 and best_c5 > 0.47:
        for cand in continuous_starts(n0, rng):
            cand = project_box_sum(cand, n0)
            val = compute_c5(cand)
            if np.isfinite(val) and val < best_c5:
                best_h, best_n, best_c5 = cand, n0, val

    rem = budget_s - (time.time() - start_time)
    if rem > 150:
        if best_n <= 200:
            max_it = 2500
        elif best_n <= 400:
            max_it = 1200
        else:
            max_it = 500
        for beta in (10.0, 50.0, 200.0, 1000.0):
            rem = budget_s - (time.time() - start_time)
            if rem < 80:
                break
            h2, c2 = slsqp_smooth_solve(best_n, best_h, beta, max_it, 1e-11)
            if h2 is not None and np.isfinite(c2) and c2 < best_c5 - 1e-14:
                best_h, best_n, best_c5 = h2, best_n, c2

        rem = budget_s - (time.time() - start_time)
        if rem > 60:
            h2, c2 = lbfgs_smooth_solve(best_n, best_h, 500.0, 100.0, 400)
            if h2 is not None and np.isfinite(c2) and c2 < best_c5 - 1e-14:
                best_h, best_n, best_c5 = h2, best_n, c2

    rem = budget_s - (time.time() - start_time)
    if rem > 80:
        allow = min(90.0, rem - 25.0)
        if allow > 10.0:
            h2, c2 = fast_pair_polish(best_h, best_n, allow, rng)
            if np.isfinite(c2) and c2 < best_c5 - 1e-14:
                best_h, best_n, best_c5 = h2, best_n, c2

    rem = budget_s - (time.time() - start_time)
    if rem > 130:
        nb = best_n if (best_n % 2) == 0 else 160
        if nb < 64:
            nb = 160
        if nb % 2 != 0:
            nb += 1

        base_for_bin = best_h if best_n == nb else interpolate_to_n(best_h, nb)
        cands = binary_candidates(nb, rng, base_for_bin)

        start_h = cands[0]
        start_c = compute_c5(start_h)
        for cd in cands[1:]:
            cd = project_box_sum(cd, nb)
            cv = compute_c5(cd)
            if np.isfinite(cv) and cv < start_c:
                start_h, start_c = cd, cv

        allow = min(90.0, rem - 25.0)
        if allow > 15.0:
            h2, c2 = binary_sa_fast(nb, rng, allow, start_h)
            if np.isfinite(c2) and c2 < best_c5 - 1e-14:
                best_h, best_n, best_c5 = h2, nb, c2

        rem = budget_s - (time.time() - start_time)
        if best_n == nb and rem > 80:
            h2, c2 = slsqp_smooth_solve(nb, best_h, 500.0, 700 if nb <= 320 else 350, 1e-11)
            if h2 is not None and np.isfinite(c2) and c2 < best_c5 - 1e-14:
                best_h, best_n, best_c5 = h2, nb, c2

    rem = budget_s - (time.time() - start_time)
    high_base = best_n
    candidates = set()
    for common in (160, 256, 320, 480, 640):
        if common > high_base and high_base >= 64:
            candidates.add(common)
    for mult in (2, 3, 4):
        nn = int(high_base) * mult
        if nn % 2 != 0:
            nn += 1
        if nn > high_base and nn <= 640:
            candidates.add(nn)

    for nn in sorted(candidates):
        rem = budget_s - (time.time() - start_time)
        if rem < 130:
            break

        h_init = interpolate_to_n(best_h, nn)
        if nn <= 320:
            max_it = 900
        elif nn <= 480:
            max_it = 500
        else:
            max_it = 300

        h2, c2 = slsqp_smooth_solve(nn, h_init, 200.0, max_it, 1e-10)
        if h2 is not None and np.isfinite(c2) and c2 < best_c5 - 1e-14:
            best_h, best_n, best_c5 = h2, nn, c2

        rem = budget_s - (time.time() - start_time)
        if rem > 80:
            h2, c2 = lbfgs_smooth_solve(nn, best_h, 1000.0, 100.0, 300)
            if h2 is not None and np.isfinite(c2) and c2 < best_c5 - 1e-14:
                best_h, best_n, best_c5 = h2, nn, c2

        rem = budget_s - (time.time() - start_time)
        if rem > 100:
            allow = min(60.0, rem - 25.0)
            if allow > 10.0:
                h2, c2 = fast_pair_polish(best_h, nn, allow, rng)
                if np.isfinite(c2) and c2 < best_c5 - 1e-14:
                    best_h, best_n, best_c5 = h2, nn, c2

    rem = budget_s - (time.time() - start_time)
    if rem > 60:
        max_it = 600 if best_n <= 320 else 350
        for beta in (2000.0, 5000.0):
            rem = budget_s - (time.time() - start_time)
            if rem < 25:
                break
            h2, c2 = slsqp_smooth_solve(best_n, best_h, beta, max_it, 1e-12)
            if h2 is not None and np.isfinite(c2) and c2 < best_c5 - 1e-14:
                best_h, best_n, best_c5 = h2, best_n, c2

    rem = budget_s - (time.time() - start_time)
    if rem > 30:
        allow = min(60.0, rem - 10.0)
        if allow > 5.0:
            h2, c2 = fast_pair_polish(best_h, best_n, allow, rng)
            if np.isfinite(c2) and c2 < best_c5 - 1e-14:
                best_h, best_n, best_c5 = h2, best_n, c2

    rem = budget_s - (time.time() - start_time)
    if rem > 20:
        h2, c2 = lbfgs_smooth_solve(best_n, best_h, 2000.0, 200.0, 200)
        if h2 is not None and np.isfinite(c2) and c2 < best_c5 - 1e-14:
            best_h, best_n, best_c5 = h2, best_n, c2

    best_h = project_box_sum(best_h, best_n)
    best_c5 = compute_c5(best_h)
    if not np.isfinite(best_c5):
        best_h = project_box_sum(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h)

    return (best_h, float(best_c5), int(best_n))
