The gradient for coordinate j is:
grad[j] = sum_k p[k] * d(corr[k])/dh[j]

Where d(corr[k])/dh[j] = (1-h[j+lag_k])*1[j+lag_k valid] - h[j-lag_k]*1[j-lag_k valid]

Let me reverse the sum:
sum_k p[k] * (1-h[j+lag_k]) * 1[j+lag_k valid]
= sum_{i: i ∈ [0,n-1], i-j ∈ ells} p[i-j] * (1-h[i])

where ells = [-(n-1), ..., n-1]

= sum_{i=j-(n-1)}^{j+(n-1)} p[i-j] * (1-h[i]) * 1[i ∈ [0,n-1]]

This is a convolution of p with (1-h), evaluated at j, but p has a specific indexing.

Let me define:
q[m] = p[m + (n-1)] for m = -(n-1), ..., n-1  (so q is indexed from -(n-1) to n-1)

Then: sum_k p[k] * (1-h[j+k]) * 1[j+k valid] = sum_i (1-h[i]) * q[i-j] * 1[i valid]

This is the correlation of (1-h) with q, evaluated at position j.

Similarly: sum_k p[k] * h[j-k] * 1[j-k valid] = sum_i h[i] * q[j-i] * 1[i valid]

This is the cross-correlation of h with q (reversed).

So grad[j] = [(1-h) ⋆ q][j] - [h ⋆ q_rev][j]

where ⋆ is convolution and q_rev[m] = q[-m].

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_opt_j(h, j, n, corr, ells):
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-14

        if sg > 1e-13:

def lse_value_and_weights(corr, t):
    max_c = np.max(corr)
    exps = np.exp((corr - max_c) / t)
    total = np.sum(exps)
    lse_v = t * (np.log(total) + max_c / t)
    p = exps / total
    return lse_v, p


def lse_obj_grad(h, n, t, mu):
    corr = get_corr(h)
    lse_v, p = lse_value_and_weights(corr, t)
    ells = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - ells
        idx2 = j + ells
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        a = np.zeros(2 * n - 1)
        if np.any(valid1):
            a[valid1] += 1.0 - h[idx1[valid1]]
        if np.any(valid2):
            a[valid2] += -h[idx2[valid2]]

    pen = mu * (np.sum(h) - n / 2.0) ** 2
    grad_pen = 2.0 * mu * (np.sum(h) - n / 2.0)
    obj = lse_v + pen

def lse_optimize(h, n, t_start, t_end, n_iter, start_time, budget_frac):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t = t_start
    mu = 1.0
    deadline = start_time + budget_frac * (time.time() - start_time + 0)
    # We'll use the global start_time passed in
    actual_deadline = start_time + budget_frac
    for it in range(n_iter):
        if time.time() - start_time > actual_deadline - 5:
            break
        # Adaptive mu
        sum_diff = np.sum(h) - n / 2.0
        if abs(sum_diff) > 1e-6:
            mu = min(1000.0, mu * 1.5)
        else:
            mu = max(0.1, mu * 0.95)
        
        def obj_func(x):
            x_clip = np.clip(x, 0.0, 1.0)
            corr = get_corr(x_clip)
            lse_v, _ = lse_value_and_weights(corr, t)
            return lse_v + mu * (np.sum(x_clip) - n / 2.0) ** 2
        
        def grad_func(x):
            x_clip = np.clip(x, 0.0, 1.0)
            corr = get_corr(x_clip)
            _, p = lse_value_and_weights(corr, t)
            ells_local = np.arange(-(n - 1), n)
            g = np.zeros(n)
            for j in range(n):
                idx1 = j - ells_local
                idx2 = j + ells_local
                v1 = (idx1 >= 0) & (idx1 < n)
                v2 = (idx2 >= 0) & (idx2 < n)
                a = np.zeros(2 * n - 1)
                if np.any(v1):
                    a[v1] += 1.0 - x_clip[idx1[v1]]
                if np.any(v2):
                    a[v2] += -x_clip[idx2[v2]]
                g[j] = np.dot(p, a)
            g += 2.0 * mu * (np.sum(x_clip) - n / 2.0)
            # Zero gradient where at bound
            g[(x_clip <= 1e-12) | (x_clip >= 1 - 1e-12)] *= 0.01
            return g
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj_func, h, jac=grad_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 200, 'ftol': 1e-14,
                                                'gtol': 1e-10, 'maxls': 50})
        h_new = make_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        
        # Decrease temperature
        progress = (it + 1) / n_iter
        t = t_start * (t_end / t_start) ** progress
        if t < 0.002:
            t = 0.002
    
    # Final feasibility check
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def binary_local_search(h, n, max_trials, time_limit, rng):
    h_bin = (h > 0.5).astype(float)
    target = n / 2.0
    s = np.sum(h_bin)
    deficit = int(round(target - s))
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:deficit]
        h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:(-deficit)]
        h_bin[ones[order]] = 0.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    for _ in range(max_trials):
        if time.time() - start > time_limit:
            break
        n_flip = rng.randint(1, 5)
        idxs = rng.choice(n, n_flip, replace=False)
        h_try = best_h.copy()
        for idx in idxs:
            h_try[idx] = 1.0 - h_try[idx]
        if abs(np.sum(h_try) - target) < 1e-10:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5

def slsqp_polish(h, n, max_iter=2000):
    def c5_obj(x):
        x_c = np.clip(x, 0.0, 1.0)
        corr = get_corr(x_c)
        return np.max(corr)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(c5_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e8:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

    start_time = time.time()
    
    def remaining():
        return max(0, budget_s - (time.time() - start_time))
    
    # Load parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except NameError:
        h_parent = None
    except Exception:
        h_parent = None
    
    if h_parent is not None:
        best_h = make_feasible(h_parent.copy(), len(h_parent))
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = (np.arange(best_n) % 2).astype(float)
        best_h = make_feasible(best_h, best_n)
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Quick coordinate descent at parent resolution
    if remaining() > 80:
        tl = min(100, remaining() - 60)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=200, time_limit=tl,
                                     start_time=start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 2: Log-sum-exp optimization at parent resolution
    if remaining() > 120:
        tl = min(250, remaining() - 80)
        h_lse, c5_lse = lse_optimize(best_h, best_n, t_start=0.05, t_end=0.003,
                                      n_iter=15, start_time=start_time, budget_frac=tl)
        if c5_lse < best_c5:
            best_c5 = c5_lse
            best_h = h_lse.copy()
    
    # Phase 2b: CD after LSE
    if remaining() > 60:
        tl = min(100, remaining() - 40)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=100, time_limit=tl,
                                     start_time=start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 3: Resolution sweep
    resolutions = sorted(set([best_n, 128, 144, 192, 224, 240, 288, 320, 384, 480, 512, 640]))
    # Filter to even numbers only (needed for sum = n/2 to be integer)
    resolutions = [r for r in resolutions if r % 2 == 0]
    # Sort by proximity to best_n first, then by size
    resolutions.sort(key=lambda r: (abs(r - best_n), r))
    
    phase3_budget = min(500, remaining() - 100)
    if phase3_budget > 50:
        per_res = max(30, phase3_budget // len(resolutions))
        for n_res in resolutions:
            if remaining() < 60:
                break
            if n_res == best_n:
                h_init = best_h.copy()
            else:
                h_init = interpolate_to_n(best_h, n_res)
            
            tl_cd = min(per_res // 2, remaining() - 40)
            if tl_cd < 10:
                continue
            h_cd, c5_cd = coord_descent(h_init, n_res, n_passes=80, time_limit=tl_cd,
                                         start_time=start_time)
            
            if remaining() > 50:
                tl_lse = min(per_res // 2 - 10, remaining() - 50)
                if tl_lse > 20:
                    h_lse, c5_lse = lse_optimize(h_cd, n_res, t_start=0.03, t_end=0.003,
                                                  n_iter=8, start_time=start_time,
                                                  budget_frac=tl_lse)
                    if c5_lse < c5_cd:
                        c5_cd = c5_lse
                        h_cd = h_lse
            
            # Quick CD polish
            if remaining() > 30:
                h_cd2, c5_cd2 = coord_descent(h_cd, n_res, n_passes=30, time_limit=15,
                                               start_time=start_time)
                if c5_cd2 < c5_cd:
                    c5_cd = c5_cd2
                    h_cd = h_cd2
            
            if c5_cd < best_c5 - 1e-13:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_res
    
    # Phase 4: SLSQP polish at best resolution
    if remaining() > 80:
        h_slq, c5_slq = slsqp_polish(best_h, best_n, max_iter=3000)
        if h_slq is not None and c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()
    
    # Phase 4b: CD after SLSQP
    if remaining() > 40:
        tl = min(60, remaining() - 20)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=80, time_limit=tl,
                                     start_time=start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 5: Simulated annealing on binary
    if remaining() > 60:
        tl = min(80, remaining() - 30)
        h_sa, c5_sa = simulated_annealing(best_h, best_n, time_limit=tl,
                                           rng=rng, temp0=0.005, cooldown=0.999)
        if c5_sa < best_c5 - 1e-13:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            # Polish the binary solution
            if remaining() > 30:
                h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=50, time_limit=25,
                                             start_time=start_time)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
    
    # Phase 6: Binary local search with larger neighborhoods
    if remaining() > 40:
        tl = min(50, remaining() - 20)
        h_bl, c5_bl = binary_local_search(best_h, best_n, max_trials=10000,
                                           time_limit=tl, rng=rng)
        if c5_bl < best_c5 - 1e-13:
            best_c5 = c5_bl
            best_h = h_bl.copy()
    
    # Phase 7: Final LSE + CD polish
    if remaining() > 80:
        tl = min(100, remaining() - 40)
        h_lse, c5_lse = lse_optimize(best_h, best_n, t_start=0.02, t_end=0.002,
                                      n_iter=8, start_time=start_time, budget_frac=tl)
        if c5_lse < best_c5:
            best_c5 = c5_lse
            best_h = h_lse.copy()
    
    if remaining() > 30:
        tl = max(10, min(40, remaining() - 15))
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=100, time_limit=tl,
                                     start_time=start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 8: Final SLSQP if time permits
    if remaining() > 50:
        h_slq, c5_slq = slsqp_polish(best_h, best_n, max_iter=2000)
        if h_slq is not None and c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()
    
    # Phase 9: Final CD polish
    if remaining() > 15:
        tl = max(5, remaining() - 8)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=50, time_limit=tl,
                                     start_time=start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
