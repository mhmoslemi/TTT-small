def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # For efficiency, compute only the positive lags (and use symmetry if possible)
    # Actually np.correlate gives all lags, and the max might be at any lag
    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_obj(x):
    # Use log-sum-exp for smoothness
    n = _N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    beta = _BETA
    # Soft max
    shifted = corr - m
    soft_max = m + np.log(np.sum(np.exp(beta * shifted))) / beta
    # Add penalty for sum constraint
    s = np.sum(x) - _TGT
    return soft_max * dx + 1e4 * s * s

def project_feasible(h, n):

    target = n / 2.0

        if cnt > 0:

        else:
            h[:] += diff / n

    return h_best, best_c5


def coordinate_descent(h, n, max_sweeps=60):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if remaining_time() < 3:
            break
        improved = False
        for i in range(n):
            for val in [0.0, 1.0]:
                if abs(val - h_best[i]) < 1e-6:
                    continue
                h_trial = h_best.copy()
                h_trial[i] = val
                h_proj = project_feasible(h_trial, n)
                c5_t = compute_c5(h_proj)
                if c5_t < best_c5 - 1e-12:
                    h_best = h_proj
                    best_c5 = c5_t
                    improved = True
            if improved:
                improved = False
        if not improved:
            break
    return h_best, best_c5


def nudge_search(h, n, max_sweeps=30):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    for sweep in range(max_sweeps):
        if remaining_time() < 4:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n, 3):
                for d in [0.03, 0.07, 0.12, 0.02]:
                    ht = h_best.copy()
                    ht[i] += d
                    ht[j] -= d
                    ht = project_feasible(ht, n)
                    c5_t = compute_c5(ht)
                    if c5_t < best_c5 - 1e-12:
                        h_best = ht
                        best_c5 = c5_t
                        improved = True
                        break
                    ht = h_best.copy()
                    ht[i] -= d
                    ht[j] += d
                    ht = project_feasible(ht, n)
                    c5_t = compute_c5(ht)
                    if c5_t < best_c5 - 1e-12:
                        h_best = ht
                        best_c5 = c5_t
                        improved = True
                        break
                if improved:
                    break
            if improved and i % 8 == 0:
                improved = False
        if not improved:
            break
    return h_best, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _N, _BETA, _TGT, _START_TIME, _BUDGET_S
    _START_TIME = time.time()
    _BUDGET_S = budget_s
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        parent_c5 = compute_c5(parent_h)
        best_h = parent_h.copy()
        best_c5 = parent_c5
        best_n = parent_n
        print("Parent: n={}, C5={:.8f}".format(parent_n, parent_c5))
    except Exception:
        best_n = 160
        best_h = np.zeros(best_n)
        idx = rng.permutation(best_n)
        best_h[idx[:best_n // 2]] = 1.0
        best_c5 = compute_c5(best_h)
        print("No parent. Default: n={}, C5={:.8f}".format(best_n, best_c5))

    # Phase 1: Multi-n L-BFGS-B with smooth objective
    n_list = [best_n]
    for nn in [128, 144, 160, 176, 192, 224, 256, 320, 384]:
        if nn not in n_list and 64 <= nn <= 384 and nn % 2 == 0:
            n_list.append(nn)
    n_list.sort()

    for n in n_list:
        if remaining_time() < 60:
            break
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        print("\n--- Phase 1: n={}, t_rem={:.0f}s ---".format(n, remaining_time()))

        starts = []
        starts.append(project_feasible(interp_to(best_h, best_n, n), n))
        starts.append(project_feasible(make_thue_morse(n), n))
        starts.append(project_feasible(make_golden(n), n))
        starts.append(project_feasible(make_vdc(n), n))
        for _ in range(4):
            hb = np.zeros(n)
            idx = rng.permutation(n)
            hb[idx[:n // 2]] = 1.0
            starts.append(project_feasible(hb, n))

        t_arr = np.arange(n) / float(n)
        for freq in [2, 3, 5, 7]:
            ph = rng.rand() * 2.0 * np.pi
            hs = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * t_arr + ph)
            starts.append(project_feasible(np.clip(hs, 0.0, 1.0), n))

        for h0 in starts:
            if remaining_time() < 15:
                break
            h0 = project_feasible(h0, n)
            c5_cur = compute_c5(h0)

            for bv in [20, 80, 250, 700]:
                if remaining_time() < 8:
                    break
                _BETA = bv
                try:
                    res = minimize(smooth_obj, h0, method='L-BFGS-B',
                                   bounds=bounds,
                                   options={'maxiter': 250, 'ftol': 1e-14, 'gtol': 1e-9})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h0 = h_new
                        c5_cur = c5_new
                except Exception:
                    pass

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n
                print("  P1 improved: n={}, C5={:.8f}".format(n, best_c5))

    # Phase 2: SLSQP refinement on best
    if remaining_time() > 20:
        n = best_n
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
        h = project_feasible(best_h, n)

        print("\n--- Phase 2: SLSQP ---")
        for bv in [100, 300, 800, 2000]:
            if remaining_time() < 8:
                break
            _BETA = bv
            try:
                res = minimize(smooth_obj, h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    print("  SLSQP b={}: C5={:.8f}".format(bv, best_c5))
            except Exception:
                pass

    # Phase 3: Bit-swap local search
    if remaining_time() > 25:
        n = best_n
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        print("\n--- Phase 3: Bit swap n={}, start C5={:.8f} ---".format(n, best_c5))
        h_sw, c5_sw = bit_swap_search(best_h, n, max_sweeps=120)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()
            print("  Bit swap: C5={:.8f}".format(best_c5))

        # SLSQP after bit swap
        if remaining_time() > 15:
            n = best_n
            _N = n
            _TGT = n / 2.0
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
            h = project_feasible(best_h, n)
            for bv in [300, 1000]:
                if remaining_time() < 6:
                    break
                _BETA = bv
                try:
                    res = minimize(smooth_obj, h, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        h = h_new.copy()
                        print("  Post-swap SLSQP: C5={:.8f}".format(best_c5))
                except Exception:
                    pass

    # Phase 4: Coordinate descent + nudge
    if remaining_time() > 15:
        n = best_n
        h_cd, c5_cd = coordinate_descent(best_h, n, max_sweeps=80)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  Coord desc: C5={:.8f}".format(best_c5))

    if remaining_time() > 15:
        n = best_n
        h_ng, c5_ng = nudge_search(best_h, n, max_sweeps=40)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()
            print("  Nudge: C5={:.8f}".format(best_c5))

    # Phase 5: Second round of bit swap + SLSQP
    if remaining_time() > 20:
        n = best_n
        h_bin2 = (best_h > 0.5).astype(float)
        h_bin2 = project_feasible(h_bin2, n)
        c5_b2 = compute_c5(h_bin2)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_bin2.copy()

        h_sw2, c5_sw2 = bit_swap_search(best_h, n, max_sweeps=80)
        if c5_sw2 < best_c5:
            best_c5 = c5_sw2
            best_h = h_sw2.copy()
            print("  Round2 bit swap: C5={:.8f}".format(best_c5))

        if remaining_time() > 10:
            n = best_n
            _N = n
            _TGT = n / 2.0
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
            h = project_feasible(best_h, n)
            for bv in [500, 1500]:
                if remaining_time() < 5:
                    break
                _BETA = bv
                try:
                    res = minimize(smooth_obj, h, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        print("  Round2 SLSQP: C5={:.8f}".format(best_c5))
                except Exception:
                    pass

    # Phase 6: Targeted perturbation of critical positions
    if remaining_time() > 10:
        n = best_n
        h = project_feasible(best_h, n)
        dx_val = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = best_c5
        # Find positions involved in top correlations
        top_k = np.argsort(corr)[::-1][:5]
        for ki in top_k:
            off = int(ki) - (n - 1)
            if off >= 0:
                involved = set(list(range(off, min(n, off + n - off))))
            else:
                ao = -off
                involved = set(list(range(0, n - ao))) | set(list(range(ao, n)))
            for i in list(involved)[:20]:
                if remaining_time() < 3:
                    break
                for d in [0.02, 0.05, -0.02, -0.05, 0.1, -0.1]:
                    ht = h.copy()
                    ht[i] += d
                    ht = project_feasible(ht, n)
                    c5_t = compute_c5(ht)
                    if c5_t < c5_cur - 1e-12:
                        h = ht
                        c5_cur = c5_t
                        break
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
            print("  Targeted: C5={:.8f}".format(best_c5))

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
