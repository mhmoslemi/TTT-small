    n = len(h)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = np.sum(free)
        if nf > 0:
            h[free] += diff / nf
        else:
            if diff > 0:
                idx = np.argsort(1.0 - h)
            else:
                idx = np.argsort(h)
            for i in idx:
                if diff > 0:

                    h[i] += add
                    diff -= add
