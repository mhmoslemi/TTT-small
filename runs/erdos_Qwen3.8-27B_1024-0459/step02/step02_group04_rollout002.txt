import time
from scipy.optimize import minimize

SM_BETA = 1.0
AT_LAGS = []
AT_DX = 0.0


def get_parent_values():
    try:
        vals = initial_h_values
        if vals is None:

def adjust_sum_feasible(out, target, n):
    for _ in range(80):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            add = diff / float(cnt)
            max_add = float(np.min(1.0 - out[mask]))
            if add > max_add:
                add = max_add
            out[mask] += add
        else:
            mask = out > 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            add = diff / float(cnt)
            max_red = float(np.min(out[mask]))
            if -add > max_red:
                add = -max_red
            out[mask] += add
        np.clip(out, 0.0, 1.0, out=out)

    diff = target - float(np.sum(out))
    if abs(diff) > 1e-13:
        if diff > 0.0:
            idxs = np.flatnonzero(out < 1.0)
        else:
            idxs = np.flatnonzero(out > 0.0)
        rem = diff
        for ix in idxs:
            if rem > 0.0:
                a = min(rem, 1.0 - float(out[ix]))
                if a <= 0.0:
                    continue
                out[ix] += a
                rem -= a
            else:
                a = max(rem, -float(out[ix]))
                if a >= 0.0:
                    continue
                out[ix] += a
                rem -= a
            if abs(rem) < 1e-14:
                break
        np.clip(out, 0.0, 1.0, out=out)
    return out


def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)

    if h.size < n:
        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    elif h.size > n:
        h = h[:n]

    target = 0.5 * float(n)

    if np.all(h >= -1e-12) and np.all(h <= 1.0 + 1e-12):
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-7:
            return adjust_sum_feasible(h, target, n)

    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    return adjust_sum_feasible(out, target, n)


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n <= 0:
        return 0.0
    if n <= 2000:
        corr = np.correlate(h, 1.0 - h, mode="full")
        return float(np.max(corr) * (2.0 / float(n)))

    N = 1
    while N < 2 * n:
        N <<= 1
    F = np.fft.rfft(h, N)
    ac = np.fft.irfft(F * np.conjugate(F), N)[:n]
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    L = np.arange(n)
    vp = pref[n - L] - ac
    vn = total - pref[L] - ac
    return float(np.max(np.maximum(vp, vn)) * (2.0 / float(n)))


def all_lag_values_c(h):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n <= 0:
        return np.empty(0, dtype=float), np.empty(0, dtype=int)
    dx = 2.0 / float(n)
    full = np.correlate(h, 1.0 - h, mode="full")
    vals = np.empty(2 * n - 1, dtype=float)
    lags = np.empty(2 * n - 1, dtype=int)
    mid = n - 1
    vals[0] = full[mid] * dx
    lags[0] = 0
    idx = 1
    for L in range(1, n):
        vals[idx] = full[mid - L] * dx
        lags[idx] = L
        idx += 1
        vals[idx] = full[mid + L] * dx
        lags[idx] = -L
        idx += 1
    return vals, lags


def add_lag_gradient_to(g, h, l, coef):
    if coef == 0.0:
        return
    n = int(h.size)
    l = int(l)
    if l == 0:
        g += coef * (1.0 - 2.0 * h)
        return
    if l > 0:
        m = n - l
        g[:m] += coef * (1.0 - h[l:])
        g[l:] -= coef * h[:m]
    else:
        lp = -l
        m = n - lp
        g[lp:] += coef * (1.0 - h[:m])
        g[:m] -= coef * h[lp:]


def lag_gradient_c(h, l):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    g = np.zeros(n, dtype=float)
    add_lag_gradient_to(g, h, l, 2.0 / float(n))
    return g


def lag_value_c(h, l, dx):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    l = int(l)
    L = abs(l)
    if L == 0:
        return dx * (float(np.sum(h)) - float(np.dot(h, h)))
    m = n - L
    low = h[:m]
    high = h[L:]
    ac = float(np.dot(low, high))
    if l > 0:
        return dx * (float(np.sum(low)) - ac)
    return dx * (float(np.sum(high)) - ac)


def sum_eq(x):
    return float(np.sum(x) - 0.5 * float(len(x)))


def sum_eq_jac(x):
    return np.ones(len(x), dtype=float)


def smooth_all_obj(x):
    h = np.asarray(x, dtype=float)
    vals, _ = all_lag_values_c(h)
    beta = max(1.0, float(SM_BETA))
    m = float(np.max(vals))
    e = np.exp(np.clip(beta * (vals - m), -745.0, 0.0))
    s = float(np.sum(e))
    if s <= 0.0:
        return m
    return float(m + np.log(s) / beta)


def smooth_all_jac(x):
    h = np.asarray(x, dtype=float)
    n = int(h.size)
    vals, lags = all_lag_values_c(h)
    beta = max(1.0, float(SM_BETA))
    m = float(np.max(vals))
    e = np.exp(np.clip(beta * (vals - m), -745.0, 0.0))
    s = float(np.sum(e))
    if s <= 0.0:
        return lag_gradient_c(h, int(lags[int(np.argmax(vals))]))
    w = e / s
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    min_coef = 1e-15
    for i in range(len(lags)):
        coef = float(w[i]) * dx
        if coef <= min_coef:
            continue
        add_lag_gradient_to(g, h, int(lags[i]), coef)
    return g


def optimize_smooth_all(h0, beta, maxiter=40, ftol=1e-11):

    h0 = np.asarray(h0, dtype=float)
    n = int(h0.size)

        return h0.copy(), 0.0
    h = project_feasible(h0, n)
    SM_BETA = max(1.0, float(beta))

            c2 = compute_c5(h2)
            return h2, c2
    except Exception:
        pass
    return h, compute_c5(h)

def choose_active_lags(vals, lags, max_count, margin=1e-6):
    if vals.size == 0:
        return []
    order = np.argsort(-vals)
    take = min(int(max_count), vals.size)
    idx = order[:take]
    if margin > 0.0:
        top = float(vals[order[0]])
        near = np.where(vals >= top - margin)[0]
        if near.size > 0:
            idx = np.unique(np.concatenate((idx, near)))
            idx = idx[np.argsort(-vals[idx])]
    cap = max(64, 2 * int(max_count))
    if idx.size > cap:
        idx = idx[:cap]
    return [int(x) for x in lags[idx]]

def resample_to_n(h_src, n_dst, mode="step"):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = int(h_src.size)
    if n_dst <= 0:
        return np.empty(0, dtype=float)
    if n_src <= 0:
        return np.full(n_dst, 0.5, dtype=float)
    if n_src == n_dst:
        return h_src.copy()

    if mode == "linear":
        if n_src <= 1:
            return np.full(n_dst, float(h_src[0]), dtype=float)
        x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
        return np.interp(x_new, x_old, h_src)

    if n_dst >= n_src and n_dst % n_src == 0:
        return np.repeat(h_src, n_dst // n_src)
    idx = np.floor(np.arange(n_dst, dtype=float) * (float(n_src) / float(n_dst))).astype(int)
    idx = np.clip(idx, 0, n_src - 1)
    return h_src[idx]


def gaussian_smooth(h, sigma=1.2):
    h = np.asarray(h, dtype=float)
    n = int(h.size)
    if n <= 1:
        return h.copy()
    length = min(7, n)
    if length % 2 == 0:
        length -= 1
    length = max(1, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * float(sigma) * float(sigma)))
    w /= w.sum()
    return np.convolve(h, w, mode="same")


def make_variants(h, n, rng, count=4):

    if n > 1:
        flip = base[::-1]
        out.append(project_feasible(0.5 * (base + flip), n))
        out.append(project_feasible(0.5 + 0.5 * (base - flip), n))
    out.append(project_feasible(gaussian_smooth(base, 1.4), n))
    out.append(project_feasible(gaussian_smooth(base, 0.7), n))
    if n > 0:
        pert = base + rng.randn(n) * 0.025
        out.append(project_feasible(pert, n))
    if count is not None and len(out) > int(count):
        out = out[:int(count)]
    return out


def random_trig_candidate(n, rng, P=4):
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    x = np.arange(n, dtype=float)
    theta = 2.0 * np.pi * x / float(n)
    h = np.full(n, 0.5, dtype=float)
    P = int(max(1, P))
    scale = 0.45 / float(np.sqrt(P))
    for p in range(1, P + 1):
        a = rng.uniform(-1.0, 1.0)
        b = rng.uniform(-1.0, 1.0)
        h += scale * (a * np.cos(p * theta) + b * np.sin(p * theta))
    return h


def random_trig_search(n, rng, count, limit):
    n = int(n)
    best = None
    best_c = np.inf
    count = int(max(0, count))
    for _ in range(count):
        if time.time() > limit:
            break
        h = project_feasible(random_trig_candidate(n, rng, 4), n)
        c = compute_c5(h)
        if c < best_c:
            best = h
            best_c = c
    return best, float(best_c)


def local_block_perturb(h0, limit, seed, step=0.02):
    h0 = project_feasible(h0)
    n = int(h0.size)
    if n <= 0:
        return h0.copy(), 0.0
    rng = np.random.RandomState(int(seed))
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.0022
    min_temp = 1e-7
    max_trials = 2500
    max_len = min(6, n)
    for _ in range(max_trials):
        if time.time() > limit:
            break
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        delta = rng.randn(L) * float(step)
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[st:st + L] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try

        d = c_try - cur_c
        if d < 1e-15:
            cur_h = h_try
            cur_c = c_try
            temp = min(0.018, max(temp * 1.15, min_temp))
        elif d < temp:
            prob = float(np.exp(-d / temp))
            if rng.rand() < prob:
                cur_h = h_try
                cur_c = c_try
        temp = max(temp * 0.995, min_temp)
    return best_h, float(best_c)


def refine_composite(h0, n, time_limit, seed):
    h0 = project_feasible(h0, int(n))
    n = int(n)
    if n <= 0:
        return h0.copy(), 0.0
    rng = np.random.RandomState(int(seed))
    h = h0.copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    if n <= 120:
        betas = (8, 18, 45, 100, 200)
        maxit = 55
    elif n <= 200:
        betas = (6, 16, 40, 90, 180)
        maxit = 45
    else:
        betas = (5, 14, 36, 80)
        maxit = 32

    for beta in betas:
        if time.time() > time_limit - 35.0:
            break
        h2, c2 = optimize_smooth_all(h, beta, maxiter=maxit, ftol=1e-11)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = float(c2)
        if c2 <= best_c + 5e-4:
            h = h2
        else:
            h = best_h.copy()

    if n >= 2:
        if n <= 180:
            acnts = (16, 32, 48)
            passes = 2
        elif n <= 260:
            acnts = (16, 32)
            passes = 2
        else:
            acnts = (12, 24)
            passes = 1

        for acnt in acnts:
            for _ in range(passes):
                if time.time() > time_limit - 35.0:
                    break
                vals, lags = all_lag_values_c(h)
                cur = float(np.max(vals))
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
                active = choose_active_lags(vals, lags, acnt, 1e-6)
                if len(active) == 0:
                    active = [int(lags[int(np.argmax(vals))])]
                t0 = cur + 1e-7
                maxit = 75 if n <= 180 else (55 if n <= 260 else 35)
                h2, c2, ok = optimize_active_t(h, active, t0, maxiter=maxit, ftol=1e-12)
                if ok:
                    if c2 < best_c:
                        best_h = h2.copy()
                        best_c = float(c2)
                    if c2 <= cur + 1e-7:
                        h = h2
                    else:
                        h = best_h.copy()

    if time.time() < time_limit - 60.0:
        vs = make_variants(best_h, n, rng, 3)
        for hvar in vs[:2]:
            if time.time() > time_limit - 35.0:
                break
            cv = compute_c5(hvar)
            if np.isfinite(cv) and cv <= best_c + 8e-4:
                h2, c2 = optimize_smooth_all(hvar, 25, maxiter=30, ftol=1e-11)
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = float(c2)

    if time.time() < time_limit - 12.0 and n <= 320:
        h2, c2 = local_block_perturb(best_h, max(time.time() + 0.1, time_limit - 2.0), seed + 777, step=0.018)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = float(c2)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def to_binary(h):
    h = project_feasible(h)
    n = int(h.size)
    if n <= 0:
        return h.copy()
    target = int(n // 2)
    order = np.argsort(-h)
    b = np.zeros(n, dtype=float)
    b[order[:target]] = 1.0
    return b


def binary_hill_climb(b0, limit, seed, max_sweeps=30):
    n = int(b0.size)
    if n <= 0:
        return b0.copy(), 0.0
    rng = np.random.RandomState(int(seed))
    b = to_binary(b0)
    c = compute_c5(b)
    best_h = b.copy()
    best_c = float(c)
    cur_h = b.copy()
    cur_c = float(c)

    for _ in range(int(max_sweeps)):
        if time.time() > limit:
            break
        ones = np.flatnonzero(cur_h == 1.0)
        zeros = np.flatnonzero(cur_h == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        zord = rng.permutation(zeros)
        improved = False
        for i in ones:
            for j in zord:
                if time.time() > limit:
                    return best_h, float(best_c)
                cur_h[i] = 0.0
                cur_h[j] = 1.0
                cn = compute_c5(cur_h)
                if cn < cur_c - 1e-15:
                    cur_c = float(cn)
                    improved = True
                    if cn < best_c:
                        best_c = float(cn)
                        best_h = cur_h.copy()
                    break
                else:
                    cur_h[j] = 0.0
                    cur_h[i] = 1.0
            if improved:
                break
        if not improved:
            break
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 35.0)
    rng = np.random.RandomState(seed)

    parent = get_parent_values()
    if parent is not None and parent.size > 0:
        best_n = int(parent.size)
        best_h = project_feasible(parent, best_n)
        best_c = compute_c5(best_h)
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    candidates = []
    candidates.append([best_h.copy(), float(best_c), int(best_n)])

    base_src = parent if parent is not None else best_h
    base_n = int(best_n)
    ns = []
    for nn in (base_n, base_n - 16, base_n - 32, base_n + 16, base_n + 32, base_n + 48,
               base_n + 64, 128, 144, 160, 176, 192, 208, 224, 240, 256, 288, 320):
        nn = int(nn)
        if 48 <= nn <= 320 and nn not in ns:
            ns.append(nn)

    phaseA_end = min(hard_end - 80.0, start_time + 180.0)
    for n in ns:
        if time.time() > phaseA_end:
            break
        h_step = resample_to_n(base_src, n, "step")
        h_lin = resample_to_n(base_src, n, "linear")
        h, c = add_candidate(h_step, n, candidates)
        if c < best_c:
            best_h = h.copy()
            best_c = c
            best_n = n
        h, c = add_candidate(h_lin, n, candidates)
        if c < best_c:
            best_h = h.copy()
            best_c = c
            best_n = n

        if time.time() > phaseA_end:
            break
        if n == base_n:
            for hvar in make_variants(best_h, n, rng, 3)[:2]:
                if time.time() > phaseA_end:
                    break
                h, c = add_candidate(hvar, n, candidates)
                if c < best_c:
                    best_h = h.copy()
                    best_c = c
                    best_n = n

    if time.time() < hard_end - 120.0:
        nt = best_n if 64 <= best_n <= 240 else 160
        h, c = random_trig_search(int(nt), rng, 260, time.time() + 45.0)
        if h is not None:
            h, c = add_candidate(h, int(nt), candidates)
            if c < best_c:
                best_h = h.copy()
                best_c = c
                best_n = int(nt)

    candidates.sort(key=candidate_key)
    to_refine = []
    seen_ns = []

    if 2 <= int(best_n) <= 300:
        to_refine.append((best_h.copy(), int(best_n)))
        seen_ns.append(int(best_n))

    for item in candidates:
        if len(to_refine) >= 2:
            break
        h, c, n = item
        n = int(n)
        if n in seen_ns:
            continue
        if 2 <= n <= 300:
            to_refine.append((h.copy(), n))
            seen_ns.append(n)

    phaseB_end = min(hard_end - 90.0, time.time() + 460.0)
    if len(to_refine) > 0:
        for idx, (h0, n0) in enumerate(to_refine):
            remain = phaseB_end - time.time()
            if remain < 55.0:
                break
            slot = min(480.0, remain)
            if idx < len(to_refine) - 1:
                slot = min(slot, 0.7 * remain)
            slot_end = time.time() + slot
            h2, c2 = refine_composite(h0, int(n0), slot_end, seed + 5000 + 17 * int(n0))
            if c2 < best_c:
                best_h = h2.copy()
                best_c = float(c2)
                best_n = int(n0)
            candidates.append([h2.copy(), float(c2), int(n0)])
            candidates.sort(key=candidate_key)
            if len(candidates) > 5:
                del candidates[5:]

    if time.time() < hard_end - 70.0 and int(best_n) <= 260:
        rb = to_binary(best_h)
        b_lim = min(hard_end - 25.0, time.time() + 90.0)
        rh, rc = binary_hill_climb(rb, b_lim, seed + 90000)
        if rc < best_c:
            best_h = rh.copy()
            best_c = float(rc)
            best_n = int(rb.size)

    final_limit = hard_end - 3.0
    if time.time() < final_limit - 30.0 and 2 <= int(best_n) <= 320:
        for beta in (90, 220):
            if time.time() > final_limit - 25.0:
                break
            h2, c2 = optimize_smooth_all(best_h, beta, maxiter=35, ftol=1e-12)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = float(c2)

    if time.time() < final_limit - 25.0 and 2 <= int(best_n) <= 320:
        vals, lags = all_lag_values_c(best_h)
        active = choose_active_lags(vals, lags, 52, 1e-6)
        if len(active) > 0:
            h2, c2, ok = optimize_active_t(
                best_h,
                active,
                float(np.max(vals)) + 1e-7,
                maxiter=75,
                ftol=1e-13,
            )
            if ok and c2 < best_c:
                best_h = h2.copy()
                best_c = float(c2)

    if time.time() < final_limit - 8.0 and 2 <= int(best_n) <= 320:
        h2, c2 = local_block_perturb(best_h, final_limit - 1.0, seed + 99999, step=0.014)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = float(c2)

    best_h = project_feasible(best_h, int(best_n))
    best_c = compute_c5(best_h)
    if not np.isfinite(best_c):
        best_h = project_feasible(np.full(max(2, int(best_n)), 0.5, dtype=float), int(best_n))
        best_c = compute_c5(best_h)

    return (best_h, float(best_c), int(best_h.size))
