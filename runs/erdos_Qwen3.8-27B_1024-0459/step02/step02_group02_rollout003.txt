def build_from_bases(bases, amps, N):

    amps=np.asarray(amps,dtype=float)

    if not np.all(np.isfinite(amps)):

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize as scipy_minimize
except Exception:
    scipy_minimize = None


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    if n <= 0:
        return 1e100
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    if h.size != n:
        if h.size > n:
            h = h[:n]
        else:
            h = np.pad(h, (0, n - h.size), mode="constant")
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    for _ in range(1000):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / float(cnt)
        else:
            mask = h > 1e-12
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            h[mask] += diff / float(cnt)
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h_src, n_dst):
    n_src = len(h_src)
    n_dst = int(n_dst)
    if n_dst <= 0:
        return np.zeros(0, dtype=float)
    if n_src <= 1:
        return np.full(n_dst, 0.5, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, np.asarray(h_src, dtype=float))


def enforce_sum_tight(h):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    if n <= 0:
        return h
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    for _ in range(80):
        s = float(h.sum())
        diff = target - s
        if diff == 0.0 or abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmin(h))
            if h[idx] >= 1.0 - 1e-14:
                idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
            if h[idx] <= 1e-14:
                idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

    for _ in range(20):
        idx = n // 2
        others = float(h.sum()) - float(h[idx])
        val = target - others
        if 0.0 <= val <= 1.0:
            h[idx] = val
        else:
            mask = (h > 1e-12) & (h < 1.0 - 1e-12)
            cnt = int(np.count_nonzero(mask))
            if cnt == 0:
                break
            diff = target - float(h.sum())
            if abs(diff) < 1e-15:
                break
            h[mask] += diff / float(cnt)
            h = np.clip(h, 0.0, 1.0)
        if abs(float(h.sum()) - target) < 1e-14:
            break

    return h


def repair_symmetric(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    if n <= 0:
        return h
    h = 0.5 * (h + h[::-1])
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    for _ in range(40):
        s = float(h.sum())
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / float(n)
        h = np.clip(h, 0.0, 1.0)

    return h


def prepare_bases(N, modes):
    N = int(N)
    idx = np.arange(N, dtype=float)
    c = 0.5 * (N - 1)
    bases = []
    for m in modes:
        m = int(m)
        bases.append(np.cos((2.0 * math.pi * float(m) * (idx - c)) / float(N)))
    return bases


def build_from_bases(bases, amps, N):
    N = int(N)
    if N <= 0:
        return np.zeros(0, dtype=float)

    amps = np.asarray(amps, dtype=float).ravel()
    if not np.all(np.isfinite(amps)):
        return np.full(N, 0.5, dtype=float)

    if len(bases) == 0:
        return np.full(N, 0.5, dtype=float)

    phi = np.zeros(N, dtype=float)
    for b, a in zip(bases, amps):
        if a != 0.0:
            phi = phi + float(a) * b

    phi -= float(phi.mean())
    h = 0.5 + phi

    maxdev = float(np.max(np.abs(h - 0.5)))
    limit = 1.0 - 1e-10
    if maxdev > limit:
        phi *= limit / maxdev
        h = 0.5 + phi

    h = 0.5 * (h + h[::-1])
    h = np.clip(h, 0.0, 1.0)
    h = repair_symmetric(h, N)
    return h


def evaluate_candidate_bases(N, bases, amps):
    h = build_from_bases(bases, amps, N)
    h = repair_symmetric(h, len(h))
    c = compute_c5(h)
    return h, float(c)


def get_hand_candidates():
    a_single = math.sqrt(0.5)
    a1_0 = math.sqrt(0.486)
    a4_0 = math.sqrt(0.034)
    a1_1 = math.sqrt(0.49)
    a4_1 = math.sqrt(0.02)
    a8_1 = math.sqrt(0.01)

    return [
        ([1], [a_single]),
        ([1, 4], [a1_0, a4_0]),
        ([1, 4], [a1_0, -a4_0]),
        ([1, 8], [a1_0, a4_0]),
        ([1, 4, 8], [a1_1, a4_1, a8_1]),
        ([1, 4, 8], [a1_1, -a4_1, a8_1]),
    ]


def objective_bases(x, N, bases):
    x = np.asarray(x, dtype=float).ravel()
    if x.size == 0:
        return 0.5
    x = np.clip(x, -0.95, 0.95)
    h = build_from_bases(bases, x, N)
    c = compute_c5(h)
    if not np.isfinite(c):
        return 1e100
    return float(c)

                    h, c = evaluate_candidate_bases(N, bases, x_opt)
                    if np.isfinite(c) and c < best_c:
                        best_h = h
                        best_c = float(c)
                        best_n = int(len(h))
                except Exception:
                    pass

    if best_h is None:
        N = 2048 if budget_s > 120 else 256
        modes = [1]
        amps = [math.sqrt(0.5)]
        bases = prepare_bases(N, modes)
        best_h, best_c = evaluate_candidate_bases(N, bases, amps)
        best_n = int(len(best_h))

    best_h = np.asarray(best_h, dtype=float).copy()
    best_n = int(len(best_h))
    best_h = enforce_sum_tight(best_h)
    final_c = compute_c5(best_h)
    if np.isfinite(final_c):
        best_c = float(final_c)
    else:
        best_c = 1e100

    return (best_h, float(best_c), best_n)
