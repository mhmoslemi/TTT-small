def remaining_time():

def set_problem(n):
    global _N, _TGT, _DX
    _N = int(n)
    _TGT = 0.5 * _N
    _DX = 2.0 / float(_N)


def project_box_sum(x):
    x = np.asarray(x, dtype=float).copy()
    n = len(x)
    if n == 0:
        return x
    target = float(_TGT)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    x = np.clip(x, 0.0, 1.0)

    # Fast feasible projection, good for small steps.
    for _ in range(30):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-13:
            break
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if np.any(free):
            x[free] += diff / float(np.sum(free))
        else:
            if diff > 0.0:
                idxs = np.argsort(1.0 - x)
                rem = diff
                for i in idxs:
                    room = 1.0 - x[i]
                    if room <= 0.0:
                        continue
                    step = room if room < rem else rem
                    if step > 0.0:
                        x[i] += step
                        rem -= step
                    if rem <= 1e-15:
                        break
            else:
                idxs = np.argsort(x)
                rem = -diff
                for i in idxs:
                    room = x[i]
                    if room <= 0.0:
                        continue
                    step = room if room < rem else rem
                    if step > 0.0:
                        x[i] -= step
                        rem -= step
                    if rem <= 1e-15:
                        break
        x = np.clip(x, 0.0, 1.0)

    # Exact box+sum projection by bisection if necessary.
    diff = target - float(np.sum(x))
    if abs(diff) > 1e-9:
        lo = float(np.min(x)) - 1.0
        hi = float(np.max(x)) + 1.0
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
            if s > target:
                lo = mid
            else:
                hi = mid
        x = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

    diff = target - float(np.sum(x))
    if abs(diff) > 1e-14:
        free = np.where((x > 1e-8) & (x < 1.0 - 1e-8))[0]
        if free.size > 0:
            x[free[0]] += diff
        else:
            x[0] += diff
        x = np.clip(x, 0.0, 1.0)

    return x


def corr_full(h):
    n = int(len(h))
    if n <= 600:
        return np.correlate(h, 1.0 - h, mode='full')
    y = 1.0 - h
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    H = np.fft.rfft(h, N)
    Y = np.fft.rfft(y[::-1], N)
    c = np.fft.irfft(H * Y, N)
    return c[:m]


def max_corr(h):
    c = corr_full(h)
    return float(np.max(c))


def compute_c5(h, mv=None):
    if mv is None:
        mv = max_corr(h)
    return float(mv * _DX)


def select_top_indices(c, K):
    m = len(c)
    if K >= m:
        return np.arange(m, dtype=int)
    if K <= 0:
        return np.array([], dtype=int)
    return np.argpartition(c, -K)[-K:].astype(int)


def softmax_values(vals, beta):
    v = np.asarray(vals, dtype=float)
    if v.size == 0:
        return np.array([], dtype=float)
    m = float(np.max(v))
    e = np.exp(beta * (v - m))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(v.size, dtype=float)
        w[int(np.argmax(v))] = 1.0
        return w
    return e / s


def grad_lag_weighted(h, D, w):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if D is None or len(D) == 0:
        return g
    D = np.asarray(D, dtype=int)
    w = np.asarray(w, dtype=float)

    for di, wi in zip(D, w):
        if wi == 0.0:
            continue
        d = int(di)
        if d == 0:
            g += wi * (1.0 - 2.0 * h)
        elif d > 0:
            m = n - d
            if m <= 0:
                continue
            g[d:] += wi
            a = h[:m]
            b = h[d:d + m]
            g[:m] -= wi * b
            g[d:d + m] -= wi * a
        else:
            aoff = -d
            m = n - aoff
            if m <= 0:
                continue
            g[:m] += wi
            a = h[:m]
            b = h[aoff:aoff + m]
            g[:m] -= wi * b
            g[aoff:aoff + m] -= wi * a
    return g


def grad_all_weights(h, wfull):
    n = len(h)
    g = np.zeros(n, dtype=float)
    off = n - 1
    w0 = wfull[off]
    if w0 != 0.0:
        g += w0 * (1.0 - 2.0 * h)

    for d in range(1, n):
        wp = wfull[off + d]
        wm = wfull[off - d]
        if wp == 0.0 and wm == 0.0:
            continue
        if wp != 0.0:
            g[d:] += wp
        if wm != 0.0:
            g[:n - d] += wm
        wsum = wp + wm
        if wsum != 0.0:
            m = n - d
            a = h[:m]
            b = h[d:d + m]
            g[:m] -= wsum * b
            g[d:d + m] -= wsum * a
    return g


def smooth_compute(x):
    x = np.asarray(x, dtype=float)
    c = corr_full(x)
    maxc = float(np.max(c))
    if _BETA <= 1e-8:
        S = maxc
        w = np.zeros(len(c), dtype=float)
        w[int(np.argmax(c))] = 1.0
    else:
        z = _BETA * (c - maxc)
        exps = np.exp(z)
        s = float(np.sum(exps))
        S = maxc + float(np.log(s)) / _BETA
        w = exps / s
    g = grad_all_weights(x, w)
    return S, g


def smooth_fun(x):

        return 1e100
    S, g = smooth_compute(x)
    _CACHE_TYPE = 1
    _CACHE_X = x.copy()
    _CACHE_G = g
    return float(S)


def smooth_jac(x):

    _CACHE_X = x.copy()
    _CACHE_G = g

def slsqp_smooth(h0, beta, maxiter):

    h0 = project_box_sum(h0)

            return h0
        x = np.asarray(res.x, dtype=float)
        if not np.all(np.isfinite(x)):

        return project_box_sum(x)
    except Exception:
        return h0

                    if np.isfinite(C) and C < best_raw - 1e-9:
                        h = p.copy()
                        best_h = p.copy()
                        best_raw = C
                        current_raw = C
                        alpha = 0.08
                        pert_done = True
                        break
                if not pert_done:
                    break
            if stall > 80:
                break

    return best_h, float(best_raw * _DX)


def polish_stage(h, deadline, rng, full):
    h = project_box_sum(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    if not np.isfinite(best_c):
        return best_h, best_c

    K = choose_K(_N)
    t = time.time()
    if t < deadline:
        rem = max(0.0, float(deadline) - t)
        pg_deadline = float(deadline) if rem < 90.0 else t + max(15.0, 0.35 * rem)
        pg_deadline = min(pg_deadline, float(deadline))
        h2, c2 = pg_active(h, pg_deadline, K, 5000, rng)
        if np.isfinite(c2) and c2 < best_c - 1e-12:
            best_h = h2.copy()
            best_c = c2
        elif np.isfinite(c2) and c2 <= best_c + 1e-10:
            best_h = h2.copy()

    t = time.time()
    if t < deadline:
        mi = slsqp_maxiter(_N)
        if full and remaining_time() > 60.0 and time.time() < deadline - 25.0:
            h2 = slsqp_smooth(best_h, 10.0, mi)
            c2 = compute_c5(h2)
            if np.isfinite(c2) and c2 < best_c - 1e-12:
                best_h = h2.copy()
                best_c = c2
            elif np.isfinite(c2) and c2 <= best_c + 1e-10:
                best_h = h2.copy()

        if remaining_time() > 50.0 and time.time() < deadline - 25.0:
            h2 = slsqp_smooth(best_h, 30.0, int(0.85 * mi) + 1)
            c2 = compute_c5(h2)
            if np.isfinite(c2) and c2 < best_c - 1e-12:
                best_h = h2.copy()
                best_c = c2
            elif np.isfinite(c2) and c2 <= best_c + 1e-10:
                best_h = h2.copy()

        if full and remaining_time() > 45.0 and time.time() < deadline - 25.0:
            h2 = slsqp_smooth(best_h, 100.0, int(0.70 * mi) + 1)
            c2 = compute_c5(h2)
            if np.isfinite(c2) and c2 < best_c - 1e-12:
                best_h = h2.copy()
                best_c = c2
            elif np.isfinite(c2) and c2 <= best_c + 1e-10:
                best_h = h2.copy()

    t = time.time()
    if t < deadline:
        mi = slsqp_maxiter(_N)
        K1 = max(4, min(8, K))
        K2 = max(8, min(16, K))
        if remaining_time() > 45.0 and time.time() < deadline - 20.0:
            h2 = slsqp_active(best_h, K1, 3.0, int(0.6 * mi) + 1)
            c2 = compute_c5(h2)
            if np.isfinite(c2) and c2 < best_c - 1e-12:
                best_h = h2.copy()
                best_c = c2
            elif np.isfinite(c2) and c2 <= best_c + 1e-10:
                best_h = h2.copy()

        if remaining_time() > 40.0 and time.time() < deadline - 20.0:
            h2 = slsqp_active(best_h, K2, 2.0, int(0.6 * mi) + 1)
            c2 = compute_c5(h2)
            if np.isfinite(c2) and c2 < best_c - 1e-12:
                best_h = h2.copy()
                best_c = c2
            elif np.isfinite(c2) and c2 <= best_c + 1e-10:
                best_h = h2.copy()

    if full and remaining_time() > 70.0 and time.time() < deadline - 60.0:
        for scale in (0.02, 0.05):
            if remaining_time() < 45.0 or time.time() >= deadline - 30.0:
                break
            noise = rng.rand(len(best_h)) * 2.0 - 1.0
            hp = project_box_sum(best_h + 0.25 * scale * noise)
            pl_deadline = min(float(deadline), time.time() + min(45.0, remaining_time() - 15.0))
            h2, c2 = pg_active(hp, pl_deadline, K, 1200, rng)
            if np.isfinite(c2) and c2 < best_c - 1e-12:
                best_h = h2.copy()
                best_c = c2

    t = time.time()
    if t < deadline:
        rem = max(0.0, float(deadline) - t)
        final_pg_deadline = t + min(35.0, rem)
        h2, c2 = pg_active(best_h, final_pg_deadline, K, 2500, rng)
        if np.isfinite(c2) and c2 < best_c - 1e-12:
            best_h = h2.copy()
            best_c = c2
        elif np.isfinite(c2) and c2 <= best_c + 1e-10:
            best_h = h2.copy()

    return best_h, best_c


def interp_to(src_h, src_n, dst_n):
    src_h = np.asarray(src_h, dtype=float)
    src_n = int(src_n)
    dst_n = int(dst_n)
    if src_n == dst_n:
        return src_h.copy()
    if src_n <= 0:
        return np.zeros(dst_n, dtype=float)
    xp = (np.arange(src_n) + 0.5) * (2.0 / src_n)
    xq = (np.arange(dst_n) + 0.5) * (2.0 / dst_n)
    return np.interp(xq, xp, src_h)


def make_block(n):
    n = int(n)
    h = np.zeros(n, dtype=float)
    m = n // 2
    if m > 0:
        h[:m] = 1.0
    if n % 2 == 1:
        h[m] = 0.5
    return h


def make_golden(n):
    n = int(n)
    h = np.zeros(n, dtype=float)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        if abs(frac - 0.5) < 0.25:
            h[i] = 1.0
    return h


def make_thue_morse(n):
    n = int(n)
    h = np.zeros(n, dtype=float)
    for i in range(n):
        v = i
        cnt = 0
        while v:
            cnt ^= (v & 1)
            v >>= 1
        if cnt == 1:
            h[i] = 1.0
    return h


def random_perm_init(n, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    if n == 0:
        return h
    idx = rng.permutation(n)
    m = n // 2
    if m > 0:
        h[idx[:m]] = 1.0
    if n % 2 == 1:
        h[idx[m]] = 0.5
    return h


def round_to_half(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    out = np.zeros(n, dtype=float)
    if n == 0:
        return out
    order = np.argsort(h)[::-1]
    m = n // 2
    if m > 0:
        out[order[:m]] = 1.0
    if n % 2 == 1 and m < n:
        out[order[m]] = 0.5
    return out


def binary_hill_search(h, deadline, rng):
    h = round_to_half(h)
    best_h = h.copy()
    best_raw = max_corr(h)
    if not np.isfinite(best_raw):
        return best_h, compute_c5(best_h)
    curr_raw = best_raw
    stall = 0

    for _ in range(50000):
        if time.time() >= deadline or remaining_time() < 5.0:
            break
        ones = np.flatnonzero(h > 0.5)
        zeros = np.flatnonzero(h < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break
        i = int(ones[int(rng.randint(ones.size))])
        j = int(zeros[int(rng.randint(zeros.size))])
        h[i], h[j] = h[j], h[i]
        raw = max_corr(h)
        if np.isfinite(raw) and raw < curr_raw - 1e-9:
            curr_raw = raw
            if raw < best_raw - 1e-9:
                best_raw = raw
                best_h = h.copy()
            stall = 0
        else:
            h[i], h[j] = h[j], h[i]
            stall += 1
        if stall > 2500:
            break

    return best_h, float(best_raw * _DX)


def run(seed=42, budget_s=1300, **kwargs):
    global _START, _BUDGET, _N, _TGT, _DX

    _START = time.time()
    _BUDGET = max(60.0, float(budget_s))
    rng = np.random.RandomState(int(seed))

    # Initialize from parent if available; otherwise use robust fallbacks.
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.size == 0:
            raise ValueError("empty parent")
        n_parent = int(len(parent))
        parent = project_box_sum(parent)
    except Exception:
        n_parent = 160
        cands = [
            make_block(n_parent),
            make_golden(n_parent),
            make_thue_morse(n_parent),
            random_perm_init(n_parent, rng),
        ]
        parent = cands[0]
        pc = compute_c5(parent)
        for cand in cands[1:]:
            cc = compute_c5(cand)
            if cc < pc:
                parent = cand
                pc = cc

    set_problem(n_parent)
    best_h = parent.copy()
    best_c = compute_c5(best_h)
    best_n = n_parent

    global_deadline = _START + _BUDGET - 8.0

    # Main polish on the parent resolution.
    first_deadline = min(global_deadline - 25.0, _START + max(80.0, 0.30 * _BUDGET))
    if time.time() < first_deadline:
        h2, c2 = polish_stage(best_h, first_deadline, rng, True)
        if np.isfinite(c2) and c2 < best_c - 1e-12:
            best_h = h2.copy()
            best_c = c2
            best_n = int(_N)
        elif np.isfinite(c2) and c2 <= best_c + 1e-10:
            best_h = h2.copy()
            best_n = int(_N)

    # Try nearby and higher resolutions.
    cand_n = []
    for off in (16, 32, 48, 64, 96, 128):
        cand_n.append(best_n + off)
        cand_n.append(best_n - off)
    fixed = [176, 192, 200, 208, 224, 240, 256, 288, 320, 384, 448]
    for nn in fixed:
        cand_n.append(int(nn))

    n_list = []
    for nn in cand_n:
        nn = int(nn)
        if 96 <= nn <= 448 and nn != int(best_n) and nn not in n_list:
            n_list.append(nn)

    keyed = []
    for nn in n_list:
        keyed.append((abs(nn - int(best_n)), nn, nn))
    keyed.sort()
    n_list = []
    for item in keyed:
        n_list.append(int(item[1]))
    if len(n_list) > 12:
        n_list = n_list[:12]

    for nn in n_list:
        if time.time() >= global_deadline - 60.0:
            break
        rem0 = global_deadline - time.time()
        if nn <= 200:
            lim = 80.0
        elif nn <= 260:
            lim = 100.0
        elif nn <= 330:
            lim = 130.0
        else:
            lim = 150.0
        lim = min(lim, rem0 - 25.0)
        if lim < 55.0:
            continue

        set_problem(nn)
        h0 = interp_to(best_h, best_n, nn)
        h0 = project_box_sum(h0)
        deadline = min(global_deadline - 12.0, time.time() + lim)
        full = (time.time() < global_deadline - 350.0) and (nn <= 300)
        h2, c2 = polish_stage(h0, deadline, rng, full)
        if np.isfinite(c2) and c2 < best_c - 1e-12:
            best_h = h2.copy()
            best_c = c2
            best_n = int(_N)
        elif np.isfinite(c2) and c2 <= best_c + 1e-10:
            best_h = h2.copy()
            best_n = int(_N)

    # Restore best resolution and do final exploitation.
    set_problem(best_n)
    best_h = project_box_sum(np.asarray(best_h, dtype=float))
    best_c = compute_c5(best_h)

    if time.time() < global_deadline - 60.0:
        h2, c2 = polish_stage(best_h, min(global_deadline - 4.0, time.time() + min(260.0, remaining_time() - 15.0)), rng, True)
        if np.isfinite(c2) and c2 < best_c - 1e-12:
            best_h = h2.copy()
            best_c = c2
            best_n = int(_N)
        elif np.isfinite(c2) and c2 <= best_c + 1e-10:
            best_h = h2.copy()
            best_n = int(_N)

    # Try a rounded binary basin, then smooth it if useful.
    set_problem(best_n)
    best_h = project_box_sum(np.asarray(best_h, dtype=float))
    if time.time() < global_deadline - 45.0:
        hr = round_to_half(best_h)
        cr = compute_c5(hr)
        if np.isfinite(cr) and cr < best_c - 1e-12:
            best_h = hr.copy()
            best_c = cr
        elif np.isfinite(cr) and cr <= best_c + 1e-10:
            best_h = hr.copy()

    if time.time() < global_deadline - 25.0 and best_n <= 360:
        h2 = slsqp_smooth(best_h, 35.0, max(40, slsqp_maxiter(best_n) // 2))
        c2 = compute_c5(h2)
        if np.isfinite(c2) and c2 < best_c - 1e-12:
            best_h = h2.copy()
            best_c = c2
        elif np.isfinite(c2) and c2 <= best_c + 1e-10:
            best_h = h2.copy()

    if time.time() < global_deadline - 35.0 and best_n <= 240:
        hb, cb = binary_hill_search(
            best_h,
            min(global_deadline - 3.0, time.time() + min(60.0, remaining_time() - 10.0)),
            rng,
        )
        if np.isfinite(cb) and cb < best_c - 1e-12:
            best_h = hb.copy()
            best_c = cb

    set_problem(best_n)
    best_h = project_box_sum(np.asarray(best_h, dtype=float))
    final_c5 = compute_c5(best_h)

    if not np.isfinite(final_c5):
        best_h = project_box_sum(make_block(best_n))
        final_c5 = compute_c5(best_h)

    try:
        ev = evaluate_erdos_solution(best_h)
        evf = float(ev)
        if np.isfinite(evf) and evf < final_c5:
            final_c5 = evf
    except Exception:
        pass

    print("FINAL: n={}, C5={:.10f}".format(int(best_n), float(final_c5)))
    return (best_h, float(final_c5), int(best_n))
