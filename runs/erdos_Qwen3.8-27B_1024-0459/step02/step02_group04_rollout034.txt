        return 0.0

    high = h[L:]
    ac = float(np.dot(low, high))
    if l > 0:
        return float(np.sum(low) - ac)
    return float(np.sum(high) - ac)

def optimize_smooth_lbfgsb(h0, n, beta, penalty, maxiter=200):

        val += 0.5 * penalty_coeff * diff * diff
        grad = smooth_max_grad(h_local, beta)
        grad += penalty_coeff * diff * np.ones(n, dtype=float)
        return val, grad
    
    res = minimize(
        obj_and_grad, h, method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'disp': False}
    )

def choose_active_lags(vals, lags, max_count, margin=1e-5):

    idx = near[order[:min(max_count, len(near))]]
    return [int(lags[i]) for i in idx]

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, 0.5, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def gaussian_smooth(h, sigma=1.5):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        return h.copy()
    length = min(11, n)
    if length % 2 == 0:
        length -= 1
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * sigma * sigma))
    w /= w.sum()
    out = np.convolve(h, w, mode="same")
    return out

def block_perturb(h0, n, time_limit, seed, step=0.02, max_trials=5000):
    """Local search via random block perturbations."""
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.003
    min_temp = 1e-7
    
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        
        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, max(temp * 1.2, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0 < d < temp * 10:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.998, min_temp)
    
    return best_h, float(best_c)

def coordinate_descent(h0, n, time_limit, seed, iters=2000):
    """Coordinate descent with random coordinate selection."""
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    step = 0.01
    
    for _ in range(iters):
        if time.time() > time_limit:
            break
        # Pick 2-4 random coordinates
        k = rng.randint(2, 5)
        idx = rng.choice(n, size=min(k, n), replace=False)
        
        for signs in (1.0, -1.0):
            h_try = h.copy()
            for i in range(0, len(idx), 2):
                h_try[idx[i]] += signs * step * rng.uniform(0.5, 1.0)
                if i+1 < len(idx):
                    h_try[idx[i+1]] -= signs * step * rng.uniform(0.5, 1.0)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < c:
                h = h_try
                c = c_try
            else:
                break
        if c < best_c:
            best_c = c
            best_h = h.copy()
            step = min(0.03, step * 1.1)
        else:
            step = max(0.001, step * 0.95)
    
    return best_h, float(best_c)

def project_feasible(h, n=None):

        n = h.size

        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))

            break

def compute_c5(h):

def all_lag_values(h):

def lag_value(h, l):
    n = len(h)
    if l == 0:
        return float(np.sum(h) - np.dot(h, h))
    L = abs(int(l))
    m = n - L
    if m <= 0:
        return 0.0
    low = h[:m]
    high = h[L:]
    ac = float(np.dot(low, high))
    if l > 0:
        return float(np.sum(low) - ac)
    return float(np.sum(high) - ac)


def lag_gradient(h, l):
    n = len(h)
    if l == 0:
        return 1.0 - 2.0 * h
    L = abs(int(l))
    m = n - L
    if m <= 0:
        return np.zeros(n, dtype=float)
    low = h[:m]
    high = h[L:]
    g = np.zeros(n, dtype=float)
    if l > 0:
        g[:m] += 1.0 - high
        g[L:] += -low
    else:
        g[L:] += 1.0 - low
        g[:m] += -high
    return g

def smooth_max_grad(x, beta):
    h = np.asarray(x, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)

    total_w = float(wpos.sum() + wneg[1:].sum())

                g += cm * lag_gradient(h, -L)
    return g


def optimize_smooth_lbfgsb(h0, n, beta, penalty=10000.0, maxiter=150):
    h = project_feasible(h0, n).copy()
    n = int(n)

    def obj_and_grad(x):
        val = smooth_max_obj(x, beta)
        diff = float(np.sum(x)) - 0.5 * float(n)
        val += 0.5 * penalty * diff * diff
        grad = smooth_max_grad(x, beta)
        grad += penalty * diff
        return val, grad

    try:
        res = minimize(
            obj_and_grad, h, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'disp': False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    return h, compute_c5(h)

def choose_active_lags(vals, lags, max_count, margin=1e-5):
    if vals.size == 0:
        return []
    top = float(np.max(vals))
    near = np.where(vals >= top - margin)[0]
    if near.size == 0:
        near = np.array([int(np.argmax(vals))])
    order = np.argsort(-vals[near])
    idx = near[order[:min(max_count, len(near))]]
    return [int(lags[i]) for i in idx]

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, 0.5, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def gaussian_smooth(h, sigma=1.5):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        return h.copy()
    length = min(11, n)
    if length % 2 == 0:
        length -= 1
    if length < 3:
        length = 3
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * sigma * sigma))
    w /= w.sum()
    return np.convolve(h, w, mode="same")

def make_variants(h, n, rng, count=4):
    base = project_feasible(h, n)
    out = [base.copy()]
    if n > 1:
        flip = base[::-1]
        out.append(project_feasible(0.5 * (base + flip), n))
        out.append(project_feasible(np.clip(0.5 + 0.5 * (base - flip), 0, 1), n))
    out.append(project_feasible(gaussian_smooth(base, 2.0), n))
    out.append(project_feasible(gaussian_smooth(base, 0.5), n))
    pert = base + rng.randn(n) * 0.03
    out.append(project_feasible(pert, n))
    if count and len(out) > count:
        out = out[:count]
    return out


def block_perturb(h0, n, time_limit, seed, step=0.015, max_trials=8000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.003
    min_temp = 1e-8
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.012, max(temp * 1.15, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            diff = c_try - cur_c
            if 0.0 < diff < temp * 10:
                prob = np.exp(-diff / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.998, min_temp)
    return best_h, float(best_c)


def coordinate_descent(h0, n, time_limit, seed, iters=5000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    step = 0.008
    patience = 0
    for _ in range(iters):
        if time.time() > time_limit:
            break
        k = rng.randint(2, 5)
        idx = rng.choice(n, size=min(k, n), replace=False)
        improved = False
        for _s in range(2):
            h_try = h.copy()
            for i in range(0, len(idx), 2):
                h_try[idx[i]] += step * rng.uniform(0.5, 1.5)
                if i + 1 < len(idx):
                    h_try[idx[i + 1]] -= step * rng.uniform(0.5, 1.5)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < c - 1e-14:
                h = h_try
                c = c_try
                improved = True
        if improved:
            best_h = h.copy()
            best_c = c
            patience = 0
            step = min(0.025, step * 1.05)
        else:
            patience += 1
            step = max(0.001, step * 0.97)
    return best_h, float(best_c)

            seq = (32, 64, 128)
        else:
            seq = (24, 48, 96)
        h2, c2 = refine_active(best_h, best_n, phase3_end - 8,
                                active_seq=seq, maxiter=65)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 4: Local search polish
    if time.time() < hard_end - 15:
        h2, c2 = block_perturb(best_h, best_n, hard_end - 12,
                                seed + 55555, step=0.012, max_trials=10000)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < hard_end - 5:
        h2, c2 = coordinate_descent(best_h, best_n, hard_end - 2,
                                     seed + 66666, iters=6000)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Final: one more tiny active set pass if time permits
    if time.time() < hard_end - 3 and best_n <= 200:
        h2, c2 = refine_active(best_h, best_n, hard_end - 1,
                                active_seq=(48,), maxiter=40)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
