import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_opt_j(h, j, n, corr):
    e = np.arange(-(n - 1), n)
    idx1 = j - e
    idx2 = j + e
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)

    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]

    c0_base = corr[n - 1] - h[j] * (1.0 - h[j])
    b = corr - h[j] * a
    b[n - 1] = c0_base

    t_grid = np.linspace(0.0, 1.0, 80)
    best_val = np.inf
    best_t = 0.5
    for t in t_grid:
        v = a * t + b
        v[n - 1] = c0_base + t - t * t
        mx = np.max(v)
        if mx < best_val:
            best_val = mx
            best_t = t

    dt = 1.0 / 80.0
    lo = max(0.0, best_t - dt)
    hi = min(1.0, best_t + dt)
    t_fine = np.linspace(lo, hi, 80)
    for t in t_fine:
        v = a * t + b
        v[n - 1] = c0_base + t - t * t
        mx = np.max(v)
        if mx < best_val:
            best_val = mx
            best_t = t

    return best_t


def coord_descent(h, n, n_passes, time_limit, global_start):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for p in range(n_passes):
        if time.time() - global_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            h[j] = coord_opt_j(h, j, n, corr)
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def perturbed_coord_descent(h, n, time_limit, global_start, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for p in range(200):
        if time.time() - global_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            h[j] = coord_opt_j(h, j, n, corr)
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()

        if p % 8 == 7:
            n_pert = max(1, n // 40)
            idxs = rng.choice(n, n_pert, replace=False)
            h_try = best_h.copy()
            for idx in idxs:
                h_try[idx] = h_try[idx] + rng.uniform(-0.15, 0.15)

            c5_try = compute_c5(h_try)
            if c5_try < best_c5 * 1.02:
                h = h_try.copy()
            else:
                h = best_h.copy()

    return best_h, best_c5


def slsqp_polish(h, n, max_iter=3000):
    n_lags = 2 * n - 1

    def obj_tc(tc):
        t = tc[0]
        hv = tc[1:]
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        violations = np.maximum(corr - t, 0)
        return t + 1e6 * np.sum(violations ** 2)

    def obj_simple(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        return np.max(corr)

    bounds_h = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]

    try:
        res = minimize(obj_simple, h, method='SLSQP', bounds=bounds_h,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10 and res.success or (hasattr(res, 'x') and np.max(np.abs(res.x - 1)) < 2):
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def slsqp_active(h, n, n_active=8, max_iter=2000):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-n_active:]

    def obj_act(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        return np.max(c[top_idx])

    bounds_h = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]

    try:
        res = minimize(obj_act, h, method='SLSQP', bounds=bounds_h,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def binary_local_search(h, n, max_trials=6000, time_limit=30, global_start=0):
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = n / 2.0 - s
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:int(np.ceil(deficit))]
        if len(order) < int(np.ceil(deficit)):
            order = zeros
            h_bin[zeros[:int(np.ceil(deficit))]] = 1.0
        else:
            h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:int(np.ceil(-deficit))]
        h_bin[ones[order]] = 0.0

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    for _ in range(max_trials):
        if time.time() - global_start > time_limit:
            break
        n_flip = rng.randint(1, 5)
        idxs = rng.choice(n, n_flip, replace=False)
        h_try = best_h.copy()
        for idx in idxs:
            h_try[idx] = 1.0 - h_try[idx]
        if abs(np.sum(h_try) - n / 2.0) < 1e-10:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5


def structured_init(n, rng):
    inits = []
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(make_feasible(h1, n))
        h2 = np.tile([1.0, 0.0, 1.0, 0.0], n // 4)
        inits.append(make_feasible(h2, n))
    if n % 2 == 0:
        h3 = np.zeros(n)
        h3[:n // 4] = 1.0
        h3[n // 2:n // 2 + n // 4] = 1.0
        inits.append(make_feasible(h3, n))
    inits.append(make_feasible(rng.rand(n), n))
    h5 = np.zeros(n)
    for i in range(n):
        h5[i] = 1.0 if ((i * i + i) // (n // 2)) % 2 == 0 else 0.0
    inits.append(make_feasible(h5, n))
    return inits
