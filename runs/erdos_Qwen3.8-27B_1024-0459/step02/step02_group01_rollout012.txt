A better approach might be **iterated local search (ILS)**:

3. Perturb (random swaps)
4. Repeat local search
5. Keep track of best

Local search for binary sequences:
- Try all O(n²) swaps, pick the best improving one
- Repeat until no improvement
- This is O(n⁴) total per local search phase (n² swaps × O(n) evaluation × O(n) iterations)
- For n=192: 192⁴ ≈ 1.3B operations... too slow in pure Python

Speedup: only try a random subset of swaps per iteration:
- Try 100 random swaps, pick the best
- This is O(100 * n) per iteration, O(100 * n²) for the whole local search (assuming O(n) iterations)
- For n=192: 100 * 192² ≈ 3.7M operations. Fast!

This is a "first-improvement" or "sampled-best-improvement" local search.

```python
def local_search(b, time_budget, rng, n_samples=200):
    n = len(b)
    t0 = time.time()
    cur_max = max_corr_of(b)
    
    while time.time() - t0 < time_budget:
        improved = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        
        best_delta = 0
        best_i, best_j = -1, -1
        
        # Sample swaps
        n_try = min(n_samples, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            b[i], b[j] = b[j], b[i]
            new_max = max_corr_of(b)
            delta = new_max - cur_max
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
            b[i], b[j] = b[j], b[i]  # Revert
        
        if best_delta < 0:
            b[best_i], b[best_j] = b[best_j], b[best_i]
            cur_max += best_delta
            improved = True
        
        if not improved:
            break
    
    return b, cur_max
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def max_corr_of(b):
    return np.max(np.correlate(b, 1.0 - b, mode='full'))

def to_binary(h, n):
    h = np.array(h, dtype=float)
    if len(h) != n:
        xp = (np.arange(len(h)) + 0.5) * 2.0 / len(h)
        xn = (np.arange(n) + 0.5) * 2.0 / n
        h = np.interp(xn, xp, h)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def fix_binary(b, n):
    b = np.array(b, dtype=float).copy()
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones > target:
        ones = np.where(b == 1)[0]
        remove = rng_global.choice(ones, n_ones - target, replace=False)
        b[remove] = 0.0
    elif n_ones < target:
        zeros = np.where(b == 0)[0]
        add = rng_global.choice(zeros, target - n_ones, replace=False)
        b[add] = 1.0
    return b

rng_global = np.random.RandomState(42)

def qr_start(n, rng):
    for p in range(n, min(n + 100, 500)):
        if p < 2:
            continue
        is_prime = True
        for d in range(2, int(np.sqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if not is_prime:
            continue
        k = (p - 1) // 2
        if k != n // 2:
            continue
        qrs = sorted(set([(x * x) % p for x in range(1, p)]))
        b = np.zeros(n)
        count = 0
        for q in qrs:
            pos = q - 1
            if 0 <= pos < n and b[pos] == 0:
                b[pos] = 1.0
                count += 1
            if count == n // 2:
                break
        if count < n // 2:
            remaining = np.where(b == 0)[0]
            extra = rng.choice(remaining, n // 2 - count, replace=False)
            b[extra] = 1.0
        return b
    return None

def local_search(b, time_budget, rng, n_samples=300):
    n = len(b)
    t0 = time.time()
    cur_max = max_corr_of(b)
    stale = 0

    while time.time() - t0 < time_budget:
        improved = False
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        best_delta = 0
        best_i, best_j = -1, -1

        n_try = min(n_samples, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            b[i], b[j] = b[j], b[i]
            new_max = max_corr_of(b)
            delta = new_max - cur_max
            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
            b[i], b[j] = b[j], b[i]

        if best_delta < 0:
            b[best_i], b[best_j] = b[best_j], b[best_i]
            cur_max += best_delta
            improved = True
            stale = 0
        else:
            stale += 1
            if stale > 5:
                break
            n_samples = max(100, n_samples // 2)

    return b, cur_max

def ils_optimize(b0, time_budget, rng):
    n = len(b0)
    best_b = b0.copy()
    best_max = max_corr_of(b0)
    t0 = time.time()
    cur = b0.copy()

    iter_count = 0
    while time.time() - t0 < time_budget:
        time_left = time_budget - (time.time() - t0)
        if time_left < 10:
            break

        ls_time = min(30.0, max(5.0, time_left / 5.0))
        cur, cur_max = local_search(cur, ls_time, rng, n_samples=400)

        if cur_max < best_max:
            best_max = cur_max
            best_b = cur.copy()

        n_kicks = rng.randint(max(3, n // 12), max(5, n // 4))
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        for _ in range(n_kicks):
            if len(ones) == 0 or len(zeros) == 0:
                break
            oi = rng.randint(len(ones))
            oj = rng.randint(len(zeros))
            i, j = ones[oi], zeros[oj]
            cur[i], cur[j] = cur[j], cur[i]
            ones[oi] = j
            zeros[oj] = i

        iter_count += 1

    return best_b, best_max

def sa_optimize(b0, time_budget, rng):
    n = len(b0)
    b = b0.copy()
    t0 = time.time()
    cur_max = max_corr_of(b)
    best_b = b.copy()
    best_max = cur_max

    T0 = 4.0
    T_floor = 0.05

    n_iters_est = 0
    t_est = time.time()
    for _ in range(min(500, time_budget * 50)):
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        b[ones[i]], b[zeros[j]] = b[zeros[j]], b[ones[i]]
        _ = max_corr_of(b)
        b[ones[i]], b[zeros[j]] = b[zeros[j]], b[ones[i]]
    elapsed = time.time() - t_est
    if elapsed > 0:
        n_iters_est = int((time_budget / elapsed) * min(500, time_budget * 50))
    n_iters_est = max(n_iters_est, 1000)

    alpha = (T_floor / T0) ** (1.0 / n_iters_est)

    for it in range(n_iters_est):
        if time.time() - t0 > time_budget:
            break
        T = max(T0 * (alpha ** it), T_floor)

        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = b[j], b[i]

        new_max = max_corr_of(b)
        delta = new_max - cur_max

        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]

    return best_b, best_max

def _smooth_obj(x, n, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    sh = beta * (c - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / beta
    return smax * (2.0 / n)

def _eq_con(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_polish(h0, time_budget, rng):
    n = len(h0)
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    stages = [
        (50.0, 200), (200.0, 300), (1000.0, 400), (5000.0, 500), (20000.0, 600)
    ]
    for beta, maxit in stages:
        if time.time() - t0 > time_budget - 5:
            break
        try:
            def obj(x):
                return _smooth_obj(x, n, beta)
            def eq(x):
                return np.sum(x) - n / 2.0
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': eq}],
                           options={'maxiter': maxit, 'ftol': 1e-15, 'iprint': 0})
            h = project_feasible(res.x, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        except Exception:
            pass

    # Raw max refinement
    if time.time() - t0 < time_budget - 5:
        try:
            def obj_raw(x):
                c = np.correlate(x, 1.0 - x, mode='full')
                return np.max(c) * (2.0 / n)
            def eq2(x):
                return np.sum(x) - n / 2.0
            res = minimize(obj_raw, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': eq2}],
                           options={'maxiter': 500, 'ftol': 1e-15, 'iprint': 0})
            h2 = project_feasible(res.x, n)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()
        except Exception:
            pass

    return best_h, best_c5

def generate_starts(n, parent_h, rng):
    starts = []

    # Parent rounded
    if parent_h is not None:
        starts.append(to_binary(parent_h, n))

    # Random balanced
    b = np.zeros(n)
    b[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(b)

    # Another random
    b2 = np.zeros(n)
    b2[rng.choice(n, n // 2, replace=False)] = 1.0
    starts.append(b2)

    # QR construction
    qr = qr_start(n, rng)
    if qr is not None:
        starts.append(qr)

    # Interleaved blocks
    b3 = np.zeros(n)
    block_size = max(2, n // 16)
    pattern = []
    while len(pattern) < n:
        pattern.extend([1] * (block_size // 2 + block_size % 2))
        pattern.extend([0] * (block_size // 2))
    b3[:n] = pattern[:n]
    b3 = b3.astype(float)
    n_ones = int(np.sum(b3))
    if n_ones > n // 2:
        ones_idx = np.where(b3 == 1)[0]
        for idx in rng.choice(ones_idx, n_ones - n // 2, replace=False):
            b3[idx] = 0.0
    elif n_ones < n // 2:
        zeros_idx = np.where(b3 == 0)[0]
        for idx in rng.choice(zeros_idx, n // 2 - n_ones, replace=False):
            b3[idx] = 1.0
    starts.append(b3)

    # Golden ratio spacing
    phi = (1 + np.sqrt(5)) / 2
    b4 = np.zeros(n)
    for k in range(n // 2):
        pos = int((k * phi) % n)
        b4[pos] = 1.0
    n_ones4 = int(np.sum(b4))
    if n_ones4 > n // 2:
        ones_idx = np.where(b4 == 1)[0]
        for idx in rng.choice(ones_idx, n_ones4 - n // 2, replace=False):
            b4[idx] = 0.0
    elif n_ones4 < n // 2:
        zeros_idx = np.where(b4 == 0)[0]
        for idx in rng.choice(zeros_idx, n // 2 - n_ones4, replace=False):
            b4[idx] = 1.0
    starts.append(b4)

    return starts

def run(seed=42, budget_s=1300, **kwargs):
    global rng_global
    t0 = time.time()
    rng = np.random.RandomState(seed)
    rng_global = np.random.RandomState(seed + 1)

    def time_left():
        return max(0.0, budget_s - (time.time() - t0) - 3.0)

    # Get parent
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if parent_h is not None and len(parent_h) > 0:
        best_h = parent_h.copy()
        best_c5 = compute_c5(parent_h)
        best_n = len(parent_h)

    # Determine resolutions to try
    n_candidates = [96, 128, 160, 192, 224, 256]
    if best_n in n_candidates:
        n_candidates.remove(best_n)
    n_candidates.sort()

    # Allocate time: give more time to best_n and nearby
    total_opt_time = budget_s * 0.7
    # Weight resolutions by proximity to parent
    weights = []
    for nn in n_candidates:
        if best_n > 0:
            w = 1.0 / (1.0 + abs(nn - best_n) * 0.01)
        else:
            w = 1.0
        weights.append(w)
    total_w = sum(weights)
    time_alloc = [total_opt_time * w / total_w for w in weights]

    for idx, n in enumerate(n_candidates):
        if time_left() < 30:
            break

        starts = generate_starts(n, parent_h, rng)

        # Deduplicate
        unique_starts = []
        for s in starts:
            is_dup = False
            for u in unique_starts:
                if np.array_equal(s, u):
                    is_dup = True
                    break
            if not is_dup:
                unique_starts.append(s)

        # Evaluate all starts
        scored = []
        for s in unique_starts:
            mc = max_corr_of(s)
            c5 = mc * (2.0 / n)
            scored.append((c5, s))
        scored.sort(key=lambda x: x[0])

        alloc_time = min(time_alloc[idx], time_left() - 10)
        if alloc_time < 15:
            alloc_time = time_left() - 5

        # Try top starts with different methods
        n_starts_to_try = min(3, len(scored))
        time_per_start = alloc_time / n_starts_to_try

        for si in range(n_starts_to_try):
            if time_left() < 10:
                break
            c5_init, b_init = scored[si]
            if c5_init < best_c5:
                best_c5 = c5_init
                best_h = b_init.copy()
                best_n = n

            method_time = min(time_per_start, time_left() - 5)
            if method_time < 5:
                break

            # Choose method: ILS for first, SA for second
            if si == 0:
                b_res, mc = ils_optimize(b_init, method_time, rng)
            elif si == 1:
                b_res, mc = sa_optimize(b_init, method_time, rng)
            else:
                # Mix: half ILS, half SA
                half = method_time / 2
                b_res, mc = ils_optimize(b_init, half, rng)
                if time_left() > half:
                    b_res2, mc2 = sa_optimize(b_res, min(half, time_left() - 5), rng)
                    if mc2 < mc:
                        b_res, mc = b_res2, mc2

            c5_res = mc * (2.0 / n)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = b_res.copy()
                best_n = n

    # SLSQP polish on best binary solution
    if time_left() > 15 and best_h is not None:
        h_cont = best_h.astype(float).copy()
        alloc_slsqp = min(time_left() - 5, budget_s * 0.2)
        h_pol, c5_pol = slsqp_polish(h_cont, alloc_slsqp, rng)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Also try SLSQP from a slightly perturbed version
    if time_left() > 15 and best_h is not None and np.all((best_h == 0) | (best_h == 1)):
        h_pert = best_h.astype(float) + 0.05 * rng.randn(len(best_h))
        h_pert = project_feasible(h_pert, best_n)
        alloc2 = min(time_left() - 5, budget_s * 0.1)
        h_pol2, c5_pol2 = slsqp_polish(h_pert, alloc2, rng)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    # Final: try rounding the continuous best to binary
    if best_h is not None:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_want = len(h_bin) // 2
        if n_ones > n_want:
            ones_idx = np.where(h_bin == 1)[0]
            order = np.argsort(best_h[ones_idx])[:n_ones - n_want]
            h_bin[ones_idx[order]] = 0.0
        elif n_ones < n_want:
            zeros_idx = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros_idx])[:n_want - n_ones]
            h_bin[zeros_idx[order]] = 1.0
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = h_bin.copy()

    # Ensure feasibility
    best_h = project_feasible(np.array(best_h, dtype=float), len(best_h))
    final_c5 = compute_c5(best_h)

    # Use evaluator if available
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, len(best_h))
```
