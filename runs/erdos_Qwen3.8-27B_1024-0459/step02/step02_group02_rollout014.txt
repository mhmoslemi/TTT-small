    h = np.array(h, dtype=float).copy()

            m = h < 1.0 - 1e-14

            h[m] += diff / c
        else:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14: break
        if diff > 0:
            m = h < 1.0 - 1e-14
            c = np.count_nonzero(m)
            if c == 0: break
            h[m] += diff / c
        else:
            m = h > 1e-14
            c = np.count_nonzero(m)
            if c == 0: break
            h[m] += diff / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def make_binary(h, n, rng):
    h = np.round(np.asarray(h, dtype=float)).astype(float)
    o = int(h.sum())
    t = n // 2
    if o > t:
        p = np.where(h == 1.0)[0]
        h[rng.choice(p, o-t, replace=False)] = 0.0
    elif o < t:
        p = np.where(h == 0.0)[0]
        h[rng.choice(p, t-o, replace=False)] = 1.0
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

    bc = compute_c5(h)
    ones = list(np.where(h==1.0)[0])
    zeros = list(np.where(h==0.0)[0])
    ni = 0
