Actually, let me think about this more carefully. If I fix all h except h[j], then:
c[k] = c0[k] + delta_c[k, j] * h[j]

where c0[k] is the correlation without the h[j] terms, and delta_c[k,j] is the coefficient.

For lag k ≥ 0: h[j] contributes (1-h[j-k]) to c[k] if j ≥ k, and -h[j+k] to c[k] if j ≤ n-1-k.
So delta_c[k,j] = (1-h[j-k]) for k ≤ j ≤ n-1, and -h[j+k] for 0 ≤ j ≤ n-1-k.

But wait, these change as h[j] changes (since they depend on other h values which are fixed, but the terms (1-h[j-k]) and h[j+k] don't depend on h[j] itself). So yes, c[k] is linear in h[j] when all other variables are fixed.

So the problem for h[j] alone is:
minimize max_k (A[k] + B[k]*h[j])
subject to 0 ≤ h[j] ≤ 1

where A[k] and B[k] are fixed (depend on other h values).

The optimal h[j] is where the max is minimized, which is a 1D problem solvable by bisection or analytical solution.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmax(h < 1.0 - 1e-6)) if diff > 0 else int(np.argmax(h > 1e-6))
        if 0 <= idx < n:
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def grad_c5(h):
    """Compute gradient of max_k corr[k] * dx analytically (subgradient)."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    kmax = int(np.argmax(corr))
    lag = kmax - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # c[lag] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
        # d/dh[j]: (1-h[j-lag]) for j in [lag, n-1]
        #          -h[j+lag]  for j in [0, n-1-lag]
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        # c[-l] = sum_{i=0}^{n-1-l} h[i]*(1-h[i+l])
        # d/dh[j]: (1-h[j+l]) for j in [0, n-1-l]
        #          -h[j-l]    for j in [l, n-1]
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] -= h[:n - l]
    return grad * dx, corr

def smooth_max_obj(h, beta):
    """Smooth max using log-sum-exp."""
    corr = compute_corr(h)
    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    return cmax + (1.0 / beta) * np.log(np.sum(exps))

def smooth_max_grad(h, beta):
    """Gradient of smooth max."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-10:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        else:
            l = -lag
            grad[:n - l] += w * (1.0 - h[l:])
            grad[l:] -= w * h[:n - l]
    return ops_smooth_max_obj(h, beta) * 0 + grad * dx, grad * dx

def ops_smooth_max_obj(h, beta):
    corr = compute_corr(h)
    cmax = np.max(corr)
    exps = np.exp(beta * (corr - cmax))
    return cmax + (1.0 / beta) * np.log(np.sum(exps))

def slsqp_with_grad(n, h_init, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return np.max(compute_corr(x))
    
    def obj_grad(x):
        g, _ = grad_c5(x)
        return g
    
    try:
        res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def coordinate_descent(h, n, n_sweeps=5, rng=None):
    """1D coordinate descent on each variable."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for j in order:
            h_orig = h[j]
            # Try several candidate values for h[j]
            t_candidates = np.array([0.0, 1.0, 0.25, 0.5, 0.75,
                                     h_orig * 0.5, h_orig * 1.5,
                                     h_orig - 0.1, h_orig + 0.1])
            t_candidates = np.clip(t_candidates, 0.0, 1.0)
            t_candidates = np.unique(np.round(t_candidates, 6))
            
            best_t = h_orig
            best_val = best_c5
            for t in t_candidates:
                if abs(t - h_orig) < 1e-8:
                    continue
                old = h[j]
                h[j] = t
                h = make_feasible(h, n)
                val = compute_c5(h)
                if val < best_val - 1e-14:
                    best_val = val
                    best_t = t
                h[j] = old
            if best_t != h_orig:
                h[j] = best_t
                h = make_feasible(h, n)
                best_c5 = best_val
    return h, best_c5

def pair_local_search(h, n, n_rounds=30, rng=None):
    """Local search: swap values between pairs, preserve sum."""
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for rd in range(n_rounds):
        improved = False
        # Try all adjacent pairs and some random pairs
        pairs = [(i, i+1) for i in range(n-1)]
        rng.shuffle(pairs)
        for step in [0.01, 0.03, 0.05, 0.1, 0.15, 0.2]:
            for (i, j) in pairs[:200]:
                if i >= n or j >= n:
                    continue
                for direction in [1, -1]:
                    amt = direction * step
                    h_i_new = h[i] + amt
                    h_j_new = h[j] - amt
                    if h_i_new < -1e-6 or h_i_new > 1 + 1e-6:
                        continue
                    if h_j_new < -1e-6 or h_j_new > 1 + 1e-6:
                        continue
                    h[i] = np.clip(h_i_new, 0, 1)
                    h[j] = np.clip(h_j_new, 0, 1)
                    # Recompute to handle clipping
                    diff = (h_i_new - h[i]) - (h_j_new - h[j])
                    if abs(diff) > 1e-10:
                        # Redistribute
                        mask = np.ones(n, dtype=bool)
                        mask[i] = False
                        mask[j] = False
                        avail = mask & ((h > 1e-6) | (h < 1 - 1e-6))
                        cnt = np.sum(avail)
                        if cnt > 0:
                            h[avail] += -diff / cnt
                            h = np.clip(h, 0, 1)
                    c5_try = compute_c5(h)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        improved = True
                    else:
                        h[i] = h_i_new - amt  # restore
                        h[j] = h_j_new + amt
                        h[i] = np.clip(h[i], 0, 1)
                        h[j] = np.clip(h[j], 0, 1)
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    h = make_feasible(h, n)
    return h, compute_c5(h)

def structured_binary_init(n, rng):
    """Create structured binary initializations."""
    inits = []
    # Greedy binary: place 1s to minimize max lag
    h = np.zeros(n)
    ones_needed = n // 2
    # Start with evenly spaced
    positions = list(range(0, n, n // ones_needed))
    while len(positions) < ones_needed:
        best_pos = 0
        best_score = float('inf')
        for p in range(n):
            if h[p] > 0:
                continue
            h[p] = 1.0
            c5_p = compute_c5(h)
            if c5_p < best_score:
                best_score = c5_p
                best_pos = p
            h[p] = 0.0
        positions.append(best_pos)
        h[best_pos] = 1.0
        ones_needed -= 1
        if ones_needed <= 0:
            break
    inits.append(h.copy())
    
    # Random binary
    for trial in range(3):
        h = np.zeros(n)
        ones = rng.choice(n, n//2, replace=False)
        h[ones] = 1.0
        inits.append(h)
    
    # Weyl quadratic
    h = np.zeros(n)
    for i in range(n):
        val = (i * i) % n
        h[i] = 1.0 if val < n // 2 else 0.0
    h = make_feasible(h, n)
    inits.append(h)
    
    return inits

def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

    if remaining() > 30:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, n_sweeps=3, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    
    if remaining() > 20:
        h_ls, c5_ls = pair_local_search(best_h, best_n, n_rounds=30, rng=rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
    
    # Phase 10: Very fine random perturbation
    if remaining() > 15:
        n_trials = min(15000, int(remaining() * 0.7 / 0.008))
        best_h, best_c5 = random_perturb_search(best_h, best_n, n_trials, 0.01, rng)
    
    # Final feasibility enforcement
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Verify
    assert np.all(best_h >= -1e-10), "h has negative values"
    assert np.all(best_h <= 1.0 + 1e-10), "h exceeds 1"
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8, f"Sum constraint violated: {np.sum(best_h)} vs {best_n/2.0}"
    
    return (best_h, best_c5, best_n)
