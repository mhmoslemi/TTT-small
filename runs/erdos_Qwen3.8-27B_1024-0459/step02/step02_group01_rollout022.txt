        else:
            h += diff / n

    if abs(diff) > 1e-13:
        if diff > 0:

        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub

def coordinate_descent(h_init, n, time_limit=60.0, max_sweeps=100):
    h = h_init.copy()
    t_start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                    break
            old_val = h[i]
            # Optimize h[i] by minimizing max_k f_k(h) where f_k is linear in h[i]
            # Golden section search on [0, 1]
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1) / 2.0  # golden ratio conjugate
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            
            h[i] = c
            fc = compute_c5_local(h, i)  # max corr (not scaled)
            h[i] = d
            fd = compute_c5_local(h, i)
            
            for _ in range(20):  # 20 iterations of golden section
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    h[i] = c
                    fc = compute_c5_local(h, i)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    h[i] = d
                    fd = compute_c5_local(h, i)
            
            h[i] = (a + b) / 2.0
            h[i] = min(max(h[i], 0.0), 1.0)
            new_val = h[i]
            if abs(new_val - old_val) > 1e-8:
                improved = True
        
        # Re-project to maintain sum constraint
        h = project_feasible(h, n)
        
        if not improved:
            break
    
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
