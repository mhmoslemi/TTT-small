            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_sum(h, n):
    """Project h to [0,1]^n with sum = n/2 using bisection on shift."""
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Find shift λ such that sum(clip(h - λ, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        h_shifted = np.clip(h - mid, 0.0, 1.0)
        s_mid = np.sum(h_shifted)
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h_out = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h_out)
    if abs(diff) > 1e-10:
        # Redistribute among free variables
        free = (h_out > 1e-10) & (h_out < 1.0 - 1e-10)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h_out[free] += diff / cnt
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def coord_descent(h, n, n_passes, time_limit):
    """Fast coordinate descent with incremental correlation update."""
    h = h.copy()
    dx = 2.0 / n
    best_c5 = np.max(get_corr(h)) * dx
    best_h = h.copy()
    corr = get_corr(h).copy()
    ell_arr = np.arange(-(n-1), n)
    
    start = time.time()
    
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        
        improved = False
        for j in range(n):
            # Compute a vector for coordinate j
            a = np.zeros(2*n-1)
            idx1 = j - ell_arr
            v1 = (idx1 >= 0) & (idx1 < n)
            if np.any(v1):
                a[v1] += 1.0 - h[idx1[v1]]
            idx2 = j + ell_arr
            v2 = (idx2 >= 0) & (idx2 < n)
            if np.any(v2):
                a[v2] += -h[idx2[v2]]
            
            b = corr - h[j] * a
            
            # Bisection: find t in [0,1] minimizing max_k(a[k]*t + b[k])
            lo, hi = 0.0, 1.0
            for _ in range(45):
                mid = 0.5 * (lo + hi)
                vals = a * mid + b
                mx = np.max(vals)
                active = vals >= mx - 1e-13
                if np.any(active):
                    sg = np.mean(a[active])
                else:
                    sg = 0.0
                if sg > 1e-13:
                    hi = mid
                elif sg < -1e-13:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            
            new_val = 0.5 * (lo + hi)
            delta = new_val - h[j]
            if abs(delta) > 1e-14:
                h[j] = new_val
                corr += delta * a
                improved = True
        
        if not improved:
            break
        
        # Project to feasible set
        h = project_sum(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    h = project_sum(h, n)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)

    h_interp = np.interp(x_new, x_old, h_src)

def perturbation_search(h, n, time_limit, rng):
    """Add random perturbations to escape local minima."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    start = time.time()

        n_perturb = rng.randint(1, 5)

        h_try = best_h.copy()
        for idx in idxs:
            h_try[idx] = min(1.0, max(0.0, h_try[idx] + rng.uniform(-0.3, 0.3)))
        h_try = project_sum(h_try, n)
        
        # Quick coordinate descent from perturbed point
        h_cd, c5_cd = coord_descent(h_try, n, n_passes=50, time_limit=5)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    return best_h, best_c5

def slsqp_polish(h, n, max_iter=2000, beta=200):
    """Polish using SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c = corr * dx
        mc = np.max(c)
        return mc + (1.0/beta) * np.log(np.sum(np.exp(beta*(c - mc))))
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c = corr * dx
        mc = np.max(c)
        probs = np.exp(beta*(c - mc)) / np.sum(np.exp(beta*(c - mc)))
        # grad of log-sum-exp w.r.t. x is probs * grad(corr) * dx
        # grad of corr w.r.t. x[j] is a[j] (as computed in coord_descent)
        # This is complex; let's use numerical gradient or just let SLSQP compute it
        return None  # Let SLSQP use numerical gradient
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10 and np.all(np.isfinite(res.x)):
            h_out = project_sum(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def binary_local_search(h, n, time_limit, rng):

    h_bin = (h > 0.5).astype(float)

            h_try[i] = 1.0 - h_try[i]
            # Fix sum

                    j = zeros[rng.randint(len(zeros))]

    h5 = np.zeros(n)
    positions = sorted(range(n), key=lambda i: (i * 104729) % n)
    for j in range(n // 2):
        h5[positions[j]] = 1.0
    inits.append(make_feasible(h5, n))
    for _ in range(2):
        inits.append(make_feasible(rng.rand(n) * 2.0, n))
    return inits


def binary_local_search(h, n, time_limit, rng):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    s = int(np.sum(h_bin))
    if s < target:
        zeros_idx = np.where(h_bin == 0)[0]
        if len(zeros_idx) > 0:
            order = np.argsort(-h[zeros_idx])[:target - s]
            h_bin[zeros_idx[order]] = 1.0
    elif s > target:
        ones_idx = np.where(h_bin == 1)[0]
        if len(ones_idx) > 0:
            order = np.argsort(h[ones_idx])[:s - target]
            h_bin[ones_idx[order]] = 0.0

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()

    while time.time() - start < time_limit:
        for _ in range(2000):
            i = rng.randint(n)
            h_try = best_h.copy()
            h_try[i] = 1.0 - h_try[i]
            s = np.sum(h_try)
            if abs(s - target) > 1e-10:
                if s < target:
                    zeros = np.where(h_try == 0)[0]
                    if len(zeros) == 0:
                        continue
                    j = zeros[rng.randint(len(zeros))]
                    h_try[j] = 1.0
                else:
                    ones = np.where(h_try == 1)[0]
                    if len(ones) == 0:
                        continue
                    j = ones[rng.randint(len(ones))]
                    h_try[j] = 0.0
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
            if time.time() - start > time_limit:
                break

    return best_h, best_c5


def two_scale_optimize(h_low, n_low, n_high, time_limit):
    h_high = interpolate_to_n(h_low, n_high)
    h_cd, c5_cd = coord_descent(h_high, n_high, n_passes=300, time_limit=time_limit * 0.5)
    if time_limit * 0.3 > 10:
        h_slq, c5_slq = slsqp_polish(h_cd, n_high, max_iter=1000, beta=100)
        if h_slq is not None and c5_slq < c5_cd:
            h_cd = h_slq
            c5_cd = c5_slq
        if time_limit * 0.2 > 10:
            h_cd2, c5_cd2 = coord_descent(h_cd, n_high, n_passes=200, time_limit=time_limit * 0.2)
            if c5_cd2 < c5_cd:
                h_cd = h_cd2
                c5_cd = c5_cd2
    return h_cd, c5_cd


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    global initial_h_values
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except (NameError, Exception):
        h_parent = None

    if h_parent is not None:
        best_h = project_sum(h_parent.copy(), len(h_parent))
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: Intensive coordinate descent at parent resolution
    if remaining() > 80:
        tl = min(180, remaining() - 80)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=2000, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: Try structured inits at parent resolution
    if remaining() > 60:
        inits = generate_structured_init(best_n, rng)
        for h_init in inits:
            if remaining() < 30:
                break
            tl = min(40, remaining() - 30)
            h_cd, c5_cd = coord_descent(h_init, best_n, n_passes=300, time_limit=tl)
            if c5_cd < best_c5 - 1e-13:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 3: Multi-resolution optimization
    resolutions = sorted(set([best_n, 128, 160, 200, 240, 288, 320, 400, 480, 512, 640]))
    for n in resolutions:
        if remaining() < 60:
            break
        if n == best_n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_to_n(best_h, n)

        tl = min(100, remaining() - 80)
        if tl < 15:
            continue

        h_cd, c5_cd = coord_descent(h_init, n, n_passes=800, time_limit=tl)

        if remaining() > 50:
            h_slq, c5_slq = slsqp_polish(h_cd, n, max_iter=1500, beta=120)
            if h_slq is not None and c5_slq < c5_cd:
                h_cd = h_slq
                c5_cd = c5_slq
            if remaining() > 30:
                h_cd2, c5_cd2 = coord_descent(h_cd, n, n_passes=200, time_limit=min(25, remaining() - 30))
                if c5_cd2 < c5_cd:
                    h_cd = h_cd2
                    c5_cd = c5_cd2

        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n

    # Phase 4: Perturbation search at best resolution
    if remaining() > 80:
        h_pert, c5_pert = perturbation_search(best_h, best_n, time_limit=min(100, remaining() - 60), rng=rng)
        if c5_pert < best_c5 - 1e-13:
            best_c5 = c5_pert
            best_h = h_pert.copy()

    # Phase 5: Try higher resolution two-scale
    if remaining() > 80:
        nHigh = best_n + best_n // 4
        nHigh = (nHigh // 4) * 4
        if nHigh > 64:
            tl = min(60, remaining() - 50)
            h_hs, c5_hs = two_scale_optimize(best_h, best_n, nHigh, time_limit=tl)
            if c5_hs < best_c5 - 1e-13:
                best_c5 = c5_hs
                best_h = h_hs.copy()
                best_n = nHigh

    # Phase 6: Binary local search
    if remaining() > 30:
        h_bin, c5_bin = binary_local_search(best_h, best_n, time_limit=min(25, remaining() - 20), rng=rng)
        if c5_bin < best_c5 - 1e-13:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            if remaining() > 20:
                h_pol, c5_pol = coord_descent(best_h.copy(), best_n, n_passes=100, time_limit=min(15, remaining() - 15))
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Phase 7: Second perturbation round
    if remaining() > 50:
        h_pert2, c5_pert2 = perturbation_search(best_h, best_n, time_limit=min(60, remaining() - 40), rng=rng)
        if c5_pert2 < best_c5 - 1e-13:
            best_c5 = c5_pert2
            best_h = h_pert2.copy()

    # Phase 8: Final SLSQP polish
    if remaining() > 40:
        h_fin, c5_fin = slsqp_polish(best_h, best_n, max_iter=3000, beta=80)
        if h_fin is not None and c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()
        if remaining() > 25:
            h_fin2, c5_fin2 = slsqp_polish(best_h, best_n, max_iter=2000, beta=400)
            if h_fin2 is not None and c5_fin2 < best_c5 - 1e-14:
                best_c5 = c5_fin2
                best_h = h_fin2.copy()

    # Phase 9: Final intensive coordinate descent
    if remaining() > 15:
        tl = max(5, remaining() - 5)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=1000, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final feasibility enforcement
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
