def binary_sa(h_init, n, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    # Start from binary version of h_init
    h_bin = (h_init >= 0.5).astype(float)
    # Fix to exactly n/2 ones
    s = np.sum(h_bin)
    if s != n/2:
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if s > n/2:
            remove = rng.choice(ones, int(s - n/2), replace=False)
            h_bin[remove] = 0
        else:
            add = rng.choice(zeros, int(n/2 - s), replace=False)
            h_bin[add] = 1
    
    dx = 2.0 / n
    corr = get_corr(h_bin)
    best_c5 = np.max(corr) * dx
    best_h = h_bin.copy()
    curr_c5 = best_c5
    
    T_start = 0.05
    T_end = 0.001
    start = time.time()
    elapsed_total = 0
    
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = elapsed / time_limit
        T = T_start * (T_end / T_start) ** progress
        
        # Choose move: swap a random 1 with a random 0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        a_idx = ones[rng.randint(len(ones))]
        b_idx = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h_bin[a_idx] = 0
        h_bin[b_idx] = 1
        
        new_corr = get_corr(h_bin)
        new_c5 = np.max(new_corr) * dx
        
        delta = new_c5 - curr_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            curr_c5 = new_c5
            corr = new_corr
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h_bin.copy()
        else:
            # Revert
            h_bin[a_idx] = 1
            h_bin[b_idx] = 0
    
    return best_h, best_c5
