       if su > target:

    n = len(h)

def project_box_sum(v, target):
    h = np.asarray(v, dtype=float).copy()
    n = h.size
    if n == 0:
        return h
    h = np.clip(h, 0.0, 1.0)

    lo = -1.0
    hi = 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        u = np.clip(h - mid, 0.0, 1.0)
        if float(u.sum()) > target:
            lo = mid
        else:

def build_d_array(h, j, lags):
    n = len(h)
    m = lags.size
    d = np.zeros(m, dtype=float)

    idx1 = j + lags
    v1 = (idx1 >= 0) & (idx1 < n)

def smooth_max(c, beta):
    if c.size == 0:
        return 0.0
    if beta <= 1e-8:
        return float(np.max(c))
    maxc = float(np.max(c))
    ex = np.exp(beta * (c - maxc))
    return maxc + float(np.log(ex.sum())) / beta

    return smooth_max(corr_raw(h), beta)


def smooth_gradient(h, n, beta, lags):
    c = corr_raw(h)
    maxc = float(np.max(c))
    grad = np.zeros(n, dtype=float)

    if beta <= 1e-8:
        active = c >= (maxc - 1e-12)
        if not np.any(active):
            idx = int(np.argmax(c))
            active[idx] = True
        w = active.astype(float)
        w /= w.sum()
    else:
        ex = np.exp(beta * (c - maxc))
        w = ex / ex.sum()

    for idx in range(lags.size):
        wt = float(w[idx])
        if wt <= 1e-16:
            continue
        ell = int(lags[idx])
        if ell == 0:
            grad += wt * (1.0 - 2.0 * h)
        elif ell > 0:
            r = n - ell
            if r > 0:
                grad[:r] += wt * (1.0 - h[ell:])
                grad[ell:] += wt * (-h[:r])
        else:
            k = -ell
            r = n - k
            if r > 0:
                grad[k:] += wt * (1.0 - h[:r])
                grad[:k] += wt * (-h[k:])

    return grad

def coord_descent(h0, n, time_limit, seed=1, mu=0.0, max_passes=1000):
    h = project_box_sum(h0, n / 2.0)
    lags = np.arange(n - 1, -n + 1, -1, dtype=int)
    idx0 = n - 1
    line_idx = np.ones(2 * n - 1, dtype=bool)
    line_idx[idx0] = False

    corr = corr_raw(h)
    current_max = float(np.max(corr))
    dx = 2.0 / n
    best_h = h.copy()
    best_c = current_max * dx

    rng = np.random.RandomState(seed)
    start = time.time()
    no_improve = 0

    for _ in range(max_passes):
        if time.time() - start >= time_limit:
            break

        order = np.arange(n, dtype=int)
        rng.shuffle(order)
        pass_start = current_max

        for j in order:
            y = float(h[j])
            d = build_d_array(h, j, lags)
            x, val = coord_min_x(corr, y, d, idx0, line_idx, float(mu), current_max)
            if x != y and val < current_max - 1e-11:
                dv = x - y
                if abs(dv) > 1e-15:
                    corr = corr + dv * d
                    corr[idx0] = corr[idx0] + x * (1.0 - x) - y * (1.0 - y)
                    h[j] = x
                    current_max = val

        corr = corr_raw(h)
        current_max = float(np.max(corr))

        if current_max < pass_start - 1e-11:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 4:
                break

        hproj = project_box_sum(h, n / 2.0)
        corrproj = corr_raw(hproj)
        maxp = float(np.max(corrproj))
        cp = maxp * dx

        if cp < best_c - 1e-13:
            best_c = cp
            best_h = hproj.copy()
            h = hproj.copy()
            corr = corrproj
            current_max = maxp
        else:
            if maxp > best_c / dx + 1e-6:
                no_improve += 1
                if no_improve > 7:
                    h = best_h.copy()
                    corr = corr_raw(h)
                    current_max = best_c / dx
                    no_improve = 0
                else:
                    h = hproj.copy()
                    corr = corrproj
                    current_max = maxp
            else:
                h = hproj.copy()
                corr = corrproj
                current_max = maxp

    return best_h, best_c


def binary_from_h(h, n):
    order = np.argsort(-h)
    b = np.zeros(n, dtype=float)
    b[order[:n // 2]] = 1.0
    return b


def binary_pair_trial(b, corr, p, q, lags, idx0):
    oldp = float(b[p])
    oldq = float(b[q])
    if oldp == oldq:
        return None

    nb = b.copy()
    nb[p] = 1.0 - oldp
    d1 = build_d_array(b, p, lags)
    delta1 = (nb[p] - oldp) * d1
    delta1[idx0] += nb[p] * (1.0 - nb[p]) - oldp * (1.0 - oldp)

    nb[q] = 1.0 - oldq
    d2 = build_d_array(nb, q, lags)
    delta2 = (nb[q] - oldq) * d2
    delta2[idx0] += nb[q] * (1.0 - nb[q]) - oldq * (1.0 - oldq)

    val = float(np.max(corr + delta1 + delta2))
    return val, nb


def binary_local_search(h0, n, time_limit, seed=123):

    best_raw = float(np.max(corr))

    lags = np.arange(n - 1, -n + 1, -1, dtype=int)

        ones = np.flatnonzero(b > 0.5)
        zeros = np.flatnonzero(b < 0.5)
        if ones.size == 0 or zeros.size == 0:
            break

        best_trial_val = best_raw - 1e-12

        nsamp = 1200 if n <= 180 else 600

            p = int(rng.choice(ones))

            if p == q:
                continue
            res = binary_pair_trial(b, corr, p, q, lags, idx0)
            if res is None:
                continue
            val, nb = res
            if val < best_trial_val - 1e-12:
                best_trial_val = val
                best_trial_b = nb

        if best_trial_b is not None:
            b = best_trial_b
            corr = corr_raw(b)
            best_raw = float(np.max(corr))
            best_b = b.copy()
            stall = 0
        else:
            stall += 1
            if n <= 180 and stall > 3:
                found = False
                trial_b = None
                for p in ones:
                    if found:
                        break
                    for q in zeros:
                        res = binary_pair_trial(b, corr, int(p), int(q), lags, idx0)
                        if res is None:
                            continue
                        val, nb = res
                        if val < best_raw - 1e-12:
                            trial_b = nb
                            found = True
                            break
                if found:
                    b = trial_b
                    corr = corr_raw(b)
                    best_raw = float(np.max(corr))
                    best_b = b.copy()
                    stall = 0
                else:
                    break
            elif stall > 15:
                break

    return best_b, best_raw * dx


def run(seed=42, budget_s=1300, **kwargs):
    run_start = time.time()
    np.random.seed(seed)
    rng = np.random.RandomState(seed)

    h_parent = None
    try:
        hv = np.asarray(initial_h_values, dtype=float)
        if hv.size > 0:
            h_parent = hv.reshape(-1)
    except Exception:
        h_parent = None

    if h_parent is not None:
        n0 = int(len(h_parent))
        if n0 < 16:
            h_parent = None

    if h_parent is not None:
        if n0 % 2 == 1:
            h_parent = interpolate_to_n(h_parent, n0 + 1)
        best_n = int(len(h_parent))
        best_h = project_box_sum(h_parent.copy(), best_n / 2.0)
    else:
        best_n = 160
        best_h = default_initial(best_n, rng)

    best_c = exact_c5(best_h)

    if remaining_time(run_start, budget_s) > 220:
        tl = min(95.0, remaining_time(run_start, budget_s) - 200.0)
        if tl > 5.0:
            h1, c1 = coord_descent(best_h, best_n, tl, seed=17)
            if c1 < best_c - 1e-13:
                best_h = h1
                best_c = c1
                best_n = int(len(best_h))

    base_res = [best_n, 160, 200, 240, 320]
    if best_n <= 400:
        base_res.append(400)
    resolution_os = []
    seen = set()
    for nn in base_res:
        nn = int(nn)
        if nn >= 128 and nn <= 512 and nn % 2 == 0 and nn not in seen:
            seen.add(nn)
            resolution_os.append(nn)
    resolution_os.sort()

    slq_calls = 0
    for n in resolution_os:
        if slq_calls >= 6:
            break
        if remaining_time(run_start, budget_s) < 110:
            break

        if len(best_h) == n:
            hcur = project_box_sum(best_h, n / 2.0)
        else:
            hcur = interpolate_to_n(best_h, n)

        cur_c = exact_c5(hcur)
        betas = [14.0, 35.0, 90.0, 180.0]
        for beta in betas:
            if slq_calls >= 6:
                break
            if remaining_time(run_start, budget_s) < 70:
                break
            maxiter = 260 if n <= 200 else 150
            hnew = slsqp_smooth(hcur, n, beta, maxiter)
            slq_calls += 1
            if hnew is None:
                continue
            cnew = exact_c5(hnew)
            if cnew < best_c - 1e-13:
                best_h = hnew
                best_c = cnew
                best_n = int(len(hnew))
            if cnew <= cur_c + 1e-10:
                hcur = project_box_sum(hnew, n / 2.0)
                cur_c = cnew

    lbfgs_calls = 0
    for n in resolution_os:
        if lbfgs_calls >= 5:
            break
        if remaining_time(run_start, budget_s) < 90:
            break

        if len(best_h) == n:
            hcur = project_box_sum(best_h, n / 2.0)
        else:
            hcur = interpolate_to_n(best_h, n)

        starts = [hcur, project_box_sum(hcur + 0.025 * rng.randn(n), n / 2.0)]
        for hs in starts:
            if lbfgs_calls >= 5:
                break
            if remaining_time(run_start, budget_s) < 55:
                break
            hs_c = exact_c5(hs)
            for beta in [40.0, 110.0]:
                if lbfgs_calls >= 5:
                    break
                if remaining_time(run_start, budget_s) < 50:
                    break
                hl = lbfgsb_penalty(hs, n, beta, alpha=250.0, maxiter=180)
                lbfgs_calls += 1
                if hl is None:
                    continue
                cn = exact_c5(hl)
                if cn < best_c - 1e-13:
                    best_h = hl
                    best_c = cn
                    best_n = int(len(best_h))
                if cn <= hs_c + 1e-10:
                    hs = project_box_sum(hl, n / 2.0)
                    hs_c = cn

    if remaining_time(run_start, budget_s) > 55:
        bn = best_n
        if bn < 128 or bn > 256:
            bn = 240
        bh = interpolate_to_n(best_h, bn)
        bh, bc = binary_local_search(bh, bn, min(70.0, remaining_time(run_start, budget_s) - 35.0), seed=777)
        if bc < best_c - 1e-13:
            best_h = bh
            best_c = bc
            best_n = int(len(best_h))

    if remaining_time(run_start, budget_s) > 30:
        tl = max(5.0, remaining_time(run_start, budget_s) - 12.0)
        h1, c1 = coord_descent(best_h, best_n, tl, seed=313)
        if c1 < best_c - 1e-13:
            best_h = h1
            best_c = c1
            best_n = int(len(best_h))

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c = exact_c5(best_h)

    return (best_h, best_c, best_n)
