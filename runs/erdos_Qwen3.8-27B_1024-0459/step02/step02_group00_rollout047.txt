def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)

def multi_subgradient(h, n, best_c5, n_iters=500, top_k=10):
    """Subgradient descent targeting top-k correlations"""
    dx = 2.0 / n
    momentum = np.zeros(n)
    beta_m = 0.85
    step = 0.05
    cur_c5 = compute_c5(h)
    best_local = cur_c5

    patience = 0

def smooth_obj_and_grad(x, beta, n, dx):
    """Compute smooth objective and its gradient"""
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = corr - m
    shifted = np.clip(shifted, -40.0 / beta, 0.0)  # prevent overflow
    exps = np.exp(beta * shifted)
    sum_exps = np.sum(exps)
    
    obj_val = (m + np.log(sum_exps) / beta) * dx
    
    # Gradient: weighted sum of correlation gradients
    weights = exps / sum_exps
    
    grad = np.zeros(n)

    grad *= dx
    return obj_val, grad
