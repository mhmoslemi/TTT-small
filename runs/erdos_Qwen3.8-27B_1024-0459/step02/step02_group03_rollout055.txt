    return np.max(corr) * dx

    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return m + np.log(total) / beta

    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    e = np.exp(shifted)
    s = np.sum(e)
    if s < 1e-300:
        e = np.ones_like(e) / len(e)
    else:
        e = e / s
    return e


def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.maximum(h, 1.0 - h)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)

    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)

def binary_local_search(h_bin, n, rng, max_iters, budget_fn):
    c5_cur = compute_c5(h_bin)
    h_ls = h_bin.copy()
    ones_all = np.where(h_ls == 1)[0]
    zeros_all = np.where(h_ls == 0)[0]
    no_improve = 0
    for it in range(max_iters):
        if budget_fn() < 15.0:
            break
        if len(ones_all) == 0 or len(zeros_all) == 0:
            break
        best_swap_delta = 0.0
        best_i1 = -1
        best_i0 = -1
        n_tries = 50
        for _ in range(n_tries):
            i1 = ones_all[rng.randint(len(ones_all))]
            i0 = zeros_all[rng.randint(len(zeros_all))]
            h_ls[i1] = 0.0
            h_ls[i0] = 1.0
            new_c5 = compute_c5(h_ls)
            delta = c5_cur - new_c5
            if delta > best_swap_delta:
                best_swap_delta = delta
                best_i1 = i1
                best_i0 = i0
            h_ls[i1] = 1.0
            h_ls[i0] = 0.0
        if best_i1 >= 0:
            h_ls[best_i1] = 0.0
            h_ls[best_i0] = 1.0
            c5_cur = c5_cur - best_swap_delta
            ones_all = np.where(h_ls == 1)[0]
            zeros_all = np.where(h_ls == 0)[0]
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Score-based restart: perturb and try again
                n_pert = min(5, len(ones_all) // 2)
                if n_pert > 0:
                    perm1 = rng.choice(len(ones_all), size=n_pert, replace=False)
                    perm0 = rng.choice(len(zeros_all), size=n_pert, replace=False)
                    for pi in range(n_pert):
                        h_ls[ones_all[perm1[pi]]] = 0.0
                        h_ls[zeros_all[perm0[pi]]] = 1.0
                    new_c5 = compute_c5(h_ls)
                    if new_c5 < c5_cur:

                            h_ls[ones_all[perm1[pi]]] = 1.0
                            h_ls[zeros_all[perm0[pi]]] = 0.0
                        break
                else:
                    break
    return h_ls, c5_cur

                    break
            # Quick binary search at alt resolution
            if remaining() > 30.0:
                h_b = (best_h >= 0.5).astype(float)
                n_want = best_n // 2
                n_ones = int(np.sum(h_b))
                if n_ones > n_want:
                    oi = np.where(h_b == 1.0)[0]
                    order = np.argsort(best_h[oi])[:n_ones - n_want]
                    h_b[oi[order]] = 0.0
                elif n_ones < n_want:
                    zi = np.where(h_b == 0.0)[0]
                    order = np.argsort(-best_h[zi])[:n_want - n_ones]
                    h_b[zi[order]] = 1.0
                h_ls2, c5_ls2 = binary_local_search(h_b, best_n, rng, 3000, remaining)
                if c5_ls2 < best_c5:
                    best_c5 = c5_ls2
                    best_h = h_ls2.copy()
                    print("  Alt binary: C5={:.6f}".format(best_c5))

    # Final SLSQP deep polish
    if remaining() > 30.0:
        n = best_n
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(_slsqp_obj, best_h.copy(),
                           args=(n, 5000), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _slsqp_constraint, 'args': (n,)}],
                           options={'maxiter': 1000, 'ftol': 1e-15, 'iprint': 0})
            h_fin = project_feasible(res.x, n)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        except Exception:
            pass

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
