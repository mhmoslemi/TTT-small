            if c_opt < best_c:
                h_opt2, c_opt2 = local_search_pipeline(h_opt, time.perf_counter() + 5.0, ni)
                if c_opt2 < best_c:
                    best_h = h_opt2.copy()
                    best_c = c_opt2
                    best_n = ni

    # Simulated annealing on best
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 30.0:
        nbest = int(best_h.size)
        sa_deadline = time.perf_counter() + rem * 0.4
        h_sa, c_sa = simulated_annealing(best_h, sa_deadline, max_iter=100000)
        if c_sa < best_c:
            best_h = h_sa.copy()
            best_c = c_sa
            best_n = nbest

    # Final intensive polish
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest

        # Binarize push
        if nbest <= 900:
            for thresh in (0.15, 0.1, 0.05):
                h_bin = binarize_push(best_h, thresh)
                c_bin = evaluate_erdos(h_bin)
                if c_bin < best_c:
                    best_h = h_bin.copy()
                    best_c = c_bin

        # Extended pair swap
        rem2 = deadline - time.perf_counter()
        if rem2 > 15.0 and nbest <= 900:
            h2, c2 = pair_swap_search(best_h, time.perf_counter() + rem2 * 0.4,
                                      max_iter=3000, top_frac=0.04, min_k=10, n_pos=12, n_neg=12)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2

        # Gradient polish
        rem3 = deadline - time.perf_counter()
        if rem3 > 10.0:
            h3, c3 = gradient_polish(best_h, time.perf_counter() + rem3 * 0.7,
                                     max_iter=1000, top_frac=0.03, min_k=12)
            if c3 < best_c:
                best_h = h3.copy()
                best_c = c3

        # One more pair swap round
        rem4 = deadline - time.perf_counter()
        if rem4 > 8.0 and nbest <= 720:
            h4, c4 = pair_swap_search(best_h, time.perf_counter() + rem4 * 0.8,
                                      max_iter=1500, top_frac=0.03, min_k=10, n_pos=8, n_neg=8)
            if c4 < best_c:
                best_h = h4.copy()
                best_c = c4

        # Brief SA on the polished result
        rem5 = deadline - time.perf_counter()
        if rem5 > 10.0:
            h5, c5 = simulated_annealing(best_h, time.perf_counter() + rem5 * 0.7, max_iter=50000)
            if c5 < best_c:
                best_h = h5.copy()
                best_c = c5

    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
