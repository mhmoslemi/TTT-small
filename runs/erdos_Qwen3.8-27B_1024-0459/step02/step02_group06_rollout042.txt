    global _G_N, _G_ALPHA
    _G_N = n
    target = 0.5 * n
    best_h = project_density(h0, target)
    best_c = evaluate_erdos(best_h)
    h = best_h.copy()
    total_it = 0
    
    for alpha in alpha_list:
        rem = deadline - time.perf_counter()
        if rem < 5.0:
            break
        _G_ALPHA = alpha
        iter_budget = min(max_total_iter, max(50, int(rem / 0.05)))
        step = 0.05
        stall = 0
        h = best_h.copy()
        for it in range(iter_budget):
            total_it += 1
            if time.perf_counter() >= deadline - 2.0:
                break
            grad = _smooth_jac(h)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            improved = False
            for s in (step, step * 0.5, step * 0.2, step * 2.0):
                h_new = project_density(h - s * grad, target)
                c = evaluate_erdos(h_new)
                if c < best_c - 1e-14:
                    best_h = h_new
                    best_c = c
                    h = h_new
                    step = s
                    improved = True
                    break
            if improved:
                stall = 0
            else:
                step *= 0.5
                stall += 1
                if stall > 20:
                    break
    return project_density(best_h, target), evaluate_erdos(best_h)

def corr_grad_single(h, k):
    """Gradient of corr[k] with respect to h"""
    n = h.size
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(k)
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g

def local_search(h, deadline, max_iter=5000, top_frac=0.05, min_k=6, n_trials=30):
    """Aggressive local search with pair swaps"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    rng = np.random.default_rng(12345)
    stall = 0
    
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        center = n - 1
        k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        range_dv = dv.max() + 1e-10
        w = np.exp(dv / range_dv)
        w /= w.sum()
        
        # Compute weighted gradient
        g = np.zeros(n)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            g += w[i] * corr_grad_single(best, k)
        
        # Greedy: find best pair swap
        neg_pool = np.argsort(g)[:n_trials]
        pos_pool = np.argsort(-g)[:n_trials]
        
        found = False
        for ii in neg_pool:
            ii = int(ii)
            for jj in pos_pool:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                for frac in (0.3, 0.6, 0.9, 1.0):
                    d = frac * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        
        # If greedy failed, try random swaps
        if not found:
            for _r in range(20):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-12:
                    continue
                d = float(rng.random()) * cap * 0.7
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    
    return project_density(best, target), evaluate_erdos(best)

def gradient_polish(h, deadline, max_iter=1000, top_frac=0.05, min_k=6):
    """Gradient-based polishing with line search"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    stall = 0
    
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")

        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / (dv.max() + 1e-10))
        w /= w.sum()
        
        g = np.zeros(n)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            g += w[i] * corr_grad_single(best, k)
        
        # Normalize and center
        d = -g
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:

                best_c = c
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1

                break
    
    return project_density(best, target), evaluate_erdos(best)

    return project_density(best, target), evaluate_erdos(best)

def gradient_polish(h, deadline, max_iter=2000, top_frac=0.04, min_k=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    stall = 0
    
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        center = n - 1
        k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / (dv.max() + 1e-10))
        w /= w.sum()
        
        g = np.zeros(n)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            g += w[i] * corr_grad_single(best, k)
        
        d = -g
        d -= np.mean(d)
        dn = np.linalg.norm(d)
        if dn < 1e-14:
            break
        d /= dn
        
        improved = False
        for step in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1):
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
    
    return project_density(best, target), evaluate_erdos(best)
