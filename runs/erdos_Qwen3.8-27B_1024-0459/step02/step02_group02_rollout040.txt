def thue_morse_binary(n):

    h = np.zeros(n)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):

        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt

  - Before: h[pos1] * dh[pos1-k] = 1 * dh[pos1-k] = dh[pos1-k]
  - After: h[pos1] * dh[pos1-k] = 0 * dh[pos1-k] = 0
  - Change: -dh[pos1-k]

- The pair (A=pos1+k, B=pos1) appears in c[k] when pos1+k - pos1 = k. ✓ (always "valid" in terms of the shift)
  - Before: h[pos1+k] * dh[pos1] = h[pos1+k] * 0 = 0
  - After: h[pos1+k] * dh[pos1] = h[pos1+k] * 1 = h[pos1+k]
  - Change: +h[pos1+k] (if pos1+k < n, i.e., k < n - pos1)

Similarly for pos0 (h goes 0→1, dh goes 1→0):
- The pair (A=pos0, B=n) in c[k] when n = pos0-k:
  - Before: h[pos0] * dh[pos0-k] = 0 * dh[pos0-k] = 0
  - After: h[pos0] * dh[pos0-k] = 1 * dh[pos0-k] = dh[pos0-k]
  - Change: +dh[pos0-k]

- The pair (A=pos0+k, B=pos0) in c[k]:
  - Before: h[pos0+k] * dh[pos0] = h[pos0+k] * 1 = h[pos0+k]
  - After: h[pos0+k] * dh[pos0] = h[pos0+k] * 0 = 0
  - Change: -h[pos0+k]

So total change in c[k]:
Δc[k] = -dh[pos1-k] + h[pos1+k] + dh[pos0-k] - h[pos0+k]

(with appropriate validity checks: 0 ≤ pos1-k < n for the first and third terms, 0 ≤ pos1+k < n for the second and fourth terms)

This is O(1) per correlation index k! But we need to track which k values change. The affected k values are those where:
- pos1-k is valid: k = pos1 - n for n in [0, n-1], i.e., k in [pos1-(n-1), pos1]
- pos1+k is valid: k in [-pos1, n-1-pos1]
- pos0-k is valid: k in [pos0-(n-1), pos0]
- pos0+k is valid: k in [-pos0, n-1-pos0]

So the affected k range is from min(pos1-(n-1), -pos1, pos0-(n-1), -pos0) to max(pos1, n-1-pos1, pos0, n-1-pos0).

In the 'full' correlate output, the index offset is (n-1), so the actual array index is k + (n-1).

The total number of affected indices is at most 2n-1 (all of them in the worst case), but typically it's about 2*max(pos1, pos0) + 2*min(n-1-pos1, n-1-pos0) indices.

For random pos1, pos0, this is about n/2 + n/2 = n indices on average. So the incremental update is O(n) per swap, compared to O(n²) for full recompute.

For n=256, O(n) = 256 operations vs O(n²) = 65536. That's a 256x speedup!

def init_correlation(h, dh):

def update_correlation(corr, h, dh, pos1, pos0, n):
    """
    Update correlation after swapping h[pos1] (1→0) and h[pos0] (0→1).
    Also updates h and dh in place.
    Returns the new max of corr * (2/n).
    """
    # Compute changes
    # Δc[k] = -dh[pos1-k] + h[pos1+k] + dh[pos0-k] - h[pos0+k]
    # where k = actual_shift, and array index = k + (n-1)
    
    # First, compute the delta for all affected k values
    # k ranges where each term is valid:
    # Term 1: -dh[pos1-k], valid when 0 <= pos1-k < n, i.e., pos1-n+1 <= k <= pos1
    # Term 2: +h[pos1+k], valid when 0 <= pos1+k < n, i.e., -pos1 <= k <= n-1-pos1
    # Term 3: +dh[pos0-k], valid when 0 <= pos0-k < n, i.e., pos0-n+1 <= k <= pos0
    # Term 4: -h[pos0+k], valid when 0 <= pos0+k < n, i.e., -pos0 <= k <= n-1-pos0
    
    k_min = max(0 - (n-1), min(pos1-n+1, -pos1, pos0-n+1, -pos0))
    k_max = min((n-1), max(pos1, n-1-pos1, pos0, n-1-pos0))
    
    # Actually k ranges from -(n-1) to (n-1) in 'full' mode
    # Array index = k + (n-1), ranging from 0 to 2n-2
    
    for k in range(k_min, k_max + 1):
        delta = 0.0
        # Term 1: -dh[pos1-k] if valid
        idx = pos1 - k
        if 0 <= idx < n:
            delta -= dh[idx]
        # Term 2: +h[pos1+k] if valid
        idx = pos1 + k
        if 0 <= idx < n:
            delta += h[idx]
        # Term 3: +dh[pos0-k] if valid
        idx = pos0 - k
        if 0 <= idx < n:
            delta += dh[idx]
        # Term 4: -h[pos0+k] if valid
        idx = pos0 + k
        if 0 <= idx < n:
            delta -= h[idx]
        corr[k + (n-1)] += delta
    
    # Update h and dh
    h[pos1] = 0.0
    h[pos0] = 1.0
    dh[pos1] = 1.0
    dh[pos0] = 0.0
    
    return np.max(corr) * (2.0 / n)

def update_correlation_vec(corr, h, dh, pos1, pos0, n):
    """Vectorized incremental update."""
    offset = n - 1
    
    # For each k in range [-(n-1), n-1]:
    # delta = -dh[pos1-k] (if valid) + h[pos1+k] (if valid) + dh[pos0-k] (if valid) - h[pos0+k] (if valid)
    
    # k = 0, 1, ..., 2n-2 in array index, corresponding to shift -(n-1) to (n-1)
    # shift = array_idx - offset

    shifts = np.arange(-(n-1), n)  # shape (2n-1,)
    
    delta = np.zeros(2*n-1)
    
    # Term 1: -dh[pos1 - shift], valid when 0 <= pos1-shift < n
    valid1 = (shifts >= pos1 - (n-1)) & (shifts <= pos1)
    if np.any(valid1):
        delta[valid1] -= dh[pos1 - shifts[valid1]]

    valid2 = (shifts >= -pos1) & (shifts <= n-1-pos1)
    if np.any(valid2):

    valid3 = (shifts >= pos0 - (n-1)) & (shifts <= pos0)

    valid4 = (shifts >= -pos0) & (shifts <= n-1-pos0)
    if np.any(valid4):
        delta[valid4] -= h[pos0 + shifts[valid4]]

    # Update h and dh
    h[pos1] = 0.0
    h[pos0] = 1.0
    dh[pos1] = 1.0
    dh[pos0] = 0.0

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def thue_morse_binary(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        excess = ones - target
        one_positions = np.where(h == 1.0)[0]
        np.random.RandomState(123).shuffle(one_positions)
        h[one_positions[:excess]] = 0.0
    elif ones < target:
        deficit = target - ones
        zero_positions = np.where(h == 0.0)[0]
        np.random.RandomState(123).shuffle(zero_positions)
        h[zero_positions[:deficit]] = 1.0
    return h

def random_binary(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        excess = ones - target
        one_positions = np.where(h == 1.0)[0]
        to_flip = rng.choice(one_positions, excess, replace=False)
        h[to_flip] = 0.0
    elif ones < target:
        deficit = target - ones
        zero_positions = np.where(h == 0.0)[0]
        to_flip = rng.choice(zero_positions, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def simulated_annealing(h_init, n, rng, t_start=1.0, t_end=0.001, n_iter=500000, timeout=200.0, restart_prob=0.0, restart_interval=50000):
    """Simulated annealing for the binary Erdős problem."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    dh = 1.0 - h
    dx = 2.0 / n
    n_ones = n // 2
    
    corr = np.correlate(h, dh, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    one_pos = np.where(h == 1.0)[0].copy()
    zero_pos = np.where(h == 0.0)[0].copy()
    
    start_t = time.time()
    
    for it in range(n_iter):
        elapsed = time.time() - start_t
        if elapsed > timeout:
            break
        
        progress = it / max(n_iter - 1, 1)
        T = t_start * (t_end / t_start) ** progress
        
        # Periodic restart from best
        if restart_prob > 0 and it > 0 and it % restart_interval == 0:
            if rng.rand() < restart_prob:
                h[:] = best_h
                dh[:] = 1.0 - h
                current_c5 = best_c5
                one_pos[:] = np.where(h == 1.0)[0]
                zero_pos[:] = np.where(h == 0.0)[0]
                continue
        
        # Random swap
        idx1 = rng.randint(n_ones)
        idx0 = rng.randint(n_ones)
        pos1 = one_pos[idx1]
        pos0 = zero_pos[idx0]
        
        if pos1 == pos0:
            continue
        
        # Apply swap
        h[pos1] = 0.0
        h[pos0] = 1.0
        dh[pos1] = 1.0
        dh[pos0] = 0.0
        
        # Evaluate
        corr = np.correlate(h, dh, mode='full')
        new_c5 = np.max(corr) * dx
        delta = new_c5 - current_c5
        
        # Accept/reject
        if delta < 0:
            current_c5 = new_c5
            one_pos[idx1] = pos0
            zero_pos[idx0] = pos1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = h
        elif T > 1e-10:
            # Compute acceptance probability (avoid overflow)
            exp_arg = -delta / T
            if exp_arg < -50:
                accept = False
            else:
                accept = rng.rand() < np.exp(exp_arg)
            if accept:
                current_c5 = new_c5
                one_pos[idx1] = pos0
                zero_pos[idx0] = pos1
            else:
                # Reject: undo
                h[pos1] = 1.0
                h[pos0] = 0.0
                dh[pos1] = 0.0
                dh[pos0] = 1.0
        else:
            # T too small, reject
            h[pos1] = 1.0
            h[pos0] = 0.0
            dh[pos1] = 0.0
            dh[pos0] = 1.0
    
    return best_h, best_c5

def iterated_local_search(h_init, n, rng, timeout=100.0, kick_sizes=(2, 3, 5), max_kicks=200):
    """ILS with varying kick sizes."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    dh = 1.0 - h
    dx = 2.0 / n
    n_ones = n // 2
    
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    start_t = time.time()
    
    for kick in range(max_kicks):
        if time.time() - start_t > timeout:
            break
        
        # Local search (hill climbing)
        no_improve = 0
        max_stall = 3000
        while no_improve < max_stall:
            if time.time() - start_t > timeout:
                break
            
            one_pos = np.where(h == 1.0)[0]
            zero_pos = np.where(h == 0.0)[0]
            pos1 = one_pos[rng.randint(n_ones)]
            pos0 = zero_pos[rng.randint(n_ones)]
            
            if pos1 == pos0:
                continue
            
            h[pos1] = 0.0
            h[pos0] = 1.0
            dh[pos1] = 1.0
            dh[pos0] = 0.0
            
            new_c5 = np.max(np.correlate(h, dh, mode='full')) * dx
            
            if new_c5 < current_c5 - 1e-15:
                current_c5 = new_c5
                no_improve = 0
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h[:] = h
            else:
                h[pos1] = 1.0
                h[pos0] = 0.0
                dh[pos1] = 0.0
                dh[pos0] = 1.0
                no_improve += 1
        
        # Kick
        kick_size = kick_sizes[kick % len(kick_sizes)]
        for _ in range(kick_size):
            one_pos = np.where(h == 1.0)[0]
            zero_pos = np.where(h == 0.0)[0]
            pos1 = one_pos[rng.randint(n_ones)]
            pos0 = zero_pos[rng.randint(n_ones)]
            if pos1 != pos0:
                h[pos1] = 0.0
                h[pos0] = 1.0
                dh[pos1] = 1.0
                dh[pos0] = 0.0
        
        current_c5 = np.max(np.correlate(h, dh, mode='full')) * dx
        
        # Occasionally restart from best
        if rng.rand() < 0.15:
            h[:] = best_h
            dh[:] = 1.0 - h
            current_c5 = best_c5
    
    return best_h, best_c5
