    """Gradient of corr at lag k w.r.t. h"""
    g = np.zeros(n, dtype=float)

import time
import numpy as np
from scipy.optimize import minimize

_G_N = 1
_G_ALPHA = 1.0
_G_TARGET = 0.5


def _project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    s = float(np.sum(x))
    if abs(s - target) < 1e-14:
        return x
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-13) & (h < 1.0 - 1e-13))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
            res2 = target - float(np.sum(h))
            if abs(res2) > 1e-10:
                free2 = np.where((h > 0.0) & (h < 1.0))[0]
                if free2.size > 0:
                    h[free2[0]] += res2
                    h = np.clip(h, 0.0, 1.0)
    return h


def _upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    if n0 <= 0:
        return _project_density(np.full(max(1, n), 0.5), 0.5 * n)
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return _project_density(h, 0.5 * n)


def _get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def _lag_grad(h, k, n):
    g = np.zeros(n, dtype=float)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        lag = abs(int(k))
        if lag >= n:
            return g
        if k > 0:
            g[lag:] += 1.0
        else:
            g[:n - lag] += 1.0
        g[:n - lag] -= h[lag:]
        g[lag:] -= h[:n - lag]
    return g


def _smo_obj(x):
    n = _G_N
    alpha = _G_ALPHA
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return 1e100
    corr = _get_corr(h)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0:
        return m * (2.0 / n)
    return (2.0 / n) * (m + np.log(z) / alpha)


def _smo_jac(x):
    n = _G_N
    alpha = _G_ALPHA
    h = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(h)):
        return np.zeros(n)
    corr = _get_corr(h)
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0.0:
        return np.zeros(n)
    w = e / z
    center = n - 1
    g = np.zeros(n, dtype=float)
    for t in range(len(corr)):
        if w[t] < 1e-12:
            continue
        k = t - center
        g += w[t] * _lag_grad(h, k, n)
    return (2.0 / n) * g


def _sum_fun(x):
    return float(np.sum(x) - _G_TARGET)


def _sum_jac(x):
    return np.ones_like(np.asarray(x, dtype=float))


def _slsqp_optimize(h, n, deadline):
    target = 0.5 * n
    best = _project_density(h, target)
    best_c = evaluate_erdos(best)
    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": _sum_fun, "jac": _sum_jac},)

    if n <= 500:
        alphas = (15.0, 50.0, 150.0, 400.0)
        maxiter = 150
    elif n <= 800:
        alphas = (15.0, 50.0, 150.0)
        maxiter = 100
    else:
        alphas = (15.0, 40.0, 100.0)
        maxiter = 60

    for alpha in alphas:
        if time.perf_counter() >= deadline - 5.0:
            break
        _G_N = n
        _G_ALPHA = alpha
        _G_TARGET = target
        try:
            res = minimize(
                _smo_obj, best,
                method="SLSQP", jac=_smo_jac,
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-12, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = _project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best = cand
                    best_c = c
        except Exception:
            pass

    return best, evaluate_erdos(best)


def _focused_local_search(h, deadline, top_frac=0.05, min_k=6, max_iter=8000, n_pos=10, n_neg=10, use_anneal=False, init_temp=0.002):
    n = h.size
    target = 0.5 * n
    best = _project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(77700)
    stall = 0
    stall_limit = 150
    temp = init_temp
    best_c_for_anneal = best_c

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break

        corr = _get_corr(best)
        center = n - 1
        k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / max(1e-10, dv.max() + 1e-10))
        w /= w.sum()

        g = np.zeros(n, dtype=float)
        for i in range(len(top_idx)):
            k = int(top_idx[i]) - center
            g += w[i] * _lag_grad(best, k, n)

        found = False

        neg_cand = np.argsort(g)[:n_neg]
        pos_cand = np.argsort(-g)[:n_pos]

        for ii in neg_cand:
            ii = int(ii)
            for jj in pos_cand:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                for frac in (0.15, 0.4, 0.7, 1.0, 0.999):
                    d = frac * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found and use_anneal:
            for _r in range(10):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * 0.8 * cap
                direction = rng.choice([-1, 1])
                if direction == 1:
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                else:
                    cand = best.copy()
                    cand[ii] -= d
                    cand[jj] += d
                if np.any(cand < -1e-12) or np.any(cand > 1 + 1e-12):
                    continue
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
                elif temp > 1e-8:
                    delta = c - best_c
                    if delta < 0.05 and rng.random() < np.exp(-delta / temp):
                        best = cand
                        best_c_for_anneal = c
                        found = True
                        break
            temp *= 0.998

        if found:
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                break

    return _project_density(best, target), evaluate_erdos(best)


def _block_move_search(h, deadline, max_iter=5000):
    n = h.size
    target = 0.5 * n
    best = _project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(31415)
    stall = 0

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break

        found = False
        for _attempt in range(6):
            block_len = int(rng.integers(2, min(12, n // 3)))
            start1 = int(rng.integers(0, n - block_len))
            start2 = int(rng.integers(0, n - block_len))
            if abs(start1 - start2) < block_len:
                continue
            cand = best.copy()
            b1 = cand[start1:start1 + block_len].copy()
            b2 = cand[start2:start2 + block_len].copy()
            cand[start1:start1 + block_len] = b2
            cand[start2:start2 + block_len] = b1
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                found = True
                break

        if found:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break

    return _project_density(best, target), evaluate_erdos(best)


def _gradient_polish(h, deadline, max_iter=500, top_frac=0.04, min_k=8):
    n = h.size
    target = 0.5 * n
    best = _project_density(h, target)
    best_c = evaluate_erdos(best)
    stall = 0

    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break

        corr = _get_corr(best)
        center = n - 1
        k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / max(1e-10, dv.max() + 1e-10))
        w /= w.sum()

        g = np.zeros(n, dtype=float)
        for i in range(len(top_idx)):
            k = int(top_idx[i]) - center
            g += w[i] * _lag_grad(best, k, n)

        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-15:
            break
        d /= norm

        improved = False
        for step in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
            cand = _project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break

    return _project_density(best, target), evaluate_erdos(best)


def _make_periodic_start(n, p, freq, phase, n_harmonics=1):
    x = np.arange(p)
    h = np.zeros(p)
    for j in range(n_harmonics):
        f = freq * (j + 1)
        amp = 0.4 / (j + 1)
        h += amp * np.cos(2.0 * np.pi * f * x / p + phase + j * 0.7)
    h += 0.5
    h = np.clip(h, 0.0, 1.0)
    h = _project_density(h, 0.5 * p)
    full = np.tile(h, n // p)
    return _project_density(full, 0.5 * n)


def _make_structured_start(n, mode, rng):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 7 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.empty(n, dtype=float)
        seg = max(1, int(n // 10))
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = float(rng.uniform(0.1, 0.9))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.7, 0.3).astype(float)
        h += rng.normal(0, 0.05, n)
    elif mode == 3:
        freq = int(rng.integers(2, 12))
        h = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x / 2)
    else:
        h = 0.5 + 0.25 * np.sin(2 * np.pi * 5 * x / 2) + 0.2 * np.cos(2 * np.pi * 11 * x / 2)
    return _project_density(h, 0.5 * n)


def _binarize_push(h, thresh=0.10):
    n = h.size
    target = 0.5 * n
    h2 = h.copy()
    changed = False
    for i in range(n):
        if h2[i] < thresh:
            h2[i] = 0.0
            changed = True
        elif h2[i] > 1.0 - thresh:
            h2[i] = 1.0
            changed = True
    if not changed:
        return h
    return _project_density(h2, target)


def _optimize_at_resolution(h0, n, deadline, seed=42):
    target = 0.5 * n
    h = _project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    rem = deadline - time.perf_counter()
    if rem > 15.0:
        h2, c2 = _slsqp_optimize(h, n, time.perf_counter() + min(rem * 0.35, 120.0))
        if c2 < best_c:
            best_h = h2
            best_c = c2

    rem = deadline - time.perf_counter()
    if rem > 10.0:
        h3, c3 = _focused_local_search(best_h, time.perf_counter() + rem * 0.4,
                                       top_frac=0.05, min_k=6, max_iter=4000,
                                       n_pos=10, n_neg=10, use_anneal=False)
        if c3 < best_c:
            best_h = h3
            best_c = c3

    rem = deadline - time.perf_counter()
    if rem > 5.0:
        h4, c4 = _gradient_polish(best_h, time.perf_counter() + rem * 0.5,
                                  max_iter=400, top_frac=0.04, min_k=8)
        if c4 < best_c:
            best_h = h4
            best_c = c4

    rem = deadline - time.perf_counter()
    if rem > 5.0:
        h5, c5 = _focused_local_search(best_h, time.perf_counter() + rem * 0.5,
                                       top_frac=0.03, min_k=8, max_iter=3000,
                                       n_pos=8, n_neg=8, use_anneal=False)
        if c5 < best_c:
            best_h = h5
            best_c = c5

    return _project_density(best_h, target), evaluate_erdos(best_h)
