                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = eval_c5(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
    return project_density(best, target), eval_c5(best)

def grad_descent_polish(h, n, deadline, max_iter=3000):
    target = 0.5 * n
    best = project_density(h, target)
    best_c = eval_c5(best)
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = get_corr(best)
        cmax = float(np.max(corr))
        n_top = max(4, int(np.ceil(2 * n * 0.04)))
        top_idx = np.argsort(corr)[-n_top:]
        top_vals = corr[top_idx]
        dv = top_vals - top_vals.min()
        w = np.exp(dv / max(1e-10, dv.max() + 1e-10))
        w /= w.sum()
        g = np.zeros(n)
        center = n - 1
        for i, m in enumerate(top_idx):
            k = int(m) - center
            if k == 0:
                g += w[i] * (1.0 - 2.0 * best)
            else:
                lag = abs(k)
                if k > 0:
                    g[lag:] += w[i]
                    g[:n-lag] -= w[i] * best[lag:]
                    g[lag:] -= w[i] * best[:n-lag]
                else:
                    g[:n-lag] += w[i]
                    g[n-lag:] -= w[i] * best[:n-lag]
                    g[:n-lag] -= w[i] * best[n-lag:]
        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-14:
            break
        d /= norm
        improved = False
        for step in (0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
            cand = project_density(best + step * d, target)
            c = eval_c5(cand)
            if c < best_c - 1e-14:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 25:
                break
    return project_density(best, target), eval_c5(best)

def binary_push(h, n, thresh=0.06):
    target = 0.5 * n
    h2 = h.copy()
    for i in range(n):
        if h2[i] < thresh:
            h2[i] = 0.0
        elif h2[i] > 1.0 - thresh:
            h2[i] = 1.0
    h2 = project_density(h2, target)
    return h2

def smooth_objective_fw(h, n, alpha=30.0):
    corr = get_corr(h)
    cmax = float(np.max(corr))
    y = alpha * (corr - cmax)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z < 1e-100:
        return cmax * (2.0 / n)
    return (cmax + np.log(z) / alpha) * (2.0 / n)

def multi_start_optimize(n, starts, deadline, seed=42):
    best_h = None
    best_c = float('inf')
    rng = np.random.default_rng(seed + n)
    per_start = (deadline - time.perf_counter()) / len(starts)
    per_start = max(15.0, per_start)
    for s in starts:
        rem = deadline - time.perf_counter()
        if rem < 15.0:
            break
        sub_dl = time.perf_counter() + min(per_start, rem - 3.0)
        target = 0.5 * n
        h = project_density(s, target)
        c = eval_c5(h)
        # Quick FW
        h, c = optimize_fw(h, n, time.perf_counter() + min(8.0, (sub_dl - time.perf_counter()) * 0.3))
        # Conflict/Repair: project
        h = project_density(h, target)
        c = eval_c5(h)
        # Paired swap
        rem2 = sub_dl - time.perf_counter()
        if rem2 > 5.0:
            h, c = paired_swap_opt(h, n, time.perf_counter() + rem2 * 0.4, max_iter=3000)
        # Gradient polish
        rem3 = sub_dl - time.perf_counter()
        if rem3 > 3.0:
            h, c = grad_descent_polish(h, n, time.perf_counter() + rem3 * 0.6, max_iter=1500)
        # Final paired
        rem4 = sub_dl - time.perf_counter()
        if rem4 > 3.0:
            h, c = paired_swap_opt(h, n, time.perf_counter() + rem4 * 0.7, max_iter=2000, top_frac=0.04)
        h = project_density(h, target)
        c = eval_c5(h)
        if c < best_c:
            best_h = h.copy()
            best_c = c
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start_time + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 10:
            parent = np.clip(np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        pn = int(parent.size)
        pp = project_density(parent, 0.5 * pn)
        cp = eval_c5(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = pn

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = eval_c5(best_h)
        best_n = n0

    # Define resolutions to try
    base_n = best_n
    resolutions = []
    for rn in [base_n, 2 * base_n, 3 * base_n, int(base_n * 1.5)]:
        if 20 <= rn <= 2000 and rn not in resolutions:
            resolutions.append(rn)
    for extra in [90, 180, 300, 450, 600, 900, 1200, 1800]:
        if 20 <= extra <= 2000 and extra not in resolutions:
            resolutions.append(extra)
    resolutions = sorted(set(resolutions))
    # Limit to at most 5 resolutions
    if len(resolutions) > 5:
        # Keep base_n and a few around it
        resolutions = sorted(set(resolutions), key=lambda x: abs(x - base_n))[:5]

    # Time allocation: weight toward resolutions near base_n and higher
    n_res = len(resolutions)
    weights = []
    for rn in resolutions:
        if rn == base_n:
            weights.append(2.0)
        elif rn <= 2 * base_n:
            weights.append(1.5)
        else:
            weights.append(1.0)
    wsum = sum(weights)
    total_avail = deadline - time.perf_counter() - 60.0  # reserve 60s for final polish
    if total_avail < 30:
        total_avail = deadline - time.perf_counter() - 10.0

    for ri, rn in enumerate(resolutions):
        rem = deadline - time.perf_counter()
        if rem < 20.0:
            break
        alloc = total_avail * weights[ri] / wsum
        # Don't overspend
        alloc = min(alloc, rem - 10.0)
        alloc = max(15.0, alloc)

        # Build starts
        starts = []
        starts.append(upsample_h(best_h, rn))
        if parent is not None:
            starts.append(upsample_h(parent, rn))
        # Add diverse starts
        n_div = 2 if rn <= 300 else 1
        for di in range(n_div):
            starts.append(make_starts(rn, rng, count=1)[0])
        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * rn)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-5):
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        # Optimize at this resolution
        sub_dl = time.perf_counter() + alloc
        h_out, c_out = multi_start_optimize(rn, uniq, sub_dl, seed)
        if h_out is not None:
            h_out = project_density(h_out, 0.5 * rn)
            c_out = eval_c5(h_out)
            if c_out < best_c:
                best_h = h_out.copy()
                best_c = c_out
                best_n = rn

    # Final intensive polish phase
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 10.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        best_c = eval_c5(h_fin)

        # Phase 1: FW
        if rem > 20.0:
            h_fin, c = optimize_fw(h_fin, nbest, time.perf_counter() + rem * 0.3, max_iter=3000)
            if c < best_c:
                best_c = c
                best_h = h_fin.copy()

        # Phase 2: Paired swaps (long)
        rem2 = deadline - time.perf_counter()
        if rem2 > 10.0:
            h_fin, c = paired_swap_opt(h_fin, nbest, time.perf_counter() + rem2 * 0.35, max_iter=8000, top_frac=0.03)
            if c < best_c:
                best_c = c
                best_h = h_fin.copy()

        # Phase 3: Binary push + re-optimize
        rem3 = deadline - time.perf_counter()
        if rem3 > 10.0 and nbest <= 1000:
            h_bin = binary_push(h_fin, nbest, thresh=0.04)
            c_bin = eval_c5(h_bin)
            if c_bin < best_c:
                best_c = c_bin
                best_h = h_bin.copy()
                h_fin = h_bin
            # Re-optimize from binary
            if rem3 > 15.0:
                h_re, c_re = paired_swap_opt(h_bin, nbest, time.perf_counter() + rem3 * 0.2, max_iter=4000)
                if c_re < best_c:
                    best_c = c_re
                    best_h = h_re.copy()

        # Phase 4: Gradient polish
        rem4 = deadline - time.perf_counter()
        if rem4 > 8.0:
            h_fin, c = grad_descent_polish(h_fin, nbest, time.perf_counter() + rem4 * 0.5, max_iter=3000)
            if c < best_c:
                best_c = c
                best_h = h_fin.copy()

        # Phase 5: Final paired swaps (fine)
        rem5 = deadline - time.perf_counter()
        if rem5 > 6.0:
            h_fin, c = paired_swap_opt(h_fin, nbest, time.perf_counter() + rem5 * 0.7, max_iter=5000, top_frac=0.02)
            if c < best_c:
                best_c = c
                best_h = h_fin.copy()

        # Phase 6: One more gradient
        rem6 = deadline - time.perf_counter()
        if rem6 > 4.0:
            h_fin, c = grad_descent_polish(h_fin, nbest, time.perf_counter() + rem6 * 0.8, max_iter=1000)
            if c < best_c:
                best_c = c
                best_h = h_fin.copy()

    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, eval_c5(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = eval_c5(final_h)
    return final_h, float(final_c), int(final_h.size)
