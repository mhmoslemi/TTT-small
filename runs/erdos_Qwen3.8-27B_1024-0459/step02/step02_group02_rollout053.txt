def smooth_max_obj(x, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = beta * (corr - cmax)
    shifted = np.clip(shifted, -50, 50)
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val

                best_c5 = c5_t
                best_h = h_t.copy()
        
        # Phase 3: Simulated annealing
        if remaining() > 50 and time.time() - phase_start < time_alloc * 0.6:
            h_sa, c5_sa = simulated_annealing(best_h, n, rng, start_T=0.4, end_T=0.003,
                                              max_iter=100000,
                                              timeout=min(30.0, remaining() * 0.03))
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
        
        # Phase 4: Intensive binary local search from best
        if remaining() > 60 and time.time() - phase_start < time_alloc * 0.7:
            h_b2, c5_b2 = multistart_binary(best_h, n, rng,
                                             timeout=min(50.0, remaining() * 0.06),
                                             n_restarts=4)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
        
        # Phase 5: Smooth max SLSQP (for continuous improvement)
        if remaining() > 60 and time.time() - phase_start < time_alloc * 0.8:
            h_sm, c5_sm = smooth_max_optimize(best_h, n, beta_start=15.0, beta_end=250.0,
                                               n_steps=4, max_iter=400,
                                               timeout=min(30.0, remaining() * 0.03))
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()
        
        # Phase 6: Binary search on the smoothed solution
        if remaining() > 40 and time.time() - phase_start < time_alloc * 0.9:
            h_b3, c5_b3 = binary_local_search_eff(best_h, n, rng,
                                                   max_trials=150000,
                                                   timeout=min(30.0, remaining() * 0.03),
                                                   stall_limit=15000)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = h_b3.copy()
        
        # Phase 7: Continuous polish
        if remaining() > 30:
            h_c, c5_c = continuous_polish(best_h, n, rng, max_trials=20000,
                                          timeout=min(20.0, remaining() * 0.02))
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()
        
        # Phase 8: SLSQP final polish
        if remaining() > 25:
            h_f, c5_f = slsqp_polish(best_h, n, max_iter=1500)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
        
        # Phase 9: Final binary search
        if remaining() > 30:
            h_b4, c5_b4 = binary_local_search_eff(best_h, n, rng,
                                                   max_trials=100000,
                                                   timeout=min(25.0, remaining() * 0.02),
                                                   stall_limit=8000)
            if c5_b4 < best_c5:
                best_c5 = c5_b4
                best_h = h_b4.copy()
        
        if best_c5 < best_overall_c5:
            best_overall_c5 = best_c5
            best_overall_h = best_h.copy()
    
    if best_overall_h is None:
        best_overall_h = thue_morse_init(256)
        best_overall_c5 = compute_c5(best_overall_h)
    
    best_overall_h = make_feasible(best_overall_h, len(best_overall_h))
    best_overall_c5 = compute_c5(best_overall_h)
    
    return (best_overall_h, best_overall_c5, len(best_overall_h))
