def all_lag_values(h):

def lag_gradient(h, L):
    """Gradient of lag L overlap integral w.r.t. h"""
    n = len(h)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        mL = abs(int(L))
        m = n - mL
        if L > 0:
            # integral h(x)(1-h(x+L)) dx
            # d/dh_i: for i < m: (1 - h[i+L]), for i >= L: -h[i-L]
            g[:m] += (1.0 - h[mL:])
            g[mL:] += -h[:m]
        else:
            # integral h(x)(1-h(x+L)) where L<0, so h(x)(1-h(x-|L|))
            # equivalent: sum h[i+|L|](1-h[i]) for i=0..m-1
            g[mL:] += (1.0 - h[:m])
            g[:m] += -h[mL:]
    return dx * g

def map_idx_to_lag(idx_arr, n):
    """Map correlation indices to lag numbers"""
    lags = []
    for i in idx_arr:
        i = int(i)
        if i == 0:
            lags.append(0)
        elif i % 2 == 1:
            lags.append(i // 2)  # positive lag
        else:
            lags.append(-(i // 2))  # negative lag
    return lags

        if time.time() < hard_end - 55:
            h2, c2 = refine_active_lags(h_init, tn, hard_end - 50, seq=(16, 32, 64))
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, tn

    # Phase 6: Final aggressive refinement at best resolution
    if time.time() < hard_end - 25 and 2 <= best_n <= 320:
        h2, c2 = refine_active_lags(best_h, best_n, hard_end - 12, seq=(32, 64, 128, 256))
        if c2 < best_c:
            best_h, best_c = h2, c2
        # Another smooth pass
        if time.time() < hard_end - 10:
            h3 = slsqp_smooth(best_h, best_n, 500, maxiter=40)
            c3 = compute_c5(h3)
            if c3 < best_c:
                best_h, best_c = h3, c3

    # Phase 7: Local search
    if time.time() < hard_end - 4 and 2 <= best_n <= 320:
        h2, c2 = local_search(best_h, best_n, hard_end - 1, seed + 4242, max_trials=50000)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
