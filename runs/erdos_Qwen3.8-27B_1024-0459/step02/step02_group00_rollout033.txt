    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 1 else 0.0
    return h


def make_golden_ratio(n):
    h = np.zeros(n)
    gr = (1.0 + np.sqrt(5.0)) / 2.0
    for i in range(n):
        frac = (i * gr) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    return h


def make_m_sequence(n, rng):
    """Generate a pseudo-random binary sequence with good autocorrelation."""
    # Use a simple LFSR-like construction
    h = np.zeros(n)
    # Start with golden ratio for positions
    gr = (1.0 + np.sqrt(5.0)) / 2.0

        h[i] = 1.0 if frac < 0.5 else 0.0
    # Fix count

    target = n // 2
    if ones > target:

        h[to_flip] = 1.0
    return h

def subgradient_descent(h, n, best_c5, max_iters=400, time_limit=60.0):

            n_pairs = n - offset
            grad[offset:offset + n_pairs] += (1.0 - h[:n_pairs])
            grad[:n_pairs] -= h[offset:offset + n_pairs]
        else:

            n_pairs = n - abs_off

            grad[abs_off:abs_off + n_pairs] -= h[:n_pairs]

        h, best_c5 = coordinated_flip(h, n, best_c5, rng, max_sweeps=10, time_limit=15.0)
        best_h = h.copy()
        print("  Coord flip: C5={:.6f}".format(best_c5))
    
    # Pair swap
    if remaining_time() > 15:
        h, best_c5 = pair_swap_search(h, n, best_c5, rng, max_sweeps=10, time_limit=25.0)
        best_h = h.copy()
        print("  Pair swap: C5={:.6f}".format(best_c5))
    
    # Phase 3: Try at higher resolution (if the solution improved at lower res)
    for n2 in [best_n + 32, best_n + 64, best_n + 128]:
        if n2 > 512 or remaining_time() < 25:
            break
        if n2 % 2 != 0:
            n2 += 1
        _N = n2
        _TGT = n2 / 2.0
        h_interp = interp_to(best_h, best_n, n2)
        h0 = project_feasible(h_interp, n2)
        c5_cur = compute_c5(h0)
        if c5_cur < best_c5 + 0.01:
            h_opt, c5_opt = slsqp_optimize(h0, n2, [50, 200, 1000], maxiter=200)
            if c5_opt < c5_cur:
                c5_cur = c5_opt
                h0 = h_opt
            # Local search
            if remaining_time() > 15:
                h0, c5_cur = subgradient_descent(h0, n2, c5_cur, max_iters=100, time_limit=15.0)
            if remaining_time() > 10:
                h_bin2 = np.where(h0 > 0.5, 1.0, 0.0)
                h_bin2 = project_feasible(h_bin2, n2)
                c5_b2 = compute_c5(h_bin2)
                if c5_b2 <= c5_cur + 0.003:
                    h_bin2, c5_cur = binary_local_search(h_bin2, n2, min(c5_cur, c5_b2), rng, max_sweeps=15, time_limit=10.0)
                    h0 = h_bin2
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n2
                print("  Higher res n={}: C5={:.6f}".format(n2, best_c5))

    # Phase 4: Final aggressive local search
    n = best_n
    _N = n
    _TGT = n / 2.0
    h = best_h.copy()
    
    # Multiple rounds of binary search
    if remaining_time() > 15:
        h_bin = np.where(h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, n)
        h_bin, best_c5 = binary_local_search(h_bin, n, best_c5, rng, max_sweeps=40, time_limit=30.0)
        if compute_c5(h_bin) <= best_c5 + 1e-10:
            best_h = h_bin.copy()
            h = h_bin.copy()
    
    # Final SLSQP polish
    if remaining_time() > 10:
        h_opt, c5_opt = slsqp_optimize(h, n, [500, 2000, 5000], maxiter=300)
        if c5_opt < best_c5 - 1e-10:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final coordinated flip
    if remaining_time() > 8:
        h, best_c5 = coordinated_flip(best_h, best_n, best_c5, rng, max_sweeps=5, time_limit=5.0)
        best_h = h.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
