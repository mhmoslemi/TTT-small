```python
def compute_grad_softmax(h, p):
    """p is the softmax weights over lags, length 2n-1"""
    n = len(h)
    grad = np.zeros(n)
    # For each lag k, the gradient contribution is:
    # (1-h)[j-k] for 0<=j-k<n, minus h[j+k] for 0<=j+k<n
    # Weighted by p[k]
    
    # Term 1: for each j, sum_k p[k] * (1-h)[j-k] * [0<=j-k<n]
    # This is a convolution of (1-h) with p (reversed appropriately)
    
    # Let me just do it with shifts
    one_minus_h = 1.0 - h
    for m in range(len(p)):
        if p[m] < 1e-15:
            continue
        k = m - (n - 1)
        # Term 1: (1-h)[j-k] for j in [max(0,k), min(n-1, n-1+k)]
        # i.e., j-k in [0, n-1], so j in [k, k+n-1] intersect [0, n-1]
        j_start = max(0, k)
        j_end = min(n, n + k)
        if j_start < j_end:
            grad[j_start:j_end] += p[m] * one_minus_h[0:j_end-j_start]
        # Term 2: -h[j+k] for j in [max(0,-k), min(n-1, n-1-k)]
        j_start2 = max(0, -k)
        j_end2 = min(n, n - k)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= p[m] * h[k+j_start2:k+j_end2]
    return grad
```

This is O(n * num_active_lags) which is fine if num_active_lags is small (at high β, only a few lags are active).

For the correlation computation, I'll use np.correlate which is O(n²) but fast in practice for n up to a few hundred.

def adam_optimize(h, n, beta, max_iters=2000, lr=0.1):
    h = h.copy().astype(float)
    target = n / 2.0
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        # Compute softmax weights
        corr = compute_corr(h)
        cmax = np.max(corr)
        exp_corr = np.exp(beta * (corr - cmax))
        p = exp_corr / np.sum(exp_corr)
        
        # Compute gradient
        grad = compute_grad_softmax(h, p)
        
        # Add constraint gradient (sum(h) = n/2)
        # Use projection instead
        
        # Adam update
        t = it + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)

                h_new = np.clip(h_new, 0, 1)
        
        h = h_new
        c5 = compute_c5(h)

            best_c5 = c5

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def corr_grad_softmax(h, p):
    """Compute gradient of softmax-smoothed max correlation.
    p: softmax weights over lags, length 2n-1
    Returns: gradient vector of length n
    """
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for m in range(len(p)):
        pm = p[m]
        if pm < 1e-14:
            continue
        k = m - (n - 1)
        # Term 1: (1-h)[j-k] for valid j
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            grad[js:je] += pm * one_minus_h[:je - js]
        # Term 2: -h[j+k] for valid j
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            grad[js2:je2] -= pm * h[k + js2:k + je2]
    return grad

def adam_step(h, grad, m, v, t, lr, b1, b2, target_sum):
    """One Adam step with projection. Returns updated h, m, v."""
    m = b1 * m + (1.0 - b1) * grad
    v = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    h_new = h - step
    h_new = np.clip(h_new, 0.0, 1.0)
    # Fix sum constraint
    diff = target_sum - np.sum(h_new)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_new < 1.0 - 1e-10
        else:
            mask = h_new > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            adj = diff / cnt
            h_new[mask] = np.clip(h_new[mask] + adj, 0.0, 1.0)
    return h_new, m, v

def adam_optimize(h, n, beta, max_iters, lr_init, time_limit):
    """Adam optimizer with softmax-smoothed max correlation objective."""
    h = h.copy().astype(float)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        t += 1
        # Decay learning rate
        lr = lr_init * (0.9995 ** it)
        
        # Compute correlation and softmax
        corr = compute_corr(h)
        cmax = np.max(corr)
        shifted = beta * (corr - cmax)
        # Clip for numerical stability
        shifted = np.clip(shifted, -500, 500)
        exp_corr = np.exp(shifted)
        p = exp_corr / np.sum(exp_corr)
        
        # Compute gradient
        grad = corr_grad_softmax(h, p)
        
        # Adam step
        h, m, v = adam_step(h, grad, m, v, t, lr, 0.9, 0.999, target)
        
        # Evaluate
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def directed_local_search(h, n, rng, n_iterations, time_limit):
    """Local search targeting the worst lag."""
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    target = n / 2.0
    
    for it in range(n_iterations):
        if time.time() > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = np.max(corr)
        worst_lags = np.where(corr > c5_val - 1e-10)[0]
        
        improved = False
        for wm in worst_lags:
            k = wm - (n - 1)
            # Try to reduce corr at lag k
            # corr[k] = sum_i h[i+k]*(1-h[i])
            # To reduce: make h[i+k] smaller or h[i] larger at positions where both contribute
            # Or: make h[i+k] and h[i] both 1 (then h[i+k]*(1-h[i])=0)
            
            # Try small adjustments at pairs (i, i+k)
            for _ in range(5):
                i = rng.randint(max(0, -k), min(n, n - abs(k)))
                j = i + k
                if 0 <= i < n and 0 <= j < n and i != j:
                    # Try to increase h[i] and h[j] (makes product h[i]*h[j] larger,
                    # which reduces h[i+k]*(1-h[i]))
                    # Actually we want to reduce h[i+k]*(1-h[i])
                    # = increase h[i] (reduces (1-h[i])) or decrease h[j]
                    # Let's try: increase h[i], decrease h[j] slightly
                    delta = rng.uniform(0.001, 0.01)
                    if h[i] + delta <= 1.0 and h[j] - delta >= 0.0:
                        h_try = h.copy()
                        h_try[i] += delta
                        h_try[j] -= delta
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5 - 1e-14:
                            h = h_try
                            best_c5 = c5_try
                            improved = True
                            break
        
        if not improved:
            # Try random pair swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                h_try = h.copy()
                h_try[i], h_try[j] = h_try[j], h_try[i]
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
    
    return h, best_c5

def pair_swap_search(h, n, rng, n_rounds, time_limit):
    """Exhaustive-ish pair swap local search."""
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    
    for _ in range(n_rounds):
        if time.time() > time_limit:
            break
        improved = False
        # Sort indices by h value for structured search
        order = np.argsort(h)
        for ii in range(0, n, max(1, n // 200)):
            for jj in range(ii + 1, min(ii + 5, n)):
                i = order[ii]
                j = order[jj]
                if abs(h[i] - h[j]) < 1e-12:
                    continue
                # Swap values
                h_try = h.copy()
                h_try[[i, j]] = h_try[[j, i]]
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    
    return h, best_c5

def small_perturb_search(h, n, rng, n_trials, base_step, time_limit):
    """Random perturbations with constraint repair."""
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    target = n / 2.0
    step = base_step
    
    for _ in range(n_trials):
        if time.time() > time_limit:
            break
        h_try = h.copy()
        n_mod = rng.randint(1, 5)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = target - np.sum(h_try)
        mask = (h_try > 0.001) & (h_try < 0.999)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        # Fix sum exactly
        diff2 = target - np.sum(h_try)
        if abs(diff2) > 1e-10:
            mask2 = (h_try > 0.001) & (h_try < 0.999)
            cnt2 = np.count_nonzero(mask2)
            if cnt2 > 0:
                h_try[mask2] += diff2 / cnt2
                h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
        elif rng.random() < 0.05:
            step *= 0.9
    
    return h, best_c5

def binary_sa(n, rng, n_steps, t_start, t_end, time_limit):
    """Simulated annealing on binary configurations."""
    h = np.zeros(n)
    ones = set(rng.choice(n, n // 2, replace=False).tolist())
    for idx in ones:
        h[idx] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        if time.time() > time_limit:
            break
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        if T < 1e-6:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        
        # Swap
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        
        if delta < 0:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-delta / max(T, 1e-15)):
            pass  # accept
        else:
            h[i], h[j] = h[j], h[i]  # reject, undo
    
    return best_h, best_c5

def slsqp_run(h, n, max_iter, ftol, time_limit):
    """SLSQP optimization."""
    bounds = [(0.0, 1.0)] * n
    
    def obj_func(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    # Load parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    
    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5
    
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Adam optimization with increasing beta
    alphas = [(10, 500, 0.08), (30, 500, 0.05), (100, 400, 0.03), 
              (300, 300, 0.02), (1000, 300, 0.01), (3000, 200, 0.005),
              (10000, 200, 0.002)]
    for beta, max_it, lr in alphas:
        if time_left() < 30:
            break
        tl = time.time() + min(120, time_left() - 10)
        best_h, best_c5 = adam_optimize(best_h, best_n, beta, max_it, lr, tl)
    
    # Phase 2: Directed local search
    if time_left() > 30:
        n_iter = min(5000, int(time_left() * 30))
        best_h, best_c5 = directed_local_search(best_h, best_n, rng, n_iter, time.time() + time_left() - 5)
    
    # Phase 3: Pair swap search
    if time_left() > 20:
        n_rounds = min(30, int(time_left() / 2))
        best_h, best_c5 = pair_swap_search(best_h, best_n, rng, n_rounds, time.time() + time_left() - 5)
    
    # Phase 4: Small perturbation search
    if time_left() > 20:
        n_trials = min(20000, int(time_left() * 100))
        best_h, best_c5 = small_perturb_search(best_h, best_n, rng, n_trials, 0.03, time.time() + time_left() - 5)
