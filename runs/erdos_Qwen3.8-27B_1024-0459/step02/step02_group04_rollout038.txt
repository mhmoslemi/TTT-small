                    best_c = c_new

                else:
                    h = _project(h, n)
        except Exception:
            h = _project(h, n)

    best_h = _project(best_h, n)
    best_c = _c5(best_h)
    return best_h, best_c


def _active_set_polish(h0, n, deadline, max_lags=64):
    h = _project(h0, n)
    best_h = h.copy()
    best_c = _c5(h)
    dx = 2.0 / n

    for _ in range(4):
        if time.time() > deadline - 3.0:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        vals = corr * dx
        maxv = np.max(vals)
        thresh = maxv - 2e-5
        active_idx = np.where(vals >= thresh)[0]
        active_idx = active_idx[:max_lags]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(vals)])

        active_lags = (active_idx - (n - 1)).astype(int)

        def t_obj(x):
            h_x = x[:-1]
            t = x[-1]
            penalty = 0.0
            for ll in active_lags:
                L = abs(int(ll))
                if L == 0:
                    v = (np.sum(h_x) - np.dot(h_x, h_x)) * dx
                else:
                    m = n - L
                    ac = float(np.dot(h_x[:m], h_x[L:]))
                    if ll > 0:
                        v = (np.sum(h_x[L:]) - ac) * dx
                    else:
                        v = (np.sum(h_x[:m]) - ac) * dx
                if v > t:
                    penalty += (v - t) ** 2
            sum_pen = 100.0 * (np.sum(h_x) - 0.5 * n) ** 2
            return t + 10.0 * penalty + sum_pen

        def t_grad(x):
            h_x = x[:-1]
            n = len(h_x)
            dx = 2.0 / n
            g = np.zeros(n + 1)
            g[-1] = 1.0
            for ll in active_lags:
                L = abs(int(ll))
                if L == 0:
                    v = (np.sum(h_x) - np.dot(h_x, h_x)) * dx
                else:
                    m = n - L
                    ac = float(np.dot(h_x[:m], h_x[L:]))
                    if ll > 0:
                        v = (np.sum(h_x[L:]) - ac) * dx
                    else:
                        v = (np.sum(h_x[:m]) - ac) * dx
                if v > x[-1]:
                    diff = v - x[-1]
                    if L == 0:
                        gl = (1.0 - 2.0 * h_x) * dx
                    else:
                        m = n - L
                        gl = np.zeros(n)
                        gl[:m] = -h_x[L:]
                        gl[L:] = 1.0 - h_x[:m]
                        gl *= dx
                    g[:-1] += 20.0 * diff * gl
                    g[-1] -= 20.0 * diff
            g[:-1] += 200.0 * (np.sum(h_x) - 0.5 * n)
            return g

        x0 = np.concatenate([h, [best_c + 1e-6]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        try:
            res = minimize(t_obj, x0, jac=t_grad, method="L-BFGS-B",
                           bounds=bounds, options={"maxiter": 200, "ftol": 1e-15, "gtol": 1e-12})
            if res.x is not None and len(res.x) == n + 1 and np.all(np.isfinite(res.x)):
                h_new = _project(res.x[:-1], n)
                c_new = _c5(h_new)
                if c_new < best_c:
                    best_h = h_new.copy()
                    best_c = c_new
                    h = h_new
                else:
                    h = _project(h, n)
        except Exception:
            h = _project(h, n)

    best_h = _project(best_h, n)
    best_c = _c5(best_h)
    return best_h, best_c


def _local_search(h0, n, deadline, seed, n_trials=20000):
    rng = np.random.RandomState(int(seed))
    h = _project(h0, n)
    best_h = h.copy()
    best_c = _c5(h)
    cur_h = h.copy()
    cur_c = best_c
    temp = 0.001
    min_temp = 1e-7
    dx = 2.0 / n

    for trial in range(n_trials):
        if time.time() > deadline:
            break
        mode = rng.randint(3)
        h_try = cur_h.copy()

        if mode == 0:
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            delta = rng.randn() * 0.02
            h_try[i] += delta
            h_try[j] -= delta
        elif mode == 1:
            L = rng.randint(1, min(5, n))
            s = rng.randint(0, n - L)
            delta = rng.randn(L) * 0.015
            delta -= delta.mean()
            h_try[s:s+L] += delta
        else:
            i = rng.randint(n)
            delta = rng.uniform(-0.03, 0.03)
            h_try[i] += delta
            diff = -delta
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            adj = mask[np.argmin(np.abs(h_try[mask] - 0.5))]
            h_try[adj] += diff

        h_try = _project(h_try, n)
        c_try = _c5(h_try)

        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.01, temp * 1.1)
        elif c_try < cur_c:
            cur_h = h_try.copy()
            cur_c = c_try
        else:
            d = c_try - cur_c
            if d < temp:
                if rng.rand() < np.exp(-d / max(temp, 1e-10)):
                    cur_h = h_try.copy()
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)

    return best_h, best_c


def _block_optimize(h0, n, deadline, seed):
    rng = np.random.RandomState(int(seed))
    h = _project(h0, n)
    best_h = h.copy()
    best_c = _c5(h)
    dx = 2.0 / n

    for trial in range(5000):
        if time.time() > deadline:
            break
        L = int(rng.randint(2, min(10, n)))
        s = int(rng.randint(0, n - L))
        idx = np.arange(s, s + L)

        h_fixed = h.copy()
        h_rest_sum = 0.5 * n - np.sum(h_fixed[idx])

        sub_res = None
        for _ in range(3):
            def sub_obj(x):
                h_sub = h_fixed.copy()
                h_sub[idx] = x
                sum_pen = 100.0 * (np.sum(h_sub) - 0.5 * n) ** 2
                corr = np.correlate(h_sub, 1.0 - h_sub, mode="full")
                mx = np.max(corr) * dx
                return mx + sum_pen

            try:
                sub_res = minimize(sub_obj, h_fixed[idx].copy(), method="Nelder-Mead",
                                   options={"maxiter": 500, "xatol": 1e-9, "fatol": 1e-12})
                if sub_res.success:
                    break
            except Exception:
                break

        if sub_res is not None and sub_res.x is not None:
            h_try = h_fixed.copy()
            h_try[idx] = sub_res.x
            h_try = _project(h_try, n)
            c_try = _c5(h_try)
            if c_try < best_c:
                best_h = h_try.copy()
                best_c = c_try
                h = h_try

    return best_h, best_c


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    start_time = time.time()
    deadline = start_time + budget_s - 25.0

    parent = _get_parent()
    if parent is not None:
        pn = int(parent.size)
        best_h = _project(parent, pn)
        best_c = _c5(best_h)
        best_n = pn
    else:
        best_n = 160
        best_h = _project(np.full(best_n, 0.5), best_n)
        best_c = _c5(best_h)

    low_sols = []

    resolutions = sorted(set([best_n, 128, 144, 160, 176, 192, 208, 224, 240, 256, 288, 320]))
    resolutions = [rn for rn in resolutions if 64 <= rn <= 384]

    if time.time() > deadline - 120.0:
        best_h = _project(best_h, best_n)
        best_c = _c5(best_h)
        return (best_h, float(best_c), int(best_n))

    n_res = len(resolutions)
    base_time_per = (deadline - start_time - 120.0) / max(1, n_res)

    for ri, n in enumerate(resolutions):
        if time.time() > deadline - 80.0:
            break
        rem = deadline - 80.0 - time.time()
        if rem < 30.0:
            break
        slot = min(base_time_per * 1.5, rem)
        sub_deadline = time.time() + slot

        inits = []
        inits.append(_interp(best_h, n))
        inits.append(_project(np.full(n, 0.5), n))

        x = np.linspace(0.0, 2.0, n, endpoint=False)
        for freq in (1, 2, 3, 4, 5, 6, 7, 8):
            amp = 0.15 / freq
            phase = seed * 0.1 + freq * 0.7
            h_pat = 0.5 + amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
            inits.append(_project(h_pat, n))
        inits = inits[:4]

        for ii, h_init in enumerate(inits):
            if time.time() > sub_deadline - 5.0:
                break
            sub_sub = time.time() + max(8.0, (sub_deadline - 5.0 - time.time()) / max(1, len(inits) - ii))
            h_opt, c_opt = _optimize_lbfgs(h_init, n, sub_sub,
                                           beta_start=3.0, beta_end=200.0, mu=200.0)
            if c_opt < best_c:
                best_h = h_opt.copy()
                best_c = c_opt
                best_n = n

    if time.time() > deadline - 60.0:
        best_h = _project(best_h, best_n)
        best_c = _c5(best_h)
        return (best_h, float(best_c), int(best_n))

    polish_deadline = time.time() + min(60.0, deadline - 50.0)
    h_pol, c_pol = _active_set_polish(best_h, best_n, polish_deadline, max_lags=80)
    if c_pol < best_c:
        best_h = h_pol
        best_c = c_pol

    if time.time() > deadline - 50.0:
        best_h = _project(best_h, best_n)
        best_c = _c5(best_h)
        return (best_h, float(best_c), int(best_n))

    ls_deadline = time.time() + min(80.0, deadline - 30.0)
    h_ls, c_ls = _local_search(best_h, best_n, ls_deadline, seed=seed + 777, n_trials=50000)
    if c_ls < best_c:
        best_h = h_ls
        best_c = c_ls

    if time.time() < deadline - 15.0:
        h_bo, c_bo = _block_optimize(best_h, best_n, deadline - 5.0, seed=seed + 555)
        if c_bo < best_c:
            best_h = h_bo
            best_c = c_bo

    if time.time() < deadline - 5.0:
        h_pol2, c_pol2 = _active_set_polish(best_h, best_n, deadline - 1.0, max_lags=128)
        if c_pol2 < best_c:
            best_h = h_pol2
            best_c = c_pol2

    best_h = _project(best_h, best_n)
    best_c = _c5(best_h)
    return (best_h, float(best_c), int(best_n))
