import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_opt_j(h, j, n, corr):
    """Find optimal h[j] given all other h values, minimizing max correlation."""
    ells = np.arange(-(n - 1), n)
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += 1.0 - h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += -h[idx2[valid2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-15
        sg = np.mean(a[active])
        if sg > 1e-14:
            hi = mid
        elif sg < -1e-14:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def coord_descent(h, n, n_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h)
    start = time.time()
    stale = 0
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr)
            if abs(t_opt - h[j]) > 1e-14:
                h[j] = t_opt
                improved = True
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 20 and p > 10:
            break
    return best_h, best_c5


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def smooth_max_f_grad(x, n, p, dx):
    """Smooth max approximation: (sum (corr*k*dx)^p)^(1/p) and its gradient."""
    h = np.clip(x, 1e-15, 1.0 - 1e-15)
    corr = get_corr(h)
    g = np.maximum(corr * dx, 1e-25)
    F = np.sum(g ** p) ** (1.0 / p)
    S = np.sum(g ** p)
    scale = S ** (1.0 / p - 1.0)
    w = g ** (p - 1)

    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)

    for s in range(1, n):
        ws_pos = w[n - 1 + s]
        if ws_pos > 1e-20:
            grad[:n - s] += ws_pos * (1.0 - h[s:])
            grad[s:] += ws_pos * (-h[:n - s])
        ws_neg = w[n - 1 - s]
        if ws_neg > 1e-20:
            grad[s:] += ws_neg * (1.0 - h[:n - s])
            grad[:n - s] += ws_neg * (-h[s:])

    grad *= scale * dx
    return F, grad


def smooth_max_obj(x, n, p, dx):
    val, _ = smooth_max_f_grad(x, n, p, dx)
    return val


def smooth_max_grad(x, n, p, dx):
    _, g = smooth_max_f_grad(x, n, p, dx)
    return g


def slq_smooth_optimize(h, n, p, dx, max_iter=2000):
    """Use SLSQP on the smooth max approximation."""
    def obj(x):
        return smooth_max_obj(x, n, p, dx)

    def grad(x):
        return smooth_max_grad(x, n, p, dx)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.success or res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def lbfgs_smooth_optimize(h, n, p, dx, max_iter=1000, penalty=50.0):
    """Use L-BFGS-B on smooth max + penalty for sum constraint."""
    def obj_pen(x):
        val, _ = smooth_max_f_grad(x, n, p, dx)
        s = np.sum(x) - n / 2.0
        return val + penalty * s * s

    def grad_pen(x):
        _, g = smooth_max_f_grad(x, n, p, dx)
        s = np.sum(x) - n / 2.0
        return g + 2.0 * penalty * s * np.ones(n)

    try:
        res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16,
                                'gtol': 1e-14, 'maxfun': 5000})
        h_out = make_feasible(res.x, n)
        return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

        pass
    return None, None


def binary_local_search(h, n, max_trials=10000, time_limit=60):
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = n / 2.0 - s
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:int(deficit)]
        h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:int(-deficit)]
        h_bin[ones[order]] = 0.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    start = time.time()
    no_improve = 0
    for _ in range(max_trials):
        if time.time() - start > time_limit:
            break
        n_flip = rng.randint(1, 5)
        idxs = rng.choice(n, n_flip, replace=False)
        h_try = best_h.copy()
        for idx in idxs:
            h_try[idx] = 1.0 - h_try[idx]
        if abs(np.sum(h_try) - n / 2.0) < 1e-10:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                no_improve = 0
            else:
                no_improve += 1
        if no_improve > 5000:
            break
    return best_h, best_c5


def perturbation_search(h, n, rng, n_perturb=20, time_limit=30):
    """Add random perturbations to escape local minima, then coord descent."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for _ in range(n_perturb):
        if time.time() - start > time_limit:
            break
        h_pert = h.copy()
        n_change = rng.randint(3, 15)
        idxs = rng.choice(n, n_change, replace=False)
        h_pert[idxs] = rng.rand(n_change)
        h_pert = make_feasible(h_pert, n)
        h_cd, c5_cd = coord_descent(h_pert, n, n_passes=50,
                                     time_limit=min(10, time_limit - (time.time() - start)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    return best_h, best_c5


def generate_structured_init(n, rng):
    inits = []
    if n % 4 == 0:
        inits.append(np.tile([1.0, 1.0, 0.0, 0.0], n // 4))
        inits.append(np.tile([1.0, 0.0, 1.0, 0.0], n // 4))
    if n % 2 == 0:
        h3 = np.zeros(n)
        h3[:n // 4] = 1.0
        h3[n // 2:n // 2 + n // 4] = 1.0
        inits.append(h3)
    # Golden ratio based
    h4 = np.zeros(n)
    for i in range(n):
        h4[i] = 1.0 if (i * 0.618033988749895) % 1.0 < 0.5 else 0.0
    inits.append(make_feasible(h4, n))
    # Random
    inits.append(make_feasible(rng.rand(n) * 2.5, n))
    return inits


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except NameError:
        h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = make_feasible(h_parent.copy(), len(h_parent))
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent at parent resolution
    if remaining() > 80:
        tl = min(100, remaining() - 60)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=500, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: Try multiple resolutions with smooth optimization
    resolutions = [320, 480, 640]
    for n in resolutions:
        if remaining() < 80:
            break

        h_init = interpolate_to_n(best_h, n)
        dx = 2.0 / n

        # Coordinate descent
        tl_cd = min(60, remaining() * 0.25)
        if tl_cd > 10:
            h_cd, c5_cd = coord_descent(h_init, n, n_passes=200, time_limit=tl_cd)
        else:
            h_cd, c5_cd = h_init, compute_c5(h_init)

        # Smooth max SLSQP
        if remaining() > 50:
            h_sm, c5_sm = slq_smooth_optimize(h_cd, n, p=80, dx=dx, max_iter=1500)
            if h_sm is not None and c5_sm < c5_cd:
                h_cd = h_sm
                c5_cd = c5_sm

        # More coordinate descent after smooth opt
        if remaining() > 40:
            h_cd2, c5_cd2 = coord_descent(h_cd, n, n_passes=200,
                                          time_limit=min(40, remaining() - 30))
            if c5_cd2 < c5_cd:
                h_cd = h_cd2
                c5_cd = c5_cd2

        # L-BFGS-B with smooth max
        if remaining() > 40:
            h_lb, c5_lb = lbfgs_smooth_optimize(h_cd, n, p=100, dx=dx,
                                                 max_iter=500, penalty=100.0)
            if h_lb is not None and c5_lb < c5_cd:
                h_cd = h_lb
                c5_cd = c5_lb

        # Final coord descent
        if remaining() > 30:
            h_cd3, c5_cd3 = coord_descent(h_cd, n, n_passes=100,
                                          time_limit=min(25, remaining() - 20))
            if c5_cd3 < c5_cd:
                h_cd = h_cd3
                c5_cd = c5_cd3

        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n

    # Phase 3: Try even higher resolution from best
    for n in [800, 1000]:
        if remaining() < 60:
            break
        h_init = interpolate_to_n(best_h, n)
        h_cd, c5_cd = coord_descent(h_init, n, n_passes=150,
                                     time_limit=min(50, remaining() - 40))
        if remaining() > 30:
            h_lb, c5_lb = lbfgs_smooth_optimize(h_cd, n, p=80, dx=2.0/n,
                                                 max_iter=300, penalty=200.0)
            if h_lb is not None and c5_lb < c5_cd:
                h_cd = h_lb
                c5_cd = c5_lb
        if remaining() > 20:
            h_cd2, c5_cd2 = coord_descent(h_cd, n, n_passes=80,
                                          time_limit=min(15, remaining() - 10))
            if c5_cd2 < c5_cd:
                h_cd = h_cd2
                c5_cd = c5_cd2
        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n

    # Phase 4: Perturbation search at best resolution
    if remaining() > 60:
        h_pert, c5_pert = perturbation_search(best_h, best_n, rng,
                                              n_perturb=30,
                                              time_limit=min(50, remaining() - 40))
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()

    # Phase 5: Binary local search
    if remaining() > 40:
        h_bin, c5_bin = binary_local_search(best_h, best_n, max_trials=15000,
                                            time_limit=min(40, remaining() - 20))
        if c5_bin < best_c5 - 1e-13:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 6: SLSQP on true max
    if remaining() > 50:
        h_fin, c5_fin = slsqp_true_max(best_h, best_n, max_iter=3000)
        if h_fin is not None and c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 7: Final coordinate descent polish
    if remaining() > 20:
        tl = max(5, remaining() - 10)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=300, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 8: One more SLSQP if time
    if remaining() > 30:
        h_fin, c5_fin = slsqp_true_max(best_h, best_n, max_iter=2000)
        if h_fin is not None and c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
