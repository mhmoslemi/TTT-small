    e_arr = np.arange(-(n - 1), n)
    start = time.time()
    no_improve = 0
    
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            ge1 = j - e_arr
            ge2 = j + e_arr
            v1 = (ge1 >= 0) & (ge1 < n)
            v2 = (ge2 >= 0) & (ge2 < n)
            
            a = np.zeros(m)
            a[v1] += 1.0 - h[ge1[v1]]
            a[v2] += -h[ge2[v2]]
            
            b = corr - h[j] * a
            
            lo, hi = 0.0, 1.0
            for _ in range(35):
                mid = 0.5 * (lo + hi)
                vals = b + a * mid
                mx = vals.max()
                active = vals >= mx - 1e-13
                sg = a[active].mean()
                if sg > 1e-12:
                    hi = mid
                elif sg < -1e-12:
                    lo = mid
                else:
                    lo = hi = mid
                    break
            h[j] = 0.5 * (lo + hi)
        
        # Project to feasible set
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * dx
        if c5 < best_c5 - tol:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break
    
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def slsqp_polish(h, n, max_iter=3000):
    """SLSQP polish using smoothed max."""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        # Smooth max approximation
        beta = 50.0
        scaled = beta * corr / (2 * len(x))  # normalize
        scaled = scaled - scaled.max()
        return np.log(np.sum(np.exp(scaled))) / beta + corr.max() / (2 * len(x))
    
    # Actually let's just use the raw max
    def obj_max(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    con_fun = lambda x: np.sum(x) - n / 2.0
    
    try:
        res = minimize(obj_max, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def binary_local_search(h, n, max_trials=20000, time_limit=60.0):
    """Local search on binary patterns."""
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = n / 2.0 - s
    if deficit > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:int(deficit)]
        h_bin[zeros[order]] = 1.0
    elif deficit < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:int(-deficit)]
        h_bin[ones[order]] = 0.0
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    start = time.time()
    
    for _ in range(max_trials):
        if time.time() - start > time_limit:
            break
        # Pick a 1 and a 0, swap them
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_try = best_h.copy()
        h_try[i] = 0.0
        h_try[j] = 1.0
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    h_parent = None
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is not None and len(h_parent) == 0:
            h_parent = None
    except (NameError, Exception):
        h_parent = None
    
    if h_parent is not None:
        best_h = make_feasible(h_parent.copy(), len(h_parent))
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = make_feasible(rng.rand(best_n) * 2.0, best_n)
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: CD at parent resolution
    if remaining() > 100:
        tl = min(180, remaining() - 80)
        h_cd, c5_cd = coord_descent(best_h, best_n, n_passes=2000, time_limit=tl)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 2: ILS - perturb and refine
    for i in range(5):
        if remaining() < 60:
            break
        h_pert = best_h.copy()
        n_pert = best_n
        # Perturb random coordinates
        n_perturb = max(3, n_pert // 20)
        idxs = rng.choice(n_pert, n_perturb, replace=False)
        for idx in idxs:
            h_pert[idx] += rng.uniform(-0.1, 0.1)
        h_pert = make_feasible(h_pert, n_pert)
        tl = min(60, remaining() - 40)
        if tl < 10:
            break
        h_cd, c5_cd = coord_descent(h_pert, n_pert, n_passes=500, time_limit=tl)
        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 3: Multiple resolutions
    resolutions = sorted(set([best_n, 128, 192, 240, 256, 320]))
    for n in resolutions:
        if remaining() < 80:
            break
        if n == best_n:
            h_init = best_h.copy()
        else:
            h_init = interpolate_to_n(best_h, n)
        tl = min(80, remaining() - 60)
        if tl < 15:
            continue
        h_cd, c5_cd = coord_descent(h_init, n, n_passes=500, time_limit=tl)
        if c5_cd < best_c5 - 1e-13:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
    
    # Phase 4: Binary local search
    if remaining() > 60:
        h_bin, c5_bin = binary_local_search(best_h, best_n, max_trials=20000,
                                             time_limit=min(40, remaining() - 30))
        if c5_bin < best_c5 - 1e-13:
            # Polish the binary solution with CD
            tl = min(40, remaining() - 20)
            if tl > 10:
                h_cd, c5_cd = coord_descent(h_bin, best_n, n_passes=500, time_limit=tl)
                if c5_cd < best_c5 - 1e-13:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
    
    # Phase 5: SLSQP polish
    if remaining() > 60:
        h_slq, c5_slq = slsqp_polish(best_h, best_n, max_iter=3000)
        if h_slq is not None and c5_slq < best_c5 - 1e-13:
            best_c5 = c5_slq
