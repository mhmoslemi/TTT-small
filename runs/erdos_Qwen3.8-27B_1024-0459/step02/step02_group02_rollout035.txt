import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0: break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0: break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary preserving as many 1s as possible."""
    h = np.array(h, dtype=float)
    threshold = 0.5
    b = (h >= threshold).astype(float)
    # Adjust to have exactly n/2 ones
    target = n // 2
    ones = np.sum(b)
    if ones > target:
        # Turn some 1s to 0s (those closest to 0.5)
        idx = np.where(b == 1)[0]
        order = np.argsort(h[idx])[::-1]  # most uncertain first
        for i in range(int(ones - target)):
            b[idx[order[i]]] = 0
    elif ones < target:
        idx = np.where(b == 0)[0]
        order = np.argsort(h[idx])  # most uncertain (highest) first
        for i in range(int(target - ones)):
            b[idx[order[i]]] = 1
    return b

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    # Adjust to have n/2 ones
    target = n // 2
    ones = np.sum(h).astype(int)
    if ones != target:
        h = random_binary(n, np.random.RandomState(0))
    return h

def random_binary(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n//2]] = 1.0
    return h

    current_c5 = best_c5
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    temp = initial_temp
    no_improve = 0
    best_no_improve = 0
    start_t = time.time()
    
    while time.time() - start_t < max_time:
        if no_improve > 50000:
            # Perturb: make 5 random swaps
            for _ in range(5):
                if len(ones) > 0 and len(zeros) > 0:
                    i = rng.choice(list(ones))
                    j = rng.choice(list(zeros))
