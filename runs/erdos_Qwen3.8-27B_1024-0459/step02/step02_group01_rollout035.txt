import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Distribute among non-saturated variables
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not np.any(mask):
            mask = np.ones(n, dtype=bool)
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h, n, time_limit, verbose=False):
    """Projected subgradient descent for min max_k c_k"""
    t_start = time.time()
    h_cur = project_feasible(h, n).copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    
    # Adaptive step size
    step = 0.05
    momentum = np.zeros(n)
    beta_mom = 0.9
    
    patience = 0
    best_patient = best_c5
    
    while time.time() - t_start < time_limit:
        corr = compute_corr(h_cur)
        max_val = np.max(corr)
        
        # Active set: correlations within 1% of max
        threshold = max_val - 0.01 * max_val if max_val > 0 else max_val - 0.001
        active = corr >= threshold
        n_active = np.sum(active)
        if n_active == 0:
            active[np.argmax(corr)] = True
            n_active = 1
        
        # Compute subgradient
        grad = np.zeros(n)
        for lag_idx in np.where(active)[0]:
            m = lag_idx - (n - 1)  # the actual lag
            if m >= 0:
                # c[m] = sum_{i=0}^{n-1-m} h[i+m] * (1-h[i])
                # d/dh[j] = (1-h[j-m]) for j>=m (from h[j-m+m]=h[j])
                #           - h[j+m] for j+m < n (from 1-h[j])
                for j in range(m, n):
                    g = 0.0
                    if j - m >= 0 and j - m < n:  # always true since j >= m
                        g += (1.0 - h_cur[j - m])  # from h[j] as h[n+m] with n=j-m
                    if j + m >= 0 and j + m < n:
                        g -= h_cur[j + m]  # from h[j] as (1-h[n]) with n=j
                    grad[j] += g
            else:
                p = -m  # p > 0
                # c[-p] = sum_{i=p}^{n-1} h[i] * (1-h[i-p])
                # d/dh[j]: h[j] appears as h[i] when i=j (j>=p): contributes (1-h[j-p])
                #           h[j] appears as (1-h[i-p]) when i-p=j, i=j+p (j+p<n): contributes h[j+p]*(-1)
                for j in range(n):
                    g = 0.0
                    if j >= p and j < n:
                        g += (1.0 - h_cur[j - p])
                    if j + p < n:
                        g -= h_cur[j + p]
                    grad[j] += g
        
        grad /= n_active
        
        # Projected gradient step with momentum
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        h_new = h_cur - step * momentum
        
        # Project to feasible set
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            patience = 0
            # Increase step slightly
            step = min(step * 1.02, 0.3)
        else:
            h_cur = h_new  # still take the step for exploration
            patience += 1
            step *= 0.95
        
        if patience > 200:
            break
        
        beta_mom = min(beta_mom + 0.001, 0.99)
    
    return best_h, best_c5

    if c5_br < best_c5:
        best_c5 = c5_br
        best_h = h_br.copy()
    
    # Final: ensure feasibility and compute exact value
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
