def get_corr(h):

def make_feasible(h, n):

    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)

def sum_constraint_fn(x, n):
    return np.sum(x) - n / 2.0

def c5_obj(x):
    return np.max(np.correlate(x, 1.0 - x, mode='full'))

def slsqp_polish(h, n, max_iter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint_fn, 'args': (n,)}]
    try:
        res = minimize(c5_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16, 'maxls': 50})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None
