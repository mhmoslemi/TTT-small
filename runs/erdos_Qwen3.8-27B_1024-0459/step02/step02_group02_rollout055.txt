                        h_work[i] = min(1, h_work[i] + rng.uniform(0.01, 0.3))
                    else:
                        h_work[j] = max(0, h_work[j] - rng.uniform(0.01, 0.3))
                    h_work = make_feasible(h_work, n)
                    c5_new = compute_c5(h_work)
                    if c5_new < best_c5 - 1e-14:
                        h = h_work.copy()
                        best_c5 = c5_new
                        best_h = h.copy()
                        improved = True
                        break
        if not improved:
            # Random fractional perturb
            h_work = h.copy()
            n_mod = rng.randint(2, 6)
            idxs = rng.choice(n, n_mod, replace=False)
            h_work[idxs] += rng.uniform(-0.1, 0.1, n_mod)
            h_work = make_feasible(h_work, n)
            c5_new = compute_c5(h_work)
            if c5_new < best_c5:
                h = h_work.copy()
                best_c5 = c5_new
                best_h = h.copy()
  
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    n = 256

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None

    # Generate initial solutions
    inits = []
    if h_parent is not None:
        inits.append(h_parent.copy())
        inits.append(thue_morse_init(n))
        inits.append(random_binary(n, rng))
        inits.append(random_binary(n, rng))
    else:
        inits.append(thue_morse_init(n))
        for _ in range(4):
            inits.append(random_binary(n, rng))
    # Phase-shifted Thue-Morse
    for shift in [1, 2, 3, 5]:
        tm = np.zeros(n)
        for i in range(n):
            tm[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
        inits.append(make_feasible(tm, n))

    # Evaluate all inits
    best_h = inits[0].copy()
    best_c5 = compute_c5(best_h)
    for h0 in inits[1:]:
        c5 = compute_c5(h0)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()

    # Phase 1: SA from the best init (use most of the time)
    if remaining() > 150:
        p1_budget = min(300, remaining() - 100)
        h_sa, c5_sa = sa_optimize(best_h, n, rng, budget_s=p1_budget, t0=0.025, t1=0.001)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 2: SA from second-best init
    if remaining() > 100:
        # Sort inits by C5 and try second best
        init_scores = [(compute_c5(h0), h0) for h0 in inits]
        init_scores.sort(key=lambda x: x[0])
        h0 = init_scores[1][1]
        p2_budget = min(150, remaining() - 60)
        if p2_budget > 30:
            h_sa, c5_sa = sa_optimize(h0, n, rng, budget_s=p2_budget, t0=0.02, t1=0.001)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 3: SA with perturbed best (escape local minima)
    if remaining() > 80:
        h_pert = best_h + rng.uniform(-0.15, 0.15, n)
        h_pert = make_feasible(h_pert, n)
        p3_budget = min(150, remaining() - 50)
        if p3_budget > 30:
            h_sa, c5_sa = sa_optimize(h_pert, n, rng, budget_s=p3_budget, t0=0.03, t1=0.002)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 4: SA again from current best (lower temperature)
    if remaining() > 80:
        p4_budget = min(200, remaining() - 50)
        if p4_budget > 30:
            h_sa, c5_sa = sa_optimize(best_h, n, rng, budget_s=p4_budget, t0=0.015, t1=0.0005)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 5: Targeted continuous optimization
    if remaining() > 60:
        p5_budget = min(120, remaining() - 30)
        if p5_budget > 20:
            h_tc, c5_tc = targeted_continuous(best_h, n, rng, budget_s=p5_budget)
            if c5_tc < best_c5:
                best_c5 = c5_tc
                best_h = h_tc.copy()

    # Phase 6: SLSQP continuous polish
    if remaining() > 50:
        p6_budget = min(150, remaining() - 20)
        if p6_budget > 20:
            h_pol, c5_pol = continuous_polish(best_h, n, p6_budget)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 7: Another SA round on the polished solution
    if remaining() > 50:
        p7_budget = min(100, remaining() - 20)
        if p7_budget > 20:
            h_sa, c5_sa = sa_optimize(best_h, n, rng, budget_s=p7_budget, t0=0.01, t1=0.0003)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 8: Final polish
    if remaining() > 30:
        h_pol, c5_pol = continuous_polish(best_h, n, remaining() - 10)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final: one more targeted pass
    if remaining() > 20:
        h_tc, c5_tc = targeted_continuous(best_h, n, rng, budget_s=remaining() - 5)
        if c5_tc < best_c5:
            best_c5 = c5_tc
            best_h = h_tc.copy()

    # Final feasibility enforcement
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
