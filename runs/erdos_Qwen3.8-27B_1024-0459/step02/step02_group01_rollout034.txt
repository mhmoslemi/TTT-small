    Compute gradient of smooth_max(correlate(h, 1-h)) w.r.t. h.
    p is the softmax weights over the correlation lags.
    c has length 2n-1.
    """

    m = np.max(c)
    sh = _g_beta * (c - m)
    sh = np.clip(sh, -500.0, 0.0)

    beta = _g_beta

    sm = np.sum(exp_sh)
    p = exp_sh / sm

        w = p[m_idx]
