f(h) = (1/beta) * log(sum_k exp(beta * corr[k]))
df/dh[j] = sum_k w_k * d(corr[k])/dh[j]
where w_k = exp(beta*corr[k]) / sum_m exp(beta*corr[m])

This can be computed efficiently:
df/dh[j] = sum_k w_k * (1-h[j+l_k])*[0≤j+l_k<n] - sum_k w_k * h[j-l_k]*[0≤j-l_k<n]
where l_k = k-(n-1)

The first sum: for a fixed j, we sum over k where 0≤j+l_k<n, i.e., -j ≤ l_k < n-j.
The second sum: for a fixed j, we sum over k where 0≤j-l_k<n, i.e., j-n < l_k ≤ j.

This is still O(n) per j, so O(n²) total. For n=160, that's 25600 operations per gradient, which is very fast.

```python
def adam_step(param, grad, lr, beta1, beta2, eps, m_dict, v_dict, t):
    m_dict[param] = beta1 * m_dict.get(param, 0.0) + (1-beta1) * grad
    v_dict[param] = beta2 * v_dict.get(param, 0.0) + (1-beta2) * grad**2
    m_hat = m_dict[param] / (1 - beta1**t)
    v_hat = v_dict[param] / (1 - beta2**t)
    return param - lr * m_hat / (np.sqrt(v_hat) + eps)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    shifted = beta * corr - beta * np.max(corr)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Value of smooth max
    val = np.max(corr) + np.log(np.sum(weights)) / beta
    
    # Gradient
    grad = np.zeros(n)
    # l_k = k - (n-1) for k = 0, ..., 2n-2

    # First term: for each j, sum over k where 0≤j+l_k<n
    # l_k = k - n + 1, so j + l_k = j + k - n + 1
    # 0 ≤ j+k-n+1 < n means n-1-j ≤ k ≤ n-1+j
    # k ranges from 0 to 2n-2
    
    # Let me compute it differently. 
    # grad[j] = sum_k w_k * (1-h[j+l_k]) * [0≤j+l_k<n] 
    #          - sum_k w_k * h[j-l_k] * [0≤j-l_k<n]

    # So m ranges from 0 to n-1 always. But k = m-j+n-1 must be in [0, 2n-2].
    # k = m-j+n-1: for m=0, k=n-1-j (≥0 when j≤n-1 ✓); for m=n-1, k=2n-2-j (≤2n-2 when j≥0 ✓)
    # So all m in [0,n-1] give valid k.

    # Second sum: let m = j-l_k = j-k+n-1. Then k = j-m+n-1.
    # m ranges: 0≤m<n, k in [0,2n-2]: 0≤j-m+n-1≤2n-2
    # j-m+n-1≥0 always (since j≥0, m≤n-1, so j-m+n-1≥j-n+1+n-1=j≥0)
    # j-m+n-1≤2n-2: -m≤n-1-j, m≥j-n+1. Since m≥0 always and j-n+1≤0, this is always satisfied.
    # So all m in [0,n-1] give valid k.
    
    # Second sum = sum_{m=0}^{n-1} w_{j-m+n-1} * h[m]
    
    # So: grad[j] = sum_{m=0}^{n-1} w_{m-j+n-1}*(1-h[m]) - sum_{m=0}^{n-1} w_{j-m+n-1}*h[m]
    #           = conv(w, 1-h)[j] (with appropriate indexing) - conv(w_reversed, h)[j]
    
    # Actually this is a convolution! Let me verify.
    # First sum: sum_m w[m-j+n-1] * (1-h[m])
    # Let p = m-j, then m=p+j, and the sum is sum_p w[p+n-1] * (1-h[p+j]) for p+j in [0,n-1]
    # = sum_p w[n-1+p] * (1-h[p+j]) for -j ≤ p ≤ n-1-j
    # This is the cross-correlation of (1-h) with a windowed w.
    
    # This is getting complex to vectorize directly. Let me just do it with numpy operations.

            grad[idx] += w * (-h[idx - l])
    
    return val, grad
```

    # (1-h[j+l]) for j in [0,n): this is (1-h) shifted right by l (if l>0) or left by -l (if l<0)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            add = diff / cnt
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            add = diff / cnt
            h[mask] = np.maximum(h[mask] + add, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax
    shifted = beta * (corr - np.max(corr))
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    val = np.max(corr) + np.log(exp_shifted.sum()) / beta
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        l = k - n + 1
        # Term 1: (1-h[j+l]) for valid j
        js = max(0, -l)
        je = min(n, n - l)
        if js < je:
            grad[js:je] += w * (1.0 - h[js+l:je+l])
        # Term 2: -h[j-l] for valid j
        js2 = max(0, l)
        je2 = min(n, l + n)
        if js2 < je2:
            grad[js2:je2] += w * (-h[js2-l:je2-l])
    
    return val, grad

def adam_smooth_max(h_init, beta_schedule, iters, lr, time_start, time_limit):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta in beta_schedule:
        for it in range(iters):
            if time.time() - time_start > time_limit:
                return best_h, best_c5
            val, grad = smooth_max_and_grad(h, beta)
            t += 1
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * grad * grad
            m_hat = m / (1 - b1**t)
            v_hat = v / (1 - b2**t)
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = make_feasible(h, n)
            if it % 200 == 0:
                c5 = compute_c5(h)
                if c5 < best_c5:

        # Reset momentum at beta transitions

        v = np.zeros(n)
        t = 0
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5
