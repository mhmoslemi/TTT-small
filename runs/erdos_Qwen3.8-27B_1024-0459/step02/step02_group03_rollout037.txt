```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_overlap_counts(h):
    """Return the raw correlation sums (not multiplied by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(np.maximum(h, 1.0 - h)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h_round = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_round))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_round == 1.0)[0]
        vals = h[ones]
        order = np.argsort(vals)[:n_ones - n_want]
        h_round[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_round == 0.0)[0]
        vals = h[zeros]
        order = np.argsort(-vals)[:n_want - n_ones]
        h_round[zeros[order]] = 1.0
    return h_round

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def run_smooth_opt(h0, n, beta, max_iter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, beta) * dx
    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-14:
                continue
            L = k - (n - 1)
            if L == 0:
                grad += w * (1.0 - 2.0 * x)
            elif L > 0:
                grad[L:] += w * (1.0 - x[:n - L])
                grad[:n - L] -= w * x[L:]
            else:
                Lp = -L
                grad[:n - Lp] += w * (1.0 - x[Lp:])
                grad[Lp:] -= w * x[:n - Lp]
        return dx * grad
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
    return project_feasible(res.x, n)

def binary_local_search(h, n, max_iter, t_start, t_end, rng, verbose=False):
    """Simulated annealing on binary space."""
    dx = 2.0 / n
    h = h.copy()
    c5 = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
    best_h = h.copy()
    best_c5 = c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        T = t_start * (t_end / t_start) ** (it / max(1, max_iter - 1))
        
        # Choose a random swap
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        a = ones[i]
        b = zeros[j]
        
        h[a] = 0
        h[b] = 1
        ones[i] = b
        zeros[j] = a
        
        new_corr = np.correlate(h, 1-h, mode='full')
        new_c5 = float(np.max(new_corr) * dx)
        
        delta = new_c5 - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[a] = 1
            h[b] = 0
            ones[i] = a
            zeros[j] = b
    
    return best_h, best_c5
```

Better approach: **Guided local search**
1. Find the bottleneck shift L*
2. Generate all swaps that would reduce the overlap at L*
3. Rank them by how much they reduce the max
4. Accept the best one
5. If no improving swap, use simulated annealing

For step 2, I need to efficiently find swaps that reduce overlap at L*.

For shift L > 0, overlap(L) = sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])

A swap (a: 1→0, b: 0→1) changes overlap(L) by:
Δ = -[(1-h[a-L]) if L≤a≤n-1 else 0] + [h[a+L] if 0≤a≤n-1-L else 0]
  + [(1-h[b-L]) if L≤b≤n-1 else 0] - [h[b+L] if 0≤b≤n-1-L else 0]

For a swap to reduce overlap(L), we need Δ < 0.

Looking at the terms:
- -(1-h[a-L]): this is ≤ 0, helps reduce (when a-L is in range and h[a-L]=0, it's -1)
- +h[a+L]: this is ≥ 0, hurts (when a+L is in range and h[a+L]=1, it's +1)
- +(1-h[b-L]): this is ≥ 0, hurts (when b-L is in range and h[b-L]=0, it's +1)
- -h[b+L]: this is ≤ 0, helps reduce (when b+L is in range and h[b+L]=1, it's -1)

So the best swaps for reducing overlap(L) are:
- a such that h[a-L]=0 (maximizes the negative term) AND h[a+L]=0 (minimizes the positive term)
- b such that h[b-L]=1 (minimizes the positive term) AND h[b+L]=1 (maximizes the negative term)

The best possible Δ is -1 + 0 + 0 - 1 = -2 (if all conditions are met).

So I should look for:
- a: position of 1 where h[a-L]=0 (and preferably h[a+L]=0)
- b: position of 0 where h[b-L]=1 (and preferably h[b+L]=1)

This gives a very targeted search!

```python
def guided_binary_search(h, n, max_iter, rng, verbose=False):
    dx = 2.0 / n
    h = h.copy()
    c5 = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
    best_h = h.copy()
    best_c5 = c5
    no_improve = 0
    
    for it in range(max_iter):
        corr = np.correlate(h, 1-h, mode='full')
        max_idx = np.argmax(corr)
        L_star = max_idx - (n-1)
        if L_star < 0:
            L_star = -L_star  # by symmetry
        
        # Find best swap for this L
        best_delta = 0
        best_swap = None
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # For efficiency, sample a subset
        if len(ones) > 50:
            sample_ones = rng.choice(ones, size=50, replace=False)
        else:
            sample_ones = ones
        if len(zeros) > 50:
            sample_zeros = rng.choice(zeros, size=50, replace=False)
        else:
            sample_zeros = zeros
        
        L = L_star
        for a in sample_ones:
            # Compute partial delta for 'a'
            delta_a = 0.0
            if L <= a <= n-1:
                delta_a -= (1.0 - h[a-L])
            if 0 <= a <= n-1-L:
                delta_a += h[a+L]
            
            if delta_a >= 0:
                # This 'a' alone doesn't help, skip
                continue
            
            for b in sample_zeros:
                delta_b = 0.0
                if L <= b <= n-1:
                    delta_b += (1.0 - h[b-L])
                if 0 <= b <= n-1-L:
                    delta_b -= h[b+L]
                
                total_delta = delta_a + delta_b
                if total_delta < best_delta:
                    best_delta = total_delta
                    best_swap = (a, b)
        
        if best_swap is not None:
            a, b = best_swap
            h[a] = 0
            h[b] = 1
            new_c5 = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                c5 = new_c5
                no_improve = 0
            else:
                # Revert
                h[a] = 1
                h[b] = 0
                no_improve += 1
        else:
            # Random swap (annealing)
            a = rng.choice(ones)
            b = rng.choice(zeros)
            h[a] = 0
            h[b] = 1
            new_c5 = float(np.max(np.correlate(h, 1-h, mode='full')) * dx)
            if new_c5 <= c5:
                c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
            else:
                h[a] = 1
                h[b] = 0
                no_improve += 1
        
        if no_improve > 200:
            break
    
    return best_h, best_c5
```
