```python
def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    h_bin = (h >= 0.5).astype(float)
    s = np.sum(h_bin)
    target = n // 2
    if s > target:
        # Need to flip some 1s to 0s
        excess = int(s - target)
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with smallest h values (closest to 0.5)
        order = np.argsort(h[ones])[:excess]
        h_bin[ones[order]] = 0
    elif s < target:
        # Need to flip some 0s to 1s
        deficit = target - int(s)
        zeros = np.where(h_bin == 0)[0]
        # Flip the zeros with largest h values
        order = np.argsort(-h[zeros])[:deficit]
        h_bin[zeros[order]] = 1
    return h_bin
```

Also, the bisection logic has a subtle issue. The function f(t) = max_m C_m(t) is:
- Convex in t if all C_m(t) are convex in t
- C_m(t) for m≠n-1 is linear (hence both convex and concave)
- C_{n-1}(t) = c0 - old*(1-old) + t*(1-t) is concave (since t-t² is concave)
- So f(t) = max(linear functions, concave function) is NOT necessarily convex

This means the bisection/subgradient approach might not work perfectly. However, in practice, the concave term is usually not the binding constraint (it's at most 0.25), so the max is usually determined by the linear terms, and the approach works well.

When h[p] goes 1→0 and h[q] goes 0→1:
Δcorr[m] = -1*(1-h[p+L])*[valid] + h[p-L]*[valid] + 1*(1-h[q+L])*[valid] - h[q-L]*[valid]

where L = m-(n-1).

```python
def swap_delta(h, p, q, n):
    """Compute Δcorr when h[p]: 1→0 and h[q]: 0→1."""
    delta = np.zeros(2*n-1)
    
    # p going 1→0:
    # As first factor: -(1-h[p+L]) for 0≤p+L<n
    # L from -p to n-1-p, m from n-1-p to 2n-2-p
    s1, e1 = n-1-p, 2*n-2-p
    if 0 <= s1 and e1 < 2*n-1:
        delta[s1:e1+1] += -(1.0 - h[0:n-p])  # h[p+L] for L=-p to n-1-p = h[0] to h[n-1-p]
    
    # As second factor: +h[p-L] for 0≤p-L<n
    # L from p-n+1 to p, m from p to p+n-1... 
    # Wait: p-L ≥ 0 and p-L < n → 0 ≤ p-L < n → p-n < L ≤ p → L from p-n+1 to p
    # m = L + (n-1) from p-n+1+n-1 = p to p+n-1
    s2, e2 = p, p+n-1
    if 0 <= s2 and e2 < 2*n-1:
        delta[s2:e2+1] += h[n-1::-1]  # h[p-L] for L=p-n+1 to p = h[n-1] to h[0]
    
    # q going 0→1:
    # As first factor: +(1-h[q+L]) for 0≤q+L<n
    s3, e3 = n-1-q, 2*n-2-q
    if 0 <= s3 and e3 < 2*n-1:
        delta[s3:e3+1] += (1.0 - h[0:n-q])  # h[q+L] for L=-q to n-1-q = h[0] to h[n-1-q]
    
    # As second factor: -h[q-L] for 0≤q-L<n
    s4, e4 = q, q+n-1
    if 0 <= s4 and e4 < 2*n-1:
        delta[s4:e4+1] += -h[n-1::-1]  # h[q-L] for L=q-n+1 to q = h[n-1] to h[0]
    
    return delta
```
