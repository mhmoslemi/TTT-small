import numpy as np
from scipy.optimize import minimize

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:
            h[free] += res / free.size
            h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_erdos(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def upsample_h(parent, n):
    p = np.asarray(parent, dtype=float)
    n0 = p.size
    if n0 == n:
        return p.copy()
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.interp(x1, x0, p, left=float(p[0]), right=float(p[-1]))
    return project_density(h, 0.5 * n)

def compute_grad_matrix(h):
    """Compute the n x (2n-1) matrix where G[i, m] = ∂corr[m]/∂h[i]"""
    n = h.size
    G = np.zeros((n, 2*n-1))
    center = n - 1
    
    for m in range(2*n-1):
        s = m - center  # shift
        if s == 0:
            G[:, m] = 1.0 - 2.0 * h
        elif s > 0:
            # grad[i] = (1-h[i-s]) for i in [s, n-1] + (-h[i+s]) for i in [0, n-1-s]
            G[s:, m] += (1.0 - h[:n-s])
            G[:n-s, m] -= h[s:]
        else:
            j = -s
            # grad[i] = (1-h[i+j]) for i in [0, n-1-j] + (-h[i-j]) for i in [j, n-1]
            G[:n-j, m] += (1.0 - h[j:])
            G[j:, m] -= h[:n-j]
    
    return G

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)
    n = x.size
    target = float(target)
    if n == 0:
        return x
    s = float(np.sum(x))
    if abs(s - target) < 1e-12:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    res = target - float(np.sum(h))
    if abs(res) > 1e-10:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size > 0:

def compute_grad_matrix(h):
    n = h.size
    G = np.zeros((n, 2 * n - 1), dtype=np.float64)
    center = n - 1
    G[:, center] = 1.0 - 2.0 * h
    for m in range(2 * n - 1):
        if m == center:
            continue
        s = m - center
        if s > 0:
            G[s:, m] += (1.0 - h[:n - s])
            G[:n - s, m] -= h[s:]
        else:
            j = -s
            G[:n - j, m] += (1.0 - h[j:])
            G[j:, m] -= h[:n - j]
    return G

def gradient_descent_search(h, deadline, seed=42, K=10):
    n = h.size
    target = 0.5 * n
    h = project_density(h, target)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / float(n)
    best_c = float(np.max(corr)) * dx
    best_h = h.copy()
    rng = np.random.default_rng(seed)
    stall = 0

    while time.perf_counter() < deadline:
        top_idx = np.argsort(corr)[-K:]
        g = np.zeros(n)
        for ki in top_idx:
            k = int(ki)
            s = k - (n - 1)
            if s == 0:
                g += 1.0 - 2.0 * h
            elif s > 0:
                g[s:] += (1.0 - h[:n - s])
                g[:n - s] -= h[s:]
            else:
                j = -s
                g[:n - j] += (1.0 - h[j:])
                g[j:] -= h[:n - j]

        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-12:
            break
        d /= norm

        improved = False

def binarize_h(h, target, thresh_lo=0.05, thresh_hi=0.95):

def smooth_logsumexp_obj(x, n, alpha, target):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return 1e100
    return (2.0 / n) * (m + np.log(z) / alpha)


def smooth_logsumexp_jac(x, n, alpha, target):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    z = float(np.sum(e))
    if z <= 0 or not np.isfinite(z):
        return np.zeros(n)
    w = e / z
    g = np.zeros(n)
    center = n - 1
    idx = np.where(w > 1e-10)[0]
    for t in idx:
        t = int(t)
        p = float(w[t])
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += p * (1.0 - h[:n - k])
            g[:n - k] -= p * h[k:]
        else:
            j = -k
            g[:n - j] += p * (1.0 - h[j:])
            g[j:] -= p * h[:n - j]
    return (2.0 / n) * g


def slsqp_optimize(h, n, deadline, target, alphas=(20.0, 60.0, 200.0), maxiter=(100, 80, 60)):
    best_h = h.copy()
    best_c = evaluate_erdos(h)
    bounds = [(0.0, 1.0)] * n

    sum_c = float(np.sum(h) - target)

    def sum_fun(x):
        return float(np.sum(x) - target)

    def sum_jac(x):
        return np.ones(n)

    constraints = ({"type": "eq", "fun": sum_fun, "jac": sum_jac},)

    for ai, alpha in enumerate(alphas):
        if time.perf_counter() >= deadline - 5.0:
            break
        def obj_func(x, a=alpha):
            return smooth_logsumexp_obj(x, n, a, target)

        def obj_jac(x, a=alpha):
            return smooth_logsumexp_jac(x, n, a, target)

        try:
            res = minimize(
                obj_func, best_h,
                method="SLSQP", jac=obj_jac,
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter[ai], "ftol": 1e-10, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
        except Exception:
            pass

    return best_h, best_c

    else:
        freq = int(rng.integers(2, 8))
        h = 0.5 + 0.45 * np.sign(np.sin(2 * np.pi * freq * x / 2 + rng.uniform(0, 6.28)))
        h = 0.3 * h + 0.7 * 0.5
    return project_density(h, 0.5 * n)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start_time + max(1.0, budget - 10.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_
    else:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    resolutions = []
    base_n = best_n
    for rn in [base_n, int(base_n * 1.3), int(base_n * 1.6), int(base_n * 2)]:
        if 40 <= rn <= 1000 and rn not in resolutions:
            resolutions.append(rn)
    for extra in [120, 180, 240, 300, 450, 600, 750, 900]:
        if extra not in resolutions and 40 <= extra <= 1000:
            resolutions.append(extra)
    resolutions = sorted(set(resolutions))

    rem = deadline - time.perf_counter()
    n_res = len(resolutions)
    time_per_res = max(30.0, rem / (n_res + 2))

    for ri, ni in enumerate(resolutions):
        rem = deadline - time.perf_counter()
        if rem < 30.0:
            break
        if ni > 700 and rem < 60.0:
            continue
        if ni > 800 and rem < 45.0:
            continue

        alloc = min(time_per_res * (1.0 + 0.3 * (ni > base_n)), rem - 5.0)
        alloc = max(20.0, alloc)

        start_h = upsample_h(best_h, ni)
        start_c = evaluate_erdos(start_h)

        sub_deadline = time.perf_counter() + alloc

        h_local = start_h.copy()
        c_local = start_c

        rem_sub = sub_deadline - time.perf_counter()
        if rem_sub > 15.0:
            n_slsqp = ni
            slsqp_deadline = time.perf_counter() + rem_sub * 0.3
            if ni <= 200:
                alphas = (15.0, 50.0, 150.0)
                maxiters = (150, 100, 80)
            elif ni <= 450:
                alphas = (15.0, 50.0)
                maxiters = (100, 70)
            else:
                alphas = (20.0, 60.0)
                maxiters = (60, 40)
            h_tmp, c_tmp = slsqp_optimize(h_local, n_slsqp, slsqp_deadline, 0.5 * n_slsqp, alphas, maxiters)
            if c_tmp < c_local:
                h_local = h_tmp
                c_local = c_tmp

        rem_sub = sub_deadline - time.perf_counter()
        if rem_sub > 10.0:
            h_tmp, c_tmp = fast_pair_search(h_local, time.perf_counter() + rem_sub * 0.5, seed=seed + ni, K=min(20, ni // 25))
            if c_tmp < c_local:
                h_local = h_tmp
                c_local = c_tmp

        rem_sub = sub_deadline - time.perf_counter()
        if rem_sub > 8.0:
            h_tmp, c_tmp = gradient_descent_search(h_local, time.perf_counter() + rem_sub * 0.4, seed=seed + ni + 1000, K=min(10, ni // 50))
            if c_tmp < c_local:
                h_local = h_tmp
                c_local = c_tmp

        rem_sub = sub_deadline - time.perf_counter()
        if rem_sub > 8.0:
            h_tmp, c_tmp = fast_pair_search(h_local, time.perf_counter() + rem_sub * 0.6, seed=seed + ni + 2000, K=min(12, ni // 30))
            if c_tmp < c_local:
                h_local = h_tmp
                c_local = c_tmp

        h_local = project_density(h_local, 0.5 * ni)
        c_local = evaluate_erdos(h_local)

        if c_local < best_c:
            best_h = h_local.copy()
            best_c = c_local
            best_n = ni

    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 20.0:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)

        if nbest <= 600:
            h_bin = binarize_h(h_fin, target, 0.03, 0.97)
            c_bin = evaluate_erdos(h_bin)
            if c_bin < best_c:
                h_fin = h_bin
                best_c = c_bin
                best_h = h_bin.copy()

        rem2 = deadline - time.perf_counter()
        if rem2 > 15.0 and nbest <= 900:
            h2, c2 = fast_pair_search(h_fin, time.perf_counter() + rem2 * 0.3, seed=seed + 10000, K=min(20, nbest // 20))
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h2.copy()
                best_n = nbest

        rem2 = deadline - time.perf_counter()
        if rem2 > 15.0 and nbest <= 700:
            h2, c2 = gradient_descent_search(h_fin, time.perf_counter() + rem2 * 0.4, seed=seed + 20000, K=min(12, nbest // 40))
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h2.copy()
                best_n = nbest

        rem2 = deadline - time.perf_counter()
        if rem2 > 10.0 and nbest <= 500:
            h2, c2 = block_move_search(h_fin, time.perf_counter() + rem2 * 0.5, seed=seed + 30000)
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h2.copy()
                best_n = nbest

        rem2 = deadline - time.perf_counter()
        if rem2 > 10.0:
            h2, c2 = fast_pair_search(h_fin, time.perf_counter() + rem2 * 0.5, seed=seed + 40000, K=min(15, nbest // 25))
            if c2 < best_c:
                h_fin = h2
                best_c = c2
                best_h = h2.copy()
                best_n = nbest

        rem2 = deadline - time.perf_counter()
        if rem2 > 8.0:
            h_bin2 = binarize_h(h_fin, 0.5 * int(h_fin.size), 0.02, 0.98)
            c_bin2 = evaluate_erdos(h_bin2)
            if c_bin2 < best_c:
                h_fin = h_bin2
                best_c = c_bin2
                best_h = h_bin2.copy()

    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15.0:
        n_up = int(best_h.size) * 2
        if n_up <= 1200:
            h_up = upsample_h(best_h, n_up)
            h_up2, c_up2 = fast_pair_search(h_up, time.perf_counter() + rem * 0.5, seed=seed + 50000, K=min(20, n_up // 25))
            if c_up2 < best_c:
                best_h = h_up2
                best_c = c_up2
                best_n = n_up

    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
