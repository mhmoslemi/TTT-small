            best_c = c
            best_h = h.copy()
    return best_h, best_c

def local_perturb(h0, n, time_limit, seed, step=0.015, max_trials=3000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h
    cur_c = best_c
    temp = 0.002
    min_temp = 1e-7
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        L = int(rng.randint(1, min(6, n) + 1))
        st = int(rng.randint(0, n - L + 1))
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[st:st + L] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / temp):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)

def make_variants(h, n, rng, count=3):
    base = project_feasible(h, n)
    out = [base.copy()]
    if n > 1:
        flip = base[::-1]
        out.append(project_feasible(0.5 * (base + flip), n))
    pert = base + rng.randn(n) * 0.04
    pert = pert - (float(np.sum(pert)) - 0.5 * n) / float(n)
    out.append(project_feasible(pert, n))
    if count is not None and len(out) > count:
        out = out[:count]
    return out

def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    hard_end = t0 + max(60.0, budget_s - 45.0)

    parent = get_parent_values()
    if parent is not None and parent.size > 0:
        pn = int(parent.size)
        best_h = project_feasible(parent, pn)
        best_c = compute_c5(best_h)
        best_n = pn
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: L-BFGS-B smooth optimization at best_n
    phase1_end = min(hard_end, t0 + 250.0)
    betas = [15, 50, 120, 300]
    for beta in betas:
        if time.time() > phase1_end - 10.0:
            break
        h2, c2 = optimize_lbfgs(best_h, best_n, float(beta), maxiter=100, ftol=1e-12, time_limit=phase1_end)
        if c2 < best_c - 1e-12:
            best_h = h2
            best_c = c2
        else:
            h = project_feasible(best_h, best_n)
            h2, c2 = optimize_lbfgs(h, best_n, float(beta), maxiter=80, ftol=1e-12, time_limit=phase1_end)
            if c2 < best_c - 1e-12:
                best_h = h2
                best_c = c2

    # Phase 2: Active-set SLSQP at best_n
    phase2_end = min(hard_end, t0 + 400.0)
    if time.time() < phase2_end - 20.0:
        h2, c2 = refine_active_multi(best_h, best_n, phase2_end, seq=(16, 32, 64, 96), maxiter=55)
        if c2 < best_c - 1e-12:
            best_h = h2
            best_c = c2
            best_n = best_n

    # Phase 3: Try other resolutions
    other_ns = [200, 240, 192, 224]
    if best_n != 160:
        other_ns = [160] + [x for x in other_ns if x != best_n]
    phase3_end = min(hard_end - 200.0, t0 + 750.0)
    for nn in other_ns:
        nn = int(nn)
        if time.time() > phase3_end - 60.0:
            break
        if not (40 <= nn <= 320):
            continue
        h_interp = interpolate_to_n(best_h, nn)
        h_interp = project_feasible(h_interp, nn)
        c_interp = compute_c5(h_interp)
        if c_interp > best_c + 0.02:
            continue
        heter_end = phase3_end
        h2, c2 = optimize_lbfgs(h_interp, nn, 50.0, maxiter=80, ftol=1e-12, time_limit=heter_end - 30.0)
        if c2 < best_c - 1e-10:
            best_h = h2
            best_c = c2
            best_n = nn
            h2, c2 = refine_active_multi(best_h, nn, heter_end, seq=(16, 32, 64), maxiter=45)
            if c2 < best_c - 1e-10:
                best_h = h2
                best_c = c2
                best_n = nn

    # Phase 4: Intensive active-set refinement on best
    phase4_end = min(hard_end - 80.0, t0 + 1100.0)
    if time.time() < phase4_end - 10.0:
        h2, c2 = refine_active_multi(best_h, best_n, phase4_end, seq=(32, 64, 128, 160), maxiter=70)
        if c2 < best_c - 1e-12:
            best_h = h2
            best_c = c2

    # Phase 5: Coordinate descent polish
    phase5_end = min(hard_end - 40.0, t0 + 1200.0)
    if time.time() < phase5_end - 5.0 and best_n <= 250:
        h2, c2 = coord_descent_polish(best_h, best_n, phase5_end, n_sweeps=10)
        if c2 < best_c - 1e-12:
            best_h = h2
            best_c = c2

    # Phase 6: Local perturbation
    phase6_end = hard_end - 5.0
    if time.time() < phase6_end - 5.0 and best_n <= 250:
        h2, c2 = local_perturb(best_h, best_n, phase6_end, seed + 7777, step=0.012, max_trials=2000)
        if c2 < best_c - 1e-12:
            best_h = h2
            best_c = c2

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)

    # Verify integral constraint
    s = float(np.sum(best_h))
    target = 0.5 * float(best_n)
    if abs(s - target) > 1e-9:
        diff = target - s
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / float(np.sum(mask))
            best_h = np.clip(best_h, 0.0, 1.0)
        best_c = compute_c5(best_h)

    return (best_h, float(best_c), int(best_n))
