        if not improved:
            for _b in range(6):
                blen = int(rng.integers(1, max_block + 1))
                i1 = int(rng.integers(0, n - blen + 1))
                i2 = int(rng.integers(0, n - blen + 1))
                if abs(i1 - i2) < blen:
                    continue
                cand = best.copy()
                tmp = cand[i1:i1 + blen].copy()
                cand[i1:i1 + blen] = cand[i2:i2 + blen]
                cand[i2:i2 + blen] = tmp
                if np.all(cand >= -1e-12) and np.all(cand <= 1 + 1e-12):
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        improved = True
                        break
        # Random perturb
        if not improved:
            for _r in range(8):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                if c < best_c - 1e-13:
                    best = cand
                    best_c = c
                    improved = True
                    break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
    return project_density(best, target), evaluate_erdos(best)


def lag_targeted_move(h, deadline, target, max_iter=1500):
    """Focus on the single worst lag and try to reduce it directly."""
    n = h.size
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    center = n - 1
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        worst_t = int(np.argmax(corr))
        worst_k = worst_t - center
        # For this lag, find which pairs contribute most
        if worst_k == 0:
            contrib = best * (1.0 - best)
        elif worst_k > 0:
            contrib = best[:n - worst_k] * (1.0 - best[worst_k:])
            # reduce by increasing best[worst_k:] where best[:n-k] is high, or decreasing best[:n-k]
            # Effect of d on h[i] (i in overlap): changes contrib by d*(1-h[i+k]) for h[i] increase
            #                         and changes by -d*h[i] for h[i+k] increase (i.e., h[j] for j in [k,n-1])
            # We want to reduce: increase h[j] where j in [k, n-1] and h[j-k] is high
            # Or decrease h[i] where i in [0, n-1-k] and h[i+k] is low
        else:
            lag = -worst_k
            contrib = best[lag:] * (1.0 - best[:n - lag])
        
        improved = False
        if worst_k > 0:
            k = worst_k
            # Increase h[j] for j in [k, n-1] where h[j-k] is high
            boost_targets = np.argsort(-best[:n - k])[:20]
            for jj in boost_targets:
                jj = int(jj) + k
                ii = jj - k
                cap = min(1.0 - best[jj], best[ii])
                if cap < 1e-10:
                    continue
                for d in (0.5 * cap, 0.8 * cap, cap):
                    cand = best.copy()
                    cand[jj] += d
                    cand[ii] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        improved = True
                        break
                if improved:
                    break
        elif worst_k < 0:
            lag = -worst_k
            boost_targets = np.argsort(-best[lag:])[:20]
            for jj in boost_targets:
                jj = int(jj) + lag
                ii = jj - lag
                cap = min(1.0 - best[jj], best[ii])
                if cap < 1e-10:
                    continue
                for d in (0.5 * cap, 0.8 * cap, cap):
                    cand = best.copy()
                    cand[jj] += d
                    cand[ii] -= d
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        improved = True
                        break
                if improved:
                    break
        else:
            # Lag 0: reduce sum h*(1-h), i.e., push h toward 0 or 1
            mid = np.argsort(-np.abs(h - 0.5))[:20]
            for ii in mid:
                ii = int(ii)
                if best[ii] < 0.5:
                    target_idx = np.random.default_rng(ii).integers(n)
                    cap = min(0.5 - best[ii], best[target_idx])
                else:
                    target_idx = np.random.default_rng(ii + 1).integers(n)
                    cap = min(best[ii] - 0.5, 1.0 - best[target_idx])
                if cap < 1e-10:
                    continue
                for d in (0.5 * cap, cap):
                    if best[ii] < 0.5:
                        cand = best.copy()
                        cand[ii] += d
                        cand[target_idx] -= d
                    else:
                        cand = best.copy()
                        cand[ii] -= d
                        cand[target_idx] += d
                    if np.any(cand < -1e-12) or np.any(cand > 1 + 1e-12):
                        continue
                    c = evaluate_erdos(cand)
                    if c < best_c - 1e-13:
                        best = cand
                        best_c = c
                        improved = True
                        break
                if improved:
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
    return project_density(best, target), evaluate_erdos(best)


def binarize_optimal(h, target, thresh_lo=0.05, thresh_hi=0.95):
    """Try to push near-binary values to exact 0/1 while maintaining sum."""
    n = h.size
    h2 = h.copy()
    n_one = int(round(target))
    # Sort by value, set top n_one to 1, rest to 0
    order = np.argsort(h2)[::-1]
    h3 = np.zeros(n)
    for i in range(n_one):
        h3[order[i]] = 1.0
    if target != int(target):
        frac = target - int(target)
        h3[order[n_one - 1]] = frac
    c3 = evaluate_erdos(h3)
    # Also try threshold approach
    h4 = np.where(h2 > 0.5, 1.0, 0.0)
    s4 = float(np.sum(h4))
    if abs(s4 - target) < 1.0:
        h4 = project_density(h4, target)
    c4 = evaluate_erdos(h4)
    if c3 <= c4:
        return h3, c3
    return h4, c4


def make_random_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.2 * np.sin(2 * np.pi * 5 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        seg = max(1, int(n // 10))
        h = np.empty(n)
        for i in range(0, n, seg):
            h[i:min(i + seg, n)] = float(rng.uniform(0.1, 0.9))
    elif mode == 2:
        u = rng.random(n)
        h = np.where(u > 0.5, 0.8, 0.2)
        h += rng.normal(0, 0.04, n)
    elif mode == 3:
        h = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 7 * x / 2 + rng.uniform(0, 6.28))
    else:
        h = np.full(n, 0.5)
        h += rng.normal(0, 0.3, n)
    return project_density(h, 0.5 * n)


def optimize_at_resolution(n, h0, deadline, seed=42):
    target = 0.5 * n
    global _GC_N, _GC_ALPHA, _GC_TARGET
    _GC_N = n
    _GC_TARGET = target

    best_h = project_density(h0, target)
    best_c = evaluate_erdos(best_h)

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_con, "jac": sum_con_jac},)

    # Phase 1: SLSQP with increasing alpha
    if n <= 200:
        alphas = (30.0, 80.0, 200.0, 500.0)
        maxiter_sq = 300
    elif n <= 500:
        alphas = (30.0, 80.0, 200.0)
        maxiter_sq = 150
    else:
        alphas = (20.0, 60.0, 150.0)
        maxiter_sq = 80

    for alpha in alphas:
        rem = deadline - time.perf_counter()
        if rem < 15.0:
            break
        _GC_ALPHA = alpha
        try:
            res = minimize(
                smooth_obj, best_h,
                method="SLSQP", jac=smooth_jac,
                bounds=bounds, constraints=constraints,
                options={"maxiter": maxiter_sq, "ftol": 1e-12, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand.copy()
                    best_c = c
        except Exception:
            pass

    # Phase 2: Directed descent
    rem = deadline - time.perf_counter()
    if rem > 20.0:
        alloc2 = min(rem * 0.35, 300.0)
        h2, c2 = directed_descent(best_h, time.perf_counter() + alloc2, target,
                                  max_iter=3000, top_k=10, temp=40.0)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 3: Lag-targeted moves
    rem = deadline - time.perf_counter()
    if rem > 15.0:
        alloc3 = min(rem * 0.3, 200.0)
        h3, c3 = lag_targeted_move(best_h, time.perf_counter() + alloc3, target,
                                   max_iter=2000)
        if c3 < best_c:
            best_h = h3
            best_c = c3

    # Phase 4: Block swap search
    rem = deadline - time.perf_counter()
    if rem > 15.0:
        alloc4 = min(rem * 0.35, 250.0)
        h4, c4 = block_swap_search(best_h, time.perf_counter() + alloc4, target,
                                   max_iter=5000, top_k=8, max_block=4, n_tries=15)
        if c4 < best_c:
            best_h = h4
            best_c = c4

    # Phase 5: Final directed descent with tighter params
    rem = deadline - time.perf_counter()
    if rem > 10.0:
        alloc5 = min(rem * 0.5, 150.0)
        h5, c5 = directed_descent(best_h, time.perf_counter() + alloc5, target,
                                  max_iter=2000, top_k=6, temp=20.0)
        if c5 < best_c:
            best_h = h5
            best_c = c5

    # Phase 6: Final swaps
    rem = deadline - time.perf_counter()
    if rem > 8.0:
        h6, c6 = block_swap_search(best_h, time.perf_counter() + rem * 0.8, target,
                                   max_iter=3000, top_k=6, max_block=3, n_tries=20)
        if c6 < best_c:
            best_h = h6
            best_c = c6

    return project_density(best_h, target), evaluate_erdos(best_h)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 25.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    # Load parent
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.clip(raw, 0.0, 1.0).astype(float, copy=True)
            parent = np.nan_to_num(parent, nan=0.5, posinf=1.0, neginf=0.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = parent.size
        pp = project_density(parent, 0.5 * np_)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Primary resolution: 2x parent (capped at 900)
    primary_n = min(900, max(90, 2 * best_n))
    # Also try parent's own resolution and a higher one

    # Build list of (n, time_fraction) pairs
    resolutions = []
    resolutions.append((primary_n, 0.45))
    if best_n != primary_n:
        resolutions.append((best_n, 0.15))
    higher_n = min(1200, 2 * primary_n)
    if higher_n > primary_n:
        resolutions.append((higher_n, 0.20))
    resolutions.append((primary_n, 0.20))  # Second pass at primary

    total_time = deadline - time.perf_counter()
    time_alloc = []
    for ni, frac in resolutions:
        time_alloc.append(max(60.0, total_time * frac))

    for (ni, frac), alloc in zip(resolutions, time_alloc):
        rem = deadline - time.perf_counter()
        if rem < 60.0:
            break
        alloc = min(alloc, rem - 20.0)
        if alloc < 40.0:
            break

        # Build starting points
        starts = []
        starts.append(upsample_h(best_h, ni))
        if parent is not None and parent.size != ni:
            starts.append(upsample_h(parent, ni))

        # Add 1-2 random starts for diversity
        n_rand = 2 if ni <= 600 else 1
        for mm in range(n_rand):
            starts.append(make_random_start(ni, rng, (mm + int(seed)) % 5))

        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * ni)
            dup = False
            for u in uniq:
                if np.linalg.norm(s - u) < 1e-5:
                    dup = True
                    break
            if not dup:
                uniq.append(s)

        per_start = alloc / len(uniq)
        per_start = max(20.0, per_start)

        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 25.0:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 10.0)
            try:
                h_out, c_out = optimize_at_resolution(ni, s, sub_dl, seed)
            except Exception:
                continue
            if h_out is not None:
                h_out = project_density(h_out, 0.5 * ni)
                c_out = evaluate_erdos(h_out)
                if c_out < best_c:
                    best_h = h_out.copy()
                    best_c = c_out
                    best_n = ni

    # Final intensive polish on the best solution
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 30.0:
        nbest = best_h.size
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        c_fin = evaluate_erdos(h_fin)

        # Try binarization
        if nbest <= 500:
            h_bin, c_bin = binarize_optimal(h_fin, target)
            if c_bin < c_fin:
                h_fin = h_bin
                c_fin = c_bin

        # Extended directed descent
        rem2 = deadline - time.perf_counter()
        if rem2 > 15.0 and nbest <= 900:
            alloc = min(rem2 * 0.4, 400.0)
            h2, c2 = directed_descent(h_fin, time.perf_counter() + alloc, target,
                                      max_iter=5000, top_k=6, temp=15.0)
            if c2 < c_fin:
                h_fin = h2
                c_fin = c2

        # Lag-targeted
        rem3 = deadline - time.perf_counter()
        if rem3 > 15.0:
            alloc3 = min(rem3 * 0.35, 300.0)
            h3, c3 = lag_targeted_move(h_fin, time.perf_counter() + alloc3, target, max_iter=3000)
            if c3 < c_fin:
                h_fin = h3
                c_fin = c3

        # Block swaps
        rem4 = deadline - time.perf_counter()
        if rem4 > 10.0:
            h4, c4 = block_swap_search(h_fin, time.perf_counter() + rem4 * 0.7, target,
                                       max_iter=5000, top_k=5, max_block=3, n_tries=20)
            if c4 < c_fin:
                h_fin = h4
                c_fin = c4

        # Final tiny steps
        rem5 = deadline - time.perf_counter()
        if rem5 > 5.0:
            h5, c5 = directed_descent(h_fin, time.perf_counter() + rem5 * 0.8, target,
                                      max_iter=2000, top_k=4, temp=10.0)
            if c5 < c_fin:
                h_fin = h5
                c_fin = c5

        if c_fin < best_c:
            best_h = h_fin.copy()
            best_c = c_fin
            best_n = nbest

    # Final validation
    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)

    # Time check
    elapsed = time.perf_counter() - start
    if elapsed > budget - 5.0:
        # We're close to timeout, just return what we have
        pass

    return final_h, float(final_c), int(final_h.size)
