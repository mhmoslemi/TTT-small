            grad[L:] -= w * x[:n-L]

            grad[Lp:] += w * (1.0 - x[:n-Lp])
            grad[:n-Lp] -= w * x[Lp:]
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_k(h, k, n):
    """Gradient of corr[k] w.r.t. h"""
    L = k - (n - 1)
    grad = np.zeros(n)
    if L == 0:
        grad = 1.0 - 2.0 * h
    elif L > 0:
        grad[:n-L] += (1.0 - h[L:])
        grad[L:] -= h[:n-L]
    else:
        Lp = -L
        grad[Lp:] += (1.0 - h[:n-Lp])
        grad[:n-Lp] -= h[Lp:]
    return grad

def smooth_max_obj(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(h, beta):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    grad = np.zeros(len(h))
    n = len(h)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        grad += w * grad_corr_k(h, k, n)
    return grad
