    g = np.zeros(n, dtype=float)
    for i in range(len(top_idx)):
        if w[i] < 1e-12:
            continue
        t = int(top_idx[i])
        k = t - center
        if k == 0:
            g += w[i] * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            p = w[i]
            if k > 0:
                g[lag:] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
            else:
                g[:n - lag] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
    return g

def gradient_descent_polish(h, deadline, max_iter=2000, top_frac=0.05, min_k=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        top_idx, corr, w = compute_active_weights(best, top_frac, min_k)
        g = gradient_from_lags(best, top_idx, w)
        d = -g
        d -= np.mean(d)
        norm = np.linalg.norm(d)
        if norm < 1e-15:
            break
        d /= norm
        improved = False
        steps = (0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2)
        for step in steps:
            cand = project_density(best + step * d, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-14:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
    return best, evaluate_erdos(project_density(best, target))

def pair_swap_search(h, deadline, max_iter=5000, top_frac=0.04, min_k=8, n_pos=12, n_neg=12):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(54321)
    stall = 0
    eval_count = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        if eval_count > 2000000:
            break
        top_idx, corr, w = compute_active_weights(best, top_frac, min_k)
        g = gradient_from_lags(best, top_idx, w)
        neg_cand = np.argsort(g)[:n_neg]
        pos_cand = np.argsort(-g)[:n_pos]
        found = False
        for ii in neg_cand:
            ii = int(ii)
            for jj in pos_cand:
                jj = int(jj)
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                for frac in (0.25, 0.5, 0.75, 1.0):
                    d = frac * cap
                    cand = best.copy()
                    cand[ii] += d
                    cand[jj] -= d
                    c = evaluate_erdos(cand)
                    eval_count += 1
                    if c < best_c - 1e-14:
                        best = cand
                        best_c = c
                        found = True
                        break
                if found:
                    break
            if found:
                break
        if not found:
            for _r in range(8):
                ii = int(rng.integers(n))
                jj = int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * 0.8 * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
                c = evaluate_erdos(cand)
                eval_count += 1
                if c < best_c - 1e-14:
                    best = cand
                    best_c = c
                    found = True
                    break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 60:
                break
    return best, evaluate_erdos(project_density(best, target))

def multi_point_perturb(h, deadline, max_iter=3000, top_frac=0.05, min_k=8):
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(99999)
    stall = 0
    for _ in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        top_idx, corr, w = compute_active_weights(best, top_frac, min_k)
        g = gradient_from_lags(best, top_idx, w)
        found = False
        for _r in range(15):
            k = int(rng.integers(2, min(5, n // 6) + 1))
            idxs = rng.choice(n, size=2 * k, replace=False)
            negs = idxs[:k]
            poss = idxs[k:]
            neg_vals = best[negs].sum()
            pos_caps = (1.0 - best[poss]).sum()
            d = min(neg_vals, pos_caps) * float(rng.uniform(0.3, 0.9))
            if d < 1e-10:
                continue
            cand = best.copy()
            for idx in negs:
                mv = d * best[idx] / neg_vals if neg_vals > 0 else 0
                cand[idx] -= mv
            for idx in poss:
                mv = d * (1.0 - best[idx]) / pos_caps if pos_caps > 0 else 0
                cand[idx] += mv
            if np.any(cand < -1e-12) or np.any(cand > 1 + 1e-12):
                continue
            c = evaluate_erdos(cand)
            if c < best_c - 1e-14:
                best = cand
                best_c = c
                found = True
                break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 40:
                break
    return best, evaluate_erdos(project_density(best, target))

def anneal_search(h, deadline, max_iter=5000, top_frac=0.05, min_k=8, T_init=0.002, T_min=1e-6):
    """Simulated annealing style: accept worse moves with probability exp(-delta/T)"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    rng = np.random.default_rng(31337)
    T = T_init
    cool = (T_min / T_init) ** (1.0 / max(1, max_iter))
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        T = max(T_min, T_init * (cool ** it))
        top_idx, corr, w = compute_active_weights(best, top_frac, min_k)
        g = gradient_from_lags(best, top_idx, w)
        found = False
        for _r in range(10):
            strategy = rng.random()
            if strategy < 0.5:
                # Gradient-guided pair swap
                neg_cand = np.argsort(g)[:8]
                pos_cand = np.argsort(-g)[:8]
                ii = int(neg_cand[int(rng.integers(8))])
                jj = int(pos_cand[int(rng.integers(8))])
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
            elif strategy < 0.8:
                # Random pair
                ii, jj = int(rng.integers(n)), int(rng.integers(n))
                if ii == jj:
                    continue
                cap = min(1.0 - best[ii], best[jj])
                if cap < 1e-10:
                    continue
                d = float(rng.random()) * cap
                cand = best.copy()
                cand[ii] += d
                cand[jj] -= d
            else:
                # Small random perturbation
                k = int(rng.integers(1, 4))
                idxs = rng.choice(n, size=2 * k, replace=False)
                negs, poss = idxs[:k], idxs[k:]
                d = float(rng.uniform(0.01, 0.08))
                cand = best.copy()
                for idx in negs:
                    cand[idx] = np.clip(cand[idx] - d, 0.0, 1.0)
                for idx in poss:
                    cand[idx] = np.clip(cand[idx] + d, 0.0, 1.0)
                cand = project_density(cand, target)
            c = evaluate_erdos(cand)
            delta = c - best_c
            if delta < -1e-14:
                best = cand
                best_c = c
                found = True
                break
            elif delta > 0 and rng.random() < math.exp(-delta / max(T, 1e-15)):
                best = cand
                best_c = c
                found = True
                break
        if found:
            stall = 0
        else:
            stall += 1
            if stall > 80:
                break
    return best, evaluate_erdos(project_density(best, target))

def optimize_at_resolution(h0, n, deadline):
    """Full optimization pipeline at fixed resolution n"""
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)
    
    # Phase A: Gradient descent
    rem = deadline - time.perf_counter()
    if rem > 15:
        tl = min(rem * 0.25, 60.0)
        h2, c2 = gradient_descent_polish(h, time.perf_counter() + tl, max_iter=1500, top_frac=0.06, min_k=6)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = gradient_descent_polish(best_h, time.perf_counter() + min(rem * 0.1, 30.0), max_iter=1000, top_frac=0.04, min_k=8)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    # Phase B: Pair swaps
    rem = deadline - time.perf_counter()
    if rem > 10:
        tl = min(rem * 0.3, 120.0)
        h2, c2 = pair_swap_search(best_h, time.perf_counter() + tl, max_iter=5000, top_frac=0.05, min_k=8, n_pos=10, n_neg=10)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    # Phase C: Multi-point perturbations
    rem = deadline - time.perf_counter()
    if rem > 8:
        h2, c2 = multi_point_perturb(best_h, time.perf_counter() + min(rem * 0.2, 40.0), max_iter=2000)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    # Phase D: Annealing
    rem = deadline - time.perf_counter()
    if rem > 10:
        h2, c2 = anneal_search(best_h, time.perf_counter() + min(rem * 0.25, 80.0), max_iter=4000, T_init=0.003)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    # Phase E: Final polish
    rem = deadline - time.perf_counter()
    if rem > 5:
        h2, c2 = gradient_descent_polish(best_h, time.perf_counter() + min(rem * 0.3, 30.0), max_iter=1000, top_frac=0.03, min_k=10)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = pair_swap_search(best_h, time.perf_counter() + min(rem * 0.3, 40.0), max_iter=3000, top_frac=0.03, min_k=10, n_pos=8, n_neg=8)
        if c2 < best_c:
            best_h, best_c = h2, c2
    
    return project_density(best_h, target), evaluate_erdos(project_density(best_h, target))

def smooth_max_objective(h, alpha=100.0):
    n = h.size
    if n <= 1:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return 1e100
    return (2.0 / float(n)) * (m + np.log(s) / alpha)

def slsqp_optimize(h0, n, deadline):
    target = 0.5 * n
    bounds = [(0.0, 1.0)] * n
    constraints = ({
        "type": "eq",
        "fun": lambda x: float(np.sum(x) - target),
        "jac": lambda x: np.ones(n)
    },)
    global_opts = {"maxiter": 100, "ftol": 1e-12, "disp": False}
    best_h = project_density(h0, target)
    best_c = evaluate_erdos(best_h)
    for alpha in (50.0, 200.0):
        rem = deadline - time.perf_counter()
        if rem < 15:
            break
        def obj(x):
            return smooth_max_objective(x, alpha)
        def obj_grad(x):
            return smooth_max_grad(x, alpha, n)
        try:
            res = minimize(obj, best_h, method="SLSQP", jac=obj_grad,
                          bounds=bounds, constraints=constraints, options=global_opts)
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
        except Exception:
            pass
    return best_h, best_c

def smooth_max_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(corr))
    y = alpha * (corr - m)
    y = np.clip(y, -700.0, 0.0)
    e = np.exp(y)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return np.zeros(n)
    w = e / s
    center = n - 1
    g = np.zeros(n, dtype=float)
    idx = np.where(w > 1e-10)[0]
    for t in idx:
        t = int(t)
        p = float(w[t])
        k = t - center
        if k == 0:
            g += p * (1.0 - 2.0 * h)
        else:
            lag = abs(k)
            if k > 0:
                g[lag:] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
            else:
                g[:n - lag] += p
                g[:n - lag] -= p * h[lag:]
                g[lag:] -= p * h[:n - lag]
    return (2.0 / float(n)) * g

def make_diverse_start(n, rng, mode):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    if mode == 0:
        h = 0.5 + 0.35 * np.sin(2 * np.pi * 3 * x / 2 + rng.uniform(0, 6.28))
        h += 0.15 * np.sin(2 * np.pi * 5 * x / 2 + rng.uniform(0, 6.28))
    elif mode == 1:
        h = np.empty(n, dtype=float)
        seg = max(2, int(n // 8))
        for i in range(0, n, seg):
            end = min(i + seg, n)
            h[i:end] = float(rng.uniform(0.1, 0.9))
    elif mode == 2:
        u = rng.random(n)
        med = float(np.median(u))
        h = np.where(u > med, 0.75, 0.25).astype(float)
        h += rng.normal(0, 0.05, n)
    else:
        h = 0.5 + 0.4 * np.sin(2 * np.pi * int(rng.integers(2, 7)) * x / 2 + rng.uniform(0, 6.28))
    return project_density(h, 0.5 * n)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(10.0, budget - 12.0)
    rng = np.random.default_rng(seed)
    
    best_h = None
    best_c = float("inf")
    best_n = 0
    
    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None
    
    if parent is not None:
        pt = 0.5 * parent.size
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = int(parent.size)
    else:
        n0 = 180
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0
    
    # Phase 1: Quick polish at parent resolution
    rem = deadline - time.perf_counter()
    if rem > 40 and best_n <= 600:
        pol_dl = time.perf_counter() + min(rem * 0.15, 50.0)
        h1, c1 = optimize_at_resolution(best_h, best_n, pol_dl)
        if c1 < best_c:
            best_h = h1
            best_c = c1
    
    # Phase 2: SLSQP at parent resolution (if time allows)
    rem = deadline - time.perf_counter()
    if rem > 50 and best_n <= 500:
        sl_dl = time.perf_counter() + min(rem * 0.15, 60.0)
        h1, c1 = slsqp_optimize(best_h, best_n, sl_dl)
        if c1 < best_c:
            best_h = h1
            best_c = c1
    
    # Phase 3: Upsample to higher resolutions and optimize
    target_resolutions = []
    pn = best_n
    for tn in [900, 1200, 1500]:
        if tn > pn:
            target_resolutions.append(tn)
    
    rem = deadline - time.perf_counter()
    for i, tn in enumerate(target_resolutions):
        rem = deadline - time.perf_counter()
        if rem < 45:
            break
        alloc = min(rem * (0.5 if i == 0 else 0.4), 180.0)
        alloc = max(40.0, alloc)
        
        # Upsample from best
        h_up = upsample_h(best_h, tn)
        c_up = evaluate_erdos(h_up)
        
        # Also try upsample from parent if different
        starts = [h_up]
        if parent is not None and parent.size != tn:
            starts.append(upsample_h(parent, tn))
        
        # Add a diverse random start
        starts.append(make_diverse_start(tn, rng, i % 4))
        
        # Deduplicate
        uniq = []
        for s in starts:
            s = project_density(s, 0.5 * tn)
            dup = False
            for u in uniq:
                if np.allclose(s, u, atol=1e-5):
                    dup = True
                    break
            if not dup:
                uniq.append(s)
        
        per_start = alloc / len(uniq)
        for s in uniq:
            rem2 = deadline - time.perf_counter()
            if rem2 < 20:
                break
            sub_dl = time.perf_counter() + min(per_start, rem2 - 5.0)
            h_opt, c_opt = optimize_at_resolution(s, tn, sub_dl)
            if c_opt < best_c:
                best_h = h_opt.copy()
                best_c = c_opt
                best_n = tn
    
    # Phase 4: Final intensive polish on the global best
    rem = deadline - time.perf_counter()
    if best_h is not None and rem > 15:
        nbest = int(best_h.size)
        target = 0.5 * nbest
        h_fin = project_density(best_h, target)
        c_fin = evaluate_erdos(h_fin)
        
        # Extended gradient polish
        if rem > 15:
            h2, c2 = gradient_descent_polish(h_fin, time.perf_counter() + min(rem * 0.2, 40.0),
                                             max_iter=3000, top_frac=0.03, min_k=10)
            if c2 < best_c:
                h_fin, best_c = h2, c2
        
        # Extended pair swap
        rem2 = deadline - time.perf_counter()
        if rem2 > 12:
            h3, c3 = pair_swap_search(h_fin, time.perf_counter() + min(rem2 * 0.3, 60.0),
                                      max_iter=8000, top_frac=0.03, min_k=10, n_pos=15, n_neg=15)
            if c3 < best_c:
                h_fin, best_c = h3, c3
        
        # Multi-point
        rem3 = deadline - time.perf_counter()
        if rem3 > 10:
            h4, c4 = multi_point_perturb(h_fin, time.perf_counter() + min(rem3 * 0.2, 30.0), max_iter=2000)
            if c4 < best_c:
                h_fin, best_c = h4, c4
        
        # Annealing
        rem4 = deadline - time.perf_counter()
        if rem4 > 10:
            h5, c5 = anneal_search(h_fin, time.perf_counter() + min(rem4 * 0.3, 60.0),
                                   max_iter=5000, T_init=0.001, T_min=1e-7)
            if c5 < best_c:
                h_fin, best_c = h5, c5
        
        # Final gradient + swap
        rem5 = deadline - time.perf_counter()
        if rem5 > 8:
            h6, c6 = gradient_descent_polish(h_fin, time.perf_counter() + min(rem5 * 0.4, 25.0),
                                             max_iter=2000, top_frac=0.02, min_k=12)
            if c6 < best_c:
                h_fin, best_c = h6, c6
            rem6 = deadline - time.perf_counter()
            if rem6 > 5:
                h7, c7 = pair_swap_search(h_fin, time.perf_counter() + min(rem6 * 0.7, 20.0),
                                          max_iter=5000, top_frac=0.02, min_k=12, n_pos=8, n_neg=8)
                if c7 < best_c:
                    h_fin, best_c = h7, c7
        
        best_h = h_fin
        best_n = nbest
    
    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45
    
    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
