```python
def compute_all_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    prefix = np.zeros(n+1)
    np.cumsum(h, out=prefix[1:])
    total = prefix[-1]
    
    # Autocorrelation via FFT
    nfft = 1
    while nfft < 2*n:
        nfft *= 2
    H = np.fft.fft(h, nfft)
    ac = np.real(np.fft.ifft(H * np.conj(H)))[:n]
    # ac[k] = sum_{i=0}^{n-k-1} h[i]*h[i+k] for k=0..n-1
    
    vals = np.empty(2*n-1)
    lags = np.empty(2*n-1, dtype=int)
    
    # Lag 0
    vals[0] = (total - ac[0]) * dx
    lags[0] = 0
    
    idx = 1
    for L in range(1, n):
        # Positive lag L: sum_{i=0}^{n-L-1} h[i]*(1-h[i+L])
        # = sum_{i=0}^{n-L-1} h[i] - sum_{i=0}^{n-L-1} h[i]*h[i+L]
        # = prefix[n-L] - ac[L]
        vals[idx] = (prefix[n-L] - ac[L]) * dx
        lags[idx] = L
        idx += 1
        # Negative lag -L: sum_{i=L}^{n-1} h[i]*(1-h[i-L])
        # = sum_{i=L}^{n-1} h[i] - sum_{i=L}^{n-1} h[i]*h[i-L]
        # = (total - prefix[L]) - ac[L]
        vals[idx] = (total - prefix[L] - ac[L]) * dx
        lags[idx] = -L
        idx += 1
    
    return vals, lags
```

```python
def all_lag_vals(h):
    n = h.size
    dx = 2.0 / float(n)
    prefix = np.empty(n + 1, dtype=float)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = float(prefix[-1])
    nfft = 1
    while nfft < 2 * n:
        nfft *= 2
    H = np.fft.fft(h, nfft)
    ac = np.real(np.fft.ifft(H * np.conj(H)))[:n]
    vals = np.empty(2 * n - 1, dtype=float)
    lags = np.empty(2 * n - 1, dtype=int)
    vals[0] = (total - ac[0]) * dx
    lags[0] = 0
    idx = 1
    for L in range(1, n):
        m = n - L
        vals[idx] = (prefix[m] - ac[L]) * dx
        lags[idx] = L
        idx += 1
        vals[idx] = (total - prefix[L] - ac[L]) * dx
        lags[idx] = -L
        idx += 1
    return vals, lags


def lag_grad(h, L):
    n = len(h)
    g = np.zeros(n, dtype=float)
    L = abs(int(L))

        g[:] = 1.0 - 2.0 * h
        return g

    if int(L) > 0:
        g[:m] += (1.0 - h[L:])
        g[L:] += -h[:m]
    else:
        g[L:] += (1.0 - h[:m])
        g[:m] += -h[L:]
    return g

def custom_projected_gradient(h0, n, time_limit, base_lr=0.005, momentum_pct=0.9):

        it += 1

        margin = 0.001
        active_mask = vals >= mx - margin
        n_active = int(np.sum(active_mask))
        if n_active == 0:
            active_mask[np.argmax(vals)] = True
            n_active = 1
        g = np.zeros(n, dtype=float)
        for i in range(active_mask.size):
            if active_mask[i]:
                g += lag_grad(h, lags[i])
        g /= float(n_active)
        g *= (2.0 / float(n))
        vel = momentum_pct * vel - lr * g
        h_new = h + vel
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)

            best_c = c_new

            vel *= 0.5
            h_new = h + vel
            h_new = project_feasible(h_new, n)
            h = h_new
        if it % sched_step == 0:
            lr = max(0.0001, lr * 0.95)
    return best_h, float(best_c)

def interpolate_to_n(h_src, n_dst):

    if n_src <= 1:

    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

                best_c = c2

        # Active set with many lags
        if time.time() < p3_end - 60:
            if best_n <= 200:
                seq = (50, 100, 200, 300)
            else:
                seq = (40, 80, 160)
            h2, c2 = refine_active_cycle(best_h, best_n, p3_end - 60, seq=seq, maxiter=80)
            if c2 < best_c:
                best_h = h2
                best_c = c2

        # Trust-constr attempt
        if time.time() < p3_end - 40:
            vals, lags = all_lag_vals(best_h)
            top = float(np.max(vals))
            near = np.where(vals >= top - 0.001)[0]
            active_tc = [int(lags[i]) for i in near[:80]]
            if len(active_tc) > 0:
                h2, c2 = optimize_active_tcnstr(best_h, active_tc, top + 1e-4, maxiter=120)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2

    # Phase 4: Custom PG + block perturb
    p4_end = hard_end - 10.0
    if time.time() < p4_end - 30:
        h2, c2 = custom_projected_gradient(best_h, best_n, p4_end - 30, base_lr=0.003)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    if time.time() < p4_end:
        h2, c2 = block_perturb(best_h, best_n, p4_end, base_step=0.008)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Final: one more active set pass
    if time.time() < hard_end - 3:
        vals, lags = all_lag_vals(best_h)
        top = float(np.max(vals))
        near = np.where(vals >= top - 0.0005)[0]
        active_fin = [int(lags[i]) for i in near[:min(60, near.size)]]
        if len(active_fin) > 0:
            h2, c2 = optimize_active_slsqp(best_h, active_fin, top + 1e-6, maxiter=40)
            if c2 < best_c:
                best_h = h2
                best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
```
