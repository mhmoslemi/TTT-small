import numpy as np

def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:
        n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    target = 0.5 * float(n)
    h = h[:n].copy()
    if h.size < n:
        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    elif h.size > n:
        h = h[:n]
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(50):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-13
        else:
            mask = out > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def lag_gradient(h, L):
    n = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    L = abs(int(L))
    if L >= n:
        return np.zeros(n, dtype=float)
    m = n - L
    g = np.zeros(n, dtype=float)
    g[:m] += (1.0 - h[L:])
    g[L:] += -h[:m]
    return g


def all_lag_values(h):
    n = len(h)
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    lags = np.arange(-(n - 1), n, dtype=int)
    return vals, lags


def choose_active_lags(vals, lags, max_count, margin=1e-5):
    if vals.size == 0:
        return [0]
    top = float(np.max(vals))
    if margin > 0:
        thr = top - margin
        idx = np.where(vals >= thr)[0]
    else:
        order = np.argsort(-vals)
        idx = order[:min(max_count, vals.size)]
    if idx.size == 0:
        idx = np.array([np.argmax(vals)])
    if idx.size > max_count * 2:
        order = np.argsort(-vals[idx])
        idx = idx[order[:max_count * 2]]
    return [int(lags[i]) for i in idx]


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    vals = []
    ac0 = float(np.dot(h, h))
    vals.append((total - ac0) * dx)
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals.append((pref[m] - ac) * dx)
        vals.append((total - pref[L] - ac) * dx)
    vals = np.array(vals)
    mval = float(np.max(vals))
    shifted = beta * (vals - mval)
    shifted = np.clip(shifted, -700.0, 0.0)
    r = float(np.sum(np.exp(shifted)))
    if r <= 0:
        return mval
    return mval + np.log(r) / beta


def smooth_obj_jac(h, beta):
    n = len(h)
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    vals = []
    lags_list = []
    ac0 = float(np.dot(h, h))
    vals.append((total - ac0) * dx)
    lags_list.append(0)
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals.append((pref[m] - ac) * dx)
        lags_list.append(L)
        vals.append((total - pref[L] - ac) * dx)
        lags_list.append(-L)
    vals = np.array(vals)
    mval = float(np.max(vals))
    shifted = beta * (vals - mval)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = float(np.sum(w))
    if w_sum <= 0:
        return np.zeros(n, dtype=float)
    w /= w_sum
    g = np.zeros(n, dtype=float)
    for i in range(len(w)):
        if w[i] > 1e-12:
            g += w[i] * dx * lag_gradient(h, lags_list[i])
    return g


def sum_eq_fun(h):
    return float(np.sum(h)) - 0.5 * len(h)


def sum_eq_jac(h):
    return np.ones(len(h), dtype=float)


def optimize_smooth_slqp(h0, n, beta, maxiter=60):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{"type": "eq", "fun": sum_eq_fun, "jac": sum_eq_jac}]
    try:
        res = minimize(
            smooth_obj, h, args=(beta,), jac=smooth_obj_jac,
            method="SLSQP", bounds=bounds, constraints=constraints,
            options={"maxiter": maxiter, "ftol": 1e-12, "disp": False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    return h, compute_c5(h)

def active_ineq_fun(x, lags):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    m = len(lags)
    out = np.empty(m, dtype=float)
    for i, L in enumerate(lags):
        g = lag_gradient(h, L)
        r_val = g @ h
        out[i] = float(x[-1]) - r_val * dx
    return out


def active_ineq_jac(x, lags):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    m = len(lags)
    J = np.zeros((m, n + 1), dtype=float)
    for i, L in enumerate(lags):
        g = lag_gradient(h, L)
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def optimize_active_slqp(h0, lags, t0, maxiter=80):

def optimize_lp_step(h0, lags, trust_radius=0.3):
    n = len(h0)
    vals, all_lags = all_lag_values(h0)
    current_max = float(np.max(vals))
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == 0.5 * n
    ]
    for L in lags:
        g = lag_gradient(h0, L)
        r_val = g @ h0
        c = g @ h0 - r_val
        constraints.append(g @ h_var - t_var <= c)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000, abort_on_infeasible=True)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return h0, current_max
    
    if h_var.value is None or t_var.value is None:
        return h0, current_max
    
    h_new = np.asarray(h_var.value, dtype=float).flatten()
    if h_new.size != n or not np.all(np.isfinite(h_new)):
        return h0, current_max
    
    step = h_new - h0
    step_norm = float(np.max(np.abs(step)))
    if step_norm > trust_radius:
        h_new = h0 + step * (trust_radius / step_norm)
    
    h_new = project_feasible(h_new, n)
    c_new = compute_c5(h_new)
    
    if c_new < current_max:
        return h_new, c_new
    return h0, current_max

def multi_active_refine(h0, n, time_limit, lags_seq=(10, 20, 40, 80), maxiter=60):

    for acnt in lags_seq:
        if time.time() > time_limit - 3.0:
            break

            if time.time() > time_limit - 3.0:
                break
            vals, lags = all_lag_values(h)
            curr_c = float(np.max(vals))
            if curr_c < best_c:
                best_c = curr_c
                best_h = h.copy()
            
            active = choose_active_lags(vals, lags, acnt, margin=1e-4)
            if len(active) == 0:

def improve_at_resolution(h0, n, time_limit, seed):

    best_h = h.copy()

            best_h = h2.copy()

            else:
                seq = (12, 24, 48)
            h2, c2 = multi_active_refine(best_h, best_n, hard_end - 5.0, lags_seq=seq, maxiter=70)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
        
        # LP polish
        if time.time() < hard_end - 8.0:
            h2, c2 = lp_refine_loop(best_h, best_n, hard_end - 3.0, max_rounds=15)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
