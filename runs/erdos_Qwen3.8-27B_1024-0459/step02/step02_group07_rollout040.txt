import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15

                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_opt_j(h, j, n, corr, ells):
    idx1 = j - ells
    idx2 = j + ells
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    a = np.zeros(2 * n - 1)
    if np.any(valid1):
        a[valid1] += -h[idx1[valid1]]
    if np.any(valid2):
        a[valid2] += 1.0 - h[idx2[valid2]]
    b = corr - h[j] * a
    lo = 0.0
    hi = 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        mx = np.max(vals)
        active = vals >= mx - 1e-15
        sg = np.mean(a[active])
        if sg > 1e-14:
            hi = mid
        elif sg < -1e-14:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def coord_descent(h, n, n_passes, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h).copy()
    ells = np.arange(-(n - 1), n)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr, ells)
            dh = t_opt - h[j]
            if abs(dh) > 1e-15:
                idx1 = j - ells
                idx2 = j + ells
                v1 = (idx1 >= 0) & (idx1 < n)
                v2 = (idx2 >= 0) & (idx2 < n)
                if np.any(v1):
                    corr[v1] += dh * (-h[idx1[v1]])
                if np.any(v2):
                    corr[v2] += dh * (1.0 - h[idx2[v2]])
            h[j] = t_opt
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def simulate_anneal_binary(h, n, time_limit):
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    deficit = n / 2.0 - s
    if deficit > 0.5:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:int(deficit) + 1]
        h_bin[zeros[order]] = 1.0
    elif deficit < -0.5:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:int(-deficit) + 1]
        h_bin[ones[order]] = 0.0
    h_bin = make_feasible(h_bin, n)
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(99)
    start = time.time()
    T = 0.05
    T_min = 0.001
    cool_rate = 0.9997
    corr = get_corr(h_bin)
    best_corr_max = np.max(corr)
    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i], h_bin[j] = 0.0, 1.0
        corr_new = get_corr(h_bin)
        new_max = np.max(corr_new)
        delta = new_max - best_corr_max
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            best_corr_max = new_max
            corr = corr_new
            h_bin_curr_max = new_max
            new_c5 = new_max * (2.0 / n)
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 1.0, 0.0
        T = max(T * cool_rate, T_min)
    return best_h, best_c5


def swap_local_search(h, n, time_limit):
    h_work = h.copy()
    best_c5 = compute_c5(h_work)
    best_h = h_work.copy()
    rng = np.random.RandomState(7)
    start = time.time()
    for it in range(5000000):
        if time.time() - start > time_limit:
            break
        i1 = rng.randint(n)
        i2 = rng.randint(n)
        if i1 == i2:
            continue
        old_i1, old_i2 = h_work[i1], h_work[i2]
        h_work[i1] = 1.0 - old_i1
        h_work[i2] = 1.0 - old_i2
        if abs(np.sum(h_work) - n / 2.0) > 1e-10:
            h_work[i1] = old_i1
            h_work[i2] = old_i2
            continue
        c5_try = compute_c5(h_work)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_work.copy()
        else:
            h_work[i1] = old_i1
            h_work[i2] = old_i2
    return best_h, best_c5


def smooth_max_obj(x, n, eps):
    corr = np.correlate(x, 1.0 - x, mode='full')
    smax = eps * np.log(np.sum(np.exp((corr - np.max(corr)) / eps))) + np.max(corr)
    return smax * (2.0 / n)


def slsqp_smooth(h, n, eps=0.005, max_iter=5000):
    def obj(x):
        return smooth_max_obj(x, n, eps)

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        shifted = corr - np.max(corr)
        weights = np.exp(shifted / eps)
        weights /= np.sum(weights)
        n_corr = len(corr)
        g = np.zeros(n)
        ells = np.arange(-(n - 1), n_corr)
        for m in range(n_corr):
            ell = int(ells[m])
            w = weights[m] * (2.0 / n) / eps
            js1 = max(0, ell)
            je1 = min(n, n + ell)
            if js1 < je1:
                g[js1:je1] += w * (1.0 - x[js1 - ell:je1 - ell])
            js2 = max(0, -ell)
            je2 = min(n, n - ell)
            if js2 < je2:
                g[js2:je2] += w * (-x[js2 + ell:je2 + ell])
        return g

    def cons_fun(x, nn):
        return np.sum(x) - nn / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            c5_out = compute_c5(h_out)
            if c5_out < compute_c5(h) + 0.01:
                return h_out, c5_out
    except Exception:
        pass
    return None, None


def momentum_coord_descent(h, n, n_passes, time_limit, momentum=0.3):
    h = h.copy()
    velocity = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = get_corr(h).copy()
    ells = np.arange(-(n - 1), n)
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        for j in range(n):
            t_opt = coord_opt_j(h, j, n, corr, ells)
            velocity[j] = momentum * velocity[j] + (1 - momentum) * (t_opt - h[j])
            h[j] = np.clip(h[j] + velocity[j], 0.0, 1.0)
        h = make_feasible(h, n)
        corr = get_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        else:
            velocity *= 0.5
    return best_h, best_c5


def generate_structured_init(n, rng):
