            # 70% random, 30% targeted
            if rng.rand() < 0.7:
                i = ones_list[rng.randint(len(ones_list))]
                j = zeros_list[rng.randint(len(zeros_list))]
            else:
                # Pick ones that are "exposed" at the worst lag
                # Position i contributes to corr at lag (i - j) if h[i]=1, h[j]=0
                # Just pick random but weighted
                i = ones_list[rng.randint(len(ones_list))]
                # Pick a zero that's at the worst lag offset from i
                target_pos = i + worst_lag
                if 0 <= target_pos < n and h[target_pos] == 0.0:
                    j = target_pos
                else:
                    j = zeros_list[rng.randint(len(zeros_list))]
        else:
            break
        
        if i == j:
            continue
            
        h[i] = 0.0
        h[j] = 1.0
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        c5_new = float(np.max(corr_new)) * dx
        
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            corr = corr_new.astype(float)
            ones_list.remove(i)
            zeros_list.append(i)
            zeros_list.remove(j)
            ones_list.append(j)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    
    return h, best_c5

def continuous_refine(h_init, n, rng, time_limit):
    """Small continuous perturbations around binary solution."""
    h = h_init.astype(float).copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    start_t = time.time()
    no_improve = 0
    stall_limit = 10000
    target = n / 2.0
    
    while time.time() - start_t < time_limit:
        if no_improve > stall_limit:
            break
        
        h_try = h.copy()
        n_mod = rng.randint(1, 4)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-0.3, 0.3)
        h_try = np.clip(h_try, 0.0, 1.0)
        
        # Fix integral constraint
        diff = target - np.sum(h_try)
        if abs(diff) > 1e-12:
            mask = (h_try > 1e-6) & (h_try < 1.0 - 1e-6)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            h = h_try
            best_c5 = c5_try
            no_improve = 0
        else:
            no_improve += 1
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global rng_global
    rng_global = np.random.RandomState(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent solution if available
    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        pass
    
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    # --- Phase 0: Quick evaluations of basic constructions ---
    for n_try in [256, 512, 1024]:
        h_tm = thue_morse_n(n_try)
        c5_tm = compute_c5(h_tm)
        if c5_tm < best_c5:
            best_c5 = c5_tm
            best_h = h_tm.copy()
            best_n = n_try
        h_rb = random_binary_n(n_try, rng)
        c5_rb = compute_c5(h_rb)
        if c5_rb < best_c5:
            best_c5 = c5_rb
            best_h = h_rb.copy()
            best_n = n_try
    
    # Evaluate parent
    if h_parent is not None:
        c5_parent = compute_c5(h_parent)
        if c5_parent < best_c5:
            best_c5 = c5_parent
            best_h = h_parent.copy()
            best_n = len(h_parent)
    
    if remaining() < 60:
        return (best_h, best_c5, best_n)
    
    # --- Phase 1: SA at n=512 (main workhorse) ---
    n_main = 512
    # Prepare initializations for n=512
    inits_512 = []
    inits_512.append(thue_morse_n(n_main))
    inits_512.append(random_binary_n(n_main, rng))
    if h_parent is not None:
        inits_512.append(upsample_binary(h_parent, n_main, rng))
    # Phase-shifted Thue-Morse
    for shift in [1, 3]:
        h_shift = np.zeros(n_main)
        for i in range(n_main):
            h_shift[i] = 1.0 if bin((i + shift) % n_main).count('1') % 2 == 0 else 0.0
        inits_512.append(h_shift)
    # Another random
    inits_512.append(random_binary_n(n_main, rng))
    
    # Run SA on best 3 initializations
    time_per_sa = min(180.0, remaining() / 3.0 - 20.0)
    if time_per_sa > 30:
        # Sort inits by initial C5
        inits_scored = [(compute_c5(h), h) for h in inits_512]
        inits_scored.sort(key=lambda x: x[0])
        
        for _, h_init in inits_scored[:3]:
            if remaining() < 60:
                break
            h_sa, c5_sa = sa_optimize(h_init, n_main, rng, time_per_sa, t_start=0.004, t_end=1e-8)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n_main
    
    if remaining() < 60:
        return (best_h, best_c5, best_n)
    
    # --- Phase 2: Local search polish at n=512 ---
    if best_n != n_main and best_n == 512:
        pass  # already at 512
    elif best_n != 512:
        # Upsample best to 512
        best_h_512 = upsample_binary(best_h, 512, rng) if best_n != 512 else best_h.copy()
        best_c5_512 = compute_c5(best_h_512)
        if best_c5_512 <= best_c5 + 0.01:
            best_h = best_h_512
            best_c5 = min(best_c5_512, best_c5)
            best_n = 512
    
    if best_n == 512 and remaining() > 40:
        h_ls, c5_ls = local_search(best_h, 512, rng, time_limit=min(80.0, remaining() - 30), stall_limit=40000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    if remaining() < 40:
        return (best_h, best_c5, best_n)
    
    # --- Phase 3: Targeted local search ---
    if best_n == 512 and remaining() > 40:
        h_ts, c5_ts = targeted_local_search(best_h, 512, rng, time_limit=min(60.0, remaining() - 30), stall_limit=20000)
        if c5_ts < best_c5:
            best_c5 = c5_ts
            best_h = h_ts.copy()
    
    if remaining() < 40:
        return (best_h, best_c5, best_n)
    
    # --- Phase 4: Try n=1024 ---
    if remaining() > 120:
        n_fine = 1024
        h_up = upsample_binary(best_h, n_fine, rng) if best_n != n_fine else best_h.copy()
        c5_up = compute_c5(h_up)
        h_tm1024 = thue_morse_n(n_fine)
        c5_tm1024 = compute_c5(h_tm1024)
        
        # Pick best starting point for n=1024 SA
        if c5_up < c5_tm1024:
            h_start_1024 = h_up
        else:
            h_start_1024 = h_tm1024
        
        time_sa_1024 = min(200.0, remaining() - 60.0)
        if time_sa_1024 > 50:
            h_sa1024, c5_sa1024 = sa_optimize(h_start_1024, n_fine, rng, time_sa_1024, t_start=0.003, t_end=1e-8)
            if c5_sa1024 < best_c5:
                best_c5 = c5_sa1024
                best_h = h_sa1024.copy()
                best_n = n_fine
    
    if remaining() < 30:
        return (make_feasible(best_h, best_n), compute_c5(make_feasible(best_h, best_n)), best_n)
    
    # --- Phase 5: Final local search polish ---
    if remaining() > 25:
        h_final, c5_final = local_search(best_h, best_n, rng, time_limit=min(50.0, remaining() - 15), stall_limit=30000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    if remaining() < 20:
        return (make_feasible(best_h, best_n), compute_c5(make_feasible(best_h, best_n)), best_n)
    
    # --- Phase 6: One more targeted search ---
    if remaining() > 25:
        h_t2, c5_t2 = targeted_local_search(best_h, best_n, rng, time_limit=min(40.0, remaining() - 15), stall_limit=15000)
        if c5_t2 < best_c5:
            best_c5 = c5_t2
            best_h = h_t2.copy()
    
    # Final: ensure feasibility
    best_h = make_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    return (best_h, final_c5, best_n)
