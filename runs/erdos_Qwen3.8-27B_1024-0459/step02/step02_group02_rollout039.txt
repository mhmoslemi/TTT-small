import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def get_parent(n_target):
    try:
        hp = np.array(initial_h_values, dtype=float)
        if hp is not None and len(hp) > 0:
            if len(hp) == n_target:
                return hp.copy()
            return interp_to_n(hp, n_target)
    except Exception:
        pass
    return None

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(h, n)

def random_binary_init(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        to_flip = rng.choice(ones, excess, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def smooth_max_fn(x, beta, n):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted = np.minimum(shifted, 50.0)
    return cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))

def gradient_optimize(h_init, n, betas=None, max_iter=800):
    if betas is None:
        betas = [15, 40, 100, 300, 800]
    h0 = make_feasible(h_init.copy(), n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        try:
            res = minimize(smooth_max_fn, h0, args=(beta, n), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15})
            candidate = make_feasible(res.x, n)
            c5 = compute_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
                h0 = candidate
        except Exception:
            pass
    return best_h, best_c5

def slsqp_optimize(h_init, n, beta, max_iter=500):
    h0 = make_feasible(h_init.copy(), n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_fn, h0, args=(beta, n), method='SLSQP',
                      bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14})
        candidate = make_feasible(res.x, n)
        c5 = compute_c5(candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate
    except Exception:
        pass
    return best_h, best_c5

def binary_local_search(h_init, n, rng, max_trials=500000, timeout=100.0, stall_limit=30000):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h < 0.5)[0])
    no_improve = 0
    start_t = time.time()
    for _ in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h.copy()
            ones_list.remove(i)
            zeros_list.append(i)
            zeros_list.remove(j)
            ones_list.append(j)
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return best_h, best_c5

def sim_anneal(h_init, n, rng, max_trials=300000, T_start=0.3, T_end=0.005, timeout=100.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    curr_c5 = best_c5
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h < 0.5)[0])
    start_t = time.time()
    for trial in range(max_trials):
        elapsed = time.time() - start_t
        if elapsed > timeout:
            break
        progress = min(elapsed / timeout, 1.0)
        T = T_start * (T_end / T_start) ** progress
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        delta = c5_try - curr_c5
        if delta < 0 or (T > 1e-12 and np.log(rng.rand()) > -delta / T):
            curr_c5 = c5_try
            ones_list.remove(i)
            zeros_list.append(i)
            zeros_list.remove(j)
            ones_list.append(j)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def focused_local_search(h_init, n, rng, max_trials=300000, timeout=100.0, stall_limit=20000):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    start_t = time.time()
    for _ in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_idx = int(np.argmax(corr))
        s = worst_idx - (n - 1)
        if s >= 0:
            bad = [j for j in range(n - s) if h[j] < 0.5 and h[j + s] > 0.5]
        else:
            bad = [j for j in range(-s, n) if h[j] > 0.5 and h[j + s] < 0.5]
        if not bad:
            no_improve += 1
            continue
        j = bad[rng.randint(len(bad))]
        if s >= 0:
            i = j + s
        else:
            i = j
        zeros_avail = [k for k in range(n) if h[k] < 0.5 and k != j]
        if not zeros_avail:
            no_improve += 1
            continue
        swap_to = zeros_avail[rng.randint(len(zeros_avail))]
        h[i] = 0.0
        h[swap_to] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h.copy()
            no_improve = 0
        else:
            h[i] = 1.0
            h[swap_to] = 0.0
            no_improve += 1
    return best_h, best_c5

def multi_pair_search(h_init, n, rng, max_trials=200000, timeout=80.0, stall_limit=15000):
    """Try swapping two pairs simultaneously for better moves."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    start_t = time.time()
    for _ in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        if i1 == i2:
            i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                no_improve += 1
                continue
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if j1 == j2:
            j2 = zeros[rng.randint(len(zeros))]
            if j1 == j2:
                no_improve += 1
                continue
        if i1 in (j1, j2) or i2 in (j1, j2):
            no_improve += 1
            continue
        h[i1] = 0.0
        h[i2] = 0.0
        h[j1] = 1.0
        h[j2] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h.copy()
            no_improve = 0
        else:
            h[i1] = 1.0
            h[i2] = 1.0
            h[j1] = 0.0
            h[j2] = 0.0
            no_improve += 1
    return best_h, best_c5

def continuous_polish(h_init, n, max_iter=3000):

def generate_inits(n, rng, h_parent=None):
    inits = []
    if h_parent is not None:
        h_p = make_feasible(interp_to_n(h_parent, n), n)
        inits.append(h_p)
    inits.append(thue_morse_init(n))
    for shift in [1, 3, 7]:
        tm = np.zeros(n)
        for i in range(n):
            tm[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
        inits.append(make_feasible(tm, n))
    for _ in range(6):
        inits.append(random_binary_init(n, rng))
    # Antisymmetric structure
    h_as = np.zeros(n)
    half = n // 2
    for i in range(half):
        h_as[i] = 1.0 if bin(i * 5 + 3).count('1') % 2 == 0 else 0.0
        h_as[i + half] = 1.0 - h_as[i]
    inits.append(make_feasible(h_as, n))
    return inits

def optimize_at_n(n, h_parent, rng, time_budget):
    """Optimize at a given resolution, respecting time budget."""
    start_t = time.time()
    best_h = None
    best_c5 = 1e10

    h_p = None
    if h_parent is not None:
        h_p = make_feasible(interp_to_n(h_parent, n), n)

    inits = generate_inits(n, rng, h_p)

    # Phase 1: Gradient optimization from each init
    for h0 in inits[:5]:
        if time.time() - start_t > time_budget * 0.3:
            break
        h_opt, c5_val = gradient_optimize(h0, n, betas=[20, 60, 200, 600], max_iter=600)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # Phase 2: SLSQP from best
    if best_h is not None and time.time() - start_t < time_budget * 0.45:
        h_s, c5_s = slsqp_optimize(best_h, n, beta=300, max_iter=1000)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 3: Binary local search (multiple rounds)
    if best_h is not None:
        for rnd in range(5):
            if time.time() - start_t > time_budget * 0.75:
                break
            timeout = min(40.0, time_budget * 0.15)
            h_b, c5_b = binary_local_search(best_h, n, rng, max_trials=500000,
                                           timeout=timeout, stall_limit=40000)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 4: Focused search
    if best_h is not None and time.time() - start_t < time_budget * 0.9:
        h_f, c5_f = focused_local_search(best_h, n, rng, max_trials=300000,
                                         timeout=min(40.0, time_budget * 0.12),
                                         stall_limit=20000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Simulated annealing
    if best_h is not None and time.time() - start_t < time_budget * 0.95:
        h_sa, c5_sa = sim_anneal(best_h, n, rng, max_trials=300000,
                                  T_start=0.2, T_end=0.003,
                                  timeout=min(50.0, time_budget * 0.15))
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Continuous polish
    if best_h is not None and time.time() - start_t < time_budget:
        h_pol, c5_pol = continuous_polish(best_h, n, max_iter=2000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if best_h is None:
        best_h = inits[0]
        best_c5 = compute_c5(best_h)

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    remaining = budget_s - 30  # Reserve 30s for safety

    # Get parent if available
    h_parent_256 = get_parent(256)

    # Strategy: Try multiple resolutions, allocate time based on expected value
    # n=256: quick refinement (200s)
    # n=384: medium effort (300s)
    # n=512: larger effort (350s)
    # Final: continuous polish of best (remaining)

    resolutions = [256, 384, 512]
    time_alloc = [200, 250, 350]

    best_overall_h = None
    best_overall_c5 = 1e10
    best_overall_n = 256

    # First, evaluate parent at its native resolution
    if h_parent_256 is not None:
        c5_parent = compute_c5(h_parent_256)
        best_overall_c5 = min(best_overall_c5, c5_parent)
        best_overall_h = h_parent_256.copy()
        best_overall_n = 256

    for n, t_alloc in zip(resolutions, time_alloc):
        elapsed = time.time() - start_time
        if elapsed > remaining:
            break
        t_left = min(t_alloc, remaining - elapsed)
        if t_left < 30:
            break

        h_prev = best_overall_h if best_overall_h is not None else None
        h_res, c5_res = optimize_at_n(n, h_prev, rng, t_left)
        if c5_res < best_overall_c5:
            best_overall_c5 = c5_res
            best_overall_h = h_res.copy()
            best_overall_n = n

    # Final: try to improve the best solution found with continuous optimization
    elapsed = time.time() - start_time
    if best_overall_h is not None and (budget_s - elapsed) > 30:
        t_final = budget_s - elapsed - 15
        h_pol, c5_pol = continuous_polish(best_overall_h, best_overall_n, max_iter=5000)
        if c5_pol < best_overall_c5:
            best_overall_c5 = c5_pol
            best_overall_h = h_pol.copy()

    # Also try a quick round of binary local search on the polished solution
    elapsed = time.time() - start_time
    if best_overall_h is not None and (budget_s - elapsed) > 20:
        h_b, c5_b = binary_local_search(best_overall_h, best_overall_n,

        if c5_b < best_overall_c5:
            best_overall_c5 = c5_b

    # Final feasibility check
    best_overall_h = make_feasible(best_overall_h, best_overall_n)
    best_overall_c5 = compute_c5(best_overall_h)

    return (best_overall_h, best_overall_c5, best_overall_n)
