        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_curr = np.max(corr) * dx
        
        if c5_curr < best_c5:

        for k in range(len(corr)):

        if time.time() - t0 > time_limit:
            return 1e10
        _, val = gradient_c5(x, beta)
        return val

    def obj_jac(x):
        if time.time() - t0 > time_limit:
            return np.zeros(n)
        g, _ = gradient_c5(x, beta)
        return g

    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_jac, bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def slsqp_direct(h, n, max_iter=5000, time_limit=90.0):
    t0 = time.time()
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    def obj(x):
        if time.time() - t0 > time_limit:
            return 1e10
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * dx

    def obj_jac(x):
        if time.time() - t0 > time_limit:
            return np.zeros(n)
        return grad_direct(x)

    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_jac, bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None


def pair_transfer_local(h, n, n_rounds=30, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        n_pairs = min(3000, n * n // 3)
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for amt in [0.005, 0.01, 0.02, 0.04]:
                h_try = h.copy()
                a_use = min(amt, h[i])
                if h[j] + a_use > 1.0:
                    a_use = max(0.0, 1.0 - h[j])
                if a_use < 1e-8:
                    continue
                h_try[i] -= a_use
                h_try[j] += a_use
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5


def random_perturb_improve(h, n, n_trials, base_step, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    for _ in range(n_trials):
        h_try = h.copy()
        n_mod = rng.randint(1, 6)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.005) & (h_try < 0.995)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
        elif rng.random() < 0.15:
            step *= 0.92
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    print(f"Initial: C5={best_c5:.6f}, n={best_n}")

    # Phase 1: Subgradient with varying beta at parent resolution
    for beta in [20, 60, 150, 400]:
        if remaining() < 30:
            break
        h_new, c5_new = subgradient_optimize(best_h, best_n, n_iters=2500,
                                              lr_start=0.02, lr_end=0.001,
                                              beta=beta, time_limit=min(40, remaining() * 0.15))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    print(f"After Phase 1: C5={best_c5:.6f}, t={time.time()-start_time:.1f}s")

    # Phase 2: SLSQP with analytical gradient (direct max)
    if remaining() > 60:
        h_new, c5_new = slsqp_direct(best_h, best_n, max_iter=5000, time_limit=min(80, remaining() * 0.5))
        if h_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    print(f"After Phase 2: C5={best_c5:.6f}, t={time.time()-start_time:.1f}s")

    # Phase 3: SLSQP with smoothed objective
    for beta in [40, 150, 500]:
        if remaining() < 30:
            break
        h_new, c5_new = slsqp_smooth(best_h, best_n, beta=beta,
                                     max_iter=2500, time_limit=min(50, remaining() * 0.4))
        if h_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    print(f"After Phase 3: C5={best_c5:.6f}, t={time.time()-start_time:.1f}s")

    # Phase 4: Multi-resolution optimization
    for n_try in [200, 240, 320, 400]:
        if remaining() < 50:
            break
        if n_try % 2 != 0:
            n_try += 1
        h_interp = interpolate_to_n(best_h, n_try)
        h_interp = make_feasible(h_interp, n_try)

        # Subgradient at new resolution
        h_new, c5_new = subgradient_optimize(h_interp, n_try, n_iters=1500,
                                              lr_start=0.012, lr_end=0.001,
                                              beta=80, time_limit=min(35, remaining() * 0.25))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            best_n = n_try

        # SLSQP direct at new resolution (only if promising)
        if remaining() > 80:
            h_new2, c5_new2 = slsqp_direct(h_new, n_try, max_iter=3000,
                                           time_limit=min(70, remaining() * 0.4))
            if h_new2 is not None and c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2
                best_n = n_try

        # SLSQP smooth at new resolution
        if remaining() > 50:
            h_new3, c5_new3 = slsqp_smooth(best_h, best_n, beta=200,
                                           max_iter=2000, time_limit=min(40, remaining() * 0.3))
            if h_new3 is not None and c5_new3 < best_c5:
                best_c5 = c5_new3
                best_h = h_new3
                best_n = n_try

        print(f"After n={n_try}: C5={best_c5:.6f}, t={time.time()-start_time:.1f}s")

    # Phase 5: Local pair transfer
    if remaining() > 25:
        n_rounds = min(25, max(5, int(remaining() / 3)))
        best_h, best_c5 = pair_transfer_local(best_h, best_n, n_rounds=n_rounds, rng=rng)
    print(f"After Phase 5: C5={best_c5:.6f}, t={time.time()-start_time:.1f}s")

    # Phase 6: Multi-start with perturbation + subgradient
    for trial in range(10):
        if remaining() < 25:
            break
        noise = 0.04 / (trial + 1)
        h_try = best_h + rng.randn(best_n) * noise
        h_try = make_feasible(h_try, best_n)
        h_new, c5_new = subgradient_optimize(h_try, best_n, n_iters=800,
                                              lr_start=0.01, lr_end=0.001,
                                              beta=100, time_limit=min(20, remaining() * 0.15))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            # Quick SLSQP polish
            if remaining() > 40:
                h_p, c5_p = slsqp_direct(best_h, best_n, max_iter=2000, time_limit=min(30, remaining() * 0.4))
                if h_p is not None and c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p
    print(f"After Phase 6: C5={best_c5:.6f}, t={time.time()-start_time:.1f}s")

    # Phase 7: Random perturbation improvement
    if remaining() > 20:
        n_trials = min(8000, int((remaining() - 10) / 0.015))
        best_h, best_c5 = random_perturb_improve(best_h, best_n, n_trials, 0.02, rng)
    print(f"After Phase 7: C5={best_c5:.6f}, t={time.time()-start_time:.1f}s")

    # Phase 8: Final SLSQP polish
    if remaining() > 30:
        h_new, c5_new = slsqp_direct(best_h, best_n, max_iter=6000, time_limit=min(60, remaining() * 0.6))
        if h_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        h_new2, c5_new2 = slsqp_smooth(best_h, best_n, beta=800,
                                       max_iter=3000, time_limit=min(30, remaining() * 0.3))
        if h_new2 is not None and c5_new2 < best_c5:
            best_c5 = c5_new2
            best_h = h_new2
    print(f"After Phase 8: C5={best_c5:.6f}, t={time.time()-start_time:.1f}s")

    # Phase 9: Final local search
    if remaining() > 15:
        best_h, best_c5 = pair_transfer_local(best_h, best_n, n_rounds=15, rng=rng)

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1 + 1e-12)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8

    print(f"Final: C5={best_c5:.6f}, n={best_n}, t={time.time()-start_time:.1f}s")
    return (best_h, best_c5, best_n)
