def binary_sa_hybrid(h_bin, n, iterations, T_start, T_end, rng):
    """SA with both single-swap and block-swap moves."""
    h = h_bin.copy()
    T = T_start
    dx = 2.0 / n
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    cooling = (T_end / T_start) ** (1.0 / iterations)
    
    for it in range(iterations):
        T *= cooling
        
        if it % 5 == 0:
            # Block swap
            block_len = rng.randint(1, max(2, n // 10))
            p = rng.randint(0, n - block_len)
            q = rng.randint(0, n - block_len)
            if abs(p - q) < 2 * block_len:
                continue
            
            # Save
            h1 = h[p:p+block_len].copy()
            h2 = h[q:q+block_len].copy()
            
            # Swap
            h[p:p+block_len] = h2
            h[q:q+block_len] = h1
            
            new_c5 = compute_c5(h)
            delta = new_c5 - cur_c5
            
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[p:p+block_len] = h1
                h[q:q+block_len] = h2
        else:
            # Single swap
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            p = ones[i]
            q = zeros[j]
            
            h[p] = 0.0

                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[p] = 1.0
                h[q] = 0.0
    
    return best_h, best_c5
