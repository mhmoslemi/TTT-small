        if vn > m:
            d = beta * (m - vn)
            r = r * (np.exp(d) if d > -700 else 0.0) + 1.0
            m = vn
        else:
            r += np.exp(beta * (vn - m)) if beta * (vn - m) > -700 else 0.0
    if r <= 0.0:
        return m
    return m + np.log(r) / beta


def smooth_jac(x):
    h = np.asarray(x, dtype=float)
    n = h.size
    beta = _SM_B
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    ac0 = float(np.dot(h, h))
    v0 = (total - ac0) * dx
    posv = np.empty(n, dtype=float)
    negv = np.empty(n, dtype=float)
    posv[0] = v0
    negv[0] = v0
    maxv = v0
    for L in range(1, n):
        mm = n - L
        ac = float(np.dot(h[:mm], h[L:]))
        vp = (pref[mm] - ac) * dx
        vn = (total - pref[L] - ac) * dx
        posv[L] = vp
        negv[L] = vn
        if vp > maxv:
            maxv = vp
        if vn > maxv:
            maxv = vn
    wpos = np.zeros(n, dtype=float)
    wneg = np.zeros(n, dtype=float)
    for i in range(n):
        d = beta * (posv[i] - maxv)
        wpos[i] = np.exp(d) if d > -700 else 0.0
    for i in range(1, n):
        d = beta * (negv[i] - maxv)
        wneg[i] = np.exp(d) if d > -700 else 0.0
    tw = float(wpos.sum() + wneg[1:].sum())
    g = np.zeros(n, dtype=float)
    if tw <= 0.0:
        return _lag_grad(h, 0, n) * dx
    inv = 1.0 / tw
    for L in range(n):
        cp = wpos[L] * inv * dx
        if cp > 0:
            g += cp * _lag_grad(h, L, n)
        if L > 0:
            cm = wneg[L] * inv * dx
            if cm > 0:
                g += cm * _lag_grad(h, -L, n)
    return g


def sum_constraint(x):
    return float(np.sum(x)) - 0.5 * len(x)


def sum_constraint_jac(x):
    return np.ones(len(x), dtype=float)


def run_smooth_opt(h0, n, beta_sched, maxiter_list, time_limit):
    global _SM_B
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac}]
    for i, beta in enumerate(beta_sched):
        if time.time() > time_limit - 5.0:
            break
        _SM_B = float(beta)
        mi = maxiter_list[min(i, len(maxiter_list) - 1)]
        try:
            res = minimize(
                smooth_obj, h, method="SLSQP", bounds=bounds,
                constraints=cons, options={"maxiter": mi, "ftol": 1e-12, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h2 = project_feasible(res.x, n)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2
                h = h2 if c2 <= best_c + 1e-6 else best_h.copy()
        except Exception:
            pass
    return best_h, best_c


def run_active_opt(h0, n, time_limit, active_counts=(16, 32, 64), maxiter=50):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    for acnt in active_counts:
        if time.time() > time_limit - 5.0:
            break
        for _ in range(2 if acnt <= 32 else 1):
            if time.time() > time_limit - 5.0:
                break
            vals, lags = all_lag_data(h)
            cur_c = float(np.max(vals))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            order = np.argsort(-vals)
            top = float(vals[order[0]])
            active_idx = set(int(x) for x in order[:acnt])
            active_idx.update(int(x) for x in order if vals[x] >= top - 5e-6)
            active_lags = [int(lags[i]) for i in sorted(active_idx)]
            if len(active_lags) == 0:
                active_lags = [0]
            h2, c2 = _solve_active(h, n, active_lags, cur_c, maxiter)
            if h2 is not None:
                if c2 < best_c:
                    best_h = h2.copy()
                    best_c = c2
                if c2 <= cur_c + 1e-9:
                    h = h2
                else:
                    h = best_h.copy()
    return best_h, best_c


def _active_obj(x):
    return float(x[-1])


def _active_obj_jac(x):
    j = np.zeros(len(x))
    j[-1] = 1.0
    return j


def _active_sum_eq(x):
    return float(np.sum(x[:-1])) - 0.5 * (len(x) - 1)


def _active_sum_eq_jac(x):
    j = np.zeros(len(x))
    j[:-1] = 1.0
    return j


def _active_ineq(x, alags):
    n = len(x) - 1
    dx = 2.0 / float(n)
    h = x[:-1]
    t = float(x[-1])
    out = np.empty(len(alags))
    for i, L in enumerate(alags):
        m = n - abs(L)
        if L == 0:
            lv = float(np.sum(h) - np.dot(h, h)) * dx
        elif L > 0:
            lv = (float(np.sum(h[:m])) - float(np.dot(h[:m], h[L:]))) * dx
        else:
            a = -L
            lv = (float(np.sum(h[a:])) - float(np.dot(h[:m], h[a:]))) * dx
        out[i] = t - lv
    return out


def _active_ineq_jac(x, alags):
    n = len(x) - 1
    dx = 2.0 / float(n)
    m_alags = len(alags)
    J = np.zeros((m_alags, n + 1))
    h = x[:-1]
    for i, L in enumerate(alags):
        if L == 0:
            g = _lag_grad(h, 0, n)
        elif L > 0:
            g = _lag_grad(h, L, n)
        else:
            g = _lag_grad(h, L, n)
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def _solve_active(h0, n, active_lags, t0, maxiter):
    x0 = np.empty(n + 1)
    x0[:n] = h0
    x0[n] = t0 + 1e-5
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{"type": "eq", "fun": _active_sum_eq, "jac": _active_sum_eq_jac}]
    if len(active_lags) > 0:
        cons.append({
            "type": "ineq",
            "fun": lambda x: _active_ineq(x, active_lags),
            "jac": lambda x: _active_ineq_jac(x, active_lags)
        })
    try:
        res = minimize(
            _active_obj, x0, method="SLSQP", bounds=bounds,
            constraints=cons, options={"maxiter": maxiter, "ftol": 1e-13, "disp": False}
        )
        if res.x is not None and res.x.size == n + 1 and np.all(np.isfinite(res.x)):
            h_out = project_feasible(res.x[:n], n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, np.inf


def local_perturb(h0, n, time_limit, seed, step=0.015):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    temp = 0.001
    min_temp = 1e-7
    max_trials = 8000
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        block_len = int(rng.randint(1, min(6, n) + 1))
        start = int(rng.randint(0, max(1, n - block_len + 1)))
        idx = np.arange(start, start + block_len)
        delta = rng.randn(block_len) * step
        delta -= delta.mean()
        trial = cur_h.copy()
        trial[idx] += delta
        trial = project_feasible(trial, n)
        c_try = compute_c5(trial)
        if c_try < best_c - 1e-15:
            best_h = trial.copy()
            best_c = c_try
            cur_h = trial
            cur_c = c_try
            temp = min(0.015, temp * 1.2)
        elif c_try < cur_c + temp:
            if c_try < cur_c or rng.rand() < np.exp(-(c_try - cur_c) / max(temp, 1e-9)):
                cur_h = trial
                cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, best_c


def structured_starts(n, rng, count=4):
    starts = []
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    # Cosine-based starts
    for arr in range(3):
        h = np.full(n, 0.5)
        ncomp = int(rng.randint(2, 6))
        for f in range(1, ncomp + 1):
            amp = rng.uniform(0.1, 0.35)
            phase = rng.uniform(0, 2 * np.pi)
            h += amp * np.cos(2 * np.pi * f * x / 2.0 + phase)
        starts.append(project_feasible(h, n))
    # Block patterns
    if n >= 4:
        for frac in (0.3, 0.4, 0.5):
            h = np.zeros(n)
            bw = max(1, int(n * frac))
            h[:bw] = 1.0
            h[n // 2:n // 2 + bw] = np.minimum(1.0, h[n // 2:n // 2 + bw] + 1.0)
            h = project_feasible(h, n)
            starts.append(h)
    # Thue-Morse-like
    if n > 1:
        tm = np.zeros(n)
        for i in range(n):
            c = 0
            v = i
            while v > 0:
                c += v & 1
                v >>= 1
            tm[i] = c % 2
        starts.append(project_feasible(tm, n))
    return starts[:count]


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    t_start = time.time()
    hard_end = t_start + max(120.0, budget_s - 30.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        raw_h = parent.copy()
        raw_c = compute_c5(raw_h)
        proj = project_feasible(parent, parent_n)
        proj_c = compute_c5(proj)
        if raw_h.size > 0 and float(np.min(raw_h)) >= -1e-12 and float(np.max(raw_h)) <= 1 + 1e-12:
            best_h = raw_h.astype(float).copy()
            best_c = raw_c
        else:
            best_h = proj
            best_c = proj_c
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Refine at parent resolution
    if 2 <= best_n <= 400 and time.time() < hard_end - 100:
        p1_end = min(hard_end - 80, t_start + 250)
        betas = (8, 20, 50, 120, 300)
        mism = [40, 40, 35, 30, 25]
        hh, cc = run_smooth_opt(best_h, best_n, betas[:3], mism, p1_end - 120)
        if cc < best_c:
            best_h, best_c = hh, cc
        hh, cc = run_active_opt(best_h, best_n, p1_end - 30, (16, 32, 48), maxiter=45)
        if cc < best_c:
            best_h, best_c = hh, cc

    # Phase 2: Try other resolutions
    trial_ns = []
    base = best_n
    for d in (16, 32, 48, 64, 80, 96):
        for nn in (base - d, base + d):
            if 80 <= nn <= 400 and nn != base:
                trial_ns.append(nn)
    for nn in (120, 144, 160, 192, 200, 224, 240, 256, 288, 320, 360, 400):
        if nn != best_n:
            trial_ns.append(nn)
    trial_ns = sorted(set(trial_ns))

    for tn in trial_ns:
        if time.time() > hard_end - 60:
            break
        if tn < 64 or tn > 420:
            continue
        rem = hard_end - 60 - time.time()
        if rem < 45:
            break
        slot = min(rem, 100.0)
        h_init = interpolate_to_n(best_h, tn)
        h_init = project_feasible(h_init, tn)
        # Quick smooth pass
        if slot > 50:
            hh, cc = run_smooth_opt(h_init, tn, (15, 50, 150), [35, 30, 25], time.time() + slot * 0.5)
        else:
            hh, cc = run_smooth_opt(h_init, tn, (30, 100), [30, 25], time.time() + slot * 0.5)
        # Active set
        if time.time() < time.time() + slot * 0.9:
            ac = (12, 24) if tn > 200 else (16, 32)
            hh2, cc2 = run_active_opt(hh, tn, time.time() + slot * 0.3, ac, maxiter=40)
            if cc2 < cc:
                hh, cc = hh2, cc2
        if cc < best_c:
            best_h = hh.copy()
            best_c = cc
            best_n = tn

    # Phase 3: Final refinement of best solution
    if time.time() < hard_end - 40 and 2 <= best_n <= 400:
        fin_end = hard_end - 20
        # High-beta smooth
        hh, cc = run_smooth_opt(best_h, best_n, (80, 200, 500), [40, 30, 20], fin_end - 60)
        if cc < best_c:
            best_h, best_c = hh, cc
        # Active set
        ac_final = (24, 48, 80) if best_n <= 200 else (16, 32, 48)
        hh, cc = run_active_opt(best_h, best_n, fin_end - 30, ac_final, maxiter=55)
        if cc < best_c:
            best_h, best_c = hh, cc

    # Phase 4: Local perturbation
    if time.time() < hard_end - 15 and 2 <= best_n <= 400:
        hh, cc = local_perturb(best_h, best_n, hard_end - 3, seed + 7777, step=0.012)
        if cc < best_c:
            best_h, best_c = hh, cc

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
