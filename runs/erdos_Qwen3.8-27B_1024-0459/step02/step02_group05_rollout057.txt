def binary_sa(h, n, t0, t1, n_steps, rng):
    """Simulated annealing for binary sequences."""
    h = h.copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for step in range(n_steps):
        T = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        if T < 1e-10:
            T = 1e-10
        
        # Pick random 1 and 0 to swap
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))

        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5

            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            # Update ones/zeros

            zeros = np.where(h == 0.0)[0]

            cur_c5 = new_c5
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]

def wiedemann_binary(n):
    """Create a Wiedemann sequence of length n (using nearest prime)."""
    def is_prime(x):
        if x < 2:
            return False
        for i in range(2, int(x**0.5) + 1):
            if x % i == 0:
                return False
        return True
    
    p = n
    while not is_prime(p):
        p += 1
    
    h = np.zeros(n)
    for i in range(n):
        t = (i * (i + 1) // 2) % p
        if 1 <= t <= (p - 1) // 2:
            h[i] = 1.0
    
    # Adjust to get exactly n/2 ones
    target = n // 2
    cnt = int(np.sum(h))
    # This might not be exactly n/2, so adjust
    if cnt > target:
        ones = np.where(h == 1.0)[0]
        for idx in ones[cnt - target:]:
            h[idx] = 0.0
    elif cnt < target:
        zeros = np.where(h == 0.0)[0]
        for idx in zeros[:target - cnt]:
            h[idx] = 1.0
    
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None
    
    if h_parent is not None:
        n0 = len(h_parent)
        best_h = h_parent.copy()
    else:
        n0 = 160
        best_h = np.ones(n0) * 0.5
    
    best_h = make_feasible(best_h, n0)
    best_c5 = compute_c5(best_h)
    best_n = n0
    
    # Phase 1: SLSQP on parent
    if remaining() > 60:
        h_opt, c5_opt = slsqp_opt(best_h, best_n, max_iter=3000, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    print(f"After SLSQP: C5 = {best_c5:.6f}, t={remaining():.0f}s")
    
    # Phase 2: Smooth annealing
    if remaining() > 60:
        h_opt, c5_opt = smooth_anneal(best_h, best_n, betas=[50, 200, 1000], max_iter=1500)
        if c5_opt < best_c5:
