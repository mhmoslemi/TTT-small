    n = len(h)

    vals, lags = all_lag_values_lags(h)
    max_val = np.max(vals)

    weights = np.exp(beta_smooth * (vals - max_val))
    weights /= weights.sum()
    
    # Compute weighted gradient
    g = np.zeros(n)

        return val, dx * g
    else:
        s_high = float(np.sum(h[L:]))
        val = (s_high - ac) * dx
        g = np.zeros(n, dtype=float)
        g[L:] += 1.0 - h[:m]
        g[:m] += -h[L:]
        return val, dx * g


def subgradient_polish(h0, n, time_limit, n_top=15, beta=30.0, max_steps=50000):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = 0.002
    momentum = np.zeros(n, dtype=float)
    beta_mom = 0.9
    stall = 0
    stall_limit = 2000
    
    for step in range(max_steps):
        if time.time() > time_limit:
            break
        vals = all_lag_values(h)
        cur_c = float(np.max(vals))
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > stall_limit:
                alpha = max(0.0005, alpha * 0.7)
                stall = 0
                if alpha < 0.0005:
                    break
        
        max_val = cur_c
        weights = np.exp(beta * (vals - max_val))
        wsum = float(np.sum(weights))
        if wsum < 1e-30:
            weights = np.zeros_like(vals)
            weights[np.argmax(vals)] = 1.0
        else:
            weights /= wsum
        
        g = np.zeros(n, dtype=float)
        n_used = len(vals)
        top_idx = np.argsort(-weights)[:n_top]
        for idx in top_idx:
            w = float(weights[idx])
            if w < 1e-10:
                continue
            L = int(idx - (n - 1))
            _, lg = lag_values_grad(h, L)
            g += w * lg
        
        momentum = beta_mom * momentum + (1 - beta_mom) * g
        h_try = project_feasible(h - alpha * momentum, n)
        
        vals_try = all_lag_values(h_try)
        c_try = float(np.max(vals_try))
        
        if c_try < cur_c:
            h = h_try
            alpha = min(0.01, alpha * 1.02)
        else:
            alpha = max(0.0003, alpha * 0.85)
    
    return best_h, float(best_c)


def smooth_objective(h, beta):
    n = h.size
    dx = 2.0 / float(n)
    vals = all_lag_values(h)
    max_v = float(np.max(vals))
    if beta < 1.0:
        beta = 1.0
    shifted = beta * (vals - max_v)
    shifted = np.clip(shifted, -500, 0)
    log_sum = float(np.log(np.sum(np.exp(shifted))))
    return max_v + log_sum / beta


def smooth_objective_grad(h, beta):
    n = h.size
    dx = 2.0 / float(n)
    vals = all_lag_values(h)
    max_v = float(np.max(vals))
    if beta < 1.0:
        beta = 1.0
    shifted = beta * (vals - max_v)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    if wsum < 1e-30:
        weights = np.zeros_like(vals)
        weights[np.argmax(vals)] = 1.0
        wsum = 1.0
    else:
        weights /= wsum
    
    g = np.zeros(n, dtype=float)
    n_used = len(vals)
    active_idx = np.where(weights > 1e-8)[0]
    for idx in active_idx:
        w = float(weights[idx])
        L = int(idx - (n - 1))
        _, lg = lag_values_grad(h, L)
        g += w * lg
    return g


def optimize_smooth(h0, n, beta, maxiter=60, ftol=1e-12):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    def obj(x):
        return smooth_objective(x, beta)
    
    def obj_grad(x):
        return smooth_objective_grad(x, beta)
    
    def c_fun(x):
        return float(np.sum(x) - 0.5 * n)
    
    def c_jac(x):
        return np.ones(n, dtype=float)
    
    constraints = [{"type": "eq", "fun": c_fun, "jac": c_jac}]
    
    best_h = h.copy()
    best_c = compute_c5(h)
    
    try:
        res = minimize(obj, h, jac=obj_grad, method="L-BFGS-B",
                       bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": ftol, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_h = h2
                best_c = c2
    except Exception:
        pass
    
    try:
        res = minimize(obj, h, jac=obj_grad, method="SLSQP",
                       bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": ftol, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_h = h2
                best_c = c2
    except Exception:
        pass
    
    return best_h, float(best_c)


def active_set_optimize(h0, n, active_lags, t0, maxiter=80):
    h = np.asarray(h0, dtype=float)
    n = int(n)
    dx = 2.0 / float(n)
    
    def obj(x):
        return float(x[-1])
    
    def obj_jac(x):
        j = np.zeros(n + 1, dtype=float)
        j[-1] = 1.0
        return j
    
    def eq_fun(x):
        return float(np.sum(x[:n]) - 0.5 * n)
    
    def eq_jac(x):
        j = np.zeros(n + 1, dtype=float)
        j[:n] = 1.0
        return j
    
    m = len(active_lags)
    
    def ineq_fun(x):
        out = np.empty(m, dtype=float)
        hh = x[:n]
        for i, ll in enumerate(active_lags):
            val, _ = lag_values_grad(hh, int(ll))
            out[i] = float(x[-1]) - val
        return out
    
    def ineq_jac(x):
        J = np.zeros((m, n + 1), dtype=float)
        hh = x[:n]
        for i, ll in enumerate(active_lags):
            _, g = lag_values_grad(hh, int(ll))
            J[i, :n] = -g
            J[i, n] = 1.0
        return J
    
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = float(t0)
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [{"type": "eq", "fun": eq_fun, "jac": eq_jac}]
    if m > 0:
        constraints.append({"type": "ineq", "fun": ineq_fun, "jac": ineq_jac})
    
    try:
        res = minimize(obj, x0, jac=obj_jac, method="SLSQP",
                       bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": 1e-13, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)) and res.x.size == n + 1:
            return project_feasible(res.x[:n], n)
    except Exception:
        pass
    
    return h


def refine_with_active_sets(h0, n, time_limit, seq=(8, 16, 32, 64, 128, 256)):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for acnt in seq:
        if time.time() > time_limit - 3.0:
            break
        for _ in range(2):
            if time.time() > time_limit - 3.0:
                break
            vals = all_lag_values(h)
            cur_c = float(np.max(vals))
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
            
            center = n - 1
            sorted_idx = np.argsort(-vals)
            active_idx = sorted_idx[:min(acnt, len(vals))]
            active_lags = [int(idx - center) for idx in active_idx]
            
            t0 = cur_c + 1e-6
            maxit = 50 if acnt <= 32 else 35
            h2 = active_set_optimize(h, n, active_lags, t0, maxiter=maxit)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            if c2 <= cur_c + 1e-8:
                h = h2
    
    return best_h, float(best_c)


def local_block_perturb(h0, n, time_limit, seed, step=0.015, max_block=6):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.001
    min_temp = 1e-7
    
    total_budget = max(1, max(0, int((time_limit - time.time()) * 100)))
    steps_done = 0
    
    while time.time() < time_limit and steps_done < total_budget:
        L = int(rng.randint(1, max(2, max_block + 1)))
        st = int(rng.randint(0, max(1, n - L + 1)))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        steps_done += 1
        
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            step = min(0.03, step * 1.03)
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    
    return best_h, float(best_c)


def gradient_direction_search(h0, n, time_limit, n_lags=20):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h
    cur_c = best_c
    alpha_base = 0.005
    
    while time.time() < time_limit:
        vals = all_lag_values(cur_h)
        cur_c = float(np.max(vals))
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = cur_h.copy()
        
        center = n - 1
        top_idx = np.argsort(-vals)[:n_lags]
        g = np.zeros(n, dtype=float)
        for idx in top_idx:
            L = int(idx - center)
            _, lg = lag_values_grad(cur_h, L)
            g += lg
        g /= float(n_lags)
        
        best_step_c = cur_c
        best_step_h = cur_h
        for mult in (1.0, 0.5, 0.25, 2.0, 0.1):
            a = alpha_base * mult
            h_try = project_feasible(cur_h - a * g, n)
            c_try = compute_c5(h_try)
            if c_try < best_step_c:
                best_step_c = c_try
                best_step_h = h_try
        
        if best_step_c < cur_c - 1e-12:
            cur_h = best_step_h
            alpha_base = min(0.02, alpha_base * 1.01)
        else:
            alpha_base = max(0.001, alpha_base * 0.8)
            if alpha_base < 0.001:
                break
    
    return best_h, float(best_c)


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def quick_optimize(h0, n, time_limit):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    
    if time.time() < time_limit - 5.0:
        h2, c2 = subgradient_polish(h, n, time_limit - 10.0, n_top=10, beta=25.0, max_steps=20000)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    if time.time() < time_limit - 3.0:
        h2, c2 = gradient_direction_search(best_h, n, time_limit - 3.0, n_lags=12)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)
    
    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        raw_h = np.clip(parent, 0.0, 1.0).copy()
        if abs(float(np.sum(raw_h)) - 0.5 * parent_n) > 1e-8:
            raw_h = project_feasible(parent, parent_n)
        best_h = raw_h
        best_c = compute_c5(raw_h)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)
    
    # Phase 1: Subgradient polish at base resolution
    if time.time() < hard_end - 200.0:
        t_limit = min(hard_end - 200.0, time.time() + 200.0)
        h1, c1 = subgradient_polish(best_h, best_n, t_limit, n_top=15, beta=30.0, max_steps=80000)
        if c1 < best_c:
            best_h = h1
            best_c = c1
    
    # Phase 2: Smooth optimization with varying beta
    if time.time() < hard_end - 150.0:
        t_limit = min(hard_end - 150.0, time.time() + 120.0)
        for beta in (20, 50, 100, 200):
            if time.time() > t_limit:
                break
            h2, c2 = optimize_smooth(best_h, best_n, beta, maxiter=40)
            if c2 < best_c:
                best_h = h2
                best_c = c2
    
    # Phase 3: Active set refinement
    if time.time() < hard_end - 100.0:
        t_limit = min(hard_end - 100.0, time.time() + 100.0)
        if best_n <= 120:
            seq = (8, 16, 32, 64, 128)
        elif best_n <= 200:
            seq = (8, 16, 32, 64)
        else:
            seq = (8, 16, 32)
        h2, c2 = refine_with_active_sets(best_h, best_n, t_limit, seq=seq)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    # Phase 4: Gradient direction search
    if time.time() < hard_end - 60.0:
        t_limit = min(hard_end - 60.0, time.time() + 50.0)
        h2, c2 = gradient_direction_search(best_h, best_n, t_limit, n_lags=20)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    # Phase 5: Multi-resolution exploration
    if time.time() < hard_end - 80.0:
        t_limit = min(hard_end - 80.0, time.time() + 80.0)
        trial_ns = []
        for d in (-20, -16, -12, -8, -4, 4, 8, 12, 16, 20, 24, 32, 40):
            nn = best_n + d
            if 80 <= nn <= 300 and nn != best_n:
                trial_ns.append(nn)
        for extra_n in (120, 140, 160, 180, 200, 220, 240):
            if extra_n != best_n and 80 <= extra_n <= 300:
                trial_ns.append(extra_n)
        
        seen = set()
        trial_ns = [x for x in trial_ns if x not in seen and not seen.add(x)]
        trial_ns = trial_ns[:6]
        
        remaining = t_limit - time.time()
        per_n = max(10.0, remaining / max(1, len(trial_ns)))
        
        for nn in trial_ns:
            if time.time() > t_limit:
                break
            h_interp = interpolate_to_n(best_h, nn)
            h2, c2 = quick_optimize(h_interp, nn, time.time() + per_n)
            if c2 < best_c:
                best_h = h2
                best_c = c2
                best_n = nn
    
    # Phase 6: Final intensive refinement at best resolution
    if time.time() < hard_end - 30.0:
        t_limit = min(hard_end - 30.0, time.time() + 40.0)
        h2, c2 = subgradient_polish(best_h, best_n, t_limit, n_top=12, beta=40.0, max_steps=30000)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    # Phase 7: Active set final push
    if time.time() < hard_end - 20.0:
        t_limit = min(hard_end - 20.0, time.time() + 18.0)
        seq_final = (16, 32, 64) if best_n <= 160 else (8, 16, 32)
        h2, c2 = refine_with_active_sets(best_h, best_n, t_limit, seq=seq_final)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    # Phase 8: Local block search
    if time.time() < hard_end - 5.0:
        t_limit = hard_end - 2.0
        h2, c2 = local_block_perturb(best_h, best_n, t_limit, seed + 7777, step=0.01, max_block=5)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
