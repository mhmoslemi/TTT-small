def thue_morse(n):
    """Generate Thue-Morse sequence of length n (must be power of 2)"""
    tm = np.zeros(n, dtype=np.float64)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return tm

def thue_morse_fast(n):
    """Fast TM generation using bit operations"""
    tm = np.zeros(n, dtype=np.float64)
    x = np.arange(n, dtype=np.int32)
    # XOR-shift trick for popcount
    x = x ^ (x >> 1)
    x = x ^ (x >> 2)
    x = x ^ (x >> 4)
    x = x ^ (x >> 8)
    x = x ^ (x >> 16)
    tm[:] = (x & 1).astype(np.float64)
    tm[:] = 1.0 - tm  # TM[i] = 1 if popcount is even
    return tm

    """Enforce h[i] + h[i+n/2] = 1"""
    half = n // 2
    h[:half] = 1.0 - h[half:]
    return h

    h = np.round(np.clip(h, 0, 1)).astype(np.float64)
    half = n // 2

    h0 = np.clip(h0, 0.0, 1.0).astype(np.float64)
    # Ensure feasibility
    diff = n / 2.0 - np.sum(h0)
    mask = (h0 > 1e-6) & (h0 < 1.0 - 1e-6)
    cnt = np.count_nonzero(mask)
    if cnt > 0:
        h0[mask] += diff / cnt
    h0 = np.clip(h0, 0.0, 1.0)

    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}

    def obj(x):
        hr = np.clip(x, 0.0, 1.0)
        cv = np.correlate(hr, 1.0 - hr, mode='full')
        return float(np.max(cv))

    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.fun < 1e15:
            h_out = np.clip(res.x, 0.0, 1.0)
            diff2 = n / 2.0 - np.sum(h_out)
            mask2 = (h_out > 1e-8) & (h_out < 1.0 - 1e-8)
            if np.count_nonzero(mask2) > 0:
                h_out[mask2] += diff2 / np.count_nonzero(mask2)
            h_out = np.clip(h_out, 0.0, 1.0)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h0, compute_c5(h0)

def simulated_anneal(h0, n, rng, timeout, t0):
    """SA to escape local optima"""
    h = make_anti_bin(h0.copy(), n)
    half = n // 2
    cur_cm = float(np.max(corr_vec(h)))
    best_cm = cur_cm
    best_h = h.copy()
    T = 4.0
    cooling = 0.9995
    stall = 0

    while True:
        if time.time() - t0 > timeout:
            break
        if stall > 200000:
            break
        if T < 0.01:
            break

        i = rng.randint(0, half)
        j = rng.randint(0, half)
        if i == j:
            continue

        h[i] = 0.0; h[i+half] = 1.0
        h[j] = 1.0; h[j+half] = 0.0
        new_cm = float(np.max(corr_vec(h)))
        delta = new_cm - cur_cm

        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-6)):
            cur_cm = new_cm
            if new_cm < best_cm:
                best_cm = new_cm
                best_h = h.copy()
                stall = 0
            T *= cooling
        else:
            h[i] = 1.0; h[i+half] = 0.0
            h[j] = 0.0; h[j+half] = 1.0
            stall += 1
            T *= cooling

    return best_h, best_cm

def random_binary_init(n, rng):
    half = n // 2
    h = np.zeros(n)
    ones_pos = rng.choice(half, half // 2, replace=False)
    h[ones_pos] = 1.0
    h[half:] = 1.0 - h[:half]
    return h

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    remaining = lambda: budget_s - (time.time() - t0)

    # Get parent if available
    h_parent = None
    try:
        hp = np.array(initial_h_values, dtype=np.float64)
        if hp is not None and len(hp) > 0:
            h_parent = hp.copy()
    except Exception:
        h_parent = None

    best_h = None
    best_c5 = 1.0
    best_n = 256

    # Try n=256 (primary target)
    n = 256
    half = n // 2

    # Generate initial solutions
    inits_256 = []
    h_tm = make_anti_bin(tm_seq(n), n)
    inits_256.append(h_tm)

    if h_parent is not None:
        h_p = make_anti_bin(interp_h(h_parent, n), n)
        inits_256.append(h_p)

    # Random inits
    for _ in range(8):
        inits_256.append(random_binary_init(n, rng))

    # Phase 1: Fast local search from each init
    for h0 in inits_256:
        if remaining() < 20:
            break
        h_opt, cm_opt = fast_ls(h0, n, rng, timeout=min(60.0, remaining() - 10), t0=t0, max_stall=200000)
        c5_val = cm_opt * 2.0 / n
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n

    # Phase 2: More aggressive search from the best
    if remaining() > 30:
        h_opt, cm_opt = fast_ls(best_h, n, rng, timeout=min(90.0, remaining() - 20), t0=t0, max_stall=500000)
        c5_val = cm_opt * 2.0 / n
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n

    # Phase 3: Targeted search
    if remaining() > 20:
        h_opt, cm_opt = targeted_ls(best_h, n, rng, timeout=min(40.0, remaining() - 10), t0=t0)
        c5_val = cm_opt * 2.0 / n
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n

    # Phase 4: Simulated annealing
    if remaining() > 30:
        h_opt, cm_opt = simulated_anneal(best_h, n, rng, timeout=min(40.0, remaining() - 10), t0=t0)
        c5_val = cm_opt * 2.0 / n
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n

    # Phase 5: More local search rounds
    for rnd in range(3):
        if remaining() < 20:
            break
        h_opt, cm_opt = fast_ls(best_h, n, rng, timeout=min(40.0, remaining() - 15), t0=t0, max_stall=300000)
        c5_val = cm_opt * 2.0 / n
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = n

    # Phase 6: Try n=128
    if remaining() > 40:
        n2 = 128
        inits_128 = []
        h_tm2 = make_anti_bin(tm_seq(n2), n2)
        inits_128.append(h_tm2)
        if h_parent is not None:
            inits_128.append(make_anti_bin(interp_h(h_parent, n2), n2))
        if best_h is not None:
            inits_128.append(make_anti_bin(interp_h(best_h, n2), n2))
        for _ in range(4):
            inits_128.append(random_binary_init(n2, rng))

        for h0 in inits_128:
            if remaining() < 15:
                break
            h_opt, cm_opt = fast_ls(h0, n2, rng, timeout=min(25.0, remaining() - 10), t0=t0, max_stall=200000)
            c5_val = cm_opt * 2.0 / n2
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n2

    # Phase 7: Try n=512
    if remaining() > 50:
        n3 = 512
        inits_512 = []
        h_tm3 = make_anti_bin(tm_seq(n3), n3)
        inits_512.append(h_tm3)
        if h_parent is not None:
            inits_512.append(make_anti_bin(interp_h(h_parent, n3), n3))
        for _ in range(4):
            inits_512.append(random_binary_init(n3, rng))

        for h0 in inits_512[:3]:
            if remaining() < 20:
                break
            h_opt, cm_opt = fast_ls(h0, n3, rng, timeout=min(20.0, remaining() - 10), t0=t0, max_stall=100000)
            c5_val = cm_opt * 2.0 / n3
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n3

    # Phase 8: SLSQP polish on best solution
    if remaining() > 15 and best_h is not None:
        n_best = best_n
        h_pol, c5_pol = slsqp_polish(best_h, n_best, max_iter=1000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_best

    # Phase 9: Final binary search after polish
    if remaining() > 20 and best_h is not None:
        h_opt, cm_opt = fast_ls(best_h, best_n, rng, timeout=min(30.0, remaining() - 10), t0=t0, max_stall=200000)
        c5_val = cm_opt * 2.0 / best_n
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()
            best_n = best_n

    # Final SLSQP if time allows
    if remaining() > 10 and best_h is not None:
        h_pol, c5_pol = slsqp_polish(best_h, best_n, max_iter=500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = best_n

    # Final feasibility check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
