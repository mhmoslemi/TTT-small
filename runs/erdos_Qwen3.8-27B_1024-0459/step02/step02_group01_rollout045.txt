def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask_lo = h > 0
        mask_hi = h < 1
        if diff > 0:
            h[mask_hi] += diff / max(1, np.sum(mask_hi))
        else:
            h[mask_lo] += diff / max(1, np.sum(mask_lo))
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)[-int(np.ceil(abs(diff)))]
            for i in idx:
                h[i] = min(1.0, h[i] + 1.0)
        else:
            idx = np.argsort(h)[-int(np.ceil(abs(diff)))]
            for i in idx:
                h[i] = max(0.0, h[i] - 1.0)
    return h

def to_binary(h, n):
    h_bin = (h > 0.5).astype(float)

        zeros = np.where(h_bin == 0)[0]

def smooth_max_objective(x, beta):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(c)
    sh = beta * (c - m)
    sh = np.clip(sh, -500.0, 0.0)
    smax = m + np.log(np.sum(np.exp(sh))) / beta
    return smax * (2.0 / len(x))

                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-14, 'iprint': 0})
            if res.success or res.x is not None:
                h = project_feasible(res.x, n)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except Exception:
            pass
    
    return best_h, best_c5

def structured_init(n, rng):
    inits = []
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Alternating blocks of various sizes
    for block in [2, 3, 4, 5]:
        h = np.zeros(n)
        pos = 0
        val = 1.0
        while pos < n:
            length = min(block, n - pos)
            h[pos:pos+length] = val
            pos += length
            val = 1.0 - val
        inits.append(project_feasible(h, n))
    # Random binary
    h = np.zeros(n)
    h[rng.choice(n, n//2, replace=False)] = 1.0
    inits.append(h)
    # Sine-based
    t = np.linspace(0, 1, n, endpoint=False)
    for freq in [3, 7, 11]:
        phase = rng.rand() * 2 * np.pi
        hs = 0.5 + 0.3 * np.sin(2 * np.pi * freq * t + phase)
        inits.append(project_feasible(hs, n))
    return inits

def interpolate_h(h_src, n_src, n_dst):
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dst) + 0.5) * 2.0 / n_dst
    return np.interp(xn, xp, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Get parent
    parent_h = None
    parent_n = 0
    try:
        ph = np.array(initial_h_values, dtype=float)
        if len(ph) > 0:
            parent_h = ph
            parent_n = len(ph)
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = parent_n
    except Exception:
        if parent_h is None:
            best_n = 96
            best_h = np.ones(best_n) * 0.5
            best_c5 = compute_c5(best_h)
    
    def remaining():
        return max(0.0, budget_s - (time.time() - t0) - 3.0)
    
    def update_best(h, n):
        nonlocal best_h, best_c5, best_n
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        return c5
    
    # Phase 1: Continuous optimization at multiple resolutions
    n_list = sorted(set([best_n, 192, 256, 384, 512]))
    n_list = [n for n in n_list if 64 <= n <= 512]
    
    for n in n_list:
        if remaining() < 60:
            break
        
        time_per_n = min(120.0, remaining() / max(1, len(n_list)))
        
        # Get starting point
        if best_n == n:
            h_start = project_feasible(best_h, n)
        elif parent_h is not None:
            h_start = project_feasible(interpolate_h(parent_h, parent_n, n), n)
        else:
            h_start = np.ones(n) * 0.5
        
        # Try continuous optimization
        h_cont, c5_cont = continuous_optimize(h_start, n, t0, time_per_n)
        update_best(h_cont, n)
        
        if remaining() < 20:
            break
        
        # Try structured inits
        inits = structured_init(n, rng)
        for h0 in inits[:4]:
            if remaining() < 30:
                break
            h_c, c5_c = continuous_optimize(h0, n, t0, 40)
            update_best(h_c, n)
    
    # Phase 2: Binary local search - main optimization
    # Convert best to binary at several resolutions
    for n in sorted(set([best_n, 256, 384, 512])):
        if remaining() < 100:
            break
        
        n = max(128, min(n, 512))
        
        if best_n == n:
            h_bin = to_binary(best_h, n)
        elif parent_h is not None:
            h_interp = interpolate_h(parent_h, parent_n, n)
            h_bin = to_binary(h_interp, n)
        else:
            h_bin = np.zeros(n)
            h_bin[rng.choice(n, n//2, replace=False)] = 1.0
        
        time_for_bls = min(200.0, remaining() * 0.6 / max(1, len([x for x in set([best_n, 256, 384, 512]) if 128 <= x <= 512])))
        time_for_bls = max(30.0, time_for_bls)
        
        # SA-based local search
        h_sa, c5_sa = binary_local_search(h_bin, n, 500000, t0, time_for_bls, rng,
                                          init_temp=0.15, cool_rate=0.99995)
        update_best(h_sa, n)
        
        if remaining() < 30:
            continue
        
        # Greedy local search (hill climbing with random restarts)
        h_greedy = h_sa.copy() if c5_sa < best_c5 else to_binary(best_h, n)
        h_g, c5_g = binary_local_search_greedy(h_greedy, n, 50000, t0, min(60.0, remaining() * 0.3), rng)
        update_best(h_g, n)
    
    # Phase 3: Multi-start binary SA
    if remaining() > 120:
        n_try = max(best_n, 256)
        n_try = min(n_try, 512)
        
        # Try multiple random binary starts
        for trial in range(8):
            if remaining() < 30:
                break
            
            # Different structured starts
            if trial == 0:
                h_start = to_binary(best_h, n_try) if best_n == n_try else to_binary(interpolate_h(parent_h, parent_n, n_try) if parent_h is not None else np.ones(n_try)*0.5, n_try)
            elif trial == 1:
                h_start = to_binary(best_h, n_try) if best_n == n_try else to_binary(interpolate_h(parent_h, parent_n, n_try) if parent_h is not None else np.ones(n_try)*0.5, n_try)
                # Perturb
                ones = np.where(h_start == 1)[0]
                zeros = np.where(h_start == 0)[0]
                for _ in range(n_try // 8):
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    h_start[i], h_start[j] = h_start[j], h_start[i]
            elif trial == 2:
                # Golden-ratio structured
                n_try2 = n_try
                h_s = np.zeros(n_try2)
                alpha = 0.6180339887
                pos = 0.0
                for i in range(n_try2 // 2):
                    h_s[int(pos) % n_try2] = 1.0
                    pos += alpha * n_try2
                h_start = to_binary(h_s, n_try2)
            elif trial == 3:
                # Fibonacci structured
                h_s = np.zeros(n_try)
                pos = 0.0
                fib_ratio = (5.0 ** 0.5 - 1.0) / 2.0
                for i in range(n_try // 2):
                    h_s[int(pos) % n_try] = 1.0
                    pos += fib_ratio * n_try
                h_start = to_binary(h_s, n_try)
            elif trial == 4:
                # Squared distribution
                h_s = np.zeros(n_try)
                positions = np.floor(np.linspace(0, n_try, n_try // 2 + 1, endpoint=False)).astype(int)
                h_s[positions[:n_try//2] % n_try] = 1.0
                h_start = to_binary(h_s, n_try)
            else:
                h_start = np.zeros(n_try)
                h_start[rng.choice(n_try, n_try // 2, replace=False)] = 1.0
            
            time_bl = min(80.0, remaining() * 0.5 / 3.0)
            h_res, c5_res = binary_local_search(h_start, n_try, 200000, t0, time_bl, rng,
                                                 init_temp=0.2, cool_rate=0.9999)
            update_best(h_res, n_try)
    
    # Phase 4: Fine-tuning - try small perturbations around best solution
    if remaining() > 30 and best_h is not None:
        n_b = best_n
        h_cur = np.array(best_h, dtype=float)
        
        # If binary, try local moves
        if np.all((h_cur == 0) | (h_cur == 1)):
            # Try swapping pairs of 1s
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            best_local = h_cur.copy()
            best_local_c5 = best_c5
            
            for _ in range(2000):
                if remaining() < 5:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                if i == j:
                    continue
                h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
                c5_try = compute_c5(h_cur)
                if c5_try < best_local_c5:
                    best_local_c5 = c5_try
                    best_local = h_cur.copy()
                else:
                    h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = best_local
                best_n = n_b
        else:
            # Continuous: small gradient step
            h_pol = project_feasible(h_cur, n_b)
            bounds = [(0.0, 1.0)] * n_b
            def eq_con(x):
                return np.sum(x) - n_b / 2.0
            
            try:
                res = minimize(lambda x: smooth_max_objective(x, 10000.0), h_pol,
                             method='SLSQP', bounds=bounds,
                             constraints=[{'type': 'eq', 'fun': eq_con}],
                             options={'maxiter': 200, 'ftol': 1e-15, 'iprint': 0})
                h_pol = project_feasible(res.x, n_b)
                c5_p = compute_c5(h_pol)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pol
            except Exception:
                pass
    
    # Phase 5: Final check with evaluate
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
