import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-14:
        idx = np.argsort(np.abs(h - 0.5))[::-1]
        for i in idx:
            if diff > 0 and h[i] < 1.0:
                h[i] = min(h[i] + diff, 1.0)
                break
            elif diff < 0 and h[i] > 0.0:
                h[i] = max(h[i] + diff, 0.0)
                break
    return h

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    result = np.interp(x_new, x_old, h_src)
    return np.clip(result, 0.0, 1.0)

def softmax_obj_and_grad(h, beta):
    """Compute soft-max-smoothed C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Soft max
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_corr)
    weights = exp_corr / sum_exp  # softmax weights
    c5_smooth = (max_corr + np.log(sum_exp) / beta) * dx
    # Gradient: dc5/dh[i] = dx * sum_k weights[k] * dcorr[k]/dh[i]
    # dcorr[k]/dh[i] = (1 - h[i-k_eff]) for the appropriate k
    # We need to figure out which lags involve h[i]
    # corr[k] = sum_j h[j] * (1-h[j + k - (n-1)])
    # dcorr[k]/dh[i] = (1 - h[i + k - (n-1)]) if i+(k-(n-1)) is in [0,n-1]
    # = -h[i] if k-(n-1) == 0 (i.e., k = n-1) -- but for binary this is 0
    # Actually: corr[k] = sum_j h[j]*(1-h[j+k-(n-1)])
    # d/dh[i]: appears in the first factor when j=i: (1-h[i+k-(n-1)])
    #           appears in the second factor when j+k-(n-1)=i, i.e., j=i+k-(n-1): -h[i+k-(n-1)] * 1 (derivative of (1-h) wrt h[i])
    # Wait, the second term: d/dh[i] of (1-h[j+k-(n-1)]) = -1 if j+k-(n-1) = i, i.e., j = i-k+(n-1)
    # So dcorr[k]/dh[i] = (1-h[i+k-(n-1)]) * [i+k-(n-1) in [0,n-1]] - h[i-k+(n-1)] * [i-k+(n-1) in [0,n-1]]
    # Let me simplify: let s = k-(n-1) be the "shift"
    # dcorr[n-1+s]/dh[i] = (1-h[i+s]) * [i+s in [0,n-1]] - h[i-s] * [i-s in [0,n-1]]
    
    # For efficiency, compute the gradient as:
    # grad[i] = dx * sum_s weights[n-1+s] * [(1-h[i+s])*valid1 - h[i-s]*valid2]
    # This is a convolution-like operation
    
    # Actually, for the purpose of L-BFGS-B, we can use a numerical gradient or just the objective.
    # Let me just return the objective for now.
    return c5_smooth

def lbfgs_smooth_opt(h_init, n, max_iter=5000, beta_start=50, beta_end=5000):
    """L-BFGS-B with annealed log-sum-exp."""
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    n_betas = 5
    for bi in range(n_betas):
        beta = beta_start * (beta_end / beta_start) ** (bi / max(n_betas - 1, 1))
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            return (cmax + np.log(np.sum(np.exp(beta * (corr - cmax)))) / beta)
        
        try:
            res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                          constraints=cons, options={'maxiter': 1000, 'ftol': 1e-15,
                                                     'gtol': 1e-12, 'disp': False})
            if res.fun < 1e10:
                h_new = make_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_c5

def binary_sa_swap(n, rng, n_steps=50000, t_start=0.5, t_end=0.01):
    """Binary SA with swap moves (preserves sum constraint automatically)."""
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    one_positions = set(ones.tolist())
    zero_positions = set(range(n)) - one_positions
    
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        # Pick a random 1 and a random 0 to swap
        a = rng.choice(sorted(one_positions)) if one_positions else 0
        b = rng.choice(sorted(zero_positions)) if zero_positions else 0
        if a == b:
            continue
        
        # Swap
        h[a] = 0
        h[b] = 1
        one_positions.remove(a)
        one_positions.add(b)
        zero_positions.remove(b)
        zero_positions.add(a)
        
        c5_new = compute_c5(h)
        
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() > np.exp(-(c5_new - best_c5) / max(T, 1e-15)):
            # Reject: swap back
            h[a] = 1
            h[b] = 0
            one_positions.add(a)
            one_positions.remove(b)
            zero_positions.add(b)
            zero_positions.remove(a)
    
    return best_h, best_c5

def binary_local_search(h_bin, n, rng, n_rounds=200):
    """Local search: try all single swaps, keep the best."""
    h = h_bin.astype(float).copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    one_idx = np.where(h == 1.0)[0]
    zero_idx = np.where(h == 0.0)[0]
    
    for round in range(n_rounds):
        improved = False
        # Shuffle the order of one positions
        rng.shuffle(one_idx)
        for a in one_idx:
            if time.time() > time.time() + 100:  # timeout placeholder
                break
            for b in zero_idx:
                if a == b:
                    continue
                h[a] = 0
                h[b] = 1
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    improved = True
                    one_idx = np.where(h == 1.0)[0]
                    zero_idx = np.where(h == 0.0)[0]
                    break
                else:
                    h[a] = 1
                    h[b] = 0
        if not improved:
            break
    return h, best_c5

def slsqp_refine(h_init, n, max_iter=5000, ftol=1e-15):
    """SLSQP refinement."""
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol,
                                                    'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def structured_binary_init(n, rng):
    """Generate structured binary initializations."""
    inits = []
    # Alternating
    h1 = np.tile([1.0, 0.0], n // 2)
    inits.append(h1)
    # Blocks of 2
    if n % 4 == 0:
        h2 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
        inits.append(h2)
    # Random
    h3 = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h3[ones] = 1.0
    inits.append(h3)
    # Quasi-periodic (Fibonacci word approximation)
    h4 = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    for i in range(n):
        h4[i] = 1.0 if ((i + 1) * phi % 1) < 0.5 else 0.0
    # Fix count
    ones_count = int(np.sum(h4))
    if ones_count > n // 2:
        # Flip some 1s to 0s
        ones_pos = np.where(h4 == 1.0)[0]
        rng.shuffle(ones_pos)
        for i in ones_pos[:ones_count - n//2]:
            h4[i] = 0.0
    elif ones_count < n // 2:
        zeros_pos = np.where(h4 == 0.0)[0]
        rng.shuffle(zeros_pos)
        for i in zeros_pos[:n//2 - ones_count]:
            h4[i] = 1.0
    inits.append(h4)
    return inits
