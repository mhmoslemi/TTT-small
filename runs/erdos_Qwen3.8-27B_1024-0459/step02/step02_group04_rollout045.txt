def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:
        n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    target = 0.5 * float(n)
    h = h[:n].copy()
    if h.size < n:
        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    elif h.size > n:
        h = h[:n]
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(90):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(50):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:

        else:

        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def lag_value(h, l):
    n = len(h)
    if l == 0:
        return float(np.sum(h) - np.dot(h, h))
    L = abs(int(l))
    m = n - L
    low = h[:m]
    high = h[L:]
    ac = float(np.dot(low, high))
    if l > 0:
        return float(np.sum(low) - ac)
    return float(np.sum(high) - ac)


def lag_gradient_only(h, l):
    n = len(h)
    if l == 0:
        return 1.0 - 2.0 * h
    L = abs(int(l))
    m = n - L
    low = h[:m]
    high = h[L:]
    g = np.zeros(n, dtype=float)
    if l > 0:
        g[:m] += 1.0 - high
        g[L:] += -low
    else:
        g[L:] += 1.0 - low
        g[:m] += -high
    return g


def add_lag_gradient_to(g, h, L, signed, coef):
    if coef == 0.0:
        return
    n = len(h)
    m = n - L
    low = h[:m]
    high = h[L:]
    if signed >= 0:
        g[:m] += coef * (1.0 - high)
        g[L:] += -coef * low
    else:
        g[L:] += coef * (1.0 - low)
        g[:m] += -coef * high


def all_lag_values_lags(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    vals = np.empty(max(1, 2 * n - 1), dtype=float)
    lags = np.empty(max(1, 2 * n - 1), dtype=int)
    ac0 = float(np.dot(h, h))
    vals[0] = (total - ac0) * dx
    lags[0] = 0
    idx = 1
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals[idx] = (pref[m] - ac) * dx
        lags[idx] = L
        idx += 1
        vals[idx] = (total - pref[L] - ac) * dx
        lags[idx] = -L
        idx += 1
    return vals, lags


def choose_active_lags(vals, lags, max_count, margin=1e-5):
    if vals.size == 0:
        return []
    order = np.argsort(-vals)
    take = min(int(max_count), int(vals.size))
    idx = order[:take]
    if margin > 0.0:
        top = float(vals[order[0]])
        near = np.where(vals >= top - margin)[0]
        if near.size > 0:
            idx = np.unique(np.concatenate((idx, near)))
            idx = idx[np.argsort(-vals[idx])]
            cap = max(int(max_count) * 3, int(vals.size))
            if idx.size > cap:
                idx = idx[:cap]
    return [int(x) for x in lags[idx]]


def smooth_sum_eq(x):

def smooth_h_objective(x):

    ac0 = float(np.dot(h, h))

    m = v0
    r = 1.0
    for L in range(1, n):
        mm = n - L
        ac = float(np.dot(h[:mm], h[L:]))
        vp = (pref[mm] - ac) * dx

def smooth_h_jac(x):
    h = np.asarray(x, dtype=float)
    n = h.size
    beta = max(1.0, float(SM_BETA))
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    ac0 = float(np.dot(h, h))
    v0 = (total - ac0) * dx
    posv = np.empty(n, dtype=float)
    negv = np.empty(n, dtype=float)
    posv[0] = v0
    negv[0] = v0
    maxv = v0
    for L in range(1, n):
        mm = n - L
        ac = float(np.dot(h[:mm], h[L:]))
        vp = (pref[mm] - ac) * dx
        vn = (total - pref[L] - ac) * dx
        posv[L] = vp
        negv[L] = vn
        if vp > maxv:
            maxv = vp
        if vn > maxv:
            maxv = vn
    wpos = np.zeros(n, dtype=float)
    wneg = np.zeros(n, dtype=float)
    for i in range(n):
        d = beta * (posv[i] - maxv)
        if d > -745.0:
            wpos[i] = np.exp(d)
    for i in range(1, n):
        d = beta * (negv[i] - maxv)
        if d > -745.0:
            wneg[i] = np.exp(d)
    total_w = float(wpos.sum() + wneg[1:].sum())
    g = np.zeros(n, dtype=float)
    if total_w <= 0.0:
        add_lag_gradient_to(g, h, 0, 0, dx)
        return g
    inv = 1.0 / total_w
    for L in range(n):
        cp = float(wpos[L]) * inv * dx
        if cp > 0.0:
            add_lag_gradient_to(g, h, L, 0, cp)
        if L > 0:
            cm = float(wneg[L]) * inv * dx
            if cm > 0.0:
                add_lag_gradient_to(g, h, L, -L, cm)
    return g


def optimize_smooth_h(h0, n, beta, maxiter=50, ftol=1e-11):
    global SM_BETA
    h = project_feasible(h0, n)
    SM_BETA = max(1.0, float(beta))
    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = [{"type": "eq", "fun": smooth_sum_eq, "jac": smooth_sum_eq_jac}]
    try:
        res = minimize(
            smooth_h_objective, h, method="SLSQP",
            bounds=bounds, constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False},
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            c2 = compute_c5(h2)
            return h2, c2
    except Exception:
        pass
    return h, compute_c5(h)

            active = choose_active_lags(vals, lags, int(acnt), margin=1e-5)
            if len(active) == 0:
                active = [int(lags[int(np.argmax(vals))])]
            t0 = current_c + 1e-4
            res = optimize_active_t(h, active, t0, maxiter=maxiter_base, ftol=1e-13)
            if res is not None:
                x = res.x
                if x is not None and x.size == len(h) + 1 and np.all(np.isfinite(x)):
                    h2 = project_feasible(x[:-1], n)
                    c2 = compute_c5(h2)
                    if c2 < best_c - 1e-12:
                        best_c = c2
                        best_h = h2.copy()
                    if c2 <= current_c + 1e-8:
                        h = h2
                    else:
                        h = best_h
    return best_h, best_c


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def gaussian_smooth(h, sigma=1.5):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 1:
        return h.copy()
    length = min(7, n)
    if length % 2 == 0:
        length -= 1
    length = max(1, length)
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * float(sigma) * float(sigma)))
    w /= w.sum()
    return np.convolve(h, w, mode="same")


def make_diverse_starts(h_base, n, rng, count=6):
    starts = [project_feasible(h_base, n)]
    if n > 1:
        flip = h_base[::-1]
        starts.append(project_feasible(0.5 * (h_base + flip), n))
        starts.append(project_feasible(0.5 + 0.4 * (h_base - flip), n))
    starts.append(project_feasible(gaussian_smooth(h_base, 1.2), n))
    starts.append(project_feasible(gaussian_smooth(h_base, 0.6), n))
    starts.append(project_feasible(h_base + rng.randn(n) * 0.04, n))
    starts.append(project_feasible(h_base + rng.randn(n) * 0.08, n))
    out = []
    seen_approx = []
    for s in starts[:count]:
        s = project_feasible(s, n)
        dup = False
        for prev in seen_approx:
            if np.max(np.abs(s - prev)) < 1e-8:
                dup = True
                break
        if not dup:
            out.append(s)
            seen_approx.append(s)
    return out


def coordinate_descent(h0, n, time_limit, seed, step=0.008, max_iters=20000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    it = 0
    stall = 0
    while it < max_iters and time.time() < time_limit:
        batch = 50 if it % 500 == 0 else 1
        improved = False
        for _b in range(batch):
            if it >= max_iters or time.time() > time_limit:
                break
            i = int(rng.randint(0, n))
            for delta in (step, -step, 2 * step, -2 * step, 0.5 * step, -0.5 * step):
                try_h = cur_h.copy()
                try_h[i] += delta
                try_h = project_feasible(try_h, n)
                try_c = compute_c5(try_h)
                if try_c < best_c - 1e-13:
                    best_h = try_h.copy()
                    best_c = try_c
                    cur_h = try_h
                    cur_c = try_c
                    improved = True
                    break
                elif try_c < cur_c:
                    cur_h = try_h
                    cur_c = try_c
        it += 1
        if improved:
            stall = 0
            step = min(0.02, step * 1.05)
        else:
            stall += 1
            step = max(0.001, step * 0.97)
            if stall > 3000:
                break
    return best_h, float(best_c)


def block_perturb_refine(h0, n, time_limit, seed, step=0.02, max_trials=3000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.003
    min_temp = 1e-7
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, temp * 1.1)
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.996, min_temp)
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)

    parent = None
    try:
        vals = initial_h_values
        if vals is not None:
            arr = np.asarray(vals, dtype=float).copy()
            if arr.size > 0:
                parent = arr
    except Exception:
        parent = None

    if parent is not None:
        parent_n = int(parent.size)
        raw_h = parent.copy()
        raw_c = compute_c5(raw_h)
        proj = project_feasible(parent, parent_n)
        proj_c = compute_c5(proj)
        if raw_c <= proj_c + 1e-12:
            best_h = raw_h
            best_c = raw_c
        else:
            best_h = proj
            best_c = proj_c
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Quick improvement at parent resolution and nearby (aim: 120s)
    phase1_end = min(hard_end - 600, start_time + 120.0)
    trial_ns = [best_n]
    for d in (-20, -10, 10, 20, 40):
        t = best_n + d
        if 40 <= t <= 320 and t != best_n:
            trial_ns.append(t)
    trial_ns = trial_ns[:4]

    for tn in trial_ns:
        if time.time() > phase1_end:
            break
        if len(trial_ns) > 1 and tn != best_n and time.time() > start_time + 50:
            break
        h_init = best_h if tn == best_n else interpolate_to_n(best_h, tn)
        h_init = project_feasible(h_init, tn)
        c_init = compute_c5(h_init)
        # Smooth max with a few betas
        for beta in (10, 40, 120):
            if time.time() > phase1_end:
                break
            h2, c2 = optimize_smooth_h(h_init, tn, beta, maxiter=40, ftol=1e-11)
            if c2 < c_init:
                h_init = h2
                c_init = c2
        if c_init < best_c:
            best_h = h_init
            best_c = c_init
            best_n = tn

    # Phase 2: Focused smooth-max + active refinement at best_n (aim: 500s)
    phase2_end = min(hard_end - 300, start_time + 120.0 + 500.0)
    n_work = best_n
    h_work = project_feasible(best_h, n_work)

    # Make diverse starts
    starts = make_diverse_starts(h_work, n_work, rng, count=5)

    for si, s_start in enumerate(starts):
        if time.time() > phase2_end:
            break
        remaining = phase2_end - time.time()
        if remaining < 20:
            break
        per_slot = remaining / max(1, len(starts) - si)
        per_slot = min(per_slot, 150.0)
        per_slot = max(15.0, per_slot)
        slot_end = time.time() + per_slot

        h_cur = s_start
        # Smooth max schedule
        betas = [5, 15, 50, 150, 400]
        for beta in betas:
            if time.time() > slot_end - 5:
                break
            h_cur, c_cur = optimize_smooth_h(h_cur, n_work, beta, maxiter=45, ftol=1e-11)
        # Active set refinement
        if time.time() < slot_end - 3:
            act_seq = (16, 32, 64, 128)
            if n_work <= 120:
                act_seq = (12, 24, 48, 96)
            h_cur, c_cur = refine_active_multi(
                h_cur, n_work, slot_end, act_seq,
                maxiter_base=60, passes_per_k=2
            )
        if c_cur < best_c:
            best_h = h_cur
            best_c = c_cur
            best_n = n_work

    # Phase 3: Heavy active set refinement (aim: 300s)
    phase3_end = min(hard_end - 150, start_time + 120.0 + 500.0 + 300.0)
    h_work = project_feasible(best_h, best_n)
    # Multiple active set rounds with increasing precision
    n_active_rounds = 4
    for rnd in range(n_active_rounds):
        if time.time() > phase3_end:
            break
        round_end = min(phase3_end - 5, time.time() + (phase3_end - time.time()) / max(1, n_active_rounds - rnd))
        act_seq = (32, 64, 128, 200)
        maxit_rnd = 80 + 20 * rnd
        h_work, c_work = refine_active_multi(
            h_work, best_n, round_end, act_seq,
            maxiter_base=maxit_rnd, passes_per_k=3
        )
        if c_work < best_c:
            best_h = h_work
            best_c = c_work

    # Phase 4: Coordinate descent + block perturb (aim: 120s)
    phase4_end = min(hard_end - 30, start_time + 120.0 + 500.0 + 300.0 + 120.0)
    h_work = project_feasible(best_h, best_n)
    if time.time() < phase4_end - 30:
        cd_end = min(phase4_end - 30, time.time() + 60.0)
        h_work, c_work = coordinate_descent(h_work, best_n, cd_end, seed + 7777, step=0.006)
        if c_work < best_c:
            best_h = h_work
            best_c = c_work

    if time.time() < phase4_end - 10:
        bp_end = min(phase4_end - 5, time.time() + 40.0)
        h_work, c_work = block_perturb_refine(h_work, best_n, bp_end, seed + 8888, step=0.015)
        if c_work < best_c:
            best_h = h_work
            best_c = c_work

    # Phase 5: Final SLSQP polish with very tight settings
    phase5_end = hard_end - 5.0
    h_work = project_feasible(best_h, best_n)
    if time.time() < phase5_end - 10:
        vals, lags = all_lag_values_lags(h_work)
        top_lags = choose_active_lags(vals, lags, 80, margin=1e-4)
        if len(top_lags) > 0:
            t0 = float(np.max(vals)) + 1e-5
            res = optimize_active_t(h_work, top_lags, t0, maxiter=200, ftol=1e-14)
            if res is not None and res.x is not None and np.all(np.isfinite(res.x)):
                h_fin = project_feasible(res.x[:-1], best_n)
                c_fin = compute_c5(h_fin)
                if c_fin < best_c:
                    best_h = h_fin
                    best_c = c_fin

    # Final validation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    # Ensure constraints are met
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(float(np.sum(best_h)) - 0.5 * best_n) < 1e-6

    return (best_h, float(best_c), int(best_n))
