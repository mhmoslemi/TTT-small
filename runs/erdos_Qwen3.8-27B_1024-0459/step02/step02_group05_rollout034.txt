def compute_c5(h):

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-10
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2} using bisection"""
    h = np.array(h, dtype=float)
    target = n / 2.0
    # Find alpha such that sum(clip(h - alpha, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def interp_resample(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    vals = np.interp(x_new, x_old, h_src)
    return make_feasible(vals, n_dst)

def lse_obj_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = (cmax + np.log(sum_exp)) / beta
    w = exp_s / sum_exp
    grad = np.zeros(n)
    for s in range(2 * n - 1):
        ws = w[s]
        if ws < 1e-15:
            continue
        shift = s - (n - 1)
        if shift >= 0:
            idx1 = np.arange(n - shift)
            grad[idx1] += ws * (1.0 - h[idx1 + shift])
            idx2 = np.arange(shift, n)
            grad[idx2] -= ws * h[idx2 - shift]
        else:
            idx1 = np.arange(-shift, n)
            grad[idx1] += ws * (1.0 - h[idx1 + shift])
            idx2 = np.arange(n + shift)
            grad[idx2] -= ws * h[idx2 - shift]
    return obj, grad

def slsqp_gradient(n, h_init, max_iter, ftol, beta):
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        o, _ = lse_obj_grad(x, beta)
        return o

    def grad_func(x):
        _, g = lse_obj_grad(x, beta)
        return g

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h_init, jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            c5 = compute_c5(h_out)
            return h_out, c5
    except Exception:
        pass
    return None, None

    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in beta_starts:

        for it in range(n_iters):
            _, grad = lse_obj_grad(h, beta)

                    else:

                best_h = h_new.copy()

def binary_local_search(h_bin, n, n_rounds, max_pairs=None):

    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break
        if max_pairs is None:

        else:
            pairs = [(oi, zi) for oi in ones for zi in zeros]
            if len(pairs) > max_pairs:
                rng_ps = np.random.RandomState(rnd)
                idx_ps = rng_ps.choice(len(pairs), max_pairs, replace=False)
                pairs = [pairs[i] for i in idx_ps]
        rng_ps = np.random.RandomState(rnd + 1000)
        if len(pairs) > 5000:
            idx_ps = rng_ps.choice(len(pairs), 5000, replace=False)
            pairs = [pairs[i] for i in idx_ps]
        best_improve = c5
        best_pair = None
        for (oi, zi) in pairs:
            h[oi] = 0.0
            h[zi] = 1.0
            c5_try = compute_c5(h)
            if c5_try < best_improve:
                best_improve = c5_try
                best_pair = (oi, zi)
            h[oi] = 1.0
            h[zi] = 0.0

            h[oi] = 0.0
            h[zi] = 1.0

def multi_start_binary(n, n_starts, n_rounds, rng, time_limit=60):
    best_h = None
    best_c5 = 1.0
    t0 = time.time()

        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-step, step)
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.01) & (h_try < 0.99)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
        else:
            step *= 0.97
        trial += 1
    return h, best_c5

def refine_resolution(h_src, n_dst, n, rng, time_limit=90):
    h_interp = interp_resample(h_src, n_dst)
    best_c5 = compute_c5(h_interp)
    best_h = h_interp.copy()
    t0 = time.time()
    # SLSQP with gradient
    if time.time() - t0 < time_limit * 0.5:
        h_opt, c5_opt = slsqp_gradient(n_dst, h_interp, 1500, 1e-13, 100.0)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    if time.time() - t0 < time_limit * 0.8:
        h_opt, c5_opt = slsqp_gradient(n_dst, best_h, 1000, 1e-13, 500.0)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    # Continuous local search
    if time.time() - t0 < time_limit:
        n_tr = min(3000, int((time_limit - (time.time() - t0)) / 0.02))
        best_h, best_c5 = continuous_local_search(best_h, n_dst, n_tr, 0.03, rng)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent, best_n)
    else:
        best_n = 160
        best_h = np.zeros(best_n)
        ones = rng.choice(best_n, best_n // 2, replace=False)
        best_h[ones] = 1.0
        best_h = make_feasible(best_h, best_n)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with exact gradient on parent resolution
    if remaining() > 100:
        h_opt, c5_opt = slsqp_gradient(best_n, best_h, 3000, 1e-15, 50.0)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Multi-beta gradient SLSQP
    if remaining() > 150:
        h_opt, c5_opt = gradient_based_multi_beta(
            best_n, best_h, [100.0, 300.0, 1000.0], 2000, time_limit=min(150, remaining() - 20))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Projected gradient with increasing beta
    if remaining() > 80:
        h_opt, c5_opt = projected_grad_solve(
            best_h, best_n, 500, [50.0, 200.0, 800.0], lr_init=0.3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Binary local search on parent
    if remaining() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, best_n)
        h_b, c5_b = binary_local_search(h_bin, best_n, n_rounds=30)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 5: Multi-start binary search
    if remaining() > 80:
        h_b, c5_b = multi_start_binary(best_n, n_starts=8, n_rounds=15, rng=rng,
                                       time_limit=min(60, remaining() - 20))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b
            best_n = best_n

    # Phase 6: Try higher resolutions
    res_list = [200, 240, 256, 320]
    for n_try in res_list:
        if remaining() < 60:
            break
        if n_try % 2 != 0:
            n_try += 1
        t_limit = min(80, remaining() - 30)
        h_new, c5_new = refine_resolution(best_h, n_try, n_try, rng, time_limit=t_limit)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            best_n = n_try

    # Phase 7: Final SLSQP polish on best resolution
    if remaining() > 40:
        h_opt, c5_opt = slsqp_gradient(best_n, best_h, 4000, 1e-16, 500.0)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 8: Continuous local search polish
    if remaining() > 30:
        n_tr = min(10000, int((remaining() - 10) / 0.015))
        best_h, best_c5 = continuous_local_search(best_h, best_n, n_tr, 0.02, rng,
                                                   time_limit=min(25, remaining() - 10))

    # Phase 9: Binary polish on final resolution
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, best_n)
        h_b, c5_b = binary_local_search(h_bin, best_n, n_rounds=20)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b

    # Phase 10: Very fine final SLSQP
    if remaining() > 15:
        h_opt, c5_opt = slsqp_gradient(best_n, best_h, 2000, 1e-16, 2000.0)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    assert np.all(best_h >= -1e-10) and np.all(best_h <= 1.0 + 1e-10)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-7

    return (best_h, best_c5, best_n)
