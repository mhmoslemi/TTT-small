def make_binary_feasible(h, n):
    """Make h binary with exactly n/2 ones."""
    h = np.round(h).astype(float)

        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c5 = new_c5
            ones.discard(i); zeros.add(i)
            zeros.discard(j); ones.add(j)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
        n_iter_actual += 1
    return best_h, best_c5
