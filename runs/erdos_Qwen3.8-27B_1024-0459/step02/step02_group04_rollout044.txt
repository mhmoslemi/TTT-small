def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_lag_values(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    vals = np.empty(2 * n - 1, dtype=float)
    lags = np.arange(-(n - 1), n, dtype=int)
    ac0 = float(np.dot(h, h))
    vals[n - 1] = (total - ac0) * dx
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals[n - 1 - L] = (total - pref[L] - ac) * dx
        vals[n - 1 + L] = (pref[m] - ac) * dx
    return vals, lags


def lag_grad(h, l):
    n = len(h)
    if l == 0:
        return 1.0 - 2.0 * h
    L = abs(int(l))
    m = n - L
    g = np.zeros(n, dtype=float)
    if l > 0:
        g[:m] += 1.0 - h[L:]
        g[L:] += -h[:m]
    else:
        g[L:] += 1.0 - h[:m]
        g[:m] += -h[L:]
    return g


def smooth_max_obj(h, beta):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    vals, lags = all_lag_values(h)
    m = float(np.max(vals))
    if beta < 1e-6:
        return m
    shifted = beta * (vals - m)
    shifted = np.clip(shifted, -700, 0)
    r = float(np.sum(np.exp(shifted)))
    return float(m + np.log(r) / beta)


def smooth_max_jac(h, beta):

    n = h.size
    dx = 2.0 / float(n)

def weighted_lag_obj(x, weights, lags, n):
    h = x
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    obj = 0.0
    g = np.zeros(n, dtype=float)
    for i, (w, l) in enumerate(zip(weights, lags)):
        if w < 1e-10:
            continue
        L = abs(int(l))
        if L == 0:
            ac = float(np.dot(h, h))
            v = (total - ac) * dx
            obj += w * v
            g += w * (1.0 - 2.0 * h) * dx
        else:
            m = n - L
            ac = float(np.dot(h[:m], h[L:]))
            if l > 0:
                v = (pref[m] - ac) * dx
            else:

            obj += w * v
            if l > 0:

                g[L:] += -w * dx * h[:m]
            else:
                g[L:] += w * dx * (1.0 - h[:m])
                g[:m] += -w * dx * h[L:]
    return obj, g

def iterative_reweight(h0, n, time_limit, max_rounds=12, top_k=40):

    best_h = h.copy()

            break

            best_h = h.copy()

        top_v = float(np.max(vals))
        shifted = (vals - top_v) * 8.0

        w_new /= w_new.sum()

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, 0.5, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def generate_starts(n, parent, best_h, best_n, rng):

    temp = 0.002

    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        L = int(rng.randint(2, min(10, n) + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)

        if c_try < best_c:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        elif c_try - cur_c < temp:
            if rng.rand() < np.exp(-(c_try - cur_c) / temp):
                cur_h = h_try
                cur_c = c_try
        temp = max(temp * 0.997, 1e-7)
        step = max(0.005, step * 0.9998)

    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)

    # Get parent
    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        best_h = project_feasible(parent, parent_n)
        best_c = compute_c5(best_h)
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Optimize at parent resolution (200s)
    p1_end = min(hard_end - 600.0, time.time() + 200.0)
    if 2 <= best_n <= 320 and time.time() < p1_end - 10.0:
        for beta in (20, 60, 150):
            if time.time() > p1_end - 5.0:
                break
            h2 = optimize_smooth(best_h, best_n, beta, maxiter=50)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        # Active set refinement
        h2, c2 = refine_active(best_h, best_n, p1_end, seq=(24, 48, 80), maxiter=50)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 2: Multi-resolution search (500s)
    target_ns = [best_n + 48, best_n + 96, 256, 320, 384, 192]
    # Deduplicate and sort by priority (closer to best_n first, then higher)
    seen = set([best_n])
    target_ns = [x for x in target_ns if x not in seen and 48 <= x <= 420]
    target_ns = list(set(target_ns))
    target_ns.sort(key=lambda x: abs(x - best_n) + 0.001 * x)
    target_ns = target_ns[:5]

    p2_end = min(hard_end - 300.0, time.time() + 500.0)
    per_res = max(60.0, (p2_end - time.time()) / max(1, len(target_ns)))

    for nn in target_ns:
        nn = int(nn)
        if time.time() > p2_end - 20.0:
            break
        slot_end = min(p2_end, time.time() + per_res)
        starts = generate_starts(nn, parent, best_h, best_n, rng)

        for j, s in enumerate(starts[:3]):
            if time.time() > slot_end - 5.0:
                break
            sub_end = min(slot_end, time.time() + (slot_end - time.time()) / max(1, 3 - j))
            # Smooth max first
            if time.time() < sub_end - 30.0:
                h2 = optimize_smooth(s, nn, 40, maxiter=40)
            else:
                h2 = s
            c2 = compute_c5(h2)
            # Iterative reweighting
            if time.time() < sub_end - 10.0:
                h2, c2 = iterative_reweight(h2, nn, sub_end - 2.0, max_rounds=6, top_k=30)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
                best_n = nn

    # Phase 3: Focused refinement at best resolution (280s)
    p3_end = min(hard_end - 60.0, time.time() + 280.0)
    if 2 <= best_n <= 400 and time.time() < p3_end - 20.0:
        # Higher beta smooth
        for beta in (80, 200, 400):
            if time.time() > p3_end - 30.0:
                break
            h2 = optimize_smooth(best_h, best_n, beta, maxiter=50)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2
        # Active set with many lags
        h2, c2 = refine_active(best_h, best_n, p3_end - 20.0,
                               seq=(32, 64, 128), maxiter=60)
        if c2 < best_c:
            best_c = c2
            best_h = h2
        # Iterative reweighting again
        if time.time() < p3_end - 10.0:
            h2, c2 = iterative_reweight(best_h, best_n, p3_end - 2.0,
                                        max_rounds=8, top_k=50)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 4: Local search polish (remaining time)
    if time.time() < hard_end - 10.0 and 2 <= best_n <= 400:
        ls_end = hard_end - 5.0
        h2, c2 = local_search(best_h, best_n, ls_end, seed + 7777, step=0.015, max_trials=5000)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Final projection and check
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)

    # Verify constraints
    assert best_h.size == best_n
    assert abs(float(np.sum(best_h)) - 0.5 * best_n) < 1e-8, f"Sum constraint violated: {np.sum(best_h)} vs {0.5*best_n}"
    assert float(np.min(best_h)) >= -1e-12
    assert float(np.max(best_h)) <= 1.0 + 1e-12

    return (best_h, float(best_c), int(best_n))
