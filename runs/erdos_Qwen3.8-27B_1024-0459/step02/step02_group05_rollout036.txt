import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_ac(h):
    return np.correlate(h, h, mode='full')

def compute_min_ac(h):
    return np.min(compute_ac(h))

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    # Keep the top n/2 values as 1
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[indices[:n//2]] = 1.0
    return h_bin

def repair(h, n):
    """Ensure sum(h) = n/2 and 0<=h<=1."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    else:
        mask = h > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

        i_old = ones[rng.randint(len(ones))]
        j_new = zeros[rng.randint(len(zeros))]
        i_new = ones[rng.randint(len(ones))]
        j_old = zeros[rng.randint(len(zeros))]
        if i_old == i_new or j_old == j_new:
            continue
        h[i_old] = 0.0
        h[j_new] = 1.0
        h[i_new] = 1.0
        h[j_old] = 0.0
    return h

def multi_start_binary_sa(n, rng, time_limit, seed_offset):
    best_h = None
    best_c5 = 1e10
    t0 = time.time()
    n_starts = 8
    for s in range(n_starts):
        if time.time() - t0 > time_limit - 10:
            break
        if s == 0:
            h0 = np.zeros(n)
            h0[rng.choice(n, n // 2, replace=False)] = 1.0
        elif s == 1:
            h0 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4)
            if n % 4 != 0:
                h0 = make_binary(h0, n)
        elif s == 2:
            h0 = np.tile([1.0, 0.0], n // 2)
        elif s == 3:
            h0 = np.zeros(n)
            for i in range(n):
                h0[i] = 1.0 if ((i * (i + 1) // 2) % n) < n // 2 else 0.0
            h0 = make_binary(h0, n)
        else:
            h0 = np.zeros(n)
            h0[rng.choice(n, n // 2, replace=False)] = 1.0
        sub_limit = min(60, (time_limit - (time.time() - t0)) / (n_starts - s))
        h_best, c5_best = binary_sa_fast(h0, n, 50000, 0.8, 0.002, sub_limit, rng)
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        n = len(h_parent)
        h = h_parent.copy()
    else:
        n = 160
        h = np.ones(n) * 0.5

    h = project_sum(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Smoothed gradient descent (600s budget)
    if time_left() > 300:
        t_phase1 = min(600, time_left() - 200)
        betas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
        h_g, c5_g = smooth_descent(h, n, t_phase1, betas, base_lr=0.02)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 2: Binary steepest descent + perturbation cycles (300s budget)
    if time_left() > 200:
        h_bin = make_binary(best_h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        t_phase2 = min(300, time_left() - 100)
        t0 = time.time()
        h_cur = h_bin.copy()
        c5_cur = compute_c5(h_cur)
        cycles = 0
        while time.time() - t0 < t_phase2:
            h_new, c5_new = binary_steepest_descent(h_cur, n, max_rounds=50)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_cur = h_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
            n_perturb = 2 + (cycles % 4)
            h_cur = perturb_binary(h_cur, n, n_perturb, rng)
            c5_cur = compute_c5(h_cur)
            cycles += 1
            if cycles > 2000:
                break

    # Phase 3: Binary SA at current n (150s budget)
    if time_left() > 100:
        t_phase3 = min(150, time_left() - 80)
        h_sa, c5_sa = multi_start_binary_sa(n, rng, t_phase3, seed)
        if h_sa is not None and c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Multi-resolution exploration (200s budget)
    if time_left() > 150:
        n_trials = [120, 140, 150, 170, 180, 200, 220, 240, 256]
        t0 = time.time()
        for n_try in n_trials:
            if time.time() - t0 > 180 or time_left() < 80:
                break
            if n_try % 2 != 0:
                n_try += 1
            if n_try < 40:
                continue
            h_interp = interpolate_to_n(best_h, n_try)
            h_interp = project_sum(h_interp, n_try)
            sub_t = min(15, time_left() - 80)
            if sub_t < 5:
                break
            # Quick gradient descent at this resolution
            betas_quick = [10, 50, 200]
            h_rq, c5_rq = smooth_descent(h_interp, n_try, sub_t, betas_quick, base_lr=0.015)
            if c5_rq < best_c5:
                best_c5 = c5_rq
                best_h = h_rq.copy()
                n = n_try
            # Binary version
            h_b = make_binary(best_h, n_try)
            c5_b = compute_c5(h_b)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                n = n_try

    # Phase 5: Final polish (remaining time)
    if time_left() > 30:
        t_polish = min(60, time_left() - 10)
        betas_final = [200, 500, 2000, 5000]
        h_fin, c5_fin = smooth_descent(best_h, n, t_polish, betas_final, base_lr=0.01)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    if time_left() > 20:
        h_bin2 = make_binary(best_h, n)
        c5_bin2 = compute_c5(h_bin2)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
        # One more round of steepest descent
        h_sd, c5_sd = binary_steepest_descent(best_h, n, max_rounds=30)
        if c5_sd < best_c5:
            best_c5 = c5_sd
            best_h = h_sd.copy()

    # Final feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum(best_h, n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
