def projected_subgradient(h, n, alpha=0.01, n_steps=5000, target_sum=None):
    """Subgradient descent with projection"""
    if target_sum is None:
        target_sum = n / 2.0
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):

        c5, grad, corr = compute_c5_and_grad_max(h, n_active=3)

        h_new = np.clip(h_new, 0, 1)
        diff = target_sum - np.sum(h_new)
