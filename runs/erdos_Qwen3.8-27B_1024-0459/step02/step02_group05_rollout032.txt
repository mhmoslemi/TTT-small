def interp_to(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def slsqp_solve(n, h_init, beta, max_iter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        if beta < 1e7:
            sm = beta * (corr - cmax)
            sm = np.clip(sm, -700, 700)
            cmax += (1.0 / beta) * np.log(np.sum(np.exp(sm)))
        return cmax
    
    try:
        res = minimize(obj_fn, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_out = make_feasible(res.x, n)
        c5 = compute_c5(h_out)
        return h_out, c5
    except Exception:
        return None, None

def smooth_anneal(n, h_init, beta_list, max_iter_per_beta):
    current = h_init.copy()
    best_h = current.copy()
    best_c5 = compute_c5(current)
    for beta in beta_list:
        h_opt, c5_opt = slsqp_solve(n, current, beta, max_iter_per_beta, 1e-13)
        if h_opt is not None:
            current = h_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        else:
            current = best_h.copy()  # reset to best if failed
    return best_h, best_c5

def transition_accept(h_cur, c5_cur, h_try, c5_try, T):
    if c5_try <= c5_cur:
        return True
    delta = c5_try - c5_cur
    if T < 1e-15:
        return False
    return np.random.random() < np.exp(-delta / T)

def continuous_sa(h_init, n, n_steps, T_start, T_end, base_step, rng):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    
    for s in range(n_steps):
        T = T_start * (T_end / T_start) ** (s / max(n_steps - 1, 1))
        h_try = h.copy()
        k = rng.randint(1, 5)
        idxs = rng.choice(n, k, replace=False)
        for i in idxs:
            h_try[i] += rng.gauss(0, base_step)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        
        if transition_accept(h, c5_cur, h_try, c5_try, T):
            h = h_try
            c5_cur = c5_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
    
    return best_h, best_c5
