def m_sequence_init(n, rng):
    """Generate an m-sequence-like initialization using LFSR."""
    # For n up to 256, use a simple LFSR
    # Find a primitive polynomial of appropriate degree
    # For n=256, degree=8, polynomial x^8+x^4+x^3+x^2+1 (0x1D)
    if n <= 256:
        degree = int(np.log2(n))
        if 2**degree < n:
            degree += 1
    else:
        degree = int(np.log2(n))
    
    # Simple LFSR with a common primitive polynomial
    # For degree 8: taps at positions 8,4,3,2
    # For degree 12: taps at positions 12,6,4,3
    # For degree 16: taps at positions 16,5,3,2
    
   Primitive_poly_map = {
        8: [8, 4, 3, 2],
        12: [12, 6, 4, 3],
        16: [16, 5, 3, 2],
        4: [4, 3],
    }
    
    taps = Primitive_poly_map.get(degree, [degree, degree//2, degree//3, 1])
    
    seq = []
    state = 1  # Non-zero initial state
    for _ in range(n):
        seq.append(1.0 if state & 1 else 0.0)
        # Shift and compute new bit
        feedback = 0
        for t in taps:
            feedback ^= (state >> (t-1)) & 1
        state = (state >> 1) | (feedback << (degree - 1))
    
    h = np.array(seq[:n])
    return make_feasible(h, n)

import numpy as np
from scipy.optimize import minimize
import time

_N_OPT = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_and_worst(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    worst_idx = np.argmax(corr)
    return np.max(corr) * dx, worst_idx, corr

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-13
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def thue_morse_init(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(tm, n)

def thue_morse_shifted(n, shift):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
    return make_feasible(tm, n)

def random_binary(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    target = n // 2
    if len(ones) > target:
        excess = len(ones) - target
        to_flip = rng.choice(ones, excess, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def _sum_con_fun(x):
    return np.sum(x) - _N_OPT / 2.0

def _sum_con_jac(x):
    return np.ones(_N_OPT)

def _smooth_max_obj(x, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    shifted[shifted > 50] = 50
    val = cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))
    return val

def _direct_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)

def smooth_max_optimize(h_init, n, beta_start=15.0, beta_end=250.0, n_steps=5, max_iter=600):
    global _N_OPT
    _N_OPT = n
    current = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(current)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    for idx in range(n_steps):
        t = idx / max(n_steps - 1, 1)
        beta = beta_start * (beta_end / beta_start) ** t
        try:
            res = minimize(_smooth_max_obj, current, args=(beta,), method='SLSQP',
                          bounds=bounds, constraints=con,
                          options={'maxiter': max_iter, 'ftol': 1e-14})
            if res.success or res.fun < 1e10:
                candidate = make_feasible(res.x, n)
                c5_val = compute_c5(candidate)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    current = candidate
        except Exception:
            pass
    return current, best_c5

def simulated_annealing(h_init, n, rng, t_start=0.08, t_end=0.001, n_iterations=300000, timeout=120.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    start_t = time.time()
    
    ones_list = np.where(h == 1.0)[0]
    zeros_list = np.where(h == 0.0)[0]
    worst_shifts = np.zeros(5, dtype=int)
    for wi in range(5):
        worst_shifts[wi] = 0
    
    for it in range(n_iterations):
        if time.time() - start_t > timeout:
            break
        progress = it / max(n_iterations - 1, 1)
        temp = t_start * (t_end / t_start) ** progress
        
        if len(ones_list) < 2 or len(zeros_list) < 2:
            break
        
        if it % 5000 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            worst_idx = np.argpartition(corr, -5)[-5:]
            worst_shifts = worst_idx - (n - 1)
        
        r = rng.rand()
        
        if r < 0.55:
            # 1-swap: targeted toward worst shifts
            if rng.rand() < 0.5:
                s = worst_shifts[rng.randint(5)]
                lo = max(0, -s)
                hi = min(n, n - s)
                if hi > lo:
                    i = int(ones_list[rng.randint(len(ones_list))])
                    # Try to find j = i + s in zeros
                    j_candidate = i + s
                    if 0 <= j_candidate < n and h[j_candidate] == 0.0:
                        j = j_candidate
                    else:
                        j = int(zeros_list[rng.randint(len(zeros_list))])
                    if h[i] != 1.0:
                        i = int(ones_list[rng.randint(len(ones_list))])
                    if h[j] != 0.0:
                        j = int(zeros_list[rng.randint(len(zeros_list))])
                else:
                    i = int(ones_list[rng.randint(len(ones_list))])
                    j = int(zeros_list[rng.randint(len(zeros_list))])
            else:
                i = int(ones_list[rng.randint(len(ones_list))])
                j = int(zeros_list[rng.randint(len(zeros_list))])
            
            if i == j:
                continue
            
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            
            delta = new_c5 - current_c5
            if delta < 0 or (temp > 1e-12 and rng.rand() < np.exp(-delta / temp)):
                current_c5 = new_c5
                # Update tracking
                ones_list = np.where(h == 1.0)[0]
                zeros_list = np.where(h == 0.0)[0]
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
                
        elif r < 0.85:
            # 2-swap
            i1 = int(ones_list[rng.randint(len(ones_list))])
            i2 = int(ones_list[rng.randint(len(ones_list))])
            j1 = int(zeros_list[rng.randint(len(zeros_list))])
            j2 = int(zeros_list[rng.randint(len(zeros_list))])
            
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            
            h[i1] = 0.0
            h[i2] = 0.0
            h[j1] = 1.0
            h[j2] = 1.0
            new_c5 = compute_c5(h)
            
            delta = new_c5 - current_c5
            if delta < 0 or (temp > 1e-12 and rng.rand() < np.exp(-delta / temp)):
                current_c5 = new_c5
                ones_list = np.where(h == 1.0)[0]
                zeros_list = np.where(h == 0.0)[0]
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                h[i1] = 1.0
                h[i2] = 1.0
                h[j1] = 0.0
                h[j2] = 0.0
        else:
            # Multi-flip (3-swap): flip 3 ones to 0 and 3 zeros to 1
            if len(ones_list) < 3 or len(zeros_list) < 3:
                continue
            idxs1 = rng.choice(len(ones_list), 3, replace=False)
            idxs2 = rng.choice(len(zeros_list), 3, replace=False)
            flips_1 = ones_list[idxs1]
            flips_2 = zeros_list[idxs2]
            if len(set(list(flips_1) + list(flips_2))) < 6:
                continue
            for idx in flips_1:
                h[idx] = 0.0
            for idx in flips_2:
                h[idx] = 1.0
            new_c5 = compute_c5(h)
            
            delta = new_c5 - current_c5
            if delta < 0 or (temp > 1e-12 and rng.rand() < np.exp(-delta / temp)):
                current_c5 = new_c5
                ones_list = np.where(h == 1.0)[0]
                zeros_list = np.where(h == 0.0)[0]
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                for idx in flips_1:
                    h[idx] = 1.0
                for idx in flips_2:
                    h[idx] = 0.0
    
    return best_h, best_c5

def greedy_binary_search(h_init, n, rng, max_trials=200000, stall_limit=25000, timeout=120.0):
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    ones = set(np.where(h == 1.0)[0].tolist())
    zeros = set(np.where(h == 0.0)[0].tolist())
    no_improve = 0
    start_t = time.time()
    ones_arr = np.array(list(ones))
    zeros_arr = np.array(list(zeros))
    
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(ones_arr[rng.randint(len(ones_arr))])
        j = int(zeros_arr[rng.randint(len(zeros_arr))])
        
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            ones.discard(i)
            zeros.add(i)
            zeros.discard(j)
            ones.add(j)
            ones_arr = np.array(list(ones))
            zeros_arr = np.array(list(zeros))
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return h, best_c5

def targeted_swap_search(h_init, n, rng, max_trials=100000, timeout=90.0):
    """Focus swaps on the worst few shifts."""
    h = np.round(h_init).astype(float)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    no_improve = 0
    stall_limit = 15000
    start_t = time.time()
    
    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if trial % 3000 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full')
            top_k = np.argpartition(corr, -6)[-6:]
        
        s = int(top_k[rng.randint(len(top_k))]) - (n - 1)
        
        # Find positions contributing to this shift
        lo = max(0, -s)
        hi = min(n, n - s)
        bad_i = []
        for i in range(lo, hi):
            if h[i] == 1.0 and h[i + s] == 0.0:
                bad_i.append(i)
        
        if len(bad_i) > 0:
            i = bad_i[rng.randint(len(bad_i))]
            # Find zero to swap with (prefer one not creating new bad pair)
            j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-13:
            best_c5 = c5_try
            no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    
    return h, best_c5

def continuous_local_search(h_init, n, rng, max_trials=30000, timeout=40.0):
    h = make_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    target = n / 2.0
    step_sizes = [0.005, 0.01, 0.03, 0.08]
    no_improve = 0
    stall_limit = 3000
    start_t = time.time()

    for trial in range(max_trials):
        if time.time() - start_t > timeout:
            break
        if no_improve > stall_limit:
            break

        h_try = h.copy()
        n_mod = rng.randint(1, 3)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-1, 1) * step_sizes[rng.randint(0, len(step_sizes))]
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = target - np.sum(h_try)
        if abs(diff) > 1e-12:
            mask = (h_try > 1e-4) & (h_try < 1.0 - 1e-4)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
            if abs(target - np.sum(h_try)) > 1e-8:
                continue
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-13:
            h = h_try
            best_c5 = c5_try
            no_improve = 0
        else:
            no_improve += 1
    return h, best_c5

def final_slsqp_polish(h_init, n, max_iter=3000):
    global _N_OPT
    _N_OPT = n
    h0 = make_feasible(h_init.copy(), n)
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': _sum_con_fun, 'jac': _sum_con_jac}
    try:
        res = minimize(_direct_obj, h0, method='SLSQP',
                      bounds=bounds, constraints=con,
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h_init, compute_c5(h_init)
