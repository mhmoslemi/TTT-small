import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h_proj = np.clip(h - t, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-14:
        mask = (h_proj > 1e-15) & (h_proj < 1.0 - 1e-15)
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
        if abs(target - h_proj.sum()) > 1e-14:
            idx = np.argmin(np.maximum(h_proj, 1.0 - h_proj))
            h_proj[idx] += target - h_proj.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def threshold_gradient(h, dx, T, top_k=30):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    excess = np.maximum(0.0, vals - T)
    if excess.max() < 1e-14:
        return np.zeros(n), 0.0
    k_active = np.where(excess > 1e-12)[0]
    if len(k_active) > top_k:
        top_idx = np.argsort(-excess)[:top_k]
        k_active = k_active[top_idx]
    total_excess_sq = np.sum(excess[k_active] ** 2)
    grad = np.zeros(n)
    center = n - 1
    for k in k_active:
        w = 2.0 * excess[k] * dx
        s = k - center
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            m = -s
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:n - m]
    return grad, total_excess_sq

def interp_h(src, n_dst):
    n_src = len(src)
    if n_src == n_dst:
        return project_feasible(src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, src)
    return project_feasible(h_dst, n_dst)

def pgd_optimize(h0, n, T_init, max_iter=5000, dt_limit=15.0, decay=0.999):
    h = h0.copy()
    T = T_init
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = dx * 0.5
    elapsed = 0.0
    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > dt_limit:
            break
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        if T - best_c5 < 1e-10:
            T = best_c5 - 1e-6
        grad, excess_sq = threshold_gradient(h, dx, T)
        if excess_sq < 1e-20:
            break
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        h_new = project_feasible(h - step * grad * n, n)
        c5_new = compute_c5(h_new)

        else:

    return h_final, compute_c5(h_final)

def slsqp_opt(h0, n, beta, max_iter=500):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -400.0, 0.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

    def con_eq(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP',
                   constraints=[{'type': 'eq', 'fun': con_eq}],
                   bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

        ones = np.where(h == 1.0)[0]

                                             dt_limit=min(15.0, remaining() * 0.5))
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    print("  Perturb+PGD: C5={:.8f}".format(best_c5))
            except Exception:
                pass

    # Final
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
