        if it % 50 == 49:
            alpha = min(0.05, alpha * 1.2)
    return best_h, best_c


def active_t_obj(x):
    return float(x[-1])


def active_t_obj_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[-1] = 1.0
    return j


def active_sum_eq(x):
    return float(np.sum(x[:-1]) - 0.5 * float(len(x) - 1))


def active_sum_eq_jac(x):
    j = np.zeros(len(x), dtype=float)
    j[:-1] = 1.0
    return j


def active_ineq(x):
    n = len(x) - 1
    dx = 2.0 / float(n)
    out = np.empty(len(AC_LAGS), dtype=float)
    h = x[:-1]
    for i, ll in enumerate(AC_LAGS):
        out[i] = float(x[-1]) - lag_value(h, int(ll)) * dx
    return out


def active_ineq_jac(x):
    n = len(x) - 1
    dx = 2.0 / float(n)
    m = len(AC_LAGS)
    J = np.zeros((m, n + 1), dtype=float)
    if m == 0:
        return J
    h = x[:-1]
    for i, ll in enumerate(AC_LAGS):
        g = lag_gradient_only(h, int(ll))
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def optimize_active_t(h0, active_lags, t0, maxiter=60, ftol=1e-13):
    global AC_LAGS
    h = np.asarray(h0, dtype=float)
    n = h.size
    AC_LAGS = [int(l) for l in active_lags]
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h
    x0[n] = float(t0)
    bounds = [(0.0, 1.0) for _ in range(n)]
    bounds.append((0.0, 1.0))
    constraints = [{"type": "eq", "fun": active_sum_eq, "jac": active_sum_eq_jac}]
    if len(AC_LAGS) > 0:
        constraints.append({"type": "ineq", "fun": active_ineq, "jac": active_ineq_jac})
    try:
        res = minimize(
            active_t_obj, x0, jac=active_t_obj_jac,
            method="SLSQP", bounds=bounds,
            constraints=constraints,
            options={"maxiter": int(maxiter), "ftol": float(ftol), "disp": False},
        )
        return res
    except Exception:
        return None


def refine_active(h0, n, time_limit, active_seq=(32, 64, 128), maxiter=50):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    for acnt in active_seq:
        if time.time() > time_limit - 3.0:
            break
        passes = 2 if int(acnt) <= 32 else 1
        for _ in range(passes):
            if time.time() > time_limit - 3.0:
                break
            vals = all_lag_values(h)
            current_c = float(np.max(vals))
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
            top_val = current_c
            margin = max(0.001, 0.01 * (top_val - 0.35))
            active_mask = vals >= top_val - margin
            active_indices = np.where(active_mask)[0]
            if len(active_indices) == 0:
                active_indices = np.array([np.argmax(vals)])
            n_lags = 2 * n - 1
            offset = n - 1
            active_lags = [(int(i) - offset) for i in active_indices]
            acnt_eff = min(int(acnt), len(active_lags))
            order = np.argsort(-vals[active_indices])
            active_lags = [active_lags[order[i]] for i in range(acnt_eff)]
            t0 = current_c + 1e-5
            res = optimize_active_t(h, active_lags, t0, maxiter=maxiter, ftol=1e-13)
            if res is not None:
                x = res.x
                if x is not None and x.size == len(h) + 1 and np.all(np.isfinite(x)):
                    h2 = project_feasible(x[:-1], n)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2.copy()
                    if c2 <= current_c + 1e-9:
                        h = h2
    return best_h, best_c


def gradient_guided_perturb(h0, n, time_limit, seed, max_trials=5000):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.002
    min_temp = 1e-7
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        vals = all_lag_values(cur_h)
        worst_idx = np.argmax(vals)
        n_lags = 2 * n - 1
        offset = n - 1
        worst_lag = int(worst_idx - offset)
        g = lag_gradient_only(cur_h, worst_lag)
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * 0.02 - 0.3 * g[idx]
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.015, max(temp * 1.1, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)


def push_borderline(h0, n, threshold_lo=0.05, threshold_hi=0.95):
    h = project_feasible(h0, n)
    h[(h > threshold_lo) & (h < 0.5)] = 0.0
    h[(h > 0.5) & (h < threshold_hi)] = 1.0
    return project_feasible(h, n)


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def make_variants(h, n, rng, count=5):
    base = project_feasible(h, n)
    out = [base.copy()]
    if n > 1:
        flip = base[::-1]
        out.append(project_feasible(0.5 * (base + flip), n))
        out.append(project_feasible(0.5 + 0.5 * (base - flip), n))
    for sigma in (1.0, 2.0, 3.0):
        length = min(2 * int(sigma) + 3, n)
        if length % 2 == 0:
            length -= 1
        length = max(3, length)
        half = length // 2
        k = np.arange(-half, half + 1, dtype=float)
        w = np.exp(-(k * k) / (2.0 * sigma * sigma))
        w /= w.sum()
        out.append(project_feasible(np.convolve(base, w, mode="same"), n))
    pert = base + rng.randn(n) * 0.03
    out.append(project_feasible(pert, n))
    if count is not None and len(out) > count:
        out = out[:count]
    return out


def full_pipeline(h0, n, time_limit, seed):
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    betas = (10, 50, 200, 500)
    for beta in betas:
        if time.time() > time_limit - 30.0:
            break
        maxit = 80 if beta <= 50 else 60
        h2, c2 = optimize_smooth_h(h, n, beta, maxiter=maxit, ftol=1e-12)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
        if c2 <= best_c + 1e-3:
            h = h2
        else:
            h = best_h
    if time.time() < time_limit - 20.0 and 2 <= n <= 250:
        h2, c2 = projected_gradient_optimize(best_h, n, 200, max_iter=200, time_limit=time_limit - 20.0)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    if time.time() < time_limit - 15.0 and 2 <= n <= 250:
        seq = (32, 64, 128)
        h2, c2 = refine_active(best_h, n, time_limit - 10.0, active_seq=seq, maxiter=55)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    if time.time() < time_limit - 10.0:
        h2, c2 = gradient_guided_perturb(best_h, n, time_limit - 3.0, seed, max_trials=4000)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    if time.time() < time_limit - 5.0 and 2 <= n <= 300:
        hp = push_borderline(best_h, n)
        cp = compute_c5(hp)
        if cp < best_c:
            best_h = hp.copy()
            best_c = cp
        h2, c2 = optimize_smooth_h(best_h, n, 300, maxiter=40, ftol=1e-12)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, float(best_c)


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 45.0)

    parent = get_parent_values()
    if parent is not None:
        parent_n = int(parent.size)
        raw_c = None
        if (parent.size > 0 and float(np.min(parent)) >= -1e-12
                and float(np.max(parent)) <= 1.0 + 1e-12
                and abs(float(np.sum(parent)) - 0.5 * parent.size) <= 1e-8):
            raw_h = parent.copy()
            raw_c = compute_c5(raw_h)
        proj = project_feasible(parent, parent_n)
        proj_c = compute_c5(proj)
        if raw_c is not None and raw_c <= proj_c + 1e-13:
            best_h = raw_h.astype(float).copy()
            best_c = raw_c
        else:
            best_h = proj
            best_c = proj_c
        best_n = parent_n
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5, dtype=float), best_n)
        best_c = compute_c5(best_h)

    primary_end = min(hard_end - 100.0, start_time + budget_s * 0.55)

    inits_primary = [best_h.copy()]
    for v in make_variants(best_h, best_n, rng, count=4):
        inits_primary.append(v)

    per_init = max(15.0, (primary_end - time.time()) / float(len(inits_primary)))
    per_init = min(per_init, 90.0)

    for j, h0 in enumerate(inits_primary):
        if time.time() > primary_end:
            break
        hh, cc = full_pipeline(h0, best_n, time.time() + per_init, seed + 100 * j)
        if cc < best_c:
            best_h = hh
            best_c = cc
            best_n = int(best_n)

    explore_end = min(hard_end - 80.0, start_time + budget_s * 0.80)
    explore_ns = []
    for nn in (128, 144, 176, 192, 200, 224):
        if nn != best_n:
            explore_ns.append(nn)

    per_explore = max(12.0, (explore_end - time.time()) / float(max(1, len(explore_ns))))
    per_explore = min(per_explore, 55.0)

    for nn in explore_ns:
        if time.time() > explore_end:
            break
        h0 = interpolate_to_n(best_h, nn)
        hh, cc = full_pipeline(h0, nn, time.time() + per_explore, seed + 5000 + 13 * nn)
        if cc < best_c:
            best_h = hh.copy()
            best_c = cc
            best_n = nn

    if time.time() < hard_end - 30.0 and 2 <= best_n <= 300:
        for beta in (300, 600, 1200):
            if time.time() > hard_end - 35.0:
                break
            h2, c2 = optimize_smooth_h(best_h, best_n, beta, maxiter=60, ftol=1e-13)
            if c2 < best_c:
                best_h = h2.copy()
                best_c = c2

    if time.time() < hard_end - 25.0 and 2 <= best_n <= 250:
        fin_seq = (48, 96, 160)
        h2, c2 = refine_active(best_h, best_n, hard_end - 20.0, active_seq=fin_seq, maxiter=70)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    if time.time() < hard_end - 15.0 and 2 <= best_n <= 300:
        h2, c2 = gradient_guided_perturb(best_h, best_n, hard_end - 5.0, seed + 99991, max_trials=6000)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    if time.time() < hard_end - 8.0 and 2 <= best_n <= 250:
        hp = push_borderline(best_h, best_n, 0.03, 0.97)
        cp = compute_c5(hp)
        if cp < best_c:
            best_h = hp.copy()
            best_c = cp
        h2, c2 = refine_active(best_h, best_n, hard_end - 3.0, active_seq=(64, 128), maxiter=50)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
