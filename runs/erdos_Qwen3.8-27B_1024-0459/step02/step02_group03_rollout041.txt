    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    # Iterative projection
    for _ in range(100):
        # Project onto sum = n/2
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h += diff / n
        # Project onto [0, 1]
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Adjust the most "adjustable" element
        slack = np.minimum(h, 1.0 - h)
        if diff > 0:
            # Need to increase: increase elements with most room
            order = np.argsort(-slack)
            for idx in order:
                room = 1.0 - h[idx]
                adj = min(diff, room)
                h[idx] += adj
                diff -= adj
                if abs(diff) < 1e-14:
                    break
        else:
            # Need to decrease: decrease elements with most room
            order = np.argsort(-h)
            for idx in order:
                adj = min(-diff, h[idx])
                h[idx] -= adj
                diff += adj
                if abs(diff) < 1e-14:
                    break
    return h

    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)

    total = np.sum(exps)

def subgradient_step(h, n, step_size, beta=100):
    """One subgradient step with projection"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = smooth_max_weights(corr, beta)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            n_eff = n - L
            if n_eff > 0:
                grad[:n_eff] += w * (1.0 - h[L:L+n_eff])
                grad[L:n] -= w * h[:n_eff]
        else:
            Lp = -L
            n_eff = n - Lp
            if n_eff > 0:
                grad[Lp:n] += w * (1.0 - h[:n_eff])
                grad[:n_eff] -= w * h[Lp:n]
    
    h_new = h - step_size * dx * grad
    h_new = project_feasible(h_new, n)
    return h_new

def corr_delta(h, j, delta):
    """Compute the change in corr when h[j] changes by delta"""
    n = len(h)
    new_cj = h[j] + delta
    # corr[k] = sum_i h[i]*(1-h[i+L]) where L = k-(n-1)
    # Terms involving h[j]:
    # For lag L > 0: h[j]*(1-h[j+L]) if j <= n-1-L, and h[j-L]*(1-h[j]) if j >= L
    # For lag L = 0: h[j]*(1-h[j])
    # For lag L < 0: similar
    
    corr = np.zeros(2*n-1)
    old_v = h[j]
    new_v = old_v + delta
    
    # Lag 0: h[j]*(1-h[j])
    # change = new_v*(1-new_v) - old_v*(1-old_v) = -delta*new_v + delta*old_v = -delta*(new_v - old_v + old_v - new_v)... 
    # = new_v - new_v² - old_v + old_v² = delta - (new_v² - old_v²) = delta - (new_v-old_v)(new_v+old_v) = delta - delta*(new_v+old_v)
    lag0_change = delta - delta * (new_v + old_v)
    corr[n-1] += lag0_change
    
    # For lag L > 0 (k = n-1+L):
    # Term 1: h[j]*(1-h[j+L]) if 0 <= j <= n-1-L, i.e., L <= n-1-j
    #   change = delta * (1-h[j+L])
    # Term 2: h[j-L]*(1-h[j]) if L <= j <= n-1, i.e., L <= j
    #   change = -delta * h[j-L]
    
    # For lag L < 0, let Lp = -L (k = n-1-Lp):
    # Term 1: h[j]*(1-h[j-Lp]) if Lp <= j <= n-1
    #   change = delta * (1-h[j-Lp])
    # Term 2: h[j+Lp]*(1-h[j]) if 0 <= j+Lp <= n-1, i.e., Lp <= n-1-j
    #   change = -delta * h[j+Lp]
    
    max_L = n - 1
    for L in range(1, max_L + 1):
        # Positive lag L
        if j <= n - 1 - L:  # h[j] appears as first factor
            corr[n-1+L] += delta * (1 - h[j+L])
        if j >= L:  # h[j] appears as second factor (1-h[j])
            corr[n-1+L] -= delta * h[j-L]
        
        # Negative lag -L (same as positive by symmetry of the problem? No, not quite)
        Lp = L
        if j >= Lp:  # h[j] appears as first factor
            corr[n-1-Lp] += delta * (1 - h[j-Lp])
        if j + Lp <= n - 1:  # h[j] appears as second factor
            corr[n-1-Lp] -= delta * h[j+Lp]
    
    return corr

    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 1: Coordinate descent polish on parent
    if remaining() > 40:
        print("Phase 1: Coordinate descent n={}".format(best_n))
        h_cd = coordinate_descent(best_h, best_n, n_sweeps=4)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CD: C5={:.8f}".format(best_c5))
        else:
            print("  CD: no improvement ({:.8f} vs {:.8f})".format(c5_cd, best_c5))

    # Phase 2: SLSQP continuation at parent resolution
    if remaining() > 60:
        print("Phase 2: SLSQP continuation n={}".format(best_n))
        cur = best_h.copy()
        for beta_val in [20, 50, 100, 200, 500, 1000]:
            if remaining() < 25:
                break
            try:
                h_new, c5_new = slsqp_optimize(cur, best_n, beta_val, max_iter=150)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  beta={}: C5={:.8f}".format(beta_val, best_c5))
                cur = h_new
            except Exception:
                break

    # Phase 3: Multi-start perturbations
    if remaining() > 80:
        print("Phase 3: Multi-start perturbations")
        for trial in range(4):
            if remaining() < 30:
                break
            h_pert = best_h.copy()
            n_pert = rng.randint(5, max(6, best_n // 6))
            pert_idx = rng.choice(best_n, size=n_pert, replace=False)
            h_pert[pert_idx] = np.clip(h_pert[pert_idx] + rng.uniform(-0.2, 0.2, n_pert), 0, 1)
            h_pert = project_feasible(h_pert, best_n)
            try:
                h_new, c5_new = slsqp_optimize(h_pert, best_n, 200, max_iter=150)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    print("  trial {}: C5={:.8f}".format(trial, best_c5))
            except Exception:
                pass

    # Phase 4: Higher resolutions
    higher_ns = [300, 400, 500, 600, 700]
    for n_try in higher_ns:
        if remaining() < 70:
            break
        print("Phase 4: n={}".format(n_try))
        h_interp = interp_h(best_h, n_try)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_try
            print("  interp: C5={:.8f}".format(best_c5))

        cur = h_interp.copy()
        for beta_val in [50, 150, 400]:
            if remaining() < 35:
                break
            try:
                h_new, c5_new = slsqp_optimize(cur, n_try, beta_val, max_iter=120)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_try
                    print("  n={}, beta={}: C5={:.8f}".format(n_try, beta_val, best_c5))
                cur = h_new
            except Exception:
                break

    # Phase 5: Coordinate descent on best (if n is small enough)
    if remaining() > 30 and best_n <= 400:
        print("Phase 5: Final CD n={}".format(best_n))
        h_cd = coordinate_descent(best_h, best_n, n_sweeps=3)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            print("  CD: C5={:.8f}".format(best_c5))

    # Phase 6: Binary rounding and local search
    if remaining() > 20:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_want = best_n // 2
        if n_ones > n_want:
            ones_idx = np.where(h_bin == 1)[0]
            vals = best_h[ones_idx]
            to_flip = ones_idx[np.argsort(vals)[:n_ones - n_want]]
            h_bin[to_flip] = 0.0
        elif n_ones < n_want:
            zeros_idx = np.where(h_bin == 0)[0]
            vals = best_h[zeros_idx]
            to_flip = zeros_idx[np.argsort(-vals)[:n_want - n_ones]]
            h_bin[to_flip] = 1.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            print("  Binary: C5={:.8f}".format(best_c5))

        if np.all((best_h == 0) | (best_h == 1)) and remaining() > 15:
            h_bs, c5_bs = binary_local_search(best_h, best_n, rng,
                                             max_rounds=150, max_trials_per_round=30)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()
            print("  Binary LS: C5={:.8f}".format(best_c5))

    # Final cleanup
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.8f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
