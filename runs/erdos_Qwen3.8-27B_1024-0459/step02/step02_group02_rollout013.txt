from scipy.optimize import minimize
import time

def compute_c5_fast(h):
    """Compute C5 using FFT for speed."""
    n = len(h)
    dx = 2.0 / n
    # Use direct correlate for small n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    """Compute full correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(h, n)

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(h, n)

def interp_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0, 2, len(h_src), endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def random_binary_init(n, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    ones_ct = int(h.sum())
    target = n // 2
    if ones_ct > target:
        flip = rng.choice(np.where(h == 1)[0], ones_ct - target, replace=False)
        h[flip] = 0
    elif ones_ct < target:
        flip = rng.choice(np.where(h == 0)[0], target - ones_ct, replace=False)
        h[flip] = 1
    return h

def binary_local_search(h0, n, rng, max_evals=2000000, time_limit=30.0):
    h = np.round(h0).copy()
    h = make_feasible(h, n)
    best_cmax = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    ones_list = list(ones_set)
    zeros_list = list(zeros_set)
    start_t = time.time()
    evals = 0
    stall = 0
    stall_limit = 120000
    while evals < max_evals and time.time() - start_t < time_limit:
        if not ones_list or not zeros_list:
            break
        if stall > stall_limit:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        cmax = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        evals += 1
        if cmax < best_cmax - 1e-12:
            best_cmax = cmax
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            stall = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    h = make_feasible(h, n)
    return h, best_cmax * (2.0 / n)

def binary_sa_search(h0, n, rng, time_limit=60.0, T0=0.3, T1=0.005, stall_limit=150000):
    h = np.round(h0).copy()
    h = make_feasible(h, n)
    best_cmax = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    cur_cmax = best_cmax
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    start_t = time.time()
    total_evals = 0
    stall = 0
    max_total = int(time_limit * 25000)
    while total_evals < max_total and time.time() - start_t < time_limit:
        if not ones_list or not zeros_list or stall > stall_limit:
            break
        progress = total_evals / max(max_total, 1)
        T = T0 * (T1 / T0) ** progress
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        cmax = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        total_evals += 1
        delta = cmax - cur_cmax
        if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
            cur_cmax = cmax
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if cmax < best_cmax - 1e-12:
                best_cmax = cmax
                stall = 0
            else:
                stall += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    h = make_feasible(h, n)
    return h, best_cmax * (2.0 / n)

def targeted_binary_search(h0, n, rng, max_evals=1000000, time_limit=40.0):
    h = np.round(h0).copy()
    h = make_feasible(h, n)
    best_cmax = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    start_t = time.time()
    evals = 0
    stall = 0
    stall_limit = 80000
    while evals < max_evals and time.time() - start_t < time_limit:
        if not ones_list or not zeros_list or stall > stall_limit:
            break
        corr_vec = np.correlate(h, 1.0 - h, mode='full')
        worst_idx = int(np.argmax(corr_vec))
        worst_shift = worst_idx - (n - 1)
        s = worst_shift
        bad_ones = []
        if s >= 0:
            for m in range(max(0, -s), min(len(ones_list), n - s)):
                if m in set(ones_list) and h[m] == 1.0 and m + s < n and h[m + s] == 0.0:
                    bad_ones.append(m)
        else:
            ls = -s
            for m in range(max(0, ls), min(len(ones_list), n)):
                if m in set(ones_list) and h[m] == 1.0 and m - ls >= 0 and h[m - ls] == 0.0:
                    bad_ones.append(m)
        if bad_ones:
            i = bad_ones[rng.randint(len(bad_ones))]
        else:
            i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0.0
        h[j] = 1.0
        cmax = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        evals += 1
        if cmax < best_cmax - 1e-12:
            best_cmax = cmax
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            stall = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
    h = make_feasible(h, n)
    return h, best_cmax * (2.0 / n)

def continuous_local_search(h0, n, rng, time_limit=60.0, step_sizes=None):
    h = make_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    target = n / 2.0
    if step_sizes is None:
        step_sizes = [0.005, 0.01, 0.02, 0.05, 0.1]
    start_t = time.time()
    stall = 0
    stall_limit = 20000
    while time.time() - start_t < time_limit and stall < stall_limit:
        h_try = h.copy()
        n_mod = rng.randint(2, 8)
        idxs = rng.choice(n, n_mod, replace=False)
        for idx in idxs:
            h_try[idx] += rng.uniform(-1, 1) * step_sizes[rng.randint(len(step_sizes))]
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = target - np.sum(h_try)
        if abs(diff) > 1e-12:
            mask = (h_try > 1e-5) & (h_try < 1.0 - 1e-5)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
            diff2 = target - np.sum(h_try)
            if abs(diff2) > 1e-10:
                stall += 1
                continue
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            h = h_try
            best_c5 = c5_try
            stall = 0
        else:
            stall += 1
    h = make_feasible(h, n)
    return h, compute_c5(h)

def subgradient_optimize(h0, n, time_limit=60.0, lr_start=0.02, lr_end=0.001):
    h = make_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    target = n / 2.0
    start_t = time.time()
    n_outer = 0
    n_max_outer = int(time_limit * 50)
    while time.time() - start_t < time_limit and n_outer < n_max_outer:
        corr_vec = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(np.max(corr_vec))
        c5_val = cmax * (2.0 / n)
        if c5_val < best_c5:
            best_c5 = c5_val
        progress = n_outer / max(n_max_outer, 1)
        lr = lr_start * (lr_end / lr_start) ** progress
        thresh = cmax - 0.5
        active = np.where(corr_vec >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(corr_vec))])
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                for m in range(n - k):
                    if m + k < n:
                        grad[m + k] += (1.0 - h[m])
                        grad[m] -= h[m + k]
            else:
                ls = -k
                for m in range(ls, n):
                    grad[m] += (1.0 - h[m - ls])
                    grad[m - ls] -= h[m]
        grad = grad / len(active)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - np.sum(h_new)
        if abs(diff) > 1e-12:
            mask = (h_new > 1e-6) & (h_new < 1.0 - 1e-6)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
            diff2 = target - np.sum(h_new)
            if abs(diff2) > 1e-8:
                h_new = h
        h = h_new
        n_outer += 1
    h = make_feasible(h, n)
    return h, compute_c5(h)

def slsqp_polish(h0, n, max_iter=2000):
    h = make_feasible(h0.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: float(np.max(np.correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full'))),
                       h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h, compute_c5(h)

def generate_inits(n, rng, h_parent=None):
    inits = []
    if h_parent is not None and len(h_parent) > 0:
        h_p = make_feasible(interp_to_n(h_parent, n), n)
        inits.append(h_p)
    inits.append(thue_morse_init(n))
    for _ in range(6):
        inits.append(random_binary_init(n, rng))
    for shift in [1, 2, 3, 5, 8, 13]:
        tm = np.zeros(n)
        for i in range(n):
            tm[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
        inits.append(make_feasible(tm, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    n = 256

    def remaining():
        return budget_s - (time.time() - start_time)

    h_parent = None
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
    except Exception:
        h_parent = None

    best_h = thue_morse_init(n)
    best_c5 = compute_c5(best_h)

    inits = generate_inits(n, rng, h_parent)

    # Evaluate all inits
    for h0 in inits:
        c5_val = compute_c5(h0)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h0.copy()

    # Phase 1: Binary local search from top inits
    sorted_inits = sorted(inits, key=lambda x: compute_c5(x))
    for h0 in sorted_inits[:5]:
        if remaining() < 20:
            break
        h_opt, c5_val = binary_local_search(h0, n, rng, max_evals=500000, time_limit=min(25, remaining() / 5))
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # Phase 2: Intensive binary local search from best
    if remaining() > 40:
        h_b, c5_b = binary_local_search(best_h, n, rng, max_evals=2000000, time_limit=min(120, remaining() - 30))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 3: Simulated annealing
    if remaining() > 50:
        h_sa, c5_sa = binary_sa_search(best_h, n, rng, time_limit=min(100, remaining() - 30), T0=0.4, T1=0.003)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Targeted binary search
    if remaining() > 40:
        h_t, c5_t = targeted_binary_search(best_h, n, rng, max_evals=1000000, time_limit=min(80, remaining() - 20))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 5: More binary search rounds with restarts
    n_rounds = 3
    for r in range(n_rounds):
        if remaining() < 50:
            break
        h_b, c5_b = binary_local_search(best_h, n, rng, max_evals=1000000, time_limit=min(60, remaining() / (n_rounds - r)))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        # Perturb and search again
        h_pert = best_h.copy()
        n_flip = rng.randint(5, 15)
        ones_idx = np.where(h_pert == 1)[0]
        zeros_idx = np.where(h_pert == 0)[0]
        for _ in range(n_flip):
            a = ones_idx[rng.randint(len(ones_idx))]
            b = zeros_idx[rng.randint(len(zeros_idx))]
            h_pert[a], h_pert[b] = h_pert[b], h_pert[a]
            rem_a = np.where(ones_idx == a)[0]
            rem_b = np.where(zeros_idx == b)[0]
            if len(rem_a) > 0:
                ones_idx = np.delete(ones_idx, rem_a[0])
            if len(rem_b) > 0:
                zeros_idx = np.delete(zeros_idx, rem_b[0])
            ones_idx = np.append(ones_idx, b)
            zeros_idx = np.append(zeros_idx, a)
        h_b2, c5_b2 = binary_local_search(h_pert, n, rng, max_evals=500000, time_limit=min(30, remaining() / (n_rounds - r)))
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 6: Subgradient optimization (continuous)
    if remaining() > 40:
        h_sg, c5_sg = subgradient_optimize(best_h, n, time_limit=min(60, remaining() - 20))
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 7: Continuous local search
    if remaining() > 40:
        h_c, c5_c = continuous_local_search(best_h, n, rng, time_limit=min(60, remaining() - 20),
                                            step_sizes=[0.003, 0.008, 0.02, 0.05])
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Phase 8: SLSQP polish
    if remaining() > 30:
        h_f, c5_f = slsqp_polish(best_h, n, max_iter=3000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 9: Final binary search from polished
    if remaining() > 30:
        h_bf, c5_bf = binary_local_search(best_h, n, rng, max_evals=1000000, time_limit=min(50, remaining() - 15))
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    # Phase 10: Final continuous polish
    if remaining() > 30:
        h_cf, c5_cf = continuous_local_search(best_h, n, rng, time_limit=min(40, remaining() - 10),
                                              step_sizes=[0.002, 0.005, 0.01, 0.03])
        if c5_cf < best_c5:
            best_c5 = c5_cf
            best_h = h_cf.copy()

    # Phase 11: Try n=512 if time permits
    if remaining() > 100:
        n2 = 512
        h2_init = make_feasible(interp_to_n(best_h, n2), n2)
        c5_2 = compute_c5(h2_init)
        h2_best = h2_init.copy()
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_best.copy()
        # Quick binary search at n=512
        if remaining() > 60:
            h2_b, c5_2b = binary_local_search(h2_best, n2, rng, max_evals=500000, time_limit=min(50, remaining() - 20))
            if c5_2b < best_c5:
                best_c5 = c5_2b
                best_h = h2_b.copy()
        if remaining() > 30:
            h2_c, c5_2c = continuous_local_search(h2_best, n2, rng, time_limit=min(30, remaining() - 10),
                                                  step_sizes=[0.005, 0.01, 0.03])
            if c5_2c < best_c5:
                best_c5 = c5_2c
                best_h = h2_c.copy()

    # Final feasibility check
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
