def subgradient_polish(h, deadline, max_iter=500, top_frac=0.05, min_k=5):
    """Subgradient method on exact max with line search"""
    n = h.size
    target = 0.5 * n
    best = project_density(h, target)
    best_c = evaluate_erdos(best)
    if n < 2:
        return best, best_c
    center = n - 1
    stall = 0
    for it in range(max_iter):
        if time.perf_counter() >= deadline:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        # Find active lags
        k_count = max(min_k, int(np.ceil(len(corr) * top_frac)))
        top_idx = np.argsort(corr)[-k_count:]
        top_vals = corr[top_idx]
        # Softmax weights
        dv = top_vals - top_vals.min()
        scale = max(1.0, dv.max() + 1e-10)
        w = np.exp(dv / scale)
        w /= w.sum()
        # Compute gradient
        g = np.zeros(n, dtype=float)
        for i, t in enumerate(top_idx):
            k = int(t) - center
            g += w[i] * corr_grad_lag(best, k)
        # Normalize
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g /= gnorm
        # Remove mean (to preserve sum constraint approximately)
        g -= np.mean(g)
        # Line search
        improved = False
        for step in (0.01, 0.03, 0.05, 0.1, 0.2, 0.4, 0.8, 1.5, 3.0):
            cand = project_density(best - step * g, target)
            c = evaluate_erdos(cand)
            if c < best_c - 1e-13:
                best = cand
                best_c = c
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
    best = project_density(best, target)
    return best, evaluate_erdos(best)

def project_density(x, target):
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    x = np.clip(x, 0.0, 1.0)

        return x

def slsqp_optimize(n, h0, deadline, seed=42):
    target = 0.5 * n
    h = project_density(h0, target)
    best_h = h.copy()
    best_c = evaluate_erdos(best_h)

    bounds = [(0.0, 1.0)] * n
    constraints = ({"type": "eq", "fun": sum_constraint, "jac": sum_constraint_jac},)

    if n <= 150:
        alphas = (20.0, 80.0, 250.0, 800.0)
        maxiter = 300
    elif n <= 400:
        alphas = (20.0, 80.0, 250.0)
        maxiter = 150
    else:
        alphas = (20.0, 60.0, 200.0)
        maxiter = 80

    for alpha in alphas:
        if time.perf_counter() >= deadline - 10.0:
            break
        h_curr = best_h.copy()
        try:
            res = minimize(
                lambda x: smooth_objective(x, n, alpha),
                h_curr,
                method="SLSQP",
                jac=lambda x: smooth_jacobian(x, n, alpha),
                bounds=bounds,
                constraints=constraints,
                options={"maxiter": maxiter, "ftol": 1e-12, "disp": False}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = project_density(res.x, target)
                c = evaluate_erdos(cand)
                if c < best_c:
                    best_h = cand
                    best_c = c
                else:
                    h_curr = best_h
        except Exception:
            pass
    return project_density(best_h, target), evaluate_erdos(best_h)

            best = h4
            best_c = c4

    rem = deadline - time.perf_counter()
    if rem > 4.0 and n <= 350:
        h5, c5 = triple_swap_search(best, time.perf_counter() + rem * 0.5, max_iter=200)
        if c5 < best_c:
            best = h5
            best_c = c5

    rem = deadline - time.perf_counter()
    if rem > 4.0:
        h6, c6 = pair_swap_search(best, time.perf_counter() + rem * 0.7, max_iter=1000, top_frac=0.03, min_k=8, n_pos=6, n_neg=6)
        if c6 < best_c:
            best = h6
            best_c = c6

    rem = deadline - time.perf_counter()
    if rem > 3.0:
        h7, c7 = subgradient_polish(best, time.perf_counter() + rem * 0.85, max_iter=300, top_frac=0.03, min_k=10)
        if c7 < best_c:
            best = h7
            best_c = c7

    return project_density(best, target), evaluate_erdos(best)


def binarize_push(h, thresh=0.1):
    n = h.size
    h2 = h.copy()
    changed = False
    for i in range(n):
        if h2[i] < thresh:
            h2[i] = 0.0
            changed = True
        elif h2[i] > 1.0 - thresh:
            h2[i] = 1.0
            changed = True
    if not changed:
        return h
    return project_density(h2, 0.5 * n)


def run(seed=42, budget_s=1300, **kwargs):
    start = time.perf_counter()
    budget = float(budget_s) if budget_s is not None else 1300.0
    deadline = start + max(1.0, budget - 15.0)
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = float("inf")
    best_n = 0

    parent = None
    try:
        raw = np.asarray(initial_h_values, dtype=float)
        if raw.ndim == 1 and raw.size >= 2:
            parent = np.nan_to_num(raw, nan=0.5, posinf=1.0, neginf=0.0).astype(float, copy=True)
            parent = np.clip(parent, 0.0, 1.0)
    except Exception:
        parent = None

    if parent is not None:
        np_ = int(parent.size)
        pt = 0.5 * np_
        pp = project_density(parent, pt)
        cp = evaluate_erdos(pp)
        best_h = pp.copy()
        best_c = cp
        best_n = np_

    if best_h is None:
        n0 = 90
        x = (np.arange(n0) + 0.5) * (2.0 / n0)
        h_init = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 5 * x / 2 + 1.0)
        best_h = project_density(h_init, 0.5 * n0)
        best_c = evaluate_erdos(best_h)
        best_n = n0

    # Phase 1: DE at low resolution for global exploration
    de_n = 90
    de_budget = min(120.0, (deadline - time.perf_counter()) * 0.15)
    if de_budget > 20.0:
        global _DE_N
        _DE_N = de_n
        bounds_de = [(0.0, 1.0)] * de_n
        de_dl = time.perf_counter() + de_budget * 0.85
        try:
            t0 = time.perf_counter()
            res_de = differential_evolution(
                de_func, bounds_de,
                seed=seed, maxiter=500, popsize=35,
                tol=1e-6, mutation=(0.3, 0.8), recombination=0.7,
                polish=False, init="latinhypercube",
                updating="deferred", workers=1
            )
            if res_de is not None and res_de.x is not None:
                h_de = project_density(res_de.x, 0.5 * de_n)
                c_de = evaluate_erdos(h_de)
                if c_de < best_c:
                    best_h_de = h_de.copy()
                    best_c_de = c_de
                else:
                    h_de = best_h
                    c_de = best_c
                # Upsample and do local search
                h_up = upsample_h(h_de, min(300, best_n if best_n >= 300 else 300))
                # Quick local search at this resolution
                nl = h_up.size
                rem_de = deadline - time.perf_counter()
                if rem_de > 30.0:
                    h_lo, c_lo = full_local_search(h_up, time.perf_counter() + min(60.0, rem_de * 0.3))
                    if c_lo < best_c:
                        best_h = h_lo
                        best_c = c_lo
                        best_n = nl
        except Exception:
            pass

    # Phase 2: SLSQP at parent resolution
    base_n = best_n if best_n >= 100 else 180
    slsqp_budget = min(250.0, (deadline - time.perf_counter()) * 0.3)
    if slsqp_budget > 30.0:
        h0 = upsample_h(best_h, base_n)
        h_out, c_out = slsqp_optimize(base_n, h0, time.perf_counter() + slsqp_budget, seed=seed)
        if c_out < best_c:
            best_h = h_out
            best_c = c_out
            best_n = base_n

    # Phase 3: Try a second resolution
    alt_n = base_n + 60 if base_n < 400 else base_n + 90
    alt_n = min(alt_n, 540)
    alt_budget = min(150.0, (deadline - time.perf_counter()) * 0.2)
    if alt_budget > 30.0:
        h0_alt = upsample_h(best_h, alt_n)
        h_out2, c_out2 = slsqp_optimize(alt_n, h0_alt, time.perf_counter() + alt_budget, seed=seed + 1)
        if c_out2 < best_c:
            best_h = h_out2
            best_c = c_out2
            best_n = alt_n

    # Phase 4: Intensive local search at best resolution
    rem = deadline - time.perf_counter()
    if rem > 30.0 and best_h is not None:
        h_loc, c_loc = full_local_search(best_h, time.perf_counter() + rem * 0.7)
        if c_loc < best_c:
            best_h = h_loc
            best_c = c_loc
            best_n = best_h.size

    # Phase 5: Try binarization + re-optimize
    rem = deadline - time.perf_counter()
    if rem > 20.0 and best_h is not None and best_n <= 400:
        h_bin = binarize_push(best_h, thresh=0.08)
        c_bin = evaluate_erdos(h_bin)
        if c_bin < best_c:
            best_h = h_bin
            best_c = c_bin
        # Re-smooth from binary
        if rem > 15.0:
            h_unb = binarize_push(best_h, thresh=0.5)
            h_unb = 0.3 * h_unb + 0.7 * best_h
            h_unb = project_density(h_unb, 0.5 * best_n)
            h_rs, c_rs = full_local_search(h_unb, time.perf_counter() + min(30.0, rem * 0.5))
            if c_rs < best_c:
                best_h = h_rs
                best_c = c_rs

    # Phase 6: Final intensive pair sweeps with annealing
    rem = deadline - time.perf_counter()
    if rem > 15.0 and best_h is not None and best_n <= 500:
        h_fin, c_fin = pair_swap_search(best_h, time.perf_counter() + rem * 0.4,
                                         max_iter=2000, top_frac=0.03, min_k=8,
                                         n_pos=8, n_neg=8, temperature=0.001)
        if c_fin < best_c:
            best_h = h_fin
            best_c = c_fin
        rem2 = deadline - time.perf_counter()
        if rem2 > 10.0:
            h_f2, c_f2 = subgradient_polish(best_h, time.perf_counter() + rem2 * 0.7,
                                             max_iter=400, top_frac=0.03, min_k=10)
            if c_f2 < best_c:
                best_h = h_f2
                best_c = c_f2

    # Phase 7: One more pass at potentially higher resolution
    rem = deadline - time.perf_counter()
    hi_n = min(best_n * 2, 720)
    if rem > 60.0 and hi_n > best_n + 50:
        h_hi = upsample_h(best_h, hi_n)
        h_hi_out, c_hi = slsqp_optimize(hi_n, h_hi, time.perf_counter() + rem * 0.3, seed=seed + 100)
        if rem > 20.0:
            h_hi2, c_hi2 = full_local_search(h_hi_out, time.perf_counter() + rem * 0.4)
            if c_hi2 < best_c:
                best_h = h_hi2
                best_c = c_hi2
                best_n = hi_n
            elif c_hi < best_c:
                best_h = h_hi_out
                best_c = c_hi
                best_n = hi_n

    # Final verification
    if best_h is None:
        fallback = project_density(np.full(45, 0.5), 22.5)
        return fallback, evaluate_erdos(fallback), 45

    final_h = project_density(best_h, 0.5 * int(best_h.size))
    final_c = evaluate_erdos(final_h)
    return final_h, float(final_c), int(final_h.size)
