import numpy as np
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible_binary(h, n, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = np.round(np.clip(h, 0.0, 1.0)).astype(float)
    target = n // 2
    ones = int(np.sum(h))
    if ones > target:
        excess = ones - target
        one_idx = np.where(h == 1.0)[0]
        to_flip = rng.choice(one_idx, excess, replace=False)
        h[to_flip] = 0.0
    elif ones < target:
        deficit = target - ones
        zero_idx = np.where(h == 0.0)[0]
        to_flip = rng.choice(zero_idx, deficit, replace=False)
        h[to_flip] = 1.0
    return h

def make_feasible_cont(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def simulated_annealing_binary(h_init, n, rng, total_time=100.0, target_c5=None):
    """Simulated annealing for binary sequences."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    dx = 2.0 / n
    T_start = 8.0 * dx
    T_end = 0.02 * dx
    
    start_t = time.time()
    no_improve_count = 0
    improve_threshold = 5000
    n_iter = 0
    n_restarts = 0
    max_restarts = 20
    
    while time.time() - start_t < total_time:
        progress = (time.time() - start_t) / total_time
        T = T_start * (T_end / T_start) ** progress
        
        if target_c5 is not None and best_c5 <= target_c5:
            break
        
        # Choose move size
        r = rng.random()
        if r < 0.65:
            k = 1
        elif r < 0.88:
            k = 2
        elif r < 0.97:
            k = 3
        else:
            k = 4 + rng.randint(0, 3)
        
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        
        if len(ones_idx) < k or len(zeros_idx) < k:
            k = min(k, len(ones_idx), len(zeros_idx))
            if k == 0:
                break
        
        flip_ones = rng.choice(ones_idx, k, replace=False)
        flip_zeros = rng.choice(zeros_idx, k, replace=False)
        
        h[flip_ones] = 0.0
        h[flip_zeros] = 1.0
        
        corr = compute_corr(h)
        new_c5 = np.max(corr) * dx
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve_count = 0
            no_improve_count = 0
        else:
            delta = new_c5 - current_c5
            if delta > 0:
                accept_prob = math.exp(-delta / T)
            else:
                accept_prob = 1.0
            if rng.random() < accept_prob:
                current_c5 = new_c5
                no_improve_count = 0
            else:
                h[flip_ones] = 1.0
                h[flip_zeros] = 0.0
                no_improve_count += 1
        
        # Reheat if stuck
        if no_improve_count > improve_threshold and n_restarts < max_restarts:
            n_restarts += 1
            no_improve_count = 0
            # Big perturbation
            n_perturb = max(4, n // 32)
            p_ones = np.where(h == 1.0)[0]
            p_zeros = np.where(h == 0.0)[0]
            p_flip_o = rng.choice(p_ones, min(n_perturb, len(p_ones)), replace=False)
            p_flip_z = rng.choice(p_zeros, min(n_perturb, len(p_zeros)), replace=False)
            nn = min(len(p_flip_o), len(p_flip_z))
            h[p_flip_o[:nn]] = 0.0
            h[p_flip_z[:nn]] = 1.0
            current_c5 = compute_c5(h)
            # Reheat temperature
            progress = max(0, progress - 0.3)
            T = T_start * (T_end / T_start) ** progress
        
        n_iter += 1
    
    return best_h, best_c5

def continuous_refine(h_init, n, rng, total_time=60.0):
    """Continuous gradient-based refinement starting from a binary solution."""
    h = make_feasible_cont(h_init.copy().astype(float), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_t = time.time()
    
    learning_rate = 0.05
    n_iter = 0
    
    while time.time() - start_t < total_time:
        corr = compute_corr(h)
        max_val = np.max(corr)
        worst_idx = np.argmax(corr)
        worst_shift = worst_idx - (n - 1)
        
        # Compute gradient of the worst correlation value
        s = worst_shift
        grad = np.zeros(n)
        # corr[worst_idx] = sum_i h[i] * (1-h[i-s]) for valid i
        # d/dh[j] = (1-h[j-s]) for j where j-s in [0,n-1]
        #           - h[j+s] for j where j+s in [0,n-1]
        if s >= 0:
            for j in range(n - s):
                grad[j + s] += (1.0 - h[j])
                grad[j] -= h[j + s]
        else:
            for j in range(-s, n):
                grad[j] += (1.0 - h[j + s])
                grad[j + s] -= h[j]
        
        # Also add smooth max contribution (soften)
        beta = 100.0
        shift_corr = beta * (corr - max_val)
        shift_corr[shift_corr > 50] = 50
        weights = np.exp(shift_corr)
        weights /= np.sum(weights)
        
        # Gradient of smooth max is weighted sum of individual gradients
        soft_grad = np.zeros(n)
        for idx in range(len(corr)):
            w = weights[idx]
            if w < 1e-10:
                continue
            sh = idx - (n - 1)
            if sh >= 0:
                for j in range(n - sh):
                    soft_grad[j + sh] += w * (1.0 - h[j])
                    soft_grad[j] -= w * h[j + sh]
            else:
                for j in range(-sh, n):
                    soft_grad[j] += w * (1.0 - h[j + sh])
                    soft_grad[j + sh] -= w * h[j]
        
        # Step
        h_new = h - learning_rate * soft_grad * (2.0 / n)
        h_new = make_feasible_cont(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            learning_rate = min(learning_rate * 1.02, 0.2)
        else:
            h = h_new
            learning_rate *= 0.95
        
        n_iter += 1
        if n_iter % 1000 == 0 and learning_rate < 1e-4:
            break
    
    return best_h, best_c5

def greedy_local_search_binary(h_init, n, rng, max_trials=100000, timeout=60.0):

    h = h_init.copy()

    zeros = list(np.where(h == 0.0)[0])

            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try a pair swap
        i = ones[rng.randint(0, len(ones))]

        h_polish = best_h_overall.copy()
        t_alloc = min(80.0, remaining() - 40.0)
        h_g, c5_g = greedy_local_search_binary(h_polish, n_polish, rng, max_trials=200000, timeout=t_alloc)
        update_best(h_g, c5_g, n_polish)
    
    # === PHASE 4: Continuous refinement at n=256 ===
    if remaining() > 80:
        n256 = 256
        # Start from best binary at n=256
        h_cont_init = thue_morse_binary(n256)
        if best_h_overall is not None and best_n_overall == n256:
            h_cont_init = best_h_overall.copy()
        elif h_parent is not None:
            h_cont_init = make_feasible_cont(interp_to_resolution(h_parent, n256), n256)
        
        t_alloc = min(100.0, remaining() - 50.0)
        h_c, c5_c = continuous_refine(h_cont_init, n256, rng, total_time=t_alloc)
        update_best(h_c, c5_c, n256)
    
    # === PHASE 5: More SA rounds if time permits ===
    if remaining() > 80:
        # Try another SA round from a perturbed version of best
        n_try = best_n_overall
        h_base = best_h_overall.copy()
        h_pert = h_base.copy()
        n_pert = max(3, n_try // 16)
        o_idx = np.where(h_pert == 1.0)[0]
        z_idx = np.where(h_pert == 0.0)[0]
        fo = rng.choice(o_idx, min(n_pert, len(o_idx)), replace=False)
        fz = rng.choice(z_idx, min(n_pert, len(z_idx)), replace=False)
        nn = min(len(fo), len(fz))
        h_pert[fo[:nn]] = 0.0
        h_pert[fz[:nn]] = 1.0
        h_pert = make_feasible_binary(h_pert, n_try, rng)
        
        t_alloc = min(100.0, remaining() - 40.0)
        h_sa2, c5_sa2 = simulated_annealing_binary(h_pert, n_try, rng, total_time=t_alloc, target_c5=target_c5)
        update_best(h_sa2, c5_sa2, n_try)
    
    # === PHASE 6: Final greedy polish ===
    if remaining() > 30:
        h_final = best_h_overall.copy()
        n_final = best_n_overall
        t_alloc = max(10.0, remaining() - 10.0)
        h_f, c5_f = greedy_local_search_binary(h_final, n_final, rng, max_trials=100000, timeout=t_alloc)
        update_best(h_f, c5_f, n_final)
    
    # Ensure we have a valid solution
    if best_h_overall is None:
        best_h_overall = thue_morse_binary(256)
        best_c5_overall = compute_c5(best_h_overall)
        best_n_overall = 256
    
    # Final feasibility check
    if best_n_overall in (256, 512, 1024):
        # Check if it's essentially binary
        if np.allclose(best_h_overall, np.round(best_h_overall), atol=1e-6):
            best_h_overall = make_feasible_binary(best_h_overall, best_n_overall, rng)
        else:
            best_h_overall = make_feasible_cont(best_h_overall, best_n_overall)
    
    best_c5_overall = compute_c5(best_h_overall)
    
    return (best_h_overall, best_c5_overall, best_n_overall)
