import numpy as np
from scipy.optimize import minimize
import time

_START_TIME = 0.0
_BUDGET_S = 1300.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h[:] += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(src, src_n, dst_n):
    if src_n == dst_n:
        return src.copy()
    xp = (np.arange(src_n) + 0.5) * 2.0 / src_n
    xq = (np.arange(dst_n) + 0.5) * 2.0 / dst_n
    return np.interp(xq, xp, src)

    return max(0.0, _BUDGET_S - (time.time() - _START_TIME) - 3.0)


def smooth_obj(h, beta, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    vals = (max_c + np.log(np.sum(np.exp(beta * shifted))) / beta) * dx
    s = np.sum(h) - n / 2.0
    vals += mu * s * s
    return vals


def smooth_grad(h, beta, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exps = np.exp(beta * shifted)
    sum_exps = np.sum(exps)
    w = exps / sum_exps

    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        wk = w[k_idx]
        if wk < 1e-15:
            continue
        if k >= 0:
            grad[0:n - k] += wk * (1.0 - h[k:n])
            grad[k:n] -= wk * h[0:n - k]
        else:
            a = -k
            grad[a:n] += wk * (1.0 - h[0:n - a])
            grad[0:n - a] -= wk * h[a:n]

    grad *= dx
    s = np.sum(h) - n / 2.0
    grad += 2.0 * mu * s
    return grad


def obj_wrap(h, beta, mu):
    return smooth_obj(h, beta, mu)


def grad_wrap(h, beta, mu):
    return smooth_grad(h, beta, mu)


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        cnt = 0
        v = i
        while v:
            cnt ^= (v & 1)
            v >>= 1
        h[i] = 1.0 if cnt == 0 else 0.0
    return h


def make_random_binary(n, rng):

def pair_flip_search(h, n, rng, max_sweeps=10, time_limit=30.0):
    """Local search: flip one 1 to 0 and one 0 to 1 simultaneously."""
    best_c5 = compute_c5(h)
    h = h.copy()
    t_start = time.time()

    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        any_imp = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        order = rng.permutation(len(ones))
        for oi in order:
            if time.time() - t_start > time_limit:
                break
            i = ones[oi]
            if h[i] < 0.5:
                continue
            h[i] = 0.0
            zeros2 = np.where(h < 0.5)[0]
            if len(zeros2) == 0:
                h[i] = 1.0
                continue
            best_j = -1
            best_c = best_c5
            # Try a subset of zeros for efficiency
            candidates = zeros2
            if len(candidates) > 20:
                candidates = candidates[rng.permutation(len(candidates))[:20]]
            for j in candidates:
                h[j] = 1.0
                h_fix = project_feasible(h, n)
                c5_test = compute_c5(h_fix)
                if c5_test < best_c:
                    best_c = c5_test
                    best_j = j
                h[j] = 0.0
            if best_j >= 0:
                h[best_j] = 1.0
                h = project_feasible(h, n)
                best_c5 = best_c
                any_imp = True
                ones = np.where(h > 0.5)[0]
        if not any_imp:
            break

    return h, best_c5


def sa_binary_search(h, n, rng, time_limit=60.0, max_iter=5000):
    """Simulated annealing on binary patterns."""
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, n)
    current_c5 = compute_c5(h_bin)
    best_c5 = current_c5
    best_h = h_bin.copy()

    t_start = time.time()
    T = 0.01
    T_min = 0.0001
    cool_rate = (T_min / T) ** (1.0 / max(1, max_iter))

    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        T *= cool_rate

        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - current_c5

        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

    return best_h, best_c5

                cb = lambda x: obj_wrap(x, beta, mu_val)
                gb = lambda x: grad_wrap(x, beta, mu_val)
                res = minimize(cb, h, jac=gb, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
                    print("  Polish beta={}: C5={:.6f}".format(beta, best_c5))
            except Exception:
                pass

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 is not None and eval_c5 < final_c5:
            final_c5 = float(eval_c5)
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
