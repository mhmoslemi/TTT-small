```python
def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def compute_grad_matrix(h, n):

    m = 2 * n - 1
```

```python
def cutplane_iteration(h0, n, dx, prev_cuts=None, delta=0.1):
    """One iteration of the cutting-plane method.
    Given current point h0, linearize the constraints and solve the LP.
    """
    corr0 = get_corr(h0)  # shape (2n-1,)
    m = 2*n - 1
    
    # Compute gradient: G[j, k] = ∂corr[k]/∂h[j]
    lags = np.arange(-(n-1), n)
    G = np.zeros((n, m))
    for k_idx, ell in enumerate(lags):
        j_lo1 = max(0, -ell)
        j_hi1 = min(n-1, n-1-ell)
        if j_lo1 <= j_hi1:
            G[j_lo1:j_hi1+1, k_idx] += (1.0 - h0[j_lo1+ell:j_hi1+ell+1])
        j_lo2 = max(0, ell)
        j_hi2 = min(n-1, n-1+ell)
        if j_lo2 <= j_hi2:
            G[j_lo2:j_hi2+1, k_idx] -= h0[j_lo2-ell:j_hi2-ell+1]
    
    # LP: minimize t
    # subject to: G[:, k] @ h + (corr0[k] - G[:, k] @ h0) <= t, for all k
    #             sum(h) = n/2
    #             0 <= h[j] <= 1
    #             h0[j] - delta <= h[j] <= h0[j] + delta (trust region)
    
    # Variables: [h(0), ..., h(n-1), t]
    # Objective: [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: G[:, k] @ h - t <= -(corr0[k] - G[:, k] @ h0)
    A_cut = np.hstack([G, -np.ones((m, 1))])  # shape (m, n+1)
    b_cut = -(corr0 - G @ h0)  # shape (m,)
    
    # Equalities: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: max(0, h0 - delta) <= h[j] <= min(1, h0 + delta), t >= 0
    bounds = [(max(0, h0[j] - delta), min(1, h0[j] + delta)) for j in range(n)]
    bounds.append((0, None))  # t is unbounded above
    
    res = linprog(c_obj, A_ub=A_cut, b_ub=b_cut, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        h_new = res.x[:n]
        t_new = res.x[n]
        return h_new, t_new, corr0
    return None
```

corr_full[m] = sum_i h[i] * (1 - h[i + ell]) where ell = m - n + 1

Let me denote the correlation at lag ell as C(ell).

C(ell) = sum_{i: 0≤i<n, 0≤i+ell<n} h[i] * (1 - h[i+ell])

∂C(ell)/∂h[j]:
- If j appears as h[i]: i = j, valid if 0 ≤ j+ell < n. Contribution: (1-h[j+ell])
- If j appears as h[i+ell]: i+ell = j, i = j-ell, valid if 0 ≤ j-ell < n. Contribution: -h[j-ell]

So ∂C(ell)/∂h[j] = [0≤j+ell<n]*(1-h[j+ell]) - [0≤j-ell<n]*h[j-ell]

Now for the coordinate descent, the optimal h[j] given current corr:

After changing h[j] from t_old to t_new, the new correlation at position m is:
corr_new[m] = corr_old[m] + (t_new - t_old) * G[j, m]

We want to minimize max_m corr_new[m] = max_m (corr_old[m] + (t_new - t_old) * G[j, m])

Let a_m = G[j, m], b_m = corr_old[m] - t_old * a_m.
Then: corr_new[m] = a_m * t_new + b_m

We want t_new ∈ [0,1] that minimizes max_m (a_m * t_new + b_m).

This is a 1D minimax problem. The function f(t) = max_m(a_m * t + b_m) is convex and piecewise linear. The minimum is at a vertex where two lines intersect (or at t=0 or t=1).

The bisection approach:
- f is convex, so f'(t) (subgradient) is monotonically increasing
- f'(t) = mean of a_m over the active set (where a_m * t + b_m = f(t))
- f'(t) < 0: decrease t (actually f is decreasing, so we want to go left... wait)

Actually, f(t) = max_m a_m * t + b_m. 
f is decreasing in t when all active a_m < 0, increasing when all active a_m > 0.
The minimum is where the active slope crosses 0.

Bisection:
- lo = 0, hi = 1
- mid = (lo + hi) / 2
- Compute f(mid) and the active set
- If mean(a_m over active) > 0: f is increasing at mid, so minimum is to the left → hi = mid
- If mean(a_m over active) < 0: f is decreasing at mid, so minimum is to the right → lo = mid
- If mean ≈ 0: we're at the minimum
