So Δcorr[k] = -(1-h[i-k]) + h[i+k] + (1-h[j-k]) - h[j+k] (for valid indices)

This is O(1) per k, so computing the full Δcorr is O(n) per flip. Then we check if max(c + Δcorr) < max(c).

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):

        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:

            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:

    it = 0
    while time.time() - start_t < time_limit:
        frac = it / max(total_iters_estimate, 1)
        T = T_init * (T_final / T_init) ** frac
        
        # Pick random 1 and 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        
        new_corr = get_corr(h)
        new_max = np.max(new_corr)
        
        delta = new_max - cur_max
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        # Kick
        if no_improve > kick_every:
            for _ in range(10):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h[ii] = 0
                h[jj] = 1
            corr = get_corr(h)
            cur_max = np.max(corr)
            no_improve = 0
        
        it += 1
    
    return best_h, best_max * 2.0 / n
