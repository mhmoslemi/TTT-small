        alloc = tl() * 0.4 if n == best_n else tl() * 0.25
        h_opt = try_optimize(n, h_init, alloc)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            best_n = n
    
    # Additional starting points at best resolution
    if tl() > 60:
        inits = [best_h.copy()]
        # Perturbed
        for s in [0.02, 0.05, 0.1]:
            inits.append(project_feasible(best_h + s * rng.randn(best_n), best_n))
        # Smooth perturbation
        t = np.arange(best_n) / best_n
        for f in [2, 3, 5]:
            ph2 = 0.7 * best_h + 0.3 * (0.5 + 0.3 * np.sin(2*np.pi*f*t))
            inits.append(project_feasible(ph2, best_n))
        
        for h0 in inits:
            if tl() < 30: break
            h_opt = try_optimize(best_n, h0, min(60.0, tl()*0.2))
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
    
    # Gradient-based final polish
    if tl() > 10:
        h = best_h.copy()
        lr = 0.01
        for step in range(200):
            if tl() < 5: break
            corr = np.correlate(h, 1.0 - h, mode='full')
