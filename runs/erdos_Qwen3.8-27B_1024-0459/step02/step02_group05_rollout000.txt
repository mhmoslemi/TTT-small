def project_to_sum(x, target):
    x = np.asarray(x, dtype=float)
    # bisection for lam
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0

        if s > target:

        else:
            hi = lam
    lam = 0.5*(lo+hi)
    y = np.clip(x-lam, 0.0, 1.0)

_SHIFT_CACHE = {}
def get_shifts(n):
    if n in _SHIFT_CACHE:
        return _SHIFT_CACHE[n]
    sh=[]
    for s in range(-(n-1), n):
        lo = 0 if s>=0 else -s
        hi = (n-1) if s<0 else (n-1-s)
        # ensure
        lo = max(0,-s); hi=min(n-1,n-1-s)
        if hi>=lo:
            sh.append((s,lo,hi))
    _SHIFT_CACHE[n]=sh
    return sh

lags_values_fast:

def lags_values_fast(h, dx, shifts):
    n=len(h)
    L=np.empty(len(shifts), dtype=float)
    k=0
    for (s,lo,hi) in shifts:
        L[k]=dx*np.dot(h[lo:hi+1], 1.0-h[lo+s:hi+1+s])
        k+=1
    return L
```

lags_grad_fast:
```
def lags_grad_fast(h, w, dx, shifts):
    n=len(h)
    g=np.zeros(n, dtype=float)
    for idx,(s,lo,hi) in enumerate(shifts):
        ws=w[idx]
        if ws<=1e-15:
            continue
        seg = ws*dx
        g[lo:hi+1] += seg*(1.0 - h[lo+s:hi+1+s])
        g[lo+s:hi+1+s] -= seg*h[lo:hi+1]
    return g
```

def soft_weight(L, beta):
    m=float(L.max())
    e=np.exp(beta*(L-m))
    Z=float(e.sum())
    if Z<=0: 
        e=np.full_like(L,1.0); Z=L.size
    w=e/Z
    val=m+np.log(Z)/beta
    return w, val
def lse_value(L,beta):
    m=float(L.max())
    e=np.exp(beta*(L-m))
    Z=float(e.sum())
    return m+np.log(Z)/beta
```

def lse_pgd(h, n, beta, t_limit, rng, lr0=0.25):
    if t_limit<=0: 
        return h, c5_exact(h)
    dx=2.0/n
    shifts=get_shifts(n)
    h=make_feasible(h,n)
    best_h=h.copy(); best_c=c5_exact(h)
    lr=lr0
    t0=time.time(); stall=0
    while (time.time()-t0) < t_limit:
        L=lags_values_fast(h,dx,shifts)
        Gcur=lse_value(L,beta)
        w,_=soft_weight(L,beta)
        g=lags_grad_fast(h,w,dx,shifts)
        gnorm2=float(np.dot(g,g))
        if gnorm2<1e-20:
            # stuck
            break
        ok=False
        lr2=lr
        for _ in range(12):
            cand=np.clip(h-lr2*g,0.0,1.0)
            candc=project_to_sum(cand, n/2.0)
            Lc=lags_values_fast(candc,dx,shifts)
            Gc=lse_value(Lc,beta)
            if Gc <= Gcur - 1e-6*lr2*gnorm2:
                h=candc; ok=True; lr=min(lr2*1.25, 1.25); break
            lr2*=0.5
        if ok:
            stall=0
        else:
            lr=max(lr*0.5,1e-4)
            stall+=1
            if stall>30:
                h=project_to_sum(h+rng.normal(0,0.02,h.size), n/2.0)
                stall=0; lr=0.05
        c=c5_exact(h)
        if c<best_c-1e-15:
            best_c=c; best_h=h.copy()
    return best_h, best_c

def local_search(h, n, sec_limit, rng):
    if sec_limit<=0:
        return h, c5_exact(h)
    h=make_feasible(h,n)
    best_c=c5_exact(h)
    t0=time.time(); it=0
    while (time.time()-t0)<sec_limit:
        it+=1
        # pair transfer
        i=rng.randint(n); j=rng.randint(n)
        if i!=j:
            if h[i]<h[j]: i,j=j,i
            mamt=min(float(h[i]), 1.0-float(h[j]))
            if mamt>1e-7:
                frac=0.1+0.9*rng.rand()
                amt=mamt*frac
                if amt>1e-9:
                    h[i]-=amt; h[j]+=amt
                    c=c5_exact(h)
                    if c<best_c: best_c=c
                    else:
                        h[i]+=amt; h[j]-=amt
        if (it & 1)==1:
            # worst-lag targeted
            L=lags_values_fast(h, 2.0/n, get_shifts(n))
            sstar=int(get_shifts(n)[int(np.argmax(L))][0])
            i=rng.randint(n)
            j=i+sstar
            if 0<=j<n:
                if h[i]>h[j]:
                    mamt=min(float(h[i])-float(h[j]), 0.5)
                    if mamt>1e-6:
                        amt=mamt*(0.1+0.9*rng.rand())
                        h[i]-=amt; h[j]+=amt
                        c=c5_exact(h)
                        if c<best_c: best_c=c
                        else:
                            h[i]+=amt; h[j]-=amt
        if (it%3)==0:
            k=1+rng.randint(0,3)
            idx=rng.choice(n,k,replace=False)
            delta=rng.uniform(-0.04,0.04,k)
            trial=h.copy(); trial[idx]+=delta
            trialp=project_to_sum(trial, n/2.0)
            c=c5_exact(trialp)
            if c<best_c-1e-15:
                h=trialp; best_c=c
    return h,best_c

def structured_inits(n, rng):
    t=np.linspace(0.0,2.0,n,endpoint=False)
    out=[]
    x=0.5+0.35*np.cos(2*np.pi*1.0*t)+0.15*np.sin(2*np.pi*2.5*t)
    out.append(np.clip(x,0,1))
    x2=0.5+0.3*np.cos(2*np.pi*2.0*t)+0.2*np.sin(2*np.pi*5.0*t)
    out.append(np.clip(x2,0,1))
    out.append(rng.rand(n))
    if n%2==0:
        out.append(np.tile([1.0,0.0], n//2))
    if n%4==0:
        out.append(np.tile([1.0,1.0,0.0,0.0], n//4))
    return [make_feasible(a, n) for a in out]

import numpy as np
import time

_SHIFT_CACHE = {}

def _get_shifts(n):
    if n in _SHIFT_CACHE:
        return _SHIFT_CACHE[n]
    sh = []
    for s in range(-(n-1), n):
        lo = 0 if s >= 0 else -s
        hi = (n-1) if s < 0 else (n-1-s)
        lo = max(0, -s)
        hi = min(n-1, n-1-s)
        if hi >= lo:
            sh.append((s, lo, hi))
    _SHIFT_CACHE[n] = sh
    return sh

def _project_to_sum(x, target):
    x = np.asarray(x, dtype=float)
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(90):
        lam = 0.5*(lo+hi)
        s = float(np.sum(np.clip(x-lam, 0.0, 1.0)))
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = 0.5*(lo+hi)
    return np.clip(x-lam, 0.0, 1.0)

def _make_feasible(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float).reshape(-1).copy()
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_sum(h, n/2.0)
    return h

def _lags_values(h, dx, shifts):
    L = np.empty(len(shifts), dtype=float)
    k = 0
    for (s, lo, hi) in shifts:
        L[k] = dx*np.dot(h[lo:hi+1], 1.0-h[lo+s:hi+1+s])
        k += 1
    return L

def _lags_grad(h, w, dx, shifts):
    n = len(h)
    g = np.zeros(n, dtype=float)
    for idx, (s, lo, hi) in enumerate(shifts):
        ws = w[idx]
        if ws <= 1e-15:
            continue
        seg = ws*dx
        g[lo:hi+1] += seg*(1.0 - h[lo+s:hi+1+s])
        g[lo+s:hi+1+s] -= seg*h[lo:hi+1]
    return g

def _lse_value(L, beta):
    m = float(L.max())
    e = np.exp(beta*(L-m))
    Z = float(e.sum())
    if Z <= 0.0:
        return m
    return m + np.log(Z)/beta

def _soft_weight(L, beta):
    m = float(L.max())
    e = np.exp(beta*(L-m))
    Z = float(e.sum())
    if Z <= 0.0:
        e = np.full_like(L, 1.0)
        Z = float(L.size)
    return e/Z, m + np.log(Z)/beta

_SHIFT_CACHE = {}


def _get_shifts(n):
    if n in _SHIFT_CACHE:
        return _SHIFT_CACHE[n]
    sh = []
    for s in range(-(n-1), n):
        lo = max(0, -s)
        hi = min(n-1, n-1-s)

            sh.append((s, lo, hi))

def _project_to_sum(x, target):
    x = np.asarray(x, dtype=float)

    hi = float(x.max()) + 1.0

            lo = lam
        else:
            hi = lam

def _lags_values(h, dx, shifts):
