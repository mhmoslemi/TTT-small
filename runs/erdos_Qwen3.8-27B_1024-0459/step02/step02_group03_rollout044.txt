def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    total = np.sum(np.exp(shifted))
    return (m + np.log(total) / beta)

def smooth_max_grad(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    return weights

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n

def interp_h(h_src, n_dst):

def slsqp_optimize(h0, n, beta, max_iter, ftol=1e-14):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, beta) * dx
    
    def jacobian(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        weights = smooth_max_grad(corr, beta)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-14:
                continue
            L = k - (n - 1)
            if L == 0:
                grad += w * (1.0 - 2.0 * x)
            elif L > 0:
                grad[L:] += w * (1.0 - x[:n-L])
                grad[:n-L] -= w * x[L:]
            else:
                Lp = -L
                grad[:n-Lp] += w * (1.0 - x[Lp:])
                grad[Lp:] -= w * x[:n-Lp]
        return dx * grad
    
    def constraint(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint}]
    
    res = minimize(obj, h0, jac=jacobian, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

            h[j] = 0.0

def active_set_subgradient(h0, n, n_top=5, n_iters=2000, step_init=0.01, step_end=0.0001, rng=None):

    dx = 2.0 / n

        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top n_top lags
        top_idx = np.argsort(corr)[-n_top:]
        top_avg = np.mean(corr[top_idx])
        
        # Subgradient of max (approximated by average of top lags)
        grad = np.zeros(n)
        for k in top_idx:
            w = 1.0 / n_top
            L = k - (n - 1)
            if L == 0:
                grad += w * (1.0 - 2.0 * h)
            elif L > 0:
                grad[L:] += w * (1.0 - h[:n-L])
                grad[:n-L] -= w * h[L:]
            else:
                Lp = -L
                grad[:n-Lp] += w * (1.0 - h[Lp:])

def compute_c5(h):
    n = len(h)

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta)

    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

    weights = smooth_max_weights(corr, beta)

            grad[:n - Lp] += w * (1.0 - x[Lp:])

def slsqp_optimize(h0, n, beta, max_iter, ftol=1e-14):
    dx = 2.0 / n

        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, beta) * dx

        return grad_smooth_max(x, beta)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)
