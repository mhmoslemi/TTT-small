    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:

            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)

            break

        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):

    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s

            break
        mask = (h > 0.0) & (h < 1.0) if abs(diff) > 1.0 / n else (h > 0.001) & (h < 0.999)
        cnt = np.sum(mask)

            mask = h < 1.0 if diff > 0 else h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to_n(h_src, n_dst):

        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def compute_subgradient(h, n):
    v = 1.0 - h
    c = np.correlate(h, v, 'full')
    k_star = int(np.argmax(c))
    d = k_star - (n - 1)
    g = np.zeros(n)
    if d >= 0:
        if d == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[d:] += (1.0 - h[:n - d])
            g[:n - d] -= h[d:]

def subgradient_momentum(h_init, n, start_time, budget_s, lr_init=0.002, momentum=0.9, n_iter_limit=200000):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    lr = lr_init
    patience = 0
    target_patience = 2000
    
    for it in range(n_iter_limit):
        if time.time() - start_time > budget_s - 20:
            break
        g, c, _ = compute_subgradient(h, n)
        m = momentum * m - lr * g
        h_new = h + m
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h = h_new

            m *= 0.5

def softmax_ls(h_init, n, start_time, budget_s, beta_start=50, betas=(50, 200, 800, 3000, 10000)):
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    
    for beta in betas:
        if time.time() - start_time > budget_s - 30:
            break
        def obj_smooth(x):
            c = np.correlate(x, 1.0 - x, 'full')
            cmax = np.max(c)
            beta_safe = min(beta, 50.0 / max(cmax, 1e-10))
            shifted = c - cmax
            logsum = cmax + (1.0 / beta_safe) * np.log(np.sum(np.exp(beta_safe * shifted)))
            return logsum * (2.0 / n)
        
        def obj_grad(x):
            c = np.correlate(x, 1.0 - x, 'full')
            cmax = np.max(c)
            beta_safe = min(beta, 50.0 / max(cmax, 1e-10))
            shifted = c - cmax
            weights = np.exp(beta_safe * shifted)
            weights /= np.sum(weights)
            g = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                g[k] = weights[k]
            # Gradient of c[k] w.r.t. h
            # c[k] = sum_i h[i+k-n+1]*(1-h[i]) for valid i
            grad_h = np.zeros(n)
            for k in range(2 * n - 1):
                w = g[k]
                if w < 1e-15:
                    continue
                d = k - (n - 1)
                if d >= 0:
                    if d == 0:
                        grad_h[:] += w * (1.0 - 2.0 * h)  # use current h as approx
                    else:
                        grad_h[d:] += w * (1.0 - h[:n-d])
                        grad_h[:n-d] -= w * h[d:]
                else:
                    dp = -d
                    grad_h[:n-dp] += w * (1.0 - h[dp:])
                    grad_h[dp:] -= w * h[:n-dp]
            return (2.0 / n) * grad_h
        
        try:
            res = minimize(obj_smooth, h, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50})
            if res.fun < 1e10:
                h = make_feasible(res.x, n)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c5

def slsqp_opt(h_init, n, start_time, budget_s, max_iter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        if time.time() - start_time > budget_s - 15:
            return 1e10
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
        if res.success or (res.fun < 1e10 and res.x is not None):
            h_out = make_feasible(res.x, n)
            c5_out = compute_c5(h_out)
            return h_out, c5_out
    except Exception:
        pass
    return None, None

def pair_swap_local(h, n, n_rounds, rng, time_fn, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        if time_fn() > deadline:
            break
        improved = False
        for i in range(n - 1):
            diff2 = h[i] - h[i + 1]
            if abs(diff2) < 1e-6:
                continue
            for step in [0.005, 0.01, 0.02, 0.05, 0.1]:
                amt = min(step, abs(diff2) / 2.0)
                if amt < 1e-7:
                    continue
                h_try = h.copy()
                if diff2 > 0:
                    h_try[i] -= amt
                    h_try[i + 1] += amt
                else:
                    h_try[i] += amt
                    h_try[i + 1] -= amt
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = make_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def binary_sa(h_init_binary, n, n_steps, t_start, t_end, rng, start_time, budget_s):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step_i in range(n_steps):
        if time.time() - start_time > budget_s - 15:
            break
        T = t_start * (t_end / t_start) ** (step_i / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:

        h[i], h[j] = h[j], h[i]

        n_rounds = min(40, int(remaining() / 0.5))
        best_overall_h, best_overall_c5 = pair_swap_local(best_overall_h, best_overall_n,
                                                           n_rounds, rng, lambda: time.time(),
                                                           time.time() + remaining() - 5)
    
    if remaining() > 15:
        n_trials = min(20000, int(remaining() / 0.003))
        best_overall_h, best_overall_c5 = random_perturb(best_overall_h, best_overall_n,
                                                          n_trials, 0.01, rng, start_time, budget_s)
    
    # Final feasibility
    best_overall_h = make_feasible(best_overall_h, best_overall_n)
    best_overall_c5 = compute_c5(best_overall_h)
    
    assert np.all(best_overall_h >= -1e-12) and np.all(best_overall_h <= 1 + 1e-12)
    assert abs(np.sum(best_overall_h) - best_overall_n / 2.0) < 1e-8
    
    return (best_overall_h, best_overall_c5, best_overall_n)
