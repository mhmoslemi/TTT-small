    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = beta * (corr - cmax)
    # Clip for numerical stability

        weights[np.argmax(corr)] = 1.0

    n_src = len(h_src)
    if n_src < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.arange(n_src) * 2.0 / n_src
    x_new = np.arange(n_dst) * 2.0 / n_dst
    return np.interp(x_new, x_old, h_src)

def proj_grad_momentum(h, n, betas, base_alpha, n_iters_per_beta, rng):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    mu = 0.9
    for beta in betas:
        n_it = n_iters_per_beta
        for it in range(n_it):
            g = lse_gradient(h_cur, beta)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-14:
                break
            alpha = base_alpha / (1.0 + 0.02 * it / max(1, n_it))
            v = mu * v - alpha * g
            h_new = project_feasible(h_cur + v, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
    c5_final = compute_c5(h_cur)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_cur.copy()
    return best_h, best_c5

def local_grad_search(h, n, n_iters, step_scale, rng):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        cmax = np.max(corr)
        near_max = np.where(corr > cmax * 0.999 - 1e-6)[0]
        g = np.zeros(n)
        for j in near_max:
            s = j - (n - 1)
            t1_lo = max(0, s)
            t1_hi = min(n - 1, n - 1 + s)
            if t1_lo <= t1_hi:
                idx = np.arange(t1_lo, t1_hi + 1)
                g[idx] += (1.0 - h_cur[idx - s])
            t2_lo = max(0, -s)
            t2_hi = min(n - 1, n - 1 - s)
            if t2_lo <= t2_hi:
                idx = np.arange(t2_lo, t2_hi + 1)
                g[idx] += (-1.0) * h_cur[idx + s]
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        step = step_scale / (1.0 + 0.01 * it / max(1, n_iters))
        improved = False
        for scale in [step, step * 0.5, step * 0.25, step * 0.1]:
            h_new = project_feasible(h_cur - scale * g / g_norm, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new
                improved = True
                break
        if not improved:
            h_try = h_cur + rng.randn(n) * step * 0.3
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_cur = h_try
    return best_h, best_c5

def pair_swap_local(h, n, n_rounds, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for i in range(n - 1):
            if abs(h[i] - h[i + 1]) < 1e-8:
                continue
            for step in [0.02, 0.05, 0.1, 0.2]:
                amt = min(step, abs(h[i] - h[i + 1]) / 2.0)
                if amt < 1e-7:
                    continue
                h_try = h.copy()
                if h[i] > h[i + 1]:
                    h_try[i] -= amt
                    h_try[i + 1] += amt
                else:
                    h_try[i] += amt
                    h_try[i + 1] -= amt
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def structured_init(n, rng):
    inits = []
    if n % 4 == 0:
        h1 = np.tile([1.0, 1.0, 0.0, 0.0], n // 4).astype(float)
        inits.append(h1.copy())
    if n % 2 == 0:
        h2 = np.tile([1.0, 0.0], n // 2).astype(float)
        inits.append(h2.copy())
    h3 = np.zeros(n)
    for i in range(n):
        h3[i] = 1.0 if (i * (i + 1) // 2) % n < n // 2 else 0.0
    inits.append(make_feasible(h3, n))
    h4 = rng.rand(n)
    inits.append(make_feasible(h4, n))
    h5 = np.ones(n) * 0.5 + rng.randn(n) * 0.2
    inits.append(project_feasible(h5, n))
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            h_parent = None
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_n = len(h_parent)
        best_h = make_feasible(h_parent.copy(), best_n)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with LSE at parent resolution
    if remaining() > 120:
        for beta in [80, 300]:
            if remaining() < 60:
                break
            h_opt, c5_opt = slsqp_lse(best_n, best_h, beta=beta, max_iter=1200, ftol=1e-14)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 2: Projected gradient with momentum at parent resolution
    if remaining() > 60:
        betas = [20, 60, 200, 800, 2000]
        h_opt, c5_opt = proj_grad_momentum(best_h, best_n, betas, 0.03, 400, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 3: Upsample to n=320 and optimize
    n_hi = 320
    h_hi = make_feasible(interpolate_to_n(best_h, n_hi), n_hi)
    if remaining() > 80:
        h_opt, c5_opt = slsqp_lse(n_hi, h_hi, beta=100, max_iter=800, ftol=1e-13)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_hi
            h_hi = h_opt
        else:
            h_hi = h_opt if h_opt is not None else h_hi

    if remaining() > 60:
        betas = [30, 100, 400, 1500]
        h_opt, c5_opt = proj_grad_momentum(h_hi, n_hi, betas, 0.015, 400, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_hi

    # Phase 4: Local gradient search
    if remaining() > 30:
        n_iters = min(8000, int(remaining() * 15))
        best_h, best_c5 = local_grad_search(best_h, best_n, n_iters, 0.015, rng)

    # Phase 5: Try n=400
    if remaining() > 60:
        n_hi2 = 400
        h_hi2 = make_feasible(interpolate_to_n(best_h, n_hi2), n_hi2)
        h_opt, c5_opt = proj_grad_momentum(h_hi2, n_hi2, [40, 150, 600], 0.01, 300, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_hi2
        if remaining() > 30:
            h_opt2, c5_opt2 = slsqp_lse(best_n, best_h, beta=200, max_iter=600, ftol=1e-13)
            if h_opt2 is not None and c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    # Phase 6: Multi-start at best_n from structured inits
    if remaining() > 40:
        for h_init in structured_init(best_n, rng):
            if remaining() < 20:
                break
            h_opt, c5_opt = slsqp_lse(best_n, h_init, beta=150, max_iter=600, ftol=1e-13)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 7: Pair swap local search
    if remaining() > 15:
        n_rounds = min(60, int(remaining() / 0.3))
        best_h, best_c5 = pair_swap_local(best_h, best_n, n_rounds, rng)

    # Phase 8: Final SLSQP polish with high beta
    if remaining() > 15:
        for beta in [500, 1500]:
            if remaining() < 8:
                break
            h_opt, c5_opt = slsqp_lse(best_n, best_h, beta=beta, max_iter=800, ftol=1e-15)
            if h_opt is not None and c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 9: Final local gradient search
    if remaining() > 10:
        n_iters = min(5000, int(remaining() * 15))
        best_h, best_c5 = local_grad_search(best_h, best_n, n_iters, 0.008, rng)

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    if np.any(best_h < -1e-12) or np.any(best_h > 1.0 + 1e-12):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-8

    return (best_h, best_c5, best_n)
