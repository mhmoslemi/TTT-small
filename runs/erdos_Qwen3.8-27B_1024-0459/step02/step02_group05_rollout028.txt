import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-11:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-8
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-8
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.maximum(h[mask] + step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_grad(h, lag_idx):
    n = len(h)
    k = lag_idx - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        grad[:m] += (1.0 - h[k:k + m])
        grad[k:] -= h[:m]
    else:
        kp = -k
        m = n - kp
        grad[kp:] += (1.0 - h[:m])
        grad[:m] -= h[kp:]
    return grad

def smooth_gradient_descent(h, betas, steps_per_beta, lr_init, time_limit):
    h = h.copy()
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_start = time.time()
    for beta in betas:
        for t in range(steps_per_beta):
            if time.time() - t_start > time_limit:
                break
            lags = compute_all_lags(h)
            shifted = beta * (lags - np.max(lags))
            weights = np.exp(shifted)
            wsum = np.sum(weights)
            if wsum <= 0:
                break
            weights /= wsum
            grad = np.zeros(n)
            active_idx = np.where(weights > 1e-5)[0]
            for idx in active_idx:
                grad += weights[idx] * subgradient_grad(h, int(idx))
            lr = lr_init / (1.0 + 0.005 * t)
            h_try = h - lr * grad
            h_try = np.clip(h_try, 0.0, 1.0)
            diff = n / 2.0 - np.sum(h_try)
            if abs(diff) > 1e-10:
                mask = (h_try > 0.001) & (h_try < 0.999)
                cnt = np.count_nonzero(mask)
                if cnt > 0:
                    h_try[mask] += diff / cnt
                h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
    return best_h, best_c5

def slsqp_optimize(h_init, n, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        if res.fun < 1e10:
            h_out = make_feasible(res.x, n)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return None, None

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)

def pair_swap_local(h, n, n_rounds, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for i in range(n - 1):
            if h[i] <= h[i + 1]:
                continue
            for step in [0.005, 0.01, 0.02, 0.05]:
                amt = min(step, (h[i] - h[i + 1]) / 2.0)
                if amt < 1e-7:
                    continue
                h_try = h.copy()
                h_try[i] -= amt
                h_try[i + 1] += amt
                h_try = np.clip(h_try, 0.0, 1.0)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def random_perturb_improve(h, n, n_trials, base_step, rng):

    step = base_step

def binary_sa(n, rng, n_steps, t_start, t_end):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h_i, h_j = h[i], h[j]
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
        elif rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-15)):
            pass
        else:
            h[i], h[j] = h_i, h_j
    return best_h, best_c5

def active_lag_optimize(h, n, n_iters, lr, time_limit):

        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)

        else:

            best_c5 = c5_opt
            best_h = h_opt
            best_h_global = best_h.copy()
            best_n_global = best_n

    # Phase 2: SLSQP refinement
    if remaining() > 150:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, max_iter=5000, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_h_global = best_h.copy()
            best_n_global = best_n

    # Phase 3: Active lag optimization (targeted subgradient)
    if remaining() > 120:
        t_alloc = min(60, remaining() - 120)
        h_opt, c5_opt = active_lag_optimize(best_h, best_n, 20000, 0.03, t_alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_h_global = best_h.copy()
            best_n_global = best_n

    # Phase 4: More smooth gradient descent on improved solution
    if remaining() > 100:
        t_alloc = min(60, remaining() - 100)
        h_opt, c5_opt = smooth_gradient_descent(
            best_h, betas=[50, 150, 500, 1500],
            steps_per_beta=2000, lr_init=0.02,
            time_limit=t_alloc
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_h_global = best_h.copy()
            best_n_global = best_n

    # Phase 5: SLSQP again after gradient descent
    if remaining() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, max_iter=4000, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_h_global = best_h.copy()
            best_n_global = best_n

    # Phase 6: Resolution exploration
    for n_try in [140, 180, 200, 220, 256]:
        if remaining() < 50:
            break
        if n_try % 2 != 0:
            n_try += 1
        h_init = make_feasible(interpolate_to_n(best_h_global, n_try), n_try)
        # Quick smooth gradient descent at new resolution
        h_tmp, c5_tmp = smooth_gradient_descent(
            h_init, betas=[50, 200],
            steps_per_beta=1500, lr_init=0.04,
            time_limit=min(30, remaining() - 50)
        )
        # SLSQP polish
        h_opt2, c5_opt2 = slsqp_optimize(h_tmp, n_try, max_iter=2000, ftol=1e-14)
        if h_opt2 is not None and c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2
            best_h_global = h_opt2.copy()
            best_n_global = n_try

    # Phase 7: Binary SA at several resolutions
    for n_bin in [best_n_global, 128, 160, 200, 256, 320]:
        if remaining() < 30:
            break
        if n_bin < 40:
            continue
        if n_bin % 2 != 0:
            n_bin += 1
        n_steps = min(25000, max(5000, int((remaining() - 30) / 0.08)))
        h_bin, c5_bin = binary_sa(n_bin, rng, n_steps, 0.5, 0.005)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.astype(float)
            best_h_global = best_h.copy()
            best_n_global = n_bin

    # Phase 8: Final SLSQP on best
    if remaining() > 30:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n_global, max_iter=6000, ftol=1e-16)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_h_global = best_h.copy()
            best_n_global = best_n

    # Phase 9: Pair swap local search
    if remaining() > 20:
        n_rounds = min(60, max(10, int(remaining() / 0.8)))
        best_h, best_c5 = pair_swap_local(best_h, best_n_global, n_rounds, rng)
        best_h_global = best_h.copy()

    # Phase 10: Random perturbation polish
    if remaining() > 15:
        n_trials = min(30000, max(500, int((remaining() - 10) / 0.005)))
        best_h, best_c5 = random_perturb_improve(best_h, best_n_global, n_trials, 0.02, rng)
        best_h_global = best_h.copy()

    # Phase 11: Final micro-optimizatio
    if remaining() > 10:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n_global, max_iter=3000, ftol=1e-16)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_h_global = best_h.copy()

    # Final feasibility
    best_h = make_feasible(best_h, best_n_global)
    best_c5 = compute_c5(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    diff_final = best_n_global / 2.0 - np.sum(best_h)
    if abs(diff_final) > 1e-10:
        mask = (best_h > 1e-6) & (best_h < 1.0 - 1e-6)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            best_h[mask] += diff_final / cnt
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n_global)
