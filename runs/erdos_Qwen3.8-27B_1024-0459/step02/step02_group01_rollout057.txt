        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def interp_to_n(h, n_dest):
    n_src = len(h)
    if n_src == n_dest:
        return h.copy()
    xp = (np.arange(n_src) + 0.5) * 2.0 / n_src
    xn = (np.arange(n_dest) + 0.5) * 2.0 / n_dest
    return np.interp(xn, xp, h)

def compute_active_gradient(h, active_lags):
    n = len(h)
    grad = np.zeros(n)
    j = np.arange(n)
    for k in active_lags:
        idx1 = j + k
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[valid1] += (1.0 - h[idx1[valid1]])
        idx2 = j - k
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid2] += (-h[idx2[valid2]])
    return grad / len(active_lags)

def slsqp_optimize(h, n, t0, budget_s):
    global _g_n, _g_beta
    _g_n = n
    h = project_feasible(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]
    
    stages = [(5, 80), (20, 120), (100, 200), (500, 300), (2000, 400), (10000, 600), (50000, 800), (200000, 1000)]
    
    for beta, maxit in stages:
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 20.0
        if remaining < 10:
            break
        _g_beta = beta
        try:
            res = minimize(_smooth_obj, h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxit, 'ftol': 1e-15, 'iprint': 0})
            h = project_feasible(res.x, n)
        except Exception:
            pass
    return h

def local_search_gradient(h, n, max_sweeps=500, t0=None, budget_s=None):
    h = project_feasible(h, n)
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if t0 is not None:
            if time.time() - t0 > budget_s - 15:
                break
        
        corr = compute_corr(h)
        max_val = np.max(corr)
        active_mask = corr >= max_val - 1e-9
        active_lags = np.where(active_mask)[0] - (n - 1)
        
        if len(active_lags) == 0:
            active_lags = np.array([0])
        
        grad = compute_active_gradient(h, active_lags)
        
        improved = False
        step_sizes = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
        
        for j in range(n):
            if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                continue
            if abs(grad[j]) < 1e-12:
                continue
            
            direction = -np.sign(grad[j])
            best_delta = 0.0
            best_val = best_c5
            best_h_local = h.copy()
            
            hi = h[j]
            for alpha in step_sizes:
                new_val = hi + direction * alpha
                if new_val < 0 or new_val > 1:
                    continue
                h[j] = new_val
                h = project_feasible(h, n)
                c = compute_c5(h)
                if c < best_val - 1e-13:
                    best_val = c
                    best_delta = direction * alpha
                    best_h_local = h.copy()
            
            h = best_h_local
            if best_delta != 0.0:
                best_c5 = best_val
                improved = True
        
        if not improved:
            # Try smaller steps
            improved2 = False
            for j in range(n):
                if h[j] <= 1e-9 or h[j] >= 1.0 - 1e-9:
                    continue
                if abs(grad[j]) < 1e-12:
                    continue
                direction = -np.sign(grad[j])
                hi = h[j]
                for alpha in [0.0001, 0.0003, 0.001]:
                    new_val = hi + direction * alpha
                    if new_val < 0 or new_val > 1:
                        continue
                    h[j] = new_val
                    h = project_feasible(h, n)
                    c = compute_c5(h)
                    if c < best_c5 - 1e-13:
                        best_c5 = c
                        improved2 = True
                        break
                if not improved2:
                    pass
            
            if not improved2:
                break
    
    return h, best_c5

def random_restart_optimize(h_base, n, seed, t0, budget_s, time_limit=60):
    rng = np.random.RandomState(seed)
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)
    
    for attempt in range(5):
        if time.time() - t0 > budget_s - 30:
            break
        scale = 0.05 + 0.05 * attempt
        h_try = h_base + scale * rng.randn(n)
        h_try = project_feasible(h_try, n)
        
        t_start = time.time()
        h_opt = slsqp_optimize(h_try, n, t0, budget_s)
        h_opt, c5_opt = local_search_gradient(h_opt, n, max_sweeps=100, t0=t0, budget_s=budget_s)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    return best_h, best_c5

def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_beta
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Load parent
    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 0:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass
    
    if best_h is None:
        best_n = 192
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)
    
    def remaining():
        return max(0.0, budget_s - (time.time() - t0) - 8.0)
    
    # Phase 1: Optimize at parent resolution
    if remaining() > 60:
        h1 = slsqp_optimize(best_h, best_n, t0, budget_s)
        h1, c5_1 = local_search_gradient(h1, best_n, max_sweeps=300, t0=t0, budget_s=budget_s)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = best_n
    
    # Phase 2: Try higher resolutions
    n_candidates = []
    if best_n <= 256:
        n_candidates = [384, 512]
    elif best_n <= 512:
        n_candidates = [512, 768]
    else:
        n_candidates = [768, 1024]
    
    for n_try in n_candidates:
        if remaining() < 80:
            break
        if n_try > 1200:
            continue
        
        h_interp = project_feasible(interp_to_n(best_h, n_try), n_try)
        h_opt = slsqp_optimize(h_interp, n_try, t0, budget_s)
        h_opt, c5_opt = local_search_gradient(h_opt, n_try, max_sweeps=200, t0=t0, budget_s=budget_s)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try
    
    # Phase 3: Random restarts at best resolution
    if remaining() > 120:
        h_rs, c5_rs = random_restart_optimize(best_h, best_n, seed + 100, t0, budget_s, time_limit=100)
        if c5_rs < best_c5:
            best_c5 = c5_rs
            best_h = h_rs.copy()
    
    # Phase 4: More local search on best
    if remaining() > 60:
        h_ls, c5_ls = local_search_gradient(best_h, best_n, max_sweeps=1000, t0=t0, budget_s=budget_s)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
    
    # Phase 5: Try different structured inits at best_n
    if remaining() > 100:
        n = best_n
        t = np.arange(n) / n
        
        inits = []
        # Cosine with different frequencies
        for freq in [2, 3, 4, 5, 6, 7, 8, 10, 12, 16]:
            for amp in [0.3, 0.4, 0.5, 0.6]:
                hf = 0.5 + amp * np.cos(2.0 * np.pi * freq * t)
                inits.append(project_feasible(hf, n))
        
        # Sawtooth-like
        for period in [2, 3, 4]:
            hs = np.mod(np.arange(n), period * 2) / (period * 2.0)
            hs = np.where(hs < 0.5, hs * 2.0, (1.0 - hs) * 2.0)
            inits.append(project_feasible(hs, n))
        
        # Binary evenly spaced
        if n % 2 == 0:
            hb = np.zeros(n)
            hb[::2] = 1.0
            inits.append(project_feasible(hb, n))
        
        # Perturbed parent
        for scale in [0.02, 0.05, 0.1]:
            hp = best_h + scale * rng.randn(n)
            inits.append(project_feasible(hp, n))
        
        # Try each init briefly
        for h_init in inits:
            if remaining() < 15:
                break
            h_try = slsqp_optimize(h_init, n, t0, budget_s)
            h_try, c5_try = local_search_gradient(h_try, n, max_sweeps=50, t0=t0, budget_s=budget_s)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n
                # Re-interpolate best_h if resolution changed
                if best_n != n:
                    best_h = project_feasible(interp_to_n(best_h, n), n)
                    best_n = n
    
    # Phase 6: Final polishing
    if remaining() > 30:
        # High-beta SLSQP
        h_pol = project_feasible(best_h, best_n)
        _g_n = best_n
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': _eq_con}]
        
        for beta in [100000, 500000, 2000000]:
            if remaining() < 10:
                break
            _g_beta = beta
            try:
                res = minimize(_smooth_obj, h_pol, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-16, 'iprint': 0})
                h_pol = project_feasible(res.x, best_n)
                c5_p = compute_c5(h_pol)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_pol.copy()
            except Exception:
                pass
        
        # Another local search
        if remaining() > 20:
            h_final, c5_final = local_search_gradient(best_h, best_n, max_sweeps=500, t0=t0, budget_s=budget_s)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    
    # Phase 7: Binary rounding check
    if remaining() > 5:
        h_bin = binary_round(best_h, best_n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    
    # Final projection and verification
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass
    
    return (best_h, final_c5, best_n)
