    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = 0.8
    T_min = 0.005
    cool = 0.99995
    stall = 0
    max_stall = 80000
    start_t = time.time()
    n2_coeff = 2.0 / n

    while time.time() - start_t < time_limit:
        if stall > max_stall:
            T = max(T, 0.3)
            stall = 0
        
        elapsed_frac = (time.time() - start_t) / time_limit
        T_floor = T_min + (0.1 - T_min) * (1.0 - elapsed_frac) ** 2
        T = max(T, T_floor)
        
        batch_size = 200
        for _ in range(batch_size):
            if time.time() - start_t > time_limit:
                break
            if len(ones_list) < 1 or len(zeros_list) < 1:
                break
            
            # Choose move type: 70% 1-swap, 30% 2-swap
            if len(ones_list) >= 2 and len(zeros_list) >= 2 and rng.random() < 0.3:
                i1, i2 = ones_list[rng.randint(len(ones_list))], ones_list[rng.randint(len(ones_list))]
                while i2 == i1:
                    i2 = ones_list[rng.randint(len(ones_list))]
                j1, j2 = zeros_list[rng.randint(len(zeros_list))], zeros_list[rng.randint(len(zeros_list))]
                while j2 == j1:
                    j2 = zeros_list[rng.randint(len(zeros_list))]
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                    ones_list.remove(i1); ones_list.remove(i2)
                    zeros_list.append(i1); zeros_list.append(i2)
                    zeros_list.remove(j1); zeros_list.remove(j2)
                    ones_list.append(j1); ones_list.append(j2)
                    T = max(T * cool, T_floor)
                elif rng.random() < np.exp(-(c5_try - best_c5) / (T * n2_coeff + 1e-12)):
                    ones_list.remove(i1); ones_list.remove(i2)
                    zeros_list.append(i1); zeros_list.append(i2)
                    zeros_list.remove(j1); zeros_list.remove(j2)
                    ones_list.append(j1); ones_list.append(j2)
                    T = max(T * cool, T_floor)
                else:
                    h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
                    stall += 1
                    T = max(T * cool, T_floor)
            else:
                i = ones_list[rng.randint(len(ones_list))]
                j = zeros_list[rng.randint(len(zeros_list))]
                h[i] = 0; h[j] = 1
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h.copy()
                    ones_list.remove(i)
                    zeros_list.append(i)
                    zeros_list.remove(j)
                    ones_list.append(j)
                    T = max(T * cool, T_floor)
                elif rng.random() < np.exp(-(c5_try - best_c5) / (T * n2_coeff + 1e-12)):
                    ones_list.remove(i)
                    zeros_list.append(i)
                    zeros_list.remove(j)
                    ones_list.append(j)
                    T = max(T * cool, T_floor)
                else:
                    h[i] = 1; h[j] = 0
                    stall += 1
                    T = max(T * cool, T_floor)
    
    return best_h, best_c5

def greedy_1swap(h_init, n, rng, time_limit=60.0, max_stall=30000):
    h = h_init.copy()
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    start_t = time.time()
    
    while time.time() - start_t < time_limit and stall < max_stall:
        if not ones_list or not zeros_list:
            break
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        h[i] = 0; h[j] = 1
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h.copy()
            ones_list.remove(i)
            zeros_list.append(i)
            zeros_list.remove(j)
            ones_list.append(j)
            stall = 0
        else:
            h[i] = 1; h[j] = 0
            stall += 1
    return best_h, best_c5

def targeted_2swap(h_init, n, rng, time_limit=120.0, max_stall=40000):
    h = h_init.copy()
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    start_t = time.time()
    
    while time.time() - start_t < time_limit and stall < max_stall:
        if len(ones_list) < 2 or len(zeros_list) < 2:
            break
        
        # Find worst shift and target it
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_k = np.argmax(corr)
        worst_s = worst_k - (n - 1)
        
        # Find bad positions at worst shift
        bad_ones = []
        if worst_s > 0:
            for i in range(worst_s, n):
                if h[i] == 1.0 and h[i - worst_s] == 0.0:
                    bad_ones.append(i)
        elif worst_s < 0:
            ss = -worst_s
            for i in range(n - ss):
                if h[i] == 1.0 and h[i + ss] == 0.0:
                    bad_ones.append(i)
        
        if bad_ones:
            i = bad_ones[rng.randint(len(bad_ones))]
        else:
            i = ones_list[rng.randint(len(ones_list))]
        
        j = zeros_list[rng.randint(len(zeros_list))]
        if i == j:
            continue
        
        # Try also a second swap for 2-swap
        if len(ones_list) >= 2 and len(zeros_list) >= 2 and rng.random() < 0.4:
            i2 = ones_list[rng.randint(len(ones_list))]
            if i2 == i:
                i2 = ones_list[(ones_list.index(i) + 1) % len(ones_list)]
            j2 = zeros_list[rng.randint(len(zeros_list))]
            if j2 == j:
                j2 = zeros_list[(zeros_list.index(j) + 1) % len(zeros_list)]
            if j2 == i:
                j2 = zeros_list[rng.randint(len(zeros_list))]
            
            h[i] = 0; h[i2] = 0; h[j] = 1; h[j2] = 1
            c5_try = compute_c5(h)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h.copy()
                ones_list.remove(i); ones_list.remove(i2)
                zeros_list.append(i); zeros_list.append(i2)
                zeros_list.remove(j); zeros_list.remove(j2)
                ones_list.append(j); ones_list.append(j2)
                stall = 0
            else:
                h[i] = 1; h[i2] = 1; h[j] = 0; h[j2] = 0
                stall += 1
        else:
            h[i] = 0; h[j] = 1
            c5_try = compute_c5(h)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_h = h.copy()
                ones_list.remove(i)
                zeros_list.append(i)
                zeros_list.remove(j)
                ones_list.append(j)
                stall = 0
            else:
                h[i] = 1; h[j] = 0
                stall += 1
    
    return best_h, best_c5

def evolutionary_search(h_init, n, rng, time_limit=200.0, pop_size=30):
    # Initialize population
    pop = [h_init.copy()]
    pop_c5 = [compute_c5(h_init)]
    
    while len(pop) < pop_size:
        h_m = h_init.copy()
        n_flip = rng.randint(2, 8)
        ones = np.where(h_m == 1.0)[0]
        zeros = np.where(h_m == 0.0)[0]
        flips_1_to_0 = rng.choice(ones, min(n_flip // 2, len(ones)), replace=False)
        flips_0_to_1 = rng.choice(zeros, min(n_flip - len(flips_1_to_0), len(zeros)), replace=False)
        h_m[flips_1_to_0] = 0.0
        h_m[flips_0_to_1] = 1.0
        h_m = make_binary(h_m, n, rng)
        pop.append(h_m)
        pop_c5.append(compute_c5(h_m))
    
    best_c5 = min(pop_c5)
    best_h = pop[pop_c5.index(best_c5)].copy()
    start_t = time.time()
    
    while time.time() - start_t < time_limit:
        # Sort by fitness
        order = np.argsort(pop_c5)
        pop = [pop[i] for i in order]
        pop_c5 = [pop_c5[i] for i in order]
        
        if pop_c5[0] < best_c5 - 1e-14:
            best_c5 = pop_c5[0]
            best_h = pop[0].copy()
        
        # Elitism: keep top half
        keep = max(2, pop_size // 2)
        new_pop = [pop[i].copy() for i in range(keep)]
        new_c5 = pop_c5[:keep]
        
        # Generate next generation
        while len(new_pop) < pop_size:
            p1 = new_pop[rng.randint(keep)].copy()
            p2 = new_pop[rng.randint(keep)].copy()
            
            # Crossover
            cross_pt = rng.randint(1, n - 1)
            child = np.concatenate([p1[:cross_pt], p2[cross_pt:]])
            
            # Mutation
            n_mut = rng.choice([1, 2, 2, 3, 4])
            child_ones = np.where(child == 1.0)[0]
            child_zeros = np.where(child == 0.0)[0]
            if len(child_ones) > 0 and len(child_zeros) > 0:
                for _ in range(n_mut):
                    if rng.random() < 0.5 and len(child_ones) > 0:
                        idx = child_ones[rng.randint(len(child_ones))]
                        child[idx] = 0.0
                        child_ones = np.where(child == 1.0)[0]
                        child_zeros = np.where(child == 0.0)[0]
                    elif len(child_zeros) > 0:
                        idx = child_zeros[rng.randint(len(child_zeros))]
                        child[idx] = 1.0
                        child_ones = np.where(child == 1.0)[0]
                        child_zeros = np.where(child == 0.0)[0]
            
            child = make_binary(child, n, rng)
            c5_val = compute_c5(child)
            new_pop.append(child)
            new_c5.append(c5_val)
        
        pop = new_pop
        pop_c5 = new_c5
        
        # Local search on best of generation
        if time.time() - start_t < time_limit - 30:
            gen_best_idx = np.argmin(pop_c5)
            h_ls, c5_ls = greedy_1swap(pop[gen_best_idx], n, rng, time_limit=min(10.0, (time_limit - (time.time() - start_t)) * 0.1))
            if c5_ls < pop_c5[gen_best_idx]:
                pop[gen_best_idx] = h_ls
                pop_c5[gen_best_idx] = c5_ls
    
    return best_h, best_c5

def slsqp_polish(h_init, n, max_iter=2000):
    h0 = np.clip(h_init.astype(float), 0.0, 1.0)
    target = n / 2.0
    # Fix sum
    diff = target - np.sum(h0)
    mask = (h0 > 1e-6) & (h0 < 1.0 - 1e-6)
    if np.count_nonzero(mask) > 0:
        h0[mask] += diff / np.count_nonzero(mask)
    h0 = np.clip(h0, 0.0, 1.0)
    
    def obj(x):
        hh = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hh, 1.0 - hh, mode='full')
        cmax = np.max(corr)
        # Smooth max
        beta = 50.0
        shifted = beta * (corr - cmax)
        shifted[shifted > 50] = 50
        return (cmax + (1.0 / beta) * np.log(np.sum(np.exp(shifted)))) * (2.0 / n)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=con,
                      options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < 10:
            h_out = np.clip(res.x, 0.0, 1.0)
            # Ensure sum constraint
            diff2 = target - np.sum(h_out)
            mask2 = (h_out > 1e-8) & (h_out < 1.0 - 1e-8)
            if np.count_nonzero(mask2) > 0:
                h_out[mask2] += diff2 / np.count_nonzero(mask2)
            h_out = np.clip(h_out, 0.0, 1.0)
            return h_out, compute_c5(h_out)
    except Exception:
        pass
    return h_init, compute_c5(h_init)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    h_parent = None
    n_parent = 256
    try:
        h_parent_arr = np.array(initial_h_values, dtype=float)
        if h_parent_arr is not None and len(h_parent_arr) > 0:
            h_parent = h_parent_arr.copy()
            n_parent = len(h_parent)
    except Exception:
        pass
    
    best_h = None
    best_c5 = float('inf')
    best_n = 256
    
    # Primary n values to try
    n_values = [256]
    if h_parent is not None and n_parent != 256:
        n_values.insert(0, n_parent)
    n_values += [320, 384]
    # Deduplicate while preserving order
    seen = set()
    n_unique = []
    for nv in n_values:
        if nv not in seen and nv % 2 == 0 and nv >= 64:
            n_unique.append(nv)
            seen.add(nv)
    n_values = n_unique
    
    for n in n_values:
        if remaining() < 60:
            break
        
        # Initialize candidates for this n
        candidates = []
        
        if h_parent is not None:
            if n == n_parent:
                hb = make_binary(h_parent, n, rng)
                candidates.append(hb)
            else:
                hb = interp_to_n(h_parent, n, rng)
                candidates.append(hb)
        
        # Thue-Morse
        candidates.append(thue_morse_binary(n, rng))
        
        # Antisymmetric Thue-Morse
        half = n // 2
        h_as = np.zeros(n)
        for i in range(half):
            h_as[i] = 1.0 if bin(i * 5 + 3).count('1') % 2 == 0 else 0.0
            h_as[i + half] = 1.0 - h_as[i]
        candidates.append(make_binary(h_as, n, rng))
        
        # Random
        for _ in range(2):
            h_r = (rng.rand(n) > 0.5).astype(float)
            h_r = make_binary(h_r, n, rng)
            candidates.append(h_r)
        
        # Phase-shifted Thue-Morse
        for shift in [1, 3]:
            h_s = np.zeros(n)
            for i in range(n):
                h_s[i] = 1.0 if bin((i + shift) % n).count('1') % 2 == 0 else 0.0
            candidates.append(make_binary(h_s, n, rng))
        
        # Evaluate all candidates
        cand_c5 = [compute_c5(c) for c in candidates]
        best_cand_idx = np.argmin(cand_c5)
        
        if cand_c5[best_cand_idx] < best_c5:
            best_c5 = cand_c5[best_cand_idx]
            best_h = candidates[best_cand_idx].copy()
            best_n = n
        
        # SA on best candidate
        if remaining() > 80:
            sa_time = min(remaining() * 0.4, 350.0)
            h_sa, c5_sa = sa_local_search(candidates[best_cand_idx], n, rng, time_limit=sa_time)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n
        
        # Also run SA on second best if time permits
        second_idx = np.argsort(cand_c5)[1] if len(cand_c5) > 1 else 0
        if remaining() > 60 and second_idx != best_cand_idx:
            sa_time2 = min(remaining() * 0.25, 150.0)
            h_sa2, c5_sa2 = sa_local_search(candidates[second_idx], n, rng, time_limit=sa_time2)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                best_n = n
        
        # Targeted search
        if remaining() > 40:
            tt_time = min(remaining() * 0.2, 100.0)
            h_t, c5_t = targeted_2swap(best_h if best_n == n else candidates[best_cand_idx], n, rng, time_limit=tt_time)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
                best_n = n
        
        # Greedy 1-swap polish
        if remaining() > 30:
            gt_time = min(remaining() * 0.1, 50.0)
            h_g, c5_g = greedy_1swap(best_h if best_n == n else candidates[best_cand_idx], n, rng, time_limit=gt_time)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
                best_n = n
    
    # Final phase: binary local search with multiple restarts on best solution
    if best_h is not None and remaining() > 50:
        # Multiple greedy restarts from best
        for restart in range(3):
            if remaining() < 30:
                break
            h_r, c5_r = greedy_1swap(best_h, best_n, rng, time_limit=min(remaining() * 0.15, 30.0))
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            
            # Perturb and re-search
            h_p = best_h.copy()
            ones_p = np.where(h_p == 1.0)[0]
            zeros_p = np.where(h_p == 0.0)[0]
            n_pert = rng.randint(2, 6)
            if len(ones_p) >= n_pert and len(zeros_p) >= n_pert:
                f1 = rng.choice(ones_p, n_pert, replace=False)
                f0 = rng.choice(zeros_p, n_pert, replace=False)
                h_p[f1] = 0.0
                h_p[f0] = 1.0
                h_p = make_binary(h_p, best_n, rng)
                if remaining() > 25:
                    h_pr, c5_pr = sa_local_search(h_p, best_n, rng, time_limit=min(remaining() * 0.15, 40.0))
                    if c5_pr < best_c5:
                        best_c5 = c5_pr
                        best_h = h_pr.copy()
    
    # SLSQP polish on best binary solution (try continuous refinement)
    if best_h is not None and remaining() > 25:
        h_sp, c5_sp = slsqp_polish(best_h, best_n, max_iter=1500)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp
    
    # One final round of greedy on the polished solution
    if best_h is not None and remaining() > 15:
        h_fg, c5_fg = greedy_1swap(best_h, best_n, rng, time_limit=min(remaining() * 0.5, 20.0))
        if c5_fg < best_c5:
            best_c5 = c5_fg
            best_h = h_fg.copy()
    
    # Ensure feasibility
    if best_h is None:
        best_h = thue_morse_binary(256, rng)
        best_n = 256
    best_h = np.clip(best_h, 0.0, 1.0)
    # Fix sum
    target = best_n / 2.0
    diff = target - np.sum(best_h)
    if abs(diff) > 1e-10:
        if best_h.min() > 0.01:
            best_h -= diff / best_n
        else:
            mask = best_h < 1.0 - 0.01
            if np.count_nonzero(mask) > 0:
                best_h[mask] += diff / np.count_nonzero(mask)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
