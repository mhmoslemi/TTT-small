        # Active set: lags within tolerance of max
        active = corr >= max_val - 1e-6
        n_active = max(1, int(np.sum(active)))
        
        # Vectorized subgradient computation
        grad = np.zeros(n)
        active_lags = np.where(active)[0]
        for lag_idx in active_lags:
            offset = int(lag_idx) - (n - 1)
            if offset >= 0:
                # corr[offset] = sum_{i=0}^{n-offset-1} h[i]*(1-h[i+offset])
                grad[:n-offset] += (1.0 - h[offset:])
                grad[offset:] += (-h[:n-offset])
            else:
                o = -offset
                # corr[-o] = sum_{i=o}^{n-1} h[i]*(1-h[i-o])
                grad[o:] += (1.0 - h[:n-o])
                grad[:n-o] += (-h[o:])
        
        grad /= n_active
        grad *= dx
        
        # Proposed step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < prev_c5 - 1e-14:
            h = h_new
            prev_c5 = c5_new
        else:
            step *= 0.85
        
        # Plateau detection
        if it % 50 == 0 and it > 0 and step * np.max(np.abs(grad)) < 1e-6:
            break
    
    return h

def binary_local_search(h, n, time_budget=30):

    t_start = time.time()

        ones = np.where(h == 1)[0]
        order = np.argsort(-h[ones])  # least "confident" ones

        h[to_zero] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h == 0)[0]
        order = np.argsort(h[zeros])  # most "confident" zeros
        to_one = order[:n_want - n_ones]
        h[to_one] = 1.0
    
    current_c5 = compute_c5(h)
    corr = get_correlation(h)
    
    for it in range(5000):
        if time.time() - t_start > time_budget:
            break
        
        corr = get_correlation(h)
        max_val = np.max(corr)
        worst = np.argmax(corr)
        offset = worst - (n - 1)
        
        # Find positions contributing to the worst lag
        if offset >= 0:
            contributing = np.where((h[:n-offset] == 1) & (h[offset:] == 0))[0]
            # These are positions i where h[i]=1 and h[i+offset]=0
            # Flipping h[i] from 1 to 0 helps, flipping h[i+offset] from 0 to 1 helps
        else:
            o = -offset
            contributing = np.where((h[o:] == 1) & (h[:n-o] == 0))[0] + o
            # These are positions i where h[i]=1 and h[i-o]=0
        
        if len(contributing) == 0:
            break
        
        improved = False
        # Try flipping pairs
        for pos in contributing[:20]:
            if time.time() - t_start > time_budget:
                break
            # Flip h[pos] (was 1, becomes 0)
            h[pos] = 0.0
            # Find best position to flip from 0 to 1
            zeros = np.where(h == 0)[0]
            zeros = zeros[zeros != pos]
            best_c5 = current_c5
            best_j = -1
            # Sample a subset of zeros for efficiency
            sample_size = min(100, len(zeros))
            sample_idx = rng_sample(zeros, sample_size, pos)
            for j in sample_idx:
                h[j] = 1.0
                c5_try = compute_c5(h)
                h[j] = 0.0
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_j = j
            
            if best_j >= 0:
                h[best_j] = 1.0
                current_c5 = best_c5
                improved = True
                break
            else:
                h[pos] = 1.0  # revert
        
        if not improved:
            # Try a random perturbation to escape local opt
            if it % 100 == 50:
                i1 = np.random.randint(n)
                i2 = np.random.randint(n)
                if h[i1] != h[i2]:
                    h[i1], h[i2] = h[i2], h[i1]
            else:
                break
    
    return h, current_c5

def rng_sample(arr, k, exclude):
    mask = arr != exclude
    arr = arr[mask]
    if len(arr) <= k:
        return arr
    return arr[np.random.choice(len(arr), k, replace=False)]

Let me think about what makes the problem hard. The function h is on [0,2] and we look at h(x) * (1-h(x+k)) for all k. The shift k can be anything, so effectively we're looking at all relative displacements.

For a step function with n points, the shifts are quantized to multiples of 2/n. The maximum correlation typically occurs at small shifts (k close to 0 or close to n).

At shift 0: corr[0] = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2 = n/2 - sum h[i]^2
For binary: n/2 - n/2 = 0. So shift 0 is always 0 for binary!
For continuous: n/2 - sum h[i]^2, which is maximized when h is uniform (all 0.5).

Wait, that's interesting. For binary h, the lag-0 correlation is 0. The interesting lags are the others.

For binary h with n/2 ones:
corr[k] for lag k (offset o = k):
= number of positions i where h[i]=1 and h[i+o]=0

The sum over all k: sum_k corr[k] = sum_i h[i] * (number of j where h[j]=0 and j is in [i+o for valid o])
= sum_i h[i] * (n/2) (approximately, ignoring boundaries)
= (n/2)*(n/2) = n^2/4

So the average corr over all lags is about n^2/4 / (2n-1) ≈ n/8.
The max is what we want to minimize.

For a "random" binary sequence, corr[k] ≈ n/4 ± sqrt(n/4 * 3n/4) ≈ n/4 ± sqrt(3)*n/4.
So max ≈ n/4 + sqrt(3)*n/4/√(2n) ≈ n/4 (1 + sqrt(3/(2n))).

For n=192: max ≈ 48 * (1 + sqrt(3/384)) ≈ 48 * (1 + 0.088) ≈ 52.3
C5 ≈ 52.3 * 2/192 ≈ 0.545

C5 ≈ (n/8) * 2/n = 0.25.

So the theoretical minimum for the "average" is 0.25, but we can't achieve it exactly due to the constraint that individual correlations must be integers (for binary) or have specific structure.

def compute_c5(h):
    n = len(h)

def get_correlation(h):

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        nmask = int(np.sum(mask))
        if nmask == 0:
            break
        adj = diff / nmask
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                adj = min(diff, 1.0 - h[i])
                h[i] += adj
                diff -= adj
            else:
                adj = max(diff, -h[i])
                h[i] += adj
                diff -= adj
            if abs(diff) < 1e-13:
                break
    return h

def slsqp_smooth(h, n, beta, maxiter, time_budget, tol=1e-14):
    t_start = time.time()
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        sh = beta * (c - m)
        sh = np.clip(sh, -500.0, 0.0)
        return (m + np.log(np.sum(np.exp(sh))) / beta) * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': tol, 'iprint': 0})
        if time.time() - t_start > time_budget:
            return h
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_uniformity(h, n, maxiter, time_budget, target_frac=0.5):

    t_start = time.time()

        return h

                                       time_budget=min(15.0, time_left() - 5), tol=1e-8)
            c5_try = compute_c5(h_cur)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_cur.copy()

        # Binary search and polish
        if time_left() > 10:
            h_bin, c5_bin = binary_local_search(best_h, best_n, time_budget=min(20.0, time_left() - 8), rng=rng)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
            # Polish the binary solution
            if time_left() > 8:
                h_pol = slsqp_smooth(best_h, best_n, 5000.0, 200, time_budget=min(10.0, time_left() - 5))
                c5_try = compute_c5(h_pol)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_pol.copy()

    # Final binary rounding attempt
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_want = best_n // 2
    if n_ones > n_want:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(best_h[ones])[:n_ones - n_want]
        h_bin[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-best_h[zeros])[:n_want - n_ones]
        h_bin[zeros[order]] = 1.0
    c5b = compute_c5(h_bin)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = h_bin.copy()

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print(f"Final: n={best_n}, C5={final_c5:.6f}")
    return (best_h, final_c5, best_n)
