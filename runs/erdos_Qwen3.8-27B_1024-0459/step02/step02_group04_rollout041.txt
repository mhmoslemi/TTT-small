def project_feasible(h, n=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.5)
    if n is None:
        n = h.size
    else:
        n = int(n)
    if n <= 0:
        return np.empty(0, dtype=float)
    target = 0.5 * float(n)
    h = h[:n].copy()
    if h.size < n:
        pad = np.zeros(n - h.size, dtype=float)
        h = np.concatenate((h, pad))
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    out = np.clip(h - tau, 0.0, 1.0)
    for _ in range(40):
        diff = target - float(np.sum(out))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            mask = out < 1.0 - 1e-13
        else:
            mask = out > 1e-13
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        out[mask] += diff / float(cnt)
        out = np.clip(out, 0.0, 1.0)
    return out


def compute_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n <= 0:
        return 0.0
    dx = 2.0 / float(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def all_lag_values(h):
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    vals = np.empty(2 * n - 1, dtype=float)
    ac0 = float(np.dot(h, h))
    vals[0] = (total - ac0) * dx
    idx = 1
    for L in range(1, n):
        m = n - L
        ac = float(np.dot(h[:m], h[L:]))
        vals[idx] = (pref[m] - ac) * dx
        vals[idx + 1] = (total - pref[L] - ac) * dx
        idx += 2
    return vals


def smooth_max_fn(h, beta):
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    ac0 = float(np.dot(h, h))
    m = (total - ac0) * dx
    r = 1.0
    for L in range(1, n):
        mm = n - L
        ac = float(np.dot(h[:mm], h[L:]))
        vp = (pref[mm] - ac) * dx
        vn = (total - pref[L] - ac) * dx
        if vp > m:
            d = beta * (m - vp)
            if d > -745.0:
                r = r * np.exp(d)
            else:
                r = 0.0
            r += 1.0
            m = vp
        else:
            d = beta * (vp - m)
            if d > -745.0:
                r += np.exp(d)
        if vn > m:
            d = beta * (m - vn)
            if d > -745.0:
                r = r * np.exp(d)
            else:
                r = 0.0
            r += 1.0
            m = vn
        else:
            d = beta * (vn - m)
            if d > -745.0:
                r += np.exp(d)
    if r <= 0.0:
        return m
    return m + np.log(r) / beta


def smooth_max_grad(h, beta):
    n = h.size
    dx = 2.0 / float(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = float(pref[-1])
    ac0 = float(np.dot(h, h))
    maxv = (total - ac0) * dx
    posv = np.empty(n, dtype=float)
    negv = np.empty(n, dtype=float)
    posv[0] = maxv
    negv[0] = maxv
    for L in range(1, n):
        mm = n - L
        ac = float(np.dot(h[:mm], h[L:]))
        vp = (pref[mm] - ac) * dx
        vn = (total - pref[L] - ac) * dx
        posv[L] = vp
        negv[L] = vn
        if vp > maxv:
            maxv = vp
        if vn > maxv:
            maxv = vn
    wpos = np.zeros(n, dtype=float)
    wneg = np.zeros(n, dtype=float)
    for i in range(n):
        d = beta * (posv[i] - maxv)
        if d > -745.0:
            wpos[i] = np.exp(d)
    for i in range(1, n):
        d = beta * (negv[i] - maxv)
        if d > -745.0:
            wneg[i] = np.exp(d)
    total_w = float(wpos.sum() + wneg[1:].sum())
    g = np.zeros(n, dtype=float)
    if total_w <= 0.0:
        return g
    inv = 1.0 / total_w
    for L in range(n):
        cp = float(wpos[L]) * inv * dx
        if cp > 1e-300:
            m = n - L
            g[:m] += cp * (1.0 - h[L:])
            g[L:] += -cp * h[:m]
        if L > 0:
            cm = float(wneg[L]) * inv * dx
            if cm > 1e-300:
                m = n - L
                g[L:] += cm * (1.0 - h[:m])
                g[:m] += -cm * h[L:]
    return g


def lbfgs_smooth_obj(x, beta, lam):
    n = len(x)
    val = smooth_max_fn(x, beta)
    s = float(np.sum(x)) - 0.5 * float(n)
    return val + lam * s * s


def lbfgs_smooth_grad(x, beta, lam):
    n = len(x)
    g = smooth_max_grad(x, beta).copy()
    s = float(np.sum(x)) - 0.5 * float(n)
    g += 2.0 * lam * s
    return g


def optimize_lbfgs(h0, n, betas, time_limit, lam=5000.0, maxiter=400):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0) for _ in range(n)]
    best_h = h.copy()
    best_c = compute_c5(h)
    for beta in betas:
        if time.time() > time_limit - 5.0:
            break
        try:
            res = minimize(
                lbfgs_smooth_obj,
                h,
                jac=lambda x: lbfgs_smooth_grad(x, beta, lam),
                method="L-BFGS-B",
                bounds=bounds,
                options={"maxiter": maxiter, "ftol": 1e-14, "gtol": 1e-10, "disp": False},
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                h2 = project_feasible(res.x, n)
                c2 = compute_c5(h2)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2
                h = h2
        except Exception:
            pass
    return best_h, best_c

ACTIVE_LAGS = []


def lag_val_signed(h, l):
    n = len(h)
    if l == 0:
        return float(np.sum(h) - float(np.dot(h, h)))
    L = abs(int(l))
    m = n - L
    low = h[:m]
    high = h[L:]
    ac = float(np.dot(low, high))
    if l > 0:
        return float(np.sum(low)) - ac
    return float(np.sum(high)) - ac


def lag_grad_signed(h, l):
    n = len(h)
    if l == 0:
        return 1.0 - 2.0 * h
    L = abs(int(l))
    m = n - L
    g = np.zeros(n, dtype=float)
    if l > 0:
        g[:m] += 1.0 - h[L:]
        g[L:] += -h[:m]
    else:
        g[L:] += 1.0 - h[:m]
        g[:m] += -h[L:]
    return g


def choose_active_lags(vals, max_count, margin=0.001):
    if vals.size == 0:
        return []
    n = (vals.size + 1) // 2
    order = np.argsort(-vals)
    take = min(int(max_count), int(vals.size))
    idx = order[:take]
    top = float(vals[order[0]])
    near = np.where(vals >= top - margin)[0]
    if near.size > 0:
        idx = np.unique(np.concatenate((idx, near)))
        idx = idx[np.argsort(-vals[idx])]
        cap = max(int(max_count) * 3, 50)
        if idx.size > cap:
            idx = idx[:cap]
    lags = []
    for i in idx:
        i = int(i)
        if i == n - 1:
            lags.append(0)
        elif i < n - 1:
            lags.append(i - (n - 1) + 1)
        else:
            lags.append(-(i - (n - 1)))
    return lags

def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def local_block_search(h0, n, time_limit, seed, step=0.025):
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.003
    min_temp = 1e-7
    max_trials = 5000
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.03, max(temp * 1.2, min_temp))
        elif c_try < cur_c + 1e-15:
            cur_h = h_try
            cur_c = c_try
            temp = min(0.03, max(temp * 1.05, min_temp))
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                prob = np.exp(-d / temp)
                if rng.rand() < prob:
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, best_c

        if c2 < best_c:
            best_h = h2
            best_c = c2
            best_n = nn

    # Phase 3: Active set refinement on best
    if time.time() < hard_end - 40.0 and 4 <= best_n <= 320:
        if best_n <= 160:
            seq3 = (16, 32, 64, 96)
        else:
            seq3 = (16, 32, 64)
        t3 = hard_end - 30.0
        h2, c2 = refine_active_set(best_h, best_n, t3, seq=seq3, maxiter=50)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 4: Another L-BFGS-B pass on updated best (if changed)
    if time.time() < hard_end - 30.0 and 4 <= best_n <= 320:
        h2, c2 = optimize_lbfgs(best_h, best_n, [200, 500], hard_end - 25.0, lam=5000.0, maxiter=200)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 5: Final local search
    if time.time() < hard_end - 10.0 and 4 <= best_n <= 320:
        h2, c2 = local_block_search(best_h, best_n, hard_end - 5.0, seed + 7777, step=0.02)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Final projection and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
