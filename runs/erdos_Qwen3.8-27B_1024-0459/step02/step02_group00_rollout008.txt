- From first term: (1-h[a-k]) if a ≥ k
- From second term: -h[a+k] if a ≤ n-1-k

For a swap (i: 0→1, j: 1→0), the change in c_k is:

                                   options={'maxiter': 400, 'gtol': 1e-12, 'iprint': 0})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h0 = h_new
                        c5_cur = c5_new
                except Exception:
                    pass

            h0_bin = to_binary(h0, n)

            h0_bin, c5_bin = fast_swap_search(h0_bin, n, best_c5, max_sweeps=200)
            if c5_bin < c5_cur:
                h0 = h0_bin
                c5_cur = c5_bin

            h0, c5_fs = fast_swap_search(h0, n, best_c5, max_sweeps=150)
            if c5_fs < c5_cur:
                h0 = h0
                c5_cur = c5_fs

            if remaining_time() > 15:
                h0, c5_ex = exhaustive_top_swap_search(h0, n, best_c5, max_sweeps=30)
                if c5_ex < c5_cur:
                    h0 = h0
                    c5_cur = c5_ex

            h0, c5_fr = fractional_refine(h0, n, best_c5, max_sweeps=8)
            if c5_fr < c5_cur:
                c5_cur = c5_fr

            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n
                print("  Improved: n={}, C5={:.7f}".format(n, best_c5))

    # Final intensive phase
    if remaining_time() > 20:
        n = best_n
        _N = n
        _TGT = n / 2.0
        h = best_h.copy()
        h = project_feasible(h, n)
        c5_curr = compute_c5(h)

        h_bin = to_binary(h, n)
        h_bin, c5_bin = fast_swap_search(h_bin, n, c5_curr, max_sweeps=400)
        if c5_bin < c5_curr:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            h = h_bin.copy()
            c5_curr = c5_bin
            print("  Final binary swap: C5={:.7f}".format(best_c5))

        h, c5_sub = subgradient_target(h, n, best_c5, max_iters=150)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h.copy()
            h = h.copy()
            print("  Subgradient: C5={:.7f}".format(best_c5))

        h_bin2 = to_binary(h, n)
        h_bin2, c5_bin2 = fast_swap_search(h_bin2, n, best_c5, max_sweeps=200)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
            print("  Post-subgradient swap: C5={:.7f}".format(best_c5))

        h, c5_fr = fractional_refine(best_h, n, best_c5, max_sweeps=10)
        if c5_fr < best_c5:
            best_c5 = c5_fr
            best_h = h.copy()
            print("  Fractional refine: C5={:.7f}".format(best_c5))

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_sum}]
        for beta_val in [3000, 8000]:
            if remaining_time() < 5:
                break
            _BETA = beta_val
            try:
                res = minimize(obj_smooth, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
