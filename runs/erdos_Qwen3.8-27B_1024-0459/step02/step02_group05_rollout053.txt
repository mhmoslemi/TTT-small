            # Binary search for minimum of max_k (intercepts[k] + t * slopes[k]) over t in [0,1]
            t_lo = 0.0
            t_hi = 1.0
            for _ in range(60):
                t_mid = 0.5 * (t_lo + t_hi)
                vals = intercepts + t_mid * slopes
                k_active = np.argmax(vals)
                if slopes[k_active] > 1e-10:
                    t_hi = t_mid
                elif slopes[k_active] < -1e-10:
                    t_lo = t_mid
                else:
                    t_lo = t_mid
                    t_hi = t_mid
                    break
            t_new = 0.5 * (t_lo + t_hi)

            h_old = h[i]
            h[i] = t_new
            c5_new = compute_c5(h)
            if c5_new < compute_c5(np.where(h[i] != t_new, h, h_old + 0)):
                pass
            # Check if improvement
            h[i] = h_old
            c5_old = compute_c5(h)
            h[i] = t_new
            c5_try = compute_c5(h)
            if c5_try < c5_old - 1e-15:
                improved = True
            else:
                h[i] = h_old
        if not improved and sweep > 5:
            break
    return h

def active_set_polish(h, n, n_iters=200):
    h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        cmax = np.max(corr) * dx
        top_k = np.argwhere(corr >= np.max(corr) - 1e-10)[:, 0]
        if len(top_k) == 0:
            top_k = [np.argmax(corr)]
        
        # For each active lag, compute gradient direction
        grad = np.zeros(n)
        for idx in top_k:
            s = idx - (n - 1)
            if s >= 0:
                if s < n:
                    grad[:n - s] += (1.0 - h[s:])
                    grad[s:] -= h[:n - s]
            else:
                sp = -s
                grad[sp:] += (1.0 - h[:n - sp])
                grad[:n - sp] -= h[sp:]
        
        # Projected gradient step: reduce h where gradient is positive (increases active corr)
        # We want to DECREASE the max correlation, so move AGAINST the gradient
        step = 0.01 / max(len(top_k), 1)
        h_try = h - step * grad
        h_try = np.clip(h_try, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_try)
        mask = (h_try > 0.001) & (h_try < 0.999)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h_try[mask] += diff / cnt
        h_try = np.clip(h_try, 0.0, 1.0)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            h = h_try
        else:
            step *= 0.5
            h_try = h - step * grad
            h_try = np.clip(h_try, 0.0, 1.0)
            diff = n / 2.0 - np.sum(h_try)
            mask = (h_try > 0.001) & (h_try < 0.999)
            cnt = np.count_nonzero(mask)
            if cnt > 0:
                h_try[mask] += diff / cnt
            h_try = np.clip(h_try, 0.0, 1.0)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                h = h_try
    return h, best_c5

def local_pair_search(h, n, n_rounds, rng):
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for trial in range(200):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for delta in [0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05]:
                h_try = h.copy()
                h_try[i] += delta
                h_try[j] -= delta
                if h_try[i] < 0 or h_try[i] > 1 or h_try[j] < 0 or h_try[j] > 1:
                    continue
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.ones(n_dst) * 0.5
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    result = np.interp(x_new, x_old, h_src)
    return result

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float)
        if h_parent is None or len(h_parent) == 0:
            raise ValueError
    except Exception:
        h_parent = None

    if h_parent is not None:
        best_h = h_parent.copy()
        best_n = len(best_h)
    else:
        best_n = 160
        best_h = np.ones(best_n) * 0.5

    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with LSE gradient at parent resolution
    if remaining() > 100:
        h_opt, c5_opt = slsqp_lse(best_h, best_n, beta=500, max_iter=3000, time_limit=remaining() - 20)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Coordinate descent polish
    if remaining() > 30:
        best_h = coord_descent_1d(best_h, beta=1000, n_sweeps=50)
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 3: Active set polish
    if remaining() > 30:
        best_h, best_c5 = active_set_polish(best_h, best_n, n_iters=300)

    # Phase 4: Higher beta SLSQP
    if remaining() > 120:
        h_opt, c5_opt = slsqp_lse(best_h, best_n, beta=2000, max_iter=3000, time_limit=remaining() - 20)
        if h_opt is not None and c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Coordinate descent again
    if remaining() > 30:
        best_h = coord_descent_1d(best_h, beta=2000, n_sweeps=40)
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 6: Try higher resolutions
    for n_try in [200, 256, 320]:
        if remaining() < 150:
            break
        if n_try % 2 != 0:
            n_try += 1
        h_interp = interpolate_to_n(best_h, n_try)
        h_interp = make_feasible(h_interp, n_try)
        c5_interp = compute_c5(h_interp)

        h_opt, c5_opt = slsqp_lse(h_interp, n_try, beta=500, max_iter=2000, time_limit=remaining() - 60)
        if h_opt is not None:
            if c5_opt < c5_interp:
                h_interp = h_opt
                c5_interp = c5_opt
            h_interp = coord_descent_1d(h_interp, beta=1000, n_sweeps=30)
            h_interp = make_feasible(h_interp, n_try)
            c5_interp = compute_c5(h_interp)

        if remaining() > 60:
            h_opt, c5_opt = slsqp_lse(h_interp, n_try, beta=2000, max_iter=2000, time_limit=remaining() - 20)
            if h_opt is not None and c5_opt < c5_interp:
                h_interp = h_opt
                c5_interp = compute_c5(h_interp)
            h_interp = coord_descent_1d(h_interp, beta=2000, n_sweeps=20)
            h_interp = make_feasible(h_interp, n_try)
            c5_interp = compute_c5(h_interp)

        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp
            best_n = n_try

    # Phase 7: Local pair search on best
    if remaining() > 30:
        n_rounds = min(30, int(remaining() / 2))
        best_h, best_c5 = local_pair_search(best_h, best_n, n_rounds, rng)

    # Phase 8: Final coordinate descent
    if remaining() > 20:
        best_h = coord_descent_1d(best_h, beta=5000, n_sweeps=30)
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 9: Ultra-fine local search
    if remaining() > 15:
        n_rounds = min(20, int(remaining() / 1.5))
        best_h, best_c5 = local_pair_search(best_h, best_n, n_rounds, rng)

    # Phase 10: Final active set
    if remaining() > 10:
        best_h, best_c5 = active_set_polish(best_h, best_n, n_iters=min(200, int(remaining() * 10)))
        best_h = make_feasible(best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Final feasibility
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
