def run_slsqp(h0, n, beta, max_iter):

    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500, 0)
        exps = np.exp(shifted)
        weights = exps / np.sum(exps)
        return dx * weighted_gradient(x, n, weights)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15, 'iprint': 0})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def make_structured_init(n, rng):
    """Create a structured initialization based on known good patterns."""
    # Try a periodic pattern with some randomness
    period = rng.choice([3, 4, 5, 6, 7, 8])
    pattern = rng.choice([0, 1], size=period, p=[0.35, 0.65])
    h = np.tile(pattern, n // period + 1)[:n].astype(float)
    # Add some noise
    noise = rng.uniform(-0.3, 0.3, n)
    h = np.clip(h + noise, 0, 1)
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed() - 5.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Get parent
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        parent_n = len(parent_h)
        if parent_n > 0:
            parent_c5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = parent_c5
            best_n = parent_n
            print("Parent: n={}, C5={:.6f}".format(parent_n, parent_c5))
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.6f}".format(best_n, best_c5))

    # Phase 1: Dual gradient descent at parent resolution
    print("Phase 1: Dual GD at n={}".format(best_n))
    n1 = best_n
    h_cur = best_h.copy()
    for eta_val in [0.2, 0.4, 0.8]:
        if remaining() < 30:
            break
        n_iter = 800 if eta_val < 0.5 else 500
        h_new, c5_new = dual_gradient_descent(h_cur, n1, n_iter, eta=eta_val, step=0.05)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            print("  Dual GD eta={}: C5={:.6f}".format(eta_val, best_c5))
        h_cur = h_new

    # Smooth max polish at parent
    if remaining() > 40:
        for beta in [100, 300, 800]:
            if remaining() < 25:
                break
            h_new, c5_new = smooth_max_opt(best_h, best_n, beta, 400, step=0.008)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  SmoothMax beta={}: C5={:.6f}".format(beta, best_c5))

    # Phase 2: Try higher resolutions
    print("Phase 2: Higher resolutions")
    for n_try in [250, 300, 400, 500]:
        if remaining() < 80:
            break
        print("  Trying n={}".format(n_try))
        h_up = interp_to_n(best_h, n_try)

        # Dual GD at new resolution
        h_new, c5_new = dual_gradient_descent(h_up, n_try, 600, eta=0.4, step=0.04)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n_try
            print("    Dual GD: C5={:.6f}".format(best_c5))
        h_cur = h_new

        # Smooth max at new resolution
        if remaining() > 40:
            h_new2, c5_new2 = smooth_max_opt(h_cur, n_try, 200, 300, step=0.006)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()
                best_n = n_try
                print("    SmoothMax: C5={:.6f}".format(best_c5))

    # SLSQP at a few resolutions
    for n_slq in [best_n, 200]:
        if remaining() < 50:
            break
        try:
            h_slq, c5_slq = run_slsqp(best_h if best_n == n_slq else interp_to_n(best_h, n_slq),
                                       n_slq, 300, 250)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()
                best_n = n_slq
                print("  SLSQP n={}: C5={:.6f}".format(n_slq, best_c5))
        except Exception:
            pass

    # Phase 3: Binary refinement
    print("Phase 3: Binary refinement")
    for n_bin in [best_n, 200, 250]:
        if remaining() < 60:
            break
        h_use = best_h if best_n == n_bin else interp_to_n(best_h, n_bin)
        h_bin = (h_use > 0.5).astype(float)
        # Fix sum
        n_ones = int(h_bin.sum())
        n_want = n_bin // 2
        if n_ones > n_want:
            ones_idx = np.where(h_bin == 1.0)[0]
            vals = h_use[ones_idx]
            order = np.argsort(vals)[:n_ones - n_want]
            h_bin[ones_idx[order]] = 0.0
        elif n_ones < n_want:
            zeros_idx = np.where(h_bin == 0.0)[0]
            vals = h_use[zeros_idx]
            order = np.argsort(-vals)[:n_want - n_ones]
            h_bin[zeros_idx[order]] = 1.0

        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n_bin
            print("  Binary n={}: C5={:.6f}".format(n_bin, c5_bin))

        # Local search
        n_rounds = min(300, int(remaining() // 0.5))
        h_bs, c5_bs = binary_local_search(h_bin, n_bin, n_rounds, rng)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()
            best_n = n_bin
            print("  LocalSearch n={}: C5={:.6f}".format(n_bin, c5_bs))

    # Phase 4: Multi-start with different initializations
    if remaining() > 100:
        print("Phase 4: Multi-start")
        n_ms = best_n if best_n in [200, 250, 300] else 200
        for trial in range(5):
            if remaining() < 40:
                break
            # Different initializations
            if trial == 0:
                h0 = interp_to_n(best_h, n_ms)
            elif trial == 1:
                h0 = make_structured_init(n_ms, rng)
            elif trial == 2:
                # Shifted parent
                h_shift = np.roll(best_h, rng.randint(1, n_ms // 2))
                h0 = project_feasible(h_shift, n_ms)
            elif trial == 3:
                # Inverted parent
                h_inv = 1.0 - best_h
                # Need to fix: sum should be n/2, currently n - n/2 = n/2. OK!
                h0 = project_feasible(h_inv, n_ms)
            else:
                # Random smooth
                freq = rng.uniform(1, 4)
                phase = rng.uniform(0, 2 * np.pi)
                h0 = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n_ms) / n_ms + phase)
                h0 = project_feasible(h0, n_ms)

            h_new, c5_new = dual_gradient_descent(h0, n_ms, 500, eta=0.5, step=0.05)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_ms
                print("  Multi-start {}: C5={:.6f}".format(trial, best_c5))

    # Phase 5: Final continuous polish
    if remaining() > 30:
        print("Phase 5: Final polish")
        for beta in [500, 1000]:
            if remaining() < 15:
                break
            h_new, c5_new = smooth_max_opt(best_h, best_n, beta, 300, step=0.004)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                print("  Final smooth beta={}: C5={:.6f}".format(beta, best_c5))

    # Final binary check
    if remaining() > 10:
        h_fbin = (best_h > 0.5).astype(float)
        n_ones = int(h_fbin.sum())
        n_want = best_n // 2
        if n_ones > n_want:
            oi = np.where(h_fbin == 1.0)[0]
            order = np.argsort(best_h[oi])[:n_ones - n_want]
            h_fbin[oi[order]] = 0.0
        elif n_ones < n_want:
            zi = np.where(h_fbin == 0.0)[0]
            order = np.argsort(-best_h[zi])[:n_want - n_ones]
            h_fbin[zi[order]] = 1.0
        c5_fbin = compute_c5(h_fbin)
        if c5_fbin < best_c5:
            best_c5 = c5_fbin
            best_h = h_fbin.copy()
            print("  Final binary: C5={:.6f}".format(best_c5))

    # Ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.6f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
