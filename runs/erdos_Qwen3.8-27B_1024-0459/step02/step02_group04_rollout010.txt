    j[:-1] = 1.0
    return j


def active_ineq(x):
    h = x[:-1]
    n = len(h)
    dx = 2.0 / float(n)
    out = np.empty(len(AC_LAGS), dtype=float)
    for i, L in enumerate(AC_LAGS):
        out[i] = float(x[-1]) - lag_gradeval(h, int(L))
    return out


def active_ineq_jac(x):
    h = x[:-1]
    n = len(h)
    dx = 2.0 / float(n)
    m = len(AC_LAGS)
    J = np.zeros((m, n + 1), dtype=float)
    for i, L in enumerate(AC_LAGS):
        g = lag_grad(h, int(L))
        J[i, :n] = -dx * g
        J[i, n] = 1.0
    return J


def optimize_active_set(h0, active_lags, t0, maxiter=150, ftol=1e-13):
    global AC_LAGS
    n = len(h0)
    AC_LAGS = [int(L) for L in active_lags]
    x0 = np.empty(n + 1, dtype=float)
    x0[:n] = h0
    x0[n] = float(t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [{"type": "eq", "fun": active_sum_eq, "jac": active_sum_eq_jac}]
    if len(AC_LAGS) > 0:
        constraints.append({"type": "ineq", "fun": active_ineq, "jac": active_ineq_jac})
    try:
        res = minimize(active_t_obj, x0, jac=active_t_obj_jac,
                       method="SLSQP", bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": ftol, "disp": False})
        return res
    except Exception:
        return None


def choose_active(h, max_lags=80, margin=3e-5):
    n = len(h)
    lv = get_lag_values(h)
    top = float(np.max(lv))
    center = n - 1
    near = np.where(lv >= top - margin)[0]
    lags = [int(p - center) for p in near]
    if len(lags) > max_lags:
        order = np.argsort(-lv)
        lags = [int(order[i] - center) for i in range(max_lags)]
    if len(lags) == 0:
        lags = [0]
    return lags, top


def refine_active(h0, time_limit, max_passes=8, max_lags=100):
    n = len(h0)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    for _ in range(max_passes):
        if time.time() > time_limit - 3.0:
            break
        lags, cur_c = choose_active(h, max_lags=max_lags, margin=max(3e-5, 0.02 * cur_c))
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        t0 = cur_c + 1e-5
        res = optimize_active_set(h, lags, t0, maxiter=130, ftol=1e-13)
        if res is not None and res.x is not None and len(res.x) == n + 1:
            h2 = project_feasible(res.x[:n], n)
            c2 = compute_c5(h2)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()
            if c2 <= cur_c + 1e-8:
                h = h2
    return best_h, best_c


def smooth_obj_fn(x):
    h = x
    n = h.size
    beta = SM_BETA
    dx = 2.0 / float(n)
    lv = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(lv))
    shifted = beta * (lv - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    r = float(np.sum(np.exp(shifted)))
    return float(m + np.log(r) / beta)


def smooth_jac_fn(x):
    h = x
    n = h.size
    beta = SM_BETA
    dx = 2.0 / float(n)
    lv = np.correlate(h, 1.0 - h, mode="full") * dx
    m = float(np.max(lv))
    shifted = beta * (lv - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    total_w = float(w.sum())
    if total_w < 1e-300:
        return dx * lag_grad(h, 0)
    center = n - 1
    g = np.zeros(n, dtype=float)
    for p in range(2 * n - 1):
        L = p - center
        cp = float(w[p]) / total_w
        if cp > 1e-10:
            g += cp * dx * lag_grad(h, L)
    return g


def smooth_sum_eq(x):
    return float(np.sum(x) - 0.5 * len(x))


def smooth_sum_eq_jac(x):
    return np.ones(len(x), dtype=float)


def optimize_smooth(h0, beta, maxiter=80):
    global SM_BETA
    h = project_feasible(h0)
    n = h.size
    SM_BETA = max(1.0, float(beta))
    bounds = [(0.0, 1.0)] * n
    constraints = [{"type": "eq", "fun": smooth_sum_eq, "jac": smooth_sum_eq_jac}]
    try:
        res = minimize(smooth_obj_fn, h, jac=smooth_jac_fn,
                       method="SLSQP", bounds=bounds, constraints=constraints,
                       options={"maxiter": maxiter, "ftol": 1e-11, "disp": False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            h2 = project_feasible(res.x, n)
            return h2, compute_c5(h2)
    except Exception:
        pass
    return h, compute_c5(h)


def fista_smooth(h0, time_limit, beta=200, maxiter=5000):
    global SM_BETA
    n = len(h0)
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    SM_BETA = max(1.0, float(beta))
    L = 30.0
    step = 1.0 / L
    t_k = 1.0
    y = h.copy()
    h_prev = h.copy()
    for it in range(maxiter):
        if time.time() > time_limit - 2.0:
            break
        f_y = smooth_obj_fn(y)
        g_y = smooth_jac_fn(y)
        z = y - step * g_y
        h_new = project_feasible(z, n)
        f_new = compute_c5(h_new)
        if f_new < best_c:
            best_c = f_new
            best_h = h_new.copy()
        if it > 0:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = h_new + ((t_k - 1.0) / t_new) * (h_new - h_prev)
            t_k = t_new
        else:
            y = h_new.copy()
        h_prev = h
        h = h_new
    return best_h, best_c


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        return np.full(n_dst, float(h_src[0]) if n_src == 1 else 0.5)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def gaussian_smooth(h, sigma=1.5):
    n = len(h)
    length = min(7, n)
    if length % 2 == 0:
        length -= 1
    half = length // 2
    k = np.arange(-half, half + 1, dtype=float)
    w = np.exp(-(k * k) / (2.0 * sigma * sigma))
    w /= w.sum()
    return np.convolve(h, w, mode="same")


def block_perturb(h0, time_limit, step=0.015, max_trials=10000, seed=99991):
    n = len(h0)
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    cur_h = h
    cur_c = c
    temp = 0.002
    min_temp = 5e-8
    for _ in range(max_trials):
        if time.time() > time_limit:
            break
        max_len = min(6, n)
        L = int(rng.randint(1, max_len + 1))
        st = int(rng.randint(0, n - L + 1))
        idx = np.arange(st, st + L)
        delta = rng.randn(L) * step
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-14:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.012, max(temp * 1.08, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp and rng.rand() < np.exp(-d / temp):
                cur_h = h_try
                cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)


def structured_start(n, rng):
    starts = []
    h = np.zeros(n, dtype=float)
    q = max(1, n // 4)
    h[:q] = 1.0
    h[n // 2:n // 2 + q] = 1.0
    starts.append(project_feasible(h, n))
    h = np.zeros(n, dtype=float)
    for i in range(n):
        h[i] = 1.0 if (i % 4 == 0 or i % 4 == 1) else 0.0
    starts.append(project_feasible(h, n))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for freq in (1, 2, 3):
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(freq * np.pi * x + phase)
    starts.append(project_feasible(h, n))
    return starts


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    start_time = time.time()
    hard_end = start_time + max(120.0, budget_s - 30.0)
    rng = np.random.RandomState(seed)

    best_h = None
    best_c = np.inf
    best_n = 0

    try:
        if initial_h_values is not None:
            parent = np.asarray(initial_h_values, dtype=float).copy()
            if parent.size > 0:
                pn = int(parent.size)
                raw_c = None
                if (float(np.min(parent)) >= -1e-12 and
                        float(np.max(parent)) <= 1.0 + 1e-12):
                    raw_c = compute_c5(parent)
                    best_h = parent.copy()
                    best_c = raw_c
                    best_n = pn
                proj = project_feasible(parent, pn)
                proj_c = compute_c5(proj)
                if raw_c is None or proj_c < raw_c:
                    best_h = proj
                    best_c = proj_c
                    best_n = pn
    except Exception:
        pass

    if best_h is None:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Aggressive active set at parent resolution
    if time.time() < hard_end - 300:
        t_end1 = min(hard_end - 280, start_time + 300)
        h2, c2 = refine_active(best_h, t_end1, max_passes=10, max_lags=60)
        if c2 < best_c:
            best_h, best_c = h2, c2
        best_n = len(best_h)

    # Phase 2: Smooth max with SLSQP at different betas
    if time.time() < hard_end - 200:
        t_end2 = min(hard_end - 180, start_time + 450)
        for beta in (30, 100, 300):
            if time.time() > t_end2 - 10:
                break
            h2, c2 = optimize_smooth(best_h, beta, maxiter=60)
            if c2 < best_c:
                best_h, best_c = h2, c2
                best_n = len(best_h)

    # Phase 3: FISTA on smooth max
    if time.time() < hard_end - 120:
        t_end3 = min(hard_end - 100, start_time + 600)
        for beta in (80, 250):
            if time.time() > t_end3 - 10:
                break
            h2, c2 = fista_smooth(best_h, t_end3, beta=beta, maxiter=4000)
            if c2 < best_c:
                best_h, best_c = h2, c2
                best_n = len(best_h)

    # Phase 4: More active set passes at parent resolution
    if time.time() < hard_end - 100:
        t_end4 = min(hard_end - 80, start_time + 750)
        h2, c2 = refine_active(best_h, t_end4, max_passes=12, max_lags=80)
        if c2 < best_c:
            best_h, best_c = h2, c2
            best_n = len(best_h)

    # Phase 5: Try neighboring resolutions
    if time.time() < hard_end - 120:
        try_ns = []
        for dn in (-32, -16, 16, 32, -48, 48, -64, 64):
            tn = best_n + dn
            if 48 <= tn <= 400 and tn != best_n:
                try_ns.append(int(tn))
        for tn in try_ns[:4]:
            if time.time() > hard_end - 100:
                break
            rem = hard_end - 100 - time.time()
            if rem < 30:
                break
            slot = min(80.0, rem)
            h_interp = interpolate_to_n(best_h, tn)
            h_interp = project_feasible(h_interp, tn)
            # Also try smoothed version
            h_smooth = project_feasible(gaussian_smooth(h_interp, 1.2), tn)
            c_int = compute_c5(h_interp)
            c_smo = compute_c5(h_smooth)
            start_h = h_interp if c_int <= c_smo else h_smooth
            h2, c2 = refine_active(start_h, time.time() + slot, max_passes=5, max_lags=70)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, tn

    # Phase 6: Final active set polish
    if time.time() < hard_end - 30:
        t_end6 = min(hard_end - 15, start_time + 1100)
        h2, c2 = refine_active(best_h, t_end6, max_passes=8, max_lags=90)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 7: Smooth max final pass
    if time.time() < hard_end - 20 and len(best_h) <= 256:
        h2, c2 = optimize_smooth(best_h, 200, maxiter=50)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 8: Block perturbation polish
    if time.time() < hard_end - 5:
        h2, c2 = block_perturb(best_h, hard_end - 2, step=0.012, max_trials=15000, seed=77777)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, best_h.size)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_h.size))
