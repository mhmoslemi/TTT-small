```python
def binary_local_search(h, n, max_iterations, time_limit, t0, budget):
    """Local search on binary sequences via pair swaps."""
    h = h.copy()
    h = (h > 0.5).astype(float)
    # Ensure exact balance
    n_ones = int(np.sum(h))
    n_want = n // 2
    if n_ones > n_want:
        ones = np.where(h == 1)[0]
        order = np.random.choice(len(ones), n_ones - n_want, replace=False)
        h[ones[order]] = 0.0
    elif n_ones < n_want:
        zeros = np.where(h == 0)[0]
        order = np.random.choice(len(zeros), n_want - n_ones, replace=False)
        h[zeros[order]] = 1.0
    
    best_corr_max = np.max(np.correlate(h, 1.0 - h, mode='full'))
    no_improve = 0
    max_no_improve = 5000
    
    while no_improve < max_no_improve:
        if time.time() - t0 > budget - 10:
            break
        # Random pair swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            no_improve += 1
            continue
        # Swap: h[i] 1-h[i], h[j] 1-h[j]
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = np.max(corr)
        if new_max < best_corr_max:
            best_corr_max = new_max
            no_improve = 0
        else:
            # Revert
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
            no_improve += 1
    
    return h, best_corr_max * (2.0 / n)
```

```python
def periodic_startup(n, p):
    """Try all periodic binary sequences with period p."""
    base_len = p
    if n % base_len != 0:
        return None
    reps = n // base_len
    best_h = None
    best_c5 = float('inf')
    
    # Enumerate all base patterns with p/2 ones
    from itertools import combinations
    for ones in combinations(range(p), p // 2):
        base = np.zeros(p)
        base[list(ones)] = 1.0
        h = np.tile(base, reps)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```
