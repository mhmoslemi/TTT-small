                                            'maxls': 50, 'maxfun': 5000})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h0 = h_new
                        c5_cur = c5_new
                except Exception:
                    pass
            
            if remaining_time() > 6:
                try:
                    res = minimize(topk_obj, h0, method='L-BFGS-B',
                                   bounds=bounds,
                                   options={'maxiter': 300, 'ftol': 1e-14,
                                            'iprint': 0, 'maxls': 50})
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur:
                        h0 = h_new
                        c5_cur = c5_new
                except Exception:
                    pass
            
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h0.copy()
                best_n = n
                print("  Improved: n={}, C5={:.6f}".format(n, c5_cur))
    
    # Phase 2: Binary simulated annealing
    if remaining_time() > 40:
        n = best_n
        print("\n--- Binary SA (n={}) ---".format(n))
        
        sa_starts = [project_feasible(best_h, n)]
        for _ in range(4):
            h_bin = np.zeros(n)
            idx = rng.permutation(n)
            h_bin[idx[:n // 2]] = 1.0
            sa_starts.append(h_bin)
        
        for sa_h in sa_starts:
            if remaining_time() < 15:
                break
            h_bin, c5_bin = binary_sa(sa_h, n, 4000, rng, t_start=0.004, t_end=0.00005)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
                best_n = n
                print("  SA: C5={:.6f}".format(c5_bin))
            
            if remaining_time() > 10:
                # Polish binary with continuous
                _N = n
                _TGT = n / 2.0
                bounds = [(0.0, 1.0)] * n
                for beta_val in [100, 300]:
                    if remaining_time() < 6:
                        break
                    _BETA = beta_val
                    try:
                        res = minimize(smooth_max_obj, h_bin, jac=smooth_max_grad,
                                       method='L-BFGS-B', bounds=bounds,
                                       options={'maxiter': 300, 'ftol': 1e-14, 'iprint': 0})
                        h_new = project_feasible(res.x, n)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            print("  Polished: C5={:.6f}".format(c5_new))
                    except Exception:
                        pass
    
    # Phase 3: Active-set gradient stepping
    if remaining_time() > 25:
        n = best_n
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        h = best_h.copy()
        print("\n--- Active-set gradient (n={}) ---".format(n))
        
        for it in range(300):
            if remaining_time() < 4:
                break
            
            grad, _ = active_set_step(h, n, top_k=10)
            if grad is None:
                break
            
            improved = False
            for step in [0.04, 0.08, 0.15, 0.02, 0.1, 0.25]:
                h_trial = h - step * grad
                h_trial = project_feasible(h_trial, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    h = h_trial.copy()
                    improved = True
                    break
            
            if not improved:
                if remaining_time() > 8:
                    _BETA = 400
                    try:
                        res = minimize(smooth_max_obj, h, jac=smooth_max_grad,
                                       method='L-BFGS-B', bounds=bounds,
                                       options={'maxiter': 150, 'ftol': 1e-14, 'iprint': 0})
                        h_new = project_feasible(res.x, n)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                            h = h_new.copy()
                        else:
                            break
                    except Exception:
                        break
    
    # Phase 4: Coordinate descent with continuous values
    if remaining_time() > 12:
        n = best_n
        _N = n
        _TGT = n / 2.0
        h = best_h.copy()
        dx_val = 2.0 / n
        
        for sweep in range(12):
            if remaining_time() < 4:
                break
            any_imp = False
            order = rng.permutation(n)
            for i in order:
                if remaining_time() < 2:
                    break
                old_val = h[i]
                best_local_c5 = compute_c5(h)
                best_local_val = old_val
                best_local_h = h.copy()
                
                candidates = [0.0, 1.0, 0.5, max(0.0, old_val - 0.2), min(1.0, old_val + 0.2)]
                if old_val > 0.3:
                    candidates.append(0.1)
                if old_val < 0.7:
                    candidates.append(0.9)
                
                for nv in candidates:
                    if abs(nv - old_val) < 1e-8:
                        continue
                    h[i] = nv
                    h_feas = project_feasible(h, n)
                    c5_test = compute_c5(h_feas)
                    if c5_test < best_local_c5:
                        best_local_c5 = c5_test
                        best_local_val = nv
                        best_local_h = h_feas.copy()
                h = best_local_h.copy()
                
                if best_local_c5 < best_c5:
                    best_c5 = best_local_c5
                    best_h = best_local_h.copy()
                    h = best_local_h.copy()
                    any_imp = True
            if not any_imp:
                break
    
    # Phase 5: Final polish
    if remaining_time() > 8:
        n = best_n
        _N = n
        _TGT = n / 2.0
        bounds = [(0.0, 1.0)] * n
        for beta_val in [800, 1500]:
            if remaining_time() < 4:
                break
            _BETA = beta_val
            try:
                res = minimize(smooth_max_obj, best_h, jac=smooth_max_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 500, 'ftol': 1e-15,
                                        'gtol': 1e-10, 'iprint': 0})
                h_new = project_feasible(res.x, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
    
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    
    try:
        eval_c5 = evaluate_erdos_solution(best_h)
        if eval_c5 < final_c5:
            final_c5 = eval_c5
    except Exception:
        pass
    
    print("\nFINAL: n={}, C5={:.7f}".format(best_n, final_c5))
    return (best_h, final_c5, best_n)
