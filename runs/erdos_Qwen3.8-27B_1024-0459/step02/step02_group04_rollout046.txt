            active_seq = (16, 32, 64, 96)
        else:
            active_seq = (12, 24, 48, 64)
    h = project_feasible(h0, n)
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c
    for acnt in active_seq:
        if time.time() > time_limit - 3.0:
            break
        passes = 3 if int(acnt) <= 32 else 2
        for _ in range(passes):
            if time.time() > time_limit - 3.0:
                break
            vals, lags = all_lag_values_lags(h)
            current_c = float(np.max(vals))
            if current_c < best_c - 1e-15:
                best_c = current_c
                best_h = h.copy()
            active = choose_active_lags(vals, lags, int(acnt), margin=5e-6)
            if len(active) == 0:
                active = [0, 1, -1]
            t0 = current_c + 1e-6
            res = optimize_active_t(h, active, t0, maxiter=maxiter, ftol=1e-14)
            if res is not None and res.x is not None:
                x = res.x
                if x.size == len(h) + 1 and np.all(np.isfinite(x)):
                    h2 = project_feasible(x[:-1], n)
                    c2 = compute_c5(h2)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2.copy()
                    if c2 <= current_c + 1e-10:
                        h = h2
                    else:
                        h = best_h
    return best_h, best_c


def interpolate_to_n(h_src, n_dst):
    h_src = np.asarray(h_src, dtype=float)
    n_dst = int(n_dst)
    n_src = h_src.size
    if n_src == n_dst:
        return h_src.copy()
    if n_src <= 1:
        val = float(h_src[0]) if n_src == 1 else 0.5
        return np.full(n_dst, val, dtype=float)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)


def binary_pattern(n, period, ones_count, offset=0):
    n = int(n)
    h = np.zeros(n, dtype=float)
    for i in range(n):
        if (i + offset) % period < ones_count:
            h[i] = 1.0
    return h


def smooth_gradient_descent(h0, n, lr=0.01, maxiter=200, time_limit=None):
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    step = lr
    n = int(n)
    for it in range(maxiter):
        if time_limit is not None and time.time() > time_limit:
            break
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        vals, lags = all_lag_values_lags(h)
        maxv = float(np.max(vals))
        active = np.where(vals >= maxv - 2e-5)[0]
        g = np.zeros(n, dtype=float)
        for idx in active:
            ll = int(lags[idx])
            g += lag_gradient_only(h, ll)
        g /= max(1, len(active))
        h_new = h - step * g
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < c:
            h = h_new
            step = min(step * 1.05, 0.05)
        else:
            step *= 0.6
    return best_h, float(best_c)


def local_block_search(h0, n, time_limit, seed, block_sizes=(2, 3, 4, 5), max_iter=8000):
    n = int(n)
    rng = np.random.RandomState(int(seed))
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h
    cur_c = best_c
    temp = 0.001
    min_temp = 1e-7
    for trial in range(max_iter):
        if time.time() > time_limit:
            break
        bs = block_sizes[int(rng.randint(0, len(block_sizes)))]
        if bs > n:
            bs = n
        st = int(rng.randint(0, max(1, n - bs + 1)))
        idx = np.arange(st, st + bs)
        delta = rng.randn(bs) * 0.02
        delta -= delta.mean()
        h_try = cur_h.copy()
        h_try[idx] += delta
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < best_c - 1e-15:
            best_h = h_try.copy()
            best_c = c_try
            cur_h = h_try
            cur_c = c_try
            temp = min(0.01, max(temp * 1.2, min_temp))
        elif c_try < cur_c:
            cur_h = h_try
            cur_c = c_try
        else:
            d = c_try - cur_c
            if 0.0 < d < temp:
                if rng.rand() < np.exp(-d / max(temp, 1e-10)):
                    cur_h = h_try
                    cur_c = c_try
        temp = max(temp * 0.997, min_temp)
    return best_h, float(best_c)


def coarsen_and_refine(h0, n, time_limit, seed):
    n = int(n)
    nc = max(16, n // 2)
    h_coarse = np.zeros(nc, dtype=float)
    ratio = n // nc
    for i in range(nc):
        h_coarse[i] = float(np.mean(h0[i * ratio:(i + 1) * ratio]))
    h_coarse = project_feasible(h_coarse, nc)
    h_coarse, cc = refine_active(h_coarse, nc, time_limit - 20.0, active_seq=(8, 16, 32), maxiter=40)
    h_fine = interpolate_to_n(h_coarse, n)
    h_fine = project_feasible(h_fine, n)
    h_fine, cf = refine_active(h_fine, n, time_limit, active_seq=(16, 32), maxiter=35)
    return h_fine, float(cf)


def periodic_optimize(n, period, time_limit, seed):
    n = int(n)
    if period < 2 or period > 20 or n % period != 0:
        return None, np.inf
    m = n // period
    rng = np.random.RandomState(int(seed))
    best_p = None
    best_c = np.inf
    starts = [np.full(period, 0.5)]
    starts.append(binary_pattern(period, period, period // 2))
    starts.append(np.arange(period, dtype=float) / period)
    starts.append(rng.rand(period) * 1.5)
    for p0 in starts[:4]:
        p = project_feasible(np.tile(p0, m), period)
        p = p[:period]
        bounds = [(0.0, 1.0) for _ in range(period)]
        rep = m
        constraints = [{"type": "eq", "fun": lambda x, r=rep: float(np.sum(x) - 0.5 * period),
                        "jac": lambda x, r=rep: np.ones(period)}]

        def obj_full(x, p0_=copy_period_to_n(x, n)):
            h_full = np.tile(x, n // period)
            return compute_c5(h_full)

        for p0 in starts[:3]:
            p_init = np.asarray(p0[:period], dtype=float)
            try:
                res = minimize(
                    lambda x: compute_c5(np.tile(x, m)),
                    p_init, jac=lambda x: -2.0 * (np.tile(x, m) - np.mean(np.tile(x, m))),
                    method="SLSQP", bounds=bounds,
                    constraints=constraints,
                    options={"maxiter": 80, "ftol": 1e-12, "disp": False},
                )
                if res.x is not None and np.all(np.isfinite(res.x)):
                    h_full = np.tile(res.x, m)
                    h_full = project_feasible(h_full, n)
                    c = compute_c5(h_full)
                    if c < best_c:
                        best_c = c
                        best_p = h_full
            except Exception:
                pass
    return best_p, best_c


def copy_period_to_n(p, n):
    return np.tile(np.asarray(p, dtype=float), n // len(p))[:n]


def diverse_starts(n, rng, parent_h=None):
    n = int(n)
    inits = []
    if parent_h is not None:
        if len(parent_h) == n:
            inits.append(parent_h.copy())
        else:
            inits.append(interpolate_to_n(parent_h, n))
    inits.append(project_feasible(binary_pattern(n, 4, 2), n))
    inits.append(project_feasible(binary_pattern(n, 4, 2, 1), n))
    inits.append(project_feasible(binary_pattern(n, 6, 3), n))
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in (2, 3, 5, 7):
        h = 0.5 + 0.3 * np.cos(freq * np.pi * x)
        inits.append(project_feasible(h, n))
    inits.append(project_feasible(rng.rand(n) * 1.2, n))
    if parent_h is not None:
        p = project_feasible(parent_h, n)
        inits.append(project_feasible(p + rng.randn(n) * 0.03, n))
        inits.append(project_feasible(p[::-1], n))
    return inits[:6]


def run(seed=42, budget_s=1300, **kwargs):
    seed = int(seed)
    budget_s = float(budget_s)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    hard_end = start_time + max(60.0, budget_s - 30.0)

    parent = get_parent_values()
    if parent is not None and parent.size > 0:
        best_n = int(parent.size)
        best_h = project_feasible(parent, best_n)
        best_c = compute_c5(best_h)
    else:
        best_n = 160
        best_h = project_feasible(np.full(best_n, 0.5), best_n)
        best_c = compute_c5(best_h)

    # Phase 1: Intensive refinement at parent resolution
    if time.time() < hard_end - 200.0:
        t1_end = min(hard_end - 200.0, time.time() + 150.0)
        if best_n <= 200:
            h2, c2 = refine_active(best_h, best_n, t1_end, active_seq=(24, 48, 80, 128), maxiter=60)
            if c2 < best_c:
                best_h, best_c = h2, c2
            for beta in (30, 100, 250):
                if time.time() > t1_end - 5:
                    break
                h2, c2 = optimize_smooth_h(best_h, best_n, beta, maxiter=50, ftol=1e-13)
                if c2 < best_c:
                    best_h, best_c = h2, c2
            h2, c2 = refine_active(best_h, best_n, t1_end, active_seq=(32, 64, 128), maxiter=70)
            if c2 < best_c:
                best_h, best_c = h2, c2

    # Phase 2: Try higher resolutions
    target_ns = [200, 240, 280, 320]
    if best_n != 200:
        target_ns.insert(0, 200)
    target_ns = [nn for nn in target_ns if nn > best_n or abs(nn - best_n) <= 40]
    target_ns = target_ns[:4]

    for nn in target_ns:
        if time.time() > hard_end - 150.0:
            break
        rem = hard_end - 150.0 - time.time()
        if rem < 60.0:
            break
        slot_end = min(hard_end - 100.0, time.time() + min(180.0, rem * 0.6))
        inits = diverse_starts(nn, rng, best_h)
        current_best_h = None
        current_best_c = np.inf
        per_init = max(15.0, (slot_end - time.time()) / float(len(inits)))
        for j, h0 in enumerate(inits):
            if time.time() >= slot_end:
                break
            per = min(per_init, slot_end - time.time())
            if per < 10.0:
                break
            h0 = project_feasible(h0, nn)
            sub_end = time.time() + per
            if nn <= 240:
                for beta in (40, 120):
                    if time.time() > sub_end - 10:
                        break
                    h2, c2 = optimize_smooth_h(h0, nn, beta, maxiter=35, ftol=1e-12)
                    if c2 < current_best_c:
                        current_best_h, current_best_c = h2, c2
                    h0 = h2
            if time.time() < sub_end - 5:
                seq = (16, 32, 64) if nn <= 200 else (12, 24, 48)
                h2, c2 = refine_active(h0, nn, sub_end, active_seq=seq, maxiter=40)
                if c2 < current_best_c:
                    current_best_h, current_best_c = h2, c2
        if current_best_h is not None and current_best_c < best_c:
            best_h, best_c, best_n = current_best_h, current_best_c, nn

    # Phase 3: Coarse-to-fine at best resolution
    if time.time() < hard_end - 80.0 and best_n >= 48:
        cf_end = min(hard_end - 50.0, time.time() + 100.0)
        h2, c2 = coarsen_and_refine(best_h, best_n, cf_end, seed + 7777)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 4: Focused active-set refinement
    if time.time() < hard_end - 40.0:
        af_end = min(hard_end - 20.0, time.time() + 120.0)
        if best_n <= 200:
            seq = (32, 64, 96, 128)
        elif best_n <= 280:
            seq = (24, 48, 80)
        else:
            seq = (16, 32, 64)
        h2, c2 = refine_active(best_h, best_n, af_end, active_seq=seq, maxiter=80)
        if c2 < best_c:
            best_h, best_c = h2, c2
        for beta in (150, 400):
            if time.time() > af_end - 8:
                break
            h2, c2 = optimize_smooth_h(best_h, best_n, beta, maxiter=40, ftol=1e-13)
            if c2 < best_c:
                best_h, best_c = h2, c2
        h2, c2 = refine_active(best_h, best_n, af_end, active_seq=seq[:2], maxiter=90)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 5: Local block search polish
    if time.time() < hard_end - 10.0 and best_n <= 320:
        ls_end = min(hard_end - 2.0, time.time() + 60.0)
        h2, c2 = local_block_search(best_h, best_n, ls_end, seed + 31337,
                                    block_sizes=(2, 3, 4), max_iter=12000)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Final gradient step
    if time.time() < hard_end - 3.0 and best_n <= 320:
        h2, c2 = smooth_gradient_descent(best_h, best_n, lr=0.002, maxiter=100,
                                         time_limit=hard_end - 2.0)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Final active-set micro-refinement
    if time.time() < hard_end - 2.0:
        h2, c2 = refine_active(best_h, best_n, hard_end, active_seq=(32, 64), maxiter=50)
        if c2 < best_c:
            best_h, best_c = h2, c2

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h)
    return (best_h, float(best_c), int(best_n))
