import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lags_array(n):
    return np.arange(-(n - 1), n)


def coord_optimize_one_pass(h, n, corr, lags):
    """One full pass of coordinate descent with exact subproblem."""
    jl_offsets = lags  # = -(n-1), ..., n-1
    for j in range(n):
        t_old = h[j]
        # d(l) = 1 - h(j+l) - h(j-l), with h=0 for out-of-range
        jl = j + jl_offsets
        valid_jl = (jl >= 0) & (jl < n)
        h_jl = np.zeros(2 * n - 1)
        h_jl[valid_jl] = h[jl[valid_jl]]
        jl2 = j - jl_offsets
        valid_jl2 = (jl2 >= 0) & (jl2 < n)
        h_jl2 = np.zeros(2 * n - 1)
        h_jl2[valid_jl2] = h[jl2[valid_jl2]]
        d = 1.0 - h_jl - h_jl2
        # corr_l(t) = corr_l + (t - t_old) * d_l
        # Minimize max_l (corr_l + (t - t_old)*d_l) over t in [0,1]
        # = max_l (d_l * t + (corr_l - t_old * d_l))
        # Let a_l = d_l, b_l = corr_l - t_old * d_l
        a = d
        b = corr - t_old * d
        lo, hi = 0.0, 1.0
        for _ in range(40):
            mid = 0.5 * (lo + hi)
            vals = a * mid + b
            mx = np.max(vals)
            active = vals >= mx - 1e-15
            sg = np.mean(a[active])
            if sg > 1e-13:
                hi = mid
            elif sg < -1e-13:
                lo = mid
            else:
                break
        new_t = 0.5 * (lo + hi)
        h[j] = new_t
        # Update corr incrementally
        delta_t = new_t - t_old
        if abs(delta_t) > 1e-15:
            corr += delta_t * d
    return h, corr


def coord_descent(h, n, n_passes, time_limit, verbose=False):
    """Run coordinate descent for n_passes or time_limit seconds."""
    h = h.copy().astype(float)
    n = len(h)
    corr = compute_corr(h)
    lags = lags_array(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        h, corr = coord_optimize_one_pass(h, n, corr, lags)
        h = make_feasible(h, n)
        corr = compute_corr(h)
        c5 = np.max(corr) * (2.0 / n)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        elif best_c5 - c5 < 1e-15 and p > 20:
            # Converged
            if (p % 50) == 0 and p > 100:
                break
    return best_h, best_c5


def interpolate_to_n(h_src, n_dst):
    if len(h_src) < 2:
        return np.full(n_dst, 0.5)
    x_old = np.linspace(0.0, 2.0, len(h_src), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_src)
    return make_feasible(h_interp, n_dst)


def c5_objective(x, n):
    return np.max(np.correlate(x, 1.0 - x, mode='full'))


def sum_constraint(x, n):
    return np.sum(x) - n / 2.0


def slsqp_polish(h, n, max_iter=2000):
    h = h.copy().astype(float)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    try:
        res = minimize(c5_objective, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        h_out = make_feasible(res.x, n)
        c5_out = compute_c5(h_out)
        if np.isfinite(c5_out):
            return h_out, c5_out
    except Exception:
        pass
    return None, None


def random_restart(h, n, sigma, rng):
    """Add noise and re-project."""
    h_noisy = h + sigma * rng.randn(n)
    h_noisy = np.clip(h_noisy, 0.0, 1.0)
    return make_feasible(h_noisy, n)


def pair_swap_search(h, n, max_trials=10000, time_limit=60):
    """Try small pairwise perturbations that preserve the sum."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = compute_corr(best_h)
    lags = lags_array(n)
    rng = np.random.RandomState(789)
    start = time.time()
    for _ in range(max_trials):
        if time.time() - start > time_limit:
            break
        i, j = rng.randint(0, n), rng.randint(0, n)
        if i == j:
            continue
        delta = rng.uniform(-0.1, 0.1)
        h_try = best_h.copy()
        hi = h_try[i] + delta
        hj = h_try[j] - delta
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            # Try smaller delta
            delta *= 0.3
            hi = h_try[i] + delta
            hj = h_try[j] - delta
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
        h_try[i] = hi
        h_try[j] = hj
        if abs(np.sum(h_try) - n / 2.0) > 1e-10:
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
    return best_h, best_c5


def structured_init_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if bin(i).count('1') % 2 == 0 else 0.0
    return make_feasible(h, n)


def structured_init_vander_corput(n):
    h = np.zeros(n)
    positions = []
    for i in range(1, n + 1):
        f = 1.0
        val = 0.0
        j = i
        while j > 0:
            j, r = divmod(j, 2)
            f *= 0.5
            val += r * f
        positions.append(val * 2.0)
    positions.sort()
    for j in range(n // 2):
        idx = int(positions[j] * n)
        idx = min(idx, n - 1)
        h[idx] = 1.0
    return make_feasible(h, n)


def structured_init_periodic(n, period):
    h = np.zeros(n)
    for i in range(n // 2):
        h[i % period] = 1.0
    # Ensure sum is n/2
    return make_feasible(h, n)
