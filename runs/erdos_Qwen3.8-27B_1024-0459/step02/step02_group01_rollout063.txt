def fast_grad(h, n, active_lags):
    """Compute gradient using shifting operations."""
    grad = np.zeros(n)
    for offset in active_lags:
        # offset = k - (n-1) where k is the lag in 0..2n-2
        if offset > 0:
            # (1-h)[j+offset] for j in 0..n-1-offset
            grad[:n-offset] += (1-h)[offset:]
            # -h[j-offset] for j in offset..n-1
            grad[offset:] -= h[:n-offset]
        elif offset < 0:
            o = -offset
            grad[o:] += (1-h)[:n-o]
            grad[:n-o] -= h[o:]
        # offset == 0: grad[j] += (1-h[j]) - h[j] = 1 - 2*h[j]
        else:
            grad += 1 - 2*h
    grad /= len(active_lags)
    return grad

    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 0.001) & (h < 0.999)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        # Add/subtract from the most room
        for i in idx:
            if diff > 0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
            else:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0:
                a = min(1.0 - h[i], diff)
                h[i] += a
                diff -= a
            else:
                a = min(h[i], -diff)
                h[i] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def make_grad(h, n, lags):
    grad = np.zeros(n)
    for k in lags:
        off = int(k) - (n - 1)
        if off > 0:
            grad[:n - off] += (1.0 - h)[off:]
            grad[off:] -= h[:n - off]
        elif off < 0:
            o = -off
            grad[o:] += (1.0 - h)[:n - o]
            grad[:n - o] -= h[o:]
        else:
            grad += 1.0 - 2.0 * h
    grad /= float(len(lags))
    return grad

                j = int(zeros[rng.randint(len(zeros))])
        else:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])

        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        c_new = compute_c5(h)

        if c_new < best_c5 - 1e-15:
            best_c5 = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
            no_improve += 1

        if no_improve > 30000:
            # Perturb: random pair of swaps
            for _ in range(3):
                on = np.where(h > 0.5)[0]
                zo = np.where(h < 0.5)[0]
                if len(on) < 1 or len(zo) < 1:
                    break
                pi = int(on[rng.randint(len(on))])
                pj = int(zo[rng.randint(len(zo))])
                h[pi], h[pj] = 1.0 - h[pi], 1.0 - h[pj]
            c_cur = compute_c5(h)
            if c_cur > best_c5 * 1.005:
                h = best_h.copy()
            no_improve = 0

    return best_h, best_c5

def structured_bin(n, rng):
    """Create structured binary initializations."""
    inits = []
    nw = n // 2
    # Alternating
    if n % 2 == 0:
        h1 = np.zeros(n)
        h1[::2] = 1.0
        inits.append(h1.copy())
    # Random
    h2 = np.zeros(n)
    h2[rng.choice(n, nw, replace=False)] = 1.0
    inits.append(h2)
    # Greedy: place ones to minimize max diff
    h3 = np.zeros(n)
    used = set()
    for _ in range(nw):
        best_pos = 0
        best_score = float('inf')
        for p in range(n):
            if p in used:
                continue
            cnt = sum(1 for q in used if abs((p - q) % n) < n // 2)
            if cnt < best_score:
                best_score = cnt
                best_pos = p
        h3[best_pos] = 1.0
        used.add(best_pos)
    inits.append(h3)
    return inits

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        ph = np.array(initial_h_values, dtype=float)
        pn = len(ph)
        if pn > 0:
            best_h = ph.copy()
            best_c5 = compute_c5(ph)
            best_n = pn
    except Exception:
        pass

    if best_h is None:
        best_n = 192
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    def tl():
        return max(0.0, budget_s - (time.time() - t0) - 3.0)

    # Phase 1: Continuous optimization at parent resolution
    if tl() > 80:
        h1, c1 = cont_opt(best_h, best_n, t0, min(80.0, tl() * 0.7), top_k=5, lr_init=0.05)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1

    if tl() > 50:
        h2, c2 = cont_opt(best_h, best_n, t0, min(50.0, tl() * 0.65), top_k=3, lr_init=0.03)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    if tl() > 35:
        h3, c3 = cont_opt(best_h, best_n, t0, min(35.0, tl() * 0.7), top_k=1, lr_init=0.08)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

    if tl() > 30:
        h4, c4 = cont_opt(best_h, best_n, t0, min(30.0, tl() * 0.7), top_k=10, lr_init=0.02)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4

    # Phase 2: Binary local search at parent resolution
    # Start 1: rounded continuous
    hb1 = round_bin(best_h, best_n)
    cb1 = compute_c5(hb1)
    if cb1 < best_c5:
        best_c5 = cb1
        best_h = hb1.copy()

    if tl() > 300:
        hbl, cbl = bin_ls(best_h, best_n, t0, min(300.0, tl() * 0.4), seed=seed)
        if cbl < best_c5:
            best_c5 = cbl
            best_h = hbl

    if tl() > 150:
        hbl2, cbl2 = bin_ls(best_h, best_n, t0, min(150.0, tl() * 0.5), seed=seed + 777)
        if cbl2 < best_c5:
            best_c5 = cbl2
            best_h = hbl2

    # Try structured inits at parent n
    struct_inits = structured_bin(best_n, rng)
    for si, hsi in enumerate(struct_inits):
        if tl() < 30:
            break
        hsi = project_feasible(hsi, best_n)
        cs = compute_c5(hsi)
        if cs < best_c5:
            best_c5 = cs
            best_h = hsi.copy()
        if tl() > 40:
            hsl, csl = bin_ls(hsi, best_n, t0, min(40.0, tl() * 0.3), seed=seed + 200 + si)
            if csl < best_c5:
                best_c5 = csl
                best_h = hsl

    # Phase 3: Multi-resolution
    for n_alt in [160, 224, 256, 288, 320]:
        if tl() < 50:
            break
        xp = (np.arange(best_n) + 0.5) / best_n * 2.0
        xn = (np.arange(n_alt) + 0.5) / n_alt * 2.0
        hi = project_feasible(np.interp(xn, xp, best_h), n_alt)

        ha, ca = cont_opt(hi, n_alt, t0, min(25.0, tl() * 0.25), top_k=5, lr_init=0.04)
        if ca < best_c5:
            best_c5 = ca
            best_h = ha.copy()
            best_n = n_alt

        hab = round_bin(ha, n_alt)
        cab = compute_c5(hab)
        if cab < best_c5:
            best_c5 = cab
            best_h = hab.copy()
            best_n = n_alt

        if tl() > 30:
            hab2, cab2 = bin_ls(hab, n_alt, t0, min(30.0, tl() * 0.3), seed=seed + n_alt)
            if cab2 < best_c5:
                best_c5 = cab2
                best_h = hab2.copy()
                best_n = n_alt

        # Structured at this n
        if tl() > 40:
            sb = structured_bin(n_alt, rng)
            for hsb in sb[:2]:
                hsb = project_feasible(hsb, n_alt)
                cbs = compute_c5(hsb)
                if cbs < best_c5:
                    best_c5 = cbs
                    best_h = hsb.copy()
                    best_n = n_alt
                if tl() > 25:
                    hsl2, csl2 = bin_ls(hsb, n_alt, t0, min(25.0, tl() * 0.3), seed=seed + n_alt + 50)
                    if csl2 < best_c5:
                        best_c5 = csl2
                        best_h = hsl2.copy()
                        best_n = n_alt

    # Phase 4: Final polish on best solution
    if tl() > 15:
        hf, cf = cont_opt(best_h, best_n, t0, min(15.0, tl() * 0.8), top_k=2, lr_init=0.01)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

        hbf = round_bin(best_h, best_n)
        cbf = compute_c5(hbf)
        if cbf < best_c5:
            best_c5 = cbf
            best_h = hbf.copy()

    if tl() > 10:
        hb_final, cb_final = bin_ls(best_h, best_n, t0, min(10.0, tl() * 0.8), seed=seed + 9999)
        if cb_final < best_c5:
            best_c5 = cb_final
            best_h = hb_final

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    return (best_h, final_c5, best_n)
