import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Adjust the most flexible coordinate
        flexibility = np.minimum(h, 1.0 - h)
        idx = np.argmax(flexibility)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exps = np.exp(shifted)
    return m + np.log(np.sum(exps)) / beta

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

    weights = exps / np.sum(exps)
    
    # Gradient of sum w_k * corr[k] wrt h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            # corr[k] = sum_{i=L}^{n-1} h[i] * (1-h[i-L])
            # d/dh[i] for i >= L: (1-h[i-L])
            # d/dh[i] for i < L: -h[i+L]
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        else:
            Lp = -L
            # corr[k] = sum_{i=0}^{n-1-Lp} h[i] * (1-h[i+Lp])
            # d/dh[i] for i < n-Lp: (1-h[i+Lp])
            # d/dh[i] for i >= n-Lp: -h[i-Lp]
            grad[:n-Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n-Lp]
    
    grad *= dx
    h_new = h - lr * grad
    return h_new, smooth_max(corr, beta) * dx

def run_opt_pgd(h0, n, beta_schedule, lr_schedule, max_iter, time_limit):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    start = time.time()
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        beta = beta_schedule[it] if it < len(beta_schedule) else beta_schedule[-1]
        lr = lr_schedule[it] if it < len(lr_schedule) else lr_schedule[-1]
        
        h_new, _ = projected_gradient_step(h, beta, lr, n)
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - float(np.sum(h))
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-15:
        flex = np.minimum(h, 1.0 - h)
        idx = int(np.argmax(flex))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) / n_src * 2.0
    x_dst = (np.arange(n_dst) + 0.5) / n_dst * 2.0
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def compute_smooth_max_and_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 0.0)
    exps = np.exp(shifted)
    total = np.sum(exps)
    smax = m + np.log(total) / beta
    weights = exps / total
    return smax, weights

def compute_gradient_weights(weights, n, h):
    dx = 2.0 / n
    grad = np.zeros(n)
    nn1 = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - nn1
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return dx * grad

def pgd_optimize(h0, n, betas, lrs, max_iters_per_phase, time_limit):
    h = h0.copy()
    best_h = h0.copy()

            break
        beta = betas[phase]
        lr = lrs[phase]
        for it in range(max_iters_per_phase):

            grad = compute_gradient_weights(weights, n, h)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)

                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new

    return best_h, best_c5

def coordinate_descent(h, n, n_iters, rng, time_limit):
    start = time.time()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        i = rng.randint(n)
        cur_val = h[i]
        # Try a few perturbations
        for dv in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:
            h_trial = h.copy()
            h_trial[i] = float(np.clip(cur_val + dv, 0.0, 1.0))
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                h[i] = h_trial[i]
                best_c5 = c5_trial
                break
    return h, best_c5

def make_initial_h(n, rng, mode):
    if mode == 0:
        # Binary random
        h = np.zeros(n)
        idx = rng.permutation(n)[:n//2]
        h[idx] = 1.0
        return h
    elif mode == 1:
        # Smooth random
        freq = rng.randint(1, 4)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.15, 0.35)
        h = 0.5 + amp * np.sin(2*np.pi*freq*np.arange(n)/n + phase)
        h = np.clip(h, 0, 1)
        return project_feasible(h, n)
    elif mode == 2:
        # Block structure
        h = np.zeros(n)
        block_size = n // (2 * rng.randint(2, 5))
        starts = sorted(rng.choice(n, size=n//block_size//2, replace=False))
        for s in range(n//block_size//2):
            s_pos = int(s * n / (n//block_size//2))
            h[s_pos:s_pos+block_size] = 1.0
        return project_feasible(h, n)
    else:
        # Alternating blocks
        h = np.zeros(n)
        period = 2 * rng.randint(4, 12)
        fill = period // 2
        for i in range(0, n, period):
            h[i:i+fill] = 1.0
        return project_feasible(h, n)
