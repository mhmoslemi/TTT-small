    exps = np.exp(shifted)
    s = np.sum(exps)
    if s < 1e-300:
        return np.zeros_like(corr)
    return exps / s

def corr_value_grad_full(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = smooth_max_val(corr, beta) * dx
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-13:
            continue
        L = k - (n - 1)
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            Lp = -L
            grad[:n - Lp] += w * (1.0 - h[Lp:])
            grad[Lp:] -= w * h[:n - Lp]
    return val, dx * grad

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.maximum(h, 1.0 - h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_reproject(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst)
    x_src = (np.arange(n_src) + 0.5) * (2.0 / n_src)
    x_dst = (np.arange(n_dst) + 0.5) * (2.0 / n_dst)
    h_dst = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_dst, n_dst)

def fourier_init(n, K, rng):
    h = np.full(n, 0.5)
    for m in range(1, K + 1):
        amp = rng.uniform(0.08, 0.35) / max(1, np.sqrt(m))
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
        phase2 = rng.uniform(0.0, 2.0 * np.pi)
        amp2 = rng.uniform(0.03, 0.2) / max(1, np.sqrt(m))
        h += amp2 * np.sin(2.0 * np.pi * m * np.arange(n) / n + phase2)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def make_random_inits(n, rng, parent_h):
    inits = []
    if parent_h is not None:
        inits.append(interp_reproject(parent_h, n))
        h_noisy = interp_reproject(parent_h, n) + rng.uniform(-0.12, 0.12, n)
        inits.append(project_feasible(np.clip(h_noisy, 0, 1), n))
    inits.append(fourier_init(n, 3, rng))
    inits.append(fourier_init(n, 5, rng))
    h_bin = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h_bin[idx] = 1.0
    inits.append(h_bin)
    freq = rng.uniform(1.0, 3.5)
    ph = rng.uniform(0.0, 2.0 * np.pi)
    h_s = 0.5 + rng.uniform(0.25, 0.45) * np.sin(2.0 * np.pi * freq * np.arange(n) / n + ph)
    inits.append(project_feasible(np.clip(h_s, 0, 1), n))
    return inits

def optimize_lbfgsb(h0, n, beta, max_iter=400, penalty_coeff=200.0):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        val, _ = corr_value_grad_full(x, beta)
        s = np.sum(x) - target
        return val + penalty_coeff * s * s

    def jac(x):
        _, g = corr_value_grad_full(x, beta)
        s = np.sum(x) - target
        return g + 2.0 * penalty_coeff * s

    res = minimize(obj, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-9, 'maxls': 40})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def optimize_topk(h0, n, beta, topk, max_rounds=5, max_iter=200):
    h = h0.copy()
    best = compute_c5(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    for r in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-topk:]

        def obj_top(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            active = c[top_idx]
            val = smooth_max_val(active, beta) * (2.0 / n)
            s = np.sum(x) - target
            return val + 100.0 * s * s

        def jac_top(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            active = c[top_idx]
            m = np.max(active)
            w = smooth_max_weights(active, beta)
            grad = np.zeros(n)
            for jj, k in enumerate(top_idx):
                wj = w[jj]
                if wj < 1e-13:
                    continue
                L = k - (n - 1)
                if L == 0:
                    grad += wj * (1.0 - 2.0 * x)
                elif L > 0:
                    grad[L:] += wj * (1.0 - x[:n - L])
                    grad[:n - L] -= wj * x[L:]
                else:
                    Lp = -L
                    grad[:n - Lp] += wj * (1.0 - x[Lp:])
                    grad[Lp:] -= wj * x[:n - Lp]
            s = np.sum(x) - target
            return (2.0 / n) * grad + 200.0 * s

        res = minimize(obj_top, h, jac=jac_top, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-8, 'maxls': 30})
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best:
            h = h_new
            best = c5_new
        else:
            break
    return h, best

def local_search_continuous(h, n, rng, n_steps=50):
    best = compute_c5(h)
    target = n / 2.0
    for _ in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        di = rng.uniform(-0.15, 0.15)
        dj = -di
        h[i] += di
        h[j] += dj
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= di
            h[j] -= dj
            continue
        c5_try = compute_c5(h)
        if c5_try < best:
            best = c5_try
        else:
            h[i] -= di
            h[j] -= dj
    return h, best

def binary_local_search(h, n, rng, max_rounds=4000):
    best = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        for _ in range(40):
            i1 = rng.randint(len(ones))
            i0 = rng.randint(len(zeros))
            p1, p0 = ones[i1], zeros[i0]
            h[p1] = 0.0
            h[p0] = 1.0
            nc = compute_c5(h)
            if nc < best:
                best = nc
                improved = True
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[p1] = 1.0
                h[p0] = 0.0
        if not improved:
            break
    return h, best

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return max(0.0, budget_s - elapsed() - 12.0)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
        pn = len(parent_h)
        if pn > 0:
            pc5 = compute_c5(parent_h)
            best_h = parent_h.copy()
            best_c5 = pc5
            best_n = pn
            print("Parent: n={}, C5={:.7f}".format(pn, pc5), flush=True)
    except Exception:
        pass

    if best_h is None:
        best_n = 200
        best_h = np.zeros(best_n)
        best_h[:best_n // 2] = 1.0
        best_c5 = compute_c5(best_h)
        print("Default: n={}, C5={:.7f}".format(best_n, best_c5), flush=True)

    # Phase 1: Multi-resolution, multi-start L-BFGS-B optimization
    n_list = sorted(set([best_n, 192, 200, 216, 224, 256, 288, 320]))
    n_list = [nn for nn in n_list if 80 <= nn <= 400]

    for n in n_list:
        if remaining() < 55:
            break
        print("=== P1 n={} rem={:.0f}s ===".format(n, remaining()), flush=True)
        inits = make_random_inits(n, rng, parent_h)

        for h0 in inits:
            if remaining() < 35:
                break
            h0 = project_feasible(h0, n)
            cur = h0.copy()
            for bv in [20, 60, 150, 400, 800]:
                if remaining() < 20:
                    break
                mi = 350 if bv <= 150 else 250
                try:
                    hnew, c5new = optimize_lbfgsb(cur, n, bv, max_iter=mi)
                    if c5new < best_c5:
                        best_c5 = c5new
                        best_h = hnew.copy()
                        best_n = n
                        print("  n={} b={}: C5={:.7f}".format(n, bv, best_c5), flush=True)
                    cur = hnew
                except Exception:
                    break

    # Phase 2: Active-set refinement at best resolution
    if remaining() > 40 and best_n > 0:
        print("=== P2 Active n={} rem={:.0f}s ===".format(best_n, remaining()), flush=True)
        for tk in [10, 20, 35]:
            if remaining() < 20:
                break
            try:
                ha, ca = optimize_topk(best_h, best_n, beta=400, topk=tk,
                                       max_rounds=4, max_iter=180)
                if ca < best_c5:
                    best_c5 = ca
                    best_h = ha.copy()
                    print("  topk={}: C5={:.7f}".format(tk, best_c5), flush=True)
            except Exception:
                pass

    # Phase 3: Perturbation + re-optimize
    if remaining() > 40:
        print("=== P3 Perturb rem={:.0f}s ===".format(remaining()), flush=True)
        n_trials = min(10, int(remaining() // 25))
        for t in range(n_trials):
            if remaining() < 18:
                break
            hp = best_h.copy()
            np_ = rng.randint(4, max(5, best_n // 5))
            pidx = rng.choice(best_n, size=np_, replace=False)
            hp[pidx] = np.clip(hp[pidx] + rng.uniform(-0.2, 0.2, np_), 0, 1)
            hp = project_feasible(hp, best_n)
            try:
                ht, ct = optimize_lbfgsb(hp, best_n, 500, max_iter=250)
                if ct < best_c5:
                    best_c5 = ct
                    best_h = ht.copy()
                    print("  Pert{}: C5={:.7f}".format(t, best_c5), flush=True)
            except Exception:
                pass

    # Phase 4: High-beta deep polish
    if remaining() > 25:
        print("=== P4 Polish rem={:.0f}s ===".format(remaining()), flush=True)
        for bv in [1200, 2500]:
            if remaining() < 12:
                break
            try:
                hpol, cpol = optimize_lbfgsb(best_h, best_n, bv, max_iter=500)
                if cpol < best_c5:
                    best_c5 = cpol
                    best_h = hpol.copy()
                    print("  b={}: C5={:.7f}".format(bv, best_c5), flush=True)
            except Exception:
                pass

    # Phase 5: Continuous local search
    if remaining() > 15:
        hls, cls = local_search_continuous(best_h.copy(), best_n, rng, n_steps=300)
        hls = project_feasible(hls, best_n)
        cls = compute_c5(hls)
        if cls < best_c5:
            best_c5 = cls
            best_h = hls.copy()
            print("  Cont LS: C5={:.7f}".format(best_c5), flush=True)

    # Phase 6: Binary rounding + local search
    if remaining() > 12:
        hr = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(hr))
        n_want = best_n // 2
        if n_ones > n_want:
            ones_idx = np.where(hr == 1.0)[0]
            order = np.argsort(best_h[ones_idx])[:n_ones - n_want]
            hr[ones_idx[order]] = 0.0
        elif n_ones < n_want:
            zero_idx = np.where(hr == 0.0)[0]
            order = np.argsort(-best_h[zero_idx])[:n_want - n_ones]
            hr[zero_idx[order]] = 1.0
        cr = compute_c5(hr)
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
            print("  Binary: C5={:.7f}".format(best_c5), flush=True)

        if remaining() > 15 and np.all((best_h == 0) | (best_h == 1)):
            hb, cb = binary_local_search(best_h.copy(), best_n, rng, max_rounds=3000)
            cb = compute_c5(hb)
            if cb < best_c5:
                best_c5 = cb
                best_h = hb.copy()
                print("  Bin LS: C5={:.7f}".format(best_c5), flush=True)

    # Finalize
    best_h = project_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)

    try:
        ec5 = evaluate_erdos_solution(best_h)
        if isinstance(ec5, float) and ec5 < final_c5:
            final_c5 = ec5
    except Exception:
        pass

    print("FINAL: n={}, C5={:.7f}".format(best_n, final_c5), flush=True)
    return (best_h, final_c5, best_n)
